    try:
        res = minimize(
            _f_and_grad, a, args=(top_k,), jac=True,
            method='L-BFGS-B',
            bounds=(0.0, 1000.0),
            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10,
                     'maxfun': max_iter * 5, 'callback': _time_cb}
        )
        x = res.x
    except Exception:
        x = a.copy()
    x = np.maximum(x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    x = _normalize(x, target_sum)
    return x, _score_fast(x)


def _momentum_run(a, deadline, lr_init=0.06, decay=0.9998, beta=0.9,
                  top_k=1, max_steps=500000):
    """Momentum-based subgradient optimizer (backup)."""
    n = len(a)
    target_sum = float(np.sum(a))
    momentum = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_s = _score_fast(a)
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        s, g = _f_and_grad(a, top_k)
        momentum = beta * momentum + (1.0 - beta) * g
        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        a = _normalize(a, target_sum)
        lr *= decay
        if step % 200 == 0:
            cur_s = _score_fast(a)
            if cur_s < best_s:
                best_s = cur_s
                best_a = a.copy()
        if lr < 1e-7:
            break
    return best_a, best_s


def _build_initializations(n, parent, rng):
    """Build diverse set of initializations for a given length n."""
    ts = np.sqrt(2.0 * n)
    k = np.arange(n)
    inits = []

    if parent is not None and len(parent) == n:
        inits.append(("parent", _normalize(np.maximum(parent, 0.0), ts)))
        inits.append(("parent_p01", _normalize(np.maximum(parent + 0.03 * rng.standard_normal(n), 0), ts)))
        inits.append(("parent_p05", _normalize(np.maximum(parent + 0.1 * rng.standard_normal(n), 0), ts)))
        inits.append(("parent_p10", _normalize(np.maximum(parent + 0.3 * rng.standard_normal(n), 0), ts)))

    inits.append(("uniform", np.full(n, ts / n)))

    for i in range(4):
        r = np.abs(rng.standard_normal(n)) + 0.1
        inits.append((f"rand{i}", _normalize(r, ts)))

    for freq in [1, 2, 3, 5, 7, 11]:
        c = _normalize(np.maximum(1.0 + 0.6 * np.cos(2 * np.pi * freq * k / n), 0.01), ts)
        inits.append((f"cos{freq}", c))

    for freq in [2, 5]:
        s = _normalize(np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * freq * k / n), 0.01), ts)
        inits.append((f"sin{freq}", s))

    # Two Gaussian peaks
    for frac in [0.2, 0.3, 0.4]:
        g1 = np.exp(-((k - n * frac) / (n * 0.08)) ** 2)
        g2 = np.exp(-((k - n * (1 - frac)) / (n * 0.08)) ** 2)
        inits.append((f"2g{frac}", _normalize(g1 + g2 + 0.01, ts)))

    # Ramp / triangle / V-shape
    inits.append(("ramp", _normalize(np.linspace(0.2, 1.8, n), ts)))
    inits.append(("inv_ramp", _normalize(np.linspace(1.8, 0.2, n), ts)))
    tri = 1.0 - 2.0 * np.abs(k / n - 0.5)
    inits.append(("triangle", _normalize(np.maximum(tri, 0.05), ts)))
    vshape = 1.0 + 0.8 * np.abs(k / n - 0.5) * 2.0
    inits.append(("vshape", _normalize(vshape, ts)))

    # Sawtooth
    for period in [3, 5, 7, 13]:
        saw = 1.0 + 0.4 * (2.0 * (k % period) / period - 1.0)
        inits.append((f"saw{period}", _normalize(np.maximum(saw, 0.05), ts)))

    # Product of cosines (higher-frequency structure)
    inits.append(("cos_prod", _normalize(np.maximum(
        1.0 + 0.3 * np.cos(2 * np.pi * k / n) * np.cos(2 * np.pi * 3 * k / n), 0.01), ts)))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2n*max(autocorr)/(sum)^2 via L-BFGS-B with analytical gradients,
    multi-restart, and momentum-based backup. Uses FFT for fast convolution.
    """
    global _GLOBAL_DEADLINE
    rng = np.random.default_rng(seed)
    _GLOBAL_DEADLINE = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # Retrieve parent sequence
    parent = None
    n_default = 5582
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = np.clip(p, 0.0, 1000.0)
            n_default = parent.size
    except Exception:
        pass

    best = None
    best_score = np.inf

    def _try_update(a):
        nonlocal best, best_score
        a = np.clip(np.maximum(a, 0.0), 0.0, 1000.0)
        if np.sum(a) < 0.01:
            return
        fs = _score_fast(a)
        if fs < best_score:
            actual = float(evaluate_sequence(a.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = a.copy()
                print(f"  ** NEW BEST: {best_score:.7f} (n={len(a)})")

    # Evaluate parent
    if parent is not None:
        p_norm = _normalize(parent, np.sqrt(2.0 * parent.size))
        _try_update(p_norm)
        print(f"Initial parent score: {best_score:.6f} (n={parent.size})")

    # Determine lengths to explore
    n_vals = sorted(set([
        n_default,
        max(2000, n_default - 500),
        n_default + 500,
        max(2000, n_default - 200),
        n_default + 200,
        4500, 5582, 6500, 7500, 8500,
    ]))
    n_vals = [n for n in n_vals if 2000 <= n <= 12000]
    print(f"Exploring lengths: {n_vals}")

    # === PHASE 1: Build and score initialization pool ===
    print("\n=== Phase 1: Initialization pool ===")
    t0 = time.monotonic()
    pool = []
    for n in n_vals:
        inits = _build_initializations(n, parent, rng)
        for name, a in inits:
            s = _score_fast(a)
            pool.append((s, name, n, a))
            _try_update(a)

    pool.sort(key=lambda x: x[0])
    print(f"  Pool: {len(pool)} inits, best fast score: {pool[0][0]:.6f}")
    print(f"  Best eval: {best_score:.6f}")
    print(f"  Time: {time.monotonic() - t0:.1f}s")

    # === PHASE 2: L-BFGS-B optimization from top candidates ===
    print("\n=== Phase 2: L-BFGS-B optimization ===")
    t0 = time.monotonic()

    max_pool_tries = 60
    for i, (s, name, n, a) in enumerate(pool[:max_pool_tries]):
        if time.monotonic() > _GLOBAL_DEADLINE - 120:
            break

        # Standard L-BFGS-B run
        a_pert = a + 0.008 * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = _normalize(a_pert, np.sqrt(2.0 * n))
        _GLOBAL_DEADLINE = max(time.monotonic() + 90.0, _GLOBAL_DEADLINE - 90)
        result, res_s = _lbfgsb_run(a_pert, top_k=1, max_iter=120)
        _try_update(result)

        # Top-k=3 smoothed gradient run
        if time.monotonic() < _GLOBAL_DEADLINE + 90 - 120:
            a_pert3 = a + 0.01 * rng.standard_normal(n)
            a_pert3 = np.maximum(a_pert3, 0.0)
            a_pert3 = _normalize(a_pert3, np.sqrt(2.0 * n))
            _GLOBAL_DEADLINE = max(time.monotonic() + 60.0, _GLOBAL_DEADLINE - 60)
            result3, res_s3 = _lbfgsb_run(a_pert3, top_k=3, max_iter=100)
            _try_update(result3)

        # If very promising, do deeper optimization
        if res_s < best_score * 1.005 and time.monotonic() < _GLOBAL_DEADLINE + 150 - 120:
            a_deep = _normalize(result + 0.003 * rng.standard_normal(n), np.sqrt(2.0 * n))
            _GLOBAL_DEADLINE = max(time.monotonic() + 150.0, _GLOBAL_DEADLINE - 150)
            result_d, res_s_d = _lbfgsb_run(a_deep, top_k=1, max_iter=250, ftol=1e-15)
            _try_update(result_d)
            if res_s_d < best_score:
                a_deep2 = _normalize(result_d + 0.001 * rng.standard_normal(n), np.sqrt(2.0 * n))
                _GLOBAL_DEADLINE = max(time.monotonic() + 100.0, _GLOBAL_DEADLINE - 100)
                result_d2, _ = _lbfgsb_run(a_deep2, top_k=1, max_iter=200, ftol=1e-15)
                _try_update(result_d2)

    # Restore deadline
    _GLOBAL_DEADLINE = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    print(f"  Phase 2 done. Best: {best_score:.6f}")
    print(f"  Time: {time.monotonic() - t0:.1f}s")

    # === PHASE 3: Momentum-based refinement ===
    print("\n=== Phase 3: Momentum refinement ===")
    t0 = time.monotonic()
    if best is not None:
        n_b = len(best)
        # Run momentum from best with different hyperparams
        for lr_i, beta, tk in [(0.04, 0.9, 1), (0.03, 0.95, 3), (0.05, 0.85, 5)]:
            if time.monotonic() > _GLOBAL_DEADLINE - 60:
                break
            a_m = best + 0.01 * rng.standard_normal(n_b)
            a_m = np.maximum(a_m, 0.0)
            a_m = _normalize(a_m, np.sqrt(2.0 * n_b))
            mom_deadline = min(time.monotonic() + 120, _GLOBAL_DEADLINE - 30)
            result, res_s = _momentum_run(a_m, mom_deadline, lr_init=lr_i,
                                          decay=0.9997, beta=beta, top_k=tk)
            _try_update(result)

    print(f"  Phase 3 done. Best: {best_score:.6f}")
    print(f"  Time: {time.monotonic() - t0:.1f}s")

    # === PHASE 4: Simulated-annealing style iterative refinement ===
    print("\n=== Phase 4: Annealing refinement ===")
    t0 = time.monotonic()
    if best is not None:
        n_b = len(best)
        n_passes = 15
        for pass_i in range(n_passes):
            if time.monotonic() > _GLOBAL_DEADLINE - 20:
                break
            cool = 1.0 - pass_i / n_passes
            pert = 0.003 * cool + 0.0005

            a_an = best + pert * rng.standard_normal(n_b)
            a_an = np.maximum(a_an, 0.0)
            a_an = _normalize(a_an, np.sqrt(2.0 * n_b))

            pass_deadline = min(time.monotonic() + min(100, len([1]) * 100), _GLOBAL_DEADLINE - 15)
            _GLOBAL_DEADLINE_SAVE = _GLOBAL_DEADLINE
            _GLOBAL_DEADLINE = pass_deadline
            result, res_s = _lbfgsb_run(a_an, top_k=1, max_iter=100, ftol=1e-15)
            _GLOBAL_DEADLINE = _GLOBAL_DEADLINE_SAVE
            _try_update(result)

            # Occasionally try top_k=5
            if pass_i % 3 == 0 and time.monotonic() < _GLOBAL_DEADLINE - 20:
                a_an5 = best + pert * rng.standard_normal(n_b)
                a_an5 = np.maximum(a_an5, 0.0)
                a_an5 = _normalize(a_an5, np.sqrt(2.0 * n_b))
                _GLOBAL_DEADLINE = min(time.monotonic() + 60, _GLOBAL_DEADLINE - 15)
                result5, _ = _lbfgsb_run(a_an5, top_k=5, max_iter=80)
                _GLOBAL_DEADLINE = _GLOBAL_DEADLINE_SAVE
                _try_update(result5)

    print(f"  Phase 4 done. Best: {best_score:.6f}")
    print(f"  Time: {time.monotonic() - t0:.1f}s")

    # === PHASE 5: Ultra-fine tuning ===
    if best is not None and time.monotonic() < _GLOBAL_DEADLINE - 5:
        print("\n=== Phase 5: Ultra-fine tuning ===")
        n_b = len(best)
        for _ in range(8):
            if time.monotonic() > _GLOBAL_DEADLINE - 3:
                break
            a_ft = best + 0.0005 * rng.standard_normal(n_b)
            a_ft = np.maximum(a_ft, 0.0)
            a_ft = _normalize(a_ft, np.sqrt(2.0 * n_b))
            _GLOBAL_DEADLINE_SAVE = _GLOBAL_DEADLINE
            _GLOBAL_DEADLINE = min(time.monotonic() + 30, _GLOBAL_DEADLINE - 2)
            result, _ = _lbfgsb_run(a_ft, top_k=1, max_iter=50, ftol=1e-16)
            _GLOBAL_DEADLINE = _GLOBAL_DEADLINE_SAVE
            _try_update(result)

    # Final output
    if best is not None:
        final = np.clip(best, 0.0, 1000.0)
        final_score = float(evaluate_sequence(final.tolist()))
        print(f"\n{'='*50}")
        print(f"FINAL SCORE: {final_score:.7f} (n={len(final)})")
        print(f"{'='*50}")
        return [float(x) for x in final]

    # Fallback
    print("FALLBACK: uniform sequence")
    n = 5582
    a = np.full(n, np.sqrt(2.0 * n) / n)
    return [float(x) for x in a]
