"""
Search for low-peak nonnegative self-convolutions using a hybrid approach:
1. LSE-smoothed gradient descent for fast initial optimization
2. LP-based refinement (SOTA approach) for precise convergence
3. Multi-length search to exploit the n-dependence of the lower bound
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones(max(a.size, 1), dtype=np.float64) / max(a.size, 1)
    return a / s

def _autocorr(a, fft_size=None):
    n = a.size
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]

def _score_norm(a):
    """Score assuming a is normalized to sum=1."""
    n = a.size
    b = _autocorr(a)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * n * mb

def _make_exponential(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _normalize(a)

def _make_symmetric_exp(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    center = 0.5
    a = np.exp(-alpha * np.abs(x - center))
    return _normalize(a)

def _make_cosine(n):
    x = np.linspace(0.0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))  # raised cosine
    return _normalize(a)

def _make_gaussian(n, sigma=0.2):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return _normalize(a)

def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = _normalize(old)
    new_n = max(int(new_n), 2)
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

def _lse_gradient_descent(a, deadline, rng, beta_init=10.0, beta_max=500.0):
    """LSE-smoothed projected gradient descent on the simplex."""
    a = _normalize(a)
    n = a.size
    if n < 8:
        a = _make_exponential(128, 1.0)
        n = a.size
    
    fft_size = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n)
    
    b = _autocorr(a, fft_size)
    maxb = float(np.max(b))
    score = 2.0 * n * maxb
    best_a = a.copy()
    best_score = score
    
    alpha = 0.1
    beta = beta_init
    it = 0
    fail = 0
    
    while time.monotonic() < deadline:
        # Compute LSE and gradient
        m = float(np.max(b))
        if m <= 1e-300:
            m = 1e-300
        
        if beta <= 1.5:
            p = np.ones(2 * n - 1) / (2 * n - 1)
            lse_val = m
        else:
            arg = beta * (b - m)
            arg = np.maximum(arg, -745.0)
            e = np.exp(arg)
            s = float(e.sum())
            if s <= 0:
                p = np.ones(2 * n - 1) / (2 * n - 1)
                lse_val = m
            else:
                p = e / s
                lse_val = m + float(np.log(s)) / beta
        
        # Gradient: d/d a_i [sum_k p_k (a*a)_k] = 2 * (p_folded * a)_i
        # where (p_folded * a)_i = sum_k p_k * a_{k-i}
        ar = a[::-1]
        A = np.fft.rfft(ar, g_fft)
        P = np.fft.rfft(p, g_fft)
        conv = np.fft.irfft(A * P, g_fft).real
        h = conv[n - 1:2 * n - 1]
        h = np.maximum(h, 0.0)
        grad = 2.0 * h
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))
        
        if scale <= 1e-300:
            # Perturb
            noise = np.exp(rng.normal(0, 0.2, n))
            a = _normalize(a * noise)
            b = _autocorr(a, fft_size)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.1
            fail = 0
            it += 1
            continue
        
        # Step
        v = a - alpha * grad_c
        a = _normalize(v)
        
        b = _autocorr(a, fft_size)
        maxb = float(np.max(b))
        score = 2.0 * n * maxb
        
        # New LSE for comparison
        m2 = float(np.max(b))
        if m2 <= 1e-300:
            m2 = 1e-300
        if beta <= 1.5:
            lse_new = m2
        else:
            arg2 = beta * (b - m2)
            arg2 = np.maximum(arg2, -745.0)
            e2 = np.exp(arg2)
            s2 = float(e2.sum())
            lse_new = m2 + (float(np.log(s2)) / beta if s2 > 0 else 0)
        
        if lse_new < lse_val - 1e-15:
            alpha = min(1.0, alpha * 1.15)
            fail = 0
        else:
            alpha *= 0.5
            fail += 1
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Increase beta periodically
        if it % 50 == 0:
            beta = min(beta_max, beta * 1.3)
        
        # Restart if stuck
        if fail > 200 and time.monotonic() < deadline - 3:
            a = _normalize(best_a * np.exp(rng.normal(0, 0.3, n)))
            b = _autocorr(a, fft_size)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.1
            beta = max(beta_init, beta * 0.5)
            fail = 0
        
        it += 1
    
    return best_a, best_score

def _lp_refine(a, deadline, rng, max_total_iters=2000):
    """LP-based refinement."""
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    
    it = 0
    stall = 0
    
    while time.monotonic() < deadline and it < max_total_iters:
        b = _autocorr(a)
        M = float(np.max(b))
        
        if M <= 1e-300:
            break
        
        # Build LP: max sum(b_j) s.t. (a*b)_k <= M for all k, b >= 0
        # (a*b)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_{k-j} * b_j
        
        # Build constraint matrix (sparse, banded)
        rows = []
        cols = []
        vals = []
        
        for k in range(2 * n - 1):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0 or res.x is None:
            break
        
        b_opt = np.maximum(res.x, 0.0)
        s_b = float(b_opt.sum())
        if s_b <= 1e-12:
            break
        
        b_norm = b_opt / s_b
        
        # Line search
        best_t = 0.05
        best_s = _score_norm((1 - 0.05) * a + 0.05 * b_norm)
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            cand = (1.0 - t) * a + t * b_norm
            cand = _normalize(cand)
            s = _score_norm(cand)
            if s < best_s - 1e-10:
                best_s = s
                best_t = t
        
        if best_t < 0.01:
            # No good step found
            stall += 1
            if stall > 10:
                # Perturb and retry
                a = _normalize(a * np.exp(rng.normal(0, 0.2, n)))
                stall = 0
                it += 1
                continue
            # Try smaller step
            a = _normalize((1 - 0.01) * a + 0.01 * b_norm)
            it += 1
            continue
        
        a = _normalize((1.0 - best_t) * a + best_t * b_norm)
        cur_score = _score_norm(a)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                a = _normalize(best_a * np.exp(rng.normal(0, 0.15, n)))
                stall = 0
        
        it += 1
        if it % 50 == 0:
            print(f"  LP it={it} score={best_score:.8f} t={time.monotonic():.1f}", flush=True)
    
    return best_a, best_score

def _get_given():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0 or arr.sum() <= 1e-300:
        return None
    return arr / arr.sum()

def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Hybrid LSE-gradient + LP refinement search for low-peak nonnegative
    self-convolutions. Returns the best sequence found within the time budget.
    """
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15  # leave 15s margin
    rng = np.random.default_rng(seed)
    
    given = _get_given()
    
    # Initial candidates
    candidates = []
    if given is not None:
        candidates.append(given)
    candidates.append(_make_exponential(256, 1.0))
    candidates.append(_make_symmetric_exp(256, 2.0))
    candidates.append(_make_cosine(256))
    candidates.append(_make_gaussian(256, 0.25))
    
    # Quick LSE at n=256
    print("Phase 1: LSE at n=256", flush=True)
    best_a = None
    best_score = 1e100
    
    for i, cand in enumerate(candidates):
        if time.monotonic() > deadline - 300:
            break
        t_seg_end = min(deadline - 280, t0 + 30 * (i + 1))
        if t_seg_end <= time.monotonic():
            break
        a_out, s_out = _lse_gradient_descent(cand.copy(), t_seg_end, rng)
        score = _score_norm(a_out)
        print(f"  candidate {i}: score={score:.8f}", flush=True)
        if score < best_score:
            best_score = score
            best_a = a_out
    
    if best_a is None:
        best_a = _make_exponential(256, 1.0)
        best_score = _score_norm(best_a)
    
    # Try at n=512
    print("Phase 2: LSE at n=512", flush=True)
    if time.monotonic() < deadline - 200:
        for shape_fn in [_make_exponential, _make_symmetric_exp, _make_cosine]:
            init = shape_fn(512, 1.0) if shape_fn != _make_cosine else shape_fn(512)
            # Also try resampling best
            if time.monotonic() > deadline - 180:
                break
            t_seg_end = min(deadline - 160, time.monotonic() + 40)
            a_out, s_out = _lse_gradient_descent(init.copy(), t_seg_end, rng)
            score = _score_norm(a_out)
            print(f"  n=512 score={score:.8f}", flush=True)
            if score < best_score:
                best_score = score
                best_a = a_out
    
    # LP refinement at increasing lengths
    lengths = [512, 1024, 2048, 4096]
    
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        
        # Create initial candidate for this length
        if best_a.size != n:
            init = _resample(best_a, n)
        else:
            init = best_a.copy()
        
        # Mix with a fresh shape to avoid getting stuck
        fresh = _make_exponential(n, rng.uniform(0.5, 2.0))
        init = _normalize(0.7 * init + 0.3 * fresh)
        
        # Also try the given sequence resampled
        if given is not None and given.size > 10:
            init_given = _resample(given, n)
            init = _normalize(0.5 * init + 0.5 * init_given)
        
        print(f"Phase 3: LP at n={n}", flush=True)
        t_seg_end = min(deadline - 30, time.monotonic() + max(30, (deadline - time.monotonic()) * 0.35))
        a_out, s_out = _lp_refine(init.copy(), t_seg_end, rng)
        score = _score_norm(a_out)
        print(f"  LP n={n}: score={score:.8f} (best={best_score:.8f})", flush=True)
        
        if score < best_score:
            best_score = score
            best_a = a_out
    
    # Final polish
    if time.monotonic() < deadline - 5:
        print(f"Phase 4: Final LP polish n={best_a.size}", flush=True)
        a_out, s_out = _lp_refine(best_a.copy(), deadline - 3, rng, max_total_iters=100)
        if s_out < best_score:
            best_score = s_out
            best_a = a_out
            print(f"  Final: score={best_score:.8f}", flush=True)
    
    # Output
    out = _normalize(best_a)
    result = [float(x) for x in out]
    print(f"Final answer: n={len(result)}, estimated score={best_score:.8f}", flush=True)
    print(f"Total time: {time.monotonic() - t0:.1f}s", flush=True)
    return result

"""
Hybrid LSE-gradient + LP-refinement search for low-peak nonnegative
self-convolutions. Uses difference-set-inspired initializations,
multi-length strategy, and a relaxed-LP variant focusing on top peaks.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones(max(a.size, 1), dtype=np.float64) / max(a.size, 1)
    return a / s


def _autocorr(a, fft_size=None):
    n = a.size
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _score_norm(a):
    n = a.size
    b = _autocorr(a)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * n * mb


def _make_exponential(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _normalize(a)


def _make_symmetric_exp(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * np.abs(x - 0.5))
    return _normalize(a)


def _make_cosine(n):
    x = np.linspace(0.0, np.pi, n)
    a = 0.5 * (1.0 + np.cos(x))
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_sparse_random(n, k, rng):
    """Place k non-zero elements at random positions."""
    a = np.zeros(n)
    if k <= 0 or n <= 0:
        return _normalize(np.ones(n))
    k = min(k, n)
    positions = rng.choice(n, size=k, replace=False)
    a[positions] = 1.0 / k
    return a


def _make_difference_set_like(n, rng):
    """Create a sequence inspired by difference sets for flat autocorrelation."""
    # Try to place elements so differences are well-spread
    k = max(4, n // 20)
    a = np.zeros(n)
    # Use a semi-structured placement
    step = n // k
    for i in range(k):
        pos = (i * step + rng.integers(0, max(1, step // 2))) % n
        a[pos] = 1.0
    # Add some random elements
    extra_k = max(1, k // 4)
    extra_pos = rng.choice(n, size=extra_k, replace=False)
    a[extra_pos] += 1.0
    return _normalize(a)


def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = _normalize(old)
    new_n = max(int(new_n), 2)
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _lse_gd(a, deadline, rng, beta_init=8.0, beta_max=400.0):
    """LSE-smoothed projected gradient descent."""
    a = _normalize(a)
    n = a.size
    if n < 8:
        a = _make_exponential(128, 1.0)
        n = a.size

    fft_size = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n)

    b = _autocorr(a, fft_size)
    maxb = float(np.max(b))
    score = 2.0 * n * maxb
    best_a = a.copy()
    best_score = score

    alpha = 0.1
    beta = beta_init
    it = 0
    fail = 0

    while time.monotonic() < deadline:
        m = float(np.max(b))
        if m <= 1e-300:
            m = 1e-300

        if beta <= 1.5:
            p = np.ones(2 * n - 1) / (2 * n - 1)
            lse_val = m
        else:
            arg = beta * (b - m)
            arg = np.maximum(arg, -745.0)
            e = np.exp(arg)
            s = float(e.sum())
            if s <= 0:
                p = np.ones(2 * n - 1) / (2 * n - 1)
                lse_val = m
            else:
                p = e / s
                lse_val = m + float(np.log(s)) / beta

        # Gradient via FFT
        ar = a[::-1]
        A = np.fft.rfft(ar, g_fft)
        P = np.fft.rfft(p, g_fft)
        conv = np.fft.irfft(A * P, g_fft).real
        h = conv[n - 1:2 * n - 1]
        h = np.maximum(h, 0.0)
        grad = 2.0 * h
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))

        if scale <= 1e-300:
            noise = np.exp(rng.normal(0, 0.25, n))
            a = _normalize(a * noise)
            b = _autocorr(a, fft_size)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.1
            fail = 0
            it += 1
            continue

        v = a - alpha * grad_c
        a = _normalize(v)

        b = _autocorr(a, fft_size)
        maxb = float(np.max(b))
        score = 2.0 * n * maxb

        # Check LSE improvement
        m2 = float(np.max(b))
        if m2 <= 1e-300:
            m2 = 1e-300
        if beta <= 1.5:
            lse_new = m2
        else:
            arg2 = beta * (b - m2)
            arg2 = np.maximum(arg2, -745.0)
            e2 = np.exp(arg2)
            s2 = float(e2.sum())
            lse_new = m2 + (float(np.log(s2)) / beta if s2 > 0 else 0.0)

        if lse_new < lse_val - 1e-15:
            alpha = min(1.0, alpha * 1.13)
            fail = 0
        else:
            alpha *= 0.5
            fail += 1

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        if it % 40 == 0:
            beta = min(beta_max, beta * 1.25)

        if fail > 300 and time.monotonic() < deadline - 3:
            a = _normalize(best_a * np.exp(rng.normal(0, 0.3, n)))
            b = _autocorr(a, fft_size)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.1
            beta = max(beta_init, beta * 0.5)
            fail = 0

        it += 1

    return best_a, best_score


def _build_lp_matrix(a):
    """Build the (2n-1) x n constraint matrix for the LP.
    A[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0.
    """
    n = len(a)
    # For column j: rows j, j+1, ..., j+n-1 get values a[0], a[1], ..., a[n-1]
    j = np.arange(n, dtype=np.int64)
    # Rows: for each column j, rows j through j+n-1
    # Vectorized: row = j + i for i in range(n)
    rows = (j[:, None] + np.arange(n)[None, :]).ravel().astype(np.int64)
    cols = j.repeat(n).astype(np.int64)
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine(a, deadline, rng, max_iters=500, top_k_ratio=0.5):
    """LP-based refinement with relaxed constraints (top-k peaks only)."""
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    start_time = time.monotonic()

    it = 0
    stall = 0
    use_relaxed = n > 512  # Only use relaxed LP for larger n
    top_frac = top_k_ratio

    while time.monotonic() < deadline and it < max_iters:
        b = _autocorr(a)
        M = float(np.max(b))

        if M <= 1e-300:
            break

        if use_relaxed:
            # Only constrain top K peaks
            k_top = max(3, int(top_frac * (2 * n - 1)))
            idx_top = np.argpartition(b, -k_top)[-k_top:]
            # Build sparse constraint matrix for only these rows
            A_full = _build_lp_matrix(a)
            A_sub = A_full[idx_top, :]
            b_sub = np.full(k_top, M)
            num_constraints = k_top
            A_ub_arr = A_sub
            b_ub_arr = b_sub
        else:
            A_full = _build_lp_matrix(a)
            A_ub_arr = A_full
            b_ub_arr = np.full(2 * n - 1, M)
            num_constraints = 2 * n - 1

        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub_arr, b_ub=b_ub_arr,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, (deadline - time.monotonic()) * 0.8)})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            # Try full LP as fallback
            c_obj = -np.ones(n)
            bounds = [(0, None)] * n
            A_full = _build_lp_matrix(a)
            b_ub_full = np.full(2 * n - 1, M)
            try:
                res = linprog(c_obj, A_ub=A_full, b_ub=b_ub_full,
                              bounds=bounds, method='highs')
            except Exception:
                break
            if res.status != 0 or res.x is None:
                break

        b_opt = np.maximum(res.x, 0.0)
        s_b = float(b_opt.sum())
        if s_b <= 1e-12:
            break

        b_norm = b_opt / s_b

        # Line search with multiple candidates
        best_t = 0.05
        best_s = _score_norm(_normalize((1 - 0.05) * a + 0.05 * b_norm))
        for t in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            cand = _normalize((1.0 - t) * a + t * b_norm)
            s = _score_norm(cand)
            if s < best_s - 1e-10:
                best_s = s
                best_t = t
            if s < best_score - 1e-6:
                break

        if best_t < 0.01:
            stall += 1
            if stall > 8:
                # Perturb to escape
                a = _normalize(best_a * np.exp(rng.normal(0, 0.2, n)))
                stall = 0
                it += 1
                continue
            a = _normalize((1.0 - 0.02) * a + 0.02 * b_norm)
            it += 1
            continue

        a = _normalize((1.0 - best_t) * a + best_t * b_norm)
        cur_score = _score_norm(a)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
            # Increase relaxation as we make progress
            if use_relaxed and top_frac < 1.0:
                top_frac = min(1.0, top_frac + 0.05)
        else:
            stall += 1
            if stall > 15:
                a = _normalize(best_a * np.exp(rng.normal(0, 0.12, n)))
                stall = 0

        it += 1
        if it % 30 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    LP it={it} score={best_score:.8f} elapsed={elapsed:.1f}s", flush=True)

    return best_a, best_score


def _get_given():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0 or arr.sum() <= 1e-300:
        return None
    return arr / arr.sum()


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Hybrid LSE-gradient + LP-refinement search for low-peak nonnegative
    self-convolution sequences. Returns the best sequence found.
    """
    t0 = time.monotonic()
    budget = float(budget_s) if np.isfinite(budget_s) else 2400.0
    deadline = t0 + budget - 20  # 20s safety margin
    rng = np.random.default_rng(seed)

    given = _get_given()

    # === Phase 0: Generate initial candidates ===
    print(f"Phase 0: Generating candidates (budget={budget:.0f}s)", flush=True)
    init_n = 256
    candidates = []

    if given is not None and given.size >= 16:
        candidates.append(_resample(given, init_n))

    candidates.append(_make_exponential(init_n, 1.0))
    candidates.append(_make_exponential(init_n, 0.5))
    candidates.append(_make_symmetric_exp(init_n, 1.5))
    candidates.append(_make_cosine(init_n))
    candidates.append(_make_sparse_random(init_n, init_n // 4, rng))
    candidates.append(_make_difference_set_like(init_n, rng))

    # === Phase 1: Quick LSE at n=256 ===
    print("Phase 1: LSE at n=256", flush=True)
    best_a = None
    best_score = 1e100

    lse_time = min(60.0, budget * 0.04)
    per_cand = lse_time / len(candidates)

    for i, cand in enumerate(candidates):
        if time.monotonic() > deadline - 100:
            break
        t_end = min(deadline - 90, t0 + (i + 1) * per_cand)
        if t_end <= time.monotonic() + 2:
            continue
        a_out, s_out = _lse_gd(cand.copy(), t_end, rng)
        score = _score_norm(a_out)
        print(f"  cand {i}: score={score:.8f}", flush=True)
        if score < best_score:
            best_score = score
            best_a = a_out.copy()

    if best_a is None:
        best_a = _make_exponential(256, 1.0)
        best_score = _score_norm(best_a)

    # === Phase 2: LSE at n=512 ===
    print(f"Phase 2: LSE at n=512 (best={best_score:.8f})", flush=True)
    if time.monotonic() < deadline - 150:
        n2 = 512
        inits_2 = [
            _resample(best_a, n2),
            _make_exponential(n2, 1.0),
            _make_sparse_random(n2, n2 // 5, rng),
        ]
        if given is not None and given.size >= 16:
            inits_2.append(_resample(given, n2))

        lse_time2 = min(80.0, (deadline - time.monotonic()) * 0.06)
        per_cand2 = lse_time2 / len(inits_2)

        for i, cand in enumerate(inits_2):
            if time.monotonic() > deadline - 120:
                break
            t_end = min(deadline - 100, t0 + 60 + (i + 1) * per_cand2)
            if t_end <= time.monotonic() + 2:
                continue
            a_out, s_out = _lse_gd(cand.copy(), t_end, rng)
            score = _score_norm(a_out)
            print(f"  n=512 cand {i}: score={score:.8f}", flush=True)
            if score < best_score:
                best_score = score
                best_a = a_out.copy()

    # === Phase 3: LP refinement at increasing lengths ===
    lengths_to_try = [512, 1024, 2048, 4096]
    remaining = deadline - time.monotonic()
    print(f"Phase 3: LP refinement (remaining={remaining:.0f}s, best={best_score:.8f})", flush=True)

    # Allocate time proportionally, favoring larger n
    weights = [1.0, 2.0, 3.5, 4.5]
    total_w = sum(weights)

    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 30:
            break

        alloc = (deadline - time.monotonic()) * (weights[li] / sum(weights[li:]))
        alloc = min(alloc, (deadline - time.monotonic()) * 0.45)
        alloc = max(alloc, 20.0)
        t_end = time.monotonic() + alloc

        # Create initial candidate for this length
        if best_a.size == n:
            init = best_a.copy()
        else:
            init = _resample(best_a, n)

        # Blend with a diverse shape to avoid local optima
        diverse = _make_exponential(n, rng.uniform(0.5, 2.5))
        sparse_init = _make_sparse_random(n, max(8, n // 10), rng)
        blend = 0.6 * init + 0.25 * diverse + 0.15 * sparse_init
        init = _normalize(blend)

        # Also try given sequence
        if given is not None and given.size >= 16:
            init_given = _resample(given, n)
            init_alt = _normalize(0.5 * init + 0.5 * init_given)
        else:
            init_alt = init

        print(f"  LP n={n} (alloc={alloc:.0f}s)", flush=True)

        # Run LP on the blended init
        a_out, s_out = _lp_refine(init.copy(), t_end, rng, max_iters=300)
        score = _score_norm(a_out)
        print(f"    blend: score={score:.8f}", flush=True)
        if score < best_score:
            best_score = score
            best_a = a_out.copy()

        # If time remains, try the given-based init
        if time.monotonic() < t_end - 10 and init_alt is not None and not np.allclose(init, init_alt):
            t_end2 = min(t_end, time.monotonic() + (t_end - time.monotonic()) * 0.4)
            a_out2, s_out2 = _lp_refine(init_alt.copy(), t_end2, rng, max_iters=150)
            score2 = _score_norm(a_out2)
            print(f"    given blend: score={score2:.8f}", flush=True)
            if score2 < best_score:
                best_score = score2
                best_a = a_out2.copy()

    # === Phase 4: Final polish on the best sequence ===
    if time.monotonic() < deadline - 10:
        print(f"Phase 4: Final LP polish n={best_a.size} (best={best_score:.8f})", flush=True)
        # Try a few short LP runs from slightly perturbed versions
        for attempt in range(3):
            if time.monotonic() > deadline - 5:
                break
            pert = _normalize(best_a * np.exp(rng.normal(0, 0.05, best_a.size)))
            t_end = min(deadline - 3, time.monotonic() + 30)
            a_out, s_out = _lp_refine(pert, t_end, rng, max_iters=30)
            if s_out < best_score:
                best_score = s_out
                best_a = a_out.copy()
                print(f"  polish attempt {attempt}: improved to {best_score:.8f}", flush=True)

    # Try a slightly longer version if time allows
    if time.monotonic() < deadline - 60 and best_a.size < 8192:
        n_final = min(8192, max(2048, 2 * best_a.size))
        print(f"Phase 5: Upscale to n={n_final}", flush=True)
        init_up = _resample(best_a, n_final)
        mixed = _normalize(0.7 * init_up + 0.3 * _make_exponential(n_final, 1.0))
        t_end = min(deadline - 3, time.monotonic() + (deadline - time.monotonic()) * 0.5)
        a_out, s_out = _lp_refine(mixed, t_end, rng, max_iters=50)
        score = _score_norm(a_out)
        print(f"  Upscaled score={score:.8f} (best={best_score:.8f})", flush=True)
        if score < best_score:
            best_score = score
            best_a = a_out.copy()

    # Final output
    out = _normalize(best_a)
    result = [float(x) for x in out]
    total_time = time.monotonic() - t0
    print(f"\nFINAL: n={len(result)}, score={best_score:.8f}, time={total_time:.1f}s", flush=True)
    return result
