        # Simpler approach: use the fact that grad = ifft(fft(p) * conj(fft(f)))
        # Then take the appropriate n samples
        FR2 = np.fft.rfft(np.concatenate([f, np.zeros(N - n)]), N)
        P2 = np.fft.rfft(np.concatenate([p, np.zeros(N - p.size)]), N)
        grad_full = np.fft.irfft(P2 * np.conjugate(FR2), N)
        # grad_full[i] = sum_k p[(k-i) mod N] * f[k]... 
        # Actually irfft(fft(p)*conj(fft(f))) = circular cross-correlation
        # x_cross[i] = sum_k p[k] * f[(k-i) mod N]
        # For linear: grad[i] = sum_k p[k] * f[k-i] where 0<=k-i<n
        # = sum_j f[j] * p[i+j]  (letting j = k-i)
        # This is a "sliding" inner product
        
        # For our purposes, let's just compute it more directly for correctness
        # grad[i] = sum_{j=0}^{n-1} f[j] * p[i+j]  (with p out of range = 0)
        # This is valid for i = 0, ..., n-1 as long as p is long enough (2n-1)
        
        # Use FFT trick: grad = ifft(fft(p_zero_padded) * conj(fft(f_zero_padded)))
        # but we need the right slice
        # Actually the cleanest: use np.fft to compute the cross-correlation
        # rfft of p (length 2n-1) and conj of rfft of f (length n), zero-padded
        # The IFFT gives circular cross-correlation which for our sizes equals linear
        
        # Let me just do it via the convolution identity:
        # cross_corr(p, f)[i] = sum_j p[i+j]*f[j] = convolve(p, reverse(f))[i + len(f) - 1]
        # For p of length 2n-1 and f of length n:
        # result length = (2n-1)+n-1 = 3n-2
        # We want indices [n-1, 2n-2] of this result
        
        # For efficiency, let's use FFT
        N_corr = _next_pow2(3 * n)
        Pfin = np.fft.rfft(p, N_corr)
        Frev = np.fft.rfft(np.concatenate([f[::-1], np.zeros(N_corr - n)]), N_corr)
        conv_res = np.fft.irfft(Pfin * Frev, N_corr)
        # conv_res[k] = sum_j p[j] * f_rev[k-j] where f_rev = reversed f
        # = sum_j p[j] * f[n-1-(k-j)] = sum_j p[j] * f[n-1-k+j]
        # We want grad[i] = sum_j p[i+j]*f[j], let j' = i+j: sum_{j'} p[j']*f[j'-i]
        # = conv_res[n-1-i]... let me verify
        # conv_res[m] = sum_j p[j] * f[n-1-m+j]
        # Set m = n-1-i: conv_res[n-1-i] = sum_j p[j] * f[n-1-(n-1-i)+j] = sum_j p[j]*f[i+j] ✓
        grad = conv_res[n-1-1::-1]  # Hmm let me be more careful
        # Actually: conv_res[m] = sum_j p[j]*f_rev[m-j] where f_rev[k]=f[n-1-k]
        # grad[i] = sum_{j=0}^{n-1} f[j]*p[i+j]
        # Let l = i+j, so j = l-i: grad[i] = sum_l f[l-i]*p[l] for l=i..i+n-1
        # = sum_l p[l]*f[l-i]
        # conv_res[m] = sum_j p[j]*f[n-1-(m-j)] = sum_j p[j]*f[n-1-m+j]
        # We want: sum_l p[l]*f[l-i], which matches conv_res[m] with n-1-m = -i, i.e., m = n-1+i
        # So grad[i] = conv_res[n-1+i]
        grad = conv_res[n-1:n-1+n].copy()
    
    # Projected gradient in log-space
    g = np.where(np.isfinite(grad), grad, 0.0)
    lay = float(np.dot(b, g))
    grad_w = b * (g - lay)
    return grad_w, h, mx if h.size > 0 else 0.0


def _self_gradient_step(a, beta, lr, N, phase):
    """
    Compute Nesterov-gradient step for self-convolution minimization.
    Returns new a, gradient_w.
    """
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2*n-1]
    c = np.maximum(c, 0.0)
    mx = float(np.max(c)) if c.size > 0 else 0.0
    
    # Smoothed weights
    if mx > 0:
        x = (beta * c) / mx
        x -= float(np.max(x))
        e = np.exp(np.clip(x, -745, 0.0))
        se = float(e.sum())
        if se > 0 and np.isfinite(se):
            p = e / se
        else:
            p = np.zeros(c.size)
            p[int(np.argmax(c))] = 1.0
    else:
        p = np.ones(c.size, dtype=np.float64) / c.size
    
    # Gradient: d/db a_i of sum_k p_k * c[k]
    # c[k] = sum_j a[j]*a[k-j], dc[k]/da[i] = a[k-i] + a[i-k] (if valid)
    # In FFT form: grad = 2 * ifft(rfft(p) * phase * conj(A))  (for the real part)
    P = np.fft.rfft(p, N)
    conv = np.fft.irfft(P * phase * np.conjugate(A), N)
    grad = 2.0 * conv[n-1:2*n-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    
    lay = float(np.dot(a, grad))
    grad_w = a * (grad - lay)
    return grad_w, c, mx


def _adam_step(grad_w, w, m, v, t, lr):
    """Adam step in log-space with clipping."""
    t += 1
    b1, b2 = 0.9, 0.999
    m = m * b1 + (1 - b1) * grad_w
    v = v * b2 + (1 - b2) * (grad_w * grad_w)
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    dw = m_hat / (np.sqrt(v_hat) + 1e-8)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    dw = np.clip(dw, -20.0, 20.0)
    w = w - lr * dw
    w = np.where(np.isfinite(w), w, 0.0)
    w -= float(np.max(w)) if w.size > 0 else 0.0
    return _from_log_w(w), w, m, v, t


def _nesterov_momentum(v_mom, grad_w, t, mu=0.9):
    """Update Nesterov momentum."""
    v_new = mu * v_mom + grad_w
    return v_new


def _init_sequence(n, rng):
    """Generate diverse initial sequences."""
    method = rng.integers(0, 5)
    if method == 0:
        # Uniform
        return np.ones(n, dtype=np.float64) / n
    elif method == 1:
        # Raised cosine
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        return _normalize(a)
    elif method == 2:
        # Triangular
        t = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(t - 0.5))
        return _normalize(a)
    elif method == 3:
        # Gaussian centered
        t = np.arange(n) - n / 2.0
        a = np.exp(-0.5 * (t / (n * 0.25))**2)
        return _normalize(a)
    else:
        # Random with structure (low-freq dominant)
        A = rng.normal(0, 1, _next_pow2(2*n-1)//2+1) + 1j*rng.normal(0, 1, _next_pow2(2*n-1)//2+1)
        # Keep low frequencies strong
        mask = np.ones(len(A))
        cutoff = _next_pow2(2*n-1)//4 + 1
        mask[cutoff:] = 0.01
        A *= mask
        a = np.fft.irfft(A, _next_pow2(2*n-1))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            return a / s
        return np.ones(n) / n


def _load_parent():
    """Load parent sequence from global if available."""
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 16:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 0 or not np.isfinite(s):
        return None
    # Downsample if too long
    if arr.size > 12000:
        target = 4096
        blocks = arr.size // target
        if blocks > 1:
            use = target * blocks
            arr = np.mean(arr[:use].reshape(target, blocks), axis=1)
        else:
            arr = arr[:target]
        s = float(arr.sum())
        if s > 0:
            arr /= s
        else:
            arr = np.ones(4096) / 4096.0
    return arr


def _main_search(a0, seed, budget_s):
    """Main optimization loop."""
    global _start_time, _budget
    _start_time[0] = time.monotonic()
    _budget[0] = budget_s
    
    rng = np.random.default_rng(seed)
    
    n = len(a0)
    if n < 256:
        a0 = np.ones(4096, dtype=np.float64) / 4096.0
        n = 4096
    if n > 12000:
        a0 = a0[:8192]
        n = 8192
    
    N = _next_pow2(3 * n - 2)
    phase = np.exp(-2j * np.pi * np.arange(N // 2 + 1) * (n - 1) / float(N))
    
    a = _normalize(a0)
    c, score = _fft_conv(a, N)
    if not np.isfinite(score):
        a = np.ones(n) / n
        c, score = _fft_conv(a, N)
    
    best_a = a.copy()
    best_score = score
    
    # Log-space parameters
    w = _log_w(a)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    v_mom = np.zeros(n)  # Nesterov momentum
    
    beta = 5.0
    lr = 0.12
    mu = 0.85  # momentum
    
    stall = 0
    outer = 0
    exact_best_score = float('inf')
    exact_best_a = a.copy()
    exact_time = 0.0
    last_exact = _elapsed()
    
    # Cross-subproblem state
    b_cross = a.copy()
    w_cross = _log_w(b_cross)
    m_cross = np.zeros(n)
    v_cross = np.zeros(n)
    t_cross = 0
    
    # Track top candidates
    top_scores = [score]
    top_seqs = [a.copy()]
    
    print(f"Init: n={n}, N={N}, score={score:.6f}")
    
    while _remaining() > 30.0:
        outer += 1
        
        # Safety check
        if not np.all(np.isfinite(a)) or not np.isfinite(score):
            a = best_a.copy()
            score = best_score
            w = _log_w(a)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            v_mom = np.zeros(n)
            c, score = _fft_conv(a, N)
            stall = 0
            continue
        
        # --- Self-convolution gradient step (Nesterov) ---
        # Compute effective gradient with momentum
        grad_w, c_new, mx_new = _self_gradient_step(a, beta, lr, N, phase)
        if np.isfinite(mx_new) and mx_new < mx_new:  # always true, just for flow
            pass
        c = c_new
        score = 2.0 * n * mx_new if mx_new > 0 else 0.0
        if not np.isfinite(score):
            score = 2.0 * n * float(np.max(c)) if c.size > 0 else 0.0
        
        # Nesterov momentum update
        v_mom = _nesterov_momentum(v_mom, grad_w, outer, mu)
        eff_grad = grad_w + mu * v_mom
        
        # Adam step
        a_new, w, m_adam, v_adam, t_adam = _adam_step(eff_grad, w, m_adam, v_adam, t_adam, lr)
        
        # Evaluate
        c_n, sc_new = _fft_conv(a_new, N)
        
        # Accept or reject (with slight tolerance)
        if np.isfinite(sc_new) and sc_new <= score * (1 + 1e-6):
            a = a_new
            c = c_n
            score = sc_new
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
                top_scores.append(score)
                top_seqs.append(a.copy())
                if len(top_scores) > 8:
                    # Remove worst
                    worst_i = np.argmax(top_scores[:7])
                    del top_scores[worst_i]
                    del top_seqs[worst_i]
                stall = 0
            elif score < best_score + 0.001:
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            lr = max(0.02, lr * 0.95)
        
        # --- Cross-subproblem step (every 3 outer iterations) ---
        if outer % 3 == 0:
            for _ in range(2):
                grad_cross, h_cross, mx_cross = _water_filling_step(a, b_cross, beta * 2, 0.08, N, n)
                if np.any(np.isfinite(grad_cross)):
                    b_cross, w_cross, m_cross, v_cross, t_cross = _adam_step(
                        grad_cross, w_cross, m_cross, v_cross, t_cross, 0.08
                    )
                    # Try blending
                    for t_val in (0.5, 0.8, 1.0):
                        blend = (1 - t_val) * a + t_val * b_cross
                        blend = _normalize(blend)
                        _, sc_blend = _fft_conv(blend, N)
                        if np.isfinite(sc_blend) and sc_blend < score:
                            a = blend
                            score = sc_blend
                            w = _log_w(a)
                            m_adam = np.zeros(n)
                            v_adam = np.zeros(n)
                            t_adam = 0
                            v_mom = np.zeros(n)
                            if score < best_score:
                                best_score = score
                                best_a = a.copy()
                                top_scores.append(score)
                                top_seqs.append(a.copy())
                                if len(top_scores) > 8:
                                    worst_i = np.argmax(top_scores[:7])
                                    del top_scores[worst_i]
                                    del top_seqs[worst_i]
                                stall = 0
            
            # Sync b_cross if too different
            if np.max(np.abs(b_cross - a)) > 1.0:
                b_cross = a.copy()
                w_cross = _log_w(b_cross)
                m_cross = np.zeros(n)
                v_cross = np.zeros(n)
                t_cross = 0
        
        # --- Adaptive beta schedule ---
        if outer % 30 == 0:
            beta = min(80.0, beta * 1.06 + 0.3)
        
        # --- Learning rate schedule ---
        if stall % 10 == 0 and stall > 0:
            lr = max(0.03, lr * 0.92)
        
        # --- Restart on long stall ---
        if stall > 50:
            # Pick restart strategy
            strat = rng.integers(0, 3)
            if strat == 0:
                # Go back to best and perturb
                a = best_a.copy()
                score = best_score
                sig = float(rng.choice([0.1, 0.2, 0.3, 0.5]))
                w_try = _log_w(a) + rng.normal(0, sig, size=n)
                a = _from_log_w(w_try)
                c, score = _fft_conv(a, N)
            elif strat == 1:
                # Blend with a new structured init
                a_new_struct = _init_sequence(n, rng)
                a = 0.5 * a + 0.5 * a_new_struct
                a = _normalize(a)
                c, score = _fft_conv(a, N)
            else:
                # Full restart with new init
                a = _init_sequence(n, rng)
                c, score = _fft_conv(a, N)
                if not np.isfinite(score) or score > 5.0:
                    a = best_a.copy()
                    score = best_score
                    c, score = _fft_conv(a, N)
            
            w = _log_w(a)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            v_mom = np.zeros(n)
            b_cross = a.copy()
            w_cross = _log_w(b_cross)
            m_cross = np.zeros(n)
            v_cross = np.zeros(n)
            t_cross = 0
            stall = 0
            lr = 0.10
            beta = max(5.0, beta * 0.5)
        
        # --- Exact validation (throttled) ---
        if N <= 32768 and _elapsed() - last_exact > 45.0:
            t0 = time.monotonic()
            sc_ex = _exact_score(best_a)
            exact_time += time.monotonic() - t0
            last_exact = _elapsed()
            if np.isfinite(sc_ex) and sc_ex < exact_best_score:
                exact_best_score = sc_ex
                exact_best_a = best_a.copy()
                if outer % 50 == 0:
                    print(f"  exact={exact_best_score:.6f} outer={outer} t={_elapsed():.0f}s")
        
        # Progress print
        if outer % 100 == 0:
            print(f"outer={outer} score={score:.6f} best={best_score:.6f} beta={beta:.1f} lr={lr:.4f} stall={stall} t={_elapsed():.0f}s")
    
    # Final phase: refine best with exact scores
    print(f"Main loop done. best_score={best_score:.6f} exact_best={exact_best_score:.6f}")
    
    # Do a few more local refinements on the best
    if _remaining() > 10.0:
        a = best_a.copy()
        w = _log_w(a)
        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
        t_adam = 0
        v_mom = np.zeros(n)
        lr_final = 0.03
        beta_final = 60.0
        
        for i in range(200):
            if _remaining() < 10.0:
                break
            grad_w, c_f, mx_f = _self_gradient_step(a, beta_final, lr_final, N, phase)
            v_mom = _nesterov_momentum(v_mom, grad_w, i, 0.9)
            eff_grad = grad_w + 0.9 * v_mom
            a_new, w, m_adam, v_adam, t_adam = _adam_step(eff_grad, w, m_adam, v_adam, t_adam, lr_final)
            _, sc_new = _fft_conv(a_new, N)
            if np.isfinite(sc_new) and sc_new < score:
                a = a_new
                score = sc_new
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        # Final exact check
        if _remaining() > 5.0:
            sc_ex = _exact_score(best_a)
            if np.isfinite(sc_ex) and sc_ex < exact_best_score:
                exact_best_score = sc_ex
                exact_best_a = best_a.copy()
            # Also check the other candidates
            for seq in top_seqs:
                if _remaining() < 5.0:
                    break
                sc_c = _exact_score(seq)
                if np.isfinite(sc_c) and sc_c < exact_best_score:
                    exact_best_score = sc_c
                    exact_best_a = seq.copy()
    
    print(f"Final: exact_best={exact_best_score:.6f}")
    
    if np.isfinite(exact_best_score):
        return exact_best_a
    return best_a


def _finalize(a):
    """Scale sequence to pass evaluator constraints."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    scale = math.sqrt(2.0 * n)
    if s > 0 and np.isfinite(s):
        a = a * scale / s
    else:
        a = np.ones(n) * (scale / n)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = a * max(1.0, 0.01 / s2)
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Entry point: runs the optimizer and returns the best sequence."""
    _start_time[0] = time.monotonic()
    budget = min(max(0.0, float(budget_s)), 2395.0)
    
    # Load parent or initialize
    parent = _load_parent()
    rng = np.random.default_rng(seed)
    
    if parent is not None and parent.size >= 256:
        a0 = _normalize(parent)
    else:
        n = 4096
        a0 = np.ones(n, dtype=np.float64) / n
        # Add small structured perturbation
        t = np.arange(n)
        a0 += 0.05 * np.sin(2 * np.pi * t / n * 3)
        a0 = _normalize(a0)
    
    print(f"Starting search: n={len(a0)}, budget={budget:.0f}s")
    
    result = _main_search(a0, seed, budget - 8.0)
    result_seq = _finalize(result)
    
    # Quick validation
    if len(result_seq) > 0:
        print(f"Output: len={len(result_seq)}, sum={sum(result_seq):.4f}, max={max(result_seq):.6f}")
    
    return result_seq
