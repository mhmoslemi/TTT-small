    # I want g[i] = sum_{k=i}^{min(i+n-1, 2n-2)} w[k] * a[k-i]
    # This is the auto/cross correlation.
    # 
    # Using FFT: Let W = IFFT(FFT(w, L) * conj(FFT(a, L))) where L >= 3n-2
    # W[m] = sum_k w[k+m] * a[k] (circular correlation)
    # For linear: W[m] = sum_k w[k+m] * a[k] for m >= 0
    # = sum_k w[k] * a[k-m] for m >= 0 (by substitution)
    # 
    # Wait, I want g[i] = sum_k w[k] * a[k-i]
    # = sum_j w[j+i] * a[j] (setting j = k-i)
    # 
    # The cross-correlation (w ⊛ a)[i] = sum_j w[j+i] * conj(a[j]) = sum_j w[j+i] * a[j] (real)
    # This equals IFFT(conj(FFT(w)) * FFT(a))[i] ... no.
    #

    # In FFT: IFFT(FFT(f) * conj(FFT(g)))

    # For L >> n: = sum_k sum_j w[k] a[j] * delta(j-k+i)
    # = sum_k w[k] a[k-i]
    # = g[i] ✓

    A_fft = np.fft.rfft(a, L)
    corr = np.fft.irfft(W_fft * np.conj(A_fft), L)
    grad_raw = 4.0 * n * corr[:n]
    
    # Project to simplex tangent space (maintain sum=1 constraint)
    # In the direction a -> a + ε*grad, we need sum(grad) = 0 or project
    # For softmax parameterization, the effective gradient is:
    # grad_eff[i] = a[i] * (grad_raw[i] - sum_j a[j]*grad_raw[j])
    
    dot = float(np.dot(a, grad_raw))
    grad_eff = a * (grad_raw - dot)
    
    return score, grad_eff

"""
Hybrid optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based smooth-max gradient descent (Adam in log-space) with
multiple diverse initializations, adaptive beta sharpening, and
periodic exact validation. Explores lengths 4096 and 8192.
"""
import numpy as np
import time

def _next_fast_len(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def _normalize_nonneg(v):
    arr = np.asarray(v, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 0.0:
        arr = np.ones(max(1, arr.size))
        s = float(arr.sum())
    return arr / s

def _fft_conv(a, N):
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2 * n - 1]
    if c.size and float(np.min(c)) < 0.0:
        c = np.maximum(c, 0.0)
    return c

def _smooth_score_grad(a, beta, N):
    """Smooth score and simplex-tangent gradient via FFT."""
    n = len(a)
    c = _fft_conv(a, N)
    if c.size == 0:
        return 999.0, np.zeros(n)
    c_max = float(np.max(c))
    if not np.isfinite(c_max):
        return 999.0, np.zeros(n)
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    smooth_val = c_max + np.log(max(s, 1e-300)) / beta
    score = 2.0 * n * smooth_val

    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    grad_raw = 4.0 * n * corr[:n]
    grad_raw = np.where(np.isfinite(grad_raw), grad_raw, 0.0)
    dot = float(np.dot(a, grad_raw))
    grad_eff = a * (grad_raw - dot)
    return score, grad_eff

def _adam_log(a, grad_eff, w, m, v, t, lr):
    t += 1
    b1, b2 = 0.9, 0.999
    m = b1 * m + (1 - b1) * grad_eff
    v = b2 * v + (1 - b2) * (grad_eff * grad_eff)
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    dw = m_hat / (np.sqrt(v_hat) + 1e-8)
    dw = np.clip(dw, -8.0, 8.0)
    w = w - lr * dw
    w = np.where(np.isfinite(w), w, 0.0)
    w -= float(np.max(w))
    ex = np.exp(np.clip(w, -700, 700))
    s = float(ex.sum())
    if s <= 0.0 or not np.isfinite(s):
        a = np.ones(len(w), dtype=np.float64) / len(w)
        w = np.zeros(len(w))
    else:
        a = ex / s
    return a, w, m, v, t

def _triangular(n):
    t = np.ones(n)
    half = n // 2
    ramp = np.linspace(0, 1, half + 1)
    t[:half + 1] = ramp
    t[half:] = 1 - ramp[:n - half][::-1]
    # Ensure non-negative
    t = np.maximum(t, 0.0)
    if t.sum() <= 0:
        t = np.ones(n)
    return t / t.sum()

def _gaussian(n, sigma_frac=0.25):
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    g = np.maximum(g, 1e-30)
    return g / g.sum()

def _resample(a, target_n):
    """Resample sequence to target length via block averaging or interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    if n < target_n:
        # Upsample: repeat with linear interp
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, target_n)
        a = np.interp(x_new, x_old, a)
    else:
        # Downsample: block average
        blocks = n // target_n
        use = blocks * target_n
        a = a[:use].reshape(target_n, blocks).mean(axis=1)
        if a.size < target_n:
            a = np.concatenate([a, a[:target_n - a.size]])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(target_n) / target_n
    return a

def _exact_score(a):
    n = len(a)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return float('inf')
    return float(2.0 * n * mx / (s * s))

        a = np.clip(a, 0.0, 1000.0)
    
    print(f"Final best score: {best_overall_score:.6f}, length={n}", flush=True)
    return [float(x) for x in a]
