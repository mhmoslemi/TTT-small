"""
Search for nonnegative coefficient sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses IRLS-inspired adaptive weighting, FFT-based evaluation, and multi-phase optimization.
"""

import numpy as np
import time

def next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def normalize(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(v.size if v.size > 0 else 1) / max(1, v.size if v.size > 0 else 1)
    return v / s

def log_prob(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.maximum(v, 1e-300)
    w = np.log(v)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= w.max()
    return w

def from_log_prob(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    w = np.where(np.isfinite(w), w, 0.0)
    w -= w.max()
    e = np.exp(w)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(w.size) / w.size
    return e / s

def fft_conv(f):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(f)
    N = next_pow2(2 * n - 1)
    F = np.fft.rfft(f, N)
    c = np.fft.irfft(F * F, N)
    return c[:2*n-1]

def fft_cross_conv(f, g):
    """Cross-convolution via FFT. Returns array of length 2n-1 (f and g same length)."""
    n = len(f)
    N = next_pow2(2 * n - 1)
    F = np.fft.rfft(f, N)
    G = np.fft.rfft(g, N)
    c = np.fft.irfft(F * G, N)
    return c[:2*n-1]

def fft_cross_corr(f, p):
    """Cross-correlation: (p * f)_j = sum_k p_k f_{k-j}, returned for j=0..n-1."""
    n = len(f)
    N = next_pow2(2 * n - 1)
    F = np.fft.rfft(f, N)
    P = np.fft.rfft(p, N)
    # cross-correlation: IFFT(conj(P) * F)
    cc = np.fft.irfft(np.conj(P) * F, N)
    return cc[:n]  # we only need the first n elements

def score_fft(f):
    n = len(f)
    c = fft_conv(f)
    if c.size == 0:
        return float('inf')
    mx = max(c.max(), 0)
    return 2.0 * n * mx  # assuming sum(f) = 1

def score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return float('inf')
    b = np.convolve(a, a)
    mx = b.max()
    val = 2.0 * n * mx / (s * s)
    return val if np.isfinite(val) else float('inf')

"""
Search for nonnegative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS-B with log-sum-exp smoothing, multi-start exploration,
and adaptive annealing.
"""

import numpy as np
import time
from scipy.optimize import minimize

# Global params for the optimizer callback
_G = {}

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(max(1, v.size)) / max(1, v.size)
    return v / s

def _log_prob(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.maximum(v, 1e-300)
    w = np.log(v)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= w.max()
    return w

def _from_log_prob(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    w = np.where(np.isfinite(w), w, 0.0)
    w -= w.max()
    e = np.exp(w)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(w.size) / w.size
    return e / s

def _fft_self_conv(f, N):
    F = np.fft.rfft(f, N)
    c = np.fft.irfft(F * F, N)
    return c[:2*len(f)-1], F

def _score_fft(f, N):
    c, _ = _fft_self_conv(f, N)
    mx = max(c.max(), 0.0)
    if not np.isfinite(mx):
        return float('inf')
    return 2.0 * len(f) * mx

def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = a.sum()
    if s < 0.01 or not np.isfinite(s):
        return float('inf')
    b = np.convolve(a, a)
    val = 2.0 * n * b.max() / (s * s)
    return val if np.isfinite(val) else float('inf')

def _obj_func(w):
    """Objective and gradient for L-BFGS-B."""
    N = _G['N']
    n = _G['n']
    beta = _G['beta']
    
    f = _from_log_prob(w)
    F = np.fft.rfft(f, N)
    c = np.fft.irfft(F * F, N)
    c = c[:2*n-1]
    c = np.maximum(c, 0.0)
    
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.minimum(shifted, 0.0)
    e = np.exp(shifted)
    e_sum = e.sum()
    
    if e_sum <= 0 or not np.isfinite(e_sum):
        return 1e10, np.zeros(n)
    
    log_se = c_max + np.log(e_sum) / beta
    p = e / e_sum
    
    val = 2.0 * n * log_se
    
    P = np.fft.rfft(p, N)
    cc = np.fft.irfft(P * np.conj(F), N)
    grad_f = cc[:n] + f
    dot_f = np.dot(f, grad_f)
    grad_w = 2.0 * n * f * (grad_f - dot_f)
    
    grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)
    
    if not np.isfinite(val):
        return 1e10, np.zeros(n)
    
    return val, grad_w

def _lbfgs_run(f, N, n, beta, max_time, max_iter=500):
    """Run L-BFGS-B with time limit."""

    _G['n'] = n

        method='L-BFGS-B',

            'ftol': 1e-12,
            'gtol': 1e-6,
            'maxfun': 2000,
            'callback': _callback
        }
    )
    
    f_new = _from_log_prob(result.x)
    return f_new, result

def _load_parent():
    obj = globals().get('height_sequence_1', None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except:

    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return arr / s

def _prepare_initial(parent, target_n, rng):
    if parent is not None:
        n_orig = parent.size
        if n_orig == target_n:
            return parent.copy()
        elif n_orig > target_n:
            # Downsample by averaging blocks
            blocks = n_orig // target_n
            if blocks > 1:
                use = target_n * blocks
                return _normalize(parent[:use].reshape(target_n, blocks).mean(axis=1))
            else:
                return _normalize(parent[:target_n])
        else:
            # Upsample by interpolation
            x_orig = np.linspace(0, 1, n_orig)
            x_new = np.linspace(0, 1, target_n)
            return _normalize(np.interp(x_new, x_orig, parent))
    else:
        return _normalize(np.ones(target_n))

def _finalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        a = np.ones(64)
        n = 64
        s = 64.0
    # Scale to a reasonable magnitude
    target_sum = np.sqrt(2.0 * n)
    a = a * target_sum / s
    a = np.clip(a, 0.0, 1000.0)
