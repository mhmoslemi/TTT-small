import numpy as np
import time
import math


EPS = 1e-300
INF = float("inf")


def safe_float(x):
    try:
        y = float(x)
    except Exception:
        return INF
    return y if np.isfinite(y) else INF


def next_fast_len(m):
    m = int(max(1, m))
    p = 1
    while p < m:
        p <<= 1
    return p


def softmax_1d(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size == 0:
        return np.ones(1, dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w = w - float(np.max(w))
    e = np.exp(np.clip(w, -745.0, 0.0))
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(w.size, dtype=np.float64) / float(w.size)
    return e / s


def softmax_2d(wm):
    wm = np.asarray(wm, dtype=np.float64)
    if wm.ndim != 2:
        wm = np.atleast_2d(wm)
    if wm.size == 0:
        return np.ones((1, 1), dtype=np.float64)
    wm = np.where(np.isfinite(wm), wm, 0.0)
    wm = wm - np.max(wm, axis=1, keepdims=True)
    e = np.exp(np.clip(wm, -745.0, 0.0))
    s = np.sum(e, axis=1, keepdims=True)
    bad = ~(np.isfinite(s) & (s > 0))
    if np.any(bad):
        s = np.where(bad, 1.0, s)
        e = np.where(bad[:, None], 0.0, e)
    out = e / s
    bad = ~(np.isfinite(s) & (s > 0))
    if np.any(bad):
        out[bad] = 1.0 / float(out.shape[1])
    return out


def log_weights_from_p(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return np.zeros(1, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, EPS)
    w = np.log(p)
    w = np.where(np.isfinite(w), w, 0.0)
    w = w - float(np.max(w))
    return w


def normalize_nonneg(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0:
        v = np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)

def interp_profile(x, target_len):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.zeros(target_len, dtype=np.float64)
    target_len = int(target_len)
    if x.size == target_len:
        return x.copy()
    if x.size == 1:
        return np.full(target_len, float(x[0]), dtype=np.float64)
    old = np.linspace(0.0, 1.0, x.size)
    new = np.linspace(0.0, 1.0, target_len)
    return np.interp(new, old, x)


def fft_score_single(p, fftlen):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return INF, np.zeros(0), np.zeros(0)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return INF, np.zeros(2 * n - 1), np.zeros(fftlen // 2 + 1)
    p = p / s
    L = 2 * n - 1
    fl = fftlen if fftlen >= L else next_fast_len(L)
    A = np.fft.rfft(p, fl)
    c = np.fft.irfft(A * A, fl)
    c = c[:L]
    if c.size:
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        mx = float(np.max(c))
    else:
        mx = 0.0
    if not np.isfinite(mx):
        return INF, c, A
    return float(2.0 * n * mx), c, A


def make_q_single(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return np.zeros(0, dtype=np.float64)
    mx = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mx) or mx <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    x = (c - mx) / mx
    x = np.where(np.isfinite(x), x, -1.0)
    x = np.clip(x, -1.0, 0.0)
    y = np.clip(float(beta) * x, -745.0, 0.0)
    e = np.exp(y)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        q = np.zeros(L, dtype=np.float64)
        q[int(np.argmax(c))] = 1.0
        return q
    return e / s


def make_q_batch(cmat, beta):
    cmat = np.asarray(cmat, dtype=np.float64)
    if cmat.ndim != 2:
        cmat = np.atleast_2d(cmat)
    B = cmat.shape[0]
    L = cmat.shape[1]
    if B == 0 or L == 0:
        return np.zeros_like(cmat, dtype=np.float64)
    mx = np.max(cmat, axis=1)
    valid = np.isfinite(mx) & (mx > 0.0)
    denom = np.where(mx > 0.0, mx, 1.0)
    x = (cmat - mx[:, None]) / denom[:, None]
    x = np.where(np.isfinite(x), x, -1.0)
    x = np.clip(x, -1.0, 0.0)
    y = np.clip(float(beta) * x, -745.0, 0.0)
    e = np.exp(y)
    s = np.sum(e, axis=1, keepdims=True)
    good = valid & (s[:, 0] > 0.0)
    e_safe = np.where(s > 0.0, s, 1.0)
    q = np.where(good[:, None], e / e_safe, 0.0)
    if not np.all(good):
        idx = np.argmax(cmat, axis=1)
        bad = ~good
        q[bad, :] = 0.0
        q[bad, idx[bad]] = 1.0
    return q


def self_grad_single(p, A, c, beta, fftlen):
    p = np.asarray(p, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    n = p.size
    L = 2 * n - 1
    q = make_q_single(c, beta)
    qr = np.zeros(fftlen, dtype=np.float64)
    qr[:L] = q[::-1]
    Qr = np.fft.rfft(qr, fftlen)
    H = np.fft.irfft(A * Qr, fftlen)
    G = 2.0 * np.asarray(H[n - 1:2 * n - 1][::-1], dtype=np.float64)
    G = np.where(np.isfinite(G), G, 0.0)
    G = np.maximum(G, 0.0)
    dot = float(np.dot(p, G))
    grad = p * (G - dot)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def adam_step_1d(w, grad, m, v, t, lr):
    t = int(t) + 1
    b1 = 0.9
    b2 = 0.999
    w = np.asarray(w, dtype=np.float64)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    m = m * b1 + (1.0 - b1) * grad
    v = v * b2 + (1.0 - b2) * (grad * grad)
    mhat = m / (1.0 - b1 ** t)
    vhat = v / (1.0 - b2 ** t)
    dw = mhat / (np.sqrt(vhat) + 1e-12)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    dw = np.clip(dw, -20.0, 20.0)
    w_new = w - float(lr) * dw
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    w_new = w_new - float(np.max(w_new))
    return w_new, m, v, t


def adam_step_2d(W, grad, m, v, t, lr):
    t = int(t) + 1
    b1 = 0.9
    b2 = 0.999
    W = np.asarray(W, dtype=np.float64)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    m = m * b1 + (1.0 - b1) * grad
    v = v * b2 + (1.0 - b2) * (grad * grad)
    mhat = m / (1.0 - b1 ** t)
    vhat = v / (1.0 - b2 ** t)
    dw = mhat / (np.sqrt(vhat) + 1e-12)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    dw = np.clip(dw, -10.0, 10.0)
    W_new = W - float(lr) * dw
    W_new = np.where(np.isfinite(W_new), W_new, 0.0)
    W_new = W_new - np.max(W_new, axis=1, keepdims=True)
    return W_new, m, v, t


def candidate_from_w(w, fftlen):
    w = np.asarray(w, dtype=np.float64).ravel()
    w = np.where(np.isfinite(w), w, 0.0)
    if w.size == 0:
        w = np.zeros(1, dtype=np.float64)
    w = w - float(np.max(w))
    p = softmax_1d(w)
    n = p.size
    L = 2 * n - 1
    fl = fftlen if fftlen >= L else next_fast_len(L)
    A = np.fft.rfft(p, fl)
    c = np.fft.irfft(A * A, fl)
    c = c[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    s = float(np.sum(p))
    if c.size and s > 0.0:
        mx = float(np.max(c))
        score = float(2.0 * n * mx / (s * s)) if mx > 0.0 else 0.0
    else:
        score = INF
    if not np.isfinite(score):
        p = np.ones(n, dtype=np.float64) / float(n)
        w = np.zeros(n, dtype=np.float64)
        A = np.fft.rfft(p, fl)
        c = np.fft.irfft(A * A, fl)
        c = c[:L]
        c = np.maximum(c, 0.0)
        score = float(2.0 * n * float(np.max(c)))
    return {
        "n": n,
        "fftlen": fl,
        "w": w.copy(),
        "p": p.copy(),
        "c": c,
        "A": A,
        "score": safe_float(score),
        "best_score": safe_float(score),
        "best_p": p.copy(),
        "best_w": w.copy(),
        "m_f": np.zeros(n, dtype=np.float64),
        "v_f": np.zeros(n, dtype=np.float64),
        "t_f": 0,
        "b": p.copy(),
        "w_b": w.copy(),
        "m_b": np.zeros(n, dtype=np.float64),
        "v_b": np.zeros(n, dtype=np.float64),
        "t_b": 0,
        "beta_self": 6.0,
        "lr_self": 0.05,
        "beta_cross": 10.0,
        "lr_cross": 0.08,
        "stall": 0,
    }


def cross_grad_b_single(f, b, A, c_h, beta, fftlen):
    b = np.asarray(b, dtype=np.float64).ravel()
    c_h = np.asarray(c_h, dtype=np.float64).ravel()
    n = b.size
    L = 2 * n - 1
    q = make_q_single(c_h, beta)
    qr = np.zeros(fftlen, dtype=np.float64)
    qr[:L] = q[::-1]
    Qr = np.fft.rfft(qr, fftlen)
    H = np.fft.irfft(A * Qr, fftlen)
    G = np.asarray(H[n - 1:2 * n - 1][::-1], dtype=np.float64)
    G = np.where(np.isfinite(G), G, 0.0)
    G = np.maximum(G, 0.0)
    dot = float(np.dot(b, G))
    grad = b * (G - dot)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def update_self_candidate(cand, temp=0.0):
    n = cand["n"]
    fl = cand["fftlen"]
    if (not np.all(np.isfinite(cand["p"]))) or (not np.all(np.isfinite(cand["w"]))):
        cand["p"] = np.ones(n, dtype=np.float64) / float(n)
        cand["w"] = np.zeros(n, dtype=np.float64)
        cand["score"] = float(2.0 * n / float(n))
        return False
    grad = self_grad_single(cand["p"], cand["A"], cand["c"], cand["beta_self"], fl)
    if not np.all(np.isfinite(grad)) or float(np.max(np.abs(grad))) == 0.0:
        cand["stall"] += 1
        return False
    new_w, m_f, v_f, t_f = adam_step_1d(
        cand["w"], grad, cand["m_f"], cand["v_f"], cand["t_f"], cand["lr_self"]
    )
    p_new = softmax_1d(new_w)
    score, c_new, A_new = fft_score_single(p_new, fl)
    score = safe_float(score)
    if np.isfinite(score) and score < cand["score"] + max(0.0, safe_float(temp)):
        cand["w"] = new_w
        cand["p"] = p_new
        cand["c"] = c_new
        cand["A"] = A_new
        cand["score"] = score
        cand["m_f"] = m_f
        cand["v_f"] = v_f
        cand["t_f"] = t_f
        cand["b"] = p_new.copy()
        cand["w_b"] = new_w.copy()
        cand["m_b"] = np.zeros(n, dtype=np.float64)
        cand["v_b"] = np.zeros(n, dtype=np.float64)
        cand["t_b"] = 0
        cand["stall"] = 0
        cand["lr_self"] = min(0.18, cand["lr_self"] * 1.035)
        cand["beta_self"] = min(220.0, cand["beta_self"] * 1.06 + 0.1)
        if score < cand["best_score"]:
            cand["best_score"] = score
            cand["best_p"] = p_new.copy()
            cand["best_w"] = new_w.copy()
        return True
    else:
        cand["stall"] += 1
        cand["lr_self"] = max(0.008, cand["lr_self"] * 0.95)
        if cand["stall"] > 8:
            cand["beta_self"] = min(300.0, cand["beta_self"] * 1.12 + 1.0)
        return False


def update_cross_candidate(cand):
    n = cand["n"]
    fl = cand["fftlen"]
    f = cand["p"]
    A = cand["A"]
    b = cand["b"]
    wb = cand["w_b"]
    if (not np.all(np.isfinite(f))) or (not np.all(np.isfinite(A))):
        return False
    if (not np.all(np.isfinite(b))) or float(np.sum(b)) <= 0.0:
        b = f.copy()
        wb = log_weights_from_p(f)
    mb = cand["m_b"]
    vb = cand["v_b"]
    tb = cand["t_b"]

    for _ in range(2):
        Bf = np.fft.rfft(b, fl)
        hh = np.fft.irfft(A * Bf, fl)
        L = 2 * n - 1
        hh = hh[:L]
        hh = np.where(np.isfinite(hh), hh, 0.0)
        hh = np.maximum(hh, 0.0)
        if float(np.max(hh)) <= 0.0:
            break
        grad = cross_grad_b_single(f, b, A, hh, cand["beta_cross"], fl)
        if not np.all(np.isfinite(grad)) or float(np.max(np.abs(grad))) == 0.0:
            break
        wb, mb, vb, tb = adam_step_1d(wb, grad, mb, vb, tb, cand["lr_cross"])
        b = softmax_1d(wb)

    cand["b"] = b
    cand["w_b"] = wb
    cand["m_b"] = mb
    cand["v_b"] = vb
    cand["t_b"] = tb

    best_t = 0.0
    best_p = f
    best_c = cand["c"]
    best_A = A
    best_score = cand["score"]

    for tval in (0.25, 0.5, 0.75, 1.0):
        p_try = (1.0 - tval) * f + tval * b
        s_try = float(np.sum(p_try))
        if not np.isfinite(s_try) or s_try <= 0.0:
            continue
        p_try = p_try / s_try
        sc, c_try, A_try = fft_score_single(p_try, fl)
        sc = safe_float(sc)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_p = p_try
            best_c = c_try
            best_A = A_try
            best_t = tval

    if best_t > 0.0 and np.isfinite(best_score) and best_score < cand["score"]:
        cand["p"] = best_p
        cand["w"] = log_weights_from_p(best_p)
        cand["c"] = best_c
        cand["A"] = best_A
        cand["score"] = best_score
        cand["m_f"] = np.zeros(n, dtype=np.float64)
        cand["v_f"] = np.zeros(n, dtype=np.float64)
        cand["t_f"] = 0
        cand["b"] = best_p.copy()
        cand["w_b"] = cand["w"].copy()
        cand["m_b"] = np.zeros(n, dtype=np.float64)
        cand["v_b"] = np.zeros(n, dtype=np.float64)
        cand["t_b"] = 0
        cand["stall"] = 0
        cand["lr_cross"] = min(0.22, cand["lr_cross"] * 1.04)
        cand["beta_cross"] = min(200.0, cand["beta_cross"] * 1.05 + 0.2)
        if best_score < cand["best_score"]:
            cand["best_score"] = best_score
            cand["best_p"] = best_p.copy()
            cand["best_w"] = cand["w"].copy()
        return True

    cand["stall"] += 1
    cand["lr_cross"] = max(0.010, cand["lr_cross"] * 0.95)
    if cand["stall"] > 6:
        cand["beta_cross"] = min(280.0, cand["beta_cross"] * 1.10 + 1.0)
    return False


def symmetrize_candidate(cand):
    n = cand["n"]
    fl = cand["fftlen"]
    p = 0.5 * (cand["p"] + cand["p"][::-1])
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return False
    p = p / s
    score, c_new, A_new = fft_score_single(p, fl)
    score = safe_float(score)
    if np.isfinite(score) and score < cand["score"] + 1e-13:
        if score < cand["score"] or score < cand["best_score"]:
            cand["p"] = p
            cand["w"] = log_weights_from_p(p)
            cand["c"] = c_new
            cand["A"] = A_new
            cand["score"] = score
            cand["m_f"] = np.zeros(n, dtype=np.float64)
            cand["v_f"] = np.zeros(n, dtype=np.float64)
            cand["t_f"] = 0
            cand["b"] = p.copy()
            cand["w_b"] = cand["w"].copy()
            cand["m_b"] = np.zeros(n, dtype=np.float64)
            cand["v_b"] = np.zeros(n, dtype=np.float64)
            cand["t_b"] = 0
            cand["stall"] = 0
            if score < cand["best_score"]:
                cand["best_score"] = score
                cand["best_p"] = p.copy()
                cand["best_w"] = cand["w"].copy()
            return True
    return False


def order_by_score(scores):
    scores = list(scores)
    n = len(scores)
    if n <= 1:
        return list(range(n))
    safe = [safe_score for safe_score in scores]
    idx = list(range(n))
    for i in range(n):
        minj = i
        minv = safe[idx[i]]
        for j in range(i + 1, n):
            v = safe[idx[j]]
            if v < minv:
                minv = v
                minj = j
        if i != minj:
            idx[i], idx[minj] = idx[minj], idx[i]
    return idx


def keep_best_candidates(cands, max_keep):
    cands = [c for c in cands if c is not None]
    if len(cands) <= max_keep:
        return cands
    scores = [c.get("best_score", INF) for c in cands]
    order = order_by_score(scores)
    out = []
    for i in order[:max_keep]:
        out.append(cands[i])
    return out


def basis_search(nb, allowed_s, seed, m=28, B=48):
    start = time.monotonic()
    deadline = start + max(0.0, float(allowed_s))
    if nb < 64 or allowed_s < 25.0:
        return []
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, nb)
    ks = np.arange(m, dtype=np.float64)
    Phi = np.cos(np.pi * ks[None, :] * x[:, None])
    targets = []
    idx = np.arange(nb, dtype=np.float64)

    for alpha in (0.25, 0.45, 0.55, 0.75):
        t1 = -alpha * np.log(idx + 1.0)
        t2 = -alpha * np.log(nb - idx)
        t1 = t1 - float(np.mean(t1))
        t2 = t2 - float(np.mean(t2))
        t3 = np.logaddexp(t1, t2) - math.log(2.0)
        t3 = t3 - float(np.mean(t3))
        targets.extend([t1, t2, t3])

    for a in (-1.5, -1.0, -0.4, 0.4, 1.0, 1.5):
        t = a * (x - 0.5)
        t = t - float(np.mean(t))
        targets.append(t)

    for a in (-1.5, -0.8, -0.2, 0.2, 0.8, 1.5):
        t = a * ((x - 0.5) ** 2)
        t = t - float(np.mean(t))
        targets.append(t)

    for a in (-0.8, -0.4, 0.4, 0.8):
        t = a * (np.sin(np.pi * x) ** 2)
        t = t - float(np.mean(t))
        targets.append(t)

    for _ in range(14):
        t = rng.normal(0.0, 0.45, nb)
        t = t - float(np.mean(t))
        targets.append(t)

    try:
        Pinv = np.linalg.pinv(Phi)
    except Exception:
        Pinv = None

    Wrows = []
    for t in targets:
        t = np.asarray(t, dtype=np.float64).ravel()
        if t.size != nb:
            continue
        t = np.where(np.isfinite(t), t, 0.0)
        t = t - float(np.mean(t))
        if Pinv is not None:
            wrow = Pinv @ t
        else:
            try:
                wrow = np.linalg.lstsq(Phi, t, rcond=None)[0]
            except Exception:
                wrow = rng.normal(0.0, 0.35, m)
        wrow = np.asarray(wrow, dtype=np.float64)
        wrow = np.where(np.isfinite(wrow), wrow, 0.0)
        norm = float(np.linalg.norm(wrow))
        if not np.isfinite(norm) or norm <= 0.0:
            wrow = rng.normal(0.0, 0.3, m)
        elif norm > 2.5:
            wrow *= 2.5 / norm
        Wrows.append(wrow)
        if len(Wrows) >= B:
            break

    while len(Wrows) < B:
        Wrows.append(rng.normal(0.0, 0.35, m))

    W = np.vstack(Wrows[:B]).astype(np.float64)
    W = W - np.max(W, axis=1, keepdims=True)
    m_adam = np.zeros_like(W)
    v_adam = np.zeros_like(W)
    fftlen = next_fast_len(2 * nb - 1)
    L = 2 * nb - 1
    beta = 6.0
    lr = 0.09
    step = 0
    best_W = W.copy()
    best_score = np.full(B, INF, dtype=np.float64)
    last_improve = start
    global_best = INF

    while time.monotonic() < deadline:
        wm = W @ Phi.T
        p = softmax_2d(wm)
        s = np.sum(p, axis=1, keepdims=True)
        s = np.where((s > 0.0) & np.isfinite(s), s, 1.0)
        p = p / s
        P = np.fft.rfft(p, fftlen)
        c = np.fft.irfft(P * P, fftlen)
        c = c[:, :L]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        scores = 2.0 * float(nb) * np.max(c, axis=1) / (s[:, 0] * s[:, 0])
        scores = np.where(np.isfinite(scores), scores, INF)
        for i in range(B):
            si = safe_float(scores[i])
            if np.isfinite(si) and si < best_score[i]:
                best_score[i] = si
                best_W[i] = W[i]
                if si < global_best:
                    global_best = si
                    last_improve = time.monotonic()

        q = make_q_batch(c, beta)
        qr = np.zeros((B, fftlen), dtype=np.float64)
        qr[:, :L] = q[:, ::-1]
        Qr = np.fft.rfft(qr, fftlen)
        H = np.fft.irfft(P * Qr, fftlen)
        G = 2.0 * H[:, nb - 1:2 * nb - 1]
        G = G[:, ::-1]
        G = np.where(np.isfinite(G), G, 0.0)
        G = np.maximum(G, 0.0)
        grad_log = p * (G - np.sum(p * G, axis=1, keepdims=True))
        grad_W = grad_log @ Phi
        grad_W = np.where(np.isfinite(grad_W), grad_W, 0.0)
        W, m_adam, v_adam = adam_step_2d(W, grad_W, m_adam, v_adam, step, lr)
        step += 1

        if time.monotonic() - last_improve < 15.0:
            beta = min(140.0, beta * 1.035 + 0.1)
            lr = min(0.22, lr * 1.02)
        else:
            beta = min(220.0, beta * 1.08 + 0.4)
            lr = max(0.025, lr * 0.96)

    out = []
    order = order_by_score(best_score)
    for i in order[:8]:
        if np.isfinite(best_score[i]):
            prof = best_W[i] @ Phi.T
            prof = np.where(np.isfinite(prof), prof, 0.0)
            prof = prof - float(np.max(prof))
            out.append(prof)
    if not out and B > 0:
        prof = np.zeros(nb, dtype=np.float64)
        out.append(prof)
    return out


def load_parent_profile():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 0.0:
        return None
    return arr / s


def exact_score_normalized(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return INF
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return INF
    p = p / s
    try:
        c = np.convolve(p, p)
    except Exception:
        return INF
    if c.size == 0:
        return INF
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return INF
    return float(2.0 * n * mx)


def finalize_sequence(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        p = np.ones(64, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    n = p.size
    s = float(np.sum(p))
    scale = float(math.sqrt(2.0 * n))
    if s > 0.0 and np.isfinite(s):
        a = p * scale / s
    else:
        a = np.ones(n, dtype=np.float64) * (scale / float(n))
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(np.sum(a))
    if not np.isfinite(s2) or s2 < 0.01:
        a = a * (0.01 / max(s2, EPS))
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    budget = min(max(1.0, safe_float(budget_s)), 2395.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    parent = load_parent_profile()

    if budget > 1500.0:
        SEQ_N = 8192
    elif budget > 600.0:
        SEQ_N = 4096
    else:
        SEQ_N = 2048

    if parent is not None and parent.size > int(0.8 * SEQ_N) and parent.size <= 16000:
        SEQ_N = int(max(2048, min(8192, parent.size + 256)))
        if (SEQ_N & (SEQ_N - 1)) != 0:
            SEQ_N = next_fast_len(SEQ_N)
            SEQ_N = max(2048, min(8192, SEQ_N))

    FFT_LEN = next_fast_len(2 * SEQ_N - 1)

    print("starting search with SEQ_N =", SEQ_N, " FFT_LEN =", FFT_LEN, flush=True)

    cands = []

    if parent is not None:
        p_par = normalize_nonneg(parent)
        w_par = log_weights_from_p(p_par)
        cands.append(candidate_from_w(interp_profile(w_par, SEQ_N), FFT_LEN))
        p_par_int = interp_profile(p_par, SEQ_N)
        cands.append(candidate_from_w(log_weights_from_p(normalize_nonneg(p_par_int)), FFT_LEN))

    idx = np.arange(SEQ_N, dtype=np.float64)
    x = idx / float(max(1, SEQ_N - 1))
    direct_targets = []
    for alpha in (0.25, 0.45, 0.55, 0.75):
        t1 = -alpha * np.log(idx + 1.0)
        t2 = -alpha * np.log(SEQ_N - idx)
        t3 = np.logaddexp(t1, t2) - math.log(2.0)
        direct_targets.extend([t1 - np.mean(t1), t2 - np.mean(t2), t3 - np.mean(t3)])
    for a in (-1.5, -0.8, 0.8, 1.5):
        direct_targets.append(a * (x - 0.5) - 0.0)
    for a in (-1.0, -0.5, 0.5, 1.0):
        direct_targets.append(a * ((x - 0.5) ** 2) - 0.0)
    for a in (0.7, 1.2, 1.8):
        direct_targets.append(a * np.sin(np.pi * x) ** 2 - 0.0)
    for _ in range(8):
        t = rng.normal(0.0, 0.35, SEQ_N)
        direct_targets.append(t - np.mean(t))
    for t in direct_targets:
        t = np.asarray(t, dtype=np.float64).ravel()
        if t.size == SEQ_N:
            t = np.where(np.isfinite(t), t, 0.0)
            t = t - float(np.max(t))
            cands.append(candidate_from_w(t, FFT_LEN))

    basis_budget = min(360.0, max(60.0, 0.16 * budget))
    basis_profiles = basis_search(1024, basis_budget, int(seed) + 17, m=28, B=48)
    for prof in basis_profiles:
        w = interp_profile(prof, SEQ_N)
        cands.append(candidate_from_w(w, FFT_LEN))

    cands = keep_best_candidates(cands, 14)
    if not cands:
        cands = [candidate_from_w(np.zeros(SEQ_N, dtype=np.float64), FFT_LEN)]

    global_best = min(c.get("best_score", INF) for c in cands)
    print("initial best =", global_best, flush=True)

    local_deadline = t0 + budget - 110.0
    outer = 0
    last_print = t0

    while time.monotonic() < local_deadline:
        outer += 1

        cands = keep_best_candidates(cands, 12 if SEQ_N <= 8192 else 9)

        top_count = min(3, len(cands))
        for i in range(top_count):
            improved = update_cross_candidate(cands[i])
            if improved:
                continue
            update_self_candidate(cands[i], temp=1e-6)

        for i in range(top_count, len(cands)):
            update_self_candidate(cands[i], temp=1e-8)

        if outer % 25 == 0:
            for i in range(min(5, len(cands))):
                symmetrize_candidate(cands[i])

        if outer % 55 == 0 and len(cands) > 3:
            best = cands[0]
            base_w = best["best_w"]
            for sig in (0.07, 0.22, 0.65, 1.6):
                wm = base_w + rng.normal(0.0, sig, SEQ_N)
                cands.append(candidate_from_w(wm, FFT_LEN))
            if best is not None:
                sym = 0.5 * (best["best_p"] + best["best_p"][::-1])
                cands.append(candidate_from_w(log_weights_from_p(normalize_nonneg(sym)), FFT_LEN))

        if outer % 180 == 0:
            for i in range(max(0, len(cands) - 3), len(cands)):
                if cands[i].get("stall", 1) > 120:
                    base = cands[0]["best_w"]
                    wm = base + rng.normal(0.0, 0.45, SEQ_N)
                    cands[i] = candidate_from_w(wm, FFT_LEN)

        if time.monotonic() - last_print > 60.0:
            g = min(c.get("best_score", INF) for c in cands)
            print("outer", outer, "t", int(time.monotonic() - t0), "best", g, flush=True)
            last_print = time.monotonic()

    cands = keep_best_candidates(cands, 10)

    best_fft_p = None
    best_fft_score = INF
    for c in cands:
        sc = safe_float(c.get("best_score", INF))
        if sc < best_fft_score:
            best_fft_score = sc
            best_fft_p = c.get("best_p", None)

    exact_start = time.monotonic()
    exact_reserve = min(85.0, max(15.0, 0.04 * budget))
    best_exact_p = best_fft_p
    best_exact_score = best_fft_score
    checked = 0

    if SEQ_N <= 16000:
        order = order_by_score([c.get("best_score", INF) for c in cands])
        for i in order:
            if checked >= 5:
                break
            if time.monotonic() - exact_start > exact_reserve:
                break
            p = cands[i].get("best_p", None)
            if p is None:
                continue
            t_e0 = time.monotonic()
            sc = exact_score_normalized(p)
            sc = safe_float(sc)
            if time.monotonic() - t_e0 > exact_reserve:
                break
            checked += 1
            if np.isfinite(sc) and sc < best_exact_score:
                best_exact_score = sc
                best_exact_p = p

    if best_exact_p is None:
        best_exact_p = np.ones(SEQ_N, dtype=np.float64) / float(SEQ_N)
        best_exact_score = INF

    print("final best exact/fft score =", best_exact_score, flush=True)

    final_seq = finalize_sequence(best_exact_p)
    return final_seq
