Search for low-peak nonnegative self-convolutions using LP-guided optimization,
LSE gradient descent, and block coordinate descent.

def _next_pow2(size):
    if size <= 1:
        return 1

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1)) / max(n, 1)
    return a / s

def _project_simplex(v):
    """True Euclidean projection onto {x : x >= 0, sum(x) = 1}."""
    n = v.size
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1.0) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    idx_max = idx[-1]
    theta = (cssv[idx_max] - 1.0) / (idx_max + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.ones(n) / n
    return w / s

def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2*n-1]

    n = a.size
    # A is (2n-1) x n
    # A[k,i] = a[k-i] if 0 <= k-i < n
    # Equivalently: for each i, A[k,i] = a[k-i] for k in [i, i+n-1]
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # k-i must be in [0, n-1], so k in [i, i+n-1]
        A[i:i+n, i] = a
    return A

    """
    Solve: max sum(g) s.t. A@g <= M*ones, g >= 0
    where A[k,i] = a[k-i] if 0<=k-i<n else 0
    """
    n = a.size
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix (sparse would be better but dense is fine for n<=2048)
    A_ub = np.zeros((2*n-1, n))
    for i in range(n):
        A_ub[i:i+n, i] = a
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5.0})

        g = np.maximum(res.x, 0.0)
        s = float(g.sum())
        if s > 1e-300:
            return g / s
    return None

def _lse_and_gradient(a, b, beta, fft_size):
    """Compute LSE-based gradient."""
    n = a.size
    m = float(np.max(b))
    arg = beta * (b - m)
    arg = np.maximum(arg, -700.0)
    e = np.exp(arg)
    s = float(e.sum())

    else:
        p = e / s

    a_flip = a[::-1]

    PF = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(AF * PF, fft_size).real
    h = conv[n-1:2*n-1]
    return 2.0 * h

"""
Search for low-peak nonnegative self-convolutions using LP-guided optimization,
LSE/top-k hybrid gradient descent, true simplex projection, and multi-scale exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
import copy

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1)) / max(n, 1)
    return a / s


def _project_simplex(v):
    n = v.size
    if n <= 1:
        return np.ones(max(n, 1))
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    idx_max = idx[-1]
    theta = (cssv[idx_max] - 1.0) / (idx_max + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.ones(n) / n
    return w / s


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _score_of(a):
    a = _normalize(a)
    n = a.size
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _build_lp_matrix(a):
    n = a.size
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _lp_direction(a, M, time_limit=3.0):
    n = a.size
    c = -np.ones(n)
    A_ub = _build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    max_a = float(np.max(a))
    upper = max(2.0 * max_a, 4.0 / n)
    bounds = [(0.0, upper)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = float(g.sum())
            if s > 1e-300:
                return g / s
    except Exception:
        pass
    return None


def _lse_gradient(a, b, beta, fft_size):
    n = a.size
    m = float(np.max(b))
    arg = beta * (b - m)
    arg = np.maximum(arg, -700.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0:
        p = np.ones(b.size) / b.size
    else:
        p = e / s
    a_flip = a[::-1]
    AF = np.fft.rfft(a_flip, fft_size)
    PF = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(AF * PF, fft_size).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _topk_gradient(a, b, fft_size, k=10):
    n = a.size
    bk = min(k, b.size)
    idx = np.argpartition(b, -bk)[-bk:]
    top_vals = b[idx]
    w_sum = float(top_vals.sum())
    if w_sum <= 0:
        weights = np.zeros(b.size)
        weights[idx] = 1.0 / bk
    else:
        weights = np.zeros(b.size)
        weights[idx] = top_vals / w_sum
    a_flip = a[::-1]
    AF = np.fft.rfft(a_flip, fft_size)
    WF = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(AF * WF, fft_size).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _perturb(a, rng, strength=0.2):
    n = a.size
    c = a * np.exp(rng.normal(0.0, strength, size=n))
    c = np.maximum(c, 1e-15)
    return _normalize(c)


def _block_perturb(a, rng, strength=0.3):
    n = a.size
    c = a.copy()
    width = max(2, n // 20)
    start = int(rng.integers(0, max(1, n - width)))
    c[start:start + width] *= np.exp(rng.normal(0.0, strength))
    if rng.random() < 0.2:
        c = 0.5 * (c + c[::-1])
    c = np.maximum(c, 1e-15)
    return _normalize(c)


def _smooth_perturb(a, rng, strength=0.5):
    n = a.size
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, n // 10)
    F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0:
        f = f / sd
    c = a * (1.0 + strength * f)
    c = np.maximum(c, 1e-15)
    return _normalize(c)

def _make_exponential(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_gaussian(n, center, width):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
    return _normalize(a)


def _make_bimodal(n, c1, c2, w):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-((x - c1) ** 2) / (2.0 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2.0 * w ** 2))
    return _normalize(a)


def _resample(old, new_n):
    old = _normalize(old)
    new_n = max(int(new_n), 2)
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)
