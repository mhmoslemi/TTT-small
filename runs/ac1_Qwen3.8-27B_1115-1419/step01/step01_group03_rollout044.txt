```python
def fista_refined(a0, n, duration, rng, step=0.95, top_k=50, weight_decay=0.5):
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    
    while time.monotonic() < deadline:
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy
        
        # Compute convolution
        c_y = np.convolve(y, y)
        
        # Weighted gradient: weight each peak by its value
        # Top-k peaks with exponential weighting
        sorted_idx = np.argsort(c_y)[::-1]
        top_idx = sorted_idx[:top_k]
        
        # Weights: exp of normalized values
        peak_vals = c_y[top_idx]
        max_peak = peak_vals[0]
        if max_peak > 1e-15:
            weights = np.exp(-weight_decay * (max_peak - peak_vals) / max_peak)
        else:
            weights = np.ones(top_k)
        weights /= weights.sum()
        
        # Compute weighted gradient
        g = np.zeros(n)
        for w, mk in zip(weights, top_idx):
            g += w * grad_c_k(y, int(mk))
        
        # FISTA step
        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic score check
        if it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            # Restart if stalled
            if it % 500 == 0 and sc > best_score * 1.001:
                # Perturb and restart
                a = best_a + rng.normal(0, 0.005/np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                a /= a.sum()
                a_prev = a.copy()
                t = 1.0
                print(f"    Restart at it={it}, score={best_score:.7f}")
    
    final_sc = fast_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    
    return best_a, best_score
```

```python
import numpy as np
import time

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    rho = np.where(cond)[0][-1] if cond.any() else 0
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s*s)
```

```python
def project_simplex(v):
```

```python
        top_idx = np.argpartition(c, -top_k)[-top_k:]

    peak_vals = c[top_idx]
    max_peak = peak_vals.max()

    if max_peak > 1e-30:
        weights = np.exp(-alpha * (max_peak - peak_vals) / max_peak)
    else:
        weights = np.ones(len(top_idx))
    weights /= weights.sum()

    g = np.zeros(n, dtype=np.float64)
    for w, mk in zip(weights, top_idx):
        mk = int(mk)
        lo = max(0, mk - n + 1)
        hi = min(n, mk + 1)
        if lo < hi:
            idx = mk - np.arange(lo, hi, dtype=np.int64)
            g[lo:hi] += 2.0 * w * a[idx]

    return g


def run_fista(a0, duration, rng, step=0.48, top_k=20, alpha=20.0, verbose=True):
    """
    Run FISTA to minimize max_k (a*a)[k] on the probability simplex.
    Uses corrected gradient (factor of 2) and exponential peak weighting.
    """
    n = len(a0)
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    stall_count = 0

    if verbose:
        print(f"  FISTA start: n={n}, score={best_score:.8f}, step={step}, top_k={top_k}, alpha={alpha}")

    while time.monotonic() < deadline:
        it += 1

        # Nesterov momentum
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        mu = (t_old - 1.0) / t
        y = a + mu * (a - a_prev)

        # Ensure y is roughly feasible
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy

        # Compute convolution at y
        c_y = np.convolve(y, y)

        # Compute weighted gradient
        g = compute_weighted_gradient(y, c_y, top_k=top_k, alpha=alpha)

        # FISTA step + project onto simplex
        z = y - step * g
        a_new = project_simplex(z)

        a_prev = a
        a = a_new

        # Periodic evaluation
        if it % 100 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

            # Restart if stalled
            if stall_count > 200:
                noise_scale = 0.005 / np.sqrt(n)
                a = best_a + rng.normal(0, noise_scale, n)
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-15:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t = 1.0
                stall_count = 0
                if verbose:
                    print(f"    Restart at it={it}, score={best_score:.8f}")

        if verbose and it % 20000 == 0:
            sc = fast_score(a)
            print(f"    it={it}, cur={sc:.8f}, best={best_score:.8f}")

    # Final check
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    if verbose:
        print(f"  FISTA done: it={it}, score={best_score:.8f}")

    return best_a, best_score


def run_adam_logspace(a_init, duration, rng, lr=0.02, top_k=50, alpha=30.0, verbose=True):
    """
    Run Adam in log-space (softmax parameterization) to minimize the score.
    No projection needed; constraint handled by softmax.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30)
    x = np.log(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a_init.copy()
    best_score = fast_score(a_init)
    deadline = time.monotonic() + duration

    if verbose:
        print(f"  Adam logspace start: n={n}, score={best_score:.8f}, lr={lr}")

    while time.monotonic() < deadline:
        t += 1

        # Softmax
        x_shift = x - x.max()
        e = np.exp(x_shift)
        a = e / e.sum()

        # Convolution
        c = np.convolve(a, a)

        # Top-k peaks with exponential weights
        if top_k >= c.size:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -top_k)[-top_k:]

        peak_vals = c[top_idx]
        max_peak = peak_vals.max()

        if max_peak > 1e-30:
            weights = np.exp(-alpha * (max_peak - peak_vals) / max_peak)
        else:
            weights = np.ones(len(top_idx))
        weights /= weights.sum()

        # Compute gradient in log-space:
        # dF/dx_j = 2n * a_j * (g_j - F_smooth)
        # where g_j = sum_k w_k * a_{k-j}, F_smooth = sum_k w_k * f_k
        g_vec = np.zeros(n, dtype=np.float64)
        F_smooth = 0.0
        for w, mk in zip(weights, top_idx):
            mk = int(mk)
            F_smooth += w * c[mk]
            lo = max(0, mk - n + 1)
            hi = min(n, mk + 1)
            if lo < hi:
                idx = mk - np.arange(lo, hi, dtype=np.int64)
                g_vec[lo:hi] += w * a[idx]

        grad_x = 2.0 * n * a * (g_vec - F_smooth)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic check
        if t % 200 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and t % 10000 == 0:
            sc = fast_score(a)
            print(f"    Adam t={t}, cur={sc:.8f}, best={best_score:.8f}")

    # Final
    x_shift = x - x.max()
    e = np.exp(x_shift)
    a_final = e / e.sum()
    sc = fast_score(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    if verbose:
        print(f"  Adam done: t={t}, score={best_score:.8f}")

    return best_a, best_score


def make_candidates(n, rng, arr):
    """Generate multiple initial candidates."""
    candidates = []
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # 1. Given sequence
    S = arr.sum()
    if S > 1e-10:
        candidates.append(("given", arr / S))

    # 2. Uniform
    candidates.append(("uniform", np.ones(n) / n))

    # 3. Gaussian bump (narrow)
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    candidates.append(("gauss_narrow", gauss / gauss.sum()))

    # 4. Gaussian bump (wide)
    sigma = n / 3.0
    gauss = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    candidates.append(("gauss_wide", gauss / gauss.sum()))

    # 5. Cosine bump
    cos_bump = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - center) / n))
    cos_bump = np.maximum(cos_bump, 1e-10)
    candidates.append(("cosine", cos_bump / cos_bump.sum()))

    # 6. Flat-top with smooth edges
    w = n // 5
    flat = np.zeros(n)
    flat[w:n - w] = 1.0
    # Smooth edges
    for i in range(w):
        t = (i + 1) / (w + 1)
        flat[i] = 0.5 * (1.0 + np.cos(np.pi * (1.0 - t)))
        flat[n - 1 - i] = flat[i]
    flat = np.maximum(flat, 1e-10)
    candidates.append(("flat_top", flat / flat.sum()))

    # 7. Random exponential
    r = rng.exponential(1.0, n)
    candidates.append(("rand_exp", r / r.sum()))

    # 8. Two-peak
    twopeak = np.zeros(n)
    w2 = max(1, n // 15)
    twopeak[n // 4 - w2:n // 4 + w2] = 1.0
    twopeak[3 * n // 4 - w2:3 * n // 4 + w2] = 1.0
    twopeak = np.maximum(twopeak, 1e-10)
    candidates.append(("two_peak", twopeak / twopeak.sum()))

    return candidates


def propose_candidate(budget_s=2400):
    """
    Main search: corrected FISTA with continuation strategy on the simplex.
    """
    rng = np.random.default_rng(12345)
    total_deadline = time.monotonic() + float(budget_s) - 20.0

    # Get initial sequence
    try:
        arr = np.array(height_sequence_1, dtype=np.float64)
        arr = np.clip(arr, 0.0, None)
        n = len(arr)
    except Exception:
        n = 4806
        arr = np.ones(n, dtype=np.float64)

    if n < 2:
        n = 4806
        arr = np.ones(n, dtype=np.float64)

    print(f"=== Starting search with n = {n} ===")

    # Generate and evaluate candidates
    candidates = make_candidates(n, rng, arr)
    best_cand = None
    best_cand_score = np.inf
    for name, c in candidates:
        sc = fast_score(c)
        print(f"  {name}: {sc:.8f}")
        if sc < best_cand_score:
            best_cand_score = sc
            best_cand = c.copy()

    print(f"Best initial: {best_cand_score:.8f}")

    best_a = best_cand.copy()
    best_score = best_cand_score

    # Phase 1: FISTA smooth (large top_k, small alpha) - fast initial convergence
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        t1 = min(time_left * 0.35, 600.0)
        print(f"\n--- Phase 1: FISTA smooth ({t1:.0f}s) ---")
        a1, s1 = run_fista(best_cand, t1, rng, step=0.48, top_k=100, alpha=5.0)
        if s1 < best_score:
            best_a, best_score = a1, s1
            print(f"  Improved to {best_score:.8f}")

    # Phase 2: FISTA medium (moderate top_k and alpha)
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        t2 = min(time_left * 0.35, 800.0)
        print(f"\n--- Phase 2: FISTA medium ({t2:.0f}s) ---")
        a2, s2 = run_fista(best_a, t2, rng, step=0.48, top_k=30, alpha=20.0)
        if s2 < best_score:
            best_a, best_score = a2, s2
            print(f"  Improved to {best_score:.8f}")

    # Phase 3: FISTA sharp (small top_k, large alpha) - fine-tuning
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        t3 = min(time_left * 0.30, 600.0)
        print(f"\n--- Phase 3: FISTA sharp ({t3:.0f}s) ---")
        a3, s3 = run_fista(best_a, t3, rng, step=0.45, top_k=10, alpha=80.0)
        if s3 < best_score:
            best_a, best_score = a3, s3
            print(f"  Improved to {best_score:.8f}")

    # Phase 4: Adam in log-space for additional refinement
    time_left = total_deadline - time.monotonic()
    if time_left > 60 and best_score > 1.503:
        t4 = time_left * 0.5
        print(f"\n--- Phase 4: Adam logspace ({t4:.0f}s) ---")
        a4, s4 = run_adam_logspace(best_a, t4, rng, lr=0.01, top_k=50, alpha=30.0)
        if s4 < best_score:
            best_a, best_score = a4, s4
            print(f"  Improved to {best_score:.8f}")

    # Phase 5: Final FISTA polish with very sharp weighting
    time_left = total_deadline - time.monotonic()
    if time_left > 20:
        print(f"\n--- Phase 5: Final polish ({time_left:.0f}s) ---")
        a5, s5 = run_fista(best_a, time_left, rng, step=0.42, top_k=5, alpha=200.0)
        if s5 < best_score:
            best_a, best_score = a5, s5
            print(f"  Improved to {best_score:.8f}")

    # Final verification
    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.8f}")

    if not np.isfinite(final_score) or final_score > best_cand_score:
        print(f"Falling back to best candidate: {best_cand_score:.8f}")
        final_list = [float(x) for x in best_cand]
        final_score = best_cand_score

    print(f"=== FINAL: {final_score:.8f} (n={n}) ===")
    return final_list
```
