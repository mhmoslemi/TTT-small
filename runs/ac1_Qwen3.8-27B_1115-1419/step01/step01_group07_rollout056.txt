    n = len(a)

def compute_obj(a):

    n = len(a)
    if a.sum() < 1e-10:

    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (a.sum() ** 2)

    """Compute gradient of LSE-smoothed max of conv(a,a)."""
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)

import time

def compute_score(a):
    """Compute the exact objective score."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def smooth_grad(a, tau):
    """Compute gradient of LSE-smoothed max of conv(a,a), assuming sum(a)=1."""
    c = np.convolve(a, a)
    c_max = c.max()
    x = (c - c_max) / tau
    e = np.exp(x)
    w = e / e.sum()
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad

def exact_subgrad(a, top_k=20):
    """Compute exact subgradient using top-k peaks."""
    c = np.convolve(a, a)
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_idx]
    # Weight by exponential of values (softmax-like)
    max_v = top_vals.max()
    weights = np.exp((top_vals - max_v) / (max_v - top_vals.min() + 1e-15))
    weights /= weights.sum()
    
    n = len(a)
    g = np.zeros(n)
    for idx, wi in zip(top_idx, weights):
        lo = max(0, idx - n + 1)
        hi = min(n - 1, idx)
        g[lo:hi+1] += wi * 2.0 * a[idx-hi:idx-lo+1][::-1]
    return g

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s

"""
Optimize step function sequence to minimize 2*n*max(a*a)/(sum(a))^2.
Uses FISTA (Nesterov-accelerated proximal gradient) with log-sum-exp smoothing,
multiple diverse initializations at various sequence lengths,
and exact subgradient polishing.
Key: treats the problem as peak sidelobe level minimization of autocorrelation.
"""

import numpy as np
import time


def compute_score(a):
    """Compute exact objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def smooth_grad(a, tau):
    """Gradient of 2*n*LSE-max of conv(a,a), assuming a is on simplex."""
    c = np.convolve(a, a)
    c_max = c.max()
    x = (c - c_max) / tau
    e = np.exp(x)
    w = e / e.sum()
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def exact_subgrad(a, top_k=20):
    """Subgradient using weighted top-k peaks of conv(a,a)."""
    c = np.convolve(a, a)
    n = len(a)
    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    spread = top_vals.max() - top_vals.min()
    if spread < 1e-15:
        weights = np.ones(k) / k
    else:
        weights = np.exp((top_vals - top_vals.max()) / spread)
        weights /= weights.sum()
    
    g = np.zeros(n)
    for idx, wi in zip(top_idx, weights):
        lo = max(0, idx - n + 1)
        hi = min(n - 1, idx)
        g[lo:hi + 1] += wi * 2.0 * a[idx - hi:idx - lo + 1][::-1]
    return g


def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s

    if tau_init is None:
        tau_init = 1.0 / np.sqrt(n)
    if step_init is None:
        step_init = 1.0

    best_score = compute_score(a)
    if verbose:
        print(f"  [{label}] Initial: score={best_score:.6f}, n={n}")
    
    t_start = time.monotonic()
    deadline = t_start + budget_s
    
    tau = tau_init
    step = step_init
    t_k = 1.0

    prev_a = a.copy()

        # Safeguard: clip extrapolation

        if np.linalg.norm(diff) > 2.0 * np.linalg.norm(prev_a - a + 1e-30):
            y = prev_a.copy()

        step *= 0.99998

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    resampled = np.interp(new_x, old_x, a)
    return resampled
