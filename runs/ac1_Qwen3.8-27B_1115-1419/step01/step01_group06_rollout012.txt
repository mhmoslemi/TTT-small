```python
"""
Search for nonnegative sequences minimizing
    E(a) = 2 * n * max(conv(a, a)) / (sum(a)^2).

The search works with the scale-invariant normalization sum(a)=1 and minimizes
    F(a) = 2 * n * max(conv(a, a)).

Strategy:
  1. Build a diverse set of initial candidates (parent sequence if present,
     resampled parent, power-law shapes, two-edge/asymmetric two-edge shapes,
     smooth random perturbations).
  2. Run a time-limited multiplicative/smoothed-Linff projected gradient
     optimizer on the best initial candidates.
  3. Periodically solve the auxiliary convex subproblem
         min_{q in simplex} max_k (p * q)_k
     by the same smoothing method and blend q back into p.
  4. Use multiple weights for the smoothed maximum (relative-softmax and
     active-set/top peaks), multiple step sizes, and smooth restarts to avoid
     weak local minima.

All scoring in the search is FFT-based. The final sequence is returned already
normalized and clipped to the evaluator's valid range.
"""

import numpy as np
import time

EPS = 1e-300
INF = float("inf")


def next_fast_len(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonneg(v):
    """Finite nonnegative vector normalized to sum 1."""
    arr = np.asarray(v, dtype=np.float64).ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones(max(1, arr.size), dtype=np.float64)
        s = float(arr.size)
    return arr / s


def log_safe(a):
    """Stable logarithm of a positive vector."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.zeros(1, dtype=np.float64)
    amin = float(a.min())
    floor = max(EPS, amin * 1e-12, 1e-300)
    return np.log(np.maximum(a, floor))


def softmax_from_log(u):
    """Stable softmax."""
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return np.ones(1, dtype=np.float64)
    u = np.where(np.isfinite(u), u, 0.0)
    u = u - float(np.max(u))
    e = np.exp(u)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(u.size, dtype=np.float64) / float(u.size)
    return e / s


def fft_score_internal(a, N):
    """
    Fast FFT score for an already normalized nonnegative vector.
    Returns (score, conv, rfft(a)).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[: 2 * n - 1]
    if c.size:
        c = np.where(c > 0.0, c, 0.0)
        mx = float(c.max())
    else:
        mx = 0.0
    if not np.isfinite(mx):
        return INF, c, A
    return float(2.0 * n * mx), c, A


def fft_score(a, N=None):
    """FFT score with normalization."""
    a = normalize_nonneg(a)
    if N is None:
        N = next_fast_len(2 * a.size)
    return fft_score_internal(a, N)


def rfft_padded(w, N):
    """Zero-pad a full convolution weight vector and rfft it."""
    w = np.asarray(w, dtype=np.float64).ravel()
    pad = np.zeros(N, dtype=np.float64)
    L = min(w.size, N)
    if L > 0:
        pad[:L] = w[:L]
    return np.fft.rfft(pad, N)


def make_weights_rel(c, beta):
    """
    Softmax weights on the normalized convolution levels.
    beta scales the relative drop from the maximum.
    """
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return np.ones(1, dtype=np.float64)
    mx = float(c.max())
    if not np.isfinite(mx) or mx <= 0.0:
        return np.ones(c.size, dtype=np.float64) / float(c.size)
    x = c / mx
    x = np.where(np.isfinite(x), x, 0.0)
    y = -float(beta) * (1.0 - x)
    y = np.where(np.isfinite(y), y, -745.0)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s


def make_weights_top(c, tol):
    """
    Uniform weights over lags whose convolution value is within tol of max.
    If too few lags are active, use the top quantile instead.
    """
    c = np.asarray(c, dtype=np.float64).ravel()
    m = c.size
    if m == 0:
        return np.ones(1, dtype=np.float64)
    mx = float(c.max())
    if not np.isfinite(mx) or mx <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)

    w = np.zeros(m, dtype=np.float64)
    thresh = mx * (1.0 - float(tol))
    mask = c >= thresh
    cnt = int(mask.sum())

    min_cnt = max(16, int(0.002 * m))
    if cnt < min_cnt:
        k = min(m, max(min_cnt, int(0.005 * m) + 1))
        if k >= m:
            w[:] = 1.0
        else:
            idx = np.argpartition(c, m - k)[-k:]
            w[idx] = 1.0
    else:
        w[mask] = 1.0

    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return w / s


def self_candidate_from_W(a, A, W, lr, N, n):
    """
    One projected log-softmax gradient step for self-convolution.
    W is the rfft of the full-lag weights for max(conv(a,a)).
    """
    r = np.fft.irfft(W * np.conjugate(A), N)
    grad = 2.0 * r[:n]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    a = np.asarray(a, dtype=np.float64).ravel()
    dot = float(np.dot(a, grad))
    g = a * (grad - dot)
    g = np.where(np.isfinite(g), g, 0.0)
    gmax = float(np.max(np.abs(g))) if g.size else 0.0
    if gmax <= 1e-300:
        return a.copy()

    g = g / gmax
    u = log_safe(a)
    u = u - float(np.max(u))
    u = u - float(lr) * g
    return softmax_from_log(u)


def cross_candidate_from_F(F, b, W, lr, N, n):
    """
    One projected log-softmax gradient step for fixed f:
        minimize max_k (f * b)_k over b in simplex.
    F is rfft(f); W is rfft of weights on f*b lags.
    """
    r = np.fft.irfft(W * np.conjugate(F), N)
    grad = r[:n]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    b = np.asarray(b, dtype=np.float64).ravel()
    dot = float(np.dot(b, grad))
    g = b * (grad - dot)
    g = np.where(np.isfinite(g), g, 0.0)
    gmax = float(np.max(np.abs(g))) if g.size else 0.0
    if gmax <= 1e-300:
        return b.copy()

    g = g / gmax
    u = log_safe(b)
    u = u - float(np.max(u))
    u = u - float(lr) * g
    return softmax_from_log(u)


def cross_score(F, b, N, n):
    """Max convolution of fixed f (via F) and candidate b."""
    B = np.fft.rfft(b, N)
    h = np.fft.irfft(F * B, N)
    h = h[: 2 * n - 1]
    if h.size:
        h = np.where(h > 0.0, h, 0.0)
        mx = float(h.max())
    else:
        mx = 0.0
    if not np.isfinite(mx):
        return INF, h
    return mx, h


def cross_optimize_fixed(F, b, N, n, beta, lr, deadline, max_inner):
    """
    Improve b for fixed f by reducing max(f*b).
    Returns (best_b, best_h, final_lr).
    """
    best_hs, h = cross_score(F, b, N, n)
    best_b = b.copy()
    local_lr = float(lr)

    for _ in range(max_inner):
        if time.monotonic() >= deadline:
            break
        if not np.isfinite(best_hs):
            b = np.ones(n, dtype=np.float64) / float(n)
            best_hs, h = cross_score(F, b, N, n)
            best_b = b.copy()
            continue

        improved = False
        weight_objs = (
            make_weights_rel(h, beta),
            make_weights_rel(h, beta * 2.0 + 1.0),
            make_weights_top(h, 0.08),
        )
        for wobj in weight_objs:
            if improved:
                break
            W = rfft_padded(wobj, N)
            for mul in (0.3, 1.0, 3.0):
                cand = cross_candidate_from_F(F, b, W, local_lr * mul, N, n)
                cand_hs, _ = cross_score(F, cand, N, n)
                if np.isfinite(cand_hs) and cand_hs < best_hs - 1e-14:
                    b = cand
                    best_b = cand.copy()
                    best_hs = cand_hs
                    improved = True
                    break

        if improved:
            local_lr = min(2.0, local_lr * 1.05)
        else:
            local_lr = max(0.01, local_lr * 0.75)
            if local_lr <= 0.012:
                wjump = make_weights_top(h, 0.2)
                Wjump = rfft_padded(wjump, N)
                cand = cross_candidate_from_F(F, b, Wjump, 2.0, N, n)
                cand_hs, _ = cross_score(F, cand, N, n)
                if np.isfinite(cand_hs) and cand_hs < best_hs - 1e-14:
                    b = cand
                    best_b = cand.copy()
                    best_hs = cand_hs
                break

    return best_b, best_hs, local_lr


def power_sequence(n, alpha, beta):
    """Power-law near both ends: ~ x^alpha (1-x)^beta."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 1.0) / float(n)
    p = np.ones(n, dtype=np.float64)
    if alpha != 0.0:
        p *= np.power(np.maximum(x, 1e-12), float(alpha))
    if beta != 0.0:
        p *= np.power(np.maximum(1.0 - x, 1e-12), float(beta))
    p = np.where(np.isfinite(p), p, 0.0)
    return normalize_nonneg(p)


def two_edge_sequence(n, a, r=1.0):
    """Two decaying edge clusters with optional asymmetry."""
    i = np.arange(n, dtype=np.float64)
    dleft = i / float(n)
    dright = (float(n) - 1.0 - i) / float(n)
    p = float(r) * np.exp(-float(a) * dleft) + np.exp(-float(a) * dright)
    p = np.where(np.isfinite(p), p, 0.0)
    return normalize_nonneg(p)


def smooth_perturb(a, sigma, rng):
    """Smooth low-frequency perturbation in log-space plus small white noise."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return normalize_nonneg(a)
    u = log_safe(a)
    i = np.arange(n, dtype=np.float64)
    sigma = float(sigma)
    for k in range(1, 13):
        amp = sigma * (0.88 ** (k - 1)) * float(rng.uniform(0.0, 1.0))
        if amp <= 0.0:
            continue
        u += amp * np.sin(2.0 * np.pi * float(k) * i / float(n) + float(rng.uniform(0.0, 2.0 * np.pi)))
    u += rng.normal(0.0, 0.02 * sigma, size=n)
    return softmax_from_log(u)


def resample_sequence(a, m):
    """Linear resample to length m, normalized."""
    a = normalize_nonneg(a)
    n = a.size
    if n == 0:
        return np.ones(m if m > 0 else 1, dtype=np.float64)
    if m == n:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=np.float64) / float(m)
    x_old = np.linspace(0.0, float(n - 1), n)
    x_new = np.linspace(0.0, float(m - 1), m)
    out = np.interp(x_new, x_old, a)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    return normalize_nonneg(out)

    if obj is None:

    except Exception:

def select_top_scored(cands, max_count):
    """Score candidates by FFT and return top tuples (score, array, N)."""
    scored = []
    for a in cands:
        a = normalize_nonneg(a)
        if a.size < 16:
            continue
        N = next_fast_len(2 * a.size)
        sc, _, _ = fft_score_internal(a, N)
        if np.isfinite(sc):
            scored.append([float(sc), a, N])

    top = []
    remaining = scored
    while remaining and len(top) < max_count:
        best_i = 0
        best_sc = remaining[0][0]
        for i in range(1, len(remaining)):
            if remaining[i][0] < best_sc:
                best_sc = remaining[i][0]
                best_i = i
        top.append(remaining.pop(best_i))
    return top


def optimize_sequence(a0, budget_s, seed=42):
    """
    Local optimizer.
    Returns (best_normalized_sequence, best_fft_score).
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 2.0)

    a = normalize_nonneg(a0)
    if a.size < 16:
        a = power_sequence(8192, 0.0, 0.0)
    n = a.size
    N = next_fast_len(2 * n)
    A, c, score = fft_score_internal(a, N)

    if not np.isfinite(score) or not np.all(np.isfinite(a)):
        a = power_sequence(8192, 0.0, 0.0)
        n = a.size
        N = next_fast_len(2 * n)
        A, c, score = fft_score_internal(a, N)
        if not np.isfinite(score):
            a = np.ones(8192, dtype=np.float64) / 8192.0
            n = a.size
            N = next_fast_len(2 * n)
            A, c, score = fft_score_internal(a, N)

    best_a = a.copy()
    best_score = float(score)

    lr_self = 0.45
    beta_self = 15.0
    lr_cross = 0.45
    beta_cross = 20.0
    b = a.copy()
    stall = 0
    outer = 0
    last_print = t0
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        outer += 1

        # Safety reset.
        if (not np.isfinite(score) or
                not np.all(np.isfinite(a)) or
                not np.all(np.isfinite(c)) or
                not np.all(np.isfinite(A))):
            a = best_a.copy()
            A, c, score = fft_score_internal(a, N)
            b = a.copy()
            stall = 0
            continue

        improved = False

        # --- Self smoothed-Linff descent ---
        weight_objs = (
            make_weights_rel(c, beta_self),
            make_weights_rel(c, beta_self * 2.5 + 1.0),
            make_weights_top(c, 0.08),
        )
        for wobj in weight_objs:
            if improved:
                break
            W = rfft_padded(wobj, N)
            for mul in (0.3, 1.0, 3.0):
                if time.monotonic() >= deadline:
                    break
                cand = self_candidate_from_W(a, A, W, lr_self * mul, N, n)
                sc, c2, A2 = fft_score_internal(cand, N)
                if np.isfinite(sc) and sc < score - 1e-12:
                    a = cand
                    c = c2
                    A = A2
                    score = sc
                    if sc < best_score - 1e-14:
                        best_score = float(sc)
                        best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
            lr_self = min(2.0, lr_self * 1.06)
            beta_self = min(300.0, beta_self * 1.005)
            b = a.copy()
        else:
            stall += 1
            lr_self = max(0.01, lr_self * 0.90)
            if stall % 7 == 0:
                beta_self = min(500.0, beta_self * 1.25)

        # --- Fixed-self auxiliary subproblem and_blend ---
        if (outer % 6 == 0 or stall > 12) and time.monotonic() < deadline:
            b, _, lr_cross_out = cross_optimize_fixed(
                A, b, N, n, beta_cross, lr_cross, deadline, 4
            )
            if np.isfinite(lr_cross_out):
                lr_cross = max(0.01, lr_cross_out)

            if np.all(np.isfinite(b)):
                for tv in (0.25, 0.5, 0.75, 1.0):
                    if time.monotonic() >= deadline:
                        break
                    cand = (1.0 - tv) * a + tv * b
                    cand = normalize_nonneg(cand)
                    sc, c2, A2 = fft_score_internal(cand, N)
                    if np.isfinite(sc) and sc < score - 1e-12:
                        a = cand
                        c = c2
                        A = A2
                        score = sc
                        improved = True
                        b = a.copy()
                        if sc < best_score - 1e-14:
                            best_score = float(sc)
                            best_a = a.copy()
                        break

                if improved:
                    stall = 0
                    lr_self = min(2.0, lr_self * 1.08)
                    beta_cross = min(300.0, beta_cross * 1.01)
                else:
                    if stall % 7 == 0:
                        beta_cross = min(500.0, beta_cross * 1.15)

        # --- Smooth restarts ---
        if stall > 70 or (outer % 250 == 0 and score > best_score + 0.01):
            if score > best_score + 0.02 or rng.random() < 0.5:
                a = best_a.copy()
            else:
                sig = float(rng.choice((0.08, 0.18, 0.35, 0.7, 1.2)))
                a = smooth_perturb(best_a, sig, rng)
            A, c, score = fft_score_internal(a, N)
            if not np.isfinite(score) or not np.all(np.isfinite(a)):
                a = best_a.copy()
                A, c, score = fft_score_internal(a, N)

            b = a.copy()
            lr_self = 0.45
            lr_cross = 0.45
            beta_self = 15.0
            beta_cross = 20.0
            stall = 0

        if time.monotonic() - last_print >= 20.0:
            print(
                f"optimize n={n} t={time.monotonic()-t0:.0f}s "
                f"current={score:.8f} best={best_score:.8f} stall={stall}"
            )
            last_print = time.monotonic()

    if not np.isfinite(best_score):
        best_score = float(score)
        best_a = a.copy()
    return best_a, best_score


def finalize_sequence(a):
    """
    Convert a normalized array to a valid list.
    Use sum = sqrt(2n) so final values are small but safely above sum threshold.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    scale = float(np.sqrt(2.0 * n))
    if s > 0.0 and np.isfinite(s):
        a = a * scale / s
    else:
        a = np.ones(n, dtype=np.float64) * (scale / float(n))
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = a * (0.01 / max(s2, 1e-300))
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Required entry point.
    Returns a nonnegative sequence of heights intended to minimize
    evaluate_sequence(sequence).
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    budget = min(max(budget, 30.0), 2394.0)

    t0 = time.monotonic()
    deadline = t0 + budget - 2.0

    parent = load_parent()
    cands = build_initial_candidates(parent, seed)
    top = select_top_scored(cands, 5)

    if not top:
        seq = power_sequence(8192, 0.0, 0.0)
        print("fallback uniform sequence")
        return finalize_sequence(seq)

    best_score = float(top[0][0])
    best_seq = top[0][1].copy()
    print(f"initial best score (FFT, sum=1): {best_score:.8f}, n={best_seq.size}")

    for idx in range(len(top)):
        rem = deadline - time.monotonic()
        if rem < 45.0:
            break

        if idx == 0:
            b = min(rem, 1500.0)
        elif idx == 1:
            b = min(rem * 0.55, 700.0)
        elif idx == 2:
            b = min(rem * 0.30, 350.0)
        else:
            b = min(rem * 0.20, 180.0)

        if b < 30.0:
            continue

        init_a, init_sc, _ = top[idx]
        print(
            f"starting candidate {idx}/{len(top)-1}: "
            f"n={init_a.size}, init_score={init_sc:.8f}, budget={b:.0f}s"
        )

        seq, score = optimize_sequence(init_a, b, seed=1000 + idx)
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_seq = seq.copy()
            print(f"updated best score: {best_score:.8f} from candidate {idx}")

    print(f"final FFT score (sum=1): {best_score:.8f}, length={best_seq.size}")
    return finalize_sequence(best_seq)
```
