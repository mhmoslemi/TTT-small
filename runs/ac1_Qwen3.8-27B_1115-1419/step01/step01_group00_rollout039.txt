import numpy as np
import time
from scipy.optimize import linprog

# Helper functions (all top-level, no closures)

def _normalize(a, S):
    s = a.sum()
    if s < 1e-12:
        return np.full_like(a, S / len(a))
    return a / s * S

def _objective(a, n):
    c = np.max(np.convolve(a, a))
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * c / (s * s)

def _lp_descent(a, n, S, K=20, max_lp_iters=200):
    """
    Sequential LP descent. Linearize conv at current point, solve LP,
    do line search. Repeat until no improvement or max_iters.
    Returns (best_a, best_obj).
    """
    best_a = a.copy()
    c = np.convolve(a, a)
    best_obj = float(np.max(c))  # With sum=S, objective = max(conv) = 2n*max/sum^2 = max (since 2n/sum^2 = 1)
    # Wait: if sum = S = sqrt(2n), then 2n*max/sum^2 = 2n*max/(2n) = max. Yes!
    
    for it in range(max_lp_iters):
        c = np.convolve(a, a)
        # Get active set: top K peaks
        top_idx = np.argsort(c)[::-1][:K]
        
        # Build LP
        # Variables: [a_0, ..., a_{n-1}, t]
        # Minimize: t
        # Subject to:
        #   sum(a_i) = S
        #   a_i >= 0
        #   t >= 2*sum_i a0[k-i]*a_i - c0[k] for k in active set
        #   equivalently: 2*sum_i a0[k-i]*a_i - t <= c0[k]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        # Equality: sum(a) = S
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        
        # Inequalities: for each k in active set
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)
        for idx, k in enumerate(top_idx):
            # 2*sum_i a0[k-i]*a_i - t <= c0[k]
            # Row: [2*a0[k-i] for valid i, -1]
            lo = max(0, k - n + 1)

                for i in range(lo, hi + 1):
                    A_ub[idx, i] = 2.0 * a[k - i]
            A_ub[idx, n] = -1.0  # coefficient of t
            b_ub[idx] = c[k]
        
        # Bounds
        bounds = [(0, None)] * n + [(0, None)]
        
        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        a_lp = res.x[:n]
        t_lp = res.x[n]

                    break

    return best_a, best_obj

= c^0_k + sum_j 2*a^0[k-j] * (a_j - a^0_j)

Population-based optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses sequential LP descent, smooth/hard gradient methods, and a population
of diverse solutions to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _norm(a, S):
    s = float(np.sum(a))
    if s < 1e-12:
        return np.full(len(a), S / len(a), dtype=np.float64)
    return np.asarray(a, dtype=np.float64) / s * S


def _mc(a):
    """Max of self-convolution."""
    return float(np.max(np.convolve(a, a)))


def _obj_full(a, n):
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * _mc(a) / (s * s)


def _lp_one(a, n, S, K):
    """One LP step. Returns (a_new, improved)."""
    c = np.convolve(a, a)
    mx = float(np.max(c))
    idx = np.argsort(c)[::-1][:K]
    
    cv = np.zeros(n + 1)
    cv[n] = 1.0
    
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    beq = np.array([S])
    
    m = len(idx)
    Aub = np.zeros((m, n + 1))
    bub = np.zeros(m)
    for ii, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ia = np.arange(lo, hi + 1)
            Aub[ii, lo:hi+1] = 2.0 * a[k - ia]
        Aub[ii, n] = -1.0
        bub[ii] = c[k]
    
    bd = [(0.0, None)] * (n + 1)
    
    try:
        r = linprog(cv, A_ub=Aub, b_ub=bub, A_eq=Aeq, b_eq=beq,
                    bounds=bd, method='highs')
    except Exception:
        return a.copy(), False
    if r.status != 0 or r.x is None:
        return a.copy(), False
    
    an = np.maximum(r.x[:n], 0.0)
    d = an - a
    if np.max(np.abs(d)) < 1e-12:
        return a.copy(), False
    
    bi = mx
    ba = a.copy()
    for al in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01, 0.005):
        at = np.maximum(a + al * d, 0.0)
        at = _norm(at, S)
        v = _mc(at)
        if v < bi - 1e-14:
            bi = v
            ba = at
    if bi < mx - 1e-12:
        return ba, True
    return a.copy(), False

    ba = a.copy()
    bo = _mc(a)
    st = 0
    it = 0
    K = 20

        it += 1

                bo = o
                ba = a.copy()
                st = 0

                st += 1

            st += 1

            K = 5 if K > 5 else 20
            st = 0
        if it % 2000 == 0:
            print(f"      lp it={it} obj={bo:.6f} K={K} t={time.monotonic()-t0:.0f}")
    return ba, bo
