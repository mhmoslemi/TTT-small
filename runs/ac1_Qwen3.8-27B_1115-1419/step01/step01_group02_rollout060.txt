```python
"""
Optimizer for the evaluation function

    2 * n * max(np.convolve(a, a)) / (sum(a) ** 2)

where a is a nonnegative real sequence.

The program uses an adaptive softmax / log-sum-exp smoothing of the maximum
convolution coefficient, a logit parameterization a = softmax(x), and
momentum line-search updates.  It tries several initial shapes, a small
low-resolution pretraining phase, then refines the best path at the parent
length, optionally tests a longer interpolation, and finally evaluates a small
archive of candidates at several scalings with the actual evaluate_sequence().
"""

import numpy as np
import time
import math
from scipy.fft import fft, ifft


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if a.size == 0:
        return a
    if s <= 1e-15 or not np.isfinite(s):
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _resize_array(a, m):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if m == n:
        return a.copy()
    if n == 0:
        return np.zeros(m, dtype=np.float64)
    if n == 1:
        return np.full(m, float(a[0]), dtype=np.float64)
    old = np.linspace(0.0, 1.0, n, dtype=np.float64)
    new = np.linspace(0.0, 1.0, m, dtype=np.float64)
    return np.interp(new, old, a)


def _softmax(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x.copy()
    x = x - np.max(x)
    x = np.maximum(x, -745.0)
    e = np.exp(x)
    s = float(np.sum(e))
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s


def _hard_score_from_a(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 1e-15 or not np.isfinite(s):
        return np.inf
    N = _next_pow2(2 * n - 1)
    A = fft(a, N)
    b = ifft(A * A).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    mb = float(np.max(b))
    return float(2.0 * n * mb / (s * s))


def _score_soft_a(a, lam):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf, np.inf
    s = float(np.sum(a))
    if s <= 1e-15 or not np.isfinite(s):
        return np.inf, np.inf
    N = _next_pow2(2 * n - 1)
    A = fft(a, N)
    b = ifft(A * A).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    mb = float(np.max(b))
    score = float(2.0 * n * mb / (s * s))
    if lam <= 0.0:
        return score, score
    L = 2 * n - 1
    u = (L / (s * s)) * b
    z = lam * u
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return score, score
    ez = np.exp(z - zmax)
    ws = float(np.sum(ez))
    if ws <= 0.0 or not np.isfinite(ws):
        return score, zmax / lam
    F = (zmax + math.log(ws)) / lam
    return score, F


def _soft_and_grad_from_x(x, lam):
    a = _softmax(x)
    n = a.size
    if n == 0:
        return 0.0, 0.0, np.array([], dtype=np.float64)

    L = 2 * n - 1
    N = _next_pow2(3 * n - 2)

    A = fft(a, N)
    b = ifft(A * A).real[:L]
    b = np.maximum(b, 0.0)

    mb = float(np.max(b))
    if mb <= 0.0 or not np.isfinite(mb):
        return 0.0, 0.0, np.zeros(n, dtype=np.float64)

    # Since x is a softmax parameterization, sum(a) == 1 mathematically.
    score = float(2.0 * n * mb)

    if lam <= 0.0:
        idx = int(np.argmax(b))
        w = np.zeros(L, dtype=np.float64)
        w[idx] = 1.0
        F = float(L * mb)
    else:
        u = L * b
        z = lam * u
        zmax = float(np.max(z))
        if not np.isfinite(zmax):
            idx = int(np.argmax(b))
            w = np.zeros(L, dtype=np.float64)
            w[idx] = 1.0
            F = float(L * b[idx])
        else:
            ez = np.exp(z - zmax)
            ws = float(np.sum(ez))
            if ws <= 0.0 or not np.isfinite(ws):
                idx = int(np.argmax(b))
                w = np.zeros(L, dtype=np.float64)
                w[idx] = 1.0
                F = float(L * b[idx])
            else:
                w = ez / ws
                F = (zmax + math.log(ws)) / lam

    # Gradient of sum_k w_k coeff_k(a*a).
    # For coefficient k, grad_j = 2 a_{k-j} when 0 <= k-j < n.
    # This is obtained by linear convolution of reversed a with w.
    C = ifft(fft(a[::-1], N) * fft(w, N)).real
    G = L * 2.0 * C[n - 1:2 * n - 1]

    # Gradient with respect to softmax logits:
    # dx_j = a_j (G_j - sum_i a_i G_i).
    bias = float(np.dot(a, G))
    gx = a * (G - bias)

    if not np.all(np.isfinite(gx)):
        gx = np.zeros(n, dtype=np.float64)

    return float(F), float(score), gx


def _run_logit_stage(x, lam, deadline, eta_init=0.08, beta=0.9, rng=None, check_exact=False):
    x = np.asarray(x, dtype=np.float64).copy()
    if x.size == 0:
        return x, np.inf, x, np.inf
    if rng is None:
        rng = np.random.default_rng(1234567)

    n = x.size
    x = x - np.mean(x)
    best_x = x.copy()
    a0 = _softmax(best_x)
    best_score = _hard_score_from_a(a0)
    current_x = x.copy()
    current_score = best_score
    current_F = best_score

    m = np.zeros(n, dtype=np.float64)
    eta = float(eta_init)
    no_improve = 0

    while time.monotonic() < deadline:
        F, sco, gx = _soft_and_grad_from_x(x, lam)
        if not np.isfinite(sco):
            x = best_x + 0.1 * rng.normal(size=n)
            x = x - np.mean(x)
            m = np.zeros(n, dtype=np.float64)
            no_improve += 1
            if no_improve > 80:
                x = best_x + 0.35 * rng.normal(size=n)
                x = x - np.mean(x)
                no_improve = 0
            continue

        current_x = x.copy()
        current_score = float(sco)
        current_F = float(F)

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_x = x.copy()

        gmax = float(np.max(np.abs(gx)))
        if gmax < 1e-14:
            x = best_x + 0.05 * rng.normal(size=n)
            x = x - np.mean(x)
            m = np.zeros(n, dtype=np.float64)
            no_improve += 1
            if no_improve > 70:
                x = best_x + 0.22 * rng.normal(size=n)
                x = x - np.mean(x)
                no_improve = 0
            continue

        d = gx / gmax
        m = beta * m + (1.0 - beta) * d
        mmax = float(np.max(np.abs(m)))
        if mmax < 1e-14:
            m = d
        else:
            m = m / mmax

        accepted = False
        score_allow = 1e-9 + 1e-7 * max(1.0, best_score)
        f_tol = 1e-8

        trial_list = [eta, 0.5 * eta, 1.5 * eta, 0.25 * eta]
        for tt in trial_list:
            if tt <= 0.0:
                continue
            xn = x - tt * m
            xn = xn - np.mean(xn)

            span = float(np.max(xn) - np.min(xn))
            if span > 60.0:
                med = float(np.median(xn))
                xn = np.clip(xn, med - 30.0, med + 30.0)
                xn = xn - med

            an = _softmax(xn)
            sn, fn = _score_soft_a(an, lam)
            if not np.isfinite(sn) or not np.isfinite(fn):
                continue

            if sn < best_score - 1e-11:
                x = xn
                current_x = x.copy()
                current_score = float(sn)
                current_F = float(fn)
                if sn < best_score:
                    best_score = float(sn)
                    best_x = x.copy()
                accepted = True
                no_improve = 0
                break

            if sn <= current_score + score_allow and fn < current_F - f_tol:
                x = xn
                current_x = x.copy()
                current_score = float(sn)
                current_F = float(fn)
                if sn < best_score:
                    best_score = float(sn)
                    best_x = x.copy()
                accepted = True
                no_improve = 0
                break

        if accepted:
            eta = max(eta_init * 0.25, min(5.0, eta * 1.05))
        else:
            no_improve += 1
            eta = max(eta_init * 0.05, eta * 0.75)
            if no_improve > 50:
                x = best_x + 0.03 * rng.normal(size=n)
                x = x - np.mean(x)
                m = np.zeros(n, dtype=np.float64)
                eta = 0.5 * eta_init
                no_improve = 0
            elif no_improve > 160:
                x = best_x + 0.12 * rng.normal(size=n)
                x = x - np.mean(x)
                m = np.zeros(n, dtype=np.float64)
                eta = eta_init
                no_improve = 0
            elif no_improve > 420:
                x = best_x + 0.35 * rng.normal(size=n)
                x = x - np.mean(x)
                m = np.zeros(n, dtype=np.float64)
                eta = eta_init
                no_improve = 0

    return best_x, best_score, current_x, current_score


def _smooth_random_init(n, rng):
    if n <= 0:
        return np.array([], dtype=np.float64)
    k = np.arange(n, dtype=np.float64)
    base = 0.5 * np.ones(n, dtype=np.float64)
    for _ in range(5):
        f = int(rng.integers(1, max(2, n // 3)))
        amp = float(rng.uniform(0.1, 0.8))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        base = base + amp * np.cos((2.0 * math.pi * f * k) / float(n) + ph)
    a = np.exp(base)
    return _normalize(a)


def _make_initial_arrays(n, parent, rng):
    if n <= 0:
        return []
    out = []
    p = _normalize(parent)
    out.append(p)
    out.append(np.ones(n, dtype=np.float64))
    out.append(_normalize(rng.random(n) + 0.05))
    out.append(_smooth_random_init(n, rng))

    k = np.arange(n, dtype=np.float64)
    denom = float(max(1, n - 1))

    for f in (1, 2, 3, 5, 8):
        out.append(_normalize(1.0 + 0.25 * np.cos((2.0 * math.pi * f * k) / float(n))))
    out.append(_normalize(1.0 + 0.35 * np.sin((2.0 * math.pi * k) / float(n))))

    out.append(_normalize(np.linspace(0.25, 1.0, n, dtype=np.float64)))
    out.append(_normalize(np.linspace(1.0, 0.25, n, dtype=np.float64)))
    out.append(_normalize(0.2 + 0.8 * np.abs(k / denom - 0.5) * 2.0))
    out.append(_normalize(np.exp(-1.35 * np.linspace(-2.0, 2.0, n, dtype=np.float64) ** 2)))
    out.append(_normalize(np.exp(-2.0 * np.linspace(-1.0, 1.0, n, dtype=np.float64) ** 4)))

    cleaned = []
    for a in out:
        aa = np.asarray(a, dtype=np.float64)
        if aa.size != n or not np.all(np.isfinite(aa)):
            continue
        aa = _normalize(aa)
        cleaned.append(aa)
    return cleaned


def _add_archive(scores, seqs, a, sc, keep=8):
    if not np.isfinite(sc):
        return
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return
    if len(scores) < keep:
        scores.append(float(sc))
        seqs.append(a.copy())
        return
    worst = 0
    worst_val = scores[0]
    for i in range(1, len(scores)):
        if scores[i] > worst_val:
            worst_val = scores[i]
            worst = i
    if sc < worst_val - 1e-12:
        scores[worst] = float(sc)
        seqs[worst] = a.copy()


def _exact_eval_list(lst):
    try:
        val = float(evaluate_sequence(lst))
    except Exception:
        return np.inf
    if not np.isfinite(val):
        return np.inf
    return val


def _exact_eval_a(a, scale=1.0):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    aa = np.maximum(a, 0.0)
    lst = [float(v) for v in (aa * float(scale))]
    return _exact_eval_list(lst)


def propose_candidate(seed=42, budget_s=2400.0):
    t0 = time.monotonic()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 2400.0
    if not np.isfinite(total_budget) or total_budget <= 0.0:
        total_budget = 2400.0
    deadline = t0 + max(5.0, total_budget - 12.0)

    rng = np.random.default_rng(seed)

    # Recover a parent construction if present.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size < 2:
            raise ValueError("bad parent")
        parent = np.clip(parent, 0.0, 1000.0)
        if float(np.sum(parent)) <= 0.0:
            raise ValueError("zero parent")
    except Exception:
        parent = np.ones(5582, dtype=np.float64)

    n = int(parent.size)
    if n < 16:
        n = 5582
        parent = np.ones(n, dtype=np.float64)
    if n > 10000:
        parent = _resize_array(parent, 8000)
        n = 8000

    parent_norm = _normalize(parent)
    parent_score = _hard_score_from_a(parent_norm)
    print(f"start: n={n}, parent internal score={parent_score:.10f}, budget={total_budget:.1f}s")

    archive_scores = []
    archive_seqs = []
    best_internal = np.inf
    best_a = parent_norm.copy()

    inits = _make_initial_arrays(n, parent_norm, rng)
    for a in inits:
        aa = _normalize(a)
        sc = _hard_score_from_a(aa)
        if np.isfinite(sc) and sc < best_internal:
            best_internal = sc
            best_a = aa.copy()
        _add_archive(archive_scores, archive_seqs, aa, sc, keep=8)

    best_internal = min(best_internal, parent_score)
    if parent_score < best_internal:
        best_a = parent_norm.copy()

    _add_archive(archive_scores, archive_seqs, best_a, best_internal, keep=8)
    print(f"initial best internal={best_internal:.10f}")

    # Low-resolution pretraining: fast, then interpolate back.
    if n > 1200 and time.monotonic() < deadline - 260:
        m = min(1024, n)
        low_inits = []
        low_inits.append(_resize_array(best_a, m))
        low_inits.append(_resize_array(parent_norm, m))
        low_inits.append(_smooth_random_init(m, rng))
        low_inits.append(np.ones(m, dtype=np.float64))
        for a in archive_seqs[:4]:
            low_inits.append(_resize_array(a, m))

        low_best_a = None
        low_best_sc = np.inf
        for a in low_inits:
            aa = _normalize(a)
            sc = _hard_score_from_a(aa)
            if np.isfinite(sc) and sc < low_best_sc:
                low_best_sc = sc
                low_best_a = aa.copy()

        if low_best_a is not None:
            x_low = np.log(np.maximum(low_best_a, 1e-12) + 1e-12)
            x_low = x_low - np.mean(x_low)
            low_stages = [(4.0, 22.0), (12.0, 26.0), (35.0, 28.0)]
            for lam, dur in low_stages:
                now = time.monotonic()
                if now > deadline - 260:
                    break
                end = min(deadline - 260.0, now + dur)
                if end <= now:
                    break
                x_low, _, cx_low, csc_low = _run_logit_stage(
                    x_low, lam, end, eta_init=0.10, beta=0.92, rng=rng
                )
                if csc_low < low_best_sc + 1e-6:
                    x_low = cx_low
                    low_best_sc = csc_low
                low_best_a = _softmax(x_low)
                sc_now = _hard_score_from_a(low_best_a)
                if sc_now < low_best_sc:
                    low_best_sc = sc_now

            up = _resize_array(low_best_a, n)
            up = _normalize(up)
            sc_up = _hard_score_from_a(up)
            if np.isfinite(sc_up):
                _add_archive(archive_scores, archive_seqs, up, sc_up, keep=8)
                if sc_up < best_internal:
                    best_internal = sc_up
                    best_a = up.copy()

    # Main refinement at parent length.
    x = np.log(np.maximum(best_a, 1e-12) + 1e-12)
    x = x - np.mean(x)

    main_stages = [
        (3.0, 70.0),
        (6.0, 130.0),
        (15.0, 220.0),
        (40.0, 320.0),
        (100.0, 440.0),
        (300.0, 440.0),
    ]

    for lam, dur in main_stages:
        now = time.monotonic()
        if now > deadline - 55:
            break
        end = min(deadline - 55.0, now + dur)
        if end <= now:
            break

        before = time.monotonic()
        bx, bsc, cx, csc = _run_logit_stage(
            x, lam, end, eta_init=0.08, beta=0.90, rng=rng
        )
        elapsed = time.monotonic() - before

        # Prefer continuing from a slightly-worse but smoother point if available.
        if csc <= bsc + 1e-7:
            x = cx
            sc_now = csc
        else:
            x = bx
            sc_now = bsc

        a_now = _softmax(x)
        sc_now = _hard_score_from_a(a_now)
        if np.isfinite(sc_now) and sc_now < best_internal:
            best_internal = sc_now
            best_a = a_now.copy()
        _add_archive(archive_scores, archive_seqs, a_now, sc_now, keep=8)
        print(f"stage lam={lam:.1f}: elapsed={elapsed:.1f}s, current={sc_now:.10f}, best={best_internal:.10f}")

        if time.monotonic() > deadline - 120:
            break

    # Optional longer-length experiment.
    if n < 9000 and time.monotonic() < deadline - 170:
        n2 = min(10000, int(n * 1.5))
        if n2 > n:
            a2 = _resize_array(best_a, n2)
            a2 = a2 + 0.004 * rng.normal(size=n2)
            a2 = _normalize(a2)
            x2 = np.log(np.maximum(a2, 1e-12) + 1e-12)
            x2 = x2 - np.mean(x2)

            for lam, dur in [(70.0, 60.0), (220.0, 70.0)]:
                now = time.monotonic()
                if now > deadline - 90:
                    break
                end = min(deadline - 90.0, now + dur)
                if end <= now:
                    break
                bx2, bsc2, cx2, csc2 = _run_logit_stage(
                    x2, lam, end, eta_init=0.06, beta=0.90, rng=rng
                )
                if csc2 <= bsc2 + 1e-7:
                    x2 = cx2
                else:
                    x2 = bx2

            a2 = _softmax(x2)
            sc2 = _hard_score_from_a(a2)
            if np.isfinite(sc2):
                _add_archive(archive_scores, archive_seqs, a2, sc2, keep=8)
                if sc2 < best_internal:
                    best_internal = sc2
                    best_a = a2.copy()
                print(f"longer n={a2.size} internal={sc2:.10f}")

    # Add the current best and parent explicitly.
    _add_archive(archive_scores, archive_seqs, best_a, best_internal, keep=8)
    _add_archive(archive_scores, archive_seqs, parent_norm, parent_score, keep=8)

    # Final exact evaluation over a small archive and several scales.
    final_seq = None
    final_score = np.inf

    unique_candidates = []
    seen = []
    for a in archive_seqs:
        aa = np.asarray(a, dtype=np.float64)
        if aa.size == 0 or not np.all(np.isfinite(aa)):
            continue
        key = (int(aa.size), float(aa[0]), float(aa[-1]), float(np.sum(aa)))
        dup = False
        for sk in seen:
            if (
                sk[0] == key[0] and
                abs(sk[1] - key[1]) < 1e-12 and
                abs(sk[2] - key[2]) < 1e-12 and
                abs(sk[3] - key[3]) < 1e-10
            ):
                dup = True
                break
        if not dup:
            seen.append(key)
            unique_candidates.append(aa)

    for candidate in unique_candidates:
        an = _normalize(candidate)
        Ln = an.size
        if Ln == 0:
            continue
        max_an = float(np.max(an))
        base_scales = [1.0, math.sqrt(2.0 * float(Ln)), 10.0, 100.0, 1000.0]
        for scale in base_scales:
            scale = float(scale)
            if scale <= 0.0 or not np.isfinite(scale):
                continue
            if max_an * scale > 1000.0:
                scale = 1000.0 / max(max_an, 1e-300) * 0.999999
                if scale <= 0.0:
                    continue
            sc = _exact_eval_a(an, scale)
            if np.isfinite(sc):
                if sc < final_score:
                    final_score = sc
                    final_seq = [float(v) for v in (an * scale)]

    if final_seq is None or not np.isfinite(final_score):
        # Safe fallback: the best internal sequence at unit sum.
        an = _normalize(best_a)
        final_seq = [float(v) for v in an]
        final_score = _exact_eval_a(an, 1.0)

    print(f"final exact score={final_score:.10f}, length={len(final_seq)}")
    return final_seq
```
