from scipy.optimize import linprog
from scipy.linalg import toeplitz

def solve_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f, g)[k] <= M for all k."""
    n = len(f)
    L = 2*n - 1
    
    # convolve(f, g)[k] = sum_i f[i] * g[k-i] 
    # This is a linear function of g for fixed f
    # A[k, j] = coefficient of g[j] in convolve(f, g)[k]
    # convolve(f, g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j]
    # So A[k, j] = f[k-j] if k-j is in [0, n-1], else 0
    
    # Build the constraint matrix
    A = np.zeros((L, n))
    for k in range(L):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = f[k-j]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M (vector of M's)
    b = np.full(L, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res.x if res.success else None
```

import numpy as np
from scipy.optimize import linprog
import time

def fft_score(a):
    """Fast score computation using FFT."""
    n = len(a)
    if n == 0 or sum(a) < 1e-15:
        return np.inf
    m = 1
    while m < 2*n - 1:
        m *= 2
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2*n-1]
    b = np.maximum(b, 0)
    bmax = b.max()
    s = sum(a)
    return 2.0 * n * bmax / (s * s)

        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A

def solve_lp(f, M):
    """Solve max sum(g) s.t. g >= 0, A @ g <= M * ones."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

For column j, the nonzeros are in rows j through j+n-1, with values f[n-1], f[n-2], ..., f[0] (reading from row j to row j+n-1, the index k-j goes from 0 to n-1, so A[k,j] = f[k-j]).

Column j has A[j+j, j]... no wait. For column j, valid rows are k where 0 <= k-j <= n-1, i.e., j <= k <= j+n-1. And A[k, j] = f[k-j].

So column j = [0, ..., 0, f[0], f[1], ..., f[n-1], 0, ..., 0]^T with f[0] at position j.

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using alternating LP iterations.
The key idea: fix one side of the convolution, solve an LP to optimize the other,
alternate, and converge to a near-flat autocorrelation profile.
Combined with FFT-based gradient polish and multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _next_pow2(x):
    p = 1
    while p < int(max(1, x)):
        p <<= 1
    return p


def fft_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if bmax <= 0 or not np.isfinite(bmax):
        return np.inf
    return float(2.0 * n * bmax / (s * s))


def convolve_direct(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    return np.convolve(a, b)


def build_conv_matrix(f):
    n = int(f.size)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_solve(f, M, time_limit=120.0):
    n = int(f.size)
    L = 2 * n - 1
    A = build_conv_matrix(f)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(L, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def lp_iterate(f, n_iters, time_deadline, step_init=0.5, step_min=0.15):
    n = int(f.size)
    target_sum = float(f.sum())
    best_f = f.copy()
    best_score = fft_score(f)
    M = float(max(convolve_direct(f, f)))

    for it in range(n_iters):
        now = time.monotonic()
        if now > time_deadline - 30:
            break

        t = max(step_min, step_init * (0.985 ** it))
        g = lp_solve(f, M, time_limit=min(60.0, max(10.0, time_deadline - now - 10.0)))
        if g is None:
            break

        g_sum = float(g.sum())
        if g_sum < 1e-12:
            break
        g = g * (target_sum / g_sum)
        g = np.maximum(g, 0.0)

        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        fs = float(f_new.sum())
        if fs < 1e-12:
            break
        f_new = f_new * (target_sum / fs)

        new_M = float(max(convolve_direct(f_new, f_new)))
        score = float(2.0 * n * new_M / (target_sum * target_sum))

        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_f = f_new.copy()

        f = f_new
        M = new_M

        if (it + 1) % 10 == 0:
            print("    LP iter %d: M=%.6f score=%.6f t=%.3f" % (it, M, best_score, t))

    return best_f, best_score


def gradient_polish(a, n_iters, time_deadline):
    n = int(a.size)
    m = _next_pow2(2 * n - 1)
    v = np.log(np.maximum(a, 1e-300))
    v = v - float(v.mean())
    v = np.clip(v, -20.0, 20.0)

    best_a = a.copy()
    best_score = fft_score(a)

    def softmax(x):
        x = x - float(x.max())
        x = np.clip(x, -745.0, 700.0)
        e = np.exp(x)
        s = float(e.sum())
        if s < 1e-300:
            return np.full(x.shape, 1.0 / max(1, x.size))
        return e / s

    rng = np.random.default_rng(12345)
    mu = np.zeros(n)
    nu = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(n_iters):
        now = time.monotonic()
        if now > time_deadline - 5:
            break

        a = softmax(v)
        F = np.fft.fft(a, m)
        b = np.fft.ifft(F * F).real[:2 * n - 1]
        b = np.maximum(b, 0.0)
        bmax = float(b.max())
        if bmax <= 0 or not np.isfinite(bmax):
            break

        tau = bmax * 0.08
        x = (b - bmax) / tau
        x = np.clip(x, -50.0, 50.0)
        x -= float(x.max())
        w = np.exp(x)
        ws = float(w.sum())
        if ws < 1e-300:
            break
        w /= ws

        top_k = max(1, int(0.03 * (2 * n - 1)))
        idx = np.argpartition(b, -top_k)[-top_k:]
        w[idx] += 1.0
        w /= float(w.sum())

        P = np.fft.fft(w, m)
        c = np.fft.ifft(P * np.conj(F)).real[:n]
        cm = float(np.dot(c, a))
        gv = a * (c - cm)
        grms = float(np.sqrt(np.mean(gv * gv)))
        if grms < 1e-14:
            v += rng.normal(0.0, 0.1, n)
            v -= float(v.mean())
            continue
        gv /= grms
        t += 1
        mu = b1 * mu + (1 - b1) * gv
        nu = b2 * nu + (1 - b2) * gv * gv
        mhat = mu / max(1e-12, 1 - b1 ** t)
        vhat = nu / max(1e-12, 1 - b2 ** t)
        lr = 0.12
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -0.4, 0.4)
        v -= lr * step
        v -= float(v.mean())
        v = np.clip(v, -20.0, 20.0)

        if (it + 1) % 15 == 0:
            an = softmax(v)
            sc = fft_score(an)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = an.copy()

    an = softmax(v)
    sc = fft_score(an)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_f = an
    return best_f, best_score


def interp_to_length(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if a.size == 0 or s < 1e-15:
        return np.ones(n) / n
    if a.size == n:
        return a / s
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss < 1e-15:
        return np.ones(n) / n
    return out / ss
