    def __init__(self, length, sequence, score, seed=None):
        self.length = int(length)
        self.sequence = np.asarray(sequence, dtype=np.float64)
        self.score = float(score)
        self.seed = seed if seed is not None else 42
        
        # Quick FFT computation for caching
        if length > 1:
            conv_result = fast_convolve(self.sequence)
            self.max_conv = np.max(conv_result) if len(conv_result) > 0 else 0
        else:
            self.max_conv = np.inf
            
        self.valid = self.length > 0 and np.min(self.sequence) >= 0.0

    def evaluate_with_clamp(self):
        """Full evaluation using the provided secure evaluator."""
        seq_list = [float(x) for x in self.sequence]
        if not self.valid or len(seq_list) == 0:
            return float('inf')
        try:
            return fast_score(seq_list, target_sum=np.sqrt(2 * self.length) if self.length > 0 else None)
        except:
            return float('inf')

# Helper functions for normalization and evaluation
def normalize_sequence(seq, target_sum):
    """Rescale a sequence to have a given sum."""
    a = np.array(seq)
    s = float(a.sum())
    if s <= 1e-15:
        a.fill(target_sum / len(a))
    else:
        a = a * (target_sum / s)
    for i in range(len(a)):
        a[i] = np.clip(a[i], 0.0, float(1e3))
    return a.tolist()

def evaluate_sequence(seq):
    """Official objective function evaluator (same as given)"""
    # Convert to float list and validate as per the problem
    if not isinstance(seq, (list, np.ndarray)):
        return float('inf')
        
    a = np.array(seq, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
        
    # Defensive clamping to enforce [0, 1000] bounds
    a = np.clip(a, 0.0, 1000.0)
    a[a == np.inf] = 0.0
    a[a == -np.inf] = 0.0
    
    # Average-out the main structure if needed
    if np.mean(a) < 0.0:
        a[a < 0.0] = np.max(a) if len(a)>0 else 0.0
        
    # Main branch of objective calculation
    if any(a > 1000.0):
        a = np.minimum(a, 1000.0)
    
    if any(a < 0.0):
        a[a < 0.0] = float(1e-5)  # Fill negatives with a tiny value to avoid division by zero
    
    conv = fast_convolve(a)  if n > 0 else np.hstack((0, 0))
    if not np.isfinite(np.max(conv)):
        return float('inf')
    
    score = 2.0 * n * np.max(conv) / (np.sum(a)**2)
    if not np.isfinite(score):
        score = float('inf')
    return score
