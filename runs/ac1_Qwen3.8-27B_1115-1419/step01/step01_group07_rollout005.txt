def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    return e / s


def lse_smoothed(c, tau):
    c_max = c.max()
    return c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))


def lse_grad_a(a, tau):
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    return 2.0 * np.correlate(w, a, mode='valid')

    a = softmax(x)
    c = np.convolve(a, a)
    return lse_smoothed(c, tau)

    a = softmax(x)
    g_a = lse_grad_a(a, tau)

    return [float(v) for v in best_a]
