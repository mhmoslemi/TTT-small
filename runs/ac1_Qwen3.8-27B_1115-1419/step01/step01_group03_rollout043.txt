import numpy as np
import time

def fast_score(a):
    """Compute the score without calling evaluate_sequence."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def softmax_gradient(theta, top_k, smooth_beta):
    """
    Compute gradient of soft-max of convolution w.r.t. theta parameters.
    Returns (a, grad_theta, score)
    """
    n = len(theta)
    # Softmax
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    a = exp_theta / exp_theta.sum()
    
    # Convolution
    c = np.convolve(a, a)
    
    # Active set with smooth weights
    # Use soft weights: w_k = exp(beta * c_k) / sum(exp(beta * c_j))
    # But for numerical stability, only compute for top candidates
    # Find top (top_k + extra) to determine the range
    extra = 5
    num_to_check = min(len(c), top_k + extra)
    
    # Get the top candidates
    # Use argpartition for efficiency
    part_idx = np.argpartition(c, -num_to_check)[-num_to_check:]
    sorted_idx = part_idx[np.argsort(c[part_idx])[::-1]]
    
    # Compute weights using softmax
    top_vals = c[sorted_idx[:top_k]]
    w = np.exp(smooth_beta * (top_vals - top_vals.max()))
    w /= w.sum()
    
    # Compute gradient g_j = 2 * sum_k w_k * a[k-j]
    g = np.zeros(n)
    for idx, w_k in zip(sorted_idx[:top_k], w):
        k = int(idx)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += 2.0 * w_k * a[k - np.arange(lo, hi)]
    
    # Gradient w.r.t. theta: a * (g - a.dot(g))
    g_bar = np.dot(a, g)
    grad_theta = a * (g - g_bar)
    
    return a, grad_theta, c.max()

def adam_optimize(a0, budget, lr_init=0.05, top_k_init=3, top_k_max=50, 
                  smooth_beta=500, beta1=0.9, beta2=0.999, eps=1e-8,
                  eval_interval=5, seed=42):
    """
    Optimize using Adam with softmax parameterization.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    
    # Initialize theta
    a_init = a0.copy()
    a_init = np.maximum(a_init, 1e-30)
    a_init = a_init / a_init.sum()
    theta = np.log(a_init)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a0.copy()
    best_score = fast_score(a0)
    stagnant = 0
    
    deadline = time.monotonic() + budget
    
    while time.monotonic() < deadline:
        t += 1
        # Adaptive parameters
        # Increase top_k over time
        progress = min(1.0, t / 5000.0)
        top_k = int(top_k_init + (top_k_max - top_k_init) * progress)
        
        # Decay learning rate
        lr = lr_init / (1.0 + 0.001 * t)
        
        a, grad_theta, _ = softmax_gradient(theta, top_k, smooth_beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation
        if t % eval_interval == 0:
            score = fast_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
            
            if t % 100 == 0:
                print(f"  Adam t={t}, score={best_score:.8f}, stagnant={stagnant}")
            
            if stagnant > 200:
                # Restart with perturbation
                print(f"  Adam restart at t={t}, score={best_score:.8f}")
                theta = np.log(np.maximum(best_a, 1e-30)) + rng.normal(0, 0.01, n)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stagnant = 0
    
    # Final check
    a_final = np.exp(theta - theta.max())
    a_final = a_final / a_final.sum()

def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def compute_gradient(a, c, top_k, smooth_beta):
    """Compute gradient of soft-max convolution w.r.t. a (not theta)."""
    n = len(a)
    # Find top-k peaks
    if top_k >= len(c):
        idx = np.arange(len(c))
        vals = c
    else:
        idx = np.argpartition(c, -top_k)[-top_k:]
        vals = c[idx]
    
    # Softmax weights
    vals_shifted = vals - vals.max()
    w = np.exp(smooth_beta * vals_shifted)
    w /= w.sum()
    
    # Gradient: g_j = 2 * sum_k w_k * a[k-j]
    g = np.zeros(n)
    for k, w_k in zip(idx, w):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += 2.0 * w_k * a[k - np.arange(lo, hi)]
    
    return g

    v = np.zeros(n)
    t = 0

        t += 1
        progress = min(1.0, t / max(1, total_iters_planned))
        
        # Interpolate parameters
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        smooth_beta = beta_start + (beta_end - beta_start) * progress
        lr_t = lr / (1.0 + 0.0005 * t)

        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        a = exp_theta / exp_theta.sum()

        g = compute_gradient(a, c, top_k, smooth_beta)

        grad_theta = a * (g - g_bar)

            if score < best_score:
                best_score = score
                best_a = a.copy()

            else:
                stagnant += 1

                # Perturb and restart

"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using FISTA with soft active-set gradients
and adaptive annealing of the peak-selection temperature. Key innovations over
the previous hard top-k approach:
  1. Exponential (soft) weights over active peaks with annealed temperature.
  2. L2-norm warm-up phase to find a good basin before switching to L-infinity.
  3. Backtracking line search for guaranteed descent.
  4. Multiple diverse initializations and restart strategies.
"""

import numpy as np
import time


def fast_score(a):
    """Compute 2n*max(conv)/(sum)^2 without calling evaluate_sequence."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {a>=0, sum(a)=1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    j = np.where(cond)[0][-1]
    theta = (cssv[j] - 1.0) / (j + 1)
    return np.maximum(v - theta, 0.0)


def soft_gradient_inf(a, c, top_k, beta):
    """
    Compute gradient of soft-max of convolution peaks.
    Uses exponential weights over top-k peaks.
    Returns gradient w.r.t. a (length n).
    """
    n = len(a)
    k_eff = min(top_k, len(c))
    idx = np.argpartition(c, -k_eff)[-k_eff:]
    vals = c[idx]
    w = np.exp(beta * (vals - vals.max()))
    w /= w.sum()
    g = np.zeros(n, dtype=np.float64)
    for ki, wi in zip(idx, w):
        k = int(ki)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += 2.0 * wi * a[k - np.arange(lo, hi)]
    return g


def l2_gradient(a, c):
    """
    Compute gradient of sum(c_k^2) = ||conv(a)||_2^2 w.r.t. a.
    This is 4 * cross_correlation(c, a) evaluated at positions 0..n-1.
    """
    n = len(a)
    # g_j = 4 * sum_k c_k * a[k-j]
    # Compute using direct summation for correctness
    g = np.zeros(n, dtype=np.float64)
    # For efficiency, only iterate over non-zero c entries (all are non-zero for non-trivial a)
    # Use the fact that sum_k c_k * a[k-j] is a cross-correlation
    # c has length 2n-1, a has length n
    # corr[j] = sum_{k=0}^{2n-2} c[k] * a[k-j] (valid when 0<=k-j<n)
    for k in range(len(c)):
        if c[k] == 0:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += c[k] * a[k - np.arange(lo, hi)]
    g *= 4.0
    return g


def fista_optimize(a0, budget, lr_init=0.5, top_k_start=3, top_k_end=80,
                   beta_start=50.0, beta_end=500.0, mode="inf",
                   use_line_search=True, max_stagnant=80, seed=42):
    """
    FISTA optimization with soft active-set gradient and adaptive parameters.
    mode='inf': minimize max(c) using soft-max gradient.
    mode='l2': minimize ||c||_2^2 (for warm-up).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = project_simplex(a0.copy())
    a_prev = a.copy()
    t_iter = 1.0
    lr = lr_init
    best_a = a.copy()
    best_score = fast_score(a)
    stagnant = 0
    it = 0

    deadline = time.monotonic() + budget

    while time.monotonic() < deadline:
        it += 1
        # Nesterov momentum
        t_old = t_iter
        t_iter = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_iter
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy

        # Compute gradient
        if mode == "l2":
            c_y = np.convolve(y, y)
            g = l2_gradient(y, c_y)
        else:
            progress = min(1.0, it / 8000.0)
            top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
            beta = beta_start + (beta_end - beta_start) * progress
            c_y = np.convolve(y, y)
            g = soft_gradient_inf(y, c_y, top_k, beta)

        # Gradient step
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        # Line search (backtracking)
        if use_line_search:
            sc = fast_score(a_new)
            sc_old = fast_score(y)
            step_factor = 1.0
            for _ in range(3):
                if sc < sc_old + 1e-12:
                    break
                step_factor *= 0.5
                a_new = y - lr * step_factor * g
                a_new = project_simplex(a_new)
                sc = fast_score(a_new)
            if lr * step_factor > lr * 0.01:
                lr = lr * step_factor * 1.0  # keep effective lr
        else:
            sc = None

        a_prev = a
        a = a_new

        # Periodic evaluation and adaptation
        if it % 5 == 0:
            if sc is None:
                sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1

            # Adapt learning rate
            if stagnant == 20:
                lr *= 0.85
            elif stagnant == 40:
                lr *= 0.7
                a_prev = best_a.copy()
                t_iter = 1.0
            elif stagnant > max_stagnant:
                # Restart with perturbation
                scale = rng.choice([0.001, 0.005, 0.01, 0.02])
                a = best_a + rng.normal(0, scale / np.sqrt(n), n)
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-10:
                    a = a / sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t_iter = 1.0
                lr = lr_init * rng.uniform(0.3, 1.0)
                stagnant = 0
                if it % 200 == 0:
                    print(f"    FISTA[{mode}] restart it={it}, score={best_score:.8f}")

            if it % 500 == 0:
                print(f"    FISTA[{mode}] it={it}, score={best_score:.8f}, lr={lr:.4f}")

    # Final check
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score
