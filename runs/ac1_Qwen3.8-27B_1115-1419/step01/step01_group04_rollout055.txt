Actually, for coordinate descent, the 1D subproblem is:

This is: minimize max_k [(B_k + a_i * G_k) / (S_0 + a_i)^2]

Let f_k(a_i) = (B_k + a_i * G_k) / (S_0 + a_i)^2
We want a_i ≥ 0 that minimizes max_k f_k(a_i).

Setting f_k'(a_i) = 0: a_i = (G_k*S_0 - 2*B_k) / G_k = S_0 - 2*B_k/G_k (if G_k ≠ 0)

"""
Searches for non-negative step-function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2.

Approach:
1. Exploit the U-shape insight: optimal sequences are symmetric with high values
   at edges and low values in the middle, compensating for the triangular
   autocorrelation envelope.
2. Parameterized shape search using scipy.optimize on low-dimensional families.
3. Full gradient-based refinement using log-sum-exp smoothed objective + Adam.
4. Coordinate descent polish for final improvements.
5. Multi-resolution: try various sequence lengths and keep the best.
"""

import numpy as np
import time
import math
from scipy.optimize import minimize, minimize_scalar


def _fast_conv(a):
    """FFT-based linear convolution a * a."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(c, 0.0)


def _score(a):
    """Fast score computation."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _fast_conv(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _smooth_score_grad(a, beta):
    """Smoothed score and gradient using log-sum-exp."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return float('inf'), np.zeros_like(a)
    c = _fast_conv(a)
    cmax = float(c.max())
    shifted = beta * (c - cmax)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    w = e / float(e.sum())
    ls = cmax + float(np.log(float(e.sum()))) / beta

    # gradient of ls w.r.t. a[i] = 2 * sum_j a[j] * w[i+j]
    N2 = 1 << (3 * n).bit_length()
    FA = np.fft.fft(a.astype(np.float64), N2)
    FW = np.fft.fft(w.astype(np.float64), N2)
    cross = np.fft.ifft(np.conj(FA) * FW).real[:n]
    grad_ls = 2.0 * cross

    sc = 2.0 * n * ls / (s * s)
    grad = 2.0 * n * grad_ls / (s * s) - 4.0 * n * ls * a / (s * s * s)
    return sc, grad


def _normalize(a, target_sum):
    """Normalize sequence to target sum, keeping non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 10000.0)
    s = float(a.sum())
    if s < 1e-15:
        a.fill(target_sum / float(a.size))
    else:
        a *= target_sum / s
    a = np.clip(a, 0.0, 1000.0)
    return a


def _make_u_shape(n, alpha, beta_p=4.0):
    """Create a U-shaped sequence: high at edges, low in middle."""
    t = np.linspace(-1.0, 1.0, n)
    a = 1.0 + alpha * np.abs(t) ** beta_p
    return _normalize(a, float(n))


def _make_two_level(n, m, ratio):
    """Two-level sequence: A at edges, B in middle."""
    m = max(1, min(m, n - 1))
    a = np.full(n, 1.0 / (m * ratio + (n - 2 * m) + m * ratio), dtype=np.float64)
    a[:m] = ratio / a[m]
    a[-m:] = ratio / a[m]
    return _normalize(a, float(n))


def _make_fourier_shape(n, coeffs):
    """Create sequence from Fourier coefficients (even harmonics for symmetry)."""
    t = np.linspace(-1.0, 1.0, n)
    a = np.ones(n, dtype=np.float64)
    for k, c in enumerate(coeffs, start=1):
        a += c * np.cos(np.pi * k * t)
    a = np.maximum(a, 0.01)
    return _normalize(a, float(n))


def _param_search_u(n, alpha_range=(-0.5, 2.0), beta_values=(2.0, 3.0, 4.0, 6.0, 8.0)):
    """Search over U-shape parameters."""
    best_a = None
    best_s = float('inf')
    for bp in beta_values:
        for alpha in np.linspace(alpha_range[0], alpha_range[1], 30):
            a = _make_u_shape(n, alpha, bp)
            s = _score(a)
            if s < best_s:
                best_s = s
                best_a = a.copy()
    return best_a, best_s


def _param_search_fourier(n, n_coeffs=4):
    """Search over Fourier shapes using scipy."""
    best_a = None
    best_s = float('inf')

    # Initial guesses
    for init in [np.zeros(n_coeffs), np.random.default_rng(0).uniform(-0.5, 0.5, n_coeffs)]:
        def obj(params):
            a = _make_fourier_shape(n, params.tolist())
            return _score(a)

        res = minimize(obj, init, method='Nelder-Mead',
                       options={'maxfev': 2000, 'xatol': 0.01, 'fatol': 1e-5})
        if res.fun < best_s:
            best_s = res.fun
            best_a = _make_fourier_shape(n, res.x.tolist())

    return best_a, best_s


def _param_search_block(n):
    """Search over block (two-level) shapes."""
    best_a = None
    best_s = float('inf')
    for frac in (0.05, 0.1, 0.15, 0.2, 0.25, 0.3):
        m = int(frac * n)
        for ratio in (1.5, 2.0, 3.0, 4.0, 5.0, 7.0, 10.0):
            a = _make_two_level(n, m, ratio)
            s = _score(a)
            if s < best_s:
                best_s = s
                best_a = a.copy()
    return best_a, best_s


def _adam_refine(a0, budget_s, seed=0, beta_start=10.0, beta_end=200.0):
    """Refine using Adam optimizer on smoothed objective."""
    a = a0.astype(np.float64).copy()
    n = a.size
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_step = 0
    best_a = a.copy()
    best_score = _score(a)

    target_sum = float(n)
    a = _normalize(a, target_sum)

    stagnation = 0
    while time.monotonic() < deadline:
        t_step += 1
        # Anneal beta
        frac = min(1.0, t_step / max(1, int(budget_s / 5.0 * 10)))
        beta = beta_start + (beta_end - beta_start) * frac

        sc, grad = _smooth_score_grad(a, beta)
        if not np.isfinite(sc) or not np.all(np.isfinite(grad)):
            break

        # Normalize gradient (remove mean to preserve sum approximately)
        grad = grad - float(np.mean(grad))

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, target_sum)

        # Evaluate
        sc_new = _score(a_new)
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        a = a_new

        # Adaptive learning rate
        if stagnation > 200:
            lr = max(0.001, lr * 0.9)
            if stagnation > 500:
                # Reset with perturbation
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = _normalize(a, target_sum)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.05
                stagnation = 0

        if stagnation % 50 == 0:
            print(f"  Adam iter {t_step}, beta={beta:.1f}, score={best_score:.6f}, lr={lr:.5f}")

    return best_a, best_score


def _coord_descent_polish(a0, budget_s):
    """Coordinate descent: optimize one element at a time."""
    a = a0.astype(np.float64).copy()
    n = a.size
    t0 = time.monotonic()
    deadline = t0 + budget_s
    target_sum = float(n)

    best_a = a.copy()
    best_score = _score(a)
    no_improve = 0

    while time.monotonic() < deadline:
        improved = False
        for i in range(n):
            if time.monotonic() > deadline - 1.0:
                break
            old_val = a[i]
            if old_val < 1e-10:
                continue

            # 1D subproblem: minimize max_k (B_k + a_i * G_k) / (S0 + a_i)^2
            # where B_k = c_k with a_i=0, G_k = dc_k/da_i, S0 = sum - a_i
            # Simplified: just try a few values around current and pick best
            # For efficiency, use a small local search

            a[i] = 0.0
            s0 = float(a.sum())
            if s0 < 0.1:
                a[i] = old_val
                continue

            # G[k] = dc_k/da_i = 2*a_{k-i} (valid) + 2*a_{i-k} (valid) -- but for self-conv
            # Actually dc_k/da_i: c_k = sum_j a_j * a_{k-j}
            # d/d a_i: a_{k-i} (from j=i term) + a_{k-i} (from k-j=i term, i.e., j=k-i)
            # = 2*a_{k-i} if 0 <= k-i < n
            # But if k-i == i (k=2i), it's 2*a_i (the same term counted once, derivative of a_i^2 is 2*a_i)
            # Wait, that's the same formula: 2*a_{k-i} when k-i=i gives 2*a_i. OK.

            # B[k] = c_k with a_i = 0
            c_zero = _fast_conv(a)  # a[i]=0 now
            a[i] = old_val

            # G as a vector: G[k] = 2*a_{k-i} if 0<=k-i<n, else 0
            G = np.zeros(2 * n - 1, dtype=np.float64)
            lo = max(0, i - (n - 1))  # not needed, k-i in [0,n-1] means k in [i, i+n-1]
            k_start = i
            k_end = min(2 * n - 1, i + n)
            # G[k] = 2 * a[k-i] for k in [i, i+n-1] (but k-i < n, so k < i+n, and k >= 0)
            valid_k = np.arange(max(0, i), min(2 * n - 1, i + n))
            G[valid_k] = 2.0 * a[valid_k - i]

            # Now minimize max_k (B[k] + x*G[k]) / (S0+x)^2 over x >= 0
            # Since denominator is the same for all k, minimize max_k (B[k] + x*G[k]) / (S0+x)^2
            # Equivalent to: find x that minimizes the max of these ratio functions.
            # Each ratio (B[k]+x*G[k])/(S0+x)^2 has at most one minimum.
            # The optimal x is where the top constraint is minimized.

            # Quick approach: try golden-section or bisection on the active constraint
            # For speed, just evaluate at a few candidate points

            s0 = float(a.sum())  # sum with a[i] at old value; we set a[i]=0 above then restored
            s0 = s0 - old_val  # sum excluding a[i]

            # Candidate x values: current, 0, and the minimizer of the top constraint
            top_k = int(np.argmax(c_zero + old_val * G))
            Bk = c_zero[top_k]
            Gk = G[top_k]
            if Gk > 1e-15:
                x_star = (Gk * s0 - 2.0 * Bk) / Gk
                if x_star < 0:
                    x_star = 0.0
            else:
                x_star = old_val

            candidates = [0.0, old_val, max(0.0, x_star), max(0.0, old_val * 0.5), max(0.0, old_val * 1.5), x_star]
            best_x = old_val
            best_sub = float('inf')

            for x in candidates:
                x = max(0.0, x)
                val = float(np.max(c_zero + x * G)) / (s0 + x) ** 2
                if val < best_sub:
                    best_sub = val
                    best_x = x

            # Fine-tune around best_x with a few more points
            if best_x > 0:
                for delta in (-0.3, -0.1, 0.1, 0.3):
                    x = best_x * (1 + delta)
                    if x < 0:
                        x = 0.0
                    val = float(np.max(c_zero + x * G)) / (s0 + x) ** 2
                    if val < best_sub:
                        best_sub = val
                        best_x = x

            a[i] = best_x
            s_new = float(a.sum())
            if s_new < 0.1:
                a[i] = old_val
                continue

            # Quick score check
            c_new = c_zero + best_x * G
            sc_est = float(np.max(c_new)) / (s0 + best_x) ** 2 * 2.0 * n
            full_score = _score(a)

            if full_score < best_score - 1e-8:
                best_score = full_score
                best_a = a.copy()
                improved = True
            else:
                a[i] = old_val  # revert

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_a, best_score


def _multi_start_gradient(a0, budget_s, n_starts=3, seed=42):
    """Run gradient refinement from multiple perturbed starts."""
    rng = np.random.default_rng(seed)
    n = a0.size
    target_sum = float(n)
    best_a = a0.copy()
    best_s = _score(a0)

    per_start = budget_s / n_starts
    for i in range(n_starts):
        if time.monotonic() > time.monotonic() + budget_s - 5:
            break
        if i == 0:
            start = a0.copy()
        else:
            start = a0 * np.exp(rng.normal(0, 0.15, n))
            start = _normalize(start, target_sum)

        budget_each = min(per_start, max(10.0, budget_s - (i * per_start) - 5.0))
        a_new, s_new = _adam_refine(start, budget_each, seed=seed + i)
        if s_new < best_s:
            best_s = s_new
            best_a = a_new

    return best_a, best_s


def _upsample(seq, n):
    """Linearly interpolate to length n and normalize."""
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    if m == 0:
        return _normalize(np.ones(n), float(n))
    if m == n:
        return _normalize(seq, float(n))
    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return _normalize(out, float(n))


def _sanitize(seq):
    """Return valid Python list."""
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return [1.0] * 1024
    scale = float(seq.size) / s
    if seq.max() * scale <= 1000.0:
        seq = seq * scale
    seq = np.clip(seq, 0.0, 1000.0)
    if float(seq.sum()) < 0.01:
        return [1.0] * 1024
    return [float(x) for x in seq]

        global_best_seq = _normalize(np.ones(1024), 1024.0)
        global_best_score = _score(global_best_seq)

    # Try refining a few of the best candidates
    n_refine = global_best_seq.size
    # Also try a nearby length
    if n_refine < 4096 and budget > 600:
        n_try = min(4096, n_refine * 2)
    else:
        n_try = n_refine

    remaining = budget - (time.monotonic() - t0) - 120.0
    stage2_budget = min(remaining * 0.5, 600.0)

    if stage2_budget > 30:
        # Refine the global best
        a_ref, s_ref = _multi_start_gradient(global_best_seq, stage2_budget,
                                             n_starts=2, seed=seed)
        _update_best(a_ref, s_ref)

    print(f"  Stage 2 done in {time.monotonic()-t2:.1f}s, best={global_best_score:.6f}")

    # === Stage 3: Try a different n for refinement ===
    t3 = time.monotonic()
    remaining = budget - (time.monotonic() - t0) - 90.0
    stage3_budget = min(remaining * 0.3, 300.0)

    if stage3_budget > 20 and budget > 800:
        # Try refining at a larger n
        for n_try in [n_refine * 2, n_refine * 3]:
            if n_try > 16384 or time.monotonic() - t3 > stage3_budget * 0.7:
                break
            a_start = _upsample(global_best_seq, n_try)
            a_ref2, s_ref2 = _adam_refine(a_start, stage3_budget * 0.5, seed=seed + 99)
            _update_best(a_ref2, s_ref2)

    print(f"  Stage 3 done in {time.monotonic()-t3:.1f}s, best={global_best_score:.6f}")

    # === Stage 4: Coordinate descent polish ===
    print("Stage 4: Coordinate descent polish")
    t4 = time.monotonic()
    remaining = budget - (time.monotonic() - t0) - 60.0
    stage4_budget = max(10.0, min(remaining * 0.6, 300.0))

    # Only do CD if n is manageable
    if global_best_seq.size <= 4096 and stage4_budget > 20:
        a_cd, s_cd = _coord_descent_polish(global_best_seq, stage4_budget)
        _update_best(a_cd, s_cd)
    elif stage4_budget > 20:
        # For large n, do a final Adam pass with higher beta
        a_ref3, s_ref3 = _adam_refine(global_best_seq, stage4_budget,
                                      seed=seed + 777, beta_start=50.0, beta_end=500.0)
        _update_best(a_ref3, s_ref3)

    print(f"  Stage 4 done in {time.monotonic()-t4:.1f}s, best={global_best_score:.6f}")

    # === Stage 5: Final validation ===
    print("Stage 5: Validation")
    result = _sanitize(global_best_seq)
    final_score = float('inf')
    try:
        final_score = evaluate_sequence(result)
        print(f"  Final exact score: {final_score:.6f}")
    except Exception as e:
        print(f"  Validation error: {e}")

    # Fallback
    if not np.isfinite(final_score) or final_score > 2.0:
        result = _sanitize(_normalize(np.ones(1024), 1024.0))
        try:
            final_score = evaluate_sequence(result)
        except Exception:
            pass
        print(f"  Fallback score: {final_score:.6f}")

    total_time = time.monotonic() - t0
    print(f"Total time: {total_time:.1f}s")
    print(f"Final: n={len(result)}, score={final_score:.6f}")

    return result
