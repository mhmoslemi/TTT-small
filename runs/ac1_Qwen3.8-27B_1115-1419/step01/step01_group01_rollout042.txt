def _soft(a, temp=1.0):
    """Soft function for log-space optimization."""
    a = np.asarray(a, dtype=np.float64)
    a = a - a.max()
    a = np.clip(a, -700, 700)
    e = np.exp(a / temp)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(a.shape, 1.0 / a.size)
    return e / s


def _adaptive_spectral(a_init, n, m, iters=40, target_flatness=0.8):
    """
    Adaptive spectral method: iteratively find a whose auto-convolution
    matches a target that is progressively flattened.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S <= 0:
        a = np.ones(n)
        S = n
    
    best_a = a.copy()
    best_sc = _score_fft(a)
    
    L = 2 * n - 1
    # Initial target: current auto-convolution normalized
    b = np.fft.ifft(np.fft.fft(a, m) ** 2).real[:L]
    b = np.maximum(b, 0)
    b_max = b.max()
    if b_max > 0:
        target = b / b_max
    else:
        target = np.ones(L)
    
    for it in range(iters):
        # Blend target with flat
        flatness = target_flatness * (it / max(1, iters - 1))
        target_new = (1 - flatness) * target + flatness * (float(target.sum()) / L)
        target_new = np.maximum(target_new, 0.01 * target_new.max())
        
        # Spectral factorization: find a such that |F_a|^2 ~ target
        T = np.fft.fft(target_new[:m], m)
        # Take square root of spectrum (real part)
        T_pos = np.maximum(T, 0)
        sqrt_T = np.fft.ifft(np.sqrt(T_pos)).real[:n]
        
        # Alternative: use the phase of current a
        F_a = np.fft.fft(a, m)
        phase = np.angle(F_a)
        mag = np.sqrt(np.maximum(T, 0))
        F_new = mag * np.exp(1j * phase)
        a_new = np.fft.ifft(F_new).real[:n]
        
        # Blend
        a_new = np.maximum(a_new, 0)
        blend = 0.5
        a = (1 - blend) * a + blend * a_new
        a = np.maximum(a, 1e-10)
        
        # Update target
        b = np.fft.ifft(np.fft.fft(a, m) ** 2).real[:L]
        b = np.maximum(b, 0)
        b_max = b.max()
        if b_max > 0:
            target = b / b_max
        
        sc = _score_fft(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _adam_optimize(a_init, n, seconds, seed):
    """Adam optimization in log-space with soft-max gradient."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-10)
    
    # Log-space parameterization
    v = np.log(a / a.sum())  # log of normalized a
    v = v - v.mean()  # center
    
    m_fft = _next_pow2(2 * n - 1)
    
    # Adam state
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    mu = np.zeros(n)
    nu = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_sc = _score_fft(a)
    
    start = time.monotonic()
    deadline = start + seconds
    last_improve = start
    
    check_every = max(1, int(n / 512))
    
    while time.monotonic() < deadline:
        # Compute a from v
        a = np.exp(v - v.max())
        a = a / a.sum()
        
        # Compute gradient
        F = np.fft.fft(a, m_fft)
        b = np.fft.ifft(F * F).real[:2*n-1]
        b = np.maximum(b, 0)
        b_max = float(b.max())
        if b_max <= 0 or not np.isfinite(b_max):
            break
        
        # Soft-max temperature (annealed)
        progress = min(1.0, (time.monotonic() - start) / seconds)
        tau = b_max * (0.15 * (1 - progress) + 0.03)
        
        x = (b - b_max) / tau
        x = np.clip(x, -80, 80)
        x = x - x.max()
        w = np.exp(x)
        w_sum = w.sum()
        if w_sum <= 0:
            break
        w = w / w_sum
        
        # Gradient: G_i = (2n/S^2) * (2*X[i] - 2*Y/S)
        # where X[i] = sum_j w[i+j]*a[j], Y = sum_k w[k]*b[k]
        S = float(a.sum())  # should be 1 since we normalized
        Y = float(np.dot(w, b))
        
        # X[i] = sum_j w[i+j] * a[j] for j=0..n-1
        # = cross-correlation of w and a at lag i
        # Compute via FFT
        m_cross = _next_pow2(2*n - 1 + n - 1)  # = next_pow2(3n-2)
        W = np.fft.fft(w, m_cross)
        A = np.fft.fft(a, m_cross)
        X = np.fft.ifft(W * np.conj(A)).real[:n]
        
        G = (2.0 * n / (S * S)) * (2.0 * X - 2.0 * Y / S)
        
        # Convert to log-space gradient
        # a = exp(v) / sum(exp(v)), so da_i/dv_j = a_i*(delta_{ij} - a_j)
        # G_log = G * a  (since d obj / dv = sum_i G_i * da_i/dv = sum_i G_i * a_i*(delta - a) ... )
        # Actually: obj = f(a), a_i = exp(v_i)/sum(exp(v))
        # df/dv_j = sum_i (df/da_i) * (da_i/dv_j) = sum_i G_i * a_i*(delta_{ij} - a_j)
        #         = G_j * a_j - a_j * sum_i G_i * a_i
        grad_log = G * a - a * np.dot(G, a)
        
        # Normalize gradient
        g_norm = np.sqrt(np.mean(grad_log**2))
        if g_norm < 1e-15:
            v += rng.normal(0, 0.1, n)
            v -= v.mean()
            continue
        grad_log = grad_log / g_norm
        
        # Adam update
        t += 1
        mu = beta1 * mu + (1 - beta1) * grad_log
        nu = beta2 * nu + (1 - beta2) * grad_log ** 2
        mu_hat = mu / (1 - beta1**t)
        nu_hat = nu / (1 - beta2**t)
        
        lr = 0.15 * (1 - 0.5 * progress)
        v -= lr * mu_hat / (np.sqrt(nu_hat) + eps)
        v -= v.mean()
        v = np.clip(v, -30, 30)
        
        # Check improvement
        if t % check_every == 0:
            a_check = np.exp(v - v.max())
            a_check = a_check / a_check.sum()
            sc = _score_fft(a_check)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a_check.copy()
                last_improve = time.monotonic()
        
        # Stuck detection
        if time.monotonic() - last_improve > min(60, 0.2 * seconds):
            sigma = 0.1 + 0.3 * (1 - progress)
            v += rng.normal(0, sigma, n)
            v -= v.mean()
            last_improve = time.monotonic()
    
    return best_a, best_sc


def _coordinate_descent(a, n, iters=5, step_fracs=(0.01, 0.05, 0.1, 0.2, -0.01, -0.05, -0.1, -0.2)):
    """
    Coordinate descent: for each coordinate, try a few perturbations and keep the best.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S <= 0:
        return a
    a = a / S
    
    m = _next_pow2(2 * n - 1)
    best_sc = _score_fft(a)
    
    for it in range(iters):
        improved = False
        for i in range(n):
            a_old = a[i]
            best_local = a_old
            best_local_sc = best_sc
            
            for frac in step_fracs:
                a_try = a.copy()
                a_try[i] = a_old * (1 + frac)
                if a_try[i] < 0:
                    a_try[i] = 0
                sc = _score_fft(a_try)
                if sc < best_local_sc - 1e-10:
                    best_local_sc = sc
                    best_local = a_try[i]
            
            # Also try setting to 0
            a_try = a.copy()
            a_try[i] = 0
            sc = _score_fft(a_try)
            if sc < best_local_sc - 1e-10:
                best_local_sc = sc
                best_local = 0.0
            
            if best_local != a_old:
                a[i] = best_local
                best_sc = best_local_sc
                improved = True
        
        if not improved:
            break
    
    return a


def _interp(a, n):
    """Interpolate sequence to length n."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if a.size == 0 or s <= 0:
        return np.ones(n) / n
    if a.size == n:
        return a / s
    x_old = np.linspace(0, 1, a.size)
    x_new = np.linspace(0, 1, n)
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0)
    ss = out.sum()
    if ss > 0:
        out = out / ss
    else:
        out = np.ones(n) / n
    return out

The ratio to the theoretical minimum (which is `2n/(2n-1) ≈ 1`) is:

        if time.monotonic() - start > max_time:
            break
        t = it / max(1, iters - 1)
        
        # Blend target with flat
        mean_val = float(target.mean())
        flatness = 0.4 + 0.5 * t
        tgt = (1.0 - flatness) * target + flatness * mean_val
        tgt = np.maximum(tgt, 0.01)
        
        # Spectral factorization: find a with |F_a|^2 ~ tgt
        T = np.fft.fft(tgt[:m], m)
        T = np.maximum(T, 0)
        
        # Use current phase
        phase = np.angle(F_a)
        mag = np.sqrt(T)
        # Normalize mag so that the resulting a has sum ~ 1
        F_new = mag * np.exp(1j * phase)
        a_cand = np.fft.ifft(F_new).real[:n]
        a_cand = np.maximum(a_cand, 0)
        S_cand = a_cand.sum()
        if S_cand > 1e-10:
            a_cand = a_cand / S_cand
        else:
            a_cand = a.copy()
        
        # Blend (stronger early, weaker late)
        blend = 0.5 * (1.0 - 0.5 * t)
        a = (1.0 - blend) * a + blend * a_cand
        a = np.maximum(a, 0)
        
        # Update
        F_a = np.fft.fft(a, m)
        b = np.fft.ifft(F_a * F_a).real[:2 * n - 1]
        b = np.maximum(b, 0)
        b_max_new = float(b.max())
        if b_max_new > 0:
            target = b / b_max_new
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _adam_opt(a_init, n, seconds, seed):
    """
    Adam optimization in log-space with soft-max gradient.
    The objective is approximated as (2n/S^2) * sum_k w[k]*b[k]
    where w is the soft-max distribution over b.
    """
    rng = np.random.default_rng(seed)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    # Log-space: a_i = exp(v_i) / sum(exp(v_j))
    v = np.log(a)
    v = v - v.mean()
    
    m = _next_pow2(2 * n - 1)
    m_cross = _next_pow2(3 * n - 2)
    
    # Adam params
    b1, b2, eps = 0.85, 0.999, 1e-8
    mu = np.zeros(n)
    nu = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_sc = _score(a)
    
    start = time.monotonic()
    deadline = start + seconds
    last_improve = start
    
    check_every = max(3, min(30, n // 256 + 3))
    
    while time.monotonic() < deadline:
        progress = min(1.0, (time.monotonic() - start) / seconds)
        
        # Get current a
        v_shifted = v - v.max()
        v_shifted = np.clip(v_shifted, -80, 80)
        a = np.exp(v_shifted)
        a_sum = a.sum()
        if a_sum < 1e-300:
            break
        a = a / a_sum
        
        # Compute convolution
        F = np.fft.fft(a, m)
        b = np.fft.ifft(F * F).real[:2 * n - 1]
        b = np.maximum(b, 0)
        b_max = float(b.max())
        if b_max <= 0 or not np.isfinite(b_max):
            break
        
        # Soft-max weights (annealed temperature)
        # Start smooth, end sharp
        tau_frac = 0.10 * (1.0 - progress) ** 1.5 + 0.025
        # Occasional random perturbation of temperature
        if rng.random() < 0.1:
            tau_frac *= rng.uniform(0.3, 3.0)
        tau = b_max * tau_frac
        tau = max(tau, b_max * 1e-4)
        
        x = (b - b_max) / tau
        x = np.clip(x, -120, 120)
        x = x - x.max()
        w = np.exp(x)
        w_sum = w.sum()
        if w_sum <= 0 or not np.isfinite(w_sum):
            break
        w = w / w_sum
        
        # Gradient: df/da[i] = (2n) * (2*X[i] - 2*Y)  [S=1]
        Y = float(np.dot(w, b))
        
        # X[i] = sum_j w[i+j] * a[j] via cross-correlation FFT
        W = np.fft.fft(w, m_cross)
        A = np.fft.fft(a, m_cross)
        X = np.fft.ifft(W * np.conj(A)).real[:n]
        
        G = 2.0 * n * (2.0 * X - 2.0 * Y)
        
        # Convert to log-space gradient
        # da_i/dv_j = a_i*(delta_ij - a_j)
        # d f / d v_j = sum_i G_i * a_i*(delta_ij - a_j) = G_j*a_j - a_j*sum_i G_i*a_i
        grad_log = G * a - a * float(np.dot(G, a))
        
        # Normalize gradient
        gnorm = float(np.sqrt(np.mean(grad_log ** 2)))
        if gnorm < 1e-14 or not np.isfinite(gnorm):
            v += rng.normal(0, 0.08, n)
            v -= v.mean()
            v = np.clip(v, -40, 40)
            continue
        grad_log = grad_log / gnorm
        
        # Adam update
        t += 1
        mu = b1 * mu + (1 - b1) * grad_log
        nu = b2 * nu + (1 - b2) * grad_log ** 2
        mu_hat = mu / (1.0 - b1 ** t)
        nu_hat = nu / (1.0 - b2 ** t)
        
        # Learning rate with cosine annealing and plateau detection
        lr_base = 0.22 * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr_base = max(lr_base, 0.02)
        
        v -= lr_base * mu_hat / (np.sqrt(nu_hat) + eps)
        v -= v.mean()
        v = np.clip(v, -50, 50)
        
        # Periodic check
        if t % check_every == 0:
            v_s = v - v.max()
            v_s = np.clip(v_s, -80, 80)
            a_check = np.exp(v_s)
            a_check = a_check / a_check.sum()
            sc = _score(a_check)
            if np.isfinite(sc) and sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a_check.copy()
                last_improve = time.monotonic()
        
        # Stuck detection: add perturbation
        stuck_limit = min(40.0, 0.12 * seconds)
        if time.monotonic() - last_improve > stuck_limit:
            sigma = 0.03 + 0.15 * (1.0 - progress)
            v += rng.normal(0, sigma, n)
            v -= v.mean()
            v = np.clip(v, -50, 50)
            last_improve = time.monotonic()
    
    # Final check
    v_s = v - v.max()
    v_s = np.clip(v_s, -80, 80)
    a_final = np.exp(v_s)
    a_final = a_final / a_final.sum()
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final
    
    return best_a, best_sc


def _coord_desc(a_init, n, passes=3, time_limit=60.0):
    """
    Efficient coordinate descent with incremental b updates.
    O(n^2) per pass.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S <= 0:
        return a
    a = a / S  # normalize to sum=1
    
    start = time.monotonic()
    
    # Initial convolution
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0)
    b_max = float(b.max())
    best_sc = float(2 * n * b_max)  # S=1
    
    step_sizes = (0.01, 0.03, 0.1, 0.25, -0.01, -0.03, -0.1, -0.25)
    
    for _pass in range(passes):
        if time.monotonic() - start > time_limit:
            break
        improved = False
        
        for i in range(n):
            a_i = a[i]
            if a_i < 1e-14:
                continue
            
            best_local_sc = best_sc
            best_local_delta = 0.0
            best_local_b = None
            best_local_bmax = b_max
            
            for frac in step_sizes:
                delta = a_i * frac
                new_val = a_i + delta
                if new_val < 0:
                    continue
                
                # Update b: b[k] += 2*delta*a[k-i] for k in [i, i+n-1]
                # Save affected region
                k_start = i
                k_end = min(i + n, 2 * n - 1)
                b_save = b[k_start:k_end].copy()
                
                # Apply
                b[k_start:k_end] += 2.0 * delta * a[:k_end - k_start]
                
                S_new = 1.0 + delta
                b_max_new = float(b.max())
                sc_new = 2.0 * n * b_max_new / (S_new * S_new)
                
                if sc_new < best_local_sc - 1e-11:
                    best_local_sc = sc_new
                    best_local_delta = delta
                    best_local_b = b.copy()
                    best_local_bmax = b_max_new
                
                # Restore
                b[k_start:k_end] = b_save
            
            if best_local_delta != 0.0:
                a[i] += best_local_delta
                b = best_local_b
                b_max = best_local_bmax
                best_sc = best_local_sc
                improved = True
        
        if not improved:
            break
    
    # Renormalize
    S_final = a.sum()
    if S_final > 0:
        a = a / S_final
    return a


def _try_length(a_prev, n, time_budget, seed):
    """Main optimization pipeline for a given length."""
    if time.monotonic() > time_budget[0]:
        return a_prev, _score(a_prev)
    
    start = time.monotonic()
    deadline = time_budget[0]
    remaining = deadline - start
    
    if remaining < 30:
        return a_prev, _score(a_prev)
    
    # Interpolate previous best to this length
    a = _interp(a_prev, n)
    sc = _score(a)
    best_a = a.copy()
    best_sc = sc
    print(f"    n={n}: interpolated score = {sc:.6f}")
    
    # Try GS initialization
    gs_time = min(30, remaining * 0.05)
    if gs_time > 10:
        a_gs = _gs_spectral_init(n, seed, iterations=40)
        sc_gs = _score(a_gs)
        print(f"    n={n}: GS init score = {sc_gs:.6f}")
        if sc_gs < best_sc:
            best_sc = sc_gs
            best_a = a_gs
            a = a_gs
        remaining = deadline - time.monotonic()
    
    # Adaptive spectral refinement
    spec_time = min(90, remaining * 0.15)
    if spec_time > 15:
        a_spec, sc_spec = _adapt_spectral(a, n, iters=50, max_time=spec_time)
        print(f"    n={n}: spectral refine score = {sc_spec:.6f}")
        if sc_spec < best_sc:
            best_sc = sc_spec
            best_a = a_spec
            a = a_spec
        remaining = deadline - time.monotonic()
    
    # Adam optimization (main workhorse)
    adam_time = min(remaining * 0.6, 1200)
    if adam_time > 20:
        # Run multiple Adam instances with different seeds
        n_adam_runs = max(1, int(adam_time / 180))
        n_adam_runs = min(n_adam_runs, 3)
        adam_per_run = adam_time / n_adam_runs
        
        for run in range(n_adam_runs):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(adam_per_run, rem - 30)
            if t_alloc < 15:
                break
            seed_r = seed + run * 7919 + n
            a_adam, sc_adam = _adam_opt(a, n, t_alloc, seed_r)
            print(f"    n={n}: Adam run {run} score = {sc_adam:.6f}")
            if sc_adam < best_sc:
                best_sc = sc_adam
                best_a = a_adam
                a = a_adam
    
    # Coordinate descent refinement
    cd_time = min(120, deadline - time.monotonic() - 10)
    if cd_time > 10:
        a_cd = _coord_desc(best_a, n, passes=min(3, int(cd_time // 20) + 1), time_limit=cd_time)
        sc_cd = _score(a_cd)
        print(f"    n={n}: after coord descent score = {sc_cd:.6f}")
        if sc_cd < best_sc:
            best_sc = sc_cd
            best_a = a_cd
    
    # Final spectral touch-up
    rem = deadline - time.monotonic() - 5
    if rem > 20:
        a_final, sc_final = _adapt_spectral(best_a, n, iters=min(30, int(rem // 2)), max_time=rem)
        if sc_final < best_sc:
            best_sc = sc_final
            best_a = a_final
        print(f"    n={n}: final score = {best_sc:.6f}")
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequences minimizing 2*n*max(a*a)/sum(a)^2.
    
    Pipeline per length:
    1. Interpolate previous best / GS spectral init
    2. Adaptive spectral factorization (flatten auto-convolution)
    3. Adam in log-space with soft-max gradient
    4. Coordinate descent with incremental convolution updates
    5. Final spectral refinement
    
    Time is allocated across multiple sequence lengths.
    """
    global_start = time.monotonic()
    global_deadline = global_start + max(60.0, budget_s - 30.0)
    
    # Get initial sequence
    best_a = None
    best_score = np.inf
    
    try:
        prev = height_sequence_1
        if isinstance(prev, list):
            arr = np.asarray(prev, dtype=np.float64)
        else:
            arr = np.asarray(prev, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.maximum(arr, 0)
            if arr.sum() > 0:
                best_a = arr / arr.sum()
                best_score = _score(best_a)
                print(f"Loaded height_sequence_1: n={arr.size}, score={best_score:.6f}")
    except Exception as e:
        print(f"Could not load previous: {e}")
    
    if best_a is None:
        n0 = 1024
        best_a = _gs_spectral_init(n0, seed, iterations=50)
        best_score = _score(best_a)
        print(f"Using GS initialization: n={n0}, score={best_score:.6f}")
    
    # Determine which lengths to try and time allocation
    n0 = best_a.size
    lengths = []
    # Always try current length
    lengths.append((n0, 0.30))
    # Try larger (better asymptotic)
    if n0 * 2 <= 16384:
        lengths.append((n0 * 2, 0.35))
    if n0 * 4 <= 16384:
        lengths.append((n0 * 4, 0.20))
    # Try smaller (faster, might find different basin)
    if n0 >= 1024:
        lengths.append((n0 // 2, 0.15))
    
    total_frac = sum(f for _, f in lengths)
    # Normalize
    for i, (n, f) in enumerate(lengths):
        lengths[i] = (n, f / total_frac)
    
    # Sort by priority (current length first, then larger)
    lengths.sort(key=lambda x: -x[1])
    
    total_opt_time = global_deadline - global_start - 30
    
    print(f"\nPlan: optimizing at lengths {[n for n, f in lengths]}")
    print(f"Total time budget: {total_opt_time:.0f}s")
    
    for n, frac in lengths:
        n = int(max(128, n))
        alloc = frac * total_opt_time
        alloc = max(30, min(alloc, total_opt_time * 0.5))
        
        rem = global_deadline - time.monotonic() - 20
        if rem < 30:
            break
        
        time_alloc = min(alloc, rem)
        print(f"\n{'='*60}")
        print(f"Stage: n={n}, time={time_alloc:.0f}s, current best={best_score:.6f}")
        print(f"{'='*60}")
        
        stage_deadline = time.monotonic() + time_alloc
        a_new, sc_new = _try_length(best_a, n, (stage_deadline,), seed + n * 13)
        
        # Check if improvement at this length, also try interpolating back
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new
            print(f"  NEW BEST: {best_score:.6f} at n={n}")
        else:
            print(f"  No improvement at n={n} (got {sc_new:.6f}, best is {best_score:.6f})")
        
        # If we got a good result at a different length, also try to use it
        # for the other lengths via interpolation
    
    # Final: try a last Adam run at the best length with remaining time
    rem = global_deadline - time.monotonic() - 30
    if rem > 30:
        print(f"\nFinal refinement: {rem:.0f}s remaining, n={best_a.size}")
        a_final, sc_final = _adam_opt(best_a, best_a.size, rem, seed + 99999)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final
            print(f"  Final improvement: {best_score:.6f}")
    
    # Validate with exact evaluation if feasible
    print(f"\n{'='*60}")
    print(f"Final result: score={best_score:.6f}, n={len(best_a)}")
    
    if len(best_a) <= 8192:
        try:
            exact = evaluate_sequence([float(x) for x in best_a])
            print(f"Exact evaluation: {exact:.6f}")
            if not np.isfinite(exact):
                print("Exact eval returned inf, falling back to n=512")
                best_a = np.ones(512) / 512
        except Exception as e:
            print(f"Validation error (using FFT score): {e}")
    
    elapsed = time.monotonic() - global_start
    print(f"Total time: {elapsed:.1f}s")
    
    return [float(x) for x in best_a]
