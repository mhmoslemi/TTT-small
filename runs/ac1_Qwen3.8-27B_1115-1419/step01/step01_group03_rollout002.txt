Minimize 2n*max(a*a)/(sum(a))^2 using smoothed minimax (log-sum-exp) 
with adaptive temperature, projected gradient with Nesterov momentum,
FFT-based convolution, and multiple diverse restarts.
"""

import numpy as np
import time

def fft_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = fft_conv(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_max_obj(a, beta, c):
    """Compute log-sum-exp(beta * c) and its gradient w.r.t. a."""
    # c = conv(a, a) already computed
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    lse = (bc_max + np.log(exp_bc.sum())) / beta
    # Weights
    p = exp_bc / exp_bc.sum()
    return lse, p

def grad_conv_k(k, a):
    """Gradient of the k-th convolution coefficient w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        idx = k - np.arange(lo, hi, dtype=np.int64)
        grad[lo:hi] = a[idx]
    if k % 2 == 0:
        j = k // 2
        if 0 <= j < n:
            grad[j] += a[j]
    return grad

    """
    Run smoothed minimax optimization with adaptive temperature.
    """
    n = len(a0)
    a = a0.copy()
    a_prev = a.copy()
    t = 1.0
    step = 0.1
    beta = beta_start
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + duration
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:

        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()
        
        # Compute convolution
        c = fft_conv(y)
        
        # Smoothed max and weights
        bc = beta * c
        bc_max = bc.max()
        # Clip for numerical stability
        bc_clipped = np.clip(bc - bc_max, -100, 100)
        exp_bc = np.exp(bc_clipped)
        exp_sum = exp_bc.sum()
        if exp_sum < 1e-300:
            p = np.full(c.shape, 1.0 / c.size)
        else:
            p = exp_bc / exp_sum
        
        # Gradient: weighted sum of individual gradients
        # Use top-m weights for efficiency
        m = min(50, c.size)
        top_idx = np.argpartition(p, -m)[-m:]
        g = np.zeros(n)
        for mk in top_idx:
            g += p[mk] * grad_conv_k(mk, y)
        
        # Gradient step
        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation and adaptation
        if it % 50 == 0:

            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

def project_simplex(v):
    """Project v onto {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
