**The gradient of the max-convolution targets the "mirror" of f around each peak.** To reduce the peak at position k*, we need to reduce the overlap between f and its shift by k*. This means f and f shifted by k* should be "anti-correlated" as much as possible (while maintaining non-negativity).

This is similar to the problem of designing "low-autocorrelation" sequences (like Zadoff-Chu, Barker, etc.), but with the constraint of non-negativity.

One approach from the signal processing literature: **iterative projection**. 
1. Project f onto the set {non-negative, sum=1} (simplex projection).
2. In the frequency domain, try to make |F(f)|^2 as flat as possible (this corresponds to flat autocorrelation).
3. Alternate between these two projections.

But step 2 is hard because we need the PEAK of the autocorrelation to be low, not the entire autocorrelation to be flat.

Actually, for a "flat" autocorrelation (all values equal), the peak would be 1/(2n-1) * sum(f)^2 = 1/(2n-1) for sum(f)=1. Then objective = 2n * 1/(2n-1) ≈ 1. So flat autocorrelation gives objective ≈ 1, which is the theoretical minimum!

But can a non-negative sequence have flat autocorrelation? The autocorrelation of a non-negative sequence has a specific structure: it's a positive definite sequence (by Bochner's theorem, it's the Fourier transform of a non-negative spectral density). The "flat" autocorrelation (delta function in frequency) corresponds to a rectangular spectrum, which means the sequence is a sinc-like function, which changes sign. So we can't have exactly flat autocorrelation with non-negative f.

But we can get close! The question is how close, and what the optimal shape looks like.

def next_fast_len(m):
    """Smallest power of 2 >= m."""
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonneg(v):
    """Normalize to sum 1, nonnegative."""
    arr = np.asarray(v, dtype=np.float64).ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones(max(1, arr.size), dtype=np.float64)
        s = float(arr.sum())
    return arr / s


def fft_self_conv(f, N):
    """Linear self-convolution via FFT."""
    A = np.fft.rfft(f, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2 * len(f) - 1]
    return np.maximum(c, 0.0)


def fft_score(f, N):
    """Objective value for normalized f: 2*n*max(conv(f,f))."""
    n = len(f)
    c = fft_self_conv(f, N)
    mx = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mx):
        return float('inf')
    return 2.0 * n * mx


def compute_topk_gradient(f, c, k, beta, N):
    """
    Compute gradient of 2n * sum_k p_k * c_k where p_k is top-k weighted.
    Returns gradient w.r.t. f (length n).
    """
    n = len(f)
    # Build importance weights
    c_flat = c
    if k >= c_flat.size:
        cmax = float(np.max(c_flat))
        x = beta * (c_flat - cmax)
        x = np.clip(x, -745.0, 0.0)
        e = np.exp(x)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            p = np.zeros(c_flat.size)
            p[int(np.argmax(c_flat))] = 1.0
        else:
            p = e / s
    else:
        # Top-k: set others to 0, exponential within top-k
        idx = np.argpartition(c_flat, -k)[-k:]
        p = np.zeros(c_flat.size)
        top_vals = c_flat[idx]
        tmax = float(np.max(top_vals))
        x_top = beta * (top_vals - tmax)
        x_top = np.clip(x_top, -745.0, 0.0)
        e_top = np.exp(x_top)
        s_top = float(e_top.sum())
        if s_top > 0.0 and np.isfinite(s_top):
            p[idx] = e_top / s_top
        else:
            p[idx[int(np.argmax(top_vals))]] = 1.0

    # Compute gradient: g_i = 4n * sum_k p_k * f_{k-i} (for valid indices)
    # Using FFT: cross[m] = sum_k p_k * f_{(k+m) mod N}
    # We need R(i) = sum_k p_k * f_{k-i} = cross[(N-i)%N]
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    cross = np.fft.irfft(np.conj(P) * F, N)

    # Index: for i=0: cross[0], for i=1..n-1: cross[N-i]
    idx_arr = np.zeros(n, dtype=np.int64)
    idx_arr[0] = 0
    if n > 1:
        idx_arr[1:] = N - np.arange(1, n)
    grad = 4.0 * n * cross[idx_arr]
    return np.where(np.isfinite(grad), grad, 0.0), p


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 0.0:
        return np.ones(len(v)) * target_sum / len(v)
    return v * (target_sum / s)


def multiplicative_update(f, grad, lr, N):
    """
    Multiplicative (exponential) update in log-space:
    f_new_i = f_i * exp(-lr * (grad_i - avg))
    where avg keeps sum(f) = 1.
    """
    n = len(f)
    # Center gradient (tangent to simplex)
    avg = float(np.dot(f, grad))
    centered = grad - avg  # This ensures sum(f_i * centered_i) = 0
    # Multiplicative update
    step = np.clip(lr * centered, -10.0, 10.0)
    f_new = f * np.exp(step)
    f_new = np.maximum(f_new, 1e-300)
    f_new = f_new / float(f_new.sum())
    return f_new

def exact_score(a):
    """Exact O(n^2) evaluation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float('inf')
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return float('inf')
    return float(2.0 * n * mx / (s * s))

                f = simplex_project(np.exp(logf))
            elif strategy == 1:
                # Mirror-reduction: specifically target the peak
                peak_idx = int(np.argmax(c))
                # The peak at position k means f overlapped with shift k
                # Reduce f_i where f_{k-i} is large
                mirror = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        mirror[i] = f[j]
                # Subtract mirror from f (reduces overlap)
                f = simplex_project(f - 0.1 * mirror + 0.05 * f)
            elif strategy == 2:
                # Frequency domain flattening: try to make spectrum more uniform
                F_f = np.fft.rfft(f, N)
                # Add small random phase in frequency domain
                phase_rand = rng.standard_normal(F_f.size) * 0.1
                F_f = F_f * np.exp(1j * phase_rand)
                f_try = np.fft.irfft(F_f, N)[:n]
                f_try = np.maximum(f_try, 0.0)
                s_try = float(f_try.sum())
                if s_try > 0.01:
                    f = f_try / s_try
                else:
                    f = best_f.copy()
            elif strategy == 3:
                # Block swap: randomly swap blocks of the sequence
                block_size = max(1, n // 16)
                b1 = int(rng.integers(0, n // block_size - 1))
                b2 = int(rng.integers(0, n // block_size - 1))
                if b1 != b2:
                    s1, e1 = b1 * block_size, (b1 + 1) * block_size
                    s2, e2 = b2 * block_size, (b2 + 1) * block_size
                    f[s1:e1], f[s2:e2] = f[s2:e2].copy(), f[s1:e1].copy()
            else:
                # Return to best
                f = best_f.copy()

            c = fft_self_conv(f, N)
            stall = 0
            lr = 0.3

        # Slowly increase beta and decrease k for sharpening
        if iters % 100 == 0:
            beta = min(500.0, beta * 1.05)

        # Exact evaluation (throttled)
        if time.monotonic() - last_exact > 30.0 and n <= 20000:
            t0 = time.monotonic()
            sc_exact = exact_score(best_f)
            exact_calls += 1
            last_exact = time.monotonic()
            if np.isfinite(sc_exact):
                exact_best = sc_exact
                exact_best_array = best_f.copy()
                if iters % 500 == 1:
                    print(f"    Exact score: {sc_exact:.6f} (iter {iters}, calls {exact_calls}, "
                          f"time {time.monotonic()-start_time:.0f}s)")

        # Print progress
        if time.monotonic() - last_print > print_every:
            print(f"    FFT score: {best_score:.6f}, iter {iters}, stall {stall}, "
                  f"lr {lr:.3f}, k {k_top}, beta {beta:.1f}")
            last_print = time.monotonic()

    return best_f, best_score, exact_best_array, exact_best


def run_cross_opt(f, N, deadline_part, N_cross):
    """
    Alternating optimization: fix f, optimize b to minimize max(conv(f,b)).
    Then update f <- (1-t)*f + t*b.
    """
    n = len(f)
    c = fft_self_conv(f, N)
    best_f = f.copy()
    best_score = fft_score(f, N)

    F = np.fft.rfft(f, N)
    b = f.copy()
    lr_b = 0.2
    beta_b = 20.0

    fr = 0
    while time.monotonic() < deadline_part and fr < 500:
        fr += 1
        # Compute conv(f, b) and gradient w.r.t. b
        c_fb = fft_self_conv(b, N_cross)  # placeholder
        # Actually need cross conv f*b:
        Ff = np.fft.rfft(f, N)
        Fb = np.fft.rfft(b, N)
        c_fb = np.fft.irfft(Ff * Fb, N)
        c_fb = c_fb[:2 * n - 1]
        c_fb = np.maximum(c_fb, 0.0)

        # Gradient w.r.t. b: d(max_c_fb)/db_i
        # Use top-1 (exact peak)
        peak = float(np.max(c_fb))
        if peak <= 0:
            break
        # Grad for b: same structure, dc/d b_i = f_{k-i} (single factor, not 2)
        # For cross conv: c_k = sum_j f_j b_{k-j}, dc/db_i = f_{k-i}
        pk = int(np.argmax(c_fb))
        # grad_b_i = f_{pk - i} for 0 <= pk-i < n
        grad_b = np.zeros(n)
        s = max(0, pk - n + 1)
        e = min(n, pk + 1)
        for i in range(s, e):
            j = pk - i
            if 0 <= j < n:
                grad_b[i] = f[j]

        # Center and update b
        avg = float(np.dot(b, grad_b))
        centered = grad_b - avg
        step = np.clip(lr_b * centered, -5.0, 5.0)
        b_new = b * np.exp(step)
        b_new = np.maximum(b_new, 1e-300)
        b_new = b_new / float(b_new.sum())

        # Try convex combos
        improved = False
        for t in (0.5, 0.8, 1.0):
            cand = (1.0 - t) * f + t * b_new
            cand = cand / float(cand.sum())
            sc = fft_score(cand, N)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = cand.copy()
                f = cand.copy()
                F = np.fft.rfft(f, N)
                b = f.copy()
                improved = True
                break

        if not improved:
            b = b_new
            lr_b *= 0.95
            if lr_b < 0.02:
                break

    return best_f, best_score


def finalize_sequence(a):
    """Scale to sum sqrt(2n), ensure valid output."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0 or not np.all(np.isfinite(a)):
        a = np.ones(64, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    scale = float(math.sqrt(2.0 * n))
    if s > 0.0 and np.isfinite(s):
        a = (a * scale) / s
    else:
        a = np.ones(n, dtype=np.float64) * (scale / n)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = a * (0.01 / max(s2, 1e-300))
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global start_time
    """
    Entry point: runs hybrid optimizer and returns best sequence.
    Strategy:
      1. Load parent, try multiple initializations.
      2. Main loop: top-k gradient + multiplicative update + backtrack.
      3. Cross-convolution alternating optimization.
      4. Fine-tuning with exact peak targeting.
      5. Return best scaled sequence.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    budget = min(max(0.0, budget), 2395.0)

    start_time = time.monotonic()
    deadline = start_time + budget - 10.0  # Leave 10s for finalization

    # Load initial
    parent = load_initial_sequence()
    if parent is None:
        parent = np.ones(4096, dtype=np.float64) / 4096.0

    n_orig = len(parent)
    # Cap length
    MAX_N = 12000
    if n_orig > MAX_N:
        # Downsample by averaging blocks
        target = 8192
        blocks = n_orig // target
        if blocks > 1:
            use = target * blocks
            parent = np.mean(parent[:use].reshape(target, blocks), axis=1)
        else:
            parent = parent[:target]
    n = len(parent)
    if n < 16:
        n = 4096
        parent = np.ones(n) / n
    parent = normalize_nonneg(parent)

    N = next_fast_len(3 * n - 2)
    print(f"Sequence length: {n}, FFT size: {N}")

    # Try to get exact score of parent
    parent_exact = exact_score(parent)
    print(f"Parent exact score: {parent_exact:.6f}")

    # Multiple initializations
    inits = [
        ("parent", parent.copy()),
    ]
    seeds_to_try = [1, 7, 13, 42, 99]
    for s in seeds_to_try:
        inits.append((f"rand_{s}", make_random_init(n, s, sig=0.5)))
    inits.append(("hann", make_hann_init(n)))
    inits.append(("blackman", make_blackman_init(n)))
    inits.append(("gauss", make_gaussian_init(n, 0.35)))
    inits.append(("two_lobe", make_two_lobe_init(n, 0.55)))

    # Evaluate all inits
    best_init = None
    best_init_score = float('inf')
    for name, init in inits:
        init = normalize_nonneg(init)
        sc = fft_score(init, N)
        if np.isfinite(sc) and sc < best_init_score:
            best_init_score = sc
            best_init = init.copy()
        print(f"  Init '{name}': {sc:.6f}")

    print(f"Best init: score {best_init_score:.6f}")

    # Phase 1: Main gradient optimization (60% of budget)
    phase1_deadline = start_time + budget * 0.55
    print(f"\nPhase 1: Gradient optimization (until {phase1_deadline - start_time:.0f}s)")
    f, fft_best, exact_arr, exact_best_score = run_gradient_opt(
        best_init, N, None, phase1_deadline, seed=seed, print_every=15.0
    )
    print(f"Phase 1 done: FFT={fft_best:.6f}, exact={exact_best_score:.6f}")

    # Phase 2: Cross-convolution optimization (20% of budget)
    phase2_deadline = start_time + budget * 0.75
    print(f"\nPhase 2: Cross-conv optimization (until {phase2_deadline - start_time:.0f}s)")
    f2, sc2 = run_cross_opt(f, N, phase2_deadline, N)
    if sc2 < fft_best:
        f = f2
        fft_best = sc2
        print(f"  Cross-opt improved: {sc2:.6f}")
    else:
        print(f"  Cross-opt: no improvement ({sc2:.6f} vs {fft_best:.6f})")

    # Phase 3: Fine-tuning with sharper top-k (remaining budget)
    phase3_deadline = deadline
    print(f"\nPhase 3: Fine-tuning (until {phase3_deadline - start_time:.0f}s)")
    f3, fft_best3, exact_arr3, exact_best3 = run_gradient_opt(
        f, N, None, phase3_deadline, seed=seed + 1000, print_every=15.0
    )

    # Determine best
    candidates = [
        (fft_best, f),
        (fft_best3, f3),
        (exact_best_score, exact_arr) if exact_best_score < float('inf') else None,
        (exact_best3, exact_arr3) if exact_best3 < float('inf') else None,
    ]
    candidates = [c for c in candidates if c is not None]

    # Final exact validation
    final_best = None
    final_best_score = float('inf')
    for sc_fft, arr in candidates:
        arr = normalize_nonneg(arr)
        n_chk = len(arr)
        if n_chk > 20000:
            continue
        sc_exact = exact_score(arr)
        if np.isfinite(sc_exact) and sc_exact < final_best_score:
            final_best_score = sc_exact
            final_best = arr.copy()

    if final_best is None:
        final_best = f
        final_best_score = fft_best

    print(f"\nFinal: FFT best={fft_best3:.6f}, exact best={final_best_score:.6f}")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    # Try one more round of gradient from the exact best if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60.0 and final_best is not None:
        print(f"\nBonus phase: {remaining:.0f}s remaining, fine-tuning exact best...")
        f_bonus, _, _, _ = run_gradient_opt(
            final_best, N, None, deadline, seed=seed + 2000, print_every=10.0
        )
        sc_bonus = exact_score(f_bonus)
        if np.isfinite(sc_bonus) and sc_bonus < final_best_score:
            final_best = f_bonus
            final_best_score = sc_bonus
        # Also check the FFT best from the bonus run
        sc_bonus2 = exact_score(f_bonus)
        if np.isfinite(sc_bonus2) and sc_bonus2 < final_best_score:
            final_best = f_bonus
            final_best_score = sc_bonus2
        print(f"After bonus: exact={final_best_score:.6f}")

    return finalize_sequence(final_best)
