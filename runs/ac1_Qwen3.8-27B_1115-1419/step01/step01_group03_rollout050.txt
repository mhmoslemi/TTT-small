"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient
descent with adaptive active set size, multiple diverse initializations,
backtracking step size adaptation, and periodic restarts.
Key improvements: adaptive K (grows when stuck), multiple n values,
structured initializations, and a dedicated polishing phase.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute full convolution a*a using FFT for speed."""
    n = len(a)
    size = 2 * n - 1
    f = np.fft.rfft(a, n=2 * size)
    return np.fft.irfft(f * f, n=2 * size)[:size]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = compute_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, active_idx):
    """Compute average subgradient over the given active index set."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(active_idx))
    return grad


def run_optimization(a0, duration, rng, k_start=3, k_max=200, step_init=0.08):
    """
    Nesterov-accelerated subgradient descent with adaptive active set.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    a_prev = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    k_active = k_start
    step = step_init

    while time.monotonic() < deadline:
        # Nesterov momentum
        t_old = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        mu = (t_old - 1.0) / t_mom
        y = a + mu * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy
        else:
            y = a.copy()

        # Compute convolution
        conv = compute_conv(y)

        # Get active set (top-k peaks)
        if k_active >= len(conv):
            active = np.arange(len(conv))
        else:
            active = np.argpartition(conv, -k_active)[-k_active:]

        # Compute subgradient
        grad = compute_subgradient(y, active)

        # Gradient step + project
        a_new = project_simplex(y - step * grad)

        # Periodic score evaluation
        if it % 5 == 0:

            if sc < best_score:

                stuck = 0
            else:

            elif stuck >= 100:

            elif stuck >= 250:
                # Restart from best with perturbation
                a = best_a + rng.normal(0.0, 0.004 / np.sqrt(n), n)
                np.maximum(a, 0.0, out=a)
                a = project_simplex(a)
