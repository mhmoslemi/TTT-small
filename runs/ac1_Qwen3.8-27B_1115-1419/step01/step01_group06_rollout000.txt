"""
Search for nonnegative coefficient sequences minimizing
    2 * n * max(np.convolve(a, a)) / (sum(a)^2)
using subgradient descent with soft-max smoothing, structured perturbations,
and adaptive annealing. Works in log-space (softmax parameterization) to
maintain nonnegativity and unit-sum constraint.
"""

import numpy as np
import time

def next_fast_len(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def softmax(w):
    w = np.asarray(w, dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= np.max(w)
    e = np.exp(np.clip(w, -745, 709))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(w.size) / w.size
    return e / s

def fft_convolve(a, b, L):
    """Linear convolution via FFT."""
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)

def compute_score_fft(a, L):
    """Compute 2*n*max(a*a/(sum a)^2) using FFT. a should be sum=1."""
    n = len(a)
    c = fft_convolve(a, a, L)
    c = c[:2*n-1]
    c = np.maximum(c, 0)  # numerical safety
    mx = c.max()
    return 2.0 * n * mx, c, mx

def compute_gradient(a, w, L):
    """
    Compute gradient of sum_k w_k * (a*a)_k w.r.t. a.
    w has length 2n-1 (soft-max weights over convolution indices).
    Returns grad of length n.
    """
    n = len(a)
    w_rev = w[::-1]  # reverse
    conv = fft_convolve(a, w_rev, L)
    # grad[j] = conv[2n-2-j] for j = 0..n-1
    grad = conv[n-1:2*n-1][::-1]
    grad = 2.0 * grad  # factor of 2 from the quadratic form
    return grad

def exact_score(a):
    """Exact score using np.convolve."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= 0:
        return float('inf')
    n = len(a)
    c = np.convolve(a, a)
    mx = c.max()
    return 2.0 * n * mx / (s * s)

def load_sota():

    if obj is None:
        return None
    try:
        arr = np.array(obj, dtype=np.float64)
    except:
        return None
    if arr.size < 16:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s <= 0:
        return None
    return arr / s

def scramble(a, rng, strength=0.1):
    """Apply random log-space perturbation."""
    n = len(a)
    a = np.maximum(a, 1e-30)
    log_a = np.log(a)
    log_a += rng.normal(0, strength, size=n)
    return softmax(log_a)

def structured_perturb(a, rng):
    """Apply a sinusoidal perturbation in log space."""
    n = len(a)
    a = np.maximum(a, 1e-30)
    log_a = np.log(a)
    # Try a few random sinusoidal patterns

    alpha = rng.uniform(0.05, 0.3)
    log_a += alpha * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    return softmax(log_a)

        sota = sota / sota.sum()
        n = target

    score, c, mx = compute_score_fft(a, L)

    best_score = score
