"""
Search for non-negative step-function heights minimizing

    2 * n * max(convolve(a, a)) / sum(a)^2.

This implementation combines improved constructive initializers, spectral
power-factor reconstruction, and log-space multiplicative (natural-gradient)
updates with adaptively chosen active-convolution weights.  Symmetric and
asymmetric local searches are run, and the best sequence over several lengths
is returned.
"""

import numpy as np
import time


def _pair_score(pair):
    return float(pair[0])


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _normalize(a, sym=False):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    else:
        arr = np.maximum(arr, 0.0)

    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-300:
        arr = np.ones(arr.size, dtype=np.float64)
        s = float(arr.size)
    else:
        arr = arr / s

    if sym:
        arr = 0.5 * (arr + arr[::-1])
        s = float(np.sum(arr))
        if not np.isfinite(s) or s <= 1e-300:
            arr = np.ones_like(arr)
            s = float(arr.size)
        arr = arr / s

    return arr


def _symmetrize(a):
    return _normalize(a, True)


def _conv_self(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = int(arr.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(arr, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    return b


def _score_from_conv(a, b):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = int(arr.size)
    if n == 0:
        return np.inf
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf
    return float(2.0 * n * bmax / (s * s))


def _score(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.inf
    return _score_from_conv(arr, _conv_self(arr))


def _a_to_log(a):
    arr = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 1e-300)
    v = np.log(arr)
    mx = float(np.max(v))
    if not np.isfinite(mx):
        return np.zeros(arr.size, dtype=np.float64)
    return v - mx


def _from_log(v, sym=False):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return _normalize(np.ones(v.size, dtype=np.float64), sym)

    mx = float(np.max(v))
    if not np.isfinite(mx):
        return _normalize(np.ones(v.size, dtype=np.float64), sym)

    w = np.exp(v - mx)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 1e-300:
        return _normalize(np.ones(v.size, dtype=np.float64), sym)

    w = w / s
    if sym:
        w = 0.5 * (w + w[::-1])
        s2 = float(np.sum(w))
        if not np.isfinite(s2) or s2 <= 1e-300:
            return _normalize(np.ones(v.size, dtype=np.float64), True)
        w = w / s2

    return w


def _interp(a, n):
    a = _normalize(a)
    n = int(max(1, n))
    if a.size == n:
        return a
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    return _normalize(out)


def _get_previous():
    obj = globals().get("height_sequence_1", None)
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None
    return _normalize(arr)


def _lambda_dense(c, p):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    cmax = float(np.max(c)) if L else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.ones(L, dtype=np.float64) / max(1, L)

    q = np.maximum(c / cmax, 0.0)
    p = float(p)
    if p <= 1e-12:
        lam = np.ones(L, dtype=np.float64)
    else:
        lam = np.power(q, p)
    lam = 0.5 * (lam + lam[::-1])
    s = float(np.sum(lam))
    if not np.isfinite(s) or s <= 1e-300:
        lam = np.ones(L, dtype=np.float64) / max(1, L)
    else:
        lam = lam / s
        floor = 1e-8
        lam = (1.0 - floor) * lam + floor / max(1, L)
    return lam


def _top_indices(c, K):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    K = int(max(0, min(K, L)))
    if K == 0:
        return np.empty(0, dtype=np.int64)
    if K == L:
        return np.arange(L, dtype=np.int64)
    return np.argpartition(c, L - K)[L - K:].astype(np.int64)


def _topk_gradient(a, idx):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    m = idx.size
    if m == 0:
        return g
    w = 1.0 / float(m)
    for pos in range(m):
        k = int(idx[pos])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        start = k - lo
        stop = k - hi - 1
        if stop >= 0:
            g[lo:hi + 1] += w * a[start:stop:-1]
        else:
            g[lo:hi + 1] += w * a[start::-1]
    return g


def _hankel_product(a, lam):
    a = np.asarray(a, dtype=np.float64).ravel()
    lam = np.asarray(lam, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    if lam.size < L:
        tmp = np.zeros(L, dtype=np.float64)
        tmp[:lam.size] = lam
        lam = tmp
    else:
        lam = lam[:L].copy()

    m = _next_pow2(n + L - 1)
    Fa = np.fft.fft(a, m)
    Fl = np.fft.fft(lam[::-1], m)
    conv = np.fft.ifft(Fl * Fa).real

    if n == 1:
        g = conv[0:1]
    else:
        g = conv[2 * n - 2:n - 1:-1]

    g = np.maximum(g, 0.0)
    return g


def _make_grad(a, c, policy, sym):
    if policy is None:
        policy = {"mode": "dense", "p": 1.0}
    mode = str(policy.get("mode", "dense"))

    if mode == "dense":
        p = float(policy.get("p", 1.0))
        lam = _lambda_dense(c, p)
        g = _hankel_product(a, lam)
    else:
        K = int(policy.get("K", 8))
        idx = _top_indices(c, K)
        g = _topk_gradient(a, idx)

    if sym:
        g = 0.5 * (g + g[::-1])
    return np.maximum(g, 0.0)


def _random_policy(rng):
    r = float(rng.random())
    if r < 0.52:
        logp = float(rng.uniform(-0.2, 2.0))
        p = float(10.0 ** logp)
        alpha = 0.14 + 0.07 * max(0.0, logp)
        return {"mode": "dense", "p": p, "alpha": alpha}

    if r < 0.92:
        e = float(rng.uniform(1.0, 6.0))
        K = int(2.0 ** e)
        K = max(2, min(128, K))
        alpha = max(0.12, 0.42 - 0.03 * e)
        return {"mode": "topk", "K": K, "alpha": alpha}

    e = float(rng.uniform(0.0, 4.0))
    K = int(max(1.0, 2.0 ** e))
    K = max(1, min(32, K))
    alpha = max(0.18, 0.50 - 0.05 * e)
    return {"mode": "topk", "K": K, "alpha": alpha}


def _param_candidates(n, prev, rng):
    n = int(max(4, n))
    out = []

    x = np.linspace(0.0, 1.0, n)
    i = np.arange(1, n + 1, dtype=np.float64)
    j = n + 1 - i
    center = np.abs(x - 0.5) * 2.0

    if prev is not None:
        p = _normalize(prev)
        out.append(_interp(p, n))
        out.append(_symmetrize(p))

    out.append(np.ones(n, dtype=np.float64) / float(n))

    for p in (0.05, 0.15, 0.25, 0.35, 0.50, 0.65, 0.80, 0.95, 1.10, 1.35):
        a = np.power(i, -p) * np.power(j, -p)
        out.append(_normalize(a))

    for A in (0.20, 0.45, 0.80, 1.5, 3.0, 6.0, 12.0):
        a = 1.0 + A * (center * center)
        out.append(_normalize(a))
        a = 1.0 + A * center + 0.25 * A * (center ** 4)
        out.append(_normalize(a))

    for g in (1.0, 3.0, 6.0, 10.0, 18.0, 32.0, 60.0):
        a = np.exp(g * center)
        out.append(_normalize(a))

    for p in (0.25, 0.45, 0.60):
        base = np.power(i, -p) * np.power(j, -p)
        for A in (0.0, 0.5, 1.5, 4.0):
            a = base * (1.0 + A * (1.0 - center * center))
            out.append(_normalize(a))

    for k in (1, 2, 3, 4, 6):
        for A in (0.2, 0.5, 0.8):
            a = 1.0 + A * np.cos(2.0 * np.pi * k * x)
            out.append(_normalize(np.maximum(a, 0.0)))

    for _ in range(4):
        a = 1.0
        for k in range(1, 9):
            amp = 0.28 / (float(k) ** 0.8)
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp * np.cos(2.0 * np.pi * k * x + ph)
        out.append(_normalize(np.maximum(a, 0.0)))

    for r in (0.25, 0.35, 0.45):
        a = (np.exp(-((x - r) * (x - r)) / 0.012) +
             np.exp(-((x - (1.0 - r)) * (x - (1.0 - r))) / 0.012) + 0.01)
        out.append(_normalize(a))

    return out


def _select_candidates(cands, max_count=6, tol=0.05, limit=32):
    scored = []
    for a in cands:
        arr = _normalize(a)
        s = _score(arr)
        if np.isfinite(s):
            scored.append((float(s), arr))

    if not scored:
        return []

    scored.sort(key=_pair_score)
    top = scored
    if len(top) > limit:
        top = top[:limit]

    selected = []
    selected_arrs = []
    tol2 = tol * tol

    for s, a in top:
        dup = False
        for b in selected_arrs:
            d = a - b
            d2 = float(np.dot(d, d))
            if d2 < tol2:
                dup = True
                break
        if not dup:
            selected.append((float(s), a))
            selected_arrs.append(a)
            if len(selected) >= max_count:
                break

    return selected


def _spectral_target_init(n, target, a0, iterations, deadline, seed):
    n = int(max(4, n))
    N = _next_pow2(2 * n - 1)
    L = 2 * n - 1

    t = np.asarray(target, dtype=np.float64).ravel()
    if t.size < L:
        tmp = np.zeros(L, dtype=np.float64)
        tmp[:t.size] = t
        t = tmp
    else:
        t = t[:L].copy()
    t = np.maximum(t, 0.0)
    ts = float(np.sum(t))
    if not np.isfinite(ts) or ts <= 1e-300:
        t = np.ones(L, dtype=np.float64) / float(L)
    else:
        t = t / ts

    Tc = np.fft.fft(t, N)

    if a0 is None:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = _normalize(a0)

    best_a = a.copy()
    best_s = _score(a)
    rng = np.random.default_rng(int(seed))
    rng._bit_generator.state  # no-op, keeps rng initialized

    for it in range(int(max(0, iterations))):
        if deadline is not None and time.monotonic() >= deadline:
            break

        A = np.fft.fft(a, N)
        absA = np.abs(A)
        amp = np.sqrt(np.abs(Tc) + 1e-12)
        phase = np.ones(N, dtype=np.complex128)
        mask = absA > 1e-14
        phase[mask] = A[mask] / absA[mask]
        Anew = 0.5 * A + 0.5 * amp * phase

        cand = np.fft.ifft(Anew).real[:n]
        cand = np.maximum(cand, 0.0)
        s = float(np.sum(cand))
        if np.isfinite(s) and s > 1e-8:
            cand = cand / s
            a = 0.62 * a + 0.38 * cand
            a = np.maximum(a, 1e-20)
            s2 = float(np.sum(a))
            if np.isfinite(s2) and s2 > 1e-300:
                a = a / s2

        if it % 25 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_s:
                best_s = float(sc)
                best_a = a.copy()

    sc = _score(a)
    if np.isfinite(sc) and sc < best_s:
        best_s = float(sc)
        best_a = a.copy()

    return _normalize(best_a), best_s


def _spectral_candidates(n, base, budget, rng):
    out = []
    n = int(max(4, n))
    if n > 4096:
        return out
    budget = float(max(0.0, budget))
    if budget < 12.0:
        return out

    deadline = time.monotonic() + min(60.0, budget)
    base = _normalize(base)
    c = _conv_self(base)
    cm = float(np.max(c))
    if np.isfinite(cm) and cm > 0.0:
        c = c / cm
    else:
        c = np.ones(2 * n - 1, dtype=np.float64) / float(2 * n - 1)

    targets = []
    targets.append(c)
    targets.append(np.maximum(c, 0.18))
    targets.append(0.70 * c + 0.30)

    L = 2 * n - 1
    u = np.linspace(0.0, 2.0, L)
    targets.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (u - 1.0)), 0.02))
    targets.append(np.maximum(1.0 - np.abs(u - 1.0), 0.02))

    iters = 34 if n <= 2048 else 22
    for target in targets:
        if time.monotonic() >= deadline:
            break
        seed = int(rng.integers(0, 2**31 - 1))
        a, s = _spectral_target_init(n, target, base, iters, deadline, seed)
        if np.isfinite(s):
            out.append(a)

    return out


def _optimize_loop(a_in, n, max_seconds, seed, sym, verbose):
    n = int(max(4, n))
    a = _normalize(np.asarray(a_in, dtype=np.float64).ravel(), sym)
    if a.size != n:
        a = _interp(a, n)

    v = _a_to_log(a)
    best_a = a.copy()
    best_s = _score(a)
    if not np.isfinite(best_s):
        best_a = np.ones(n, dtype=np.float64) / float(n)
        best_s = 2.0

    rng = np.random.default_rng(int(seed))
    deadline = time.monotonic() + max(1.0, float(max_seconds))
    alpha = 0.25
    policy = None
    it = 0
    last_improve = time.monotonic()
    last_print = time.monotonic()
    stuck_limit = min(45.0, max(6.0, 0.20 * float(max_seconds)))
    current_s = best_s

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        if it % 35 == 0:
            policy = _random_policy(rng)
            alpha = float(policy.get("alpha", 0.25))
            alpha = float(np.clip(alpha, 0.06, 0.85))

        c = _conv_self(a)
        s_cur = _score_from_conv(a, c)
        if not np.isfinite(s_cur):
            a = best_a.copy()
            v = _a_to_log(a)
            it = 0
            continue

        current_s = float(s_cur)
        if s_cur < best_s - 1e-12:
            best_s = float(s_cur)
            best_a = a.copy()
            last_improve = now

        if np.isfinite(best_s) and s_cur > best_s * 1.10:
            a = best_a.copy()
            v = _a_to_log(a)
            it = 0
            continue

        if now - last_improve > stuck_limit:
            base_a = best_a if (np.isfinite(best_s) and s_cur > best_s * 1.02) else a
            v = _a_to_log(base_a)
            sigma = 0.12 + 0.35 * float(rng.random())
            v = v + rng.normal(0.0, sigma, n)
            if sym:
                v = 0.5 * (v + v[::-1])
            v = v - float(np.mean(v))
            v = np.clip(v, -60.0, 60.0)
            a = _from_log(v, sym)
            last_improve = now
            it = 0
            continue

        g = _make_grad(a, c, policy, sym)
        gm = float(np.max(g)) if g.size else 0.0
        if not np.isfinite(gm) or gm <= 1e-300:
            v = v + rng.normal(0.0, 0.08, n)
            if sym:
                v = 0.5 * (v + v[::-1])
            v = v - float(np.mean(v))
            v = np.clip(v, -60.0, 60.0)
            a = _from_log(v, sym)
            it = 0
            continue

        floor = max(1e-8 * gm, np.finfo(np.float64).tiny)
        lg = np.log(np.maximum(g, floor))
        lg = lg - float(np.mean(lg))

        vt = v - alpha * lg
        vt = vt - float(np.mean(vt))
        vt = np.clip(vt, -60.0, 60.0)
        at = _from_log(vt, sym)
        st = _score(at)

        accepted = False
        if np.isfinite(st):
            if st < best_s - 1e-12:
                a = at
                v = vt
                best_s = float(st)
                best_a = a.copy()
                last_improve = time.monotonic()
                alpha = min(0.90, alpha * 1.05)
                accepted = True
            elif st < s_cur:
                a = at
                v = vt
                last_improve = time.monotonic()
                alpha = min(0.90, alpha * 1.02)
                accepted = True

        if not accepted and alpha > 0.15:
            vt2 = v - 0.5 * alpha * lg
            vt2 = vt2 - float(np.mean(vt2))
            vt2 = np.clip(vt2, -60.0, 60.0)
            at2 = _from_log(vt2, sym)
            st2 = _score(at2)
            if np.isfinite(st2) and st2 < s_cur:
                a = at2
                v = vt2
                if st2 < best_s - 1e-12:
                    best_s = float(st2)
                    best_a = a.copy()
                last_improve = time.monotonic()
                alpha = max(0.06, 0.5 * alpha)
                accepted = True

        if not accepted:
            alpha = max(0.06, 0.72 * alpha)
            if alpha < 0.10 and it % 7 == 0:
                sigma = 0.05 + 0.12 * float(rng.random())
                v = v + rng.normal(0.0, sigma, n)
                if sym:
                    v = 0.5 * (v + v[::-1])
                v = v - float(np.mean(v))
                v = np.clip(v, -60.0, 60.0)
                a = _from_log(v, sym)
                it = 0

        it += 1
        if verbose and now - last_print > 30.0:
            print("      opt n=%d sym=%s it=%d best=%.6f cur=%.6f alpha=%.3f" %
                  (n, bool(sym), it, best_s, current_s, alpha))
            last_print = now

    out = _normalize(best_a, sym)
    return out, float(best_s)


def _process_length(n, deadline, prev_best, rng, verbose=True):
    n = int(max(4, n))
    if prev_best is None:
        prev_best = np.ones(n, dtype=np.float64) / float(n)

    now = time.monotonic()
    if now >= deadline:
        return prev_best, _score(prev_best)

    p = _normalize(prev_best)
    p = _interp(p, n)

    cands = []
    cands.append(p)
    cands.append(_symmetrize(p))
    cands.extend(_param_candidates(n, p, rng))

    unique = _select_candidates(cands, 8, 0.045, 36)
    if not unique:
        a = np.ones(n, dtype=np.float64) / float(n)
        s = _score(a)
        unique = [(float(s), a)]

    local_best = unique[0][1]
    local_score = float(unique[0][0])

    rem = max(0.0, float(deadline) - time.monotonic())
    if rem > 40.0 and n <= 4096:
        spec_budget = min(42.0, max(12.0, 0.16 * rem))
        spec_cands = _spectral_candidates(n, local_best, spec_budget, rng)
        if spec_cands:
            all_cands = list(cands) + list(spec_cands)
            unique2 = _select_candidates(all_cands, 8, 0.045, 36)
            if unique2:
                if unique2[0][0] < local_score:
                    local_best = unique2[0][1]
                    local_score = float(unique2[0][0])
                unique = unique2

    top_count = min(5, len(unique))
    for pos in range(top_count):
        now = time.monotonic()
        rem = max(0.0, float(deadline) - now)
        if rem < 25.0:
            break

        sc, a = unique[pos]
        sym = (pos % 2 == 0)
        a0 = _symmetrize(a) if sym else a

        if pos == 0:
            alloc = min(max(15.0, rem * 0.34), 1200.0 if n <= 4096 else 700.0)
        elif pos == 1:
            alloc = min(max(10.0, rem * 0.25), 800.0 if n <= 4096 else 450.0)
        elif pos == 2:
            alloc = min(max(8.0, rem * 0.18), 500.0 if n <= 4096 else 320.0)
        elif pos == 3:
            alloc = min(max(6.0, rem * 0.15), 360.0 if n <= 4096 else 240.0)
        else:
            alloc = min(max(5.0, rem * 0.12), 240.0 if n <= 4096 else 180.0)

        alloc = min(alloc, max(1.0, rem - 12.0))
        if alloc <= 0.0:
            break

        seed = int(rng.integers(0, 2**31 - 1))
        a_opt, sc_opt = _optimize_loop(a0, n, alloc, seed, sym, verbose=(pos == 0 and verbose))

        if np.isfinite(sc_opt) and sc_opt < local_score:
            local_score = float(sc_opt)
            local_best = a_opt

        opp_bool = (not sym)
        if opp_bool:
            a_opp = _symmetrize(a_opt)
            s_opp = _score(a_opp)
            if np.isfinite(s_opp) and s_opp < local_score:
                local_score = float(s_opp)
                local_best = a_opp

        if verbose:
            print("    n=%d cand %d sym=%s score=%.6f best=%.6f" %
                  (n, pos, bool(sym), sc_opt, local_score))

    now = time.monotonic()
    rem = max(0.0, float(deadline) - now)
    if rem > 30.0 and local_best is not None:
        a_sym = _symmetrize(local_best)
        s_sym = _score(a_sym)
        sym_polish = (np.isfinite(s_sym) and s_sym <= local_score)

        alloc = min(max(10.0, 0.65 * rem), 300.0 if n <= 4096 else 200.0)
        alloc = min(alloc, max(1.0, rem - 12.0))
        seed = int(rng.integers(0, 2**31 - 1))
        a_opt, sc_opt = _optimize_loop(local_best, n, alloc, seed, sym_polish, False)

        if np.isfinite(sc_opt) and sc_opt < local_score:
            local_score = float(sc_opt)
            local_best = a_opt

        a_sym2 = _symmetrize(local_best)
        s_sym2 = _score(a_sym2)
        if np.isfinite(s_sym2) and s_sym2 < local_score:
            local_score = float(s_sym2)
            local_best = a_sym2

    return local_best, float(local_score)


def _call_exact(a):
    ev = globals().get("evaluate_sequence", None)
    if ev is None:
        return np.inf
    if len(a) > 8192:
        return np.inf
    try:
        return float(ev([float(x) for x in a]))
    except Exception:
        return np.inf


def _sanitize(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s < 0.01:
        arr = np.ones(2048, dtype=np.float64) / 2048.0
    else:
        arr = arr / s
    return arr


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    seed = int(seed)
    budget_s = float(max(120.0, budget_s))
    start = time.monotonic()
    deadline = start + max(60.0, budget_s - 20.0)
    rng = np.random.default_rng(seed)

    best_a = _get_previous()
    if best_a is None:
        best_a = np.ones(2048, dtype=np.float64) / 2048.0

    best_score = _score(best_a)
    sym_prev = _symmetrize(best_a)
    sym_prev_score = _score(sym_prev)
    if np.isfinite(sym_prev_score) and sym_prev_score < best_score:
        best_a = sym_prev
        best_score = sym_prev_score

    print("Initial best: n=%d score=%.6f" % (len(best_a), best_score))

    base = max(60.0, budget_s - 25.0)
    stage_defs = [
        (512, 0.025),
        (1024, 0.060),
        (2048, 0.240),
        (4096, 0.340),
        (8192, 0.185),
        (16384, 0.100),
    ]

    for n, frac in stage_defs:
        now = time.monotonic()
        stage_budget = max(18.0, base * frac)
        stage_deadline = min(float(deadline) - 12.0, now + stage_budget)
        if stage_deadline <= now + 18.0:
            break

        print("Stage n=%d budget=%.1fs current_best=%.6f" % (n, stage_budget, best_score))
        a_new, s_new = _process_length(int(n), stage_deadline, best_a, rng, True)

        if a_new is not None and np.isfinite(s_new) and s_new < best_score:
            best_a = a_new
            best_score = float(s_new)
            print("  improved n=%d score=%.6f" % (n, best_score))

    now = time.monotonic()
    if now < float(deadline) - 35.0 and best_a is not None:
        nbest = int(len(best_a))
        for sym in (True, False):
            now = time.monotonic()
            rem = max(0.0, float(deadline) - now - 12.0)
            if rem < 25.0:
                break
            alloc = min(max(10.0, 0.55 * rem), 260.0 if nbest <= 4096 else 180.0)
            seed = int(rng.integers(0, 2**31 - 1))
            a_opt, s_opt = _optimize_loop(best_a, nbest, alloc, seed, sym, False)
            if np.isfinite(s_opt) and s_opt < best_score:
                best_score = float(s_opt)
                best_a = a_opt
        sym_check = _symmetrize(best_a)
        sym_check_score = _score(sym_check)
        if np.isfinite(sym_check_score) and sym_check_score < best_score:
            best_score = float(sym_check_score)
            best_a = sym_check

    final = _sanitize(best_a)
    fft_final = _score(final)
    exact_final = _call_exact(final)

    if np.isfinite(exact_final):
        print("Final FFT score=%.6f exact score=%.6f n=%d" % (fft_final, exact_final, len(final)))
        if not np.isfinite(best_score):
            best_score = exact_final
    else:
        print("Final FFT score=%.6f n=%d exact skipped/invalid" % (fft_final, len(final)))

    if not np.isfinite(fft_final) or len(final) == 0:
        final = _sanitize(np.ones(2048, dtype=np.float64) / 2048.0)

    return [float(x) for x in final]
