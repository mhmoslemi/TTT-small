def _lp_complement(f, margin=1.0, max_n=1024):
   if scipy_opt is None: return None
   f=_normalize(f)
   n=len(f)
   if n>max_n: f=_resample(f,max_n); n=max_n
   M = _score(f)/(2*n) * margin
   if not finite or M<=0: return None
   A=np.zeros((2*n-1,n), dtype=np.float64)
   for k in range(2*n-1):
      s=max(0,k-n+1); e=min(n,k+1)
      if e>s: A[k,s:e] = f[k-s:k+1]
   c=-np.ones(n)
   try:
      res=scipy_opt.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0,None), method='highs', options={'presolve':True})
   except TypeError:
      res=scipy_opt.linprog(c, A_ub=A, b_ub=..., bounds=..., method='interior-point')
   if res.success:
      x=res.x
      x=np.maximum(x,0)
      if x.sum()>1e-12: return x/x.sum()
   return None

"""
Search for nonnegative step-function heights minimizing
score(a) = 2 n max(a * a) / (sum a)^2.

The search combines:
  * FFT-based exact scoring and gradients for a smoothed log-sum-exp surrogate,
  * cosine/top-lag active-set projected gradient moves,
  * multiplicative correction moves,
  * local polynomial/gaussian-mixture parametric starts,
  * coarse LP-based complementary updates (primal packing updates),
  * log-softmax L-BFGS polishing,
  * adaptive length staging (256 -> 4096) and resampling.
"""

import numpy as np
import time

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")

# Globals used by top-level objective/jacobian calls for L-BFGS.
_LOGIT_BETA = 1.0
_LOGIT_N1 = 1
_LOGIT_N2 = 1
_LOGIT_END = 0.0


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _conv_full(a, fft_len):
    n = a.size
    A = np.fft.rfft(a, fft_len)
    return np.fft.irfft(A * A, fft_len)


def _conv_score(a, fft_len=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return None, np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / float(n)
        s = 1.0
    a = a / s
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    b = _conv_full(a, fft_len)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return b, np.inf
    return b, float(2.0 * n * mb)


def _score(a):
    _, s = _conv_score(a, None)
    return s


def _conv_weighted(a, weights, fft_len):
    """
    For b_k = sum_i a_i a_{k-i}, gradient direction weighted by p_k:
        h_i = 2 * sum_k p_k a_{k-i}
    """
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_len)
    W = np.fft.rfft(weights, fft_len)
    conv = np.fft.irfft(A * W, fft_len).real
    return conv[a.size - 1:a.size - 1 + a.size]


def _weights_lse(b, beta):
    size = b.size
    if size == 0:
        return np.ones(1), 0.0
    if beta <= 1.0:
        return np.full(size, 1.0 / size), float(np.max(b))
    m = float(np.max(b))
    if not np.isfinite(m):
        p = np.zeros(size)
        p[0] = 1.0
        return p, 0.0
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 1e-300:
        p = np.zeros(size)
        p[int(np.argmax(b))] = 1.0
        return p, m
    p = e / s
    val = float(m + np.log(s) / beta)
    return p, val


def _lse_value(b, beta):
    if b.size == 0:
        return 0.0
    m = float(np.max(b))
    if not np.isfinite(m) or beta <= 1.0:
        return m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 1e-300:
        return m
    return float(m + np.log(s) / beta)


def _weights_top(b, frac):
    size = b.size
    if size == 0:
        return np.ones(1)
    k = int(max(1, size * float(frac)))
    if k >= size:
        return np.full(size, 1.0 / size)
    idx = np.argpartition(b, size - k)[-k:]
    p = np.zeros(size)
    p[idx] = 1.0 / float(k)
    return p


def _resample(a, new_n):
    a = _normalize(a)
    new_n = max(2, int(new_n))
    old_n = a.size
    if old_n == 0:
        return _normalize(np.ones(new_n))
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, 1.0 / float(new_n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    y = np.interp(x_new, x_old, a)
    return _normalize(y)


def _make_uniform(n):
    n = max(2, int(n))
    return np.ones(n, dtype=np.float64) / float(n)


def _make_exponential(n, alpha=1.0):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-float(alpha) * x))


def _make_mixture(n, alpha=1.0, gamma=0.5):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-float(alpha) * x) + float(gamma) * np.exp(-float(alpha) * (1.0 - x))
    a = a + 0.125
    return _normalize(a)


def _make_sine_basis(n, m=3, eps=0.6, phase=0.0):
    n = max(2, int(n))
    m = max(1, int(m))
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + float(eps) * np.sin(2.0 * np.pi * float(m) * x + float(phase))
    a = np.maximum(a, 1e-8)
    return _normalize(a)


def _make_random_smooth(n, rng, base=None, amp=0.5):
    n = max(2, int(n))
    if base is None:
        base = _make_uniform(n)
    base = _normalize(base)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cut = max(2, len(F) // max(2, n // 12))
    if cut < len(F):
        F[cut:] = 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 1e-300:
        f = f / sd
    a = base * (1.0 + float(amp) * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _parametric_sequence(params, n):
    """
    Three Gaussian bumps plus a base, with parameters:
      0:2 centers, 3:5 log widths, 6:8 log amplitudes.
    """
    n = max(2, int(n))
    p = np.asarray(params, dtype=np.float64).ravel()
    if p.size < 9:
        p = np.concatenate([p, np.zeros(9 - p.size)])
    centers = np.clip(p[0:3], 0.0, 1.0)
    log_w = np.clip(p[3:6], -4.0, 0.5)
    log_a = np.clip(p[6:9], -4.0, 2.0)
    x = np.linspace(0.0, 1.0, n)
    y = 0.35 * np.ones(n, dtype=np.float64)
    for j in range(3):
        w = float(np.exp(log_w[j]))
        amp = float(np.exp(log_a[j]))
        y += amp * np.exp(-0.5 * ((x - centers[j]) / max(1e-6, w)) ** 2)
    return _normalize(y)


def _parametric_score(params, n, end):
    if time.monotonic() > end:
        return 1e100
    a = _parametric_sequence(params, n)
    s = _score(a)
    if not np.isfinite(s):
        return 1e100
    return float(s)


def _logits_to_a(l):
    l = np.asarray(l, dtype=np.float64).ravel()
    l = np.nan_to_num(l, nan=0.0, posinf=0.0, neginf=0.0)
    if l.size == 0:
        return np.ones(1, dtype=np.float64)
    m = float(np.max(np.maximum(l, 0.0)))
    e = np.exp(l - m)
    last = float(np.exp(-m))
    s = float(e.sum() + last)
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(l.size + 1, dtype=np.float64) / float(l.size + 1)
    a = np.empty(l.size + 1, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = last / s
    return a


def _logit_obj(l):
    if time.monotonic() > _LOGIT_END:
        return 1e100
    a = _logits_to_a(l)
    b = _conv_full(a, _LOGIT_N1)
    val = _lse_value(b, _LOGIT_BETA)
    if not np.isfinite(val):
        return 1e100
    return float(val)


def _logit_jac(l):
    if time.monotonic() > _LOGIT_END:
        return np.zeros(max(0, int(np.asarray(l).size)), dtype=np.float64)
    a = _logits_to_a(l)
    b = _conv_full(a, _LOGIT_N1)
    p, _ = _weights_lse(b, _LOGIT_BETA)
    h = _conv_weighted(a, p, _LOGIT_N2)
    dot = float(np.dot(a, h))
    g = a[:-1] * (h[:-1] - dot)
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    return g


def _perturb(a, rng, mode=None, strength=0.2):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return np.ones(1)
    if mode is None:
        mode = int(rng.integers(0, 5))
    c = a.copy()
    strength = max(0.05, float(strength))
    if mode == 0:
        lo = max(2, n // 64)
        hi = max(lo + 1, n // 8 + 3)
        width = int(rng.integers(lo, hi))
        width = max(2, min(n, width))
        if n >= width:
            start = int(rng.integers(0, n - width + 1))
        else:
            start = 0
        c[start:start + width] *= np.exp(rng.normal(0.0, strength))
    elif mode == 1:
        z = rng.normal(size=n)
        F = np.fft.rfft(z)
        cut = max(2, len(F) // max(2, n // 12))
        if cut < len(F):
            F[cut:] = 0.0
        f = np.fft.irfft(F, n)
        f = f - float(np.mean(f))
        sd = float(np.std(f))
        if sd > 1e-300:
            f = f / sd
        c = a * (1.0 + 0.5 * f)
    elif mode == 2:
        c = a * np.exp(rng.normal(0.0, strength, size=n)) + 1e-12
    elif mode == 3:
        c = 0.5 * a + 0.5 * a[::-1]
    else:
        count = max(1, n // 64)
        width = max(2, n // 128)
        for _ in range(count):
            if n >= width:
                start = int(rng.integers(0, n - width + 1))
            else:
                start = 0
            stop = min(n, start + width)
            c[start:stop] *= float(rng.uniform(0.35, 1.7))
    c = np.maximum(c, 0.0)
    return _normalize(c)


def _generate_inits(n, given, best, rng):
    n = max(16, int(n))
    inits = []
    exp1 = _make_exponential(n, 1.0)
    exp2 = _make_exponential(n, 0.6)
    inits.append(exp1)
    inits.append(exp2)
    inits.append(_make_mixture(n, 0.85, 0.55))
    if best is not None:
        inits.append(_resample(best, n))
    if given is not None:
        inits.append(_resample(given, n))
    base_rand = _resample(best, n) if best is not None else exp1
    inits.append(_make_random_smooth(n, rng, base_rand, 0.55))
    m = int(rng.integers(1, max(2, min(32, max(2, n // 8)))))
    inits.append(_make_sine_basis(n, m, float(rng.uniform(0.25, 0.85)), float(rng.uniform(0.0, 2.0 * np.pi))))
    centers = list(rng.uniform(0.0, 1.0, 3))
    widths = list(max(0.03, rng.uniform(0.03, 0.35, 3)))
    amps = list(max(0.25, rng.uniform(0.25, 2.0, 3)))
    params = np.concatenate([np.asarray(centers), np.log(np.asarray(widths)), np.log(np.asarray(amps))])
    inits.append(_parametric_sequence(params, n))

    seen = []
    out = []
    for arr in inits:
        arr = _normalize(arr)
        step = max(1, arr.size // 16)
        sig = tuple(np.round(arr[::step], 6))
        if sig not in seen:
            seen.append(sig)
            out.append(arr)
    return out


def _refine_sequence(a0, end, rng, beta_start=15.0):
    a = _normalize(a0)
    n = a.size
    if n < 16:
        a = _make_exponential(64, 1.0)
        n = a.size

    n1 = _next_pow2(2 * n - 1)
    n2 = _next_pow2(3 * n - 2)

    b, current_score = _conv_score(a, n1)
    if b is None or not np.isfinite(current_score):
        a = _make_exponential(128, 1.0)
        n = a.size
        n1 = _next_pow2(2 * n - 1)
        n2 = _next_pow2(3 * n - 2)
        b, current_score = _conv_score(a, n1)

    best_a = a.copy()
    best_score = float(current_score)
    current = a.copy()

    eta = 0.12
    T = max(1e-7, 0.002 * max(1e-4, best_score - 1.0))
    beta_param = float(beta_start)
    active_frac = 0.02
    it = 0
    fail = 0
    last_best = 0

    while time.monotonic() < end:
        if b is None or not np.isfinite(current_score):
            break
        maxb = float(np.max(b))
        if not np.isfinite(maxb) or maxb <= 1e-300:
            current = _make_exponential(n, 1.0)
            b, current_score = _conv_score(current, n1)
            fail = 0
            it += 1
            continue

        beta = min(1e7, max(1.0, beta_param / maxb))
        step_class = it % 8
        if step_class < 5:
            p, _ = _weights_lse(b, beta)
        elif step_class == 5:
            p = _weights_top(b, active_frac)
        elif step_class == 6:
            p = _weights_top(b, min(0.50, active_frac * 3.0))
        else:
            p = np.zeros(b.size, dtype=np.float64)
            p[int(np.argmax(b))] = 1.0

        h = _conv_weighted(current, p, n2)
        h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
        center = h - float(np.mean(h))
        max_center = float(np.max(np.abs(center)))
        if (not np.isfinite(max_center)) or max_center <= 1e-300:
            current = _perturb(best_a, rng, mode=1, strength=0.25)
            b, current_score = _conv_score(current, n1)
            fail = 0
            it += 1
            continue

        cand = current - eta * center
        cand = _normalize(cand)
        cand_b, cand_score = _conv_score(cand, n1)

        accepted = False
        if cand_score < current_score:
            accepted = True
            eta = min(1.1, eta * 1.12)
        else:
            diff = cand_score - current_score
            if diff > -1e-14 and diff < T:
                if diff / T < 700.0:
                    prob = float(np.exp(-(diff / T)))
                else:
                    prob = 0.0
                if prob > 0.0 and rng.random() < prob:
                    accepted = True

        if accepted:
            current = cand
            b = cand_b
            current_score = cand_score
            fail = 0
        else:
            eta = max(0.02, eta * 0.75)
            fail += 1

        # Multiplicative correction occasionally.
        if (it % 5 == 0 or fail > 40) and time.monotonic() < end - 1.0:
            h_mean = max(1e-300, float(np.mean(h)))
            cand2 = current / (1.0 + 0.5 * (h / h_mean))
            cand2 = _normalize(cand2)
            cand2_b, cand2_score = _conv_score(cand2, n1)
            if cand2_score < current_score:
                current = cand2
                b = cand2_b
                current_score = cand2_score
                fail = 0
            elif cand2_score < best_score + T and rng.random() < 0.12:
                current = cand2
                b = cand2_b
                current_score = cand2_score

        if current_score < best_score:
            best_a = current.copy()
            best_score = float(current_score)
            last_best = it
            T = max(1e-8, T * 0.9)

        if it % 100 == 0:
            beta_param = min(250.0, beta_param * 1.12)
            active_frac = min(0.45, max(0.02, active_frac * 1.08 + 0.002))
            T = max(1e-8, T * 0.97)

        if fail > 320 or (it - last_best > 750 and it > 120):
            if current_score > best_score + max(1e-7, 1e-4 * max(1e-4, best_score - 1.0)):
                current = best_a.copy()
            else:
                current = _perturb(best_a, rng, mode=int(rng.integers(0, 4)), strength=0.35)
            b, current_score = _conv_score(current, n1)
            eta = 0.12
            beta_param = max(8.0, beta_param * 0.70)
            active_frac = max(0.02, active_frac * 0.70)
            fail = 0
            T = max(1e-7, 0.002 * max(1e-4, best_score - 1.0))

        it += 1

    return best_a, float(best_score)


def _refine_llogit(a0, end, rng, beta_start=50.0):
    a = _normalize(a0)
    n = a.size
    if n < 16:
        a = _make_exponential(64, 1.0)
        n = a.size

    rem = max(1.0, float(end - time.monotonic()))
    b, score = _conv_score(a, None)
    best_a = a.copy()
    best_score = float(score)

    if _scipy_opt is not None and n <= 1200:
        maxb = float(np.max(b)) if b is not None else np.inf
        if np.isfinite(maxb) and maxb > 1e-300:
            n1 = _next_pow2(2 * n - 1)
            n2 = _next_pow2(3 * n - 2)
            beta = min(1e7, max(1.0, float(beta_start) / maxb))
            l0 = np.zeros(max(0, n - 1), dtype=np.float64)
            if a[-1] > 1e-300:
                l0 = np.log(np.maximum(a[:-1], 1e-300) / a[-1])
            else:
                l0 = rng.normal(size=max(0, n - 1)) * 0.2
            l0 = np.nan_to_num(l0, nan=0.0, posinf=0.0, neginf=0.0)

            global _LOGIT_BETA, _LOGIT_N1, _LOGIT_N2, _LOGIT_END
            _LOGIT_BETA = beta
            _LOGIT_N1 = n1
            _LOGIT_N2 = n2
            _LOGIT_END = end

            max_iter = int(max(20, min(90, 20 + 70 * rem / 60.0)))
            options = {
                "maxiter": max_iter,
                "maxfun": max_iter * 3,
                "gtol": 1e-8,
                "ftol": 1e-12,
                "disp": False,
            }
            if maxb > 1e-300 and np.isfinite(maxb):
                options["fenscale"] = float(1.0 / maxb)
            try:
                res = _scipy_opt.minimize(_logit_obj, l0, method="L-BFGS-B", jac=_logit_jac, options=options)
                if res is not None and hasattr(res, "x") and res.x is not None:
                    cand = _logits_to_a(res.x)
                    cand = _normalize(cand)
                    cand_score = _score(cand)
                    if np.isfinite(cand_score) and cand_score < best_score:
                        best_a = cand
                        best_score = float(cand_score)
            except Exception:
                pass

    if time.monotonic() < end - 5.0:
        ra, rs = _refine_sequence(best_a.copy(), end, rng, beta_start=max(20.0, float(beta_start) * 0.6))
        if np.isfinite(rs) and rs < best_score:
            best_a = ra
            best_score = float(rs)

    return best_a, float(best_score)


def _lp_complement(f, target_score=1.49, max_n=256):
    if _scipy_opt is None:
        return None
    f = _normalize(f)
    n = int(max(64, min(max_n, f.size)))
    fc = _resample(f, n)
    n = fc.size

    fc = np.maximum(fc, 1e-6)
    s = float(fc.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    fc = fc / s

    _, fc_score = _conv_score(fc, None)
    fc_max = fc_score / (2.0 * n)
    target_max = max(1e-12, float(target_score) / (2.0 * n))
    if not np.isfinite(fc_max) or fc_max <= 0.0:
        return None

    M = 1.05 * min(fc_max, target_max)
    M = max(M, 0.60 * fc_max)
    if not np.isfinite(M) or M <= 0.0:
        return None

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s0 = max(0, k - n + 1)
        e0 = min(n, k + 1)
        if e0 > s0:
            A[k, s0:e0] = fc[k - e0 + 1:k - s0 + 1][::-1]

    c = -np.ones(n, dtype=np.float64)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    res = None
    try:
        res = _scipy_opt.linprog(c, A_ub=A, b_ub=b, bounds=(0.0, None), method="highs", options={"presolve": True})
    except Exception:
        try:
            res = _scipy_opt.linprog(c, A_ub=A, b_ub=b, bounds=(0.0, None), method="interior-point",
                                     options={"maxiter": 2000})
        except Exception:
            return None

    if res is None or not hasattr(res, "x") or res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    x = np.maximum(x, 0.0)
    if not np.all(np.isfinite(x)) or x.sum() <= 1e-12:
        return None
    # Feasibility sanity check.
    Ax = A @ x
    if not np.all(np.isfinite(Ax)):
        return None
    if np.max(Ax) > M + max(1e-9, 1e-6 * M):
        return None
    return x / x.sum()


def _lp_combo(a_full, end, rng, target_score=1.49):
    a = _normalize(a_full)
    score = _score(a)
    best = a.copy()
    best_score = score

    variants = [a.copy()]
    if a.size > 20:
        variants.append(_normalize(0.75 * a + 0.25 * _make_exponential(a.size, 1.0)))
        if score > float(target_score) + 0.01:
            variants.append(_perturb(a, rng, mode=0, strength=0.35))

    for f in variants:
        if time.monotonic() > end - 25.0:
            break
        max_n = 256
        if a.size <= 512 and time.monotonic() < end - 60.0:
            max_n = 512
        g = _lp_complement(f, float(target_score), max_n)
        if g is None:
            continue
        g_full = _resample(g, a.size)
        for mix in (1.0, 0.8, 0.55, 0.35, 0.2):
            cand = (1.0 - mix) * best + mix * g_full
            cand = _normalize(cand)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
                print(f"LP-combo mix={mix:.2f} score={sc:.8f}", flush=True)
    return best, float(best_score)


def _lp_iterate(f_full, end, rng, target_score=1.49):
    if _scipy_opt is None:
        return _lp_combo(f_full, end, rng, target_score)

    f_full = _normalize(f_full)
    full_size = f_full.size
    n = 256

    f = _resample(f_full, n)
    _, score = _conv_score(f, None)
    best = f.copy()
    best_score = float(score)

    # Short direct refinement at coarse length before LP iteration.
    if time.monotonic() < end - 20.0:
        r_end = min(end - 10.0, time.monotonic() + 12.0)
        rf, rs = _refine_sequence(f.copy(), r_end, rng, beta_start=15.0)
        if np.isfinite(rs) and rs < best_score:
            best = rf.copy()
            best_score = float(rs)

    for it in range(60):
        now = time.monotonic()
        if now > end - 18.0:
            break
        cur_target = min(float(target_score), max(1.0, best_score - 0.015))
        g = _lp_complement(best, cur_target, max_n=n)
        if g is None:
            # Relax once.
            g = _lp_complement(best, best_score, max_n=n)
            if g is None:
                break

        best_mix = None
        best_mix_score = best_score
        for mix in (1.0, 0.8, 0.6, 0.4, 0.25, 0.12):
            cand = (1.0 - mix) * best + mix * g
            cand = _normalize(cand)
            _, sc = _conv_score(cand, None)
            if np.isfinite(sc) and sc < best_mix_score:
                best_mix = cand
                best_mix_score = sc

        if best_mix is None:
            if it > 4:
                break
            best = _perturb(best, rng, mode=0, strength=0.25)
            _, best_score = _conv_score(best, None)
            continue

        best = best_mix
        best_score = float(best_mix_score)
        print(f"LP-iter coarse n={n} it={it} score={best_score:.8f}", flush=True)

        # Occasional short local refinement to avoid LP plateau.
        if time.monotonic() < end - 35.0 and it % 3 == 0:
            r_end = min(end - 15.0, time.monotonic() + 8.0)
            rf, rs = _refine_sequence(best.copy(), r_end, rng, beta_start=20.0)
            if np.isfinite(rs) and rs < best_score:
                best = rf.copy()
                best_score = float(rs)

    if time.monotonic() < end - 5.0:
        rf, rs = _refine_sequence(best.copy(), end, rng, beta_start=25.0)
        if np.isfinite(rs) and rs < best_score:
            best = rf.copy()
            best_score = float(rs)

    if full_size != n:
        full = _resample(best, full_size)
        _, fsc = _conv_score(full, None)
        if np.isfinite(fsc) and fsc < best_score:
            return full, float(fsc)
    return best, float(best_score)


def _parametric_search(num_starts, end, rng, n=256):
    if _scipy_opt is None or time.monotonic() > end - 20.0:
        return None, np.inf
    best = None
    best_score = np.inf
    for s in range(max(1, int(num_starts))):
        if time.monotonic() > end - 10.0:
            break
        centers = rng.uniform(0.0, 1.0, 3)
        widths = np.clip(rng.uniform(0.04, 0.35, 3), 0.02, 0.9)
        amps = np.clip(rng.uniform(0.25, 2.0, 3), 0.05, 10.0)
        x0 = np.concatenate([centers, np.log(widths), np.log(amps)])
        x0 = np.nan_to_num(x0, nan=0.0, posinf=0.0, neginf=0.0)
        try:
            res = _scipy_opt.minimize(
                _parametric_score,
                x0,
                args=(n, end),
                method="Nelder-Mead",
                options={"maxiter": 600, "xatol": 1e-5, "fatol": 1e-8, "adaptive": True},
            )
        except Exception:
            res = None
        if res is not None and hasattr(res, "x") and res.x is not None:
            seq = _normalize(_parametric_sequence(res.x, n))
            sc = _score(seq)
            if np.isfinite(sc) and sc < best_score:
                best = seq
                best_score = float(sc)
                print(f"parametric start {s} score={sc:.8f}", flush=True)
    return best, float(best_score)


def _get_given_sequence():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return arr / s


def _update_best(best_a, best_score, cand, score, label=""):
    if cand is None:
        return best_a, best_score
    a = np.asarray(cand, dtype=np.float64).ravel()
    if a.size == 0:
        return best_a, best_score
    a = _normalize(a)
    if not np.isfinite(score):
        score = _score(a)
    if np.isfinite(score) and score < best_score:
        return a, float(score)
    return best_a, best_score


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """Run the search while respecting the time budget and return the best list."""
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 60.0:
        budget = 2400.0
    deadline = t0 + max(60.0, budget) - 45.0
    rng = np.random.default_rng(int(seed))

    best_a = None
    best_score = np.inf

    given = _get_given_sequence()
    if given is not None:
        given = _normalize(given)
        gs = _score(given)
        best_a, best_score = _update_best(best_a, best_score, given, gs, "given")
        if np.isfinite(best_score):
            print(f"initial given len={best_a.size} score={best_score:.8f}", flush=True)

    if best_a is None:
        best_a = _make_exponential(256, 1.0)
        best_score = _score(best_a)
        print(f"initial fallback len={best_a.size} score={best_score:.8f}", flush=True)

    scale = float(min(1.0, max(0.25, budget / 2400.0)))

    # Lightweight parametric starts.
    if time.monotonic() < deadline - 180.0:
        p_end = min(deadline - 180.0, time.monotonic() + 45.0 * scale)
        pseq, pscore = _parametric_search(3, p_end, rng, 256)
        if pseq is not None:
            best_a, best_score = _update_best(best_a, best_score, pseq, pscore, "parametric")
            if np.isfinite(pscore) and pscore <= best_score and time.monotonic() < deadline - 120.0:
                r_end = min(deadline - 120.0, time.monotonic() + 60.0 * scale)
                res, rscore = _refine_sequence(pseq.copy(), r_end, rng, beta_start=15.0)
                best_a, best_score = _update_best(best_a, best_score, res, rscore, "parametric_ref")
                print(f"after parametric len={best_a.size} score={best_score:.8f}", flush=True)

    # Coarse LP iteration at length 256.
    if best_score > 1.48 and time.monotonic() < deadline - 240.0:
        lp_end = min(deadline - 240.0, time.monotonic() + 90.0 * scale)
        res, rscore = _lp_iterate(best_a.copy(), lp_end, rng, target_score=1.49)
        best_a, best_score = _update_best(best_a, best_score, res, rscore, "lp_iter")
        print(f"after LP iteration len={best_a.size} score={best_score:.8f}", flush=True)

    # Staged refinement and resampling.
    configs = [
        (256, 2, 50),
        (512, 2, 80),
        (1024, 3, 120),
        (2048, 3, 200),
        (4096, 2, 260),
    ]
    for n, num, sec in configs:
        now = time.monotonic()
        if now > deadline - 120.0:
            break
        if best_score < 1.505 and n >= 2048:
            # If we are already near/below target, spend less time on very large lengths.
            continue
        max_total = float(num * sec * scale)
        max_total = min(max_total, max(0.0, deadline - now - 120.0))
        if max_total < 25.0:
            break
        per = max(15, int(max_total / max(1, num)))
        inits = _generate_inits(n, given, best_a, rng)
        for idx, init in enumerate(inits[:num]):
            now = time.monotonic()
            t_end = min(deadline - 30.0, now + float(per))
            if now >= t_end - 5.0:
                break
            res, rs = _refine_sequence(init.copy(), t_end, rng, beta_start=15.0 + 3.0 * idx)
            before = best_score
            best_a, best_score = _update_best(best_a, best_score, res, rs, f"n={res.size}")
            msg = f"n={res.size} init={idx} score={rs:.8f} best={best_score:.8f}"
            if res.size and res.size != n:
                msg += f" (returned_len={res.size})"
            print(msg, flush=True)
            if np.isfinite(best_score) and before > 1.51 and best_score < 1.505:
                # Push immediately into larger lengths if a good basin appears.
                if n < 2048 and time.monotonic() < deadline - 180.0:
                    break

    # A second LP attempt if search is still slow to drop below the target.
    if best_score > 1.495 and time.monotonic() < deadline - 150.0:
        lp_end = min(deadline - 150.0, time.monotonic() + 70.0 * scale)
        res, rs = _lp_combo(best_a.copy(), lp_end, rng, target_score=1.48)
        best_a, best_score = _update_best(best_a, best_score, res, rs, "lp_combo")
        print(f"after LP combo len={best_a.size} score={best_score:.8f}", flush=True)
        if np.isfinite(rs) and rs <= best_score and time.monotonic() < deadline - 90.0:
            r_end = min(deadline - 90.0, time.monotonic() + 70.0 * scale)
            res2, rs2 = _refine_sequence(best_a.copy(), r_end, rng, beta_start=25.0)
            best_a, best_score = _update_best(best_a, best_score, res2, rs2, "rc_lop")
            print(f"after LP-follow len={best_a.size} score={best_score:.8f}", flush=True)

    # Upscale once if still not close to target.
    if best_score > 1.515 and time.monotonic() < deadline - 180.0:
        n0 = max(16, int(best_a.size))
        n1 = min(4096, max(1024, 2 * n0))
        if n1 > n0:
            init = _resample(best_a, n1)
            init = _normalize(0.85 * init + 0.15 * _make_exponential(n1, 1.0))
            t_end = min(deadline - 120.0, time.monotonic() + 150.0 * scale)
            res, rs = _refine_sequence(init, t_end, rng, beta_start=25.0)
            best_a, best_score = _update_best(best_a, best_score, res, rs, "upscale")
            print(f"after upscale len={best_a.size} score={best_score:.8f}", flush=True)

    # L-BFGS on logits for final polish if feasible.
    if best_a.size <= 1200 and time.monotonic() < deadline - 140.0:
        t_end = min(deadline - 140.0, time.monotonic() + 90.0 * scale)
        res, rs = _refine_llogit(best_a.copy(), t_end, rng, beta_start=55.0)
        best_a, best_score = _update_best(best_a, best_score, res, rs, "llogit")
        print(f"after llogit len={best_a.size} score={best_score:.8f}", flush=True)

    # Final deep refinement.
    if best_a is None:
        best_a = _make_exponential(256, 1.0)
    if time.monotonic() < deadline - 3.0:
        res, rs = _refine_sequence(best_a.copy(), deadline - 3.0, rng, beta_start=35.0)
        best_a, best_score = _update_best(best_a, best_score, res, rs, "final")
        print(f"after final refine len={best_a.size} score={best_score:.8f}", flush=True)

    out = _normalize(best_a)
    out_list = [float(x) for x in out]

    # Final report.  Our own FFT score is used throughout; evaluate_sequence is the official objective.
    print(f"final own_fft score={_score(out_list):.10f} length={len(out_list)}", flush=True)
    try:
        official = evaluate_sequence(out_list)
        print(f"final official score={official:.10f}", flush=True)
    except Exception:
        pass

    return out_list
