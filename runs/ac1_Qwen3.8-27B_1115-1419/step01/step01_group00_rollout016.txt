```
def _simplex_project(v, S):
   v=np.array(v, copy=True)
   v[v<0]=0
   s=v.sum()
   if s<=0: return np.full(n,S/n)
   # waterfilling: find rho such that max(v-rho,0) sum=S
   # sort v descending
   u=np.sort(v)[::-1]
   n=len(u)
   cs=np.cumsum(u)
   rho=(cs - S)/np.arange(1,n+1)
   # find smallest i where u[i] >= rho[i]
   # vectorized
   valid = u >= rho
   idx = np.argmax(valid)
   rho = rho[idx]
   return np.maximum(v-rho,0)
```

```
    gmax = float(np.max(np.abs(g)))
```

```
def _coeff_weights(C, tau):
    Cmax = float(np.max(C))
    if not np.isfinite(Cmax):
        p = np.zeros_like(C)
        p[0] = 1.0
        return p, Cmax
    if tau <= 1e-10:
        mask = C >= Cmax - max(1e-12, 1e-9*abs(Cmax))
        cnt = int(np.count_nonzero(mask))
        p = np.zeros_like(C)
        if cnt == 0:
            p[int(np.argmax(C))] = 1.0
        else:
            p[mask] = 1.0 / cnt
        return p, Cmax
    v = (C - Cmax) / tau
    # clamp v to avoid underflow? exp handles, but if v < -745, exp 0.
    expv = np.exp(v)

    if s <= 1e-300 or not np.isfinite(s):

        p[int(np.argmax(C))] = 1.0
        return p, Cmax
    p = expv / s
    J = Cmax + tau * math.log(s)
```

```python
"""
Optimizer for the nonnegative self-convolution objective

    2 * n * max(conv(a, a)) / sum(a)^2

The code works in the simplex sum(a)=n, where the objective is 2*max(conv(a,a))/n.
It uses FFT for fast convolutions/gradients, log-space multiplicative projected
updates, soft-max active-set smoothing, multi-initialization, bank-based
time-sliced refinement, random perturbations, and a final direct-convolution
polish.
"""

import numpy as np
import time
import math


def _score_pair_key(x):
    return x[0]


def _bank_key(x):
    return x[1]


def _next_power_of_two(n):
    m = 1
    while m < n:
        m <<= 1
    return m


def _conv_fft(a, m):
    n = len(a)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _objective_from_C(C, n):
    cmax = float(np.max(C))
    if not np.isfinite(cmax):
        return float("inf")
    if cmax < 0.0 and cmax > -1e-12:
        cmax = 0.0
    return 2.0 * cmax / float(n)


def _objective_direct(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return float("inf")
    if cmax < 0.0 and cmax > -1e-12:
        cmax = 0.0
    return 2.0 * float(n) * cmax / (s * s)


def _coeff_weights(C, tau):
    """Soft-max weights exp((C - max C)/tau) and smooth max value."""
    cmax = float(np.max(C))
    if not np.isfinite(cmax):
        p = np.zeros_like(C)
        p[0] = 1.0
        return p, cmax

    if tau <= 1e-10:
        tol = max(1e-12, 1e-9 * abs(cmax))
        mask = C >= cmax - tol
        cnt = int(np.count_nonzero(mask))
        p = np.zeros_like(C)
        if cnt <= 0:
            p[int(np.argmax(C))] = 1.0
        else:
            p[mask] = 1.0 / float(cnt)
        return p, cmax

    v = (C - cmax) / tau
    expv = np.exp(v)
    s = float(np.sum(expv))
    if (not np.isfinite(s)) or s <= 1e-300:
        p = np.zeros_like(C)
        p[int(np.argmax(C))] = 1.0
        return p, cmax

    p = expv / s
    jval = cmax + tau * math.log(s)
    return p, jval


def _grad_convolution_weights(a, w, m):
    """
    Gradient of sum_k w_k * (a*a)_k with respect to a.
    Uses linear convolution length convention.
    """
    n = len(a)
    ar = a[::-1]
    fa = np.fft.rfft(ar, m)
    fw = np.fft.rfft(w, m)
    h = np.fft.irfft(fa * fw, m)
    return 2.0 * h[n - 1:2 * n - 1]


def _simplex_normalize(a, n):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    np.maximum(a, 0.0, out=a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return np.ones(n, dtype=np.float64)
    a *= float(n) / s
    return a


def _positive_regularize(a, n):
    """Normalize to sum n and enforce tiny positivity for log-space updates."""
    b = _simplex_normalize(a, n)
    floor = 1e-12
    b = np.maximum(b, floor)
    b = _simplex_normalize(b, n)
    b = np.minimum(b, 1000.0)
    b = _simplex_normalize(b, n)
    return b


def _mul_update(a, g, lr, clip):
    """Multiplicative simplex step, equivalent to log-space gradient step."""
    gmax = float(np.max(np.abs(g)))
    if not np.isfinite(gmax) or gmax <= 1e-300:
        return a
    delta = -lr * (g - float(np.mean(g))) / gmax
    np.clip(delta, -clip, clip, out=delta)
    an = a * np.exp(delta)
    s = float(np.sum(an))
    if not np.isfinite(s) or s <= 0.0:
        return a
    an *= float(len(a)) / s
    return an


def _box_smooth(a, n, w):
    if w <= 1:
        return a.copy()
    w = min(int(w), n)
    kernel = np.ones(w, dtype=np.float64) / float(w)
    b = np.convolve(a, kernel, mode="same")
    np.maximum(b, 0.0, out=b)
    return b


def _resample_length(x, n):
    if x.size == n:
        return x.copy()
    x_old = np.linspace(0.0, 1.0, x.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, x)


def _parent_resampled(n):
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        return None
    if p.ndim != 1 or p.size < 4:
        return None
    p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
    np.maximum(p, 0.0, out=p)
    if float(np.sum(p)) <= 0.0:
        return None
    if p.size != n:
        p = _resample_length(p, n)
    return p


def _thue_bits(n):
    arr = np.empty(n, dtype=np.float64)
    for i in range(n):
        y = i
        parity = 0
        while y:
            parity ^= 1
            y &= y - 1
        arr[i] = 0.0 if parity == 0 else 1.0
    return arr


def _init_uniform(n):
    return np.ones(n, dtype=np.float64)


def _init_cos(n, freq, amp, phase, power):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    b = 1.0 + amp * np.cos(2.0 * math.pi * float(freq) * x + phase)
    np.maximum(b, 0.0, out=b)
    if power != 1.0:
        b = np.power(b, power)
    return b


def _init_random_smooth(n, rng):
    j = int(rng.integers(3, 9))
    freqs = rng.choice(np.arange(1, 21, dtype=int), size=j, replace=False)
    amps = rng.uniform(0.25, 0.95, size=j) / np.power(np.arange(1, j + 1, dtype=float), 0.6)
    phases = rng.uniform(0.0, 2.0 * math.pi, size=j)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    b = 1.8
    for f, amp, phi in zip(freqs, amps, phases):
        b = b + amp * np.cos(2.0 * math.pi * float(f) * x + phi)
    return np.maximum(b, 0.0)


def _init_gaussians(n, rng):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    k = int(rng.integers(1, 4))
    b = np.zeros(n, dtype=np.float64)
    for _ in range(k):
        c = float(rng.uniform(0.03, 0.97))
        w = float(rng.uniform(0.02, 0.25))
        amp = float(rng.uniform(0.5, 2.5))
        b += amp * np.exp(-0.5 * np.power((x - c) / w, 2.0))
    return np.maximum(b, 0.0)


def _init_bits_smooth(n, rng, tm):
    if tm is None:
        mask = (rng.random(n) < 0.5).astype(np.float64)
    else:
        if rng.random() < 0.7:
            mask = tm.copy()
        else:
            noisy = tm + rng.normal(0.0, 0.25, size=n)
            mask = (noisy > 0.5).astype(np.float64)
    low = float(rng.uniform(0.0, 0.35))
    high = float(rng.uniform(1.0, 2.2))
    b = low + (high - low) * mask
    w = int(rng.integers(1, max(2, n // 80) + 1))
    b = _box_smooth(b, n, w)
    return np.maximum(b, 0.0)


def _block_perturb(a, n, rng, sigma, width_frac):
    b = a.copy()
    w = max(2, int(n * float(width_frac)))
    if w >= n:
        b = b * np.exp(rng.normal(0.0, sigma, size=n))
    else:
        s = int(rng.integers(0, n - w + 1))
        b[s:s + w] = b[s:s + w] * np.exp(rng.normal(0.0, sigma, size=w))
    s = float(np.sum(b))
    if not np.isfinite(s) or s <= 0.0:
        b = np.ones(n, dtype=np.float64)
    b *= float(n) / s
    return np.maximum(b, 0.0)

            else:
                p = p_soft

            g = _grad_convolution_weights(a, p, m)
            gmax = float(np.max(np.abs(g)))
            if not np.isfinite(gmax) or gmax < 1e-15:
                break

            a_try = _mul_update(a, g, lr, 2.0)
            if a_try is a:
                break

            C_try = _conv_fft(a_try, m)
            o_try = _objective_from_C(C_try, n)
            accept = False
            if np.isfinite(o_try):
                _, jtry = _coeff_weights(C_try, tau)
                jtol = max(1e-9 * max(1.0, tau), 1e-12)
                accept = (o_try + 1e-13 < o) or (jtry + jtol < jcurr)

            if accept:
                a, C, o = a_try, C_try, o_try
                stale = 0
                lr = min(1.6, lr * 1.02)
                if o < best_o - 1e-12:
                    best = a.copy()
                    best_o = o
                    if (it & 1023) == 0:
                        print(f"[{tag}] hard it={it} o={o:.6f} best={best_o:.6f}")
            else:
                stale += 1
                lr = max(0.02, lr * 0.80)
                if stale > 50:
                    a = best.copy()
                    sigma = 0.06 + 0.02 * min(1.0, float(stale) / 120.0)
                    a = a * np.exp(rng.normal(0.0, sigma, size=n))
                    a = _simplex_normalize(a, n)
                    C = _conv_fft(a, m)
                    o = _objective_from_C(C, n)
                    if not np.isfinite(o) or o > best_o + 0.03:
                        a = best.copy()
                        C = _conv_fft(a, m)
                        o = best_o
                    stale = 0
                    lr = 0.25

            tau = max(tau_floor, tau * 0.9998)
            it += 1
            if it % 3000 == 0:
                print(f"[{tag}] hard it={it} o={o:.6f} best={best_o:.6f} tau={tau:.2f} lr={lr:.3f}")

    return best, best_o


def _postprocess_direct(a, n):
    base = _positive_regularize(a, n)
    best_a = base
    best_o = _objective_direct(base)

    thresholds = (0.0, 1e-10, 1e-6, 0.05)
    for thr in thresholds:
        b = base.copy()
        if thr > 0.0:
            b[b < thr] = 0.0
        s = float(np.sum(b))
        if not np.isfinite(s) or s <= 0.0:
            continue
        b = _simplex_normalize(b, n)
        o = _objective_direct(b)
        if np.isfinite(o) and o < best_o:
            best_o = o
            best_a = b

    return best_a, best_o


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-start log-space gradient search with soft/hard active-set refinement.
    Returns a nonnegative list optimized for the provided evaluator.
    """
    start = time.monotonic()
    deadline = start + max(30.0, float(budget_s) - 75.0)
    rng = np.random.default_rng(seed)

    # Larger lengths usually help, while still keeping direct final checks fast.
    if budget_s > 1200:
        n = 8192
    elif budget_s > 600:
        n = 4096
    else:
        n = 2048

    m = _next_power_of_two(2 * n - 1)

    candidates = []

    parent = _parent_resampled(n)
    if parent is not None:
        candidates.append(parent)
        candidates.append(_box_smooth(parent, n, max(3, n // 100)))
        candidates.append(parent + 0.08 * rng.standard_normal(size=n))
        candidates.append(parent + 0.20 * rng.standard_normal(size=n))

    candidates.append(_init_uniform(n))

    tm = _thue_bits(n)
    for amp in (0.35, 0.65, 0.90, 1.0):
        candidates.append(np.maximum(1.0 + amp * (2.0 * tm - 1.0), 0.0))
    for amp in (0.7, 0.95):
        candidates.append(np.maximum(1.0 + amp * (1.0 - 2.0 * tm), 0.0))

    for _ in range(7):
        freq = int(rng.integers(1, 5))
        amp = float(rng.uniform(0.55, 1.0))
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        power = float(rng.choice([0.5, 1.0, 1.5, 2.0]))
        candidates.append(_init_cos(n, freq, amp, phase, power))

    for _ in range(5):
        candidates.append(_init_random_smooth(n, rng))

    for _ in range(4):
        candidates.append(_init_gaussians(n, rng))

    for _ in range(5):
        candidates.append(_init_bits_smooth(n, rng, tm))

    x = np.linspace(-1.0, 1.0, n, endpoint=True)
    candidates.append(np.maximum(1.0 - x * x, 0.0))
    x0 = np.linspace(0.0, 1.0, n, endpoint=False)
    candidates.append(np.maximum(1.0 - np.abs(2.0 * x0 - 1.0), 0.0))
    candidates.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * x0), 0.0))

    scored = []
    for arr in candidates:
        if arr is None:
            continue
        a = _positive_regularize(arr, n)
        C = _conv_fft(a, m)
        o = _objective_from_C(C, n)
        if np.isfinite(o):
            scored.append((float(o), a))

    if not scored:
        a = np.ones(n, dtype=np.float64)
        return [float(x) for x in a]

    scored.sort(key=_score_pair_key)

    # Quick refinement of the most promising raw initializations.
    quick = scored[:10]
    quick_deadline = min(deadline - 120.0, start + max(60.0, 0.12 * max(60.0, float(budget_s) - 60.0)))
    if quick_deadline < start + 2.0:
        quick_deadline = start + 2.0

    quick_refined = []
    if quick:
        slice_quick = max(5.0, min(45.0, max(0.0, quick_deadline - start) / float(len(quick))))
        for qi, (qo, qa) in enumerate(quick):
            t_end = min(quick_deadline, time.monotonic() + slice_quick)
            if time.monotonic() >= quick_deadline - 1.0:
                break
            na, no = _refine(qa, n, m, t_end, f"q{qi}", rng)
            quick_refined.append((float(no), na))

    if quick_refined:
        quick_refined.sort(key=_score_pair_key)
        bank = [(qa, qo, f"q{i}") for i, (qo, qa) in enumerate(quick_refined[:6])]
    else:
        bank = [(qa, qo, f"raw{i}") for i, (qo, qa) in enumerate(scored[:6])]

    if not bank:
        qo, qa = scored[0]
        bank = [(qa, qo, "fallback")]

    # Main bank-based optimization: repeatedly allocate short refinement slices
    # to the current best settings and occasionally inject block perturbations.
    pass_no = 0
    while time.monotonic() < deadline - 60.0:
        bank.sort(key=_bank_key)
        if len(bank) > 6:
            bank = bank[:6]

        remaining = deadline - time.monotonic() - 35.0
        if remaining <= 0.0:
            break

        slice_s = min(240.0, max(15.0, remaining / (float(len(bank)) * 1.5)))
        if slice_s * float(len(bank)) > 0.95 * remaining:
            slice_s = max(10.0, 0.95 * remaining / float(len(bank)))

        for bi in range(len(bank)):
            a, o, tag = bank[bi]
            t_end = min(deadline - 25.0, time.monotonic() + slice_s)
            if time.monotonic() >= t_end:
                continue
            na, no = _refine(a, n, m, t_end, f"{tag}p{pass_no}", rng)
            if np.isfinite(no) and no < o:
                bank[bi] = (na, float(no), tag)
            else:
                bank[bi] = (a, float(o), tag)

        pass_no += 1

        if pass_no % 2 == 0 and time.monotonic() < deadline - 180.0:
            bank.sort(key=_bank_key)
            best_a = bank[0][0]
            for sigma, wf in ((0.04, 0.02), (0.12, 0.05), (0.25, 0.10), (0.50, 0.18)):
                pa = _block_perturb(best_a, n, rng, sigma, wf)
                pa = _positive_regularize(pa, n)
                pc = _conv_fft(pa, m)
                po = _objective_from_C(pc, n)
                if np.isfinite(po):
                    bank.append((pa, float(po), "pert"))
            bank = bank[:6]

    bank.sort(key=_bank_key)
    best_a = bank[0][0]
    best_o = bank[0][1]

    # Final dedicated refinement.
    if time.monotonic() < deadline - 15.0:
        best_a, best_o = _refine(best_a, n, m, deadline - 10.0, "final", rng)

    # Final direct-convolution polish and validation.
    best_a, direct_o = _postprocess_direct(best_a, n)
    print(f"direct objective before evaluate: {direct_o:.6f} n={n}")

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        eval_score = float(evaluate_sequence(result))
        print(f"evaluate_sequence score: {eval_score:.6f}")
        if not np.isfinite(eval_score):
            result = [1.0] * n
            print("fallback to uniform")
    except Exception:
        print(f"evaluate_sequence unavailable/failed; using direct objective {direct_o:.6f}")

    return result
```
