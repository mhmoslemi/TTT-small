Optimize step function to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses Adam with active-set LSE smoothing, adaptive parameters,
and exact subgradient polishing.
"""

import numpy as np
import time


def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def active_lse_objective(a, tau, active_indices):
    """Compute LSE-smoothed max over active indices only."""
    c = np.convolve(a, a)
    c_active = c[active_indices]
    c_max = c_active.max()
    shifted = c_active - c_max
    vals = np.exp(shifted / tau)
    return tau * np.log(vals.sum()) + c_max


def active_lse_grad_x(x, tau, active_indices):
    """Gradient of active-set LSE-smoothed max w.r.t. x (log-param)."""
    a = softmax(x)
    c = np.convolve(a, a)
    c_active = c[active_indices]
    c_max = c_active.max()
    w_active = np.exp((c_active - c_max) / tau)
    w_active /= w_active.sum()
    
    # Full weight vector (zero outside active set)
    n = len(a)
    w = np.zeros(2 * n - 1)
    w[active_indices] = w_active
    
    # grad_a = 2 * correlate(w, a, 'valid')
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    
    # Chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x


def exact_subgradient_step(a, n_active=50, step=0.01):
    """One exact subgradient step using true max of a*a."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    if max_c < 1e-15:
        return a
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

    best_a = a_init.copy()
    best_score = float('inf')
    deadline = time.monotonic() + max_time
    
    x = np.log(a_init + 1e-30)
    
    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Adaptive parameters
    K = 20  # number of active lags
    tau = 0.05 / n
    lr = 1.0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Initial active set
    c = np.convolve(a_init, a_init)
    active_indices = np.argsort(c)[::-1][:K]
    
    step_count = 0
    last_improve = time.monotonic()
    stall_count = 0
    
    while time.monotonic() < deadline:
        t += 1
        step_count += 1
        
        # Compute gradient
        grad_x = active_lse_grad_x(x, tau, active_indices)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Get current a
        a = softmax(x)
        
        # Every 100 steps, re-identify active set and check score
        if t % 100 == 0:
            c = np.convolve(a, a)
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve = time.monotonic()
                stall_count = 0
            else:
                stall_count += 1

            active_indices = np.argsort(c)[::-1][:K]

            if stall_count > 3:

                K = min(K + 5, 100)

                x += rng.normal(0, 0.01, n)

                # Switch to exact subgradient
                break
        
        # Time check

            break
    
    return best_a, best_score, x

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_gradient(x, tau, active_indices):
    """Compute gradient of active-set LSE objective w.r.t. x."""
    a = softmax(x)
    c = np.convolve(a, a)
    c_active = c[active_indices]
    c_max_active = c_active.max()
    w_active = np.exp((c_active - c_max_active) / tau)
    w_sum = w_active.sum()
    if w_sum < 1e-300:
        return a * (1.0 / len(a))  # uniform gradient (fallback)
    w_active /= w_sum
    
    n = len(a)
    w = np.zeros(2 * n - 1)
    w[active_indices] = w_active
    
    # grad_a[i] = 2 * sum_j w[i+j] * a[j]  (correlation)
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    
    # Chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x

    """Compute the score and autocorrelation."""
    c = np.convolve(a, a)
    n = len(a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf'), c
    score = 2.0 * n * c.max() / (sum_a ** 2)
    return score, c


def exact_subgradient_step(a, n_active, step):
    """One exact subgradient step."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    if max_c < 1e-15:
        return a
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

```python
def compute_gradient_active(x, tau, active_indices):
    """Fast gradient using only active lags."""
    a = softmax(x)
    n = len(a)
    c = np.convolve(a, a)
    
    c_active = c[active_indices]
    c_max = c_active.max()
    w_active = np.exp((c_active - c_max) / tau)
    w_sum = w_active.sum()
    w_active /= w_sum
    
    # Direct computation of grad_a without full correlation
    # grad_a[i] = 2 * sum_{k in active} w[k] * a[k-i] (for valid k-i)
    grad_a = np.zeros(n)
    for idx, k in enumerate(active_indices):
        w_k = w_active[idx]
        # a[k-i] is valid when 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad_a[lo:hi+1] += 2 * w_k * a[k-hi : k-lo+1][::-1]
        grad_a[lo:hi+1] += 2.0 * w_k * a[k-hi:k-lo+1][::-1]
    
    # Chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x
```

Let me think about option 4. The autocorrelation theorem says that the DFT of the autocorrelation is the power spectrum: `|DFT(a)|^2 = DFT(c)` where `c = autocorrelation(a)`.

Actually, that's for the circular autocorrelation. For the linear autocorrelation used here, the relationship is different.

But there's still a connection: if we zero-pad `a` to length `2n-1` and compute its autocorrelation, the DFT of the zero-padded `a` gives the power spectrum, and the inverse DFT of the power spectrum gives the circular autocorrelation of the zero-padded `a`, which equals the linear autocorrelation of `a`.

So: `c = ifft(|fft(a_padded)|^2)` where `a_padded` is `a` zero-padded to length `2n-1`.

This means we can compute the autocorrelation in O(n log n) using FFT! And the gradient can also be computed in O(n log n).

This is a significant speedup over O(n^2). Let me use this.

```python
def autocorr_fft(a):
    """Compute linear autocorrelation (full convolution a*a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    nfft = 1
    while nfft < size:
        nfft *= 2
    A = np.fft.rfft(a, nfft)
    c = np.fft.irfft(A * A, nfft)
    return c[:size]
```

And the gradient:
```python
def grad_autocorr_fft(a, w):
    """Compute grad_a = 2 * correlate(w, a, 'valid') using FFT.
    w has length 2n-1, a has length n, result has length n.
    """
    n = len(a)
    nfft = 1
    while nfft < len(w) + n - 1:
        nfft *= 2
    W = np.fft.rfft(w, nfft)
    A = np.fft.rfft(a, nfft)
    # correlate(w, a, 'full') = ifft(fft(w) * conj(fft(a)))
    corr = np.fft.irfft(W * np.conj(A), nfft)
    # 'valid' part starts at index n-1 and has length n
    return 2.0 * corr[n-1:n-1+n]
```
