```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

Algorithm:
- Use FFT-based convolution for very fast score/gradient evaluation.
- Maintain sequences normalized to sum 1 (objective is scale invariant).
- Use a dense softmax-active-set mirror-descent update on the non-negative
  simplex.  The weight given to convolution coefficients is a temperature
  annealed softmax centered at the current maximum coefficient.
- Multi-resolution search: refine promising sequences at several lengths,
  upsampling the current global best to longer lengths.
- Random smooth/blockshape starts and occasional reheating smooth perturbations
  are used to escape local minima.
- Final candidates are verified with the exact evaluator before returning.
"""

import numpy as np
import time
import math


def pair_score(item):
    """Sort key for (score, sequence) tuples."""
    return item[0]


def next_pow2(x):
    """Smallest power of two >= x (widgets x to integer logic)."""
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def to_float_list(arr):
    """Convert an array-like to a plain Python list of floats."""
    out = []
    for x in np.asarray(arr, dtype=np.float64).reshape(-1):
        out.append(float(x))
    return out


def prepare_local(a, target_sum=1.0):
    """
    Clean a sequence, clip to non-negative finite values, and normalize
    to a target sum.  A tiny floor is added to keep mirror-descent active.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.sum())
    else:
        a *= float(target_sum) / s

    a = np.clip(a, 0.0, 1000.0)

    if a.size > 0 and float(a.min()) < 1e-15:
        a = a + 1e-15
        s = float(a.sum())
        if s > 0.0:
            a *= float(target_sum) / s

    return a


def positive_normalize(a, target_sum=1.0):
    """Prepare and force strictly positive values."""
    a = prepare_local(a, target_sum)
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if s <= 0.0:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.sum())
    a *= float(target_sum) / s
    return a


def fft_transform_only(a):
    """Return (rfft, length) used for convolution of a with itself."""
    n = int(np.asarray(a).size)
    L = 2 * n - 1
    N = next_pow2(L)
    fa = np.fft.rfft(a, N)
    return fa, N


def fast_conv_square(a):
    """FFT-based linear convolution a*a, truncated to length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)

    L = 2 * n - 1
    N = next_pow2(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def fast_max_conv_square(a):
    """Return only max coefficient of a*a."""
    c = fast_conv_square(a)
    if c.size == 0:
        return 0.0
    return float(np.max(c))


def fft_square_with_transform(a):
    """Return (convolution, rfft array, FFT length) for a positive sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    L = 2 * n - 1
    N = next_pow2(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c, fa, N


def fast_score_array(a):
    """
    Fast analogue of evaluate_sequence(), assuming scale invariance and
    applying the same clipping behavior.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return float("inf")

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")

    maxc = fast_max_conv_square(a)
    if not np.isfinite(maxc):
        return float("inf")

    return float(2.0 * float(a.size) * float(maxc) / (s * s))


def smooth_direction(a, C, q, lam, fa, N):
    """
    Direction in log-probability space from a softmax-active-set gradient of
    max_k C_k.  Returns a normalized centered direction, or None if inert.
    """
    n = int(np.asarray(a).size)
    if n == 0 or q <= 0.0 or not np.isfinite(q):
        return None

    ratio = C / q
    z = float(lam) * (ratio - 1.0)
    z = np.where(np.isfinite(z), z, -50.0)
    z = np.clip(z, -50.0, 0.0)

    w = np.exp(z)
    wz = float(w.sum())
    if not np.isfinite(wz) or wz <= 1e-300:
        w = np.zeros_like(C, dtype=np.float64)
        w[int(np.argmax(C))] = 1.0
    else:
        w *= 1.0 / wz

    L = 2 * n - 1
    wr = np.zeros(N, dtype=np.float64)
    wr[:L] = w[::-1]

    fw = np.fft.rfft(wr, N)
    corr = np.fft.irfft(fa * fw, N)

    if n == 1:
        grad = np.array([2.0 * float(corr[0])], dtype=np.float64)
    else:
        grad = 2.0 * corr[2 * n - 2:n - 2:-1].copy()

    grad = np.where(np.isfinite(grad), grad, 0.0)
    np.maximum(grad, 0.0, out=grad)

    g = grad / float(q)
    g = g - float(np.mean(g))
    g = np.where(np.isfinite(g), g, 0.0)

    norm = float(np.max(np.abs(g)))
    if not np.isfinite(norm) or norm <= 1e-15:
        return None

    return g / norm


def multiplicative_update(a, g, eta):
    """
    Mirror-descent multiplicative update:
        p <- p * exp(-eta*g), then renormalize to sum 1.
    """
    v = np.log(np.maximum(a, 1e-300))
    v -= float(eta) * g
    v -= float(np.max(v))
    v = np.clip(v, -50.0, 0.0)
    p = np.exp(v)
    p += 1e-300
    p /= float(p.sum())
    return p


def smooth_perturbation(a, rng, strength=0.30):
    """
    Random low-frequency multiplicative perturbation, used for reheating.
    """
    a = positive_normalize(np.asarray(a, dtype=np.float64), 1.0)
    n = int(a.size)
    if n < 4:
        return a

    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)

    for m in range(1, 5):
        amp = strength / (0.75 + 0.25 * float(m))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(2.0 * np.pi * float(m) * t + ph)

    # Include skewing modes to encourage asymmetry.
    for m in range(1, 4):
        amp = strength * 0.5 / (0.5 + float(m))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(float(m) * np.pi * (2.0 * t - 1.0) + ph)

    a *= np.exp(z)

    if n > 8 and float(rng.random()) < 0.45:
        width = int(rng.integers(max(1, n // 150), max(2, n // 6 + 1)))
        width = min(width, n)
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        a[start:stop] *= np.exp(rng.normal(0.0, strength, stop - start))

    return positive_normalize(a, 1.0)


def refine_local_sequence(a, budget, seed):
    """
    Local mirror-descent refinement at a fixed length.
    Returns (best_sequence_sum1, best_fast_score).
    """
    a = positive_normalize(np.asarray(a, dtype=np.float64), 1.0)
    n = int(a.size)
    if n == 0:
        return np.ones(1, dtype=np.float64), float("inf")

    C, fa, N = fft_square_with_transform(a)
    q = float(np.max(C))
    score = float(2.0 * float(n) * q) if q > 0.0 else float("inf")

    best = a.copy()
    best_score = score

    t_start = time.monotonic()
    budget = max(0.0, float(budget))
    t_end = t_start + budget

    if budget < 0.5 or score < 0.0:
        return best, best_score

    rng = np.random.default_rng(int(seed))

    lam = 20.0
    eta = 0.12
    last_accept = t_start
    no_improve_limit = max(8.0, min(150.0, 0.10 * budget))
    stagnant = 0
    it = 0

    while time.monotonic() < t_end:
        it += 1

        if (not np.isfinite(score)) or (not np.isfinite(q)) or q <= 0.0:
            break

        g = smooth_direction(a, C, q, lam, fa, N)
        if g is None:
            if it > 10:
                break
            a = smooth_perturbation(best, rng, 0.25)
            C, fa, N = fft_square_with_transform(a)
            q = float(np.max(C))
            score = float(2.0 * float(n) * q)
            continue

        accepted = False
        step = float(eta)
        first_try = True

        for _ in range(5):
            cand = multiplicative_update(a, g, step)
            cand_C = fast_conv_square(cand)
            q2 = float(np.max(cand_C))
            sc2 = float(2.0 * float(n) * q2)

            if np.isfinite(sc2) and sc2 < score - 1e-12:
                a = cand
                C = cand_C
                q = q2
                score = sc2
                fa, N = fft_transform_only(a)

                if score < best_score - 1e-12:
                    best = a.copy()
                    best_score = score

                last_accept = time.monotonic()
                accepted = True
                break

            step *= 0.5
            first_try = False
            if step < 1e-5:
                break

        if accepted:
            stagnant = 0
            if first_try:
                eta = min(0.80, eta * 1.05 + 0.001)
            else:
                eta = max(0.003, step * 1.05)

            if lam < 2500.0:
                lam = min(2500.0, lam * 1.015 + 0.05)
        else:
            stagnant += 1
            eta = max(0.002, eta * 0.68)

            if stagnant == 24:
                lam = max(6.0, lam * 0.72)

            if stagnant >= 60:
                a = smooth_perturbation(best, rng, 0.30 + 0.25 * float(rng.random()))
                C, fa, N = fft_square_with_transform(a)
                q = float(np.max(C))
                score = float(2.0 * float(n) * q)

                if (not np.isfinite(score)) or score > best_score * 1.6:
                    a = best.copy()
                    C, fa, N = fft_square_with_transform(a)
                    q = float(np.max(C))
                    score = float(2.0 * float(n) * q)

                eta = 0.10
                lam = max(10.0, lam * 0.85)
                stagnant = 0

        if (it % 1000) == 0:
            print(
                "  refine n=%d lam=%.2f eta=%.4f score=%.10f best=%.10f"
                % (n, lam, eta, score, best_score),
                flush=True,
            )

        if time.monotonic() - last_accept > no_improve_limit:
            break

    best = positive_normalize(best, 1.0)
    return best, best_score


def upsample_sequence(seq, n):
    """Linear interpolation to length n, normalized to sum 1."""
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    n = int(n)

    if n <= 0:
        return np.ones(1, dtype=np.float64)
    if seq.size == 0:
        return np.ones(n, dtype=np.float64)
    if seq.size == 1:
        return np.full(n, float(seq[0]), dtype=np.float64)

    xold = np.linspace(0.0, 1.0, seq.size, dtype=np.float64)
    xnew = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(xnew, xold, seq)
    return positive_normalize(out, 1.0)


def start_uniform(n):
    """Uniform start."""
    return np.ones(int(n), dtype=np.float64)


def start_exponential(n, alpha):
    """Exponential tilt."""
    n = int(n)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    return np.exp(float(alpha) * x)


def start_quad_exp(n, alpha, beta):
    """Quadratic exponential tilt."""
    n = int(n)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    z = float(alpha) * x + float(beta) * x * (x - 1.0)
    z = np.clip(z, -50.0, 50.0)
    return np.exp(z)


def start_power(n, alpha):
    """Power-law edge weighting."""
    n = int(n)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x = np.maximum(x, 1e-6)
    return np.power(x, float(alpha))


def start_poly(n, alpha, beta):
    """Low-degree polynomial shape."""
    n = int(n)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    u = 2.0 * x - 1.0
    vals = 1.0 + float(alpha) * u + float(beta) * (u * u - 1.0 / 3.0)
    return np.maximum(vals, 1e-4)


def start_cos(n, alpha, phase):
    """Cosine shape with asymmetry."""
    n = int(n)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    vals = 1.0 + float(alpha) * np.cos(np.pi * (2.0 * x - 1.0) + float(phase))
    return np.maximum(vals, 1e-4)


def start_two_block(n, frac, ratio):
    """Two-level block start."""
    n = int(n)
    m = int(round(float(frac) * float(n)))
    if m < 1:
        m = 1
    if m >= n:
        m = n - 1

    a = np.empty(n, dtype=np.float64)
    a[:m] = float(ratio)
    a[m:] = 1.0
    return a


def start_random_fourier(n, rng):
    """Random low-frequency smooth sequence."""
    n = int(n)
    if n < 2:
        return np.ones(n, dtype=np.float64)

    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    z = np.zeros(n, dtype=np.float64)

    for m in range(1, 5):
        amp = rng.uniform(0.08, 1.20) / math.sqrt(float(m))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(2.0 * np.pi * float(m) * x + ph)

    for m in range(1, 4):
        amp = rng.uniform(0.0, 0.90) / math.sqrt(float(m))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(float(m) * np.pi * (2.0 * x - 1.0) + ph)

    vals = 1.0 + z
    vals = np.maximum(vals, 0.0)
    if float(vals.max()) <= 0.0:
        vals = np.ones(n, dtype=np.float64)
    return vals


def start_random_blocks(n, rng):
    """Random block-shaped sequence."""
    n = int(n)
    if n < 4:
        return np.ones(n, dtype=np.float64)

    blocks = int(rng.integers(5, 25))
    blocks = int(np.clip(blocks, 2, max(2, n // 2)))

    cuts = np.sort(rng.integers(1, n, blocks - 1))
    edges = np.unique(np.concatenate((np.array([0], dtype=int), cuts, np.array([n], dtype=int))))
    lengths = np.diff(edges)

    if lengths.size < 1 or float(np.sum(lengths)) <= 0:
        return np.ones(n, dtype=np.float64)

    vals = rng.uniform(0.05, 3.0, lengths.size)
    if float(rng.random()) < 0.2:
        vals[int(rng.integers(0, lengths.size))] = 0.0

    return np.repeat(vals, lengths)


def add_start_to_list(starts, arr):
    """Evaluate and append a candidate start if finite."""
    try:
        a = positive_normalize(arr, 1.0)
        sc = fast_score_array(a)
        if np.isfinite(sc):
            starts.append((sc, a))
    except Exception:
        pass


def generate_starts(n, rng, parent, random_count=35):
    """Generate a diverse pool of start sequences of length n."""
    n = int(n)
    starts = []

    add_start_to_list(starts, start_uniform(n))

    for alpha in (-4.0, -2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0, 4.0):
        add_start_to_list(starts, start_exponential(n, alpha))

    for alpha in (-2.0, -1.0, -0.5, -0.25, 0.25, 0.5, 1.0, 2.0, 3.0):
        add_start_to_list(starts, start_power(n, alpha))

    for alpha in (-3.0, -1.5, -0.8, 0.0, 0.8, 1.5, 3.0):
        for beta in (-3.0, -1.2, 0.0, 1.2, 3.0):
            add_start_to_list(starts, start_quad_exp(n, alpha, beta))

    for alpha in (-0.8, -0.5, -0.25, 0.25, 0.5, 0.8):
        for beta in (-0.35, 0.0, 0.35):
            add_start_to_list(starts, start_poly(n, alpha, beta))

    for alpha in (-0.7, -0.4, -0.2, 0.2, 0.4, 0.7):
        for phase in (0.0, 0.45, 0.9):
            add_start_to_list(starts, start_cos(n, alpha, phase))

    for frac in (0.15, 0.25, 0.40):
        for ratio in (0.25, 0.5, 2.0, 4.0):
            add_start_to_list(starts, start_two_block(n, frac, ratio))
            add_start_to_list(starts, start_two_block(n, 1.0 - frac, 1.0 / ratio))

    if parent is not None:
        try:
            add_start_to_list(starts, upsample_sequence(parent, n))
        except Exception:
            pass

    for _ in range(max(0, int(random_count))):
        try:
            if float(rng.random()) < 0.55:
                add_start_to_list(starts, start_random_fourier(n, rng))
            else:
                add_start_to_list(starts, start_random_blocks(n, rng))
        except Exception:
            pass

    if not starts:
        a = positive_normalize(start_uniform(n), 1.0)
        starts.append((fast_score_array(a), a))

    starts.sort(key=pair_score)
    return starts


def add_to_pool(pool, seq, score, max_size=25):
    """Maintain a small sorted pool of candidates normalized to sum 1."""
    try:
        sc = float(score)
    except Exception:
        return
    if not np.isfinite(sc):
        return

    try:
        arr = positive_normalize(seq, 1.0)
    except Exception:
        return

    if arr.size == 0:
        return

    pool.append((sc, arr.copy()))
    pool.sort(key=pair_score)
    if len(pool) > int(max_size):
        pool.pop()


def sanitize_for_return(seq):
    """
    Return a valid Python list representing a non-negative finite sequence.
    The sequence is normalized to sum 1, which is safe and score-equivalent.
    """
    try:
        arr = positive_normalize(seq, 1.0)
    except Exception:
        return [1.0]

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    s = float(arr.sum())
    if s < 0.01:
        return [1.0]

    arr *= 1.0 / s

    return to_float_list(arr)


def choose_initial_length(parent_arr, budget):
    """Choose the first length to refine."""
    if parent_arr is not None:
        p = int(parent_arr.size)
        if 128 <= p <= 4096:
            return p
    if budget < 700:
        return 512
    return 1024


def choose_max_length(budget):
    """Choose the largest length considered."""
    if budget < 500:
        return 1024
    if budget < 1300:
        return 2048
    return 4096


def length_weights(lengths):
    """Allocate more time to longer sequences."""
    m = len(lengths)
    if m <= 1:
        return [1.0]
    if m == 2:
        return [1.0, 1.8]
    if m == 3:
        return [1.0, 1.5, 2.8]
    if m == 4:
        return [1.0, 1.3, 1.9, 3.0]
    return [1.0, 1.2, 1.6, 2.4, 3.4, 4.6][:m]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point.  Runs multi-resolution mirror-descent refinement
    and returns the best valid sequence found before the time limit.
    """
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 2400.0

    if budget < 1.0:
        return [1.0]

    # Leave room for exact validation and final cleanup.
    hard_deadline = t0 + max(5.0, budget - 5.0)
    search_deadline = t0 + max(10.0, budget - 45.0)

    # Use externally provided previously best construction if available.
    parent = None
    try:
        parent = height_sequence_1
    except Exception:
        parent = None

    parent_arr = None
    if parent is not None:
        try:
            tmp = positive_normalize(np.asarray(parent, dtype=np.float64).reshape(-1), 1.0)
            if tmp.size >= 8:
                parent_arr = tmp
        except Exception:
            parent_arr = None

    base_n = choose_initial_length(parent_arr, budget)
    max_len = choose_max_length(budget)
    max_len = max(max_len, min(base_n, 4096))

    if base_n > max_len:
        base_n = max_len

    # Build a small set of resolutions to optimize.
    lengths = []
    cur = int(base_n)
    while True:
        cur = int(min(cur, max_len))
        if cur not in lengths:
            lengths.append(cur)
        if cur >= max_len:
            break
        nxt = int(min(max_len, max(cur * 2, cur + 256)))
        if nxt <= cur:
            nxt = cur + 256
        cur = nxt

    if not lengths:
        lengths = [512]

    weights = length_weights(lengths)
    remaining_weights = float(sum(weights))

    pool = []

    # Initial pool: parent if available, and uniform at base length.
    if parent_arr is not None:
        add_to_pool(pool, parent_arr, fast_score_array(parent_arr), 25)
    add_to_pool(pool, start_uniform(lengths[0]), fast_score_array(start_uniform(lengths[0])), 25)

    if not pool:
        pool.append((2.0, positive_normalize(start_uniform(lengths[0]), 1.0)))

    prev_best = pool[0][1].copy()
    prev_best_score = float(pool[0][0])

    print("start base_n=%d max_len=%d budget=%.1f initial_best=%.10f"
          % (base_n, max_len, budget, prev_best_score), flush=True)

    for idx, n in enumerate(lengths):
        if time.monotonic() >= search_deadline - 25.0:
            break

        if remaining_weights <= 1e-12:
            share = 1.0
        else:
            share = float(weights[idx]) / remaining_weights
        remaining_weights -= float(weights[idx])

        elapsed_already_at_level = 0.0
        level_start = time.monotonic()

        max_level_budget = max(0.0, search_deadline - time.monotonic() - 30.0)
        level_budget = max(20.0, share * max_level_budget)
        level_budget = min(level_budget, max_level_budget)

        rng = np.random.default_rng(int(seed) + 10000 * idx + 7)

        starts = generate_starts(n, rng, prev_best, random_count=45 if n <= 2048 else 60)

        # Add top pool candidates filtered/upsampled to this length.
        for ps, seq in pool[:8]:
            try:
                if int(seq.size) == int(n):
                    add_start_to_list(starts, seq)
                else:
                    add_start_to_list(starts, upsample_sequence(seq, n))
            except Exception:
                pass

        starts.sort(key=pair_score)

        exploration_budget = min(0.45 * level_budget, 350.0)
        if exploration_budget < 25.0:
            exploration_budget = 0.5 * level_budget
        exploration_budget = min(exploration_budget, max(0.0, level_budget - 25.0))

        if exploration_budget >= 70.0:
            k = 2
        elif exploration_budget >= 30.0:
            k = 1
        else:
            k = 0

        k = max(0, min(k, len(starts)))
        if k > 0:
            time_each = exploration_budget / float(k)

            for i in range(k):
                if time.monotonic() >= search_deadline - 35.0:
                    break

                sc, sq = starts[i]

                # Refine only starts that are already competitive or modestly worse;
                # avoid spending full time on obviously poor shapes.
                if sc < prev_best_score * 1.08 + 1e-6:
                    time_this = max(
                        5.0,
                        min(time_each, search_deadline - time.monotonic() - 45.0)
                    )
                    if time_this > 3.0:
                        try:
                            seq, s = refine_local_sequence(
                                sq,
                                time_this,
                                seed=int(seed) + 1000000 + 1000 * idx + i
                            )
                        except Exception:
                            seq, s = None, float("inf")

                        if seq is not None and np.isfinite(s):
                            add_to_pool(pool, seq, s, 25)
                            if s < prev_best_score:
                                prev_best = seq.copy()
                                prev_best_score = float(s)
                                print("level %d n=%d improved to %.10f"
                                      % (idx, n, prev_best_score), flush=True)

                add_to_pool(pool, sq, sc, 25)

        # Polish the best available sequence at this resolution.
        if time.monotonic() < search_deadline - 45.0:
            best_at_n = None
            best_at_n_score = float("inf")

            for ps, seq in pool:
                if int(seq.size) == int(n) and float(ps) < best_at_n_score:
                    best_at_n = seq
                    best_at_n_score = float(ps)

            if best_at_n is None:
                try:
                    best_at_n = upsample_sequence(prev_best, n)
                except Exception:
                    best_at_n = start_uniform(n)
                best_at_n_score = fast_score_array(best_at_n)

            add_to_pool(pool, best_at_n, best_at_n_score, 25)

            level_elapsed = time.monotonic() - level_start
            polish_budget = max(0.0, level_budget - level_elapsed)
            if polish_budget > 25.0:
                polish_budget = min(polish_budget, search_deadline - time.monotonic() - 45.0)
                if polish_budget > 25.0:
                    try:
                        seq, s = refine_local_sequence(
                            best_at_n,
                            polish_budget,
                            seed=int(seed) + 2000000 + 1000 * idx + 555
                        )
                    except Exception:
                        seq, s = None, float("inf")

                    if seq is not None and np.isfinite(s):
                        add_to_pool(pool, seq, s, 25)
                        if s < prev_best_score:
                            prev_best = seq.copy()
                            prev_best_score = float(s)
                            print("polish level %d n=%d improved to %.10f"
                                  % (idx, n, prev_best_score), flush=True)

    # Final exact verification of a few top fast-score candidates.
    pool.sort(key=pair_score)

    exact_best_score = float("inf")
    exact_best_arr = None
    checked = 0

    max_checked = 6
    for i in range(min(len(pool), max_checked)):
        if time.monotonic() > hard_deadline - 8.0:
            break

        fs, seq = pool[i]

        # Skip very poor candidates if we already have a good exact value.
        if np.isfinite(exact_best_score) and float(fs) > exact_best_score * 1.02 + 1e-6:
            break

        try:
            cand_list = sanitize_for_return(seq)
            ex = float(evaluate_sequence(cand_list))
        except Exception:
            ex = float("inf")

        if np.isfinite(ex) and ex < exact_best_score:
            exact_best_score = ex
            exact_best_arr = np.asarray(cand_list, dtype=np.float64).reshape(-1)

        checked += 1

    if exact_best_arr is None or not np.isfinite(exact_best_score):
        exact_best_arr = positive_normalize(prev_best, 1.0)
        exact_best_score = fast_score_array(exact_best_arr)

    print("final fast_best=%.10f exact_best=%.10f checked=%d"
          % (prev_best_score, exact_best_score, checked), flush=True)

    return sanitize_for_return(exact_best_arr)
```
