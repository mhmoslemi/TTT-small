"""
Optimize step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LSE-smoothed Adam optimization, active-set subgradient polishing,
momentum-based projected gradients, and diverse structured initializations.
Exploits non-symmetric structures to beat the symmetric lower bound of 2.
"""

import numpy as np
import time


def compute_convolution(a):
    """Compute full self-convolution."""
    return np.convolve(a, a)


def lse_weights(conv_vals, tau):
    """Compute softmax weights for LSE approximation of max."""
    m = conv_vals.max()
    w = np.exp((conv_vals - m) / tau)
    s = w.sum()
    if s < 1e-300:
        return np.full(len(conv_vals), 1.0 / len(conv_vals))
    return w / s


def lse_gradient(a, tau):
    """
    Gradient of LSE-smoothed max(conv(a,a)) w.r.t. a.
    conv[k] = sum_j a[j]*a[k-j]
    d(conv[k])/d(a[i]) = 2*a[k-i] for valid indices
    LSE grad[i] = sum_k w_k * 2*a[k-i] = 2 * sum_j w[j+i]*a[j]
    """
    conv = compute_convolution(a)
    w = lse_weights(conv, tau)
    n = len(a)
    # cross-correlation: sum_j w[j+i] * a[j] for i = 0..n-1
    full_corr = np.correlate(w, a, mode='full')
    return 2.0 * full_corr[:n], conv


def project_to_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(len(v)) / len(v)
    return v / s


def subgradient_from_active(a, active_indices):
    """Compute averaged subgradient from active set of convolution peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(active_indices)
    return grad


def run_lse_adam(a0, max_time, initial_tau, n, rng):
    """Adam optimizer on LSE-smoothed max convolution objective."""
    a = a0.copy()
    m_ema = np.zeros(n)
    v_ema = np.zeros(n)
    t = 0
    lr = 0.05
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    tau = initial_tau
    best_local = a.copy()
    best_local_score = evaluate_sequence(a.tolist())
    t_start = time.monotonic()
    no_improve = 0
    eval_count = 0

    while time.monotonic() < t_start + max_time:
        grad, conv = lse_gradient(a, tau)

        t += 1
        m_ema = beta1 * m_ema + (1 - beta1) * grad
        v_ema = beta2 * v_ema + (1 - beta2) * (grad * grad)
        m_hat = m_ema / (1 - beta1 ** t)
        v_hat = v_ema / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a_new)

        # Decrease tau over time for annealing
        tau = max(initial_tau * 0.999 ** t, initial_tau * 0.005)

        eval_count += 1
        if eval_count % 30 == 0:
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_local_score:
                best_local_score = score
                best_local = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= 15:
                lr *= 0.6
                no_improve = 0
                if lr < 1e-5:
                    break

    return best_local, best_local_score


def run_subgradient_polish(a0, max_time, n, rng):
    """Active-set subgradient with momentum and adaptive step."""
    a = a0.copy()
    momentum = np.zeros(n)
    beta_m = 0.85
    lr = 0.08
    best_local = a.copy()
    best_local_score = evaluate_sequence(a.tolist())
    t_start = time.monotonic()
    no_improve = 0
    round_num = 0

    while time.monotonic() < t_start + max_time:
        conv = compute_convolution(a)
        # Active set: top K peaks
        K = min(200, len(conv))
        sorted_idx = np.argpartition(conv, -K)[-K:]

        grad = subgradient_from_active(a, sorted_idx)
        momentum = beta_m * momentum + grad
        a_new = a - lr * momentum
        a = project_to_simplex(a_new)

        round_num += 1
        if round_num % 20 == 0:
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_local_score:
                best_local_score = score
                best_local = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= 8:
                lr *= 0.7
                no_improve = 0
                if lr < 0.0005:
                    lr = 0.03
                    a = a + rng.normal(0, 0.003, n)
                    a = project_to_simplex(a)

    return best_local, best_local_score


def run_single_max_targeting(a0, max_time, n, rng):
    """Target only the single maximum convolution peak with small steps."""
    a = a0.copy()
    step = 0.01
    best_local = a.copy()
    best_local_score = evaluate_sequence(a.tolist())
    t_start = time.monotonic()
    no_improve = 0

    while time.monotonic() < t_start + max_time:
        conv = compute_convolution(a)
        max_idx = int(np.argmax(conv))
        lo = max(0, max_idx - n + 1)
        hi = min(n - 1, max_idx)
        grad_seg = 2.0 * a[max_idx - hi:max_idx - lo + 1][::-1]

        a_new = a.copy()
        a_new[lo:hi + 1] -= step * grad_seg
        a_new = project_to_simplex(a_new)

        score = evaluate_sequence(a_new.tolist())
        if np.isfinite(score) and score < best_local_score:
            best_local_score = score
            best_local = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 10:
            step *= 0.7
            no_improve = 0
            if step < 0.0001:
                step = 0.005
                a = a + rng.normal(0, 0.001, n)
                a = project_to_simplex(a)

    return best_local, best_local_score


def generate_structured_init(n, rng, variant):
    """Generate diverse structured initializations."""
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)

    if variant == 0:
        return np.ones(n) / n
    elif variant == 1:
        return rng.dirichlet(np.full(n, 0.3))
    elif variant == 2:
        return rng.dirichlet(np.full(n, 1.0))
    elif variant == 3:
        return rng.dirichlet(np.full(n, 3.0))
    elif variant == 4:
        x = 1.0 + 0.6 * np.cos(2 * t)
        x = np.maximum(x, 0.01)
        return x / x.sum()
    elif variant == 5:
        x = 1.0 + 0.6 * np.cos(5 * t) + 0.3 * np.cos(13 * t)
        x = np.maximum(x, 0.01)
        return x / x.sum()
    elif variant == 6:
        # Asymmetric ramp
        x = np.linspace(3.0, 0.2, n)
        return x / x.sum()
    elif variant == 7:
        # Exponential decay
        x = np.exp(-2.0 * np.linspace(0, 1, n))
        return x / x.sum()
    elif variant == 8:
        # Two peaks
        mid = n // 2
        peak_w = n // 8
        x = np.ones(n) * 0.3
        x[mid - peak_w:mid + peak_w] = 1.5
        x[0:peak_w] = 1.2
        return x / x.sum()
    elif variant == 9:
        # Random Fourier
        n_freq = 7
        freqs = rng.uniform(1, 15, n_freq)
        phases = rng.uniform(0, 2 * np.pi, n_freq)
        amps = rng.uniform(0.1, 0.5, n_freq)
        x = np.ones(n)
        for f, p, a in zip(freqs, phases, amps):
            x += a * np.cos(f * t + p)
        x = np.maximum(x, 0.01)
        return x / x.sum()
    elif variant == 10:
        # Sparse blocks
        n_blocks = rng.integers(3, 8)
        x = np.zeros(n)
        for _ in range(n_blocks):
            start = rng.integers(0, n - 10)
            width = rng.integers(5, n // 4)
            height = rng.uniform(0.5, 2.0)
            x[start:start + width] += height
        x = np.maximum(x, 0.01)
        return x / x.sum()
    elif variant == 11:
        # Wedge (peaked at one edge)
        x = np.linspace(0.1, 2.0, n)
        x += 0.3 * np.cos(3 * t)
        x = np.maximum(x, 0.01)
        return x / x.sum()
    elif variant == 12:
        # Anti-wedge
        x = np.linspace(1.5, 0.1, n)
        x += 0.3 * np.sin(7 * t)
        x = np.maximum(x, 0.01)
        return x / x.sum()
    elif variant == 13:
        # Periodic with random phase
        period = rng.integers(7, 23)
        pattern = rng.uniform(0.3, 2.0, period)
        x = np.tile(pattern, n // period + 1)[:n]
        return x / x.sum()
    elif variant == 14:
        # Gaussian window
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        i = np.arange(n)
        x = np.exp(-0.5 * ((i - center) / width) ** 2)
        x += 0.05
        return x / x.sum()
    else:
        x = rng.random(n)
        return x / x.sum()
