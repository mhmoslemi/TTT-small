def solve_lp(a, M, max_time=120):
    """Solve LP: max sum(b) s.t. b >= 0, convolve(a,b) <= M"""
    n = len(a)
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: A @ b <= M (2n-1 constraints), b >= 0 (added via bounds)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def peak_shave(a, n_shave=20, n_active=5, n_quiet=5):
    """Reduce the peak of convolve(a,a) by redistributing mass."""
    n = len(a)
    c = np.convolve(a, a)
    
    # Find top peaks
    top_peaks = np.argsort(c)[::-1][:n_active]
    
    # Find quiet positions (lowest convolution values)
    # But be careful: reducing a[i] affects multiple convolution positions
    # Let's just find positions with low a[i] that are not in the peak-contributing set
    
    # For peak k, contributing pairs are (i, k-i) for valid i
    peak_indices = set()
    for k in top_peaks:
        j_lo = max(0, k - n + 1)
        j_hi = min(n-1, k)
        for j in range(j_lo, j_hi+1):
            peak_indices.add(j)
        # Also k-j for j in range (the other element)
        for j in range(j_lo, j_hi+1):
            peak_indices.add(k - j)
    
    # Find quiet positions (not in peak_indices, low a values)
    quiet_candidates = [i for i in range(n) if i not in peak_indices]
    if not quiet_candidates:
        quiet_candidates = list(range(n))
    
    # Redistribute
    a_new = a.copy()
    total_removed = 0
    
    for k in top_peaks:
        j_lo = max(0, k - n + 1)
        j_hi = min(n-1, k)
        # Contribution of each pair
        contributions = []
        for j in range(j_lo, j_hi+1):
            other = k - j
            contributions.append((a[j] * a[other], j, other))
        contributions.sort(reverse=True)
        
        # Remove from the top contributors
        removal = 0.01 * a.sum() / len(top_peaks)  # remove 1% of total mass per peak
        for contrib, j, other in contributions[:n_shave]:
            if removal <= 0:
                break
            # Remove proportionally from j and other
            frac = min(removal / (2 * contrib) if contrib > 0 else 0, 0.1)
            a_new[j] -= a[j] * frac
            if other != j:
                a_new[other] -= a[other] * frac
            removal -= 2 * contrib * frac
            total_removed += 2 * contrib * frac
    
    a_new = np.maximum(a_new, 0)
    
    # Add removed mass to quiet positions
    if total_removed > 0 and len(quiet_candidates) > 0:
        # Add to the n_quiet quietest positions
        quiet_sorted = sorted(quiet_candidates, key=lambda i: a_new[i])
        add_each = total_removed / n_quiet
        for i in quiet_sorted[:n_quiet]:
            a_new[i] += add_each
    
    a_new = a_new / a_new.sum()
    return a_new

Optimize step function to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Combines FFT-based convolution, LP projection iterations, subgradient
polishing, peak shaving, and simulated annealing for global search.
"""

import numpy as np
import time

def fft_conv_full(a, b=None, L=None):
    """Compute full linear convolution a*b (or a*a if b is None) via FFT."""
    if b is None:
        b = a
    n = len(a)
    if L is None:
        L = 1 << max(n.bit_length() - 1, 1) * 2  # power of 2 >= 2n
        while L < 2*n - 1:
            L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:2*n-1]

def quick_score(a):
    """Compute score without calling evaluate_sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fft_conv_full(a)
    return 2.0 * n * c.max() / (s * s)

def subgradient_step(a, step_size, n_active=10, L=None):
    """One subgradient step to reduce max of convolve(a,a)."""
    n = len(a)
    c = fft_conv_full(a, L=L)
    max_c = c.max()
    if max_c < 1e-15:
        return a

    sorted_idx = np.argsort(c[::max(1, len(c)//n_active)][1 - len(c)//n_active::max(1, len(c)//n_active)])  # This is getting complicated

        active = top_k

        g[j_lo:j_hi+1] += 2.0 * a[k - j_hi:k - j_lo + 1][::-1]
    g /= len(active)

    a_new = a - step_size * g / max_c
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new /= s

    Uses iterative constraint projection (Gauss-Seidel style)."""

        L = 1 << (2*n - 1).bit_length()

    b = np.ones(n)  # Start with all ones (maximize sum)
    
    for it in range(n_iters):

            current = np.dot(coeffs, b[j_lo:j_hi+1])
            if current > M:
                scale = M / current * 0.999
                b[j_lo:j_hi+1] *= scale
                b[j_lo:j_hi+1] = np.maximum(b[j_lo:j_hi+1], 0)
        
        # Periodic check
        if it % 20 == 19:
            max_cb = cb.max() if 'cb' in dir() else 0
    
    return b

def peak_shave(a, intensity=0.05, L=None):
    """Reduce peak by moving mass from peak-contributing positions to quiet ones."""
    n = len(a)
    c = fft_conv_full(a, L=L)
    max_c = c.max()
    
    if max_c < 1e-15:
        return a
    
    # Find the single peak position
    k_star = c.argmax()
    
    # Find contributing pairs for this peak
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    
    # Find positions that are NOT contributing significantly to any top-5 peak
    top_peaks = np.argsort(c)[::-1][:5]
    in_peak = np.zeros(n, dtype=bool)
    for k in top_peaks:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        in_peak[jl:jh+1] = True
        in_peak[k-jh:k-jl+1] = True
    
    # Positions NOT in any peak
    quiet = np.where(~in_peak)[0]
    if len(quiet) == 0:
        quiet = np.where(a < np.median(a))[0]
    if len(quiet) == 0:
        return a
    
    # Remove mass from peak-contributing positions
    a_new = a.copy()
    total_remove = intensity * a.sum()
    
    # Remove from the positions with highest a[j]*a[k-j] at the peak
    contributions = np.array([a[j] * a[k_star - j] for j in range(j_lo, j_hi+1)])
    js = np.arange(j_lo, j_hi+1)
    top_contrib = np.argsort(contributions)[::-1][:min(20, len(js))]
    
    for idx in top_contrib:
        j = js[idx]
        other = k_star - j
        rem = min(total_remove / len(top_contrib) * 2, 0.1 * a[j])
        a_new[j] -= rem
        if other != j:
            a_new[other] -= rem
        total_remove -= 2 * rem
        if total_remove <= 0:
            break
    
    a_new = np.maximum(a_new, 0)
    
    # Add to quiet positions
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s  # Renormalize (this adds mass back)
    
    return a_new
