```python
"""
Search for low-peak nonnegative self-convolutions.

The score is
    2 * n * max_k (a*a)_k / (sum a)^2,
with a_i >= 0.  The search is scale invariant, so all internal sequences are
normalized to sum 1.

Main method:
  - Many cheap parametric/smooth starts, resampling of the supplied sequence,
    and resampling of the best sequence found so far.
  - A strong descent operator: for fixed x, solve the LP
        max   sum(y)
        s.t. (x*y)_k <= max(x*x),  y >= 0.
    The optimal y is the best-response direction.  Then perform an exact
    one-dimensional line search over (1-t)x + t y; the objective is convex
    along this line because every convolution coefficient is a convex quadratic
    in t.
  - Periodic mirror-descent polishing using a log-sum-exp approximation of the
    max convolution peak.
"""

import time
import math
import numpy as np

try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    HAS_SCIPY_LP = True
except Exception:
    HAS_SCIPY_LP = False

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")

MAX_LP_N = 1024
LP_CACHE = {}


def _next_pow2(n):
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _clean_prob(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def _autocorr_fft(a, fft_size):
    a = np.asarray(a, dtype=np.float64)
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * a.size - 1]


def _convolve_fft(a, b, fft_size):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    out = np.fft.irfft(A * B, fft_size)
    return out[:a.size + b.size - 1]


def _score_fft(p, fft_size=None):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return 1e100, np.array([], dtype=np.float64)
    if p.size == 1:
        return 2.0, np.array([1.0], dtype=np.float64)
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, fft_size)
    b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    mb = float(b.max())
    if not np.isfinite(mb):
        return 1e100, b
    return 2.0 * float(n) * mb, b


def _score_direct(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return 1e100
    if n == 1:
        return 2.0
    b = np.convolve(p, p)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _make_uniform(n):
    n = max(int(n), 2)
    return np.full(n, 1.0 / float(n), dtype=np.float64)


def _make_exp(n, alpha, reverse=False):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    if reverse:
        a = a[::-1]
    a = a + 1e-12
    return _clean_prob(a)


def _make_poly_power(n, p, q):
    n = max(int(n), 2)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    if p > 0.0:
        a = a * np.power(np.clip(t, 1e-9, 1.0), float(p))
    if q > 0.0:
        a = a * np.power(np.clip(1.0 - t, 1e-9, 1.0), float(q))
    a = a + 1e-12
    return _clean_prob(a)


def _make_mixture_exp(n, alpha, gamma):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x) + float(gamma) * np.exp(-float(alpha) * (1.0 - x))
    a = a + 0.15
    return _clean_prob(a)


def _make_fourier_random(n, rng):
    n = max(int(n), 2)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    f = np.ones(n, dtype=np.float64)
    for m in range(1, 5):
        f = f + rng.normal(0.0, 0.12) * np.cos(2.0 * np.pi * m * t)
    for m in range(1, 5):
        f = f + rng.normal(0.0, 0.12) * np.cos(float(np.pi) * float(m) * t)
    mn = float(np.min(f))
    if mn < 1e-4:
        f = f + (1e-3 - mn)
    f = np.maximum(f, 1e-9)
    return _clean_prob(f)


def _smooth_random(n, rng, base=None, amplitude=0.45):
    n = max(int(n), 2)
    if base is None:
        base = _make_uniform(n)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 16 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 1e-12:
        f = f / sd
    a = base * (1.0 + float(amplitude) * f)
    a = np.maximum(a, 1e-12)
    return _clean_prob(a)


def _resample_sequence(old, new_n):
    old = _clean_prob(old)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _make_uniform(new_n)
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / float(new_n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _clean_prob(new)


def _perturb(p, rng, sigma=0.12):
    p = _clean_prob(p)
    n = p.size
    noise = np.exp(rng.normal(0.0, float(sigma), size=n))
    q = p * noise + (1e-12 / float(n))
    if rng.random() < 0.20:
        q = 0.65 * q + 0.35 * p
    return _clean_prob(q)


def _get_given_sequence():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if float(arr.sum()) <= 1e-300:
        return None
    return arr / float(arr.sum())


def _make_starts(n, given, rng, best):
    n = max(int(n), 2)
    starts = []

    starts.append(_make_uniform(n))

    alphas = (0.35, 0.60, 0.90, 1.20, 1.60, 2.20, 3.00, 4.50, 7.00)
    for a in alphas:
        starts.append(_make_exp(n, a, False))
        starts.append(_make_exp(n, a, True))

    poly_pairs = (
        (1.0, 0.0), (0.0, 1.0),
        (2.0, 0.0), (0.0, 2.0),
        (1.0, 1.0), (1.5, 0.0), (0.0, 1.5),
        (0.5, 0.0), (0.0, 0.5),
        (1.0, 2.0), (2.0, 1.0),
    )
    for pp, qq in poly_pairs:
        starts.append(_make_poly_power(n, pp, qq))

    mixtures = (
        (0.80, 0.30), (1.00, 0.50), (1.20, 0.70),
        (1.50, 0.40), (0.50, 0.60), (2.00, 0.80),
    )
    for a, g in mixtures:
        starts.append(_make_mixture_exp(n, a, g))

    for _ in range(8):
        starts.append(_make_fourier_random(n, rng))

    starts.append(_smooth_random(n, rng, None, 0.50))
    starts.append(_smooth_random(n, rng, _make_uniform(n), 0.70))
    starts.append(_smooth_random(n, rng, _make_exp(n, 1.0), 0.45))

    if best is not None and best.size > 4:
        starts.append(_resample_sequence(best, n))

    if given is not None and given.size > 4:
        starts.append(_resample_sequence(given, n))

    return starts


def _top_starts(n, starts, k):
    n = max(int(n), 2)
    fft_size = _next_pow2(2 * n - 1)
    scored = []
    for idx, s in enumerate(starts):
        s = _clean_prob(s)
        sc, _ = _score_fft(s, fft_size)
        if np.isfinite(sc):
            scored.append((sc, idx, s))
    scored.sort()
    out = []
    for sc, _idx, s in scored[:max(int(k), 1)]:
        out.append((sc, s))
    return out


def _get_lp_structure(n):
    if n in LP_CACHE:
        return LP_CACHE[n]
    n = int(n)
    K = 2 * n - 1
    total = n * n
    indices = np.empty(total, dtype=np.int32)
    base = np.arange(n, dtype=np.int32)
    for j in range(n):
        indices[j * n:(j + 1) * n] = base + j
    indptr = np.arange(0, total + 1, n, dtype=np.int64)
    LP_CACHE[n] = (K, indices, indptr)
    return LP_CACHE[n]


def _lp_best_response(p, M, n, K, indices, indptr):
    if not HAS_SCIPY_LP:
        return None
    if n < 4 or n > MAX_LP_N:
        return None
    if M <= 1e-300:
        return _clean_prob(p), 1.0

    data = np.tile(p, n)
    try:
        A = sp.csc_matrix((data, indices, indptr), shape=(K, n), dtype=np.float64)
    except Exception:
        try:
            rows = indices
            cols = np.repeat(np.arange(n, dtype=np.int32), n)
            A = sp.coo_matrix((data, (rows, cols)), shape=(K, n)).csc()
        except Exception:
            return None

    c = -np.ones(n, dtype=np.float64)
    b_ub = M * np.ones(K, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return None

    if not getattr(res, "success", False):
        return None

    y = np.asarray(res.x, dtype=np.float64)
    if y.size != n:
        return None
    y = np.maximum(y, 0.0)
    S = float(y.sum())
    if S <= 1e-12:
        return None
    y = y / S
    return y, max(1.0, S)


def _mix_conv_max(bxx, byy, bxy, t):
    u = 1.0 - float(t)
    c = (u * u) * bxx + (2.0 * u * float(t)) * bxy + (float(t) * float(t)) * byy
    c = np.maximum(c, 0.0)
    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        return 0.0
    return m


def _line_search_mix(p, q, bxx, byy, bxy, n, iterations=28):
    p = _clean_prob(p)
    q = _clean_prob(q)
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    a = 0.0
    b = 1.0

    c = b - (b - a) / gr
    d = a + (b - a) / gr
    fc = _mix_conv_max(bxx, byy, bxy, c)
    fd = _mix_conv_max(bxx, byy, bxy, d)

    for _ in range(max(8, int(iterations))):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / gr
            fc = _mix_conv_max(bxx, byy, bxy, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / gr
            fd = _mix_conv_max(bxx, byy, bxy, d)

    t = 0.5 * (a + b)
    best_m = _mix_conv_max(bxx, byy, bxy, t)
    best_t = t

    for te in (0.0, 1.0, 0.01, 0.02, 0.05, 0.10, 0.25, 0.50):
        me = _mix_conv_max(bxx, byy, bxy, te)
        if me < best_m:
            best_m = me
            best_t = te

    new = (1.0 - best_t) * p + best_t * q
    new = _clean_prob(new)
    score = 2.0 * float(n) * best_m
    return new, score


def _lse_weights(b, beta):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return np.ones(1, dtype=np.float64)
    m = float(b.max())
    if beta <= 1.0 or m <= 1e-300:
        return np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 1e-300:
        return np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
    return e / s


def _weighted_gradient(p, weights, fft_size, n):
    p = np.asarray(p, dtype=np.float64)
    weights = np.asarray(weights, dtype=np.float64)
    pr = p[::-1]
    P = np.fft.rfft(pr, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(P * W, fft_size)
    h = conv[n - 1:2 * n - 1]
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    return 2.0 * h


def _mirror_polish(p, deadline, rng, fft_size):
    p = _clean_prob(p)
    n = p.size
    if n < 8:
        sc, _ = _score_fft(p, fft_size)
        return p, sc
    if time.monotonic() >= deadline:
        sc, _ = _score_fft(p, fft_size)
        return p, sc

    current = p.copy()
    current_score, _ = _score_fft(current, fft_size)
    best = current.copy()
    best_score = current_score

    g_fft = _next_pow2(3 * n - 2)
    beta_target = 18.0
    eta = 0.25
    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 0.5:
        it += 1

        _, b = _score_fft(current, fft_size)
        mb = float(b.max())
        if mb <= 1e-300:
            break
        beta = float(min(1e7, max(1.0, beta_target / mb)))
        weights = _lse_weights(b, beta)

        gp = _weighted_gradient(current, weights, g_fft, n)
        mean_gp = float(np.dot(current, gp))
        h = gp - mean_gp
        scale = float(np.max(np.abs(h)))
        if scale <= 1e-300:
            current = _perturb(best, rng, 0.10)
            current_score, _ = _score_fft(current, fft_size)
            eta = 0.25
            beta_target = min(900.0, beta_target * 0.55)
            no_improve = 0
            continue

        h = h / scale
        q = current * np.exp(-eta * h)
        q = _clean_prob(q)
        qscore, _ = _score_fft(q, fft_size)

        if qscore < current_score - 1e-14:
            current = q
            current_score = qscore
            if qscore < best_score:
                best = q.copy()
                best_score = qscore
            eta = float(min(0.80, eta * 1.18))
            no_improve = 0
        else:
            eta *= 0.70
            no_improve += 1
            if eta < 1e-4:
                current = _perturb(best, rng, 0.12)
                current_score, _ = _score_fft(current, fft_size)
                beta_target = max(12.0, beta_target * 0.45)
                eta = 0.25
                no_improve = 0

        if it % 8 == 0:
            beta_target = float(min(1200.0, beta_target * 1.04))

        if no_improve > 120:
            current = best.copy()
            current_score = best_score
            beta_target = 18.0
            eta = 0.25
            no_improve = 0

    return best, best_score


def _refine_sequence_lp(p, deadline, rng, try_mirror=True):
    p = _clean_prob(p)
    n = p.size
    if n < 2:
        return p, 2.0
    if n == 1:
        return p, 2.0

    fft_size = _next_pow2(2 * n - 1)
    current = p.copy()
    current_score, bxx = _score_fft(current, fft_size)
    best = current.copy()
    best_score = current_score

    structure = None
    if HAS_SCIPY_LP and 8 <= n <= MAX_LP_N:
        structure = _get_lp_structure(n)

    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 2.0:
        it += 1

        if it % 5 == 1 and current_score > best_score + 1e-10:
            current = best.copy()
            current_score, bxx = _score_fft(current, fft_size)
            no_improve = 0

        M = current_score / (2.0 * float(n))
        if not np.isfinite(M) or M <= 0.0:
            break

        if structure is not None:
            K, indices, indptr = structure
            res = _lp_best_response(current, M, n, K, indices, indptr)
            did_lp = False
            if res is not None:
                y, S = res
                did_lp = True
                if S > 1.0 + 1e-8:
                    byy, _ = _score_fft(y, fft_size)
                    bxy = _convolve_fft(current, y, fft_size)
                    bxy = np.where(np.isfinite(bxy), bxy, 0.0)
                    bxy = np.maximum(bxy, 0.0)
                    new, new_score = _line_search_mix(current, y, bxx, byy, bxy, n, 26)
                    if np.isfinite(new_score):
                        if new_score < best_score:
                            best = new.copy()
                            best_score = new_score
                        if new_score < current_score + 1e-13:
                            current = new.copy()
                            current_score = new_score
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

            if try_mirror and did_lp and (no_improve >= 2 or it % 7 == 0):
                t_now = time.monotonic()
                if t_now < deadline - 6.0:
                    emphend = min(deadline - 4.0, t_now + 4.0)
                    pol, pol_score = _mirror_polish(
                        best if best_score < current_score else current,
                        emphend,
                        rng,
                        fft_size,
                    )
                    if np.isfinite(pol_score) and pol_score < best_score:
                        best = pol.copy()
                        best_score = pol_score
                        no_improve = 0
                    else:
                        no_improve += 1
        else:
            if not try_mirror:
                break
            t_now = time.monotonic()
            emphend = min(deadline - 2.0, t_now + 12.0)
            pol, pol_score = _mirror_polish(current, emphend, rng, fft_size)
            if np.isfinite(pol_score) and pol_score < best_score:
                best = pol.copy()
                best_score = pol_score
                current = poly.copy()
                current_score, bxx = _score_fft(current, fft_size)
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 18 and time.monotonic() < deadline - 12.0:
            current = _perturb(best, rng, 0.10)
            current_score, bxx = _score_fft(current, fft_size)
            if current_score < best_score:
                best = current.copy()
                best_score = current_score
            no_improve = 0

    return best, best_score


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    deadline = t0 + max(budget, 60.0) - 35.0

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(0)

    given = _get_given_sequence()

    # Quick robust fallback: a simple decreasing exponential.
    best_a = _make_exp(256, 1.0, False)
    best_score, _ = _score_fft(best_a)

    if time.monotonic() < deadline - 20.0:
        starts0 = _make_starts(256, given, rng, None)
        top0 = _top_starts(256, starts0, 1)
        if top0:
            s0, p0 = top0[0]
            if np.isfinite(s0) and s0 < best_score:
                best_a = p0
                best_score = s0
    print("initial best len=%d score=%.10f" % (best_a.size, best_score), flush=True)

    scale = min(1.0, max(0.18, budget / 2400.0))

    # Stage allocation: concentrate on 256 first, then resample/extend.
    configs = [
        (256, int(980 * scale), 4),
        (512, int(650 * scale), 4),
        (1024, int(720 * scale), 3),
    ]
    fracs4 = (0.50, 0.25, 0.15, 0.10)

    for n_st, total_time, k in configs:
        t_now = time.monotonic()
        if t_now >= deadline - 25.0:
            break
        total_time = max(25, int(total_time))
        total_time = min(total_time, int(max(25, deadline - t_now - 25)))
        if total_time < 20:
            break

        print("stage n=%d budget=%.1fs" % (n_st, total_time), flush=True)
        starts = _make_starts(int(n_st), given, rng, best_a)
        top = _top_starts(int(n_st), starts, max(1, int(k)))
        if not top:
            continue

        fr = fracs4[:len(top)]
        if len(fr) < len(top):
            extra = 1.0 - sum(fr)
            if extra > 0.0:
                fr = fr + (extra,)

        for frac, (s0, p0) in zip(fr, top):
            t_now = time.monotonic()
            if t_now >= deadline - 15.0:
                break
            t_end = min(deadline - 10.0, t_now + max(10.0, total_time * float(frac)))
            if t_end <= t_now:
                continue
            res_a, res_score = _refine_sequence_lp(p0.copy(), t_end, rng, True)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).ravel()
                best_score = float(res_score)
                print(
                    "improved len=%d score=%.10f t=%.1fs"
                    % (len(best_a), best_score, time.monotonic() - t0),
                    flush=True,
                )

    # Upscale / polish the current best if there is still meaningful time.
    if time.monotonic() < deadline - 70.0:
        target因您的 = None
        if best_a.size < 1024:
            up_n = 1024
            init = _resample_sequence(best_a, up_n)
            # Blend a little exponential base to avoid an overly degenerate resample.
            init = _clean_prob(0.82 * init + 0.18 * _make_exp(up_n, 1.0, False))
            t_end = min(deadline - 40.0, time.monotonic() + int(min(320.0, (deadline - time.monotonic()) * 0.55)))
            res_a, res_score = _refine_sequence_lp(init, t_end, rng, True)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).ravel()
                best_score = float(res_score)
                print("upscale len=%d score=%.10f" % (len(best_a), best_score), flush=True)
        elif best_a.size == 1024 and time.monotonic() < deadline - 20.0 and best_score > 1.5020:
            # Try 2048 with mirror only; if it does not improve, discard.
            init = _resample_sequence(best_a, 2048)
            t_end = min(deadline - 25.0, time.monotonic() + 120.0)
            res_a, res_score = _refine_sequence_lp(init, t_end, rng, False)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).ravel()
                best_score = float(res_score)
                print("2048 upside len=%d score=%.10f" % (len(best_a), best_score), flush=True)

    # Final polish on the best sequence found so far.
    if time.monotonic() < deadline - 5.0:
        t_end = min(deadline - 2.0, time.monotonic() + max(5.0, deadline - time.monotonic() - 3.0))
        res_a, res_score = _refine_sequence_lp(best_a.copy(), t_end, rng, True)
        if np.isfinite(res_score) and res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).ravel()
            best_score = float(res_score)
            print("final polish len=%d score=%.10f" % (len(best_a), best_score), flush=True)
        else:
            print("final polish no improvement len=%d score=%.10f" % (len(best_a), best_score), flush=True)

    out = _clean_prob(best_a)
    out = np.asarray(out, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s > 1e-300:
        out = out / s

    # As a safety net, verify with the external scorer if it is available and cheap.
    try:
        if out.size <= 4096 and time.monotonic() < t0 + budget - 5.0:
            direct = _score_direct(out)
            print("internal/direct final len=%d score=%.10f" % (out.size, direct), flush=True)
            if not np.isfinite(direct) or direct > 2.0:
                out = _make_uniform(512)
                print("fallback to uniform len=%d" % out.size, flush=True)
    except Exception:
        pass

    try:
        if out.size <= 4096 and time.monotonic() < t0 + budget - 2.0:
            ext = evaluate_sequence([float(x) for x in out])
            if np.isfinite(ext):
                print("external final score=%r" % (ext,), flush=True)
            else:
                out = _make_uniform(512)
                print("external fallback to uniform" , flush=True)
    except Exception:
        pass

    return [float(x) for x in out]
```
