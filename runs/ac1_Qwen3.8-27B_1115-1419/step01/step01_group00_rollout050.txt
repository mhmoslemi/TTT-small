internal optimizer uses probabilities p_i = a_i / sum(a).  The core loop is
a multiplicative / log-softmax descent with FFT-accelerated convolutions and
gradients.  It anneals several surrogate weightings of the convolution
coefficients (Lq-norm, top-r, softmax), uses backtracking line search, and
periodically perturbs the best known point to escape local minima.
"""

import numpy as np
import time


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _prepare_p(p):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    if p.size == 0:
        return np.ones(1, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(p.size, dtype=np.float64) / p.size
    p /= s
    # A tiny floor keeps the multiplicative update able to move all coordinates.
    p = np.maximum(p, 1e-15)
    s = float(p.sum())
    p /= s
    return p


def _convolve_self(p, L):
    ft = np.fft.rfft(p, L)
    c = np.fft.irfft(ft * ft, L)
    return c[:2 * p.size - 1]


def _convolve_and_obj_A(p, L, n):
    ft = np.fft.rfft(p, L)
    c = np.fft.irfft(ft * ft, L)
    v = np.maximum(c[:2 * n - 1], 0.0)
    score = 2.0 * float(n) * float(np.max(v))
    return v, score, ft


def _score_base(base, n, L):
    p = np.asarray(base, dtype=np.float64).ravel().copy()
    if p.size != n:
        return np.inf, None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.inf, None
    p /= s
    v = np.maximum(_convolve_self(p, L), 0.0)
    return 2.0 * float(n) * float(np.max(v)), p


def _grad_half_from_A(p, w, L, A):
    # h_i = sum_j w_{i+j} p_j, which is one half of the gradient of
    # sum_k w_k (p*p)_k.  The formula is a circular cross-convolution and
    # can be computed with one FFT of W and the existing FFT of p.
    Wf = np.fft.rfft(w, L)
    h = np.fft.irfft(Wf * np.conjugate(A), L)
    return h[:len(p)]


def _weights_from_v(v, q):
    m = len(v)
    vmax = float(np.max(v))
    if vmax <= 0.0 or not np.isfinite(vmax):
        return np.ones(m, dtype=np.float64) / m
    u = v / vmax
    if q <= 1.0:
        w = u.copy()
    else:
        w = np.power(u, float(q - 1.0))
    sw = float(np.sum(w))
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(v))] = 1.0
        return w
    return w / sw


def _weights_top(v, r):
    m = len(v)
    r = int(np.clip(r, 1, m))
    if r >= m:
        return np.ones(m, dtype=np.float64) / m
    idx = np.argpartition(v, -r)[-r:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0 / r
    return w


def _weights_softmax(v, rel):
    m = len(v)
    vmax = float(np.max(v))
    if vmax <= 0.0 or not np.isfinite(vmax):
        return np.ones(m, dtype=np.float64) / m
    z = (v - vmax) / (max(1e-300, rel * vmax))
    z = np.clip(z, -50.0, 50.0)
    w = np.exp(z)
    sw = float(np.sum(w))
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(v))] = 1.0
        return w
    return w / sw


def _perturb_p(p, rng, sigma, kind="global"):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    sigma = float(np.clip(sigma, 1e-6, 0.6))
    n = len(p)
    if kind == "block":
        w = max(2, int(n * np.clip(sigma, 0.01, 0.25)))
        if w >= n:
            s = 0
            w = n
        else:
            s = int(rng.integers(0, n - w + 1))
        p[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    else:
        p *= np.exp(rng.normal(0.0, sigma, size=n))
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 1e-300)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return p
    p /= s
    p = np.maximum(p, 1e-15)
    p /= p.sum()
    return p


def _resample_sequence(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n_new, dtype=np.float64)
    if m == n_new:
        return a.copy()
    if m == 1:
        return np.full(n_new, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, a)


def _best_param_init(n, seed=0, max_polys=80, max_smooth=40):
    L = _next_pow2(2 * n - 1)
    best_p = None
    best_F = np.inf

    t = np.linspace(0.0, 1.0, n)
    x = t - 0.5

    for a in np.linspace(-4.0, 4.0, 17):
        base = 1.0 + a * x
        F, p = _score_base(base, n, L)
        if F < best_F:
            best_F = F
            best_p = p

    for a in np.linspace(-4.0, 4.0, 9):
        raw = 1.0 + a * x
        for pw in (0.5, 1.0, 2.0, 3.0):
            base = np.maximum(raw, 0.0)
            if pw != 1.0:
                base = np.power(base, pw)
            F, p = _score_base(base, n, L)
            if F < best_F:
                best_F = F
                best_p = p

    rng = np.random.default_rng(seed)
    for _ in range(max_polys):
        c1 = rng.uniform(-4.0, 4.0)
        c2 = rng.uniform(-3.0, 3.0)
        c3 = rng.uniform(-3.0, 3.0)
        c4 = rng.uniform(-2.0, 2.0)
        base = 1.0 + c1 * x + c2 * x * x + c3 * x**3 + c4 * x**4
        F, p = _score_base(base, n, L)
        if F < best_F:
            best_F = F
            best_p = p

        F2, p2 = _score_base(-base, n, L)
        if F2 < best_F:
            best_F = F2
            best_p = p2

    for _ in range(max_smooth):
        f = np.zeros(n, dtype=np.float64)
        for mode in range(1, 5):
            amp = rng.uniform(0.0, 0.8)
            phase = rng.uniform(0.0, 2.0 * np.pi)
            f += amp * np.cos(2.0 * np.pi * mode * t + phase)
        base = np.exp(f)
        F, p = _score_base(base, n, L)
        if F < best_F:
            best_F = F
            best_p = p

    if best_p is None:
        best_p = np.ones(n, dtype=np.float64) / n
    return _prepare_p(best_p)


def _optimize_p(p, n, deadline, tag, rng, q_start=4.0, q_max=512.0):
    p = _prepare_p(p)
    L = _next_pow2(2 * n - 1)

    v, F, A = _convolve_and_obj_A(p, L, n)
    best_p = p.copy()
    best_F = F

    q = float(q_start)
    mode = "q"
    top_frac = 0.25
    soft_rel = 0.15

    base_step = 0.5
    min_step = 1e-4
    max_step = 2.0
    stagnation_limit = 650
    stall = 0
    adapt_count = 0

    start_time = time.monotonic()
    span = max(1e-3, deadline - start_time)
    strict_after = start_time + 0.62 * span
    last_print = start_time - 20.0

    while time.monotonic() < deadline:
        now = time.monotonic()
        if now >= deadline:
            break

        if now - last_print > 15.0:
            print(
                f"[{tag}] t={now - start_time:6.1f}s "
                f"best={best_F:.8f} cur={F:.8f} mode={mode} q={q:.1f} "
                f"top={top_frac:.3f} soft={soft_rel:.4f} step={base_step:.4f}"
            )
            last_print = now

        if F < best_F:
            best_F = F
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        strict = now > strict_after

        if stall > stagnation_limit:
            adapt_count += 1
            if mode == "q":
                if q < q_max:
                    q = float(min(q * 2.0, q_max))
                else:
                    mode = "top"
                    top_frac = 0.25
            elif mode == "top":
                top_frac *= 1.5
                if top_frac > 1.0:
                    mode = "softmax"
                    soft_rel = 0.15
            else:
                soft_rel *= 0.6
                if soft_rel < 0.004:
                    mode = "q"
                    q = float(max(q_start, q_max / 8.0))

            sigma = min(0.25, 0.025 * (adapt_count + 1))
            if adapt_count % 2 == 0:
                p = _perturb_p(best_p, rng, sigma, kind="global")
            else:
                p = _perturb_p(best_p, rng, sigma, kind="block")
            v, F, A = _convolve_and_obj_A(p, L, n)
            base_step = 0.5
            stall = 0
            continue

        if mode == "q":
            w = _weights_from_v(v, q)
        elif mode == "top":
            w = _weights_top(v, max(1, int(n * top_frac)))
        else:
            w = _weights_softmax(v, soft_rel)

        h = _grad_half_from_A(p, w, L, A)
        if not np.all(np.isfinite(h)):
            h = np.zeros(n, dtype=np.float64)

        dot = float(np.dot(p, h))
        d = h - dot
        dmax = float(np.max(np.abs(d)))

        if dmax < 1e-15:
            p = _perturb_p(best_p, rng, 0.03, kind="global")
            v, F, A = _convolve_and_obj_A(p, L, n)
            base_step = 0.5
            stall = 0
            continue

        u = -d / dmax
        u = np.clip(u, -2.0, 2.0)

        step = base_step
        accepted = False
        cur_F = F
        U_cur = float(np.dot(w, v))

        for trial in range(8):
            p_new = p * np.exp(step * u)
            p_new = np.maximum(p_new, 0.0)
            s_new = float(p_new.sum())
            if not np.isfinite(s_new) or s_new <= 0.0:
                step *= 0.5
                continue
            p_new /= s_new

            A_new = np.fft.rfft(p_new, L)
            c_new = np.fft.irfft(A_new * A_new, L)
            v_new = np.maximum(c_new[:2 * n - 1], 0.0)
            F_new = 2.0 * float(n) * float(np.max(v_new))
            U_new = float(np.dot(w, v_new))

            accept_F = F_new < cur_F - 1e-15
            accept_U = (
                (not strict)
                and U_new < U_cur - 1e-12
                and F_new < best_F + max(1e-8, 0.002 * best_F)
            )

            if accept_F or accept_U:
                p = p_new
                v = v_new
                F = F_new
                A = A_new
                accepted = True

                if trial <= 1:
                    base_step = min(max_step, base_step * 1.05)
                else:
                    base_step = max(min_step, base_step * 0.85)
                break

            step *= 0.5
            if step < min_step:
                break

        if not accepted:
            if base_step <= min_step * 2.0:
                p = _perturb_p(best_p, rng, 0.035, kind="global")
                v, F, A = _convolve_and_obj_A(p, L, n)
                base_step = 0.5
            else:
                base_step = max(min_step, base_step * 0.5)
                if rng.random() < 0.08:
                    p = _perturb_p(p, rng, 0.012, kind="global")
                    v, F, A = _convolve_and_obj_A(p, L, n)
            stall += 1

    return best_p, best_F


def _get_parent_sequence():
    seq = None
    try:
        seq = height_sequence_1
    except Exception:
        seq = globals().get("height_sequence_1")
    if seq is None:
        return None
    try:
        a = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    if float(a.sum()) <= 0.0:
        return None
    return a


def _internal_sequence_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    p = a / s
    n = len(p)
    L = _next_pow2(2 * n - 1)
    v, F, _ = _convolve_and_obj_A(p, L, n)
    return F


def _finalize_sequence(p, n):
    p = _prepare_p(p)
    base = p * float(n)
    base = np.where(np.isfinite(base), base, 0.0)
    base = np.clip(base, 0.0, 1000.0)

    candidates = [base.copy()]
    for eps in (1e-12, 1e-10, 1e-8, 1e-6):
        r = p.copy()
        r[r < eps] = 0.0
        s = float(r.sum())
        if s > 0.0 and np.isfinite(s):
            a = r / s * float(n)
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == n:
                candidates.append(a)

    best_a = base
    best_score = np.inf
    for a in candidates:
        sc = _internal_sequence_score(a)
        try:
            esc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(esc):
                sc = min(sc, esc)
        except Exception:
            pass
        if sc < best_score:
            best_score = sc
            best_a = a

    b = best_a.copy()
    if b.size == 0:
        return [1.0]
    mx = float(np.max(b))
    if mx > 1000.0:
        b *= 1000.0 / mx
    sm = float(np.sum(b))
    if sm < 0.01:
        b *= 0.01 / sm
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.clip(b, 0.0, 1000.0)
    return b.tolist()


def propose_candidate(seed=42, budget_s=2400):
    start = time.monotonic()
    total_usable = max(60.0, float(budget_s) - 20.0)
    hard_deadline = start + total_usable - 8.0
    final_reserve = min(300.0, 0.12 * total_usable)
    work_deadline = start + (total_usable - final_reserve) - 5.0

    rng = np.random.default_rng(seed)
    parent = _get_parent_sequence()

    primary_n = 4096
    tasks = []

    if parent is not None and 500 <= parent.size <= 8000:
        p_parent = _prepare_p(parent)

        p_prim = _resample_sequence(parent, primary_n)
        p_prim = _prepare_p(p_prim)
        L_prim = _next_pow2(2 * primary_n - 1)
        _, F_prim, _ = _convolve_and_obj_A(p_prim, L_prim, primary_n)

        p_par = _best_param_init(primary_n, int(seed) + 101, max_polys=60, max_smooth=20)
        _, F_par, _ = _convolve_and_obj_A(p_par, L_prim, primary_n)

        if F_par < F_prim:
            p_prim = _prepare_p(p_par)

        tasks.append((primary_n, p_prim, 0.55, "primary4096"))
        tasks.append((int(parent.size), p_parent, 0.45, "parent"))
    elif parent is not None:
        p_prim = _resample_sequence(parent, primary_n)
        p_prim = _prepare_p(p_prim)
        tasks.append((primary_n, p_prim, 1.0, "parent_resampled4096"))
    else:
        p4096 = _best_param_init(primary_n, int(seed) + 7, max_polys=100, max_smooth=50)
        p3072 = _best_param_init(3072, int(seed) + 17, max_polys=40, max_smooth=20)
        tasks.append((primary_n, p4096, 0.75, "param4096"))
        tasks.append((3072, p3072, 0.25, "param3072"))

    best_p = None
    best_n = 0
    best_F = np.inf

    used_work = sum(max(0.0, dead - start) for dead in [work_deadline])
    remaining_work = max(30.0, work_deadline - time.monotonic())
    frac_sum = sum(float(fr) for _, _, fr, _ in tasks)
    if frac_sum <= 0.0:
        frac_sum = 1.0

    for n, p0, fr, tag in tasks:
        dur = (float(fr) / frac_sum) * remaining_work
        task_deadline = min(work_deadline, max(time.monotonic() + 1.0, time.monotonic() + dur))
        if task_deadline <= time.monotonic():
            continue

        p_best, F_best = _optimize_p(
            p0, int(n), task_deadline, tag, rng,
            q_start=4.0, q_max=512.0
        )
        print(f"[{tag}] done F={F_best:.9f} n={n}")

        if F_best < best_F:
            best_F = F_best
            best_n = int(n)
            best_p = p_best

    if best_p is None:
        best_p = np.ones(primary_n, dtype=np.float64) / primary_n
        best_n = primary_n
        best_F = 2.0

    if time.monotonic() < hard_deadline - 10.0:
        final_start = time.monotonic()
        final_deadline = max(final_start + 30.0, hard_deadline)
        # Final refinement starts from a sharper surrogate; previous runs usually
        # are close enough that high-q weights are productive.
        p_final, F_final = _optimize_p(
            best_p, best_n, final_deadline, "final_refine", rng,
            q_start=16.0, q_max=256.0
        )
        print(f"[final_refine] done F={F_final:.9f} n={best_n}")
        if F_final < best_F:
            best_F = F_final
            best_p = p_final

    if best_p is None:
        best_p = np.ones(primary_n, dtype=np.float64) / primary_n
        best_n = primary_n

    return _finalize_sequence(best_p, best_n)
