import numpy as np
import time

def project_simplex(v):
    """Project v onto the standard probability simplex."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def fast_score(a):

Minimize 2n*max(a*a)/(sum(a))^2 using LP-based iterations combined with FISTA polishing.
The LP step finds the "best response" direction, and FISTA polishes the result.

from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def project_simplex(v):
    """Project v onto the standard probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    cols = []
    vals = []

        j_max = min(n - 1, k)

                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])

                best_a = a_new.copy()
                a_norm = a_new
                break
    
    return best_a, best_score

def fista_polish(a0, n, duration, rng, threshold_frac=0.01):
    """FISTA polishing with adaptive threshold for active peaks."""
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    step = 1.5  # Start with a reasonable step (Lipschitz constant ≈ 1)
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # FISTA momentum
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif abs(sy - 1.0) > 0.1:
            y = y / sy

        c_y = np.convolve(y, y)
        M = c_y.max()

        threshold = M * (1.0 - threshold_frac)
        active = np.where(c_y >= threshold)[0]

            active = np.array([np.argmax(c_y)])
        
        # Compute subgradient (average over active peaks)
        g = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                g[j] += y[k - j]
                if k - j != j:
                    g[k - j] += y[j]
        g /= len(active)
        
        # Gradient step + project
        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation

            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adapt step size
            if no_improve == 30:
                step *= 0.8
            elif no_improve == 100:
                step *= 0.6
                a_prev = best_a.copy()
                t = 1.0
            elif no_improve > 300:
                # Restart with perturbation
                a = best_a + rng.normal(0, 0.005 / np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                a = a / a.sum()
                a_prev = a.copy()
                t = 1.0
                step = 1.5
                no_improve = 0
        
        if it % 500 == 0:
            print(f"  FISTA it={it}, score={best_score:.8f}, |active|={len(active)}, step={step:.4f}")
    
    # Final check
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

Minimize 2n*max(a*a)/(sum(a))^2 using LP-based iterations and FISTA polishing.
Uses FFT-based gradient computation for efficiency.
"""

import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def project_simplex(v):
    """Project v onto the standard probability simplex."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def compute_subgradient(y, active_idx):
    """Compute the subgradient of max(a*a) using FFT-based correlation."""
    n = len(y)
    # Build the active indicator
    I = np.zeros(2 * n - 1)
    I[active_idx] = 1.0
    # Cross-correlation gives the gradient
    g = 2.0 * correlate(I, y, mode='valid') / len(active_idx)
    return g

def solve_lp_b(f, M, n):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            vals.append(f[k - j])
            rows.append(k)
            cols.append(j)
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return f.copy()

def run_lp_phase(a, n, max_iters=15, time_budget=300):
    """Phase: LP-based iterations."""
    best_a = a.copy()
    best_score = fast_score(a)
    a_norm = a / a.sum()
    deadline = time.monotonic() + time_budget
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = np.max(np.convolve(a_norm, a_norm))
        if M < 1e-15:
            break
        
        print(f"  LP iter {it}: M={M:.8f}, score={best_score:.8f}")
        
        b = solve_lp_b(a_norm, M, n)
        sb = b.sum()
        if sb < 1e-10:
            break
        b = b / sb
        
        # Try different blend rates
        best_blend_a = None
        best_blend_score = np.inf
        for t in [0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
            a_try = (1 - t) * a_norm + t * b
            sc = fast_score(a_try)
            if sc < best_blend_score:
                best_blend_score = sc
                best_blend_a = a_try
            if sc < best_score:
                break
        
        if best_blend_score < best_score and best_blend_a is not None:
            best_score = best_blend_score
            best_a = best_blend_a.copy()
            a_norm = best_blend_a
        else:
            print(f"  LP iter {it}: no improvement, stopping.")
            break
    
    return best_a, best_score

def run_fista(a0, n, duration, rng, threshold_frac=0.005):
    """FISTA with FFT-based subgradient."""
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    step = 1.5
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif abs(sy - 1.0) > 0.05:
            y = y / sy
        
        c_y = np.convolve(y, y)
        M = c_y.max()
        
        # Find active peaks
        threshold = M * (1.0 - threshold_frac)
        active = np.where(c_y >= threshold)[0]
        
        g = compute_subgradient(y, active)
        
        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1
        
        if it % 10 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve == 50:
                step *= 0.85
            elif no_improve == 150:
                step *= 0.7
                a_prev = best_a.copy()
                t = 1.0
            elif no_improve > 400:
                a = best_a + rng.normal(0, 0.003 / np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a = a / sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t = 1.0
                step = 1.5
                no_improve = 0
                print(f"    FISTA restart at it={it}, score={best_score:.8f}")
            
            if it % 500 == 0:
                print(f"    FISTA it={it}, score={best_score:.8f}, |act|={len(active)}, step={step:.4f}")
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

"""
Minimize 2n*max(a*a)/(sum(a))^2 using FFT-accelerated FISTA on the probability simplex.
Combines LP-based fast iterations with accelerated projected gradient descent.
Key improvements: FFT-based subgradient computation, adaptive step size, 
and multi-threshold active set selection.
"""

import numpy as np
import time
from scipy.signal import correlate as fft_correlate

def project_simplex(v):
    """Project v onto standard probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n <= 1:
        return np.array([max(0.0, v[0])] if n == 1 else [], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 efficiently."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(c.max()) / (s * s)


def compute_subgrad(y, active_idx):
    """
    Compute subgradient of max_k (y*y)[k] using FFT-based correlation.
    The subgradient is the average of gradients at active peaks.
    
    For (y*y)[k], gradient w.r.t. y[i] = 2*y[k-i] (for valid k-i).
    Sum over active peaks: g[i] = (2/|A|) * sum_{k in A} y[k-i] * 1{0<=k-i<n}
    = (2/|A|) * (I_A cross-correlate y)[i]
    """
    n = len(y)
    I = np.zeros(2 * n - 1, dtype=np.float64)
    I[active_idx] = 1.0
    g = 2.0 * fft_correlate(I, y, mode='valid')
    g /= len(active_idx)
    return g


def run_fista(a0, n, duration, rng, threshold=0.002, init_step=1.0):
    """
    Accelerated projected gradient (FISTA) to minimize max(conv) on simplex.
    Uses FFT-based subgradient and adaptive step size.
    """
    a = a0.copy()
    a_prev = a0.copy()
    t_k = 1.0
    step = init_step
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # FISTA momentum update
        t_old = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        alpha = (t_old - 1.0) / t_k
        y = a + alpha * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy  # Keep on simplex for stability
        
        # Compute convolution and find active peaks
        c_y = np.convolve(y, y)
        M = c_y.max()
        
        # Active set: peaks within threshold of max
        thr = M * (1.0 - threshold)
        active = np.where(c_y >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_y)], dtype=np.int64)
        
        # Compute subgradient
        g = compute_subgrad(y, active)
        
        # Gradient step + project onto simplex
        a_new = y - step * g
        a_new = project_simplex(a_new)
        
        a_prev = a
        a = a_new
        it += 1
        
        # Evaluate and adapt (periodically)
        if it % 5 == 1:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Step size adaptation
            if no_improve == 40:
                step *= 0.85
            elif no_improve == 120:
                step *= 0.7
                a_prev = best_a.copy()
                t_k = 1.0
            elif no_improve >= 300:
                # Perturb and restart
                noise = rng.normal(0.0, 0.002 / np.sqrt(n), n)
                a = best_a + noise
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t_k = 1.0
                step = init_step
                no_improve = 0
            
            if it % 2000 == 1:
                print(f"    it={it}, score={best_score:.8f}, |A|={len(active)}, step={step:.4f}")
    
    # Final evaluation
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def run_lp_phase(a, n, max_iters=8, time_budget=200):
    """
    LP-based iterations: at each step, solve the LP to find best response
    direction, then blend.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    best_a = a.copy()
    best_score = fast_score(a)
    a_norm = a / a.sum()
    deadline = time.monotonic() + time_budget
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = float(np.max(np.convolve(a_norm, a_norm)))
        if M < 1e-15:
            break
        
        # Build sparse constraint matrix
        rows = np.empty(n * n, dtype=np.int32)  # upper bound on nnz
        cols = np.empty(n * n, dtype=np.int32)
        vals = np.empty(n * n, dtype=np.float64)
        count = 0
        
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows[count] = k
                cols[count] = j
                vals[count] = a_norm[k - j]
                count += 1
        
        A = csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(2*n-1, n))
        
        # Solve LP: max sum(b) s.t. A*b <= M*1, b >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'presolve': True})
        
        if not res.success or res.x is None:
            print(f"  LP iter {it}: solver failed")
            break
        
        b = np.maximum(res.x, 0.0)
        sb = b.sum()
        if sb < 1e-10:
            break
        b /= sb
        
        # Try blending
        improved = False
        for t in [0.7, 0.5, 0.3, 0.2, 0.1]:
            a_try = (1.0 - t) * a_norm + t * b
            sc = fast_score(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
                a_norm = a_try
                improved = True
                break
        
        print(f"  LP iter {it}: M={M:.8f}, LP_sum={sb:.4f}, score={best_score:.8f}")
        
        if not improved:
            break
    
    return best_a, best_score
