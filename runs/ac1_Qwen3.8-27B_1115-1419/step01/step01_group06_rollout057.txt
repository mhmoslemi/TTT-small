    return np.maximum(v - theta, 0)

    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences
using smoothed subgradient descent on the probability simplex with FFT.
"""

import numpy as np
import time

def next_fast_len(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def project_simplex(v):
    """Hard projection onto the standard probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if rho.size == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def load_parent():
    try:
        obj = globals().get("height_sequence_1", None)
        if obj is None:
            return None
        arr = np.asarray(obj, dtype=np.float64).ravel()

            return None
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = arr.sum()
        if s <= 0 or not np.isfinite(s):
            return None
        return arr / s
    except:
        return None

def exact_objective(a):
    """Compute the exact objective 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    mx = c.max()
    return float(2.0 * n * mx / (s * s))

def fft_convolution(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2 * len(a) - 1]

def smoothed_grad_step(a, N, beta, lr):
    """One smoothed subgradient step on the simplex."""
    n = len(a)
    # Compute convolution
    c = fft_convolution(a, N)
    # Compute softmax weights
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(np.clip(bc - bc_max, -745, 709))
    p = exp_bc / exp_bc.sum()
    # Compute gradient: grad_j = 2n * 2 * sum_k p_k * a_{k-j}
    # = 4n * (reverse(p) * a)[2n-2-j]
    Rp = p[::-1]
    Rp_fft = np.fft.rfft(Rp, N)
    a_fft = np.fft.rfft(a, N)
    conv = np.fft.irfft(Rp_fft * a_fft, N)
    grad = 4.0 * n * conv[n - 1:2 * n - 1][::-1]
    # Gradient step with projection
    a_new = project_simplex(a - lr * grad)
    # Compute smoothed objective for tracking
    smoothed_obj = 2.0 * n * (bc_max + np.log(exp_bc.sum())) / beta
    return a_new, smoothed_obj, p, c
