    return e / s


def compute_smooth_grad(x, tau):
    """Gradient of LSE-smoothed max(convolve(softmax(x), softmax(x))) w.r.t. x"""
    a = softmax(x)
    c = np.convolve(a, a)
    c_max = c.max()
    c_shifted = (c - c_max) / tau
    w = np.exp(c_shifted)
    w /= w.sum()
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x


def smooth_obj(x, tau):
    a = softmax(x)
    c = np.convolve(a, a)
    c_max = c.max()
    c_shifted = (c - c_max) / tau
    return c_max + tau * np.log(np.exp(c_shifted).sum())


def exact_subgradient_a(a, n_active=None, tol=1e-8):
    """Compute exact subgradient of max(convolve(a,a)) w.r.t. a.
    Returns grad_a where grad_a[j] = 2 * sum_k w[k] * a[k-j] for active k."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    if n_active is not None:
        sorted_idx = np.argsort(c)[::-1][:n_active]
        active = np.zeros(len(c))
        active[sorted_idx] = 1.0 / n_active
    else:
        active = (c >= c_max - tol * max(c_max, 1.0))
        active = active.astype(float)
        active_sum = active.sum()
        if active_sum == 0:
            active = np.zeros(len(c))
            active[np.argmax(c)] = 1.0
        else:
            active /= active_sum
    grad_a = 2.0 * np.correlate(active, a, mode='valid')
    return grad_a


def mirror_descent_step(x, grad_x, step, momentum=0.9):
    """One mirror descent step with Nesterov momentum.
    x is in log-space, grad_x is gradient w.r.t. x."""
    # Nesterov: evaluate gradient at x + momentum * (x - x_prev)
    # Simplified: just use momentum on the update
    return x - step * grad_x


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.where(cond)[0][-1] + 1
    theta = cssv[k - 1] / k
    return np.maximum(v - theta, 0)

    t = np.arange(n) / n
    x = (t - 0.5) ** 2 + 0.05
    inits['u_shaped'] = x / x.sum()

    # Two-block
    for ratio in [0.3, 0.4, 0.6]:
        x = np.ones(n)
        x[:int(n * ratio)] = 2.0
        x[int(n * ratio):] = 0.5
        inits[f'twoblock_{ratio}'] = x / x.sum()

    # Three-block (high-low-high)
    x = np.ones(n)
    x[:n // 4] = 2.0
    x[n // 4:n - n // 4] = 0.5
    x[n - n // 4:] = 2.0
    inits['three_block'] = x / x.sum()

    # Periodic patterns
    for period in [2, 3, 4, 5, 7]:
        x = np.zeros(n)
        for i in range(n):
            x[i] = 1.0 + 0.4 * ((i % period) < period // 2)
        inits[f'periodic_{period}'] = x / x.sum()

    # Fourier-based random (good autocorrelation structure)
    for trial in range(4):
        nfreq = n
        phases = rng.random(nfreq) * 2 * np.pi
        amps = rng.random(nfreq)
        amps[:n // 10] = 0  # Kill low frequencies
        spectrum = amps * np.exp(1j * phases)
        spectrum = np.concatenate([spectrum, np.conj(spectrum[:n // 2][::-1])])
        x = np.abs(np.fft.ifft(spectrum)) + 0.01
        inits[f'fourier_{trial}'] = x / x.sum()

    # Random with smooth structure
    for trial in range(3):
        coarse = rng.random(20)
        x = np.interp(np.arange(n), np.arange(20), coarse)
        x = x ** rng.uniform(0.5, 3.0)
        x = np.maximum(x, 0.01)
        inits[f'smooth_rand_{trial}'] = x / x.sum()

    return list(inits.values())


def upsample_sequence(a, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    old_t = np.linspace(0, 1, old_n)
    new_t = np.linspace(0, 1, new_n)
    new_a = np.interp(new_t, old_t, a)
    new_a = np.maximum(new_a, 0)
    new_a = new_a / new_a.sum()
    return new_a


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search: minimize 2*n*max(convolve(a,a))/(sum(a))^2.
    Multi-phase: screen initializations, L-BFGS anneal, subgradient polish,
    perturbation escapes, final ultra-fine optimization.
    """
    deadline = time.monotonic() + budget_s - 20
    rng = np.random.default_rng(20240101)

    # Get parent
    parent = None
    if 'height_sequence_1' in globals():
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 1e-15)
        parent = parent / parent.sum()
        n_parent = len(parent)
    else:
        n_parent = 4806

    best_score = float('inf')
    best_a = None

    # Phase 1: Multi-length screening
    print("=" * 60)
    print("PHASE 1: Multi-length screening")
    print("=" * 60)

    # Try a few lengths near the parent
    if n_parent > 2000:
        candidate_lengths = [n_parent, n_parent, n_parent]  # Focus on parent length
    else:
        candidate_lengths = [n_parent]

    n = candidate_lengths[0]
    print(f"  Primary length: n={n}")

    # Generate and screen initializations
    inits = generate_initializations(n, rng, parent)
    print(f"  Generated {len(inits)} initializations")

    scored_inits = []
    for i, init in enumerate(inits):
        score = evaluate_sequence(init.tolist())
        scored_inits.append((score, i, init))

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Best init: {scored_inits[0][0]:.6f}")
    print(f"  Top 5: {[f'{s:.4f}' for s, _, _ in scored_inits[:5]]}")

    if scored_inits[0][0] < best_score:
        best_score = scored_inits[0][0]
        best_a = scored_inits[0][2].copy()

    # Phase 2: L-BFGS optimization of top initializations
    print("\n" + "=" * 60)
    print("PHASE 2: L-BFGS optimization")
    print("=" * 60)

    n_top = min(10, len(scored_inits))
    remaining = deadline - time.monotonic()
    time_per = max(60.0, remaining / (n_top + 2) - 120)
    time_per = min(200.0, time_per)

    for idx in range(n_top):
        if time.monotonic() > deadline - 240:
            break

        score_init, init_idx, init = scored_inits[idx]
        x = np.log(init + 1e-30)

        tau_sched = [0.08 / n, 0.04 / n, 0.02 / n, 0.01 / n, 0.005 / n, 0.002 / n]
        a_opt, score, x = run_lbfgs_anneal(x, n, max_time=time_per, tau_schedule=tau_sched)

        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  [{idx+1}/{n_top}] New best: {score:.6f} (init was {score_init:.6f})")
        else:
            print(f"  [{idx+1}/{n_top}] Score: {score:.6f} (init was {score_init:.6f})")

    # Phase 3: Subgradient polishing
    print("\n" + "=" * 60)
    print("PHASE 3: Subgradient polishing")
    print("=" * 60)
    print(f"  Starting score: {best_score:.6f}")

    remaining = deadline - time.monotonic()
    sg_time = min(300.0, remaining * 0.3)
    a_polished, score = subgradient_polish(best_a, max_time=sg_time, initial_step=0.003)
    if score < best_score:
        best_score = score
        best_a = a_polished
        print(f"  After subgradient: {score:.6f}")

    # Phase 4: Perturbation + re-optimization cycles
    print("\n" + "=" * 60)
    print("PHASE 4: Perturbation escapes")
    print("=" * 60)

    n_cycles = 5
    cycle_time = min(120.0, (deadline - time.monotonic()) / n_cycles - 60)
    cycle_time = max(30.0, cycle_time)

    for cyc in range(n_cycles):
        if time.monotonic() > deadline - 120:
            break

        # Create perturbed version
        noise_scale = 0.02 * (1 - cyc / n_cycles) + 0.005
        a_pert = best_a + rng.normal(0, noise_scale, n)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Also try a "mix" with a random initialization
        a_mix = 0.7 * best_a + 0.3 * rng.dirichlet(np.ones(n))
        a_mix = a_mix / a_mix.sum()

        candidates = [a_pert, a_mix]
        x_cands = [np.log(c + 1e-30) for c in candidates]

        for x_c in x_cands:
            if time.monotonic() > deadline - 120:
                break
            tau_sched = [0.02 / n, 0.01 / n, 0.005 / n, 0.002 / n]
            a_opt, score, x_new = run_lbfgs_anneal(x_c, n, max_time=cycle_time, tau_schedule=tau_sched)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  Cycle {cyc+1}: New best {score:.6f}")

        print(f"  Cycle {cyc+1}: best={best_score:.6f}")

    # Phase 5: Final ultra-fine L-BFGS
    print("\n" + "=" * 60)
    print("PHASE 5: Final ultra-fine optimization")
    print("=" * 60)

    remaining = deadline - time.monotonic() - 10
    if remaining > 30:
        x_final = np.log(best_a + 1e-30)
        tau_final = [0.01 / n, 0.005 / n, 0.002 / n, 0.001 / n, 0.0005 / n, 0.0002 / n]
        a_final, score, _ = run_lbfgs_anneal(x_final, n, max_time=remaining, tau_schedule=tau_final)
        if score < best_score:
            best_score = score
            best_a = a_final
            print(f"  Final L-BFGS: {score:.6f}")

        # One more subgradient pass
        if time.monotonic() < deadline - 10:
            a_sg, score = subgradient_polish(best_a, max_time=min(60, deadline - time.monotonic() - 5),
                                             initial_step=0.001)
            if score < best_score:
                best_score = score
                best_a = a_sg
                print(f"  Final subgradient: {score:.6f}")

    print("\n" + "=" * 60)
    print(f"FINAL: score={best_score:.6f}, n={len(best_a)}")
    print(f"  max_conv={np.convolve(best_a, best_a).max():.8f}")
    print(f"  sum={best_a.sum():.8f}")
    print("=" * 60)

    return [float(val) for val in best_a]
