def _next_pow2(x):

    return 1 << (x - 1).bit_length()


def _softmax(v):

    v = v - float(v.max())

    return e / s

Algorithm:

  - Adam optimization in softmax/log parameterization

import numpy as np

import math


def _next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def _softmax(v):
    v = np.asarray(v, dtype=np.float64).copy()
    v = v - float(v.max())
    v = np.clip(v, -745.0, 700.0)
    e = np.exp(v)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.full(v.shape, 1.0 / max(1, v.size), dtype=np.float64)
    return e / s


def _compute_b(a, m):
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * int(a.size) - 1]
    return b


def _score_from_b(a, b):
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-30:
        return np.inf
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax < 1e-300:
        return np.inf
    return float(2.0 * n * bmax / (S * S))


def _score(a):
    n = int(a.size)
    if n == 0:
        return np.inf
    m = _next_pow2(2 * n - 1)
    b = _compute_b(a, m)
    return _score_from_b(a, b)


def _grad_weighted(a, b, m, w):
    """Gradient of sum_k w_k * b_k with respect to a.
    b_k = (a*a)_k, grad = 2 * IFFT(FFT(w) * conj(FFT(a)))"""
    n = int(a.size)
    P = np.fft.fft(w, m)
    F = np.fft.fft(a, m)
    g = 2.0 * np.fft.ifft(P * np.conj(F)).real[:n]
    return g


def _make_weights(b, tau, top_k_frac=0.05):
    """Create soft-max weights on b."""
    L = int(b.size)
    bmax = float(b.max())
    if bmax < 1e-300:
        return np.ones(L) / L
    x = (b - bmax) / max(tau, 1e-30)
    x = np.clip(x, -80.0, 0.0)
    w = np.exp(x)
    ws = float(w.sum())
    if ws < 1e-30 or not np.isfinite(ws):
        w = np.zeros(L)
        w[np.argmax(b)] = 1.0
    else:
        w /= ws
    if top_k_frac > 0.0:
        k = max(1, int(top_k_frac * L))
        if k < L:
            idx = np.argpartition(b, -k)[-k:]
        else:
            idx = np.arange(L)
        w[idx] += 0.5
        w /= float(w.sum())
    return w


def _proj_nonneg_sum(a, s_target=None):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        a.fill(1.0 / max(1, a.size))
        s = 1.0
    if s_target is not None:
        a *= s_target / s
    else:
        a /= s
    return a


def _interp(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = int(n)
    if a.size == 0 or s < 1e-30:
        return np.ones(n) / n
    if a.size == n:
        return a / s
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss < 1e-30:
        out.fill(1.0 / n)
    else:
        out /= ss
    return out


def _init_spectral(n, rng, target_b=None):
    """Spectral sqrt initialization: try to realize a target autocorrelation."""
    n = int(n)
    m = _next_pow2(2 * n - 1)
    L = 2 * n - 1
    
    if target_b is None:
        # Default target: slightly tapered (higher in the middle)
        t = np.linspace(0.0, 1.0, L)
        target_b = 1.0 - 0.3 * (2.0 * t - 1.0) ** 2
    
    target_b = np.maximum(target_b, 0.0)
    tmax = float(target_b.max())
    if tmax > 0:
        target_b = target_b / tmax
    else:
        target_b = np.ones(L)
    
    T = np.fft.fft(target_b, m)
    # Take "square root" in frequency domain
    # Handle sign: use magnitude with phase
    mag = np.abs(T)
    ph = np.angle(T)
    sqrt_T = np.sqrt(mag) * np.exp(0.5j * ph)
    a = np.fft.ifft(sqrt_T).real[:n]
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _init_parametric(n, rng):
    """Generate diverse parametric initializations."""
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    inits.append(_proj_nonneg_sum(1.0 + 0.5 * np.cos(2.0 * np.pi * t)))
    
    # Gaussian bumps
    for sigma in (0.15, 0.25, 0.35, 0.5):
        inits.append(_proj_nonneg_sum(np.exp(-((t - 0.5) / sigma) ** 2)))
    
    # Edge-weighted
    inits.append(_proj_nonneg_sum(1.0 / (0.1 + np.abs(t - 0.5) * 2.0)))
    
    # Two peaks
    for d in (0.2, 0.25, 0.3):
        inits.append(_proj_nonneg_sum(np.exp(-((t - d) / 0.1) ** 2) + np.exp(-((t - (1-d)) / 0.1) ** 2)))
    
    # Random Fourier
    kc = min(20, n // 2)
    for _ in range(3):
        a = 1.0 + 0.3
        for k in range(1, kc + 1):
            amp = 0.25 / (k ** 0.7)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + ph)
        inits.append(_proj_nonneg_sum(a))
    
    # Triangle / tent
    inits.append(_proj_nonneg_sum(1.0 - 2.0 * np.abs(t - 0.5)))
    
    # Beta-like
    inits.append(_proj_nonneg_sum((t * (1 - t)) ** 0.5 + 0.01))
    inits.append(_proj_nonneg_sum((t * (1 - t)) ** 1.0 + 0.01))
    inits.append(_proj_nonneg_sum((t * (1 - t)) ** 2.0 + 0.01))
    
    return inits


def _adam_step(v, grad_v, t, mu, nu, beta1, beta2, eps, lr):
    """Standard Adam update. Returns new v, mu, nu."""
    mu = beta1 * mu + (1 - beta1) * grad_v
    nu = beta2 * nu + (1 - beta2) * (grad_v * grad_v)
    b1 = beta1 ** t
    b2 = beta2 ** t
    mhat = mu / max(1e-12, 1 - b1)
    vhat = nu / max(1e-12, 1 - b2)
    step = mhat / (np.sqrt(vhat) + eps)
    v = v - lr * step
    v = v - float(v.mean())
    v = np.clip(v, -30.0, 30.0)
    return v, mu, nu


def _optimize(a_init, n, budget, seed, deadline):
    """Main optimization loop using Adam on softmax parameters."""
    n = int(n)
    m = _next_pow2(2 * n - 1)
    L = 2 * n - 1
    rng = np.random.default_rng(seed)
    
    # Initialize parameters
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    v = np.log(a)
    v = v - float(v.mean())
    v = np.clip(v, -25.0, 25.0)
    
    # Optimizer state
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    mu = np.zeros(n)
    nu = np.zeros(n)
    t = 0
    
    # Score tracking
    a_curr = _softmax(v)
    b = _compute_b(a_curr, m)
    best_score = _score_from_b(a_curr, b)
    best_a = a_curr.copy()
    last_improve = time.monotonic()
    start = time.monotonic()
    
    # Temperature schedule parameters
    tau_start = 0.25
    tau_end = 0.02
    # LR schedule
    lr_start = 0.08
    lr_end = 0.005
    
    check_interval = max(2, min(10, 500 // max(1, n // 256)))
    stuck_threshold = 60.0
    
    n_iter = 0
    while time.monotonic() < min(deadline, start + budget):
        now = time.monotonic()
        elapsed = max(1e-9, now - start)
        progress = min(1.0, elapsed / max(1.0, budget))
        
        # Check for stuck
        if now - last_improve > stuck_threshold:
            sigma = 0.1 + 0.3 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(v.mean())
            v = np.clip(v, -25.0, 25.0)
            last_improve = now
            continue
        
        # Current a
        a_curr = _softmax(v)
        
        # Compute b and weights
        b = _compute_b(a_curr, m)
        b = np.maximum(b, 0.0)
        bmax = float(b.max())
        
        if not np.isfinite(bmax) or bmax < 1e-300:
            break
        
        # Temperature (anneal from tau_start to tau_end)
        tau = (tau_start + (tau_end - tau_start) * (progress ** 0.7)) * bmax
        tau = max(tau, 1e-15)
        
        # In the last 15% of the run, use very sharp temperature
        if progress > 0.85:
            tau = 0.03 * bmax
        
        w = _make_weights(b, tau, top_k_frac=0.03)
        
        # Gradient of weighted objective w.r.t. a
        grad_a = _grad_weighted(a_curr, b, m, w)
        
        # Gradient w.r.t. v (softmax chain rule)
        cm = float(np.dot(grad_a, a_curr))
        grad_v = a_curr * (grad_a - cm)
        
        # Normalize gradient
        grms = float(np.sqrt(np.mean(grad_v * grad_v)))
        if not np.isfinite(grms) or grms < 1e-14:
            v += rng.normal(0.0, 0.05, n)
            v = v - float(v.mean())
            continue
        
        grad_v = grad_v / grms
        
        # Learning rate (cosine schedule)
        t += 1
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1.0 + math.cos(math.pi * progress))
        
        # Adam step
        v, mu, nu = _adam_step(v, grad_v, t, mu, nu, beta1, beta2, eps, lr)
        
        # Score check
        n_iter += 1
        if n_iter % check_interval == 0:
            a_check = _softmax(v)
            b_check = _compute_b(a_check, m)
            sc = _score_from_b(a_check, b_check)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                best_score = sc
                best_a = a_check.copy()
                last_improve = now
    
    # Final score
    a_final = _softmax(v)
    b_final = _compute_b(a_final, m)
    sc_final = _score_from_b(a_final, b_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final
    
    return best_a, float(best_score)
