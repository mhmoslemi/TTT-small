from scipy.optimize import linprog

def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]

def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _normalize(a, target_sum):
    """Normalize array to target sum."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def _lp_direction(a, M):
    """Solve LP to find best direction g.
    max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
    """
    n = len(a)
    # Build constraint matrix A (sparse): A[k, j] = a[k-j] if valid
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid (k-j) in [0, n-1]

    rows = []
    cols = []
    vals = []

        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            idx = k - j  # index into a
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)

    if res.status == 0:

    return None

    n = len(a)
    n_rows = 2 * n - 1
    # For each column j, place a[0:min(n, 2n-1-j)] at rows j to j+len-1

    rows = []
    cols = []
    vals = []
    for j in range(n):

        rows.extend(range(j, j + length))
        cols.extend([j] * length)
        vals.extend(a[:length])
    from scipy.sparse import coo_matrix
    return coo_matrix((vals, (rows, cols)), shape=(n_rows, n)).tocsr()

    """Solve simplified LP with only top K constraints."""
    n = len(a)
    peak_positions = sorted(peak_positions)[:K]
    
    # Build constraint matrix: K rows, n columns
    # A[k_i, j] = a[k_i - j] if 0 <= k_i - j < n
    A = np.zeros((K, n))
    for idx, k in enumerate(peak_positions):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[idx, j] = a[k - j]

    # Min -sum(g) s.t. A @ g <= M, g >= 0

    if res.status == 0:

lengths, diverse initializations, and Nesterov-like momentum with backtracking.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum):
    """Normalize array to target sum, preserving non-negativity."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _compute_subgrad(a, peak_positions):
    """Compute subgradient of max(a*a) at given peak positions."""
    n = len(a)
    grad = np.zeros(n)
    for m in peak_positions:
        m = int(m)
        k_lo = max(0, m - n + 1)
        k_hi = min(n - 1, m)
        if k_hi >= k_lo:
            js = np.arange(k_lo, k_hi + 1)
            grad[js] += a[m - js]
    return grad


def _lp_direction(a, peak_positions, M, max_k=25, time_limit=3.0):
    """Solve simplified LP: max sum(g) s.t. (a*g)[k] <= M for top peaks, g >= 0."""
    n = len(a)
    K = min(len(peak_positions), max_k)
    if K == 0:
        return None
    peaks = sorted(peak_positions, key=lambda x: M - M)  # keep order
    peaks = peaks[:K]

    A = np.zeros((K, n))
    for idx, k in enumerate(peaks):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            js = np.arange(j_lo, j_hi + 1)
            A[idx, js] = a[k - js]

    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(K, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 and float(np.sum(res.x)) > 1e-10:
            return res.x
    except Exception:
        pass
    return None


def _optimize_from_start(a, deadline, use_lp=True, lp_every=30,
                         beta=0.85, lr_init=0.3, verbose=False, tag=""):
    """Main optimization loop with hybrid LP + subgradient."""
    n = len(a)
    S = float(np.sum(a))
    a = _normalize(a, S).copy()

    best = a.copy()
    best_score = _score_fast(a)

    momentum = np.zeros(n)
    lr = lr_init
    step = 0
    stuck = 0
    lp_count = 0

    while time.monotonic() < deadline:
        b = _fft_conv(a, a)
        max_b = float(np.max(b))

        # Find active peaks
        threshold = 0.75 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(b)[-3:]
        if len(active) > 30:
            top_vals = b[active]
            order = np.argsort(top_vals)[-30:]
            active = active[order]

        # --- LP direction (periodic) ---
        if use_lp and step > 0 and step % lp_every == 0:
            g = _lp_direction(a, active, max_b, max_k=25)
            if g is not None:
                g_norm = _normalize(g.copy(), S)
                # Try blending
                for t in [0.5, 0.3, 0.15, 0.05]:
                    a_try = (1 - t) * a + t * g_norm
                    a_try = np.maximum(a_try, 0.0)
                    a_try = _normalize(a_try, S)
                    sc = _score_fast(a_try)
                    if sc < _score_fast(a) - 1e-12:
                        a = a_try
                        lp_count += 1
                        if sc < best_score:
                            best = a.copy()
                            best_score = sc
                            stuck = 0
                        break

        # --- Subgradient step ---
        grad = _compute_subgrad(a, active)
        obj_grad = S * grad - max_b

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * obj_grad

        # Step with adaptive lr
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S)

        sc_new = _score_fast(a_new)
        sc_cur = _score_fast(a)

        if sc_new < sc_cur - 1e-12:
            a = a_new
            lr = min(lr * 1.05, 2.0)  # increase lr slightly if progressing
            if sc_new < best_score:
                best = a.copy()
                best_score = sc_new
                stuck = 0
            else:
                stuck += 1
        else:
            a = best + 0.01 * np.random.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = _normalize(a, S)
            lr = max(lr * 0.7, 0.01)
            momentum = momentum * 0.5
            stuck += 1

        step += 1

        # Restart if stuck too long
        if stuck > 40:
            a = best + 0.08 * np.random.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = _normalize(a, S)
            momentum = np.zeros(n)
            lr = lr_init * 0.5
            stuck = 0

        if verbose and step % 500 == 0:
            print(f"    [{tag}] step={step}, best={best_score:.6f}, lr={lr:.3f}, lp={lp_count}")

    return best, best_score


def _try_n_values(n_list, parent, deadline, rng, verbose=True):
    """Try optimization from multiple initializations across different n values."""
    results = []

    for n in n_list:
        if time.monotonic() > deadline - 30:
            break
        tag = f"n={n}"
        if verbose:
            print(f"\n  Trying {tag}:")

        # Create initializations for this n
        inits = []

        # From parent (interpolated or truncated)
        if parent is not None and len(parent) > 0:
            p = np.asarray(parent, dtype=np.float64)
            p = np.clip(p, 0, 1000)
            if len(p) == n:
                inits.append(("parent", p.copy()))
            else:
                # Interpolate
                x_old = np.linspace(0, 1, len(p))
                x_new = np.linspace(0, 1, n)
                p_interp = np.interp(x_new, x_old, p)
                inits.append(("parent_interp", p_interp))

        inits.append(("uniform", np.ones(n)))
        inits.append(("random", rng.random(n) * 5 + 0.5))

        # Cosine modulation
        k = np.arange(n)
        for freq in [1, 3]:
            inits.append((f"cos{freq}", 1.0 + 0.5 * np.cos(2 * np.pi * freq * k / n)))

        # Anti-symmetric-ish
        inits.append(("ramp", 1.0 + 0.5 * k / n))
        inits.append(("inv_ramp", 1.0 + 0.5 * (1.0 - k / n)))

        # Gaussian bump
        inits.append(("gauss", np.exp(-4.0 * ((k - n // 2) / n) ** 2) + 0.3))

        # Bimodal
        if n % 2 == 0:
            inits.append(("bimodal", np.concatenate([np.ones(n // 2) * 1.5,
                                                      np.ones(n // 2) * 0.5])))

        # Time allocation per initialization
        remaining = deadline - time.monotonic()
        n_inits = len(inits)
        budget_per = max(10.0, min(120.0, remaining / max(n_inits, 1) * 0.7))

        local_best = None
        local_score = np.inf

        for name, init in inits:
            if time.monotonic() > deadline - 20:
                break
            a0 = _normalize(np.maximum(init, 0.0), np.sqrt(2.0 * n))
            individual_deadline = time.monotonic() + budget_per
            res_a, res_score = _optimize_from_start(
                a0, individual_deadline, use_lp=True, lp_every=40,
                beta=0.85, lr_init=0.3, verbose=False, tag=f"{tag}/{name}")
            if res_score < local_score:
                local_score = res_score
                local_best = res_a
                if verbose:
                    print(f"    {name}: {local_score:.6f}")

        if local_best is not None:
            results.append((n, local_best, local_score))
            if verbose:
                print(f"  Best for {tag}: {local_score:.6f}")

    return results


def _polish(a, deadline, verbose=False):
    """Fine-tuning with small steps and frequent LP."""
    n = len(a)
    S = float(np.sum(a))
    a = _normalize(a, S).copy()
    best = a.copy()
    best_score = _score_fast(a)

    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.05
    step = 0
    stuck = 0

    while time.monotonic() < deadline:
        b = _fft_conv(a, a)
        max_b = float(np.max(b))

        threshold = 0.9 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(b)[-2:]
        if len(active) > 15:
            active = active[np.argsort(b[active])[-15:]]

        # LP every 15 steps
        if step > 0 and step % 15 == 0:
            g = _lp_direction(a, active, max_b, max_k=20, time_limit=2.0)
            if g is not None:
                g_norm = _normalize(g.copy(), S)
                for t in [0.2, 0.1, 0.03]:
                    a_try = (1 - t) * a + t * g_norm
                    a_try = np.maximum(a_try, 0.0)
                    a_try = _normalize(a_try, S)
                    sc = _score_fast(a_try)
                    if sc < _score_fast(a) - 1e-14:
                        a = a_try
                        if sc < best_score:
                            best = a.copy()
                            best_score = sc
                            stuck = 0
                        break

        grad = _compute_subgrad(a, active)
        obj_grad = S * grad - max_b
        momentum = beta * momentum + (1.0 - beta) * obj_grad

        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S)
        sc_new = _score_fast(a_new)
        sc_cur = _score_fast(a)

        if sc_new < sc_cur - 1e-14:
            a = a_new
            if sc_new < best_score:
                best = a.copy()
                best_score = sc_new
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
            lr = max(lr * 0.9, 0.005)

        if stuck > 30:
            a = best + 0.02 * np.random.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = _normalize(a, S)
            momentum *= 0.3
            stuck = 0

        step += 1
        if verbose and step % 300 == 0:
            print(f"    polish step={step}, score={best_score:.8f}")

    return best, best_score
