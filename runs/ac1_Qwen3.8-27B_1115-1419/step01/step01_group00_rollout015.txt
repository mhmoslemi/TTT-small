"""
Search for a non-negative sequence minimizing
    2*n*max(conv(a,a)) / (sum(a)^2).

The program uses a multi-resolution projected gradient search with:
  - FFT-based self-convolution and gradient evaluations
  - smoothed max-convolution weights (active-set detector)
  - direct/FFT hybrid gradient computation
  - line-search step control and occasional conditional-gradient restarts
  - coarse-to-fine profile initialization from the previous best sequence
  - repeated perturbing/refinement cycles to escape local minima
"""

import numpy as np
import time


def _next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s


def _project_sum(v, target):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 1e-300 or not np.isfinite(s):
        v = np.ones(v.shape, dtype=np.float64)
        s = float(v.size)
    v = v * (float(target) / s)

    mx = float(np.max(v))
    if mx > 1000.0:
        v = np.minimum(v, 1000.0)
        s = float(v.sum())
        if s > 1e-300 and np.isfinite(s):
            v = v * (float(target) / s)
        else:
            v = np.ones(v.shape, dtype=np.float64) * (float(target) / v.size)
    return v


def _resample_vec(v, m):
    v = np.asarray(v, dtype=np.float64).flatten()
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    n = v.size
    if n == 0:
        return np.ones(m, dtype=np.float64) * m
    if n == m:
        a = v.copy()
    else:
        xp = np.arange(n, dtype=np.float64)
        fp = np.linspace(0.0, n - 1.0, m)
        a = np.interp(fp, xp, v)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        a = np.ones(m, dtype=np.float64)
        s = float(m)
    return a * (float(m) / s)


def _objective_fft(a, N=None):
    if N is None:
        N = len(a)
    N = int(N)
    L = _next_pow2(2 * N - 1)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    c = np.maximum(c[:2 * N - 1], 0.0)
    return 2.0 * float(np.max(c)) / float(N)


def _convolve_with_f(a, L, N):
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    c = np.maximum(c[:2 * N - 1], 0.0)
    return c, fa


def _cross_fft(weights, a):
    N = len(a)
    L = _next_pow2(3 * N - 2)
    P = np.fft.rfft(weights, L)
    R = np.fft.rfft(a[::-1], L)
    out = np.fft.irfft(P * R, L)
    return out[N - 1:N - 1 + N]


def _active_gradient(p, a, N):
    pmax = float(np.max(p))
    if pmax <= 1e-300:
        return np.zeros(N, dtype=np.float64)

    thr = max(1e-12, 1e-5 * pmax)
    idx = np.flatnonzero(p >= thr)

    direct_limit = 150 if N > 256 else 400
    if idx.size > direct_limit:
        return _cross_fft(p, a)

    if idx.size == 0:
        return _cross_fft(p, a)

    mass = float(np.sum(p[idx]))
    if mass < 0.85:
        return _cross_fft(p, a)

    h = np.zeros(N, dtype=np.float64)
    for kk in idx:
        k = int(kk)
        w = float(p[k])
        if w <= 0.0:
            continue
        start = k - (N - 1)
        if start < 0:
            start = 0
        end = k + 1
        if end > N:
            end = N
        if start >= end:
            continue
        seg = a[k - end + 1:k - start + 1][::-1]
        h[start:end] += w * seg

    if mass < 0.999:
        h *= (1.0 / mass)
    return h


def _fw_point(a, h, N):
    cst = h - float(np.min(h))
    scale = float(np.max(cst))
    if scale <= 1e-12:
        return None
    z = -0.5 * cst / scale
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return None
    return (e / s) * float(N)


def _perturb(a, N, rng, sigma):
    b = np.array(a, dtype=np.float64, copy=True)
    if N < 4:
        b *= np.exp(rng.normal(0.0, max(1e-6, sigma), size=N))
        return _project_sum(b, N)

    nblocks = int(rng.integers(1, 4))
    low = 2
    high = max(3, N // 25)
    for _ in range(nblocks):
        w = int(rng.integers(low, high + 1))
        if w > N:
            w = N
        s = int(rng.integers(0, N - w + 1))
        b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    if rng.random() < 0.25:
        b *= np.exp(rng.normal(0.0, 0.4 * max(1e-6, sigma), size=N))
    return _project_sum(b, N)


def _to_list(v):
    v = np.asarray(v, dtype=np.float64).flatten()
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.clip(v, 0.0, 1000.0)
    return [float(x) for x in v]


def _top_by_score(items, K):
    arr = []
    for i, item in enumerate(items):
        o = float(item[0])
        v = item[1]
        if np.isfinite(o) and v is not None:
            arr.append((o, i, v))
    arr.sort()
    out = []
    for o, i, v in arr[:K]:
        out.append((o, v))
    return out


def _add_bank(bank, v, o):
    if v is None or not np.isfinite(o):
        return
    items = []
    for o2, v2 in bank:
        items.append((float(o2), v2))
    items.append((float(o), v))
    bank[:] = _top_by_score(items, 10)


def _make_starts(N, rng, parent_vec=None, count=8):
    starts = []
    try:
        if parent_vec is not None:
            p = np.asarray(parent_vec, dtype=np.float64).flatten()
            if p.size > 1:
                starts.append(_resample_vec(p, N))
    except Exception:
        pass

    starts.append(np.ones(N, dtype=np.float64) * float(N))
    if N <= 0:
        return starts

    x = np.linspace(0.0, 1.0, N, endpoint=False)
    for _ in range(count):
        kind = int(rng.integers(0, 6))
        if kind == 0:
            w = rng.normal(0.0, 0.6, N)
        elif kind == 1:
            w = np.zeros(N, dtype=np.float64)
            w += rng.normal(0.0, 0.6) * x
            nm = int(rng.integers(2, 7))
            for _ in range(nm):
                f = float(rng.integers(1, 8))
                amp = float(rng.uniform(0.3, 1.2))
                ph = float(rng.uniform(0.0, 2.0 * np.pi))
                if rng.random() < 0.5:
                    w += amp * np.cos(2.0 * np.pi * f * x + ph)
                else:
                    w += amp * np.sin(2.0 * np.pi * f * x + ph)
        elif kind == 2:
            drift = float(rng.uniform(-3.0, 3.0))
            bow = float(rng.uniform(-0.5, 3.0))
            y = 1.0 - 2.0 * x
            w = drift * x + bow * y * y
        elif kind == 3:
            k = int(rng.integers(2, 6))
            raw = np.zeros(N, dtype=np.float64)
            for _ in range(k):
                c = float(rng.uniform(0.0, 1.0))
                ww = float(rng.uniform(0.04, 0.25))
                st = float(rng.uniform(2.0, 10.0))
                raw += np.exp(-st * ((x - c) / ww) ** 2)
            raw += 0.1
            w = np.log(raw)
        elif kind == 4:
            f = float(rng.integers(1, 10))
            amp = float(rng.uniform(0.0, 0.95))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            raw = 1.0 + amp * np.sin(2.0 * np.pi * f * x + ph)
            raw = np.maximum(raw, 1e-3)
            w = np.log(raw)
        else:
            w = rng.uniform(-1.5, 1.5) * x + rng.normal(0.0, 0.6)

        z = w - float(np.max(w))
        z = np.clip(z, -8.0, 0.0)
        arr = np.exp(z)
        starts.append(_project_sum(arr, N))
    return starts


def _optimize_vector(a, N, deadline, tag, rng, coarse=False, hard=False):
    a = _project_sum(a, N)
    best = np.array(a, dtype=np.float64, copy=True)
    best_o = _objective_fft(a, N)
    cur_o = best_o

    if hard:
        delta = 0.035
        delta_min = 1e-5
        decay = 0.9998
        rho = 0.05
        rho_max = 0.20
    elif coarse:
        delta = 0.20
        delta_min = 0.0003
        decay = 0.999
        rho = 0.09
        rho_max = 0.25
    else:
        delta = 0.12
        delta_min = 0.0005
        decay = 0.9995
        rho = 0.045
        rho_max = 0.14

    Lc = _next_pow2(2 * N - 1)
    full_len = 2 * N - 1
    it = 0
    stall = 0

    print(f"[{tag}] start obj={best_o:.7f} N={N}", flush=True)

    while time.monotonic() < deadline and it < 1000000:
        c, fa = _convolve_with_f(a, Lc, N)
        maxc = float(np.max(c))
        cur_o = 2.0 * maxc / float(N)

        if cur_o < best_o - 1e-11:
            best_o = cur_o
            best = np.array(a, dtype=np.float64, copy=True)

        tau = delta * maxc
        if tau < 1e-12:
            tau = 1e-12
        z = (c - maxc) / tau
        p = np.exp(z)
        ps = float(p.sum())
        if ps <= 1e-300 or not np.isfinite(ps):
            p = np.zeros(full_len, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
        else:
            p /= ps

        h = _active_gradient(p, a, N)
        g = 2.0 * h
        gt = g - float(np.mean(g))
        scale = float(np.max(np.abs(gt)))

        if not np.isfinite(scale) or scale < 1e-12:
            a = _perturb(best, N, rng, 0.30)
            delta = max(delta, 0.10)
            rho = 0.05
            stall += 20
            it += 1
            continue

        d = -gt / scale
        d -= float(np.mean(d))
        d[np.isnan(d)] = 0.0

        neg = d < -1e-10
        if np.any(neg):
            alpha = float(np.min(-a[neg] / d[neg]))
        else:
            alpha = 1.0
        alpha *= 0.999

        if alpha <= 1e-10:
            v = _fw_point(a, h, N)
            moved = False
            if v is not None:
                fv = np.fft.rfft(v, Lc)
                best_fs_o = cur_o
                best_fs_b = 0.0
                for beta in (0.02, 0.05, 0.10, 0.20, 0.40, 0.70, 1.0):
                    ft = fa + beta * (fv - fa)
                    cc = np.fft.irfft(ft * ft, Lc)
                    oo = 2.0 * float(np.max(cc[0:full_len])) / float(N)
                    if np.isfinite(oo) and oo < best_fs_o:
                        best_fs_o = oo
                        best_fs_b = beta
                if best_fs_b > 0.0:
                    a = a + best_fs_b * (v - a)
                    a = _project_sum(a, N)
                    cur_o = best_fs_o
                    if cur_o < best_o - 1e-11:
                        best_o = cur_o
                        best = np.array(a, dtype=np.float64, copy=True)
                    stall = 0
                    moved = True
            if not moved:
                a = _perturb(best, N, rng, 0.25)
                delta = max(delta, 0.08)
                stall += 10
            it += 1
            continue

        tbase = rho * alpha
        fd = np.fft.rfft(d, Lc)
        best_fs_o = cur_o
        best_fs_t = 0.0

        if stall < 35:
            factors = (1.0, 2.0)
        else:
            factors = (1.0, 0.5, 2.0, 4.0)

        for fac in factors:
            t = tbase * float(fac)
            if t <= 1e-14 or t > alpha:
                continue
            ft = fa + t * fd
            cc = np.fft.irfft(ft * ft, Lc)
            oo = 2.0 * float(np.max(cc[0:full_len])) / float(N)
            if np.isfinite(oo) and oo < best_fs_o - 1e-13:
                best_fs_o = oo
                best_fs_t = t

        if best_fs_t > 0.0:
            a = a + best_fs_t * d
            a = _project_sum(a, N)
            cur_o = best_fs_o
            if cur_o < best_o - 1e-11:
                best_o = cur_o
                best = np.array(a, dtype=np.float64, copy=True)
            stall = max(0, stall - 5)
            rho = min(rho_max, rho * 1.08)
        else:
            stall += 1
            rho = max(0.005, rho * 0.78)

            if stall >= 30 and rng.random() < 0.35:
                v = _fw_point(a, h, N)
                if v is not None:
                    fv = np.fft.rfft(v, Lc)
                    best_fw_o = cur_o
                    best_fw_b = 0.0
                    for beta in (0.02, 0.05, 0.10, 0.20, 0.40, 0.70, 1.0):
                        ft = fa + beta * (fv - fa)
                        cc = np.fft.irfft(ft * ft, Lc)
                        oo = 2.0 * float(np.max(cc[0:full_len])) / float(N)
                        if np.isfinite(oo) and oo < best_fw_o:
                            best_fw_o = oo
                            best_fw_b = beta
                    if best_fw_b > 0.0:
                        a = a + best_fw_b * (v - a)
                        a = _project_sum(a, N)
                        cur_o = best_fw_o
                        if cur_o < best_o - 1e-11:
                            best_o = cur_o
                            best = np.array(a, dtype=np.float64, copy=True)
                        stall = 0
                        delta = max(delta, 0.05)

            if stall > 85:
                sig = 0.20 + 0.30 * min(1.0, float(stall) / 500.0)
                a = _perturb(best, N, rng, sig)
                cur_o = _objective_fft(a, N)
                delta = max(delta, 0.12)
                rho = 0.05
                stall = 0

        delta *= decay
        if delta < delta_min:
            delta = delta_min
        it += 1

    print(f"[{tag}] done obj={best_o:.7f} iters={it}", flush=True)
    return best, best_o

    total_budget = deadline - start
    coarse_budget = min(700.0, max(80.0, 0.25 * total_budget))
    weights = [float(s) for s in scales]
    sumw = sum(weights)
    stage_list = []
    for M, w in zip(scales, weights):
        stage_list.append((M, max(5.0, coarse_budget * w / sumw)))
    stage_sum = 0.0
    for _, b in stage_list:
        stage_sum += b
    if stage_sum > coarse_budget:
        sc = coarse_budget / stage_sum if stage_sum > 0 else 1.0
        stage_list = [(M, b * sc) for M, b in stage_list]

    current = []
    for M, sb in stage_list:
        rem = deadline - time.monotonic()
        if rem < 20.0:
            break
        budget = min(sb, 0.45 * rem)
        if budget < 4.0:
            continue

        selected = []
        if current:
            for _, v in _top_by_score(current, 3):
                selected.append(_resample_vec(v, M))
        if parent is not None:
            selected.append(_resample_vec(parent, M))

        target_total = 8 if M <= 128 else 6
        nnew = max(0, target_total - len(selected))
        if nnew > 0:
            selected.extend(_make_starts(M, rng, parent, nnew))
        if not selected:
            selected.append(_make_starts(M, rng, parent, 1)[0])

        per = budget / float(len(selected))
        while len(selected) > 4 and per < 1.5:
            selected = selected[:-2]
            per = budget / float(len(selected))

        results = []
        per = budget / float(len(selected))
        for si, sv in enumerate(selected):
            subd = time.monotonic() + max(1.0, per)
            if subd > deadline - 10.0:
                subd = deadline - 10.0
            bvec, ovec = _optimize_vector(sv, M, subd, f"ms{M}_{si}", rng, coarse=True)
            if np.isfinite(ovec):
                results.append((ovec, si, bvec))
        results.sort()
        for o, si, v in results:
            current.append((o, v))
        current = _top_by_score(current, 6)
        if current:
            print(f"coarse scale {M}: top={current[0][0]:.7f}", flush=True)

    # Full-length candidates.
    full_items = []
    if parent_full is not None:
        full_items.append((_objective_fft(parent_full, N), parent_full.copy()))

    for _, v in current:
        rv = _resample_vec(v, N)
        full_items.append((_objective_fft(rv, N), rv))

    if N > 256:
        for rv in _make_starts(N, rng, parent_full, 2):
            full_items.append((_objective_fft(rv, N), rv))

    full_items = _top_by_score(full_items, 5)
    for o, v in full_items:
        _add_bank(bank, v.copy(), o)

    # First full-length refinement pass.
    if time.monotonic() < deadline - 70.0:
        rem = deadline - time.monotonic()
        nstart = len(full_items)
        if nstart > 0:
            per = min(190.0, max(25.0, (rem - 70.0) / max(1, nstart) * 0.7))
            for idx, (_, v0) in enumerate(full_items):
                if time.monotonic() + 25.0 > deadline:
                    break
                subd = min(time.monotonic() + per, deadline - 45.0)
                b, o = _optimize_vector(np.array(v0, copy=True), N, subd, f"full{idx}", rng, coarse=False)
                _add_bank(bank, b.copy(), o)
                if bank and o < bank[0][0]:
                    print(f"full{idx}: obj={o:.7f}", flush=True)

    if not bank:
        best_vec = np.ones(N, dtype=np.float64) * float(N)
        best_o = _objective_fft(best_vec, N)
    else:
        best_o, best_vec = bank[0][0], bank[0][1]

    # Perturbation + refinement loop.
    cycle = 0
    while time.monotonic() < deadline - 55.0:
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break
        cycle += 1

        if len(bank) > 1 and rng.random() < 0.40:
            bi = int(rng.integers(0, len(bank)))
            base = np.array(bank[bi][1], dtype=np.float64, copy=True)
        else:
            base = np.array(best_vec, dtype=np.float64, copy=True)

        sigma = 0.07 + 0.35 * min(1.0, float(cycle) / 25.0)

        if cycle % 8 == 0 and parent_full is not None:
            a0 = _resample_vec(parent_full, N)
            a0 = _perturb(a0, N, rng, 0.12 * sigma + 0.03)
        else:
            a0 = _perturb(base, N, rng, sigma)

        cyc_time = min(85.0, max(22.0, rem * 0.18))
        if cycle % 4 == 0:
            cyc_time = min(cyc_time + 55.0, rem * 0.35)
        subd = min(time.monotonic() + cyc_time, deadline - 35.0)

        b, o = _optimize_vector(a0, N, subd, f"cycle{cycle}", rng, coarse=False)
        _add_bank(bank, b.copy(), o)
        if o < best_o - 1e-11:
            best_o = o
            best_vec = b.copy()
            print(f"cycle {cycle}: new best obj={best_o:.7f}", flush=True)

    # Final polish.
    if time.monotonic() < deadline - 12.0:
        subd = deadline - 2.0
        b, o = _optimize_vector(np.array(best_vec, copy=True), N, subd, "final", rng, hard=True)
        if np.isfinite(o) and o < best_o:
            best_o = o
            best_vec = b
    else:
        b, o = None, None

    # Exact evaluation of a few bank entries can rescue from tiny FFT/rounding differences.
    final_best_vec = np.asarray(best_vec, dtype=np.float64).copy()
    final_best_score = float(best_o)
    for _, v in _top_by_score(bank, 4):
        lst = _to_list(v)
        try:
            sc = float(evaluate_sequence(lst))
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < final_best_score:
            final_best_score = sc
            final_best_vec = np.asarray(v, dtype=np.float64).copy()

    final_list = _to_list(final_best_vec)
    try:
        sc = float(evaluate_sequence(final_list))
    except Exception:
        sc = float("inf")

    if not np.isfinite(sc):
        if parent_full is not None:
            fb = _to_list(parent_full)
            try:
                sc2 = float(evaluate_sequence(fb))
            except Exception:
                sc2 = float("inf")
            if np.isfinite(sc2):
                return fb
        final_list = [1.0] * N

    print(f"FINAL estimated obj={final_best_score:.7f} n={len(final_list)}", flush=True)
    return final_list
