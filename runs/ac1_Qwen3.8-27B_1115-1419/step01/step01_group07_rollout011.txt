    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

        else:

            w_top /= w_top.sum()
            w[sorted_idx] = w_top

            obj_new = obj_val(a_new)

                stall_count = 0

        c = get_conv(a)
        obj = c.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            stall = 0

        # Adaptive k_top: starts large (smooth), decreases to 1 (exact)
        progress = min(1.0, iter_count / 8000.0)
        k_top = max(1, int(300.0 * (1.0 - progress) ** 2.5 + 1.0))
        if iter_count > 12000:
            k_top = 1

        g = compute_subgrad(a, c, k_top)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break

        # Backtracking line search
        improved = False
        eta_trial = max(eta, eta_min)
        best_candidate = None
        best_candidate_obj = obj
        for _ in range(25):
            a_candidate = mirror_step(a, g, eta_trial)
            c_cand = get_conv(a_candidate)
            obj_cand = c_cand.max()
            if obj_cand < best_candidate_obj:
                best_candidate_obj = obj_cand
                best_candidate = a_candidate
            if obj_cand < obj - 1e-14:
                improved = True
                break
            eta_trial *= 0.7
            if eta_trial < eta_min:
                break

        if improved and best_candidate is not None:
            a = best_candidate
            obj = best_candidate_obj
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            eta = min(eta * 1.03, 10.0)
        else:
            stall += 1
            eta = max(eta * 0.4, eta_min)
            if stall > 300:
                # Perturb to escape local minimum
                perturb_type = rng.integers(3)
                if perturb_type == 0:
                    a_pert = a + rng.normal(0, 0.003, n)
                elif perturb_type == 1:
                    # Block perturbation
                    a_pert = a.copy()
                    blk = n // 4
                    sign = 1 if rng.random() > 0.5 else -1
                    a_pert[rng.integers(blk):blk + blk] += sign * 0.005
                else:
                    # Random swap
                    a_pert = a.copy()
                    i1, i2 = rng.integers(0, n, 2)
                    swap_amt = 0.01 * min(a_pert[i1], a_pert[i2]) * 5
                    a_pert[i1] -= swap_amt
                    a_pert[i2] += swap_amt
                a = normalize(a_pert)
                c = get_conv(a)
                obj = c.max()
                stall = 0
                eta = 2.0 / np.sqrt(n)

        iter_count += 1
        total_iters += 1
        if verbose and total_iters % 5000 == 0:
            score_est = 2.0 * n * best_obj
            print(f"  MD iter {total_iters}: obj={obj:.8f}, best={best_obj:.8f}, "
                  f"eta={eta:.4f}, k_top={k_top}, est_score={score_est:.6f}")

    return best_a, best_obj


def run_subgradient_polish(a, time_budget, rng, verbose=True):
    """Projected subgradient with adaptive step size. Returns (best_a, best_obj)."""
    n = len(a)
    a = a.copy()
    c = get_conv(a)
    best_obj = c.max()
    best_a = a.copy()
    deadline = time.monotonic() + time_budget
    step = 0.05
    iter_count = 0

    while time.monotonic() < deadline:
        c = get_conv(a)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Use small active set for sharper gradient
        k_top = min(3, len(c))
        g = compute_subgrad(a, c, k_top)

        a_new = a - step * g
        a_new = project_simplex(a_new)

        c_new = get_conv(a_new)
        obj_new = c_new.max()

        if obj_new < obj - 1e-15:
            a = a_new
            obj = obj_new
            step = min(step * 1.01, 0.15)
        else:
            step *= 0.75
            if step < 1e-7:
                a_pert = a + rng.normal(0, 0.001, n)
                a = normalize(a_pert)
                step = 0.02
                c = get_conv(a)
                obj = c.max()

        iter_count += 1
        if verbose and iter_count % 10000 == 0:
            print(f"  SG iter {iter_count}: obj={obj:.8f}, best={best_obj:.8f}, step={step:.6f}")

    return best_a, best_obj


def lse_smoothing(x, tau, maxiter=200):
    """L-BFGS with LSE smoothing in log-space. Returns optimized x."""
    n = len(x)

    def obj_and_grad(xv):
        xv_s = xv - xv.max()
        e = np.exp(xv_s)
        s = e.sum()
        if s < 1e-300:
            return 1e10, np.zeros_like(xv)
        a = e / s
        c = np.convolve(a, a)
        cmax = c.max()
        shifted = (c - cmax) / tau
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            return cmax, np.zeros_like(xv)
        obj_val = cmax + tau * np.log(ws)
        w_use = w / ws
        grad_a = 2.0 * np.correlate(w_use, a, mode='valid')
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        return obj_val, grad_x

    try:
        result = minimize(obj_and_grad, x, jac=True, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxfun': maxiter * 3})
        return result.x
    except Exception:
        return x


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search function. Minimizes 2*n*max(conv(a))/(sum(a))^2.
    Uses: L-BFGS warm-start -> Mirror descent -> Subgradient polish.
    """
    start_time = time.monotonic()
    rng = np.random.default_rng(777)

    # Load initial sequence
    if 'height_sequence_1' in globals():
        a_init = np.array(height_sequence_1, dtype=np.float64)
    else:
        a_init = np.ones(2048, dtype=np.float64)

    n = len(a_init)
    a = normalize(a_init)
    c = get_conv(a)
    best_obj = c.max()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial: n={n}, obj={best_obj:.8f}, score={best_score:.6f}")

    # === Phase 1: L-BFGS warm polish (up to 300s) ===
    print("\n=== Phase 1: L-BFGS warm-start ===")
    phase1_time = min(300, int(budget_s - 1900))
    p1_deadline = time.monotonic() + phase1_time
    x = np.log(a + 1e-30)
    taus = [8.0 / n, 4.0 / n, 2.0 / n, 1.0 / n, 0.5 / n]
    for tau in taus:
        if time.monotonic() > p1_deadline:
            break
        x = lse_smoothing(x, tau, maxiter=150)
        xv_s = x - x.max()
        e = np.exp(xv_s)
        s = e.sum()
        if s > 1e-300:
            a_tmp = e / s
            c_tmp = get_conv(a_tmp)
            if c_tmp.max() < best_obj:
                best_obj = c_tmp.max()
                best_a = a_tmp.copy()
                a = a_tmp.copy()
                print(f"  L-BFGS tau={tau:.5f}: obj={best_obj:.8f} (score~{2*n*best_obj:.4f})")
    a = best_a.copy()
    print(f"  Phase 1 done: obj={best_obj:.8f}")

    # === Phase 2: Mirror descent (main workhorse, ~1400s) ===
    print(f"\n=== Phase 2: Mirror descent ===")
    elapsed = time.monotonic() - start_time
    phase2_time = min(1400, int(budget_s - elapsed - 700))
    print(f"  Budget: {phase2_time}s")
    a_md, obj_md = run_mirror_descent(a, phase2_time, rng, verbose=True)
    if obj_md < best_obj:
        best_obj = obj_md
        best_a = a_md.copy()
    a = best_a.copy()
    score_check = evaluate_sequence(a.tolist())
    if np.isfinite(score_check):
        print(f"  Post-MD: score={score_check:.6f}")
        if score_check < best_score:
            best_score = score_check
    print(f"  Phase 2 done: obj={best_obj:.8f}")

    # === Phase 3: Subgradient polishing (~400s) ===
    print(f"\n=== Phase 3: Subgradient polish ===")
    elapsed = time.monotonic() - start_time
    phase3_time = min(400, int(budget_s - elapsed - 300))
    print(f"  Budget: {phase3_time}s")
    a_sg, obj_sg = run_subgradient_polish(a, phase3_time, rng, verbose=True)
    if obj_sg < best_obj:
        best_obj = obj_sg
        best_a = a_sg.copy()
    a = best_a.copy()
    score_check = evaluate_sequence(a.tolist())
    if np.isfinite(score_check):
        print(f"  Post-SG: score={score_check:.6f}")
        if score_check < best_score:
            best_score = score_check
    print(f"  Phase 3 done: obj={best_obj:.8f}")

    # === Phase 4: Fine-tuning with alternating k_top (remaining time) ===
    elapsed = time.monotonic() - start_time
    phase4_time = max(0, int(budget_s - elapsed - 150))
    if phase4_time > 30:
        print(f"\n=== Phase 4: Fine polish ({phase4_time}s) ===")
        a = best_a.copy()
        p4_deadline = time.monotonic() + phase4_time
        step = 0.02
        stagnation = 0
        while time.monotonic() < p4_deadline:
            c = get_conv(a)
            obj = c.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 1

            # Alternate between k_top values
            k_top = [1, 2, 3, 5][stagnation % 4]
            g = compute_subgrad(a, c, k_top)
            a_new = a - step * g
            a_new = project_simplex(a_new)
            c_new = get_conv(a_new)
            if c_new.max() < obj:
                a = a_new
                step = min(step * 1.02, 0.1)
            else:
                step *= 0.8
                if step < 5e-6:
                    # Perturb and reset
                    a_pert = a + rng.normal(0, 0.002, n)
                    a = normalize(a_pert)
                    step = 0.01
                    if stagnation > 500:
                        break

        score_check = evaluate_sequence(a.tolist())
        if np.isfinite(score_check) and score_check < best_score:
            best_score = score_check
            best_a = a.copy()
        print(f"  Phase 4 done: best_obj={best_obj:.8f}")

    # === Final validation ===
    final_seq = [float(v) for v in best_a]
    for v in final_seq:
        assert v >= 0.0, f"Negative value: {v}"
    final_score = evaluate_sequence(final_seq)
    print(f"\n{'='*50}")
    print(f"FINAL RESULT: score={final_score:.6f}")
    print(f"Sequence length: {len(final_seq)}")
    print(f"max(conv)={best_obj:.8f}, sum={sum(final_seq):.6f}")
    print(f"{'='*50}")

    if not np.isfinite(final_score):
        print("ERROR: invalid sequence, using fallback")
        fallback = [1.0 / n] * n
        return fallback

    return final_seq
