```python
def _fft_conv(a):
    n = len(a)
    fa = np.fft.rfft(a, n=2 * n - 1)
    return np.fft.irfft(fa * fa, n=2 * n - 1)


def _project(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def _softmax_w(c, tau):
    mc = np.max(c)
    if tau < 1e-10:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    e = np.exp((c - mc) / tau)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return e / s


def _grad_smooth(a, p, n):
    """Gradient of smooth max conv w.r.t. a. p is softmax weights of conv."""
    return 2.0 * np.correlate(p, a, mode='valid')


def _grad_peak(a, k, n):
    """Exact gradient of conv(a,a)[k] w.r.t. a."""
    g = np.zeros(n)
    js = max(0, k - (n - 1))
    je = min(n, k + 1)
    if js < je:
        g[js:je] = 2.0 * a[k - np.arange(js, je)]
    return g


def _obj_val(a, n, c=None):
    if c is None:
        c = _fft_conv(a)
    return 2.0 * float(np.max(c)) / n


def _adam_update(m, v, t, a, g, lr, beta1, beta2, eps, n):
    m = beta1 * m + (1 - beta1) * g
    v = beta2 * v + (1 - beta2) * g * g
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v, _project(a_new, n)


def _smooth_descent(a, n, deadline, tag, tau0=None, lr=0.05, max_iter=500000):
    """Adam-based smooth descent with annealing."""
    if tau0 is None:
        tau0 = n / 2.0
    best_a = a.copy()
    best_o = _obj_val(a, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        t += 1
        c = _fft_conv(a)
        mc = float(np.max(c))
        o = 2.0 * mc / n
        if o < best_o - 1e-10:
            best_o = o
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        p = _softmax_w(c, tau)
        g = _grad_smooth(a, p, n)
        g -= g.mean()  # maintain sum constraint more naturally
        mg = np.max(np.abs(g))
        if mg < 1e-14:
            break
        g = g / mg  # normalize
        m, v, a = _adam_update(m, v, t, a, g, lr, beta1, beta2, eps, n)
        tau = max(tau * (1 - 0.00005), n * 0.002)
        if stall > 5000:
            lr *= 0.5
            stall = 0
            if lr < 1e-6:

    return best_a, best_o


def _hard_refine(a, n, deadline, tag, K=5, max_iter=200000):
    """Multi-peak active set descent with backtracking line search."""
    best_a = a.copy()
    best_o = _obj_val(a, n)
    lr = 0.05
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        c = _fft_conv(a)
        mc = float(np.max(c))
        o = 2.0 * mc / n
        if o < best_o - 1e-10:
            best_o = o
            best_a = a.copy()
        # Find top K peaks
        idx = np.argsort(c)[-K:][::-1]
        vals = c[idx]
        w = np.maximum(vals - c.mean(), 0)
        if w.sum() < 1e-15:
            w = np.ones(K) / K
        else:
            w = w / w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, idx):
            g += wi * _grad_peak(a, int(ki), n)
        g -= g.mean()
        mg = np.max(np.abs(g))
        if mg < 1e-14:
            break
        g = g / mg
        # Backtracking line search
        accepted = False
        for trial in range(10):
            a_try = _project(a - lr * g, n)
            o_try = _obj_val(a_try, n)
            if o_try < o - 1e-12:
                a = a_try
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            # Try smaller
            lr *= 0.1
            if lr < 1e-10:
                break
    return best_a, best_o

    b = a * np.exp(rng.normal(0, sigma, size=n))
```

```python
    """Compute gradient = 2 * correlate(p, a, 'valid') via FFT.
    p has length 2n-1, a has length n.
    Result has length n: result[m] = 2 * sum(0..n-1) a[j] * p[m+j]

    # This is the cross-correlation of p and a
    # corr[k] = sum_j p[j+k] * a[j]  (this is what we want, with k=m)

    A = np.fft.rfft(a, n=size)

    # In FFT domain: ifft(P * conj(A)) gives circular cross-correlation
    # With zero-padding to size >= len(p)+len(a)-1 = 3n-2, we'd get linear.
    # But size = 2n-1 is not enough for full linear cross-correlation.
    
    # Hmm, this is tricky. Let me think differently.
    # 
    # result[m] = sum_{j=0}^{n-1} a[j] * p[m+j]
    # 
    # Let's define q[j] = p[j] for j=0..2n-2 (length 2n-1)
    # result[m] = sum_{j=0}^{n-1} a[j] * q[m+j]
    # 
    # This is (a * q_rev)[n-1+m] where q_rev is q reversed? No...
    # 
    # Actually, (a * q)[k] = sum_j a[j] * q[k-j]
    # We want sum_j a[j] * q[m+j] = sum_j a[j] * q[m-j + 2j]... no that doesn't work.
    #
    # Let's try: sum_j a[j] * q[m+j]
    # Let i = m+j, so j = i-m: sum_i a[i-m] * q[i] for i=m..m+n-1
    # = sum_i a[i-m] * q[i]
    # = (a_rev * q)[m-1 + something]... this is getting confusing.

    # 
    # Define: f[k] = sum_{j=0}^{n-1} a[j] * p[m+j] where m=k
    # 
    # This is the cross-correlation R_{ap}[k] = sum_j a[j+k] * p[j]  (standard definition)
    # Wait no. R_{ap}[k] = sum_j a[j+k] * conj(p[j])
    # Our expression: sum_j a[j] * p[k+j] = sum_j a[j] * conj(p[k+j]) (for real)
    # = R_{pa}[-k] = sum_j p[j-k] * a[j] ... I'm getting confused with conventions.
    
    # Let me just verify with a small example and use the FFT that works.
    
    # Alternative: use the fact that this is a "sliding dot product" and can be
    # computed as the first n samples of the linear convolution of a with reversed p.
    # 
    # conv(a, p_rev)[k] = sum_j a[j] * p_rev[k-j] = sum_j a[j] * p_rev[k-j]
    # p_rev[i] = p[len(p)-1-i] = p[2n-2-i]
    # conv(a, p_rev)[k] = sum_j a[j] * p[2n-2-(k-j)] = sum_j a[j] * p[2n-2-k+j]
    # 
    # We want: sum_j a[j] * p[m+j]
    # So: 2n-2-k+j = m+j => 2n-2-k = m => k = 2n-2-m
    # 
    # So result[m] = conv(a, p_rev)[2n-2-m]
    # 
    # The convolution has length n + (2n-1) - 1 = 3n-2
    # We need indices k = 2n-2-m for m=0..n-1, i.e., k = 2n-2, 2n-3, ..., n-1
    # These are in the range [n-1, 2n-2], which is the middle of the convolution.
    
    # OK this is getting too complicated. Let me just use np.correlate.
    return 2.0 * np.correlate(p, a, mode='valid')
```

```python
"""
FFT-accelerated multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Phases: Adam smooth descent -> multi-peak active set -> diverse perturbation+refine -> final polish.
Key innovations: FFT convolution, Adam optimizer, anti-peak perturbations, population crossover,
and backtracking line search.
"""

import numpy as np
import time


def _conv(a):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    fa = np.fft.rfft(a, n=2 * n - 1)
    return np.fft.irfft(fa * fa, n=2 * n - 1)


def _proj(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def _softmax_w(c, tau):
    """Stable softmax of c with temperature tau."""
    mc = float(np.max(c))
    if tau < 1e-10:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    e = np.exp((c - mc) / tau)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return e / s


def _grad(a, p, n):
    """Gradient of weighted smooth max: 2*corr(p, a, valid). p has length 2n-1."""
    return 2.0 * np.correlate(p, a, mode='valid')


def _grad_peak(a, k, n):
    """Exact gradient of conv(a,a)[k] w.r.t. a (length n)."""
    g = np.zeros(n)
    js = max(0, k - (n - 1))
    je = min(n, k + 1)
    if js < je:
        g[js:je] = 2.0 * a[k - np.arange(js, je)]
    return g


def _obj(a, n, c=None):
    """Objective: 2*max(conv)/n (valid when sum(a)=n)."""
    if c is None:
        c = _conv(a)
    return 2.0 * float(np.max(c)) / n


def _smooth_run(a, n, deadline, tag, tau0, lr, max_it):
    """Adam-based smooth descent with annealing. Returns (best_a, best_obj)."""
    best_a = a.copy()
    best_o = _obj(a, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    tau = tau0
    cur_lr = lr
    for it in range(max_it):
        if time.monotonic() >= deadline:
            break
        t += 1
        c = _conv(a)
        mc = float(np.max(c))
        o = 2.0 * mc / n
        if o < best_o - 1e-12:
            best_o = o
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        p = _softmax_w(c, tau)
        g = _grad(a, p, n)
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        g = g / mg
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        a = _proj(a - cur_lr * m_hat / (np.sqrt(v_hat) + eps), n)
        tau = max(tau * 0.99995, n * 0.003)
        if stall > 4000:
            cur_lr *= 0.5
            stall = 0
            if cur_lr < 1e-7:
                break
    return best_a, best_o


def _hard_run(a, n, deadline, tag, K, max_it):
    """Multi-peak active-set descent with backtracking. Returns (best_a, best_obj)."""
    best_a = a.copy()
    best_o = _obj(a, n)
    lr = 0.08
    for it in range(max_it):
        if time.monotonic() >= deadline:
            break
        c = _conv(a)
        mc = float(np.max(c))
        o = 2.0 * mc / n
        if o < best_o - 1e-12:
            best_o = o
            best_a = a.copy()
        # Top-K peaks
        idx = np.argsort(c)[-K:][::-1]
        vals = c[idx]
        w = np.maximum(vals - c.mean(), 0.0)
        ws = float(w.sum())
        if ws < 1e-15:
            w = np.full(K, 1.0 / K)
        else:
            w = w / ws
        g = np.zeros(n)
        for wi, ki in zip(w, idx):
            g += wi * _grad_peak(a, int(ki), n)
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        g = g / mg
        accepted = False
        for _ in range(12):
            a_try = _proj(a - lr * g, n)
            o_try = _obj(a_try, n)
            if o_try <= o + 1e-14:
                a = a_try
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            lr *= 0.05
            if lr < 1e-12:
                break
        if it % 50000 == 49999:
            lr = max(lr * 1.2, 0.08)
    return best_a, best_o


def _perturb_gauss(a, n, rng, sigma):
    """Multiplicative Gaussian perturbation."""
    b = a * np.exp(rng.normal(0.0, sigma, size=n))
    return _proj(b, n)


def _perturb_block(a, n, rng, sigma, width):
    """Perturb a random contiguous block."""
    b = a.copy()
    s = int(rng.integers(0, max(1, n - width + 1)))
    b[s:s + width] *= np.exp(rng.normal(0.0, sigma, size=width))
    return _proj(b, n)


def _perturb_antipeak(a, n, rng, strength):
    """Gradient-based perturbation targeting convolution peaks."""
    c = _conv(a)
    tau = max(n / 10.0, 1.0)
    p = _softmax_w(c, tau)
    g = _grad(a, p, n)
    mg = float(np.max(np.abs(g)))
    if mg < 1e-14:
        return a.copy()
    # Apply to random fraction of coordinates
    fraction = float(rng.uniform(0.2, 0.7))
    mask = rng.random(n) < fraction
    b = a.copy()
    b[mask] *= np.exp(-strength * g[mask] / mg)
    return _proj(b, n)


def _perturb_sparse(a, n, rng, magnitude):
    """Add sparse positive spikes."""
    b = a.copy()
    n_spikes = max(1, n // 200)
    for _ in range(n_spikes):
        idx = int(rng.integers(0, n))
        b[idx] += magnitude
    return _proj(b, n)


def _cross_seqs(a1, a2, n, rng):
    """Random mask crossover."""
    mask = rng.random(n) > 0.5
    b = np.where(mask, a1, a2)
    return _proj(b, n)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2n*max(conv(a,a))/sum(a)^2. Uses FFT, Adam, multi-peak active set,
    and diverse structured perturbations with population maintenance.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 5.0
    n = 3809
    S = float(n)

    # --- Initialization ---
    a = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n and np.all(np.isfinite(parent)):
            a = np.clip(parent, 0.0, 1000.0)
    except Exception:
        pass
    if a is None:
        a = np.ones(n, dtype=np.float64)
    a = _proj(a, S)

    best_a = a.copy()
    best_o = _obj(a, n)
    print(f"Init: n={n} obj={best_o:.6f}")

    # --- Phase 1: Adam smooth descent (200s) ---
    p1_end = min(t0 + 200.0, deadline - 10.0)
    best_a, best_o = _smooth_run(a, n, p1_end, "s1", tau0=n / 2.0, lr=0.10, max_it=1000000)
    print(f"P1: obj={best_o:.6f} t={time.monotonic()-t0:.0f}s")

    # --- Phase 2: Second smooth pass with lower tau (150s) ---
    p2_end = min(t0 + 350.0, deadline - 10.0)
    best_a, best_o = _smooth_run(best_a, n, p2_end, "s2", tau0=n / 8.0, lr=0.05, max_it=1000000)
    print(f"P2: obj={best_o:.6f} t={time.monotonic()-t0:.0f}s")

    # --- Phase 3: Multi-peak hard refine (300s) ---
    p3_end = min(t0 + 650.0, deadline - 10.0)
    best_a, best_o = _hard_run(best_a, n, p3_end, "h3", K=5, max_it=500000)
    print(f"P3: obj={best_o:.6f} t={time.monotonic()-t0:.0f}s")

    # --- Phase 4: Perturbation + refinement loop (remaining - 300s) ---
    pop = [best_a.copy()]
    attempt = 0
    phase4_end = deadline - 300.0

    while time.monotonic() < phase4_end:
        attempt += 1
        # Progressively increase perturbation strength
        if attempt <= 15:
            sigma = float(rng.uniform(0.04, 0.12))
        elif attempt <= 40:
            sigma = float(rng.uniform(0.08, 0.25))
        elif attempt <= 70:
            sigma = float(rng.uniform(0.12, 0.40))
        else:
            sigma = float(rng.uniform(0.20, 0.60))

        rtype = int(rng.integers(0, 6))
        if rtype == 0:
            a_p = _perturb_gauss(best_a, n, rng, sigma)
        elif rtype == 1:
            w = max(3, int(n * float(rng.choice([0.005, 0.01, 0.02, 0.05, 0.1, 0.2]))))
            a_p = _perturb_block(best_a, n, rng, sigma, w)
        elif rtype == 2:
            a_p = _perturb_antipeak(best_a, n, rng, sigma * 3)
        elif rtype == 3:
            a_p = _perturb_sparse(best_a, n, rng, sigma * 1.5)
        elif rtype == 4 and len(pop) > 1:
            a2 = pop[int(rng.integers(0, len(pop)))]
            a_p = _cross_seqs(best_a, a2, n, rng)
        else:
            # Fresh restart with slight structure
            base = np.ones(n) * float(rng.uniform(0.7, 1.3))
            base += rng.normal(0, 0.2, size=n)
            a_p = _proj(base, S)

        # Quick smooth refine (15-20s)
        sub1 = min(time.monotonic() + 18.0, phase4_end - 5.0)
        a_q, o_q = _smooth_run(a_p, n, sub1, f"s{attempt}", tau0=n / 5.0, lr=0.06, max_it=50000)

        # Quick hard refine (10s)
        sub2 = min(time.monotonic() + 10.0, phase4_end - 3.0)
        a_q2, o_q2 = _hard_run(a_q, n, sub2, f"h{attempt}", K=3, max_it=30000)

        if o_q2 < best_o:
            imp = best_o - o_q2
            best_o = o_q2
            best_a = a_q2.copy()
            pop.insert(0, a_q2.copy())
            if len(pop) > 5:
                pop.pop()
            print(f"  [{attempt}] BEST obj={best_o:.6f} (+{imp:.2e}) t={time.monotonic()-t0:.0f}s")
        elif o_q2 < best_o + 0.003 and len(pop) < 5:
            pop.append(a_q2.copy())

        # Re-optimize a population member occasionally
        if attempt % 8 == 0 and len(pop) > 1:
            idx = int(rng.integers(1, len(pop)))
            sub3 = min(time.monotonic() + 15.0, phase4_end - 3.0)
            a_r, o_r = _hard_run(pop[idx], n, sub3, f"r{attempt}", K=5, max_it=20000)
            if o_r < best_o:
                best_o = o_r
                best_a = a_r
                pop[0] = a_r.copy()
                print(f"  [{attempt}] PopReopt obj={best_o:.6f} t={time.monotonic()-t0:.0f}s")

    # --- Phase 5: Final long refinement (300s) ---
    p5_end = deadline - 5.0
    best_a, best_o = _hard_run(best_a, n, p5_end, "final", K=7, max_it=1000000)
    print(f"P5: obj={best_o:.6f} t={time.monotonic()-t0:.0f}s")

    # --- Final smooth polish if time remains ---
    if time.monotonic() < p5_end - 20.0:
        best_a, best_o = _smooth_run(best_a, n, p5_end - 3.0, "polish", tau0=n / 10.0, lr=0.02, max_it=100000)
        print(f"Polish: obj={best_o:.6f} t={time.monotonic()-t0:.0f}s")

    # --- Validate and return ---
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')
    if not np.isfinite(sc):
        result = [1.0] * n
        try:
            sc = float(evaluate_sequence(result))
        except Exception:
            sc = 2.0
    print(f"FINAL: internal_obj={best_o:.6f} evaluated={sc:.6f} n={n} time={time.monotonic()-t0:.0f}s")
    return result
```
