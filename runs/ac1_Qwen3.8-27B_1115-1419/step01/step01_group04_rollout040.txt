def fast_convolution(a):

    n = a.size

    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def normalize_sequence(a, target_sum):

            return a
