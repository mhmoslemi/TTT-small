"""
FFT-accelerated multi-phase optimizer with multi-peak tracking, adaptive
temperature scheduling, and structural diversification for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multiple n values, diverse initializations, and a smooth->hard->perturb
pipeline with time-bounded phases.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def _nfft_for(n):
    """Smallest power-of-2 FFT size for full self-convolution of length n."""
    need = 2 * n - 1
    x = 1
    while x < need:
        x <<= 1
    return x


def _conv_fft(a, nfft):
    """Self-convolution via FFT. Returns length 2n-1."""
    A = rfft(a, n=nfft)
    return irfft(A * A, n=nfft)[:2 * len(a) - 1]


def _softmax_stable(v, tau):
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp((v - m) / tau)
    s = float(e.sum())
    if s < 1e-300:
        return np.full_like(v, 1.0 / len(v), dtype=np.float64)
    return e / s


def _project(a, target_sum):
    """Project onto non-negative simplex with given sum."""
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a), dtype=np.float64)
    return a * (target_sum / s)


def _obj(a):
    """Objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    nfft = _nfft_for(n)
    c = _conv_fft(a, nfft)
    return 2.0 * n * float(np.max(c)) / (S * S)


def _obj_cached(a, nfft):
    """Objective with precomputed nfft, assumes sum ~ n."""
    n = len(a)
    c = _conv_fft(a, nfft)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def _grad_smooth(a, c, tau, n):
    """Gradient of softmax-smoothed max of conv w.r.t. a."""
    p = _softmax_stable(c, tau)
    # grad[i] = 2 * sum_j p[i+j] * a[j] for j=0..n-1, i=0..n-1
    # This is 2 * np.correlate(p, a, 'valid')
    g = 2.0 * np.correlate(p, a, mode='valid')
    g -= g.mean()
    return g


def _grad_hard(a, k, n):
    """Exact gradient of conv(a,a)[k] w.r.t. a."""
    g = np.zeros(n, dtype=np.float64)
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:
        idx = np.arange(js, je)
        g[idx] = 2.0 * a[k - idx]
        g -= g.mean()
    return g


def _top_k_grad(a, c, n, K=3):
    """Weighted gradient from top-K peaks (soft active set)."""
    idx = np.argpartition(c, -K)[-K:]
    vals = c[idx]
    w = np.exp(vals / (0.5 * float(np.std(c) + 1e-10)))
    w /= w.sum()
    g = np.zeros(n, dtype=np.float64)
    for wj, k in zip(w, idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k) + 1
        if js < je:
            idx2 = np.arange(js, je)
            g[js:je] += wj * 2.0 * a[k - idx2]
    g -= g.mean()
    return g


def _smooth_opt(a, n, nfft, deadline, tag="s", it_limit=500000):
    """Entropy-smoothed gradient descent with adaptive step and temperature."""
    tau = n / 3.0
    tau_min = n * 0.003
    eta = 0.06
    best = a.copy()
    best_o = _obj_cached(a, nfft)
    stall = 0
    t0 = time.monotonic()
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _conv_fft(a, nfft)
        o = 2.0 * float(np.max(c)) / float(a.sum())
        if o < best_o - 1e-9:
            best_o = o
            best = a.copy()
            stall = 0
        else:
            stall += 1
        g = _grad_smooth(a, c, tau, n)
        mg = float(np.max(np.abs(g)))
        if mg < 1e-13:
            break
        step = eta / mg
        a = _project(a - step * g, n)
        # Adaptive temperature: slow decay early, faster later
        decay = 0.99998 if it < 5000 else 0.99994
        tau = max(tau * decay, tau_min)
        if stall > 6000:
            eta = max(eta * 0.5, 1e-7)
            stall = 0
        if stall > 20000:
            break
    return best, best_o


def _hard_opt(a, n, nfft, deadline, tag="h", it_limit=500000):
    """Hard + multi-peak active-set descent."""
    best = a.copy()
    best_o = _obj_cached(a, nfft)
    eta = 0.03
    use_topk = False
    K = 3
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _conv_fft(a, nfft)
        o = 2.0 * float(np.max(c)) / float(a.sum())
        if o < best_o - 1e-10:
            best_o = o
            best = a.copy()
            if it % 50000 == 0:
                print(f"    [{tag}] it={it} obj={o:.6f}")
        # Switch to top-K after initial convergence
        if it > 5000 and not use_topk:
            use_topk = True
        if use_topk:
            g = _top_k_grad(a, c, n, K)
        else:
            k = int(np.argmax(c))
            g = _grad_hard(a, k, n)
        mg = float(np.max(np.abs(g)))
        if mg < 1e-12:
            break
        # Adaptive step
        step = eta / mg
        a_new = _project(a - step * g, n)
        # Backtrack: verify improvement
        c_new = _conv_fft(a_new, nfft)
        o_new = 2.0 * float(np.max(c_new)) / float(a_new.sum())
        if o_new > o * 1.02 + 1e-6:
            # Shrink step
            a_new = _project(a - step * 0.25 * g, n)
            c_new = _conv_fft(a_new, nfft)
            o_new = 2.0 * float(np.max(c_new)) / float(a_new.sum())
        a = a_new
        if it > 50000 and it % 50000 == 0:
            eta = max(eta * 0.85, 1e-8)
    return best, best_o


def _diversify(a, rng, strength=0.3):
    """Apply a structural perturbation and re-project."""
    n = len(a)
    b = a.copy()
    r = rng.random()
    if r < 0.3:
        # Block multiply
        w = max(3, int(rng.integers(n // 50, n // 5)))
        s = int(rng.integers(0, n - w))
        b[s:s+w] *= np.exp(rng.normal(0.0, strength, size=w))
    elif r < 0.6:
        # Block shift (move mass)
        w = max(3, int(rng.integers(n // 60, n // 10)))
        s1 = int(rng.integers(0, n - w))
        s2 = int(rng.integers(0, n - w))
        amt = b[s1:s+w].sum() * strength
        b[s1:s+w] -= amt / w
        b[s2:s2+w] += amt / w
    elif r < 0.8:
        # Gaussian noise block
        w = max(5, n // 8)
        s = int(rng.integers(0, n - w))
        b[s:s+w] += rng.normal(0.0, strength * 0.5 * float(b[s:s+w].mean()), size=w)
    else:
        # Swap sub-blocks
        w = max(3, n // 100)
        s1 = int(rng.integers(0, n - w))
        s2 = int(rng.integers(0, n - w))
        b[s1:s1+w], b[s2:s2+w] = b[s2:s2+w].copy(), b[s1:s1+w].copy()
    return _project(b, n)


def _quick_refine(a, n, nfft, time_budget=15.0):
    """Quick smooth + hard refinement."""
    deadline = time.monotonic() + time_budget * 0.6
    a2, o2 = _smooth_opt(a, n, nfft, deadline, "qs", it_limit=50000)
    deadline2 = time.monotonic() + time_budget * 0.4
    a2, o2 = _hard_opt(a2, n, nfft, deadline2, "qh", it_limit=50000)
    return a2, o2
