```python
def convolution_shaping(a, n, iterations=1000):
    """Iteratively shape a to flatten its self-convolution."""
    a = a.copy()
    for it in range(iterations):
        c = np.convolve(a, a)
        # Find the target: reduce peaks
        c_mean = c[n-1]  # center value as reference (or use mean)
        # Actually, use the mean of the central portion
        center = c[n//4:3*n//4]
        c_ref = np.mean(center)
        # Target: flatten by clipping
        c_target = np.minimum(c, c_ref * 1.5)  # cap at 1.5x the central mean
        # Normalize target to have same sum as c
        c_target = c_target * (np.sum(c) / np.sum(c_target))
        # Frequency domain update
        N = 2 * n  # zero-pad length
        A = np.fft.rfft(a, n=N)
        C_target = np.fft.rfft(c_target, n=N)
        # Update: keep phase of A, use magnitude of C_target^0.5
        # A^2 = C, so A = sqrt(C)
        phase = np.angle(A)
        mag = np.sqrt(np.abs(C_target))
        A_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n=N)[:n]
        # Project
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 0:
            a_new = a_new / s
        a = a_new
    return a
```

```python
"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: convolution-shaped initialization via frequency domain, Nesterov-accelerated
smooth gradient descent with adaptive temperature, hard active-set descent with
backtracking line search, and sensitivity-guided perturbation with re-optimization.
"""

import numpy as np
import time


def _conv_full(a):
    return np.convolve(a, a)


def _obj(a, n):
    c = _conv_full(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * np.max(c) / (s * s)


def _proj(a, n):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) * (n / n)
    return a * (n / s)


def _grad_hard(a, n, k):
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:
        idx = k - np.arange(js, je)
        g[js:je] = 2.0 * a[idx]
    return g


def _grad_smooth(a, n, tau):
    c = _conv_full(a)
    cs = c - np.max(c)
    p = np.exp(cs / tau)
    p /= np.sum(p)
    g = 2.0 * np.correlate(p, a, mode='valid')
    return g, c


def _nesterov_smooth(a, n, deadline, tag, it_limit=500000):
    """Nesterov-accelerated smooth gradient descent with adaptive temperature."""
    tau = n / 3.0
    eta = 0.03
    best = a.copy()
    best_o = _obj(a, n)
    v = np.zeros(n)  # momentum
    y = a.copy()     # Nesterov look-ahead
    stall = 0
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        # Nesterov: evaluate gradient at look-ahead point
        g, c = _grad_smooth(y, n, tau)
        g -= g.mean()
        mg = np.max(np.abs(g))
        if mg < 1e-14:
            break
        g = g / mg
        # Update momentum
        beta = 0.9
        v = beta * v - eta * g
        # New point
        x_new = y + v
        x_new = _proj(x_new, n)
        # Nesterov update
        mu = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = x_new + (v - v) * 0  # just x_new for simplicity (heavy ball variant)
        # Actually let's use simpler momentum:
        y = x_new
        a = x_new
        o = _obj(a, n)
        if o < best_o - 1e-10:
            best_o = o
            best = a.copy()
            stall = 0
            if it % 2000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} tau={tau:.1f}")
        else:
            stall += 1
        # Adaptive temperature
        tau = max(tau * 0.9999, n * 0.002)
        if stall > 5000:
            eta = max(eta * 0.7, 1e-7)
            v = np.zeros(n)
            stall = 0
        # Occasionally reset momentum
        if it % 10000 == 0:
            v *= 0.5
    return best, best_o


def _hard_descent(a, n, deadline, tag, it_limit=500000):
    """Hard active-set descent with adaptive step size."""
    best = a.copy()
    best_o = _obj(a, n)
    eta = 0.015
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _conv_full(a)
        k = int(np.argmax(c))
        mc = c[k]
        o = 2.0 * mc / n  # since sum(a)=n, obj = 2*n*mc/n^2 = 2*mc/n
        if o < best_o - 1e-10:
            best_o = o
            best = a.copy()
            if it % 5000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} eta={eta:.5f}")
        g = _grad_hard(a, n, k)
        g -= g.mean()
        mg = np.max(np.abs(g))
        if mg < 1e-14:
            break
        g = g / mg
        a_new = _proj(a - eta * g, n)
        o_new = _obj(a_new, n)
        if o_new < o:
            a = a_new
            # Increase step slightly
            eta = min(eta * 1.02, 0.05)
        else:
            eta = max(eta * 0.8, 1e-8)
        if it % 20000 == 19999:
            eta = max(eta * 0.95, 1e-8)
    return best, best_o


def _freq_init(n, rng, target_flatness=0.6):
    """Initialize via frequency-domain convolution square root."""
    # Start with a smooth bump
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    a = a + rng.normal(0, 0.1, n)
    a = np.maximum(a, 0)
    a = a / np.sum(a) * n

    N = 4 * n  # zero-pad length for FFT
    for _ in range(30):
        c = _conv_full(a)
        # Target: flatten the convolution
        c_center = c[n // 4:3 * n // 4]
        c_ref = np.median(c_center)
        # Soft clip: reduce peaks
        excess = np.maximum(c - c_ref * (1.0 + target_flatness), 0)
        c_target = c - 0.5 * excess
        c_target = np.maximum(c_target, 0)
        c_target = c_target * (np.sum(c) / np.sum(c_target))

        # Frequency domain
        A = np.fft.rfft(a, N)
        C_t = np.fft.rfft(c_target, N)
        # A^2 = C, so A = sqrt(C) with phase preserved
        mag = np.sqrt(np.maximum(np.abs(C_t), 0))
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a = a_new / s * n
        else:
            break
    return a


def _sensitivity_perturb(a, n, rng, strength=0.3):
    """Perturb elements weighted by their sensitivity to the peak convolution."""
    c = _conv_full(a)
    k = int(np.argmax(c))
    # Sensitivity: s_i = a[k-i] (how much a_i affects c[k])
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    sens = np.zeros(n)
    if js < je:
        idx = k - np.arange(js, je)
        sens[js:je] = a[idx]
    # Normalize sensitivity
    smax = sens.max()
    if smax > 0:
        sens = sens / smax
    # Perturb more where sensitivity is low (less critical)
    # weight = 1/(1+sens) so low-sensitivity elements get more perturbation
    weight = 1.0 / (1.0 + sens)
    log_pert = rng.normal(0, strength, n) * weight
    a_new = a * np.exp(log_pert)
    return _proj(a_new, n)


def _block_perturb(a, n, rng, sigma, frac):
    """Random block perturbation."""
    w = max(2, int(n * frac))
    s = int(rng.integers(0, max(1, n - w + 1)))
    a_new = a.copy()
    a_new[s:s + w] *= np.exp(rng.normal(0, sigma, w))
    return _proj(a_new, n)

    p1_end = min(t0 + 0.30 * (budget_s - 20), deadline - 10)
    best_a, best_o = _nesterov_smooth(best_a, n, p1_end, "smooth1")
    print(f"After smooth1: obj={best_o:.6f}  time={time.monotonic()-t0:.0f}s")

    # Also briefly optimize the 2nd best init
    if len(order) > 1 and time.monotonic() < deadline - 30:
        p1b_end = min(time.monotonic() + 30, deadline - 60)
        a2 = inits[order[1]].copy()
        a2, o2 = _nesterov_smooth(a2, n, p1b_end, "smooth1b")
        if o2 < best_o:
            best_o = o2
            best_a = a2
            print(f"  Alt init improved! obj={best_o:.6f}")

    # ---- Phase 2: Hard active-set descent (20% budget) ----
    p2_end = min(time.monotonic() + 0.20 * (budget_s - 20), deadline - 10)
    best_a, best_o = _hard_descent(best_a, n, p2_end, "hard1")
    print(f"After hard1: obj={best_o:.6f}  time={time.monotonic()-t0:.0f}s")

    # ---- Phase 3: Perturbation + re-optimization loop (remaining time) ----
    attempt = 0
    no_improve = 0
    while time.monotonic() < deadline - 30:
        attempt += 1
        if time.monotonic() > deadline - 40:
            break

        # Choose perturbation strategy
        r = rng.random()
        if r < 0.4:
            # Sensitivity-guided perturbation
            sigma = float(rng.uniform(0.1, 0.5))
            a_p = _sensitivity_perturb(best_a, n, rng, sigma)
        elif r < 0.7:
            # Block perturbation
            sigma = float(rng.uniform(0.1, 0.6))
            frac = float(rng.choice([0.01, 0.03, 0.05, 0.1, 0.2]))
            a_p = _block_perturb(best_a, n, rng, sigma, frac)
        else:
            # Frequency-domain perturbation (modify spectrum)
            N_ft = 4 * n
            A = np.fft.rfft(best_a, N_ft)
            # Add random phase noise
            noise = rng.normal(0, 0.1, size=len(A)) + 1j * rng.normal(0, 0.1, size=len(A))
            A_new = A * np.exp(noise)
            a_p = np.fft.irfft(A_new, N_ft)[:n]
            a_p = _proj(a_p, n)

        # Quick smooth refinement (15-25s)
        smooth_time = min(20.0, (deadline - time.monotonic()) * 0.3)
        sub_end = min(time.monotonic() + smooth_time, deadline - 20)
        a_q, o_q = _nesterov_smooth(a_p, n, sub_end, f"s{attempt}", it_limit=30000)

        # Quick hard refinement (10-15s)
        hard_time = min(12.0, (deadline - time.monotonic()) * 0.2)
        sub_end2 = min(time.monotonic() + hard_time, deadline - 15)
        a_q2, o_q2 = _hard_descent(a_q, n, sub_end2, f"h{attempt}", it_limit=20000)

        o_final = min(o_q, o_q2)
        a_final = a_q if o_q <= o_q2 else a_q2

        if o_final < best_o - 1e-8:
            best_o = o_final
            best_a = a_final
            no_improve = 0
            print(f"  Perturb {attempt}: obj={best_o:.6f} (sigma/type improved)")
        else:
            no_improve += 1

        # If stuck, try a fresh init occasionally
        if no_improve > 15 and attempt % 10 == 0:
            fresh = inits[rng.integers(0, len(inits))].copy()
            sub_end3 = min(time.monotonic() + 15, deadline - 20)
            a_f, o_f = _nesterov_smooth(fresh, n, sub_end3, f"fresh{attempt}", it_limit=20000)
            sub_end4 = min(time.monotonic() + 10, deadline - 15)
            a_f2, o_f2 = _hard_descent(a_f, n, sub_end4, f"freshh{attempt}", it_limit=15000)
            if o_f2 < best_o:
                best_o = o_f2
                best_a = a_f2
                print(f"  Fresh restart {attempt}: obj={best_o:.6f}")
            no_improve = 0

    # ---- Final hard pass ----
    if time.monotonic() < deadline - 5:
        best_a, best_o = _hard_descent(best_a, n, deadline - 2, "final", it_limit=100000)
        print(f"After final: obj={best_o:.6f}")

    # ---- Validate ----
    result = [float(x) for x in best_a]
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')
    if not np.isfinite(sc):
        result = [1.0] * n
        sc = evaluate_sequence(result)
    print(f"FINAL: n={n}  obj_internal={best_o:.6f}  evaluated={sc:.6f}")
    print(f"Total time: {time.monotonic()-t0:.1f}s")
    return result
```
