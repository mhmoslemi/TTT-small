def exact_subgradient_step(a, n_active=40, step=0.01):
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

"""
Optimize step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based convolution, log-softmax parameterization, LSE smoothing,
L-BFGS optimization with multiple asymmetric restarts, and subgradient polishing.
Key insight: asymmetric sequences can achieve scores below 2 by flattening
the convolution peak relative to the (sum)^2/n baseline.
"""

import numpy as np
import time
from scipy.optimize import minimize


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def fft_self_convolve(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    conv_len = 2 * n - 1
    size = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:conv_len]


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    c = fft_self_convolve(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return float(2.0 * n * c.max() / (s * s))


def smooth_obj(x, tau):
    """LSE-smoothed objective in log-param space."""
    a = softmax(x)
    c = fft_self_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    lse = c_max + tau * np.log(np.sum(np.exp(shifted)))
    n = len(a)
    return 2.0 * n * lse  # sum(a) = 1 by softmax


def smooth_grad(x, tau):
    """Gradient of smoothed objective w.r.t. x (log-param)."""
    n = len(x)
    a = softmax(x)
    c = fft_self_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    w = np.exp(shifted)
    w /= w.sum()
    
    # grad_c[i] = sum_k w[k] * (a[k-i] + a[i-k]) with a[·]=0 outside [0,n-1]
    # Part 1: sum_k w[k]*a[k-i] = correlate(w, a, 'valid')[i]
    # Part 2: sum_k w[k]*a[i-k] = convolve(w, a, 'full')[i] for i in [0, n-1]
    
    # Using FFT for efficiency
    size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    
    # Part 1: cross-correlation of w and a
    # correlate(w, a)[i] = sum_j w[j+i]*a[j]
    # = ifft(fft(w) * conj(fft(a))) appropriately indexed
    corr = np.fft.irfft(W * np.conj(A), size)
    # correlate(w,a,'full') has length len(w)+len(a)-1 = 3n-1
    # 'valid' starting position: we want correlate(w,a,'valid') which is length n
    # 'full' = ['anti-valid' | 'valid' | 'post-valid']

    part2 = conv[:n]
    
    grad_a = part1 + part2
    
    # Chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

"""
Optimize step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based convolution, log-softmax parameterization, LSE smoothing,
L-BFGS with multiple asymmetric restarts at varying lengths, and
subgradient polishing. Key insight: asymmetric sequences flatten the
convolution peak below the (sum)^2/n baseline, achieving scores < 2.
"""

import numpy as np
import time
from scipy.optimize import minimize


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def fft_self_conv(a):
    """Full self-convolution using FFT. Returns array of length 2n-1."""
    n = len(a)
    conv_len = 2 * n - 1
    size = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:conv_len]


def grad_smoothed_max(a, tau):
    """
    Compute gradient of LSE-smoothed max(a*a) w.r.t. a.
    a has length n, returns grad_a of length n.
    """
    n = len(a)
    c = fft_self_conv(a)
    conv_len = len(c)  # 2n-1
    c_max = c.max()
    shifted = (c - c_max) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.ones(n) / n
    w /= w_sum

    # grad_a[i] = sum_k w[k]*(a[k-i] + a[i-k])
    # Part 1: correlate(w, a, 'valid') - length n
    # Part 2: first n elements of convolve(w, a, 'full')
    size = 1 << (conv_len + n - 1).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)

    # Cross-correlation: ifft(fft(w) * conj(fft(a)))
    corr = np.fft.irfft(W * np.conj(A), size)
    # 'valid' part starts at index n-1, length n
    part1 = corr[n - 1:n - 1 + n]

    # Convolution: ifft(fft(w) * fft(a))
    conv = np.fft.irfft(W * A, size)
    part2 = conv[:n]

    grad_a = part1 + part2
    return grad_a


def smooth_obj_fn(x, tau):
    """Smoothed objective in log-param space. sum(softmax(x))=1."""
    a = softmax_fn(x)
    c = fft_self_conv(a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    lse = c_max + tau * np.log(np.sum(np.exp(shifted)))
    n = len(a)
    return 2.0 * n * lse


def smooth_grad_fn(x, tau):
    """Gradient of smoothed obj w.r.t. x in log-param space."""
    n = len(x)
    a = softmax_fn(x)
    grad_a = grad_smoothed_max(a, tau)
    # Chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x


def subgradient_step(a, step, n_top=30):
    """Projected subgradient step using top peaks of convolution."""
    n = len(a)
    c = fft_self_conv(a)
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_top, len(c))]

    # Compute subgradient from active set using FFT
    # For each active k, the subgradient contribution at position i is a[k-i] + a[i-k]
    # We can compute this as: build a "gradient convolution"
    grad_a = np.zeros(n)
    for k in active:
        # a[k-i] for valid i: this is a reversed and shifted version
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # Contribution to grad_a[lo:hi+1] from a[k-i]
        segment = a[k - hi:k - lo + 1][::-1]
        grad_a[lo:hi + 1] += 2.0 * segment
    grad_a /= len(active)

    a_new = a - step * grad_a
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def run_lbfgs_for_n(n, time_budget, rng, start_a=None):
    """
    Run L-BFGS optimization for a given sequence length n.
    Returns (best_a, best_score).
    """
    deadline = time.monotonic() + time_budget
    best_score = float('inf')
    best_a = None

    # Generate initializations
    inits = []

    if start_a is not None and len(start_a) == n:
        a = start_a.copy()
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        inits.append(('start', a))

    # Linear ramp (asymmetric - key for beating 2)
    for beta in [1.0, 2.0, 3.0]:
        x = np.ones(n)
        for i in range(n):
            x[i] = 1.0 + beta * (1.0 - i / (n - 1))
        x = np.maximum(x, 1e-8)
        inits.append(('ramp', x / x.sum()))

    # Exponential decay
    for alpha in [1.0, 2.0, 4.0]:
        x = np.exp(-alpha * np.linspace(0, 1, n))
        inits.append(('exp', x / x.sum()))

    # Cosine taper (asymmetric)
    x = np.cos(np.pi * np.linspace(0, 0.7, n)) ** 2
    x = np.maximum(x, 1e-8)

    x = np.maximum(x, 1e-8)
    inits.append(('raised_cos', x / x.sum()))

    # Quadratic (asymmetric)
    t = np.linspace(0, 1, n)
    x = (1.0 - t) ** 2 + 0.1
    inits.append(('quad', x / x.sum()))

    # Random Dirichlet (asymmetric concentration)
    for conc in [0.3, 0.7, 1.5]:
        x = rng.dirichlet(np.ones(n) * conc)
        inits.append(('dirichlet', x))

    # Time per init
    time_per_init = max(10.0, time_budget / (len(inits) + 1) * 0.7)

    for idx, (name, a_init) in enumerate(inits):
        if time.monotonic() > deadline - 30:

        x = np.log(a_init)

        # Scale tau appropriately

        taus = [base_tau * (0.5 ** i) for i in range(n_tau)]

            if time.monotonic() > deadline - 30:

                pass

                    args=(tau,),

                x = result.x
            except Exception:
                break

            a = softmax_fn(x)
            score = compute_obj_direct(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if time.monotonic() > deadline - 30:
            break

    return best_a, best_score


def compute_obj_direct(a):
    """Compute the objective directly (for final evaluation)."""
    n = len(a)
    c = fft_self_conv(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * c.max() / (s * s))


def polish_subgradient(a, time_budget, rng):
    """Polish a sequence using projected subgradient steps."""
    deadline = time.monotonic() + time_budget
    best_score = compute_obj_direct(a)
    best_a = a.copy()
    step = 0.05
    n = len(a)
    stuck_count = 0

    while time.monotonic() < deadline:
        a = subgradient_step(a, step, n_top=min(50, n // 20))
        score = compute_obj_direct(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 50:
                # Try perturbation
                a_pert = a + rng.normal(0, 0.01, n)
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
                score_p = compute_obj_direct(a_pert)
                if score_p < best_score:
                    best_score = score_p
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    stuck_count = 0
                else:
                    a = best_a.copy()

        step *= 0.999
        if step < 0.001:
            step = 0.003

    return best_a, best_score


def upsample(a, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 1e-10)
    a_new /= a_new.sum()
    return a_new


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search: find sequence minimizing 2*n*max(conv)/(sum)^2.
    Strategy: multi-length L-BFGS with asymmetric inits + subgradient polish.
    """
    total_deadline = time.monotonic() + budget_s - 30
    rng = np.random.default_rng(42)
    best_score = float('inf')
    best_a = None

    # Try to use parent sequence
    parent_n = 4806
    if 'height_sequence_1' in globals():
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 1e-15)
        parent = parent / parent.sum()
        parent_n = len(parent)
        print(f"Using parent sequence: n={parent_n}")
        parent_score = compute_obj_direct(parent)
        print(f"Parent score: {parent_score:.6f}")
        best_score = parent_score
        best_a = parent.copy()
    else:
        print("No parent sequence available, starting fresh.")
        parent = None

    # Try multiple lengths
    # Start with lengths around the parent, plus some larger ones
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(parent_n)
    lengths_to_try.extend([4096, 6144, 8192, 12288, 16384])
    # Remove duplicates, sort
    lengths_to_try = sorted(set(lengths_to_try))

    # Time allocation: more time for promising lengths
    # First pass: quick evaluation of each length
    time_per_length = max(60, (budget_s - 300) / len(lengths_to_try))

    print(f"\nTrying lengths: {lengths_to_try}")
    print(f"Time per length: {time_per_length:.0f}s")

    for n in lengths_to_try:
        if time.monotonic() > total_deadline:
            break

        print(f"\n--- n = {n} ---")
        phase_start = time.monotonic()
        time_left = min(time_per_length, total_deadline - phase_start)

        # Get starting point: upsample parent if available
        start_a = None
        if parent is not None and n != len(parent):
            start_a = upsample(parent, n)
            s = compute_obj_direct(start_a)
            print(f"  Upsampled parent to n={n}: score={s:.6f}")
            if s < best_score:
                best_score = s
                best_a = start_a.copy()
        elif parent is not None and n == len(parent):
            start_a = parent.copy()

        # Run L-BFGS
        a, score = run_lbfgs_for_n(n, time_left, rng, start_a)
        if a is not None and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  L-BFGS improved to: {score:.6f}")
        elif a is not None:
            print(f"  L-BFGS best: {score:.6f} (not an improvement)")

    # Final polishing phase
    remaining = max(60, total_deadline - time.monotonic())
    print(f"\n=== Final polishing: {remaining:.0f}s remaining, best={best_score:.6f} ===")

    if best_a is not None:
        # Split time: 60% subgradient, 40% more L-BFGS from best
        polish_time = remaining * 0.6
        lbfgs_time = remaining * 0.4

        # Subgradient polishing
        a, score = polish_subgradient(best_a.copy(), polish_time, rng)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  After subgradient: {score:.6f}")

        # One more L-BFGS round from the polished point
        if lbfgs_time > 60:
            n = len(best_a)
            a2, score2 = run_lbfgs_for_n(n, lbfgs_time, rng, best_a)
            if a2 is not None and score2 < best_score:
                best_score = score2
                best_a = a2.copy()
                print(f"  After final L-BFGS: {score:.6f}")

            # Quick final subgradient
            a3, score3 = polish_subgradient(best_a.copy(), 30, rng)
            if score3 < best_score:
                best_score = score3
                best_a = a3.copy()

    # Verify with the actual evaluator
    final_score = evaluate_sequence([float(v) for v in best_a])
    print(f"\nFinal score (evaluator): {final_score:.6f}")
    print(f"Sequence length: {len(best_a)}")
    print(f"Sum: {best_a.sum():.6f}, Max: {best_a.max():.6f}")

    if not np.isfinite(final_score) or final_score < best_score - 0.01:
        # Fallback to uniform if something went wrong
        n = len(best_a)
        best_a = np.ones(n) / n
        final_score = evaluate_sequence([float(v) for v in best_a])
        print(f"Fallback to uniform: {final_score:.6f}")

    return [float(v) for v in best_a]
