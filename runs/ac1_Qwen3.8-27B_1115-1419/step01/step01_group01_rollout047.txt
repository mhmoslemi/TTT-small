"""
Search for non-negative step-function heights minimizing

    2 * n * max(convolve(a, a)) / sum(a)^2.

The code uses:
  - FFT/RFFT accelerated convolution and FFT-based gradients.
  - A global best carried through increasing lengths (2048, 4096, 8192, 16384).
  - Many diverse starts: interpolated prior best, random Fourier mutants,
    parametric asymmetric starts, spectral-factor starts, and LP-like starts.
  - Fast Adam-like log-space optimization with soft/top active-lag gradients.
  - Monotone line-search polishing using exact score evaluations.

The function propose_candidate returns the best sequence found before the
time budget expires.
"""

import numpy as np
import time


def _pair_score(pair):
    return pair[0]


def _next_pow2(x):
    x = int(max(2, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _proj_nonneg_sum(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.zeros(0, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _softmax(v, clip=35.0):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return np.zeros(0, dtype=np.float64)
    vmax = float(np.max(v))
    if not np.isfinite(vmax):
        return np.full(v.size, 1.0 / float(v.size), dtype=np.float64)
    v = v - vmax
    v = np.clip(v, -clip, 0.0)
    e = np.exp(v)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(v.size, 1.0 / float(v.size), dtype=np.float64)
    return e / s


def _v_from_p(p, clip=35.0):
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return np.zeros(0, dtype=np.float64)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros(p.size, dtype=np.float64)
    v = np.log(np.maximum(p / s, 1e-300))
    v = v - float(np.mean(v))
    return np.clip(v, -clip, clip)


def _center_clip(v, clip=35.0):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.zeros(v.size, dtype=np.float64)
    v = v - float(np.mean(v))
    return np.clip(v, -clip, clip)


def _conv_self_rfft(a, m):
    a = np.asarray(a, dtype=np.float64)
    F = np.fft.rfft(a, m)
    b = np.fft.irfft(F * F, m)[:2 * a.size - 1]
    b = np.maximum(b, 0.0)
    return F, b


def _score_from_b(a, b):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return float("inf")
    bm = float(np.max(b))
    if not np.isfinite(bm) or bm < 0.0:
        return float("inf")
    return float(2.0 * n * bm / (s * s))


def _score_fft(a, m=None):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf"), None
    if m is None:
        m = _next_pow2(2 * n - 1)
    F, b = _conv_self_rfft(a, m)
    return _score_from_b(a, b), b


def _interp(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    n = int(max(1, n))
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = a / s
    if a.size == 1:
        return np.full(n, float(a[0]), dtype=np.float64)
    if a.size == n:
        return a
    x_old = np.linspace(0.0, 1.0, int(a.size))
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if not np.isfinite(ss) or ss <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return out / ss


def _weights_soft(b, tau):
    b = np.asarray(b, dtype=np.float64)
    L = int(b.size)
    if L == 0:
        return np.zeros(0, dtype=np.float64)
    bx = float(np.max(b))
    if not np.isfinite(bx) or bx <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    tau = max(1e-300, float(tau))
    x = (b - bx) / tau
    x = np.clip(x, -60.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    return e / s


def _top_weights(b, top_k):
    b = np.asarray(b, dtype=np.float64)
    L = int(b.size)
    out = np.zeros(L, dtype=np.float64)
    if L == 0:
        return out
    top_k = int(max(1, min(L, top_k)))
    if top_k == L:
        out.fill(1.0 / float(L))
    else:
        idx = np.argpartition(b, -top_k)[-top_k:]
        out[idx] = 1.0 / float(top_k)
    return out


def _param_inits(n, rng):

def _mutate_parent(p, rng, amp):

    n = int(p.size)

    v = np.log(np.maximum(p, 1e-12))

def _lp_like_start(p, n, iters=50, eta=2.0, tau_frac=0.2):
    n = int(max(4, n))
    p = _proj_nonneg_sum(p)
    m = _next_pow2(2 * n - 1)
    L = 2 * n - 1

    Fa, b = _conv_self_rfft(p, m)
    M = float(b.max())
    if not np.isfinite(M) or M <= 0.0:
        return p

    g = np.ones(n, dtype=np.float64)
    Fg = np.fft.rfft(g, m)
    iters = int(max(1, iters))

    for it in range(iters):
        prog = float(it) / float(max(1, iters - 1))
        u = np.fft.irfft(Fa * Fg, m)[:L]
        u = np.maximum(u, 0.0)
        umax = float(u.max())
        if np.isfinite(umax) and umax > 0.0:
            factor = M / umax
            if np.isfinite(factor) and factor > 0.0:
                g = g * factor
                Fg = Fg * factor
                u = u * factor

        tau = M * max(0.03, tau_frac * (1.0 - 0.5 * prog))
        lam = _weights_soft(u, tau)
        W = np.fft.rfft(lam, m)
        h = np.fft.irfft(W * np.conj(Fa), m)[:n]
        h = np.maximum(h, 0.0)
        hmax = float(h.max())
        if np.isfinite(hmax) and hmax > 0.0:
            g = g * np.exp(-eta * h / hmax)

        g = np.maximum(g, 1e-300)
        Fg = np.fft.rfft(g, m)

    return _proj_nonneg_sum(g)


def _spectral_starts(p_best, n, m, rng):
    p_best = _proj_nonneg_sum(p_best)
    n = int(p_best.size)
    if n == 0:
        return []

    Fa, b = _conv_self_rfft(p_best, m)
    L = int(b.size)
    bm = float(b.max())
    if np.isfinite(bm) and bm > 0.0:
        b_norm = b / bm
    else:
        b_norm = np.ones(L, dtype=np.float64) / float(L)

    x = np.linspace(0.0, 1.0, L)
    edge = np.abs(x - 0.5) * 2.0
    out = []

    prefs = (
        (1.0, 0.0),
        (0.85, 0.5),
        (0.65, 1.0),
        (0.45, 1.5),
    )
    for ff, eb in prefs:
        base = 1.0 + eb * (edge ** 1.5)
        base_max = float(np.max(base))
        if np.isfinite(base_max) and base_max > 0.0:
            base = base / base_max
        else:
            base = np.ones(L, dtype=np.float64)

        target = ff * base + (1.0 - ff) * b_norm
        target = np.maximum(target, 0.0)
        tm = float(np.max(target))
        if np.isfinite(tm) and tm > 0.0:
            target = target / tm
        else:
            target = np.ones(L, dtype=np.float64)

        T = np.fft.fft(target, m)
        S = np.fft.ifft(np.sqrt(T + 0j)).real[:n]

        pos = np.maximum(S, 0.0)
        s1 = float(pos.sum())
        if np.isfinite(s1) and s1 > 0.0:
            out.append(_proj_nonneg_sum(pos))

        absS = np.abs(S)
        s2 = float(absS.sum())
        if np.isfinite(s2) and s2 > 0.0:
            out.append(_proj_nonneg_sum(absS))

    return out

            mhat = mu / max(1e-12, 1.0 - b1)
            vhat = nu / max(1e-12, 1.0 - b2)

            lr = lr_start + (lr_end - lr_start) * progress
            step = mhat / (np.sqrt(vhat) + eps)
            step = np.clip(step, -1.5, 1.5)

            v = v - lr * step
            v = _center_clip(v, max_v)
            fail = 0

    pf = _softmax(v, max_v)
    pf = _proj_nonneg_sum(pf)
    scf, _ = _score_fft(pf, m)
    if np.isfinite(scf) and scf < best_score:
        best_p = pf
        best_score = float(scf)

    return best_p, float(best_score)


def _stage(n, deadline, best_in, best_score_in, rng):
    n = int(max(4, n))
    now = time.monotonic()
    if now >= deadline or deadline - now < 12.0:
        if best_in is None:
            return np.ones(n, dtype=np.float64) / float(n), best_score_in
        return _interp(best_in, n), float(best_score_in)

    m = _next_pow2(2 * n - 1)

    if best_in is None:
        parent = np.ones(n, dtype=np.float64) / float(n)
    else:
        parent = _interp(best_in, n)

    cands = []

    sc, _ = _score_fft(parent, m)
    cands.append((sc, parent))

    for amp in (0.08, 0.20, 0.45):
        p = _mutate_parent(parent, rng, amp)
        sc, _ = _score_fft(p, m)
        cands.append((sc, p))

    for p in _param_inits(n, rng):
        sc, _ = _score_fft(p, m)
        cands.append((sc, p))

    for p in _spectral_starts(parent, n, m, rng):
        sc, _ = _score_fft(p, m)
        cands.append((sc, p))

    lp_specs = (
        (50, 1.5, 0.20),
        (70, 3.0, 0.10),
        (40, 0.8, 0.35),
    )
    for iters, eta, tau_frac in lp_specs:
        p = _lp_like_start(parent, n, iters=iters, eta=eta, tau_frac=tau_frac)
        sc, _ = _score_fft(p, m)
        cands.append((sc, p))

    scored = []
    seen = []
    for sc, p in cands:
        if not np.isfinite(sc):
            continue
        p = _proj_nonneg_sum(p)
        dup = False
        for q in seen:
            d = float(np.sqrt(np.sum((p - q) * (p - q))))
            if d < 0.07:
                dup = True
                break
        if not dup:
            scored.append((sc, p))
            seen.append(p)

    if not scored:
        scored = [(2.0, parent)]
        seen = [parent]

    scored.sort(key=_pair_score)
    top = scored[:3]

    if len(scored) > 3:
        ref1 = top[0][1]
        ref2 = top[1][1] if len(top) > 1 else ref1
        div = None
        for sc, p in scored:
            d1 = float(np.sqrt(np.sum((p - ref1) * (p - ref1))))
            if d1 <= 0.12:
                continue
            d2 = float(np.sqrt(np.sum((p - ref2) * (p - ref2))))
            if d2 > 0.10:
                div = (sc, p)
                break
        if div is not None:
            top.append(div)

    cur_score = float(top[0][0])
    cur_p = top[0][1]

    stage_rem = max(10.0, float(deadline - time.monotonic() - 5.0))
    if stage_rem <= 15.0:
        return cur_p, cur_score

    seed_base = int(rng.integers(0, 1000000000))

    if stage_rem > 600.0:
        fracs = (0.36, 0.28, 0.18, 0.18)
    elif stage_rem > 250.0:
        fracs = (0.45, 0.30, 0.25)
    else:
        fracs = (0.60, 0.40)

    run = 0
    for frac in fracs:
        rt = stage_rem * frac
        if time.monotonic() >= deadline - 8.0:
            break
        if rt < 12.0:
            break

        if run == 0:
            init = top[0][1]
            p_out, sc_out = _optimize_search(
                init, n, rt, seed_base + 101,
                line=False, top_frac=0.0, mix_soft=1.0,
                tau_start=0.45, tau_end=0.02,
                lr_start=0.18, lr_end=0.015
            )
        elif run == 1:
            init = cur_p
            p_out, sc_out = _optimize_search(
                init, n, rt, seed_base + 202,
                line=True, top_frac=0.01, mix_soft=0.25,
                tau_start=0.35, tau_end=0.02,
                lr_start=0.55, lr_end=0.06
            )
        elif run == 2:
            init = top[1][1] if len(top) > 1 else top[0][1]
            p_out, sc_out = _optimize_search(
                init, n, rt, seed_base + 303,
                line=False, top_frac=0.005, mix_soft=0.5,
                tau_start=0.30, tau_end=0.015,
                lr_start=0.16, lr_end=0.012
            )
        else:
            init = top[2][1] if len(top) > 2 else cur_p
            p_out, sc_out = _optimize_search(
                init, n, rt, seed_base + 404,
                line=True, top_frac=0.0008, mix_soft=0.0,
                tau_start=0.25, tau_end=0.01,
                lr_start=0.45, lr_end=0.04
            )

        if np.isfinite(sc_out) and sc_out < cur_score:
            cur_score = float(sc_out)
            cur_p = p_out

        run += 1

    if deadline - time.monotonic() > 25.0:
        p_out, sc_out = _optimize_search(
            cur_p, n, deadline - time.monotonic() - 8.0, seed_base + 505,
            line=False, top_frac=0.003, mix_soft=0.3,
            tau_start=0.20, tau_end=0.01,
            lr_start=0.12, lr_end=0.008
        )
        if np.isfinite(sc_out) and sc_out < cur_score:
            cur_score = float(sc_out)
            cur_p = p_out

    return cur_p, float(cur_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return the best non-negative coefficient sequence found."""
    start = time.monotonic()
    budget = int(max(60, budget_s))
    hard_deadline = start + max(60.0, float(budget) - 5.0)
    seed = int(seed)

    try:
        best_p = None
        best_score = float("inf")

        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if arr.size > 0:
                arr = np.maximum(arr, 0.0)
                s = float(arr.sum())
                if np.isfinite(s) and s > 0.0:
                    best_p = arr / s
        except Exception:
            best_p = None

        if best_p is None:
            best_p = np.ones(2048, dtype=np.float64) / 2048.0

        best_p = _proj_nonneg_sum(best_p)
        best_score, _ = _score_fft(best_p, None)
        if not np.isfinite(best_score) or best_score > 2.0 + 1e-9:
            best_p = np.ones(2048, dtype=np.float64) / 2048.0
            best_score, _ = _score_fft(best_p, None)
            if not np.isfinite(best_score):
                best_score = 2.0

        print("Initial best score: %.8f len=%d" % (best_score, int(best_p.size)))

        rng = np.random.default_rng(seed)
        usable = max(60.0, float(budget) - 80.0)

        if best_score > 1.55:
            plan = (
                (2048, 0.05),
                (4096, 0.15),
                (8192, 0.28),
                (16384, 0.52),
            )
        else:
            plan = (
                (2048, 0.04),
                (4096, 0.13),
                (8192, 0.30),
                (16384, 0.53),
            )

        for n, frac in plan:
            now = time.monotonic()
            if now + 30.0 >= hard_deadline:
                break
            stage_time = usable * float(frac)
            stage_deadline = min(hard_deadline - 5.0, now + stage_time)
            if stage_deadline - now < 15.0:
                continue

            print("Stage n=%d budget=%.1fs current_best=%.8f" %
                  (int(n), stage_time, best_score), flush=True)

            best_p, best_score = _stage(
                int(n), stage_deadline, best_p, best_score, rng
            )

            print("After stage n=%d best=%.8f len=%d" %
                  (int(n), best_score, int(best_p.size)), flush=True)

        best_p = _proj_nonneg_sum(best_p)
        final_score, _ = _score_fft(best_p, None)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score

        print("Final FFT score: %.8f len=%d" % (best_score, int(best_p.size)),
              flush=True)

        return [float(x) for x in best_p]

    except Exception as e:
        print("Fatal error in propose_candidate:", repr(e), flush=True)
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if arr.size > 0:
                arr = np.maximum(arr, 0.0)
                s = float(arr.sum())
                if np.isfinite(s) and s > 0.0:
                    return [float(x) for x in arr / s]
        except Exception:
            pass
        return [1.0 / 1024.0] * 1024
