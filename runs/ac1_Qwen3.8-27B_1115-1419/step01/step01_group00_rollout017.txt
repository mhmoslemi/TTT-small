It uses FFT-based evaluation and gradients, multiplicative log-space updates,

shape refinement, occasional perturbations, and a secondary length attempt.
"""

import numpy as np
import time

try:
    from scipy.fft import rfft, irfft, next_fast_len
except Exception:
    from numpy.fft import rfft, irfft

    def next_fast_len(n):
        if n <= 1:
            return 1
        p = 2
        while p < n:
            p <<= 1
        return p

try:
    from scipy.optimize import minimize
    _HAVE_MINIMIZE = True
except Exception:
    _HAVE_MINIMIZE = False


def _next_fast_len(n):
    m = int(max(1, n))
    return int(next_fast_len(m))


def _normalize_array(a, n):
    """Normalize to nonnegative finite array of length n with sum exactly n."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)

    if a.size != n:
        if a.size == 0:
            a = np.ones(n, dtype=np.float64)
        else:
            a = np.where(np.isfinite(a), a, 0.0)
            s = float(a.sum())
            if s <= 1e-300:
                a = np.ones(n, dtype=np.float64)
            else:
                loga = np.log(np.maximum(a, 1e-300))
                old = np.linspace(0.0, 1.0, a.size)
                new = np.linspace(0.0, 1.0, n)
                loga = np.interp(new, old, loga)
                a = np.exp(loga)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64)
    else:
        a *= n / s
    return a


def _normalize_loga(loga, n, cap=50.0):
    """Normalize log-values, cap dynamic range, return (a, loga) with sum(a)=n."""
    loga = np.asarray(loga, dtype=np.float64).reshape(-1)
    if loga.size != n:
        if loga.size == 0:
            a = np.ones(n, dtype=np.float64)
            return a, np.log(a)
        loga = np.interp(
            np.linspace(0.0, 1.0, n),
            np.linspace(0.0, 1.0, loga.size),
            loga,
        )

    loga = np.where(np.isfinite(loga), loga, 0.0)
    mx = float(loga.max())
    loga = loga - mx

    mn = float(loga.min())
    if -mn > cap:
        loga = np.maximum(loga, -cap)

    vals = np.exp(loga)
    s = float(vals.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64)
        return a, np.log(a)

    a = vals * (n / s)
    loga = np.log(a)
    return a, loga


def _lowpass_noise(rng, n, fft_len, keep=20):
    """Smooth normalized zero-mean noise with only the lowest.keep frequency bins."""
    x = rng.normal(size=n)
    F = rfft(x, fft_len)
    k = min(len(F), max(1, int(keep)))
    if k < len(F):
        F[k:] = 0.0
    y = irfft(F, n=fft_len)[:n]
    y -= float(y.mean())
    sd = float(np.std(y))
    if sd > 1e-15:
        y /= sd
    else:
        y = rng.normal(size=n)
        y -= float(y.mean())
        sd = float(np.std(y))
        if sd > 1e-15:
            y /= sd
    return y


def _add_perturb(loga, rng, n, fft_len, sigma=0.20):
    """Add low-frequency and block noise to log-values."""
    noise = _lowpass_noise(rng, n, fft_len, keep=max(3, n // 300))
    new = loga + sigma * noise

    w = max(3, n // 80)
    s = int(rng.integers(0, max(1, n - w + 1)))
    new[s:s + w] += sigma * 0.7 * rng.normal(size=w)
    return new


def _convolve_fft(a, L):
    """Full self-convolution via FFT. Returns (conv, rfft(a, L))."""
    A = rfft(a, L)
    c = irfft(A * A, n=L)[:2 * a.size - 1]
    c = np.maximum(c, 0.0)
    return c, A


def _grad_half(A, p, L, n):
    """
    Half-gradient of sum_k p_k conv(a,a)_k:
        gh_i = sum_k p_k a_{k-i}
    (the derivative of conv wrt a_i is 2*gh_i).
    """
    P = rfft(p, L)
    gh = irfft(np.conjugate(A) * P, n=L)[:n]
    gh = np.maximum(gh, 0.0)
    return gh


def _band_weights(c, mc, margin, beta, n):
    """
    Nonnegative probability weights on convolution lags.

    Use a top band above
        avg + (1 - margin)*(mc - avg),
    with optional sharpening beta.
    """
    m = c.size
    avg = float(n) / float(2 * n - 1)
    if mc <= avg + 1e-12:
        return np.full(m, 1.0 / m, dtype=np.float64)

    lo = mc - margin * (mc - avg)
    if lo < 0.0:
        lo = 0.0

    w = np.maximum(c - lo, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        idx = int(np.argmax(c))
        w = np.zeros(m, dtype=np.float64)
        w[idx] = 1.0
        s = 1.0

    if beta > 1.0 + 1e-12:
        w = np.power(w, beta)
        s = float(w.sum())
        if s <= 1e-300:
            idx = int(np.argmax(c))
            w = np.zeros(m, dtype=np.float64)
            w[idx] = 1.0
            s = 1.0

    return w / s


def _make_starts(n, parent, rng, fft_len, max_starts=6):
    """Create a small pool of diverse starting sequences."""
    starts = []

    def _add(arr):
        a = _normalize_array(arr, n)
        starts.append(a)

    # Parent if useful.
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).reshape(-1)
        if p.size > 0 and np.any(np.isfinite(p) & (p > 0)):
            _add(p)

    if len(starts) == 0:
        _add(np.ones(n, dtype=np.float64))

    # Smooth random log starts.
    for sigma in (0.35, 0.90):
        noise = _lowpass_noise(rng, n, fft_len, keep=max(8, n // 120))
        _add(np.exp(sigma * noise))

    # One-sided exponential envelopes.
    t = np.linspace(0.0, 1.0, n)
    for alpha in (0.4, 1.0, 2.5, 6.0):
        _add(np.exp(-alpha * t))
        _add(np.exp(alpha * t))
    for beta in (4.0, 12.0):
        _add(np.exp(-beta * (1.0 - t)))
        _add(np.exp(-beta * t))

    # Low-frequency sinusoidal log shapes.
    for amp1, amp2 in ((0.55, 0.25), (0.80, 0.35), (1.10, 0.55)):
        ph1 = float(rng.uniform(0.0, 2.0 * np.pi))
        ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
        loga = amp1 * np.sin(2.0 * np.pi * t + ph1) + amp2 * np.sin(4.0 * np.pi * t + ph2)
        _add(np.exp(loga))

    # Balanced anti-palindromic 0/2 start.
    try:
        if n % 2 == 0:
            k = n // 2
            s = rng.integers(0, 2, size=k) * 2 - 1
            x = np.empty(n, dtype=np.float64)
            x[:k] = s
            x[k:] = -s[::-1]
        else:
            k = n // 2
            s = rng.integers(0, 2, size=k) * 2 - 1
            x = np.empty(n, dtype=np.float64)
            x[:k] = s
            x[k] = 0.0
            x[k + 1:] = -s[::-1]
        _add(1.0 + x)
    except Exception:
        pass

    # Deduplicate-ish by limited selection.
    if len(starts) > max_starts:
        keep = [0]
        rest = list(range(1, len(starts)))
        rng.shuffle(rest)
        keep.extend(rest[:max_starts - 1])
        starts = [starts[i] for i in keep]

    return starts


def _optimize_band(a_init, n, L, deadline, tag="", rng=None, hard_only=False):
    """
    Main local optimizer.

    Uses log-space multiplicative updates driven by half-gradient
        v_i = a_i * sum_k p_k a_{k-i}
    where p weights the active high convolution lags.
    """
    a = _normalize_array(a_init, n)
    loga = np.log(np.maximum(a, 1e-300))

    if rng is None:
        rng = np.random.default_rng(123456789)

    start_time = time.monotonic()
    avail = max(1.0, deadline - start_time)

    c, A = _convolve_fft(a, L)
    best_mc = float(c.max())
    best_obj = 2.0 * best_mc / n
    best_a = a.copy()

    if avail <= 2.0:
        return best_a, best_obj, best_mc

    if hard_only:
        phase1_until = start_time
    else:
        phase_frac = 0.55 if avail > 90.0 else 0.35
        phase1_until = start_time + phase_frac * avail

    fft_n = _next_fast_len(n)

    eta = 0.04 if hard_only else 0.10
    margin = 0.04 if hard_only else 0.70
    beta = 2.0 if hard_only else 1.0

    stall = 0
    bad = 0
    prev_mc = None
    it = 0
    last_print = start_time

    # Progress log at start.
    print(f"[{tag}] start obj={best_obj:.6f} n={n} hard_only={hard_only}")

    while time.monotonic() < deadline - 0.75:
        now = time.monotonic()
        if now >= deadline:
            break

        c, A = _convolve_fft(a, L)
        mc = float(c.max())
        obj = 2.0 * mc / n

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_mc = mc
            best_a = a.copy()
            stall = 0
            bad = 0
            eta = min(0.20, eta * 1.02)
        else:
            stall += 1
            if obj > best_obj * 1.004:
                bad += 1
            else:
                bad = max(0, bad - 1)

        if prev_mc is not None:
            if mc > prev_mc * 1.008:
                eta = max(0.012, eta * 0.86)
            elif mc < prev_mc * 0.998:
                eta = min(0.22, eta * 1.02)
        prev_mc = mc

        # Adaptive schedule.
        if hard_only or now >= phase1_until:
            denom = max(1e-9, deadline - start_time)
            hf = min(1.0, (now - start_time) / denom)
            if hard_only:
                margin = 0.05 * (1.0 - 0.75 * hf) + 0.002
                eta = max(0.012, 0.05 * (1.0 - 0.45 * hf))
            else:
                margin = 0.15 * (1.0 - 0.80 * hf) + 0.003
                eta = max(0.018, 0.08 * (1.0 - 0.40 * hf))
            beta = 1.0 + 2.5 * hf
        else:
            denom = max(1e-9, phase1_until - start_time)
            frac = min(1.0, (now - start_time) / denom)
            margin = 0.70 - 0.55 * frac
            beta = 1.0 + 0.5 * frac
            eta = max(0.05, 0.10 - 0.03 * frac)

        # Stall handling.
        if stall > 1200:
            eta = max(0.012, eta * 0.85)
            margin = min(0.95, margin * 1.15)
            beta = min(5.0, beta + 0.25)
            stall = 0

        if stall > 6000:
            loga = _add_perturb(loga, rng, n, fft_n, sigma=0.22)
            a, loga = _normalize_loga(loga, n, 50.0)
            eta = max(eta, 0.08)
            margin = min(0.85, margin + 0.20)
            beta = max(1.0, beta - 0.5)
            stall = 0
            bad = 0

        if bad > 2500 and obj > best_obj * 1.002:
            a = best_a.copy()
            loga = np.log(np.maximum(a, 1e-300))
            loga = _add_perturb(loga, rng, n, fft_n, sigma=0.28)
            a, loga = _normalize_loga(loga, n, 50.0)
            bad = 0
            stall = 0
            eta = 0.07
            margin = 0.60
            beta = 1.0

        p = _band_weights(c, mc, margin, beta, n)
        gh = _grad_half(A, p, L, n)

        # Log-space half-gradient contribution: v_i = a_i * gh_i.
        v = a * gh
        wmean = float(np.dot(a, v)) / n
        d = -(v - wmean)

        scale2 = float(np.dot(a, d * d))
        if not np.isfinite(scale2) or scale2 <= 1e-30:
            break
        scale = np.sqrt(scale2 / n)
        if not np.isfinite(scale) or scale <= 1e-15:
            break

        delta = eta * d / scale
        delta = np.clip(delta, -1.0, 1.0)

        loga = loga + delta
        a, loga = _normalize_loga(loga, n, 50.0)

        it += 1
        if tag and it % 4000 == 0 and time.monotonic() - last_print > 1.5:
            last_print = time.monotonic()
            print(
                f"[{tag}] it={it} obj={obj:.6f} best={best_obj:.6f} "
                f"eta={eta:.4f} margin={margin:.3f} beta={beta:.2f}"
            )

    return best_a, best_obj, best_mc


def _shape_basis(n, nharmonic=6):
    """Low-dimensional smooth basis for log-space shape refinement."""
    t = np.linspace(0.0, 1.0, n)
    B = [t.astype(np.float64)]
    for m in range(1, nharmonic + 1):
        B.append(np.sin(2.0 * np.pi * m * t))
        B.append(np.cos(2.0 * np.pi * m * t))
    return np.asarray(B, dtype=np.float64)


def _shape_array(params, base_log, basis, n):
    params = np.asarray(params, dtype=np.float64).reshape(-1)
    loga = base_log + params @ basis
    a, _ = _normalize_loga(loga, n, 50.0)
    return a


def _shape_objective(params, base_log, basis, n, L):
    a = _shape_array(params, base_log, basis, n)
    c, _ = _convolve_fft(a, L)
    return 2.0 * float(c.max()) / n


def _refine_shape(base_a, n, L, deadline, tag="", rng=None):
    """
    Short deterministic refinement of the low-frequency logarithmic shape.
    """
    if not _HAVE_MINIMIZE:
        c, _ = _convolve_fft(base_a, L)
        return base_a.copy(), 2.0 * float(c.max()) / n

    base_log = np.log(np.maximum(base_a, 1e-300))
    basis = _shape_basis(n, 6)
    coef = basis.shape[0]

    c, _ = _convolve_fft(base_a, L)
    best_obj = 2.0 * float(c.max()) / n
    best_a = base_a.copy()

    if rng is None:
        rng = np.random.default_rng(987654321)

    trials = 3
    for trial in range(trials):
        if time.monotonic() >= deadline:
            break
        if trial == 0:
            x0 = np.zeros(coef, dtype=np.float64)
        else:
            x0 = rng.normal(size=coef) * (0.15 if trial == 1 else 0.35)

        try:
            res = minimize(
                _shape_objective,
                x0,
                args=(base_log, basis, n, L),
                method="Nelder-Mead",
                options={
                    "maxfev": 2500,
                    "xatol": 1e-3,
                    "fatol": 1e-9,
                    "adaptive": True,
                },
            )
        except Exception:
            res = None

        if res is not None and res.x is not None:
            a = _shape_array(res.x, base_log, basis, n)
            c, _ = _convolve_fft(a, L)
            obj = 2.0 * float(c.max()) / n
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"[{tag}] shape trial {trial} obj={obj:.6f}")

    return best_a, best_obj


def _finalize(a):
    """Return a safe Python list: nonnegative, finite, max(abs) == 1 if possible."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return [1.0]

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    s = float(a.sum())
    if s <= 1e-300:
        return [1.0]

    mx = float(a.max())
    if mx > 0.0:
        a = a / mx

    a = np.where(np.isfinite(a), a, 0.0)
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400):
    try:
        t0 = time.monotonic()
        deadline = t0 + max(1.0, float(budget_s) - 10.0)
        rng = np.random.default_rng(seed)

        parent = globals().get("height_sequence_1", None)
        if parent is not None:
            p = np.asarray(parent, dtype=np.float64).reshape(-1)
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size == 0 or p.sum() <= 1e-300:
                parent = None
        else:
            parent = None

        # Choose primary length. Prefer the provided parent when reasonable.
        if parent is not None and 1500 <= parent.size <= 9000:
            n1 = int(parent.size)
        else:
            n1 = 3809

        L1 = _next_fast_len(2 * n1 - 1)
        fft1 = _next_fast_len(n1)

        global_best = None
        global_obj = float("inf")
        global_n = n1
        global_L = L1
        global_fft = fft1

        # ---------------- Pilot multi-start at primary length ----------------
        starts = _make_starts(n1, parent, rng, fft1, max_starts=6)
        pilot_total = int(min(180.0, max(70.0, 0.08 * budget_s)))
        per = max(12, pilot_total // max(1, len(starts)))

        for i, st in enumerate(starts):
            end = min(time.monotonic() + per, deadline - 420.0)
            if time.monotonic() >= deadline - 420.0:
                break
            a, o, mc = _optimize_band(st, n1, L1, end, tag=f"pilot{i}", rng=rng)
            if o < global_obj:
                global_obj = o
                global_best = a.copy()
                global_n = n1
                global_L = L1
                global_fft = fft1
                print(f"  new global after pilot{i}: obj={global_obj:.6f} n={global_n}")

        if global_best is None:
            global_best = _normalize_array(np.ones(n1, dtype=np.float64), n1)
            c, _ = _convolve_fft(global_best, L1)
            global_obj = 2.0 * float(c.max()) / n1

        # ---------------- Main refinement at primary length ----------------
        end_main = min(deadline - 380.0, time.monotonic() + 0.45 * budget_s)
        if time.monotonic() < end_main - 10.0:
            a, o, mc = _optimize_band(global_best, global_n, global_L, end_main, tag="main", rng=rng)
            if o < global_obj:
                global_obj = o
                global_best = a.copy()
                print(f"  new global after main: obj={global_obj:.6f} n={global_n}")

        # ---------------- Restart / perturb local searches ----------------
        att = 0
        while time.monotonic() < deadline - 260.0:
            slice_s = max(30.0, min(120.0, (deadline - time.monotonic()) / 4.0))
            if slice_s <= 15.0:
                break
            base_log = np.log(np.maximum(global_best, 1e-300))
            init_log = _add_perturb(base_log, rng, global_n, global_fft, sigma=0.26)
            init, _ = _normalize_loga(init_log, global_n, 50.0)
            end = min(time.monotonic() + slice_s, deadline - 200.0)
            a, o, mc = _optimize_band(init, global_n, global_L, end, tag=f"restart{att}", rng=rng)
            if o < global_obj:
                global_obj = o
                global_best = a.copy()
                print(f"  new global after restart{att}: obj={global_obj:.6f} n={global_n}")
            att += 1

        # ---------------- Low-frequency shape refinement ----------------
        if time.monotonic() < deadline - 220.0:
            shape_deadline = min(deadline - 150.0, time.monotonic() + 130.0)
            a, o = _refine_shape(global_best, global_n, global_L, shape_deadline, tag="shape", rng=rng)
            if o < global_obj:
                global_obj = o
                global_best = a.copy()
                print(f"  new global after shape: obj={global_obj:.6f} n={global_n}")

        # ---------------- Optional secondary length attempt ----------------
        if time.monotonic() < deadline - 100.0:
            if global_n < 5000:
                n2 = 6400
            elif global_n < 8000:
                n2 = 8000
            else:
                n2 = global_n

            if n2 != global_n:
                L2 = _next_fast_len(2 * n2 - 1)
                fft2 = _next_fast_len(n2)
                starts2 = _make_starts(n2, global_best, rng, fft2, max_starts=4)
                rem = max(0.0, deadline - time.monotonic())
                per2 = max(15.0, min(90.0, (rem - 70.0) / len(starts2)))

                for i, st in enumerate(starts2):
                    if time.monotonic() >= deadline - 70.0:
                        break
                    end = min(time.monotonic() + per2, deadline - 60.0)
                    a, o, mc = _optimize_band(st, n2, L2, end, tag=f"len2_{i}", rng=rng)
                    if o < global_obj:
                        global_obj = o
                        global_best = a.copy()
                        global_n = n2
                        global_L = L2
                        global_fft = fft2
                        print(f"  new global from n={n2}: obj={global_obj:.6f}")

                if global_n == n2 and time.monotonic() < deadline - 45.0:
                    end = min(deadline - 35.0, time.monotonic() + 80.0)
                    a, o, mc = _optimize_band(global_best, global_n, global_L, end, tag="len2_ref", rng=rng)
                    if o < global_obj:
                        global_obj = o
                        global_best = a.copy()
                        print(f"  new global after len2_ref: obj={global_obj:.6f}")

        # ---------------- Final short hard polish ----------------
        if time.monotonic() < deadline - 3.0:
            a, o, mc = _optimize_band(
                global_best,
                global_n,
                global_L,
                deadline - 1.0,
                tag="final_hard",
                rng=rng,
                hard_only=True,
            )
            if o < global_obj:
                global_obj = o
                global_best = a.copy()

        result = _finalize(global_best)

        try:
            score = float(evaluate_sequence(result))
            print(f"FINAL internal obj={global_obj:.6f} evaluated score={score:.6f} n={len(result)}")
            if not np.isfinite(score):
                try:
                    alt = globals().get("height_sequence_1")
                    if alt is not None and np.any(np.asarray(alt, dtype=np.float64) > 0):
                        result = _finalize(alt)
                        score = float(evaluate_sequence(result))
                        print(f"FALLBACK parent evaluated score={score:.6f}")
                except Exception:
                    result = [1.0] * global_n
        except Exception as exc:
            print("evaluate_sequence failed at final check:", exc)

        return result

    except Exception as exc:
        print("Top-level error in propose_candidate:", exc)
        try:
            alt = globals().get("height_sequence_1")
            if alt is not None:
                return _finalize(alt)
        except Exception:
            pass
        return [1.0] * 3809
