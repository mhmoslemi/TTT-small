```python
def evaluate_score(a):
    """Fast evaluation of the objective."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a ** 2)

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

def lp_update(a, M, n_active=None):
    """
    Solve LP: maximize sum(b) subject to b >= 0, (a*b)[k] <= M for all k (or active k).
    """
    n = len(a)
    
    # Constraint: (a*b)[k] <= M for all k = 0, ..., 2n-2
    # (a*b)[k] = sum_i a[i] * b[k-i] where valid indices
    # For each k, the constraint is: sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i] <= M
    # Let j = k-i, then i = k-j, and j ranges from max(0, k-n+1) to min(n-1, k)
    # So the constraint is: sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j] * b[j] <= M
    
    # This gives a (2n-1) x n matrix A where A[k,j] = a[k-j] if valid
    # But this matrix is large. Let's use a sparse representation or just use the full matrix.
    
    # Actually, for n up to ~10000, the matrix is (2n-1) x n which is about 2n^2 elements.
    # For n=4806, that's about 46M elements, which is ~370MB. That's a lot.
    # Let me use a different approach.
    
    # Alternative: use the constraint in matrix form with a Toeplitz-like matrix.
    # (a*b)[k] for k=0,...,2n-2 is the convolution of a and b.
    # We can express this as a matrix-vector product where the matrix is a 
    # (2n-1) x n Toeplitz matrix.

        n_active = min(2*n-1, 200)  # Use at most 200 active constraints
    
    # Find the most active constraints
    c = np.convolve(a, a)
    sorted_idx = np.argsort(c)[::-1]
    active_k = sorted_idx[:n_active]

    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((len(active_k), n))

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A[idx, i] = a[k - i]

    # In linprog form: minimize -sum(b) subject to A @ b <= M, b >= 0
    c_obj = -np.ones(n)
    A_ub = A
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    
    try:

        if res.success:
            b = res.x
            return b
        else:

    except Exception:

def subgradient_step(a, n_active, step):

    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    if max_c < 1e-15:
        return a
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(active)
    a_new = a - step * g
    return normalize(a_new)

        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))
        a = normalize(a_new)
    return a
```

```python
    """Project onto probability simplex (non-negative, sums to 1)."""

    """LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for top n_cons k."""
    n = len(a)
    c = np.convolve(a, a)
    idx = np.argsort(c)[::-1][:n_cons]
    
    A = np.zeros((n_cons, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1)
        A[i, j] = a[k - j]
```

The constraint is: (a * b)[k] <= M for all k.

Where (a * b)[k] = sum_{i} a[i] * b[k-i] (with the understanding that a[j] = b[j] = 0 for j outside [0, n-1]).

For a given k, the valid i range is: max(0, k-(n-1)) <= i <= min(n-1, k).

So (a * b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i].

Let j = k-i, then i = k-j, and j ranges from max(0, k-n+1) to min(n-1, k)... wait, that's not right.

If i ranges from lo to hi, then j = k-i ranges from k-hi to k-lo.

lo = max(0, k-n+1), hi = min(n-1, k)
j ranges from k-hi to k-lo.

k-hi = k - min(n-1, k)
k-lo = k - max(0, k-n+1)

If k < n: hi = k, lo = 0, so j ranges from 0 to k.
If k >= n: hi = n-1, lo = k-n+1, so j ranges from k-(n-1) to k-(k-n+1) = n-1.

So in general:
- If k < n: (a*b)[k] = sum_{j=0}^{k} a[k-j] * b[j]
- If k >= n: (a*b)[k] = sum_{j=k-n+1}^{n-1} a[k-j] * b[j]

```python

    """Score assuming sum(a)=1."""
    return 2.0 * len(a) * np.convolve(a, a).max()

    """Project to probability simplex."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 1e-30 else np.ones(len(a)) / len(a)


def lp_refine(a, M, n_cons):
    """LP best-response: max sum(b) s.t. b>=0, (a*b)[k]<=M (top n_cons)."""
    n = len(a)
    c = np.convolve(a, a)
    idx = np.argsort(c)[::-1][:n_cons]
    A = np.zeros((n_cons, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1)
        A[i, j] = a[k - j]
    try:

                      bounds=[(0, None)] * n, method='highs',
                      options={"presolve": True, "time_limit": 15})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        mag = np.abs(A)
```

```python
                if step < 0.002 or stall > 60:
                    step = 0.04
                    a = normalize(best_a + rng.normal(0, 0.012, n))
                    stall = 0
    s = norm_eval(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return a, best_score, best_a


def run_longer_phase(best_a, best_score, deadline, max_time, rng):
    """Phase 4: Try longer sequences via interpolation + polish."""
    n = len(best_a)
    end = time.monotonic() + min(max_time, deadline - time.monotonic())
    
    for mult in [2, 3, 4]:
        n_new = n * mult
        if n_new > 60000 or time.monotonic() > end - 30:
            break
        a_new = np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, n), best_a)
        a_new = normalize(a_new)
        st = 0.012
        n_act_new = min(150, 2 * n_new - 1)
        for _ in range(5000):
            if time.monotonic() > end - 30:
                break
            a_new = subgrad_update(a_new, n_act_new, st)
            st *= 0.999
            if st < 0.001:
                st = 0.012
        s = norm_eval(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
            print(f"  Long n={n_new}: {best_score:.6f} ***")
        else:
            print(f"  Long n={n_new}: {s:.6f}")
    return best_a, best_score


def propose_candidate(budget_s=2400):
    """Find best non-negative sequence minimizing the evaluation function."""
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        a = np.ones(4096, dtype=np.float64)
    
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_score = norm_eval(a)
    print(f"Init: n={n}, score={best_score:.6f}")
    
    deadline = time.monotonic() + budget_s - 40
    rng = np.random.default_rng(42)
    
    # Phase 1: LP iterations (~35% of time)
    print("\n[Phase 1] LP-based refinement")
    a, best_score, best_a = run_lp_phase(a, best_score, best_a, deadline,
                                          (deadline - time.monotonic()) * 0.35)
    print(f"  After P1: {best_score:.6f}")
    
    # Phase 2: Spectral flattening (~15% of time)
    print("\n[Phase 2] Spectral flattening")
    a, best_score, best_a = run_spectral_phase(best_a, best_score, best_a, deadline,
                                                (deadline - time.monotonic()) * 0.15)
    print(f"  After P2: {best_score:.6f}")
    
    # Phase 3: Subgradient polish (~35% of time)
    print("\n[Phase 3] Subgradient polishing")
    a, best_score, best_a = run_subgrad_phase(best_a, best_score, best_a, deadline,
                                               (deadline - time.monotonic()) * 0.35, rng)
    print(f"  After P3: {best_score:.6f}")
    
    # Phase 4: Longer sequences (~10% of time)
    print("\n[Phase 4] Longer sequences")
    best_a, best_score = run_longer_phase(best_a, best_score, deadline,
                                           (deadline - time.monotonic()) * 0.12, rng)
    print(f"  After P4: {best_score:.6f}")
    
    # Phase 5: Final LP polish on best found
    print("\n[Phase 5] Final LP polish")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a, best_score, best_a = run_lp_phase(best_a, best_score, best_a, deadline,
                                              remaining * 0.8)
        print(f"  After P5: {best_score:.6f}")
    
    # Final: one more subgradient sweep with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n[Final] Extra subgradient sweep")
        a, best_score, best_a = run_subgrad_phase(best_a, best_score, best_a,
                                                   time.monotonic() + remaining * 0.9,
                                                   remaining * 0.9, rng)
    
    # Safety check
    result = [float(max(0.0, x)) for x in best_a]
    if sum(result) < 0.01:
        result = [1.0 / len(best_a)] * len(best_a)
    
    final_score = norm_eval(np.array(result))
    print(f"\nFINAL: n={len(result)}, norm_eval={final_score:.6f}")
    return result
```
