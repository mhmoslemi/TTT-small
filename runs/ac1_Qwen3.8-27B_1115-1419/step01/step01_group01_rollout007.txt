```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_lp_peak_shave(a, M):
    """Solve max sum(g) s.t. g >= 0, (a*g)_j <= M for all j"""
    n = len(a)
    L = 2*n - 1
    
    # Build constraint matrix A: L x n
    # (a*g)_j = sum_k a[k] * g[j-k] for valid k, j-k
    # A[j, k] = a[k] if 0 <= j-k <= n-1, else 0
    A = np.zeros((L, n))
    for j in range(L):
        k_min = max(0, j - (n-1))
        k_max = min(n-1, j)
        for k in range(k_min, k_max+1):
            A[j, k] = a[k]
    
    # Objective: maximize -sum(g) → minimize -sum(g)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        return res.x
    return None
```

```python
        v = v - lr * step
        v = v - v.mean()
        v = np.clip(v, -30, 30)
        if t_adam % 12 == 0:
            an = _expsum(v)
            sc = _score_fft(an)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = an.copy()
                last_improve = time.monotonic()
                if verbose and t_adam % 200 == 0:
                    print("  [AD] t=%d sc=%.8f" % (t_adam, best_sc))
    if verbose:
        print("  [AD] done n=%d sc=%.8f" % (n, best_sc))
    return _norm_nonneg(best_a), best_sc


def _make_inits(n, rng):
    res = []
    t_arr = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    res.append(np.ones(n, dtype=np.float64))
    for c in (0.5, 1.0, 2.0, 4.0):
        res.append(np.exp(-c * t_arr))
    L = i_arr + 1.0
    R = n - i_arr
    res.append(1.0 / np.sqrt(L * R))
    for p, q in ((0.5, 0.5), (1.0, 0.2), (0.2, 1.0)):
        res.append((L / n) ** p * (1.0 - t_arr + 1e-9) ** q)
    a_cos = 1.0 + 0.4 * np.cos(2 * np.pi * t_arr) + 0.2 * np.cos(4 * np.pi * t_arr)
    res.append(np.maximum(a_cos, 0.01))
    kc = min(10, n)
    a_r = 1.0
    for k in range(1, kc + 1):
        a_r += (0.3 / (k ** 0.8)) * np.cos(2 * np.pi * k * t_arr + rng.uniform(0, 2 * np.pi))
    res.append(np.maximum(a_r, 0.01))
    return [np.asarray(x, dtype=np.float64) for x in res]


def _interp_to(a_src, n):
    a_src = np.asarray(a_src, dtype=np.float64)
    if a_src.size == n:
        return a_src.copy()
    s = a_src.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a_src = a_src / s
    x_old = np.linspace(0.0, 1.0, a_src.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a_src)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        out = np.ones(n) / n
    else:
        out = out / ss
    return out


def _process_stage(n, budget, prev_best, rng, seed=42):
    if prev_best is not None:
        a = _interp_to(prev_best, n)
    else:
        a = np.ones(n) / n
    best_a = a.copy()
    best_sc = _score_fft(a)
    t0 = time.monotonic()

    # LP
    lp_time = min(budget * 0.45, 600.0)
    if lp_time > 3.0:
        a_lp, sc_lp = _lp_phase(a, lp_time, verbose=True)
        if sc_lp < best_sc:
            best_a, best_sc = a_lp, sc_lp

    remaining = max(0.0, budget - (time.monotonic() - t0))

    # Spectral polish
    sp_time = min(remaining * 0.25, 45.0)
    if sp_time > 1.0:
        a_sp, sc_sp = _spectral_polish(best_a, sp_time, verbose=True)
        if sc_sp < best_sc:
            best_a, best_sc = a_sp, sc_sp

    remaining = max(0.0, budget - (time.monotonic() - t0))

    # LP round 2
    lp2_time = min(remaining * 0.30, 300.0)
    if lp2_time > 3.0:
        a_lp2, sc_lp2 = _lp_phase(best_a, lp2_time, verbose=True)
        if sc_lp2 < best_sc:
            best_a, best_sc = a_lp2, sc_lp2

    remaining = max(0.0, budget - (time.monotonic() - t0))

    # Adam
    ad_time = max(2.0, remaining * 0.50)
    if ad_time > 2.0:
        a_ad, sc_ad = _adam_polish(best_a, ad_time, seed=seed, verbose=True)
        if sc_ad < best_sc:
            best_a, best_sc = a_ad, sc_ad

    remaining = max(0.0, budget - (time.monotonic() - t0))

    # Final LP
    lp3_time = max(2.0, remaining * 0.6)
    if lp3_time > 2.0:
        a_final, sc_final = _lp_phase(best_a, lp3_time, verbose=True)
        if sc_final < best_sc:
            best_a, best_sc = a_final, sc_final

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    seed = int(seed)
    budget_s = float(max(60.0, budget_s))
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0
    rng = np.random.default_rng(seed)

    prev = None
    try:
        cand = height_sequence_1
        arr = np.asarray(cand, dtype=np.float64).ravel()
        if arr.size > 0 and np.isfinite(arr).all():
            arr = np.maximum(arr, 0.0)
            if arr.sum() > 0.0:
                prev = arr / arr.sum()
    except Exception:
        prev = None

    best_a = prev.copy() if prev is not None else np.ones(512, dtype=np.float64) / 512.0
    best_sc = _score_fft(best_a)
    print("Initial: n=%d score=%.8f" % (len(best_a), best_sc))

    stage_defs = [
        (512, 0.03),
        (1024, 0.09),
        (2048, 0.34),
        (4096, 0.32),
        (8192, 0.22),
    ]

    for n, frac in stage_defs:
        now = time.monotonic()
        if now + 25.0 >= deadline:
            break
        budget = min(frac * (budget_s - 15.0), deadline - now - 5.0)
        if budget < 8.0:
            break
        print("\n=== Stage n=%d budget=%.1fs (best=%.8f) ===" % (n, budget, best_sc))
        a_new, sc_new = _process_stage(n, budget, best_a, rng, seed=int(rng.integers(0, 2**31)))
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_sc:
            best_a = a_new
            best_sc = sc_new
            print("  Updated best: n=%d sc=%.8f" % (len(best_a), best_sc))
        else:
            print("  No improvement at n=%d, keeping n=%d sc=%.8f" % (n, len(best_a), best_sc))

    # Final try with remaining time on best length
    rem = deadline - time.monotonic()
    if rem > 15.0 and best_a is not None:
        print("\n=== Final polish: n=%d budget=%.1fs ===" % (len(best_a), rem))
        a_pol, sc_pol = _lp_phase(best_a, min(rem * 0.6, 200.0), verbose=True)
        if sc_pol < best_sc:
            best_a, best_sc = a_pol, sc_pol
        rem2 = deadline - time.monotonic()
        if rem2 > 5.0:
            a_ad, sc_ad = _adam_polish(best_a, max(1.0, rem2 * 0.8), seed=int(rng.integers(0, 2**31)), verbose=True)
            if sc_ad < best_sc:
                best_a, best_sc = a_ad, sc_ad

    best_a = _norm_nonneg(best_a)
    print("\nFinal: n=%d score=%.8f" % (len(best_a), best_sc))

    # Validation
    try:
        if len(best_a) <= 10000:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("Exact evaluation: %.8f" % exact)
            if not np.isfinite(exact):
                best_a = np.ones(512, dtype=np.float64)
    except Exception as e:
        print("Validation error: %s" % e)

    return [float(x) for x in best_a]
```
