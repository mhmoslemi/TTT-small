def simplex_project(v, s=1.0):
    """Project v onto {a >= 0, sum(a) = s}."""
    n = len(v)
    # Binary search for threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)

    """Compute full convolution of a with itself."""
    return np.convolve(a, a)

    c = np.convolve(a, a)
    c_max = c.max()
    return c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))

import numpy as np
import time

def simplex_project(v, target_sum=1.0):
    """Project vector v onto {a >= 0, sum(a) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)

def simplex_project(v, target_sum=1.0):
    """Project vector v onto {a >= 0, sum(a) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
