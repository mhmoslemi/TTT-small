import numpy as np
import time
from scipy.fft import fft, ifft

def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]

def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _normalize(a, S):
    """Normalize to target sum."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, S / len(a))
    return a * (S / s)

def _subgrad_peak(a, m):
    """Subgradient of b_m w.r.t. a: g[j] = 2*a[m-j] if valid."""
    n = len(a)
    g = np.zeros(n)
    j_lo = max(0, m - n + 1)
    j_hi = min(n - 1, m)
    if j_hi >= j_lo:
        g[j_lo:j_hi+1] = 2.0 * a[m-j_lo : m-j_hi-1 : -1]
    return g

def _mirror_step(a, b, eta, n_peaks=50, mode='excess'):
    """One mirror descent step."""
    n = len(a)
    
    # Select peaks
    if mode == 'excess':
        mean_b = float(np.mean(b))
        excess = b - mean_b
        # Top peaks by excess
        if n_peaks >= 2*n - 1:
            indices = np.arange(len(b))
        else:
            indices = np.argsort(excess)[-n_peaks:]
        weights = np.maximum(excess[indices], 0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(len(indices))
        else:
            weights = weights / wsum
    elif mode == 'softmax':
        temp = 0.01 * float(np.max(b) - np.min(b)) + 1e-10
        exp_b = np.exp(b / temp - np.max(b) / temp)
        weights_all = exp_b / exp_b.sum()
        # Use all peaks
        indices = np.arange(len(b))
        weights = weights_all
    else:  # 'topk'
        if n_peaks >= 2*n - 1:
            indices = np.arange(len(b))
        else:
            indices = np.argsort(b)[-n_peaks:]
        weights = np.ones(len(indices)) / len(indices)
    
    # Compute weighted subgradient
    g = np.zeros(n)
    for w, m in zip(weights, indices):
        if w < 1e-12:

        g += w * _subgrad_peak(a, int(m))

    g -= mean_g
