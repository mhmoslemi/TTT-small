    """Compute linear convolution a * a."""
    n = len(a)
    if n > 512:
        # Use FFT

        fa = np.fft.rfft(a, n=size)

def compute_subgradient(a, active_indices):
    """Compute subgradient of max(c) at a, given active set."""
    n = len(a)
    g = np.zeros(n)
    m = len(active_indices)
    if m == 0:
        return g
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

        # So a[k-hi : k-lo+1] in reverse
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= m
    return g

def project_simplex(v):
    """Project v onto the simplex {x >= 0 : sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Optimize step function sequence to minimize 2*n*max(a*a)/(sum(a))^2.
Uses adaptive subgradient method with active-set selection, projected onto simplex,
combined with LSE-smoothed L-BFGS for initial refinement and local search for escaping
local minima. Exploits scale invariance by working on the probability simplex.
"""

import numpy as np
import time
from scipy.optimize import minimize


def compute_conv(a):
    """Compute linear convolution a*a, using FFT for larger sequences."""
    n = len(a)
    if n <= 512:
        return np.convolve(a, a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, n=size)
    return np.fft.irfft(fa * fa, n=size)


def local_score(a):
    """Compute the objective score locally (faster than evaluate_sequence)."""
    n = len(a)
    c = compute_conv(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * c.max() / (s * s))


def subgradient(a, active):
    """Compute subgradient of max(convolve(a,a)) w.r.t. a at given active indices."""
    n = len(a)
    g = np.zeros(n)
    m = len(active)
    if m == 0:
        return g
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return g / m


def proj_simplex(v):
    """Project vector onto the probability simplex {x >= 0 : sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cumsum - 1.0))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cumsum[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_smoothed_max_conv(a, tau):
    """LSE-smoothed max of convolve(a,a)."""
    c = compute_conv(a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    exp_vals = np.exp(shifted)
    return c_max + tau * np.log(exp_vals.sum())


def lse_gradient(a, tau):
    """Gradient of LSE-smoothed max of convolve(a,a) w.r.t. a (not log-param)."""
    c = compute_conv(a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    w = np.exp(shifted)
    w /= w.sum()
    # Gradient of smoothed max w.r.t. a
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += w[k] * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def run_lbfgs_smooth(a, n, max_time=120):
    """Run L-BFGS on LSE-smoothed objective. Returns best a found."""
    best_a = a.copy()
    best_score = local_score(a)
    deadline = time.monotonic() + max_time

    # Work in log-parameterization: a = softmax(x)
    a_safe = np.maximum(a, 1e-15)
    a_safe = a_safe / a_safe.sum()
    x = np.log(a_safe)

    n_params = len(x)
    taus = [0.05 / n_params, 0.025 / n_params, 0.01 / n_params, 0.005 / n_params,
            0.002 / n_params, 0.001 / n_params]

    for tau in taus:
        if time.monotonic() > deadline:
            break

        def obj(x_):
            a_ = np.exp(x_ - x_.max())
            a_ = a_ / a_.sum()
            return lse_smoothed_max_conv(a_, tau)

        def grad(x_):
            a_ = np.exp(x_ - x_.max())
            a_ = a_ / a_.sum()
            ga = lse_gradient(a_, tau)
            dot = np.dot(a_, ga)
            return a_ * (ga - dot)

        try:
            result = minimize(obj, x, jac=grad, method='L-BFGS-B',
                              options={'maxiter': 300, 'maxfun': 600,
                                       'ftol': 1e-16, 'gtol': 1e-14})
            x = result.x
        except Exception:
            break

        a = np.exp(x - x.max())
        a = a / a.sum()
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score

def local_search(a, deadline, rng):

    iter_count = 0

            a_new = a_new / s

def make_diverse_inits(n, rng):

    inits = []

    inits.append(u / u.sum())

    # Cosine window
    x = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
    x = np.maximum(x, 0.01)
    inits.append(x / x.sum())

    # Two peaks
    p1 = n // 4
    p2 = 3 * n // 4
    x = np.ones(n) * 0.3
    width = max(3, n // 20)
    for p in [p1, p2]:
        for d in range(-width, width + 1):
            idx = (p + d) % n
            x[idx] += 1.0 * np.exp(-((d / width) ** 2))
    inits.append(x / x.sum())

    # Linear ramp (asymmetric)
    x = np.linspace(0.2, 2.0, n)
    inits.append(x / x.sum())

    # Sparse clusters
    x = np.zeros(n)
    n_clusters = max(3, n // 50)
    for _ in range(n_clusters):
        center = rng.integers(0, n)
        width = max(2, n // (4 * n_clusters))
        for d in range(-width, width + 1):
            idx = (center + d) % n
            x[idx] += rng.uniform(0.5, 2.0)
    x = np.maximum(x, 0.001)
    inits.append(x / x.sum())

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Alternating high/low
    x = np.ones(n)
    x[::2] = 1.5
    x[1::2] = 0.6

    final_score = evaluate_sequence(best_a.tolist())
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)
    print(f"Final score: {best_score:.6f} (n={len(best_a)})")

    # Ensure valid output
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()

    return [float(val) for val in best_a]
