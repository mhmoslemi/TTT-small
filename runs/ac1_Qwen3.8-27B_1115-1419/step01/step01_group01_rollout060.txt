```python
"""
Search algorithm for non-negative step functions minimizing
    2 * n * max_k (a * a)_k / (sum a)^2.

The routine works at several lengths and uses a hybrid of:
  - spectral cap-and-rebuild iterations (match a capped convolution target),
  - smooth log-sum-exp Adam-style optimization in log probability space,
  - active-lag multiplicative response updates,
  - small linear-program direction searches (when scipy is available),
  - diverse constructive initializers, including dyadic Bernoulli products
    and smooth bump combinations.

The best sequence found within the time limit is returned.
"""

import numpy as np
import time
import math


def _project_nonneg(a):
    """Project a one-dimensional array to non-negative probabilities."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = a.copy()
    a[~np.isfinite(a)] = 0.0
    np.maximum(a, 0.0, out=a)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a.fill(1.0 / float(a.size))
    else:
        a /= s
    return a


def _next_pow2(x):
    """Smallest power of two >= x."""
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _interp_len(a, n):
    """Interpolate a sequence to length n in the unit interval."""
    n = int(max(1, n))
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64)
    a = a / s
    if a.size == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss <= 1e-300:
        out.fill(1.0 / float(n))
    else:
        out /= ss
    return out


def _conv_self(a, m):
    """Full self-convolution of a, using FFT. Returns length 2n-1."""
    n = int(a.size)
    F = np.fft.rfft(a, m)
    b = np.fft.irfft(F * F, m)
    b = b[:2 * n - 1]
    b = np.maximum(b, 0.0)
    return b


def _score_unit(a, m=None):
    """Score for a sequence (normalization invariant)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    a = a / s
    if m is None:
        m = _next_pow2(2 * n - 1)
    b = _conv_self(a, m)
    bmax = float(b.max()) if b.size else 0.0
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf
    return float(2.0 * n * bmax)


def _correlate(w, a, m):
    """
    Return c_i = sum_j w_{i+j} a_j for valid i, j.
    This is the derivative factor for weighted self-convolution.
    """
    n = int(a.size)
    Fw = np.fft.rfft(w, m)
    Fa = np.fft.rfft(a, m)
    c = np.fft.irfft(Fw * np.conj(Fa), m)
    c = c[:n]
    c = np.maximum(c, 0.0)
    return c


def _log_plus(a):
    """Safe log for positive/non-negative arrays."""
    a = np.asarray(a, dtype=np.float64)
    return np.log(np.maximum(a, 1e-300))


def _soft(v):
    """Softmax to probabilities."""
    v = np.asarray(v, dtype=np.float64)
    v = v - float(np.max(v))
    v = np.clip(v, -745.0, 700.0)
    e = np.exp(v)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(v.shape, 1.0 / max(1, v.size), dtype=np.float64)
    return e / s


def _dyadic_sequence(L, p):
    """
    Sequence from product prod_{j=0}^{L-1} ((1-p) + p z^{2^j}).
    Length is 2^L and sums to 1.
    """
    L = int(max(0, L))
    a = np.array([1.0], dtype=np.float64)
    q = 1.0 - float(p)
    for _ in range(L):
        a = np.concatenate((q * a, float(p) * a))
    s = float(a.sum())
    if s > 0.0:
        a /= s
    else:
        a.fill(1.0 / max(1, a.size))
    return a


def _lacunary_sequence(base, p, target_len):
    """
    Generalized Bernoulli product with shifts base^j.
    Returns a normalized sparse sequence, suitable for interpolation.
    """
    base = int(max(2, base))
    target_len = int(max(16, target_len))
    a = np.array([1.0], dtype=np.float64)
    shift = 1
    p = float(p)
    q = 1.0 - p
    guard = 0
    while a.size < 2 * target_len and a.size < 200000 and guard < 40:
        new = np.zeros(a.size + shift, dtype=np.float64)
        new[:a.size] = q * a
        new[shift:shift + a.size] = p * a
        s = float(new.sum())
        if s > 0.0:
            new /= s
        a = new
        shift *= base
        guard += 1
    s = float(a.sum())
    if s > 0.0:
        a /= s
    else:
        a.fill(1.0 / max(1, a.size))
    return a


def _bump_init(n, K, rng):
    """Smooth non-negative bump initializer."""
    n = int(max(8, n))
    K = int(max(1, K))
    a = np.zeros(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64)
    if K == 1:
        a[:] = 1.0
        return _project_nonneg(a)
    if K == 3:
        pos = np.array([0.05, 0.5, 0.95], dtype=np.float64)
        pos += np.array([0.0, rng.uniform(-0.05, 0.05), 0.0], dtype=np.float64)
    else:
        pos = np.sort(rng.uniform(0.0, 1.0, K))
    pos = np.clip(pos, 0.002, 0.998)
    widths = rng.uniform(0.02, 0.24, K)
    weights = rng.uniform(0.5, 1.5, K)
    centers = pos * float(n - 1)
    wids = widths * float(max(1, n - 1))
    for c, w, wt in zip(centers, wids, weights):
        if w <= 0.5:
            idx = int(min(n - 1, max(0, int(c))))
            a[idx] += max(0.0, float(wt))
            continue
        d = x - c
        mask = np.abs(d) <= w
        if not np.any(mask):
            continue
        dm = d[mask]
        a[mask] += float(wt) * 0.5 * (1.0 + np.cos(np.pi * dm / w))
    return _project_nonneg(a)


def _param_inits(n, rng, include_dyadic=True):
    """Generate a list of diverse initial sequences of length n."""
    n = int(max(16, n))
    out = []
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    li = i + 1.0
    ri = float(n) - i

    out.append(np.ones(n, dtype=np.float64))

    for c in (0.4, 0.9, 1.8, 3.5, 7.0):
        out.append(np.exp(-c * t))
        out.append(np.exp(-c * (1.0 - t)))

    out.append(1.0 / np.sqrt(np.maximum(li * ri, 1.0)))

    for p in (-0.45, -0.15, 0.15, 0.45, 1.0):
        a = np.power(np.maximum(li, 1e-6), p) * np.power(np.maximum(ri, 1e-6), p)
        out.append(a)

    kc = min(28, n)
    for style in range(4):
        a = 3.5 * np.ones(n, dtype=np.float64)
        for k in range(1, kc + 1):
            amp = 0.95 / (float(k) ** 0.75)
            ph = rng.normal(0.0, 0.6)
            if style % 2 == 0:
                a += amp * np.cos(2.0 * np.pi * k * t + ph)
            else:
                a += amp * np.sin(2.0 * np.pi * k * t + ph)
        out.append(np.maximum(a, 0.0))

    for K in (3, 5, 8, 12):
        out.append(_bump_init(n, K, rng))

    if include_dyadic and (n & (n - 1)) == 0 and 64 <= n <= 16384:
        L = int(round(math.log2(n)))
        if (1 << L) == n:
            for p in (0.10, 0.18, 0.26, 0.34, 0.42, 0.48,
                      0.52, 0.58, 0.66, 0.74, 0.82, 0.90):
                out.append(_dyadic_sequence(L, p))

    if n <= 16384:
        for base, p in ((3, 0.30), (3, 0.45), (3, 0.55), (4, 0.40), (4, 0.60)):
            a = _lacunary_sequence(base, p, n)
            if a.size >= 16 and a.size <= 8 * n:
                out.append(_interp_len(a, n))

    return out


def _first_item(pair):
    return pair[0]


def _score_and_dedupe(cands, max_count, threshold=0.055):
    """Score candidates and keep diverse ones."""
    scored = []
    seen = []
    max_seen = max(2 * int(max_count), 12)
    for a in cands:
        a = _project_nonneg(a)
        sc = _score_unit(a)
        if not np.isfinite(sc):
            continue
        ok = True
        for b in seen:
            d = float(np.sqrt(np.sum((a - b) * (a - b))))
            if d < threshold:
                ok = False
                break
        if ok:
            scored.append((float(sc), a))
            seen.append(a)
            if len(seen) >= max_seen:
                break
    scored.sort(key=_first_item)
    return scored


def _unique_candidates(cands, max_count, threshold=0.055):
    """Keep a diverse subset of candidates."""
    out = []
    for a in cands:
        a = _project_nonneg(a)
        if a.size == 0:
            continue
        ok = True
        for b in out:
            if b.size != a.size:
                continue
            d = float(np.sqrt(np.sum((a - b) * (a - b))))
            if d < threshold:
                ok = False
                break
        if ok:
            out.append(a)
            if len(out) >= int(max_count):
                break
    return out


def _adam_smooth(a_init, max_seconds, seed, fast=False):
    """
    Smooth log-sum-exp gradient/Adam optimization in log-probability space.
    Works with a normalized probability vector.
    """
    a = _project_nonneg(a_init)
    n = int(a.size)
    m = _next_pow2(2 * n - 1)
    v = _log_plus(a)
    v = v - float(np.mean(v))
    v = np.clip(v, -45.0, 45.0)

    best = a.copy()
    best_sc = _score_unit(a, m)
    rng = np.random.default_rng(seed)

    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    den1 = 1.0
    den2 = 1.0

    start = time.monotonic()
    deadline = start + max(1.0, float(max_seconds))
    last_improve = start

    tau_start = 0.50 if not fast else 0.80
    tau_end = 0.018 if not fast else 0.07
    lr_start = 0.12 if not fast else 0.08
    max_step = 0.34
    stuck_limit = 45.0 if not fast else 18.0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = max(1e-9, now - start)
        progress = min(1.0, elapsed / max_seconds)

        if now - last_improve > min(stuck_limit, 0.30 * max_seconds):
            sigma = (0.12 + 0.32 * (1.0 - progress)) if not fast else 0.18
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -45.0, 45.0)
            mu.fill(0.0)
            nu.fill(0.0)
            den1 = 1.0
            den2 = 1.0
            last_improve = now
            continue

        a = _soft(v)
        b = _conv_self(a, m)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 1e-300:
            break

        sc = 2.0 * float(n) * bmax
        if sc < best_sc - 1e-12:
            best_sc = float(sc)
            best = a.copy()
            last_improve = time.monotonic()

        tau = bmax * (tau_start + (tau_end - tau_start) * progress)
        if rng.random() < 0.40:
            tau *= float(rng.uniform(0.45, 2.2))
        if tau <= 1e-300:
            tau = bmax * 1e-3

        x = (b - bmax) / tau
        x = np.clip(x, -745.0, 0.0)
        p = np.exp(x)
        z = float(p.sum())
        if not np.isfinite(z) or z <= 1e-300:
            break
        p /= z

        c = _correlate(p, a, m)
        mean_ac = float(np.dot(a, c))
        gv = a * (c - mean_ac)

        grms = float(np.sqrt(np.mean(gv * gv)))
        if not np.isfinite(grms) or grms < 1e-14:
            v += rng.normal(0.0, 0.10, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -45.0, 45.0)
            last_improve = time.monotonic()
            continue

        gv /= grms

        mu = b1 * mu + (1.0 - b1) * gv
        nu = b2 * nu + (1.0 - b2) * (gv * gv)
        den1 = den1 * b1 + (1.0 - b1)
        den2 = den2 * b2 + (1.0 - b2)

        mhat = mu / max(1e-12, den1)
        vhat = nu / max(1e-12, den2)

        lr = lr_start * (1.0 - 0.65 * progress)
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -max_step, max_step)

        v = v - lr * step
        v = v - float(np.mean(v))
        v = np.clip(v, -45.0, 45.0)

    a = _soft(v)
    sc = _score_unit(a, m)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
        best = a
    return _project_nonneg(best), float(best_sc)


def _response_smooth(a_init, max_seconds, seed):
    """
    Active-lag multiplicative response updates.  Given soft top-lag weights p,
    reduce log-a where these lags respond strongly.
    """
    a = _project_nonneg(a_init)
    n = int(a.size)
    m = _next_pow2(2 * n - 1)
    v = _log_plus(a)
    v = v - float(np.mean(v))
    v = np.clip(v, -45.0, 45.0)

    best = a.copy()
    best_sc = _score_unit(a, m)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(1.0, float(max_seconds))
    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        progress = min(1.0, (now - start) / max_seconds)

        if now - last_improve > min(45.0, 0.35 * max_seconds):
            v += rng.normal(0.0, 0.16, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -45.0, 45.0)
            last_improve = now
            continue

        a = _soft(v)
        b = _conv_self(a, m)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 1e-300:
            break
        sc = 2.0 * float(n) * bmax
        if sc < best_sc - 1e-12:
            best_sc = float(sc)
            best = a.copy()
            last_improve = time.monotonic()

        tau = bmax * (0.090 + 0.460 * progress)
        if rng.random() < 0.5:
            tau *= float(rng.uniform(0.5, 2.0))
        if tau <= 1e-300:
            tau = bmax * 1e-3

        x = (b - bmax) / tau
        x = np.clip(x, -745.0, 0.0)
        p = np.exp(x)
        z = float(p.sum())
        if not np.isfinite(z) or z <= 1e-300:
            break
        p /= p.sum()

        # Emphasize a few strongest lags, but keep smooth mass.
        top_k = max(1, int(0.025 * p.size))
        if top_k < p.size:
            idx = np.argpartition(b, -top_k)[-top_k:]
            p[idx] *= 2.0
            z2 = float(p.sum())
            if z2 > 0.0:
                p /= z2

        c = _correlate(p, a, m)
        cmean = float(c.mean())
        if not np.isfinite(cmean) or cmean <= 1e-300:
            break

        q = c / max(cmean, 1e-300)
        eta = 0.22 * (1.0 - 0.55 * progress)
        delta = -eta * np.log(np.maximum(q, 1e-300))
        delta = np.clip(delta, -1.0, 1.0)

        v = v + delta
        v = v - float(np.mean(v))
        v = np.clip(v, -45.0, 45.0)

    a = _soft(v)
    sc = _score_unit(a, m)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
        best = a
    return _project_nonneg(best), float(best_sc)


def _capped_target(b, P):
    """
    Build a target convolution with entries <= P and sum ~ 1 by redistributing
    capped mass into lower lags.
    """
    bt = np.asarray(b, dtype=np.float64).copy()
    if bt.size == 0:
        return bt
    s = float(bt.sum())
    if s <= 1e-300:
        bt.fill(1.0 / float(bt.size))
        return bt
    bt /= s
    P = float(max(P, 1.0 / float(bt.size) * 1.02))

    for _ in range(60):
        excess = bt > P
        D = float(np.sum(bt[excess] - P)) if np.any(excess) else 0.0
        if D <= 1e-14:
            break
        free = bt < P - 1e-15
        if not np.any(free):
            break
        wt = bt[free] + 1e-20
        wt_sum = float(wt.sum())
        if wt_sum <= 1e-300:
            break
        bt[free] += D * (wt / wt_sum)

    bt = np.maximum(bt, 0.0)
    s = float(bt.sum())
    if s > 0.0:
        bt /= s
    return bt


def _spectral_smooth(a_init, target_score, max_seconds, seed):
    """
    Homotopy spectral matching:
      a -> a such that a*a approximates a capped target with peak P.
    """
    a = _project_nonneg(a_init)
    n = int(a.size)
    m = _next_pow2(2 * n - 1)
    L = 2 * n - 1

    best = a.copy()
    best_sc = _score_unit(a, m)

    target_P = max(1.0, float(n) * 0.0)  # placeholder, replaced below
    target_P = max(target_score / (2.0 * float(n)), (1.0 / L) * 1.02)

    b = _conv_self(a, m)
    cur_P = float(b.max())
    if not np.isfinite(cur_P) or cur_P <= 0.0:
        return best, best_sc
    if cur_P <= target_P * (1.0 + 1e-7):
        return best, best_sc

    start = time.monotonic()
    deadline = start + max(1.0, float(max_seconds))
    steps = 7
    ratio = (target_P / cur_P) ** (1.0 / float(steps))
    P = cur_P

    for _ in range(steps):
        P *= ratio
        if P < target_P:
            P = target_P

        b = _conv_self(a, m)
        bt = _capped_target(b, P)
        if time.monotonic() >= deadline:
            break

        T = np.fft.fft(bt, m)
        F = np.fft.fft(a, m)
        fmax = max(1.0, float(np.max(np.abs(F))))
        eps = 1e-12 * fmax

        inner_limit = 34
        for inner in range(inner_limit):
            if time.monotonic() >= deadline:
                break

            absF = np.abs(F)
            denom = np.where(absF > eps, F, eps + 0.0j)
            Fr = 0.5 * (F + T / denom)

            step = 0.75 - 0.45 * (float(inner) / float(max(1, inner_limit - 1)))
            step = float(np.clip(step, 0.15, 1.0))
            Fnew = F + step * (Fr - F)

            anew = np.fft.ifft(Fnew, m).real[:n]
            anew = np.maximum(anew, 0.0)
            s = float(anew.sum())
            if not np.isfinite(s) or s <= 1e-300:
                break
            anew /= s
            anew = _project_nonneg(anew)

            blend = 0.80 if inner < 6 else (0.55 if inner < 16 else 0.32)
            a = (1.0 - blend) * a + blend * anew
            a = _project_nonneg(a)
            F = np.fft.fft(a, m)
            fmax = max(1.0, float(np.max(np.abs(F))))
            eps = 1e-12 * fmax

            if inner % 5 == 0:
                sc = _score_unit(a, m)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = float(sc)
                    best = a.copy()

    b = _conv_self(a, m)
    sc = 2.0 * float(n) * float(np.max(b)) if b.size else np.inf
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
        best = a
    return _project_nonneg(best), float(best_sc)


def _lp_direction(f, M, ub, time_limit):
    """
    Van-off direction from LP:
      maximize sum g
      s.t. 0 <= g <= ub, f*g <= M for all lags.
    Returns normalized g or None.
    """
    try:
        import scipy.optimize as opt
    except Exception:
        return None

    n = int(f.size)
    if n < 96 or n > 2048:
        return None
    L = 2 * n - 1

    f_use = np.asarray(f, dtype=np.float64).copy()
    s = float(f_use.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    f_use /= s
    if f_use.size:
        fmax = float(f_use.max())
    else:
        fmax = 1.0
    eps = 1e-12 * max(1e-12, fmax)
    f_use = f_use + eps
    f_use /= f_use.sum()

    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f_use

    b_ub = np.full(L, float(M), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, float(ub))] * n

    try:
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(1.0, float(time_limit))},
        )
    except Exception:
        return None

    if res is None:
        return None
    try:
        g = np.asarray(res.x, dtype=np.float64)
    except Exception:
        return None
    if g.size != n or not np.all(np.isfinite(g)):
        return None
    gs = float(g.sum())
    if gs <= 1e-15:
        return None
    return g / gs


def _lp_refine(a_init, max_seconds, max_n=2048):
    """
    Use LP directions to move inside the feasible sublevel set of current
    convolution peak.  This follows the active-constraint LP idea from the
    previous state of the art, but with short capped time budgets.
    """
    a = _project_nonneg(a_init)
    n = int(a.size)
    if n < 96 or n > max_n:
        return a, _score_unit(a)

    m = _next_pow2(2 * n - 1)
    start = time.monotonic()
    deadline = start + max(1.0, float(max_seconds))
    best = a.copy()
    best_sc = _score_unit(a, m)
    fails = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break
        tl = min(18.0, rem / 3.0)
        b = _conv_self(a, m)
        M = float(b.max())
        if not np.isfinite(M) or M <= 1e-300:
            break
        g = _lp_direction(a, M, 100.0, tl)
        if g is None:
            fails += 1
            if fails > 3:
                break
            a = _project_nonneg(a + 1e-3 * np.ones(n))
            continue

        g = _project_nonneg(g)
        improved_here = False
        old_a = a
        old_sc = best_sc
        for t in (0.05, 0.10, 0.18, 0.28, 0.42, 0.60, 0.80):
            cand = (1.0 - t) * a + t * g
            cand = _project_nonneg(cand)
            sc = _score_unit(cand, m)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = float(sc)
                best = cand.copy()
                a = cand
                improved_here = True
        if not improved_here:
            fails += 1
            a = old_a
            if fails > 5:
                break
            # small perturbation to escape a bad LP direction
            a = _project_nonneg(a + 1e-4 * np.random.default_rng(int(a.sum() * 1e6) & 0xffff).random(n))

    sc = _score_unit(a, m)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
        best = a
    return _project_nonneg(best), float(best_sc)


def _process_stage(n, stage_deadline, current_a, target_score, rng):
    """Run all local/global refinement methods for one length."""
    n = int(max(96, n))
    now = time.monotonic()
    stage_time = max(0.0, float(stage_deadline) - now)
    if stage_time <= 10.0:
        if current_a is None:
            return None, np.inf
        return current_a, _score_unit(current_a)

    cands = []
    if current_a is not None:
        cands.append(_interp_len(current_a, n))
    cands.extend(_param_inits(n, rng, include_dyadic=True))
    scored = _score_and_dedupe(cands, 8)

    if scored:
        best_score, best_a = scored[0]
    else:
        best_a = np.ones(n, dtype=np.float64)
        best_a = _project_nonneg(best_a)
        best_score = _score_unit(best_a)

    print("    stage n=%d initial best=%.6f" % (n, best_score))

    # LP refinement at small lengths, if available and time permits.
    if n <= 2048 and stage_time > 140.0:
        lp_budget = min(130.0, 0.20 * stage_time)
        lp_budget = max(0.0, lp_budget - (time.monotonic() - now))
        if lp_budget > 12.0:
            seed = int(rng.integers(0, 2**31 - 1))
            a_lp, sc_lp = _lp_refine(best_a, lp_budget, max_n=2048)
            if np.isfinite(sc_lp) and sc_lp < best_score - 1e-12:
                best_score = float(sc_lp)
                best_a = a_lp
                print("      LP improved to %.6f" % best_score)

    # Spectral capped matching on the best few starts.
    spec_budget = min(stage_time * 0.26, 300.0 if n <= 4096 else 220.0)
    spec_budget = max(0.0, spec_budget - (time.monotonic() - now))
    if spec_budget > 15.0:
        spec_starts = []
        if best_a is not None:
            spec_starts.append(best_a)
        for _, cand in scored[:2]:
            spec_starts.append(cand)
        spec_starts = _unique_candidates(spec_starts, 2)
        if spec_starts:
            per = spec_budget / float(len(spec_starts))
            for cand in spec_starts:
                if time.monotonic() >= stage_deadline:
                    break
                alloc = min(per, max(5.0, stage_deadline - time.monotonic() - 5.0))
                if alloc > 8.0:
                    seed = int(rng.integers(0, 2**31 - 1))
                    a_sp, sc_sp = _spectral_smooth(cand, target_score, alloc, seed)
                    if np.isfinite(sc_sp) and sc_sp < best_score - 1e-12:
                        best_score = float(sc_sp)
                        best_a = a_sp
                        print("      spectral improved to %.6f" % best_score)
    stage_time = max(0.0, stage_deadline - time.monotonic())

    # Adam smoothing on several diverse starts.
    adam_starts = []
    if best_a is not None:
        adam_starts.append(best_a)
    for _, cand in scored[:4]:
        adam_starts.append(cand)
    adam_starts = _unique_candidates(adam_starts, 4)
    if adam_starts:
        weights = (0.50, 0.28, 0.16, 0.06)
        used = 0.0
        for idx, cand in enumerate(adam_starts):
            if time.monotonic() >= stage_deadline:
                break
            rem = max(0.0, stage_deadline - time.monotonic())
            w = weights[idx] if idx < len(weights) else 0.05
            alloc = rem * w
            if idx == 0:
                alloc = min(alloc + used * 0.5, rem)
            alloc = max(0.0, min(alloc, rem - 10.0))
            used += alloc
            if alloc > 25.0:
                seed = int(rng.integers(0, 2**31 - 1))
                fast = stage_time < 240.0
                a_ad, sc_ad = _adam_smooth(cand, alloc, seed, fast=fast)
                if np.isfinite(sc_ad) and sc_ad < best_score - 1e-12:
                    best_score = float(sc_ad)
                    best_a = a_ad
                    print("      adam %d improved to %.6f" % (idx, best_score))

    # Response polish on the stage best.
    rem = max(0.0, stage_deadline - time.monotonic())
    if rem > 45.0 and best_a is not None:
        alloc = min(160.0, rem - 15.0)
        if alloc > 20.0:
            seed = int(rng.integers(0, 2**31 - 1))
            a_rp, sc_rp = _response_smooth(best_a, alloc, seed)
            if np.isfinite(sc_rp) and sc_rp < best_score - 1e-12:
                best_score = float(sc_rp)
                best_a = a_rp
                print("      response improved to %.6f" % best_score)

    # Final quick Adam polish if time remains.
    rem = max(0.0, stage_deadline - time.monotonic())
    if rem > 60.0 and best_a is not None:
        alloc = min(150.0, rem - 15.0)
        if alloc > 25.0:
            seed = int(rng.integers(0, 2**31 - 1))
            a_ad, sc_ad = _adam_smooth(best_a, alloc, seed, fast=True)
            if np.isfinite(sc_ad) and sc_ad < best_score - 1e-12:
                best_score = float(sc_ad)
                best_a = a_ad
                print("      final adam improved to %.6f" % best_score)

    if best_a is None:
        best_a = np.ones(n, dtype=np.float64)
        best_score = _score_unit(best_a)
    return _project_nonneg(best_a), float(best_score)


def _validate_and_make_output(best_a, history, base_deadline):
    """Final validation with evaluate_sequence (when feasible) and safe output."""
    cands = []
    if best_a is not None:
        cands.append(best_a)
    for _, a in history:
        cands.append(a)

    cands = _unique_candidates(cands, 4)
    scored = []
    for a in cands:
        sc = _score_unit(a)
        if np.isfinite(sc):
            scored.append((float(sc), a))
    scored.sort(key=_first_item)

    if not scored:
        out = np.ones(512, dtype=np.float64)
        return [float(x) for x in out]

    best_out = scored[0][1]
    best_exact = np.inf

    for pair in scored[:3]:
        a = pair[1]
        if a.size > 10000:
            continue
        if time.monotonic() > base_deadline - 12.0:
            break
        seq = [float(x) for x in a]
        try:
            ex = evaluate_sequence(seq)
        except Exception:
            ex = np.inf
        if np.isfinite(ex):
            if ex < best_exact:
                best_exact = float(ex)
                best_out = a
        print("final exact score: %.8f  n=%d" % (float(ex) if np.isfinite(ex) else np.inf, a.size))

    out = np.asarray(best_out, dtype=np.float64).ravel()
    if out.size == 0:
        out = np.ones(512, dtype=np.float64)
    out[~np.isfinite(out)] = 0.0
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if s <= 1e-12:
        out.fill(1.0 / float(out.size))
    else:
        out /= s
    # Keep values safely finite and non-negative.
    out = np.minimum(out, 1000.0)
    return [float(x) for x in out]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point required by the harness.
    Runs the hybrid search and returns the best non-negative sequence found.
    """
    seed = int(seed)
    budget_s = float(max(120.0, float(budget_s)))
    start = time.monotonic()
    base_total = budget_s - 70.0
    base_deadline = start + max(60.0, base_total)
    rng = np.random.default_rng(seed)

    # Load previous construction if available.
    parent = None
    try:
        cand = height_sequence_1
        if isinstance(cand, list):
            arr = np.asarray(cand, dtype=np.float64)
        else:
            arr = np.asarray(cand, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size > 0:
            arr = np.asarray(arr, dtype=np.float64)
            arr[~np.isfinite(arr)] = 0.0
            np.maximum(arr, 0.0, out=arr)
            s = float(arr.sum())
            if s > 1e-15:
                parent = arr / s
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(2048, dtype=np.float64) / 2048.0
    parent = _project_nonneg(parent)
    best_a = parent
    best_score = _score_unit(parent)
    print("Initial FFT score: %.8f n=%d" % (best_score, parent.size))

    history = [(float(best_score), parent)]

    # Choose stage lengths based on current bound.
    if not np.isfinite(best_score) or best_score > 1.55:
        stage_defs = [(1024, 0.20), (2048, 0.35), (4096, 0.45)]
    elif best_score > 1.515:
        stage_defs = [(1024, 0.12), (2048, 0.25), (4096, 0.63)]
    elif best_score > 1.505:
        stage_defs = [(2048, 0.25), (4096, 0.55), (8192, 0.20)]
    else:
        stage_defs = [(2048, 0.15), (4096, 0.45), (8192, 0.40)]

    main_deadline = base_deadline - 60.0
    for n, frac in stage_defs:
        now = time.monotonic()
        if now + 50.0 >= main_deadline:
            break
        # Do not over-reserve the last stage; leave some slack.
        stage_time = max(20.0, min((main_deadline - start) * frac, main_deadline - now - 35.0))
        stage_deadline = min(main_deadline - 20.0, now + stage_time)
        stage_target = min(1.5005, max(1.0, best_score - 0.001))
        print("Stage n=%d budget=%.1fs current_best=%.8f target=%.6f" %
              (n, stage_time, best_score, stage_target))
        a_new, sc_new = _process_stage(n, stage_deadline, best_a, stage_target, rng)
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_score - 1e-12:
            best_score = float(sc_new)
            best_a = a_new
            history.append((best_score, best_a))
            print("After stage n=%d best=%.8f" % (n, best_score))

    # If there is still meaningful time and the bound is not yet clearly below
    # the target, try one extra longer sequence.
    now = time.monotonic()
    if now + 260.0 < base_deadline and best_score > 1.5028 and best_a is not None:
        extra_target = min(1.5005, max(1.0, best_score - 0.001))
        n_extra = 8192 if best_a.size < 8192 else 10240
        # Use a non-power length only if not too odd; FFT padding handles it.
        stage_time = min(900.0, base_deadline - now - 70.0)
        if stage_time > 200.0:
            a_new, sc_new = _process_stage(n_extra, base_deadline - 70.0, best_a, extra_target, rng)
            if a_new is not None and np.isfinite(sc_new) and sc_new < best_score - 1e-12:
                best_score = float(sc_new)
                best_a = a_new
                history.append((best_score, best_a))
                print("Extra n=%d best=%.8f" % (n_extra, best_score))

    # Final safe output.
    out = _validate_and_make_output(best_a, history, base_deadline)
    print("Returning sequence length=%d fft_score=%.8f" % (len(out), best_score))
    return out
```
