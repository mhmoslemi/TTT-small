```python
"""
Search for nonnegative coefficient sequences minimizing

    2 * n * max(np.convolve(a, a)) / (sum(a)^2)

using a multi-state local search in log-probability space.

The search uses:
  - FFT-based convex-conjugate evaluation of the self-convolution,
  - Euclidean-projected and mirror-descent style gradient steps,
  - alternating "best response" minimization of max(f*b) over b,
  - adaptive active-set / log-sum-exp weights for the maximum coefficient,
  - multiple initializations, including parent sequence, resampled lengths,
    perturbed parent, and random smooth profiles,
  - throttled exact np.convolve validation and an optional final L-BFGS polish.
"""

import time
import numpy as np

INF = float("inf")
EPS = 1e-300
MAX_N = 10000


def next_fast_len(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonneg(v):
    """Return a finite nonnegative vector normalized to sum 1."""
    arr = np.asarray(v, dtype=np.float64).ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones(max(1, arr.size), dtype=np.float64)
        s = float(arr.size)
    return arr / s


def a_from_w(w):
    """Stable softmax to a probability vector."""
    w = np.asarray(w, dtype=np.float64).ravel()
    n = w.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    if n:
        w -= float(np.max(w))
    w = np.clip(w, -745.0, 0.0)
    e = np.exp(w)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def log_from_a(a):
    """Centered log of a positive-dimensional probability vector."""
    a = normalize_nonneg(a)
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w = np.where(np.isfinite(w), w, 0.0)
    if w.size:
        w -= float(np.max(w))
    return w


def resample_prob(v, m):
    """Coventa-like resampling of a probability mass via its CDF."""
    v = normalize_nonneg(v)
    m = int(max(1, m))
    n = v.size
    if m == n:
        return v.copy()
    F = np.empty(n + 1, dtype=np.float64)
    F[0] = 0.0
    if n:
        F[1:] = np.cumsum(v)
    else:
        F[1:] = 0.0
    x_src = np.arange(n + 1, dtype=np.float64)
    x_tgt = np.linspace(0.0, float(n), m + 1, dtype=np.float64)
    F_tgt = np.interp(x_tgt, x_src, F)
    q = np.diff(F_tgt)
    q = np.where(np.isfinite(q), q, 0.0)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 0.0:
        q = np.ones(m, dtype=np.float64) / float(m)
    else:
        q = q / s
    return q


def score_from_A(A, n, N):
    """Full linear self-convolution from DFT A and normalized score."""
    c = np.fft.irfft(A * A, N)
    c = c[: max(0, 2 * n - 1)]
    if c.size == 0:
        return np.zeros(0, dtype=np.float64), INF
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return c, INF
    return c, float(2.0 * n * mx)


def fft_score_only(a, N):
    """FFT-based objective for a normalized nonnegative sequence."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return INF
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[: 2 * n - 1]
    if c.size == 0:
        return INF
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return INF
    return float(2.0 * n * mx)


def corr_from_q(q, A_fixed, N, phase, n):
    """
    Compute g_i = sum_k q_k f_{k-i} for fixed f with DFT A_fixed.
    This is the unprojected gradient of q dot (f * variable) with respect to variable.
    """
    P = np.fft.rfft(q, N)
    g = np.fft.irfft(P * phase * np.conjugate(A_fixed), N)
    g = g[n - 1: 2 * n - 1]
    return np.where(np.isfinite(g), g, 0.0)


def make_q(b, beta, top_mix=0.0):
    """
    Distribution over convolution coefficients.

    Uses a scaled log-sum-exp around the maximum coefficient. Optionally blends
    with a uniform distribution over the top K coefficients.
    """
    b = np.asarray(b, dtype=np.float64).ravel()
    m = b.size
    if m == 0:
        return np.ones(1, dtype=np.float64)
    L = float(np.max(b))
    if not np.isfinite(L) or L <= 0.0:
        return np.full(m, 1.0 / float(m), dtype=np.float64)

    x = (b - L) / L
    x = np.where(np.isfinite(x), x, -1e300)
    x = np.clip(x, -1000.0, 0.0)
    y = np.clip(float(beta) * x, -745.0, 0.0)
    e = np.exp(y)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        q = np.zeros(m, dtype=np.float64)
        q[int(np.argmax(b))] = 1.0
        return q
    q = e / s

    if top_mix > 0.0 and m > 2:
        K = max(16, min(512, int(0.02 * m)))
        if K >= m:
            qt = q.copy()
        else:
            if K == 1:
                idx = np.array([int(np.argmax(b))], dtype=np.int64)
            else:
                idx = np.argpartition(b, -K)[-K:]
            qt = np.zeros(m, dtype=np.float64)
            qt[idx] = 1.0 / float(K)
        mm = float(np.clip(top_mix, 0.0, 0.7))
        q = (1.0 - mm) * q + mm * qt
    return q


def adam_update(w, grad, m, v, t, lr, beta1=0.9, beta2=0.95):
    """One Adam step in log-coordinate space."""
    t += 1
    m = m * beta1 + (1.0 - beta1) * grad
    g2 = grad * grad
    v = v * beta2 + (1.0 - beta2) * g2

    mhat = m / (1.0 - beta1 ** t)
    vhat = v / (1.0 - beta2 ** t)
    d = mhat / (np.sqrt(vhat) + 1e-12)
    d = np.where(np.isfinite(d), d, 0.0)
    d = np.clip(d, -3.0, 3.0)

    w_new = w - float(lr) * d
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    if w_new.size:
        w_new -= float(np.max(w_new))
    w_new = np.clip(w_new, -745.0, 0.0)
    return w_new, m, v, t


def apply_a(state, a, w=None, reset_b=False, reset_moments=False):
    """Set state.a to a and recompute spectra, convolution, score."""
    n = state["n"]
    N = state["N"]
    a = normalize_nonneg(a)
    if a.size != n:
        a = resample_prob(a, n)
        a = normalize_nonneg(a)

    if w is None:
        w = log_from_a(a)
    else:
        w = np.asarray(w, dtype=np.float64).ravel()
        if w.size != n:
            w = log_from_a(a)
        else:
            w = np.where(np.isfinite(w), w, 0.0)
            if n:
                w -= float(np.max(w))
            w = np.clip(w, -745.0, 0.0)

    state["a"] = a
    state["w"] = w

    A = np.fft.rfft(a, N)
    bconv, score = score_from_A(A, n, N)
    state["A"] = A
    state["F"] = A
    state["FR"] = state["phase"] * np.conjugate(A)
    state["bconv"] = bconv
    state["score"] = score

    if reset_moments:
        state["m"] = np.zeros(n, dtype=np.float64)
        state["v"] = np.zeros(n, dtype=np.float64)
        state["t"] = 0

    if reset_b or state["b"].size != n or not np.all(np.isfinite(state["b"])):
        state["b"] = a.copy()
        state["wb"] = w.copy()
        state["mb"] = np.zeros(n, dtype=np.float64)
        state["vb"] = np.zeros(n, dtype=np.float64)
        state["tb"] = 0


def make_state(a_init, seed=0):
    """Create a search state from an initial nonnegative sequence."""
    rng_seed = int(seed)
    a = normalize_nonneg(a_init)
    if a.size < 16:
        a = np.ones(4096, dtype=np.float64) / 4096.0
    if a.size > MAX_N:
        a = resample_prob(a, 8192)

    n = int(a.size)
    N = next_fast_len(3 * n - 2)
    ph = np.arange(N // 2 + 1, dtype=np.float64)
    phase = np.exp(-2j * np.pi * ph * (n - 1) / float(N))

    state = {
        "n": n,
        "N": N,
        "phase": phase,
        "a": a,
        "w": np.zeros(n, dtype=np.float64),
        "m": np.zeros(n, dtype=np.float64),
        "v": np.zeros(n, dtype=np.float64),
        "t": 0,
        "b": a.copy(),
        "wb": np.zeros(n, dtype=np.float64),
        "mb": np.zeros(n, dtype=np.float64),
        "vb": np.zeros(n, dtype=np.float64),
        "tb": 0,
        "A": np.zeros(N // 2 + 1, dtype=np.complex128),
        "F": np.zeros(N // 2 + 1, dtype=np.complex128),
        "FR": np.zeros(N // 2 + 1, dtype=np.complex128),
        "bconv": np.zeros(max(0, 2 * n - 1), dtype=np.float64),
        "score": INF,
        "beta_s": 5.0,
        "beta_c": 5.0,
        "lr_s": 0.1,
        "lr_m": 0.3,
        "lr_b": 0.2,
        "stall": 0,
        "cross_stall": 0,
        "tick": 0,
        "elapsed_frac": 0.0,
        "rng": np.random.default_rng(rng_seed),
    }
    apply_a(state, a, reset_b=True, reset_moments=True)
    return state


def self_step(state, mode="euclid", elapsed_frac=0.0):
    """
    One self-convolution improvement step.

    mode='euclid': Adam step on the projected Euclidean log-gradient.
    mode='hard':   mirror step using only the top active convolution coefficients.
    """
    a = state["a"]
    w = state["w"]
    bconv = state["bconv"]
    N = state["N"]
    n = state["n"]
    phase = state["phase"]

    if not np.isfinite(state["score"]):
        apply_a(state, a, reset_moments=True)
        return False

    beta = state["beta_s"]
    top_mix = 0.0
    if beta > 10.0:
        top_mix = min(0.5, (beta - 10.0) / 80.0)

    if mode == "hard":
        q = np.zeros(bconv.size, dtype=np.float64)
        K = max(8, min(64, int(0.005 * max(1, bconv.size))))
        if bconv.size == 0:
            return False
        if K >= bconv.size:
            q[:] = 1.0 / float(bconv.size)
        elif K == 1:
            q[int(np.argmax(bconv))] = 1.0
        else:
            idx = np.argpartition(bconv, -K)[-K:]
            q[idx] = 1.0 / float(K)
    else:
        q = make_q(bconv, beta, top_mix)

    g = 2.0 * corr_from_q(q, state["A"], N, phase, n)
    g = np.where(np.isfinite(g), g, 0.0)

    best_trial = None

    if mode == "euclid":
        lam = float(np.dot(a, g))
        grad = a * (g - lam)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        nw, mnew, vnew, tnew = adam_update(
            w, grad, state["m"], state["v"], state["t"], state["lr_s"]
        )

        for scale in (1.0, 0.5, 0.25):
            trial_w = w + scale * (nw - w)
            trial_a = a_from_w(trial_w)
            sc = fft_score_only(trial_a, N)
            if np.isfinite(sc):
                if best_trial is None or sc < best_trial[0]:
                    best_trial = (sc, trial_w, trial_a)
                if sc < state["score"] - 1e-12:
                    apply_a(state, trial_a, trial_w)
                    state["m"] = mnew
                    state["v"] = vnew
                    state["t"] = tnew
                    if scale >= 0.99:
                        state["lr_s"] = min(1.0, state["lr_s"] * 1.07)
                    else:
                        state["lr_s"] = max(0.01, state["lr_s"] * 0.85)
                    state["stall"] = 0
                    return True

        if best_trial is not None:
            sc0, tw, ta = best_trial
            if state["stall"] >= 8 and elapsed_frac < 0.8:
                temp = 1e-5 + 6e-4 * max(0.0, 1.0 - elapsed_frac)
                if sc0 < state["score"] + temp and state["rng"].random() < 0.22:
                    apply_a(state, ta, tw)
                    state["m"] = mnew
                    state["v"] = vnew
                    state["t"] = tnew
                    state["stall"] = 5
                    return True

        state["lr_s"] = max(0.01, state["lr_s"] * 0.72)
        state["stall"] += 1
        return False

    # Mirror / hard step.
    raw = g - float(np.mean(g))
    rmax = float(np.max(np.abs(raw)))
    if not np.isfinite(rmax) or rmax <= 1e-300:
        state["stall"] += 1
        return False
    raw = raw / rmax

    lr = state["lr_m"]
    cand = []
    for scale in (lr, 0.5 * lr, 1.5 * lr):
        trial_w = w - scale * raw
        trial_a = a_from_w(trial_w)
        sc = fft_score_only(trial_a, N)
        if np.isfinite(sc):
            cand.append((sc, scale, trial_w, trial_a))

    if cand:
        cand.sort(key=lambda x: x[0])
        sc0, scale0, tw0, ta0 = cand[0]
        if sc0 < state["score"] - 1e-12:
            apply_a(state, ta0, tw0)
            if sc0 < state["score"] - 1e-8:
                state["lr_m"] = min(3.0, lr * 1.15)
            else:
                state["lr_m"] = max(0.05, lr * 0.9)
            state["stall"] = 0
            return True

        if state["stall"] >= 12 and elapsed_frac < 0.7:
            temp = 1e-5 + 4e-4 * max(0.0, 1.0 - elapsed_frac)
            if sc0 < state["score"] + temp and state["rng"].random() < 0.18:
                apply_a(state, ta0, tw0)
                state["stall"] = 6
                return True

    state["lr_m"] = max(0.05, state["lr_m"] * 0.8)
    state["stall"] += 1
    return False


def cross_step_b(state):
    """
    One step toward solving min_{b>=0,sum b=1} max_k (f*b)_k for fixed f.
    Uses mirror-descent style Adam in log(b).
    """
    n = state["n"]
    N = state["N"]
    phase = state["phase"]
    F = state["F"]
    b = state["b"]
    wb = state["wb"]

    if b.size != n or not np.all(np.isfinite(b)):
        b = state["a"]
        wb = state["w"]
        state["b"] = b.copy()
        state["wb"] = wb.copy()
        state["mb"] = np.zeros(n, dtype=np.float64)
        state["vb"] = np.zeros(n, dtype=np.float64)
        state["tb"] = 0
        return False

    B = np.fft.rfft(b, N)
    h = np.fft.irfft(F * B, N)
    h = h[: 2 * n - 1]
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    hmx = float(np.max(h)) if h.size else INF
    if not np.isfinite(hmx):
        return False

    beta = state["beta_c"]
    top_mix = 0.0
    if beta > 10.0:
        top_mix = min(0.5, (beta - 10.0) / 80.0)
    q = make_q(h, beta, top_mix)

    g = corr_from_q(q, F, N, phase, n)
    g = np.where(np.isfinite(g), g, 0.0)

    raw = g - float(np.mean(g))
    rmax = float(np.max(np.abs(raw)))
    if not np.isfinite(rmax) or rmax <= 1e-300:
        return False
    raw = raw / rmax

    lw, mnew, vnew, tnew = adam_update(
        wb, raw, state["mb"], state["vb"], state["tb"], state["lr_b"]
    )

    for scale in (1.0, 0.5):
        tw = wb + scale * (lw - wb)
        trial_b = a_from_w(tw)
        B2 = np.fft.rfft(trial_b, N)
        h2 = np.fft.irfft(F * B2, N)
        h2 = h2[: 2 * n - 1]
        h2 = np.where(np.isfinite(h2), h2, 0.0)
        h2 = np.maximum(h2, 0.0)
        mx2 = float(np.max(h2)) if h2.size else INF
        if np.isfinite(mx2) and mx2 < hmx - 1e-13:
            state["b"] = trial_b
            state["wb"] = tw
            state["mb"] = mnew
            state["vb"] = vnew
            state["tb"] = tnew
            state["lr_b"] = min(2.0, state["lr_b"] * 1.05)
            state["cross_stall"] = 0
            return True

    state["lr_b"] = max(0.01, state["lr_b"] * 0.75)
    state["cross_stall"] += 1
    return False


def cross_mix(state):
    """
    Try convex combinations of current a and the current best-response b.
    Accept only if the self-convolution score improves.
    """
    a = state["a"]
    b = state["b"]
    N = state["N"]
    n = state["n"]

    if b.size != n or not np.all(np.isfinite(b)):
        state["b"] = a.copy()
        state["wb"] = state["w"].copy()
        state["mb"] = np.zeros(n, dtype=np.float64)
        state["vb"] = np.zeros(n, dtype=np.float64)
        state["tb"] = 0
        return False

    best_score = state["score"]
    best_a = None
    best_w = None

    for t in (1.0, 0.8, 0.6, 0.4, 0.2, 0.1):
        cand = (1.0 - t) * a + t * b
        s = float(np.sum(cand))
        if not np.isfinite(s) or s <= 0.0:
            continue
        cand = cand / s
        sc = fft_score_only(cand, N)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = cand
            best_w = log_from_a(cand)

    if best_a is not None:
        apply_a(state, best_a, best_w, reset_moments=True)
        state["stall"] = 0
        return True

    state["stall"] += 1
    state["cross_stall"] += 1
    return False


def tick_state(state, elapsed_frac):
    """Run a small batch of local search operations on one state."""
    state["elapsed_frac"] = float(elapsed_frac)
    state["tick"] += 1
    t = state["tick"]

    if t % 2 == 1:
        self_step(state, "euclid", elapsed_frac)

    if t % 5 == 0:
        inner = 3
        if state["stall"] > 10:
            inner = 5
        if state["cross_stall"] > 10:
            inner = 2
        for _ in range(inner):
            cross_step_b(state)

    if t % 10 == 0:
        cross_mix(state)

    if state["stall"] > 6 and t % 3 == 0:
        self_step(state, "hard", elapsed_frac)

    state["beta_s"] = min(
        250.0,
        state["beta_s"] + 0.03 + 0.01 * min(1.0, float(state["tick"]) / 1000.0),
    )
    state["beta_c"] = min(150.0, state["beta_c"] + 0.08)


def state_score_key(state):
    return state["score"]


def top_score_key(item):
    return item[0]


def add_top(top, a, score, max_items=32):
    """Maintain a small list of best FFT-score candidates."""
    if not np.isfinite(score):
        return top
    a = np.asarray(a, dtype=np.float64).copy()
    if len(top) < max_items:
        top.append((float(score), a))
        return top

    worst_idx = 0
    worst_score = float(top[0][0])
    for i in range(1, len(top)):
        if float(top[i][0]) > worst_score:
            worst_score = float(top[i][0])
            worst_idx = i
    if float(score) < worst_score:
        top[worst_idx] = (float(score), a)
    return top


def exact_score_array(a):
    """Exact objective using np.convolve. Input may be normalized or not."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return INF
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return INF
    b = np.convolve(a, a)
    if b.size == 0:
        return INF
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return INF
    val = float(2.0 * n * mx / (s * s))
    return val if np.isfinite(val) else INF


def reweight_states(states, global_best_ff):
    """Credit weights: favor near-best states, slightly favor shorter FFTs."""
    m = len(states)
    if m == 0:
        return np.zeros(0, dtype=np.float64)
    raw = []
    total = 0.0
    for st in states:
        s = st["score"]
        if np.isfinite(global_best_ff) and np.isfinite(s):
            diff = max(0.0, s - global_best_ff)
        else:
            diff = 1.0
        w = (0.2 + 0.8 * float(np.exp(-diff / 0.05))) * (1.0 / float(np.sqrt(st["N"])))
        if not np.isfinite(w) or w <= 0.0:
            w = 0.05
        raw.append(float(w))
        total += float(w)
    if not np.isfinite(total) or total <= 0.0:
        arr = np.ones(m, dtype=np.float64) / float(m)
    else:
        arr = np.asarray(raw, dtype=np.float64) / total
    return arr


def load_initial_sequence():
    """Load the SOTA parent sequence if available; otherwise return uniform."""
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return np.ones(4096, dtype=np.float64) / 4096.0
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return np.ones(4096, dtype=np.float64) / 4096.0

    if arr.size < 16:
        return np.ones(4096, dtype=np.float64) / 4096.0

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(4096, dtype=np.float64) / 4096.0

    arr = arr / s
    if arr.size > MAX_N:
        arr = resample_prob(arr, 8192)
    return arr


def build_initial_candidates(parent, seed):
    """Create several initial sequences for the multi-state search."""
    rng = np.random.default_rng(seed)
    p = normalize_nonneg(parent)
    if p.size < 16:
        p = np.ones(4096, dtype=np.float64) / 4096.0
    if p.size > MAX_N:
        p = resample_prob(p, 8192)

    arrays = [p.copy()]

    # Resampled main lengths.
    if 1000 <= p.size < 5461:
        arrays.append(resample_prob(p, 5461))
    if 1000 <= p.size < 8192:
        arrays.append(resample_prob(p, 8192))
    elif p.size > 8192:
        arrays.append(resample_prob(p, 8192))

    # Pick a main array to perturb: prefer moderate length.
    main = p
    if p.size < 5461 and len(arrays) > 1:
        main = arrays[-2]
    elif p.size >= 5461 and len(arrays) > 1:
        main = arrays[-1]

    # Perturbed parent.
    w = log_from_a(main)
    sig = float(rng.choice([0.35, 0.6, 1.0]))
    w = w + rng.normal(0.0, sig, size=main.size)
    arrays.append(a_from_w(w))

    # Random smooth profile at a moderate length.
    n_rand = 5461
    if p.size >= 8192:
        n_rand = 8192
    elif p.size < 2000:
        n_rand = 4096

    t = np.linspace(0.0, 1.0, n_rand, endpoint=False, dtype=np.float64)
    ph1 = float(rng.normal())
    ph2 = float(rng.normal())
    ph3 = float(rng.normal())
    amp1 = float(rng.choice([0.25, 0.4, 0.6]))
    amp2 = float(amp1 * rng.choice([0.5, 0.8, 1.2]))
    wRnd = rng.normal(0.0, 0.18, size=n_rand)
    wRnd += amp1 * np.sin(2.0 * np.pi * t + ph1)
    wRnd += amp2 * np.sin(4.0 * np.pi * t + ph2)
    wRnd += 0.5 * amp2 * np.sin(8.0 * np.pi * t + ph3)
    arrays.append(a_from_w(wRnd))

    return arrays


def run_search_multi(candidates, budget_s, seed):
    """
    Run the main multi-state local search.
    Returns (best_array, best_score) where best_score is exact if validated,
    otherwise FFT-based.
    """
    start = time.monotonic()
    budget = max(0.0, float(budget_s))
    deadline = start + max(0.0, budget - 12.0)

    states = []
    for a_init in candidates:
        try:
            st = make_state(a_init, seed=seed + len(states))
            if np.isfinite(st["score"]):
                states.append(st)
        except Exception:
            pass

    if not states:
        st = make_state(np.ones(4096, dtype=np.float64) / 4096.0, seed=seed)
        states.append(st)

    states.sort(key=state_score_key)
    global_best_ff = float(states[0]["score"])
    global_best_a = states[0]["a"].copy()

    top = []
    for st in states:
        top = add_top(top, st["a"], st["score"])

    best_valid_score = INF
    best_valid_a = global_best_a.copy()
    exact_time = 0.0
    last_exact = start
    exact_limit = max(20.0, 0.06 * max(60.0, budget))  # reserve a modest amount for exact checks

    if global_best_a.size <= 10000 and budget > 30.0:
        t0 = time.monotonic()
        val = exact_score_array(global_best_a)
        dt = time.monotonic() - t0
        exact_time += dt
        last_exact = t0 + dt
        if np.isfinite(val):
            best_valid_score = float(val)
            best_valid_a = global_best_a.copy()

    S = len(states)
    weights = reweight_states(states, global_best_ff)
    credit = np.zeros(S, dtype=np.float64)
    rng = np.random.default_rng(seed + 12345)
    loop_count = 0

    while time.monotonic() < deadline:
        loop_count += 1

        credit += weights
        i_choose = -1
        best_credit = -1.0
        for k in range(S):
            if credit[k] >= 1.0 and credit[k] > best_credit:
                best_credit = credit[k]
                i_choose = k

        if i_choose < 0:
            # No state accrued a full credit yet; continue.
            if loop_count % 500 == 0:
                weights = reweight_states(states, global_best_ff)
                credit = np.zeros(S, dtype=np.float64)
            continue

        credit[i_choose] -= 1.0
        st = states[i_choose]

        try:
            tick_state(st, (time.monotonic() - start) / max(1.0, budget))
        except Exception:
            # If a state corrupts, reset it from the current global best.
            try:
                base = resample_prob(global_best_a, st["n"])
                w = log_from_a(base)
                w = w + rng.normal(0.0, 0.5, size=st["n"])
                new_st = make_state(a_from_w(w), seed=seed + 777 + i_choose)
                states[i_choose] = new_st
                weights = reweight_states(states, global_best_ff)
                credit = np.zeros(S, dtype=np.float64)
            except Exception:
                pass
            continue

        # Update the global FFT-best candidate.
        if np.isfinite(st["score"]) and st["score"] < global_best_ff - 1e-12:
            global_best_ff = float(st["score"])
            global_best_a = st["a"].copy()
            top = add_top(top, st["a"], st["score"])

            if (
                global_best_a.size <= 10000
                and exact_time < exact_limit
                and time.monotonic() - last_exact > 25.0
            ):
                t0 = time.monotonic()
                val = exact_score_array(global_best_a)
                dt = time.monotonic() - t0
                exact_time += dt
                last_exact = time.monotonic()
                if np.isfinite(val) and val < best_valid_score:
                    best_valid_score = float(val)
                    best_valid_a = global_best_a.copy()
                print(
                    "fft_best = %.9f  exact_best = %.9f  n=%d  elapsed=%.1f"
                    % (
                        global_best_ff,
                        best_valid_score if np.isfinite(best_valid_score) else INF,
                        global_best_a.size,
                        time.monotonic() - start,
                    )
                )
                flush = globals().get("flush", None)
                if callable(flush):
                    try:
                        flush()
                    except Exception:
                        pass

        # Replace states that have fallen far behind and stagnated.
        if (
            S > 1
            and np.isfinite(global_best_ff)
            and st["stall"] > 130
            and st["score"] > global_best_ff + 0.08
            and (deadline - time.monotonic()) > 150.0
        ):
            try:
                base = resample_prob(global_best_a, st["n"])
                w = log_from_a(base)
                sig = float(rng.choice([0.15, 0.35, 0.7, 1.1]))
                w = w + rng.normal(0.0, sig, size=st["n"])
                new_a = a_from_w(w)
                new_st = make_state(new_a, seed=int(seed + 100000 + rng.integers(0, 1000000)))
                if (
                    np.isfinite(new_st["score"])
                    and new_st["score"] < st["score"] + 0.75
                ):
                    states[i_choose] = new_st
                    credit = np.zeros(S, dtype=np.float64)
                    weights = reweight_states(states, global_best_ff)
                    if new_st["score"] < global_best_ff:
                        global_best_ff = float(new_st["score"])
                        global_best_a = new_st["a"].copy()
                        top = add_top(top, new_st["a"], new_st["score"])
            except Exception:
                pass

        if loop_count % 250 == 0:
            weights = reweight_states(states, global_best_ff)
            credit = np.zeros(S, dtype=np.float64)

        if loop_count % 2000 == 0:
            print(
                "loop=%d  best_fft=%.9f  best_valid=%.9f  elapsed=%.1f"
                % (
                    loop_count,
                    global_best_ff,
                    best_valid_score if np.isfinite(best_valid_score) else INF,
                    time.monotonic() - start,
                )
            )

    # Final exact validation of top FFT candidates.
    if np.isfinite(global_best_ff):
        if global_best_a.size <= 10000 and exact_time < exact_limit:
            t0 = time.monotonic()
            val = exact_score_array(global_best_a)
            exact_time += time.monotonic() - t0
            if np.isfinite(val) and val < best_valid_score:
                best_valid_score = float(val)
                best_valid_a = global_best_a.copy()

    top.sort(key=top_score_key)
    for sc, a in top:
        if time.monotonic() > deadline - 1.0:
            break
        if exact_time >= exact_limit:
            break
        if a is None or a.size == 0 or a.size > 10000:
            continue
        if np.isfinite(best_valid_score) and float(sc) > best_valid_score + 0.015:
            break
        t0 = time.monotonic()
        val = exact_score_array(a)
        dt = time.monotonic() - t0
        exact_time += dt
        if np.isfinite(val) and val < best_valid_score:
            best_valid_score = float(val)
            best_valid_a = a.copy()

    if np.isfinite(best_valid_score):
        return best_valid_a, best_valid_score
    return global_best_a, global_best_ff


def smooth_obj_grad(w, n, N, phase, beta):
    """
    Smoothed objective and gradient for L-BFGS polish.

    Returns (F, grad_w) where F is the scaled smoothed max self-convolution.
    """
    a = a_from_w(w)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[: 2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)

    L = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(L) or L <= 0.0:
        return 0.0, np.zeros(n, dtype=np.float64)

    x = (c - L) / L
    x = np.where(np.isfinite(x), x, -1e300)
    x = np.clip(x, -1000.0, 0.0)
    y = np.clip(float(beta) * x, -745.0, 0.0)
    e = np.exp(y)
    s = float(e.sum())

    if not np.isfinite(s) or s <= 0.0:
        q = np.zeros(c.size, dtype=np.float64)
        q[int(np.argmax(c))] = 1.0
        val = L
    else:
        q = e / s
        val = L + (L / float(beta)) * float(np.log(s))

    g = 2.0 * corr_from_q(q, A, N, phase, n)
    lam = float(np.dot(a, g))
    grad_w = a * (g - lam)
    grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)

    scale = float(2.0 * n)
    return scale * float(val), scale * grad_w


def try_lbfgs_polish(a0, budget_s, seed):
    """
    Optional final L-BFGS polish on the smoothed objective.
    Returns a polished probability vector if it lowers the exact FFT max score,
    otherwise None.
    """
    if budget_s < 25.0:
        return None
    try:
        from scipy.optimize import minimize
    except Exception:
        return None

    a = normalize_nonneg(a0)
    n = a.size
    if n < 16 or n > MAX_N:
        return None

    N = next_fast_len(3 * n - 2)
    ph = np.arange(N // 2 + 1, dtype=np.float64)
    phase = np.exp(-2j * np.pi * ph * (n - 1) / float(N))
    w = log_from_a(a)
    init_score = fft_score_only(a, N)
    if not np.isfinite(init_score):
        return None

    beta = 120.0
    if n > 6000:
        maxiter = 45
    else:
        maxiter = 80
    if budget_s < 80.0:
        maxiter = min(maxiter, 40)

    try:
        res = minimize(
            smooth_obj_grad,
            w,
            args=(n, N, phase, beta),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-12,
                "gtol": 1e-7,
                "maxls": 25,
                "disp": False,
            },
        )
    except Exception:
        return None

    if res is None:
        return None

    try:
        a2 = a_from_w(np.asarray(res.x, dtype=np.float64).ravel())
    except Exception:
        return None

    sc2 = fft_score_only(a2, N)
    if np.isfinite(sc2) and sc2 < init_score - 1e-9:
        return a2
    return None


def finalize_sequence(a):
    """Scale and return a valid Python list of floats."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(64, dtype=np.float64) / 64.0
        s = 1.0

    n = a.size
    scale = float(np.sqrt(2.0 * float(n)))
    a = (a * scale) / s
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if not np.isfinite(s2) or s2 < 0.01:
        a = a * (0.01 / max(s2, EPS))
        a = np.clip(a, 0.0, 1000.0)

    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point required by the harness.
    Runs the multi-state optimizer and returns the best sequence found.
    """
    try:
        budget = max(5.0, min(float(budget_s), 2400.0))
    except Exception:
        budget = 2400.0

    global_start = time.monotonic()
    parent = load_initial_sequence()
    candidates = build_initial_candidates(parent, seed=seed)

    main_budget = max(120.0, budget - 75.0)
    best_a, best_score = run_search_multi(candidates, main_budget, seed=seed)
    elapsed = time.monotonic() - global_start

    # Final polish, if time remains.
    remaining = max(0.0, budget - 15.0 - elapsed)
    if remaining > 40.0:
        try:
            polished = try_lbfgs_polish(best_a, remaining - 12.0, seed=seed + 999983)
            if polished is not None:
                n = polished.size
                if n <= 10000:
                    sc_base = best_score if np.isfinite(best_score) else INF
                    if not np.isfinite(sc_base):
                        sc_base = exact_score_array(best_a)
                    sc_pol = exact_score_array(polished)
                else:
                    N = next_fast_len(3 * n - 2)
                    sc_base = best_score
                    sc_pol = fft_score_only(polished, N)
                if np.isfinite(sc_pol) and sc_pol < sc_base - 1e-10:
                    best_a = polished
                    best_score = float(sc_pol)
        except Exception:
            pass

    # Last-resort validation.
    if not np.isfinite(best_score):
        try:
            if best_a.size <= 10000:
                best_score = exact_score_array(best_a)
            else:
                N = next_fast_len(3 * best_a.size - 2)
                best_score = fft_score_only(best_a, N)
        except Exception:
            best_score = INF

    if not np.isfinite(best_score):
        best_a = parent if parent is not None else np.ones(4096, dtype=np.float64) / 4096.0

    return finalize_sequence(best_a)
```
