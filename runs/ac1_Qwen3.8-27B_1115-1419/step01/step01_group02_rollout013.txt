    a = np.asarray(a, dtype=np.float64).reshape(-1)

                if p.size < 200:
                    p = np.ones(5582, dtype=np.float64)
                return p, p.size
    except Exception:
        pass
    return np.ones(default_n, dtype=np.float64), default_n


def _actual_evaluate(seq):
    try:
        arr = np.asarray(seq, dtype=np.float64).reshape(-1)
        out = [float(x) for x in arr]
        val = evaluate_sequence(out)
        return float(val)
    except Exception:
        return float("inf")


def _top_k_indices(values, K):
    values = np.asarray(values, dtype=np.float64).reshape(-1)
    m = len(values)
    K = int(K)
    if K <= 0:
        return np.array([], dtype=np.int64)
    if K >= m:
        return np.arange(m, dtype=np.int64)
    if K == 1:
        return np.array([int(np.argmax(values))], dtype=np.int64)
    return np.argpartition(values, -K)[-K:].astype(np.int64)


def _peak_weights(vals, max_val, beta_rel):
    vals = np.asarray(vals, dtype=np.float64).reshape(-1)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    m = vals.size
    if m == 0:
        return np.array([], dtype=np.float64)
    max_val = float(max_val)
    if not np.isfinite(max_val) or max_val <= 1e-300:
        return np.full(m, 1.0 / m, dtype=np.float64)
    scale = max_val
    z = (max_val - vals) / scale
    z = np.where(np.isfinite(z), z, 50.0)
    z = np.clip(z, 0.0, 50.0 / max(1.0, float(beta_rel)))
    w = np.exp(-float(beta_rel) * z)
    w = np.where(np.isfinite(w), w, 0.0)
    w += 1e-300
    s = float(np.sum(w))
    if s <= 1e-300:
        return np.full(m, 1.0 / m, dtype=np.float64)
    return w / s


def _gradient_shift(f, idx, w, factor=2.0):
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    n = len(f)
    g = np.zeros(n, dtype=np.float64)
    for k, wi in zip(np.asarray(idx).astype(int), w):
        if float(wi) <= 0.0:
            continue
        k = int(k)
        if k < 0:
            k = 0
        if k > 2 * n - 2:
            k = 2 * n - 2

        if k < n:
            lo = 0
            hi = k
        else:
            lo = k - n + 1
            hi = n - 1

        if hi < lo:
            continue

        start = k - lo
        stop = k - hi - 1
        if stop < 0:
            stop = None

        seg = f[start:stop:-1]
        if seg.size:
            g[lo:hi + 1] += float(factor) * float(wi) * seg
    return g


def _run_normgrad(start, deadline, rng, rel_init, beta, topK, topK_max, beta_rel):
    start = np.asarray(start, dtype=np.float64).reshape(-1)
    n = len(start)
    if n <= 0:
        return np.array([], dtype=np.float64), float("inf")
    target_sum = float(np.sqrt(2.0 * n))
    a = _normalize_target(start, target_sum)

    best = a.copy()
    best_score, b = _fast_info(a)
    if not np.isfinite(best_score):
        a = np.full(n, target_sum / n, dtype=np.float64)
        best = a.copy()
        best_score, b = _fast_info(a)

    mom = np.zeros(n, dtype=np.float64)
    rel = float(rel_init)
    no_improve = 0
    no_improve_limit = 5000
    avg_b = (2.0 * n) / max(1.0, (2.0 * n - 1.0))

    while time.monotonic() < deadline:
        max_b, b = _fast_info(a)
        if not np.isfinite(max_b):
            break

        if max_b < best_score - 1e-12:
            best_score = max_b
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > no_improve_limit:
            noise_scale = 0.035 * max(1e-12, float(np.linalg.norm(best))) / np.sqrt(max(1, n))
            a = best + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0.0)
            a = _normalize_target(a, target_sum)
            mom = np.zeros(n, dtype=np.float64)
            rel = min(0.25, max(0.02, rel_init * 1.4))
            no_improve = 0

        K = int(min(topK, len(b)))
        diff = max_b - avg_b
        if diff > 1e-14:
            thresh = max_b - 0.18 * diff
            cnt = int(np.sum(b >= thresh))
            if cnt > K:
                K = min(cnt + 8, int(topK_max), len(b))

        idx = _top_k_indices(b, K)
        vals = b[idx]
        w = _peak_weights(vals, max_b, beta_rel)

        g = _gradient_shift(a, idx, w)
        g -= np.mean(g)
        gn = float(np.linalg.norm(g))

        if gn < 1e-14:
            noise_scale = 0.025 * max(1e-12, float(np.linalg.norm(best))) / np.sqrt(max(1, n))
            a = best + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0.0)
            a = _normalize_target(a, target_sum)
            mom = np.zeros(n, dtype=np.float64)
            rel = min(0.25, max(0.02, rel_init * 1.3))
            continue

        d = g / gn
        mom = float(beta) * mom + (1.0 - float(beta)) * d
        mn = float(np.linalg.norm(mom))

        if mn < 1e-14:
            step = np.zeros(n, dtype=np.float64)
        else:
            an = float(np.linalg.norm(a))
            step = (rel * an / mn) * mom
            mean_a = float(np.mean(a))
            comp_limit = max(25.0 * mean_a, 1e-12)
            step = np.clip(step, -comp_limit, comp_limit)

        cand = a - step
        cand = np.maximum(cand, 0.0)
        s = float(np.sum(cand))
        if (not np.isfinite(s)) or s <= target_sum * 1e-7:
            a = best.copy()
            mom = np.zeros(n, dtype=np.float64)
        else:
            a = cand * (target_sum / s)
            if not np.all(np.isfinite(a)):
                a = best.copy()
                mom = np.zeros(n, dtype=np.float64)

        rel = max(0.0035, rel * 0.99985)

    return best, best_score


def _select_active_set(b, cg, prev, maxC, n):
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    L = len(b)
    maxC = int(maxC)
    if maxC <= 0 or L == 0:
        return np.array([], dtype=np.int64)
    maxC = min(maxC, L)

    top_b = _top_k_indices(b, maxC)
    prev = np.asarray(prev, dtype=np.int64).reshape(-1)
    prev = prev[(prev >= 0) & (prev < L)]

    parts = [top_b, prev]
    if cg is not None:
        cg = np.asarray(cg, dtype=np.float64).reshape(-1)
        if cg.size == L:
            top_cg = _top_k_indices(cg, maxC)
            parts.append(top_cg)

    guard = []
    e = min(8, n)
    guard.extend(range(0, e))
    guard.extend(range(max(0, L - e), L))
    for kk in (n - 1, n, n + 1):
        if 0 <= kk < L:
            guard.append(int(kk))
    if guard:
        parts.append(np.array(guard, dtype=np.int64))

    cand = np.concatenate(parts)
    cand = np.unique(cand)
    cand = cand[(cand >= 0) & (cand < L)].astype(np.int64)

    if cand.size <= maxC:
        return cand

    b_scale = max(float(np.max(b)), 1e-300)
    score = b[cand] / b_scale
    if cg is not None and np.size(cg) == L:
        c_scale = max(float(np.max(cg)), 1e-300)
        score = 0.55 * score + 0.45 * (cg[cand] / c_scale)

    keep = _top_k_indices(score, maxC)
    return cand[keep]


def _convex_update(f, g, t, target_sum):
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    g = np.asarray(g, dtype=np.float64).reshape(-1)
    sg = float(np.sum(g))
    if sg <= 1e-300:
        return f.copy()
    g = g * (float(target_sum) / sg)
    cand = (1.0 - float(t)) * f + float(t) * g
    cand = np.maximum(cand, 0.0)
    return _normalize_target(cand, target_sum)


def _solve_lp(f, active, M, cap, time_limit=4.0):
    if not _LP_AVAILABLE:
        return None
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    n = len(f)
    active = np.asarray(active, dtype=np.int64).reshape(-1)
    active = active[(active >= 0) & (active < (2 * n - 1))]
    if active.size == 0:
        return None

    m = active.size
    row_parts = []
    col_parts = []
    val_parts = []

    for r, k in enumerate(active):
        k = int(k)
        if k < n:
            lo = 0
            hi = k
        else:
            lo = k - n + 1
            hi = n - 1
        if hi < lo:
            continue
        start = k - lo
        stop = k - hi - 1
        if stop < 0:
            stop = None
        vals = np.asarray(f[start:stop:-1], dtype=np.float64).copy()
        l = int(vals.size)
        row_parts.append(np.full(l, r, dtype=np.int32))
        col_parts.append(np.arange(lo, hi + 1, dtype=np.int32))
        val_parts.append(vals)

    if not val_parts:
        return None

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    data = np.concatenate(val_parts)

    try:
        A = csr_matrix((data, (rows, cols)), shape=(m, n), dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)
        bubs = np.full(m, float(M), dtype=np.float64)
        res = linprog(
            c,
            A_ub=A,
            b_ub=bubs,
            bounds=(0.0, float(cap)),
            method="highs",
            options={"presolve": True, "time_limit": max(1.0, float(time_limit))},
        )
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=bubs, bounds=(0.0, float(cap)))
        except Exception:
            return None

    if res is None:
        return None
    if not getattr(res, "success", False):
        return None
    g = np.asarray(res.x, dtype=np.float64).reshape(-1)
    if g.size != n:
        return None
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    if np.sum(g) <= 1e-300:
        return None
    return g


def _run_lp_search(start, deadline, rng, maxC=380, initialK=150, time_per_lp=3.5, stall_limit=7):
    start = np.asarray(start, dtype=np.float64).reshape(-1)
    n = len(start)
    if n <= 0:
        return np.array([], dtype=np.float64), float("inf")
    target_sum = float(np.sqrt(2.0 * n))
    a = _normalize_target(start, target_sum)
    best = a.copy()
    best_score, b = _fast_info(a)
    if not np.isfinite(best_score):
        return best, float("inf")

    active = _select_active_set(b, None, np.array([], dtype=np.int64), min(initialK, maxC), n)
    last_cg = None
    stalls = 0
    cap = 120.0 * target_sum / max(1, n)

    while time.monotonic() < deadline:
        max_b, b = _fast_info(a)
        if not np.isfinite(max_b):
            break

        if max_b < best_score - 1e-12:
            best_score = max_b
            best = a.copy()
            stalls = 0

        active = _select_active_set(b, last_cg, active, maxC, n)
        if active.size == 0:
            break

        Meff = max_b * (1.0 + 1e-8)
        g = None
        cg = None

        for _ in range(3):
            g = _solve_lp(a, active, Meff, cap, time_per_lp)
            if g is None:
                break
            if float(np.sum(g)) <= 1e-300:
                break
            cg = _fft_conv2(a, g)
            cg_max = float(np.max(cg)) if cg.size else float("inf")
            if not np.isfinite(cg_max):
                break
            if cg_max <= Meff * (1.0 + 1e-6) + 1e-10:
                break
            active = _select_active_set(b, cg, active, maxC, n)

        if g is None or cg is None or float(np.sum(g)) <= 1e-300:
            stalls += 1
            if stalls > stall_limit:
                break
            continue

        improved = False
        for t in (0.45, 0.30, 0.22, 0.15, 0.10, 0.07, 0.04):
            cand = _convex_update(a, g, t, target_sum)
            sc, _ = _fast_info(cand)
            if np.isfinite(sc) and sc < max_b - 1e-12:
                a = cand
                improved = True
                break

        if not improved:
            cand = _convex_update(a, g, 0.03, target_sum)
            noise = 0.015 * max(1e-12, float(np.linalg.norm(a))) * rng.standard_normal(n) / np.sqrt(max(1, n))
            cand = np.maximum(cand + noise, 0.0)
            cand = _normalize_target(cand, target_sum)
            sc, _ = _fast_info(cand)
            if np.isfinite(sc) and sc < max_b * 1.01 + 1e-10:
                a = cand
            else:
                a = best.copy()
            stalls += 1
            if stalls > stall_limit:
                noise = 0.045 * max(1e-12, float(np.linalg.norm(best))) * rng.standard_normal(n) / np.sqrt(max(1, n))
                a = np.maximum(best + noise, 0.0)
                a = _normalize_target(a, target_sum)
                stalls = 0
        else:
            stalls = 0

        last_cg = cg
        if not np.all(np.isfinite(a)):
            a = best.copy()

    return best, best_score


def propose_candidate(seed=1, budget_s=2400):
    try:
        seed = int(seed)
    except Exception:
        seed = 1
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 18.0)

    parent, n = _get_parent()
    if n < 250:
        parent = np.ones(5582, dtype=np.float64)
        n = parent.size

    target_sum = float(np.sqrt(2.0 * n))
    best = _normalize_target(parent, target_sum)
    best_score, _ = _fast_info(best)
    if not np.isfinite(best_score):
        best = np.full(n, target_sum / n, dtype=np.float64)
        best_score, _ = _fast_info(best)

    actual_best = _actual_evaluate(best)
    actual_seq = best.copy()

    print(f"Initial n={n}, fast score={best_score:.6f}, actual score={actual_best:.6f}")

    if _LP_AVAILABLE and time.monotonic() < deadline - 500:
        lp_deadline = min(deadline - 500, time.monotonic() + 330.0)
        print("Phase 1: LP-guided direction search")
        lp_best, lp_score = _run_lp_search(best, lp_deadline, rng, maxC=380, initialK=150, time_per_lp=3.5)
        if np.isfinite(lp_score) and lp_score < best_score - 1e-12:
            best = lp_best
            best_score = lp_score
            act = _actual_evaluate(best)
            if np.isfinite(act) and act < actual_best:
                actual_best = act
                actual_seq = best.copy()
            print(f"  LP improved to fast={best_score:.6f}, actual={act:.6f}")
        else:
            print("  LP did not improve; continuing.")

    configs = (
        # alloc_seconds, rel_init, beta, topK, topK_max, beta_rel
        (330.0, 0.055, 0.85, 128, 320, 32.0),
        (260.0, 0.025, 0.92, 256, 512, 60.0),
        (180.0, 0.100, 0.74, 48, 128, 12.0),
        (230.0, 0.035, 0.90, 192, 448, 45.0),
    )

    for idx, (alloc, rel_init, beta, topK, topK_max, beta_rel) in enumerate(configs):
        now = time.monotonic()
        if now >= deadline - 25:
            break
        remaining = deadline - now
        t = min(max(30.0, alloc), max(30.0, remaining * 0.38))
        if remaining - t < 25:
            t = max(20.0, remaining - 25)
        if t <= 0:
            break
        d = now + t
        print(f"Phase 2.{idx+1}: normgrad rel={rel_init}, beta={beta}, K={topK}/{topK_max}, beta_rel={beta_rel}, t={t:.0f}s")
        cand, sc = _run_normgrad(best, d, rng, rel_init, beta, topK, topK_max, beta_rel)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = cand
            best_score = sc
            act = _actual_evaluate(best)
            if np.isfinite(act) and act < actual_best:
                actual_best = act
                actual_seq = best.copy()
            print(f"  improved to fast={best_score:.6f}, actual={act:.6f}")
        else:
            print(f"  no improvement, best fast remains {best_score:.6f}")

    if time.monotonic() < deadline - 12:
        print("Phase 3: final polish")
        cand, sc = _run_normgrad(best, deadline - 6, rng, 0.010, 0.95, 256, 256, 80.0)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = cand
            best_score = sc
            act = _actual_evaluate(best)
            if np.isfinite(act) and act < actual_best:
                actual_best = act
                actual_seq = best.copy()
            print(f"  final improved to fast={best_score:.6f}, actual={act:.6f}")

    final_act_best = _actual_evaluate(best)
    final_act_saved = _actual_evaluate(actual_seq)

    if np.isfinite(final_act_best) and (not np.isfinite(final_act_saved) or final_act_best < final_act_saved):
        out = best
    else:
        out = actual_seq

    out = np.asarray(out, dtype=np.float64).reshape(-1)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s <= 0.0:
        out = np.full(n, target_sum / n, dtype=np.float64)
    else:
        out = out * (target_sum / s)

    final = _actual_evaluate(out)
    if not np.isfinite(final):
        out = actual_seq
        s = float(np.sum(out))
        if s > 0:
            out = out * (target_sum / s)

    print(f"Returned fast target-scale score={best_score:.6f}, actual score={_actual_evaluate(out):.6f}, n={len(out)}")
    return [float(x) for x in out]
