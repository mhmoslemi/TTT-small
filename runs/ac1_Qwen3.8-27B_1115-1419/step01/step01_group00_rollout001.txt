"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based convolution, Adam optimizer with smooth-to-hard transition,
structured perturbations (mirror, swap, shift), and simulated annealing.
Explores multiple n values and initializations for robustness.
"""

import numpy as np
import time


def _conv_fft(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize <<= 1
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]


def _soft_weights(c, tau):
    """Stable softmax over convolution values."""
    mx = np.max(c)
    e = np.exp((c - mx) / tau)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return e / s


def _smoothed_grad(a, w, n):
    """Smoothed gradient = 2 * correlate(weights, a, valid)."""
    return 2.0 * np.correlate(w, a, mode='valid')[:n]


def _project(a, S):
    """Project to {x>=0, sum(x)=S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def _obj_val(a, n, S):
    """Objective given fixed sum S."""
    c = _conv_fft(a)
    return 2.0 * n * float(c.max()) / (S * S)


def _adam_smooth(a, n, S, deadline, tag, it_max=3000000):
    """Adam optimizer with entropy smoothing, transitioning to hard."""
    tau = n * 0.3
    tau_min = 1.0
    lr = 0.5 / n
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best = a.copy()
    best_o = _obj_val(a, n, S)
    stall = 0

    for it in range(it_max):
        if time.monotonic() >= deadline:
            break
        t += 1
        c = _conv_fft(a)
        mc = float(c.max())
        o = 2.0 * n * mc / (S * S)
        if o < best_o - 1e-12:
            best_o = o
            best = a.copy()
            stall = 0
            if it % 20000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} tau={tau:.1f} lr={lr:.2e}")
        else:
            stall += 1

        # Smooth weights (transition to hard)
        if tau > tau_min * 5:
            p = _soft_weights(c, tau)
        else:
            # Hard: top-3 peaks
            p = np.zeros(len(c))
            topk = np.argsort(c)[-3:]
            p[topk] = 1.0 / len(topk)

        g = _smoothed_grad(a, p, n)
        g -= g.mean()  # Maintain sum constraint

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project(a, S)

        # Schedule
        tau = max(tau * 0.9999, tau_min)
        if stall > 10000:
            lr = max(lr * 0.5, 1e-6 / n)
            stall = 0

    return best, best_o


def _hard_refine(a, n, S, deadline, tag, it_max=2000000):
    """Hard active-set with top-k peaks and momentum."""
    eta = 0.03
    mom = 0.85
    vel = np.zeros(n)
    best = a.copy()
    best_o = _obj_val(a, n, S)
    k_top = 5

    for it in range(it_max):
        if time.monotonic() >= deadline:
            break
        c = _conv_fft(a)
        mc = float(c.max())
        o = 2.0 * n * mc / (S * S)
        if o < best_o - 1e-12:
            best_o = o
            best = a.copy()
            if it % 10000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f}")

        # Top-k weighting
        p = np.zeros(len(c))
        topk = np.argsort(c)[-k_top:]
        # Exponential weighting on rank
        ranks = np.array([c[i] for i in topk])
        wts = np.exp(np.linspace(0, 1, len(topk)))
        wts /= wts.sum()
        for i, idx in enumerate(topk):
            p[idx] = wts[i]

        g = _smoothed_grad(a, p, n)
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        vel = mom * vel - (eta * g) / mg
        a = _project(a + vel, S)

        if it % 30000 == 29999:
            eta = max(eta * 0.93, 1e-7)
            k_top = min(k_top + 1, 10)

    return best, best_o


def _perturb_mirror(a, rng):
    """Mirror a random block."""
    n = len(a)
    w = max(3, n // 30)
    s = int(rng.integers(0, n - w + 1))
    b = a.copy()
    b[s:s + w] = a[s:s + w][::-1]
    return b


def _perturb_swap(a, rng, n_swaps=20):
    """Swap mass between random pairs."""
    b = a.copy()
    for _ in range(n_swaps):
        i = int(rng.integers(0, len(b)))
        j = int(rng.integers(0, len(b)))
        if i != j:
            # Partial swap
            amt = min(b[i], b[j]) * float(rng.uniform(0.1, 0.7))
            b[i] -= amt
            b[j] += amt
    return b


def _perturb_gaussian_block(a, rng, sigma=0.3):
    """Gaussian multiplicative perturbation on a random block."""
    n = len(a)
    w = max(3, int(n * float(rng.choice([0.02, 0.05, 0.1, 0.2]))))
    s = int(rng.integers(0, n - w + 1))
    b = a.copy()
    b[s:s + w] *= np.exp(rng.normal(0, sigma, size=w))
    return b


def _perturb_shift(a, rng):
    """Cyclic shift a block."""
    n = len(a)
    w = max(3, n // 40)
    s = int(rng.integers(0, n - w))
    shift = int(rng.integers(1, w))
    b = a.copy()
    block = a[s:s + w]
    b[s:s + w] = np.roll(block, shift)
    return b


def _coordinate_descent(a, n, S, max_sweeps=3):
    """A few sweeps of coordinate descent on the max conv index."""
    for sweep in range(max_sweeps):
        c = _conv_fft(a)
        k = int(np.argmax(c))
        # Try moving mass away from positions that contribute to conv[k]
        # Positions i where a[k-i] is large contribute most
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idxs = np.arange(lo, hi + 1)
        contrib = a[k - idxs]  # Contribution of a[i] to conv[k]
        # Find the top contributor and try reducing it
        # Move some mass to a random other position
        top_idx = idxs[np.argmax(contrib)]
        if a[top_idx] > S / n * 2:
            target = int(np.random.randint(0, n))
            while target == top_idx:
                target = int(np.random.randint(0, n))
            move = a[top_idx] * 0.1
            a[top_idx] -= move
            a[target] += move
    return a
