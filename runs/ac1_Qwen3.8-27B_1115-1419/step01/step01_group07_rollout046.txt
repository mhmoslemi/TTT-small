1. Find the threshold `theta` such that `sum(max(a - step*g - theta, 0)) = 1`.
2. `a_new = max(a - step*g - theta, 0)`.

This can be done efficiently using sorting.

```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search for theta
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv[-1] - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

def project_simplex(v):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv[-1] - 1)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv[-1] - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    w has length 2n-1 (same as convolution output).

    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad

            grad_a = 2.0 * np.correlate(w, a_, mode='valid')
            dot = a_.dot(grad_a)
            return a_ * (grad_a - dot)
        
        try:

        a0 = np.ones(4000, dtype=np.float64)

    n = len(a0)
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    c = np.convolve(a, a)
    best_score = score_from_conv(c, n)
    print(f"Start: n={n}, score={best_score:.6f}")

    total_time = budget_s - 60
    deadline = time.monotonic() + total_time
    rng = np.random.default_rng(42)

    # Phase 1: Aggressive subgradient (50% of time)
    p1 = 0.50 * total_time
    print(f"\nPhase 1: Subgradient ({p1:.0f}s)")
    a1, s1 = run_subgradient(a, time.monotonic() + p1, rng,
                             start_step=3.0, tau_frac=0.02, n_active=100)
    if s1 < best_score:
        best_score = s1
        best_a = a1.copy()
    print(f"  After Phase 1: {best_score:.6f}")

    # Phase 2: L-BFGS polishing (20% of time)
    p2 = 0.20 * total_time
    print(f"\nPhase 2: L-BFGS ({p2:.0f}s)")
    tau_fracs = [0.1, 0.05, 0.02, 0.01, 0.005]
    a2, s2 = run_lbfgs(best_a, p2, tau_fracs)
    if s2 < best_score:
        best_score = s2
        best_a = a2.copy()
    print(f"  After Phase 2: {best_score:.6f}")

    # Phase 3: Fine subgradient with small steps (20% of time)
    p3 = 0.20 * total_time
    print(f"\nPhase 3: Fine subgradient ({p3:.0f}s)")
    a3, s3 = run_subgradient(best_a, time.monotonic() + p3, rng,
                             start_step=0.5, tau_frac=0.005, n_active=200,
                             use_momentum=True, beta=0.5)
    if s3 < best_score:
        best_score = s3
        best_a = a3.copy()
    print(f"  After Phase 3: {best_score:.6f}")

    # Phase 4: Final L-BFGS with very small tau (10% of time)
    p4 = max(30.0, deadline - time.monotonic())
    print(f"\nPhase 4: Final L-BFGS ({p4:.0f}s)")
    tau_fracs = [0.01, 0.005, 0.002, 0.001]
    a4, s4 = run_lbfgs(best_a, p4, tau_fracs)
    if s4 < best_score:
        best_score = s4
        best_a = a4.copy()

    # Verify with actual evaluator
    final = evaluate_sequence(best_a.tolist())
    if np.isfinite(final):
        best_score = final

    print(f"\nFinal score: {best_score:.6f}")
    return [float(v) for v in best_a]
