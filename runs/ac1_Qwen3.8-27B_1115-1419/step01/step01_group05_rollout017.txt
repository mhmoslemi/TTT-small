"""
Search for a low-peak nonnegative self-convolution sequence of length 256.

Main ideas:
  * Score is scale-invariant, so all internal sequences are normalized to sum 1.
  * Use squared-FFT for autocorrelation/convolution and LSE-smoothed gradient
    descent on the simplex as a cheap polisher.
  * Main refinement implements the LP/Frank-Wolfe construction:
      - given a, solve  max sum y  subject to  (a * y)_k <= 1, y >= 0;
      - use the normalized LP solution as a descent direction a -> (1-t)a + t y.
  * Periodically restart / perturb to escape local minima.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    _HAS_LINPROG = True
except Exception:
    _HAS_LINPROG = False


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def _autocorr(a, size):
    n = a.size
    r = np.fft.rfft(a, size)
    b = np.fft.irfft(r * r, size)
    b = b[: 2 * n - 1]
    return np.maximum(b, 0.0)


def _max_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return 0.0
    b = _autocorr(a, _pow2(2 * n - 1))
    m = float(np.max(b))
    if not np.isfinite(m) or m < 0.0:
        return 0.0
    return m


def _score_of(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    m = _max_conv(a)
    if not np.isfinite(m):
        return float("inf")
    return 2.0 * float(n) * m


def _make_uniform(n):
    n = max(int(n), 1)
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_power(n, alpha, reverse=False, eps=0.75):
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64) + eps
    v = np.exp(-float(alpha) * np.log(i))
    if reverse:
        v = v[::-1]
    return _normalize(v)


def _make_exp(n, alpha, reverse=False):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    v = np.exp(-float(alpha) * x)
    if reverse:
        v = v[::-1]
    return _normalize(v)


def _mix_two(x, y, w):
    x = np.asarray(x, dtype=np.float64).ravel()
    y = np.asarray(y, dtype=np.float64).ravel()
    if x.size == 0:
        x = _make_uniform(2)
    if y.size == 0 or y.size != x.size:
        y = _resample(y, x.size) if y.size else _make_uniform(x.size)
    return _normalize(float(w) * x + float(1.0 - w) * y)


def _smooth_random(n, rng, base=None, amp=0.40):
    n = max(int(n), 2)
    if base is None:
        base = _make_uniform(n)
    base = _normalize(base)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 16 + 2))
    if cutoff < len(F):
        F[cutoff:] = 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 1e-300:
        f = f / sd
    a = base * (1.0 + float(amp) * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _make_uniform(new_n)
    s = float(old.sum())
    if s <= 1e-300:
        return _make_uniform(new_n)
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, old))


def _get_given_sequence():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if arr.size == 0 or s <= 1e-300:
        return None
    return arr / s


def _perturb(a, rng, strength=0.18):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n < 2:
        return _make_uniform(2)
    h = a.copy()
    width = max(2, n // 24)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    h[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    if rng.random() < 0.35:
        h = 0.5 * (h + h[::-1])
    return _normalize(h)


def _lse_value(b, beta):
    if b.size == 0:
        return 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        return m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 1e-300:
        return m
    return m + float(np.log(s)) / beta


def _lse_and_p(b, beta):
    if b.size == 0:
        return np.ones(1, dtype=np.float64), 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        return np.full(b.size, 1.0 / b.size, dtype=np.float64), m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 1e-300:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _weighted_gradient(a, weights, fft_size, n):
    # d/d a_i of sum_k weights_k (a*a)_k
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    h = np.where(h > 0.0, h, 0.0)
    return 2.0 * h


def _project_soft(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    w = np.maximum(v, 0.0)
    s = float(w.sum())
    n = w.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return w / s


def _sym_dir(d):
    d = np.asarray(d, dtype=np.float64).ravel()
    return _normalize(0.5 * (d + d[::-1]))


def _smooth_dir(d, frac=0.12):
    d = np.asarray(d, dtype=np.float64).ravel()
    n = d.size
    u = _make_uniform(n)
    return _normalize((1.0 - float(frac)) * d + float(frac) * u)


def _line_search(a, d, current_score):
    a = np.asarray(a, dtype=np.float64).ravel()
    d = _normalize(d)
    best_a = a
    best_score = current_score

    ts = (0.10, 0.05, 0.20, 0.35, 0.50, 0.70, 1.00, 0.02, 0.005, 0.001)
    for t in ts:
        if t <= 0.0:
            continue
        h = (1.0 - t) * a + t * d
        h = np.maximum(h, 0.0)
        s = float(h.sum())
        if s <= 1e-300:
            continue
        h = h / s
        sc = _score_of(h)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = h.copy()
    return best_a, best_score


def _lp_direction(a):
    if not _HAS_LINPROG:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n < 2 or n > 1024:
        return None

    # Tiny offset for numerical stability; effect is negligible.
    a_lp = a + 1e-12
    s = float(a_lp.sum())
    if s <= 1e-300:
        return None
    a_lp = a_lp / s

    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    cols = np.arange(n)
    for lag in range(n):
        A[lag + cols, cols] = a_lp[lag]

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(rows, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = None
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True})
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds)
        except Exception:
            res = None

    if res is None or not getattr(res, "success", False):
        return None
    y = np.asarray(getattr(res, "x", None), dtype=np.float64) if res.x is not None else None
    if y is None:
        return None
    y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
    y = np.maximum(y, 0.0)
    sy = float(y.sum())
    if sy <= 1e-12:
        return None
    return y / sy


def _try_lp_step(a, current_score):
    a = np.asarray(a, dtype=np.float64).ravel()
    y = _lp_direction(a)
    if y is None:
        return a, current_score, False

    if float(np.dot(a, y)) > 1.0 - 1e-9:
        return a, current_score, False

    directions = [y]
    y2 = _sym_dir(y)
    if not np.allclose(y2, y, rtol=1e-8, atol=1e-12):
        directions.append(y2)
    y3 = _smooth_dir(y, 0.12)
    if not np.allclose(y3, y, rtol=1e-8, atol=1e-12):
        directions.append(y3)

    for d in directions:
        cand, sc = _line_search(a, d, current_score)
        if np.isfinite(sc) and sc < current_score - 1e-14:
            return cand, sc, True

    return a, current_score, False


def _try_reverse_step(a, current_score):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size < 2:
        return a, current_score, False
    rev = _normalize(a[::-1])
    if float(np.dot(a, rev)) > 1.0 - 1e-9:
        return a, current_score, False
    cand, sc = _line_search(a, rev, current_score)
    if np.isfinite(sc) and sc < current_score - 1e-14:
        return cand, sc, True
    return a, current_score, False


def _lse_refine(a, deadline, rng):
    a = _normalize(a)
    n = a.size
    if n < 2:
        a = _make_uniform(256)
        n = 256

    b_fft = _pow2(2 * n - 1)
    g_fft = _pow2(3 * n - 2)

    b = _autocorr(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    if not np.isfinite(score):
        a = _make_uniform(n)
        b = _autocorr(a, b_fft)
        maxb = float(np.max(b))
        score = 2.0 * float(n) * maxb

    best_a = a.copy()
    best_score = score

    alpha = 0.25
    beta_target = 8.0
    fail = 0
    it = 0

    while time.monotonic() < deadline:
        if maxb <= 1e-300:
            maxb = 1e-300
        beta = float(min(1e7, max(1.0, beta_target / maxb)))

        p, val_old = _lse_and_p(b, beta)
        grad = _weighted_gradient(a, p, g_fft, n)
        grad = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad)))

        if scale <= 1e-300:
            a = _perturb(best_a, rng, 0.15)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.25
            fail = 0
            it += 1
            continue

        if alpha < 0.001:
            alpha = 0.25

        v = a - alpha * grad
        a = _project_soft(v)

        b = _autocorr(a, b_fft)
        maxb = float(np.max(b))
        score = 2.0 * float(n) * maxb
        val_new = _lse_value(b, beta)

        if np.isfinite(val_new) and val_new < val_old - 1e-13:
            alpha = float(min(2.0, alpha * 1.10))
            fail = 0
        else:
            alpha *= 0.5
            fail += 1

        if np.isfinite(score) and score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()

        if it % 100 == 0:
            beta_target = float(min(1000.0, beta_target * 1.05 + 2.0))

        if fail > 300:
            a = _perturb(best_a, rng, 0.12)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.20
            beta_target = float(max(12.0, beta_target * 0.5))
            fail = 0

        if fail > 2500:
            break

        it += 1

    return best_a, best_score


def _refine_sequence(a, deadline, rng):
    a = _normalize(a)
    current_score = _score_of(a)
    if not np.isfinite(current_score):
        a = _make_uniform(256)
        current_score = _score_of(a)

    best_a = a.copy()
    best_score = current_score
    fail = 0
    it = 0

    while time.monotonic() < deadline:
        if not np.isfinite(current_score) or current_score <= 0.0:
            a = best_a.copy()
            current_score = best_score

        # If the LP direction has stopped helping, use a short LSE burst.
        if fail > 60:
            end_lse = min(deadline - 1.0, time.monotonic() + 15.0)
            if end_lse > time.monotonic() + 2.0:
                av, sv = _lse_refine(a, end_lse, rng)
                if np.isfinite(sv) and sv < current_score:
                    a = av
                    current_score = sv
                fail = 0
                if np.isfinite(sv) and sv < best_score:
                    best_score = sv
                    best_a = av.copy()
            else:
                fail = 0
        else:
            new_a, new_score, improved = _try_lp_step(a, current_score)
            if improved:
                a = new_a
                current_score = new_score
                fail = 0
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
            else:
                fail += 1

                if fail == 10:
                    a = _perturb(a, rng, 0.07)
                    current_score = _score_of(a)

                if fail > 20:
                    ra, rs, rok = _try_reverse_step(a, current_score)
                    if rok:
                        a = ra
                        current_score = rs
                        fail = 0
                        if current_score < best_score:
                            best_score = current_score
                            best_a = a.copy()

                if fail > 220:
                    a = _perturb(best_a, rng, 0.25)
                    current_score = _score_of(a)
                    if current_score > best_score + 0.05:
                        a = best_a.copy()
                        current_score = best_score

        it += 1

    if time.monotonic() < deadline - 2.0:
        av, sv = _lse_refine(best_a, deadline - 1.0, rng)
        if np.isfinite(sv) and sv < best_score:
            best_a = av.copy()
            best_score = sv

    return best_a, best_score


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """Return the best 256-height nonnegative sequence found within budget_s."""
    start_t = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0

    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    budget = max(20.0, budget)
    deadline = start_t + budget - 25.0
    if deadline <= start_t + 10.0:
        deadline = start_t + max(20.0, budget)

    rng = np.random.default_rng(seed)
    N = 256

    given = _get_given_sequence()

    inits = []
    if given is not None:
        inits.append(_resample(given, N))
    inits.append(_make_power(N, 0.50))
    inits.append(_make_power(N, 0.38))
    inits.append(_make_power(N, 0.62))
    inits.append(_make_exp(N, 1.0))
    inits.append(_make_exp(N, 2.5))
    if given is not None:
        inits.append(_mix_two(_resample(given, N), _make_power(N, 0.50), 0.80))
    inits.append(_smooth_random(N, rng, _make_power(N, 0.50), 0.35))
    inits.append(_smooth_random(N, rng, _make_uniform(N), 0.65))

    # De-duplicate obvious starts.
    seen = set()
    unique_inits = []
    for arr in inits:
        arr = _normalize(arr)
        step = max(1, arr.size // 32)
        sig = tuple(arr[::step])
        if sig not in seen:
            seen.add(sig)
            unique_inits.append(arr)
    inits = unique_inits

    if not inits:
        inits = [_make_uniform(N)]

    best_a = inits[0]
    best_score = _score_of(best_a)
    print(f"init best len={best_a.size} score={best_score:.10f}", flush=True)

    # Stage 1: cheap LSE polishing of several starts.
    init_budget = min(750.0, max(60.0, 0.30 * budget))
    init_deadline = start_t + init_budget

    if given is not None:
        end1 = min(init_deadline, start_t + min(150.0, max(30.0, 0.08 * budget)))
        if end1 > start_t + 5.0:
            av, sv = _lse_refine(best_a, end1, rng)
            if np.isfinite(sv) and sv < best_score:
                best_a = av.copy()
                best_score = sv
                print(f"given LSE score={best_score:.10f}", flush=True)
    else:
        for idx, arr in enumerate(inits):
            if time.monotonic() >= init_deadline:
                break
            end1 = min(init_deadline, time.monotonic() + 55.0)
            if end1 <= time.monotonic():
                continue
            av, sv = _lse_refine(arr.copy(), end1, rng)
            if np.isfinite(sv) and sv < best_score:
                best_a = av.copy()
                best_score = sv
                print(f"init {idx} LSE score={best_score:.10f}", flush=True)

    # Stage 2: main LP/Frank-Wolfe + LSE refinement from best point.
    main_end = deadline - 90.0
    if time.monotonic() < main_end:
        av, sv = _refine_sequence(best_a.copy(), main_end, rng)
        if np.isfinite(sv) and sv < best_score:
            best_a = av.copy()
            best_score = sv
        print(f"main refine score={best_score:.10f}", flush=True)

    # Stage 3: occasional resets from robust starts / perturbations.
    last_print = time.monotonic()
    while time.monotonic() < deadline - 50.0:
        if rng.random() < 0.55:
            a0 = _perturb(best_a, rng, 0.25)
        elif given is not None:
            a0 = _resample(given, N)
            if rng.random() < 0.5:
                a0 = _perturb(a0, rng, 0.25)
        else:
            alpha = float(rng.uniform(0.30, 0.75))
            if rng.random() < 0.5:
                a0 = _make_power(N, alpha)
            else:
                a0 = _make_exp(N, float(rng.uniform(0.7, 4.0)))
            a0 = _smooth_random(N, rng, a0, 0.25)

        end = min(deadline - 30.0, time.monotonic() + 120.0)
        if end <= time.monotonic():
            break

        av, sv = _refine_sequence(a0, end, rng)
        if np.isfinite(sv) and sv < best_score:
            best_a = av.copy()
            best_score = sv

        if time.monotonic() - last_print > 30.0:
            print(f"restart best={best_score:.10f} t={time.monotonic()-start_t:.1f}s", flush=True)
            last_print = time.monotonic()

    # Final polish.
    if time.monotonic() < deadline - 3.0:
        av, sv = _lse_refine(best_a.copy(), deadline - 1.0, rng)
        if np.isfinite(sv) and sv < best_score:
            best_a = av.copy()
            best_score = sv
        print(f"final polish score={best_score:.10f}", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-3:
        out = _make_uniform(N)
    else:
        out = out / s

    # Final external validation if available.
    if "evaluate_sequence" in globals():
        try:
            ev = evaluate_sequence([float(x) for x in out])
            if not np.isfinite(ev):
                out = _make_uniform(N)
        except Exception:
            pass

    print(f"returned len={len(out)} internal_score={_score_of(out):.10f}", flush=True)
    return [float(x) for x in out]
