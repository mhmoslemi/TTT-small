    def objective(h_vec):
        h = h_vec
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 100
    start_time = time.time()
    
    # Use SLSQP for optimization
    result = scipy.optimize.minimize(
        objective,
        h,
