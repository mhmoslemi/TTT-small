        h_initial,
        method=method,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=tol,
        options={'maxiter': max_iter}
    )

    # Check if optimization completed within budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return initial_h, evaluate_c5(initial_h), n_points

    best_h = result.x
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n_points
