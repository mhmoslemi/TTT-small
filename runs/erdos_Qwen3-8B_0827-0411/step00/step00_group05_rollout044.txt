        return (best_h, c5_bound, n_points)
    
    # Additional refinement if time allows
    # Use Nelder-Mead for further improvement
    result_nm = minimize(
        objective,
        best_h,
        args=(n_points,),
        method='Nelder-Mead',
        bounds=bounds,
        constraints=cons,
        options={'xatol': 1e-6, 'fatol': 1e-6}
    )
    
    best_h = result_nm.x
    c5_bound = objective(best_h, n_points)
    
    return (best_h, c5_bound, n_points)
