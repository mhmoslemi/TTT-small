import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    # Import the initial_h_values from the global scope
    global initial_h_values
    np.random.seed(seed)
    
    # Initial setup
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to normalize h to ensure sum is n_points / 2 and values are in [0, 1]
    def normalize_h(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scaling = (n_points / 2) / total
        h_scaled = h * scaling
        h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped
    
    # Function to compute C5
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initialize population
    population_size = 50
    population = [normalize_h(initial_h.copy())]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        population.append(normalize_h(h))
    
    # Evaluate initial population
    population_fitness = [compute_c5(h) for h in population]
    
    # Genetic algorithm parameters
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Select parents based on fitness
        parent_indices = np.argsort(population_fitness)[:population_size // 2]
        parents = [population[i] for i in parent_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - len(parents)):
            # Crossover
            if random.random() < crossover_rate:
                p1, p2 = random.sample(parents, 2)
                idx = random.randint(1, n_points - 1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            else:
                # Mutation
                child = population[random.randint(0, len(population) - 1)].copy()
            
            # Mutation

            offspring.append(child)
        
        # Replace population

    # Find the best individual
    best_idx = np.argmin(population_fitness)
