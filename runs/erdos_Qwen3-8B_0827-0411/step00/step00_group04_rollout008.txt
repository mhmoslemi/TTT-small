def run(seed=42, budget_s=100, **kwargs):
    # Import the initial_h_values

    # Ensure the initial h satisfies the integral constraint
    sum_h = np.sum(initial_h)

    initial_h = initial_h * scaling_factor
    initial_h = np.clip(initial_h, 0, 1)

    # Function to compute C5

        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5

    # Simulated Annealing parameters
    temp = 100.0
    cooling_rate = 0.995
    max_iterations = int(budget_s / 0.1)  # 10 iterations per second
    start_time = time.time()
