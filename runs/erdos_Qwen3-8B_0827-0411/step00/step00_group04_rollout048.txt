    import random

    # Initial h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    temp = 100.0  # Initial temperature
    cooling_rate = 0.995
    min_temp = 1e-3

        for _ in range(iterations_per_temp):
            # Generate a new solution by perturbing a few points
            indices = random.sample(range(n_points), 5)
            new_h = h.copy()
            for i in indices:
                delta = random.uniform(-0.1, 0.1)
                new_val = max(0.0, min(1.0, new_h[i] + delta))
                new_h[i] = new_val
            
            # Compute new C5
            new_c5 = compute_c5(new_h, n_points)
            
            # Acceptance probability
            if new_c5 < best_c5:
                # Better solution
                best_c5 = new_c5
                best_h = new_h.copy()
            else:
                # Worse solution, accept with probability
                delta_c5 = new_c5 - best_c5
                prob = np.exp(-delta_c5 / temp)
                if random.random() < prob:
                    best_c5 = new_c5
                    best_h = new_h.copy()
        
        # Cool the temperature
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
