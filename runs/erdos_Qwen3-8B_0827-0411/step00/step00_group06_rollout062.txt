def run(seed=42, budget_s=100, **kwargs):

        corr = np.correlate(h_array, one_minus_h, mode='full')

        max_corr = np.max(corr)

        c5 = max_corr * dx

        return c5

    # Function to normalize h to have sum target_sum

    def normalize_h(h):

        current_sum = np.sum(h)

        if current_sum == 0:

            return h

        scaling_factor = target_sum / current_sum

        return h * scaling_factor

    # Function to mutate h

    def mutate(h, mutation_rate=0.1):

        h = h.copy()

        for i in range(n_points):

            if np.random.rand() < mutation_rate:

                delta = np.random.uniform(-0.1, 0.1)

                h[i] += delta

                h[i] = np.clip(h[i], 0.0, 1.0)

        h = normalize_h(h)

        return h

        child = np.zeros_like(parent1)

                child[i] = parent2[i]
