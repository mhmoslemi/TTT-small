        population.append(new_h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    
    while time.time() - start_time < budget_s:
        # Evaluate all individuals
        evaluations = []
        for h in population:
            c5 = evaluate_c5(h)
            evaluations.append(c5)
        # Find the best
        best_index = np.argmin(evaluations)
        best_c5 = evaluations[best_index]
        best_h = population[best_index].copy()
        
        # Select top individuals
        sorted_indices = np.argsort(evaluations)
        selected = [population[i] for i in sorted_indices[:20]]
        
        # Generate new population
        new_population = []
        for h in selected:
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            new_h = h + perturbation
            new_sum = np.sum(new_h)
            scaling = target_sum / new_sum
            new_h = new_h * scaling
            new_h = np.clip(new_h, 0.0, 1.0)
            new_population.append(new_h)
        
        # Add some random individuals
        for _ in range(len(selected) - len(new_population)):
            random_h = np.random.uniform(0.0, 1.0, size=n_points)
            random_sum = np.sum(random_h)
            scaling = target_sum / random_sum
            random_h = random_h * scaling
            random_h = np.clip(random_h, 0.0, 1.0)
            new_population.append(random_h)
        
        # Update population
        population = new_population
    
    return (best_h, best_c5, n_points)
