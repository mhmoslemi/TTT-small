        # Ensure initial_guess is within [0, 1]
        initial_guess = np.clip(initial_guess, 0, 1)
        # Enforce sum constraint
        sum_initial = np.sum(initial_guess)
        initial_guess = initial_guess * (n_points / 2) / sum_initial if sum_initial != 0 else initial_guess
        
        # Define objective with penalty
        def objective(h_values):
            penalty = 1e6 * abs(np.sum(h_values) - n_points / 2)
            return compute_c5(h_values, n_points) + penalty
        
        # Run differential evolution
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(
            objective, bounds, strategy='best1bin', 
            maxiter=100, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8,
            callback=lambda x: None, disp=False
        )
        
        # Check if within budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        current_c5 = compute_c5(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Fallback to initial guess if no improvement
    if best_h is None:
        best_h = initial_h_values
    
    # Final check with initial guess
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
