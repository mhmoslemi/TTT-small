    elapsed_time = time.time() - start_time
    
    if result.success and elapsed_time < budget_s:
        h_values = result.x
        c5_bound = result.fun
    else:
        # Fallback to initial_h_values if optimization fails or time is exceeded
        h_values = initial_h_values
        c5_bound = objective(initial_h_values, dx)
    
    return h_values, c5_bound, n_points
