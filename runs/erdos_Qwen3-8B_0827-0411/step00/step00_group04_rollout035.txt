import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    # Initial guess based on random distribution
    h_initial = np.random.rand(n_points)
    h_initial = h_initial * (n_points / 2.0) / np.sum(h_initial)
    
    def objective(h_flat, dx, penalty_coeff):
        h = h_flat.reshape(n_points)
        # Penalty for sum constraint
        penalty = penalty_coeff * (np.sum(h) - n_points / 2) ** 2
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + penalty
    
    bounds = [(0, 1) for _ in range(n_points)]
    # Run differential evolution with penalty
    result = differential_evolution(
        lambda h_flat: objective(h_flat, dx, 1000),
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    best_h = result.x
    # Compute the actual C5 value by subtracting the penalty
    best_c5 = objective(best_h, dx, 1000) - 1000 * (np.sum(best_h) - n_points / 2) ** 2
    return (best_h, best_c5, n_points)
