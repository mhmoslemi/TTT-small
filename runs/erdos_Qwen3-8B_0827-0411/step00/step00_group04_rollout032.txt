import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100  # balance between speed and resolution
    dx = 2.0 / n_points
    # Initialize with a random guess that satisfies the integral constraint
    h_initial = np.random.rand(n_points)
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Constraints: sum(h) = n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with COBYLA method
    result = minimize(objective, h_initial, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return result.x, result.fun, n_points
