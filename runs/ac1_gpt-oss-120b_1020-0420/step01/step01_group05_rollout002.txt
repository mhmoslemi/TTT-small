def _project_simplex(v):
    # project v onto simplex {x >=0, sum x = 1}
    # Use algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

# --------------------------------------------------------------
# Fast AC1 bound (matrix–free) using FFT.
# --------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a))²  –  FFT based, O(n log n)."""
    n = a.shape[0]
    m = 2 * n - 1                     # full convolution length
    nfft = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float("inf")
    return 2.0 * n * M / (S * S)

    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    return c


# --------------------------------------------------------------
# Gradient (subgradient) of max‑convolution w.r.t. a.
# --------------------------------------------------------------
def _max_conv_grad(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return g_i = 2·a_{kmax-i} for indices that participate,
    else 0, where kmax is the index attaining max(conv)."""
    k = int(np.argmax(conv))
    n = a.shape[0]
    idx = np.arange(n)
    partner = k - idx
    mask = (partner >= 0) & (partner < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[partner[mask]]
    return g


# --------------------------------------------------------------
# Main entry point required by the harness.
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Parent fallback (already available in the execution environment).
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:                     # any problem -> simple uniform fallback
        parent = np.ones(256, dtype=np.float64)

    n = parent.shape[0]
    parent = np.clip(parent, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Build a modest collection of analytic seeds (all normalized to sum=1).
    # ------------------------------------------------------------------
    seeds = []

    # 0) Uniform (baseline)
    seeds.append(np.ones(n, dtype=np.float64) / n)

    # 1) Normalized parent
    if np.sum(parent) >= 0.01:
        seeds.append(parent / np.sum(parent))

    i = np.arange(n, dtype=np.float64)

    # 2) Power‑law families
    for p in (0.3, 0.4, 0.5, 0.6, 0.7):
        a = (i + 1.0) ** (-p)
        seeds.append(a / np.sum(a))
        a_rev = (n - i) ** (-p)
        seeds.append(a_rev / np.sum(a_rev))

    # 3) Inverse‑square‑root families
    for off in (0.1, 0.5, 1.0, 2.0, 4.0):
        a = 1.0 / np.sqrt(i + off)
        seeds.append(a / np.sum(a))
        a_rev = 1.0 / np.sqrt((n - 1 - i) + off)
        seeds.append(a_rev / np.sum(a_rev))

    # ------------------------------------------------------------------
    # Keep the best seed so far (fast evaluation only).
    # ------------------------------------------------------------------
    best_seq = seeds[0].copy()
    best_score = _fast_score(best_seq)
    for a in seeds[1:]:
        s = _fast_score(a)
        if s < best_score:
            best_seq = a.copy()
            best_score = s

    # ------------------------------------------------------------------
    # Gradient‑descent refinement on the most promising seeds.
    # ------------------------------------------------------------------
    # Sort seeds by their fast scores (ascending) and keep the top few.
    scored = [(a, _fast_score(a)) for a in seeds]
    scored.sort(key=lambda x: x[1])
    top_seeds = [a for a, _ in scored[:5]]

    for seed_vec in top_seeds:
        seq = seed_vec.copy()
        # number of inner iterations – modest to stay linear‑memory.
        for it in range(250):
            conv = _conv_full(seq)
            grad = _max_conv_grad(seq, conv)

            # step size decays with iteration number
            step = 0.8 / (it + 1.0)

            seq = seq - step * grad
            seq = np.clip(seq, 0.0, 1000.0)
            total = seq.sum()
            if total <= 1e-12:                     # avoid division by zero
                seq = np.full_like(seq, 1.0 / n)
            else:
                seq = seq / total                  # renormalise to sum=1

            s = _fast_score(seq)
            if s < best_score:
                best_score = s
                best_seq = seq.copy()

    # ------------------------------------------------------------------
    # Final short stochastic polishing (tiny multiplicative jitter).
    # ------------------------------------------------------------------
    for _ in range(500):
        perturb = np.exp(rng.normal(0.0, 0.018, size=n))
        cand = best_seq * perturb
        cand = np.clip(cand, 0.0, 1000.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand = cand / cand_sum
        s = _fast_score(cand)
        if s < best_score:
            best_score = s
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
