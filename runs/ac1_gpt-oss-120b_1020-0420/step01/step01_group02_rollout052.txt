def fast_ac1(seq):
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    L = 1 << ((2*n-1).bit_length())
    # compute conv
    fftseq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fftseq*fftseq, n=L)[:2*n-1]
    maxc = conv.max()
    return 2.0 * n * maxc / (s * s)

import numpy as np

# ------------------------------------------------------------
# Generate the truncated series of (1 - z)^(-1/2)
# a_0 = 1,  a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return a length‑`length` array a_i = binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound using FFT‑based convolution.
# ------------------------------------------------------------
def fast_ac1(seq: np.ndarray) -> float:
    """Return the AC1 value 2*n*max(conv)/sum(seq)**2 using FFT."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    # length for zero‑padded FFT (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)
