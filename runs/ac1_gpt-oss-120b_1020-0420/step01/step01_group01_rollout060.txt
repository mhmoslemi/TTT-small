def _make_seq(n, exponent):
    i = np.arange(1, n+1, dtype=np.float64)
    raw = 1.0 / np.power(i, exponent)
    s = raw.sum()
    scale = np.sqrt(2.0 * n) / s
    seq = raw * scale
    # Ensure bounds
    seq = np.clip(seq, 0.0, 1000.0)
    return seq

def propose_candidate(seed: int = 42, **kwargs):

    # Try a handful of exponents

        seq = _make_seq(n, exp)
        # evaluate using provided function

    return [float(v) for v in best_seq]

    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

def propose_candidate(seed: int = 42, **kwargs):

    # Determine length n
    if "height_sequence_1" in globals():

    # destinations, keep any improvement.

                cand[:k] = new_early

                sc = evaluate_sequence(cand.tolist())

        k0 = int(np.argmax(conv))                       # location of the current maximum
        i_start = max(0, k0 - (n - 1))
        i_end = min(n - 1, k0)
        idx = np.arange(i_start, i_end + 1)
        contrib = best[idx] * best[k0 - idx]            # contribution of each index to conv[k0]
        src = int(idx[np.argmax(contrib)])               # heavy‑contributor
        # choose a destination among the smallest entries (far from src)
        dst_candidates = np.where(best == best.min())[0]
        dst = int(rng.choice(dst_candidates))
        if src == dst:
            continue
        # transfer up to 30 % of the source entry
        delta = best[src] * rng.uniform(0.01, 0.30)
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # --------------------------------------------------------
    return [float(v) for v in best]
```
