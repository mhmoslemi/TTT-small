def propose_candidate(seed=42):
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, np.ndarray):
        a = height_sequence_1.astype(np.float64).copy()
    else:
        # generate half power series length maybe 2048? use half_power_series as fallback
        n = 2048
        a = generate_half_power_series(n)
    # Ensure non-negativity and clip
    np.clip(a, 0.0, 1000.0, out=a)

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    # hill climbing
    for step in range(2000):
        # compute convolution and find index of max
        conv = np.convolve(best, best)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]
        # pick one random location among them
        k = int(rng.choice(max_idxs))
        # compute contributions: indices i where partner j=k-i in [0,n)
        # Let idx_i_range = max(0, k-(n-1)) .. min(n-1, k)
        i_low = max(0, k - (best.size - 1))
        i_high = min(best.size - 1, k)
        # choose random i within this range weighted by product a_i * a_j
        # compute for each i product
        # To avoid O(n) heavy each iteration, we could randomly sample few i's.
        # Simpler: choose random i from that range.
        i = int(rng.integers(i_low, i_high+1))
        j = k - i
        # Transfer epsilon from max(a_i, a_j) to some other index far from both
        # Choose a donor index d (i or j) with larger a
        if best[i] > best[j]:
            donor = i
        else:
            donor = j
        # Choose recipient index r random but not in contributions for same k? or any.
        r = int(rng.integers(0, best.size))
        # Ensure r not equal donor
        if r == donor:
            r = (r + 1) % best.size
        # compute epsilon = min( best[donor] * 0.05, a small fraction)
        eps = best[donor] * 0.05
        eps = max(eps, 1e-6)  # small non-zero
        # transfer
        cand = best.copy()
        cand[donor] = max(0.0, cand[donor] - eps)
        cand[r] = cand[r] + eps
        # compute score
        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
        # else maybe try other modifications? optionally continue.

    # ensure result proper list
    return [float(x) for x in best]

import numpy as np

def _compute_score(seq: np.ndarray) -> float:
    """Exact copy of the evaluator's objective (without clipping safety)."""
    # Ensure non‑negative and bounded as the evaluator would do.
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    m = np.max(conv)
    return float(2.0 * seq.size * m / (s * s))


def _half_power_series(length: int) -> np.ndarray:
    """Coefficients of (1‑z)^(-1/2): a_0=1, a_i = a_{i-1}*(2i‑1)/(2i)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that (heuristically)
    minimises the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Initialise from the parent if present, otherwise use the half‑power series.
    # ------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2048
        base = _half_power_series(n)

    n = base.size
    np.clip(base, 0.0, 1000.0, out=base)

    # ------------------------------------------------------------
    # Keep the best sequence seen so far.
    # ------------------------------------------------------------
    best = base
    best_score = _compute_score(best)

    # ------------------------------------------------------------
    # Hill‑climbing: move a small amount of mass from a donor that
    # participates in the current maximal convolution to a random
    # recipient elsewhere, preserving the total sum.
    # ------------------------------------------------------------
    max_iter = 1500
    no_improve = 0
    for step in range(max_iter):
        # locate a (random) shift attaining the maximum convolution value
        conv = np.convolve(best, best)
        max_val = conv.max()
        max_positions = np.where(conv == max_val)[0]
        k = int(rng.choice(max_positions))

        # indices that contribute to this shift
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        if i_low > i_high:
            continue   # should not happen, but protect against degenerate n=0

        # pick a random contributing index
        i = int(rng.integers(i_low, i_high + 1))
        j = k - i
        # choose the larger of the pair as donor (more impact on the max)
        donor = i if best[i] >= best[j] else j

        # pick a recipient far from the donor (anywhere else)
        recipient = int(rng.integers(0, n))
        if recipient == donor:
            recipient = (recipient + 1) % n

        # transfer a modest fraction of the donor's mass
        if best[donor] <= 0.0:
            continue
        fraction = float(rng.uniform(0.015, 0.035))   # 1.5 %–3.5 %
        epsilon = best[donor] * fraction

        cand = best.copy()
        cand[donor] = max(0.0, cand[donor] - epsilon)
        cand[recipient] = cand[recipient] + epsilon
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = _compute_score(cand)

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            no_improve = 0

            # occasional random restart if stuck for long

                # small random reshuffle to escape flat regions
                perm = rng.permutation(n)
                best = best[perm]
                no_improve = 0

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
