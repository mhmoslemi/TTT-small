import numpy as np
def half_binomial_coeffs(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0*i - 1.0) / (2.0 * i)
    return coeffs

def propose_candidate(seed=42, **kwargs):
    # Determine n
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256
    coeffs = half_binomial_coeffs(n)
    # Scale so sum = sqrt(2*n)
    sum0 = np.sum(coeffs)
    target_sum = np.sqrt(2.0 * n)
    if sum0 > 0:
        scale = target_sum / sum0
    else:
        scale = 1.0
    coeffs = coeffs * scale
    # clip to bounds if needed
    np.clip(coeffs, 0.0, 1000.0, out=coeffs)
    return coeffs.tolist()

import numpy as np

# ------------------------------------------------------------
# Helper: half‑binomial coefficients (1‑z)^‑1/2 series
# ------------------------------------------------------------
def _half_binomial_coeffs(n: int) -> np.ndarray:
    """Return the first n coefficients of (1‑z)^‑1/2.
    C_i = comb(2i,i)/4**i, generated by recurrence.
    """
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs

        # evaluate candidate
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            break

        # keep improvement
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
        else:
            # if no improvement, reduce the fraction and retry
            fraction *= 0.5
            if fraction < 1e-4:
                break

    # -----------------------------------------------------------------
    # Fallback: if the supplied parent construction is better, use it.
    # -----------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent
            best_score = parent_score
    except Exception:
        pass

    # guarantee a finite, non‑empty result
    if best_score == float('inf') or best.size == 0:
        # as a last resort return a simple uniform vector (never fails)
        best = np.full(n, 1.0 / n, dtype=np.float64)
        np.clip(best, 0.0, 1000.0, out=best)

    return [float(x) for x in best]
