def make_power_law(p, offset, n, rev=False):
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    if rev:
        a = a[::-1]
    return a

import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                                 # pointwise product → convolution
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

# ----------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound.

    The routine first evaluates a safe fallback (the supplied parent or a uniform
    vector) and then explores a variety of compact analytic families:
      * power‑law a_i = (i+offset)^{-p} (and its reverse)
      * mixtures of a power‑law and its reverse
      * exponential decay a_i = exp(-γ·i/(n‑1))
      * random exponentials of low‑degree polynomials
    Random jitter around the current best provides a final refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Parent fallback – a pre‑existing construction is available as
    # `height_sequence_1`.  Use it if it looks valid, otherwise fall back
    # to a modest uniform vector.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    parent = np.clip(parent, 0.0, 1000.0)

    # start from a uniform vector (classical bound 2.0)
    best_seq = np.ones(n, dtype=np.float64)
    best_score = _fast_score(best_seq)

    # also try the supplied parent – it could already be good
    if parent.sum() >= 0.01:
        s = _fast_score(parent)
        if s < best_score:
            best_seq = parent.copy()
            best_score = s

    i = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------

    offsets = np.concatenate([

    for off in offsets:

        for p in powers:
