import numpy as np

# ------------------------------------------------------------
# Convolution (full) of a vector with itself
# ------------------------------------------------------------
def compute_full_conv(vec: np.ndarray) -> np.ndarray:
    """Return the full convolution vec * vec."""
    return np.convolve(vec, vec, mode="full")


# ------------------------------------------------------------
# Contribution of each index to the convolution value at k
# ------------------------------------------------------------
def contribution_vector(vec: np.ndarray, k: int) -> np.ndarray:
    """
    For a given sum-index k, return an array `c` where
    c[i] = vec[i] * vec[k-i] if both indices are valid, else 0.
    """
    n = vec.shape[0]
    i = np.arange(n, dtype=np.int64)
    j = k - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(vec)
    contrib[mask] = vec[mask] * vec[j[mask]]
    return contrib
