import numpy as np

def compute_grad(a):
    n = a.size
    b = np.convolve(a, a)
    max_b = np.max(b)
    idx = np.nonzero(b == max_b)[0]   # indices
    grad_b = np.zeros_like(a)
    for k in idx:
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # vectorized: for j in range(lo, hi+1):
        # partner = k - j
        # if j == partner: grad_b[j] += a[partner]
        # else: grad_b[j] += 2*a[partner]
        # use slices?
        # Could compute using loops.
        for j in range(lo, hi+1):
            partner = k - j
            if partner < 0 or partner >= n:
                continue
            if j == partner:
                grad_b[j] += a[partner]
            else:
                grad_b[j] += 2.0 * a[partner]
    sum_a = a.sum()

    factor = 2.0 * n / (sum_a**2)
    grad = factor * grad_b - (4.0 * n * max_b / (sum_a**3)) * a
    return grad, max_b, sum_a

import numpy as np

# Gradient computation for bound B = 2 n max(b) / sum(a)^2
def compute_grad(a):
    n = a.shape[0]
    # convolution (full)
    b = np.convolve(a, a)
    max_b = b.max()
    # indices where max occurs
    max_idx = np.where(b == max_b)[0]
    grad_b = np.zeros_like(a)
    # for each index k where max occurs, accumulate derivative contributions
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # vector of j values
        j = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k - j
        # partner is guaranteed between 0 and n-1
        weight = np.where(j == partner, 1.0, 2.0)
        grad_b[j] += a[partner] * weight
    S = a.sum()
    if S <= 0.0:
        S = 1e-12
    factor = 2.0 * n / (S * S)
    grad = factor * grad_b - (4.0 * n * max_b / (S * S * S)) * a
    return grad, max_b, S
