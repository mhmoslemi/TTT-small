def _deterministic_reallocation(seq):
    best = seq.copy()
    best_score = _score(best)
    n = seq.size
    prefix_ratios = (0.05, 0.10, 0.20)
    # shrink_factors using np.linspace as before.
    shrink_factors = np.linspace(0.5, 1.0, 11)
    for r in prefix_ratios:
        k = max(1, int(n * r))
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()
            g = 1.0 + delta / tail_sum
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                early = best[:k]
                tail = best[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()
    return best, best_score

import numpy as np

# ------------------------------------------------------------
# Score function (exactly matches the evaluator's computation)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the AC1 bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine the problem size – prefer a provided parent.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        ):
            seq = np.clip(parent, 0.0, 1000.0)
            n = seq.size
        else:
            n = 256
            seq = _catalan_seq(n)
    else:
        n = 256
        seq = _catalan_seq(n)

    # Ensure a minimal total mass (the evaluator rejects sum < 0.01)
    if seq.sum() < 0.01:
        seq += (0.01 - seq.sum()) / n

    best_seq = seq.copy()
    best_score = _score(best_seq)

    # --------------------------------------------------------
    # Deterministic reallocation (shrink prefix, inflate suffix)
    # --------------------------------------------------------
    prefix_ratios = (0.05, 0.10, 0.20)
    shrink_factors = np.linspace(0.5, 1.0, 11)  # 0.5 … 1.0

    for r in prefix_ratios:
        k = max(1, int(n * r))
        early = best_seq[:k]
        tail = best_seq[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()
            g = 1.0 + delta / tail_sum
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if sc < best_score:
                best_seq = cand
                best_score = sc
                # recompute early / tail for further tweaks
                early = best_seq[:k]
                tail = best_seq[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()

    # --------------------------------------------------------
    # Greedy peak‑reduction loop
    # --------------------------------------------------------
    steps = int(kwargs.get("num_steps", num_steps))
    no_improve = 0
    max_no_improve = steps // 10 if steps >= 10 else 0

    for it in range(steps):
        conv = np.convolve(best_seq, best_seq)
        kmax = int(conv.argmax())                 # index of the current peak
        # indices (i,j) with i+j = kmax
        i0 = max(0, kmax - (n - 1))
        i1 = min(kmax, n - 1)
        i_range = np.arange(i0, i1 + 1)
        j_range = kmax - i_range
        contrib = best_seq[i_range] * best_seq[j_range]
        idx = int(contrib.argmax())
        i_sel = i_range[idx]
        j_sel = j_range[idx]

        # choose the larger of the two as the source of mass
        source = i_sel if best_seq[i_sel] >= best_seq[j_sel] else j_sel

        # pick a random destination different from source
        dest = rng.integers(0, n)
        while dest == source:
            dest = rng.integers(0, n)

        # decreasing step size (from ≈5 % down to ≈0.1 %)
        step_frac = 0.05 * (1.0 - it / steps) + 0.001
        delta = best_seq[source] * step_frac
        if delta <= 0.0:
            continue

        cand = best_seq.copy()
        cand[source] -= delta
        cand[dest] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if max_no_improve and no_improve >= max_no_improve:
                break   # stop early if progress stalls

    # --------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # --------------------------------------------------------
    return [float(v) for v in best_seq]
