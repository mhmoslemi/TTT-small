import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """
    Generate the first ``length`` terms of the sequence a_i = C(2i,i)/4**i.
    This sequence satisfies (a * a)[k] == 1 for all 0 <= k < length.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _smooth_once(arr: np.ndarray) -> np.ndarray:
    """
    One step of nearest‑neighbour averaging.
    New[i] = (arr[i-1] + arr[i+1]) / 2, with zero outside the interval.
    """
    left = np.concatenate((arr[1:], [0.0]))
    right = np.concatenate(([0.0], arr[:-1]))
    return (left + right) * 0.5


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function sequence that improves the AC1 score.
    The method starts from the classic root‑constant sequence and repeatedly
    smooths it while renormalising the total sum.  The best candidate seen
    during the short search is returned.
    """
    # Determine the length (use the provided parent if present)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256  # safe fallback length

    # Base construction
    base = _root_constant_seq(n)
    original_sum = float(base.sum())

    # Evaluate the initial candidate (fallback handling later)
    try:
        best_score = float(evaluate_sequence(base.tolist()))
    except Exception:
        best_score = np.inf
    best_seq = base.copy()

    # If the initial score is already invalid, replace with a simple uniform fallback
    if not np.isfinite(best_score):
        base = np.full(n, 1.0, dtype=np.float64)
        original_sum = float(base.sum())
        best_seq = base.copy()
        best_score = float(evaluate_sequence(base.tolist()))

    # Perform a few smoothing steps; after each step rescale to keep the original sum.
    max_iters = 12
    cur = base.copy()
    for _ in range(max_iters):
        # smoothing
        smoothed = _smooth_once(cur)

        # keep values within allowed range before scaling
        smoothed = np.clip(smoothed, 0.0, 1000.0)

        # renormalise to preserve the original total mass
        cur_sum = float(smoothed.sum())
        if cur_sum == 0.0:

        scale = original_sum / cur_sum
        cur = smoothed * scale
        cur = np.clip(cur, 0.0, 1000.0)
