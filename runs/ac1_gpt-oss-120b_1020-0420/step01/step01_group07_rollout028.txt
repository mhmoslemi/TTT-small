def compute_conv_max(a):
    n = a.shape[0]
    nfft = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[:2*n-1]
    max_c = conv.max()
    return float(max_c)
```

import numpy as np

# ------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n) memory O(n))
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray, nfft: int) -> float:
    """
    Return max entry of the full convolution a * a using FFT.
    a is one‑dimensional, non‑negative. nfft is a power of two >= 2*len(a)-1.
    """
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * a.size - 1]
    return float(conv.max())


def _bound(a: np.ndarray, nfft: int) -> float:
    """AC1 upper bound for a (with clipping to the evaluator's limits)."""
    # clipping follows evaluate_sequence's policy
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    max_c = _conv_max(a, nfft)
    return 2.0 * a.size * max_c / (sum_a * sum_a)


# ------------------------------------------------------------
# Simple descent that flattens the convolution peak
# ------------------------------------------------------------
def _flatten_descent(start: np.ndarray, steps: int, nfft: int, rng: np.random.Generator):
    """
    Perform a few gradient‑like steps aiming to reduce the maximum
    convolution entry.  Returns the best vector seen.
    """
    n = start.size
    a = start.copy()
    # keep total mass constant (scale after each update)
    target_sum = a.sum()
    best_vec = a.copy()
    best_sc = _bound(best_vec, nfft)

    for it in range(steps):
        # convolution and location of its maximum
        af = np.fft.rfft(a, n=nfft)
        conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # gradient of the peak (ignoring the factor 2, absorbed in step size)
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        jdx = k_max - idx
        grad[idx] = a[jdx]

        # step size slowly decreasing
        eta = 0.6 / (it + 1.0)

        a -= eta * grad
        a = np.clip(a, 0.0, 1000.0)

        # restore total sum (scale does not change the bound)
        cur_sum = a.sum()
        if cur_sum > 0:
            a *= target_sum / cur_sum

        # occasional random jitter to avoid stagnation
        if it % 15 == 0:
            mask = rng.random(n) < 0.015
            if mask.any():
                a[mask] *= np.exp(rng.normal(0.0, 0.018, size=mask.sum()))
                a = np.clip(a, 0.0, 1000.0)
                # re‑balance the sum again
                cur_sum = a.sum()
                if cur_sum > 0:
                    a *= target_sum / cur_sum

        sc = _bound(a, nfft)
        if sc < best_sc:
            best_sc = sc
            best_vec = a.copy()
            # early exit if we beat the target bound
            if best_sc <= 1.50274349:
                break

    return best_vec, best_sc
