        # if range too small, break
        if i_max - i_min < 1:
            # random perturbation fallback
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue
            delta = seq[src] * rng.uniform(0.01, 0.05)
            cand = seq.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand = np.clip(cand, 0.0, 1000.0)
            new_conv = np.convolve(cand, cand)
            new_sc = 2 * n * new_conv.max() / (cand.sum()**2)
            if new_sc < cur_score:
                seq = cand
                conv = new_conv
                cur_score = new_sc
                if new_sc < best_sc:
                    best_sc = new_sc
                    best_seq = cand.copy()
            continue

        # compute contributions for each i in range
        # products = seq[i] * seq[kmax - i]
        idxs = np.arange(i_min, i_max+1)
        partners = kmax - idxs
        prods = seq[idxs] * seq[partners]
        # ignoring diagonal factor 2, just use product magnitude
        # choose source index weighted by product
        total = prods.sum()
        if total == 0.0:
            # fallback random
            src = rng.integers(0, n)
        else:
            # sample index proportionally to product
            probs = prods / total
            src = rng.choice(idxs, p=probs)
        partner = kmax - src
        # ensure src != partner
        if src == partner:
            # pick another
            if i_max - i_min >= 2:
                # pick different
                alt = src + 1 if src < i_max else src - 1
                src = alt
                partner = kmax - src
            else:
                # fallback random
                src = rng.integers(0, n)
                partner = (kmax - src) % n

        # choose destination indices different from src and partner
        dst1 = rng.integers(0, n)
        while dst1 == src or dst1 == partner:
            dst1 = rng.integers(0, n)
        dst2 = rng.integers(0, n)
        while dst2 == src or dst2 == partner or dst2 == dst1:
            dst2 = rng.integers(0, n)

        # compute transfer amount
        delta_max = min(seq[src], seq[partner])
        if delta_max <= 0.0:
            continue
        frac = rng.uniform(0.01, 0.05)
        delta = delta_max * frac

        cand = seq.copy()
        cand[src] -= delta
        cand[partner] -= delta
        cand[dst1] += delta
        cand[dst2] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        new_conv = np.convolve(cand, cand)
        new_sc = 2 * n * new_conv.max() / (cand.sum()**2)

        if new_sc < cur_score:
            seq = cand
            conv = new_conv
            cur_score = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_seq = cand.copy()
        # else reject and keep old seq

    return [float(v) for v in best_seq]

import numpy as np

# ------------------------------------------------------------
# Utility: compute the exact objective used by the harness
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """2*n*max(conv(seq,seq))/sum(seq)^2 with clipping."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ------------------------------------------------------------
# Catalan‐like initial sequence (a_i = C(2i,i)/4^i)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a
