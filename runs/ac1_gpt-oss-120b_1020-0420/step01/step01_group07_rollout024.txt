import numpy as np

# ------------------------------------------------------------
# Helper: evaluate bound quickly for binary mask (optional)
# ------------------------------------------------------------
def _bound_binary(mask, n):
    """
    mask : 1‑D array of 0/1 (int or bool)
    returns bound = 2*n*max_c / (m**2) where
        max_c = max ordered‑pair counts,
        m    = number of ones.
    """
    m = int(mask.sum())
    if m == 0:
        return float("inf")
    cnt = np.convolve(mask.astype(np.int32), mask.astype(np.int32))
    max_c = int(cnt.max())
    return 2.0 * n * max_c / (m * m)


# ------------------------------------------------------------
# Gradient step (projected) for a real‑valued vector
# ------------------------------------------------------------
def _gradient_step(x, lr):
    """
    One projected gradient step that reduces the peak of the autocorrelation.
    x : non‑negative float array (sum is positive)
    lr: learning rate (float)
    returns the updated vector (still non‑negative, renormalised to sum 1)
    """
    n = x.shape[0]
    c = np.convolve(x, x)
    k_max = int(np.argmax(c))

    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    i_range = np.arange(i_start, i_end + 1)
    j = k_max - i_range

    # gradient of the active convolution entry
    grad_vals = np.where(i_range == j, x[i_range], 2.0 * x[j])
    grad = np.zeros_like(x)
    grad[i_range] = grad_vals

    y = x - lr * grad
    y = np.clip(y, 0.0, 1000.0)
    s = y.sum()
    if s > 1e-12:
        y = y / s
    return y
