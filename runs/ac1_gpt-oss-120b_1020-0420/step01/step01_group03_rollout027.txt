import numpy as np

# ----------------------------------------------------------------------
#  Root‑constant sequence a_i = C(2i,i)/4**i.
#  It satisfies (a*a)[k] == 1 for 0 <= k < len(a).
# ----------------------------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    # safety clamping – the evaluator also clips, but we keep it.
    return np.clip(seq, 0.0, 1000.0)

#  Helper to safely evaluate a candidate sequence.
# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:

    Produce a non‑negative step‑function (list of floats) whose
    AC1 upper‑bound score is lower than the reference 1.570878.

    The routine starts from the classic root‑constant sequence and
    performs a few iterations of a simple linear‑programming update
    (the “exchange” step described in the problem statement).  The
    LP is solved with CVXPY in a matrix‑free fashion – only the

    """
    # --------------------------------------------------------------
    #  Determine a manageable length.
    #  The parent sequence (height_sequence_1) may be very long; we
    #  cap the optimisation to at most 1024 entries so that the LP
    #  stays modest in size while still giving a good score.
    # --------------------------------------------------------------
    try:
        full_len = int(len(height_sequence_1))
        seq_len = min(full_len, 1024)
        base = np.array(height_sequence_1[:seq_len], dtype=np.float64)
    except Exception:

        base = _root_constant_seq(seq_len)

    # ensure the base sequence respects the evaluator's bounds
    base = np.clip(base, 0.0, 1000.0)

    # start from the base sequence
    f = base.copy()
    best_seq = f.copy()
    best_score = _evaluate(best_seq)
    if not np.isfinite(best_score):
        best_score = float('inf')

    # ------------------------------------------------------------------
    #  Import CVXPY lazily – if it is unavailable we fall back to the
    #  original root‑constant sequence.
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        cp = None

    if cp is None:
        # No CVXPY – return the plain root‑constant sequence.
        return best_seq.tolist()

    # ------------------------------------------------------------------
    #  Parameters for the iteration.
    # ------------------------------------------------------------------
    mixing = 0.15                # step size t in the description
    max_iters = 8                # a few cheap LP solves
    target_sum = np.sqrt(2.0 * seq_len)   # normalisation used in the paper

    for _ in range(max_iters):
        # --------------------------------------------------------------
        #  Current convolution bound M = max (f*f)
        # --------------------------------------------------------------
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # --------------------------------------------------------------
        #  Linear programme:
        #      maximise   sum(g_j)
        #      subject to   g >= 0
        #                    conv(f,g) <= M   (elementwise)
        # --------------------------------------------------------------
        g = cp.Variable(seq_len)
        constraints = [
            g >= 0,
            cp.conv(cp.Constant(f), g) <= M
        ]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:

        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_val = np.asarray(g.value, dtype=np.float64)
        g_val = np.clip(g_val, 0.0, 1000.0)

        # --------------------------------------------------------------
        #  Rescale g to the normalisation used by the original algorithm.
        # --------------------------------------------------------------

        if sum_g == 0.0:

        #  Convex combination f <- (1‑t) f + t g
        # --------------------------------------------------------------
        f_new = (1.0 - mixing) * f + mixing * g_val
        f_new = np.clip(f_new, 0.0, 1000.0)

        # --------------------------------------------------------------
        #  Evaluate the new candidate.
        # --------------------------------------------------------------
        score_new = _evaluate(f_new)
        if np.isfinite(score_new) and score_new < best_score:
            best_seq = f_new.copy()
            best_score = score_new

        # use the new f for the next iteration (even if it did not improve)
        f = f_new

        if best_score < 1.5028:

    #  Safety net – if everything failed, fall back to the root constant.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best_seq = _root_constant_seq(seq_len)

    # Ensure the final sequence is a plain Python list of floats.
    return best_seq.tolist()
