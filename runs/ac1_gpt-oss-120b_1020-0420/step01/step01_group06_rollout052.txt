def fast_score(a):
    n = a.shape[0]
    L = 1 << (2*n-1).bit_length()
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a <= 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

def _fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the AC1 upper bound using FFT."""
    n = a.shape[0]
    # length for zero‑padded FFT (next power of two >= 2n-1)
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale vector to have sum == 1 (preserves positivity)."""
    total = v.sum()
    if total <= 0.0:
        # return a small uniform vector to avoid division by zero
        return np.full_like(v, 1.0 / v.size)
    return v / total


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 upper
    bound is hopefully below the target ~1.5027.
    The routine works by:
      1. building a modest catalogue of analytic families,
      2. picking the best of them,
      3. performing a cheap stochastic hill‑climb with FFT‑based scoring.
    """
    # ------------------------------------------------------------------
    # 1. Base length and optional parent sequence
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            parent = np.clip(tmp, 0.0, 1000.0)

    n = parent.size if parent is not None else 256

    # ------------------------------------------------------------------
    # 2. Helper to generate analytic candidates
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)           # 1‑based indices
    rev_idx = np.arange(n, 0, -1, dtype=np.float64)      # reversed indices
    ar = np.arange(n, dtype=np.float64)                  # 0‑based

    def power(alpha):
        w = np.power(idx, -alpha)
        return _normalize(w)

    def power_rev(alpha):
        w = np.power(rev_idx, -alpha)
        return _normalize(w)

    def exponential(beta):
        w = np.exp(-beta * ar)
        return _normalize(w)

    def power_exp(alpha, beta):
        w = np.power(idx, -alpha) * np.exp(-beta * ar)
        return _normalize(w)

    def linear():
        w = np.linspace(1.0, 0.0, n, dtype=np.float64)
        return _normalize(w)

    # ------------------------------------------------------------------
    # 3. Assemble initial catalogue and pick the best by fast scoring
    # ------------------------------------------------------------------
    candidates = []

    if parent is not None:
        candidates.append(_normalize(parent))

    # power‑law family
    for a in np.linspace(0.10, 0.90, 9):
        candidates.append(power(a))
        candidates.append(power_rev(a))

    # exponential family
    for b in np.linspace(1e-4, 5e-3, 7):
        candidates.append(exponential(b))

    # mixed power‑exponential family (coarse grid)
    for a in np.linspace(0.20, 0.70, 6):
        for b in np.linspace(0.0, 4e-3, 5):
            candidates.append(power_exp(a, b))

    # simple linear ramp
    candidates.append(linear())

    best = None
    best_score = np.inf
    for cand in candidates:
        sc = _fast_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 4. Stochastic local search (hill‑climb + occasional uphill steps)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    # current point for the Metropolis‑style walk
    cur = best.copy()
    cur_score = best_score

    max_iters = max(2000, n)                 # enough iterations for large n
    sigma_block = 0.07                       # noise on a local block
    sigma_global = 0.12                      # noise on the whole vector
    block_max = max(2, n // 200)             # maximal block size

    target_score = 1.50274349

    for it in range(max_iters):
        # temperature linearly decays to zero
        T = 1.0 - it / max_iters

        # --------------------------------------------------------------
        # generate a proposal
        # --------------------------------------------------------------
        cand = cur.copy()
        r = rng.random()
        if r < 0.60:                           # local multiplicative perturbation
            width = rng.integers(1, block_max + 1)
            start = rng.integers(0, n - width + 1)
            noise = np.exp(rng.normal(0.0, sigma_block, size=width))
            cand[start:start + width] *= noise
        elif r < 0.85:                         # global multiplicative perturbation
            noise = np.exp(rng.normal(0.0, sigma_global, size=n))
            cand *= noise
        else:                                  # tiny mass transfer between two positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                eps = rng.random() * 0.04      # up to 4 % of mass at i
                transfer = eps * cand[i]
                cand[i] -= transfer
                cand[j] += transfer

        # renormalise (preserves the scale‑invariant objective)
        cand = _normalize(cand)

        # --------------------------------------------------------------
        # evaluate proposal
        # --------------------------------------------------------------
        sc = _fast_score(cand)

        # Metropolis acceptance (allows occasional uphill moves)
        accept = False
        if sc < cur_score:

        else:

            if T > 0 and rng.random() < np.exp((cur_score - sc) / (T + 1e-12)):

    # ------------------------------------------------------------------
    # 5. Return the best candidate as a plain Python list
    # ------------------------------------------------------------------
    # Guarantee finite numbers and clip to the allowed range (0 … 1000)
    out = np.clip(best, 0.0, 1000.0).astype(np.float64)
    return [float(v) for v in out]
