    n = v.shape[0]
    L = 1 << ((2*n-1).bit_length())
    V = np.fft.rfft(v, L)
    conv = np.fft.irfft(V*V, L)
    max_b = conv[:2*n-1].max()
    return max_b

def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale vector to have L1 sum = 1 (if possible)."""
    s = v.sum()
    if s == 0.0:
        return v
    return v / s


def _fast_score(v: np.ndarray, n: int, L: int) -> float:
    """Score using the same formula as evaluate_sequence, but via FFT."""
    V = np.fft.rfft(v, L)

    """Move a random fraction of mass from one entry to another."""
    n = v.size

    out = v.copy()
    out[i] -= amount
    out[j] += amount
