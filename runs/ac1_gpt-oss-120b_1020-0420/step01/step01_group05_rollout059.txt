# Fast evaluation of the AC1 upper bound using FFT.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1                     # length of full convolution
    nfft = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

            or parent.size == 0
            or not np.isfinite(parent).all()
        ):
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    parent = np.clip(parent, 0.0, 1000.0)

    # ---- start from a few analytic families -------------------------------
    best_seq = np.ones(n, dtype=np.float64)            # uniform = bound 2.0
    best_score = _fast_score(best_seq)

    # parent candidate
    if np.sum(parent) >= 0.01:
        s = _fast_score(parent)
        if s < best_score:
            best_seq = parent.copy()
            best_score = s

    i = np.arange(n, dtype=np.float64)

    # 1) Inverse‑square‑root family (a_i = 1/sqrt(i+off))
    offsets = np.linspace(0.01, 10.0, 40)               # finer search
    for off in offsets:
        a = 1.0 / np.sqrt(i + off)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = 1.0 / np.sqrt((n - 1 - i) + off)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # 2) Power‑law family (a_i = (i+1)^{-p})
    powers = np.linspace(0.35, 0.80, 20)
    for p in powers:
        a = (i + 1) ** (-p)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = (n - i) ** (-p)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # 3) Exponential decay (a_i = exp(-τ·i))
    taus = np.linspace(0.0005, 0.012, 15)
    for tau in taus:
        a = np.exp(-tau * i)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = np.exp(-tau * (n - 1 - i))
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ---- random multiplicative jitter (broad exploration) ---------------
    for _ in range(300):
        sigma = 0.1 * (0.95 ** _)          # gradually shrinking variance
        pert = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_seq * pert
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = _fast_score(cand)
        if s < best_score:
            best_seq, best_score = cand, s

    # ---- targeted mass‑transfer refinement -------------------------------
    # The idea: locate the term that dominates the maximal convolution value
    # and move a fraction of its mass to an index that does not participate
    # in that convolution, hoping to lower the peak.
    max_iters = 2000
    fractions = (0.3, 0.2, 0.1, 0.05, 0.02, 0.01)

    for _ in range(max_iters):
        conv = _conv_full(best_seq)
        k = int(np.argmax(conv))          # shift where max occurs
        # overlapping region for this shift
        start = max(0, k - n + 1)
        end = min(k, n - 1)

        # if the whole vector overlaps (central peak), skip targeted move
        if start == 0 and end == n - 1:
            # occasional random jitter as a fallback
            if rng.random() < 0.01:
                sigma = 0.03
                pert = np.exp(rng.normal(0.0, sigma, size=n))
                cand = best_seq * pert
                np.clip(cand, 0.0, 1000.0, out=cand)
                s = _fast_score(cand)
                if s < best_score:
                    best_seq, best_score = cand, s
            continue

        # products contributing to the current maximum
        idxs = np.arange(start, end + 1)
        partners = k - idxs
        products = best_seq[idxs] * best_seq[partners]
        i_max = int(idxs[np.argmax(products)])   # index with biggest term

        # pick a destination outside the overlapping region
        while True:
            j = rng.integers(n)
            if j < start or j > end:
                break

        improved = False
        for frac in fractions:
            delta = best_seq[i_max] * frac
            if delta <= 0.0:
                continue
            cand = best_seq.copy()
            cand[i_max] -= delta
            cand[j] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = np.sum(cand)
            if s < 0.01:
                continue
            score = _fast_score(cand)
            if score < best_score:
                best_seq, best_score = cand, score
                improved = True
                break
        if not improved:
            # rare chance to try a larger random shuffle
            if rng.random() < 0.005:
                perm = rng.permutation(n)
                cand = best_seq[perm]
                s = _fast_score(cand)
                if s < best_score:
                    best_seq, best_score = cand, s

        # early exit if we already beat the target bound
        if best_score < 1.5028:
            break

    # ----------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    # ----------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
