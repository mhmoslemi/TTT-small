import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2: C_i = (2i choose i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _make_block_candidate(arr, rng):
    """Scale a random contiguous block (anywhere) by a log‑normal factor."""
    n = arr.size
    min_len = max(1, n // 200)      # ≈0.5 % of n
    max_len = max(2, n // 10)       # ≈10 % of n
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.30)))   # ≈ log‑normal perturbation
    arr[start:start + width] *= factor
    np.clip(arr, 0.0, 1000.0, out=arr)
    return arr


def _make_head_block_candidate(arr, rng):
    """Scale a random block located in the first quarter of the vector."""
    n = arr.size
    quarter = max(1, n // 4)
    min_len = max(1, n // 200)
    max_len = max(2, n // 10)
    width = int(rng.integers(min_len, max_len + 1))
    # ensure the whole block lies inside [0, quarter)
    start_limit = max(0, quarter - width)
    start = int(rng.integers(0, start_limit + 1))
    factor = float(np.exp(rng.normal(0.0, 0.30)))
    arr[start:start + width] *= factor
    np.clip(arr, 0.0, 1000.0, out=arr)
    return arr


def _gradient_step(arr, lr, rng):
    """
    One gradient‑like move: increase entries whose partner in the
    max‑convolution term is small and decrease the others.
    """
    # current convolution data
    conv = np.convolve(arr, arr)                     # full convolution
    idx_max = int(conv.argmax())
    max_val = float(conv[idx_max])
    sum_a = float(arr.sum())

    # partner values a_{idx_max-i}
    i = np.arange(arr.size)
    j = idx_max - i
    mask = (j >= 0) & (j < arr.size)
    partner = np.zeros_like(arr)
    partner[mask] = arr[j[mask]]

    # direction (max_val - partner * sum_a) is positive where partner is small
    delta = max_val - partner * sum_a
    arr += lr * delta
    np.clip(arr, 0.0, 1000.0, out=arr)
    return arr

    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The algorithm mixes deterministic gradient‑style updates with a suite of
    cheap random perturbations, keeping the best candidate seen.
    """
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).shape[0])
        except Exception:
            n = 256
    else:
        n = 256

    # ----- initial vector -----------------------------------------------
    # start from the provided parent if it exists, otherwise from the analytic series
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        base = _coefficients_binom_half(n)

    base = np.clip(base, 0.0, 1000.0)
    best = base.copy()
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = np.inf

    # Keep a reference fallback (in case the search degrades)
    fallback = best.copy()
    fallback_score = best_score

    target = 1.50274349          # early‑exit threshold
    lr = 0.001                    # learning‑rate for gradient step

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        if best_score <= target:
            break

        # pick a mutation type
        r = rng.random()
        cand = best.copy()

        if r < 0.30:                                 # gradient‑style move
            cand = _gradient_step(cand, lr, rng)

        elif r < 0.55:                               # random block scaling (anywhere)
            cand = _make_block_candidate(cand, rng)

        elif r < 0.70:                               # head‑block scaling (first quarter)
            cand = _make_head_block_candidate(cand, rng)

        elif r < 0.80:                               # swap two entries
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        elif r < 0.90:                               # small positive bump
            idx = rng.integers(0, n)
            bump = max(0.0, rng.normal(0.0, 0.05))
            cand[idx] += bump
            np.clip(cand, 0.0, 1000.0, out=cand)

        else:                                        # full shuffle (pure permutation)
            rng.shuffle(cand)

        except Exception:

            fallback_score = score

    # if the search never improved, return the fallback (always valid)
    if best_score == np.inf:
        best = fallback
        best_score = fallback_score

    return [float(x) for x in best]
