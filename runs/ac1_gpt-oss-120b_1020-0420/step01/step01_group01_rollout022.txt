    best_score = 2.0 * n * max_b / (total * total)
    best_f = f.copy()

    # Precompute sorted indices for selection each iteration (maybe recompute after each improvement)
    steps = kwargs.get("num_steps", num_steps)
    # Use top and bottom percent sizes
    top_percent = 0.05  # top 5%
    bottom_percent = 0.05
    top_k = max(1, int(n * top_percent))

    no_improve = 0

        # recompute sorted indices each iteration to reflect changes

        if it % 100 == 0:

import numpy as np

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0*(i+1)*(i+1))
    return a

def _conv_max(arr, size):
    # arr: 1d np.ndarray
    X = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(X * X, n=size)
    # full convolution length = 2*len(arr)-1
    L = 2*arr.shape[0] - 1
    return conv[:L].max()

def _score_from_max(n, max_b, total):
    if total < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (total * total))

    new_tail = tail * g

import numpy as np

# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Convolution (full) peak using FFT – O(n log n) and memory O(n)
# ------------------------------------------------------------
def _conv_max(arr: np.ndarray, size: int) -> float:
    X = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(X * X, n=size)
    L = 2 * arr.shape[0] - 1
    return float(conv[:L].max())


# ------------------------------------------------------------
# Score from known quantities (the same formula the evaluator uses)
# ------------------------------------------------------------
def _score_from_max(n: int, max_b: float, total: float) -> float:
    if total < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (total * total))


# ------------------------------------------------------------
# Deterministic reallocation: shrink a prefix by factor `fac`
# and uniformly inflate the remaining tail so that the total mass
# stays unchanged.
# ------------------------------------------------------------
def _shrink_prefix(seq: np.ndarray, ratio: float, fac: float) -> np.ndarray | None:
    n = seq.shape[0]
    k = max(1, int(n * ratio))
    early = seq[:k]
    tail = seq[k:]
    early_sum = early.sum()
    tail_sum = tail.sum()
    if tail_sum == 0.0:
        return None
    new_early = early * fac
    delta = early_sum - new_early.sum()                # mass taken from the prefix
    g = 1.0 + delta / tail_sum                         # uniform inflation factor
    new_tail = tail * g
    cand = np.concatenate([new_early, new_tail])
    return np.clip(cand, 0.0, 1000.0)
