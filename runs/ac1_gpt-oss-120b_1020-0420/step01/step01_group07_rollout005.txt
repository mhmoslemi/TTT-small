import numpy as np

# ------------------------------------------------------------
# Helper: ordered‑pair counts for a binary mask (sparse subset)
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices (0 … n‑1) where the coefficient is 1
    returns  : np.ndarray of length 2*n‑1 with ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    sel = list(selected)
    L = len(sel)
    for a in range(L):
        i = sel[a]
        cnt[2 * i] += 1                       # (i,i)
        for b in range(a + 1, L):
            j = sel[b]
            s = i + j
            cnt[s] += 2                       # (i,j) and (j,i)
    return cnt


# ------------------------------------------------------------
# Incremental update utilities (swap one index for another)
# ------------------------------------------------------------
def _apply_swap(selected, cnt, rem, add):
    """Remove index `rem`, add index `add`, update cnt in‑place."""
    # remove contributions of `rem`
    cnt[2 * rem] -= 1
    for j in selected:
        if j != rem:
            cnt[rem + j] -= 2
    selected.remove(rem)

    # add contributions of `add`
    cnt[2 * add] += 1
    for j in selected:
        cnt[add + j] += 2
    selected.add(add)


def _revert_swap(selected, cnt, rem, add):
    """Undo the swap performed by _apply_swap."""
    # remove contributions of `add`
    cnt[2 * add] -= 1
    for j in selected:
        if j != add:
            cnt[add + j] -= 2
    selected.remove(add)

    # restore contributions of `rem`
    cnt[2 * rem] += 1
    for j in selected:
        cnt[rem + j] += 2
    selected.add(rem)
