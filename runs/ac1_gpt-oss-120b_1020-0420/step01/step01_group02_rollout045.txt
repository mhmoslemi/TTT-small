    # Also evaluate baseline half_power_series if not same length? we may compute.
    # Not needed.

    # Parameters
    sigma = 0.2  # lognormal sigma factor
    smoothing_prob = 0.3
    block_prob = 0.5
    no_improve_counter = 0

    for iter_idx in range(max_iters):
        cand = best.copy()
        # Choose a perturbation type randomly
        r = rng.random()
        if r < block_prob:
            # block multiplicative perturbation similar to previous
            max_width = max(1, n // 20)
            width = int(rng.integers(1, max_width + 1))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))  # lognormal
            cand[start_idx:start_idx+width] *= factor
        elif r < block_prob + smoothing_prob:
            # smoothing operation
            # simple moving average width 3
            padded = np.pad(cand, (1,1), mode='edge')
            smoothed = (padded[:-2] + padded[1:-1] + padded[2:]) / 3.0
            # mix with original
            alpha = rng.random()
            cand = alpha * cand + (1-alpha) * smoothed
        else:
            # per-element multiplicative noise
            factors = np.exp(rng.normal(0.0, sigma, size=n))
            cand *= factors

        # clip to allowed range
        np.clip(cand, 0.0, 1000.0, out=cand)

        # compute score
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # decrease sigma gradually
        sigma *= 0.9995  # small decay

        if no_improve_counter >= 200:
            # early stop if no improvement for many iter
            break

    return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# Helper: generate the truncated series of (1 - z)^(-1/2)
# a_0 = 1, a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

# Simple moving‑average smoothing (window size=3)
# ------------------------------------------------------------
def smooth(arr: np.ndarray) -> np.ndarray:
    # Pad with edge values to keep length unchanged
    padded = np.pad(arr, (1, 1), mode='edge')
    return (padded[:-2] + padded[1:-1] + padded[2:]) / 3.0


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iters: int = 1500, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size and initial sequence
    # --------------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        n = int(height_sequence_1.size)
        current = height_sequence_1.astype(np.float64).copy()
    else:
        n = 256
        current = generate_half_power_series(n)

    # Safety clipping (the evaluator also clips, but keep numbers bounded)
    np.clip(current, 0.0, 1000.0, out=current)

    # --------------------------------------------------------
    # Keep the best sequence seen so far
    # --------------------------------------------------------
    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------
    # Search parameters
    # --------------------------------------------------------
    sigma = 0.3                     # log‑normal mutation scale
    block_prob = 0.40               # probability of a block multiplicative change
    smooth_prob = 0.30              # probability of applying smoothing
    swap_prob = 0.15                # probability of swapping two entries
    # remaining probability => per‑element multiplicative noise
    no_improve = 0                  # counter for early stopping

    for it in range(max_iters):
        cand = best.copy()
        r = rng.random()

        if r < block_prob:
            # ---- random contiguous block scaling ----
            max_width = max(1, n // 20)
            width = int(rng.integers(1, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor

        elif r < block_prob + smooth_prob:
            # ---- smoothing operation ----
            sm = smooth(cand)
            # mix original and smoothed versions (mix factor uniform)
            alpha = rng.random()
            cand = alpha * cand + (1.0 - alpha) * sm

        elif r < block_prob + smooth_prob + swap_prob:
            # ---- swap two random positions ----
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            # ensure distinct indices
            while j == i:
                j = int(rng.integers(0, n))
            cand[i], cand[j] = cand[j], cand[i]

        else:
            # ---- per‑element log‑normal perturbation ----
            factors = np.exp(rng.normal(0.0, sigma, size=n))
            cand *= factors

        # Clip to admissible range before evaluation
        np.clip(cand, 0.0, 1000.0, out=cand)

        # ----------------------------------------------------
        # Evaluate candidate
        # ----------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        # Gradually reduce mutation strength
        sigma *= 0.9995

        # Early termination if stagnating
        if no_improve >= 200:
            break

    # Final safety clip
    np.clip(best, 0.0, 1000.0, out=best)

    # Return a plain Python list of floats
    return [float(x) for x in best]
