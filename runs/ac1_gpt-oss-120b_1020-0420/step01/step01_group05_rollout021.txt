import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------
# Generate a candidate sequence from a three‑parameter family.
# ----------------------------------------------------------
def _gen_seq(i: np.ndarray, p: float, off: float, beta: float,
             n: int, rev: bool = False) -> np.ndarray:
    """
    a_i = (idx + off)^{-p} * exp(-beta * idx)
    where idx = i (or reversed) and i = np.arange(n).
    The result is clipped to [0,1000].
    """
    if rev:
        idx = n - 1 - i
    else:
        idx = i
    base = idx + off
    # ensure positivity for the power; 0**0 is treated as 1 by NumPy.
    base = np.where(base <= 0, np.finfo(float).tiny, base)
    a = np.power(base, -p) * np.exp(-beta * idx)
    return np.clip(a, 0.0, 1000.0)
