    best = seq.copy()

    # Hyper-parameters

    for it in range(max_iters):

import numpy as np

# ------------------------------------------------------------
# Scoring – identical to the provided evaluator (faster locally)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """2*n*max(conv(seq,seq))/sum(seq)^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full linear convolution (FFT for large n)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like base sequence a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Helper: partner values for the lag that attains the max convolution
# ------------------------------------------------------------
def _partner(arr: np.ndarray, lag: int) -> np.ndarray:
    """Array p where p[i] = arr[lag-i] if index in range, else 0."""
    n = arr.size
    p = np.zeros_like(arr)
    if lag < n:
        # indices 0 .. lag map to reversed slice
        p[: lag + 1] = arr[lag::-1]
    else:
        start = lag - (n - 1)                # first index that has a partner
        p[start:] = arr[: n - start][::-1]
    return p

    else:

            if sc < best_score:
                best = cand
                best_score = sc
                # a good flattening move – gently increase step size
                step_eta = min(step_eta * 1.02, 0.20)
                continue

        # ---- occasional random perturbation to escape shallow plateaus ----------
        if it % 73 == 0:                         # pseudo‑random schedule
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src != dst and best[src] > 1e-12:
                delta = best[src] * rng.uniform(0.01, 0.10)
                cand = best.copy()
                cand[src] -= delta
                cand[dst] += delta
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc
                    step_eta = min(step_eta * 1.05, 0.20)

        # optional early exit if step size becomes negligible
        if step_eta < 1e-5:
            break

    # --------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # --------------------------------------------------------
    return [float(v) for v in best]
