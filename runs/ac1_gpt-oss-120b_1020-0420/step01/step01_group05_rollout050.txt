    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


# ---------------------------------------------------------
# Full self‑convolution via FFT (matrix‑free).
# ---------------------------------------------------------
def _conv_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=nfft)
    return np.fft.irfft(fa * fa, n=nfft)[: 2 * a.shape[0] - 1]


# ---------------------------------------------------------
# Gradient of the AC1 bound B(a) = 2 n max(c) / sum(a)^2.
# ---------------------------------------------------------
def _grad_B(a: np.ndarray, conv: np.ndarray, M: float, S: float) -> np.ndarray:
    n = a.shape[0]
    # indices where the convolution attains its maximum (within tolerance)
    max_inds = np.where(conv >= M - 1e-12)[0]
    dM = np.zeros_like(a)
    # use only a few maximal positions to keep work linear
    for k in max_inds[:5]:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        p = np.arange(start, end + 1, dtype=np.int64)
        dM[p] += 2.0 * a[k - p]
    factor1 = 2.0 * n / (S * S)
    factor2 = 4.0 * n * M / (S * S * S)
    return factor1 * dM - factor2


# ---------------------------------------------------------
# Local refinement by gradient descent (matrix‑free).
# ---------------------------------------------------------
def _refine_sequence(seq: np.ndarray,
                     max_iter: int = 250,
                     lr: float = 0.2,
                     decay: float = 0.95) -> tuple:
    a = seq.astype(np.float64).copy()
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    target_sum = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = _fast_score(a)

    for _ in range(max_iter):
        conv = _conv_fft(a, nfft)
        M = conv.max()
        S = a.sum()
        if S < 0.01:
            break
        grad = _grad_B(a, conv, M, S)
        a -= lr * grad
        a = np.clip(a, 0.0, 1000.0)

        # renormalise the sum (scale invariance of the objective)
        S = a.sum()
        if S > 0.0:
            a *= target_sum / S
            a = np.clip(a, 0.0, 1000.0)

        score = _fast_score(a)
        if score < best_score:
            best = a.copy()
            best_score = score
        lr *= decay

    return best, best_score


# ---------------------------------------------------------
# Main entry point.
# ---------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # -------------------------------------------------
    # Parent fallback (height_sequence_1 if available).
    # -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)  # generic fallback

    n = parent.shape[0]
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 0.01:
        parent = np.ones(n, dtype=np.float64)

    best_seq = parent.copy()
    best_score = _fast_score(best_seq)

    # -------------------------------------------------
    # Simple uniform candidate (classic bound 2.0).
    # -------------------------------------------------
    uniform = np.ones(n, dtype=np.float64)
    s = _fast_score(uniform)
    if s < best_score:
        best_seq, best_score = uniform, s

    # -------------------------------------------------
    # Analytic families (inverse‑square‑root, power laws).
    # -------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    # 1) Inverse‑square‑root a_i = 1 / sqrt(i + offset)
    for off in np.linspace(0.1, 4.0, 20):
        a = 1.0 / np.sqrt(i + off)
        a = np.clip(a, 0.0, 1000.0)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s

        a_rev = 1.0 / np.sqrt((n - 1 - i) + off)
        a_rev = np.clip(a_rev, 0.0, 1000.0)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # 2) Power‑law a_i = (i+1)^{-p}
    for p in np.linspace(0.35, 0.65, 9):
        a = (i + 1) ** (-p)
        a = np.clip(a, 0.0, 1000.0)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s

        a_rev = (n - i) ** (-p)
        a_rev = np.clip(a_rev, 0.0, 1000.0)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # -------------------------------------------------
    # Small random multiplicative tweaks around the best.
    # -------------------------------------------------
    for _ in range(100):
        pert = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = best_seq * pert
        cand = np.clip(cand, 0.0, 1000.0)
        s = _fast_score(cand)
        if s < best_score:
            best_seq, best_score = cand, s

    # -------------------------------------------------
    # Gradient‑descent refinement (matrix‑free).
    # -------------------------------------------------
    refined, refined_score = _refine_sequence(best_seq,
                                              max_iter=250,
                                              lr=0.2,
                                              decay=0.95)
    if refined_score < best_score:
        best_seq = refined
        best_score = refined_score

    # Final safety clip and conversion to plain Python list.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq.tolist()]
