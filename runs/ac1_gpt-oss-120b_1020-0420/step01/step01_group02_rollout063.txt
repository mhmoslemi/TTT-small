def propose_candidate(seed=42, **kwargs):
    import numpy as np
    # length
    n = 256
    if 'height_sequence_1' in globals():
        seq = height_sequence_1
        if isinstance(seq, np.ndarray):
            if seq.ndim == 1 and seq.size > 0:
                n = int(seq.size)
    # base half-power series
    base = generate_half_power_series(n)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # parameter search
    exponents = [0.5, 0.55, 0.58, 0.6, 0.62]
    max_offset = min(500, n//4)   # e.g., up to 500 or n/4
    step_offset = max(1, max_offset // 20)
    for p in exponents:
        for offset in range(0, max_offset+1, step_offset):
            idx = np.arange(n, dtype=np.float64)
            a = (idx + offset + 1) ** (-p)   # +1 to avoid zero
            np.clip(a, 0.0, 1000.0, out=a)
            # no scaling needed
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best = a
                best_score = score
    # additional random block refinement (as before)
    rng = np.random.default_rng(seed)
    for _ in range(300):

import numpy as np

# ------------------------------------------------------------
# Helper: generate the truncated series of (1 - z)^(-1/2)
#   a_0 = 1,  a_{i} = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return (1‑z)^(-1/2) coefficients of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Small random block perturbation (keeps O(n) memory)
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block of ``base`` by a log‑normal factor."""
    n = base.size
    if n == 1:
        return base.copy()
    # block width between 1 and roughly n/20
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    # log‑normal factor (allows both increase and decrease)
    factor = float(np.exp(rng.normal(0.0, 0.2)))   # larger variance than before
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # Determine length of the problem (use provided parent if possible)
    n = 256
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        n = int(height_sequence_1.size)

    # Baseline: half‑power series (the prior construction)
    base = generate_half_power_series(n)
    np.clip(base, 0.0, 1000.0, out=base)

    # Start with the best candidate seen so far
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Consider the provided parent sequence as a possible fallback
    if "height_sequence_1" in globals():
        parent = height_sequence_1
        if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size == n:
            score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(score) and score < best_score:
                best = parent.astype(np.float64)
                best_score = score

    # ------------------------------------------------------------
    # Structured two‑copy construction:
    #   a = f + λ * shift(f, offset)
    # This keeps the maximal convolution under control while
    # increasing the total sum, yielding a lower AC1 value.
    # ------------------------------------------------------------
    lam_vals = [1.5, 2.0, 1.0 + np.sqrt(2.0), 2.8]  # λ candidates
    offset_fracs = [0.25, 0.33, 0.5, 0.66, 0.75]     # offset as a fraction of n

        for frac in offset_fracs:
            offset = int(frac * n)
            if offset <= 0 or offset >= n:
                continue
            cand = base.copy()

            cand[offset:] += lam * base[:n - offset]
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:

                best_score = score

        if np.isfinite(score) and score < best_score:

            best_score = score
