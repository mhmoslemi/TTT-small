def _grad_B(a, M, k):
    n = a.shape[0]

    # compute g vector
    idx = np.arange(n)

import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the objective.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = float(conv.max())
    S = float(a.sum())
    if S < 0.01:
        return float("inf")
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------
# Full convolution (FFT) – also returns the index of the max.
# ----------------------------------------------------------
def _conv_max_and_index(a: np.ndarray):
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    k = int(np.argmax(conv))
    return float(conv[k]), k


# ----------------------------------------------------------
# Gradient of the bound B(a) = 2 n max(conv) / (sum a)^2.
# ----------------------------------------------------------
def _grad_B(a: np.ndarray, M: float, k: int):
    n = a.shape[0]
    S = a.sum()
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    if mask.any():
        g[mask] = a[j[mask]]
    double_mask = (2 * idx == k)
    if double_mask.any():
        g[double_mask] += a[double_mask]
    # grad = dB/da
    grad = (2.0 * n / (S ** 3)) * (g * S - 2.0 * M)
    return grad
