import numpy as np

# ----------------------------------------------------------------------
# Scoring – identical to the evaluator used by the harness
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the AC1 bound 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution, FFT based for long vectors
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Catalan‑like seed (C(2i,i)/4^i) – the classic starting point
# ----------------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (if SciPy is present)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
except Exception:                 # SciPy may be unavailable – we fall back gracefully
    linprog = None


def _lp_candidate(seq: np.ndarray, max_val: float, rng: np.random.Generator,
                  max_constraints: int = 10) -> np.ndarray | None:
    """Return a vector g that maximises sum(g) under   (seq * g)[k] ≤ max_val .
    Only a small set of the most active shift indices is used to keep the LP tiny."""
    n = seq.size
    conv = np.convolve(seq, seq)
    # shifts where the current bound is active (within 1 % of the maximum)
    active = np.where(conv >= max_val * 0.99)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)], dtype=int)
    if active.size > max_constraints:
        active = rng.choice(active, size=max_constraints, replace=False)

    K = active.size
    A = np.zeros((K, n), dtype=np.float64)
    for r, k in enumerate(active):
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        js = np.arange(start, end + 1, dtype=int)
        A[r, js] = seq[k - js]
    b = np.full(K, max_val, dtype=np.float64)

    # simple per‑variable caps: each entry appears multiplied by seq[0] in some constraint
    max_a = seq.max()
    ub = max_val / max_a if max_a > 0 else max_val * 10.0
    bounds = [(0.0, ub) for _ in range(n)]

    c = -np.ones(n, dtype=np.float64)      # maximise sum(g)  ⇔  minimise –sum(g)
    if linprog is None:
        return None
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    return None if not res.success else res.x
