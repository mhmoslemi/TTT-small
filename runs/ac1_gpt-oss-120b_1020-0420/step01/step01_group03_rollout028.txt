import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the classic root‑of‑constant sequence (fallback)
def _root_constant_seq(length: int) -> np.ndarray:
    """
    a_i = C(2i,i) / 4**i   (i = 0 .. length‑1)
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


# ----------------------------------------------------------------------
# Convolution (full) using an FFT – O(n log n) and linear memory
def _full_self_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    c = np.fft.irfft(fa * fa, n=size)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
# Compute the AC1 score (fast, uses the same formula as the evaluator)
def _ac1_score(a: np.ndarray, conv: np.ndarray = None) -> float:
    n = a.shape[0]
    if conv is None:
        conv = _full_self_convolution(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 score
    improves on the classic root‑constant sequence.  The method starts from
    the provided parent `height_sequence_1` (if present) and performs a
    simple hill‑climbing that moves a small amount of mass from early
    positions to later positions while never increasing the maximal
    convolution value.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise sequence ------------------------------------------------
    try:
        # parent is available as a NumPy array at runtime
        seq = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:

        n_fallback = 256
        seq = _root_constant_seq(n_fallback)

    # Ensure validity for the evaluator (clip, non‑negative)
    np.clip(seq, 0.0, 1000.0, out=seq)

    if n == 0:

        seq = np.full(1, 1.0, dtype=np.float64)
        n = 1

    # ------------------------------------------------------------------
    # Pre‑compute the convolution and the current best score
    conv = _full_self_convolution(seq)
    best_score = _ac1_score(seq, conv)
    best_seq = seq.copy()
    best_conv = conv.copy()

    # ------------------------------------------------------------------
    # Parameters for the hill‑climbing
    early_len = max(1, n // 6)          # first sixth of the array
    late_start = n - early_len          # last sixth of the array
    max_iter = 12000                    # number of attempted moves
    # The step size is a small fraction of the source coefficient.
    # It will be drawn uniformly from [1 %, 5 %] of a[i].
    min_frac, max_frac = 0.01, 0.05

    # ------------------------------------------------------------------
    # Main optimisation loop (linear‑time updates)
    for _ in range(max_iter):
        i = rng.integers(0, early_len)           # early index (to decrease)
        j = rng.integers(late_start, n)          # late  index (to increase)
        if seq[i] <= 0.0:
            continue

        # choose a modest transfer amount
        frac = rng.random() * (max_frac - min_frac) + min_frac
        delta = seq[i] * frac
        if delta <= 0.0:
            continue

        # ----- compute the updated convolution in O(n) -----
        # copy the old convolution (length 2n‑1)
        new_conv = best_conv.copy()

        # contribution of 2*delta*(shift_j - shift_i)

        new_conv[j : j + n] += 2.0 * delta * seq

        d2 = delta * delta
