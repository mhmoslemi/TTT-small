import numpy as np

# --------- convolution and score (FFT based, linear memory) ----------
def _conv_fft(vec: np.ndarray) -> np.ndarray:
    """Full convolution of vec with itself using FFT (linear memory)."""
    n = vec.size
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score_and_conv(vec: np.ndarray):
    """Return (score, full‑conv) for a non‑negative vector."""
    conv = _conv_fft(vec)
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf, conv
    score = 2.0 * vec.size * max_c / (s * s)
    return score, conv
