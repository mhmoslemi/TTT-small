import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                                 # convolution in freq. domain
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

    # 4) Combined power‑law & exponential
    for lam in np.linspace(0.0, 0.003, 6):
        for p in np.linspace(0.15, 0.55, 7):
            a = (i + 1.0) ** (-p) * np.exp(-lam * i)
            s = _fast_score(a)
            if s < best_score:
                best_seq, best_score = a, s
            a_rev = a[::-1]
            s = _fast_score(a_rev)
            if s < best_score:
                best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # Stochastic refinement: multiplicative jitter + occasional reverse mixing.
    # ------------------------------------------------------------------
    sigma = 0.12                     # starting jitter (≈12 % log‑scale)
    max_iter = 2500                 # total refinement steps
    for it in range(max_iter):
        # multiplicative jitter (log‑normal)
        pert = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_seq * pert
        cand = np.clip(cand, 0.0, 1000.0)
        s = _fast_score(cand)
        if s < best_score:
            best_seq, best_score = cand, s
            # shrink jitter a little after a success
            sigma *= 0.995
        # Occasionally try mixing with the reversed sequence.
        if (it + 1) % 71 == 0:
            rev = best_seq[::-1]
            t = rng.uniform(0.0, 0.3)          # convex combination weight
            cand = (1.0 - t) * best_seq + t * rev
            cand = np.clip(cand, 0.0, 1000.0)
            s = _fast_score(cand)
            if s < best_score:
                best_seq, best_score = cand, s
                sigma *= 0.995

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
