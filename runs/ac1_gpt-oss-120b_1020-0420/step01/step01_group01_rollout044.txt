import numpy as np

# ----------------------------------------------------------------------
# Helper: score identical to the evaluator (clipping, sum, convolution)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution
    m = np.max(conv)
    return float(2.0 * seq.size * m / (s * s))


# ----------------------------------------------------------------------
# Catalan‑like (central binomial) sequence – used when no parent is present
# ----------------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

def _prefix_reallocation(seq: np.ndarray) -> np.ndarray:
    """Try a few shrink‑and‑inflate transformations and keep the best."""
    best = seq.copy()
    best_score = _score(best)
    n = best.size
    prefix_ratios = [0.05, 0.10, 0.20]
    shrink_factors = np.linspace(0.5, 1.0, 11)

    for r in prefix_ratios:
        k = max(1, int(n * r))
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue

        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()
            if delta <= 0.0:
                continue
            g = 1.0 + delta / tail_sum
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                early = best[:k]
                tail = best[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()
    return best

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):
    """
    Return a list of non‑negative heights with a low AC1 bound.
    The function attempts a cheap deterministic reallocation followed by
    a gradient‑guided random hill‑climbing loop.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size – use the provided parent if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
            seq = parent.copy()
        else:
            n = 256
            seq = _catalan_seq(n)
    else:
        n = 256
        seq = _catalan_seq(n)

    seq = np.clip(seq, 0.0, 1000.0)

    # --------------------------------------------------------------
    # 1) Deterministic early‑prefix reshaping (fast, cheap)
    # --------------------------------------------------------------
    seq = _prefix_reallocation(seq)

    best_seq = seq.copy()
    best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Helper to recompute convolution and the index of its maximum
    # --------------------------------------------------------------
    def _conv_and_max(arr):
        c = np.convolve(arr, arr)
        return c, int(np.argmax(c))

    conv, max_idx = _conv_and_max(best_seq)

    # --------------------------------------------------------------
    # 2) Gradient‑guided random hill climbing
    # --------------------------------------------------------------
    steps = kwargs.get("num_steps", num_steps)
    n = best_seq.size
    last_improve = 0

    for step in range(steps):
        # Recompute convolution and max index occasionally (cheap vs.
        # per‑iteration expense)
        if step % 10 == 0:
            conv, max_idx = _conv_and_max(best_seq)

        # ----------------------------------------------------------
        # Build the partner vector for the current maximal index.
        # partner[i] = best_seq[max_idx - i] if in range, else 0.
        # ----------------------------------------------------------
        idxs = np.arange(n)
        partner = np.zeros_like(best_seq)
        valid = (max_idx - idxs >= 0) & (max_idx - idxs < n)
        partner[valid] = best_seq[max_idx - idxs[valid]]

        # Weighted importance of each position for the current peak.
        weighted = best_seq * partner
        wsum = weighted.sum()
        if wsum > 0:
            src = rng.choice(n, p=weighted / wsum)
        else:
            src = rng.integers(n)

        # Destination: pick an index with relatively small partner value.
        thresh = np.median(partner)
        low_idxs = np.where(partner <= thresh)[0]
        if low_idxs.size == 0:
            dst = rng.integers(n)
        else:
            dst = rng.choice(low_idxs)

        if src == dst:
            continue

        # Move a random fraction (1 %–20 %) of mass from src to dst.
        if best_seq[src] <= 0.0:
            continue
        delta = best_seq[src] * rng.uniform(0.01, 0.20)
        cand = best_seq.copy()
        cand[src] -= delta
        cand[dst] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc
            last_improve = step
            # Refresh convolution and max index immediately.
            conv, max_idx = _conv_and_max(best_seq)
            continue

        # ----------------------------------------------------------
        # Occasionally try a local smoothing move: take mass from a
        # local maximum and spread it to neighbours.
        # ----------------------------------------------------------
        if step % 7 == 0:
            i = int(np.argmax(best_seq))
            if best_seq[i] > 0.0:
                delta2 = best_seq[i] * rng.uniform(0.01, 0.10)
                cand2 = best_seq.copy()
                cand2[i] -= delta2
                share = delta2 / 2.0
                if i > 0:
                    cand2[i - 1] += share
                if i + 1 < n:
                    cand2[i + 1] += share
                np.clip(cand2, 0.0, 1000.0, out=cand2)
                sc2 = _score(cand2)
                if sc2 < best_score:
                    best_seq = cand2
                    best_score = sc2
                    last_improve = step
                    conv, max_idx = _conv_and_max(best_seq)
                    continue

        # Simple simulated‑annealing style occasional acceptance to escape stalls
        if rng.random() < 0.001:
            best_seq = cand
            conv, max_idx = _conv_and_max(best_seq)

        # Stop early if no improvement for a while.
        if step - last_improve > 1000:
            break

    # --------------------------------------------------------------
    # Final safety net: if everything failed, return a simple uniform vector.
    # --------------------------------------------------------------
    if best_score == np.inf or best_seq.size == 0:
        best_seq = np.full(n, 1.0, dtype=np.float64)

    # Return a plain Python list as required by the harness.
    return [float(v) for v in best_seq]
