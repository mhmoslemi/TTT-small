def max_convolution(seq):
    m = 1 << ((2*len(seq)-1).bit_length())
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)
    # conv length m, but valid part length: 2*n-1
    return conv[:2*len(seq)-1]
```

import numpy as np

# ------------------------------------------------------------
# Helper: fast convolution‑max using a single FFT (linear memory)
# ------------------------------------------------------------
def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Return full convolution seq*seq (real values) using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    fft = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft * fft, m)
    return conv[: 2 * n - 1]                # relevant part only


def _score(seq: np.ndarray) -> float:
    """Exact score used by the evaluator (2·n·max(conv)/sum²)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # use FFT‑based convolution – O(n log n), linear memory
    conv = _conv_full(seq)
    m = np.max(conv)                       # max of a*a
    n = seq.size
    return float(2.0 * n * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central‑binomial) sequence, the classic start
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 20000, **kwargs):
    """
    Return a list of non‑negative heights with a low AC1 bound.

    The routine starts from the Catalan sequence (which gives the
    classic bound π/2) and then performs a greedy “mass‑transfer”
    search: repeatedly move a small fraction of the largest‑contributing
    entry (as identified from the current maximal convolution index)
    toward the far end of the vector.  This lowers the maximal
    autocorrelation while keeping the total sum (hence the denominator)
    unchanged, driving the bound towards the theoretical optimum 1.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Problem size: use provided parent if available, else 256.
    # --------------------------------------------------------

    # --------------------------------------------------------
    # Initialise with Catalan (exactly gives max_b = 1).
    # --------------------------------------------------------
    best = _catalan_seq(n)
    best = np.clip(best, 0.0, 1000.0)
    best_score = _score(best)

    # --------------------------------------------------------
    # Main greedy loop – move mass from the entry that most
    # contributes to the current maximal convolution toward the
    # last index (which currently contributes almost nothing).
    # --------------------------------------------------------
    steps = kwargs.get("num_steps", num_steps)
    for it in range(steps):
        # full convolution and location of current maximum
        conv_full = _conv_full(best)
        max_val = np.max(conv_full)
        max_idx = int(np.argmax(conv_full))

        # compute products a[i] * a[max_idx-i] for all valid i
        i_arr = np.arange(n)
        j_arr = max_idx - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        if not np.any(mask):
            # fallback: random move (should virtually never happen)
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
        else:
            products = best[i_arr[mask]] * best[j_arr[mask]]
            src = int(i_arr[mask][np.argmax(products)])

            # destination: far end of the vector (different from src)
            dst = n - 1 if src != n - 1 else 0

            continue

        cand[dst] += delta

    # --------------------------------------------------------
