import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """
    Generate a_i = C(2i,i)/4**i for i = 0..length-1.
    This sequence satisfies (a*a)[k] == 1 for 0 <= k < length (up to round‑off).
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _score(arr: np.ndarray) -> float:
    """Convenient wrapper around the provided evaluator."""
    # evaluate_sequence is supplied by the harness
    return float(evaluate_sequence(arr.tolist()))

    # --------------------------------------------------------------------
    if not np.isfinite(best_val):
        best_seq = np.full(n, 1.0, dtype=np.float64)

    return best_seq.tolist()
