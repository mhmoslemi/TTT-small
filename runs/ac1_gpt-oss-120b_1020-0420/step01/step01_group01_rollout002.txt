import numpy as np

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0*(i+1)*(i+1))
    return a

def propose_candidate(seed=42, num_iters=10000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # Initialize sequence
    use_parent = False
    if 'height_sequence_1' in globals():
        seq = np.clip(parent, 0.0, 1000.0)
        if seq.sum() >= 0.01:
            use_parent = True
    if not use_parent:
        seq = _catalan_seq(n)
        seq = np.clip(seq, 0.0, 1000.0)

    sum_seq = float(seq.sum())
    # ensure sum decent
    if sum_seq < 0.01:
        # fallback to catalan
        seq = _catalan_seq(n)
        seq = np.clip(seq, 0.0, 1000.0)
        sum_seq = float(seq.sum())

    conv = np.convolve(seq, seq)
    max_conv = float(conv.max())
    best_score = 2.0 * n * max_conv / (sum_seq * sum_seq)
    best_seq = seq.copy()
    best_conv = conv.copy()

    # Precompute some indices for biased moves
    quarter = max(1, n // 4)
    half = max(1, n // 2)

    # number of iterations
    iters = int(kwargs.get('num_iters', num_iters))
    for step in range(iters):
        # pick source and destination indices
        if rng.random() < 0.7:
            src = rng.integers(0, quarter)  # early index
            dst = rng.integers(half, n)     # later index

        src_val = seq[src]

            continue
        # choose delta, up to 80% of src value, but ensure dst limit
        max_delta = src_val

import numpy as np

# ------------------------------------------------------------
# Catalan-like sequence (a_i = binom(2i,i)/4^i)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_iters: int = 10000, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n (use parent if present, otherwise 256)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = int(parent.size)
        else:
            n = 256
    else:
        n = 256

    # ------------------------------------------------------------------
    # Initialise sequence: parent (clipped) if feasible, else Catalan
    # ------------------------------------------------------------------
    use_parent = False
    if "height_sequence_1" in globals():
        seq = np.clip(parent, 0.0, 1000.0)
        if seq.sum() >= 0.01:
            use_parent = True
    if not use_parent:
        seq = _catalan_seq(n)
        seq = np.clip(seq, 0.0, 1000.0)

    sum_seq = float(seq.sum())
    if sum_seq < 0.01:                     # safety fallback
        seq = _catalan_seq(n)
        seq = np.clip(seq, 0.0, 1000.0)
        sum_seq = float(seq.sum())

    # ------------------------------------------------------------------
    # Full convolution and initial score
    # ------------------------------------------------------------------
    conv = np.convolve(seq, seq)           # length 2*n-1
    max_conv = float(conv.max())
    best_score = 2.0 * n * max_conv / (sum_seq * sum_seq)
    best_seq = seq.copy()
    best_conv = conv.copy()

    # ------------------------------------------------------------------
    # Helper constants for biased moves
    # ------------------------------------------------------------------
    quarter = max(1, n // 4)
    half = max(1, n // 2)

    # ------------------------------------------------------------------
    # Hill‑climbing with biased random mass transfers
    # ------------------------------------------------------------------
    total_iters = int(kwargs.get("num_iters", num_iters))
    for step in range(total_iters):
        # choose source and destination (biased towards moving mass forward)
        if rng.random() < 0.7:                         # 70 % biased moves
            src = rng.integers(0, quarter)            # early index
            dst = rng.integers(half, n)               # later index
        else:                                          # occasional fully random
            src = rng.integers(0, n)
            dst = rng.integers(0, n)

        if src == dst:
            continue

        src_val = seq[src]
        if src_val <= 0.0:
            continue

        # amount to move: between 1 % and 80 % of the source mass
        frac = rng.uniform(0.01, 0.80)
        delta = src_val * frac

        # respect upper bound 1000 on the destination
        if seq[dst] + delta > 1000.0:
            delta = 1000.0 - seq[dst]
            if delta <= 0.0:
                continue

        # ------------------------------------------------------------------
        # Build candidate sequence and updated convolution (matrix‑free)
        # ------------------------------------------------------------------
        cand_seq = seq.copy()
        cand_seq[src] -= delta
        cand_seq[dst] += delta

        cand_conv = conv.copy()

        # shift contributions using the *original* sequence (seq)
        cand_conv[src:src + n] -= 2.0 * delta * seq
        cand_conv[dst:dst + n] += 2.0 * delta * seq

        # point‑mass contributions from the d*d term
        idx_src = 2 * src
        idx_dst = 2 * dst
        idx_mid = src + dst
        cand_conv[idx_src] += delta * delta
        cand_conv[idx_dst] += delta * delta
        cand_conv[idx_mid] -= 2.0 * delta * delta

        # numerical safety: convolution must stay non‑negative
        cand_conv = np.maximum(cand_conv, 0.0)

        # ------------------------------------------------------------------
        # Evaluate candidate
        # ------------------------------------------------------------------
        new_max = float(cand_conv.max())
        new_score = 2.0 * n * new_max / (sum_seq * sum_seq)

        if new_score < best_score:          # accept only improvements
            best_score = new_score
            best_seq = cand_seq
            best_conv = cand_conv
            seq = cand_seq
            conv = cand_conv

        # occasional full recompute to curb drift (cheap, O(n log n))
        if (step + 1) % 2000 == 0:
            conv = np.convolve(seq, seq)
            conv = np.maximum(conv, 0.0)

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
