def propose_candidate(seed=42):
    import numpy as np

    # initialize
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        # clip to [0,1000] just in case
        a = np.clip(a, 0.0, 1000.0)
    else:
        # fallback uniform
        n = 256
        a = np.ones(n, dtype=np.float64) / n # sum =1
    # Ensure sum>0
    if a.sum() < 0.01:
        a = np.ones_like(a) * (0.01 / a.size)

    # normalize to sum=1 (optional scaling invariant)
    a = a / a.sum()

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Helper evaluation
    def evaluate(arr):
        return float(evaluate_sequence(arr.tolist()))

    # Families search (powers, exponent 0.5 with offset)
    n = a.size
    idx = np.arange(n, dtype=np.float64)

    # 1. power-law exponent near 0.5
    alphas = np.linspace(0.3,0.7,41) # 0.01 step approx
    for alpha in alphas:
        w = np.power(idx + 1.0, -alpha)
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w = w / w_sum
        sc = evaluate(w)
        if np.isfinite(sc) and sc < best_score:
            best = w.copy()
            best_score = sc

    # 2. offset power law
    offsets = [0.0, 0.1, 0.5, 1.0, 2.0, 5.0, 10.0]
    for off in offsets:
        w = np.power(idx + 1.0 + off, -0.5)
        w_sum = w.sum()
        w = w / w_sum
        sc = evaluate(w)
        if np.isfinite(sc) and sc < best_score:
            best = w.copy()
            best_score = sc

    # 3. refined random multiplicative perturbations
    rng = np.random.default_rng(seed)
    cand = best.copy()
    for it in range(500):
        # pick random block
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 200))
        stop = min(n, start + width)
        # perturb multiplicatively
        noise = np.exp(rng.normal(0.0, 0.1, size=stop-start))
        cand[start:stop] *= noise
        # ensure nonnegativity
        cand = np.clip(cand, 0.0, None)
        # renormalize to sum 1
        total = cand.sum()
        if total == 0:
            continue
        cand = cand / total
        sc = evaluate(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc
            # update cand to best for further modifications
            cand = best.copy()
        # occasional greedy peak reduction step
        if it % 50 == 0:
            # compute convolution using fft for speed
            # but evaluate_sequence uses full convolution; we compute approximate
            b = np.convolve(cand, cand)
            k_max = int(np.argmax(b))
            # overlapping indices for k_max
            i_start = max(0, k_max - (n - 1))
            i_end = min(n-1, k_max)
            # compute products
            i_vals = np.arange(i_start, i_end+1)
            j_vals = k_max - i_vals
            prods = cand[i_vals] * cand[j_vals]
            imax = i_vals[np.argmax(prods)]
            jmax = k_max - imax
            # shrink these two entries a bit
            eps = 0.05  # 5% shrink
            cand[imax] *= (1 - eps)
            cand[jmax] *= (1 - eps)
            # renormalize again
            cand = cand / cand.sum()
    # final best check
    return [float(v) for v in best]

import numpy as np

# ------------------------------------------------------------
# Helper: compute the AC1 upper bound using an FFT convolution
# (much faster than full np.convolve for large n)
# ------------------------------------------------------------
def _ac1_score(vec: np.ndarray) -> float:
    """Return the value of evaluate_sequence(vec.tolist()) without the extra checks."""
    n = vec.size
    if n == 0:
        return float("inf")
    # FFT length: next power of two ≥ 2*n‑1 for speed and exactness
    m = 1 << ((2 * n - 1).bit_length())
    # real FFT for speed (vec is real)
    f = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_b = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound
    is as small as reasonably possible.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine length and a generic fallback.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        # basic sanity checks – keep the values but clip to allowed range
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
        else:
            parent = None
    else:
        parent = None

    if parent is None:
        # generic fallback – a moderate size for quick testing
        n = 256
        parent = np.ones(n, dtype=np.float64) / n
    else:
        n = parent.size

    # ------------------------------------------------------------------
    # 2. Normalise a starting candidate.
    #    We begin with the known near‑optimal “arcsine” shape:
    #        a_i ∝ 1 / sqrt(i+1)
    # ------------------------------------------------------------------
    base = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    base /= base.sum()          # total mass = 1

    best = base.copy()
    best_score = _ac1_score(best)

    # ------------------------------------------------------------------
    # 3. Small systematic search over simple parametric families.
    #    (a) Power‑law exponents around the optimal 0.5
    #    (b) A modest offset added to the index before the power‑law
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # (a) power‑law exponents
    for alpha in np.linspace(0.40, 0.60, 41):
        cand = np.power(idx + 1.0, -alpha)
        cand /= cand.sum()

        if sc < best_score:
            best, best_score = cand, sc

    # (b) offset + 0.5‑power‑law (the classic arcsine)
    for off in (0.0, 0.5, 1.0, 2.0, 5.0):
        cand = 1.0 / np.sqrt(idx + 1.0 + off)
        cand /= cand.sum()
        sc = _ac1_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 4. Random multiplicative block perturbations (global exploration)
    # ------------------------------------------------------------------
    cand = best.copy()
    for _ in range(400):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 150))
        stop = min(n, start + width)

        # log‑normal multiplicative noise on the block
        noise = np.exp(rng.normal(0.0, 0.12, size=stop - start))
        cand[start:stop] *= noise
        cand = np.clip(cand, 0.0, None)
