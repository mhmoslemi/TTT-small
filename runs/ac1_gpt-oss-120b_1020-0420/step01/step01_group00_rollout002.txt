import numpy as np

# ---------- fast AC1 upper bound (scale‑invariant) ----------
def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative vector (sum need not be 1)."""
    n = vec.size
    if n == 0:
        return np.inf
    # length for FFT (next power of two)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
