    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0 * i - 1.0) / (2.0 * i)

def compute_greedy(f, M):
    n = len(f)
    a = f # rename
    # cross term f*g currently zero
    cross = np.zeros(2*n - 1, dtype=np.float64)
    g = np.zeros(n, dtype=np.float64)
    # iterate j from n-1 down to 0
    for j in reversed(range(n)):
        # compute slack slice for k=j..j+n-1
        slack = M - cross[j:j+n]  # shape n
        # compute candidate deltas: slack / a
        # a is f, positive
        # delta_max = min(slack / a)
        # handle case a=0 (should not happen)
        ratios = slack / a
        delta_max = np.min(ratios)
        if delta_max <= 0:

        delta = delta_max
        g[j] = delta
        # update cross slice
        cross[j:j+n] += delta * a
    return g

import numpy as np

# ----- helper to generate the analytic half‑binomial coefficients -----
def _coefficients_binom_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


# ----- linear‑programming helper (sparse convolution matrix) -----
def _solve_lp_max_sum(f, M):
    """
    Maximise sum(g) subject to:
        conv(f, g) <= M  (element‑wise)
        0 <= g_i <= M
    Returns the optimal g as a NumPy array, or None on failure.
    """
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception:          # SciPy not available – signal to caller
        return None

    n = f.shape[0]
    # Build sparse matrix for the convolution constraints.
    # Column j contains the vector f shifted by j (i.e. a_i = f_i).
    # This is exactly a set of n diagonals with constant values f[i].
    diags = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=np.int32)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')

    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: maximise sum(g)  == minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # Variable bounds 0 <= g_i <= M
    bounds = [(0.0, float(M))] * n

    # Use HiGHS interior point solver (sparse‑aware)
    opt = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 30.0}
    )
    if opt.success:
        return np.clip(opt.x, 0.0, float(M))
    return None


# ----- main candidate constructor -----
def propose_candidate(seed=42, num_steps=8000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine attempts a linear‑programming construction followed by a
    small convex combination with the original sequence.
    """
    rng = np.random.default_rng(seed)

    # ----- initialise with the supplied parent or analytic start -----
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    else:
        n = 256
        base = _coefficients_binom_half(n)

    base = np.clip(base, 0.0, 1000.0)

    # ----- evaluate the starting point -----
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    target = 1.50274349
    if best_score <= target:
        return [float(x) for x in best]

    # ----- iterative improvement (LP + mixing) -----
    f = base.copy()
    for iteration in range(3):                     # a few global refinements
        # current convolution peak
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if M <= 0:
            break

        # solve LP for a vector g
        g = _solve_lp_max_sum(f, M)
        if g is None:
            break                     # fallback to simpler scheme

        # if g turned out zero, no progress possible
        if np.all(g == 0):
            break

        # size of g*g (controls safe mixing weight)
        conv_gg = np.convolve(g, g)
        Mgg = float(np.max(conv_gg))

        # safe mixing coefficient – keep the new max ≤ M
        # (1‑t)^2·M , 2t(1‑t)·M , t^2·Mgg  are the three contributions
        if Mgg > 0:
            t = np.sqrt(M / (Mgg + 1e-12)) * 0.9
        else:

        t = min(0.5, max(t, 1e-3))   # stay away from extremes

        # blended candidate

        except Exception:

        # keep the best

            best_score = score

        f = cand.copy()
        if best_score <= target:

    if not np.isfinite(best_score):
        best = base
    return [float(x) for x in best]
