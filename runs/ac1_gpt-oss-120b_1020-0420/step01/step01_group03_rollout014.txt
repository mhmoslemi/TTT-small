def compute_g_from_f(f, M):

    slack = np.full(2*n-1, M, dtype=np.float64)

    order = np.argsort(f)  # ascending
    for j in order:
        # Determine limit delta
        # We need for i from 0..n-1: row = i + j
        # Compute ratio = slack[row] / f[i]
        # Use vectorized: rows = np.arange(j, j+n)   # shape n

def _compute_g(seq, M):
    n = len(seq)

    g = np.zeros(n, dtype=np.float64)

def _compute_g(seq, M):
   n= len(seq)
   slack = np.full(2*n-1, M, dtype=np.float64)
   g = np.zeros(n, dtype=np.float64)
   order = np.argsort(seq)  # ascending (small first)
   for j in order:
       rows = np.arange(j, j+n)
       # compute allowed delta
       # compute ratio = slack[rows] / seq
       # Use np.where for zero division
       ratio = np.where(seq > 0, slack[rows] / seq, np.inf)
       delta = ratio.min()
       if delta <= 0:
           continue
       g[j] = delta
       # update slack
       slack[rows] -= seq * delta
       # clip slack to nonnegative
       slack[rows] = np.maximum(slack[rows], 0.0)
   return g

def _root_constant_seq(length: int) -> np.ndarray:
    """
    Generate a_i = C(2i,i)/4**i for i=0..length-1.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _compute_g(seq: np.ndarray, M: float) -> np.ndarray:
    """
    Greedy construction of a vector g that maximises its L1 norm
    under the linear constraints   (seq * g)[k] <= M  for all shifts k.
    This is a fast, matrix‑free approximation of the LP described in the
    problem statement.
    """
    n = seq.shape[0]
    slack = np.full(2 * n - 1, M, dtype=np.float64)   # remaining capacity per shift
    g = np.zeros(n, dtype=np.float64)

    # Process indices with smallest seq values first – they consume less capacity.
    order = np.argsort(seq)  # ascending
    for j in order:
        rows = np.arange(j, j + n)                # shifts where g_j participates
        # allowable increase for each i: slack[row] / seq[i]
        # seq contains no zeros (root‑constant), but guard against division by zero.
        denom = seq
        ratio = np.where(denom > 0, slack[rows] / denom, np.inf)
        delta = ratio.min()
        if not np.isfinite(delta) or delta <= 0.0:
            continue
        g[j] = delta
        slack[rows] -= denom * delta
        # Numerical safety – avoid tiny negatives.
        slack[rows] = np.maximum(slack[rows], 0.0)

    return g
