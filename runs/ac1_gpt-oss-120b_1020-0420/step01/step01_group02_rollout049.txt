def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = length (height_sequence_1 size?) else 256
    # base half-power series
    base = generate_series(0.5, n)
    np.clip(base,0,1000,out=base)
    best = base
    best_score = float(evaluate_sequence(best.tolist()))
    # try different alphas
    for alpha in np.linspace(0.3, 0.9, 7):
        cand = generate_series(alpha, n)
        np.clip(cand,0,1000,out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
    # mass redistribution search
    for frac in [0.01,0.03,0.05,0.1,0.15,0.2]:
        front = max(1, int(frac * n))
        sum_base = np.sum(best)
        front_sum = np.sum(best[:front])
        back_sum = np.sum(best[front:])
        for factor in [0.6,0.7,0.8,0.85,0.9]:
            # scale front down
            front_new_sum = factor * front_sum
            # determine scaling for back to keep total sum same
            if back_sum == 0:
                continue
            back_factor = (sum_base - front_new_sum) / back_sum
            cand = best.copy()
            cand[:front] *= factor
            cand[front:] *= back_factor
            np.clip(cand,0,1000,out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand.copy()
                best_score = score
    # random local perturbations
    for _ in range(200):
        cand = random_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    # final return list
    return [float(x) for x in best]

def random_block_perturbation(base, rng):
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand = base.copy()
    cand[start:start+width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

import numpy as np

# ------------------------------------------------------------
# Generate the power‑series coefficients of (1 - z)^(-alpha)
#   a_0 = 1,  a_i = a_{i-1} * (alpha + i - 1) / i
# ------------------------------------------------------------
def generate_series(alpha: float, length: int) -> np.ndarray:
    """Return a non‑negative step‑function of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (alpha + i - 1) / i
    return a


# ------------------------------------------------------------
# Multiply a random contiguous block by a log‑normal factor.
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a copy of ``base`` with one contiguous block scaled."""
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))   # modest log‑normal variation
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that attempts
    to minimise the AC1 upper bound defined by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine the length of the problem (use the parent if present)
    # --------------------------------------------------------
    n = 256
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)

    # --------------------------------------------------------
    # Baseline: half‑power series (alpha = 0.5)
    # --------------------------------------------------------
    base = generate_series(0.5, n)
    np.clip(base, 0.0, 1000.0, out=base)

    # Keep the best candidate seen so far
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------
    # 1) Scan a modest range of alphas (different power‑law tails)
    # --------------------------------------------------------
    for alpha in np.linspace(0.3, 0.9, 7):          # 0.3, 0.4, ..., 0.9
        cand = generate_series(float(alpha), n)
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

    # --------------------------------------------------------
    # 2) Mass‑redistribution: shrink an early block, compensate
    #    by scaling the remaining tail so that the total sum stays
    #    unchanged (scaling does not affect the bound, but the
    #    relative profile does).
    # --------------------------------------------------------
    for frac in (0.01, 0.03, 0.05, 0.10, 0.15, 0.20):
        front_sz = max(1, int(frac * n))
        front_sum = np.sum(best[:front_sz])
        back_sum = np.sum(best[front_sz:])
        total_sum = front_sum + back_sum
        if back_sum == 0:
            continue
        for front_factor in (0.6, 0.7, 0.8, 0.85, 0.9):
            new_front_sum = front_factor * front_sum
            back_factor = (total_sum - new_front_sum) / back_sum

            best_score = score

    return [float(x) for x in best]
