import numpy as np

# ------------------------------------------------------------
# Helper: half‑power series (coefficients of (1‑z)^(-1/2))
#   a_0 = 1,   a_i = a_{i‑1} * (2*i‑1)/(2*i)
# ------------------------------------------------------------
def half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Fast AC1 bound using FFT (avoids the heavy np.convolve in the search)
# ------------------------------------------------------------
def fast_ac1_bound(seq: np.ndarray, fft_len: int) -> float:
    s = seq.sum()
    if s == 0.0:
        return float('inf')
    # convolution via FFT (zero‑padded to fft_len)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    max_val = conv[: 2 * seq.size - 1].max()
    return 2.0 * seq.size * max_val / (s * s)
