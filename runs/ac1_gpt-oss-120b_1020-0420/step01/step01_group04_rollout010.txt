import numpy as np

# ----------------------------------------------------------------------
# Helper: first n coefficients of (1‑z)^‑1/2  (C_i = comb(2i,i)/4^i)
# ----------------------------------------------------------------------
def _coefficients_binom_half(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


# ----------------------------------------------------------------------
# Fast max‑convolution using FFT (real inputs)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray):
    """Return (max value, index of max, full convolution) of a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv
