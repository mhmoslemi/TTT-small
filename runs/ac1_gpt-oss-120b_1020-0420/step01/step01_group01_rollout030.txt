        a = np.empty(m, dtype=np.float64)

import numpy as np

# ----------------------------------------------------------------------
# Simple score identical to the evaluator (full convolution, max term)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    c = np.convolve(seq, seq)
    return float(2.0 * seq.size * c.max() / (s * s))


# ----------------------------------------------------------------------
# Catalan‑like central‑binomial sequence a_i = C(2i,i)/4^i
# ----------------------------------------------------------------------
def _catalan_seq(m: int) -> np.ndarray:
    a = np.empty(m, dtype=np.float64)
    a[0] = 1.0
    for i in range(m - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a
