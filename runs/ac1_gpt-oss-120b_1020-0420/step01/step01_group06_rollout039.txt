import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the AC1 upper bound for a binary (0/1) candidate without
# calling the expensive convolution.  The result is identical to
# evaluate_sequence for such vectors.
# ----------------------------------------------------------------------
def binary_score(selected, n):
    """
    selected : list of indices where the vector is 1
    n        : length of the full vector
    Returns the value 2 * n * max_b / (|selected|**2) where
    max_b is the maximum coefficient of the self‑convolution.
    """
    m = len(selected)
    if m == 0:
        return float("inf")
    # ordered‑pair counts per sum index (0 .. 2n-2)
    cnt = np.zeros(2 * n - 1, dtype=np.int16)

    # fill counts: self‑pairs contribute 1, distinct pairs contribute 2
    for i, idx in enumerate(selected):
        cnt[idx + idx] += 1                     # a_i*a_i
        for prev in selected[:i]:
            s = idx + prev
            cnt[s] += 2                         # a_i*a_prev + a_prev*a_i
    max_b = int(cnt.max())
    return float(2.0 * n * max_b / (m * m))


# ----------------------------------------------------------------------
# Build a set of indices whose pairwise sums never exceed max_allowed
# ordered‑pair count.  This is a matrix‑free greedy Sidon‑like construction.
# ----------------------------------------------------------------------
def build_sidon(order, max_allowed):
    """
    order        : 1‑D ndarray of a permutation of range(n)
    max_allowed  : maximum allowed value of any convolution coefficient
    Returns (selected_indices, max_b) where max_b <= max_allowed.
    """
    n = order.shape[0]
    cnt = np.zeros(2 * n - 1, dtype=np.int16)
    selected = []

    for i in order:
        # check self‑pair
        idx_self = i + i
        if cnt[idx_self] + 1 > max_allowed:
            continue

        # check all pairs with already selected indices
        ok = True
        for s in selected:
            idx = i + s
            if cnt[idx] + 2 > max_allowed:
                ok = False
                break
        if not ok:
            continue

        # accept i
        selected.append(i)
        cnt[idx_self] += 1
        for s in selected[:-1]:
            idx = i + s
            cnt[idx] += 2

    max_b = int(cnt.max())
    return selected, max_b

# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 upper
    bound is as low as possible.  The routine first uses any supplied
    `height_sequence_1` as a fallback, then tries several greedy
    Sidon‑like constructions with different collision caps.
    """
    # ------------------------------------------------------------------
    # Determine length n and obtain a valid parent candidate for fallback.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            tmp.ndim == 1
            and tmp.size > 0
            and np.isfinite(tmp).all()
        ):
            n = int(tmp.size)
            parent = np.clip(tmp, 0.0, 1000.0)
        else:
            n = 256
            parent = np.ones(n, dtype=np.float64)
    else:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # initial best candidate
    best_vec = parent.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Try a few Sidon‑like constructions.
    #   max_allowed = 2  (strict Sidon)
    #   max_allowed = 3  (allow a few collisions)
    #   max_allowed = 4  (more flexibility, still small max_b)
    # ------------------------------------------------------------------
    for max_allowed, attempts in ((2, 6), (3, 4), (4, 3)):
        for _ in range(attempts):
            order = rng.permutation(n)
            selected, max_b = build_sidon(order, max_allowed)
            if not selected:
                continue

            # fast score based on counts
            fast_score = binary_score(selected, n)

            # Only keep if promising (fast_score already beats best)
            if fast_score < best_score:
                cand = np.zeros(n, dtype=np.float64)
                cand[selected] = 1.0
                # final verification with the official evaluator
                eval_score = float(evaluate_sequence(cand.tolist()))
                if eval_score < best_score:
                    best_vec = cand
                    best_score = eval_score

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python list of floats).
    # ------------------------------------------------------------------
    return [float(v) for v in best_vec]
