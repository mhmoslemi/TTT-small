import numpy as np
import math

def _fast_score(vec: np.ndarray) -> float:
    """AC1 upper bound for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _compute_conv(vec: np.ndarray) -> np.ndarray:
    """Convolution of a vector with itself (FFT, length 2n‑1)."""
    n = vec.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    return np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
