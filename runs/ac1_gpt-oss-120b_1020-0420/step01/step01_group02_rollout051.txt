import numpy as np

def ac1_cost(arr):
    n = arr.size
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(arr, m)
    conv = np.fft.irfft(A * A, m)[:2 * n - 1]
    max_b = conv.max()
    return float(2 * n * max_b / (sum_a * sum_a))

def exponential_candidate(n, alpha):
    # return ndarray
    # maybe use np.exp(-alpha * np.arange(n))
    return np.exp(-alpha * np.arange(n, dtype=np.float64))

def find_best_alpha(n, iterations=30):
    lo = -5.0
    hi = 0.0
    phi = (np.sqrt(5) - 1) / 2  # = 0.618...
    best_alpha = None
    best_score = np.inf
    for _ in range(iterations):
        t1 = lo + (1 - phi) * (hi - lo)
        t2 = lo + phi * (hi - lo)
        a1 = 10.0 ** t1
        a2 = 10.0 ** t2
        arr1 = exponential_candidate(n, a1)
        arr2 = exponential_candidate(n, a2)
        s1 = ac1_cost(arr1)
        s2 = ac1_cost(arr2)
        if s1 < s2:
            hi = t2
            if s1 < best_score:
                best_score = s1
                best_alpha = a1
        else:
            lo = t1
            if s2 < best_score:
                best_score = s2
                best_alpha = a2
    # final refine at middle
    mid_alpha = 10.0 ** ((lo + hi) / 2)
    arr_mid = exponential_candidate(n, mid_alpha)
    mid_score = ac1_cost(arr_mid)
    if mid_score < best_score:
        best_alpha = mid_alpha
        best_score = mid_score
    return best_alpha, best_score

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length n
    n = 256
    if 'height_sequence_1' in globals():
        h = globals()['height_sequence_1']
        if isinstance(h, np.ndarray) and h.ndim == 1 and h.size > 0:
            n = int(h.size)
    # Base candidates
    # 1. exponential shape
    best_alpha, best_score = find_best_alpha(n, iterations=27)
    best_arr = exponential_candidate(n, best_alpha)
    # ensure within [0,1000]
    np.clip(best_arr, 0.0, 1000.0, out=best_arr)
    # 2. parent if exists
    if 'height_sequence_1' in globals():
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent_score = ac1_cost(parent_arr)
        if parent_score < best_score:
            best_arr = parent_arr
            best_score = parent_score
    # random block perturbations for further refinement
    max_width = max(1, n // 20)
    sigma_initial = 0.2  # for lognormal factor
    for step in range(num_steps):
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        # decreasing sigma
        sigma = sigma_initial * (1 - step / num_steps)
        factor = float(np.exp(rng.normal(0.0, sigma)))
        cand = best_arr.copy()
        cand[start:start+width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        # no need to renorm since scaling invariant
        cand_score = ac1_cost(cand)
        if cand_score < best_score:
            best_arr = cand
            best_score = cand_score
    # Return as list
    return [float(x) for x in best_arr]
