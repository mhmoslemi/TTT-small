import numpy as np

def _compute_score(arr: np.ndarray) -> float:
    """
    Fast evaluation of the AC1 bound:
        2 * n * max(conv(arr, arr)) / (sum(arr) ** 2)
    using a real FFT (linear memory, O(n log n) time).
    """
    n = arr.size
    # length of the full convolution
    m = 2 * n - 1
    # real FFT of the zero‑padded sequence
    fft_seq = np.fft.rfft(arr, n=m)
    # convolution via element‑wise multiplication in the frequency domain
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    max_b = float(conv.max())
    sum_a = float(arr.sum())
    # sum_a is never zero for the candidates we generate
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) whose
    AC1 upper bound is as low as possible.
    The routine combines a systematic search over simple parametric families
    with a modest stochastic local optimiser.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Parent / fallback sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
    else:
        # generic safe fallback – uniform of moderate length
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # ------------------------------------------------------------------
    # 2.  Initialise the best candidate
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = _compute_score(best)

    # ------------------------------------------------------------------
    # 3.  Helper to try a candidate and keep it if it improves
    # ------------------------------------------------------------------
    def _try(candidate: np.ndarray):
        nonlocal best, best_score
        sc = _compute_score(candidate)
        if sc < best_score:
            best = candidate.copy()
            best_score = sc
        return sc

    # ------------------------------------------------------------------
    # 4.  Simple families ------------------------------------------------
    # ------------------------------------------------------------------
    indices = np.arange(1, n + 1, dtype=np.float64)   # 1‑based for power laws

    # ---- uniform (baseline) -------------------------------------------
    uniform = np.ones(n, dtype=np.float64) / n
    _try(uniform)

    # ---- power‑law family a_i ∝ (i)^‑α  -------------------------------
    alphas_coarse = np.linspace(0.10, 0.90, 81)       # step ≈0.01
    best_alpha = None
    for a in alphas_coarse:
        w = np.power(indices, -a)
        w /= w.sum()
        sc = _try(w)
        if sc < best_score:
            best_alpha = a

    # ---- refined grid around the best coarse α -------------------------
    if best_alpha is not None:
        lo = max(0.01, best_alpha - 0.02)
        hi = min(0.99, best_alpha + 0.02)
        alphas_fine = np.linspace(lo, hi, 41)       # step ≈0.001
        for a in alphas_fine:
            w = np.power(indices, -a)
            w /= w.sum()
            _try(w)

    # ---- exponential decay a_i ∝ exp(-β·i) -----------------------------
    betas = np.linspace(1e-5, 0.010, 100)            # fine grid of decays
    for b in betas:
        w = np.exp(-b * np.arange(n, dtype=np.float64))
        w /= w.sum()
        _try(w)

    # ---- random mixtures of two power‑laws ----------------------------
    for _ in range(200):
        a1 = rng.uniform(0.05, 0.95)
        a2 = rng.uniform(0.05, 0.95)
        lam = rng.random()
        w1 = np.power(indices, -a1)
        w2 = np.power(indices, -a2)
        w = lam * w1 + (1.0 - lam) * w2
        w /= w.sum()
        _try(w)

    # ------------------------------------------------------------------
    # 5.  Stochastic local optimisation ---------------------------------
    # ------------------------------------------------------------------
    max_iters = 1500               # budget of further evaluations
    sigma = 0.25                    # initial noise level for multiplicative perturbations

    for it in range(max_iters):
        cand = best.copy()

        # two mutation families: block scaling or mass transfer
        if rng.random() < 0.65:
            # --- multiplicative log‑normal noise on a random block -----
            start = rng.integers(0, n)
            # block width up to about 1 % of the length, never zero
            max_width = max(2, n // 100)
            width = rng.integers(1, max_width)
            stop = min(n, start + width)
            noise = np.exp(rng.normal(0.0, sigma, size=stop - start))
            cand[start:stop] *= noise
        else:
            # --- move a fraction of mass from i to j ------------------
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and cand[i] > 0.0:
                # move up to half of the source entry
                delta = cand[i] * rng.random() * 0.5
                cand[i] -= delta
                cand[j] += delta

        # re‑normalise to keep the total mass fixed (scale‑invariant)
        total = cand.sum()
        if total > 0.0:
            cand /= total
            _try(cand)

        # anneal the noise amplitude slowly
        sigma = max(0.01, sigma * 0.997)

    # ------------------------------------------------------------------
    # 6.  Optional final symmetry check (reversed order) ---------------
    # ------------------------------------------------------------------
    rev = best[::-1]
    _try(rev)

    # ------------------------------------------------------------------
    # 7.  Return the best candidate found as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
