import numpy as np

# ------------------------------------------------------------
# Helper: compute ordered‑pair sum counts for a set of indices
# ------------------------------------------------------------
def _pair_counts(indices, n):
    """
    indices : 1‑D np.ndarray of distinct integers in [0, n)
    returns : np.ndarray length 2*n‑1 with ordered‑pair sum frequencies
    """
    if indices.size == 0:
        return np.zeros(2 * n - 1, dtype=np.int64)
    # outer sum produces all i+j (ordered)
    sums = np.add.outer(indices, indices)          # shape (m,m)
    cnt = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return cnt.astype(np.int64)


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Returns a list of non‑negative heights that (hopefully) yields a
    low AC1 upper bound according to the pre‑imported evaluate_sequence.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine the problem length (use the provided parent if any)
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size) if parent.ndim == 1 and parent.size > 0 else 256
    except NameError:          # height_sequence_1 not defined
        n = 256

    TARGET   = 1.50274349          # desired bound
    MIN_M    = 80                    # we never drop below this set size
    TRIES    = 30                    # independent random starts
    ADD_TRIES = 6                    # attempts to re‑insert after a removal

    # --------------------------------------------------------
    # Simple uniform fallback (always feasible)
    # --------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # --------------------------------------------------------
    # Core search: binary mask with incremental updates
    # --------------------------------------------------------
    # start with a reasonably dense random subset
    init_frac = 0.13                     # about 13 % of the domain
    INIT_M = max(MIN_M, int(init_frac * n))
    if INIT_M > n:
        INIT_M = n

    for _ in range(TRIES):
        # random initial subset of the requested size
        selected = set(rng.choice(n, size=INIT_M, replace=False))
        sel_arr = np.fromiter(selected, dtype=np.int64)
        cnt = _pair_counts(sel_arr, n)
        max_c = int(cnt.max())
        m = len(selected)

        # -----------------------------------------------------------------
        # Reduce the maximal convolution count while trying to keep |S| large
        # -----------------------------------------------------------------
        while True:
            bound_est = 2.0 * n * max_c / (m * m)
            if bound_est <= TARGET or m <= MIN_M:
                break

            # choose a sum k that currently attains the maximum count
            ks = np.flatnonzero(cnt == max_c)
            k = int(rng.choice(ks))

            # an index that participates in that sum
            involved = [i for i in selected if (k - i) in selected]
            if involved:
                rem = int(rng.choice(involved))
            else:
                rem = int(rng.choice(list(selected)))

            # ----------- incremental removal of `rem` --------------------
            # subtract its contributions from the count histogram
            cnt[2 * rem] -= 1                              # (rem,rem)
            if m > 1:                                      # there are others
                other = sel_arr[sel_arr != rem]
                sums = other + rem                         # unordered sums
                dec = np.bincount(sums, minlength=2 * n - 1)
                cnt[: len(dec)] -= 2 * dec                 # (rem,i) and (i,rem)

            # update the set / array
            selected.remove(rem)
            m -= 1
            sel_arr = np.fromiter(selected, dtype=np.int64) if m else np.empty(0, dtype=np.int64)

            # recompute the current maximum after removal
            max_c = int(cnt.max())

            # ----------- attempt to add a replacement without raising max_c ---
            for __ in range(ADD_TRIES):
                cand = int(rng.integers(0, n))
                if cand in selected:
                    continue

                # prospective count histogram after insertion of `cand`
                new_cnt = cnt.copy()
                new_cnt[2 * cand] += 1                      # (cand,cand)
                if m > 0:
                    sums_add = sel_arr + cand
                    inc = np.bincount(sums_add, minlength=2 * n - 1)
                    new_cnt[: len(inc)] += 2 * inc          # (cand,i),(i,cand)

                new_max = int(new_cnt.max())
                if new_max <= max_c:                       # no worsening
                    selected.add(cand)
                    m += 1
                    cnt = new_cnt
                    max_c = new_max
                    sel_arr = np.fromiter(selected, dtype=np.int64)
                    break

        # -----------------------------------------------------------------
        # Evaluate the binary‑mask candidate produced by this trial
        # -----------------------------------------------------------------
        if not selected:
            continue
        cand_vec = np.zeros(n, dtype=np.float64)
        cand_vec[list(selected)] = 1.0
        score = float(evaluate_sequence(cand_vec.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand_vec
            best_score = score
            if best_score <= TARGET:
                return best_vec.tolist()

    # -----------------------------------------------------------------
    # Light power‑law sweep as a cheap secondary fallback
    # -----------------------------------------------------------------
    p_vals = np.linspace(0.45, 0.80, 71)        # 0.45 … 0.80 in steps of 0.005
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                break

    # ------------------------------------------------------------
    # Return the best candidate found (always non‑empty)
    # ------------------------------------------------------------
    return best_vec.tolist()
