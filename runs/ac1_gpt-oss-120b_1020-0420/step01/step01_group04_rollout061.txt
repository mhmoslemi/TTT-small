            j = i

            delta = rng.uniform(0.01, 0.2) * best[i]  # amount to move
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
        elif move_type == 'pair_transfer':
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            # transfer delta from i to j
            delta = rng.uniform(0.01, 0.2) * best[i]
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
        elif move_type == 'scale_block':
            cand = best.copy()
            min_len = max(1, n // 200)  # 0.5%
            max_len = max(2, n // 10)   # up to 10%
            width = rng.integers(min_len, max_len+1)
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.15))
            cand[start:start+width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
        elif move_type == 'bump':
            cand = best.copy()
            idx = rng.integers(0, n)
            bump = max(0.0, rng.normal(0.0, 0.03))
            cand[idx] += bump
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:  # swap
            cand = best.copy()
            i,j = rng.integers(0,n,size=2)
            cand[i],cand[j] = cand[j],cand[i]
        # evaluate
        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        # acceptance: if better, accept; else accept with prob exp((best_score - score)/T)
        if np.isfinite(score):
            if score < best_score:
                best = cand
                best_score = score
            else:
                # maybe accept worse for exploration
                if rng.random() < np.exp((best_score - score) / max(T, 1e-12)):
                    best = cand
                    best_score = score
        if best_score < 1.0: break
        # early termination if below target
        if best_score < 1.50274349:
            break
    return best.tolist()

def reduce_peak_move(a, rng):
    n = len(a)
    b = np.convolve(a, a)
    k = int(np.argmax(b))
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    # compute contributions
    i_vals = np.arange(i_low, i_high+1)
    contrib = a[i_vals] * a[k - i_vals]
    # pick index with maximum contribution
    idx = int(np.argmax(contrib))
    i_star = int(i_vals[idx])
    # choose target index not in [i_low, i_high]
    # generate random index until outside range
    while True:
        t = int(rng.integers(0, n))
        if t < i_low or t > i_high:
            break
    # transfer fraction delta from i_star to t
    delta = rng.uniform(0.01, 0.2) * a[i_star]
    a_new = a.copy()
    a_new[i_star] -= delta
    a_new[t] += delta
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new

import numpy as np

# ------------------------------------------------------------
# Helper utilities (matrix‑free convolution, scoring)
# ------------------------------------------------------------
def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2 : C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _auto_conv_max(a: np.ndarray) -> float:
    """
    Return max_k (a*a)_k using an FFT‑based convolution.
    Works in O(L log L) where L is the next power‑of‑two ≥ 2·len(a).
    """
    n = a.shape[0]
    # length for a zero‑padded FFT (next power of two)
    L = 1 << ((2 * n - 1).bit_length())
    # real‑FFT of the padded array
    fa = np.fft.rfft(a, n=L)
    # element‑wise square (auto‑convolution in the frequency domain)
    conv = np.fft.irfft(fa * fa, n=L)
    # truncate to the true convolution length
    conv = conv[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Exact objective used by the official evaluator (but computed fast)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    max_b = _auto_conv_max(a)
    n = a.shape[0]
    return 2.0 * n * max_b / (s * s)

def _move_max_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of the current maximum entry to a random partner."""
    n = a.shape[0]
    i = int(np.argmax(a))
    # choose a different index as recipient
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    # fraction to move (1 % – 8 % of the source value)
    frac = rng.uniform(0.01, 0.08)
    delta = float(frac * a[i])
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _move_random_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:

def _move_block_scale(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a contiguous block by a mild log‑normal factor."""
    n = a.shape[0]
    min_len = max(1, n // 200)          # ≈0.5 % of n

    out = a.copy()
    out[start:start + width] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _move_bump(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:

def _move_swap(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
