def propose_candidate(...):
    try:
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        n = 1024
        a = np.full(n, 1.0)
    # ensure non-negative and within bounds
    a = np.clip(a, 0.0, 1000.0)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # main loop
    max_iters = 100
    for _ in range(max_iters):
        S = a.sum()
        conv = np.convolve(a, a)
        M = conv.max()
        k = int(conv.argmax())
        # compute a_rev at shifted indices
        idx = k - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        a_shifted = np.zeros_like(a)
        a_shifted[mask] = a[idx[mask]]
        # direction = M - a_shifted * S
        direction = M - a_shifted * S
        # step size
        max_abs = np.max(np.abs(direction))
        if max_abs == 0:
            break
        eta = 0.01 / max_abs
        # update
        a = a + eta * direction
        # enforce nonnegativity and caps
        a = np.clip(a, 0.0, 1000.0)
        # compute new score
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            # fallback: break
            break
        if not np.isfinite(score):
            break
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # early stop if improvement small
        if best_score < 1.50274349:
            break
    # final candidate
    candidate = best_seq.tolist()
    # final safety check
    try:
        final_score = evaluate_sequence(candidate)
        if not np.isfinite(final_score) or final_score >= np.inf:
            # fallback uniform
            candidate = np.full_like(best_seq, 1.0).tolist()
    except Exception:
        candidate = np.full_like(best_seq, 1.0).tolist()
    return candidate

import numpy as np

def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a non‑negative step‑function sequence that improves the AC1 score.
    The method starts from the provided parent sequence (height_sequence_1)
    and performs a simple projected gradient descent that transfers mass from
    coefficients that heavily influence the maximum of the self‑convolution
    to coefficients that have little influence.  The step size is limited so
    that no single convolution entry grows by more than a tiny amount per
    iteration, keeping the maximum roughly bounded while the total sum
    increases.  The best sequence seen during the walk is returned.
    """
    # ------------------------------------------------------------------
    #  Initialise from the parent if it exists, otherwise use a modest fallback.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        n = 256
        a = np.full(n, 1.0, dtype=np.float64)

    # Enforce the evaluator’s hard bounds
    np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    #  Helper to compute the AC1 score without invoking the evaluator
    #  (the evaluator is used only for a final sanity check).
    # ------------------------------------------------------------------
    def _score(arr: np.ndarray) -> float:
        s = arr.sum()
        if s == 0.0:
            return np.inf
        c = np.convolve(arr, arr)
        m = c.max()
        return float(2.0 * n * m / (s * s))

    # Initial score and best‑so‑far record
    best_seq = a.copy()
    best_score = _score(a)

    # Target bound we would like to beat
    TARGET = 1.50274349

    # ------------------------------------------------------------------
    #  Gradient‑descent walk.
    #  direction_i = M - a_{k*‑i} * S  (see analysis for derivation)
    #  The step size is chosen so that any convolution entry can change
    #  by at most EPS per iteration.
    # ------------------------------------------------------------------
    EPS = 0.001                 # max permitted change of any conv entry per step
    MAX_ITER = 300
    no_improve = 0

    for _ in range(MAX_ITER):
        # current convolution, its maximum and the index where it occurs
        conv = np.convolve(a, a)
        M = conv.max()
        k = int(conv.argmax())
        S = a.sum()

        # a_{k-i} for all i (zero where index out of bounds)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        a_shift = np.zeros_like(a)
        a_shift[mask] = a[idx[mask]]

        # direction of steepest descent for the AC1 score
        direction = M - a_shift * S
        max_abs_dir = np.abs(direction).max()
        if max_abs_dir == 0.0:
            break

        # bound the step so that any convolution entry changes by ≤ EPS
        max_a = a.max()
        step = EPS / (2.0 * max_a * max_abs_dir + 1e-12)

        step = min(step, 0.02)

        # gradient step and projection onto the feasible box
        a = a + step * direction
        np.clip(a, 0.0, 1000.0, out=a)

        # evaluate the new score
        cur_score = _score(a)

        # keep the best sequence seen so far
        if cur_score + 1e-12 < best_score:

            best_seq = a.copy()
            no_improve = 0

    # ------------------------------------------------------------------

        if not np.isfinite(final_check):

    except Exception:
