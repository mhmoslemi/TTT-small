def flatten(a, steps=20, eta=0.5, eps=1e-12):
    best_a = a.copy()
    best_score = eval_score(best_a)
    for s in range(steps):
        b = np.convolve(a, a)
        k_max = int(b.argmax())
        # compute contributions
        i_start = max(0, k_max - (len(a) - 1))
        i_end = min(len(a) - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        partner = k_max - idx
        contrib = a[idx] * a[partner]
        # compute scaling factor
        avg = contrib.mean()
        # avoid divide by zero
        factor = (avg / (contrib + eps))**eta
        a[idx] = a[idx] * factor
        # renorm
        total = a.sum()
        if total > 0:
            a = a / total
        # evaluate
        sc = eval_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def _score(vec: np.ndarray) -> float:
    """
    Evaluate a candidate using the provided `evaluate_sequence` function.
    The input is a 1‑D NumPy array; it is converted to a list for the evaluator.
    """
    return float(evaluate_sequence(vec.tolist()))


def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale the vector to have L¹‑norm equal to 1 (sum‑to‑one)."""
    s = v.sum()
    if s == 0.0:
        return v
    return v / s


def _powerlaw(n: int, alpha: float) -> np.ndarray:
    """Return a normalized power‑law sequence a_i ∝ (i+1)^{-alpha}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = np.power(idx, -alpha)
    return _normalize(w)


def _exponential(n: int, beta: float) -> np.ndarray:
    """Return a normalized exponential decay sequence a_i ∝ exp(-beta·i)."""
    w = np.exp(-beta * np.arange(n, dtype=np.float64))
    return _normalize(w)


def _flatten_optimize(a: np.ndarray,
                     steps: int = 15,
                     eta: float = 0.5,
                     eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Iteratively reduce the peak of the self‑convolution.
    Only the entries that participate in the current maximal convolution
    coefficient are re‑weighted.
    """
    n = a.size
    cur = _normalize(a.copy())
    best = cur.copy()
    best_sc = _score(best)

    for _ in range(steps):
        conv = np.convolve(cur, cur)
        k = int(conv.argmax())                     # index of the current maximum
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        idx = np.arange(i0, i1 + 1)
        partner = k - idx
        contrib = cur[idx] * cur[partner]          # contributions to the max coefficient
        avg = contrib.mean() + eps
        factor = (avg / (contrib + eps)) ** eta    # shrink large contributions, enlarge small
        cur[idx] = cur[idx] * factor
        cur = _normalize(cur)

        sc = _score(cur)
        if sc < best_sc:
            best_sc = sc
            best = cur.copy()

    return best, best_sc


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          attempts: int = 200) -> tuple[np.ndarray, float]:

    Apply many small multiplicative log‑normal perturbations to random blocks.

    """

    best = _normalize(a.copy())

    for _ in range(attempts):
