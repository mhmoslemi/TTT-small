def _reduce_peak(seq, rng, reduction_factor=0.9, top_frac=0.02):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    # compute contributions
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    contrib = np.zeros(n)
    contrib[mask] = seq[idx[mask]] * seq[j[mask]]
    # pick top positions
    num = max(1, int(top_frac * n))
    # indices with highest contrib
    top_idx = np.argpartition(contrib, -num)[-num:]
    # Reduce them
    cand = seq.copy()
    # compute removed mass sum
    reduction = (1 - reduction_factor) * cand[top_idx]
    cand[top_idx] *= reduction_factor
    # Distribute removed mass uniformly to random low indices
    # Choose random indices (not in top_idx) of same size or maybe all indices.
    low_idx = np.setdiff1d(np.arange(n), top_idx, assume_unique=True)
    if low_idx.size == 0:
        # fallback: just add to random indices
        low_idx = np.arange(n)
    # pick random destination indices (maybe same number as top)
    dest = rng.choice(low_idx, size=top_idx.size, replace=True)
    # Add proportional mass
    cand[dest] += reduction / top_idx.size
    # Clip
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate_sequence is provided by the harness.
# ----------------------------------------------------------------------


def _initial_candidate():
    """
    Build a starting vector.
    If the parent construction `height_sequence_1` is present, use it.
    Otherwise fall back to a simple uniform vector.
    """
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
    else:
        # default length when no parent is supplied
        n = 256
        seq = np.full(n, 1.0, dtype=np.float64)
    # guarantee a tiny positive sum (the evaluator needs sum>0.01)
    if seq.sum() < 0.02:
        seq += 0.01
    return seq


def _random_block(seq, rng, min_frac=0.01, max_frac=0.25):
    """
    Randomly scale a contiguous block of the sequence.
    """
    n = seq.size
    min_len = max(1, int(n * min_frac))
    max_len = max(2, int(n * max_frac))
    length = rng.integers(min_len, max_len + 1)
    start = rng.integers(0, n - length + 1)
    factor = np.exp(rng.normal(0.0, 0.20))
    out = seq.copy()
    out[start:start + length] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _reduce_peak(seq, rng, top_frac=0.02, reduction=0.85):
    """
    Identify the convolution peak and pull a little mass away from the
    contributing indices.  The removed mass is redistributed to random
    low‑contributing positions, preserving the total sum.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    # contributions of each index to the peak (a_i * a_{k-i})
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros(n, dtype=np.float64)
    contrib[mask] = seq[i[mask]] * seq[j[mask]]

    # indices with largest contribution
    n_top = max(1, int(top_frac * n))
    top_idx = np.argpartition(contrib, -n_top)[-n_top:]

    out = seq.copy()
    removed = (1.0 - reduction) * out[top_idx]
    out[top_idx] *= reduction

    # choose destinations that are *not* in top_idx
    all_idx = np.arange(n)
    low_idx = np.setdiff1d(all_idx, top_idx, assume_unique=True)
    if low_idx.size == 0:          # extremely unlikely, but guard
        low_idx = all_idx
    dest = rng.choice(low_idx, size=top_idx.size, replace=True)
    # distribute the removed mass uniformly to the destinations
    out[dest] += removed / top_idx.size

    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _swap_two(seq, rng):
    """Swap two random positions."""
    n = seq.size
    i, j = rng.integers(0, n, size=2)
    out = seq.copy()
    out[i], out[j] = out[j], out[i]
    return out


def _bump_one(seq, rng, sigma=0.05):
    """Add a small non‑negative bump to a single entry."""
    out = seq.copy()
    idx = rng.integers(0, out.size)
    bump = max(0.0, rng.normal(0.0, sigma))
    out[idx] += bump
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def propose_candidate(seed=42, num_steps=12000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low
    AC1 upper bound.

    The search starts from the supplied parent sequence (if any) and
    repeatedly applies inexpensive stochastic moves.  In addition to the
    generic moves of the previous baseline, a *peak‑reduction* move is
    used: the index of the maximal convolution entry is located and a
    fraction of the mass that contributes most to that peak is shifted
    to low‑contributing locations.  This directly targets the
    objective 2·n·max(conv)/(sum²) and quickly drives the score below
    the target 1.50274349 when possible.
    """
    rng = np.random.default_rng(seed)

    # initialise
    best = _initial_candidate()
    best_score = float(evaluate_sequence(best.tolist()))

    # if a parent is present and better, use it
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent
            best_score = parent_score
    except Exception:
        pass

    target = 1.50274349
    n = best.size

    # ------------------------------------------------------------------
    # search loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # early exit if we already beat the target
        if best_score < target:
            break

        # pick an operation
        op = rng.integers(0, 5)
        if op == 0:
