import numpy as np

def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2:  C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _full_convolution(arr):
    """Return the full auto‑convolution of a non‑negative vector."""
    n = arr.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(arr, m)
    conv = np.fft.irfft(f * f, m)[: 2 * n - 1]
    return conv


def _gradient_step(arr, rng, alpha):
    """Scale entries according to a simple gradient of the bound."""
    n = arr.shape[0]
    conv = _full_convolution(arr)
    M = conv.max()
    S = arr.sum()
    # pick a random shift that attains the maximum
    max_pos = np.where(np.isclose(conv, M, atol=1e-12))[0]
    k = int(rng.choice(max_pos))
    d = k - (n - 1)                 # signed shift

    # contribution of each coordinate to the active convolution
    D = np.zeros_like(arr)
    if d >= 0:
        D[: n - d] += arr[d:]      # a_i * a_{i+d}
        D[d:]      += arr[: n - d] # a_{i-d} * a_i
    else:
        d = -d
        D[: n - d] += arr[d:]
        D[d:]      += arr[: n - d]

    # target contribution from the first‑order optimality condition
    T = 2.0 * M / S if S != 0 else np.inf

    cand = arr.copy()
    inc = D < T
    dec = ~inc
    cand[inc] *= (1.0 + alpha)
    cand[dec] *= max(0.0, 1.0 - alpha)
    return cand


def _random_block_scale(arr, rng):
    """Multiply a random contiguous block by a log‑normal factor."""
    n = arr.shape[0]
    cand = arr.copy()
    # block length between ~0.5 % and 20 % of the vector
    min_len = max(1, n // 200)
    max_len = max(2, n // 5)
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.25)))   # moderate perturbation
    cand[start:start + width] *= factor
    return cand


def propose_candidate(seed=42, max_steps=15000, **kwargs):
    """
    Return a non‑negative height sequence with a low AC1 upper bound.
    The routine starts from the supplied parent (if any) and then
    performs a mixed gradient / random‑block hill‑climbing search.
    """
    rng = np.random.default_rng(seed)

    # --- initialise -------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        a = _coefficients_binom_half(n)

    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    target = 1.50274349
    alpha = 0.02                     # step size for gradient updates

    for step in range(max_steps):
        if best_score < target:
            break

        # Decide which mutation to apply
        if rng.random() < 0.25:
            cand = _random_block_scale(best, rng)
        else:
            cand = _gradient_step(best, rng, alpha)

        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate the candidate
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        # keep the best improvement
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

        # slowly shrink the gradient step size
        if (step + 1) % 2000 == 0:
            alpha *= 0.7

    # final fallback – ensure a finite returning value
    if not np.isfinite(best_score):
        best = np.maximum(best, 0.0)
        best = np.clip(best, 0.0, 1000.0)

    return [float(x) for x in best]
