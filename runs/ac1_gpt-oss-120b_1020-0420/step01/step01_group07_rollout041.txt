import numpy as np

# ----------------------------------------------------------------------
# Convolution via FFT (full, linear convolution)
# ----------------------------------------------------------------------
def _full_convolve_fft(a: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Return full linear convolution of a with itself using FFT.
    a : 1‑d real array
    fft_size : length of zero‑padded FFT (power of two)
    """
    fa = np.fft.rfft(a, n=fft_size)
    conv = np.fft.irfft(fa * fa, n=fft_size)
    return conv[: 2 * a.shape[0] - 1]
