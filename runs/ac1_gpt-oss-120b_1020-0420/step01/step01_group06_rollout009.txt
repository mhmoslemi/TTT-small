import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate a numpy vector using the provided scorer
# ----------------------------------------------------------------------
def _score_vec(vec: np.ndarray) -> float:
    """Return the AC1 upper bound for a numpy 1‑D array."""
    return float(evaluate_sequence(vec.tolist()))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a non‑negative step‑function sequence of length ``n``.
    The method starts from the uniform distribution (which is provably
    optimal for the given objective) and performs a tiny stochastic
    local search; any improvement is kept.
    """
    # --------------------------------------------------------------
    # Determine length ``n`` – use the supplied parent if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0 \
                and np.isfinite(parent_arr).all():
            n = parent_arr.size
        else:
            n = 256
    else:
        n = 256

    # --------------------------------------------------------------
    # Initialise with the uniform vector (sum == 1, optimal baseline)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best = uniform.copy()
    best_score = _score_vec(best)

    # --------------------------------------------------------------
    # Random block perturbations (hill‑climbing). Expected not to beat
    # the uniform baseline, but kept for completeness.
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)
    max_iter = 200  # modest budget, linear memory usage
    for _ in range(max_iter):
        # pick a random contiguous block
        start = rng.integers(0, n)
        max_width = max(2, n // 100)
        width = rng.integers(1, max_width + 1)
        stop = min(n, start + width)

        cand = best.copy()
        # multiplicative log‑normal noise (mean 1, modest std)
        noise = np.exp(rng.normal(0.0, 0.05, size=stop - start))
        cand[start:stop] *= noise

        # re‑normalise to keep the ℓ₁‑norm unchanged (scale‑invariant)
        total = cand.sum()
        if total > 0:
            cand /= total

        sc = _score_vec(cand)
        # keep strictly improving candidates
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------------
    # Return as a plain Python list of floats
    # --------------------------------------------------------------
    return [float(v) for v in best]
