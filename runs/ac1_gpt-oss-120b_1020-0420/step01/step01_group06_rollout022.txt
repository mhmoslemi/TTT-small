def fft_full_convolution(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]
    return conv
```

def golden_section(func, lo, hi, tol=1e-6, max_iter=30):
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    a = lo
    b = hi
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
    # return best point (c if fc < fd else d)
    return c if fc < fd else d, fc if fc < fd else fd
```

def golden_section(func, lo, hi, tol=1e-6, max_iter=35):
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    a = lo
    b = hi
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if (b - a) < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd

import numpy as np

# Global placeholder for the current length – set inside propose_candidate
_CURRENT_N = None


def fft_full_convolve(v: np.ndarray) -> np.ndarray:
    """Full linear convolution of v with itself using an FFT."""
    n = v.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fv = np.fft.rfft(v, size)
    conv = np.fft.irfft(fv * fv, size)[: 2 * n - 1]
    return conv


def score_vector(v: np.ndarray) -> float:
    """Exact AC1 upper bound for a non‑negative vector v."""
    conv = fft_full_convolve(v)
    n = v.shape[0]
    s = v.sum()
    if s == 0.0:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def make_power(alpha: float, n: int) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-alpha}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = np.power(idx, -alpha)
    w = np.clip(w, 0.0, 1000.0)
    total = w.sum()
    if total == 0.0:
        w = np.ones(n, dtype=np.float64)
        total = n
    return w / total


def make_exponential(beta: float, n: int) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-beta*i), normalised to sum 1."""
    w = np.exp(-beta * np.arange(n, dtype=np.float64))
    w = np.clip(w, 0.0, 1000.0)
    total = w.sum()
    if total == 0.0:
        w = np.ones(n, dtype=np.float64)
        total = n
    return w / total


def golden_section(func, lo: float, hi: float,
                  tol: float = 1e-6, max_iter: int = 35):
    """One‑dimensional golden‑section search for a minimum."""
    phi = (np.sqrt(5) - 1) / 2.0           # ≈0.618
    a, b = lo, hi
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if (b - a) < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
    if fc < fd:
        return c, fc
    else:
        return d, fd

            if (it + 1) % noise_freq == 0:
                s = rng.integers(0, n)
                w = rng.integers(1, max(2, n // 50))
                e = min(n, s + w)
                factor = np.exp(rng.normal(0.0, 0.12, size=e - s))
                a[s:e] *= factor
                a = normalize_vector(a)

        if best_score < target:
            break

    # ------------------------------------------------------------------
    # 8. Fallback safety (should never trigger).
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best_vec = normalize_vector(np.ones(n, dtype=np.float64))

    return [float(v) for v in best_vec]
