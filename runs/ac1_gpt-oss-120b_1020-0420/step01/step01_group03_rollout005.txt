import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _next_pow2(x: int) -> int:
    """Small helper for the FFT size."""
    return 1 << ((x - 1).bit_length())


def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum of the full self‑convolution using FFT (real input)."""
    fa = np.fft.rfft(a, n=fft_len)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact AC1 upper bound for a given sequence."""
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    m = _conv_max(a, fft_len)
    return 2.0 * n * m / (s * s)
