    n = len(f)
    m = 2*n - 1
    A = lil_matrix((m, n), dtype=np.float64)
    for k in range(m):
        start = max(0, k - (n-1))
        end = min(n-1, k)
        # indices i from start to end inclusive; j = k - i.
        # coefficients are f[i]

        # So fill row k as A[k, j] = f[k - j] for j such that 0 <= j < n and 0 <= k-j < n.
