Implementation details of fast_score:

```
def fast_score(arr):
    n = arr.shape[0]
    m = 1 << ((2*n - 2).bit_length())   # next power of two >= 2n-1
    af = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(af * af, n=m)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf

import numpy as np

# --------------------------------------------------------------
# Helper: half‑binomial coefficients (C_i = comb(2i,i)/4^i)
def _coefficients_binom_half(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


# --------------------------------------------------------------
# Fast evaluation (FFT based) – used during the search.
# The evaluator supplied to the harness (evaluate_sequence) is
# called only for the final answer.
def _fast_score(arr: np.ndarray, n: int, fft_len: int) -> float:
    """Return the AC1 bound using an FFT convolution (no clipping)."""
    af = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 20000, **kwargs):
    """
    Return a list of non‑negative heights whose AC1 bound is as low
    as possible.  The routine starts from the supplied parent
    (height_sequence_1) if present, otherwise from the analytic
    half‑binomial sequence, and then performs a cheap simulated‑annealing
    style optimisation that favours increasing the total sum while
    keeping the convolution peak low.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------
    # Determine length and initialise the starting candidate.
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.shape[0])
        parent = np.clip(parent, 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
    else:
        parent = None
        n = 256
        parent_score = np.inf

    # Analytic half‑binomial start (always available)
    analytic = _coefficients_binom_half(n)
    analytic = np.clip(analytic, 0.0, 1000.0)
    analytic_score = float(evaluate_sequence(analytic.tolist()))

    # Choose the best of the baselines
    if parent_score < analytic_score:
        best = parent.copy()
        best_score = parent_score
    else:
        best = analytic.copy()
        best_score = analytic_score

    # ----------------------------------------------------------
    # Pre‑compute FFT length (next power of two ≥ 2n‑1)
    fft_len = 1 << ((2 * n - 2).bit_length())

    # Fast score of the current incumbent
    cur = best.copy()
    cur_score = _fast_score(cur, n, fft_len)

    target = 1.50274349

    # ----------------------------------------------------------
    # Main optimisation loop
    for step in range(num_steps):
        if best_score < target:
            break

        # Temperature for occasional uphill moves (linear cooling)
        temp = max(0.0, 1.0 - step / num_steps)

        op = rng.integers(0, 5)

        cand = cur.copy()

        if op == 0:
            # --------------------------------------------------
            # Tail block scaling – try to boost the low‑valued tail.
            tail_start = int(rng.integers(n // 2, n))
            width = int(rng.integers(1, max(2, n - tail_start + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.25)))  # log‑normal factor >0
            cand[tail_start : tail_start + width] *= factor

        elif op == 1:
            # --------------------------------------------------
            # General block scaling (anywhere, modest factor)
            min_len = max(1, n // 200)          # 0.5 %–0.5 % of n
            max_len = max(2, n // 20)           # up to 5 %
            width = int(rng.integers(min_len, max_len + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.15)))
            cand[start : start + width] *= factor

        elif op == 2:
            # --------------------------------------------------
            # Transfer a chunk of mass from a high‑value region to a low‑value region.
            src = int(rng.integers(0, n // 2))          # source in the first half
            dst = int(rng.integers(n // 2, n))          # destination in the second half
            if cand[src] > 0.0:
                amount = cand[src] * rng.uniform(0.01, 0.08)
                cand[src] -= amount
                cand[dst] += amount

        elif op == 3:
            # --------------------------------------------------
            # Small positive bump on a random entry.
            idx = int(rng.integers(0, n))
            bump = max(0.0, rng.normal(0.0, 0.06))
            cand[idx] += bump

        else:
            # --------------------------------------------------
            # Swap two random entries (pure permutation move).
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # Apply the hard bounds required by the evaluator.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Fast evaluation of the proposal.
        new_score = _fast_score(cand, n, fft_len)

        # Acceptance: greedy downhill, occasional uphill when temperature > 0.
        accept = False
        if new_score < cur_score:
            accept = True
        elif temp > 0.0:
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = new_score
            if new_score < best_score:
                best = cand.copy()
                best_score = new_score

    # ----------------------------------------------------------
    # Final validation with the exact evaluator (guard against FFT rounding).
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score) or final_score > best_score:
        # As a safety net, fall back to the analytic start.
        best = analytic.copy()

    return [float(x) for x in best]
