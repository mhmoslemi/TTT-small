def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine n

    best_score = float(evaluate_sequence(best_vec.tolist()))
    
    # Parameter settings
    MAX_G = 6                # max allowed count per sum
    NUM_TRIALS = 50
    EPS_VALUES = np.concatenate([np.array([0.0]), np.logspace(-4, -0.3, 30)])  # 0 to 0.5 approx
    # also include linear low values
    EPS_VALUES = np.unique(np.sort(np.concatenate([EPS_VALUES, np.linspace(0, 0.2, 21)])))
    
    for trial in range(NUM_TRIALS):
        order = rng.permutation(n)
        cnt = np.zeros(2*n-1, dtype=np.int16)  # counts per sum
        selected = []
        # Greedy insertion
        for i in order:
            # fast early break: if cnt.max() >= MAX_G, maybe skip? no.
            # Check contributions
            # self
            s_self = i + i
            if cnt[s_self] + 1 > MAX_G:
                continue
            ok = True
            # loop over existing selected
            for j in selected:
                s = i + j
                if cnt[s] + 2 > MAX_G:
                    ok = False
                    break
            if not ok:
                continue
            # Add i
            selected.append(i)
            cnt[s_self] += 1
            for j in selected[:-1]:  # previously added ones (including j before i)
                s = i + j
                cnt[s] += 2
        m = len(selected)

            continue

        mask = np.zeros(n, dtype=np.float64)
        mask[selected] = 1.0
        for eps in EPS_VALUES:
            cand = mask + eps   # uniform epsilon added to all entries (including masks)
            # ensure clipping bounds 0-1000 (automatically true)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_vec = cand.copy()
                # early termination if below target
                if best_score <= 1.50274349:
                    return [float(x) for x in best_vec]
    
    # fallback: power law sweep (same from previous code)
    p_vals = np.linspace(0.45, 0.80, 61)

import numpy as np

# ------------------------------------------------------------
# Helper: ordered‑pair counts for a binary mask
# ------------------------------------------------------------
def _pair_counts(selected_set, n):
    """
    selected_set : set of indices (0 … n-1) where the coefficient is 1
    returns      : np.ndarray of length 2*n-1 with ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int16)
    idx = list(selected_set)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt

    """
