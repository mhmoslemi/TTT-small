def project_simplex(v):
    # ensure nonnegative and sum to 1
    # using algorithm from https://arxiv.org/abs/1309.1541
    # Input v (numpy array), possibly negative
    # Output w such that w >=0 and sum w = 1, minimizing ||w - v||_2.
    # We can also support sum=1.

Implementation steps:
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    exponential and their product), performs a short block‑perturbation
    local search and finishes with a few gradient‐descent refinements.
    The best candidate encountered is returned.
    """
    # ---- 1.  Initialise from the external parent or a fallback ----
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
    else:
        parent = np.ones(256, dtype=np.float64)            # generic fallback

    n = int(parent.size)
    total = parent.sum()
    if total > 0.0:
        a = parent / total                                   # normalise to sum 1
    else:
        a = np.full(n, 1.0 / n)

    # ---- 2.  Helper to evaluate a candidate ----------------------
    def _score(v: np.ndarray) -> float:
        return float(evaluate_sequence(v.tolist()))

    # current best
    best = a.copy()
    best_score = _score(best)

    # ---- 3.  Parametric families ----------------------------------
    idx = np.arange(n, dtype=np.float64)          # 0 … n‑1
    # ----- power‑law ------------------------------------------------
    alphas = np.linspace(0.05, 3.0, 60)           # coarse grid
    best_alpha = None
    for alpha in alphas:
        w = np.power(idx + 1.0, -alpha)
        w = w / w.sum()
        sc = _score(w)
        if np.isfinite(sc) and sc < best_score:
            best, best_score, best_alpha = w.copy(), sc, alpha

    # fine‑tune around the best α
    if best_alpha is not None:
        lo = max(0.001, best_alpha - 0.2)
        hi = best_alpha + 0.2
        fine = np.linspace(lo, hi, 60)
        for alpha in fine:
            w = np.power(idx + 1.0, -alpha)
            w = w / w.sum()
            sc = _score(w)
            if np.isfinite(sc) and sc < best_score:
                best, best_score = w.copy(), sc

    # ----- exponential decay ----------------------------------------
    betas = np.linspace(1e-4, 0.02, 30)
    for beta in betas:
        w = np.exp(-beta * idx)
        w = w / w.sum()
        sc = _score(w)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = w.copy(), sc

    # ----- product of power‑law and exponential --------------------
    for alpha in np.linspace(0.5, 3.0, 8):
        for beta in np.linspace(5e-4, 5e-3, 8):
            w = np.power(idx + 1.0, -alpha) * np.exp(-beta * idx)
            w = w / w.sum()
            sc = _score(w)
            if np.isfinite(sc) and sc < best_score:
                best, best_score = w.copy(), sc

    # ---- 4.  Light stochastic block perturbations ----------------
    rng = np.random.default_rng(seed)
    for _ in range(300):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 200))
        stop = min(n, start + width)

        cand = best.copy()
        # multiplicative log‑normal noise on a contiguous block
        noise = np.exp(rng.normal(0.0, 0.1, size=stop - start))
        cand[start:stop] *= noise
        cand = np.clip(cand, 0.0, None)

        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / n)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand.copy(), sc

    # ---- 5.  Gradient‑descent style refinement (max‑conv reduction) --
    # pre‑compute FFT length (next power of two)
    L = 1 << ((2 * n - 1).bit_length())
    for it in range(500):
        # full convolution via FFT (ordered pairs)
        fa = np.fft.rfft(best, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        k = int(np.argmax(conv))                     # index of the current max

        # gradient of conv[k] w.r.t. a_i is 2 * a_{k‑i} (ordered‑pair count)
        grad = np.zeros_like(best)
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        i_range = np.arange(i0, i1 + 1, dtype=np.int64)
        j_range = k - i_range
        grad[i_range] = best[j_range]                # factor 2 absorbed in step‑size

        # simple projected gradient step (step size decays)
        eta = 0.5 / (1.0 + it * 0.01)
        cand = best - eta * grad
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / n)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand.copy(), sc

        # early stop if we already beat the target bound
        if best_score < 1.50274349:
            break

    # ---- 6.  Return the best candidate found --------------------
    return [float(v) for v in best]
