import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2: C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _build_conv_matrix(vec):
    """Sparse matrix A (2n‑1)×n with A[k,j]=vec[k‑j] for 0≤j≤k<2n‑1."""
    n = vec.size
    # each diagonal i (offset i) contains the constant vec[i]
    diags = [np.full(n, vec[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    return sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')


def propose_candidate(seed=42, **kwargs):
    """Return a height sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --- problem size -------------------------------------------------
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256

    # --- analytical start (binomial half) -----------------------------
    base = _coefficients_binom_half(n)
    # target total sum = sqrt(2n); scaling makes the score equal to max(conv)
    target_sum = np.sqrt(2.0 * n)
    base = base * (target_sum / base.sum())
    np.clip(base, 0.0, 1000.0, out=base)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --- optional parent fallback --------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    target_score = 1.50274349
    max_iters = 5          # a few LP‑mix steps are enough
    mix_factor = 0.5       # convex‑combination weight

    for _ in range(max_iters):
        # current maximal autocorrelation
        conv_best = np.convolve(best, best)
        M = conv_best.max()

        # build linear constraints: (best * g)[k] ≤ M for all k
        A = _build_conv_matrix(best)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # LP: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            break

        g = res.x
        np.clip(g, 0.0, 1000.0, out=g)

        # convex mix of the current best and the LP solution
        cand = (1.0 - mix_factor) * best + mix_factor * g

        s = cand.sum()
        if s <= 0.0:
