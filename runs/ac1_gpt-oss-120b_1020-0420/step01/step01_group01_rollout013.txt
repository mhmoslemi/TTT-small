import numpy as np

# ------------------------------------------------------------
# Helper: compute the AC1 bound using the supplied evaluator.
# ------------------------------------------------------------
def _bound(seq):
    """Return the evaluator score for a NumPy 1‑D array."""
    # The evaluator clips internally, but we do it here for safety.
    seq = np.clip(seq, 0.0, 1000.0)
    # Convert to plain Python list as required by evaluate_sequence.
    return evaluate_sequence(seq.tolist())


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Deterministic re‑allocation: shrink a prefix and inflate the tail
# while preserving the total L1 mass.
# ------------------------------------------------------------
def _reallocate(seq, ratio, factor):
    """Return a new array where the first int(ratio*n) entries are multiplied by factor,
    and the removed mass is spread uniformly over the remaining tail."""
    n = seq.size
    k = max(1, int(n * ratio))
    early = seq[:k]
    tail = seq[k:]

    early_sum = early.sum()
    tail_sum = tail.sum()
    if tail_sum == 0.0:
        return seq  # nothing to do

    new_early = early * factor
    delta = early_sum - new_early.sum()         # mass taken from the prefix
    # distribute the taken mass uniformly over the tail
    new_tail = tail * (1.0 + delta / tail_sum)
    return np.concatenate([new_early, new_tail])


# ------------------------------------------------------------
# Localised hill‑climbing move: move a small fraction of mass from a
# source index that participates in the current maximal convolution
# entry to a random destination outside that set.
# ------------------------------------------------------------
def _hill_move(seq, rng):
    """Perform one stochastic hill‑climbing step. Returns (new_seq, improved)."""
    n = seq.size
    # full convolution to locate the current maximum
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    max_positions = np.flatnonzero(conv == max_val)

    # pick one of the maximal positions
    k = rng.choice(max_positions)

    # all indices i that contribute to conv[k] (i + j = k)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    contributors = np.arange(i_min, i_max + 1)  # each i pairs with j = k - i

    # pick a source among contributors that actually holds some mass
    src_candidates = contributors[seq[contributors] > 1e-12]
    if src_candidates.size == 0:
        return seq, False
    src = rng.choice(src_candidates)

    # pick a destination that is *not* a contributor (to avoid immediate increase of this max)
    non_contrib = np.setdiff1d(np.arange(n), contributors, assume_unique=True)
    if non_contrib.size == 0:
        dst = rng.integers(0, n)     # fallback – any index
    else:
        dst = rng.choice(non_contrib)

    # move a random fraction (1–20 %) of the source mass
    frac = rng.uniform(0.01, 0.20)
    delta = seq[src] * frac
    if delta <= 0.0:
        return seq, False

    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    # enforce the absolute limits required by the evaluator
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)

    # accept only if the bound improves
    new_score = _bound(new_seq)
    old_score = _bound(seq)
    if new_score < old_score:
        return new_seq, True
    return seq, False


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 5000, **kwargs):
    """
    Return a list of non‑negative heights with a low AC1 bound.
    The function works without any external data, but will use the
    runtime variable `height_sequence_1` as an initial candidate when
    it is present and valid.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine the problem size.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        # sanity check – the evaluator will reject NaNs/Inf anyway
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
        else:
            parent = None
            n = 256
    else:
        parent = None
        n = 256

    # --------------------------------------------------------
    # Initialise the candidate.
    # --------------------------------------------------------
    if parent is not None:
        init = np.clip(parent, 0.0, 1000.0)
        init_score = _bound(init)
        # If the parent is somehow invalid for the objective, fall back.
        if not np.isfinite(init_score):
            init = _catalan_seq(n)
            init_score = _bound(init)
    else:
        init = _catalan_seq(n)
        init_score = _bound(init)

    best = init
    best_score = init_score

    # --------------------------------------------------------
    # Deterministic re‑allocation sweep (more extensive than before).
    # --------------------------------------------------------
    # probe a grid of prefix ratios and shrink factors.
    prefix_ratios = np.linspace(0.01, 0.50, 10)          # 1 % … 50 %
    shrink_factors = np.linspace(0.30, 1.00, 15)        # 30 % … 100 %

    for ratio in prefix_ratios:
        for fac in shrink_factors:
            cand = _reallocate(best, ratio, fac)
            cand_score = _bound(cand)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                # continue improving from this new baseline
                # (restart the inner loops with the refined sequence)
                # – we simply keep looping; the grid will be revisited.

    # --------------------------------------------------------
    # Stochastic hill‑climbing refinement.
    # --------------------------------------------------------
    steps = kwargs.get("num_steps", num_steps)
    no_improve = 0
    for _ in range(steps):
        best, improved = _hill_move(best, rng)
        if improved:
            best_score = _bound(best)
            no_improve = 0
        else:
            no_improve += 1
        # optional early stop if stuck for a long time
        if no_improve >= 1000:
            break

    # --------------------------------------------------------
    # Return the result as a plain Python list.
    # --------------------------------------------------------
    return [float(v) for v in best]
