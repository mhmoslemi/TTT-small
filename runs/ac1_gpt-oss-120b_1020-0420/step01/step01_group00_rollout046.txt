import numpy as np

# ------------------------------------------------------------
# Fast AC1 bound (identical to the official evaluator but using FFT)
# ------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    # length for linear convolution (next power of two)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ------------------------------------------------------------
# Helper: clip to allowed interval and renormalise to sum = 1
# ------------------------------------------------------------
def _norm(vec: np.ndarray) -> np.ndarray:
    v = np.clip(vec, 0.0, 1000.0)
    total = v.sum()
    if total < 1e-12:
        v[:] = 1.0
        total = v.sum()
    return v / total


# ------------------------------------------------------------
# Main candidate proposal
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 20000, **kwargs):
    """Search for a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)

    # ---- initialise from the supplied parent if possible ----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        # fallback: a modest length uniform vector
        parent = np.full(1024, 1.0, dtype=np.float64)

    n = parent.size
    candidates = []

    # ---- add the (clipped) parent as a starting candidate ----
    candidates.append(_norm(parent))

    # ---- generate a few analytic shapes (sinusoidal, polynomial) ----
    i = np.arange(n, dtype=np.float64)

    # sinusoid raised to various powers
    base = np.sin(np.pi * (i + 0.5) / n)
    for p in (0.5, 1.0, 2.0, 3.0):
        cand = np.power(base, p)
        candidates.append(_norm(cand))

    # a simple triangular “hat”
    cand = (i + 1) * (n - i)
    candidates.append(_norm(cand))

    # linear ramps (forward and backward)
    candidates.append(_norm(i + 1.0))
    candidates.append(_norm(n - i))

    # pick the best of the cheap analytic candidates
    cur = min(candidates, key=_fast_score).copy()
    cur_score = _fast_score(cur)

    # ------------------------------------------------------------
    # Adam‑style projected gradient descent on the max‑convolution peak
    # ------------------------------------------------------------
    steps = int(kwargs.get('num_steps', num_steps))
    # FFT length (power of two) – constant through the run
    m_fft = 1 << ((2 * n - 1).bit_length())

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.18                      # initial learning rate
    m_vec = np.zeros_like(cur)     # first moment
    v_vec = np.zeros_like(cur)     # second moment

    best = cur.copy()
    best_score = cur_score
    no_improve = 0
    top_peaks = 3                  # number of large convolution entries to use

    for step in range(steps):
        # ---- convolution via FFT (linear, not circular) ----
        fa = np.fft.rfft(cur, n=m_fft)
        conv = np.fft.irfft(fa * fa, n=m_fft)[: 2 * n - 1]
