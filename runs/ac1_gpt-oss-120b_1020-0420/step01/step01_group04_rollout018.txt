import numpy as np

def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2: C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _evaluate(arr):
    """Helper converting ndarray to list and calling the provided evaluator."""
    return float(evaluate_sequence(arr.tolist()))


def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step‑function that aims to minimise the AC1 bound.

    The strategy starts from the analytic (1‑z)^‑1/2 coefficients,
    applies systematic tail‑scalings, then performs a modest random
    block search.  The best sequence found (or a parent fallback) is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size.
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256

    # ----- initialise with analytic coefficients -----
    base = _coefficients_binom_half(n)
    base = np.clip(base, 0.0, 1000.0)

    best = base.copy()
    best_score = _evaluate(best)

    # ----- optional parent fallback -----
    if 'height_sequence_1' in globals():
        try:
            parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)
            parent_score = _evaluate(parent)
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent
                best_score = parent_score
        except Exception:
            pass

    # ----- deterministic tail‑scaling sweep -----
    # fractions of the tail that will be scaled, and a geometric grid of factors.
    tail_fractions = [0.05, 0.10, 0.15, 0.20, 0.25, 0.33, 0.50, 0.66, 0.75, 0.90]
    scale_factors = np.geomspace(1.0, 500.0, num=40)  # from 1 up to 500

            cand = best.copy()

            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _evaluate(cand)

                best_score = sc
