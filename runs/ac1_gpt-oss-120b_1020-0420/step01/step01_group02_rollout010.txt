import numpy as np

# ------------------------------------------------------------
# Helper: generate the half‑power series a_i = binom(2i,i)/4^i
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Simple random block perturbation (stays O(n) memory)
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    if n <= 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))   # mild log‑normal factor
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Transfer mass from an early prefix to a late suffix.
# This keeps the total sum (approximately) but reshapes the profile.
# ------------------------------------------------------------
def transfer_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = seq.size
    if n <= 2:
        return seq.copy()
    # choose modest prefix / suffix lengths (≤5 % of n)
    max_len = max(1, n // 20)
    p = int(rng.integers(1, max_len + 1))   # prefix length
    q = int(rng.integers(1, max_len + 1))   # suffix length
    # scaling factor for the early part (reduce, never below 0.5)
    factor = float(0.5 + 0.5 * rng.random())   # ∈[0.5,1.0)
    cand = seq.copy()
    # reduce the prefix
    prefix = cand[:p]
    before = prefix.sum()
    prefix *= factor
    after = prefix.sum()
    reduction = before - after
    # distribute the removed mass uniformly over the suffix
    if reduction > 0.0:
        delta = reduction / q
        cand[-q:] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
