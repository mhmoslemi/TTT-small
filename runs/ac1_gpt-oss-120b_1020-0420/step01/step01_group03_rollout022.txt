import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Root‑constant sequence a_i = C(2i,i)/4**i, clipped to [0,1000]."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _generate_seq_alpha(alpha: float, n: int):
    """Generate the sequence that satisfies
       sum_{i=0}^{k} a_i a_{k-i} = 1  (0 ≤ k < n)
    with a_0 = alpha.  Returns None if any entry becomes negative."""
    if alpha <= 0.0:
        return None
    a = np.empty(n, dtype=np.float64)
    a[0] = alpha
    for k in range(1, n):
        # s = sum_{i=1}^{k-1} a_i * a_{k-i}
        if k == 1:
            s = 0.0
        else:
            # dot of a[1:k] with a[k-1:0:-1] (reversed part)
            s = np.dot(a[1:k], a[k-1:0:-1])
        val = (1.0 - s) / (2.0 * alpha)
        if val < 0.0 or not np.isfinite(val):
            return None
        a[k] = val
    return a
