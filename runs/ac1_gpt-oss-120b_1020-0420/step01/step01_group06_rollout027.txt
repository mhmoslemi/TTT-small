            width = rng.integers(1, max_width + 1)
            stop = min(n, start + width)
            noise = np.exp(rng.normal(0.0, sigma_block, size=stop - start))
            cand[start:stop] *= noise
        elif r < 0.85:
            # Global log‑normal perturbation
            noise = np.exp(rng.normal(0.0, sigma_global, size=n))
            cand *= noise
        else:
            # Transfer a random fraction from one index to another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and cand[i] > 0.0:
                transfer = cand[i] * rng.random() * sigma_swap
                cand[i] -= transfer
                cand[j] += transfer

        # Renormalise (scale‑invariant objective)
        tot = cand.sum()
        if tot > 0.0:
            cand /= tot

        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc
            if best_score < 1.45:  # early stop if clearly good
                break

    # ------------------------------------------------------------------
    # Return the best candidate as a plain list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
