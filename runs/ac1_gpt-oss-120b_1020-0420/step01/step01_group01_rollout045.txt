import numpy as np

# ------------------------------------------------------------
# Convolution (linear, full) via FFT – O(n log n) memory O(n)
# ------------------------------------------------------------
def _conv(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself using FFT."""
    n = seq.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f * f, size)
    return conv


# ------------------------------------------------------------
# Score exactly matching the provided evaluator
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    m = _conv(seq).max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence – used when no parent
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

def propose_candidate(seed: int = 42, num_steps: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights with a low AC1 bound.
    The routine combines a cheap deterministic re‑allocation with a
    targeted hill‑climbing that moves mass from entries contributing most
    to the current maximal autocorrelation toward the far edge (where the
    contribution to that peak is zero).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the supplied parent if present, else Catalan.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = kwargs.get("n", 256)
        seq = _catalan_seq(n)
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.size

    # ------------------------------------------------------------------
    # Keep the best candidate seen so far.
    # ------------------------------------------------------------------
    best_seq = seq.copy()
    best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Deterministic prefix‑shrink / tail‑inflate search (same idea as
    # the earlier program but kept as a fast pre‑filter).
    # ------------------------------------------------------------------
    prefix_ratios = [0.05, 0.10, 0.20, 0.30]
    shrink_factors = np.linspace(0.5, 1.0, 11)   # 0.5 … 1.0 in steps of 0.05
    for r in prefix_ratios:
        k = max(1, int(n * r))          # number of early entries
        early = best_seq[:k]
        tail = best_seq[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()          # mass removed
            g = 1.0 + delta / tail_sum                    # uniform inflate factor
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best_seq = cand
                best_score = sc
                # refresh early / tail for further refinements
                early = best_seq[:k]
                tail = best_seq[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()

    # ------------------------------------------------------------------
    # Hill‑climbing: repeatedly move a small fraction of mass from the

    # ------------------------------------------------------------------
