import numpy as np

def _fast_score(a: np.ndarray) -> float:
    """Fast AC1 bound using FFT. Returns inf if sum(a) < 0.01."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]      # convolution a*a
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * np.max(c) / (S * S)
