def autocorr_fft(a):
    n = a.size
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)  # rfft length size//2+1
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

import numpy as np

# ------------------------------------------------------------
# Helper: generate the (1 - z)^(-1/2) series coefficients
#   a_0 = 1, a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# FFT‑based linear convolution (real input, linear padding)
# ------------------------------------------------------------
def convolve_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Fast surrogate for the evaluator (same formula, FFT conv)
# ------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = convolve_fft(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

        # linear convolution via FFT (cheap) – used only for guidance
        conv = convolve_fft(best)
