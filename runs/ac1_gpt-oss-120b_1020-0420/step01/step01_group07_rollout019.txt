import numpy as np

def _can_add(idx, selected, cnt, max_allowed):
    """Check whether index ``idx`` can be added without exceeding max_allowed."""
    # self‑pair
    if cnt[2 * idx] + 1 > max_allowed:
        return False
    # pairs with existing elements
    for j in selected:
        if cnt[idx + j] + 2 > max_allowed:
            return False
    return True


def _add_index(idx, selected, cnt):
    """Add ``idx`` to ``selected`` and update the ordered‑pair count array."""
    selected.append(idx)
    cnt[2 * idx] += 1                # (idx,idx)
    for j in selected[:-1]:
        s = idx + j
        cnt[s] += 2                 # (idx,j) and (j,idx)


def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise the AC1 bound.
    The routine builds a large binary mask whose ordered‑pair counts never exceed 3.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------
    # Determine problem length and initialise a fallback best candidate
    # --------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0:
            n = int(parent_arr.size)
            best_vec = parent_arr.copy()
            best_score = float(evaluate_sequence(best_vec.tolist()))
        else:
            raise NameError
    except NameError:
        n = 256
        best_vec = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_vec.tolist()))

    TARGET = 1.50274349          # desired bound
    MAX_C = 3                    # maximum allowed ordered‑pair count per sum
    TRIALS = 80                  # random permutations to attempt

    # --------------------------------------------------------------------
    # Random greedy construction of a large binary mask (max count ≤ MAX_C)
    # --------------------------------------------------------------------
    for _ in range(TRIALS):
        order = rng.permutation(n)
        selected = []
        cnt = np.zeros(2 * n - 1, dtype=np.int16)

        for idx in order:
            idx = int(idx)
            if _can_add(idx, selected, cnt, MAX_C):
                _add_index(idx, selected, cnt)
                m = len(selected)
                # bound estimate using current max count (≤ MAX_C)
                bound_est = 2.0 * n * cnt.max() / (m * m)
                if bound_est <= TARGET:
                    cand = np.zeros(n, dtype=np.float64)
                    cand[selected] = 1.0
                    return cand.tolist()

        # full pass finished – evaluate the candidate
        if selected:
            cand = np.zeros(n, dtype=np.float64)
            cand[selected] = 1.0
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_vec = cand
                best_score = score

    # --------------------------------------------------------------------
    # Power‑law sweep as a cheap generic fallback
    # --------------------------------------------------------------------
    p_vals = np.linspace(0.45, 0.80, 71)               # 0.45 … 0.80
    base = np.arange(1, n + 1, dtype=np.float64)      # 1 … n
    for p in p_vals:
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score

    # --------------------------------------------------------------------
    # Tiny random perturbations to polish the best vector found so far
    # --------------------------------------------------------------------
    num_steps = int(kwargs.get('num_steps', 200))
    for _ in range(num_steps):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.05)))   # ≤ ~5 % change

        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score

    # --------------------------------------------------------------------
    # Return the best candidate (as a plain Python list of floats)
    # --------------------------------------------------------------------
    return [float(x) for x in best_vec]
