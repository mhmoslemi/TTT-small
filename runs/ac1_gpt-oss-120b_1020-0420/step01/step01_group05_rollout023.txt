import numpy as np

# ----------------------------------------------------------
# Fast matrix‑free evaluation of the AC1 bound (identical to the
# evaluator but uses FFT for speed).
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                                 # pointwise product
    conv = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------
# Candidate factories – each returns a NumPy array of length n.
# ----------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64)


def _inverse_sqrt(n: int, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + offset)
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p)
    return a


def _exponential(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    return a


def _combined(n: int, p: float, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p) * np.exp(-alpha * i)
    return a
