import numpy as np

def generate_half_power_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def fast_ac1_score(seq, n, fft_len):
    # Clip values to allowed range
    np.clip(seq, 0.0, 1000.0, out=seq)
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    # convolution using FFT
    a_fft = np.fft.rfft(seq, n=fft_len)
    b = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
    max_b = b.max()
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed: int = 42, num_iter: int = 5000, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine length n
    n = 256
    if 'height_sequence_1' in globals():
        try:
            arr_parent = np.asarray(height_sequence_1)
            if arr_parent.ndim == 1 and arr_parent.size > 0:
                n = int(arr_parent.size)
        except Exception:
            pass

    # Base construction: half‑power series (known low AC1)
    base = generate_half_power_series(n)
    np.clip(base, 0.0, 1000.0, out=base)

    # FFT length (next power of two for speed)
    fft_len = 1 << ((2 * n - 1).bit_length())

    best = base.copy()
    best_score = fast_ac1_score(best, n, fft_len)

    half = n // 2
    sigma = 0.15  # log‑normal perturbation width

    for _ in range(num_iter):
        cand = best.copy()
        # choose early and late indices
        i = rng.integers(0, half)
        j = rng.integers(half, n)

import numpy as np

# ------------------------------------------------------------
# Half‑power series (coefficients of (1‑z)^(-1/2))
#   a_0 = 1, a_i = a_{i-1} * (2*i‑1)/(2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Fast evaluation of the AC1 upper bound using FFT.
# Mirrors the evaluator’s logic (clipping, sum‑check, convolution).
# ------------------------------------------------------------
def fast_ac1_score(seq: np.ndarray, n: int, fft_len: int) -> float:
    # clamp to allowed interval
    np.clip(seq, 0.0, 1000.0, out=seq)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    # convolution via FFT (real → real)
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    m = conv.max()
    return 2.0 * n * m / (s * s)


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_iter: int = 5000, **kwargs):
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size and initial sequence.
    # Prefer the parent construction if it exists.
    # --------------------------------------------------------
    n = 256
    if "height_sequence_1" in globals():
        try:
            parent = np.asarray(height_sequence_1, dtype=np.float64)
            if parent.ndim == 1 and parent.size > 0:
                n = int(parent.size)
                base = parent.copy()
            else:
                base = generate_half_power_series(n)
        except Exception:
            base = generate_half_power_series(n)
    else:
        base = generate_half_power_series(n)

    # safety clamp (also matches evaluator’s limits)
    np.clip(base, 0.0, 1000.0, out=base)

    # --------------------------------------------------------
    # Prepare FFT parameters (next power of two for speed)
    # --------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    best = base.copy()
    best_score = fast_ac1_score(best, n, fft_len)

    half = n // 2
    sigma = 0.30          # log‑normal spread for multiplicative moves

    # --------------------------------------------------------
    # Stochastic hill‑climbing:
    # - pick an early index and a late index,
    # - transfer a multiplicative factor between them,
    # - keep the sequence inside the allowed box,
    # - accept only strictly improving candidates.
    # --------------------------------------------------------
    for _ in range(num_iter):
        cand = best.copy()

        i = rng.integers(0, half)        # early (large) coefficient
        j = rng.integers(half, n)        # late  (small) coefficient
        factor = float(np.exp(rng.normal(0.0, sigma)))  # >0

        # move mass from i toward j (overall sum changes slightly)
        cand[i] = cand[i] / factor
        cand[j] = cand[j] * factor

        np.clip(cand, 0.0, 1000.0, out=cand)

        score = fast_ac1_score(cand, n, fft_len)
        if score < best_score:
            best = cand
            best_score = score

    # --------------------------------------------------------
    # Return a plain Python list of floats.
    # --------------------------------------------------------
    return [float(x) for x in best]
