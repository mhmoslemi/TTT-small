import numpy as np

def _fast_score(vec):
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _conv_and_max(vec, m):
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*vec.size-1]
    max_c = conv.max()
    k_max = int(conv.argmax())
    return max_c, k_max, conv

def _generate_exponential(alpha, n):
    # build shape exp(-alpha*i)
    arr = np.exp(-alpha * np.arange(n, dtype=np.float64))
    return arr

def _generate_power(p, n):
    # shape (i+1)^(-p)
    idx = np.arange(1, n+1, dtype=np.float64)
    arr = idx ** (-p)
    return arr

def propose_candidate(seed=42, num_grad_steps=8000, **kwargs):
    rng = np.random.default_rng(seed)
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        # fallback uniform
        parent = np.ones(256, dtype=np.float64)
    # clip to allowed range
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 1e-12:
        parent[:] = 1.0
    parent = parent / parent.sum()  # normalise sum to 1

    n = parent.size
    m = 1 << ((2 * n - 1).bit_length())

    best = parent.copy()
    best_score = _fast_score(best)

    # Phase 1: parametric families
    # Exponential alphas
    alphas = np.logspace(-4, 1, num=30)  # 1e-4 to 10
    for alpha in alphas:
        cand = _generate_exponential(alpha, n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum()
        s = _fast_score(cand)
        if s < best_score:
            best = cand
            best_score = s

    # Power law p values
    ps = np.linspace(0.0, 2.5, num=30)  # 0 to 2.5
    for p in ps:
        cand = _generate_power(p, n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum()
        s = _fast_score(cand)
        if s < best_score:
            best = cand
            best_score = s

    # Phase 2: gradient descent with momentum from best
    cur = best.copy()
    cur_score = best_score
    momentum = np.zeros_like(cur)

    init_eta = 0.2
    init_T = 0.2

    for step in range(num_grad_steps):
        # compute convolution of current
        # using same m
        fa = np.fft.rfft(cur, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
        max_c = conv.max()
        k_max = int(conv.argmax())
        # gradient of max convolution
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        b = np.zeros_like(cur)
        b[mask] = cur[idx[mask]]
        grad = 2.0 * (b - b.mean())
        # momentum update
        momentum = 0.9 * momentum + grad
        eta = init_eta * (1.0 - step / num_grad_steps)
        T = init_T * (1.0 - step / num_grad_steps)

        cand = cur - eta * momentum
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = cur.copy()
            s = cand.sum()
        cand = cand / s

        cand_score = _fast_score(cand)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-16)):
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # occasional random block perturbation
        if (step + 1) % 400 == 0 and rng.random() < 0.1:
            width = int(rng.integers(max(1, n // 250), max(2, n // 50)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            block = cur.copy()
            factor = np.exp(rng.normal(0.0, 0.15))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block = block / block.sum()
            block_score = _fast_score(block)
            if block_score < cur_score:
                cur = block
                cur_score = block_score
                if block_score < best_score:
                    best = block.copy()
                    best_score = block_score

    # final verification using official evaluator
    try:
        final_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(final_score):
            raise ValueError
    except Exception:
        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
        best = fallback

    return [float(v) for v in best]

import numpy as np

# ------------------------------------------------------------
# Fast AC1 upper bound (uses FFT, no extra memory beyond O(n))
# ------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ------------------------------------------------------------
# Families of shapes (parameterised step functions)
# ------------------------------------------------------------
def _gen_exponential(alpha: float, n: int) -> np.ndarray:
    """a_i = exp(-alpha * i)  (i = 0..n-1)"""
    return np.exp(-alpha * np.arange(n, dtype=np.float64))


def _gen_power(p: float, n: int) -> np.ndarray:
    """a_i = (i+1)^(-p)  (i = 0..n-1)"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)
