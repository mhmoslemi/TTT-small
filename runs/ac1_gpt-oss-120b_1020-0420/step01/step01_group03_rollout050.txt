import scipy.optimize as opt
import scipy.sparse as sp

def propose_candidate(seed=42, **kwargs):
    try:
        n = int(len(height_sequence_1))
        f = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 256
        # Compute root constant sequence
        f = np.empty(n, dtype=np.float64)
        f[0] = 1.0
        for i in range(1, n):
            f[i] = f[i-1] * (2*i - 1) / (2*i)
    # Clip for safety
    f = np.clip(f, 0.0, 1000.0)

    # compute M
    conv_f = np.convolve(f, f)
    M = float(np.max(conv_f))

    # Build constraint rows K
    step = max(1, n // 800)  # ~6 for n~4800
    K = list(range(0, 2*n-1, step))
    # Ensure centre and last are included
    centre = n-1
    if centre not in K:
        K.append(centre)
    if (2*n-2) not in K:
        K.append(2*n-2)
    K.sort()
    m = len(K)

    # Build sparse matrix A
    row_inds = []
    col_inds = []
    data = []
    for row_idx, k in enumerate(K):
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # j_start..j_end inclusive
        # coefficient for variable g_j is f[k - j]
        # compute slice
        # Direct loop
        for j in range(j_start, j_end+1):
            row_inds.append(row_idx)
            col_inds.append(j)
            data.append(f[k - j])
    A = sp.csr_matrix((data, (row_inds, col_inds)), shape=(m, n))

    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n
    b = np.full(m, M, dtype=np.float64)

    # Solve LP
    result = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not result.success:
        # fallback to f
        best = f
    else:
        g = result.x
        # Scale g to not exceed max value 1000 after possible scaling later
        # Compute self-convolution bound of g
        conv_g = np.convolve(g, g)
        M_g = float(np.max(conv_g))
        if M_g <= 0:
            lamb = 1.0
        else:
            # Scale to keep max <= M (original bound) to avoid increasing score too much
            lamb = np.sqrt(M / M_g) if M_g > M else 1.0
        candidate = g * lamb

        try:

            score_f = float(evaluate_sequence(f.tolist()))
            best = candidate if score_candidate < score_f else f
        except Exception:
            best = f

    # Return as list

def _root_constant_seq(length: int) -> np.ndarray:
    """Root‑of‑constant sequence a_i = C(2i,i)/4**i (non‑negative, clipped)."""
    seq = np.empty(length, dtype=np.float64)

    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)
