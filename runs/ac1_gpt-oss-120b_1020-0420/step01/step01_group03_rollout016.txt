def conv_full(a, b):
    m = 1 << ((len(a) + len(b) - 2).bit_length())
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)

    return conv

import numpy as np
import math

# -----------------------------------------------------------
# Helper functions
# -----------------------------------------------------------

def _root_constant_seq(length: int) -> np.ndarray:
    """
    First ``length`` terms of a_i = C(2i,i)/4**i.
    This satisfies (a*a)[k] == 1 for 0 ≤ k < length (in exact arithmetic).
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution (full) using FFT. Returns a real array of length
    len(a)+len(b)-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())               # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv


def _greedy_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Greedy construction of a vector g that maximises sum(g) under
    the linear constraints (f * g)[k] ≤ M for all shifts k.
    The algorithm processes indices from the tail (small f_i) to the
    head, always taking the largest admissible increment.
    """
    n = f.shape[0]
    cap = np.full(2 * n - 1, M, dtype=np.float64)   # remaining capacity per shift
    g = np.zeros(n, dtype=np.float64)

    for i in range(n - 1, -1, -1):                   # small f[i] first
        fi = f[i]
        if fi <= 0.0:
            continue
        start = i
        end = i + n
        # minimal residual capacity over the window where f[i] participates
        min_cap = cap[start:end].min()
        if min_cap <= 0.0:
            continue
        delta = min_cap / fi
        if delta <= 0.0:
            continue
        g[i] = delta
        # update the capacities
        cap[start:end] -= fi * delta
        np.maximum(cap, 0.0, out=cap)                # avoid tiny negatives
    return g
