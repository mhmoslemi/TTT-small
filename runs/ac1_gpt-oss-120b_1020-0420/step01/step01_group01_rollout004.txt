import numpy as np

# ------------------------------------------------------------
# Helpers
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    """Catalan‑like central‑binomial sequence a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


def _smooth_seq(seq: np.ndarray, window: int) -> np.ndarray:
    """Uniform moving‑average (mode='same') with integer window size."""
    if window <= 1:
        return seq.copy()
    kernel = np.ones(window, dtype=np.float64) / window
    return np.convolve(seq, kernel, mode='same')
