    coeffs = np.empty(n, dtype=np.float64)

    return coeffs

import numpy as np

def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2  (C_i = comb(2i,i)/4^i)."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs

def _conv_max(arr):
    """Maximum entry of the full autocorrelation (conv with itself)."""
    return np.max(np.convolve(arr, arr))

def propose_candidate(seed=42, num_steps=12000, **kwargs):
    """Return a non‑negative height sequence with a low AC1 bound.

    The routine starts from the analytic (1‑z)^‑1/2 coefficients, optionally
    keeps a provided parent, and then performs a mix of random and
    targeted moves.  A dedicated move reduces the first entry (which heavily
    influences the convolution peak) and expands a tail block as far as the
    current maximal convolution permits.  Accepted moves strictly improve the
    evaluated score.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # Determine problem size and obtain an initial candidate.
    # ------------------------------------------------------------------ #
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        n = 256
        parent = None

    analytic = np.clip(_coefficients_binom_half(n), 0.0, 1000.0)

    # pick the best among analytic and optional parent
    best = analytic.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    best_max = _conv_max(best)

    if parent is not None:
        try:
            sc = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = parent.copy()
                best_score = sc
                best_max = _conv_max(best)
        except Exception:
            pass

    # ------------------------------------------------------------------ #
    # Search parameters
    # ------------------------------------------------------------------ #
    target = 1.50274349
    head_idx = 0                     # position we will shrink
    tail_len = max(1, n // 20)       # 5 % of the sequence, used for expansion

    # ------------------------------------------------------------------ #
    # Main optimisation loop
    # ------------------------------------------------------------------ #
    for step in range(num_steps):
        if best_score < target:
            break

        cand = best.copy()
        op = rng.integers(0, 5)

        if op == 0:
            # --- strategic head reduction + tail expansion -----------------
            # shrink a single leading entry (most influential for the peak)
            h = float(np.exp(rng.normal(-0.15, 0.05)))   # around 0.86
            cand[head_idx] *= h

            # after shrinking, the peak cannot exceed the previous best peak
            max_allowed = best_max * (1.0 + 1e-6)

            # binary search the largest tail scaling that respects the bound
            low, high = 1.0, 6.0
            best_tail = cand.copy()          # feasible with scale = 1
            for _ in range(12):

            cand = best_tail

            best_score = score
