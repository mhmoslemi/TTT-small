    n = a.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m-1).bit_length())

    import time

    rng = np.random.default_rng(seed)

    # fallback parent

            raise ValueError
    except Exception:
        parent_arr = np.ones(256, dtype=np.float64)

    n = int(parent_arr.size)
    parent_arr = np.clip(parent_arr, 0.0, 1000.0)

    # fast score helper
    def _fast_score(a):
        n_local = a.shape[0]
        m = 2 * n_local - 1
        nfft = 1 << ((m - 1).bit_length())
        fa = np.fft.rfft(a, n=nfft)
        c = np.fft.irfft(fa * fa, n=nfft)[:m]
        M = np.max(c)
        S = np.sum(a)
        if S < 0.01:
            return float('inf')
        return 2.0 * n_local * M / (S * S)

    best_seq = np.ones(n, dtype=np.float64)  # uniform fallback
    best_score = _fast_score(best_seq)

    # evaluate parent
    if np.sum(parent_arr) >= 0.01:

        if s < best_score:
            best_seq = parent_arr.copy()
            best_score = s

    # base index array
    i = np.arange(n, dtype=np.float64)

    # candidate families
    # power laws
    for p in np.linspace(0.35, 0.70, 15):
        a = (i + 1.0) ** (-p)
        s = _fast_score(a)
        if s < best_score:

        # reversed
        a_rev = a[::-1]
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq = a_rev; best_score = s
        # mix with reverse
        for t in (0.2, 0.5, 0.8):
            mixed = (1.0 - t) * a + t * a_rev
            s = _fast_score(mixed)

import numpy as np
import time

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    conv_full = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(conv_full)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)
