def compute_gradient(a):
    n = a.size
    # compute convolution via FFT
    # pad length to next power of two
    N = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)[:2*n-1]
    M = conv.max()
    # find indices where conv == M (within small tolerance)
    # using np.isclose may be robust:
    eps = 1e-12 * M
    max_idxs = np.where(conv >= M - eps)[0]
    g = np.zeros_like(a)
    for k in max_idxs:
        lo = max(0, k - (n-1))
        hi = min(n-1, k) + 1
        i = np.arange(lo, hi)
        j = k - i
        g[i] += 2 * a[j]
    S = a.sum()
    C = 2.0 * n
    grad = C * (g / (S*S) - (2.0 * M) / (S**3))
    return grad, M, S, conv
```

def fft_convolve(x, y, out_len=None):
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    result = np.fft.irfft(X * Y, n=N)[:n]
    if out_len is not None:
        return result[:out_len]
    return result

```
def _gradient_step(a, target_sum, eta, rng):
    n = a.size
    # compute convolution
    Nfft = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(fa * fa, n=Nfft)[:2*n - 1]
    M = conv.max()
    # tolerance
    tol = 1e-12 * max(1.0, M)
    active = np.where(conv >= M - tol)[0]
    if active.size == 0:
        return a, M
    k = rng.choice(active)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k) + 1
    # vectorized
    idx = np.arange(lo, hi, dtype=np.int64)
    jdx = k - idx
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[jdx]
    S = a.sum()
    C = 2.0 * n
    grad = C * (g / (S * S) - (2.0 * M) / (S ** 3))
    new_a = a - eta * grad
    new_a = np.maximum(new_a, 0.0)
    sum_new = new_a.sum()
    if sum_new > 0:
        new_a *= (target_sum / sum_new)
    # clip values
    new_a = np.clip(new_a, 0.0, 1000.0)
    return new_a, M
```

import numpy as np

# ------------------------------------------------------------
# Basic utilities – all operations are O(n log n) or linear
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    """Catalan–like sequence a_i = C(2i,i) / 4^i for i=0..n-1 (non‑negative)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


def _full_conv_fft(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself using a power‑of‑two FFT."""
    n = x.size
    length = 2 * n - 1
    N = 1 << ((length - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=N)
    conv = np.fft.irfft(fx * fx, n=N)[:length]
    return conv


def _score_fast(x: np.ndarray) -> float:
    """Exact AC1 bound (same as evaluator) but using FFT for speed."""
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return np.inf
    n = x.size
    conv = _full_conv_fft(x)
    m = conv.max()
    return float(2.0 * n * m / (s * s))


def _gradient_step(a: np.ndarray, target_sum: float, eta: float,
                   rng: np.random.Generator) -> np.ndarray:
    """Perform one sub‑gradient descent step on the AC1 bound."""
    n = a.size
    conv = _full_conv_fft(a)
    M = conv.max()

    # pick a random active index (where the max is attained)
    tol = 1e-12 * max(1.0, M)
    active = np.where(conv >= M - tol)[0]
    if active.size == 0:                     # should not happen
        return a
    k = rng.choice(active)

    # sub‑gradient of the max‑convolution term
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k) + 1
    idx = np.arange(lo, hi, dtype=np.int64)
    jdx = k - idx
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[jdx]                     # derivative of conv_k w.r.t. a

    S = a.sum()
    C = 2.0 * n
    # gradient of the whole bound (positive = increase bound)
    grad = C * (g / (S * S) - (2.0 * M) / (S ** 3))

    # descent step, keep non‑negative, then renormalise the ℓ1‑norm
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new > 0.0:
        a_new *= (target_sum / s_new)
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

    The function respects the optional ``num_steps`` argument (default 2000).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size – use the provided parent if possible.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)

        a = _catalan_seq(256)

    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    target_sum = a.sum()
    if target_sum < 0.01:                     # sanity – should never happen
        a = np.full(n, 1.0, dtype=np.float64)
        target_sum = a.sum()

    # --------------------------------------------------------
    # Initial best candidate
    # --------------------------------------------------------
    best = a.copy()
    best_score = _score_fast(best)

    # --------------------------------------------------------
    # Parameters for the optimisation loop
    # --------------------------------------------------------
    max_iters = int(kwargs.get("num_steps", 2000))
    eta = 0.5                                 # initial step length

    for it in range(max_iters):
        # schedule step size: halve every 500 iterations
        if it and it % 500 == 0:
            eta *= 0.5

        # ---- gradient descent step ----
        a = _gradient_step(a, target_sum, eta, rng)

        # ---- occasional random re‑allocation (helps escape plateaus) ----
        if it % 100 == 0:
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src != dst and a[src] > 0.0:
                delta = a[src] * rng.uniform(0.01, 0.05)
                a[src] -= delta
                a[dst] += delta
                a = np.clip(a, 0.0, 1000.0)
                s = a.sum()
                if s > 0.0:
                    a *= (target_sum / s)

        if sc < best_score:
