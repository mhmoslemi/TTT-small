```python
def _fast_score(a):
    n = a.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=nfft)[:m]
    M = conv.max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))
```
