def directed_update(a, lr, rng):
    n = len(a)
    b = np.convolve(a, a)
    k = int(np.argmax(b))
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    w = np.zeros(n, dtype=a.dtype)
    idx = np.arange(lo, hi+1)
    w[idx] = a[k - idx]
    # Determine high and low index sets
    # choose quantiles
    high_mask = w > np.quantile(w, 0.9)   # top 10%
    low_mask = w < np.quantile(w, 0.1)    # bottom 10%
    # Ensure at least one element each
    if not high_mask.any() or not low_mask.any():
        return a  # no change
    # reduction amount: total reduction = lr * sum(a[high])
    total_high = np.sum(a[high_mask])
    total_low = np.sum(a[low_mask])
    # compute delta to move from high to low proportionally
    # we will subtract fraction from high, add fraction to low
    delta = lr * total_high / np.count_nonzero(high_mask)   # per-element reduction
    # reduce high
    cand = a.copy()
    cand[high_mask] = np.maximum(cand[high_mask] - delta, 0.0)
    # compute amount moved = total_before - total_after from high
    moved = total_high - np.sum(cand[high_mask])
    # distribute moved equally to low
    inc = moved / np.count_nonzero(low_mask)
    cand[low_mask] = np.minimum(cand[low_mask] + inc, 1000.0)
    return cand
```

```
import numpy as np

def generate_half_power_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1,length):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def compute_score(seq):
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * seq.size * max_b / (sum_a * sum_a)

def propose_candidate(seed: int = 42, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine length n
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, np.ndarray):
        n = int(height_sequence_1.shape[0])
    else:
        n = 256

    base = generate_half_power_series(n)
    np.clip(base, 0.0, 1000.0, out=base)

    best = base.copy()
    best_score = compute_score(best)

    # precompute linear tilt vector t in [-1,1]
    idx = np.arange(n, dtype=np.float64)
    t = (idx - (n - 1) / 2.0) / ((n - 1) / 2.0)   # -1 ... 1

    # Search over linear tilt
    for lam in np.linspace(0.0, 0.8, 17):   # 0,0.05,...0.8
        # linear tilt
        cand = base * (1.0 + lam * t)
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = compute_score(cand)
        if score < best_score:
            best = cand
            best_score = score

        # power tilt (increase later entries)
        power = ((idx + 1.0) / n) ** lam
        cand2 = base * power
        np.clip(cand2, 0.0, 1000.0, out=cand2)
        score2 = compute_score(cand2)
        if score2 < best_score:
            best = cand2
            best_score = score2

    # optional random refinement around best
    # use block perturbations with decreasing factor
    max_width = max(1, n // 20)
    for _ in range(300):
        # pick random block
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        # factor lognormal centered at 1 with modest sigma
        factor = float(np.exp(rng.normal(0.0, 0.05)))
        cand = best.copy()
        cand[start:start+width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = compute_score(cand)
        if score < best_score:
            best = cand
            best_score = score

    # final fallback: ensure non-empty and finite
    if best_score == np.inf or not np.isfinite(best_score):
        best = base
    return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return the (1‑z)^(-1/2) coefficients a_i = binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def compute_score(seq: np.ndarray) -> float:
    """Calculate the AC1 upper bound for a candidate sequence."""
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    n = seq.size
    return 2.0 * n * max_b / (sum_a * sum_a)


def linear_tilt(base: np.ndarray, lam: float) -> np.ndarray:
    """Tilt the sequence linearly (shift mass from the start to the end)."""
    n = base.size
    t = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / ((n - 1) / 2.0)  # -1 … 1
    cand = base * (1.0 + lam * t)
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def power_tilt(base: np.ndarray, lam: float) -> np.ndarray:
    """Weight the tail with a power law ((i+1)/n)**lam."""
    n = base.size
    power = ((np.arange(n, dtype=np.float64) + 1.0) / n) ** lam
    cand = base * power
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def shift_mass_forward(base: np.ndarray, m: int, frac: float) -> np.ndarray:
    """
    Reduce the first ``m`` entries by ``frac`` (multiply by 1‑frac)
    and redistribute the removed mass uniformly to the remaining entries.
    """
    n = base.size
    if m <= 0 or m >= n:
        return base.copy()
    early = base[:m]
    later = base[m:]
    sum_early = early.sum()
    sum_later = later.sum()
    if sum_later == 0.0:
        return base.copy()
    reduction = frac * sum_early
    new_early = early * (1.0 - frac)
    scale = 1.0 + reduction / sum_later
    new_later = later * scale
    cand = np.concatenate([new_early, new_later])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def shift_mass_backward(base: np.ndarray, m: int, frac: float) -> np.ndarray:
    """
    Reduce the last ``m`` entries by ``frac`` and give the mass to the prefix.
    (mirror of ``shift_mass_forward``)
    """
    n = base.size
    if m <= 0 or m >= n:
        return base.copy()
    early = base[: n - m]
    later = base[n - m :]
    sum_early = early.sum()
    sum_later = later.sum()
    if sum_early == 0.0:
        return base.copy()
    reduction = frac * sum_later
    new_later = later * (1.0 - frac)
    scale = 1.0 + reduction / sum_early
    new_early = early * scale
    cand = np.concatenate([new_early, new_later])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def random_block_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = arr.size
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand = arr.copy()
    cand[start : start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size (use the parent sequence if present)
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):
        n = int(height_sequence_1.size)
    else:
        n = 256

    # Baseline: the (1‑z)^(-1/2) power series
    base = generate_half_power_series(n)
    np.clip(base, 0.0, 1000.0, out=base)

    best = base.copy()
    best_score = compute_score(best)

    # --------------------------------------------------------
    # 1. Systematic parametric families
    # --------------------------------------------------------
    # Linear tilt (move mass towards the end)
    for lam in np.linspace(0.0, 0.8, 17):  # 0 … 0.8 in steps of 0.05
        cand = linear_tilt(base, lam)
        score = compute_score(cand)
        if score < best_score:
            best, best_score = cand, score

    # Power tilt (grow the tail)
    for lam in np.linspace(0.0, 1.0, 11):  # 0 … 1.0
        cand = power_tilt(base, lam)
        score = compute_score(cand)
        if score < best_score:
            best, best_score = cand, score

    # Shift mass from the head to the tail
    for m_frac in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        m = max(1, int(n * m_frac))
        for frac in (0.2, 0.4, 0.6, 0.8):
            cand = shift_mass_forward(base, m, frac)
            score = compute_score(cand)
            if score < best_score:
                best, best_score = cand, score

    # Shift mass from the tail to the head (mirror operation)
    for m_frac in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        m = max(1, int(n * m_frac))
        for frac in (0.2, 0.4, 0.6, 0.8):
            cand = shift_mass_backward(base, m, frac)
            score = compute_score(cand)
            if score < best_score:
                best, best_score = cand, score

    # --------------------------------------------------------
    # 2. Stochastic refinement around the current best
    # --------------------------------------------------------
    max_width = max(1, n // 20)
    for _ in range(300):
        # occasional aggressive block, occasional gentle
        if rng.random() < 0.6:
            cand = random_block_perturbation(best, rng)
        else:
            # tiny multiplicative tweak on a random element
            idx = int(rng.integers(0, n))
            factor = float(np.exp(rng.normal(0.0, 0.03)))
            cand = best.copy()
            cand[idx] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = compute_score(cand)
        if score < best_score:
            best, best_score = cand, score

    # Final safety: ensure we return a valid finite candidate
    if not np.isfinite(best_score):
        best = base.copy()
    return [float(x) for x in best]
