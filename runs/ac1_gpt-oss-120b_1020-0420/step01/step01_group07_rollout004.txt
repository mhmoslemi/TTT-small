        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            s = i + idx[b]
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt


# ------------------------------------------------------------
# Build a set of indices keeping every sum count ≤ max_allowed
# ------------------------------------------------------------
def _build_set(max_allowed, rng, n):
    """
    Greedy construction of a sparse index set.
    Returns the list of chosen indices and the final count array.
    """
    selected = []                     # list of kept indices
    cnt = np.zeros(2 * n - 1, dtype=np.int32)

    perm = rng.permutation(n)        # random order of candidates
    for i in perm:
        # self‑contribution check
        if cnt[2 * i] + 1 > max_allowed:
            continue

        # pair‑contribution checks
        ok = True
        for j in selected:
            if cnt[i + j] + 2 > max_allowed:
                ok = False
                break
        if not ok:
            continue

        # accept index i
        for j in selected:
            cnt[i + j] += 2
        cnt[2 * i] += 1
        selected.append(i)

    return selected, cnt


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed=42, num_attempts=200, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that seeks to
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ---- problem size -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
        else:
            n = 256
    except NameError:
        n = 256

    # ---- simple uniform fallback (guaranteed feasible) ---------------
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))

    TARGET = 1.50274349
    best_est = best_score
    best_selected = None

    # ---- repeated sparse constructions --------------------------------
    for _ in range(num_attempts):
        max_allowed = int(rng.integers(2, 6))          # try 2‒5
        selected, cnt = _build_set(max_allowed, rng, n)
        m = len(selected)
        if m < 2:                                     # need at least two points
            continue
        max_c = int(cnt.max())
        est = float(2.0 * n * max_c / (m * m))        # exact bound for binary mask
        if est < best_est:
            best_est = est
            best_selected = selected
            if best_est <= TARGET:
                break

    # ---- if a good sparse set was found, materialise it -------------
    if best_selected is not None:
        cand = np.zeros(n, dtype=np.float64)
        cand[best_selected] = 1.0
        best_vec = cand
        best_score = best_est

    # ---- secondary cheap fallback: power‑law family -------------------
    if best_score > TARGET:
        base = np.arange(1, n + 1, dtype=np.float64)
        for p in np.linspace(0.45, 0.80, 71):         # 0.45 … 0.80 step 0.005
            cand = base ** (-p)
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_score:
                best_score = score
                best_vec = cand
                if best_score <= TARGET:
                    break

    return [float(x) for x in best_vec]
