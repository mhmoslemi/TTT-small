import numpy as np

def _root_constant_seq(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(a, 0.0, 1000.0)


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    # eliminate tiny negative rounding artefacts
    if conv.min() < 0:
        np.maximum(conv, 0.0, out=conv)
    return conv


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step function (list of floats) whose AC1
    upper‑bound score is intended to beat the root‑constant baseline.
    """
    # ------------------------------------------------------------------
    # 1. Initialise ``a`` – try to reuse the parent if present, otherwise
    #    fall back to the classic root‑constant sequence.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        n = 256
        a = _root_constant_seq(n)

    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:                     # safety net
        a = _root_constant_seq(n)

    # ------------------------------------------------------------------
    # 2. Helper to compute the AC1 score directly (avoids an extra call
    #    to ``evaluate_sequence`` during the search).
    # ------------------------------------------------------------------
    def _score(arr: np.ndarray, conv: np.ndarray) -> float:
        return 2.0 * n * conv.max() / (arr.sum() ** 2)

    # ------------------------------------------------------------------
    # 3. Gradient‑descent style refinement that preserves the total sum.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    conv = _full_convolution(a)
    best_score = _score(a, conv)

    max_iters = 400
    for _ in range(max_iters):
        conv = _full_convolution(a)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # gradient of the active (max) convolution entry
        g = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        if hi >= lo:
            idx = np.arange(lo, hi + 1)
            g[idx] = 2.0 * a[kmax - idx]

        # direction that reduces the max while keeping the sum unchanged
        avg_g = g.mean()
        d = -g + avg_g                     # sum(d) == 0

        step = 0.5 * step_max                # safe fraction
        if step <= 0 or not np.isfinite(step):

        a += step * d
        np.clip(a, 0.0, 1000.0, out=a)

        conv = _full_convolution(a)
        cur_score = _score(a, conv)
        if cur_score < best_score:
