def fast_score(vec):
    n = vec.size
    # compute convolution via FFT
    # length = 2*n-1
    # use np.fft.rfft, zero-pad to next power of two for speed.
    size = 1 << (2*n-1).bit_length()
    # but for small n, we can just use np.fft.fft to size.
    # Actually use np.fft.rfft size.
    f = np.fft.rfft(vec, size)
    c = np.fft.irfft(f * f, size)[:2*n-1]  # real result
    max_b = np.max(c)
    sum_a = np.sum(vec)
    if sum_a < 1e-12:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

def normalize(vec):
    total = vec.sum()
    if total <= 0:
        return vec
    return vec / total

def fast_score(vec):
   n = vec.size
   # using next power of two
   L = 1 << ((2*n-1).bit_length())
   # FFT using rfftn for real -> use rfft
   f = np.fft.rfft(vec, L)
   conv = np.fft.irfft(f * f, L)[:2*n-1]
   maxb = np.max(conv)
   sum_a = np.sum(vec)
   if sum_a == 0:
       return np.inf
   return float(2 * n * maxb / (sum_a*sum_a))

def generate_powerlaw(alpha, n):
   idx = np.arange(n, dtype=np.float64) + 1
   w = np.power(idx, -alpha)
   return normalize(w)

def generate_exponential(beta, n):
   w = np.exp(-beta * np.arange(n, dtype=np.float64))
   return normalize(w)

def generate_random_desc(n, rng):
   # generate random positive numbers (exponential)
   w = rng.exponential(scale=1.0, size=n)
   w = np.sort(w)[::-1]  # descending
   return normalize(w)

def local_mutation(vec, sigma, rng):
   n = vec.size
   start = rng.integers(0, n)
   max_width = max(2, n // 20)  # up to 5% of n
   width = rng.integers(1, max_width+1)
   stop = min(n, start+width)
   cand = vec.copy()
   # multiply block
   factor = np.exp(rng.normal(0.0, sigma, size=stop-start))
   cand[start:stop] *= factor
   # renorm
   return normalize(cand)

def global_mutation_random(vec, sigma, rng):
   n = vec.size
   # elementwise lognormal perturbation
   factor = np.exp(rng.normal(0.0, sigma, size=n))
   cand = vec * factor
   # optionally sort descending to keep shape monotonic? Might improve.
   cand = np.sort(cand)[::-1]
   return normalize(cand)

import numpy as np
import math

def _normalize(v):
    s = v.sum()
    if s <= 0:
        return v
    return v / s

def _fast_score(v):
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(v, L)
    conv = np.fft.irfft(ft * ft, L)[: 2 * n - 1]
    maxb = np.max(conv)
    s = v.sum()
    if s <= 0:
        return np.inf
    return float(2 * n * maxb / (s * s))

def _powerlaw(alpha, n):
    idx = np.arange(n, dtype=np.float64) + 1.0
    w = np.power(idx, -alpha)
    return _normalize(w)

def _exponential(beta, n):
    w = np.exp(-beta * np.arange(n, dtype=np.float64))
    return _normalize(w)

def _random_descending(n, rng):
    w = rng.exponential(scale=1.0, size=n)
    w = np.sort(w)[::-1]
    return _normalize(w)

def _local_mutation(v, sigma, rng):
    n = v.size
    start = rng.integers(0, n)

    cand = v.copy()
    factor = np.exp(rng.normal(0.0, sigma, size=stop - start))
    cand[start:stop] *= factor
    return _normalize(cand)

def _global_mutation(v, sigma, rng):

import numpy as np
import math

def _normalize(v: np.ndarray) -> np.ndarray:
    s = v.sum()
    return v / s if s > 0 else v

def _fast_score(v: np.ndarray) -> float:
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(v, L)
    conv = np.fft.irfft(ft * ft, L)[: 2 * n - 1]
    maxb = np.max(conv)
    s = v.sum()
    return float(2 * n * maxb / (s * s)) if s > 0 else np.inf

def _powerlaw(alpha: float, n: int) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    w = np.power(idx, -alpha)
    return _normalize(w)

def _exponential(beta: float, n: int) -> np.ndarray:
    w = np.exp(-beta * np.arange(n, dtype=np.float64))
    return _normalize(w)

def _random_descending(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.exponential(scale=1.0, size=n)
    w = np.sort(w)[::-1]
    return _normalize(w)

def _local_mutation(v: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    n = v.size
    start = rng.integers(0, n)
    max_width = max(2, n // 20)
    width = rng.integers(1, max_width + 1)
    stop = min(n, start + width)
    cand = v.copy()
    factor = np.exp(rng.normal(0.0, sigma, size=stop - start))
    cand[start:stop] *= factor
    return _normalize(cand)

def _global_mutation(v: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    factor = np.exp(rng.normal(0.0, sigma, size=v.size))
    cand = v * factor
    cand = np.sort(cand)[::-1]            # keep decreasing order
    return _normalize(cand)

def _shuffle_mutation(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = v.copy()
    rng.shuffle(cand)
    return _normalize(cand)
