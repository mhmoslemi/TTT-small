import numpy as np

def _coeff_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2 : C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _score(seq: np.ndarray) -> float:
    """
    Compute the AC1 bound exactly as evaluate_sequence does,
    but without the safety overhead of converting to a list.
    """
    # enforce the same clipping used by the evaluator
    np.clip(seq, 0.0, 1000.0, out=seq)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    n = seq.shape[0]
    return float(2.0 * n * max_b / (s * s))

        idx_reduce = i_big if best[i_big] >= best[j_big] else j_big

        # target position: the mirror index, i.e. farthest from the source
        idx_target = n - 1 - idx_reduce
        if idx_target == idx_reduce:
            # fall back to the globally smallest entry
            idx_target = int(best.argmin())
            if idx_target == idx_reduce:
                # second smallest if we hit the same index
                idx_target = int(best.argsort()[1])

        # move a fraction of the source value to the target
        move_frac = 0.07  # 7 % per successful step
        delta = best[idx_reduce] * move_frac
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[idx_reduce] -= delta
        cand[idx_target] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        new_score = _score(cand.copy())
        if np.isfinite(new_score) and new_score < best_score:
            best = cand
            best_score = new_score
            continue

        # Occasionally try a random small transfer to escape flat spots
        if rng.random() < 0.15:
            i_rand, j_rand = rng.integers(0, n, size=2)
            if i_rand != j_rand and best[i_rand] > 0.0:
                delta_rand = best[i_rand] * 0.05
                cand2 = best.copy()
                cand2[i_rand] -= delta_rand
                cand2[j_rand] += delta_rand
                np.clip(cand2, 0.0, 1000.0, out=cand2)
                score2 = _score(cand2.copy())
                if np.isfinite(score2) and score2 < best_score:
                    best = cand2
                    best_score = score2

    # Return a plain Python list of floats as required by the harness
    return [float(x) for x in best]
```
