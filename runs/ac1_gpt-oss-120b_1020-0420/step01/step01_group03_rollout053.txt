import numpy as np

# -----------------------------------------------------------------------------
# Helper: generate the classic root‑constant sequence (a_i = C(2i,i)/4**i)
# -----------------------------------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    # Clamp to the evaluator's limits
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq

def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 upper

    construction is present) and adds a modest uniform bump to a collection

    raises the total L1 mass significantly while increasing the maximal
    convolution value by only the size of the bump (plus a tiny quadratic
    correction), which is enough to beat the target.
    """
    rng = np.random.RandomState(seed)

    # ----------------------------------------------------------------------
    # 1. Obtain the base sequence (root‑constant).  Use the supplied parent
    #    if it exists; otherwise generate one of modest size.
    # ----------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, copy=True, dtype=np.float64)
        n = base.shape[0]
    except Exception:
        n = 256
        base = _root_constant_seq(n)

    # ----------------------------------------------------------------------
    # 2. Baseline score (kept for fallback)
    # ----------------------------------------------------------------------
    try:
        best_score = float(evaluate_sequence(base.tolist()))
    except Exception:
        best_score = float('inf')
    best_seq = base.copy()

    # ----------------------------------------------------------------------
    # 3. Parameters of the augmentation
    # ----------------------------------------------------------------------
    TARGET = 1.50274349          # required upper bound
    MAX_MOD = min(n, 300)        # maximum number of positions we will touch
    # evenly spaced indices, deliberately skipping the very first entries
    # (modifying a0 would raise the peak by >delta because of the a0² term).
    start_offset = max(1, n // 1000)   # tiny offset to stay clear of index 0
