def compute_score(a):
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())  # next power of two greater or equal to 2n-1
    fft_a = np.fft.rfft(a, m)
    conv = np.fft.irfft(fft_a * fft_a, m)[:2*n-1]
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_conv / (sum_a * sum_a)

import numpy as np

# ------------------------------------------------------------
# Helper: a compact half‑power series (the classic baseline)
# ------------------------------------------------------------
def half_power_series(length: int) -> np.ndarray:
    """Generate a_i = binom(2i,i)/4**i for i=0..length‑1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Fast convolution‑based AC1 score (scale‑invariant)
# ------------------------------------------------------------
def ac1_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq))/sum(seq)^2 using FFT (no clipping)."""
    n = seq.shape[0]
    # length for a power‑of‑two FFT covering 2n‑1 points
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    max_conv = np.max(conv)
    s = np.sum(seq)
    return float("inf") if s <= 0 else 2.0 * n * max_conv / (s * s)

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iter: int = 1200, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    lower the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Initialise from the provided parent if present,
    # otherwise fall back to the classical half‑power series.
    # --------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(
        height_sequence_1, np.ndarray
    ):
        a = height_sequence_1.astype(np.float64).copy()
    else:
        n = 256
        a = half_power_series(n)

    # Safety clamp – evaluator also clips, but we keep the same range.
    np.clip(a, 0.0, 1000.0, out=a)

    n = a.shape[0]
    best = a.copy()
    best_score = ac1_score(best)

    # --------------------------------------------------------
    # Iterative improvement:
    #   – locate the shift k where the convolution attains its maximum,
    #   – move a small fraction of mass from an index that contributes
    #     heavily to that maximum to an index that does not.
    #   – occasionally perform a random block perturbation to escape
    #     shallow local minima.
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())          # FFT size, reused each iteration
    eps = 0.2                                      # initial step size (fraction of source)

    for it in range(max_iter):
        # --- convolution of the current best sequence ---
        fa = np.fft.rfft(best, L)
        conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # --- indices that participate in the k_max sum ---
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
        partner = best[k_max - idx]                # a_{k_max - i}
        contrib = best[idx] * partner               # contribution of each i to b_{k_max}

        # source index: one with the largest contribution (and non‑zero)
        src_local = int(np.argmax(contrib))
        i_src = idx[src_local]

        # destination index: any index outside the support of k_max
        mask_outside = np.ones(n, dtype=bool)
        mask_outside[i_low : i_high + 1] = False

        if outside.size == 0:

                i_dst = rng.integers(0, n)
        else:

        # Transfer a fraction of the source mass to the destination

        cand = best.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        # Guard against accidental negativity (should not happen)
        if cand[i_src] < 0:

        np.clip(cand, 0.0, 1000.0, out=cand)

    # --------------------------------------------------------
