def propose_candidate(seed: int = 42, **kwargs):

    as possible.  The routine builds a series of simple parametric families
    (power‑law, exponential) and then performs a modest number of local
    perturbations.  Candidates are scored with an FFT‑based convolution
    (O(n log n)) which is much faster than the naïve O(n²) convolution
    used by the evaluator.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1.  Parent (fallback) sequence -------------------------------------------------
    # --------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():

    if parent is None:

    n = parent.size

    # 2.  Helper: fast convolution‑based scoring ------------------------------------
    # --------------------------------------------------------------
    # length for FFT – next power of two that accommodates full convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    def _score(vec: np.ndarray) -> float:
        """Return 2·n·max(conv(vec,vec)) / (sum(vec)²)."""
        s = vec.sum()

        max_b = conv.max()
        return float(2.0 * n * max_b / (s * s))
