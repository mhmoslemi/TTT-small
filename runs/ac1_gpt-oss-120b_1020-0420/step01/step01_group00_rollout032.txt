import numpy as np

# ---- fast AC1 upper bound (same as evaluator) ----
def _fast_score(vec: np.ndarray) -> float:
    """Return AC1 bound for a non‑negative vector (no required normalization)."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # power‑of‑two length for FFT
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ---- main candidate proposal -------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound evaluated by `evaluate_sequence`.  The routine follows a
    projected gradient descent with momentum, occasional simulated‑annealing,
    and random block perturbations.  It returns the best vector found as a
    plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ----- initialise from the provided parent sequence ----------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        # fallback uniform initialisation (length unknown -> use 256)
        parent = np.full(256, 1.0, dtype=np.float64)

    cur = np.clip(parent, 0.0, 1000.0)
    if cur.sum() < 1e-12:
        cur[:] = 1.0
    cur = cur / cur.sum()                     # normalise to sum = 1

    # a tiny random perturbation to escape the uniform stationary point
    cur = cur * np.exp(rng.normal(0.0, 0.05, size=cur.shape))
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur / cur.sum()

    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    n = cur.size
    m_fft = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    # ---- optimisation hyper‑parameters ------------------------------------
    total_steps = 25000
    init_eta = 0.55          # initial step magnitude
    init_T = 0.07            # initial temperature for annealing
    momentum = 0.92          # exponential decay factor for velocity
    velocity = np.zeros_like(cur)

    # ---- main loop ---------------------------------------------------------
    for step in range(total_steps):
        # ----- convolution and index of current maximal coefficient ----------
        fa = np.fft.rfft(cur, n=m_fft)
        conv = np.fft.irfft(fa * fa, n=m_fft)[: 2 * n - 1]
        max_c = conv.max()
        k_max = int(conv.argmax())

        # ----- sub‑gradient of the max‑convolution term --------------------
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        i_valid = np.arange(n)[mask]
        idx_valid = idx[mask]

        grad = np.zeros_like(cur)
        # derivative: 2*a[k_max-i] except when i == k_max-i (diagonal)
        diag = (2 * i_valid == k_max)
        grad[i_valid] = np.where(
            diag,
            cur[i_valid],                 # single contribution on the diagonal
            2.0 * cur[idx_valid]          # two ordered contributions otherwise
        )

        # ----- momentum & step‑size schedule --------------------------------
        eta = init_eta * (1.0 - step / total_steps)          # linear cooling
        T = init_T * (1.0 - step / total_steps)

        velocity = momentum * velocity + eta * grad
        cand = cur - velocity

        # ----- projection onto feasible set (non‑negative, sum = 1) ---------
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:

            s = cand.sum()

    # ----- final safety check (official evaluator) -------------------------
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):

        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
        best = fallback

    return [float(v) for v in best]
