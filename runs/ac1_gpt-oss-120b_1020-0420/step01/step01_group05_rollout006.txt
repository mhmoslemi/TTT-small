import numpy as np

# ----------------------------------------------------------
# Utility: fast evaluation of the AC1 bound (FFT‑based)
# ----------------------------------------------------------
def _fast_score(arr: np.ndarray) -> float:
    """Return the AC1 bound 2*n*max(conv(arr,arr))/sum(arr)^2 using FFT."""
    n = arr.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    f = np.fft.rfft(arr, nfft)
    conv = np.fft.irfft(f * f, nfft)[:m]
    M = np.max(conv)
    S = np.sum(arr)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _score_and_max(arr: np.ndarray):
    """Return (bound, index_of_max) for the current sequence."""
    n = arr.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    f = np.fft.rfft(arr, nfft)
    conv = np.fft.irfft(f * f, nfft)[:m]
    max_idx = int(np.argmax(conv))
    M = conv[max_idx]
    S = np.sum(arr)
    if S < 0.01:
        return float('inf'), max_idx
    score = 2.0 * n * M / (S * S)
    return score, max_idx
