def _score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = seq.size
    # compute full conv using FFT
    L = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    conv = conv[:2*n-1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (total * total))

import numpy as np

# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Fast full convolution using FFT (linear, not circular)
# ------------------------------------------------------------
def _conv_full(seq: np.ndarray) -> np.ndarray:
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)
    return conv[: 2 * n - 1]


# ------------------------------------------------------------
# Exact score used by the harness (same as evaluate_sequence)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    n = seq.size
    conv = _conv_full(seq)
    m = conv.max()
    return float(2.0 * n * m / (s * s))
