import numpy as np

# ----- fast AC1 upper bound (uses FFT, linear memory) -----
def _fast_score(vec: np.ndarray) -> float:
    """Compute the AC1 bound for a non‑negative vector (sum assumed >0)."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            raise ValueError

        init = np.ones(256, dtype=np.float64)

        cur[:] = 1.0

    # small random perturbation to leave any stationary point
    cur = cur * np.exp(rng.normal(0.0, 0.05, size=cur.shape))
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur / cur.sum()

    n = cur.size

    # best candidate so far (validated with the official evaluator)
    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best_score = np.inf

    # momentum buffer for smoother updates
    momentum = np.zeros_like(cur)
    beta = 0.9               # momentum decay
    eta0 = 0.18              # initial step length

    for step in range(max_iter):

        idx_f = k_max - np.arange(n)

                    best_score = full

        # ----- occasional local block scaling (exploration) -----
        if (step + 1) % 500 == 0 and rng.random() < 0.25:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            block = cur.copy()
            factor = np.exp(rng.normal(0.0, 0.12))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block = block / block.sum()
            block_score = _fast_score(block)
            if block_score < cur_score:
                cur = block

        # ----- occasional full permutation (decorrelate) -----
        if (step + 1) % 2000 == 0 and rng.random() < 0.30:
            perm = rng.permutation(n)
            cur = cur[perm]

    # final safety check
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # fallback to a simple uniform distribution (guaranteed finite)
        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
        best = fallback

    return [float(v) for v in best]
