def projected_gradient_descent(x, max_iter=200, eps=0.01):
   for iter in range(max_iter):
       conv = np.convolve(x, x)
       k = np.argmax(conv)
       M = conv[k]
       S = x.sum()
       idx = k - np.arange(n)
       mask = (idx >= 0) & (idx < n)
       gradM = np.zeros_like(x)
       gradM[mask] = 2.0 * x[idx[mask]]
       grad = (2.0 * n / (S * S)) * (gradM - (2.0 * M / S) * np.ones_like(x))
       # project to sum-zero direction:
       grad_proj = grad - grad.mean()
       # step size to limit L1 change:
       norm = np.linalg.norm(grad_proj, 1)
       if norm == 0:
           break
       step = -eps * grad_proj / norm  # ensures sum of absolute changes = eps
       x_new = x + step
       # projection to nonnegatives:
       x_new = np.maximum(x_new, 0.0)
       # renormalize sum to 1 (since projection may change sum)
       total = x_new.sum()
       if total == 0:
           x_new = np.full_like(x, 1.0 / n)
           total = 1.0
       x_new = x_new / total
       # evaluate
       M_new = np.max(np.convolve(x_new, x_new))
       score_new = (2.0 * n * M_new) / (x_new.sum() ** 2)
       if score_new < best_score:
           best = x_new.copy()
           best_score = score_new
           x = x_new.copy()
       else:
           # optional: keep some steps;
           # we keep x for next iteration maybe with some probability
           if np.random.rand() < 0.2:
               x = x_new.copy()
   return best, best_score

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _obj(x: np.ndarray) -> float:
    """AC1 upper bound for a non‑negative vector (scale‑invariant)."""
    n = x.shape[0]
    conv = np.convolve(x, x)
    m = conv.max()
    s = x.sum()
    # s is forced >0 in all callers
    return float((2.0 * n * m) / (s * s))

def _grad(x: np.ndarray):
    """Gradient of the objective w.r.t. x (before projection)."""
    n = x.shape[0]
    conv = np.convolve(x, x)
    k = int(conv.argmax())
    m = conv[k]
    s = x.sum()
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad_m = np.zeros_like(x)
    grad_m[mask] = 2.0 * x[idx[mask]]
    coeff = (2.0 * n) / (s * s) if s != 0 else 0.0
    grad = coeff * (grad_m - (2.0 * m / s) * np.ones_like(x))
    return grad, k, m

def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale to unit sum, protecting against zero total."""
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total

    otherwise 256) that aims to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Acquire parent length; fall back to a generic size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
        else:
            parent = None
    else:
        parent = None

    if parent is None:
        n = 256
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size

    # ------------------------------------------------------------------
    # 2.  Initialise best candidate (unit‑sum, non‑negative).
    # ------------------------------------------------------------------
    best = _normalize(parent.copy())
    best_score = _obj(best)

    # ------------------------------------------------------------------
    # 3.  Families of analytic shapes (power‑law ± exponential).
    # ------------------------------------------------------------------
    indices = np.arange(n, dtype=np.float64)

    # --- Power‑law alone ------------------------------------------------
    alphas_coarse = np.linspace(0.10, 1.20, 57)          # step ≈0.02
    best_alpha = None
    for a in alphas_coarse:
        w = (indices + 1.0) ** (-a)
        w = _normalize(w)
        sc = _obj(w)
        if sc < best_score:
            best, best_score = w, sc
            best_alpha = a

    # --- Refine around the best α ---------------------------------------
    if best_alpha is not None:
        lo = max(0.01, best_alpha - 0.05)
        hi = min(1.50, best_alpha + 0.05)
        alphas_fine = np.linspace(lo, hi, 41)
        for a in alphas_fine:
            w = (indices + 1.0) ** (-a)
            w = _normalize(w)
            sc = _obj(w)
            if sc < best_score:
                best, best_score = w, sc
                best_alpha = a

    # --- Power‑law with exponential cut‑off ------------------------------
    betas = np.linspace(0.0, 0.010, 11)   # 0,0.001,…,0.010
    for b in betas:
        if best_alpha is None:
            # try a few arbitrary alphas if none found yet
            for a in (0.3, 0.5, 0.8):
                w = (indices + 1.0) ** (-a) * np.exp(-b * indices)
                w = _normalize(w)
                sc = _obj(w)
                if sc < best_score:
                    best, best_score = w, sc
        else:
            w = (indices + 1.0) ** (-best_alpha) * np.exp(-b * indices)
            w = _normalize(w)
            sc = _obj(w)
            if sc < best_score:
                best, best_score = w, sc

    # --- Pure exponential decay -----------------------------------------
    for b in np.linspace(1e-4, 5e-3, 12):
        w = np.exp(-b * indices)
        w = _normalize(w)
        sc = _obj(w)
        if sc < best_score:
            best, best_score = w, sc

    # --- Simple linear fall‑off -----------------------------------------
    lin = np.arange(n, 0, -1, dtype=np.float64)
    lin = _normalize(lin)
    sc = _obj(lin)
    if sc < best_score:
        best, best_score = lin, sc

    # ------------------------------------------------------------------
    # 4.  Projected gradient descent (mass‑preserving, non‑negative).
    # ------------------------------------------------------------------
    x = best.copy()
    eps = 0.01                    # total L1 movement per iteration
    max_iter = 200
    for _ in range(max_iter):
        grad, k, m = _grad(x)
        # project onto hyperplane sum(x)=const  (remove mean)
        grad_proj = grad - grad.mean()
        norm_l1 = np.linalg.norm(grad_proj, 1)
        if norm_l1 == 0.0:
            break
        step = -eps * grad_proj / norm_l1
        x_new = x + step
        x_new = np.maximum(x_new, 0.0)
        x_new = _normalize(x_new)

        sc = _obj(x_new)
        if sc < best_score:
            best, best_score = x_new.copy(), sc
            x = x_new.copy()
        else:
            # occasional acceptance to escape flat regions
            if rng.random() < 0.15:
                x = x_new.copy()

    # ------------------------------------------------------------------
    # 5.  Light stochastic local perturbations (log‑normal block noise).
    # ------------------------------------------------------------------
    for _ in range(300):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 200))
        stop = min(n, start + width)

        cand = best.copy()
        noise = np.exp(rng.normal(0.0, 0.03, size=stop - start))
        cand[start:stop] *= noise
        cand = _normalize(cand)

        sc = _obj(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 6.  Return the best vector found (as a plain Python list).
    # ------------------------------------------------------------------
    return [float(v) for v in best]
