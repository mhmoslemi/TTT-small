def compute_conv_and_max(vec):
    n=vec.size
    m=1 << ((2*n-1).bit_length())
    fa=np.fft.rfft(vec, n=m)
    conv=np.fft.irfft(fa*fa, n=m)[:2*n-1]
    max_c=conv.max()
    k_max=int(conv.argmax())
    return conv, max_c, k_max

import numpy as np

# ------------------------------------------------------------
# Helper: fast AC1 bound using convolution via FFT (single call)
# ------------------------------------------------------------
def _bound_from_max(max_c: float, total: float, n: int) -> float:
    """Return the bound 2*n*max_c/(total**2)."""
    return float(2.0 * n * max_c / (total * total))


def _fast_score(vec: np.ndarray) -> float:
    """Score used during the search (same formula as the official evaluator)."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return _bound_from_max(float(max_c), float(s), n)
