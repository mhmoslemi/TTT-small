import numpy as np

# ------------------------------------------------------------
# Helper: Catalan‑like sequence (fallback if no parent provided)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    """Return a_i = C(2i,i)/4^i for i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Convolution utilities (full convolution, max value and index)
# ------------------------------------------------------------
def _max_conv(seq: np.ndarray) -> tuple[float, int]:
    """Return (maximum value, index) of full self‑convolution of seq."""
    conv = np.convolve(seq, seq)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ------------------------------------------------------------
# Contribution of each entry to the convolution value at a given index
# ------------------------------------------------------------
def _contrib_at_index(seq: np.ndarray, k: int) -> np.ndarray:
    """
    For a fixed convolution index k, return the array
    c[i] = seq[i] * seq[k-i] (zero where partner out of bounds).
    """
    n = seq.size
    i = np.arange(n)
    partner = k - i
    valid = (partner >= 0) & (partner < n)
    contrib = np.zeros_like(seq)
    contrib[valid] = seq[valid] * seq[partner[valid]]
    return contrib

    and then performs a focused hill‑climbing that moves mass from
    indices heavily contributing to the maximal convolution entry to
    indices contributing little.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Initialise candidate
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
