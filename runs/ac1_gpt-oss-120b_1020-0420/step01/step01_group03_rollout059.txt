def compute_score(seq):
    # use FFT size >= 2*n
    m = 1
    while m < 2*n: m <<= 1
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]   # real
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a**2)

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _next_pow2(x: int) -> int:
    """Small utility to get the next power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fast_score(seq: np.ndarray) -> float:
    """
    Compute the AC1 upper bound 2*n*max(conv)/sum^2 using FFT.
    Returns np.inf if the sum is too small.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.size
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    m = _next_pow2(2 * n - 1)          # size for zero‑padded FFT
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a * sum_a))
