import numpy as np

# ------------------------------------------------------------
# Helper: count ordered pair sums for a set of indices
# ------------------------------------------------------------
def count_pair_sums(idx, n):
    """
    idx : 1‑D np.ndarray of distinct integers in [0, n‑1]
    Returns np.ndarray of length 2*n‑1 where entry k is the number of
    ordered pairs (i, j) from idx with i + j == k.
    """
    # pairwise sum matrix (ordered pairs)
    sums = idx[:, None] + idx[None, :]               # shape (m, m)
    # flatten and histogram
    cnt = np.bincount(sums.ravel(), minlength=2 * n - 1)
    return cnt
