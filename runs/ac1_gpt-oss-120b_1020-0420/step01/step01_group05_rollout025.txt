def refine_subgradient(a, steps=200, step0=0.1):
    best = a.copy()
    best_score = _fast_score(best)
    S = np.sum(best)  # sum maybe not 1; but we keep sum constant.
    for t in range(steps):
        c = convolve_self(best)
        idx = int(np.argmax(c))
        # compute d:
        j = idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        d = np.zeros_like(best)
        d[mask] = best[j[mask]]
        mean_d = np.mean(d)
        v = -d + mean_d
        # step schedule:
        eta = step0 / (1 + t * 0.02)
        cand = best + eta * v
        # clip:
        cand = np.clip(cand, 0.0, 1000.0)
        # renorm sum:
        sum_c = np.sum(cand)
        if sum_c > 0:
            cand = cand * (S / sum_c)
            # after scaling maybe some >1000? but scaling reduces them.
            cand = np.clip(cand, 0.0, 1000.0)
            # If scaling changed sum due to clipping, adjust:
            sum_c2 = np.sum(cand)
            if sum_c2 > 0 and abs(sum_c2 - S) > 1e-9:
                cand = cand * (S / sum_c2)
        # evaluate:
        s = _fast_score(cand)
        if s < best_score:
            best = cand
            best_score = s

def _conv_full(a):
    n = a.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    return c

import numpy as np

# ----------------------------------------------------------
# Fast evaluation of the AC1 bound using FFT (exact).
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using FFT (real input)."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    return c


def _refine_subgradient(a: np.ndarray, steps: int = 200, step0: float = 0.05):
    """Iterative sub‑gradient descent that tries to lower the max convolution."""
    n = a.shape[0]
    best = a.copy()
    best_score = _fast_score(best)
    total = np.sum(best)

    for t in range(steps):
        conv = _conv_full(best)
        idx = int(np.argmax(conv))

        # d_i = a_{idx-i} where defined, else 0
        j = idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        d = np.zeros_like(best)
        d[mask] = best[j[mask]]

        mean_d = np.mean(d)
        # zero‑sum descent direction
        v = -d + mean_d

        eta = step0 / (1.0 + t * 0.02)
        cand = best + eta * v
        cand = np.clip(cand, 0.0, 1000.0)

        # renormalise sum (scaling does not affect the bound)
        s = np.sum(cand)
        if s > 0:
            cand = cand * (total / s)
            cand = np.clip(cand, 0.0, 1000.0)
            # ensure the sum is exactly the original one after clipping
            s2 = np.sum(cand)
            if s2 > 0 and abs(s2 - total) > 1e-12:
                cand = cand * (total / s2)

        sc = _fast_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    return best, best_score
