def _normalize(v):
    """Clip to non‑negative and renormalise to sum 1."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.ones_like(v) / v.size
    return v / s


def _score(v):
    """Wrapper around the supplied evaluator."""
    return float(evaluate_sequence(v.tolist()))


def _subgradient(a):
    """
    Sub‑gradient of  max_k (a*a)[k]  w.r.t. a.
    Returns (g, k_max) where g is the gradient at the active shift.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_max = int(conv.argmax())
    shift = k_max - (n - 1)          # shift that realises the max

        s = shift
        if s < n:
            # a[j-s] term
            g[s:] += a[:-s]
            # a[j+s] term
            g[:-s] += a[s:]
    else:                             # negative shift
        s = -shift
        if s < n:
            g[: n - s] += a[s:]
            g[s:] += a[: n - s]

    return g, k_max

    for alpha in np.linspace(0.10, 0.90, 17):
        w = np.power(indices, -alpha)
        w = _normalize(w)
        sc = _score(w)
        if sc < best_score:
            best = w.copy()
            best_score = sc

    # --------------------------------------------------------------
    # 3️⃣  Stochastic local search (hill‑climbing + sub‑gradient steps)
    # --------------------------------------------------------------
    cur = best.copy()
    cur_score = best_score
    max_iters = 1800          # number of further evaluations

    for it in range(max_iters):
        move = rng.integers(0, 3)   # 0:block,1:swap,2:gradient
        cand = cur.copy()

        if move == 0:
            # ---- block perturbation -------------------------------------------------
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 50))
            end = min(n, start + width)
            seg_sum = cand[start:end].sum()
            if seg_sum > 0.0:
                # replace the block by a fresh Dirichlet‑distributed chunk
                rnd = rng.random(end - start)
                if rnd.sum() == 0.0:
                    rnd = np.ones(end - start)
                rnd = rnd / rnd.sum() * seg_sum
                cand[start:end] = rnd
            else:
                # give the block a tiny uniform mass
                rnd = rng.random(end - start)
                cand[start:end] = rnd / rnd.sum()

        elif move == 1:
            # ---- pairwise mass exchange -------------------------------------------
            i, j = rng.choice(n, size=2, replace=False)
            # move up to 5 % of the donor’s mass
            if rng.random() < 0.5:
                delta = rng.uniform(0.0, 0.05) * cand[i]
                delta = min(delta, cand[i])
                cand[i] -= delta
                cand[j] += delta
            else:
                delta = rng.uniform(0.0, 0.05) * cand[j]
                delta = min(delta, cand[j])
                cand[j] -= delta
                cand[i] += delta

        else:
            # ---- sub‑gradient descent step -----------------------------------------
            g, _ = _subgradient(cand)
            step = 0.30 / np.sqrt(it + 1)   # diminishing step size
            cand = cand - step * g
            cand = np.clip(cand, 0.0, None)

        # ----- normalise after every move -----------------------------------------
        cand = _normalize(cand)

        # ----- evaluate ------------------------------------------------------------
        sc = _score(cand)
        if not np.isfinite(sc):
            continue

        # accept if it improves the current walk
        if sc < cur_score:
            cur = cand
            cur_score = sc

        # keep the globally best solution found
        if sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------------
    # 4️⃣  Return the best candidate (plain Python list)
    # --------------------------------------------------------------
    return [float(v) for v in best]
