import numpy as np

# ------------------------------------------------------------
# Helper: Fast full convolution via FFT (real sequences)
# ------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Return the linear (full) convolution of real vectors a and b using FFT.
    The result length is len(a)+len(b)-1.
    """
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


# ------------------------------------------------------------
# Score identical to the evaluator (but without its checks)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Compute 2*n*max(conv)/sum(seq)^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = _conv_full_fft(seq, seq)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        # a_{i+1} = a_i * (2i+2)(2i+1) / (4 (i+1)^2)
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a
