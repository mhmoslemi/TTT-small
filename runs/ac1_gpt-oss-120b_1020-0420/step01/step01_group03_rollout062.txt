    # Build lower triangular toeplitz matrix A where A[k, j] = f[k-j] for j<=k
    data = []
    offsets = []

        length = n - d
        data.append(np.full(length, f[d]))
        offsets.append(-d)
    A = sp.diags(data, offsets, shape=(n, n), format='csr')

import numpy as np
from scipy import sparse
from scipy.optimize import linprog

def _root_constant_seq(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(a, 0.0, 1000.0)


def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the lower‑triangular Toeplitz matrix A such that
    (A @ b)[k] = (f * b)[k]  for k = 0 … n‑1.
    """
    n = f.shape[0]
    data = []
    offsets = []
    for d in range(n):
        length = n - d
        data.append(np.full(length, f[d], dtype=np.float64))
        offsets.append(-d)                       # column = row - d
    return sparse.diags(data, offsets, shape=(n, n), format='csr')


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a deterministic non‑negative step‑function that (hopefully)
    yields an AC1 score below the target 1.50274349.
    The routine starts from the classic root‑constant sequence,
    adds a tiny random perturbation to break symmetry,
    then performs a few LP‑based improvement steps described in the
    problem statement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise from the provided parent or from the root‑constant.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 2000                     # safe fallback length
        base = _root_constant_seq(n)

    # tiny perturbation to avoid exact degeneracy of the LP
    eps = 1e-4 * np.max(base)
    base = np.clip(base + rng.uniform(-eps, eps, size=n), 0.0, 1000.0)

    # normalise the sum to sqrt(2n); this makes the score equal to max‑conv.
    target_sum = np.sqrt(2.0 * n)
    a = base * (target_sum / base.sum())

    # ------------------------------------------------------------------
    # 2.  Iterative LP‑guided improvement.
    # ------------------------------------------------------------------
    max_iter = 6          # a few iterations are enough and cheap
    t = 0.25              # convex‑combination weight (0 < t < 1)

    prev_M = np.max(np.convolve(a, a))

    for _ in range(max_iter):
        # current max convolution (the quantity we want to shrink)
        M = float(prev_M)

        # build the linear constraints: (a * b)[k] <= M   for k = 0..n‑1
        A = _build_conv_matrix(a)
        b_ub = np.full(n, M, dtype=np.float64)

        # linear program: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            break

        b_opt = res.x
        if b_opt.sum() == 0.0:
            break

        # rescale to keep the sum exactly target_sum (as in the description)
        scale = target_sum / b_opt.sum()
        g = b_opt * scale

        # convex combination
        a = (1.0 - t) * a + t * g
        a = np.clip(a, 0.0, 1000.0)

        new_M = np.max(np.convolve(a, a))
