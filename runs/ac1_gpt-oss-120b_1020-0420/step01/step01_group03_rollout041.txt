import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate the first ``length`` terms of a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)

    Return a deterministic non‑negative step‑function that improves on the
    root‑constant construction.  The method builds several simple
    modifications of the base sequence (zeroing early or late entries,
    right‑shifting) and linearly mixes them with the base.  Each mixed
    candidate is rescaled to keep the original total sum, then scored with
    ``evaluate_sequence``.  The best finite candidate is returned.
    """
    # ------------------------------------------------------------------

        try:
            score = float(evaluate_sequence(candidate_arr.tolist()))
        except Exception:
            return current_best, best_score
        if np.isfinite(score) and score < best_score:
            return candidate_arr.copy(), score
        return current_best, best_score

    # Initialise with the base sequence.
    best_seq = base.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = float("inf")

    # ------------------------------------------------------------------
    # Generate candidates ------------------------------------------------
    # ------------------------------------------------------------------
    sum_base = base.sum()
    # Fractions of the vector to mask / shift.
    L_values = [n // 4, n // 2, (3 * n) // 4]
