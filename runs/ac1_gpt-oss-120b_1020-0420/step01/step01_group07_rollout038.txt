import numpy as np

# ------------------------------------------------------------
# Helper: compute the AC1 bound for a candidate vector
# ------------------------------------------------------------
def _score(vec):
    """Wrapper around the pre‑imported evaluate_sequence."""
    # evaluate_sequence expects a list, not a NumPy array.
    return float(evaluate_sequence(vec.tolist()))


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step‑function (vector) that minimizes the
    AC1 upper bound defined by `evaluate_sequence`.  The routine uses a
    fast random‑mask heuristic followed by simple hill‑climbing
    refinements and a power‑law fallback.  It never allocates O(n²)
    auxiliary structures and works for vectors with thousands of entries.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Determine length n and an initial candidate (fallback)
    # ----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
            best_vec = np.clip(parent, 0.0, 1000.0).astype(np.float64)
        else:
            raise NameError
    except Exception:
        n = 256
        best_vec = np.ones(n, dtype=np.float64)

    best_score = _score(best_vec)

    # ----------------------------------------------------------------
    # Random binary‑mask search (large density)
    # ----------------------------------------------------------------
    TARGET = 1.50274349          # stop early if we reach the target
    RANDOM_TRIALS = 200
    for _ in range(RANDOM_TRIALS):
        # pick a density between 0.30 and 0.70 (inclusive)
        density = rng.uniform(0.30, 0.70)
        mask = rng.random(n) < density
        if not mask.any():
            continue
        cand = mask.astype(np.float64)   # binary 0/1 vector
        score = _score(cand)
        if score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                return [float(x) for x in best_vec]

    # ----------------------------------------------------------------
    # Simple hill‑climbing: flip bits while improvement is found
    # ----------------------------------------------------------------
    HILL_ITERS = 300
    # Work with a boolean mask for easier flipping
    mask = best_vec > 0.5
    for _ in range(HILL_ITERS):
        idx = rng.integers(0, n)
        new_mask = mask.copy()
        new_mask[idx] = not new_mask[idx]          # flip the bit
        cand = new_mask.astype(np.float64)
        score = _score(cand)
        if score < best_score:
            mask = new_mask
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                break

    # ----------------------------------------------------------------
    # Power‑law sweep (compact analytic family)
    # ----------------------------------------------------------------
    p_vals = np.linspace(0.45, 0.80, 71)  # 0.45 … 0.80 step 0.005

    for p in p_vals:
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        score = _score(cand)
        if score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:

    # ----------------------------------------------------------------

    # ----------------------------------------------------------------
    return [float(x) for x in best_vec]
