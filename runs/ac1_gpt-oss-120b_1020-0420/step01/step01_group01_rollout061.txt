import numpy as np

# ------------------------------------------------------------
# Scoring function – exactly the same as the evaluator uses
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # O(n log n) via FFT
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) unnormalised sequence:
#   a_i = C(2i,i) / 4^i   (i = 0 .. n‑1)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Core optimiser – starts from a seed sequence and tries simple
# deterministic re‑weightings and a cheap gradient‑like mass move.
# ------------------------------------------------------------
def _optimise(seq: np.ndarray, rng: np.random.Generator, **kwargs):
    n = seq.size
    best = np.clip(seq, 0.0, 1000.0)
    best_score = _score(best)

    ar = np.arange(n)

    # ----- deterministic weight sweeps ---------------------------------
    # a_i ← a_i * (i+1)^(-lam)   (lam from 0 … 0.6)
    for lam in np.linspace(0.0, 0.6, 13):
        # forward weighting
        w = (ar + 1) ** (-lam)
        cand = np.clip(best * w, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

        # reverse weighting (large indices keep their size)
        w_rev = (n - ar) ** (-lam)
        cand = np.clip(best * w_rev, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

        # symmetric weighting – damp both ends
        w_mid = ((ar + 1) * (n - ar)) ** (-lam / 2.0)
        cand = np.clip(best * w_mid, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ----- cheap gradient‑like mass relocation -------------------------
    max_iters = kwargs.get("num_steps", 2000)
    for _ in range(max_iters):
        conv = np.convolve(best, best)
        max_idx = int(conv.argmax())
        shift = max_idx - (n - 1)                     # shift that yields the max

        # gradient weight: how much each position participates in this conv term
        w = np.zeros(n, dtype=np.float64)
        if shift >= 0:
            limit = n - shift
            i = np.arange(limit)
            w[i] += best[i + shift]
            w[i + shift] += best[i]
        else:
            s = -shift
            i = np.arange(s, n)
            w[i] += best[i - s]
            w[i - s] += best[i]

        src = int(np.argmax(w))
        dst = int(np.argmin(w))
        if src == dst:
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue

        # move a modest fraction of mass from src to dst
        delta = best[src] * rng.uniform(0.001, 0.02)
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc
            # early stop if we already beat the target
            if best_score < 1.5027435:
                break

    # ----- final random fine‑tuning ------------------------------------
    for _ in range(500):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        delta = best[src] * rng.uniform(0.01, 0.10)
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    return best, best_score

def propose_candidate(seed: int = 42, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size – use the provided parent if present.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # --------------------------------------------------------
    # Build a short list of starting candidates.
    # --------------------------------------------------------
    candidates = []

    # 1) Catalan‑like sequence (the classic starting point)
    cat = _catalan_seq(n)
    candidates.append(cat)

    # 2) Use the supplied parent sequence if it exists
    if "height_sequence_1" in globals():

        if par.ndim == 1 and par.size == n and np.isfinite(par).all():
            candidates.append(par)

    # --------------------------------------------------------
    # Run the optimiser on every start and keep the best result.
    # --------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for start in candidates:

        if sc < best_score:
            best_seq, best_score = seq, sc

    # -----------------------------------------------------------------
    # Fallback: in the extremely unlikely case everything failed,
    # return a simple uniform sequence (guaranteed valid and finite).
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_score = _score(best_seq)

    # --------------------------------------------------------
    # Return plain Python list as required by the harness.
    # --------------------------------------------------------
    return [float(v) for v in best_seq]
