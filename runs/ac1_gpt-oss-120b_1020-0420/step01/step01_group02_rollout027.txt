def compute_score(arr):
    n = len(arr)
    L = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fft_a*fft_a, n=L)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # length
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, np.ndarray):
        n = int(height_sequence_1.size)
    else:
        n = 256
    # init: half-series
    base = generate_half_power_series(n)
    # clip to [0,1000] (already)
    current = base.copy()
    best = current.copy()
    best_score = compute_score(best)

    # parameters
    early_end = max(1, n // 5)
    late_start = max(early_end+1, n * 2 // 3)
    max_iters = 20000
    # optional temperature for acceptance
    temp0 = 1e-3
    for it in range(max_iters):
        # select indices
        i = rng.integers(0, early_end)
        j = rng.integers(late_start, n)
        if current[i] <= 0:
            continue
        delta = rng.random() * current[i] * 0.5  # up to half
        new_val_i = current[i] - delta
        new_val_j = current[j] + delta
        if new_val_j > 1000:
            # reduce delta
            delta = 1000 - current[j]
            if delta <= 0:
                continue
            new_val_i = current[i] - delta
            new_val_j = current[j] + delta
        # create candidate
        cand = current.copy()
        cand[i] = new_val_i
        cand[j] = new_val_j
        cand_score = compute_score(cand)
        # Simulated annealing acceptance
        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
            current = cand
        else:
            # accept with probability exp(-(cand_score - best_score)/temp)
            # temperature schedule decreasing
            temp = temp0 * (1 - it / max_iters)
            # if temp <=0: skip acceptance
            if temp > 0:
                prob = np.exp(-(cand_score - best_score) / temp)
                if rng.random() < prob:
                    current = cand
    # final fallback ensure not infinite
    if not np.isfinite(best_score):
        # fallback to base
        best = base
    # Return as list
    return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# Generate the truncated series of (1 - z)^(-1/2)
#   a_0 = 1,  a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return an array of size ``length`` of the half‑power series."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Fast evaluation of the AC1 upper bound (exact formula)
# ------------------------------------------------------------
def compute_ac1_score(vec: np.ndarray) -> float:
    """Return the objective 2*n*max(conv(vec,vec))/sum(vec)^2."""
    n = vec.size
    # length for FFT – next power of two ≥ 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(vec.sum())
    if s == 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # ---- determine problem size ------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        n = int(height_sequence_1.size)
        base = height_sequence_1.astype(np.float64).copy()
    else:
        n = 256
        base = generate_half_power_series(n)

    # ---- initial candidate -------------------------------------------------------
    best = base.copy()
    best_score = compute_ac1_score(best)

    # We will work on a mutable current candidate
    current = best.copy()

    # Parameters for the simple hill‑climbing search
    early_end = max(1, n // 5)                 # first 20 % of indices
    late_start = max(early_end + 1, n * 2 // 3)  # last 1/3 of indices
    max_iters = 12000                           # ample but still fast
    min_frac = 0.05                              # minimum transferred fraction
    max_frac = 0.30                              # maximum transferred fraction
    eps = 1e-12

    for _ in range(max_iters):
        # pick a source from the “early’’ part and a destination from the “late’’ part
        i = rng.integers(0, early_end)
        j = rng.integers(late_start, n)

        ai = current[i]
        if ai <= eps:
            continue

        # propose moving a random fraction of a_i to a_j
        frac = rng.random() * (max_frac - min_frac) + min_frac
        delta = ai * frac

        new_i = ai - delta
        new_j = current[j] + delta

        # enforce the hard upper bound of the evaluator
        if new_j > 1000.0:
            # reduce the transfer so that we stay in range
            delta = 1000.0 - current[j]
            if delta <= 0.0 or delta > ai:
                continue
            new_i = ai - delta
            new_j = current[j] + delta

        cand = current.copy()
        cand[i] = new_i
