def compute_score(a):
    n = a.shape[0]
    # compute conv via FFT: Using rfft
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s*s)
```

import numpy as np

# ------------------------------------------------------------
# Half‑power series (1‑z)^(-1/2), used as a deterministic baseline.
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Convolution using a power‑of‑two FFT (matrix‑free, O(n log n)).
# ------------------------------------------------------------
def full_convolution(vec: np.ndarray) -> np.ndarray:
    n = vec.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    ft = np.fft.rfft(vec, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Compute the AC1 bound for a given vector (same formula as the
# evaluator but without the final clipping checks).
# ------------------------------------------------------------
def ac1_score(vec: np.ndarray) -> float:
    n = vec.size
    s = vec.sum()
    if s < 0.01:
        return float("inf")
    conv = full_convolution(vec)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)
