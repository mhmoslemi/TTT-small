    for r in (0.05, 0.1, 0.2, 0.3):
        k = max(1, int(n * r))
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0:
            continue
        for f in np.linspace(0.1, 0.9, 9):
            new_early = early * f
            delta = early_sum - new_early.sum()

            w = np.arange(1, tail.size+1, dtype=np.float64)
            w = w / w.sum()
            new_tail = tail + delta * w
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                # recompute early/tail for next loops
                early = best[:k]; tail = best[k:]; early_sum = early.sum(); tail_sum = tail.sum()
    
    # Main local search
    cur = best.copy()
    cur_score = best_score
    steps = kwargs.get('num_steps', num_steps)
    no_improve = 0
    for step in range(steps):
        # compute conv and max lag
        conv = np.convolve(cur, cur)
        max_idx = int(conv.argmax())

        if max_idx == 0:

            dst_idx = rng.integers(0, n)

            if cur[i] >= cur[j]:
                src_idx = i
                other_idx = j
            else:
                src_idx = j
                other_idx = i

import numpy as np

# --------------------------------------------------------------------
# Score function identical to the provided evaluator (but local, fast)
# --------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# --------------------------------------------------------------------
# Catalan‑like (central binomial) sequence – convenient fallback init.
# --------------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """Return a list of non‑negative heights with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------

    # ----------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:
        n = 256
        a = _catalan_seq(n)

    n = a.size
    best = a.copy()
    best_score = _score(best)

    # ----------------------------------------------------------------
    # Deterministic re‑allocation: shrink an early prefix and pour the
    # removed mass into the tail with an increasing weight.  Try a few
    # prefix sizes and shrink factors – keep any improvement.
    # ----------------------------------------------------------------
    prefix_ratios = [0.05, 0.10, 0.20, 0.30]
    shrink_factors = np.linspace(0.10, 0.90, 9)   # 10 % … 90 % in steps of 10 %
    for r in prefix_ratios:
        k = max(1, int(n * r))
        early = best[:k]
        tail = best[k:]
        if tail.size == 0:
            continue
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()               # mass taken from early part

            w = np.arange(1, tail.size + 1, dtype=np.float64)
            w = w / w.sum()
            new_tail = tail + delta * w
            cand = np.concatenate([new_early, new_tail])

            if sc < best_score:

                early = best[:k]
                tail = best[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()

    # destination.  Accept moves that improve the current score, keep the
    # overall best, and occasionally apply a mild smoothing step.
    # ----------------------------------------------------------------
    cur = best.copy()

    steps = kwargs.get("num_steps", num_steps)

    no_improve = 0
