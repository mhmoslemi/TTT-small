def generate_piecewise_random(n, rng):
    # number of segments random from 3..7
    k = rng.integers(3, 8)
    # positions: sorted unique indices (including 0 and n-1)
    # ensure first=0, last=n-1
    pos = np.sort(rng.choice(np.arange(1, n-1), size=k-2, replace=False))
    pos = np.concatenate(([0], pos, [n-1]))
    values = rng.random(pos.shape) # random heights
    # ensure nonnegative; set first and last maybe small?
    # Use interpolation:
    x = np.arange(n)
    w = np.interp(x, pos, values)
    # After interpolation, maybe some small negative; clip.
    w = np.clip(w, 0, None)
    s = w.sum()
    if s==0:
        w = np.ones_like(w)
        s = w.sum()
    w /= s
    return w

def propose_candidate(seed=42):
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000)
    else:
        # fallback: generate uniform vector
        n = 500
        parent = np.ones(n, dtype=np.float64)
    n = parent.shape[0]
    # Normalize parent
    if parent.sum() == 0:
        parent = np.ones_like(parent)
    parent = parent / parent.sum()

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Phase 1: shape exploration
    for _ in range(400):
        # randomly pick shape type
        r = rng.random()
        if r < 0.4:
            # powerlaw
            alpha = rng.uniform(0.2, 1.2)
            cand = np.arange(1, n+1, dtype=np.float64) ** (-alpha)
        elif r < 0.7:
            # exponential
            beta = rng.uniform(1e-4, 5e-3)
            cand = np.exp(-beta * np.arange(n, dtype=np.float64))
        else:
            # piecewise linear
            cand = generate_piecewise_random(n, rng)

        # sometimes mix with best
        if rng.random() < 0.3:
            p_weight = rng.random()
            cand = p_weight * cand + (1-p_weight) * best

        # Normalize
        s = cand.sum()
        if s == 0:
            continue
        cand = cand / s

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

    # Phase 2: local perturbations
    for _ in range(1500):
        start = rng.integers(0, n)
        max_len = max(1, n // 100)  # up to 1% length
        length = rng.integers(1, max_len+1)
        stop = min(n, start + length)

        cand = best.copy()
        sigma = rng.uniform(0.05, 0.25)  # lognormal sigma
        noise = np.exp(rng.normal(0.0, sigma, size=stop-start))
        cand[start:stop] *= noise

        # occasional global scaling (smooth)
        if rng.random() < 0.1:
            # apply moving average smoothing with window size random
            win = rng.integers(3, 11)
            kernel = np.ones(win, dtype=np.float64) / win
            cand = np.convolve(cand, kernel, mode='same')
        # Renormalize
        total = cand.sum()
        if total == 0:
            continue
        cand = cand / total
        # Clip to avoid tiny negative due to convolution noise
        cand = np.clip(cand, 0, None)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

    # Return best as list of floats
    return [float(v) for v in best]

    mid = rng.uniform(0.2, 0.8) * n
    steep = rng.uniform(0.5, 5.0) * (n / 100)  # typical width
    x = np.arange(n)
    w = 1.0 / (1.0 + np.exp((x - mid) / steep))

    w = np.clip(w, 0, None)
    s = w.sum()

        return np.ones(n) / n
    return w / s

def normalize(vec):
    total = vec.sum()
    if total <= 0:
        return np.ones_like(vec) / vec.size

def exponential_vec(n, beta):
    w = np.exp(-beta * np.arange(n, dtype=np.float64))
    return normalize(w)

def piecewise_vec(n, rng):
    # number of breakpoints between 3 and 7 inclusive
    k = rng.integers(3, 8)
    # generate positions
    if n-2 > 0:
        pos = np.sort(rng.choice(np.arange(1, n-1), size=k-2, replace=False))
        pos = np.concatenate(([0], pos, [n-1]))
    else:
        pos = np.array([0, n-1], dtype=int)
    values = rng.random(pos.shape)
    x = np.arange(n, dtype=np.float64)
    w = np.interp(x, pos, values)
    return normalize(np.clip(w, 0, None))

def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale vector to have L1‑norm 1 (avoid zero‑sum)."""
    tot = v.sum()
    if tot <= 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / tot


def _score(v: np.ndarray) -> float:
    """Evaluate a NumPy vector via the harness evaluator."""
    return float(evaluate_sequence(v.tolist()))


def _powerlaw_vec(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-α}  (indices start at 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** (-alpha)
    return _normalize(w)


def _exp_vec(n: int, beta: float) -> np.ndarray:
    """a_i ∝ exp(-β·i)."""
    w = np.exp(-beta * np.arange(n, dtype=np.float64))
    return _normalize(w)


def _piecewise_vec(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑linear shape with 3‑7 break points."""
    k = rng.integers(3, 8)  # number of vertices
    if n > 2:
        inner = rng.choice(np.arange(1, n - 1), size=k - 2, replace=False)
        pos = np.sort(inner)
        pos = np.concatenate(([0], pos, [n - 1]))
    else:
        pos = np.array([0, n - 1], dtype=int)
    vals = rng.random(pos.shape)
    x = np.arange(n, dtype=np.float64)
    w = np.interp(x, pos, vals)
    return _normalize(np.clip(w, 0.0, None))


def _logistic_vec(n: int, rng: np.random.Generator) -> np.ndarray:
    """Logistic (sigmoid) decreasing curve."""
    mid = rng.uniform(0.2, 0.8) * (n - 1)
    steep = rng.uniform(0.5, 5.0) * (n / 100.0) + 1e-9
    x = np.arange(n, dtype=np.float64)
    w = 1.0 / (1.0 + np.exp((x - mid) / steep))
    return _normalize(w)


def _ternary_search(func, lo: float, hi: float, iters: int = 20) -> float:
    """Simple bounded ternary search for a scalar minimiser."""
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = func(m1)
        f2 = func(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
