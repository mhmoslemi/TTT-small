        a = a * (1000.0 / max_val)

def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)
    # fallback uniform
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        parent = parent.copy()
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    # fast score
    def fast_score(a):
        nlen = a.shape[0]
        m = 2*nlen - 1
        nfft = 1 << ((m-1).bit_length())
        fa = np.fft.rfft(a, n=nfft)
        c = np.fft.irfft(fa*fa, n=nfft)[:m]
        M = np.max(c)
        S = np.sum(a)
        if S < 0.01:
            return np.inf
        return 2.0 * nlen * M / (S*S)
    # best init
    best_seq = np.ones(n, dtype=np.float64)
    best_seq = best_seq * (1.0)  # unchanged
    best_score = fast_score(best_seq)

    # Evaluate parent
    if np.sum(parent) >= 0.01:
        s = fast_score(parent)
        if s < best_score:
            best_seq = parent.copy()
            best_score = s

    i = np.arange(n, dtype=np.float64)

    # family 1: power law with exponent p near 0.5 and offset
    p_vals = np.linspace(0.45, 0.55, 11)  # step .01
    off_vals = np.concatenate([
        np.linspace(0.01, 0.5, 20),
        np.linspace(0.6, 5.0, 30)
    ])
    for p in p_vals:
        for off in off_vals:
            a = (i + off) ** (-p)
            # scale to avoid >1000
            max_val = a.max()
            if max_val > 1000:
                a = a * (1000.0 / max_val)
            s = fast_score(a)
            if s < best_score:
                best_seq = a.copy(); best_score = s

            # reversed
            a_rev = ((n-1 - i) + off) ** (-p)
            max_val = a_rev.max()
            if max_val > 1000:
                a_rev = a_rev * (1000.0 / max_val)
            s = fast_score(a_rev)
            if s < best_score:
                best_seq = a_rev.copy(); best_score = s

import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound using FFT.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]       # convolution a*a
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------
# Helper to keep values in the permitted range.
# ----------------------------------------------------------
def _scale_to_range(a: np.ndarray) -> np.ndarray:
    """Scale uniformly so that max(a) ≤ 1000, keep non‑negative."""
    a = np.maximum(a, 0.0)
    mx = a.max()
    if mx > 1000.0:
        a = a * (1000.0 / mx)
    return a
