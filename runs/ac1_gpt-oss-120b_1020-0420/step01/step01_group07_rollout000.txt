import numpy as np

# ------------------------------------------------------------
# Utility: smallest power of two >= x
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


# ------------------------------------------------------------
# Full convolution via FFT (real input -> real output)
# ------------------------------------------------------------
def _full_convolution(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    n_fft = _next_pow2(2 * n - 1)
    # FFT of real sequence
    freq = np.fft.rfft(seq, n=n_fft)
    prod = freq * freq
    conv = np.fft.irfft(prod, n=n_fft)
    return conv[: 2 * n - 1]

def _bound_and_grad(seq: np.ndarray):

    Returns: (bound, gradient, k_max, maxc)

    """

        t_eq = k_max // 2
        if 0 <= t_eq < n:
            factor[t_eq] = 1.0
    grad_conv *= factor

    # Bound = 2 * n * maxc / S^2
    bound = 2.0 * n * maxc / (S * S)

    # Gradient of bound:
    #   ∂/∂a_i [2 n maxc / S^2] = 2 n (grad_conv_i * S - 2 maxc) / S^3
    grad = 2.0 * n * (grad_conv * S - 2.0 * maxc) / (S ** 3)

    return bound, grad, k_max, maxc


# ------------------------------------------------------------
# Simple projected gradient descent on the bound
# ------------------------------------------------------------
def _optimize(seq: np.ndarray, max_iter: int = 150, step0: float = 0.1):
    """
    Improves `seq` by gradient steps. Returns (best_seq, best_bound).
    """
    # Initial projection to feasible region
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.sum() < 0.01:
        seq[0] += 0.01 - seq.sum()

    best_seq = seq.astype(np.float64, copy=True)
    best_bound, _, _, _ = _bound_and_grad(best_seq)

    step = step0
    for _ in range(max_iter):
        bound, grad, _, _ = _bound_and_grad(best_seq)

        # Stop if gradient is (practically) zero
        if np.all(np.abs(grad) < 1e-12):
            break

        improved = False
        for _ in range(10):  # backtracking line search
            cand = best_seq - step * grad
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                cand[0] += 0.01 - cand.sum()

            cand_bound, _, _, _ = _bound_and_grad(cand)

            if cand_bound < best_bound - 1e-12:
                best_seq = cand
                best_bound = cand_bound
                step *= 1.3   # modestly enlarge step after success
                improved = True
                break
            else:
                step *= 0.5   # shrink step and retry

        if not improved:
            # No progress – terminate early
            break

    return best_seq, best_bound


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Returns a non‑empty list of non‑negative coefficients that (hopefully)
    achieves a low AC1 upper bound when evaluated with `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem dimension
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
        else:
            n = 256
    except NameError:
        n = 256

    # --------------------------------------------------------
    # Fallback uniform candidate (guaranteed feasible)
    # --------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64)
    best_bound = float(evaluate_sequence(best_vec.tolist()))

    # --------------------------------------------------------
    # Helper to possibly update the global best
    # --------------------------------------------------------
    def _maybe_update(candidate):
        nonlocal best_vec, best_bound
        cand_bound = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(cand_bound) and cand_bound < best_bound:
            best_vec = candidate.copy()
            best_bound = cand_bound

    # --------------------------------------------------------
    # Candidate generators
    # --------------------------------------------------------
    candidates = []

    # 1. Parent if available (already clipped)
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.sum() < 0.01:
            parent[0] += 0.01 - parent.sum()
        candidates.append(parent)
    except NameError:
        pass

    # 2. Binary sparse random subsets (several sizes)
    for size_frac in (0.06, 0.04, 0.02):  # relative to n
        m = max(5, int(size_frac * n))
        for _ in range(3):
            sel = rng.choice(n, size=m, replace=False)
            vec = np.zeros(n, dtype=np.float64)
            vec[sel] = 1.0
            candidates.append(vec)

    # 3. Power‑law families
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.45, 0.80, 9):
        vec = base ** (-p)
        vec = np.clip(vec, 0.0, 1000.0)
        candidates.append(vec)

    # 4. Exponential decay family
    for alpha in np.linspace(0.0003, 0.0030, 8):
        vec = np.exp(-alpha * np.arange(n, dtype=np.float64))
        vec = np.clip(vec, 0.0, 1000.0)
        candidates.append(vec)

    # 5. Random positive vectors (log‑normal)
    for _ in range(3):
        vec = rng.lognormal(mean=2.0, sigma=1.0, size=n)
        vec = np.clip(vec, 0.0, 1000.0)
        candidates.append(vec)

    # --------------------------------------------------------
    # Optimize each candidate (gradient descent) and keep the best
    # --------------------------------------------------------
    for init in candidates:
        opt_vec, opt_bound = _optimize(init, max_iter=120, step0=0.2)
        # Use our cheap bound to discard clearly bad ones early
        if opt_bound >= best_bound:
            # still evaluate via evaluate_sequence for safety on ties
            _maybe_update(opt_vec)
            continue
        # Confirm with the official evaluator (cheap call)
        _maybe_update(opt_vec)

    # --------------------------------------------------------
    # Return the best vector found as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
