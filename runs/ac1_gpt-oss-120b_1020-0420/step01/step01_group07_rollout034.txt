        cand = np.arange(1,n+1) ** (-p)

        s = float(evaluate_sequence(cand.tolist()))
        if s < best_score:
            best_score = s; best_vec = cand

    # binary mask hill climbing
    TARGET = 1.50274349
    # Choose m_range
    min_m = max(80, int(np.sqrt(2*n/TARGET)))  # about 80
    max_m = int(0.5 * n)  # but cap at maybe 300
    max_m = min(max_m, 400)
    RESTARTS = 6
    ITER_PER = 1500  # total 9000

    for _ in range(RESTARTS):
        m = rng.integers(min_m, max_m+1)
        # initial random set

        selected.sort()
        # compute max_c

        bound = 2.0 * n * max_c / (m*m)
        # hill climbing
        for _ in range(ITER_PER):

            out_idx = rng.choice(selected)
            # find new idx not selected
            while True:
                in_idx = rng.integers(0, n)
                if in_idx not in selected:

            # create new array
            new_sel = selected.copy()

            # find position of out_idx to replace
            pos = np.where(new_sel == out_idx)[0][0]  # Since values unique

            # accept if better

                selected = new_sel
                max_c = new_max_c
                bound = new_bound

def compute_max_c(selected, n):
    if selected.size == 0:
        return 0
    sums = selected[:, None] + selected[None, :]   # shape (m,m)
    # flatten
    counts = np.bincount(sums.ravel(), minlength=2*n-1)
    return int(counts.max())

import numpy as np

# ------------------------------------------------------------
# Helper: compute ordered‑pair sum histogram for a set of indices
# ------------------------------------------------------------
def compute_counts(arr, n):
    """
    arr : 1‑d numpy array of distinct integer indices (0 … n‑1)
    returns: np.ndarray of length 2*n‑1, where entry k is the number of
             ordered pairs (i,j)∈arr² with i+j = k.
    """
    # outer sum creates all i+j; flatten and bin‑count
    sums = np.add.outer(arr, arr)          # shape (m, m)
    return np.bincount(sums.ravel(), minlength=2 * n - 1)

    """
    Upper‑bound value 2·n·max(cnt) / m² for a binary mask of size m.
    """
    if m == 0:
        return float("inf")
    max_c = int(cnt.max())
    return 2.0 * n * max_c / (m * m)

    TARGET = 1.50274349

    # --------------------------------------------------------
    # Binary‑mask hill‑climbing search
    # --------------------------------------------------------
    # Minimal size needed for a realistic chance to beat TARGET
    min_m = max(80, int(np.sqrt(2.0 * n / TARGET)))
    max_m = min(300, n)          # keep the problem tractable
    RESTARTS = 8
    ITER_PER = 1200

    for _ in range(RESTARTS):
        m = rng.integers(min_m, max_m + 1)          # size of the mask
        sel = rng.choice(n, size=m, replace=False) # distinct indices
        sel = sel.astype(np.int64)
        mask = np.zeros(n, dtype=bool)
        mask[sel] = True

        cnt = compute_counts(sel, n)
        cur_bound = bound_from_counts(cnt, n, m)

        for _ in range(ITER_PER):
            # ----------------------------------------------------
            # Focus on a sum where the count is maximal
            # ----------------------------------------------------
            max_c = int(cnt.max())
            ks = np.where(cnt == max_c)[0]
            k = int(rng.choice(ks))

            # indices that actually participate in this sum
            involved = [int(i) for i in sel
                        if 0 <= k - i < n and mask[k - i]]
            if involved:
                out_idx = rng.choice(involved)
            else:
                # random element if no involvement (rare)
                out_idx = int(rng.choice(sel))

            # locate the position of out_idx inside the array
            pos = int(np.flatnonzero(sel == out_idx)[0])

            # propose a fresh index not yet selected
            in_idx = int(rng.integers(0, n))
            while mask[in_idx]:
                in_idx = int(rng.integers(0, n))

            # ----------------------------------------------------
            # Perform the swap and evaluate the new bound
            # ----------------------------------------------------
            sel[pos] = in_idx
            mask[out_idx] = False
            mask[in_idx] = True

            new_cnt = compute_counts(sel, n)
            new_bound = bound_from_counts(new_cnt, n, m)

            if new_bound < cur_bound:
                # accept improvement
                cnt = new_cnt
                cur_bound = new_bound

                # evaluate the true objective (cheap for binary vectors)
                cand_vec = np.zeros(n, dtype=np.float64)
                cand_vec[sel] = 1.0
                score = float(evaluate_sequence(cand_vec.tolist()))
                if score < best_score:
                    best_score = score
                    best_vec = cand_vec
                    if best_score <= TARGET:
                        return best_vec.tolist()
            else:
                # revert the change
                sel[pos] = out_idx
                mask[in_idx] = False
                mask[out_idx] = True

        # final evaluation for this restart
        cand_vec = np.zeros(n, dtype=np.float64)
        cand_vec[sel] = 1.0
        score = float(evaluate_sequence(cand_vec.tolist()))
        if score < best_score:
            best_score = score
            best_vec = cand_vec
            if best_score <= TARGET:
                return best_vec.tolist()

    # --------------------------------------------------------
    # Return the best candidate found (fallback is uniform)
    # --------------------------------------------------------
    return best_vec.tolist()
