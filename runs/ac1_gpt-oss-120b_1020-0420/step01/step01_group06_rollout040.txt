import numpy as np

# The evaluator `evaluate_sequence` is provided by the harness.

def _clip_and_normalize(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and rescale to have the given sum."""
    v = np.clip(v, 0.0, 1000.0)
    s = v.sum()
    if s == 0.0:
        # avoid division by zero – return a tiny uniform vector
        v = np.full_like(v, target_sum / v.size)
    else:
        v = v * (target_sum / s)
        v = np.clip(v, 0.0, 1000.0)  # clip again after scaling
    return v


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    The routine uses a small family scan (power‑law & exponential) followed
    by a lightweight gradient‑descent / simulated‑annealing phase.
    """
    # ------------------------------------------------------------------
    # 1.  Obtain a parent sequence if available, otherwise fall back.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(height_sequence_1, dtype=np.float64)
            if (
                tmp.ndim == 1
                and tmp.size > 0
                and np.isfinite(tmp).all()
            ):
                parent = np.clip(tmp, 0.0, 1000.0)
            else:
                parent = None
        except Exception:
            parent = None
    else:
        parent = None

    if parent is None:
        # generic fallback – short uniform vector (any non‑empty vector works)
        n_fallback = 256
        parent = np.full(n_fallback, 1.0, dtype=np.float64)

    n = int(parent.size)
    total_sum = float(parent.sum())
    if total_sum < 0.01:
        total_sum = 0.01
        parent = parent * (total_sum / parent.sum())

    # ------------------------------------------------------------------
    # 2.  Initialise best candidate using the parent.
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        # replace with a safe uniform candidate
        best = np.full(n, total_sum / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 3.  Quick scan of simple analytic families.
    # ------------------------------------------------------------------
    indices = np.arange(1, n + 1, dtype=np.float64)

    # power‑law family a_i ∝ (i+1)^{-α}
    for alpha in np.linspace(0.05, 1.20, 33):
        cand = np.power(indices, -alpha)
        cand = _clip_and_normalize(cand, total_sum)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # exponential decay family a_i ∝ exp(-β·i)
    for beta in np.linspace(1e-4, 5e-3, 30):
        cand = np.exp(-beta * np.arange(n, dtype=np.float64))
        cand = _clip_and_normalize(cand, total_sum)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 4.  Gradient‑descent / annealing refinement.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    # start the refinement from the current best
    a = best.copy()
    mean_val = total_sum / n
    max_iter = 720                     # enough for n≈5k with fast conv
    grad_phase_len = int(max_iter * 0.70)

    for it in range(max_iter):
        # clip and keep the sum constant (the evaluator also clips)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, total_sum / n)
            s = total_sum
        a = a * (total_sum / s)        # renormalise to the original sum

        # convolution and score (exact same formula as the evaluator)
        conv = np.convolve(a, a)       # full convolution, length 2n‑1
        max_b = float(conv.max())
        score = 2.0 * n * max_b / (total_sum * total_sum)

        # keep the best ever seen
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score

        # ------------------------------------------------------------------
        # 4a.   Gradient step (first part of the iterations)
        # ------------------------------------------------------------------
        if it < grad_phase_len:
            # index where convolution attains its maximum
            k = int(conv.argmax())
            # gradient of conv[k] w.r.t. a (2 * a_{k‑j} where defined)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[k - idx]

            # modest step size, proportional to the typical entry magnitude
            lr = 0.5 * mean_val
            a = a - lr * g
        else:
            # ------------------------------------------------------------------
            # 4b.   Simulated‑annealing style multiplicative perturbation.
            # ------------------------------------------------------------------
            # noise level linearly decays to zero
            sigma = 0.20 * (1.0 - (it - grad_phase_len) / (max_iter - grad_phase_len))
            if sigma > 0.0:
                noise = np.exp(rng.normal(0.0, sigma, size=n))
                a = a * noise

        # after any modification clip and re‑scale
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s > 0.0:
            a = a * (total_sum / s)
        else:
            a = best.copy()   # fall back to a known good state

    # ------------------------------------------------------------------
    # 5.  Light local block perturbations (final polishing).
    # ------------------------------------------------------------------
    for _ in range(200):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 80))
        stop = min(n, start + width)

        cand = best.copy()
        # log‑normal multiplicative noise on a single block
        block_noise = np.exp(rng.normal(0.0, 0.05, size=stop - start))
        cand[start:stop] *= block_noise
        cand = _clip_and_normalize(cand, total_sum)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 6.  Return the best candidate found (as a plain Python list).
    # ------------------------------------------------------------------
    # Ensure we return plain floats, not numpy scalars.
    return [float(v) for v in best]
