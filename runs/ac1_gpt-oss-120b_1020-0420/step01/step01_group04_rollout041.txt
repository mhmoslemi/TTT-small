def _coefficients_binom_half(n: int) -> np.ndarray:
    """Analytic coefficients of (1‑z)^‑1/2, i.e. C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _build_candidate(n: int) -> np.ndarray:
    """
    Construct a sequence whose convolution maximum stays 1
    (by separating a dense analytic block from a Sidon set of spikes).
    """
    # size of the analytic block – we keep it well before the end
    m = max(1, n // 4)
    while 2 * m >= n and m > 1:          # guarantee a gap after the block
        m //= 2

    # analytic block (first m entries)
    cand = np.zeros(n, dtype=np.float64)
    cand[:m] = _coefficients_binom_half(m)

    # start of the region where we may place extra spikes
    start = 2 * m
    if start >= n:                         # very small n – fall back to a simple split
        start = m

    # greedy construction of a Sidon set in the tail region.
    # No two distinct sums of two selected positions coincide.
    positions = []
    used_sums = set()

        positions.append(pos)

            used_sums.add(pos + q)

    # set each selected position to the maximal safe value 1.0
    for p in positions:
        cand[p] = 1.0

    return cand
