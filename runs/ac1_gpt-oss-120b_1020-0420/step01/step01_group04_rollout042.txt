        elif op == 1:

            width = int(rng.integers(n // 20, n // 5))

            cand = best.copy()

            reduced = block * factor
            delta_sum = float(block.sum() - reduced.sum())
            cand[start:start+width] = reduced
            # distribute delta to ends
            # split equally to left and right ends
            delta_each = delta_sum * 0.5
            cand[0:n//20] += delta_each / (n//20)
            cand[n - n//20 : n] += delta_each / (n//20)
            np.clip(cand, 0.0, 1000.0, out=cand)
        else: # op == 2
            # boost a tail block (similar to previous program) but also shrink middle slightly
            # select tail
            tail_frac = 0.1

                # left tail
                start = 0

                # right tail

                tail_idx = slice(n-width, n)

            # also shrink middle slightly
            mid_width = int(rng.integers(n // 20, n // 10))
            mid_start = int(rng.integers(middle_start, middle_end - mid_width + 1))

        try:

            best_score = score

            if best_score < 1.50274349:
                break

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a step‑function height sequence whose AC1 bound does not exceed
    the target 1.50274349.  The routine first tries a short grid over power‑law
    exponents (a_i = (i+1)^{-p}) which are cheap to evaluate, then refines the
    best result with tiny random perturbations.
    """
    rng = np.random.default_rng(seed)

    # length of the problem (provided parent if available)
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256

    # keep a valid fallback (the supplied parent, if any)
    best = None
    best_score = np.inf
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        try:
            score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(score):
                best = parent.copy()
                best_score = score
        except Exception:
            pass

    # ------------------------------------------------------------
    # 1) grid search over power‑law exponents p  (a_i = (i+1)^{-p})
    # ------------------------------------------------------------
    # p range chosen empirically; it includes the analytically optimal
    # exponent 0.5 (≈1.57 score) and shallower decays which give larger sums.
    p_vals = np.linspace(0.20, 0.70, 26)          # 0.20, 0.22, …, 0.70
    for p in p_vals:
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        np.clip(a, 0.0, 1000.0, out=a)           # respects the hard bounds
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            # early exit if already under the target
            if best_score < 1.50274349:
                break

    # ------------------------------------------------------------
    # 2) tiny stochastic tweaks around the best candidate found
    # ------------------------------------------------------------
    # A few hundred random local moves can sometimes improve the bound.
    # They are inexpensive because the candidate is already close to optimal.
    steps = 2000
    for _ in range(steps):
        if best_score < 1.50274349:
            break

        cand = best.copy()
        idx = int(rng.integers(0, n))
        # add a small zero‑mean perturbation, scaled to the current magnitude
        delta = float(rng.normal(0.0, 0.05 * max(1.0, cand[idx])))
        cand[idx] = max(0.0, min(1000.0, cand[idx] + delta))

        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------
    # 3) final safety: if nothing succeeded, fall back to a simple analytic vector
    # ------------------------------------------------------------
    if best is None:
        # analytical (1‑z)^‑1/2 coefficients – a reliable always‑valid fallback
        coeffs = np.empty(n, dtype=np.float64)
        coeffs[0] = 1.0
        for i in range(1, n):
            coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
        best = np.clip(coeffs, 0.0, 1000.0)

    return [float(x) for x in best]
