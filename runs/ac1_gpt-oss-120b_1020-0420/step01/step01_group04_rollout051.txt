import numpy as np

def _coefficients_binom_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0*i - 1.0) / (2.0*i)
    return coeffs

def approx_score(seq):
    n = seq.shape[0]
    L = 2*n - 1
    # use rfft
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return (2.0 * n * max_conv) / (s * s)

import numpy as np

# ----------------------------------------------------------------------
# Helper: coefficients of (1 - z)^(-1/2)  (C_i = comb(2i,i)/4^i)
def _coefficients_binom_half(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


# ----------------------------------------------------------------------
# Fast (FFT based) evaluation of the AC‑1 score.
def _approx_score(seq: np.ndarray) -> float:
    """Return the AC‑1 bound 2·n·max(conv)/sum² using FFT (exact up to FP)."""
    n = seq.shape[0]
    L = 2 * n - 1
    # real FFT of the zero‑padded sequence
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return (2.0 * n * max_conv) / (s * s)

        # Accept only improving moves (greedy descent).
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # If we have been stuck for a while, inject a random jitter.
        if no_improve >= 250:
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            best_seq = best_seq * jitter
            np.clip(best_seq, 0.0, 1000.0, out=best_seq)
            best_score = _approx_score(best_seq)
            no_improve = 0

    # --------------------------------------------------------------------
    # Final safety: guarantee a positive sum for the external evaluator.
    if best_seq.sum() < 0.01:
        best_seq += (0.01 - best_seq.sum()) / n

    # Return as plain Python floats.
    return [float(v) for v in best_seq]
