import numpy as np

def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as defined by the provided `evaluate_sequence` function.

    The theoretical minimum of the bound is attained by a uniform distribution,
    because for any non‑negative vector a of length n:

        2·n·max(a∗a) / (sum a)²  ≥  2

    with equality when all entries are equal.  The routine therefore starts
    from a uniform candidate and performs a brief random mass‑transfer search
    that respects the total sum (conserves mass).  Any improvement would be
    captured; otherwise the uniform vector is returned.

    The search is inexpensive (O(n·log n) per evaluation via NumPy's FFT‑based
    convolution inside `evaluate_sequence`) and respects the linear memory
    budget.
    """
    # ----------------------------------------------------------------------
    # 1. Determine length n from the optional parent sequence `height_sequence_1`
    # ----------------------------------------------------------------------
    n = 256  # default length
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        ):
            n = int(parent.size)

    # ----------------------------------------------------------------------
    # 2. Uniform initialization (the provably optimal shape for the bound)
    # ----------------------------------------------------------------------
    best = np.full(n, 1.0, dtype=np.float64)          # each entry = 1.0
    best_score = float(evaluate_sequence(best.tolist()))  # should be exactly 2.0

    # ----------------------------------------------------------------------
    # 3. Light stochastic refinement (mass transfer that keeps total sum constant)
    # ----------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for _ in range(200):
        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a random fraction (up to 50 %) of the mass at i to position j
        delta = best[i] * rng.random() * 0.5
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # enforce the hard bounds required by the evaluator
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate the candidate (the evaluator normalises internally)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # 4. Return the best candidate found as a plain Python list
    # ----------------------------------------------------------------------
