    b = np.convolve(a, a)
    k = b.argmax()
    # compute gradient
    g = np.zeros_like(a)
    # vectorized: for i from 0..n-1:
    # partner = k - i
    # we need to fill g[i] = 2*a[partner] if partner in [0,n)

    partners = k - np.arange(n)
    mask = (partners >= 0) & (partners < n)
    g[mask] = 2 * a[partners[mask]]
    # update a
    a_new = a - eta * g
    # enforce nonneg
    a_new = np.clip(a_new, 0, None)
    # renorm sum to 1 (or any constant)
    s = a_new.sum()
    if s < 1e-12:
        # fallback: uniform small
        a_new = np.ones_like(a) / len(a)
        s = 1.0
    a_new = a_new / s
    return a_new, b.max()

import numpy as np

# ----------------------------------------------------------------------
# Fast ordered‑pair count of a selected index set (ordered pairs)
# ----------------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices (0 … n‑1)
    returns  : np.ndarray of length 2*n‑1 containing ordered‑pair counts
    """
    idx = np.fromiter(selected, dtype=np.int64, count=len(selected))
    if idx.size == 0:
        return np.zeros(2 * n - 1, dtype=np.int64)

    # all pairwise sums (including i=j)
    sums = idx[:, None] + idx[None, :]               # shape (m, m)
    # weight: 1 on the diagonal (i=j), 2 elsewhere (i≠j)
    m = idx.size
    w = np.full((m, m), 2, dtype=np.int64)
    np.fill_diagonal(w, 1)
    cnt = np.bincount(sums.ravel(), weights=w.ravel(), minlength=2 * n - 1)
    return cnt


def _bound_from_selected(selected, n):
    """
    Compute the AC1 bound for a binary vector with ones at `selected`.
    """
    m = len(selected)
    if m == 0:
        return float("inf")
    cnt = _pair_counts(selected, n)
    max_c = int(cnt.max())
    # 2*n*max_c / (m*m)  (sum a_i = m because each entry is 1)
    return float(2.0 * n * max_c / (m * m))

    TRIES = 8                               # independent random starts
    SWAP_ITER = 400                         # local refinement steps per start

    for _ in range(TRIES):
        # random initial subset of the chosen size
        selected = set(rng.choice(n, size=approx_m, replace=False))
        cnt = _pair_counts(selected, n)
        max_c = int(cnt.max())
        m = len(selected)
        bound = 2.0 * n * max_c / (m * m)

        # local swap‑based refinement (focus on the current worst sum)
        local_best_set = selected.copy()
        local_best_bound = bound
        for __ in range(SWAP_ITER):
            if bound <= TARGET:
                break

            # sums attaining the current maximum count
            ks = np.where(cnt == max_c)[0]
            k = int(rng.choice(ks))

            # indices that actually participate in this sum
            involved = [i for i in selected if (k - i) in selected]
            if involved:
                out = int(rng.choice(involved))
            else:
                out = int(rng.choice(list(selected)))

            # propose a new index not currently selected
            while True:
                inp = int(rng.integers(0, n))
                if inp not in selected:
                    break

            new_set = selected.copy()
            new_set.remove(out)
            new_set.add(inp)

            new_cnt = _pair_counts(new_set, n)
            new_max_c = int(new_cnt.max())
            new_bound = 2.0 * n * new_max_c / (m * m)

            # accept if improvement; occasionally accept worse moves
            if new_bound < bound or rng.random() < 0.01:
                selected = new_set
                cnt = new_cnt
                max_c = new_max_c
                bound = new_bound
                if bound < local_best_bound:
                    local_best_bound = bound
                    local_best_set = selected.copy()

        # evaluate the binary mask obtained from this start
        cand = np.zeros(n, dtype=np.float64)
        cand[list(local_best_set)] = 1.0
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand
            if best_score <= TARGET:
                return [float(x) for x in best_vec]

    # --------------------------------------------------------------
    # Power‑law sweep (cheap continuous family)
    # --------------------------------------------------------------
    p_vals = np.linspace(0.45, 0.80, 71)   # 0.45 … 0.80
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand

    # --------------------------------------------------------------
    # Small random local perturbations (optional polishing)
    # --------------------------------------------------------------
    for _ in range(300):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.05)))   # ~5 % change
        cand[start:stop] *= factor

        if rng.random() < 0.20:                        # occasional smoothing
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand

    return [float(x) for x in best_vec]
