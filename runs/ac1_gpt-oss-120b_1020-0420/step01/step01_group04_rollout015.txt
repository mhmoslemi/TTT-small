import numpy as np
from scipy import signal

def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2: comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _fast_score(seq: np.ndarray) -> float:
    """Compute the AC1 bound without the heavy Python wrapper."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = signal.fftconvolve(seq, seq, mode="full")
    max_b = conv.max()
    return 2.0 * len(seq) * max_b / (s * s)
