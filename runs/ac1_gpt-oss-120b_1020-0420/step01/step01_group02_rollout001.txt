    # generate half power series
    hp = generate_half_power_series(n)

    # prepare base candidates
    candidates = []

        base1 = parent.copy()
        np.clip(base1, 0.0, 1000.0, out=base1)
        if base1.sum() < 0.01:
            base1 += (0.01 - base1.sum())/n
        candidates.append(base1)
    base2 = hp
    np.clip(base2, 0.0, 1000.0, out=base2)
    if base2.sum() < 0.01:
        base2 += (0.01 - base2.sum())/n
    candidates.append(base2)

    # pick the best by fast bound
    best = None
    best_score = np.inf
    for cand in candidates:
        score = fast_ac1_bound(cand)
        if score < best_score:
            best = cand
            best_score = score

    # hill climbing
    sigma = 0.25
    for step in range(num_steps):
        cand = best.copy()
        max_width = max(1, n // 20)
        width = int(rng.integers(1, max_width+1))
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, sigma)))
        cand[start:start+width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s < 0.01:
            cand += (0.01 - s) / n
        cand_score = fast_ac1_bound(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # optionally reduce sigma?
        sigma *= 0.9995  # slowly reduce exploration

    # final verification with official evaluator
    final_score = float(evaluate_sequence(best.tolist()))
    # If final_score is inf or nan or not finite, fall back to first candidate
    if not np.isfinite(final_score):
        best = candidates[0]
        final_score = float(evaluate_sequence(best.tolist()))
        # If still invalid, just return a safe sequence e.g., constant
        if not np.isfinite(final_score):
            safe = np.full(n, 0.01/n)
            np.clip(safe, 0.0, 1000.0, out=safe)
            best = safe
    return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# Helper: generate the truncated series of (1 - z)^(-1/2)
#   a_0 = 1, a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return array of size ``length`` with entries binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound using FFT (linear convolution)
# ------------------------------------------------------------
def fast_ac1_bound(a: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative sequence a using FFT."""
    n = a.shape[0]
    if n == 0:
        return np.inf
    # size = next power of two >= 2n-1
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    max_b = conv[: 2 * n - 1].max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
