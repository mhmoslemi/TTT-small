import numpy as np

# ------------------------------------------------------------
# Helper: ordered‑pair counts for a binary mask
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """
    Compute ordered‑pair counts of indices in `selected`.
    Returns an ndarray of length 2*n-1 where entry k is the number of ordered
    pairs (i,j) with i+j == k and i,j in selected.
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt
