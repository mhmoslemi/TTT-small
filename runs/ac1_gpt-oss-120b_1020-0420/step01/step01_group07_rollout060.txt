# ------------------------------------------------------------
# Helper: ordered‑pair counts for a given index set (binary mask)
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices where the coefficient is 1
    returns   : np.ndarray length 2*n-1 with ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            cnt[i + j] += 2                 # (i,j) and (j,i)
    return cnt

    # ----------------------------------------------------------------
    # Return the best candidate found as a plain Python list
    # ----------------------------------------------------------------
    return [float(x) for x in best_vec.tolist()]
