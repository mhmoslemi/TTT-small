def mass_transfer_move(arr, km, w, rng, step_factor):

    # pick i from w distribution
    probs_i = w / w.sum()
    i = rng.choice(len(arr), p=probs_i)

    delta = arr[i] * step_factor
    if delta <= 0:
        return None

    # compute inv_w = max_w - w
    max_w = w.max()
    inv_w = max_w - w
    inv_w = np.where(inv_w < 0, 0, inv_w)  # not needed
    inv_sum = inv_w.sum()
    if inv_sum == 0:
        return None
    probs_j = inv_w / inv_sum
    j = rng.choice(len(arr), p=probs_j)

    # don't choose same index; if same, can skip
    if i == j:
        # pick another j maybe
        possible = np.arange(len(arr))
        possible = possible[possible != i]
        if possible.size == 0:
            return None
        j = rng.choice(possible)

    cand = arr.copy()
    cand[i] -= delta  # ensure nonnegative later
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, num_steps=1500, **kwargs):

    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, np.ndarray):

    else:
        n = 1024  # fallback
        a = generate_half_power_series(n)
    np.clip(a, 0.0, 1000.0, out=a)
    # Ensure sum >= 0.01
    if a.sum() < 0.01:
        a += (0.01 - a.sum()) / a.size
    best = a.copy()
    best_score, best_max, best_sum = compute_metric(best)
    best_score = float(best_score)
    n = best.size
    eps = 1e-12
    # Precompute contribution of probability arrays each iteration
    for step in range(num_steps):

                alt = alt[alt != i_reduce]

import numpy as np

# ------------------------------------------------------------
# Helpers
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Series for (1‑z)^(-1/2): a[0]=1, a[i]=a[i‑1]*(2*i‑1)/(2*i)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def random_block_perturbation(base: np.ndarray, rng: np.random.Generator,
                             sigma: float = 0.1) -> np.ndarray:
    """Multiply a random contiguous block of ``base`` by a log‑normal factor."""
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, sigma)))   # modest multiplicative noise
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def compute_metric(arr: np.ndarray):
    """Return (score, max_convolution, sum) exactly as evaluate_sequence does."""
    a = np.clip(arr, 0.0, 1000.0)          # safety clamp, does not modify original
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf, np.inf, sum_a
    b = np.convolve(a, a)
    max_b = b.max()
    n = a.size
    score = 2.0 * n * max_b / (sum_a * sum_a)
    return float(score), float(max_b), float(sum_a)

def propose_candidate(seed: int = 42, num_steps: int = 1500, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 bound
    is as low as the inexpensive stochastic optimisation can achieve.
    """
    rng = np.random.default_rng(seed)

    # ------------------- initialise -------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        a = height_sequence_1.astype(np.float64).copy()
    else:
        # fallback: half‑power series of modest length
        a = generate_half_power_series(1024)

    np.clip(a, 0.0, 1000.0, out=a)
    # ensure a non‑trivial sum (the evaluator would reject otherwise)
    if a.sum() < 0.01:
        a += (0.01 - a.sum()) / a.size

    best = a.copy()
    best_score, best_max, best_sum = compute_metric(best)

    n = best.size
    eps = 1e-12
    # parameters for the mass‑transfer move
    max_frac = 0.5        # initial fraction of a[i] to move
    min_frac = 0.01       # final fraction

    for step in range(num_steps):
        # linearly decay the transfer fraction
        frac = max_frac * (1.0 - step / num_steps) + min_frac

        if rng.random() < 0.4:
            # ---- mass transfer: shift a small amount from a large entry
            #       to a small entry, preserving the total sum ----------
            prob_reduce = best / (best.sum() + eps)          # favour large entries
            i_reduce = rng.choice(n, p=prob_reduce)

            delta = best[i_reduce] * frac
            if delta <= 0.0:
                continue

            # prefer indices that are currently small
            inv = 1.0 / (best + eps)
            prob_increase = inv / inv.sum()
            i_inc = rng.choice(n, p=prob_increase)

            # avoid a no‑op (same index)
            if i_inc == i_reduce:
                # pick a different index uniformly
                alt = np.arange(n)
                alt = alt[alt != i_reduce]
                if alt.size == 0:
                    continue
                i_inc = rng.choice(alt)

            cand = best.copy()
            cand[i_reduce] -= delta
            cand[i_inc] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:
            # ---- random block perturbation (exploratory) ---------------
            cand = random_block_perturbation(best, rng)

        cand_score, cand_max, cand_sum = compute_metric(cand)

        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            best_max = cand_max
            best_sum = cand_sum

            # early exit if we have already passed the target bound
            if best_score <= 1.5028:
                break

    # Return a plain Python list as required by the harness
    return [float(x) for x in best]
