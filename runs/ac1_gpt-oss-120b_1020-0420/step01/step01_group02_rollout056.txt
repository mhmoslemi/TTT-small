import numpy as np
import time

def generate_half_power_series(length: int) -> np.ndarray:
    """Coefficients a_i = binom(2i,i)/4**i, a_0=1, a_i = a_{i-1}*(2i-1)/(2i)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution(seq: np.ndarray) -> float:
    """Linear convolution of seq with itself, returning the maximum entry."""
    n = seq.shape[0]
    N = _next_pow2(2 * n - 1)
    fft_seq = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(fft_seq * fft_seq, n=N)
    max_val = float(conv[: 2 * n - 1].max())
    return max_val if max_val > 0.0 else 0.0


def _ac1_score(seq: np.ndarray) -> float:
    """Fast AC1 upper bound (same formula as evaluate_sequence)."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = float(seq.sum())
    if total < 0.01:
        return float("inf")
    maxb = _max_convolution(seq)
    n = seq.shape[0]
    return float(2.0 * n * maxb / (total * total))
