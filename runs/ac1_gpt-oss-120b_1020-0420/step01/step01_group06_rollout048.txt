import numpy as np

# evaluate_sequence is provided by the harness; it is imported in the execution environment.

def _to_score(vec: np.ndarray) -> float:
    """Convert a numpy vector to a float score using the provided evaluator."""
    # The evaluator expects a plain Python list of floats.
    return float(evaluate_sequence(vec.tolist()))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound defined by evaluate_sequence.
    The search proceeds by random dense constructions followed by a brief
    local refinement.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine a moderate problem size.
    # If the parent sequence is available, we cap the working length to keep
    # runtime cheap; otherwise we use a default length.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent_len = int(np.asarray(height_sequence_1).size)
        n = min(600, max(200, parent_len // 4))  # keep n reasonable
    else:
        n = 500

    # ----------------------------------------------------------------------
    # Baseline: uniform positive sequence (score ≈ 2).  This ensures we
    # always have a valid fallback.
    # ----------------------------------------------------------------------
    uniform = np.ones(n, dtype=np.float64)
    uniform /= uniform.sum()                       # normalise sum to 1
    best_vec = uniform.copy()
    best_score = _to_score(best_vec)

    # ----------------------------------------------------------------------
    # Random dense search.
    # We draw many candidates with a random fill‑density and values from an
    # exponential distribution (Dirichlet‑like after normalisation).
    # ----------------------------------------------------------------------
    attempts = 3000
    for _ in range(attempts):
        # Choose a density between 20 % and 90 % of the entries.
        density = rng.uniform(0.20, 0.90)
        m = max(2, int(density * n))

        # Random distinct positions.
        idx = rng.choice(n, size=m, replace=False)

        # Positive magnitudes (exponential is simple and gives variability).
        vals = rng.exponential(scale=1.0, size=m)

        vec = np.zeros(n, dtype=np.float64)
        vec[idx] = vals

        total = vec.sum()
        if total == 0.0:
            continue
        vec /= total                     # normalise sum to 1

        sc = _to_score(vec)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_vec = vec.copy()

    # ----------------------------------------------------------------------
    # Light local refinement: move a small fraction of mass between two
    # coordinates while preserving the total sum.
    # ----------------------------------------------------------------------
    for _ in range(500):
        i, j = rng.choice(n, size=2, replace=False)
        # Only move mass that is present.
        if best_vec[i] <= 1e-12:
            continue
        # Transfer a random proportion of the mass at i to j.
        delta = rng.uniform(0.01, 0.10) * best_vec[i]
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = _to_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_vec = cand

    # ----------------------------------------------------------------------
    # Final clean‑up: ensure bounds and a non‑empty list.
    # ----------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Defensive fallback: if for any reason we lost a valid score.
    if not np.isfinite(best_score) or best_vec.size == 0:
        best_vec = np.ones(n, dtype=np.float64)
        best_vec /= best_vec.sum()

    return [float(v) for v in best_vec]
