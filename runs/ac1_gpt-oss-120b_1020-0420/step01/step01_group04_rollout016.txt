def _coeff_binom_half(n):

import numpy as np

def _coeff_binom_half(n: int) -> np.ndarray:
    """Return the first n coefficients of (1‑z)^‑1/2:
       C_0 = 1, C_i = C_{i‑1} * (2i‑1)/(2i) for i≥1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * ((2.0 * i - 1.0) / (2.0 * i))
    return coeffs


def propose_candidate(seed: int = 42, **kwargs):

    low AC1 upper bound.  The construction uses the exact coefficients of the
    series (1‑z)^‑1/2, whose self‑convolution is identically 1 on the interior,
    giving a score ≈ 0.5 regardless of length.
    """
    # Length of the problem: use the supplied parent if present,
    # otherwise fall back to a modest default.

        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256

    # Analytic candidate (non‑negative, already bounded by 1 < 1000)
    cand = _coeff_binom_half(n)
    np.clip(cand, 0.0, 1000.0, out=cand)

    # Return as a plain Python list of floats for the evaluator.
    return [float(x) for x in cand]
