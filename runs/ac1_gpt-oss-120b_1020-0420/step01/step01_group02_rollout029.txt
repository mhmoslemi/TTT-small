def propose_candidate(seed=42, max_iter=2000, epsilon=0.01, **kwargs):
    rng = np.random.default_rng(seed)

    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, np.ndarray):
        a = height_sequence_1.astype(np.float64).copy()
    else:
        n = 3000   # choose moderate length
        a = np.full(n, 1.0, dtype=np.float64)
    # clamp to bounds
    np.clip(a, 0.0, 1000.0, out=a)
    # ensure sum > 0.01
    if a.sum() < 0.01:
        a += 0.01 / a.size

    best = a.copy()
    best_score = compute_score(best)

    n = best.size

    # early stagnation tracking
    no_improve = 0
    for iter_idx in range(max_iter):
        # convolution
        b = np.convolve(best, best)
        max_b = b.max()
        k_max = int(np.argmax(b))

        # partner contributions
        idxs = np.arange(n)
        partner_idx = k_max - idxs
        mask = (partner_idx >= 0) & (partner_idx < n)
        partner_vals = np.empty_like(best)
        partner_vals.fill(0.0)
        partner_vals[mask] = best[partner_idx[mask]]

        # pick i_high from top quantile of partner_vals
        # choose a random index weighted by partner_vals + epsilon to avoid zero
        prob = partner_vals + 1e-12
        prob_sum = prob.sum()
        if prob_sum == 0:
            i_high = rng.integers(0, n)
        else:
            prob = prob / prob_sum
            i_high = int(rng.choice(n, p=prob))

        # pick i_low from indices where partner_vals is minimal
        # approach: sample from inverted weights
        inv_prob = (prob_sum - prob) + 1e-12   # larger weight for small partner
        inv_sum = inv_prob.sum()
        inv_prob = inv_prob / inv_sum
        i_low = int(rng.choice(n, p=inv_prob))

        # ensure i_high != i_low
        if i_high == i_low:
            continue

        # transfer amount
        avg = best.mean()
        delta = epsilon * avg
        # cap delta
        delta = min(delta, best[i_high])
        # also ensure not exceed max bound
        if best[i_low] + delta > 1000.0:
            delta = 1000.0 - best[i_low]
        if delta <= 0:
            continue

        cand = best.copy()
        cand[i_high] -= delta
        cand[i_low] += delta
        # clip just in case
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = compute_score(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        # early exit if no improvement long time
        if no_improve >= 200 and iter_idx > 0:
            break

    # final fallback: if best_score is inf or not finite, use uniform
    if not np.isfinite(best_score):
        best = np.full_like(best, 1.0)
        best_score = compute_score(best)

    return best.tolist()

import numpy as np

# ------------------------------------------------------------
# Helper: evaluate AC1 score (same formula as the harness)
# ------------------------------------------------------------
def _ac1_score(a: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative step function."""
    n = a.shape[0]
    # convolution (full)
    b = np.convolve(a, a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2 * n * max_b / (s * s))


# ------------------------------------------------------------
# Core optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Determine length (use parent if present, otherwise a moderate size)
    # -----------------------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        n = int(height_sequence_1.size)
    else:
        n = 3000                     # a size that stays comfortably fast

    # -----------------------------------------------------------------
    # Initialise a candidate.
    # Uniform with a tiny random perturbation works well as a neutral start.
    # -----------------------------------------------------------------
    a = np.ones(n, dtype=np.float64)
    a += rng.normal(0.0, 0.05, size=n)      # break perfect symmetry
    np.clip(a, 0.0, 1000.0, out=a)
    if a.sum() < 0.01:
        a += 0.01 / a.size

    best = a.copy()
    best_score = _ac1_score(best)

    # -----------------------------------------------------------------
    # Parameters for the stochastic local search
    # -----------------------------------------------------------------
    max_iter = 5000                     # total iterations
    no_improve_limit = 300              # early stop if stuck
    eps_factor = 0.10                   # relative amount transferred
    block_prob = 0.12                   # probability to do a block move
    block_sigma = 0.25                  # log‑normal width for block scaling

    no_improve = 0

    for it in range(max_iter):
        # -------------------------------------------------------------
        # Compute convolution, its maximum and the index where it occurs
        # -------------------------------------------------------------
        conv = np.convolve(best, best)
        max_b = conv.max()
        k_max = int(conv.argmax())
        # -------------------------------------------------------------
        # Prepare partner values for the peak index
        # -------------------------------------------------------------
        idx = np.arange(n)
        partner_idx = k_max - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        partner = np.zeros_like(best)
        partner[mask] = best[partner_idx[mask]]

        # -------------------------------------------------------------
        # Choose a move type
        # -------------------------------------------------------------
        if rng.random() < block_prob:
            # ----- block scaling move ---------------------------------
            max_w = max(1, n // 10)
            width = int(rng.integers(1, max_w + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, block_sigma)))
            cand = best.copy()
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:
            # ----- directed mass transfer move -----------------------
            # contribution of each index to the current peak
            contrib = best * partner
            i_high = int(contrib.argmax())
            i_low = int(contrib.argmin())
            if i_high == i_low:
                continue

            # amount to move: a fraction of the larger value
            delta = eps_factor * best[i_high]
            delta = min(delta, best[i_high])                     # keep non‑negative
            delta = min(delta, 1000.0 - best[i_low])             # respect upper bound
            if delta <= 0:
                continue

            cand = best.copy()
            cand[i_high] -= delta
            cand[i_low] += delta

        # -------------------------------------------------------------
        # Evaluate candidate
        # -------------------------------------------------------------
        score = _ac1_score(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        # -------------------------------------------------------------
        # Early termination
        # -------------------------------------------------------------
        if no_improve >= no_improve_limit:
            break

    # -----------------------------------------------------------------
    # Final safety checks – ensure the result respects the evaluator's
    # clipping and sum constraints.
    # -----------------------------------------------------------------
    np.clip(best, 0.0, 1000.0, out=best)
    if best.sum() < 0.01:
        # tiny boost to avoid the sum‑zero guard
        best += (0.01 - best.sum()) / best.size

    # Return plain Python list as required by the harness
    return [float(x) for x in best]
