def conv_max(seq):
    n = seq.shape[0]
    size = int(2**np.ceil(np.log2(2*n-1)))  # zero-pad length
    fft_seq = np.fft.rfft(seq, n=size)
    prod = fft_seq * fft_seq
    conv = np.fft.irfft(prod, n=size)[:2*n-1]
    return conv.max()

# Helpers

def _catalan_seq(n: int) -> np.ndarray:
    """Catalan‑like coefficients a_i = C(2i,i)/4^i for i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


def _fft_conv(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of seq with itself using an FFT of power‑of‑2 length."""
    n = seq.size
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    f = np.fft.rfft(seq, n=size)
    prod = f * f
    conv = np.fft.irfft(prod, n=size)[:m]

    s = seq.sum()

    return float(2.0 * seq.size * conv.max() / (s * s))

    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
