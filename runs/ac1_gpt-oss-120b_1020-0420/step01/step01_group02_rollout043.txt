def generate_half_power_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _score(arr):

    return evaluate_sequence(arr.tolist())

def propose_candidate(seed: int = 42, **kwargs):

    # Determine problem length

# Helper: (1 - z)^(-1/2) series – the classic low‑AC1 baseline

def generate_half_power_series(length: int) -> np.ndarray:
    """Return the first *length* coefficients of (1‑z)^(-1/2)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a
