import numpy as np

# ------------------------------------------------------------
# Helper: gradient of the currently maximal convolution term
# ------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Returns the gradient (sub‑gradient) of the term (a*a)[k] with respect to a.
    The gradient is 2*a[k-i] for every i such that 0 ≤ i < n and 0 ≤ k-i < n,
    otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (approximately)
    minimises the AC1 bound evaluated by `evaluate_sequence`.
    The routine uses a simple gradient‑descent on the maximal convolution term,
    followed by a short random‑perturbation polishing phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------
    # Determine the problem length and initialise a starting vector
    # --------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
            a0 = np.clip(parent, 0.0, 1000.0)
        else:
            raise Exception
    except Exception:
        n = 256
        # power‑law fallback (exponent chosen experimentally)
        idx = np.arange(1, n + 1, dtype=np.float64)
        a0 = 1.0 / (idx ** 0.6)

    # Ensure a tiny positive sum (required by the evaluator)
    if a0.sum() < 0.01:
        a0 = np.ones(n, dtype=np.float64) * 0.01

    # --------------------------------------------------------------------
    # Possibly improve the initial vector with a simple power‑law candidate
    # --------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    alt = 1.0 / (idx ** 0.6)                # identical to a0 for the fallback case
    alt = np.clip(alt, 0.0, 1000.0)
    if alt.sum() < 0.01:
        alt = np.ones(n, dtype=np.float64) * 0.01

    # Pick the better of the two as the starting point
    score0 = float(evaluate_sequence(a0.tolist()))
    score_alt = float(evaluate_sequence(alt.tolist()))
    if np.isfinite(score_alt) and score_alt < score0:
        a = alt.copy()
        best_score = score_alt
    else:
        a = a0.copy()
        best_score = score0

    best_vec = a.copy()

    # --------------------------------------------------------------------
