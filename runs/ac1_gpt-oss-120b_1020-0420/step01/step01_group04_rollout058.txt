import numpy as np

def _coefficients_binom_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0*i - 1.0) / (2.0*i)
    return coeffs

def _max_convolution(x):
    n = x.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[:2*n-1]
    return float(conv.max())

def propose_candidate(seed=42, num_steps=12000, **kwargs):

    else:
        n = 256

    np.clip(cur, 0.0, 1000.0, out=cur)

    # fallback to provided parent if better
    best = cur.copy()
    sum_c = float(best.sum())
    max_c = _max_convolution(best)
    best_score = 2.0 * n * max_c / (sum_c * sum_c)

        max_p = _max_convolution(parent)
        parent_score = 2.0 * n * max_p / (sum_p * sum_p)
        if parent_score < best_score:
            best = parent.copy()
            cur = best.copy()
            best_score = parent_score
    except Exception:
        pass

    tail_start = n * 3 // 4

    for step in range(num_steps):
        # compute convolution max and its position

        size = 1 << ((2*nlen-1).bit_length())
        fx = np.fft.rfft(cur, n=size)

def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs

def _max_convolution(x):
    """Maximum value of the full convolution x*x using an FFT."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    return float(conv.max())
