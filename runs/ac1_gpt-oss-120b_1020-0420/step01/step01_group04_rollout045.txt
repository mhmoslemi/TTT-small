def _generate_binomial_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0

        coeffs[i] = coeffs[i-1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs
