    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # full convolution length = 2*len(arr)-1
    return float(conv[: 2 * arr.size - 1].max())


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iter: int = 2000, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound defined by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial sequence: use the provided parent if present, otherwise half‑power.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):
        a = np.array(height_sequence_1, dtype=np.float64, copy=True)
    else:
        n = 256
        a = generate_half_power_series(n)

    n = a.size
    np.clip(a, 0.0, 1000.0, out=a)

    # safeguard against a zero sum (should never happen)
    if a.sum() < 0.01:
        a += 0.01 / n

    # --------------------------------------------------------------
    # Pre‑compute FFT length (next power of two ≥ 2n‑1) – reused each call.
    # --------------------------------------------------------------
    size = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Score function – AC1 bound using the fast FFT max‑convolution.
    # --------------------------------------------------------------
    def score(arr: np.ndarray) -> float:
        s = arr.sum()
        if s == 0.0:
            return np.inf
        M = _fft_max_convolution(arr, size)
        return 2.0 * n * M / (s * s)

    # --------------------------------------------------------------
    # Keep the best candidate seen so far.
    # --------------------------------------------------------------
    best = a.copy()
    best_score = score(best)

    # --------------------------------------------------------------
    # Simple stochastic hill‑climbing / simulated‑annealing loop.
    # --------------------------------------------------------------
    current = best.copy()
    current_score = best_score
    no_improve = 0
    eps = 1e-12

    for _ in range(max_iter):
        # source index – biased toward larger entries (they contribute most to M)
        prob_src = current / (current.sum() + eps)
        i = rng.choice(n, p=prob_src)

        # destination index – biased toward smaller entries
        inv = (current.max() - current) + eps
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)

        if i == j:
            continue

        # move a random fraction (up to 20 %) of the source mass to the destination
        delta = current[i] * rng.uniform(0.0, 0.2)
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = score(cand)

        # record a strict improvement (global best)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            current = cand
            current_score = cand_score
            no_improve = 0
            continue

        # allow non‑worsening steps to help escape plateaus
        if np.isfinite(cand_score) and cand_score < current_score:
            current = cand
            current_score = cand_score
            no_improve = 0
            continue

        no_improve += 1
        if no_improve >= 300:   # stagnation → early exit
            break

    # ------------------------------------------------------------------
    # Final safety clipping; ensure the sum constraint required by the evaluator.
    # ------------------------------------------------------------------
    np.clip(best, 0.0, 1000.0, out=best)
    s = best.sum()
    if s < 0.01:
        best *= 0.01 / (s + eps)

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best.tolist()]
