def simplex_projection(v):
    # Ensure nonnegative and sum=1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def conv_full(a):
    n = a.shape[0]
    m = 2*n-1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    return c

def max_conv_info(a):
    c = conv_full(a)
    max_val = np.max(c)
    max_idx = int(np.argmax(c))
    return max_val, max_idx

def simplex_projection(v):
    # based on algorithm by Duchi et al.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

        # all nonpositive, return uniform? Actually if v has sum <=1 and all <=0, projection is uniform? But we can fallback to uniform.
        return np.full_like(v, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

# ----------------------------------------------------------
# Fast convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                                 # element‑wise product
    c = np.fft.irfft(fc, n=nfft)[:m]
    return c

def _fast_score(a: np.ndarray) -> float:
    """Compute the AC1 upper bound 2*n*max(conv)/sum(a)^2."""
    n = a.shape[0]
    c = _conv_full(a)
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

# ----------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum=1)
# ----------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    # quick exit
    if np.all(v >= 0) and abs(np.sum(v) - 1.0) <= 1e-12:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------
# Subgradient step (gradient of the maximal convolution entry)
# ----------------------------------------------------------
def _subgradient(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Subgradient of max_k (a*a)_k at active index idx.
    Returns g where g_j = 2*a_{idx-j} for valid indices, else 0.
    """
    n = a.shape[0]
    indices = idx - np.arange(n)
    mask = (indices >= 0) & (indices < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[indices[mask]]
    return g
