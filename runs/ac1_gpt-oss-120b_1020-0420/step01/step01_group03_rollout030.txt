def get_score(a):
    # input numpy array (float64) length n
    # clip a to [0,1000]
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    # compute conv using FFT:
    # pad length to next power of two >= 2n
    n = a.shape[0]
    m = 1 << (2*n-1).bit_length()
    # Actually we can use np.fft.rfft: next_pow2 = 1 << (2*n-1).bit_length()
    # compute pad
    a_pad = np.zeros(m, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    B = A * A
    conv = np.fft.irfft(B)[:2*n-1]
    max_b = conv.max()
    # score formula
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full) of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    m = _next_pow2(n)
    A = np.fft.rfft(a, m)
    B = np.fft.rfft(b, m)
    C = A * B
    conv = np.fft.irfft(C, m)[:n]
    return conv
