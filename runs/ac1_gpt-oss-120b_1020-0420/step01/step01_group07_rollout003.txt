import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: count ordered pairs (i,j) with i+j = k for a binary mask
# ----------------------------------------------------------------------
def _pair_counts(selected, n):
    """selected : set of indices (0 … n‑1) where the coefficient is 1
       returns   : np.ndarray of length 2*n‑1 with ordered‑pair counts"""
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = np.array(sorted(selected), dtype=np.int32)
    L = idx.size
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt


# ----------------------------------------------------------------------
# Helper: build sparse matrix A such that (A @ g)[k] = (f * g)[k]
# ----------------------------------------------------------------------
def _build_A(f):
    """Return a CSR matrix of shape (2*len(f)-1, len(f))"""
    n = f.shape[0]
    rows = 2 * n - 1
    if np.all(f == 0):
        return sp.csr_matrix((rows, n), dtype=np.float64)

    cols = np.arange(n, dtype=np.int32)
    data_blocks = []
    row_blocks = []
    col_blocks = []

    for i, val in enumerate(f):
        if val == 0.0:
            continue
        rows_i = i + cols                     # rows where offset=i appears
        data_i = np.full(n, val, dtype=np.float64)
        data_blocks.append(data_i)
        row_blocks.append(rows_i)
        col_blocks.append(cols)

    data = np.concatenate(data_blocks)
    row_ind = np.concatenate(row_blocks)
    col_ind = np.concatenate(col_blocks)
    return sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n), dtype=np.float64)
