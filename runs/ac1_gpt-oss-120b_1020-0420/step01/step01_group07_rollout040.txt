import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    the AC1 upper bound defined by `evaluate_sequence`.  The routine uses a
    binary mask optimisation (simulated annealing on the set of indices) and
    a cheap continuous refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem length n (use the provided parent if available)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.shape[0]) if parent.ndim == 1 and parent.size > 0 else 256
    except NameError:
        n = 256

    # ------------------------------------------------------------------
    # Simple uniform fallback – always feasible (score = 2)
    # ------------------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # ------------------------------------------------------------------
    # Parameters for the binary‑mask simulated annealing
    # ------------------------------------------------------------------
    TARGET_BOUND = 1.50274349          # stop early if reached
    m = min(n, max(50, int(0.07 * n))) # size of the support set
    max_iters = 20000                  # total SA steps
    temp_start = 0.4
    temp_end = 0.01

    # ------------------------------------------------------------------
    # Initialise a random support set and its pair‑count array
    # ------------------------------------------------------------------
    selected = set(rng.choice(n, size=m, replace=False))
    cnt = np.zeros(2 * n - 1, dtype=np.int32)

    sel_list = list(selected)
    for idx, i in enumerate(sel_list):
        cnt[2 * i] += 1                # (i,i)
        for j in sel_list[idx + 1:]:
            cnt[i + j] += 2            # (i,j) and (j,i)

    max_c = int(cnt.max())
    cur_bound = 2.0 * n * max_c / (m * m)
    best_bound = cur_bound
    best_selected = set(selected)

    # ------------------------------------------------------------------
    # Simulated‑annealing loop: swap one index with another
    # ------------------------------------------------------------------
    for it in range(max_iters):
        # temperature schedule (linear)
        temp = temp_start + (temp_end - temp_start) * (it / max_iters)

        # pick an element to remove and a new element to insert
        rem = rng.choice(list(selected))
        while True:
            add = rng.integers(0, n)
            if add not in selected:
                break

        # ---- remove contributions of `rem` --------------------------------
        for i in list(selected):               # copy because we mutate afterwards
            s = i + rem
            if i == rem:
                cnt[s] -= 1
            else:
                cnt[s] -= 2
        selected.remove(rem)

        # ---- add contributions of `add` ----------------------------------
        for i in selected:                     # `selected` now lacks `rem` and `add`
            cnt[i + add] += 2
        cnt[2 * add] += 1
        selected.add(add)

        # evaluate new bound
        new_max = int(cnt.max())
        new_bound = 2.0 * n * new_max / (m * m)

        # ---- acceptance test ---------------------------------------------
        accept = False
        if new_bound < cur_bound:
            accept = True
        else:
            # Metropolis rule
            if rng.random() < np.exp(-(new_bound - cur_bound) / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_bound = new_bound
            max_c = new_max
            if cur_bound < best_bound:
                best_bound = cur_bound
                best_selected = set(selected)
                if best_bound <= TARGET_BOUND:
                    break
        else:
            # ---- revert the swap -----------------------------------------
            # remove the added contributions
            for i in selected:
                if i != add:
                    cnt[i + add] -= 2
            cnt[2 * add] -= 1
            selected.remove(add)

            # re‑add the removed contributions
            for i in selected:
                cnt[i + rem] += 2
            cnt[2 * rem] += 1
            selected.add(rem)

    # ------------------------------------------------------------------
    # Build the best binary vector found
    # ------------------------------------------------------------------
    cand_vec = np.zeros(n, dtype=np.float64)
    if best_selected:
        cand_vec[list(best_selected)] = 1.0
    cand_score = float(evaluate_sequence(cand_vec.tolist()))
    if np.isfinite(cand_score) and cand_score < best_score:
        best_vec = cand_vec
        best_score = cand_score

    # ------------------------------------------------------------------
    # Small continuous refinement (multiplicative hill‑climb)
    # ------------------------------------------------------------------
    for _ in range(300):
        width = int(rng.integers(1, max(2, n // 50)))
        start = int(rng.integers(0, n - width + 1))
        stop = start + width
        trial = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.04)))   # ≈ 4 % random change
        trial[start:stop] *= factor
        np.clip(trial, 0.0, 1000.0, out=trial)
        sc = float(evaluate_sequence(trial.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_vec = trial
            best_score = sc

    return [float(x) for x in best_vec]
