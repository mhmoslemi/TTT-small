    steps = 5000

        max_idx = conv.argmax()

        partner_idx = max_idx - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]
        # source: max contribution index
        src_candidates = np.where(contrib > 0)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[contrib[mask].argmax()] # maybe compute directly.
        # target: minimal contribution (or zero)
        # choose dst where contrib is minimal and best[dst] < 1000
        dst_candidates = np.where(best < 1000.0)[0]
        if dst_candidates.size == 0:
            break

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0 * (i+1)**2)
    return a

def _score(seq):
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    maxb = conv.max()
    return float(2.0 * seq.size * maxb / (s * s))

import numpy as np

# ------------------------------------------------------------
# Helper: score function identical to the evaluator (no clipping here,
# the candidate is already clipped before scoring)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Compute 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution (FFT for large n)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Catalan–like central‑binomial sequence (a0 = 1, a{i+1}=a{i}*(2i+2)*(2i+1)/(4*(i+1)^2))
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Deterministic refinement: shrink a prefix and inflate the tail.
# ------------------------------------------------------------
def _deterministic_refine(seq: np.ndarray, n: int, cur_score: float):
    best = seq.copy()
    best_sc = cur_score

    # explore many prefix sizes and shrink factors (still cheap – O(250) scores)
    ratios = np.linspace(0.02, 0.50, 25)          # proportion taken as “early” block
    shrink_vals = np.linspace(0.10, 1.00, 20)    # how much to shrink the early block

    for r in ratios:
        k = max(1, int(n * r))                  # early block length
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue

        for f in shrink_vals:
            new_early = early * f
            delta = early_sum - new_early.sum()          # mass removed from early part
            g = 1.0 + delta / tail_sum                    # uniform inflation factor for the tail
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_sc:
                best = cand
                best_sc = sc
                # reuse the improved candidate for subsequent shrink factors
                early = best[:k]
                tail = best[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()
                if tail_sum == 0.0:
                    break

    return best, best_sc

    # ------------------------------------------------------------------
