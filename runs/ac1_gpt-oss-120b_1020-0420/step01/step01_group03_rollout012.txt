import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (size is next power of two)."""
    n = x.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=m)
    prod = fx * fx
    conv = np.fft.irfft(prod, n=m)
    return conv[: 2 * n - 1]


def _objective(x: np.ndarray) -> float:
