def compute_grad(a):
    n = a.shape[0]
    c = np.convolve(a, a)
    M = c.max()
    idxs = np.where(c == M)[0]
    S = a.sum()
    grad_M = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # range i_start..i_end inclusive
        # compute a_sub = a[k - np.arange(i_start, i_end+1)]

        i_range = np.arange(i_start, i_end+1)
        a_sub = a[k - i_range]
        grad_M[i_start:i_end+1] += 2.0 * a_sub
    factor = (2.0 * n) / (S**3)
    grad = factor * (S * grad_M - 2.0 * M)
    return grad, M, S

def compute_score(a):

    c = np.convolve(a, a)
    M = c.max()
    S = a.sum()
    n = a.shape[0]
    # avoid division by zero: if S < 0.01, return inf
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

import numpy as np

def _objective(a):
    """Return the AC1 bound for a non‑negative sequence a."""
    # a is a 1‑D numpy array of floats
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    M = c.max()
    n = a.shape[0]
    return 2.0 * n * M / (s * s)

def _grad_and_metrics(a):
    """Return gradient of the AC1 bound w.r.t. a, together with the current bound."""
    n = a.shape[0]
    c = np.convolve(a, a)
    M = c.max()
    S = a.sum()
    # gradient of M (sub‑gradient) – only indices attaining the max matter
    idxs = np.where(c == M)[0]
    grad_M = np.zeros_like(a)
    # For each critical convolution index add 2*a[k‑i] to the corresponding i
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vector of i values
        i_vals = np.arange(i_start, i_end + 1)
        # a[k-i] for each i
        a_sub = a[k - i_vals]
        grad_M[i_start:i_end + 1] += 2.0 * a_sub
    # gradient of the bound f = 2 n M / S^2
    # df/da_i = 2 n (S * dM/da_i - 2 M) / S^3
    if S == 0.0:
        factor = 0.0
    else:
        factor = (2.0 * n) / (S ** 3)
    grad = factor * (S * grad_M - 2.0 * M)
    # current bound value
    if S < 0.01:
        bound = float('inf')
    else:
        bound = 2.0 * n * M / (S * S)
    return grad, bound

import numpy as np

def _objective(a: np.ndarray) -> float:
    """AC1 upper bound for a non‑negative vector a."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)          # full linear convolution
    m = c.max()
    n = a.shape[0]
    return 2.0 * n * m / (s * s)


def _grad_and_metrics(a: np.ndarray):
    """
    Return the gradient of the AC1 bound w.r.t. a, together with the
    current bound value.  The gradient is a sub‑gradient that treats the
    maximal convolution entry as active.
    """
    n = a.shape[0]
    c = np.convolve(a, a)
    m = c.max()
    s = a.sum()

    # indices where the convolution attains its maximum
    idxs = np.where(c == m)[0]

    # sub‑gradient of the max‑convolution
    grad_m = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        # a[k - i] corresponds to the partner element in the pair product
        grad_m[i_start:i_end + 1] += 2.0 * a[k - i_range]

    # gradient of f(a) = 2 n m / s²  (see analysis)
    if s == 0.0:
        factor = 0.0
    else:
        factor = (2.0 * n) / (s ** 3)
    grad_f = factor * (s * grad_m - 2.0 * m)

    # current bound value
    bound = _objective(a)
    return grad_f, bound
