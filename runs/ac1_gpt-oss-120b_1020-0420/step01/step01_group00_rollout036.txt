import numpy as np

# ------------------------------------------------------------
# Fast AC1 score using FFT (same as the reference implementation)
# ------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ------------------------------------------------------------
# Full convolution (linear) of a vector with itself, using FFT
# ------------------------------------------------------------
def _full_conv(vec: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * vec.size - 1]
    return conv

            factor = np.exp(rng.normal(0.0, 0.15))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block = block / block.sum()
            blk_score = _fast_score(block)
            if blk_score < cur_score:
                cur = block
                cur_score = blk_score
                if blk_score < best_score:
                    best = block.copy()
                    best_score = blk_score

    # final validation with the official evaluator
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # guaranteed fallback: uniform distribution of the same length
        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
        best = fallback

    return [float(v) for v in best]
