def conv_full_fft(x, y):
    n = x.shape[0] + y.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, nfft)
    fy = np.fft.rfft(y, nfft)
    fz = fx * fy
    z = np.fft.irfft(fz, nfft)[:n]
    return z

import numpy as np

def _fast_score(a: np.ndarray) -> float:
    n = a.shape[0]
    m = 2*n -1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))

def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = x.shape[0] + y.shape[0] -1
    nfft = 1 << ((n-1).bit_length())
    fx = np.fft.rfft(x, nfft)
    fy = np.fft.rfft(y, nfft)
    fz = fx * fy
    z = np.fft.irfft(fz, n=nfft)[:n]
    return z

def _gradient_max_conv(a: np.ndarray, beta: float) -> np.ndarray:
    # a is 1D, length n, non‑negative, sum=1
    n = a.shape[0]
    # convolution a*a
    c = _conv_full_fft(a, a)   # length 2n-1
    c_max = np.max(c)
    # smooth weights for each lag
    w = np.exp(beta * (c - c_max))
    w_sum = np.sum(w)
    if w_sum == 0:

        w = np.full_like(w, 1.0 / w.size)

        w = w / w_sum

    idx = (2*n - 2) - np.arange(n)   # indices where gradient is located
    g = grad_full[idx]
    return g

    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


# ----------------------------------------------------------
# Convolution via FFT (full linear convolution)
# ----------------------------------------------------------
def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = x.shape[0] + y.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, nfft)
    fy = np.fft.rfft(y, nfft)
    fz = fx * fy
    z = np.fft.irfft(fz, n=nfft)[:n]
    return z


# ----------------------------------------------------------
# Gradient of the (smoothed) max‑convolution w.r.t. a
# ----------------------------------------------------------
def _gradient_max_conv(a: np.ndarray, beta: float) -> np.ndarray:
    n = a.shape[0]
    # convolution a*a
    c = _conv_full_fft(a, a)               # length 2n‑1
    c_max = np.max(c)
    # weights concentrating on large lags
    w = np.exp(beta * (c - c_max))
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # compute g_j = sum_i a_i * w_{i+j}
    a_ext = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    w_rev = w[::-1]
    grad_full = _conv_full_fft(a_ext, w_rev)          # length 4n‑3
    idx = (2 * n - 2) - np.arange(n)                  # positions of the gradient
    g = grad_full[idx]
    return g


# ----------------------------------------------------------
# Simple projected gradient descent on max‑convolution
# ----------------------------------------------------------
def _descend(
    a0: np.ndarray,
    iters: int = 250,
    lr0: float = 0.8,
    beta_start: float = 0.5,
    beta_end: float = 5000.0,
) -> (np.ndarray, float):
    a = np.maximum(a0, 0.0)
    s = np.sum(a)
    if s < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    best_a = a.copy()
    best_score = _fast_score(a)
    n = a.size
    for t in range(iters):
        beta = beta_start + (beta_end - beta_start) * (t / max(iters - 1, 1))
        g = _gradient_max_conv(a, beta)
        lr = lr0 * (1.0 - t / iters)
        a = a - lr * g
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-12:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        cur_score = _fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------
    # Obtain parent sequence if available.
    # ------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None

    # ------------------------------------------------------
    # Determine length.
    # ------------------------------------------------------
    if parent is not None:
        n = parent.size
    else:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------
    # Baseline candidate (uniform, normalized).
    # ------------------------------------------------------
    best_seq = np.ones(n, dtype=np.float64) / n
    best_score = _fast_score(best_seq)

    # ------------------------------------------------------
    # Check parent (normalized).
    # ------------------------------------------------------
    if np.sum(parent) >= 0.01:
        cand = parent / np.sum(parent)
        s = _fast_score(cand)
        if s < best_score:
            best_seq = cand.copy()
            best_score = s

    # ------------------------------------------------------
    # Compact analytic families.
    # ------------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    # Inverse‑square‑root family.
    offsets = np.linspace(0.1, 6.0, 40)
    for off in offsets:
        a = 1.0 / np.sqrt(i + off)
        a = a / np.sum(a)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = 1.0 / np.sqrt((n - 1 - i) + off)
        a_rev = a_rev / np.sum(a_rev)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # Power‑law family.
    powers = np.linspace(0.2, 0.9, 35)
    for p in powers:
        a = (i + 1.0) ** (-p)
        a = a / np.sum(a)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = (n - i) ** (-p)
        a_rev = a_rev / np.sum(a_rev)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # Exponential family.
    rates = np.linspace(0.0005, 0.015, 30)
    for r in rates:
        a = np.exp(-r * i)
        a = a / np.sum(a)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = np.exp(-r * (n - 1 - i))
        a_rev = a_rev / np.sum(a_rev)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------
    # Gradient‑descent refinement on the best candidate.
    # ------------------------------------------------------
    refined_seq, refined_score = _descend(
        best_seq,
        iters=300,
        lr0=0.8,
        beta_start=0.5,
        beta_end=5000.0,
    )
    if refined_score < best_score:
        best_seq, best_score = refined_seq, refined_score

    # ------------------------------------------------------
    # Final random multiplicative jitter.
    # ------------------------------------------------------
    for _ in range(300):
        perturb = np.exp(rng.normal(0.0, 0.015, size=n))
        cand = best_seq * perturb
        cand = cand / np.sum(cand)
        s = _fast_score(cand)
        if s < best_score:
            best_seq = cand
            best_score = s

    # ------------------------------------------------------
    # Return a plain Python list.
    # ------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
