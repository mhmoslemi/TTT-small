def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)

    # Determine length n
    if 'height_sequence_1' in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
        n = parent_arr.size
    else:
        n = 1024
        parent_arr = np.ones(n, dtype=np.float64)

    # Helper scoring
    def score(arr):
        s = evaluate_sequence(arr.tolist())
        return float('inf') if not np.isfinite(s) else s

    # Keep best candidate
    best = parent_arr.copy()
    best_score = score(best)

    # Phase 1: try power-law and exponential families (like before) but with finer grid & more points
    indices = np.arange(1, n+1, dtype=np.float64)

    # Explore powerlaw exponents from 0.05 to 1.0
    for a in np.linspace(0.05, 1.0, 96):
        w = np.power(indices, -a)
        w /= w.sum()
        sc = score(w)
        if sc < best_score:
            best = w.copy()
            best_score = sc

    # Explore exponential decays from 1e-5 to 5e-3
    for b in np.linspace(1e-5, 5e-3, 100):
        w = np.exp(-b * np.arange(n, dtype=np.float64))
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w /= w_sum
        sc = score(w)
        if sc < best_score:
            best = w.copy()
            best_score = sc

    # Phase 2: random permutation search on best-found weighting distribution shape
    # Use weights from best as base set to permute
    base_weights = best.copy()
    # Shuffle to start
    perm = rng.permutation(n)
    cur = base_weights[perm]
    cur_score = score(cur)
    if cur_score < best_score:
        best = cur.copy()
        best_score = cur_score

    # Simulated annealing / hill climb using swaps
    max_iters = 2000
    for it in range(max_iters):
        # pick two indices to swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # do swap in copy
        cand = cur.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = score(cand)
        # Accept if improvement or with small probability decreasing over time
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(1e-12, (it+1))):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # occasional small multiplicative noise (global)
        if it % 200 == 0:
            noise = np.exp(rng.normal(0.0, 0.03, size=n))
            cand_noise = cur * noise
            cand_noise = np.clip(cand_noise, 0.0, 1000.0)
            cand_noise /= cand_noise.sum()  # renorm to keep sum unchanged (scale invariance)
            cand_noise_score = score(cand_noise)

                cur = cand_noise
                cur_score = cand_noise_score
                if cur_score < best_score:
                    best = cur.copy()
                    best_score = cur_score

    # Return best as list
    return [float(v) for v in best]

    indices = np.arange(1, n+1, dtype=np.float64)

            best_score = sc

    for b in np.linspace(1e-5, 5e-3, 30):

        w_sum = w.sum()

            continue
        w /= w_sum
        sc = score(w)
        if sc < best_score:
            best = w.copy()
            best_score = sc

    # Gamma family
    for shape in np.linspace(0.2, 3.0, 15):
        w = rng.gamma(shape, 1.0, size=n)
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w /= w_sum
        sc = score(w)
        if sc < best_score:
            best = w.copy()
            best_score = sc

    # Phase 2: Random permutation + hill climbing
    # Take current best as weight multiset
    base_weights = best.copy()
    # initial random permutation
    cur = rng.permutation(base_weights)
    cur_score = score(cur)
    if cur_score < best_score:
        best = cur.copy()
        best_score = cur_score

    # hill climbing
    max_iters = 2000
    for it in range(max_iters):
        # swap two random positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = cur.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = score(cand)
        # accept if better
        if cand_score < cur_score:

                best_score = cur_score
        # occasional block multiplicative perturb
        if it % 250 == 0:

                cur = cand2
                cur_score = cand2_score
                if cur_score < best_score:
                    best = cur.copy()
                    best_score = cur_score

    # Ensure final result is non-empty and finite
    # Clip just in case
    best = np.clip(best, 0.0, 1000.0)
    # Return as list
    return [float(v) for v in best]
