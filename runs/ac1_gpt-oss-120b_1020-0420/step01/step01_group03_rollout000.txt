def _next_pow2(x):
    return 1 << (int(x - 1).bit_length())

def _conv_self(a):
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2 * n - 1]
    return conv

def _maxconv_gradient(a, conv):
    n = a.shape[0]
    k = int(np.argmax(conv))
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - idx
    g = np.zeros_like(a)

    if (k & 1) == 0:
        i_mid = k // 2
        if 0 <= i_mid < n:
            g[i_mid] = a[i_mid]  # adjust diagonal
    return g, k

def _score_from_conv(conv, s, n):
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)

def _improve(a, max_iter=50, lr=0.02):
    n = a.shape[0]
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = _score_from_conv(conv, a.sum(), n)
    for _ in range(max_iter):
        s = a.sum()
        if s < 1e-12:
            break
        conv = _conv_self(a)
        max_b = np.max(conv)
        g, k = _maxconv_gradient(a, conv)
        factor = 2.0 * n / (s * s)
        const = 2.0 * max_b / s
        grad = factor * (g - const)
        # take opposite direction
        step = lr * grad
        a_new = a - step

import numpy as np

# ---------- utility ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x (x > 0)."""
    return 1 << (int(x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _maxconv_gradient(a: np.ndarray, conv: np.ndarray):
    """
    Gradient of the maximal convolution entry with respect to a.
    Returns the gradient vector g and the index k where the max occurs.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))

    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - idx

    g = np.zeros_like(a)
    g[idx] = a[j] * 2.0

    # diagonal (i = k-i) appears only once
    if (k & 1) == 0:
        i_mid = k // 2
        if 0 <= i_mid < n:
            g[i_mid] = a[i_mid]

    return g, k


def _score_from_conv(conv: np.ndarray, s: float, n: int) -> float:
    """Exact AC1 bound given a convolution and sum."""
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)


# ---------- root‑constant sequence ----------
def _root_constant_seq(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


# ---------- improvement routine ----------
def _improve(a: np.ndarray, max_iter: int = 50, lr: float = 0.02):
    """Simple gradient‑descent on the AC1 bound (matrix‑free)."""
    n = a.shape[0]
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = _score_from_conv(conv, a.sum(), n)

    for _ in range(max_iter):
        s = a.sum()
        if s < 1e-12:
            break

        conv = _conv_self(a)
        max_b = np.max(conv)

        g, _ = _maxconv_gradient(a, conv)

        factor = 2.0 * n / (s * s)          # 2n / S^2
        const = 2.0 * max_b / s            # 2M / S
        grad = factor * (g - const)        # ∇B

        # step opposite to gradient (descent)
        step = lr * grad
        a_new = a - step
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        # evaluate new bound without calling the external evaluator
        s_new = a_new.sum()
        conv_new = _conv_self(a_new)
        score_new = _score_from_conv(conv_new, s_new, n)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.2, 0.5)    # gently accelerate
        else:
            lr *= 0.5                  # back‑off
            if lr < 1e-8:
                break

    return best_a, best_score
