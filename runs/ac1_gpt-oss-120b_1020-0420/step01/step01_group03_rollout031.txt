import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate the first ``length`` terms of a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full linear convolution arr * arr."""
    n = arr.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    ft = np.fft.rfft(arr, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * n - 1]
    return float(conv.max())
