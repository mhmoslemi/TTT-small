def correlation(a, w):
    n = len(a)
    m = len(w)
    N = 1 << ((n + m - 1).bit_length())
    fft_a = np.fft.rfft(a, N)

    start = m - 1

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """
    a_i = C(2i,i)/4**i  (central binomial / 4^i),  a_0 = 1,
    satisfies (a*a)[k] == 1 for 0 <= k < length.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _next_pow2(x: int) -> int:
    """small helper: next power of two >= x."""
    return 1 << (x - 1).bit_length()
