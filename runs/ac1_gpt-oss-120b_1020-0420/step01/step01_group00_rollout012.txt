import numpy as np

def _fast_score(vec):
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)

def propose_candidate(seed=42, num_steps=12000, top_k=5, eta0=0.08, T0=0.08, **kwargs):
    rng = np.random.default_rng(seed)
    # initial vector
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            raise ValueError
    except Exception:
        # fallback uniform length 2048 maybe?
        init = np.ones(256, dtype=np.float64)
    cur = np.clip(init, 0.0, 1000.0)
    # ensure sum > 0
    if cur.sum() < 1e-12:
        cur[:] = 1.0
    cur = cur / cur.sum()
    # random small perturbation to break symmetry
    cur = cur * np.exp(rng.normal(0.0, 0.03, size=cur.shape))
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur / cur.sum()
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.size
    v = np.zeros_like(cur)  # momentum
    beta = 0.9
    eps = 1e-12
    for step in range(num_steps):
        # convolution via FFT
        mlen = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(cur, n=mlen)
        conv = np.fft.irfft(fa * fa, n=mlen)[:2*n-1]
        # find top_k peaks (including zero lag)
        # we can include zero lag automatically
        # get indices of largest values
        # use argpartition for efficiency
        top_idx = np.argpartition(-conv, top_k)[:top_k]
        # compute gradient
        grad = np.zeros_like(cur)
        for k in top_idx:
            # overlap slice
            start = max(0, k - n + 1)
            end = min(n, k + 1)
            # indices in cur that overlap: idx = np.arange(start, end)
            # Corresponding indices of cur for a_{k-i}: rev_idx = k - idx
            # So grad[idx] += cur[rev_idx]
            # Use view slicing for speed
            idx = slice(start, end)
            rev = slice(k - end + 1, k - start + 1)  # careful: Python slice exclusive end
            # Actually rev range: for i from start to end-1, rev_i = k - i, which runs from k - start down to k - (end-1)
            # So rev slice order reversed relative to idx; but assignment order not matter since sum.
            # We can assign using np.add.at perhaps. However slice reversed can be reversed.
            # Simpler: loop over i (slower). But K small, n moderate; we can use vectorized.
            i_range = np.arange(start, end)
            rev_i = k - i_range
            grad[i_range] += cur[rev_i]
        # gradient is sum of contributions (not multiplied by 2 yet)
        grad = 2.0 * grad
        # momentum update
        v = beta * v + (1 - beta) * grad
        # step size schedule
        eta = eta0 * (1.0 - step / num_steps)
        # candidate
        cand = cur - eta * v
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < eps:
            cand = cur.copy()
            s = cand.sum()
        cand = cand / s
        cand_score = _fast_score(cand)

    return [float(v) for v in best]

import numpy as np

# fast AC1 upper bound (same formula as the official evaluator, but using FFT)
def _fast_score(vec: np.ndarray) -> float:
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Returns a list of non‑negative step‑function heights that (hopefully) lower
    the AC1 upper bound.  The implementation is a gradient‑momentum search that
    smooths the largest autocorrelation peaks while keeping the vector normalized.
    """
    rng = np.random.default_rng(seed)

    # ----- initialise from provided parent if possible -----------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            raise ValueError
    except Exception:
        # fallback: short uniform vector (the optimiser works for any length)
        init = np.ones(256, dtype=np.float64)

    cur = np.clip(init, 0.0, 1000.0)
    if cur.sum() < 1e-12:
        cur[:] = 1.0
    cur = cur / cur.sum()                     # normalise sum = 1

    # small random perturbation to break symmetry
    cur = cur * np.exp(rng.normal(0.0, 0.03, size=cur.shape))
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur / cur.sum()

    n = cur.size
    uniform = np.full_like(cur, 1.0 / n)       # reference uniform distribution

    # hyper‑parameters
    num_steps = 12000          # total iterations
    top_k = 5                  # number of strongest autocorrelation lags to flatten
    eta0 = 0.02                # initial step size (will decay linearly)
    T0 = 0.02                  # initial temperature for occasional uphill moves
    beta = 0.9                 # momentum coefficient
    lam = 0.04                 # regularisation strength towards uniformity
    eps = 1e-12

    # state variables
    v = np.zeros_like(cur)                     # momentum vector
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    # pre‑compute FFT length (constant throughout)
    fft_len = 1 << ((2 * n - 1).bit_length())

    for step in range(num_steps):
        # ----- autocorrelation via FFT -----
        fa = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

        # ----- identify the top_k largest lags -----
        # argpartition gives unsorted top indices efficiently
        top_idx = np.argpartition(-conv, top_k)[:top_k]

        # ----- gradient: sum of shifted copies of cur corresponding to those lags -----
        grad = np.zeros_like(cur)
        for k in top_idx:
            start = max(0, k - n + 1)
            end = min(n, k + 1)
            i = np.arange(start, end, dtype=int)
            grad[i] += cur[k - i]

        # ----- combine with uniform regularisation -----
        direction = 2.0 * grad - lam * (cur - uniform)

        # ----- momentum update -----
        v = beta * v + (1.0 - beta) * direction

        # ----- annealed step size -----
        eta = eta0 * (1.0 - step / num_steps)          # linear cooling
        cand = cur - eta * v
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < eps:                                   # safeguard against degeneration
            cand = cur.copy()
            s = cand.sum()
        cand = cand / s

        cand_score = _fast_score(cand)

        # ----- simulated‑annealing acceptance -----
        T = T0 * (1.0 - step / num_steps)
        accept = False
        if cand_score < cur_score:
            accept = True
        elif T > 0 and rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-16)):
            accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # ----- occasional random block perturbation (exploration) -----
        if (step + 1) % 400 == 0 and rng.random() < 0.2:
            width = int(rng.integers(max(1, n // 200), max(2, n // 30)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            block = cur.copy()
            factor = np.exp(rng.normal(0.0, 0.07))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block = block / block.sum()
            block_score = _fast_score(block)
            if block_score < cur_score:
                cur = block
                cur_score = block_score
                if block_score < best_score:
                    best = block.copy()
                    best_score = block_score

    # ----- final verification with the official evaluator -----
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # guaranteed finite fallback: uniform distribution
        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
        best = fallback

    # return a plain Python list of floats
    return [float(v) for v in best]
