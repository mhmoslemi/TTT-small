def _eval(vec: np.ndarray) -> float:
    """Return evaluate_sequence result for a NumPy vector."""
    # evaluate_sequence internally clips to [0,1000] and enforces non‑negativity.
    return float(evaluate_sequence(vec.tolist()))


def _conv_stats(vec: np.ndarray):

    conv = np.convolve(vec, vec)
    max_val = conv.max()
    argmax = int(conv.argmax())
    return conv, max_val, argmax
