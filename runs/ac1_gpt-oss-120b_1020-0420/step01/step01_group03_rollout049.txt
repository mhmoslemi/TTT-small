    g = cp.Variable(n, nonneg=True)
    constraints = []
    # compute convolution constraints
    # We can precompute for each k the list of (i) where f[i]!=0
    for k in range(2*n - 1):
        # i range
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_min > i_max:
            continue
        # Build linear expression
        expr = 0
        # vectorized: use g[i_idx] terms scaling.
        # but we can compute slice positions
        indices_i = np.arange(i_min, i_max + 1)
        indices_g = k - indices_i

        # Use cp.sum(cp.multiply(f[indices_i], g[indices_g]))

    prob = cp.Problem(objective, constraints)

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Root‑of‑constant sequence a_i = C(2i,i)/4**i for i=0..length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _conv_max(a: np.ndarray) -> float:
    """Maximum value of the full convolution a*a using an FFT (O(n log n))."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[:size]
    return float(conv.max())

    total = float(a.sum())
    if total < 0.01:
        return np.inf
    M = _conv_max(a)
    return 2.0 * n * M / (total * total)
