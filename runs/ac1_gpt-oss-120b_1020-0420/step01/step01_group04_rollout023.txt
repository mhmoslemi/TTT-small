import numpy as np

def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2  (C_i = comb(2i,i)/4^i)."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _scale_low_block(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Scale a contiguous block that is guaranteed to start after the
    very first few (large) entries.  The block is chosen uniformly,
    its length is modest (≈1‑5 % of n) and the multiplicative factor
    follows a log‑normal law with a relatively large spread to raise
    the total sum noticeably while keeping the peak convolution
    increase limited.
    """
    n = seq.size
    # avoid the first few dominant coefficients
    low_start = max(1, n // 100)
    # block length: between 0.5 % and 5 % of n
    min_len = max(1, n // 200)
    max_len = max(2, n // 20)
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(low_start, n - width + 1))
    # log‑normal factor, sigma large enough to give >1 expectation
    sigma = max(0.05, 0.5 * (1.0 - rng.random()))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand = seq.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

            cand = _scale_low_block(best, rng)
        elif op == 1:
            cand = _scale_random_block(best, rng)
        elif op == 2:
            cand = _add_one_bump(best, rng)
        elif op == 3:
            cand = _swap_two(best, rng)
        else:
            cand = _shuffle_slice(best, rng)

        # evaluate with the provided safe evaluator
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        # accept only if strictly better (non‑increasing sequence)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Return the best-found candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
