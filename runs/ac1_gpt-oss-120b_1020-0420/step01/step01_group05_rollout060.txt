import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Compute 2 * n * max(conv(a, a)) / (sum(a) ** 2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                     # convolution in frequency domain
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)
