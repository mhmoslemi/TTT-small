    return [float(x) for x in best_seq.tolist()]

def _fast_score(a):

    # 1) Power‑law with an offset: a_i = (i+off)^{-p}
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.2, 1.5, 14)          # 14 trial exponents
    off_vals = np.linspace(0.01, 20.0, 20)      # 20 offsets
    for p in p_vals:
        for off in off_vals:
            a = (i + off) ** (-p)
            s = _fast_score(a)
            if s < best_score:
                best_seq, best_score = a, s
            a_rev = (i_rev + off) ** (-p)
            s = _fast_score(a_rev)
            if s < best_score:
                best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 2) Pure exponential decay: a_i = exp(-γ * i)
    # ------------------------------------------------------------------
    gamma_vals = np.logspace(-4, -1, 20)        # γ from 1e‑4 to 1e‑1
    for gamma in gamma_vals:
        a = np.exp(-gamma * i)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = np.exp(-gamma * i_rev)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 3) Combined power‑law and exponential (coarse grid)
    # ------------------------------------------------------------------
    for p in p_vals[::3]:                     # every third exponent
        for gamma in gamma_vals[::4]:          # every fourth decay factor
            a = (i + 1) ** (-p) * np.exp(-gamma * i)
            s = _fast_score(a)
            if s < best_score:
                best_seq, best_score = a, s
            a_rev = (i_rev + 1) ** (-p) * np.exp(-gamma * i_rev)
            s = _fast_score(a_rev)
            if s < best_score:
                best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 4) Inverse‑sqrt family with finer offsets
    # ------------------------------------------------------------------
    offsets = np.linspace(0.001, 10.0, 30)      # 30 offsets
    for off in offsets:
        a = 1.0 / np.sqrt(i + off)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = 1.0 / np.sqrt(i_rev + off)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 5) Random multiplicative jitter around the current best.
    # ------------------------------------------------------------------
    for _ in range(300):
        perturb = np.exp(rng.normal(0.0, 0.015, size=n))
        cand = best_seq * perturb
        cand = np.clip(cand, 0.0, 1000.0)
        s = _fast_score(cand)
        if s < best_score:
            best_seq, best_score = cand, s

    # ------------------------------------------------------------------
    # 6) Simple hill‑climbing moves (mass transfer between two indices).
    # ------------------------------------------------------------------
    eps = 0.02
    for step in range(200):
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        if i1 == i2:
            continue
        delta = eps * best_seq[i1]
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i1] -= delta
        new_seq[i2] += delta
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        s = _fast_score(new_seq)
        if s < best_score:
            best_seq, best_score = new_seq, s
        if step % 50 == 0 and eps > 0.005:
            eps *= 0.9

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
