import numpy as np

def _full_conv(x):
    n = x.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2 * n - 1]
    return conv

def _fast_score(x):
    n = x.size
    conv = _full_conv(x)
    max_c = conv.max()
    s = x.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def propose_candidate(seed: int = 42, num_steps: int = 25000, top_k: int = 8, **kwargs):
    rng = np.random.default_rng(seed)
    # load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.full(256, 1.0, dtype=np.float64)
    n = parent.size
    cur = np.clip(parent, 0.0, 1000.0)
    cur = np.maximum(cur, 0.0)
    if cur.sum() < 1e-12:
        cur[:] = 1.0
    cur = cur / cur.sum()   # normalize sum to 1

    # add small random noise to escape uniform
    cur = cur * np.exp(rng.normal(0.0, 0.05, size=n))
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur / cur.sum()

    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    # momentum
    velocity = np.zeros_like(cur)

    # schedule parameters
    init_eta = 0.12
    momentum = 0.9

    m_fft = 1 << ((2 * n - 1).bit_length())

    for step in range(num_steps):
        # compute conv
        fa = np.fft.rfft(cur, n=m_fft)
        conv = np.fft.irfft(fa * fa, n=m_fft)[:2 * n - 1]
        max_c = conv.max()
        # compute gradient using top_k peaks
        if top_k >= conv.size:
            top_indices = np.arange(conv.size)
        else:
            # get indices of largest top_k values
            part = np.argpartition(conv, -top_k)[-top_k:]
            top_indices = part
        b = np.zeros_like(cur)
        for k_idx in top_indices:
            idx = k_idx - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            b[mask] += cur[idx[mask]]
        # subtract mean to keep sum unchanged
        grad = 2.0 * (b - b.mean())

        # step size schedule: linear decay
        eta = init_eta * (1.0 - step / num_steps)
        # update velocity and candidate
        velocity = momentum * velocity - eta * grad
        cand = cur + velocity

        # projection onto feasible set: clip and renorm
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = cur.copy()
            s = cand.sum()
        cand = cand / s

        cand_score = _fast_score(cand)

        # acceptance: always accept if improves, else accept with probability
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            # simulated annealing style temperature decays similarly to eta
            T = 0.02 * (1.0 - step / num_steps)  # small temperature
            if T > 0 and rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-16)):
                cur = cand
                cur_score = cand_score
                # not updating best unless better

                    best_score = block_score

import numpy as np

# ----- fast convolution (full) using FFT -----
def _full_conv(x: np.ndarray) -> np.ndarray:
    """Return full linear convolution of x with itself using FFT."""
    n = x.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fx * fx, n=m)[: 2 * n - 1]
    return conv


# ----- fast AC1 upper bound (sum is not required to be 1) -----
def _fast_score(x: np.ndarray) -> float:
    n = x.size
    conv = _full_conv(x)
    max_c = conv.max()
    s = x.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
