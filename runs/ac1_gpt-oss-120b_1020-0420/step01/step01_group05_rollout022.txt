import numpy as np

def _fast_score(a):
    n = a.shape[0]
    m = 2*n-1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa*fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _gradient_step(a, step):
    n = a.shape[0]
    # compute convolution via FFT for speed
    m = 2*n - 1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv_full = np.fft.irfft(fa*fa, n=nfft)[:m]
    k = np.argmax(conv_full)
    M = conv_full[k]
    S = np.sum(a)
    # compute gradient of max conv at shift k
    # grad_conv[j] = 2*a[k-j] if 0 <= k-j < n else 0
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad_conv = np.zeros_like(a)
    grad_conv[mask] = 2.0 * a[idx[mask]]
    scale1 = 2.0 * n / (S * S)
    scale2 = -4.0 * n * M / (S * S * S)
    grad = scale1 * grad_conv + scale2 * np.ones_like(a)
    new_a = a - step * grad
    # projection
    new_a = np.clip(new_a, 0.0, 1000.0)
    # optional: if sum becomes too small after clipping, re-normalize to keep sum
    if np.sum(new_a) < 0.01:
        # fallback: revert to a
        return a
    return new_a

import numpy as np

# ----------------------------------------------------------
# Fast AC1 bound evaluation (FFT based, O(n log n))
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------
# One gradient step for the max‑convolution objective.
# ----------------------------------------------------------
def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """Take a projected sub‑gradient step to lower max(conv(a,a))."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv_full = np.fft.irfft(fa * fa, n=nfft)[:m]
    k = int(np.argmax(conv_full))               # shift where max occurs
    M = conv_full[k]
    S = np.sum(a)
    if S < 0.01:
        return a

    # gradient of the max‑convolution term
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad_conv = np.zeros_like(a)
    grad_conv[mask] = 2.0 * a[idx[mask]]

    # full gradient of the objective
    scale1 = 2.0 * n / (S * S)                  # from max(conv) term
    scale2 = -4.0 * n * M / (S * S * S)         # from denominator term
    grad = scale1 * grad_conv + scale2 * np.ones_like(a)

    new_a = a - step * grad
    new_a = np.clip(new_a, 0.0, 1000.0)
    # keep a viable (non‑zero) sum
    if np.sum(new_a) < 0.01:
        return a
    return new_a
