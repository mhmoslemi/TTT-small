def compute_ac(seq):
    n = seq.size
    # next power of two for length at least 2*n-1
    m = 1 << ((2*n-1).bit_length())
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)
    max_b = np.max(conv[:2*n-1])  # only first part
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

# ------------------------------------------------------------
# Generate truncated series of (1 - z)^(-1/2)
# a_0 = 1,  a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Compute the AC1 upper bound as defined by the harness.
# Uses FFT for O(n log n) convolution.
# ------------------------------------------------------------
def compute_ac(seq: np.ndarray) -> float:
    n = seq.shape[0]
    # length needed for linear convolution
    m = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)
    max_b = conv[: 2 * n - 1].max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

# Random multiplicative perturbation of a contiguous block.
# ------------------------------------------------------------
def perturb_block(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.03) -> np.ndarray:
    n = seq.shape[0]
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand = seq.copy()
    cand[start : start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Smooth the sequence with a uniform moving average.
# Afterwards rescale to keep the total sum unchanged.
# ------------------------------------------------------------
def smooth_sequence(seq: np.ndarray, rng: np.random.Generator,
                    min_width: int = 3, max_frac: float = 0.15) -> np.ndarray:
    n = seq.shape[0]
    max_width = max(min_width, int(n * max_frac))
    width = int(rng.integers(min_width, max_width + 1))
    if width % 2 == 0:
        width += 1
    kernel = np.full(width, 1.0 / width, dtype=np.float64)
    smoothed = np.convolve(seq, kernel, mode="same")
    sum_orig = seq.sum()
    sum_smooth = smoothed.sum()
    if sum_smooth == 0.0:
        return seq.copy()
    cand = smoothed * (sum_orig / sum_smooth)
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial sequence: use the provided parent if present, otherwise
    # fall back to the half‑power series.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):
        n = int(height_sequence_1.shape[0])
        base = height_sequence_1.astype(np.float64).copy()
    else:
        n = 256
        base = generate_half_power_series(n)

    np.clip(base, 0.0, 1000.0, out=base)

    # also consider a fresh half‑power series as an alternative start
    alt = generate_half_power_series(n)
    np.clip(alt, 0.0, 1000.0, out=alt)

    best_seq = base
    best_score = compute_ac(best_seq)
    alt_score = compute_ac(alt)
    if alt_score < best_score:
        best_seq = alt
        best_score = alt_score

    # ------------------------------------------------------------
    # Simple stochastic hill‑climbing with two move types.
    # ------------------------------------------------------------
    stagnation = 0
    TARGET = 1.50274349  # target bound to stop early if reached
    for _ in range(num_steps):
        if rng.random() < 0.7:
            cand = perturb_block(best_seq, rng, sigma=0.03)
        else:

        score = compute_ac(cand)

        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            stagnation = 0
            if best_score <= TARGET:

        else:
            stagnation += 1

        # occasional global scaling to escape flat regions
        if stagnation >= 250:

            global_factor = float(np.exp(rng.normal(0.0, 0.05)))
            g *= global_factor
            np.clip(g, 0.0, 1000.0, out=g)
            score_g = compute_ac(g)
            if np.isfinite(score_g) and score_g < best_score:
                best_seq = g
                best_score = score_g
                stagnation = 0

    # ------------------------------------------------------------
    # Guard against pathological sum (should not happen).
    # ------------------------------------------------------------
    if best_seq.sum() < 0.01:
