def proj_simplex(v, s=1):
    # v: np.ndarray, shape (n,)
    # Returns projection onto simplex {x >=0, sum x = s}
    # Implementation based on https://doi.org/10.1137/070710111
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all are nonpositive after shift; return zero then adjust
        theta = 0
    else:
        rho = np.argmax(cond)  # last true? Actually we need max i satisfying cond; we can compute np.where(cond)[0][-1]
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # ensure sum is s (due to numeric errors)
    w_sum = w.sum()
    if w_sum > 0:
        w = w * (s / w_sum)
    else:
        # all zero; assign uniform distribution
        w = np.full_like(v, s / v.size)
    return w

import numpy as np

def _fast_score(vec: np.ndarray) -> float:
    """Fast AC1 bound for a non‑negative vector (sum not required to be 1)."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Projection of v onto the simplex {x >= 0, sum x = s}."""
    # Numerical stability: work with float64
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, return uniform
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑scale very small numeric drift
    w_sum = w.sum()
    if w_sum > 0:
        w *= s / w_sum
    else:
        w = np.full_like(v, s / n)
    return w
