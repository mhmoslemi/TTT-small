def _root_constant_seq(length: int) -> np.ndarray:

    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Greedy construction of a feasible g such that f * g ≤ M (pointwise)
def _greedy_g(f: np.ndarray, M: float) -> np.ndarray:
    n = f.shape[0]
    cur = np.zeros(2 * n - 1, dtype=np.float64)   # current f * g values
    g = np.zeros(n, dtype=np.float64)
    f_inv = 1.0 / f                                 # f > 0 for all entries
    for j in range(n - 1, -1, -1):                  # right‑to‑left ordering
        window = cur[j:j + n]                       # view into cur
        # maximal addition without breaking any constraint
        remaining = M - window
        # guard against tiny negative round‑off
        remaining = np.where(remaining > 0, remaining, 0.0)
        ratios = remaining * f_inv
        max_add = ratios.min()
        if max_add > 1e-12:
            g[j] = max_add
            window += max_add * f                   # update the affected window
    return g


# ----------------------------------------------------------------------
# Scale a vector to a prescribed ℓ₁‑norm (capped to the evaluator limits)
def _scale_to_target(v: np.ndarray, target: float) -> np.ndarray:
    s = v.sum()
    if s == 0:
        return v
    factor = target / s
    v_scaled = v * factor
    # evaluator caps values to 1000 – clipping is safe and still yields a
    # feasible (non‑negative) candidate.
    np.clip(v_scaled, 0.0, 1000.0, out=v_scaled)
    return v_scaled


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 score
    improves on the classic root‑constant construction.
    The method follows a few iterations of the LP‑inspired update:
        1. normalise the current sequence to √(2·n);
        2. build a feasible g with a greedy solver for the linear constraints
           f·g ≤ M (where M = max(f*f));
        3. rescale g to the same ℓ₁‑norm and convex‑combine it with f.
    The process is repeated a few times (deterministically) and the best
    candidate seen is returned.
    """
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1) Obtain length and initialise f (the current sequence)
    try:
        n = int(len(height_sequence_1))
        f = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        n = 256
        f = _root_constant_seq(n)

    # ------------------------------------------------------------------
    # 2) Normalise to the prescribed total √(2·n)
    target_sum = np.sqrt(2.0 * n)
    f = _scale_to_target(f, target_sum)

    # ------------------------------------------------------------------
    # evaluate the initial candidate; keep it as the current best
    best = f.copy()
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = float('inf')
    if not np.isfinite(best_score):
        # catastrophic fallback – uniform heights
        best = np.full(n, 1.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
        return best.tolist()

    # ------------------------------------------------------------------
    # 3) Iterative improvement (few deterministic passes)
    start_time = time.time()
    max_seconds = 4.5                     # stay well inside the execution budget
    t_values = (0.5, 0.3, 0.2, 0.1, 0.05)  # mixing coefficients tried each round

    while time.time() - start_time < max_seconds:
        # current bound M = max(f * f)
        conv_ff = np.convolve(best, best, mode='full')
        M = float(conv_ff.max())

        # greedy feasible g for the linear constraints f * g ≤ M
        g0 = _greedy_g(best, M)
        if g0.sum() == 0:
            break

        # rescale g0 to the same ℓ₁‑norm as best (i.e. target_sum)
        g = _scale_to_target(g0, target_sum)

        improved = False
        # try several mixing weights; keep the best improvement this round
        for t in t_values:
            cand = (1.0 - t) * best + t * g
            # assure bounds before evaluation (the evaluator will clip anyway)
            np.clip(cand, 0.0, 1000.0, out=cand)
            try:
                cand_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue
            if np.isfinite(cand_score) and cand_score < best_score:
                best = cand
                best_score = cand_score
                improved = True

        if not improved:
            # no improvement with any mixing weight – stop early
            break

    # ------------------------------------------------------------------
    # 4) Return the best sequence found (as a plain Python list)
    return best.tolist()
