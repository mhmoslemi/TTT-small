import numpy as np

# -------------------------------------------------
# Helper: fast convolution (full) via FFT
def _conv_fft(x: np.ndarray) -> np.ndarray:
    """Return full linear convolution of x with itself using FFT."""
    n = x.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fx * fx, n=m)[: 2 * n - 1]
    return conv


# Helper: evaluate the AC1 upper bound (fast, uses the same formula as the official checker)
def _fast_score(v: np.ndarray) -> float:
    n = v.size
    if n == 0:
        return np.inf
    conv = _conv_fft(v)
    max_c = conv.max()
    s = v.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
