import numpy as np

# ----------------------------------------------------------------------
# Analytic baseline: coefficients of (1-z)^(-1/2)
def _coeff_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2: c_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


# Random contiguous‑block scaling (used as occasional perturbation)
def _make_block_candidate(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    min_len = max(1, n // 100)          # 1‑% of length
    max_len = max(2, n // 4)            # up to 25‑%
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.20)))  # log‑normal perturbation
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed: int = 42,
                     num_steps: int = 4000,
                     target: float = 1.50274349,
                     **kwargs) -> list:
    """
    Search for a non‑negative step‑function with a low AC1 bound.
    The algorithm repeatedly transfers a fraction of the mass from the
    coefficient that currently contributes most to the peak of the
    self‑convolution to a far‑away index, thereby shrinking that peak.
    A few random block‑scaling moves are interleaved to escape plateaus.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem length
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256

    # --------------------------------------------------------------
    # Initial candidates: analytic baseline and optional parent
    analytic = _coeff_binom_half(n)
    analytic = np.clip(analytic, 0.0, 1000.0)

    best = analytic.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # parent fallback (if supplied)
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    # Early exit if already good enough
    if best_score < target:
        return [float(x) for x in best]

    # --------------------------------------------------------------
    # Parameters for the mass‑transfer move
    early_len = max(1, n // 50)          # ~2 % of the vector (source side)
    late_start = n - early_len          # far side for destination
    frac_low, frac_high = 0.1, 0.35     # fraction of mass to move
    block_probability = 0.03           # occasional random block

    for step in range(num_steps):
        # ----------------------------------------------------------
        # Decide on move type
        if rng.random() < block_probability:
            cand = _make_block_candidate(best, rng)
        else:
            # Identify the index that contributes most to the current peak
            conv_full = np.convolve(best, best)
            max_val = conv_full.max()
            km = int(conv_full.argmax())          # position of the peak
            # i ranges over indices participating in that convolution entry
            left = max(0, km - (n - 1))
            right = min(n - 1, km)
            idxs = np.arange(left, right + 1, dtype=np.int64)
            partners = km - idxs
            products = best[idxs] * best[partners]
            i_max = int(idxs[products.argmax()])   # one endpoint of the max pair

            # Choose a far destination index

            while j == i_max:                     # avoid trivial move

            best = cand
            best_score = score
            if best_score < target:

    return [float(x) for x in best]
