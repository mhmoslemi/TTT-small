import numpy as np

# ------------------------------------------------------------
# Helper: try to build a binary mask whose ordered‐pair counts
# never exceed ``max_allowed``.  The goal is to obtain at least
# ``m_needed`` ones; if successful the bound 2·n·max_c/m² ≤ target.
# ------------------------------------------------------------
def _attempt_mask(n, max_allowed, m_needed, rng, max_attempts=100000):
    selected = set()
    cnt = np.zeros(2 * n - 1, dtype=np.int32)      # ordered‑pair counts
    attempts = 0
    while len(selected) < m_needed and attempts < max_attempts:
        i = int(rng.integers(0, n))
        if i in selected:
            attempts += 1
            continue
        # self‑pair check
        if cnt[i + i] + 1 > max_allowed:
            attempts += 1
            continue
        ok = True
        for j in selected:
            s = i + j
            if cnt[s] + 2 > max_allowed:          # (i,j) and (j,i)
                ok = False
                break
        if ok:
            # commit addition
            for j in selected:
                cnt[i + j] += 2
            cnt[i + i] += 1
            selected.add(i)
        attempts += 1
    if not selected:
        return None
    vec = np.zeros(n, dtype=np.float64)
    vec[list(selected)] = 1.0
    return vec
