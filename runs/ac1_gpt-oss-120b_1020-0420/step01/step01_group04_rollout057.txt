    base = _coefficients_binom_half(n)
    base_score = float(evaluate_sequence(base.tolist()))
    if base_score < best_score:
        best = base.copy()
        best_score = base_score
        a = best.copy()

    target = 1.50274349
    # early exit possible
    if best_score < target:
        return best.tolist()

    # parameters
    block_frac = 0.02  # 2% of entries
    min_block = 1

    # main loop
    for step in range(num_steps):
        # compute convolution and max index
        conv = np.convolve(a, a)
        M = conv.max()
        k0 = int(conv.argmax())
        # compute partner values
        idx = np.arange(n)
        partner_idx = k0 - idx
        partner = np.zeros_like(a)
        mask = (partner_idx >= 0) & (partner_idx < n)
        partner[mask] = a[partner_idx[mask]]

        # choose block of indices with smallest partner
        order = np.argsort(partner)
        block_len = max(min_block, int(n * block_frac))

        # compute scaling factor based on partner (smaller partner larger factor)
        # Normalize partner in chosen block to [0,1]
        if block_len > 0:
            sub_partner = partner[chosen]
            # avoid division by zero if all zero
            max_sub = sub_partner.max()
            if max_sub > 0:
                norm = sub_partner / max_sub
            else:
                norm = np.zeros_like(sub_partner)
            # scaling factor: 1 + alpha * (1 - norm)

            factor = 1.0 + alpha * (1.0 - norm) * rng.random(block_len)

            factor = np.ones(0)

        try:
            cand_score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

            best_score = cand_score
            a = cand  # continue search from improved candidate

                break

import numpy as np

def _coefficients_binom_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0*i - 1.0) / (2.0*i)
    return coeffs

    conv = np.fft.irfft(Y, n=size)

import numpy as np

def _coefficients_binom_half(n):
    """Coefficients of (1‑z)^‑½ : C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _full_convolve(x):
    """FFT‑based full convolution x*x (real input, linear‐time)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, n=size)
    Y = X * X
    conv = np.fft.irfft(Y, n=size)
    return conv[: 2 * n - 1]
