import numpy as np

# ------------------------------------------------------------
# Helper: generate a decreasing power‑law sequence a_i = (i+1)^(-p)
# ------------------------------------------------------------
def _power_sequence(p, n):
    """Return an n‑vector a_i = (i+1)^{-p} as float64."""
    idx = np.arange(1, n + 1, dtype=np.float64)   # i = 1 … n
    a = np.power(idx, -p, dtype=np.float64)
    # enforce problem limits (the values are far below 1000)
    np.clip(a, 0.0, 1000.0, out=a)
    return a

        _update_if_better(cand)

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
