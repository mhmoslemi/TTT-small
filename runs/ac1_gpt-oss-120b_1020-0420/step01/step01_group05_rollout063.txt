import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    if n == 0:
        return float("inf")
    # length of the full convolution
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float("inf")
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------
# Helper: alternate placement of sorted values.
# ----------------------------------------------------------
def _alternate_ends(arr: np.ndarray) -> np.ndarray:
    """Place values in descending order alternating from left and right."""
    sorted_desc = np.sort(arr)[::-1]
    out = np.empty_like(arr)
    left, right = 0, arr.size - 1
    toggle = True
    for v in sorted_desc:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out
