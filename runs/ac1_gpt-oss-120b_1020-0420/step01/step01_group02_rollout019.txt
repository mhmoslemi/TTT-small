def full_convolution(a):
    n = a.shape[0]
    N = 2 * n
    A = np.fft.rfft(a, n=N)

    S = np.sum(a)
    if S < 1e-12:

    B = full_convolution(a)

    # Clip values

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # gradient-based refinement
    for iteration in range(num_steps):
        n = best.shape[0]
        S = np.sum(best)

        # compute gradient

import numpy as np

# ------------------------------------------------------------
# Convolution using real FFT (O(n log n), memory O(n))
# ------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return full discrete convolution a*a (length 2*len(a)-1)."""
    n = a.shape[0]
    N = 2 * n
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# AC1 objective (exactly the evaluator's formula, but cheap)
# ------------------------------------------------------------
def ac1_score(a: np.ndarray) -> float:
    """Compute 2*n*max(a*a)/(sum(a)^2)."""
    n = a.shape[0]
    s = np.sum(a)
    if s <= 0.0:
        return np.inf
    b = full_convolution(a)
    max_b = np.max(b)
    return 2.0 * n * max_b / (s * s)


# ------------------------------------------------------------
# Simple half‑power series (fallback generator)
# ------------------------------------------------------------
def half_power_series(length: int) -> np.ndarray:
    """Generate a_i = binom(2i,i)/4**i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 200, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # ---- initial vector -------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        base = height_sequence_1.astype(np.float64).copy()
    else:
        n = 1024
        base = half_power_series(n)

    # enforce bounds early
    np.clip(base, 0.0, 1000.0, out=base)

    best = base.copy()
    best_score = ac1_score(best)

    # ---- gradient‑descent style refinement -------------------------------
    no_improve = 0
    for it in range(num_steps):
        n = best.shape[0]
        s = np.sum(best)
        if s <= 0.0:
            break

        # convolution and location of current maximum
        conv = full_convolution(best)
        max_b = np.max(conv)
        t_max = int(np.argmax(conv))

        # gradient of the max‑lag term w.r.t. a_i (only the active lag)
        grad = np.zeros_like(best)
        start = max(0, t_max - (n - 1))
        end = min(t_max, n - 1)
        idx = np.arange(start, end + 1, dtype=np.int64)
        partner = t_max - idx
        grad[idx] = 2.0 * best[partner]

        # gradient of the AC1 objective
        factor1 = (2.0 * n) / (s * s)
        factor2 = (4.0 * n * max_b) / (s * s * s)
        grad_J = factor1 * grad - factor2   # vector of size n

        # try a few step sizes, accept first improvement
        improved = False
        for step in (0.1, 0.05, 0.02, 0.01, 0.005, 0.001):
            cand = best - step * grad_J
            np.clip(cand, 0.0, 1000.0, out=cand)
            if np.sum(cand) < 0.01:
                continue
            cand_score = ac1_score(cand)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                improved = True
                break

        if improved:
            no_improve = 0
            continue

        no_improve += 1

        cand = best.copy()

            best_score = cand_score
            no_improve = 0

        if no_improve >= 30:
            break

    # ------------------------------------------------------------
    # Return plain Python list (the harness expects this)
    # ------------------------------------------------------------
    return [float(x) for x in best]
