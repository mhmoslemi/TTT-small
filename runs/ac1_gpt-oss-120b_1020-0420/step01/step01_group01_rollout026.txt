import numpy as np

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0*(i+1)*(i+1))
    return a

def _score_from_max(max_val, sum_sq, n):
    return float(2.0 * n * max_val / sum_sq)

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine n, starting seq
    if 'height_sequence_1' in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if seq.ndim != 1 or seq.size == 0 or not np.isfinite(seq).all():
            n = 256
            seq = _catalan_seq(n)
    else:
        n = 256
        seq = _catalan_seq(n)

    # Ensure non-negative and clip at 1000
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.size

    total_sum = seq.sum()
    if total_sum < 0.01:
        # guard, fallback to Catalan
        seq = _catalan_seq(n)
        total_sum = seq.sum()
    sum_sq = total_sum * total_sum

    # Compute initial convolution and best score
    conv = np.convolve(seq, seq)  # full conv length 2n-1
    max_conv = conv.max()
    best_score = _score_from_max(max_conv, sum_sq, n)
    best_seq = seq.copy()
    best_conv = conv.copy()

    # Current working sequence (will be updated on acceptance)
    cur_seq = seq.copy()
    cur_conv = conv.copy()

    # Parameters
    max_iters = kwargs.get('num_steps', 20000)  # default maybe 20000
    # but we can also incorporate extra steps
    # Use early region size for moves
    half = n // 2
    no_improve = 0
    # We'll interleave prefix scaling occasionally
    for it in range(max_iters):
        # occasional scaling move every 500 iterations
        if (it % 500) == 0 and it > 0:
            # prefix shrink/expand move
            r = rng.uniform(0.05, 0.3)   # proportion
            k = max(1, int(r * n))
            early = cur_seq[:k]
            tail = cur_seq[k:]
            early_sum = early.sum()
            tail_sum = tail.sum()

                new_tail = tail * g
                cand_seq = np.concatenate([new_early, new_tail])
                # safety clamp
                if cand_seq.max() <= 1000.0:
                    cand_conv = np.convolve(cand_seq, cand_seq)
                    cand_max = cand_conv.max()
                    cand_score = _score_from_max(cand_max, sum_sq, n)
                    if cand_score < best_score:
                        best_seq = cand_seq.copy()
                        best_conv = cand_conv.copy()
                        best_score = cand_score
                        cur_seq = cand_seq.copy()
                        cur_conv = cand_conv.copy()
                        no_improve = 0
                        continue
        # mass transfer move
        # bias early to late
        src = rng.integers(0, half)
        dst = rng.integers(half, n)
        if src == dst:
            continue
        src_val = cur_seq[src]
        if src_val < 1e-12:
            continue
        # fraction to move
        frac = rng.uniform(0.1, 0.5)
        delta = src_val * frac
        # create candidate copies
        cand_seq = cur_seq.copy()
        cand_conv = cur_conv.copy()

        # compute score

        cand_score = _score_from_max(cand_max, sum_sq, n)
        if cand_score < best_score:

            best_conv = cand_conv.copy()
            best_score = cand_score
            cur_seq = cand_seq
            cur_conv = cand_conv
            no_improve = 0
        else:

            if no_improve > 2000:

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]

import numpy as np

def _catalan_seq(n: int) -> np.ndarray:
    """Central‑binomial (Catalan‑like) sequence a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


def _score_from_max(max_val: float, sum_sq: float, n: int) -> float:
    """Exact AC1 score from max convolution value."""
    return float(2.0 * n * max_val / sum_sq)
