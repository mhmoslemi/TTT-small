import numpy as np

# ------------------------------------------------------------
# Half‑power series a_i = binom(2i,i)/4^i  (coefficients of (1‑z)^‑½)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return the first ``length`` terms of the half‑power series."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Transfer a fraction of mass from an early index to a later one.
# ------------------------------------------------------------
def early_to_late_transfer(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Pick an early position and a late position.
    Move a random fraction of the early entry to the late entry.
    """
    n = vec.size
    # define early/late zones (roughly first/last 10 % of the vector)
    early_zone = max(1, n // 10)
    late_zone_start = max(early_zone, n - early_zone)

    early_idx = int(rng.integers(0, early_zone))
    late_idx = int(rng.integers(late_zone_start, n))

    # fraction of the early value to move (between 1 % and 30 %)
    frac = float(rng.uniform(0.01, 0.30))
    delta = vec[early_idx] * frac

    cand = vec.copy()
    cand[early_idx] -= delta
    cand[late_idx] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Reduce a contiguous early block and redistribute the removed mass
# uniformly over a distant late block.
# ------------------------------------------------------------
def block_transfer(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Choose a small early block, shrink it, and add the removed mass
    (distributed equally) to a distant late block.
    """
    n = vec.size
    max_width = max(1, n // 30)
    width = int(rng.integers(1, max_width + 1))

    # early block starts somewhere in the first third
    early_start_max = max(1, n // 3 - width)
    early_start = int(rng.integers(0, early_start_max + 1))

    # late block starts somewhere in the last third
    late_start_min = 2 * n // 3
    late_start_max = max(late_start_min, n - width)
    late_start = int(rng.integers(late_start_min, late_start_max + 1))

    # shrink factor for the early block (reduce by 10‑%…40‑%)
    factor = float(rng.uniform(0.60, 0.90))

    cand = vec.copy()
    early_slice = slice(early_start, early_start + width)
    late_slice = slice(late_start, late_start + width)

    old_sum = np.sum(cand[early_slice])
    cand[early_slice] *= factor
    removed = old_sum - np.sum(cand[early_slice])

    # distribute the removed mass evenly over the late block
    cand[late_slice] += removed / width
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 250, **kwargs):
    """
    Generate a non‑negative step‑function whose AC1 upper bound is
    (heuristically) low.  The routine starts from the classic half‑power
    series, optionally seeds from an existing parent ``height_sequence_1``,
    and performs a lightweight stochastic mass‑transfer search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and an initial candidate
    # ------------------------------------------------------------------
    n = 256
    if "height_sequence_1" in globals():
        hs = globals()["height_sequence_1"]
        if isinstance(hs, np.ndarray) and hs.ndim == 1 and hs.size > 0:
            n = int(hs.size)

    base = generate_half_power_series(n)                     # O(n) memory
    np.clip(base, 0.0, 1000.0, out=base)

    # start from the half‑power series; if a parent exists, keep the better
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    if "height_sequence_1" in globals():
        parent = np.array(height_sequence_1, dtype=np.float64)
        np.clip(parent, 0.0, 1000.0, out=parent)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent
            best_score = parent_score

    # ------------------------------------------------------------------
    # Stochastic refinement: early–late mass transfers
    # ------------------------------------------------------------------

    for _ in range(num_steps):

        if np.isfinite(score) and score < best_score:

            if best_score <= TARGET:

    # Return a plain Python list as required by the harness
