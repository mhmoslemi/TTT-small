import numpy as np

# ------------------------------------------------------------
# Helper: fast linear convolution via FFT (linear, not circular)
# ------------------------------------------------------------
def _conv_fft(x, size):
    """
    Linear convolution of x with itself using zero‑padded FFT.
    Returns a real array of length 2*len(x)-1.
    """
    fx = np.fft.rfft(x, size)
    conv_full = np.fft.irfft(fx * fx, size)[: 2 * x.size - 1]
    return conv_full


# ------------------------------------------------------------

# ------------------------------------------------------------
def propose_candidate(seed=42, max_iters=300, restarts=3, **kwargs):
    """
    Gradient‑based search for a non‑negative sequence that minimizes the
    AC1 upper bound used by `evaluate_sequence`.  The algorithm works on the
    simplex (fixed L1 norm) and repeatedly flattens the peak of the
    self‑convolution using a simple projected gradient step based on the
    index of the current maximal convolution value.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem length (use parent if present)
    # --------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        parent_vec = np.asarray(height_sequence_1, dtype=np.float64)
        parent_vec = np.clip(parent_vec, 0.0, 1000.0)
        if parent_vec.sum() < 0.01:
            parent_vec = np.ones_like(parent_vec) / parent_vec.size
    except NameError:
        n = 256
        parent_vec = None

    # --------------------------------------------------------
    # Pre‑compute FFT size (next power of two >= 2*n-1)
    # --------------------------------------------------------
    fft_size = 1 << ((2 * n - 1).bit_length())

    # Helper: compute the AC1 bound for a vector whose L1 norm is 1

    def _bound_one_norm(v):
        conv = _conv_fft(v, fft_size)
        mb = conv.max()
        return 2.0 * n * mb  # denominator = 1 because sum(v)=1

    # --------------------------------------------------------

    best_vec = np.ones(n, dtype=np.float64) / n
    best_score = _bound_one_norm(best_vec)

    if parent_vec is not None:

        p = parent_vec / parent_vec.sum()
        sc = _bound_one_norm(p)
        if sc < best_score:

            best_vec = p.copy()

    for _ in range(restarts):
