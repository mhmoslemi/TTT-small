import numpy as np
import math

def _fast_score(vec):
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _conv_full(vec):
    n = vec.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    return conv

def _gradient_of_max(vec, k):
    n = vec.size
    grad = np.zeros_like(vec)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad[mask] = vec[idx[mask]]
    idx2 = np.arange(n) - k
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] += vec[idx2[mask2]]
    grad *= 2.0
    return grad

import numpy as np
import math

# -------------------------------------------------------------
# Fast AC1 upper bound (scale‑invariant, uses FFT)
# -------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())           # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# -------------------------------------------------------------
# Full convolution via FFT (returns length 2n‑1 array)
# -------------------------------------------------------------
def _conv_full(vec: np.ndarray) -> np.ndarray:
    n = vec.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
# Gradient of a single convolution peak (index k)
# -------------------------------------------------------------
def _gradient_of_max(vec: np.ndarray, k: int) -> np.ndarray:
    n = vec.size
    grad = np.zeros_like(vec)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad[mask] = vec[idx[mask]]
    idx2 = np.arange(n) - k
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] += vec[idx2[mask2]]
    grad *= 2.0
    return grad


# -------------------------------------------------------------
# Generate a candidate window of a given type
# -------------------------------------------------------------
def _candidate_from_window(name: str, n: int) -> np.ndarray:
    if name == 'uniform':
        return np.ones(n, dtype=np.float64)
    if name == 'hanning':
        return np.hanning(n).astype(np.float64)
    if name == 'hamming':
        return np.hamming(n).astype(np.float64)
    if name == 'blackman':
        return np.blackman(n).astype(np.float64)
    if name == 'triangular':
        i = np.arange(n, dtype=np.float64)
        return 1.0 - np.abs(i - (n - 1) / 2) / ((n - 1) / 2)
    if name == 'gaussian':
        i = np.arange(n, dtype=np.float64)
        sigma = n / 6.0
        return np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    if name == 'gaussian_wide':
        i = np.arange(n, dtype=np.float64)
        sigma = n / 3.0
        return np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    if name == 'exponential':
        i = np.arange(n, dtype=np.float64)
        beta = 5.0 / n
        return np.exp(-beta * i)
    # fallback
    return np.ones(n, dtype=np.float64)
