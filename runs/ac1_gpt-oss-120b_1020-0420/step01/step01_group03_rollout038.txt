import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _root_constant_seq(n: int) -> np.ndarray:
    """Root‑of‑constant sequence a_i = C(2i,i)/4**i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the full convolution of seq with itself."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())        # next power of two
    fft = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft * fft, m)
    # only the first 2*n-1 entries belong to the full convolution
    return float(conv[: 2 * n - 1].max())


def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound for a given non‑negative step function."""
    n = seq.shape[0]
    s = float(seq.sum())
    if s < 0.01:
        return float('inf')
    maxb = _conv_max(seq)
    return 2.0 * n * maxb / (s * s)


def propose_candidate(seed: int = 42, **kwargs):

    Return a non‑negative step‑function (list of floats) whose
    AC1 upper bound improves on the root‑constant baseline.
    The routine performs a few iterations of the
    linear‑programming refinement described in the problem statement.

        n = 256

    # target L1‑norm (the normalising constant used in the literature)
