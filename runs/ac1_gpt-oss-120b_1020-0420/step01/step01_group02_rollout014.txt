    best = None

                best_score = sc

    # Simple stochastic refinement: multiplicative perturbation on random indices
    if best is None:
        # fallback: uniform small values
        best = np.full(n, 1.0 / n, dtype=np.float64)
        best = clip_sequence(best)
        best_score = compute_score(best)

    # number of refinement steps
    num_refine = 300

def generate_half_power_series(length: int) -> np.ndarray:
    """Return array a_i = binom(2i,i)/4**i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a

    # --------------------------------------------------------
    best = None
    best_score = float("inf")

    # Parent sequence (if supplied)
    if "height_sequence_1" in globals():
        cand = globals()["height_sequence_1"].astype(np.float64)
        cand = clip_sequence(cand)
        sc = compute_score(cand)
        if np.isfinite(sc):
            best = cand
            best_score = sc

    # Baseline half‑power series (always a valid competitor)
    baseline = generate_half_power_series(n)
    baseline = clip_sequence(baseline)
    sc_base = compute_score(baseline)
    if np.isfinite(sc_base) and sc_base < best_score:
        best = baseline
        best_score = sc_base

    # --------------------------------------------------------
    # Grid search over power‑law exponents and small integer offsets
    # --------------------------------------------------------
    alphas = np.linspace(0.55, 0.85, 31)          # 0.55 … 0.85 in steps of 0.01
    offsets = range(0, 6)                        # 0 … 5

    for alpha in alphas:
        for offset in offsets:
            cand = generate_power_sequence(n, float(alpha), offset)
            cand = clip_sequence(cand)
            sc = compute_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc

    # --------------------------------------------------------
    # Simple stochastic refinement (multiply a random index)
    # --------------------------------------------------------
    if best is None:  # extremely unlikely, but keep a fallback
        best = np.full(n, 1.0 / n, dtype=np.float64)

    num_refine = 300
    sigma = 0.03                     # log‑normal multiplicative step size

    for _ in range(num_refine):
        total = float(best.sum())
        if total == 0.0:
            probs = None
        else:
            probs = best / total   # bias towards larger entries
        idx = rng.choice(n, p=probs)
        factor = math.exp(rng.normal(0.0, sigma))
        cand = best.copy()
        cand[idx] *= factor
        cand = clip_sequence(cand)
        sc = compute_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------
    # Return the final candidate as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
