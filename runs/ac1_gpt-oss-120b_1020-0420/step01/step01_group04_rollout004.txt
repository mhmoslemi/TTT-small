def _coefficients_binom_half(n: int) -> np.ndarray:
    """
    First n coefficients of the expansion (1‑z)^‑1/2.
    They satisfy a_0 = 1 and a_i = a_{i‑1}·(2i‑1)/(2i).
    All entries lie in (0, 1] and the convolution self‑product
    attains the value 1 on the first n positions, giving the
    theoretical AC1 constant 2/π ≈ 0.637.
    """

    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs
