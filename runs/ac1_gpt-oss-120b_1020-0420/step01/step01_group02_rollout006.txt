# ------------------------------------------------------------
# Half‑power series: a_i = binom(2i,i)/4**i  (coefficients of (1‑z)^(-1/2))
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

# Two‑block move:
#   - multiply a random early block by a factor f
#   - multiply a random late block by 1/f
#   - renormalise the total sum (scaling does not change the bound)
#   - clip to the allowed range
# ------------------------------------------------------------
def two_block_move(base: np.ndarray, rng: np.random.Generator, sigma: float) -> np.ndarray:
    n = base.size
    new = base.copy()

    max_width = max(1, n // 20)
    w = int(rng.integers(1, max_width + 1))

    early_end = max(1, n // 5)                 # first 20 % of indices

    start_inc = int(rng.integers(late_start, max(late_start, n - w) + 1))

    new_sum = np.sum(new)

    return [float(x) for x in best]
