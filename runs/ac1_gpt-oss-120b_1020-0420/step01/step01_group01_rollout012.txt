import numpy as np

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1))/(4.0*(i+1)*(i+1))
    return a

def _max_convolution_lag(seq):

def propose_candidate(seed=42, num_steps=5000, **kwargs):
    rng = np.random.default_rng(seed)
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size if parent.ndim == 1 and parent.size>0 and np.isfinite(parent).all() else 256
    else:
        n = 256
    # Use Catalan start
    seq = _catalan_seq(n)
    seq = np.clip(seq, 0.0, 1000.0)   # ensure bounds
    best_score = evaluate_sequence(seq.tolist())
    best_seq = seq.copy()
    # Optimization loop
    max_iter = kwargs.get("num_steps", num_steps)
    move_factor_start = 0.5
    move_factor_end = 0.01
    for it in range(max_iter):
        # linearly fade move factor
        move_factor = move_factor_start - (move_factor_start-move_factor_end)*it/max_iter
        conv = np.convolve(seq, seq)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        # Determine overlapping index range
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        # Compute products for this lag
        a_sub = seq[i_start:i_end+1]
        b_sub = seq[max_idx - np.arange(i_start, i_end+1)]
        prod = a_sub * b_sub
        # Pick the larger contributor
        sub_idx = int(prod.argmax())
        i_max = i_start + sub_idx
        j_max = max_idx - i_max
        src = i_max if seq[i_max] >= seq[j_max] else j_max
        if seq[src] <= 0:
            continue
        delta = seq[src] * move_factor * rng.random()
        if delta <= 0:
            continue
        # Choose far destination
        # Ensure distance at least n//4
        dist = max(n//4, 1)
        dst = (src + rng.integers(dist, n-dist)) % n
        if dst == src:
            continue
        cand = seq.copy()
        cand[src] -= delta
        cand[dst] += delta
        # Clip
        cand = np.clip(cand, 0.0, 1000.0)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            seq = cand   # continue from improved
    return [float(v) for v in best_seq]

import numpy as np

# ------------------------------------------------------------
# Catalan‑like base sequence: a_i = C(2i,i)/4^i  (i = 0 .. n‑1)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Score identical to the provided evaluator (but vectorised)
# ------------------------------------------------------------
def _score_seq(x: np.ndarray) -> float:
    s = x.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(x, x)
    return float(2.0 * x.size * conv.max() / (s * s))

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size – use the parent if it exists.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (parent.ndim == 1 and parent.size > 0 and
                np.isfinite(parent).all()):
            seq = parent.copy()
            n = int(parent.size)
        else:
            n = 256
            seq = _catalan_seq(n)
    else:
        n = 256
        seq = _catalan_seq(n)

    # --------------------------------------------------------
    # Normalise the initial vector (clip – the evaluator does this too)
    # --------------------------------------------------------
    seq = np.clip(seq, 0.0, 1000.0)
    best_seq = seq.copy()
    best_score = _score_seq(best_seq)

    # --------------------------------------------------------
    # Parameters of the search
    # --------------------------------------------------------
    max_iter = int(kwargs.get("num_steps", 5000))
    # move fraction decays from ~0.5 to ~0.01
    start_frac = 0.5
    end_frac = 0.01
    min_dist = max(1, n // 4)   # we always move mass far away

    # --------------------------------------------------------
    # Greedy hill‑climbing – each step tries to lower the peak of
    # the autocorrelation by moving mass from the most contributing
    # entry of the current peak to a far location.
    # --------------------------------------------------------
    for it in range(max_iter):
        # current convolution & peak information
        conv = np.convolve(seq, seq)
        k_max = int(conv.argmax())
        # early exit if the peak is already low enough
        if conv.max() < 0.94:      # corresponds roughly to bound < 1.5
            break

        # indices that actually contribute to this lag
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1)
        # product a_i * a_{k-i} for this lag
        prod = seq[idx] * seq[k_max - idx]
        sub = int(prod.argmax())
        i_max = i_start + sub
        j_max = k_max - i_max

        # choose the larger of the two contributors as source
        src = i_max if seq[i_max] >= seq[j_max] else j_max
        if seq[src] <= 1e-15:
            continue

        # fraction of the source mass to relocate (annealed)
        move_frac = start_frac - (start_frac - end_frac) * it / max_iter
        delta = seq[src] * move_frac * rng.random()
        if delta <= 0.0:
            continue

        # pick a destination far from the source
        offset = rng.integers(min_dist, n - min_dist)
        dst = (src + int(offset)) % n
        if dst == src:
            continue

        cand = seq.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _score_seq(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand   # continue from the improved state

    # --------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # --------------------------------------------------------
    return [float(v) for v in best_seq]
