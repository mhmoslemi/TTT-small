def _conv_fft(seq):
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, m)
    prod = fft_seq * fft_seq
    conv_full = np.fft.irfft(prod, m)
    return conv_full[:2 * n - 1]

import numpy as np

# ------------------------------------------------------------
# Helper: Catalan‑like (central binomial) sequence
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Helper: fast full convolution using FFT (real‑valued)
# ------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, m)
    prod = f * f
    conv_full = np.fft.irfft(prod, m)
    return conv_full[: 2 * n - 1]


# ------------------------------------------------------------
# Helper: compute the same score as the provided evaluator
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:
        return float("inf")
    conv = _conv_fft(seq)
    max_b = conv.max()
    return float(2.0 * seq.size * max_b / (total * total))

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size – use the provided parent if available.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        seq = _catalan_seq(n)

    seq = np.clip(seq, 0.0, 1000.0)
    if seq.sum() < 0.01:                # safety guard
        seq[0] = 1.0

    best_seq = seq
    best_score = _score(best_seq)
    n = best_seq.size

    # --------------------------------------------------------
    # Deterministic re‑allocation: shrink a prefix and inflate the tail.
    # Both the original shape and a uniformised shape are tried.
    # --------------------------------------------------------
    ratios = np.linspace(0.01, 0.5, 30)           # prefix size as proportion of n
    factors = np.linspace(0.0, 1.0, 21)           # shrink factor for the prefix

    for r in ratios:
        k = max(1, int(r * n))                   # size of the early block
        # keep a copy of the current best to serve as the baseline for this r
        baseline = best_seq.copy()
        for f in factors:
            early = baseline[:k]
            tail = baseline[k:]
            early_sum = early.sum()
            tail_sum = tail.sum()
            if tail_sum <= 0.0:
                continue

            # ---- Variant 1: keep original early shape, only scale ----
            early_scaled = early * f
            delta = early_sum - early_scaled.sum()
            new_tail_sum = tail_sum + delta
            scale = new_tail_sum / tail_sum
            cand1 = np.concatenate([early_scaled, tail * scale])
            cand1 = np.clip(cand1, 0.0, 1000.0)
            sc1 = _score(cand1)
            if sc1 < best_score:
                best_seq, best_score = cand1, sc1
                baseline = best_seq.copy()     # update baseline for further f

            # ---- Variant 2: flatten the early block uniformly ----
            new_early_sum = early_sum * f
            if k > 0:

                if sc2 < best_score:
                    best_seq, best_score = cand2, sc2

    # --------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # --------------------------------------------------------
    return [float(v) for v in best_seq]
