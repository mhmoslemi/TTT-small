    cand = np.clip(cand, 0.0, 1000.0)
    score = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(score) and score < best_score:

    return best_vec, best_score


# ------------------------------------------------------------
# Gradient step for reducing the L∞ convolution peak
# ------------------------------------------------------------
def _gradient_step(vec, eta):
    n = vec.size
    # full convolution
    conv = np.convolve(vec, vec)
    max_val = conv.max()
    peak_idxs = np.where(conv == max_val)[0]

    # sub‑gradient: contribution of each entry to the peak sums
    grad = np.zeros_like(vec)
    for k in peak_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        grad[i_range] += vec[j_range]

    # multiplicative descent (scale invariant, keep sum = 1)
    vec = vec / (1.0 + eta * grad)
    s = vec.sum()
    if s > 0:
        vec = vec / s
    return vec
