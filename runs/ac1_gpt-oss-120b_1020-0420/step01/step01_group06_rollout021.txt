def fast_score(a):
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:
        return np.inf
    # Normalize for better numeric stability:
    w = a / total
    n = a.size
    # compute convolution of w with w using FFT
    size = 2*n - 1
    N = 1 << (size - 1).bit_length()  # next power of two
    # rfft length N
    A = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(A * A, n=N)[:size]
    maxC = conv.max()
    return 2.0 * n * maxC

def propose_candidate(seed: int = 42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)

    # Determine parent or fallback
    parent = None
    if 'height_sequence_1' in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            parent = np.clip(tmp, 0.0, 1000.0)
    if parent is None:

        parent = np.ones(n, dtype=np.float64)
    n = parent.size

    def normalize(v):
        v = np.clip(v, 0.0, 1000.0)
        total = v.sum()
        if total < 1e-12:
            v = np.ones_like(v)
            total = v.sum()
        v = v / total
        return v

    # fast scoring function
    def fast_score(v):
        w = normalize(v)  # returns normalized
        # compute convolution via FFT
        size = 2 * n - 1
        N = 1 << (size - 1).bit_length()
        A = np.fft.rfft(w, n=N)
        conv = np.fft.irfft(A * A, n=N)[:size]
        maxC = conv.max()
        return 2.0 * n * maxC

    # uniform

        idx = np.arange(1, n+1, dtype=np.float64)
        w = idx ** (-a)
        candidates.append(normalize(w))
    # exponential

        w = np.exp(-b * np.arange(n, dtype=np.float64))
        candidates.append(normalize(w))
    # sine-squared shape

    w = np.sin(t) ** 2

import numpy as np

def normalize_vector(vec):
    """Clip to the allowed range and rescale to sum 1."""
    v = np.clip(vec, 0.0, 1000.0)
    total = v.sum()
    if total < 1e-12:                       # degenerate case – fall back to uniform
        v = np.full_like(v, 1.0 / v.size)
        total = v.sum()
    return v / total


def fast_score(vec):
    """Score used during the search (identical to the official metric)."""
    w = normalize_vector(vec)
    n = w.size
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()       # next power‑of‑two for a fast FFT
    A = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(A * A, n=N)[:size]
    return 2.0 * n * conv.max()


def max_convolution_norm(w):
    """Full convolution of a normalized vector with itself (FFT, no scaling)."""
    n = w.size
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(w, n=N)
    return np.fft.irfft(A * A, n=N)[:size]


def subgradient_max(w, conv):
    """
    Sub‑gradient of the maximal autocorrelation value.
    For each lag achieving the maximum we add its contribution;
    the result is averaged over those lags.
    """
    n = w.size
    max_val = conv.max()
    ks = np.where(conv == max_val)[0]
    grad = np.zeros_like(w)
    for k in ks:
        if k == 0:
            grad += 2.0 * w                     # derivative of ∑ w_i²
        else:
            grad[: n - k] += w[k:]               # w_i w_{i+k}
            grad[k:]      += w[: n - k]          # w_{i-k} w_i
    grad /= len(ks)
    return grad


def propose_candidate(seed: int = 42, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 upper bound.
    The routine builds a pool of analytic families, refines the best by gradient
    descent on the maximal autocorrelation and finally performs random
    perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Acquire a parent length (prefer the supplied sequence)
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            parent = np.clip(tmp, 0.0, 1000.0)

    if parent is None:                     # generic fallback
        n = 1024
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size

    # ------------------------------------------------------------------
    # 2.  Build an initial portfolio of analytic candidates
    # ------------------------------------------------------------------
    candidates = []

    # a) parent (if any)
    candidates.append(normalize_vector(parent))

    # b) uniform distribution
    candidates.append(np.full(n, 1.0 / n))

    # c) power‑law families
    for alpha in (0.2, 0.4, 0.6, 0.8):
        idx = np.arange(1, n + 1, dtype=np.float64)
        cand = np.power(idx, -alpha)
        candidates.append(normalize_vector(cand))

    # d) exponential decay families
    for beta in (3e-4, 7e-4, 1.5e-3, 3e-3):
        cand = np.exp(-beta * np.arange(n, dtype=np.float64))
        candidates.append(normalize_vector(cand))

    # e) sine‑power families (peaked in the middle)
    base_sin = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand = np.power(base_sin, p)
        candidates.append(normalize_vector(cand))

    # f) Gaussian‑like families
    centre = (n - 1) / 2.0
    for sigma_factor in (0.2, 0.35, 0.5):
        sigma = sigma_factor * centre
        cand = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        candidates.append(normalize_vector(cand))

    # g) a few random Dirichlet‑type draws
    for _ in range(5):
        cand = rng.random(n)
        candidates.append(normalize_vector(cand))

    # ------------------------------------------------------------------
    # 3.  Choose the best among the analytic pool
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 4.  Gradient‑descent phase that directly reduces the maximal lag
    # ------------------------------------------------------------------
    a = best.copy()                        # always kept normalised
    max_grad_iters = int(min(5000, max(2000, n // 2)))
    eta0 = 0.5                              # initial step size

    for it in range(max_grad_iters):
        conv = max_convolution_norm(a)
        grad = subgradient_max(a, conv)
        eta = eta0 * (0.99 ** it)           # slowly decreasing step
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)   # enforce non‑negativity
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= s                       # renormalise to sum 1

        sc = fast_score(a_new)
        if sc < best_score:
            best = a_new.copy()
            best_score = sc
        a = a_new

    # ------------------------------------------------------------------
    # 5.  Stochastic refinement (local block moves, pairwise transfers,
    #    global log‑normal noise)
    # ------------------------------------------------------------------
    for _ in range(3000):
        cand = best.copy()
        r = rng.random()
        if r < 0.5:                         # block scaling
            start = rng.integers(0, n)
            width = int(rng.exponential(scale=n / 100.0)) + 1
            width = min(width, n - start)
            factor = np.exp(rng.normal(0.0, 0.1))
            cand[start:start + width] *= factor
        elif r < 0.8:                       # pairwise mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            src, dst = (i, j) if cand[i] > cand[j] else (j, i)
            delta = rng.random() * 0.1 * cand[src]
            cand[src] -= delta
            cand[dst] += delta
        else:                               # global multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cand *= noise

        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        sc = fast_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 6.  Return the best candidate found as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
