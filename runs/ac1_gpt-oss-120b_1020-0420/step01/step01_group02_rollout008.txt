    # ---- local search: mass transfer between two positions -----------
    max_factor = 0.1          # maximal relative transfer size
    cooling = 0.9995          # exponential decay of the step size
    stagnation = 0            # counter for iterations without improvement

    for step in range(max_iter):
        # pick two distinct indices
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # maximum admissible transfer that respects the bounds [0,1000]
        delta_max = min(best[i], 1000.0 - best[j])
        if delta_max <= 0.0:
            continue

        # relative magnitude decays with the iteration count
        rel = max_factor * (cooling ** step)
        delta = delta_max * rng.random() * rel
        if delta <= 1e-12:
            continue

        # incremental update of the convolution (O(n))
        conv_new = best_conv.copy()
        # contribution of the linear term 2*delta*(a shifted by j − a shifted by i)
        conv_new[j:j + n] += 2.0 * delta * best
        conv_new[i:i + n] -= 2.0 * delta * best
        # contribution of the quadratic term delta^2*(δ_{2i}+δ_{2j}−2δ_{i+j})
        idx_2i = 2 * i
        idx_2j = 2 * j
        idx_ij = i + j
        conv_new[idx_2i] += delta * delta
        conv_new[idx_2j] += delta * delta
        conv_new[idx_ij] -= 2.0 * delta * delta

        # evaluate the new bound
        max_b = conv_new.max()
        score = float(2 * n * max_b / (sum_a * sum_a))

        # accept only if the bound strictly improves
        if score < best_score:
            best[i] -= delta
            best[j] += delta
            best_conv = conv_new
            best_score = score
            stagnation = 0
        else:
            stagnation += 1

        # stop early if the search has stalled
        if stagnation >= 1000:
            break

    # ------------------------------------------------------------
    # Return a plain Python list of floats (the harness expects this)
    # ------------------------------------------------------------
    return [float(x) for x in best]
```
