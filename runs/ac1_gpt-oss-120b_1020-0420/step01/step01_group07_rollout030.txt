def build_sidon_set(n, rng):
    # random permutation order of indices
    order = rng.permutation(n)
    selected = []
    used_sums = set()
    for i in order:
        conflict = False
        # check sums with existing elements
        for j in selected:
            s = i + j
            if s in used_sums:
                conflict = True
                break
        if conflict:
            continue
        # check self sum
        s_self = i + i
        if s_self in used_sums:
            continue
        # else add i
        for j in selected:
            used_sums.add(i + j)
        used_sums.add(s_self)
        selected.append(i)
    return selected

import numpy as np

TARGET = 1.50274349

def _sidon_set(n, rng):
    order = rng.permutation(n)
    used = np.zeros(2*n-1, dtype=np.bool_)
    selected = []
    for i in order:
        i = int(i)
        # self sum conflict
        if used[i+i]:
            continue
        if selected:
            cur = np.array(selected, dtype=np.int64)
            s = i + cur
            if used[s].any():
                continue
            # add i
            used[i + cur] = True
        used[i+i] = True
        selected.append(i)
    return selected

def propose_candidate(seed=42, trials=500, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine length and baseline
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.shape[0])

    except NameError:

        base_vec = np.ones(n, dtype=np.float64)

    best_vec = base_vec.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # Upper bound on theoretical Sidon size
    max_possible = int(np.floor((-1 + np.sqrt(1 + 8 * (2 * n - 1))) / 2))

    best_sel = []

    for _ in range(trials):

        cand[selected] = 1.0
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                break

    # -------------------- tiny local refinement --------------------
    for _ in range(100):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        cand = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.05)))
        cand[start:stop] *= factor
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                break

    return [float(x) for x in best_vec]
