import numpy as np

def _convolution_update(a_old, b_old, i, j, delta):
    """
    Return a new convolution array after moving `delta` from index i to j.
    Uses the old vector a_old for the linear part of the update.
    """
    n = a_old.shape[0]
    b_new = b_old.copy()
