import numpy as np
import time

# ------------------------------------------------------------
# Convolution‑based score (identical to evaluate_sequence)
# ------------------------------------------------------------
def _score(arr: np.ndarray) -> float:
    """Return 2*n*max(conv(arr,arr))/sum(arr)^2."""
    n = arr.size
    conv = np.convolve(arr, arr)
    maxc = np.max(conv)
    s = np.sum(arr)
    # sum is guaranteed >0 for all candidates we keep
    return 2.0 * n * maxc / (s * s)


# ------------------------------------------------------------
# Generate the (1‑z)^(-1/2) half‑power series
# ------------------------------------------------------------
def _half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Propose a candidate sequence
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that
    tries to beat the AC1 bound pi/2 ≈ 1.5708.
    """
    rng = np.random.default_rng(seed)

    # ---- obtain problem size and an initial vector -----------------
    if "height_sequence_1" in globals():
        hs = globals()["height_sequence_1"]
        if isinstance(hs, np.ndarray) and hs.ndim == 1 and hs.size > 0:
            n = int(hs.size)
            base = hs.astype(np.float64).copy()
        else:
            n = 256
            base = _half_power_series(n)
    else:
        n = 256
        base = _half_power_series(n)

    # safety clamp (required by the evaluator)
    np.clip(base, 0.0, 1000.0, out=base)

    # ---- keep the best sequence seen so far -----------------------
    best = base.copy()
    best_score = _score(best)

    # ------------------------------------------------------------
    # Simple deterministic re‑weighting of early / late entries
    # ------------------------------------------------------------
    early_fracs = (0.25, 0.33, 0.40)               # part of the array we shrink
    factors = (0.9, 0.8, 0.7, 0.6, 0.5)          # how much we shrink it
    for frac in early_fracs:
        early = int(n * frac)
        if early == 0 or early * 2 > n:
            continue
        for f in factors:
            cand = best.copy()
            # mass removed from the early block
            removed = np.sum(cand[:early] * (1.0 - f))
            cand[:early] *= f
            # spread removed mass uniformly over the last block
            cand[-early:] += removed / early
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if sc < best_score:
                best, best_score = cand, sc

    # ------------------------------------------------------------
    # A few smoothing passes (3‑point moving average)
    # ------------------------------------------------------------
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    for _ in range(3):
        cand = np.convolve(best, kernel, mode="same")
        np.clip(cand, 0.0, 1000.0, out=cand)
        # renormalise to keep the same total mass (scale, not shift)
        cand *= np.sum(best) / np.sum(cand)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------
    # Stochastic hill‑climbing: transfer a small fraction of mass
    # ------------------------------------------------------------
    # current is the sequence we perturb; start from the best we have
    current = best.copy()
    start_time = time.time()
    max_seconds = 1.8                # keep runtime modest
    target_score = 1.50274349        # we stop once we beat this
    max_eps = 0.07                   # maximal fraction of a[i] to move

    while time.time() - start_time < max_seconds:
        if best_score <= target_score:

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:

        cand[j] += delta

        if sc < best_score:
            best, best_score = cand, sc
            current = cand          # continue climbing from the improvement
        # occasional non‑improving moves are ignored – pure hill‑climb

    # ------------------------------------------------------------
    # Return the final candidate as a plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best]
