import numpy as np, import time

def _score_and_conv(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf, None, np.inf
    conv = np.convolve(seq, seq)
    maxc = conv.max()
    n = seq.size
    score = float(2.0 * n * maxc / (s * s))
    return score, conv, maxc

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0 * (i+1)*(i+1))
    return a

def _powerlaw_seq(n, alpha):
    i = np.arange(n, dtype=np.float64) + 1.0
    return i ** (-alpha)

import numpy as np

# ------------------------------------------------------------
# Scoring helper (same formula as the provided evaluator)
# ------------------------------------------------------------
def _score_and_conv(seq: np.ndarray):
    """Return (score, full convolution, max(conv)) for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf, None, np.inf
    conv = np.convolve(seq, seq)            # full convolution (FFT for large n)
    m = conv.max()
    n = seq.size
    score = float(2.0 * n * m / (s * s))
    return score, conv, m


# ------------------------------------------------------------
# Catalan‑like sequence (a_i = C(2i,i)/4^i)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Power‑law family: a_i = (i+1)^(-alpha)
# ------------------------------------------------------------
def _powerlaw_seq(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    return i ** (-alpha)


# ------------------------------------------------------------
# Deterministic “shrink‑early” refinement (multiple passes)
# ------------------------------------------------------------
def _shrink_seq(seq: np.ndarray,
                prefix_ratios,
                shrink_factors) -> tuple[np.ndarray, float]:
    """Iteratively shrink an early prefix and inflate the tail."""
    best = seq
    best_score, _, _ = _score_and_conv(best)
    n = seq.size

    for _ in range(3):                     # up to three passes
        improved = False
        for r in prefix_ratios:
            k = max(1, int(n * r))
            early = best[:k]
            tail = best[k:]
            early_sum = early.sum()
            tail_sum = tail.sum()
            if tail_sum == 0.0:
                continue
            for f in shrink_factors:
                new_early = early * f
                delta = early_sum - new_early.sum()
                g = 1.0 + delta / tail_sum
                new_tail = tail * g
                cand = np.concatenate([new_early, new_tail])
                cand = np.clip(cand, 0.0, 1000.0)
                sc, _, _ = _score_and_conv(cand)
                if sc < best_score - 1e-12:
                    best = cand
                    best_score = sc
                    improved = True
        if not improved:
            break
    return best, best_score
