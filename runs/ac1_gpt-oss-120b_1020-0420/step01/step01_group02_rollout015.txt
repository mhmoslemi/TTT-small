import numpy as np
import time

# ----------------------------------------------------------------------
# Generate the (1 - z)^(-1/2) series (used when no parent sequence exists)
# ----------------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return array a_i = binom(2i,i)/4**i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ----------------------------------------------------------------------
# Random block perturbation (keeps O(n) memory)
# ----------------------------------------------------------------------
def random_block_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block of ``arr`` by a log‑normal factor."""
    n = arr.size
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    out = arr.copy()
    out[start:start + width] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out


# ----------------------------------------------------------------------
# Helper: locate the pair (i,j) that contributes most to convolution at idx
# ----------------------------------------------------------------------
def max_contributor(arr: np.ndarray, idx: int) -> tuple[int, int]:
    """Return indices (i,j) with i+j==idx maximising a_i*a_j."""
    n = arr.shape[0]
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_vals = idx - i_vals
    contrib = arr[i_vals] * arr[j_vals]
    best_local = int(np.argmax(contrib))
    return int(i_vals[best_local]), int(j_vals[best_local])


# ----------------------------------------------------------------------
# Compute the AC1 upper bound directly (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def compute_score(arr: np.ndarray) -> float:
    n = arr.shape[0]
    sum_a = arr.sum()
    if sum_a < 0.01:
        return float("inf")
    size = 2 * n - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:size]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    max_iters: int = 2000,
    time_budget: float = 6.5,
    **kwargs,
) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) aiming to minimise
    the AC1 bound.  The routine starts from the provided parent sequence
    (``height_sequence_1``) if available, otherwise from the half‑power series,
    and then performs a greedy flattening of the autocorrelation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise a starting vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):
        a = height_sequence_1.astype(np.float64).copy()
    else:
        a = generate_half_power_series(256)
    np.clip(a, 0.0, 1000.0, out=a)

    # Guarantee a non‑trivial sum (evaluate_sequence requires sum>=0.01)
    if a.sum() < 0.01:
        a += 0.01 / a.size

    current = a
    best = current.copy()
    best_score = compute_score(best)
    target_score = 1.50274349

    start_time = time.time()

    # ------------------------------------------------------------------
    # Iterative greedy improvement (plus occasional random jumps)
    # ------------------------------------------------------------------
    for it in range(max_iters):
        # Time guard
        if time.time() - start_time > time_budget:
            break

        n = current.shape[0]
        size = 2 * n - 1
        N = 1 << ((size - 1).bit_length())

        # Convolution of the current candidate
        fa = np.fft.rfft(current, n=N)
        conv = np.fft.irfft(fa * fa, n=N)[:size]
        max_idx = int(np.argmax(conv))

        # --------------------------------------------------------------
        # Greedy step: weaken the pair that dominates the current maximum
        # --------------------------------------------------------------
        i_star, j_star = max_contributor(current, max_idx)

        # epsilon slowly decays from 0.2 down to 0.001
        eps = max(0.001, 0.2 * (1.0 - it / max_iters))

        cand = current.copy()
        ai = cand[i_star]
        if i_star != j_star:
            aj = cand[j_star]
            mass_removed = eps * (ai + aj)
            cand[i_star] = ai * (1.0 - eps)
            cand[j_star] = aj * (1.0 - eps)

            other_cnt = n - 2
            if other_cnt > 0:
                add = mass_removed / other_cnt
                cand += add
                cand[i_star] -= add
                cand[j_star] -= add
        else:   # i_star == j_star
            mass_removed = eps * ai
            cand[i_star] = ai * (1.0 - eps)

            other_cnt = n - 1
            if other_cnt > 0:
                add = mass_removed / other_cnt
                cand += add
                cand[i_star] -= add

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = compute_score(cand)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            current = cand
            if best_score <= target_score:
                break
            continue  # accept greedy improvement and skip random trial

        # --------------------------------------------------------------
        # Occasionally try a random block perturbation to escape stagnation
        # --------------------------------------------------------------
        if it % 10 == 0:
            rand_cand = random_block_perturbation(current, rng)
            rand_score = compute_score(rand_cand)
            if rand_score < best_score:
                best = rand_cand.copy()
                best_score = rand_score
                current = rand_cand
                if best_score <= target_score:
                    break

    # ------------------------------------------------------------------
    # Return the best vector found as plain Python floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
