import numpy as np

def _score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0*(i+1)*(i+1))
    return a

def propose_candidate(seed: int = 42, num_steps: int = 1000, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            seq = parent.copy()
            n = parent.size
        else:
            n = 256
            seq = _catalan_seq(n)
    else:
        n = 256
        seq = _catalan_seq(n)

    seq = np.clip(seq, 0.0, 1000.0)
    best = seq.copy()
    best_score = _score(best)

    # Deterministic reallocation (shrink early prefix, inflate tail)
    prefix_ratios = [0.05, 0.10, 0.20]
    shrink_factors = np.linspace(0.5, 1.0, 11)
    for r in prefix_ratios:
        k = max(1, int(n * r))
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()
            if delta <= 0.0:
                continue

# ------------------------------------------------------------
# Helper: compute the same score as the provided evaluator
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------

#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)

    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a
