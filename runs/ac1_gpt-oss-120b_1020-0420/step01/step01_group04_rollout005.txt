import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^-1/2: C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _fft_score(a, fft_len):
    """Return the AC1 bound for a non‑negative vector using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2 * a.size - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * a.size * max_b / (s * s)


def _make_transfer(a, rng, max_frac):
    """Move a random fraction of mass from i to j."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a.copy()
    frac = rng.random() * max_frac
    amount = a[i] * frac
    b = a.copy()
    b[i] -= amount
    b[j] += amount
    np.clip(b, 0.0, 1000.0, out=b)
    return b


def _make_block_scale(a, rng, max_sigma):
    """Scale a random contiguous block by a log‑normal factor."""
    n = a.size
    min_len = max(1, n // 200)          # 0.5 % of n
    max_len = max(2, n // 10)           # up to 10 % of n
    width = rng.integers(min_len, max_len + 1)
    start = rng.integers(0, n - width + 1)
    sigma = rng.random() * max_sigma
    factor = np.exp(rng.normal(0.0, sigma))
    b = a.copy()
    b[start:start + width] *= factor
    np.clip(b, 0.0, 1000.0, out=b)
    return b


def _make_bump(a, rng, sigma):
    """Add a small positive bump to a single entry."""
    n = a.size
    idx = rng.integers(0, n)
    bump = max(0.0, rng.normal(0.0, sigma))
    b = a.copy()
    b[idx] += bump
    np.clip(b, 0.0, 1000.0, out=b)
    return b

def propose_candidate(seed=42, num_steps=5000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound is
    as small as possible.  The routine starts from the provided parent
    sequence ``height_sequence_1`` (if present) or from the analytic
    (1‑z)^‑1/2 coefficients, then performs a cheap stochastic optimisation
    using FFT‑based evaluation.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size and initial vector
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
        cur = np.clip(parent, 0.0, 1000.0)
    else:
        n = 256
        cur = _coefficients_binom_half(n)

    # pre‑compute FFT length (next power of two >= 2n‑1)
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()

    # --------------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------------
    best = cur.copy()
    best_score = _fft_score(best, fft_len)

    # optional fallback to analytic start if parent is worse
    if n != cur.size:
        pass
    else:
        # when a parent exists we already used it; otherwise try analytic
        if best_score == np.inf:
            best = _coefficients_binom_half(n)
            best_score = _fft_score(best, fft_len)

    # --------------------------------------------------------------
    # Search parameters
    # --------------------------------------------------------------
    target = 1.50274349
    max_frac = 0.5          # max fraction transferred in a move
    max_sigma = 0.6         # for log‑normal block scaling
    bump_sigma = 0.08       # stddev for single‑entry bump

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    for step in range(num_steps):
        if best_score <= target:
            break

        # slowly anneal the aggressiveness of moves
        if step and step % 1000 == 0:
            max_frac = max(0.02, max_frac * 0.85)
            max_sigma = max(0.02, max_sigma * 0.85)
            bump_sigma = max(0.01, bump_sigma * 0.90)

        op = rng.integers(0, 4)
        if op == 0:
            cand = _make_transfer(best, rng, max_frac)
        elif op == 1:
            cand = _make_block_scale(best, rng, max_sigma)
        elif op == 2:
            cand = _make_bump(best, rng, bump_sigma)
        else:
            # cheap permutation of a short slice
            width = int(rng.integers(2, min(100, n // 4) + 1))
            start = int(rng.integers(0, n - width + 1))
            sub = best[start:start + width].copy()
            rng.shuffle(sub)
            cand = best.copy()
            cand[start:start + width] = sub

        score = _fft_score(cand, fft_len)
        if score < best_score:
            best = cand
            best_score = score

    # --------------------------------------------------------------
    # Final validation with the official evaluator
    # --------------------------------------------------------------
    try:
        final_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(final_score):
            raise ValueError
    except Exception:
        # Fall back to the original parent if available, else to analytic start
        if 'height_sequence_1' in globals():
            best = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        else:
            best = _coefficients_binom_half(n)
        final_score = float(evaluate_sequence(best.tolist()))

    # The function must return a plain Python list of floats
    return [float(x) for x in best]
