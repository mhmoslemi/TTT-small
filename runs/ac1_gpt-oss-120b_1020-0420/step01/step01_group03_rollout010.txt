def compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)   # to ensure
    n = seq.size
    # compute full convolution via fft
    size = int(2 ** np.ceil(np.log2(2* n - 1)))
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[:2*n-1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a ** 2)

def compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    # Enforce non-negatives, clip
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.shape[0]
    # Use FFT for convolution
    size = 1 << ( (2*n - 1).bit_length() )  # next power of two
    # compute rfft
    A = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a * sum_a))

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """
    Generate the first ``length`` terms of the sequence
    a_i = C(2i,i)/4**i, which satisfies (a * a)[k] == 1 for
    0 <= k < length.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _score_of(seq: np.ndarray) -> float:
    """
    Compute the AC1 score 2*n*max(conv)/sum(seq)^2 efficiently via FFT.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.shape[0]
    # next power of two for full convolution length (2n-1)
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))
