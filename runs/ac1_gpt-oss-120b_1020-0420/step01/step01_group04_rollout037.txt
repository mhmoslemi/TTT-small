import numpy as np

def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2, i.e. C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def propose_candidate(seed=42, max_steps=2000, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) whose AC1 upper bound
    is as small as possible.  The routine starts from the analytic
    (1‑z)^‑1/2 coefficients and repeatedly moves mass from the largest entry
    to the smallest one, accepting moves that lower the bound.
    """
    rng = np.random.default_rng(seed)

    # length of the problem (use the supplied parent if it exists)
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256

    # analytic starting point (max conv = 1)
    base = _coefficients_binom_half(n)
    best = np.clip(base, 0.0, 1000.0)

    # try the provided parent as a fallback
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        best_score = float(evaluate_sequence(best.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    except Exception:
        best_score = float(evaluate_sequence(best.tolist()))

    target = 1.50274349

    # -----------------------------------------------------------------
    # Mass‑transfer hill climbing: move a fraction of the largest entry
    # to the smallest entry while the bound improves.
    # -----------------------------------------------------------------
    for _ in range(max_steps):
        if best_score < target:
            break

        # indices of current extreme values
        i_max = int(np.argmax(best))
        i_min = int(np.argmin(best))

        # nothing to move
        if best[i_max] <= 1e-12:
            break

        improved = False
        # try decreasing fractions of the donor
        for frac in (0.5, 0.25, 0.1, 0.05):
            delta = best[i_max] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)

            try:
                score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue

            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                improved = True
                break  # accept first improving fraction

        if not improved:
            # No improvement possible with the current extremes – try a random
            # small perturbation to escape possible plateaus.
            op = rng.integers(0, 3)
            cand = best.copy()
            if op == 0:
                # bump a random tail entry
                idx = rng.integers(int(0.75 * n), n)
                bump = max(0.0, rng.normal(0.0, 0.02))
                cand[idx] += bump
            elif op == 1:
                # scale a random tail block slightly
                start = rng.integers(int(0.5 * n), n)
                width = int(rng.integers(1, max(2, n - start)))
                factor = float(np.exp(rng.normal(0.0, 0.03)))
                cand[start:start + width] *= factor
            else:
                # swap two random positions (pure permutation)
                i, j = rng.integers(0, n, size=2)
                cand[i], cand[j] = cand[j], cand[i]

            np.clip(cand, 0.0, 1000.0, out=cand)
            try:
                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = cand
                    best_score = score
            except Exception:
                pass

    return [float(x) for x in best]
