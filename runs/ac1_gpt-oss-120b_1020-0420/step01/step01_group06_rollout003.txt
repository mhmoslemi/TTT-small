import numpy as np

def _max_conv_info(a: np.ndarray):
    """Return (index, value) of the maximal entry of a*a (full convolution)."""
    n = a.shape[0]
    size = np.fft.next_fast_len(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    conv = conv[: 2 * n - 1]
    idx = int(conv.argmax())
    val = float(conv[idx])
    return idx, val


def _score_normed(a: np.ndarray) -> float:
    """Score assuming the vector will be normalized to sum‑1 (scale invariant)."""
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:
        return np.inf

    _, max_val = _max_conv_info(a)
