import numpy as np

def _fast_score(a):
    n = a.shape[0]
    # zero pad to power of two
    L = 1 << ((2*n - 2).bit_length())
    # compute FFT
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    max_b = conv[:2*n-1].max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

def _fast_score(a: np.ndarray) -> float:
    """
    Fast approximation of the AC1 upper bound using FFT.
    The value is identical to evaluate_sequence up to floating‑point error.
    """
    n = a.shape[0]
    if n == 0:
        return float('inf')
    total = a.sum()
    if total < 0.01 or not np.isfinite(total):
        return float('inf')
    # Length for linear convolution via zero‑padding (next power of two)
    L = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    max_b = conv[: 2 * n - 1].max()
    return 2.0 * n * max_b / (total * total)
