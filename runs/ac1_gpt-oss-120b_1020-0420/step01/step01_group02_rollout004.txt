    n = x.size
    size = 1 << (2*n-1).bit_length() # next power of 2
    # Actually easier: size = int(2**np.ceil(np.log2(2*n-1)))

    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Fast full convolution via FFT (real inputs).
# ------------------------------------------------------------
def full_convolution_fft(x: np.ndarray) -> np.ndarray:
    n = x.size
    m = 2 * n - 1
    # next power of two for efficient FFT
    size = 1
    while size < m:
        size <<= 1
    X = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(X * X, n=size)[:m]
    return conv


# ------------------------------------------------------------
# Multiply a random contiguous block by a log‑normal factor.
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))  # modest perturbation
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 400, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem length and load parent if available
    # --------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):
        n = int(height_sequence_1.size)
        parent = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    else:
        n = 256
        parent = None

    # --------------------------------------------------------
    # Build two simple initial candidates: parent (if any) and
    # the half‑power series. Pick the better one.
    # --------------------------------------------------------
    init_cands = []

    if parent is not None:
        if parent.sum() < 0.01:
            parent = parent + (0.01 - parent.sum()) / n
        init_cands.append(parent)

    half = generate_half_power_series(n)
    half = np.clip(half, 0.0, 1000.0)
    if half.sum() < 0.01:
        half = half + (0.01 - half.sum()) / n
    init_cands.append(half)

    best = None
    best_score = np.inf
    for cand in init_cands:
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

    # fallback safety
    if best is None:
        best = half.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    cur = best.copy()
    step_size = 0.05          # initial learning rate
    no_improve = 0

    # --------------------------------------------------------
    # Optimization loop: gradient‑descent step + occasional
    # random block perturbation as a fallback.
    # --------------------------------------------------------
    max_iter = int(num_steps)
    for it in range(max_iter):
        # full convolution and locating the maximal entry
        conv = full_convolution_fft(cur)
        idx = int(np.argmax(conv))
        max_b = float(conv[idx])
        sum_a = float(cur.sum())

        # gradient of the AC1 objective w.r.t. each coefficient
        delta = 4.0 * n / (sum_a * sum_a)
        const = 4.0 * n * max_b / (sum_a * sum_a * sum_a)

        # shift = idx - i  (i = 0..n-1)
        i_arr = np.arange(n)
        shift = idx - i_arr
        mask = (shift >= 0) & (shift < n)

        grad = -const * np.ones_like(cur)
        grad[mask] += delta * cur[shift[mask]]

        # take a gradient step
        cand = cur - step_size * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        if cand.sum() < 0.01:
            cand += (0.01 - cand.sum()) / n

        cand_score = float(evaluate_sequence(cand.tolist()))
        improved = False

        if np.isfinite(cand_score) and cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score
            cur = cand.copy()
            step_size = min(step_size * 1.1, 1.0)
            no_improve = 0
            improved = True
        else:
            # fallback random block perturbation
            blk = random_block_perturbation(cur, rng)
            blk_score = float(evaluate_sequence(blk.tolist()))
            if np.isfinite(blk_score) and blk_score < best_score - 1e-12:
                best = blk.copy()
                best_score = blk_score
                cur = blk.copy()
                step_size = max(step_size * 0.9, 1e-4)
                no_improve = 0
                improved = True

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                step_size = max(step_size * 0.7, 1e-4)

        if no_improve >= 200:
            break

    # final safety clamp
    np.clip(best, 0.0, 1000.0, out=best)
    if best.sum() < 0.01:
        best += (0.01 - best.sum()) / n

    return [float(v) for v in best]
