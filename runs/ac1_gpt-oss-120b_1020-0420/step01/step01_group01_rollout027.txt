        b[2 * dst] += delta * delta
        b[src + dst] -= 2.0 * delta * delta

        # ---- apply the mass move to the sequence --------------------
        a[src] = a_src - delta
        a[dst] = a_dst + delta

        # ---- evaluate the new score ---------------------------------
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        else:
            # ---- revert: restore a and b --------------------------------
            a[src] = a_src
            a[dst] = a_dst
            # undo the convolution updates (now a is back to original)
            b[src : src + n] += 2.0 * delta * a
            b[dst : dst + n] -= 2.0 * delta * a
            b[2 * src] -= delta * delta
            b[2 * dst] -= delta * delta
            b[src + dst] += 2.0 * delta * delta

    # --------------------------------------------------------
    # Return the best candidate as a plain Python list.
    # --------------------------------------------------------
    return [float(v) for v in best]
```
