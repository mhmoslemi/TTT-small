def ac1_score(seq):
    n = len(seq)
    seq = np.asarray(seq, dtype=np.float64)
    # Zero pad to length m = 1 << ceil_pow2(2*n-1)
    m = 1 << ((2*n-1).bit_length())
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    # conv length m, but we want full convolution length 2*n-1
    conv = conv[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    return 2.0 * n * max_b / (sum_a * sum_a)
```

def analytic_coeffs(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs

def ac1_score(seq):
    n = len(seq)
    # pad to power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)

    conv = conv[:2 * n - 1]

    sum_a = np.sum(seq)
    if sum_a == 0:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    Returns 2·n·max(conv(seq,seq))/(sum(seq)^2).
