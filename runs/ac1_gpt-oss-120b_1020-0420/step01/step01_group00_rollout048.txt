import numpy as np
import math

def _fast_score(vec: np.ndarray) -> float:
    """Fast evaluation of the AC1 upper bound using FFT (no clipping)."""
    n = vec.shape[0]
    if n == 0:
        return np.inf
    # length for linear convolution via FFT
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
