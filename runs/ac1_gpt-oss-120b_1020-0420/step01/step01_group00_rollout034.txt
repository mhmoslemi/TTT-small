import numpy as np

def _fast_score(vec: np.ndarray) -> float:
    n = vec.size
    if n == 0:
        return np.inf
    # next power of two >= 2n-1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _conv_self(a: np.ndarray, m: int) -> np.ndarray:
    """Return full self‑convolution a*a using FFT of length m."""
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * a.size - 1]
    return conv

def _gradient(a: np.ndarray, conv: np.ndarray, alpha: float, m2: int) -> np.ndarray:
    """Gradient of smooth‑max objective w.r.t. a."""
    # soft‑max weights on convolution coefficients
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    w /= w.sum()
    # convolution of w with reversed a (full linear convolution)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=m2)
    fr = np.fft.rfft(a_rev, n=m2)
    full = np.fft.irfft(fw * fr, n=m2)[: 3 * a.size - 2]

    return grad

import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the AC1 upper bound (uses FFT, O(n log n))
# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Linear‑time self‑convolution via FFT (full length 2n‑1)
# ----------------------------------------------------------------------
def _self_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective
#   L(a) = log Σ_k exp(α·(a*a)[k])
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray,
                     conv_full: np.ndarray,
                     alpha: float,
                     size_fft2: int) -> np.ndarray:
    # soft‑max weights on the convolution coefficients
    maxc = conv_full.max()
    w = np.exp(alpha * (conv_full - maxc))
    w /= w.sum()

    # convolution of w with the reversed a (linear, not circular)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size_fft2)
    fr = np.fft.rfft(a_rev, n=size_fft2)
    lin = np.fft.irfft(fw * fr, n=size_fft2)[: 3 * a.size - 2]

    # gradient = 2 * (w ⋆ a)[j] = 2 * lin[n-1 : 2n-1]
    grad = 2.0 * lin[a.size - 1: 2 * a.size - 1]
    return grad

# ----------------------------------------------------------------------

    init_eta = 0.09

    alpha_start = 4.0
