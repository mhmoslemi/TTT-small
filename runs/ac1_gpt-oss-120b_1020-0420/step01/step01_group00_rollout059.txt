def compute_gradient(a):
    n = a.shape[0]
    # compute convolution via FFT
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    max_c = conv.max()
    # find all indices with conv approx max
    eps = max_c * 1e-12 + 1e-14
    ks = np.where(conv >= max_c - eps)[0]
    grad = np.zeros_like(a)
    for k in ks:
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] += 2.0 * a[idx[mask]]
    grad /= len(ks)  # average
    return grad, max_c

    # ---- parameters ---------------------------------------------------------
    max_allowed = 5          # max allowed ordered‑pair count per sum index
    target_k = 200           # desired number of non‑zero entries

        S = []                                      # chosen indices

            if counts[2 * i] + 1 > max_allowed:
