def _next_pow2(x):
    # smallest power of two >= x
    return 1 << (x - 1).bit_length()

def _compute_bound(vec, n, nfft):
    # fast bound using FFT
    # vec: 1-D np.ndarray of float64
    fa = np.fft.rfft(vec, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:2 * n - 1]
    max_b = conv.max()
    sum_a = vec.sum()
    if sum_a == 0.0:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _compute_bound(vec: np.ndarray, n: int, nfft: int) -> float:
    """
    Fast evaluation of the AC1 upper bound using FFT.
    vec – 1‑D array of non‑negative coefficients (sum may be arbitrary).
    """
    fa = np.fft.rfft(vec, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[: 2 * n - 1]
    max_b = conv.max()
    s = vec.sum()
    if s == 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)
