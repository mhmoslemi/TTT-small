import numpy as np

# ----------------------------------------------------------------------
# Helper: classic root‑constant sequence a_i = C(2i,i)/4**i
def _root_constant_seq(length: int) -> np.ndarray:
    """Generate first ``length`` terms of the sequence a_i = binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    # safety clipping – matches evaluator limits
    np.clip(a, 0.0, 1000.0, out=a)
    return a


# ----------------------------------------------------------------------
# Convolution via FFT (matrix‑free, O(n log n))
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Return full convolution seq * seq using a real FFT."""
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    return conv[: 2 * n - 1]                     # exact length


# ----------------------------------------------------------------------
# Gradient of the AC1 score for a fixed maximal lag (kmax)
def _ac1_gradient(seq: np.ndarray, max_val: float, kmax: int) -> np.ndarray:
    """
    Gradient of   f(a) = 2*n*max(a*a) / (sum a)^2
    assuming the maximum is attained at lag ``kmax``.
    """
    n = seq.shape[0]
    s = seq.sum()
    # coefficients of the derivative
    coeff_max = (2.0 * n) / (s * s)                # d/d a of the prefactor
    coeff_den = (4.0 * n * max_val) / (s ** 3)    # d/d a of the denominator term

    # grad of max_convolution for the active lag (2*seq shifted)
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)

    grad = np.full_like(seq, -coeff_den)          # constant part from denominator
    # add the part coming from the active convolution term
    grad[mask] += coeff_max * 2.0 * seq[j[mask]]
    return grad

    cur = np.clip(cur, 0.0, 1000.0)
    if cur.sum() < 0.01:
        cur = _root_constant_seq(n)

    # ------------------------------------------------------------------
    # Compute the initial AC1 score using the exact formula (no full convolve).
    conv = _conv_fft(cur)
    max_val = float(conv.max())
    cur_score = 2.0 * n * max_val / (cur.sum() ** 2)

    best_seq = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Gradient‑descent loop (projected, with simple back‑tracking line search).
    step = 0.1                     # initial learning rate
    max_iters = 2000
    for _ in range(max_iters):
        # recompute convolution and gradient for the current iterate
        conv = _conv_fft(cur)
        max_val = float(conv.max())
        kmax = int(conv.argmax())
        s = cur.sum()
        if s < 0.01:                       # should never happen, break just in case
            break

        grad = _ac1_gradient(cur, max_val, kmax)

        improved = False
        # try a few step‑sizes (back‑tracking) until improvement is seen
        for _ls in range(10):
            nxt = cur - step * grad
            np.clip(nxt, 0.0, 1000.0, out=nxt)
            if nxt.sum() < 0.01:
                step *= 0.5
                continue

            # compute AC1 score directly (no call to heavy evaluator)
            conv_nxt = _conv_fft(nxt)
            max_nxt = float(conv_nxt.max())
            nxt_score = 2.0 * n * max_nxt / (nxt.sum() ** 2)

            if nxt_score < best_score * 0.9999:      # require a tiny improvement
                cur = nxt
                best_seq = nxt.copy()
                best_score = nxt_score
                improved = True
                break
            else:
                step *= 0.5

        if not improved:
            # could not improve – shrink step and eventually give up
            step *= 0.5
            if step < 1e-9:
                break
            continue
        else:
            # on success, gently increase step to accelerate convergence
            step *= 1.05

    # ------------------------------------------------------------------
    # Final sanity check with the authoritative evaluator.
    try:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if not np.isfinite(final_score):
            raise ValueError
    except Exception:
        # fallback to a safe uniform candidate if something went wrong
        best_seq = np.full(n, 1.0, dtype=np.float64)

    return best_seq.tolist()
