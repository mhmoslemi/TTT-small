import numpy as np

def _fast_score(vec):
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def propose_candidate(seed=42, num_steps=12000, **kwargs):
    rng = np.random.default_rng(seed)
    # Load parent if possible
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = np.full(256, 1.0, dtype=np.float64)

    cur = np.clip(parent, 0.0, 1000.0)
    if cur.sum() < 1e-12:
        cur[:] = 1.0
    cur = cur / cur.sum()
    # small perturb to avoid exact uniform stationary point
    cur = cur * np.exp(rng.normal(0.0, 0.04, size=cur.shape))
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur / cur.sum()

    n = cur.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    best = cur.copy()
    best_score = _fast_score(best)
    cur_score = best_score
    momentum = np.zeros_like(cur)

    # schedule parameters
    init_eta = 0.09
    init_T = 0.12

    for step in range(num_steps):
        # convolution via FFT
        fa = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2 * n - 1]
        # index of maximal convolution entry
        k = int(conv.argmax())
        # subgradient built from the shift that attains the peak
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        b = np.zeros_like(cur)
        b[mask] = cur[idx[mask]]
        grad = 2.0 * (b - b.mean())

        # annealed step size and temperature
        eta = init_eta * (1.0 - step / num_steps)
        T = init_T * (1.0 - step / num_steps)

        # momentum update
        momentum = 0.9 * momentum - eta * grad
        cand = cur + momentum
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = cur.copy()
            s = cand.sum()
        cand = cand / s
        cand_score = _fast_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur = cand
            cur_score = cand_score

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # occasional block scaling to escape plateaus
        if (step + 1) % 2500 == 0 and rng.random() < 0.3:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            block = best.copy()
            factor = np.exp(rng.normal(0.0, 0.1))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block = block / block.sum()
            block_score = _fast_score(block)
            if block_score < best_score:
                best = block.copy()
                best_score = block_score
                cur = block.copy()
                cur_score = block_score

    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
