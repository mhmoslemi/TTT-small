```python
import numpy as np
from scipy.optimize import linprog

def catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0]=1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1))/(4.0*(i+1)*(i+1))
    return a

def conv_max(f):
    conv = np.convolve(f, f)
    return conv.max(), conv

def linear_optimize(f, M, top_k=10):
    n = f.size
    # compute convolution f*f to get per-shift contributions weight?
    # Actually constraints are (f * g)[k] <= M.
    # Determine top_k indices with largest (f*f) because they are most binding may also produce higher constraints?
    # We'll just select top_k indices with highest (f * f) but we want those where current (f*f) is near M.
    conv = np.convolve(f, f)
    # Get indices sorted by conv descending
    idx_sorted = np.argsort(-conv) # descending
    # pick first top_k distinct indices
    idx = idx_sorted[:top_k]
    # Build constraint matrix A shape (top_k, n)
    A = np.zeros((top_k, n), dtype=np.float64)
    for row, k in enumerate(idx):
        # We need row vector where weight at j is f[k-j] if inside.
        # Equivalent to extracting f shifted.
        # For j from 0..n-1:
        # weight = f[k-j] if 0 <= k-j < n else 0.
        # That is f[k - j] for j <= k and k - j < n => j >= k-n+1.
        # Simpler: use slice.
        lo = max(0, k - n + 1)
        hi = min(k+1, n)  # exclusive hi
        # For j in [lo, hi):
        # weight = f[k - j]
        # assign.
        # vectorized:
        js = np.arange(lo, hi)
        A[row, js] = f[k - js]
    # Solve LP: max 1^T g  => min -1^T g
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # constraints: A @ g <= M
    # use linprog with method='highs'
    res = linprog(c, A_ub=A, b_ub=np.full(top_k, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: zeros
        return np.zeros_like(f)
    g = res.x
    return g
```

import numpy as np

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0 * (i+1) * (i+1))
    return a

def _score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    n = seq.size
    size = 1 << ((2*n - 1).bit_length())
    # FFT
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:2*n-1]
    m = conv.max()
    return float(2.0 * n * m / (s * s))

    orig = base.copy()

    best_score = _score(best_seq)

    for r in prefix_ratios:

        early = best_seq[:k]
        tail = best_seq[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:

        for f in shrink_factors:

            delta = early_sum - new_early.sum()
            g = 1.0 + delta / tail_sum

import numpy as np

# ------------------------------------------------------------
# Catalan‑like base sequence (the classical start point)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Fast score identical to the harness evaluator (FFT convolution)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    n = seq.size
    # next power of two for an efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    f_fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f_fft * f_fft, n=size)[: 2 * n - 1]
    m = conv.max()
    return float(2.0 * n * m / (s * s))


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_trials: int = 1500, **kwargs):
    """
    Return a list of non‑negative heights with a low AC1 bound.
    The algorithm starts from a Catalan‑like vector (or a provided parent)
    and performs cheap deterministic tweaks followed by many random
    sparse augmentations.  All operations stay linear in the length of the
    vector.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size – use the provided parent if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.clip(base, 0.0, 1000.0)
        n = base.size
    else:
        n = 256
        base = _catalan_seq(n)

    # Keep a clean copy of the starting vector (used for random trials)
    start_vec = base.copy()

    best_seq = start_vec.copy()
    best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Small deterministic mass‑reallocation (identical to the previous program)
    # ------------------------------------------------------------------
    prefix_ratios = [0.05, 0.10, 0.20]                # proportion taken as “early”
    shrink_factors = np.linspace(0.5, 1.0, 11)        # 50 % → 100 %

    for r in prefix_ratios:
        k = max(1, int(n * r))                       # size of the early block
        early = best_seq[:k]
        tail = best_seq[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()
            g = 1.0 + delta / tail_sum
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best_seq = cand
                best_score = sc
                # update early/tail for possible further tweaks
                early = best_seq[:k]
                tail = best_seq[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()

    # ------------------------------------------------------------------
    # Random sparse augmentation:
    #   pick a random subset of positions (avoiding index 0) and add a tiny
    #   constant ε.  The subset size and total added mass are sampled.
    # ------------------------------------------------------------------
    # indices that are safe to increase (exclude the first entry which
    # controls the convolution peak)
    safe_indices = np.arange(1, n, dtype=np.int64)

    # for extra safety we also prefer positions where the current value
    # is modest; this reduces the linear cross‑term with the large a₀.
    low_mask = np.where(start_vec <= 0.2)[0]
    low_mask = low_mask[low_mask != 0]   # drop index 0 if present

    min_m = max(2, int(0.05 * n))
    max_m = int(0.60 * n)

    for _ in range(num_trials):
        m = rng.integers(min_m, max_m + 1)

        # pick from low‑value positions when enough of them exist
        if low_mask.size >= m:
            idx = rng.choice(low_mask, size=m, replace=False)
        else:
            idx = rng.choice(safe_indices, size=m, replace=False)

        # target total increase of the sum (t) – loosely tuned
        t = rng.uniform(5.0, 30.0)           # we want to add roughly 5‑30 mass units
        eps = t / float(m)

        # cap epsilon to keep the peak of the convolution modest
        eps = min(eps, 0.02)

        cand = best_seq.copy()
        cand[idx] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc

    # ------------------------------------------------------------------
    # Deterministic low‑value augmentation:
    #   take the currently smallest entries (excluding the first) and
    #   raise them uniformly – this is cheap and often beneficial.
    # ------------------------------------------------------------------
    sorted_idx = np.argsort(best_seq)
    sorted_idx = sorted_idx[sorted_idx != 0]   # never touch the first entry
    for m in (int(0.10 * n), int(0.20 * n), int(0.40 * n)):
        if m == 0:
            continue
        idx = sorted_idx[:m]
        # aim for a modest average increase (≈0.5 per entry)
        t = 0.5 * m
        eps = t / float(m)
        eps = min(eps, 0.015)
        cand = best_seq.copy()
        cand[idx] += eps
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
