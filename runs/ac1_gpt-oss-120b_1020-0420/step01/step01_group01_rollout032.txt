def propose_candidate(seed=42, num_steps=1000, **kwargs):
    rng = np.random.default_rng(seed)
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        if seq.ndim != 1 or seq.size == 0:
            seq = _catalan_seq(256)
    else:
        seq = _catalan_seq(256)
    # Clip initial sequence.
    seq = np.clip(seq, 0.0, 1000.0)
    best = seq
    best_score = evaluate_sequence(best.tolist())
    n = best.size
    # Deterministic reallocation
    prefix_ratios = np.linspace(0.05, 0.5, 10)  # include up to 0.5
    shrink_factors = np.linspace(0.0, 1.0, 21)  # include 0
    for r in prefix_ratios:
        k = max(1, int(n * r))
        if k >= n:
            continue
        # recompute each iteration with current best
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0:
            continue
        for f in shrink_factors:
            new_early = early * f
            new_early_sum = early_sum * f
            delta = early_sum - new_early_sum
            # avoid division by zero
            if tail_sum == 0:
                continue
            factor = 1.0 + delta / tail_sum
            new_tail = tail * factor
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best = cand
                best_score = sc
                # update early/tail sums for subsequent f? Keep recomputed later.
                early = best[:k]
                tail = best[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()
    # Random local search (optional)
    steps = kwargs.get("num_steps", num_steps)
    for _ in range(steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # move a fraction of src mass
        max_move = best[src] * 0.2
        if max_move <= 0:
            continue
        delta = rng.uniform(0.01, 0.20) * best[src]
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best = cand
            best_score = sc
    return [float(v) for v in best]

import numpy as np

# ------------------------------------------------------------
# Helper: Catalan-like (central binomial) sequence
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        # a_{i+1} = a_i * (2i+2)(2i+1) / (4 (i+1)^2)
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 1000, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine size and initial sequence – use the provided parent if
    # available, otherwise fall back to a Catalan‑like sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            seq = parent.copy()
        else:
            seq = _catalan_seq(256)
    else:
        seq = _catalan_seq(256)

    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.size

    # ------------------------------------------------------------------
    # Score the current candidate using the supplied evaluator.
    # ------------------------------------------------------------------
    best = seq
    best_score = evaluate_sequence(best.tolist())

    # ------------------------------------------------------------------
    # Deterministic reallocation: zero (or shrink) an early prefix and
    # redistribute the removed mass uniformly over the remaining tail.
    # This drastically reduces the convolution peak for sufficiently
    # large prefixes, driving the bound far below the target.
    # ------------------------------------------------------------------
    # Prefix ratios to try – from 5 % up to 50 % of the vector.
    prefix_ratios = np.linspace(0.05, 0.5, 10)
    # Shrink factors – include the extreme case f = 0 (complete removal).
    shrink_factors = np.linspace(0.0, 1.0, 21)

    for r in prefix_ratios:
        k = max(1, int(round(n * r)))
        if k >= n:
            continue

        # Use the current best sequence for each new prefix size.
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue

        for f in shrink_factors:
            # Shrink the early block.
            new_early = early * f
            new_early_sum = early_sum * f

            # Mass removed from the early block.
            delta = early_sum - new_early_sum
            if delta == 0.0:
                # No change – skip expensive evaluation.
                continue

            # Uniformly inflate the tail to keep total sum constant.
            factor = 1.0 + delta / tail_sum
            new_tail = tail * factor

            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)

            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best = cand
                best_score = sc
                # Refresh early/tail after an improvement.
                early = best[:k]
                tail = best[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()
                if tail_sum == 0.0:
                    break  # cannot continue for this prefix

    # ------------------------------------------------------------------
    # Stochastic refinement: random mass transfers that keep the bound
    # non‑increasing. This can surf occasional local improvements.
    # ------------------------------------------------------------------
    steps = kwargs.get("num_steps", num_steps)
    for _ in range(steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if best[src] <= 0.0:
            continue
        # Transfer a random fraction (1 %–20 %) of the source entry.
        delta = best[src] * rng.uniform(0.01, 0.20)
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
