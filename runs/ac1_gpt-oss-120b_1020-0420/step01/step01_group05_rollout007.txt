def refine_gradient(a, steps=200, eta=0.005):
    a = np.maximum(a, 0.0)
    if np.sum(a) == 0:
        a = np.ones_like(a) * (1.0 / a.size)
    # Normalize sum to 1
    a = a / np.sum(a)
    best_a = a.copy()
    best_score = _fast_score(a)
    for t in range(steps):
        # compute convolution
        n = a.size
        m = 2*n - 1
        nfft = 1 << ((m-1).bit_length())
        fa = np.fft.rfft(a, n=nfft)
        c = np.fft.irfft(fa*fa, n=nfft)[:m]
        k_max = int(np.argmax(c))
        M = float(c[k_max])
        # compute gradient
        # base term
        base = -2.0 * M
        # compute valid indices
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        # generate indices
        idx = np.arange(i_start, i_end+1)
        jdx = k_max - idx
        g = np.full_like(a, base)
        g[idx] = a[jdx] + base  # because base negative; Actually we need a[j] - 2M = a[j] + base
        # Update
        a = a - eta * g
        # Clip nonnegatives
        a = np.maximum(a, 0.0)
        # Renorm
        sum_a = np.sum(a)
        if sum_a < 1e-12:
            a = np.ones_like(a) * (1.0 / a.size)
        else:
            a = a / sum_a
        # Evaluate
        s = _fast_score(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
        # Possibly adjust eta if no improvement after some steps, reduce step size
    return best_a, best_score

    s = np.sum(a)
    if s == 0:

    else:
        a = a * (target_sum / s)

    current_eta = eta

    m = 2 * n - 1

    fa = np.fft.rfft(a, n=nfft)
