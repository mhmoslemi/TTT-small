Define function compute_bound(p: np.ndarray) -> float:

    n = p.shape[0]
    m = 1 << ((2 * n - 1).bit_length())  # next power of 2 >= 2n-1
    # compute FFT
    f = np.fft.rfft(p, n=m)
    conv = np.fft.irfft(f * f, n=m)
    # conv length m, but only first 2*n-1 entries matter
    max_b = conv[:2*n-1].max()
    # bound
    return 2.0 * n * max_b / (p.sum()**2)  # p sum should be 1, but we keep generic.

def perturb(p, rng, sigma=0.1, max_width_frac=0.05):
    n = len(p)
    width = int(rng.integers(1, max(1, int(n * max_width_frac)) + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = np.exp(rng.normal(0.0, sigma))
    new = p.copy()
    new[start:start+width] *= factor
    # ensure positivity; but scaling by exp makes >0.
    # renormalize to sum 1
    new_sum = new.sum()
    if new_sum == 0:
        return p
    new /= new_sum
    return new

    """next power of two >= x (x>0)"""
    return 1 << (x - 1).bit_length()


def compute_bound(seq: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative sequence."""
    n = seq.shape[0]
    m = _next_pow2(2 * n - 1)               # length for linear convolution via FFT
    ft = np.fft.rfft(seq, n=m)              # forward real FFT

    max_b = conv.max()
    s = seq.sum()

            sigma: float = 0.12, width_frac: float = 0.02) -> np.ndarray:
