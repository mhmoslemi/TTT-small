def simplex_projection(v):
    # projects onto simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

# Fast computation of the AC1 upper bound (scale invariant)
# -------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Return 2*n*max(conv(vec,vec))/sum(vec)^2 for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    # length for FFT (next power of two)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

                # occasional uphill move to escape shallow local minima
                T = 0.05 * (1.0 - step / steps_per) + 1e-12
                if rng.random() < np.exp(-(cand_score - cur_score) / T):
                    accept = True
            if accept:
                cur = cand
                cur_score = cand_score

            # -------------------------------------------------
            # Keep the best over all restarts
            # -------------------------------------------------
            if cur_score < best_score:
                best_vec = cur.copy()
                best_score = cur_score

            # -------------------------------------------------
            # Periodic random block perturbation (exploration)
            # -------------------------------------------------
            if (step + 1) % 750 == 0 and rng.random() < 0.20:
                width = max(1, n // 120)
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.18))
                blk = cur.copy()
                blk[start:start + width] *= factor
                blk = np.clip(blk, 0.0, 1000.0)
                blk_sum = blk.sum()
                if blk_sum < 1e-12:
                    blk = cur.copy()
                else:
                    blk = blk / blk_sum
                blk_score = _fast_score(blk)
                if blk_score < cur_score:
                    cur = blk
                    cur_score = blk_score
                    if cur_score < best_score:
                        best_vec = cur.copy()
                        best_score = cur_score

    # ---------------------------------------------------------
    # Final validation with the official evaluator
    # ---------------------------------------------------------
    if best_vec is None:
        # safety net: uniform vector
        best_vec = np.full(n, 1.0, dtype=np.float64) / n
    final_score = float(evaluate_sequence(best_vec.tolist()))
    if not np.isfinite(final_score):
        best_vec = np.full(n, 1.0, dtype=np.float64) / n

    return [float(v) for v in best_vec]
