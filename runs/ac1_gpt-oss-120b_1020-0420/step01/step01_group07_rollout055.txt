def conv_fft(x):
    n = x.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[:2*n-1]
    return conv

import numpy as np

# ------------------------------------------------------------
# Fast AC1 bound using FFT (linear memory, O(n log n) time)
# ------------------------------------------------------------
def _fast_ac1(seq: np.ndarray) -> float:
    """
    Compute the AC1 upper bound 2*n*max(conv(seq,seq))/sum(seq)^2
    using FFT for speed.  `seq` is assumed to be a 1‑D float array.
    """
    n = seq.shape[0]
    # length of FFT – next power of two ≥ 2*n‑1
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    maxb = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * maxb / (s * s)
