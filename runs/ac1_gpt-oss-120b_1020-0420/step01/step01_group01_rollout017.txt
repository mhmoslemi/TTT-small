def _full_convolution(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, size)
    B = A * A
    conv = np.fft.irfft(B, size)[:2*n-1]
    return conv
```

def propose_candidate(seed=42, max_moves=8000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size>0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # initial sequence: central binomial
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0 * (i+1)*(i+1))
    a = np.clip(a, 0.0, 1000.0)

    best = a.copy()
    best_score, conv = _score_and_conv(best)
    # if parent present, maybe incorporate as candidate as fallback? probably not needed

    # deterministic central: shrink early prefix and expand tail
    # Not needed now maybe.

    # Main hill climbing
    max_iters = max_moves
    move_frac_range = (0.01, 0.10)  # fraction of src mass to move
    for it in range(max_iters):
        # Update conv to reflect current best if not available?
        # conv currently for best.
        k_max = int(np.argmax(conv))
        # compute i range for contributions
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k_max - i_vals
        # contributions
        prod = best[i_vals] * best[j_vals]
        idx_local = int(np.argmax(prod))
        src = int(i_vals[idx_local])
        # choose destination far enough to have zero contribution at k_max
        if k_max < n:
            dst = n - 1
        else:
            dst = 0
        if dst == src:
            # pick alternative far index
            dst = (dst + n // 2) % n
        # compute delta
        if best[src] <= 0.0:
            continue
        # fraction random within range
        frac = rng.random() * (move_frac_range[1] - move_frac_range[0]) + move_frac_range[0]
        delta = best[src] * frac
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        # clip
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score, cand_conv = _score_and_conv(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            conv = cand_conv
            if best_score < 1.5028:
                break
        else:
            # optionally try random move as fallback
            # random small transfer
            src_r = rng.integers(0, n)
            dst_r = rng.integers(0, n)
            if src_r == dst_r:
                continue
            if best[src_r] <= 0.0:
                continue
            delta_r = best[src_r] * (rng.random() * 0.05 + 0.01)
            cand2 = best.copy()
            cand2[src_r] -= delta_r
            cand2[dst_r] += delta_r
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            cand2_score, cand2_conv = _score_and_conv(cand2)
            if cand2_score < best_score:
                best = cand2
                best_score = cand2_score
                conv = cand2_conv

    # at end, ensure valid sequence (nonnegative)
    best = np.clip(best, 0.0, 1000.0)
    # return list
    return [float(v) for v in best]

import numpy as np

# ------------------------------------------------------------
# Convolution utilities – matrix‑free, O(n log n) memory
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)[: 2 * n - 1]
    return conv


def _score_and_conv(a: np.ndarray):
    """Score = 2·n·max(conv)/(sum(a)^2) and accompanying convolution."""
    s = a.sum()
    if s < 0.01:
        return np.inf, None
    conv = _full_convolution(a)
    m = conv.max()
    score = float(2.0 * a.size * m / (s * s))
    return score, conv
