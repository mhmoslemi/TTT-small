def init_pair_counts(selected, n):
    """Return counts array for selected indices."""
    cnt = np.zeros(2*n-1, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2*i] += 1
        for b in range(a+1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2
    return cnt

def update_counts_add(cnt, i, selected):
    """Add contribution of new index i to cnt, assuming selected is set before addition."""
    for j in selected:
        cnt[i + j] += 2
    cnt[2*i] += 1

def update_counts_remove(cnt, i, selected):
    """Remove contribution of index i from cnt, assuming selected does not contain i after removal."""
    for j in selected:
        cnt[i + j] -= 2
    cnt[2*i] -= 1

    if m == 0:
        return float('inf')
    max_c = cnt.max()
    return float(2.0 * n * max_c / (m*m))

import numpy as np

# ------------------------------------------------------------
# Helpers for binary (0/1) step functions
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """Return ordered‑pair count array for the current set."""
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt


def _bound_from_counts(cnt, m, n):
    """Exact AC1 bound for a binary mask (size m, length n)."""
    if m == 0:
        return float("inf")
    max_c = int(cnt.max())
    return float(2.0 * n * max_c / (m * m))
