import numpy as np

# ----------------------------------------------------------
# Fast evaluation of the AC1 bound using FFT (scale‑invariant).
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float("inf")
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------
# Full convolution (FFT based) returning the array and argmax.
# ----------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    idx = int(np.argmax(c))
    return c, idx


# ----------------------------------------------------------
# Gradient of the convolution value at a given shift index.
# ----------------------------------------------------------
def _grad_at_k(a: np.ndarray, k: int):
    """∂/∂a  Σ_i a_i a_{k-i}  (ordered product version)."""
    n = a.shape[0]
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] += a[idx[mask]]
    g[mask] += a[mask]
    return g
