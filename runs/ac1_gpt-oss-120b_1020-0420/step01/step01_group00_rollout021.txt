def _conv_grad(vec):
    n = vec.size
    mlen = 1 << ((2 * n - 1).bit_length())          # power‑of‑two length for FFT

    shift = k - np.arange(n)
    mask = (shift >= 0) & (shift < n)
    b = np.zeros_like(vec)
    b[mask] = vec[shift[mask]]
    # zero‑mean gradient keeps the total mass unchanged before renormalisation
