def _fast_score(a):
    n = a.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

import numpy as np

def _fast_score(a: np.ndarray) -> float:
    """FFT‑based AC1 bound 2·n·max(conv(a,a)) / (sum(a))²."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]       # full convolution
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)
