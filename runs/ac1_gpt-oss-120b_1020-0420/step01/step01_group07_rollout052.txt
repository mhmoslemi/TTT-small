def greedy_limit(n, max_c_allowed, rng):
    selected = []
    cnt = np.zeros(2*n-1, dtype=np.int32)
    perm = rng.permutation(n)
    for i in perm:
        # check if can add without exceeding max_c_allowed
        # self sum
        s_self = 2*i
        can = cnt[s_self] + 1 <= max_c_allowed
        if can:
            for j in selected:
                s = i + j
                if cnt[s] + 2 > max_c_allowed:
                    can = False
                    break
        if can:
            # update cnt
            cnt[s_self] += 1
            for j in selected:
                s = i + j
                cnt[s] += 2
            selected.append(i)
    return selected, cnt

import numpy as np

def _greedy_set(n: int, max_c: int, rng: np.random.Generator):
    """
    Build a set of indices such that the ordered pair count for any sum
    does not exceed max_c.
    Returns the list of chosen indices and the full count array.
    """
    cnt = np.zeros(2 * n - 1, dtype=np.int16)   # ordered‑pair counts
    selected = []
    order = rng.permutation(n)
    for i in order:
        s_self = 2 * i
        if cnt[s_self] + 1 > max_c:            # self‑pair would overflow
            continue
        conflict = False
        for j in selected:
            s = i + j
            if cnt[s] + 2 > max_c:              # (i,j) & (j,i) would overflow
                conflict = True
                break
        if conflict:
            continue
        # accept i
        cnt[s_self] += 1
        for j in selected:
            s = i + j
            cnt[s] += 2
        selected.append(i)
    return selected, cnt

    for C in C_values:
        for _ in range(num_trials):
            sel, cnt = _greedy_set(n, C, rng)
            m = len(sel)
            if m < 2:
                continue
            max_c = int(cnt.max())
            # exact bound for a binary mask of ones
            score = 2.0 * n * max_c / (m * m)
            if score < best_score:
                cand = np.zeros(n, dtype=np.float64)
                cand[sel] = 1.0
                best_vec = cand
                best_score = score
                if best_score <= TARGET:
                    return [float(x) for x in best_vec.tolist()]

    # --------------------------------------------------------------------
    # If the mask search did not improve enough, try a simple power law
    # --------------------------------------------------------------------
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.45, 0.80, 71):
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                break

    # --------------------------------------------------------------------
    # Final safety: guarantee a positive sum
    # --------------------------------------------------------------------
    if best_vec.sum() < 0.01:
        best_vec = np.full(n, 0.01 / n, dtype=np.float64)

    return [float(x) for x in best_vec.tolist()]
