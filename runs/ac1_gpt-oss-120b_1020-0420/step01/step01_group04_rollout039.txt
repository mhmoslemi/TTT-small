    # compute convolution a*a
    n = a.size
    # length for FFT
    size = 1 << (2*n - 1).bit_length()  # next power of 2 >= 2n-1

    conv = np.fft.irfft(fft_a * fft_a, n=size)
    # take first 2n-1 entries

def propose_candidate(seed=42, num_outer=30, t=0.2, random_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256

    # base shape
    def binom_half():
        coeff = np.empty(n, dtype=np.float64)
        coeff[0] = 1.0
        for i in range(1, n):
            coeff[i] = coeff[i-1] * (2.0*i - 1.0) / (2.0*i)
        return coeff

    base = binom_half()
    base = np.clip(base, 0.0, 1000.0)

    # parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = None

    # evaluate initial candidates
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if parent is not None:
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # current f as best
    f = best.copy()
    # iterative improvement
    for outer in range(num_outer):
        # compute max convolution of f with itself
        conv_self = np.convolve(f, f, mode='full')
        M = conv_self.max()
        if M <= 0:
            break
        # compute diff = M - f, ensure nonnegative
        diff = M - f
        np.clip(diff, 0.0, None, out=diff)
        # compute max of f * diff
        conv_fd = np.convolve(f, diff, mode='full')
        max_fd = conv_fd.max()
        if max_fd <= 0:
            break
        scale = M / max_fd
        g = diff * scale
        # blend
        # optional decreasing t schedule
        tt = t * (1.0 - outer / max(1, num_outer - 1))
        f_new = (1.0 - tt) * f + tt * g
        np.clip(f_new, 0.0, 1000.0, out=f_new)
        # evaluate new candidate
        score = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f_new.copy()
            best_score = score
        # update f for next outer iteration regardless of improvement
        f = f_new

        # early stop if bound already low
        if best_score < 1.50274349:
            break

    # random local search (mass transfer)
    current = best.copy()
    cur_score = best_score
    target = 1.50274349
    # number of random steps
    steps = random_steps
    for step in range(steps):
        if cur_score < target:
            break
        # temperature slowly decreasing
        T = max(0.01, 1.0 - step / steps)
        # propose perturbation: transfer mass from i to j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # delta up to fraction of a_i
        max_delta = current[i] * T * 0.5
        if max_delta <= 0:
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)
        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue
        # accept if better, else maybe accept downhill with prob
        if score < cur_score:
            current = cand
            cur_score = score
            if score < best_score:
                best = cand.copy()
                best_score = score
        else:
            # uphill acceptance
            prob = np.exp((cur_score - score) / (T + 1e-12))
            if rng.random() < prob:
                current = cand
                cur_score = score

    return [float(x) for x in best]

import numpy as np

def _coefficients_binom_half(n):
    """(1‑z)^‑1/2 coefficients C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step‑function whose AC1 bound is low.

    The routine starts from the classic (1‑z)^‑1/2 coefficients (or a
    provided parent), applies a deterministic “LP‑like’’ iteration that
    mixes the current vector with a direction that respects the convolution
    bound, and finishes with a short targeted hill‑climb that moves mass
    away from the most offending lag.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Determine length n (use supplied parent if present)
    # ------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256

    # ------------------------------------------------------------
    # Initial candidates: analytic (1‑z)^‑1/2 and optional parent
    # ------------------------------------------------------------
    base = _coefficients_binom_half(n)
    base = np.clip(base, 0.0, 1000.0)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # parent fallback
    parent = None
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        pass
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(p_score) and p_score < best_score:
            best = parent.copy()
            best_score = p_score

    # ------------------------------------------------------------
    # Deterministic improvement loop (LP‑style)
    # ------------------------------------------------------------
    f = best.copy()                     # current vector
    target = 1.50274349
    num_outer = 30                      # limited outer iterations
    for outer in range(num_outer):
        # self‑convolution peak
        conv_self = np.convolve(f, f, mode="full")
        M = conv_self.max()
        if M <= 0.0:
            break

        # direction that respects f·g ≤ M : diff = M - f (non‑negative part)
        diff = M - f
        np.clip(diff, 0.0, None, out=diff)

        # maximal value of f * diff
        conv_fd = np.convolve(f, diff, mode="full")
        max_fd = conv_fd.max()
        if max_fd <= 0.0:
            break

        # scale diff so that f·g stays ≤ M
        scale = M / max_fd
        g = diff * scale

        # blending factor (decreases linearly)
        t = 0.2 * (1.0 - outer / max(1, num_outer - 1))
        f_new = (1.0 - t) * f + t * g
        np.clip(f_new, 0.0, 1000.0, out=f_new)

        # evaluate candidate
        sc = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = f_new.copy()
            best_score = sc
        # continue with the updated vector regardless of improvement
        f = f_new

        if best_score < target:
            break

    # ------------------------------------------------------------
    # Short targeted hill‑climb: move mass away from the peak lag
    # ------------------------------------------------------------
    cur = best.copy()
    cur_score = best_score
    steps = 2000                       # modest number of refinements
    for step in range(steps):
        if cur_score < target:
            break

        # temperature for stochastic acceptance (linear cooling)
        T = max(0.01, 1.0 - step / steps)

        # compute convolution to locate the offending lag
        conv_cur = np.convolve(cur, cur, mode="full")
        k_max = int(np.argmax(conv_cur))

        # contribution of each index to that lag
        w = np.zeros_like(cur)
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        idxs = np.arange(i_start, i_end, dtype=np.int64)
        jdxs = k_max - idxs
        w[idxs] = cur[idxs] * cur[jdxs]

        # indices to move mass from / to
        i_from = int(np.argmax(w))                 # heavy contributor
        i_to = int(np.argmin(cur))                  # lightest entry

        # propose a small transfer
        delta = cur[i_from] * (0.04 * T)            # up to a few percent
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i_from] -= delta
        cand[i_to] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(sc):
            continue

        # deterministic accept if improvement, else Metropolis accept
        if sc < cur_score:
            cur = cand
            cur_score = sc
            if sc < best_score:
                best = cand.copy()
                best_score = sc
        else:
            # uphill acceptance probability
            prob = np.exp((cur_score - sc) / (T + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = sc

    # ------------------------------------------------------------
    # Final safeguard: ensure a valid (finite) return
    # ------------------------------------------------------------
    if not np.isfinite(best_score):
        # fallback to a simple uniform vector if everything failed
        fallback = np.full(n, 1.0, dtype=np.float64)
        fallback = np.clip(fallback, 0.0, 1000.0)
        return [float(x) for x in fallback]

    return [float(x) for x in best]
