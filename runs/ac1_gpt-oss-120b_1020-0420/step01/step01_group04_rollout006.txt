    except Exception:

        best_score = np.inf

    # If a parent vector exists, evaluate it and possibly replace the best.
    if parent_vec is not None:
        parent = np.clip(parent_vec, 0.0, 1000.0)
        try:
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent
                best_score = parent_score
        except Exception:
            pass

    # Safety fallback: ensure a finite, non‑trivial sequence.
    if not np.isfinite(best_score) or np.sum(best) < 0.01:
        best = np.full(n, 0.01 / n, dtype=np.float64)
        np.clip(best, 0.0, 1000.0, out=best)

    return [float(x) for x in best]
