import numpy as np

def _fast_score(vec):
    n = vec.shape[0]
    if n == 0:
        return np.inf
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

import numpy as np

# ----------------------------------------------------------------------
# Fast AC1 upper bound (uses FFT, scale‑invariant)
# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Return AC1 bound for a non‑negative vector (sum normalisation not required)."""
    n = vec.size
    if n == 0:
        return np.inf
    # length for convolution via FFT (next power of two)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Small local gradient‑based refinement (flattens the highest peak)
# ----------------------------------------------------------------------
def _local_optimize(vec: np.ndarray, steps: int = 200) -> tuple:
    """Run a few gradient‑descent‑like updates starting from vec."""
    cur = vec.copy()
    cur_score = _fast_score(cur)
    n = cur.size
    m = 1 << ((2 * n - 1).bit_length())

    for it in range(steps):
        # convolution and index of maximal coefficient
        fa = np.fft.rfft(cur, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
        k_max = int(conv.argmax())

        # gradient that reduces overlap at k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(cur)
        grad[mask] = cur[idx[mask]]
        grad = 2.0 * (grad - grad.mean())

        # decreasing step size
        eta = 0.06 * (1.0 - it / steps)
        cand = cur - eta * grad
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total < 1e-12:
            cand = cur.copy()
            total = cand.sum()
        cand = cand / total

        cand_score = _fast_score(cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score

    return cur, cur_score

                     Cr: float = 0.7, **kwargs):

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise parent (fallback to uniform if missing or bad)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.full(256, 1.0, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 1e-12:
        parent[:] = 1.0
    parent = parent / parent.sum()        # normalise to sum = 1

    n = parent.size

    # ------------------------------------------------------------------
    # Build initial population (first individual is the parent)
    # ------------------------------------------------------------------
    pop = np.empty((pop_size, n), dtype=np.float64)
    scores = np.empty(pop_size, dtype=np.float64)

    pop[0] = parent
    scores[0] = _fast_score(parent)

    for i in range(1, pop_size):
        # exponential draws give a broad range of shapes
        vec = rng.exponential(scale=1.0, size=n)
        vec = np.clip(vec, 0.0, 1000.0)
        vec = vec / vec.sum()
        pop[i] = vec
        scores[i] = _fast_score(vec)

    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    # ------------------------------------------------------------------
    # Differential Evolution core
    # ------------------------------------------------------------------
    for gen in range(max_gens):
        for i in range(pop_size):

            a, b, c = idxs

            mutant = np.clip(mutant, 0.0, 1000.0)

            trial = np.where(cross, mutant, pop[i])

        best = np.full(n, 1.0 / n)

    return [float(v) for v in best]
