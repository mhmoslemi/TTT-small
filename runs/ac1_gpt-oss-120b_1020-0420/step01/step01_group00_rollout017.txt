def propose_candidate(seed=42, num_steps=8000, **kwargs):
    rng = np.random.default_rng(seed)
    try: parent = np.asarray(height_sequence_1, dtype=np.float64)
    except: parent = np.full(256, 1.0)

    cur = np.clip(parent, 0, 1000)
    if cur.sum() < 1e-12: cur[:] = 1.0

    cur = cur / cur.sum()
    cur = cur * np.exp(rng.normal(0, 0.05, size=cur.shape))
    cur = np.clip(cur,0,1000)
    cur = cur / cur.sum()

    n = cur.size

    cur_score = _fast_score(cur)   # using FFT
    best = cur.copy()
    best_score = cur_score

    # schedule scalars
    init_eta = 0.25
    init_T = 0.2
    beta = 0.9 # momentum (maybe use)
    v = np.zeros_like(cur)

    for step in range(num_steps):
        # compute convolution and max
        conv, max_c = _conv_and_max(cur)
        # get top peaks indices
        thr = max_c * 0.9995   # near max
        ks = np.flatnonzero(conv >= thr)
        # compute avg rev gradient
        grad = np.zeros_like(cur)
        for k in ks:
            # compute rev contribution
            start = max(0, k - (n - 1))
            end = min(k + 1, n)
            # rev[start:end] = cur[k - np.arange(start, end)]
            # we will add to grad
            idx = np.arange(start, end)
            rev_vals = cur[k - idx]
            grad[idx] += rev_vals
        grad /= max(1, ks.size)  # to avoid division by zero (should never)
        # subtract mean to keep zero sum direction
        grad -= grad.mean()

        # update step size and temperature
        eta = init_eta * (1.0 - step / num_steps)
        T = init_T * (1.0 - step / num_steps)

        # momentum update
        v = beta * v - eta * grad
        cand = cur + v   # note sign: v includes negative gradient direction

        # clip and renorm
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = cur.copy()
            s = cand.sum()
        cand = cand / s

        cand_score = _fast_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            accept = (T > 0) and (rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-16)))

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # occasional random block perturbation
        if (step + 1) % 400 == 0 and rng.random() < 0.2:
            # choose block size relative to n
            width = int(rng.integers(max(1, n // 300), max(2, n // 15)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            block = cur.copy()
            factor = np.exp(rng.normal(0, 0.12))
            block[start:start+width] *= factor
            block = np.clip(block,0.0,1000.0)
            block = block / block.sum()
            block_score = _fast_score(block)
            if block_score < cur_score or (T > 0 and rng.random() < np.exp(-(block_score - cur_score) / (T + 1e-16))):
                cur = block
                cur_score = block_score
                if block_score < best_score:
                    best = block.copy()
                    best_score = block_score

    # final evaluation using official evaluator
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # fallback uniform
        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
        best = fallback

    return [float(v) for v in best]

def _fast_score(vec):
