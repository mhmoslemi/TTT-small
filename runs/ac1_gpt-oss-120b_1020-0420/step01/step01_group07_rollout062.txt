import numpy as np, time

def _pair_counts(selected, n):
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = sorted(selected)  # list of ints
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2*i] += 1
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2
    return cnt

def _pair_counts(selected, n):
    """
    Ordered‑pair sum counts for a binary mask.
    selected : set of indices where a_i = 1
    Returns   : np.ndarray of length 2*n‑1 with integer counts.
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = sorted(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt
