import numpy as np

# ----------------------------------------------------------------------
# Fast AC1 bound (identical to the one used in the previous solution)
# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector (no normalization)."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

    """Return (max value, index of max) of the linear self‑convolution of vec."""
    n = vec.size

    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    idx = int(conv.argmax())
    return max_c, idx

    cur /= cur.sum()                         # normalise to sum = 1

    # small random perturbation to avoid a symmetric stationary point
    cur *= np.exp(rng.normal(0.0, 0.05, size=cur.shape))
    cur = np.clip(cur, 0.0, 1000.0)
    cur /= cur.sum()

    n = cur.size
    # initial evaluation
    cur_max, cur_k = _conv_max(cur)
    cur_score = 2.0 * n * cur_max               # sum(cur) == 1
    best = cur.copy()
    best_score = cur_score

    # optimisation hyper‑parameters
    steps = 3000
    eta0 = 0.22          # initial step size (gradient magnitude scale)
    eta_min = 0.001      # lower bound for step size
    beta = 0.9           # momentum coefficient
    T0 = 0.22            # initial temperature for simulated annealing

    momentum = np.zeros_like(cur)

    # early‑stop counters
    no_improve = 0
    max_no_improve = 800

    for step in range(steps):
        # ---- gradient based on the current maximal convolution entry ----
        cur_max, cur_k = _conv_max(cur)            # O(n log n)
        # derivative of conv(k) w.r.t. a_i is a_{k-i} (if in range)
        idx = cur_k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(cur)
        grad[mask] = cur[idx[mask]]
        # make gradient zero‑sum (projection onto the hyper‑plane sum=0)
        grad = 2.0 * (grad - grad.mean())

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- step‑size schedule (linear decay to eta_min) ----
        eta = eta_min + (eta0 - eta_min) * (1.0 - step / steps)

        # ---- propose candidate ----
        cand = cur - eta * momentum
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:                      # protect against collapse
            cand = cur.copy()
            s = cand.sum()
        cand /= s                           # re‑normalise (sum = 1)

        # ---- evaluate candidate (fast path) ----
        cand_score = _fast_score(cand)

        # ---- simulated‑annealing acceptance ----
        T = T0 * (1.0 - step / steps)
        accept = False
        if cand_score < cur_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-(cand_score - cur_score) / T):
            accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- occasional random local perturbation (block scaling) ----
        if (step + 1) % 300 == 0 and rng.random() < 0.25:
            width = int(rng.integers(max(2, n // 200), max(3, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            block = cur.copy()
            factor = np.exp(rng.normal(0.0, 0.14))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block /= block.sum()
            block_score = _fast_score(block)
            if block_score < cur_score:
                cur = block
                cur_score = block_score
                if block_score < best_score:
                    best = block.copy()
                    best_score = block_score
                    no_improve = 0

        # ---- occasional pairwise swap ----
        if (step + 1) % 500 == 0 and rng.random() < 0.15:
            i, j = rng.choice(n, size=2, replace=False)
            swap = cur.copy()
            swap[i], swap[j] = swap[j], swap[i]
            swap_score = _fast_score(swap)
            if swap_score < cur_score:
                cur = swap
                cur_score = swap_score
                if swap_score < best_score:
                    best = swap.copy()
                    best_score = swap_score
                    no_improve = 0

        # ---- early termination if stagnated ----
        if no_improve >= max_no_improve:
            break

    # final safety check with the official evaluator
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # fallback to a plain uniform vector (guaranteed finite)
        fallback = np.full_like(best, 1.0)
        fallback /= fallback.sum()
        best = fallback

    return [float(v) for v in best]
