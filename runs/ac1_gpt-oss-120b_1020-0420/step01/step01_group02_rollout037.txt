def project_to_simplex(v, s):
    # v: numpy array
    # returns projection onto {w >= 0, sum w = s}
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - s
    # find rho: largest i where u_i > css_i / (i+1)
    rho = np.where(u - css / (np.arange(1, n+1)) > 0)[0]

        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Coefficients of (1‑z)^(-1/2) : a[i] = a[i‑1]*(2*i-1)/(2*i)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of a with itself using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(fa * fa, n=n_fft)[: 2 * n - 1]
    return conv


def ac1_score(a: np.ndarray) -> float:
    """Exact AC1 score as used by the evaluator (no clipping inside)."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    n = a.shape[0]
    max_conv = full_convolution(a).max()
    return float(2.0 * n * max_conv / (S * S))


def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Mild log‑normal perturbation of a random contiguous block."""
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 30)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand = base.copy()
    cand[start : start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    # safety

    if a.sum() < 0.01:
        a = np.full_like(a, 1.0)

    n = a.shape[0]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    cur_score = ac1_score(a)
