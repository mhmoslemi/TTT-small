# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative vector (no L1 normalisation required)."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # power‑of‑two length for FFT
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
