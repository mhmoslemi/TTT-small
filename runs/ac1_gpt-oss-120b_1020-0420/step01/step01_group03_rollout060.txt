import numpy as np

# --------------------------------------------------------------
# Helper to compute the root‑constant (Catalan) sequence.
# --------------------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)

# Evaluate a candidate and return its (finite) score.
# --------------------------------------------------------------
def _score_of(seq: np.ndarray) -> float:
    # evaluate_sequence is supplied by the harness
    return float(evaluate_sequence(seq.tolist()))

    attempts to minimise the AC1 upper‑bound score.
    The routine explores a few analytically motivated families of
    sequences and keeps the best one found.
    """
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # Determine the working length.  Use the parent sequence if present.
    # ------------------------------------------------------------------
    try:

    except Exception:
        n = 256  # safe fallback length

    # ------------------------------------------------------------------
    # Baseline: the classic root‑constant sequence.
    # ------------------------------------------------------------------
    best_seq = _root_constant_seq(n)
    best_score = _score_of(best_seq)

    # ------------------------------------------------------------------
    # Family 1 – power‑law decay a_i = (i+1)^{-alpha}.
    # Theoretical analysis shows that for alpha>0.5 the AC1 score can
    # drop below the π/2 ≈ 1.57 bound.
    # ------------------------------------------------------------------
    alphas = np.linspace(0.55, 0.85, 7)   # 0.55 … 0.85 in steps of 0.05
    indices = np.arange(1, n + 1, dtype=np.float64)  # i+1 values

    for alpha in alphas:
        seq = indices ** (-alpha)
        # keep within the allowed range
        seq = np.clip(seq, 0.0, 1000.0)
        sc = _score_of(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq

    # ------------------------------------------------------------------
    # Family 2 – weighted root‑constant: a_i * w_i,
    # where w_i = ((i+1)/n)^{beta} tilts mass towards the tail.
    # ------------------------------------------------------------------
    betas = np.linspace(-0.3, 0.3, 7)   # modest tilts in both directions
    root = _root_constant_seq(n)
    norm = np.arange(1, n + 1, dtype=np.float64) / float(n)

    for beta in betas:
        weight = norm ** beta
        seq = root * weight
        seq = np.clip(seq, 0.0, 1000.0)
        sc = _score_of(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq

    # ------------------------------------------------------------------
    # Family 3 – random Gaussian perturbations around the best so far.
    # A tiny number of trials helps escape possible local plateaus.
    # ------------------------------------------------------------------
    for _ in range(12):
        # perturb around the current best sequence
        noise = rng.normal(scale=0.02, size=n)   # 2 % relative noise
        cand = best_seq * (1.0 + noise)
        cand = np.clip(cand, 0.0, 1000.0)
