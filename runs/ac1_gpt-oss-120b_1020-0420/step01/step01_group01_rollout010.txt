import numpy as np

# ------------------------------------------------------------
# Utility: full convolution via FFT (linear convolution)
# ------------------------------------------------------------
def _fft_conv(seq: np.ndarray, pad_len: int) -> np.ndarray:
    """Return full linear convolution of seq with itself using FFT."""
    fft = np.fft.rfft(seq, pad_len)
    conv = np.fft.irfft(fft * fft, pad_len)
    # conv length is pad_len; the useful part is the first 2*len(seq)-1 entries
    return conv[: 2 * seq.size - 1]


# ------------------------------------------------------------
# AC1 score (same formula as the harness evaluator)
# ------------------------------------------------------------
def _score(seq: np.ndarray, pad_len: int) -> float:
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    conv = _fft_conv(seq, pad_len)
    m = float(conv.max())
    n = seq.size
    return 2.0 * n * m / (s * s)


# ------------------------------------------------------------
# Catalan‑like central‑binomial sequence
# a_i = C(2i,i) / 4^i
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

        if rng.random() < 0.7:
            # ---- gradient‑guided move ---------------------------------
            # Compute convolution and pick a k where it attains the maximum
            fft = np.fft.rfft(cur_seq, pad_len)
            conv = np.fft.irfft(fft * fft, pad_len)[: 2 * n - 1]
            k_max = int(conv.argmax())

            # Derivative of C_k w.r.t a_i for this k
            grad = np.zeros_like(cur_seq)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            idx = np.arange(i_start, i_end + 1)
            grad[idx] = cur_seq[k_max - idx]
            if k_max % 2 == 0:
                diag = k_max // 2
                if 0 <= diag < n:
                    grad[diag] = 2.0 * cur_seq[diag]

            src = int(grad.argmax())
            dst = int(grad.argmin())
            if src == dst or cur_seq[src] <= 0.0:
                # fall back to a tiny random move
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
                if src == dst or cur_seq[src] <= 0.0:
                    continue
            delta = cur_seq[src] * rng.uniform(0.02, 0.20)
            cand = cur_seq.copy()
            cand[src] -= delta
            cand[dst] += delta
        else:
            # ---- pure random move ------------------------------------
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst or cur_seq[src] <= 0.0:
                continue
            delta = cur_seq[src] * rng.uniform(0.01, 0.20)
            cand = cur_seq.copy()
            cand[src] -= delta
            cand[dst] += delta

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate
        cand_score = _score(cand, pad_len)

        # Simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # uphill move accepted probabilistically
            diff = cand_score - cur_score
            if temperature > 0.0 and rng.random() < np.exp(-diff / temperature):
                accept = True

        if accept:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score

    # --------------------------------------------------------
    # Return the best candidate found as a plain Python list.
    # --------------------------------------------------------
    return [float(v) for v in best_seq]
