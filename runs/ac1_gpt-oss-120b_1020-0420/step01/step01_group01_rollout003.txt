import numpy as np

# ------------------------------------------------------------
# Scoring utility – identical to the harness evaluator
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution (FFT for large n)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ------------------------------------------------------------
# Catalan‑like central‑binomial coefficients (fallback generator)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size and initial sequence
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256
            best = _catalan_seq(n)
        else:
            n = base.size
            best = np.clip(base, 0.0, 1000.0)
    else:
        n = 256
        best = _catalan_seq(n)

    best = np.clip(best, 0.0, 1000.0)
    best_score = _score(best)

    # --------------------------------------------------------
    # Phase 1 – deterministic prefix / tail re‑allocation
    # --------------------------------------------------------
    # explore a richer grid of ratios and scaling factors
    ratios = np.linspace(0.02, 0.50, 10)          # proportion of vector taken as prefix
    shrink_factors = np.linspace(0.20, 1.00, 21) # how much to shrink the prefix
    tail_scales = np.linspace(0.80, 2.00, 13)    # uniform factor applied to the tail

    for r in ratios:
        k = max(1, int(n * r))                 # size of the early block
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum <= 0.0:
            continue
        for sf in shrink_factors:
            new_early = early * sf
            delta = early_sum - new_early.sum()  # mass removed from prefix
            # distribute the removed mass uniformly before applying tail scaling
            if tail_sum > 0:
                base_factor = 1.0 + delta / tail_sum
            else:
                base_factor = 1.0
            for ts in tail_scales:
                new_tail = tail * (base_factor * ts)
                cand = np.concatenate([new_early, new_tail])
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc
                    # refresh sums for possible chained improvements
                    early = best[:k]
                    tail = best[k:]
                    early_sum = early.sum()
                    tail_sum = tail.sum()

    # --------------------------------------------------------
    # Phase 2 – guided mass relocation using the current worst
    #           convolution index (k_max)
    # --------------------------------------------------------
    max_phase_iters = max(1, (kwargs.get("num_steps", num_steps) // 2))
    for _ in range(max_phase_iters):
        conv = np.convolve(best, best)
        k_max = int(conv.argmax())
        # indices that contribute to k_max
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        # pick the index with the largest entry among contributors

        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)

            best_score = sc

    # --------------------------------------------------------

    # --------------------------------------------------------

    # --------------------------------------------------------
    return [float(v) for v in best]
