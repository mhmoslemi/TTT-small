import numpy as np

# ------------------------------------------------------------
# Helper: half‑power series (coefficients of (1‑z)^‑1/2)
# ------------------------------------------------------------
def half_power_series(length: int) -> np.ndarray:
    """Generate a non‑negative sequence a_i = C(2i,i)/4^i for i=0..length‑1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Core scoring routine (identical to evaluate_sequence up to scaling)
# ------------------------------------------------------------
def _ac1_score(x: np.ndarray, fft_len: int) -> float:
    """Return the AC1 upper bound for a sequence x (sum(x)=1)."""
    # convolution via real FFT (size fft_len ≥ 2*len(x)-1)
    fx = np.fft.rfft(x, fft_len)
    conv = np.fft.irfft(fx * fx, fft_len)[: 2 * len(x) - 1]
    max_conv = np.max(conv)
    # sum(x) == 1  →  AC1 = 2·n·max_conv
    return 2.0 * len(x) * max_conv


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iter: int = 6000, **kwargs):
    """
    Produce a list of non‑negative heights whose AC1 bound is
    (hopefully) better than the reference construction.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Determine problem length and starting vector (use parent if present)
    # -----------------------------------------------------------------
    n = 256
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        n = int(height_sequence_1.size)
        base = height_sequence_1.astype(np.float64).copy()
    else:
        base = half_power_series(n)

    # -----------------------------------------------------------------
    # Normalise to unit L1 norm (scale invariance of the objective)
    # -----------------------------------------------------------------
    base = np.clip(base, 0.0, 1000.0, out=base)
    total = base.sum()
    if total <= 0.0:
        base[:] = 1.0
        total = base.sum()
    a = base / total                     # now sum(a) == 1

    # -----------------------------------------------------------------
    # Pre‑compute FFT length (next power of two ≥ 2n‑1)
    # -----------------------------------------------------------------
    conv_len = int(2 * n - 1)
    fft_len = 1 << (conv_len - 1).bit_length()   # power‑of‑2 ≥ conv_len

    # -----------------------------------------------------------------
    # Initial score and bookkeeping
    # -----------------------------------------------------------------
    best = a.copy()
    best_score = _ac1_score(best, fft_len)

    # parameters for the two simple search moves
    eta = 0.02                      # step size for gradient‑like move
    transfer_frac = 0.05            # max fraction of a[i] moved in a transfer
    no_improve = 0                  # counter for adaptive decay

    for it in range(max_iter):
        # ------------------------------------------------------------
        # Decide which move to try: 0 = gradient‑style, 1 = mass transfer
        # ------------------------------------------------------------
        if rng.random() < 0.5:
            # ---- Gradient–like move: reduce coefficients that form the current maximum ----
            fx = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fx * fx, fft_len)[:conv_len]
            k_max = int(np.argmax(conv))

            # build the sub‑gradient of (a*a)[k_max] w.r.t. a
            grad = np.zeros_like(a)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            jdx = k_max - idx
            grad[idx] = a[jdx]                     # ∂/∂a_i (a*a)[k_max] = a_{k_max-i}
            cand = a - eta * grad
        else:
            # ---- Random mass transfer from early (large) to late (small) indices ----
            if n < 2:
                continue
            i = rng.integers(0, n // 2)               # from first half
            j = rng.integers(n // 2, n)               # to second half
            step = a[i] * rng.uniform(0.0, transfer_frac)
            if step <= 0.0:
                continue
            cand = a.copy()
            cand[i] -= step
            cand[j] += step

        # ------------------------------------------------------------
        # Enforce feasibility (non‑negativity) and renormalise to unit sum
        # ------------------------------------------------------------
        np.maximum(cand, 0.0, out=cand)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s

        # ------------------------------------------------------------
        # Evaluate candidate with the cheap FFT based score
        # ------------------------------------------------------------
        score = _ac1_score(cand, fft_len)

        # ------------------------------------------------------------
        # Accept if strictly better (record best)
        # ------------------------------------------------------------
        if score < best_score:
            best = cand
            best_score = score
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # adapt step sizes if stagnating
            if no_improve > 500:
                eta *= 0.9
                transfer_frac *= 0.9
                no_improve = 0

    # -----------------------------------------------------------------
    # Return the best candidate found as a plain Python list
    # -----------------------------------------------------------------
    return [float(v) for v in best]
