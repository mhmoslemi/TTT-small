def flatten_step(a, lr):
    n = a.shape[0]
    # compute convolution
    m = 2*n - 1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa*fa, n=nfft)[:m]
    max_val = conv.max()
    # indices where max occurs
    k = np.flatnonzero(conv == max_val)[0]  # take first
    # compute gradient
    # g_i = a_i + a_{k-i} if 0 <= k-i < n else a_i
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = a.copy()
    g[mask] += a[j[mask]]
    # update
    a_new = a - lr * g
    a_new = np.clip(a_new, 0, None)
    # renormalize to sum = 1
    s = a_new.sum()
    if s < 1e-12:
        return a  # fallback
    a_new = a_new * (1.0 / s)
    return a_new
```

def _refine_sequence(a, target_sum, steps=200, lr0=0.5):
    a = a.astype(np.float64)
    # Ensure nonnegative cutoff and sum target
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        # fallback uniform
        a = np.ones_like(a) * (target_sum / a.shape[0])
    else:
        a = a * (target_sum / s)
    best = a.copy()
    best_score = _fast_score(best)
    n = a.shape[0]
    for t in range(steps):
        # compute convolution via FFT
        m = 2 * n - 1
        nfft = 1 << ((m - 1).bit_length())
        fa = np.fft.rfft(a, n=nfft)
        conv = np.fft.irfft(fa * fa, n=nfft)[:m]
        k_max = int(conv.argmax())
        # compute gradient g
        g = np.zeros_like(a)
        # valid i range:
        i0 = max(0, k_max - (n - 1))
        i1 = min(n - 1, k_max)
        # i from i0 to i1 inclusive
        idx = np.arange(i0, i1 + 1, dtype=np.int64)
        j = k_max - idx
        # j guaranteed in [0,n-1]
        g[idx] = 2.0 * a[j]   # gradient factor 2
        # step size:
        lr = lr0 / np.sqrt(t + 1.0)
        a_new = a - lr * g
        a_new = np.clip(a_new, 0.0, None)
        # renormalize sum to target_sum
        s_new = a_new.sum()
        if s_new < 1e-12:
            # cannot renorm; fallback to best
            break
        a_new = a_new * (target_sum / s_new)
        # compute score
        score = _fast_score(a_new)
        if score < best_score:
            best = a_new.copy()
            best_score = score
        a = a_new
    return best, best_score

import numpy as np
import math

# ----------------------------------------------------------
# Fast evaluation of the AC1 bound (identical to the evaluator's
# formula but using FFT for speed).
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)^2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = conv.max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))

# One step of projected sub‑gradient descent on the max‑convolution.
# ----------------------------------------------------------
def _refine_step(a: np.ndarray, target_sum: float, lr: float) -> np.ndarray:
    """
    Perform a single iteration that reduces the maximum entry of
    conv(a, a).  a is assumed to sum to target_sum.
    """
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    k = int(conv.argmax())

    # gradient of the maximal entry (2 * a_{k-i} where defined)
    g = np.zeros_like(a)
    i0 = max(0, k - (n - 1))
    i1 = min(n - 1, k)
    if i0 <= i1:
        idx = np.arange(i0, i1 + 1, dtype=np.int64)
        j = k - idx
        g[idx] = 2.0 * a[j]

    # gradient step, projection onto non‑negative orthant
    a_new = a - lr * g
    a_new = np.clip(a_new, 0.0, None)

    # re‑scale to keep the L1‑norm fixed (scale invariance of the bound)
    s = a_new.sum()
    if s < 1e-12:          # degenerated vector – keep the old one
        return a
    a_new *= (target_sum / s)
    return a_new


# ----------------------------------------------------------
# Full refinement loop (sub‑gradient descent with diminishing step).
# ----------------------------------------------------------
def _refine_sequence(a: np.ndarray,
                     target_sum: float,
                     steps: int = 120,
                     lr0: float = 0.6) -> (np.ndarray, float):
    """
    Starting from a (already scaled to target_sum), run a few
    sub‑gradient iterations and return the best vector seen together
    with its fast score.
    """
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, target_sum / a.size)
    else:
        a = a * (target_sum / s)

    best = a.copy()
    best_score = _fast_score(best)
    n = a.shape[0]

    for t in range(steps):
        lr = lr0 / math.sqrt(t + 1.0)
        a = _refine_step(a, target_sum, lr)

        # fast score using the already computed convolution if possible
        # (re‑compute it – cheap compared to other work)
        nfft = 1 << ((2 * n - 2).bit_length())
        fa = np.fft.rfft(a, n=nfft)
        conv = np.fft.irfft(fa * fa, n=nfft)[:2 * n - 1]
        cur_score = float(2.0 * n * conv.max() / (target_sum * target_sum))

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score

    return best, best_score


# ----------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain a parent construction if it exists.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)   # safe fallback

    n = parent.shape[0]
    target_sum = math.sqrt(2.0 * n)               # normalization used in theory

    # ------------------------------------------------------------------
    # A small pool of candidates generated analytically.
    # ------------------------------------------------------------------
    candidates = []

    # uniform baseline (already normalized)
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    candidates.append((_fast_score(uniform), uniform))

    # parent (scaled to target sum)
    if parent.sum() >= 0.01:
        p = parent * (target_sum / parent.sum())
        candidates.append((_fast_score(p), p))

    i = np.arange(n, dtype=np.float64)

    # 1) Inverse‑square‑root family
    for off in np.linspace(0.1, 4.0, 12):
        a = 1.0 / np.sqrt(i + off)
        a = a * (target_sum / a.sum())
        candidates.append((_fast_score(a), a))

        a_rev = 1.0 / np.sqrt((n - 1 - i) + off)
        a_rev = a_rev * (target_sum / a_rev.sum())
        candidates.append((_fast_score(a_rev), a_rev))

    # 2) Power‑law family
    for pwr in np.linspace(0.35, 0.65, 9):
        a = (i + 1) ** (-pwr)
        a = a * (target_sum / a.sum())
        candidates.append((_fast_score(a), a))

        a_rev = (n - i) ** (-pwr)
        a_rev = a_rev * (target_sum / a_rev.sum())
        candidates.append((_fast_score(a_rev), a_rev))

    # 3) Simple exponential decay
    for alp in np.linspace(0.0005, 0.010, 7):
        a = np.exp(-alp * i)
        a = a * (target_sum / a.sum())
        candidates.append((_fast_score(a), a))

        a_rev = np.exp(-alp * (n - 1 - i))
        a_rev = a_rev * (target_sum / a_rev.sum())
        candidates.append((_fast_score(a_rev), a_rev))

    # ------------------------------------------------------------------
    # Keep the best few raw candidates and refine them more thoroughly.
    # ------------------------------------------------------------------
    candidates.sort(key=lambda x: x[0])
    top_k = 5
    best_seq, best_score = candidates[0][1].copy(), candidates[0][0]

    for _, seq in candidates[:top_k]:
        refined, score = _refine_sequence(seq, target_sum,
                                          steps=180, lr0=0.45)
        if score < best_score:
            best_seq, best_score = refined, score

    # ------------------------------------------------------------------
    # Random jitter around the current best (small multiplicative noise).
    # ------------------------------------------------------------------
    for _ in range(150):
        jitter = np.exp(rng.normal(0.0, 0.018, size=n))
        cand = best_seq * jitter
        cand = cand * (target_sum / cand.sum())
        s = _fast_score(cand)
        if s < best_score:
            best_seq, best_score = cand, s

    # ------------------------------------------------------------------
    # Final short refinement on the best found vector.
    # ------------------------------------------------------------------
    best_seq, best_score = _refine_sequence(best_seq, target_sum,
                                            steps=250, lr0=0.35)

    # ------------------------------------------------------------------
    # Enforce hard bounds required by the official evaluator.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return a plain Python list.
    return [float(x) for x in best_seq.tolist()]
