import numpy as np

def _greedy_subset(max_count, rng, n):
    """
    Construct a subset of {0,…,n-1} such that the ordered‑pair count
    for any sum does not exceed max_count.
    Returns the list of selected indices and the full count array.
    """
    cnt = np.zeros(2 * n - 1, dtype=np.int32)
    selected = []
    # random permutation ensures stochasticity
    perm = rng.permutation(n)
    for i in perm:
        # test self‑pair
        if cnt[2 * i] + 1 > max_count:
            continue
        # test all existing pairs
        ok = True
        for j in selected:
            if cnt[i + j] + 2 > max_count:
                ok = False
                break
        if not ok:
            continue
        # add i – update the count array
        for j in selected:
            cnt[i + j] += 2
        cnt[2 * i] += 1
        selected.append(i)
    return selected, cnt

def propose_candidate(seed=42, attempts_per_c=5, **kwargs):
    """
    Search for a non‑negative step‑function (binary mask) that minimises
    the AC1 upper bound defined by ``evaluate_sequence``.
    Returns a list of floats of length n.
    """
    rng = np.random.default_rng(seed)
    # --------------------------------------------------------------------
    # Determine the problem length
    # --------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
        else:
            n = 256
    except NameError:
        n = 256

    # --------------------------------------------------------------------
    # Baseline: uniform vector (guaranteed feasible)
    # --------------------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # --------------------------------------------------------------------
    # Greedy subset construction for several maximal count limits
    # --------------------------------------------------------------------
    for max_c in range(4, 13):                     # 4 … 12
        for _ in range(attempts_per_c):
            sel, cnt = _greedy_subset(max_c, rng, n)
            m = len(sel)
            if m < 2:
                continue
            cur_max = int(cnt.max())
            # explicit bound (identical to evaluate_sequence for a binary mask)
            score = float(2.0 * n * cur_max / (m * m))
            if score < best_score:
                vec = np.zeros(n, dtype=np.float64)
                vec[sel] = 1.0
                best_vec = vec
                best_score = score
                if best_score <= 1.50274349:
                    return best_vec.tolist()

    # --------------------------------------------------------------------
    # Power‑law sweep – cheap but often decent
    # --------------------------------------------------------------------
    p_vals = np.linspace(0.45, 0.80, 71)   # 0.45 … 0.80 in steps of 0.005
    base = np.arange(1, n + 1, dtype=np.float64)   # 1 … n
    for p in p_vals:
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= 1.50274349:
                return best_vec.tolist()

    # --------------------------------------------------------------------
    # Fallback to the parent construction if it beats everything else
    # --------------------------------------------------------------------
    try:
        parent_vec = height_sequence_1.astype(np.float64)
        parent_vec = np.clip(parent_vec, 0.0, 1000.0)
        score = float(evaluate_sequence(parent_vec.tolist()))
        if score < best_score:
            best_vec = parent_vec
            best_score = score
    except NameError:
        pass

    return best_vec.tolist()

import numpy as np

def _greedy_subset(max_allowed, rng, n):
    """
    Build a subset S ⊆ {0,…,n‑1} such that for every sum k the ordered‑pair
    count cnt[k] does not exceed max_allowed.
    Returns (list_of_indices, cnt_array).
    """
    cnt = np.zeros(2 * n - 1, dtype=np.int32)   # convolution counts
    selected = []
    # Random ordering gives a stochastic construction.
    for i in rng.permutation(n):
        # self‑pair (i,i) contributes 1
        if cnt[2 * i] + 1 > max_allowed:
            continue
        # each existing j contributes two ordered pairs (i,j) and (j,i)
        ok = True
        for j in selected:
            if cnt[i + j] + 2 > max_allowed:
                ok = False
                break
        if not ok:
            continue
        # Accept i: update counts
        for j in selected:
            cnt[i + j] += 2
        cnt[2 * i] += 1
        selected.append(i)
    return selected, cnt
