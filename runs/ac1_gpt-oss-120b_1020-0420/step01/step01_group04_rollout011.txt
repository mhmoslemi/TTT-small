import numpy as np

def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2: C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _score(seq: np.ndarray) -> float:

    return float(evaluate_sequence(seq.tolist()))

            parent = np.clip(parent_arr, 0.0, 1000.0)
            parent_score = _score(parent)
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent.copy()
                best_score = parent_score
        except Exception:
            pass

    target = 1.50274349

    # ------------------------------------------------------------------ #
    # Optimization loop – a handful of cheap, structured mutations.
    # ------------------------------------------------------------------ #
    for step in range(num_steps):
        if best_score < target:
            break

        cand = best.copy()
        op = rng.integers(0, 5)

        if op == 0:
            # ---------------------------------------------------------- #
            # Swap two entries – cheap permutation move.
            # ---------------------------------------------------------- #
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        elif op == 1:
            # ---------------------------------------------------------- #
            # Scale a random contiguous block (up or down).
            # ---------------------------------------------------------- #
            start = rng.integers(0, n)
            max_len = max(2, n // 20)                     # ≈5 % of n
            width = int(rng.integers(1, max_len + 1))
            factor = float(np.exp(rng.normal(0.0, 0.15)))  # log‑normal step
            cand[start:start + width] *= factor

        elif op == 2:
            # ---------------------------------------------------------- #
            # Tail‑up / head‑down combined move.
            # ---------------------------------------------------------- #
            # tail (increase)
            tail_start = rng.integers(int(n * 0.6), n)
            tail_len = int(rng.integers(1, n - tail_start + 1))
            up_factor = float(np.exp(rng.normal(0.25, 0.08)))   # >1
            cand[tail_start:tail_start + tail_len] *= up_factor

            # head (decrease)
            head_len = int(rng.integers(1, int(n * 0.2) + 1))
            down_factor = float(np.exp(rng.normal(-0.25, 0.08)))  # <1
            cand[0:head_len] *= down_factor

        elif op == 3:
            # ---------------------------------------------------------- #
            # Add a tiny positive bump to a random entry.
            # ---------------------------------------------------------- #
            idx = rng.integers(0, n)
            bump = max(0.0, rng.normal(0.0, 0.02))
            cand[idx] += bump

        else:
            # ---------------------------------------------------------- #
            # Shuffle a modest slice.
            # ---------------------------------------------------------- #
            seg_len = int(rng.integers(2, min(200, n // 5) + 1))
            start = rng.integers(0, n - seg_len + 1)
            sub = cand[start:start + seg_len].copy()
            rng.shuffle(sub)
            cand[start:start + seg_len] = sub

        # Clip to the allowed range before evaluation.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # -------------------------------------------------------------- #
        # Evaluate – accept only improving moves.
        # -------------------------------------------------------------- #
        try:
            sc = _score(cand)
        except Exception:
            continue

        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------ #
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------ #
    return [float(x) for x in best]
