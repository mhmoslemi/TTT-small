def compute_score(arr):
    n = len(arr)
    m = 1 << ((2*n - 1) - 1).bit_length()
    # Actually (2*n-1).bit_length() gives number of bits needed for value; we need next power of two >= 2n-1.
    m = 1 << (2*n - 1).bit_length()
    # compute rfft
    fft_a = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(arr)
    return float(2 * n * max_b / (sum_a ** 2))

def propose_candidate(seed=42, **kwargs):
    np.random.seed(seed)
    try:
        n = int(len(height_sequence_1))
        base = _root_constant_seq(n)
    except Exception:
        n = 256

    # Ensure base as float64
    a = base.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(best_a)
    # iterations
    max_iters = 2000
    # step size relative to amount moved
    for it in range(max_iters):
        # select two different indices
        i, j = np.random.choice(n, size=2, replace=False)
        # compute transfer amount delta
        # pick delta as fraction of a[i]
        if a[i] <= 0:
            continue
        frac = np.random.rand() * 0.5  # up to 50% of a_i
        delta = frac * a[i]
        # apply transfer: a_i - delta, a_j + delta (respect bounds)
        new_i = a[i] - delta
        new_j = a[j] + delta
        if new_i < 0:
            delta = a[i]  # remove all
            new_i = 0.0
            new_j = a[j] + delta
        if new_j > 1000.0:
            # limit delta to keep within bound
            delta = 1000.0 - a[j]
            new_i = a[i] - delta
            new_j = 1000.0
            if new_i < 0:
                continue
        # create candidate copy
        cand = a.copy()
        cand[i] = new_i
        cand[j] = new_j
        # compute score
        cand_score = compute_score(cand)
        # Accept if score improved enough
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand  # move to new state
        # else maybe accept with small probability (annealing) to escape local minima
        else:
            # occasional random acceptance
            if np.random.rand() < 0.001:
                a = cand

    # safety: final candidate check
    try:
        final_score = float(evaluate_sequence(best_a.tolist()))
        if not np.isfinite(final_score):
            raise ValueError
    except Exception:
        # fallback to base
        best_a = base
    return best_a.tolist()

import numpy as np

# ----- Helper: generate the classic root‑constant sequence -----------------
def _root_constant_seq(length: int) -> np.ndarray:
    """
    a_i = C(2i,i)/4**i  (i = 0,1,...,length‑1)
    This satisfies (a*a)[k] == 1 for 0 <= k < length.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


# ----- Scoring routine (matrix‑free) ------------------------------------
#   score = 2*n*max(convolution)/sum^2
def _score_of(arr: np.ndarray) -> float:
    n = arr.shape[0]
    # next power of two >= 2*n-1  (FFT length)
    m = 1 << ((2 * n - 1) - 1).bit_length()
    fft = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_b = float(np.max(conv))
    sum_a = float(np.sum(arr))
    # defensive: avoid division by zero (should never happen for our candidates)
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed: int = 42, **kwargs):
    """

    best_score = _score_of(best_seq)

    idx = np.arange(n, dtype=np.float64)
