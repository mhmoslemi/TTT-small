```python
def _full_convolution(a):
    n = a.shape[0]
    m = 2*n-1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, nfft)
    c = np.fft.irfft(fa*fa, nfft)[:m]
    return c
```

import numpy as np

# ------------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound using FFT.
# ------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


# ------------------------------------------------------------
# Full convolution (needed for the reduction heuristic).
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full, length 2n-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    return c


# ------------------------------------------------------------
# Heuristic: reduce the contribution to the current maximal
# convolution entry by slightly shrinking the involved coefficients.
# ------------------------------------------------------------
def _reduce_max_convolution(a: np.ndarray, step: float = 0.1, iters: int = 3) -> np.ndarray:
    a = a.copy()
    n = a.shape[0]
    idx = np.arange(n)
    for _ in range(iters):
        conv = _full_convolution(a)
        k0 = int(np.argmax(conv))
        j = k0 - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]
        # avoid division by zero – conv[k0] is the sum of contrib
        w = contrib / (conv[k0] + 1e-12)
        a = a * (1.0 - step * w)
        a = np.clip(a, 0.0, 1000.0)
    # rescale to keep a reasonable total mass (scale does not affect the bound)
    total = np.sum(a)
    if total > 0:
        a = a / total
    return np.clip(a, 0.0, 1000.0)
