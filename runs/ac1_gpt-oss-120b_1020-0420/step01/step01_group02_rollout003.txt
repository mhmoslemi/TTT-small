import numpy as np

# Target bound we try to beat (a little safety margin is applied later)
TARGET_AC1 = 1.50274349
# Tiny epsilon to keep numbers strictly non‑negative
_EPS = 1e-12


def generate_half_power_series(length: int) -> np.ndarray:
    """Return the first ``length`` coefficients of (1‑z)^(-1/2)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def propose_candidate(seed: int = 42, **kwargs):
    """
    Build a step‑function whose AC1 upper bound is below the target.
    The method starts from a half‑power series (or the provided parent
    sequence) and transfers a small amount of mass from the largest entry
    to the smallest one so that the self‑convolution peak becomes
    ≈TARGET_AC1·S²/(2n) < 1, which yields an AC1 bound below the target.
    """
    # ------------------------------------------------------------------
    # 1️⃣  Choose a starting sequence
    # ------------------------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        a = height_sequence_1.astype(np.float64).copy()
    else:
        # fallback length – the harness will normally provide the parent
        n_fallback = 256
        a = generate_half_power_series(n_fallback)

    n = a.size
    # ensure the allowed range (the evaluator also clips, but we stay safe)
    np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    # 2️⃣  Evaluate the baseline (needed for comparison)
    # ------------------------------------------------------------------
    best = a
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 3️⃣  One deterministic mass‑transfer move:
    #     move mass from the current maximum entry to the current minimum.
    # ------------------------------------------------------------------
    cand = best.copy()
    i_from = int(np.argmax(cand))          # index with largest height
    i_to = int(np.argmin(cand))            # index with smallest height

    # current total sum (invariant after the transfer)
    S = float(cand.sum())

    # required peak of the convolution to meet the target bound
    target_peak = TARGET_AC1 * S * S / (2.0 * n)

    # we stay a little below to be robust against rounding
    a_target = np.sqrt(max(0.0, target_peak) * 0.999)

    # Perform the transfer only if it actually reduces the maximal entry
    if a_target > cand[i_to] + _EPS and cand[i_from] > a_target + _EPS:
        needed = a_target - cand[i_to]                 # how much we need at i_to
        delta = min(needed, cand[i_from] - _EPS)       # take it from i_from
        cand[i_from] -= delta
        cand[i_to] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score

    # ------------------------------------------------------------------
    # 4️⃣  Return a plain Python list of finite floats
    # ------------------------------------------------------------------
    return [float(x) for x in best]
