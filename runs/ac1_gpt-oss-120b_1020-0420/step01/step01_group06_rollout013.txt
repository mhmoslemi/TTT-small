import numpy as np

def _eval_score(vec):
    """Return the AC1 upper bound for a numpy 1‑D vector."""
    return float(evaluate_sequence(vec.tolist()))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that lowers
    the AC1 upper bound.  The routine:
      * starts from the supplied `height_sequence_1` if available,
      * explores a shifted power‑law family (a_i ∝ (i+δ)^‑α),
      * refines the best parameters on a finer grid,
      * applies a few exponential‑decay tweaks,
      * finishes with cheap stochastic block‑perturbations.
    The best vector found is returned as a plain Python list.
    """

    # 1.  Load a parent sequence or fall back to a uniform one.
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                parent = np.clip(arr, 0.0, 1000.0)
        except Exception:
            parent = None

    if parent is None:
        # generic fallback – length 1024, uniform mass
        n = 1024
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size

    # Normalise to unit mass (scale‑invariant for the objective)
    total = parent.sum()
    if total <= 0.0:
        parent = np.ones(n, dtype=np.float64)
        total = n
    parent = parent / total

    # ------------------------------------------------------------------
    # 2.  Initial best candidate.
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = _eval_score(best)
    best_alpha = None
    best_delta = None

    # Index array used throughout the search
    idx = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3.  Coarse search over shifted power‑law shapes.
    # ------------------------------------------------------------------
    start_time = time.time()
    time_limit = 25.0                     # seconds total budget
