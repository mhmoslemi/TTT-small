import numpy as np

def _fast_score_vec(a, n, m, nfft):
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa*fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def _evaluate_vec(a):
    # convert to list for evaluate_sequence
    return evaluate_sequence([float(x) for x in a.tolist()])

def propose_candidate(seed: int = 42, **kwargs):
    rng = np.random.default_rng(seed)
    # Try to load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    # Clip parent
    parent = np.clip(parent, 0.0, 1000.0)
    # baseline candidates
    best_seq = parent.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # uniform fallback in case parent is invalid sum<0.01
    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
    # also evaluate uniform if better
    uniform = np.ones(n, dtype=np.float64)
    u_score = evaluate_sequence(uniform.tolist())
    if u_score < best_score:
        best_seq = uniform
        best_score = u_score

    # Prepare FFT parameters
    m = 2 * n - 1

import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound using FFT.
# ----------------------------------------------------------
def _fast_score_vec(a: np.ndarray, n: int, m: int, nfft: int) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    if a.size != n:
        return float('inf')
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound.

    The routine starts from the supplied parent (height_sequence_1) or a
    uniform fallback, then performs a refined search over inverse‑square‑root
    families, a small power‑law neighbourhood and a final jitter sweep.
    The best candidate according to the exact evaluator is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Load the parent construction if it exists; otherwise fall back.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)
    parent = np.clip(parent, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Baselines: uniform and the parent (if its sum is sufficient).
    # ------------------------------------------------------------------
    best_seq = parent.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    uniform = np.ones(n, dtype=np.float64)
    u_score = evaluate_sequence(uniform.tolist())
    if u_score < best_score:
        best_seq = uniform.copy()
        best_score = u_score

    # --------------------------------------------------------------
    # Pre‑compute FFT parameters used by all fast scores.
    # --------------------------------------------------------------
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    i = np.arange(n, dtype=np.float64)

    best_fast = _fast_score_vec(best_seq, n, m, nfft)

    # --------------------------------------------------------------
    # 1) Inverse‑square‑root family  a_i = (i + off)^{-0.5}
    #    Search a fine grid of offsets.
    # --------------------------------------------------------------
    best_candidate = None
    best_offset = None
    best_rev = False

    offsets = np.linspace(1e-3, 5.0, 2000)  # fine grid
    for off in offsets:
        a = (i + off) ** -0.5
        s_fast = _fast_score_vec(a, n, m, nfft)
        if s_fast < best_fast:
            best_fast = s_fast
            best_candidate = a.copy()
            best_offset = off
            best_rev = False

        a_rev = a[::-1]
        s_fast_rev = _fast_score_vec(a_rev, n, m, nfft)
        if s_fast_rev < best_fast:
            best_fast = s_fast_rev
            best_candidate = a_rev.copy()
            best_offset = off
            best_rev = True

    # --------------------------------------------------------------
    # Evaluate the best inverse‑square‑root candidate exactly.
    # --------------------------------------------------------------
    if best_candidate is not None:
        exact = evaluate_sequence(best_candidate.tolist())
        if exact < best_score:
            best_seq = best_candidate.copy()
            best_score = exact
            best_fast = _fast_score_vec(best_seq, n, m, nfft)

    # --------------------------------------------------------------

    # --------------------------------------------------------------

        s_fast = _fast_score_vec(a, n, m, nfft)
        if s_fast < best_fast:

            if exact < best_score:

                best_score = exact
                best_fast = s_fast

            if exact < best_score:
                best_seq = a_rev.copy()
                best_score = exact
                best_fast = s_fast

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    # We use the fast score to pre‑filter candidates and resort to the

                best_score = exact
                best_fast = s_fast

    # --------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
