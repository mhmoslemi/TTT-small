        """Linear convolution of x with itself via FFT."""
        X = np.fft.rfft(x, n=L_fft)
        y = np.fft.irfft(X * X, n=L_fft)
        return y[: 2 * n - 1]

    def _ac1_score(conv, total):
        """Exact AC1 upper bound from convolution and total sum."""
        # total > 0 by construction.
        return float(2.0 * n * conv.max() / (total * total))

    # ------------------------------------------------------------------
    # 5.  Pick the best starter candidate.
    # ------------------------------------------------------------------
    best_vec = candidates[0]
    best_score = np.inf
    for v in candidates:
        conv = _conv_full(v)
        sc = _ac1_score(conv, v.sum())
        if sc < best_score:
            best_score = sc
            best_vec = v.copy()

    # ------------------------------------------------------------------
    # 6.  Gradient‑descent refinement of the best starter.
    # ------------------------------------------------------------------
    a = best_vec.copy()
    cur_score = best_score

    # Pre‑compute index array for shift construction.
    idx_arr = np.arange(n, dtype=np.int64)

    max_iters = 300
    lr = 0.2                # initial step size
    decay_up = 1.02
    decay_down = 0.5

    for it in range(max_iters):
        # convolution and argmax (max lag)
        conv = _conv_full(a)
        k = int(conv.argmax())
        max_val = conv[k]
        total = a.sum()
        if total == 0.0:
            break

        # shift array: a_{k‑j} where valid, else 0
        j = k - idx_arr
        mask = (j >= 0) & (j < n)
        shift = np.zeros_like(a)
        shift[mask] = a[j[mask]]

        # gradient of f = 2·n·max_b / total²
        grad = (4.0 * n / (total * total)) * shift \
               - (4.0 * n * max_val / (total ** 3)) * a

        # propose a step and enforce feasibility
        a_prop = a - lr * grad
        a_prop = np.maximum(a_prop, 0.0)
        s = a_prop.sum()
        if s == 0.0:
            break
        a_prop /= s

        # compute new score (using convolution from FFT)
        conv_prop = _conv_full(a_prop)
        sc_prop = _ac1_score(conv_prop, a_prop.sum())

        if sc_prop < cur_score:          # accept
            a = a_prop
            cur_score = sc_prop
            if sc_prop < best_score:
                best_score = sc_prop
                best_vec = a.copy()
            lr = min(lr * decay_up, 1.0)
        else:                            # reject, shrink step size
            lr *= decay_down

        # occasional random jitter to escape shallow minima
        if (it + 1) % 30 == 0:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = np.maximum(a * jitter, 0.0)
            a /= a.sum()
            cur_score = _ac1_score(_conv_full(a), a.sum())
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.copy()

    # ------------------------------------------------------------------
    # 7.  Final sanity checks and return.
    # ------------------------------------------------------------------
    if best_vec.sum() == 0.0:
        best_vec = np.full(n, 1.0 / n)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec.tolist()]
