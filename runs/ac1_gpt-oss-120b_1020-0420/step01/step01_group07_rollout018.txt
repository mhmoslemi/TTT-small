    selected = set()
    counts = {}
    # generate random order of indices
    candidates = rng.permutation(n)
    for cand in candidates:
        # quick early termination: if len(selected) >= target_m maybe break? Not needed.

        ok = True

        s_self = cand + cand
        if counts.get(s_self, 0) + 1 > max_allowed:
            ok = False
        else:
            for i in selected:
                s = i + cand
                if counts.get(s, 0) + 1 > max_allowed:
                    ok = False
                    break
        if ok:
            # update
            selected.add(cand)
            counts[s_self] = counts.get(s_self, 0) + 1
            for i in selected - {cand}:
                s = i + cand
                counts[s] = counts.get(s, 0) + 1
    # compute max_c
    max_c = max(counts.values()) if counts else 0
    return selected, max_c

import numpy as np

# ------------------------------------------------------------
# Helper: build a random subset of indices with a bound on the
# ordered pair count for each possible sum (i + j).
# ------------------------------------------------------------
def _build_random_set(max_allowed: int, rng: np.random.Generator, n: int):
    """
    Greedy construction of a set S ⊆ {0,…,n‑1}.
    For every possible sum s = i + j (ordered pairs) the number of
    contributions a_i * a_j with a_i = a_j = 1 is kept ≤ max_allowed.
    Returns the set and the achieved maximum ordered count.
    """
    selected = set()
    # ordered count per sum (0 … 2*n-2)
    cnt = {}
    # random ordering of candidate indices
    cand_order = rng.permutation(n)
    for cand in cand_order:
        # check self‑pair (i,i)
        s_self = cand + cand
        if cnt.get(s_self, 0) + 1 > max_allowed:
            continue
        # check all pairs with already selected indices
        ok = True
        for i in selected:
            s = i + cand
            if cnt.get(s, 0) + 2 > max_allowed:   # two ordered contributions (i,cand) & (cand,i)
                ok = False
                break
        if not ok:
            continue
        # Accept candidate, update counts
        cnt[s_self] = cnt.get(s_self, 0) + 1
        for i in selected:
            s = i + cand
            cnt[s] = cnt.get(s, 0) + 2
        selected.add(cand)
    max_c = max(cnt.values()) if cnt else 0
    return selected, max_c

    # Baseline candidate (uniform ones) – always feasible
    # --------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # --------------------------------------------------------
    # Try the provided parent sequence as a candidate as well
    # --------------------------------------------------------
    if 'parent' in locals():
        cand = np.clip(np.maximum(parent, 0.0), 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score

    # --------------------------------------------------------
    # Random sparse‑set search with a bound on ordered pair counts
    # --------------------------------------------------------
    TARGET = 1.50274349
    # try several allowances for the ordered count per sum
    allowance_options = [4, 5, 6, 7, 8]
    attempts_per_allowance = 30

    for max_allowed in allowance_options:
        for _ in range(attempts_per_allowance):
            sel, max_c = _build_random_set(max_allowed, rng, n)
            m = len(sel)
            if m < 2:                     # need at least two non‑zeros to improve
                continue
            # quick bound estimate using the binary‑mask formula
            bound_est = 2.0 * n * max_c / (m * m)
            if bound_est >= best_score:   # cannot beat current best
                continue
            # Build the binary vector and evaluate exactly
            cand = np.zeros(n, dtype=np.float64)
            cand[list(sel)] = 1.0
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_vec = cand
                best_score = score
                if best_score <= TARGET:
                    return [float(x) for x in best_vec]

    # --------------------------------------------------------
    # Power‑law family (cheap cheap alternative)
    # --------------------------------------------------------
    p_vals = np.linspace(0.45, 0.80, 71)          # 0.45 … 0.80
    base = np.arange(1, n + 1, dtype=np.float64)  # 1 … n
    for p in p_vals:
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                break

    # --------------------------------------------------------
    # Tiny local perturbations (optional polishing)
    # --------------------------------------------------------
    for _ in range(num_steps):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.05)))   # modest multiplicative noise
        cand[start:stop] *= factor

        if rng.random() < 0.20:                        # occasional mirroring smoothing
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                break

    # --------------------------------------------------------
    # Return the best vector found as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
