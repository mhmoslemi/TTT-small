import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^(-1/2):  C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _fast_bound(a, fft_len):
    """Return the AC1 bound using an FFT‑based convolution (fast, accurate)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2 * a.size - 1]
    M = conv.max()
    S = a.sum()
    if S < 1e-12:
        return np.inf
    n = a.size
    return float(2.0 * n * M / (S * S))


def _random_block_move(a, rng):
    """Multiply a random contiguous block by a log‑normal factor."""
    n = a.size
    min_len = max(1, n // 200)          # ≈0.5 % of n
    max_len = max(2, n // 5)            # ≈20 % of n
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.15)))  # can be <1 or >1
    cand = a.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _transfer_move(a, rng):
    """Transfer a fraction of mass from an early index to a late one."""
    n = a.size
    early = int(0.12 * n)
    late = int(0.88 * n)
    if early == 0 or late >= n:
        return a.copy()
    i = rng.integers(0, early)
    j = rng.integers(late, n)
    delta = min(a[i], rng.uniform(0.0, a[i] * 0.25))
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _tail_boost_move(a, rng):
    """Boost the last quarter of the vector by a log‑normal factor."""
    n = a.size
    start = int(rng.integers(int(0.75 * n), n))

    factor = float(np.exp(rng.normal(0.0, 0.25)))
    cand = a.copy()
    cand[start:] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _bump_move(a, rng):

    n = a.size

def _swap_move(a, rng):
    """Swap two random positions."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    cand = a.copy()
    cand[i], cand[j] = cand[j], cand[i]

def _permute_slice_move(a, rng):

    n = a.size
    max_len = min(120, n // 4)
