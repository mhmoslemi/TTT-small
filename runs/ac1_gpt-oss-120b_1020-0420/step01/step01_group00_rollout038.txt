def _conv_and_max(vec):
    n = vec.size
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2*n - 1]
    max_c = conv.max()
    k_max = int(conv.argmax())
    return conv, max_c, k_max

# ------------------------------------------------------------
# Fast AC1 upper bound using FFT (scale‑invariant)
# ------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def _conv_and_max(vec: np.ndarray):
    """Return full convolution, its maximum, and index of that maximum."""
    n = vec.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    k_max = int(conv.argmax())
    return conv, max_c, k_max
