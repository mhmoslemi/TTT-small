            continue
        cand = best.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] = min(1000.0, cand[j] + delta)
        # re‑normalise to keep scale‑invariance (sum does not matter for the score)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s
        score = _eval(cand.tolist())
        # keep any improvement (unlikely) and update best
        if score < best_score:
            best, best_score = cand, score

    # ------------------------------------------------------------------
    # Final safety: if for any reason the best candidate is invalid,
    # fall back to a safe uniform construction.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.full(n, 1.0 / n, dtype=np.float64)

    # Return as plain Python list of floats.
    return [float(v) for v in best]
