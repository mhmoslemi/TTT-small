import numpy as np

# ------------------------------------------------------------
# Half‑power series (1‑z)^(-1/2) – a compact baseline
# ------------------------------------------------------------
def half_power_series(n: int) -> np.ndarray:
    """Return the first n coefficients of (1‑z)^(-1/2)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Fast convolution max using FFT (linear‑time, O(n log n))
# ------------------------------------------------------------
def conv_max_fft(x: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the full convolution x*x via FFT."""
    # rfft returns N/2+1 complex numbers for real input of length fft_len
    Fx = np.fft.rfft(x, fft_len)
    # element‑wise product corresponds to convolution of x with itself
    conv = np.fft.irfft(Fx * Fx, fft_len)
    # only the first 2*len(x)-1 entries are valid
    return conv[: 2 * x.shape[0] - 1].max()


# ------------------------------------------------------------
# Scoring function identical to evaluate_sequence (but cheaper)
# ------------------------------------------------------------
def score_of(x: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(x,x))/sum(x)^2 after clipping to [0,1000]."""
    np.clip(x, 0.0, 1000.0, out=x)
    s = x.sum()
    if s < 0.01:
        return np.inf
    m = conv_max_fft(x, fft_len)
    n = x.shape[0]
    return float(2.0 * n * m / (s * s))


# ------------------------------------------------------------
# Random block perturbation – multiplicative log‑normal factor
# ------------------------------------------------------------
def random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = a.shape[0]
    if n == 1:
        return a.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    # moderate variance; allows both up‑ and down‑scaling
    factor = float(np.exp(rng.normal(0.0, 0.20)))
    b = a.copy()
    b[start:start + width] *= factor
    np.clip(b, 0.0, 1000.0, out=b)
    return b

    else:
        n = 256

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two

    # ----- initial best -------------------------------------------------
    best = base.copy()
    best_score = score_of(best, fft_len)

    for start_ratio in (0.5, 0.6, 0.7, 0.8):
