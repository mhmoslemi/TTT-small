def proj_simplex(v, z):
    # v: array shape (n,)
    # projects onto {x >=0, sum x = z}
    # Implementation from reference.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x*x using real FFT (size power of two)."""
    n = x.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[:m]
    return conv


def _proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto the simplex {w>=0, sum w = s}."""
    if s < 0:
        raise ValueError
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho

        nz = w > 0
        if nz.any():
            w[nz] += diff / np.count_nonzero(nz)
            w = np.maximum(w, 0.0)
    return w
