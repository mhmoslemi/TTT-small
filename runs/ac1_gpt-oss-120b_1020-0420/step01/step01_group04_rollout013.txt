import numpy as np

def propose_candidate(seed=42, num_steps=8000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound evaluated by the provided ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Initialise sequence – use the supplied parent if present, else analytic.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = seq.shape[0]
    else:
        # analytic (1‑z)^‑1/2 coefficients, length 4806 (typical problem size)
        n = 4806
        seq = np.empty(n, dtype=np.float64)
        seq[0] = 1.0
        for i in range(1, n):
            seq[i] = seq[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    seq = np.clip(seq, 0.0, 1000.0)

    # ----------------------------------------------------------------------
    # Basic evaluation of the starting point.
    # ----------------------------------------------------------------------
    try:
        best_score = float(evaluate_sequence(seq.tolist()))
    except Exception:                     # pragma: no cover – safety net
        best_score = np.inf
    if not np.isfinite(best_score):
        # fallback to a tiny random vector if the parent is unusable
        seq = rng.random(n) * 0.1
        best_score = float(evaluate_sequence(seq.tolist()))
    best_seq = seq.copy()

    target = 1.50274349

    # ----------------------------------------------------------------------
    # Main hill‑climbing / mass‑transfer loop.
    # ----------------------------------------------------------------------
    for step in range(num_steps):
        if best_score < target:
            break

        # current convolution and the index attaining the maximum
        conv = np.convolve(best_seq, best_seq)
        max_idx = int(conv.argmax())

        # contribution of each entry to the current maximal convolution value
        idx = np.arange(n)
        mate = max_idx - idx
        valid = (mate >= 0) & (mate < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = best_seq[valid] * best_seq[mate[valid]]

        # ------------------------------------------------------------------
        # Choose a move.
        # ------------------------------------------------------------------
        op = rng.integers(0, 2)          # 0 – directed transfer, 1 – random transfer
        if op == 0:
            # Move mass from a high‑contributing position to a low‑contributing one.
            i_high = int(np.argmax(contrib))
            if contrib[i_high] == 0.0:

            # low‑contribution candidates (strictly smaller than 10 % of the high value)
            low_mask = (contrib < 0.1 * contrib[i_high])

            low_candidates = np.flatnonzero(low_mask)
