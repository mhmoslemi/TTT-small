import numpy as np

# ------------------------------------------------------------
# Helper: Catalan-like (central binomial) sequence
#   a_i = C(2i,i) / 4^i
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

# ------------------------------------------------------------
# Linear convolution via FFT (full, length 2n-1)
# ------------------------------------------------------------
def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Return full convolution x*x using FFT of length L (power of two)."""
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)[: 2 * x.size - 1]
    return conv

# ------------------------------------------------------------
# Compute the AC1 score (identical to evaluate_sequence)
# ------------------------------------------------------------
def _ac1_score(seq: np.ndarray, conv: np.ndarray) -> float:
    s = seq.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * seq.size * conv.max() / (s * s))

# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iter: int = 2000, **kwargs):
    """
    Return a list of non‑negative heights with a low AC1 bound.
    The algorithm starts from the Catalan‑like sequence and
    performs greedy pairwise mass moves that never increase the
    maximal autocorrelation.
    """
    rng = np.random.default_rng(seed)

    # ----- determine problem size -------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # ----- initial sequence --------------------------------------------------------
    a = _catalan_seq(n)
    a = np.clip(a, 0.0, 1000.0)

    # ----- FFT plan ---------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two >= 2n‑1
    conv = _conv_fft(a, L)
    best_score = _ac1_score(a, conv)
    best = a.copy()

    # parameters for the greedy search
    step_ratio = 0.20        # fraction of source mass moved each successful step
    no_improve = 0
    max_no_improve = 300     # after this many useless steps we stop

    for _ in range(max_iter):
        # locate the current maximal convolution entry
        max_idx = int(conv.argmax())
        max_val = float(conv[max_idx])

        # partner[i] = a[max_idx - i] when the index is inside the sequence,
        # otherwise zero.  This is the contribution of a[i] to conv[max_idx].
        idx = np.arange(n)
        mask = (max_idx - idx >= 0) & (max_idx - idx < n)
        partner = np.zeros_like(a)
        partner[mask] = a[max_idx - idx[mask]]

        # choose a source index that most influences the maximum
        src = int(np.argmax(a * partner))
        if a[src] <= 0 or partner[src] <= 0:
            break  # nothing can improve the peak any more

        # choose a destination that does NOT influence the current maximum
        zero_mask = partner == 0
        if zero_mask.any():
            dst_candidates = np.where(zero_mask)[0]
            # avoid moving to the same location
            if dst_candidates.size > 1:
                dst_candidates = dst_candidates[dst_candidates != src]
            dst = int(rng.choice(dst_candidates))
        else:
            dst = int(np.argmin(partner))

        # move a fraction of mass from src to dst
        eps = a[src] * step_ratio
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate candidate
        conv_cand = _conv_fft(cand, L)
        cand_score = _ac1_score(cand, conv_cand)

        if cand_score < best_score:
            # accept improvement
            a = cand
            conv = conv_cand
            best = cand
            best_score = cand_score
            no_improve = 0
            # gradually tighten step size as we approach a good solution
            step_ratio = max(0.02, step_ratio * 0.995)
        else:
            # try a smaller move before giving up on this pair
            eps2 = a[src] * (step_ratio * 0.2)
            if eps2 > 0:
                cand2 = a.copy()
                cand2[src] -= eps2
                cand2[dst] += eps2
                cand2 = np.clip(cand2, 0.0, 1000.0)
                conv_cand2 = _conv_fft(cand2, L)
                cand2_score = _ac1_score(cand2, conv_cand2)
                if cand2_score < best_score:
                    a = cand2
                    conv = conv_cand2
                    best = cand2
                    best_score = cand2_score
                    no_improve = 0

            # no improvement this round
            no_improve += 1
            if no_improve >= max_no_improve:

        # early exit if we already beat the target bound
        if best_score <= 1.50274349:
            break

    # ----- final fallback: ensure a valid non‑empty list -------------------------
    if best.size == 0:
        best = np.ones(1, dtype=np.float64)

    return [float(v) for v in best]
