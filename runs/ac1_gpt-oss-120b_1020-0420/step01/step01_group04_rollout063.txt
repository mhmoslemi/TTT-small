import numpy as np

# ----------------------------------------------------------------------
# Analytic binomial‑half coefficients:  C_i = comb(2i,i)/4^i  (generating function (1‑z)^‑1/2)
def _coefficients_binom_half(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs

def propose_candidate(seed=42, num_steps=2500, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine starts from the analytic (1‑z)^‑1/2 coefficients and
    performs a cheap parametric search: scale an early block down and a
    later block up.  A small local hill‑climbing phase refines the best
    candidate further.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and a reasonable starting point
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.shape[0])
        # analytic baseline
        base = _coefficients_binom_half(n)
        # clip both to respect the evaluator’s limits
        base = np.clip(base, 0.0, 1000.0)
        parent = np.clip(parent, 0.0, 1000.0)

        # pick the better of analytic vs supplied parent as initial best
        best = base.copy()
        best_score = float(evaluate_sequence(best.tolist()))
        try:
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent.copy()
                best_score = parent_score
        except Exception:
            pass
    else:
        # No parent available – fall back to analytic coefficients
        n = 256
        best = _coefficients_binom_half(n)
        best = np.clip(best, 0.0, 1000.0)
        best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Parametric two‑block search:
    #   - early block (indices < early_len) is scaled down (factor s <= 1)
    #   - later block (indices >= early_len) is scaled up (factor t >= 1)
    early_len = max(1, int(n * 0.25))          # 25 % of the vector
    s_vals = np.linspace(0.55, 1.0, 21)       # reduction factors
    t_vals = np.linspace(1.0, 3.5, 21)        # amplification factors

    for s in s_vals:
        for t in t_vals:
            cand = best.copy()
            cand[:early_len] *= s
            cand[early_len:] *= t
            # clipping is cheap and guarantees evaluator safety
            np.clip(cand, 0.0, 1000.0, out=cand)
            try:
                score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue
            if np.isfinite(score) and score < best_score:
                best = cand.copy()
                best_score = score
                # early exit if we already beat the target
                if best_score < 1.50274349:
                    break
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Local refinement: small random multiplicative tweaks on modest slices
    # (this mimics the hill‑climbing moves of the reference solution)
    refinement_steps = max(1, num_steps - (len(s_vals) * len(t_vals)))
    for _ in range(refinement_steps):
        cand = best.copy()
        # choose a random slice (up to ~1 % of the vector)
        start = int(rng.integers(0, n))

        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            if best_score < 1.50274349:
                break

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list of floats
    return [float(x) for x in best]
