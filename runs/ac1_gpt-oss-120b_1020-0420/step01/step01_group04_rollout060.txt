import numpy as np

def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2,  a_i = C(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _make_block_candidate(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block of ``base`` by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    # block length between 1 % and 25 % of n
    min_len = max(1, n // 100)
    max_len = max(2, n // 4)
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.20)))   # log‑normal perturbation
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _early_tail_trade(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce a random early block and use the freed mass to boost a random tail block.
    The net effect often lowers the convolution peak while increasing the total sum.
    """
    n = base.size
    cand = base.copy()

    # sizes (≈5 % of n, bounded below)
    L_e = max(1, n // 20)          # early block length
    L_t = max(1, n // 20)          # tail block length

    # choose early block inside first 10 % of the sequence
    early_start_max = max(0, n // 10 - L_e)
    early_start = int(rng.integers(0, early_start_max + 1))

    # choose tail block inside last 10 % of the sequence
    tail_start_min = n - n // 10
    tail_start_max = max(tail_start_min, n - L_t)
    tail_start = int(rng.integers(tail_start_min, tail_start_max + 1))

    # sums of the two blocks
    sum_e = float(np.sum(cand[early_start:early_start + L_e]))
    sum_t = float(np.sum(cand[tail_start:tail_start + L_t]))

    if sum_e == 0.0:
        return cand  # nothing to trade

    # reduction factor for early block (median around 0.75, never >1)
    red_factor = np.exp(rng.normal(np.log(0.75), 0.25))
    red_factor = float(min(1.0, max(0.01, red_factor)))

    freed = (1.0 - red_factor) * sum_e                     # mass released
    # extra multiplier for tail block (≥1, up to ~2)
    extra_mult = float(rng.uniform(1.0, 2.0))
    # safe denominator

    up_factor = 1.0 + extra_mult * freed / denom
    up_factor = float(min(up_factor, 20.0))               # avoid overflow

    cand[tail_start:tail_start + L_t] *= up_factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
