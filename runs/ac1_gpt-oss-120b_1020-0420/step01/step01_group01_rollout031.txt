import numpy as np

# ------------------------------------------------------------
# Scoring – matches the evaluator exactly
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Compute 2*n*max(conv(seq,seq))/sum(seq)**2 after clipping."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution (FFT for large n)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Central‑binomial (Catalan‑like) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Deterministic re‑allocation of mass from an early prefix
# ------------------------------------------------------------
def _deterministic_reallocation(seq: np.ndarray, cur_score: float) -> tuple:
    n = seq.size
    # try many prefix lengths (1 % … 30 %) and shrink factors (0.1 … 1.0)
    pref_fracs = np.arange(0.01, 0.31, 0.01)
    shrink_factors = np.linspace(0.10, 1.00, 19)   # 0.10,0.15,…,1.00
    improved = True
    while improved:
        improved = False
        for frac in pref_fracs:
            k = max(1, int(n * frac))
            early = seq[:k]
            tail = seq[k:]
            sum_e = early.sum()
            sum_t = tail.sum()
            if sum_t == 0.0:
                continue
            for f in shrink_factors:
                new_early = early * f
                delta = sum_e - new_early.sum()           # mass taken from early part
                g = 1.0 + delta / sum_t                   # inflate tail uniformly
                new_tail = tail * g
                cand = np.concatenate([new_early, new_tail])
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _score(cand)
                if sc < cur_score - 1e-12:
                    seq = cand
                    cur_score = sc
                    improved = True
                    break
            if improved:
                break
    return seq, cur_score


# ------------------------------------------------------------
# Simple smoothing pass (local averaging, sum preserving)
# ------------------------------------------------------------
def _smooth_pass(seq: np.ndarray, cur_score: float, max_passes: int = 5) -> tuple:
    total = seq.sum()
    for _ in range(max_passes):
        left = np.concatenate(([0.0], seq[:-1]))
        right = np.concatenate((seq[1:], [0.0]))
        cand = 0.5 * seq + 0.25 * left + 0.25 * right
        cs = cand.sum()
        if cs > 0.0:
            cand *= total / cs
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < cur_score - 1e-12:
            seq = cand
            cur_score = sc
        else:
            break
    return seq, cur_score


# ------------------------------------------------------------
# Random hill‑climbing moves (mass transfer between two indices)
# ------------------------------------------------------------
def _hill_climb(seq: np.ndarray, cur_score: float, rng: np.random.Generator,
                steps: int) -> tuple:
    n = seq.size
    for _ in range(steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # move a modest fraction of the source entry
        max_delta = seq[src] * 0.20
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.01 * seq[src], max_delta)
        cand = seq.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < cur_score - 1e-12:
            seq = cand
            cur_score = sc
    return seq, cur_score
