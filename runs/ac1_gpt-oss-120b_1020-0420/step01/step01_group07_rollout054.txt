def full_convolve_fft(x, y):
    n = len(x) + len(y) - 1
    N = 1 << (n-1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    con = np.fft.irfft(X*Y, N)
    return con[:n]

import numpy as np

# ------------------------------------------------------------
# Helper: full convolution using FFT (real inputs)
# ------------------------------------------------------------
def _full_convolve_fft(x, y):
    """Return full convolution of real vectors x and y using FFT."""
    n = len(x) + len(y) - 1
    N = 1 << ((n - 1).bit_length())       # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv

        if p.sum() > 0:
            p = p / p.sum()
            a = 0.7 * a + 0.3 * p
            a = a / a.sum()

    best = a.copy()
    # Direct bound (scale invariant, sum = 1)
    conv_len = 2 * n - 1
    N_fft = 1 << ((conv_len - 1).bit_length())
    best_conv = _full_convolve_fft(best, best)
    best_score = 2.0 * n * best_conv.max()

    # --------------------------------------------------------
    # Sub‑gradient descent on max(convolution)
    # --------------------------------------------------------
    TARGET = 1.50274349          # early‑stop target
    for _ in range(max_iter):
        # full convolution a*a
        A = np.fft.rfft(a, N_fft)
        conv = np.fft.irfft(A * A, N_fft)[:conv_len]

        max_c = conv.max()
        cur_score = 2.0 * n * max_c

        # keep the best seen so far
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= TARGET:
                break

        # index of (one of) the maximal sums
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal quadratic form
        grad = np.zeros(n, dtype=np.float64)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
        j = k - idx
        grad[idx] = 2.0 * a[j]

        max_grad = grad.max()
        if max_grad == 0.0:
            # stagnation – random restart
            a = rng.random(n) + 1e-12
            a = a / a.sum()
            continue

        # multiplicative step (preserves positivity)
        alpha = step_coeff / max_grad            # ensures max shrink factor == exp(-step_coeff)
        a = a * np.exp(-alpha * grad)

        # project back to the simplex (sum = 1)
        total = a.sum()
        if total <= 0.0:
            a = rng.random(n) + 1e-12
            a = a / a.sum()
        else:
            a = a / total

        # occasional random jitter to escape shallow plateaus
        if rng.random() < 0.03:
            jitter = np.exp(0.1 * (rng.random(n) - 0.5))
            a = a * jitter
            a = a / a.sum()

    # --------------------------------------------------------
    # Return the best candidate found (as a plain list)
    # --------------------------------------------------------
    return [float(v) for v in best]
