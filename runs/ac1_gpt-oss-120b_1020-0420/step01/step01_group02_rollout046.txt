import numpy as np

# ------------------------------------------------------------
# Helper: half‑power series (1 - z)^(-1/2) truncation.
# a[i] = binom(2i,i) / 4**i, decreasing like 1/sqrt(i).
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

# ------------------------------------------------------------
# Helper: random multiplicative block perturbation (log‑normal factor).
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# ------------------------------------------------------------
# Compute the AC1 bound directly (scale‑invariant).
# ------------------------------------------------------------
def compute_score(arr: np.ndarray) -> float:
    n = arr.shape[0]
    b = np.convolve(arr, arr)
    max_b = float(b.max())
    sum_a = float(arr.sum())
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)
