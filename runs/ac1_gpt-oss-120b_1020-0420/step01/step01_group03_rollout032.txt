import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate the root‑constant sequence a_i = C(2i,i)/4**i for i<length."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    improve the AC1 upper bound over the classic root‑constant sequence.
    The method uses a simple projected gradient descent on the exact
    AC1 score, followed by a short random local search.
    """
    np.random.seed(seed)

    # ---------- initialise ----------
    try:
        n = int(len(height_sequence_1))
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        n = 256
        a = _root_constant_seq(n)

    # enforce evaluator bounds
    np.clip(a, 0.0, 1000.0, out=a)

    # ---------- baseline score ----------
    # evaluate via the same formula used inside evaluate_sequence
    def _score(seq):
        s = seq.sum()
        if s < 1e-8:
            return np.inf
        conv = np.convolve(seq, seq)
        m = conv.max()
        return float(2.0 * n * m / (s * s))

    best = a.copy()
    best_score = _score(best)
    if not np.isfinite(best_score):
        # safest fallback: uniform sequence with sum = sqrt(2n)
        c = np.sqrt(2.0 * n) / n
        best = np.full(n, c, dtype=np.float64)
        best_score = _score(best)

    a = best.copy()

    # ---------- projected gradient descent ----------
    max_iter = 200
    lr = 0.02  # initial learning rate
    for it in range(max_iter):
        # convolution of current iterate
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]
        S = a.sum()
        if S < 1e-8:
            break

        # constants for gradient of s = 2n*M/S^2
        const_term = 4.0 * n * M / (S ** 3)
        denom_factor = 2.0 * n / (S ** 2)

        # derivative of M wrt each a_j (approx 2*a_{k*-j})
        idx = k_star - np.arange(n)
        deriv = np.zeros_like(a)
        valid = (idx >= 0) & (idx < n)
        deriv[valid] = 2.0 * a[idx[valid]]

        grad = denom_factor * deriv - const_term

        # tentative step and projection
        a_new = a - lr * grad
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        # evaluate new candidate (manual formula, identical to evaluate_sequence)
        score_new = _score(a_new)

        if not np.isfinite(score_new):
            lr *= 0.5
            continue

        if score_new < best_score:
            best_score = score_new
            best = a_new.copy()
            a = a_new
            lr = min(lr * 1.1, 1.0)  # slowly widen step if improving
        else:
            lr *= 0.5  # shrink step on failure

        if lr < 1e-6:
            break

    # ---------- short random local search ----------
    for _ in range(30):
        cand = best.copy()
        # perturb a small random subset (≈5% of entries)
        num = max(1, n // 20)
        idxs = np.random.choice(n, size=num, replace=False)
        noise = np.random.normal(loc=0.0, scale=0.05, size=num)
        cand[idxs] *= (1.0 + noise)
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = cand

    # ---------- final validation ----------
    try:
        final = evaluate_sequence(best.tolist())
        if not np.isfinite(final):
            raise ValueError
    except Exception:
        # fallback uniform if anything went wrong
        c = np.sqrt(2.0 * n) / n
        best = np.full(n, c, dtype=np.float64)

    return best.tolist()
