import numpy as np

# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Fast AC1 upper bound using FFT (no normalization required)."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

        # ----- occasional random block perturbation ----------------------
        if (step + 1) % 1000 == 0 and rng.random() < 0.25:
            min_w = max(2, n // 200)
            max_w = max(min_w + 1, n // 50)
            width = int(rng.integers(min_w, max_w))
            start = int(rng.integers(0, n - width + 1))
            block = cur.copy()
            factor = np.exp(rng.normal(0.0, 0.2))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block /= block.sum()
            block_score = _fast_score(block)
            if block_score < cur_score:
                cur = block
                cur_score = block_score
                if block_score < best_score:
                    best = block.copy()
                    best_score = block_score

        # ----- early exit if target achieved -----------------------------
        if best_score <= target_score:
            break

    # ----- final validation with the official scorer --------------------
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # fallback to a simple uniform distribution (guaranteed finite)
        fallback = np.full_like(best, 1.0 / n)
        best = fallback

    return [float(v) for v in best]
