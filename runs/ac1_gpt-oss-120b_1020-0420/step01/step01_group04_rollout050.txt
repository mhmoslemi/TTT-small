def _move_head_to_tail(a, rng):

    n = a.size
    # choose proportion p between 0.05 and 0.2
    p = float(rng.uniform(0.05, 0.2))
    head_len = int(p * n)
    tail_len = int(p * n)
    # define block indices
    head_start = 0
    tail_start = n - tail_len
    # get sums
    head_sum = a[head_start:head_start+head_len].sum()
    tail_sum = a[tail_start:tail_start+tail_len].sum()
    if head_sum == 0:
        return a.copy()
    # choose downscale factor r_down ~ uniform(0.5,0.9)
    r_down = float(rng.uniform(0.5, 0.9))
    removed = head_sum * (1 - r_down)
    # choose epsilon factor for net increase: random small positive amount (0-0.1 of removed)
    epsilon = float(rng.uniform(0.0, 0.2))
    added_target = removed * (1 + epsilon)
    # compute up factor
    if tail_sum == 0:
        r_up = 1.0 + added_target / (tail_len * 1e-6)  # avoid zero but unrealistic
    else:
        r_up = 1.0 + added_target / tail_sum
    cand = a.copy()
    cand[head_start:head_start+head_len] *= r_down
    cand[tail_start:tail_start+tail_len] *= r_up
    # clip to max 1000
    np.clip(cand, 0, 1000, out=cand)
    return cand

    # analytic base
    analytic = _coefficients_binom_half(n)
    analytic = np.clip(analytic, 0, 1000)
    # pick best initial candidate
    best = analytic.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if 'height_sequence_1' in globals():
        parent = np.clip(parent_seq, 0, 1000)
        try:
            cscore = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(cscore) and cscore < best_score:
                best = parent.copy()
                best_score = cscore
        except Exception:
            pass

    target = 1.50274349
    # main loop
    for step in range(num_steps):
        # early exit
        if best_score < target:
            break
        # decide operation
        op = rng.integers(0, 5)
        if op == 0:
            # small bump
            cand = best.copy()
            idx = rng.integers(0, n)
            bump = max(0.0, rng.normal(0.0, 0.02))
            cand[idx] += bump
        elif op == 1:
            # head-tail move
            cand = _move_head_to_tail(best, rng)
        elif op == 2:
            # block scale up random block modestly (range)
            cand = best.copy()
            block_len = int(rng.integers(max(1, n // 200), max(2, n // 10)))
            start = rng.integers(0, n - block_len + 1)
            factor = float(np.exp(rng.normal(0.0, 0.15)))
            cand[start:start+block_len] *= factor
        elif op == 3:
            # block scale down random block (reduce)
            cand = best.copy()
            block_len = int(rng.integers(max(1, n // 200), max(2, n // 10)))
            start = rng.integers(0, n - block_len + 1)
            factor = float(np.exp(rng.normal(-0.15, 0.05)))  # <1
            cand[start:start+block_len] *= factor
        else:
            # shuffle segment
            cand = best.copy()
            seg_len = int(rng.integers(2, min(200, n // 5)))
            start = rng.integers(0, n - seg_len + 1)
            sub = cand[start:start+seg_len].copy()
            rng.shuffle(sub)
            cand[start:start+seg_len] = sub

        # clip to bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        # evaluate
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    # return as list
    return [float(x) for x in best]

import numpy as np

def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2: C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _move_head_to_tail(a, rng):
    """
    Take mass from a leading block and give it to a trailing block.
    The amount moved is random; the head block is shrunk, the tail block
    is enlarged (optionally with a net sum increase).  This tends to lower
    the autocorrelation peak while slightly raising the total sum.
    """
    n = a.shape[0]
    # proportion of the sequence used for the two blocks (5‑20 %)
    p = float(rng.uniform(0.05, 0.20))
    head_len = max(1, int(p * n))
    tail_len = max(1, int(p * n))

    head_slice = slice(0, head_len)
    tail_slice = slice(n - tail_len, n)

    head_sum = a[head_slice].sum()
    tail_sum = a[tail_slice].sum()
    if head_sum <= 0.0 or tail_sum <= 0.0:
        return a.copy()

    # shrink the head block (factor between 0.30 and 0.90)
    r_down = float(rng.uniform(0.30, 0.90))
    removed = head_sum * (1.0 - r_down)

    # decide how much of the removed mass we actually re‑inject
    # epsilon < 0 means a net loss, epsilon > 0 a net gain
    epsilon = float(rng.uniform(-0.10, 0.30))
    added_target = removed * (1.0 + epsilon)

    # enlarge the tail block; cap the factor to keep values reasonable
    r_up = 1.0 + added_target / tail_sum
    r_up = min(r_up, 5.0)          # safety cap

    cand = a.copy()
    cand[head_slice] *= r_down
    cand[tail_slice] *= r_up
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    perturbations are explored; any improvement is kept.
    """
    rng = np.random.default_rng(seed)
