import numpy as np

def _score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)

    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0 * (i+1)*(i+1))
    return a

import numpy as np

# ------------------------------------------------------------
# Helper: fast score (uses FFT for convolution)
# ------------------------------------------------------------
def _score_and_conv(seq: np.ndarray):
    """Return (score, full convolution) for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf, None
    n = seq.size
    size = 2 * n - 1
    # FFT‑based convolution (real inputs → real output)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)
    m = conv.max()
    sc = float(2.0 * n * m / (s * s))
    return sc, conv


# ------------------------------------------------------------
# Central‑binomial (Catalan‑like) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size – use the provided parent if any.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
            start = np.clip(parent, 0.0, 1000.0)
        else:
            n = 256
            start = _catalan_seq(n)
    else:
        n = 256
        start = _catalan_seq(n)

    # --------------------------------------------------------
    # Initialise the current best candidate.
    # --------------------------------------------------------
    best = start.copy()
    best_score, best_conv = _score_and_conv(best)

    # --------------------------------------------------------
    # Optional: quick scan over exponentiation of the base Catalan
    # sequence (a_i ** gamma).  This may find a better starting
    # point without any heavy optimisation.
    # --------------------------------------------------------
    base = _catalan_seq(n)
    # grid of exponents around 1 (1 gives the original bound)
    gammas = np.concatenate([
        np.linspace(0.70, 0.95, 7),   # a bit flatter
        np.linspace(1.00, 1.30, 7),   # a bit steeper
    ])
    for g in np.unique(np.round(gammas, 5)):
        cand = np.power(base, g, dtype=np.float64)
        cand = np.clip(cand, 0.0, 1000.0)
        sc, conv = _score_and_conv(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            best_conv = conv

    # --------------------------------------------------------
    # Guided hill‑climb: repeatedly move a fraction of the largest
    # entry to the smallest entry.  The move is kept only if it
    # strictly improves the AC1 bound.
    # --------------------------------------------------------
    max_steps = kwargs.get("num_steps", 1500)
    for _ in range(max_steps):
        # indices of current extremal values
        src = int(np.argmax(best))          # largest entry (typically at the front)
        dst = int(np.argmin(best))          # smallest entry (in the tail)
        if src == dst:
            break
        # move up to a modest fraction of the source mass
        delta = best[src] * rng.uniform(0.02, 0.18)
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc, conv = _score_and_conv(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            best_conv = conv

    # --------------------------------------------------------
    # Final safety: ensure we have a valid, non‑empty sequence.
    # --------------------------------------------------------
    if best.sum() < 0.01:
        # fallback to the plain Catalan sequence
        best = _catalan_seq(n)

    return [float(v) for v in best]
