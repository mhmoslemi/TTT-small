import numpy as np

# ------------------------------------------------------------
# Helper: compute the AC1 bound (same as the provided evaluator)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    a = np.clip(seq, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.size
    # convolution via FFT (full length 2*n-1)
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Propose a candidate sequence.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 1000, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size (use the provided parent if possible).
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        ):
            n = int(parent.size)
        else:
            n = 256
    else:
        n = 256

    # --------------------------------------------------------
    # Baseline candidates: parent (if present) and Catalan.
    # --------------------------------------------------------
    best = _catalan_seq(n)
    best_score = _score(best)

    if "height_sequence_1" in globals():
        cand = np.asarray(height_sequence_1, dtype=np.float64)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------
    # Random parametric families.
    # --------------------------------------------------------
    num_random = kwargs.get("num_random", 300)
    for _ in range(num_random):
        fam = rng.integers(0, 4)
        if fam == 0:  # power‑law
            alpha = rng.uniform(0.0, 0.8)
            a = (np.arange(1, n + 1, dtype=np.float64) ** -alpha)
        elif fam == 1:  # Beta‑like (zero at edges)
            p = rng.uniform(0.1, 5.0)
            q = rng.uniform(0.1, 5.0)
            x = np.linspace(0.0, 1.0, n, dtype=np.float64)
            a = (x ** p) * ((1.0 - x) ** q)
        elif fam == 2:  # Sine‑power
            power = rng.uniform(0.5, 5.0)
            a = np.sin(np.pi * (np.arange(n) + 1) / (n + 1)) ** power
        else:  # Gaussian bump
            sigma_factor = rng.uniform(0.05, 0.4)
            sigma = sigma_factor * n
            mu = (n - 1) / 2.0
            a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)

        if a.sum() < 1e-6:
            continue
        a = np.clip(a, 0.0, 1000.0)
        sc = _score(a)
        if sc < best_score:
            best = a
            best_score = sc

        # ----------------------------------------------------
        # Optional deterministic early‑prefix re‑allocation.
        # ----------------------------------------------------
        for __ in range(2):
            pref = rng.uniform(0.0, 0.5)          # proportion taken as “early”
            k = max(1, int(n * pref))
            early = a[:k]
            tail = a[k:]
            early_sum = early.sum()
            tail_sum = tail.sum()
            if tail_sum == 0.0:
                continue
            factor = rng.uniform(0.0, 1.0)        # shrink factor for the early block
            new_early = early * factor
            delta = early_sum - new_early.sum()
            inc = 1.0 + delta / tail_sum
            new_tail = tail * inc
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc2 = _score(cand)
            if sc2 < best_score:
                best = cand
                best_score = sc2

    # --------------------------------------------------------
    # Local refinement: focus on the lag where the convolution peaks.
    # --------------------------------------------------------
    hill_steps = kwargs.get("hill_steps", 1500)
    for _ in range(hill_steps):
        # compute full convolution (FFT based)
        m = 1
        while m < 2 * n - 1:
            m <<= 1
        fa = np.fft.rfft(best, m)
        conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # indices that actually contribute to the peak lag
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        src_candidates = np.where(best[i_min : i_max + 1] > 0.0)[0]
        if src_candidates.size:
            src = i_min + int(rng.choice(src_candidates))
        else:
            src = rng.integers(0, n)

        dst = rng.integers(0, n)
        if src == dst:
            continue
        delta = best[src] * rng.uniform(0.01, 0.20)
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------
    # Final small stochastic jitter (as a safety net).
    # --------------------------------------------------------
    jitter_steps = kwargs.get("jitter_steps", 500)
    for _ in range(jitter_steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        delta = best[src] * rng.uniform(0.005, 0.10)
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # --------------------------------------------------------
    return [float(v) for v in best]
