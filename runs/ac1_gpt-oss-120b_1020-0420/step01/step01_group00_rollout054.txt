def _fast_score(vec: np.ndarray, n: int, m: int) -> float:
    """Return the AC1 upper bound for a non‑negative vector (sum not required)."""
    if vec.size == 0:
        return np.inf
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
