        a = _root_constant_seq(n)
        # Evaluate baseline
        best = a.copy()

        # if best_score is already below target, can return early
        target = 1.50274349

        # We'll attempt limited attempts
        max_iters = 500  # adjust
        # Pre-calc some random indexes for moves
        # Choose early indexes (0..k) and tail indexes.
        early_idx = np.arange(min(20, n))  # first 20 indices may dominate
        tail_idx = np.arange(n - 20, n)   # last 20 indices
        # We'll repeatedly transfer epsilon from early to tail.

        for it in range(max_iters):
            # choose random early index and random tail index
            i = np.random.choice(early_idx)
            j = np.random.choice(tail_idx)

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """
    Catalan‐type sequence a_i = C(2i,i)/4**i  (root‑of‑constant).
    Satisfies (a*a)[k] = 1 for 0 ≤ k < length.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _max_convolution_fft(a: np.ndarray) -> float:
    """Full convolution max using a power‑of‑two FFT (linear memory)."""
    n = a.shape[0]
    m = 2 * n - 1
    N = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)
    return float(conv[:m].max())


def _score(a: np.ndarray) -> float:
    """Exact AC1 upper bound score for a non‑negative sequence."""
    s = a.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * a.shape[0] * _max_convolution_fft(a) / (s * s)
