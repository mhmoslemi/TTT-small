import numpy as np

# ---------- utilities ----------
def _full_conv(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Fast full convolution using FFT (real inputs)."""
    n = x.size + y.size - 1
    m = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=m)
    fy = np.fft.rfft(y, n=m)
    return np.fft.irfft(fx * fy, n=m)[:n]


def _fast_score(v: np.ndarray) -> float:
    """Exact AC1 upper bound for a non‑negative vector."""
    n = v.size
    if n == 0:
        return np.inf
    conv = _full_conv(v, v)
    max_c = conv.max()
    s = v.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ---------- main search ----------
def propose_candidate(seed: int = 42, num_iters: int = 800, **kwargs):
    """
    Gradient‑descent / simulated‑annealing search for a step‑function
    that flattens its self‑convolution.
    Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)

    # ---- initialise from the provided parent ----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.full(256, 1.0, dtype=np.float64)

    a = np.clip(parent, 0.0, 1000.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                 # keep scale‑invariant (sum ≈ 1)

    n = a.size
    len_conv = 2 * n - 1
    len_grad_full = len_conv + n - 1          # = 3n‑2
    m_conv = 1 << (len_conv - 1).bit_length()
    m_grad = 1 << (len_grad_full - 1).bit_length()

    # initial score
    cur_conv = _full_conv(a, a)
    cur_score = 2.0 * n * cur_conv.max() / (a.sum() ** 2)
    best = a.copy()
    best_score = cur_score

    # schedules
    eta0 = 0.25                # initial step size (will be normalised)
    temp0 = 0.12               # annealing temperature
    pert_prob = 0.12           # chance of a random block perturbation

    for step in range(num_iters):
        # ----- convolution of current iterate -----
        a_f = np.fft.rfft(a, n=m_conv)
        conv = np.fft.irfft(a_f * a_f, n=m_conv)[:len_conv]

        s = a.sum()
        avg = (s * s) / (2 * n - 1)           # target constant value for each entry
        excess = conv - avg
        excess_pos = excess.clip(min=0.0)     # only positive part contributes to penalty
        max_excess = excess_pos.max()
        if max_excess < 1e-12:                # already flat enough
            break

        # ----- gradient of sum_{k} (excess_k)^2 -----
        a_rev = a[::-1]
        fe = np.fft.rfft(excess_pos, n=m_grad)
        frev = np.fft.rfft(a_rev, n=m_grad)
        grad_full = np.fft.irfft(fe * frev, n=m_grad)[:len_grad_full]
        grad = 4.0 * grad_full[n - 1:n - 1 + n]   # factor 4 from derivative (see analysis)

        # normalise gradient to keep step size reasonable
        grad_norm = np.linalg.norm(grad) + 1e-16
        eta = eta0 * (1.0 - step / num_iters)

        cand = a - eta * grad / grad_norm
        cand = np.clip(cand, 0.0, 1000.0)
        sc = cand.sum()
        if sc < 1e-12:
            cand = a.copy()
            sc = cand.sum()
        cand = cand / sc

        # ----- evaluate candidate (fast) -----
        cf = np.fft.rfft(cand, n=m_conv)
        conv_c = np.fft.irfft(cf * cf, n=m_conv)[:len_conv]
        cand_score = 2.0 * n * conv_c.max() / (sc * sc)

        # ----- acceptance (simulated annealing) -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            T = temp0 * (1.0 - step / num_iters)
            if T > 0 and rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-16)):
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score

        # ----- occasional random block perturbation (exploration) -----
        if (step + 1) % 100 == 0 and rng.random() < pert_prob:
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            block = a.copy()
            factor = np.exp(rng.normal(0.0, 0.18))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block = block / block.sum()
            # fast score for the perturbed block
            bf = np.fft.rfft(block, n=m_conv)
            conv_b = np.fft.irfft(bf * bf, n=m_conv)[:len_conv]
            block_score = 2.0 * n * conv_b.max() / (block.sum() ** 2)
            if block_score < cur_score:
                a = block
                cur_score = block_score
                if cur_score < best_score:
                    best = a.copy()
                    best_score = cur_score

    # ----- final validation with the official evaluator -----
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # safe fallback – uniform distribution
        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
        best = fallback

    return [float(v) for v in best]
