            try:

            except:
                continue

                best_score = score
            continue

        except Exception:
            continue

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # keep for next iteration
    return [float(x) for x in best]

import numpy as np

def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2  (C_i = comb(2i,i)/4^i)."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _bound(a: np.ndarray) -> float:
    """Exact AC1 bound for a non‑negative vector a."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = float(conv.max())
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)


def propose_candidate(seed: int = 42, max_iter: int = 2000, **kwargs):
    """Return a height sequence with a low AC1 bound.

    The method starts from the analytic (1‑z)^‑1/2 coefficients (or the
    supplied parent) and performs directed mass‑transfer moves:
    mass is taken from positions that contribute most to the current maximal
    autocorrelation value and deposited at positions that do not affect that
    maximum.  The bound is monotone decreasing, and the best vector found is
    returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial vector (use the provided parent if present)
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if parent is not None:
        a = np.asarray(parent, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
    else:
        n = 256
        a = _coefficients_binom_half(n)
        a = np.clip(a, 0.0, 1000.0)

    n = a.shape[0]
    best = a.copy()
    best_score = _bound(best)

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    target = 1.50274349
    no_improve = 0

    for _ in range(max_iter):
        if best_score <= target:
            break

        # Compute current autocorrelation max and its location
        conv = np.convolve(best, best)
        M = float(conv.max())
        m = int(conv.argmax())
        S = float(best.sum())
        sqrt_M = np.sqrt(M)

        # Indices that actually participate in the max entry
        low_src = max(0, m - (n - 1))
        high_src = min(m, n - 1)
        src_idx = np.arange(low_src, high_src + 1, dtype=int)
        partner = m - src_idx  # partner indices inside the vector

        # Contribution weights for each source index (how much it affects the max)
        weights = best[src_idx] * best[partner]
        w_sum = weights.sum()
        if w_sum <= 0.0:
            # fallback: random block perturbation
            cand = best.copy()
            width = max(1, n // 100)

                best_score = score
                no_improve = 0

        src = int(rng.choice(src_idx, p=weights / w_sum))
