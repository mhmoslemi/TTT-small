import numpy as np

# ----------------------------------------------------------
# Fast bound evaluation using FFT (matrix‑free).
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """2 * n * max(conv(a, a)) / (sum(a) ** 2) via FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

        if sc < best_score:
            best_seq, best_score = a, sc
        a_rev = (n - i) ** (-p)
        sc = _fast_score(a_rev)
        if sc < best_score:
            best_seq, best_score = a_rev, sc

    # ------------------------------------------------------------------
    # 3) Exponential decay families.
    # ------------------------------------------------------------------
    for c in np.linspace(5e-4, 2e-2, 12):
        a = np.exp(-c * i)
        sc = _fast_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
        a_rev = np.exp(-c * (n - 1 - i))
        sc = _fast_score(a_rev)
        if sc < best_score:
            best_seq, best_score = a_rev, sc

    # ------------------------------------------------------------------
    # 4) Gaussian‑shaped families (centered).
    # ------------------------------------------------------------------
    centre = (n - 1) * 0.5
    for sf in np.linspace(0.20, 0.60, 9):
        sigma = sf * n
        a = np.exp(-((i - centre) ** 2) / (2.0 * sigma * sigma))
        sc = _fast_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc

    # ------------------------------------------------------------------
    # 5) Mixture of two power laws (α*(i+1)^-p + (1-α)*(n-i)^-q).
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.35, 0.65, 7)
    q_vals = np.linspace(0.35, 0.65, 7)
    alphas = np.linspace(0.0, 1.0, 7)
    n_i = n - i  # pre‑compute for speed
    for p in p_vals:
        term1 = (i + 1.0) ** (-p)
        rev_term1 = n_i ** (-p)
        for q in q_vals:
            term2 = n_i ** (-q)
            rev_term2 = (i + 1.0) ** (-q)
            for a_mix in alphas:
                a = a_mix * term1 + (1.0 - a_mix) * term2
                sc = _fast_score(a)
                if sc < best_score:
                    best_seq, best_score = a, sc
                a_rev = a_mix * rev_term1 + (1.0 - a_mix) * rev_term2
                sc = _fast_score(a_rev)
                if sc < best_score:
                    best_seq, best_score = a_rev, sc

    # ------------------------------------------------------------------
    # 6) Stochastic refinement: multiplicative jitter with cooling.
    # ------------------------------------------------------------------
    iters = 2000
    sigmas = np.linspace(0.10, 0.001, iters)  # exponential‑like cooling
    for sigma in sigmas:
        # multiplicative log‑normal perturbation (~sigma relative)
        perturb = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_seq * perturb
        # keep within allowed range
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # ------------------------------------------------------------------
    # Ensure the sum constraint required by the evaluator.
    # ------------------------------------------------------------------
    total = best_seq.sum()
    if total < 0.01:
        # fallback to a safe uniform vector
        best_seq = np.ones(n, dtype=np.float64)
        total = best_seq.sum()
    # Scaling does not affect the bound, but guard against overflow.
    if best_seq.max() > 1000.0:
        scale = 1000.0 / best_seq.max()
        best_seq *= scale
        total = best_seq.sum()
    # Final clipping for robustness.
    np.clip(best_seq, 0.0, 1000.0, out=best_seq)

    # ------------------------------------------------------------------
    # Return as plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
