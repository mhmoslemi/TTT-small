import numpy as np

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    the AC1 upper bound given by the pre‑imported `evaluate_sequence`.
    The routine works for any length n (e.g. the parent vector
    `height_sequence_1` of length 4806) without building large dense matrices.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------
    # Determine length n
    # --------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
        else:
            n = 256
    except NameError:
        n = 256

    # --------------------------------------------------------------------
    # Helpers for candidate evaluation
    # --------------------------------------------------------------------
    best_vec = None
    best_score = float('inf')

    def consider(vec: np.ndarray):
        """Evaluate a vector and keep it if it improves the best score."""
        nonlocal best_vec, best_score
        # avoid degenerate zero vector
        if vec.sum() < 1e-12:
            return
        # evaluate (evaluate_sequence expects a plain Python list)
        score = float(evaluate_sequence(vec.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = vec.copy()

    # --------------------------------------------------------------------
    # Basic candidates
    # --------------------------------------------------------------------
    # Uniform distribution (sum = 1)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    consider(uniform)

    # Parent vector, if present
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.ndim == 1 and hs.size == n:
            hs = np.clip(hs, 0.0, 1000.0)
            s = hs.sum()
            if s > 0:
                consider(hs / s)
    except NameError:
        pass

    # --------------------------------------------------------------------
    # Power‑law families
    # --------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)        # 1 … n
    rev_idx = n - idx + 1                               # n … 1

    # plain decreasing power law
    for beta in np.linspace(0.30, 1.20, 20):
        vec = idx ** (-beta)
        consider(vec / vec.sum())

    # symmetric ends‑heavy power law (sum of two tails)
    for beta in np.linspace(0.20, 1.50, 15):
        vec = idx ** (-beta) + rev_idx ** (-beta)
        consider(vec / vec.sum())

    # --------------------------------------------------------------------
    # U‑shaped family: inverse of product (i·(n‑i+1))
    # --------------------------------------------------------------------
    prod = idx * rev_idx                               # i·(n‑i+1)
    for beta in np.linspace(0.10, 2.00, 20):
        vec = prod ** (-beta)
        consider(vec / vec.sum())

    # --------------------------------------------------------------------
    # Random Dirichlet–like samples
    # --------------------------------------------------------------------
    for _ in range(10):
        vec = rng.gamma(shape=1.0, scale=1.0, size=n)
        consider(vec / vec.sum())

    # --------------------------------------------------------------------
    # If nothing succeeded (should not happen), fall back to uniform
    # --------------------------------------------------------------------
    if best_vec is None:
        best_vec = uniform.copy()
        best_score = float(evaluate_sequence(best_vec.tolist()))

    # --------------------------------------------------------------------
    # Local hill‑climbing refinements
    # --------------------------------------------------------------------
    for _ in range(num_steps):
        cand = best_vec.copy()

        # 1) Random multiplicative stretch on a contiguous segment
        seg_len = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - seg_len + 1)))
        stop = start + seg_len
        factor = float(np.exp(rng.normal(0.0, 0.07)))   # ≈ ± 7 % change
        cand[start:stop] *= factor

        # 2) Occasionally smooth / symmetrize
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        # 3) Occasionally move a tiny amount of mass between two points
        if rng.random() < 0.20:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.uniform(0.0, min(cand[i], 0.02))  # ≤ 2 % of source mass
            cand[i] -= eps
            cand[j] += eps

        # 4) Project back onto the probability simplex (sum = 1)
        total = cand.sum()
        if total > 0:
            cand /= total
        else:
            continue

        # 5) Clip extreme values (safety, though they stay ≤ 1)
        cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate and possibly accept
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score

    # --------------------------------------------------------------------
    # Return the best discovered vector as a plain Python list
    # --------------------------------------------------------------------
    final_vec = np.clip(best_vec, 0.0, 1000.0)
    # Ensure a tiny sum in the pathological unlikely case
    if final_vec.sum() < 1e-12:
        final_vec = np.full_like(final_vec, 1.0 / n)

    return [float(x) for x in final_vec.tolist()]
