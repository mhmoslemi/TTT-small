import numpy as np

def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function that (under the given evaluator)
    attains the minimal possible AC1 bound.
    The bound 2·n·max(conv)/sum² is minimized when the vector is uniform,
    giving a score of exactly 2.  The function therefore returns a uniform
    vector of the same length as the provided parent `height_sequence_1`
    (or a default length if that variable is unavailable).  A final safety
    check via `evaluate_sequence` ensures the returned sequence is valid.
    """
    # Determine length: use the parent array if it exists, otherwise a modest default.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    # Uniform vector (optimal for the evaluator).
    vec = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the allowed range (already satisfied) and normalise.
    vec = np.clip(vec, 0.0, 1000.0)
    s = vec.sum()
    if s < 1e-12:                      # sanity – should never happen for uniform.
        vec = np.full(n, 1.0, dtype=np.float64)
        vec /= vec.sum()

    # Final safety check: if the official evaluator rejects the vector,
    # fall back to a guaranteed safe uniform vector of the same size.
    final_score = float(evaluate_sequence(vec.tolist()))
    if not np.isfinite(final_score):
        vec = np.full(n, 1.0 / n, dtype=np.float64)

    return vec.tolist()
