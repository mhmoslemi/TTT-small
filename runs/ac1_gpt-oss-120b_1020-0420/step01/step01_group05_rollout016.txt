def _fast_score(a):
    n = a.shape[0]
    m = 2*n-1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa*fa, n=nfft)[:m]
    M = c.max()
    S = a.sum()
    return float('inf') if S < 0.01 else 2.0*n*M/(S*S)

def propose_candidate(seed: int = 42, **kwargs):

        if s < best_score:

import numpy as np

# ----------------------------------------------------------
# Fast AC1 bound evaluation using FFT (matrix‑free).
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    # FFT of a, point‑wise square gives convolution in time domain
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)
