def _score_fft(vec):
    vec = np.asarray(vec, dtype=np.float64)
    n = vec.size
    sum_a = vec.sum()
    if sum_a <= 0: return np.inf
    # Optionally we can skip scaling because scale invariance; but keep for later.
    # compute linear convolution via FFT:
    m = 1 << ((2*n-1).bit_length())  # next power of 2 >= 2n-1
    fft_len = m
    # Actually compute convolve using np.fft.rfft
    # zero pad to length fft_len
    f = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(f * f, fft_len)[:2*n-1]
    max_b = conv.max()
    # compute objective
    return 2 * n * max_b / (sum_a**2)

import numpy as np

def _fast_score(v: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector v."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return float('inf')
    s = v.sum()
    if s == 0.0:
        return float('inf')
    m = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    f = np.fft.rfft(v, m)
    conv = np.fft.irfft(f * f, m)[: 2 * n - 1]
    return float(2 * n * conv.max() / (s * s))

def _make_power(n: int, alpha: float) -> np.ndarray | None:
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** -alpha
    tot = w.sum()
    return w / tot if tot > 0 else None

def _make_exp(n: int, beta: float) -> np.ndarray | None:
    w = np.exp(-beta * np.arange(n, dtype=np.float64))
    tot = w.sum()
    return w / tot if tot > 0 else None

def _make_mix(n: int, alpha: float, beta: float, lam: float) -> np.ndarray | None:
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = lam * (idx ** -alpha) + (1.0 - lam) * np.exp(-beta * np.arange(n, dtype=np.float64))
    tot = w.sum()
    return w / tot if tot > 0 else None

def _make_twopow(n: int, split: int, a1: float, a2: float) -> np.ndarray | None:
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = np.empty(n, dtype=np.float64)
    w[:split] = idx[:split] ** -a1
    w[split:] = idx[split:] ** -a2
    tot = w.sum()
    return w / tot if tot > 0 else None

def _make_randdec(n: int, rng: np.random.Generator) -> np.ndarray | None:
    w = np.sort(rng.random(n))[::-1]   # descending random numbers
    tot = w.sum()
    return w / tot if tot > 0 else None


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 bound.  The routine starts from a parent vector,
    explores several parametric families, and finishes with a simple
    hill‑climbing refinement using FFT‑based scoring (O(n log n) per call).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.   Acquire a parent vector (fallback if unavailable)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.shape[0])
    # Normalise once – the objective is scale‑invariant
    best = parent / parent.sum()
    best_score = _fast_score(best)

    # ------------------------------------------------------------------
    # 2.   Systematic grid search over simple families
    # ------------------------------------------------------------------
    # Power‑law grid
    for a in np.linspace(0.2, 2.0, 81):
        cand = _make_power(n, a)
        if cand is None:
            continue
        sc = _fast_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # Exponential grid
    for b in np.linspace(1e-5, 5e-3, 81):
        cand = _make_exp(n, b)
        if cand is None:
            continue
        sc = _fast_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 3.   Random exploration of richer families
    # ------------------------------------------------------------------
    modes = ["pow", "exp", "mix", "twopow", "rand"]
    for _ in range(2500):
        mode = rng.choice(modes)
        if mode == "pow":
            a = rng.uniform(0.05, 3.0)
            cand = _make_power(n, a)
        elif mode == "exp":
            b = rng.uniform(1e-6, 0.02)
            cand = _make_exp(n, b)
        elif mode == "mix":
            a = rng.uniform(0.05, 3.0)
            b = rng.uniform(1e-6, 0.02)
            lam = rng.random()
            cand = _make_mix(n, a, b, lam)
        elif mode == "twopow":
            split = rng.integers(int(0.2 * n), int(0.8 * n))
            a1 = rng.uniform(0.05, 3.0)
            a2 = rng.uniform(0.05, 3.0)
            cand = _make_twopow(n, split, a1, a2)
        else:  # "rand"
            cand = _make_randdec(n, rng)

        if cand is None:
            continue
        sc = _fast_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 4.   Local hill‑climbing (multiplicative log‑normal noise)
    # ------------------------------------------------------------------
    sigma = 0.3                # initial noise strength
    for it in range(3000):
        cand = best.copy()
        if rng.random() < 0.5:
            # global multiplicative perturbation
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            cand *= noise
        else:

            block_noise = np.exp(rng.normal(0.0, sigma, size=stop - start))
            cand[start:stop] *= block_noise

        cand /= cand.sum()
        sc = _fast_score(cand)
        if sc < best_score:
            best, best_score = cand, sc
            sigma = max(0.01, sigma * 0.95)   # tighten as we improve

    # ------------------------------------------------------------------
    # 5.   Final safety check with the authoritative evaluator
    # ------------------------------------------------------------------
    try:
        final = evaluate_sequence(best.tolist())
        if np.isfinite(final):
            # keep best (fast_score already reflects shape)
            pass
        else:
            # fallback to uniform if something went wrong
            best = np.ones(n, dtype=np.float64) / n
    except Exception:
        best = np.ones(n, dtype=np.float64) / n

    return [float(v) for v in best]
