import numpy as np

def _coefficients_binom_half(n: int) -> np.ndarray:
    """Analytic (1‑z)^‑1/2 coefficients: C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _bound(seq: np.ndarray) -> float:
    """Return the AC1 upper bound for a sequence (same formula as evaluate_sequence)."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = float(np.sum(seq))
    if total < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    mx = float(np.max(conv))
    n = seq.shape[0]
    return 2.0 * n * mx / (total * total)


def propose_candidate(seed: int = 42, num_steps: int = 5000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound is low.
    The routine starts from the analytic (1‑z)^‑1/2 coefficients,
    optionally from the supplied parent, and performs a cheap hill‑climb
    that moves mass from the front of the vector to its tail.
    """

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    cand_binom = np.clip(cand_binom, 0.0, 1000.0)
