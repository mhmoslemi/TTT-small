import numpy as np
from scipy.optimize import linprog

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0 * (i+1)*(i+1))
    return a

def _score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    m = conv.max()
    n = seq.size
    return float(2.0 * n * m / (s * s))
```

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)

    else:

    # base Catalan sequence
    f = _catalan_seq(n)
    f = np.clip(f, 0.0, 1000.0)
    best_seq = f.copy()
    best_score = _score(best_seq)
    # compute M = max conv for first n elements (no need full)
    conv_f = np.convolve(f, f)
    M = conv_f[:n].max()
    # Setup intervals
    m_intervals = min(20, n)
    # Compute interval boundaries (int)
    base_len = n // m_intervals
    rem = n % m_intervals
    starts = np.empty(m_intervals, dtype=int)
    ends = np.empty(m_intervals, dtype=int)
    cur = 0
    for j in range(m_intervals):
        length = base_len + (1 if j < rem else 0)
        starts[j] = cur
        cur += length
        ends[j] = cur
    # Build A_ub matrix (n x m)
    prefix_f = np.concatenate(([0.0], np.cumsum(f)))
    A = np.empty((n, m_intervals), dtype=np.float64)
    for k in range(n):
        for j in range(m_intervals):
            lo = k - ends[j] + 1
            if lo < 0:
                lo = 0
            hi = k - starts[j]
            if hi > n-1:
                hi = n-1
            if lo <= hi:
                A[k, j] = prefix_f[hi+1] - prefix_f[lo]
            else:
                A[k, j] = 0.0
    # Linear programming: maximize sum_j h_j*len_j  => minimize negative
    obj = -(ends - starts).astype(np.float64)   # length per interval
    bounds = [(0.0, 1000.0) for _ in range(m_intervals)]
    res = linprog(c=obj, A_ub=A, b_ub=np.full(n, M), bounds=bounds, method='highs')
    if res.success:
        h = res.x
        # Build g
        g = np.empty(n, dtype=np.float64)
        for j in range(m_intervals):
            g[starts[j]:ends[j]] = h[j]
        g = np.clip(g, 0.0, 1000.0)
        # Evaluate g alone
        sc_g = _score(g)
        if sc_g < best_score:
            best_seq = g.copy()
            best_score = sc_g
        # Try convex combos
        for t in [0.2, 0.5, 0.8]:
            cand = (1 - t) * f + t * g
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best_seq = cand.copy()
                best_score = sc
    # fallback: if best_score still high, do a few random moves as before
    steps = kwargs.get("num_steps", 500)
    for _ in range(steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        delta = best_seq[src] * rng.uniform(0.01, 0.2)

        cand = best_seq.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc

def _make_intervals(n, m):
    lengths = [n // m] * m
    for i in range(n % m):
        lengths[i] += 1
    starts = np.empty(m, dtype=int)
    ends = np.empty(m, dtype=int)
    cur = 0
    for i, length in enumerate(lengths):
        starts[i] = cur
        cur += length
        ends[i] = cur
    return starts, ends, np.array(lengths)
```

- Build A_ub for given f and intervals:

```
def _build_A(f, starts, ends):
    n = f.size
    m = starts.size
    prefix = np.concatenate(([0.0], np.cumsum(f)))
    A = np.empty((n, m), dtype=np.float64)
    for k in range(n):
        for j in range(m):
            lo = k - ends[j] + 1
            if lo < 0:
                lo = 0
            hi = k - starts[j]
            if hi > n - 1:
                hi = n - 1
            if lo <= hi:
                A[k, j] = prefix[hi + 1] - prefix[lo]
            else:
                A[k, j] = 0.0
    return A
```

def _solve_lp(f, M, starts, ends, lengths):
    A = _build_A(f, starts, ends)
    obj = -lengths.astype(np.float64)  # maximize sum lengths*h
    bounds = [(0.0, 1000.0) for _ in range(len(lengths))]
    b = np.full(f.size, M, dtype=np.float64)
    res = linprog(c=obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    h = res.x
    g = np.empty(f.size, dtype=np.float64)
    for j in range(len(lengths)):
        g[starts[j]:ends[j]] = h[j]
    return g
```

import numpy as np

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    """Catalan‑like sequence a_i = C(2i,i)/4^i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


def _score(seq: np.ndarray) -> float:
    """Exact score used by the evaluator."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    m = float(conv.max())
    n = seq.size
    return 2.0 * n * m / (s * s)


# ------------------------------------------------------------
# Interval handling (used for a tiny linear program)
# ------------------------------------------------------------
def _make_intervals(n: int, m: int):
    """Return (starts, ends, lengths) for m non‑empty intervals covering 0..n."""
    base = n // m
    rem = n % m
    lengths = np.full(m, base, dtype=int)
    lengths[:rem] += 1
    starts = np.empty(m, dtype=int)
    ends = np.empty(m, dtype=int)
    cur = 0
    for i, L in enumerate(lengths):
        starts[i] = cur
        cur += L
        ends[i] = cur
    return starts, ends, lengths


def _build_A(f: np.ndarray, starts: np.ndarray, ends: np.ndarray) -> np.ndarray:
    """Linear‑convolution matrix: (A @ h)[k] = Σ_i f_i * h_{k-i}."""
    n = f.size
    m = starts.size
    pref = np.concatenate(([0.0], np.cumsum(f)))
    A = np.empty((n, m), dtype=np.float64)
    for k in range(n):
        for j in range(m):
            lo = k - ends[j] + 1
            if lo < 0:
                lo = 0
            hi = k - starts[j]
            if hi > n - 1:
                hi = n - 1
            if lo <= hi:
                A[k, j] = pref[hi + 1] - pref[lo]
            else:
                A[k, j] = 0.0
    return A


# ------------------------------------------------------------
# Tiny LP: maximise Σ h_j·len_j  subject to  f*g ≤ M
# ------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float, starts, ends, lengths):
    """Return a piecewise‑constant g (as full array) solving the LP."""
    try:
        from scipy.optimize import linprog
    except Exception:                         # SciPy unavailable – signal failure
        return None

    A = _build_A(f, starts, ends)           # shape (n, m)
    n, m = A.shape
    c = -lengths.astype(np.float64)         # maximise → minimise negative
    bounds = [(0.0, 1000.0) for _ in range(m)]
    b = np.full(n, M, dtype=np.float64)

    res = linprog(c=c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None

    h = res.x
    g = np.empty(n, dtype=np.float64)
    for j in range(m):
        g[starts[j]:ends[j]] = h[j]
    return g
