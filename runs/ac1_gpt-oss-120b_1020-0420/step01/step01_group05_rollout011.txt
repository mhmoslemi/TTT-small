def evaluate_sequence(sequence):
    if not isinstance(sequence, list): return inf
    if not sequence: return inf
    for x in sequence:
        if isinstance(x,bool) or not isinstance(x,(int,float)): return inf
        if np.isnan(x) or np.isinf(x): return inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01: return inf
    return float(2 * n * max_b / (sum_a**2))
```

import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 upper bound via FFT.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                                 # pointwise product
    c = np.fft.irfft(fc, n=nfft)[:m]            # linear convolution
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)
