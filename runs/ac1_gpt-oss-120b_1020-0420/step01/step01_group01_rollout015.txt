import numpy as np

# ------------------------------------------------------------
# Helper: fast full convolution using FFT (real inputs)
# ------------------------------------------------------------
def _full_convolution(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    # rfft for real input, then element‑wise square, then irfft
    f = np.fft.rfft(seq, m)
    conv = np.fft.irfft(f * f, m)[: 2 * n - 1]
    # remove spurious tiny negatives
    return np.maximum(conv, 0.0)


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence, used when no parent
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

def propose_candidate(seed: int = 42, num_steps: int = 300, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size – use the provided parent if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if seq.ndim != 1 or seq.size == 0:
            seq = None
    else:
        seq = None

    if seq is None:
        n = 256
        seq = _catalan_seq(n)
    else:
        n = seq.size

    # ------------------------------------------------------------------
    # Basic sanitisation and a fixed total sum (preserved during updates)
    # ------------------------------------------------------------------
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:                     # fallback to a modest uniform vector
        seq = np.full(n, 1.0, dtype=np.float64)
        total = seq.sum()
    seq = seq * (total / seq.sum())      # ensure exact total (numerical safety)

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    conv = _full_convolution(seq)
    max_b = conv.max()
    best_score = 2.0 * n * max_b / (total * total)
    best_seq = seq.copy()

    # ------------------------------------------------------------------
    # Gradient‑descent style refinement targeting the index that attains
    # the maximal convolution value.  The step size decays geometrically.
    # ------------------------------------------------------------------
    for it in range(num_steps):
        conv = _full_convolution(seq)
        max_b = conv.max()
        L = conv.argmax()

        # Build the vector a_{L‑i} with zero padding outside the range.
        neighbor = np.zeros_like(seq)
        low = max(0, L - (n - 1))
        high = min(n - 1, L)
        if low <= high:
            # seq[L‑high : L‑low+1] is increasing; reverse to align.
            chunk = seq[L - high : L - low + 1]
            neighbor[low : high + 1] = chunk[::-1]

        avg = max_b / total                     # target contribution per entry
        # geometric decay of step size – larger at the start, tiny later.
        eta = 0.20 * (0.95 ** it)

        cand = seq - eta * (neighbor - avg)
        cand = np.maximum(cand, 0.0)            # enforce non‑negativity
        # Renormalise to preserve the total sum (scale‑invariant objective)
        new_sum = cand.sum()
        if new_sum > 0:
            cand *= total / new_sum

        # Evaluate the candidate
        conv_new = _full_convolution(cand)
        max_b_new = conv_new.max()
        score_new = 2.0 * n * max_b_new / (total * total)

        # Keep the best ever seen
        if score_new < best_score:
            best_score = score_new
            best_seq = cand.copy()

        # Accept only monotone improvements (helps stability)
        if max_b_new <= max_b:
            seq = cand

    # ------------------------------------------------------------------
    # Small random mass‑transfer refinement – allows occasional
    # escapes from shallow local minima.
    # ------------------------------------------------------------------
    for _ in range(80):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        amount = seq[src] * rng.uniform(0.01, 0.15)
        if amount <= 0.0:
            continue
        cand = seq.copy()
        cand[src] -= amount
        cand[dst] += amount
        # No need to renorm: total sum unchanged.
        conv_new = _full_convolution(cand)
        max_b_new = conv_new.max()
        score_new = 2.0 * n * max_b_new / (total * total)
        if score_new < best_score:
            best_score = score_new
            best_seq = cand.copy()
            seq = cand  # keep the improvement for subsequent steps

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
