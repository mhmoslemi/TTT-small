import numpy as np

def _full_conv_max(arr):
    """Return (max_value, argmax_index, full_convolution) of arr with itself."""
    n = arr.shape[0]
    size = 2 * n - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    fa = np.fft.rfft(arr, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:size]   # autocorrelation (full)
    maxv = conv.max()
    idx = int(conv.argmax())
    return maxv, idx, conv


def _score_from_max(maxv, n):
    """Score when sum(arr) == 1."""
    return 2.0 * n * maxv


def _grad_at_k(arr, k):
    """
    Gradient of the k‑th autocorrelation entry with respect to arr.
    For a_i >= 0 the gradient is non‑negative.
    """
    n = arr.shape[0]
    g = np.zeros_like(arr)
    if k < n:
        # left (j <= k) part
        g[:k + 1] += arr[k::-1]
        # right (j >= k) part
        g[k:] += arr[: n - k]
    else:
        start = k - (n - 1)
        # only the left part exists when k >= n
        g[start:] += arr[start:][::-1]
    return g


def propose_candidate(seed: int = 42, **kwargs):
    """
    Construct a non‑negative step‑function whose AC1 upper bound
    improves on the classic root‑constant sequence.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise from the parent sequence if it exists; otherwise use a
    # simple geometric decay as a deterministic fallback.
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=np.float64)   # parent provided by harness
    except Exception:
        n_fallback = 2000
        base = np.geomspace(1.0, 0.001, n_fallback, dtype=np.float64)

    n = base.shape[0]
    a = np.maximum(base, 0.0)               # enforce non‑negativity
    total = a.sum()
    if total < 1e-12:                       # avoid degenerate zero vector
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total                        # normalise to L1‑norm = 1

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    maxv, kmax, _ = _full_conv_max(a)
    cur_score = _score_from_max(maxv, n)
    best_score = cur_score
    best_seq = a.copy()

    # ------------------------------------------------------------------
    # Sub‑gradient descent on the max‑autocorrelation (keeping sum = 1)
    # ------------------------------------------------------------------
    max_iters = 2000
    eta0 = 0.5                               # initial step size factor

    for it in range(max_iters):
        # gradient of the active lag
        g = _grad_at_k(a, kmax)

        # project gradient to a zero‑sum direction (preserves L1 norm)
        g_mean = g.mean()
        d = -g + g_mean

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)
        a_new = a + eta * d
        a_new = np.maximum(a_new, 0.0)       # enforce non‑negativity

        s = a_new.sum()
        if s <= 0.0:
            break
        a_new = a_new / s                    # renormalise to sum = 1

        # evaluate the new point
        maxv_new, kmax_new, _ = _full_conv_max(a_new)
        new_score = _score_from_max(maxv_new, n)

        # accept only non‑worsening moves (guarantees monotone descent)
        if new_score <= cur_score:
            a = a_new
            cur_score = new_score
            kmax = kmax_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()

        # early termination if the target bound is reached
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # A short random local‑search phase to potentially squeeze out a bit more.
    # ------------------------------------------------------------------
    for _ in range(300):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer a tiny amount of mass from i to j
        delta = min(0.1 * a[i], 1e-3)
        if delta <= 0.0:
            continue
        a_tmp = a.copy()
        a_tmp[i] -= delta
        a_tmp[j] += delta
        a_tmp = np.maximum(a_tmp, 0.0)
        s = a_tmp.sum()
        if s <= 0.0:
            continue
        a_tmp = a_tmp / s

        maxv_tmp, _, _ = _full_conv_max(a_tmp)
        score_tmp = _score_from_max(maxv_tmp, n)
        if score_tmp < best_score:
            best_score = score_tmp
            best_seq = a_tmp.copy()
            a = a_tmp                      # continue searching from the improved point

        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found, respecting the evaluator's bounds.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
