    f = np.fft.rfft(seq, size)

def ac1_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    # zero-pad to length power of two >= 2*n
    size = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(a, size)
    conv = np.fft.irfft(f * f, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:  # avoid division
        return np.inf
    return 2 * n * max_b / (sum_a * sum_a)
```

import numpy as np
import math

def generate_half_power_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def max_convolution_fft(arr):
    n = arr.size
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    return np.max(conv)

def ac1_score(arr):
    arr = np.asarray(arr, dtype=np.float64)
    n = arr.size
    max_b = max_convolution_fft(arr)
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def perturb_block(arr, rng, max_width):
    n = arr.size
    if n == 1:
        return arr.copy()
    # Choose block width
    if rng.random() < 0.7:
        width = 1
    else:
        width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    # Log‑normal factor
    sigma = 0.12
    factor = float(math.exp(rng.normal(0.0, sigma)))
    cand = arr.copy()
    cand[start:start+width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed: int = 42, max_iters: int = 5000, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine length and initial vector
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, np.ndarray):
        base = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        base = generate_half_power_series(n)

    np.clip(base, 0.0, 1000.0, out=base)

    # Simple parametric family exploration
    n = base.size
    best = base
    best_score = ac1_score(best)

    # Try a handful of parameterised shapes (power law + sinusoid)
    for _ in range(200):
        p = rng.uniform(0.35, 0.75)          # exponent
        eps = rng.uniform(0.0, 0.25)         # modulation amplitude
        period_frac = rng.uniform(0.02, 0.4) # fraction of length
        period = max(2, int(n * period_frac))
        i = np.arange(n, dtype=np.float64)
        shape = (i + 1) ** (-p)
        if eps > 0:
            shape *= (1.0 + eps * np.sin(2.0 * math.pi * i / period))
        shape = np.maximum(shape, 0.0)
        np.clip(shape, 0.0, 1000.0, out=shape)
        sc = ac1_score(shape)
        if sc < best_score:
            best = shape
            best_score = sc

    # Hill‑climbing / simulated‑annealing refinement
    cur = best.copy()
    cur_score = best_score
    max_width = max(1, n // 10)

    temperature = 0.5
    decay = 0.9997

    for it in range(max_iters):
        cand = perturb_block(cur, rng, max_width)
        cand_score = ac1_score(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
        else:
            # occasional uphill move
            prob = math.exp(-(cand_score - cur_score) / temperature) if temperature > 0 else 0.0
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score

        temperature *= decay
        # optional early stop if temperature small and no improvement
        if it % 500 == 0 and best_score < 1.4:
            break

    # Final safety clip and conversion to list
    np.clip(best, 0.0, 1000.0, out=best)
    return [float(v) for v in best]

import numpy as np
import math

# ------------------------------------------------------------
# Basic building blocks
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """(1‑z)^(-1/2) coefficients a_i = binom(2i,i)/4**i,  i>=0."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def max_convolution_fft(arr: np.ndarray) -> float:
    """Linear convolution of arr with itself, return the maximum entry."""
    n = arr.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(np.max(conv))


def ac1_score(arr: np.ndarray) -> float:
    """Exact AC1 upper‑bound used by the evaluator (scale‑invariant)."""
    arr = np.asarray(arr, dtype=np.float64)
    n = arr.size
    sum_a = float(np.sum(arr))
    if sum_a < 0.01:
        return float('inf')
    max_b = max_convolution_fft(arr)
    return 2.0 * n * max_b / (sum_a * sum_a)


# ------------------------------------------------------------
# Simple stochastic perturbations (block‑wise log‑normal scaling)
# ------------------------------------------------------------
def perturb_block(arr: np.ndarray, rng: np.random.Generator, max_width: int) -> np.ndarray:
    n = arr.size
    if n == 1:
        return arr.copy()
    # most moves touch a single entry – occasionally a larger block
    width = 1 if rng.random() < 0.7 else int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(math.exp(rng.normal(0.0, 0.12)))      # log‑normal factor ~[0.7,1.4]
    cand = arr.copy()
    cand[start : start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Structured scaling: shrink an early segment, enlarge the tail
# ------------------------------------------------------------
def split_scale_candidate(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = arr.size
    # split somewhere away from the boundaries
    split = int(rng.integers(n // 10, 9 * n // 10 + 1))
    early_factor = float(rng.uniform(0.2, 1.0))          # shrink early part
    total = float(np.sum(arr))
    early_sum = float(np.sum(arr[:split]))
    tail_sum = total - early_sum
    if tail_sum <= 0.0:
        return arr.copy()
    # keep total sum unchanged (approximately) by expanding the tail
    new_early_sum = early_factor * early_sum
    new_tail_sum = total - new_early_sum
    tail_factor = new_tail_sum / tail_sum
    cand = arr.copy()
    cand[:split] *= early_factor
    cand[split:] *= tail_factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iters: int = 5000, **kwargs):
    """
    Return a list of non‑negative heights that aims to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the parent construction if it exists; otherwise use
    # the analytic half‑power series (length 256 is a safe fallback).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):
        base = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        base = generate_half_power_series(256)

    np.clip(base, 0.0, 1000.0, out=base)

    n = base.size
    best = base.copy()
    best_score = ac1_score(best)

    # ------------------------------------------------------------------
    # 1) Explore a modest family of analytic shapes (power law + sinusoid)
    # ------------------------------------------------------------------
    for _ in range(300):
        p = rng.uniform(0.35, 0.75)                       # exponent of decay
        eps = rng.uniform(0.0, 0.25)                      # sinusoid amplitude
        per_frac = rng.uniform(0.02, 0.4)                 # period as fraction of n
        period = max(2, int(n * per_frac))
        i = np.arange(n, dtype=np.float64)
        shape = (i + 1.0) ** (-p)
        if eps > 0.0:
            shape *= 1.0 + eps * np.sin(2.0 * math.pi * i / period)
        np.clip(shape, 0.0, 1000.0, out=shape)
        sc = ac1_score(shape)
        if sc < best_score:
            best = shape
            best_score = sc

    # ------------------------------------------------------------------
    # 2) Structured split‑scale candidates (early shrink / tail expand)
    # ------------------------------------------------------------------
    for _ in range(200):
        cand = split_scale_candidate(base, rng)
        sc = ac1_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 3) Dedicated hill‑climbing / simulated‑annealing refinement
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_score = best_score
    max_width = max(1, n // 10)

    temperature = 0.5
    decay = 0.9997

    for it in range(max_iters):
        # propose a block‑scaled perturbation
        cand = perturb_block(cur, rng, max_width)
        cand_score = ac1_score(cand)

        if cand_score < cur_score:
            # always accept improvement
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand, cand_score
        else:
            # occasional uphill move (simulated annealing)
            prob = math.exp(-(cand_score - cur_score) / temperature) if temperature > 0 else 0.0
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        temperature *= decay

        # early stop if we already beat the target comfortably
        if best_score < 1.4:
            break

    # ------------------------------------------------------------------
    # Final safety clamp and conversion to plain Python list
    # ------------------------------------------------------------------
    np.clip(best, 0.0, 1000.0, out=best)
    return [float(v) for v in best]
