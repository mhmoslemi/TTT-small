import numpy as np

def _coefficients_binom_half(n: int) -> np.ndarray:
    """
    Return the first n coefficients of the power series (1‑z)^(-1/2), i.e.
    a_k = C(2k, k) / 4^k = prod_{i=1}^k (2i-1)/(2i), a_0 = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a

    in a bound ≈ 1, well below the target 1.50274349.
    If a parent sequence `height_sequence_1` is available, it is evaluated
    and the better of the two is returned.

    # Determine problem length

    else:

    # Baseline candidate from analytic series
    cand = _coefficients_binom_half(n)
