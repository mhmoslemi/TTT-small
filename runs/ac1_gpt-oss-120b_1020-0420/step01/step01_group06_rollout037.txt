import numpy as np

def _max_convolution(v: np.ndarray, n: int) -> float:
    """Return max_{k} (v*v)[k] using a real FFT (size power of two)."""
    sz = 1 << ((2 * n - 1).bit_length())   # next power‑of‑2 ≥ 2n‑1
    fv = np.fft.rfft(v, n=sz)
    conv = np.fft.irfft(fv * fv, n=sz)[: 2 * n - 1]
    # tiny negative values can appear from round‑off; ignore them
    return float(np.max(conv))

def _score(v: np.ndarray, n: int) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    s = float(v.sum())
    if s < 0.01:
        return float("inf")
    maxb = _max_convolution(v, n)
    return 2.0 * n * maxb / (s * s)


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 upper bound
    is hoped to be below the target 1.5027.  The method builds a few analytic
    seeds and then runs a cheap hill‑climbing search that mutates contiguous
    blocks using a log‑normal factor.  Scoring is performed with an
    FFT‑based convolution, so thousands of candidates can be examined quickly.
    """
    # ------------------------------------------------------------------
    # 1.  Determine length and a sensible starting vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
        else:
            parent = None
    else:
        parent = None

    if parent is None or parent.size == 0:
        n = 256
        parent = np.ones(n, dtype=np.float64) / n
    else:
        n = int(parent.size)

    # ------------------------------------------------------------------
    # 2.  Helper to evaluate and keep the best candidate seen so far
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = _score(best, n)

    def _maybe_update(candidate: np.ndarray):
        nonlocal best, best_score
        sc = _score(candidate, n)
        if np.isfinite(sc) and sc < best_score:
            best = candidate.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 3.  Analytic seed families
    # ------------------------------------------------------------------
    # (a) Power‑law a_i ∝ (i+1)^{-α}
    idx = np.arange(1, n + 1, dtype=np.float64)
    for α in (0.10, 0.15, 0.20, 0.25, 0.30, 0.35):
        w = np.power(idx, -α)
        w /= w.sum()
        _maybe_update(w)

    # (b) Exponential a_i ∝ exp(-β·i)
    for β in (5e-5, 1e-4, 2e-4, 5e-4, 1e-3):
        w = np.exp(-β * np.arange(n, dtype=np.float64))
        w /= w.sum()
        _maybe_update(w)

    # (c) Linear “triangular” shape a_i ∝ (n-i)
    tri = (n - np.arange(n, dtype=np.float64))
    tri = np.clip(tri, 0, None)
    tri /= tri.sum()
    _maybe_update(tri)

    # (d) Difference‑of‑square‑roots shape a_i = sqrt(i+1)-sqrt(i)
    sq = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    sqm1 = np.sqrt(np.arange(0, n, dtype=np.float64))
    diff = sq - sqm1
    diff /= diff.sum()
    _maybe_update(diff)

    # ------------------------------------------------------------------
    # 4.  Hill‑climbing search (log‑normal block perturbations)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    MAX_ITER = 2000                       # enough iterations for n≈5k
    max_block = max(2, n // 20)           # block length ≤ 5 % of n
    sigma_start = 0.5                     # initial log‑normal spread
    sigma_end = 0.03                      # final spread

    for it in range(MAX_ITER):
        # slowly anneal the perturbation magnitude
        sigma = sigma_start * (1.0 - it / MAX_ITER) + sigma_end
        cand = best.copy()

        # random contiguous block
        blk_len = rng.integers(1, max_block + 1)
        start = rng.integers(0, n - blk_len + 1)
        factor = np.exp(rng.normal(0.0, sigma))
        cand[start:start + blk_len] *= factor

        # keep values inside the allowed range
        np.clip(cand, 0.0, 1000.0, out=cand)

        # normalise sum – this does not affect the objective (scale‑invariant)
        total = cand.sum()
        if total > 0.0:
            cand /= total

        # evaluate
        sc = _score(cand, n)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

        # occasional re‑seed with a fresh power‑law to escape plateaus
        if (it + 1) % 500 == 0:
            α = rng.uniform(0.08, 0.40)
            w = np.power(idx, -α)
            w /= w.sum()
            _maybe_update(w)

    # ------------------------------------------------------------------
    # 5.  Return the best vector found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
