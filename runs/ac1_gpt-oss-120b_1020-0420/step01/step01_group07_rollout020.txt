def evaluate_sequence(sequence: list) -> float:
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf

import numpy as np

# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 500, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    If a parent construction `height_sequence_1` exists it is used as a
    starting point; otherwise a uniform vector is used.
    A tiny stochastic hill‑climb refines the candidate.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Initialise from the parent construction if it is present
    # ----------------------------------------------------------------
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n = int(seq.shape[0])
    except Exception:
        n = 256
        seq = np.ones(n, dtype=np.float64)

    # Ensure feasibility with respect to the evaluator's checks
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.sum() < 0.01:                     # avoid the sum‑zero rejection
        seq[0] = max(seq[0], 0.01)

    best_vec = seq.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))
    TARGET = 1.50274349

    # If the parent already satisfies the target we can stop early
    if best_score <= TARGET:
        return best_vec.tolist()

    # ----------------------------------------------------------------
    # Tiny stochastic hill‑climb – local perturbations
    # ----------------------------------------------------------------
    for _ in range(num_steps):
        # pick a random contiguous window and perturb its heights
        width = int(rng.integers(1, max(2, n // 200)))   # modest window size
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        cand = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.05)))    # ≤ 5 % multiplicative change
        cand[start:stop] *= factor

        # occasional mild smoothing (mirroring) – helps escape plateaus
        if rng.random() < 0.1:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:            # maintain feasibility
            cand[0] = max(cand[0], 0.01)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                break

    # ----------------------------------------------------------------
    # Simple power‑law fallback (cheap cheap) – only if needed
    # ----------------------------------------------------------------
    if best_score > TARGET:
        p_vals = np.linspace(0.45, 0.80, 71)  # same grid as the reference program
        base = np.arange(1, n + 1, dtype=np.float64)
        for p in p_vals:
            cand = base ** (-p)
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_vec = cand
                best_score = score
                if best_score <= TARGET:
                    break

    # ----------------------------------------------------------------
    # Return the best-found candidate as a plain Python list
    # ----------------------------------------------------------------
    return best_vec.tolist()
