def _local_optimize(seq, rng, steps):
    best = seq.copy()
    best_score = _score(best)
    n = best.size

        cand = _random_move(best, rng)

            best = cand
            best_score = sc
            continue
        # try reduction move

            best = cand2
            best_score = sc2

import numpy as np

# ----------------------------------------------------------------------
# Scoring function (identical to the reference evaluator)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Compute the AC1 bound for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)            # full convolution, FFT based
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))

# Simple analytic families

def _catalan_seq(n: int) -> np.ndarray:
    """Central‑binomial (Catalan‑like) sequence a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


def _alpha_seq(n: int, alpha: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)^{-alpha}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-alpha)

def _random_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:

    src = rng.integers(0, n)
    dst = rng.integers(0, n)

        dst = rng.integers(0, n)

        return seq  # nothing to move
    delta = seq[src] * rng.uniform(0.01, 0.20)
    cand = seq.copy()
    cand[src] -= delta
    cand[dst] += delta
    return cand
