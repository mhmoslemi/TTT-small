import numpy as np

def _fast_score(a: np.ndarray) -> float:
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

import numpy as np

# ----------------------------------------------------------
# Fast (FFT‑based) evaluation of the AC1 bound.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a, a)) / (sum(a) ** 2) using FFT."""
    n = int(a.shape[0])
    if n == 0:
        return float('inf')
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                                 # pointwise product
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The construction tries a compact exponential family (and a few variants),
    compares against the supplied parent (height_sequence_1) and a uniform fallback,
    and performs a tiny random refinement.
    """
    # ------------------------------------------------------------------
    # Obtain length n from the runtime parent if possible.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)   # safe fallback
    n = parent.size
    parent = np.clip(parent, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Baseline candidates: parent and uniform.
    # ------------------------------------------------------------------
    best_seq = parent.copy()
    best_score = _fast_score(best_seq)

    uniform = np.ones(n, dtype=np.float64)
    uscore = _fast_score(uniform)
    if uscore < best_score:
        best_seq, best_score = uniform, uscore

    # ------------------------------------------------------------------
    # Exponential families a_i = exp(lam * i/(n-1)).
    # lam is varied to explore a few shapes (full range, 75%, 50%, 25%).
    # ------------------------------------------------------------------
    if n == 1:
        # trivial case – any positive number works
        return [float(best_seq[0])]

    idx = np.arange(n, dtype=np.float64)
    lam_full = np.log(1000.0)                     # makes a_{n‑1}=1000
    for factor in (1.0, 0.75, 0.5, 0.25):
        lam = lam_full * factor
        # increasing exponential
        cand = np.exp(lam * idx / (n - 1))
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

        # decreasing exponential (reversal) – mathematically identical bound,
        # but evaluated explicitly for robustness.
        cand_rev = np.exp(lam * (n - 1 - idx) / (n - 1))
        cand_rev = np.clip(cand_rev, 0.0, 1000.0)
        sc_rev = _fast_score(cand_rev)
        if sc_rev < best_score:
            best_seq, best_score = cand_rev, sc_rev

    # ------------------------------------------------------------------
    # Tiny random multiplicative refinement around the current best.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for _ in range(100):
        perturb = np.exp(rng.normal(0.0, 0.02, size=n))   # ~2 % jitter
        cand = best_seq * perturb
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
