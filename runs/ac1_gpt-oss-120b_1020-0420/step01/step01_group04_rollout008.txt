import numpy as np

def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2 : C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _full_conv_fft(a):
    """Full convolution a*a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    # tiny negative values may appear because of round‑off
    conv = np.maximum(conv, 0.0)
    return conv


def _evaluate(a):
    """Return (bound, full_convolution) for a non‑negative sequence a."""
    conv = _full_conv_fft(a)
    cmax = float(conv.max())
    s = float(a.sum())
    if s < 0.01:
        return np.inf, conv
    n = a.shape[0]
    bound = 2.0 * n * cmax / (s * s)
    return bound, conv

            else:
                if rng.random() < np.exp(-(cand_score - best_score) / temp):
                    best_seq, best_score, best_conv = cand, cand_score, cand_conv

    # final safety clipping
    np.clip(best_seq, 0.0, 1000.0, out=best_seq)
    if best_seq.sum() < 0.01 and 'height_sequence_1' in globals():
        best_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    return [float(x) for x in best_seq]
