import numpy as np

# --------------------------------------------------------------
# Fast evaluation of the AC1 bound (identical to the official one)
# --------------------------------------------------------------
def _fast_score(v: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector."""
    n = v.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())           # FFT length (next power of two)
    fa = np.fft.rfft(v, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = float(conv.max())
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# --------------------------------------------------------------
# Main search routine
# --------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     num_steps: int = 25000,
                     **kwargs):
    """Return a list of non‑negative heights that (approximately) minimise the AC1 bound."""
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)  # tiny fallback if no parent is present

    cur = np.clip(parent, 0.0, 1000.0)
    if cur.sum() < 1e-12:
        cur[:] = 1.0
    cur = cur / cur.sum()                       # normalise to sum = 1

    # tiny random kick to move off the uniform stationary point
    cur = cur * np.exp(rng.normal(0.0, 0.05, size=cur.shape))
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur / cur.sum()

    n = cur.size
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    # Adam‑style moments (projected onto the simplex later)
    m1 = np.zeros_like(cur)
    m2 = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # schedule constants
    init_eta = 0.12
    init_T   = 0.12

    # pre‑compute FFT length (constant throughout the run)
    fft_len = 1 << ((2 * n - 1).bit_length())

    for step in range(num_steps):
        # -------- convolution & peak index --------------------------------
        fa = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())

        # -------- sub‑gradient of the peak (zero‑sum) --------------------
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        b = np.zeros_like(cur)
        b[mask] = cur[idx[mask]]
        grad = 2.0 * (b - b.mean())               # ∑ grad = 0  (keeps the sum constant)

        # -------- Adam update (step‑size decays linearly) -----------------
        t += 1
        eta = init_eta * (1.0 - step / num_steps)   # linear cooling of the base step
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)
        step_vec = eta * m1_hat / (np.sqrt(m2_hat) + eps)

        cand = cur - step_vec

        if s < 1e-12:

            s = cand.sum()
        cand = cand / s                               # re‑project onto the simplex

        else:
