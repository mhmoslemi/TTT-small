    # Score function

    # Parameter families

        w /= w.sum()
        sc = _score(w)
        if sc < best_score:

            best_score = sc
    # 2. exponential decay
    betas = rng.uniform(1e-5, 0.02, size=200)
    for b in betas:
        w = np.exp(-b * np.arange(n, dtype=np.float64))
        w /= w.sum()
        sc = _score(w)
        if sc < best_score:
            best = w.copy()
            best_score = sc
    # 3. combined power*exp
    # sample p in [-3,3] and beta in range
    for _ in range(150):
        p = rng.uniform(-3.0, 3.0)
        beta = rng.uniform(1e-5, 0.02)
        w = np.power(indices, p) * np.exp(-beta * np.arange(n, dtype=np.float64))
        # set negative values to zero, because power with negative may cause huge, but still positive
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s == 0:
            continue
        w /= s
        sc = _score(w)
        if sc < best_score:
            best = w.copy()
            best_score = sc
    # Local refinement
    max_iters = 2000
    for it in range(max_iters):
        cand = best.copy()
        op = rng.integers(0, 4)
        if op == 0:
            # block multiplicative noise
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 200))
            stop = min(n, start + width)
            noise = np.exp(rng.normal(0.0, 0.05, size=stop - start))
            cand[start:stop] *= noise
        elif op == 1:
            # swap small fraction from high to low

            if cand[i_max] > 0:

        if sc < best_score:
            best = cand.copy()
            best_score = sc
