import numpy as np

def _evaluate(vec: np.ndarray) -> float:
    """
    Helper that forwards a NumPy array to the authority evaluator.
    Returns a Python float; the evaluator clips and sanitises the values.
    """
    return float(evaluate_sequence(vec.tolist()))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that minimises the
    AC1 upper bound as measured by `evaluate_sequence`.

    Strategy
    --------
    * Use the supplied `height_sequence_1` as a possible parent if it exists.
    * Generate a few simple analytic families (uniform, power‑law, exponential)
      normalised to unit ℓ¹‑mass.
    * Evaluate all candidates with the exact evaluator and return the best.
    * The algorithm runs in O(n) memory and O(k·n) time where k is the number
      of trial families (tiny, well below the limits).
    """
    # ------------------------------------------------------------------ #
    # 1. Determine length n and initialise a parent candidate (if any).   #
    # ------------------------------------------------------------------ #
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            tmp.ndim == 1
            and tmp.size > 0
            and np.isfinite(tmp).all()
        ):
            # the evaluator also clips to [0,1000]; we do it here as well.
            parent = np.clip(tmp, 0.0, 1000.0)

    # fallback length when no parent is available
    n = parent.size if parent is not None else 256

    # ------------------------------------------------------------------ #
    # 2. Build a short list of analytically generated candidates.         #
    # ------------------------------------------------------------------ #
    candidates = []

    # (a) Uniform distribution – the provably optimal ℓ²‑norm for a fixed ℓ¹‑mass.
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    candidates.append(uniform)

    # (b) Parent sequence, if present.
    if parent is not None:
        candidates.append(parent)

    # (c) Power‑law families a_i ∝ (i+1)^{-α} for a few exponents.
    indices = np.arange(1, n + 1, dtype=np.float64)
    for alpha in (0.3, 0.5, 0.7):
        w = np.power(indices, -alpha)
        s = w.sum()
        if s > 0.0:
            candidates.append(w / s)

    # (d) Simple exponential decays a_i ∝ exp(‑β·i) for a few rates.
    for beta in (1e-3, 5e-3, 1e-2):
        w = np.exp(-beta * np.arange(n, dtype=np.float64))
        s = w.sum()
        if s > 0.0:
            candidates.append(w / s)

    # ------------------------------------------------------------------ #
    # 3. Evaluate all candidates and keep the best (lowest score).        #
    # ------------------------------------------------------------------ #
    best_vec = None
    best_score = np.inf
    for cand in candidates:
        sc = _evaluate(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_vec = cand

    # ------------------------------------------------------------------ #
    # 4. Return a plain Python list of floats.                            #
    # ------------------------------------------------------------------ #
    if best_vec is None:
        # Very unlikely – fall back to uniform.
        best_vec = uniform
    return [float(v) for v in best_vec]
