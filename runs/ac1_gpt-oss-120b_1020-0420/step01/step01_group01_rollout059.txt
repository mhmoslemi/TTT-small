    n = seq.size
    size = 1 << ((2*n-1).bit_length())
    # zero padding
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)

    conv_full = conv[:2*n-1]
    return conv_full

import numpy as np

# ------------------------------------------------------------
# Helper: Catalan‑like central binomial sequence (a_i = C(2i,i)/4^i)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Linear‑time convolution via real FFT (exact for our sizes)
# ------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    n = seq.size
    # smallest power of two ≥ 2·n‑1
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)
    return conv[: 2 * n - 1]


# ------------------------------------------------------------
# Score identical to the evaluator (2·n·max(conv)/sum²)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_fft(seq)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))

        delta = early.sum() - new_val.sum()
        cand = seq.copy()
        cand[0] = new_val[0]
        cand[-1] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            seq = cand
            best_seq = cand
            best_score = sc
            early = new_val

    # --------------------------------------------------------
    # Stochastic hill‑climbing: move a random fraction of mass from an
    # index that heavily participates in the current maximal convolution
    # to a location that contributes little to that same convolution.
    # --------------------------------------------------------
    steps = int(kwargs.get("num_steps", num_steps))
    seq = best_seq.copy()

    for _ in range(steps):
        conv = _conv_fft(seq)
        M = conv.max()
        k_max = int(conv.argmax())

        # Indices that appear in the maximal convolution term
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k_max - i_vals
        contributions = seq[i_vals] * seq[j_vals]
        tot = contributions.sum()
        if tot <= 0.0:
            break

        # Source: pick proportionally to its contribution
        src = rng.choice(i_vals, p=contributions / tot)

        # Destination: try a few random candidates and keep the one
        # with the smallest contribution to the same convolution index.
        best_dst = None
        best_contrib = None
        for _ in range(5):
            dst = rng.integers(0, n)
            if dst == src:
                continue
            if 0 <= k_max - dst < n:
                contrib = seq[dst] * seq[k_max - dst]
            else:
                contrib = 0.0
            if (best_contrib is None) or (contrib < best_contrib):
                best_dst = dst
                best_contrib = contrib
        if best_dst is None:
            continue

        # Move up to 20 % of the source entry
        max_delta = seq[src] * 0.20
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)
        if delta <= 0.0:
            continue

        cand = seq.copy()
        cand[src] -= delta
        cand[best_dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)

        if sc < best_score:
            seq = cand
            best_seq = cand
            best_score = sc
            # No need to recompute conv here – the loop starts anew.

    # --------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # --------------------------------------------------------
    return [float(v) for v in best_seq]
