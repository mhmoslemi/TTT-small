import numpy as np

def _score(arr):
    """Return the AC1 upper bound for a 1‑D sequence."""
    # `evaluate_sequence` expects a list of Python floats.
    return float(evaluate_sequence(arr.tolist()))

def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) whose
    AC1 upper bound is (hopefully) below the target ~1.5027.
    The routine works in three phases:
      1. Initialise from the provided `height_sequence_1` or a uniform fallback.
      2. Scan a family of power‑law / exponential shapes and keep the best.
      3. Perform cheap stochastic local search to fine‑tune the best candidate.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise candidate from the harness (if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            a = None
    else:
        a = None

    if a is None:
        # generic fallback – length 5582 (the length of the usual parent)
        n = 5582
        a = np.ones(n, dtype=np.float64)
    else:
        n = a.size

    # Normalise to unit L¹ mass (the score is scale invariant)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.ones(n, dtype=np.float64) / n

    best = a.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 2.  Scan parametrised families (power‑law, exponential, mixed)
    # ------------------------------------------------------------------
    indices = np.arange(1, n + 1, dtype=np.float64)   # 1‑based for power law

    # ---- coarse power‑law scan ------------------------------------------------
    alphas = np.linspace(0.10, 3.00, 121)   # step ≈0.024
    for alpha in alphas:
        cand = np.power(indices, -alpha)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ---- mixed power‑law × exponential scan ----------------------------------
    betas = np.linspace(0.0, 0.010, 21)    # up to 1 % decay per step
    for alpha in np.linspace(0.10, 3.00, 61):
        for beta in betas:
            cand = np.power(indices, -alpha) * np.exp(-beta * (indices - 1.0))
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                continue
            cand /= cand_sum
            sc = _score(cand)
            if sc < best_score:
                best = cand.copy()
                best_score = sc

    # ---- random draws in the same parameter space (exploration) ---------------
    for _ in range(200):
        alpha = rng.uniform(0.05, 3.5)
        beta = rng.uniform(0.0, 0.015)
        cand = np.power(indices, -alpha) * np.exp(-beta * (indices - 1.0))
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 3.  Stochastic local search (budgeted)
    # ------------------------------------------------------------------
    max_iter = 1500
    no_improve = 0
    for it in range(max_iter):
        # Choose a random neighbourhood operation
        op = rng.integers(0, 4)        # 0: transfer, 1: block mult, 2: block avg, 3: swap
        cand = best.copy()

        if op == 0:
            # Transfer a fraction of mass from i to j
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            eps = rng.uniform(0.0, 0.5) * cand[i]   # up to 50 % of source
            cand[i] -= eps
            cand[j] += eps

        elif op == 1:
            # Multiplicative log‑normal noise on a short block
            start = rng.integers(0, n)
            length = rng.integers(1, max(2, n // 250))
            stop = min(n, start + length)
            factor = np.exp(rng.normal(0.0, 0.08))
            cand[start:stop] *= factor

        elif op == 2:
            # Replace a block by its average (smoothing)
            start = rng.integers(0, n)
            length = rng.integers(2, max(3, n // 200))
            stop = min(n, start + length)
            avg = cand[start:stop].mean()
            cand[start:stop] = avg

        else:  # op == 3
            # Swap two entries (preserves L¹ mass)
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            cand[i], cand[j] = cand[j], cand[i]

        # Enforce feasibility: non‑negative, clamp to 0‑1000
        np.clip(cand, 0.0, 1000.0, out=cand)
        # Re‑normalise L¹ mass to keep scaling invariant
        s = cand.sum()
        if s <= 0.0:

        cand /= s

        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 400:
                # stagnation – stop early
                break

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
