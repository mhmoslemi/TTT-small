    if total == 0:
        return v
    return v / total

def _score_vec(v):
    # assume v nonnegative
    n = v.size
    conv = np.convolve(v, v)
    max_b = conv.max()
    sum_a = v.sum()
    if sum_a == 0:
        return math.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _perturb_pair(v, rng):
    n = v.size
    i = rng.integers(0, n)
    # avoid self
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    if v[i] <= 0:
        return v.copy()
    delta = rng.random() * 0.2 * v[i]  # up to 20% of mass at i
    new = v.copy()
    new[i] -= delta
    new[j] += delta
    return new

def _perturb_block(v, rng):

    start = rng.integers(0, n)
    width = rng.integers(1, max(2, n // 200))

import numpy as np
import math
import time


def _normalize(v: np.ndarray) -> np.ndarray:
    total = v.sum()
    if total == 0.0:
        return v
    return v / total


def _score_vec(v: np.ndarray) -> float:
    """Exact AC1 upper‑bound score for a non‑negative vector."""
    n = v.size
    conv = np.convolve(v, v)
    max_b = conv.max()
    s = v.sum()
    if s == 0.0:
        return math.inf
    return 2.0 * n * max_b / (s * s)


def _perturb_pair(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = v.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    if v[i] <= 0.0:
        return v.copy()
    delta = rng.random() * 0.2 * v[i]          # up to 20 % of the source mass
    new = v.copy()
    new[i] -= delta
    new[j] += delta
    return new


def _perturb_block(v: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal noise on a short contiguous block."""
    n = v.size
    start = rng.integers(0, n)
    width = rng.integers(1, max(2, n // 200))
    stop = min(n, start + width)

        return v.copy()
    new = v.copy()
    noise = np.exp(rng.normal(0.0, 0.1, size=stop - start))
    new[start:stop] *= noise
    s = new.sum()
    if s > 0.0:
        new = new * (v.sum() / s)                # keep total mass unchanged
    return new
