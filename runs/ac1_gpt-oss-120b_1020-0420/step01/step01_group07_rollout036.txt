def compute_cnt(selected, n):
    idx = np.array(list(selected), dtype=np.int64)
    sums = idx[:,None] + idx[None,:]   # shape (m,m)
    cnt = np.bincount(sums.ravel(), minlength=2*n-1)
    return cnt

def compute_bound(n, cnt, m):
    max_c = cnt.max()

import numpy as np

# ----------------------------------------------------------------------
# Helper: count ordered pairs (i,j) with i+j = k for a set of indices
# ----------------------------------------------------------------------
def _pair_counts(selected_set, n):
    """
    selected_set : set of indices (0 … n-1) where the coefficient is non‑zero
    Returns       : np.ndarray of length 2*n-1 with ordered‑pair counts
    """
    if not selected_set:
        return np.zeros(2 * n - 1, dtype=np.int64)
    # convert set to a dense 1‑D array
    idx = np.fromiter(selected_set, dtype=np.int64, count=len(selected_set))
    # all ordered sums (including i=j)
    sums = idx[:, None] + idx[None, :]
    return np.bincount(sums.ravel(), minlength=2 * n - 1)


def _bound_from_counts(cnt, m, n):
    """
    Fast bound for a binary vector whose ones are at `selected_set`.
    cnt : pair‑count array (output of _pair_counts)
    m   : number of non‑zero entries
    n   : length of the full sequence
    """
    if m == 0:
        return float("inf")
    max_c = int(cnt.max())
    return 2.0 * n * max_c / (m * m)
