```
def _fast_score(vec):
    n = vec.shape[0]
    if n == 0:
        return np.inf
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
```

```
def propose_candidate(seed=42, num_iters=8000, **kwargs):
    rng = np.random.default_rng(seed)
    # Initialize cur
    init = globals().get('height_sequence_1')
    if init is None:
        n = 256
        cur = np.ones(n, dtype=np.float64)
    else:
        cur = np.asarray(init, dtype=np.float64).copy()
    # Clip and enforce nonnegative, 0-1000
    cur = np.clip(cur, 0.0, 1000.0)
    if cur.sum() < 1e-12:
        cur[:] = 1.0
    cur = cur / cur.sum()
    # Random perturbation to break symmetry
    cur = cur * np.exp(rng.normal(0.0, 0.05, size=cur.shape))
    cur = cur / cur.sum()
    n = cur.size
    # precompute m for FFT
    m = 1 << ((2*n - 1).bit_length())
    # evaluate initial
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    eta0 = 0.6
    for step in range(num_iters):
        # compute conv
        fa = np.fft.rfft(cur, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
        # find top peaks
        # get k_max = index of max
        # we can get top few
        # Using argpartition
        num_peaks = 3
        if num_peaks >= conv.size:
            top_idxs = np.arange(conv.size)
        else:
            # note argpartition returns unsorted
            part = np.argpartition(-conv, num_peaks)[:num_peaks]
            top_idxs = part
        # compute average b
        b = np.zeros_like(cur)
        for k in top_idxs:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            b[mask] += cur[idx[mask]]
        b = b / len(top_idxs)
        # gradient as subgradient of max: 2*(b - b.mean())
        grad = 2.0 * (b - b.mean())
        # schedule eta
        eta = eta0 * (1.0 - step / num_iters)
        # exponentiated update
        cur = cur * np.exp(-eta * grad)
        # normalize sum to 1
        s = cur.sum()
        if s < 1e-12:
            cur[:] = 1.0 / n
            s = cur.sum()
        cur = cur / s

        # occasional random noise perturbation
        if (step + 1) % 500 == 0:
            if rng.random() < 0.2:
                noise = np.exp(rng.normal(0.0, 0.1, size=cur.shape))
                cur = cur * noise
                cur = cur / cur.sum()
        # compute score
        cur_score = _fast_score(cur)
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score

    # final evaluation using evaluate_sequence
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # fallback to uniform
        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
        best = fallback
    return [float(v) for v in best]
```
