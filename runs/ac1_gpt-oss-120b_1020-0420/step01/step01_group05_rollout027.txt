import numpy as np

def _conv_max(a,b):
    n = a.shape[0]
    m = 2*n-1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(fa*fb, n=nfft)[:m]
    return np.max(c)

def _score(a):
    return 2.0*a.shape[0]*_conv_max(a,a)/(np.sum(a)**2)

def _conv_max(a,b):
    n = a.shape[0]
    m = 2*n-1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(fa*fb, n=nfft)[:m]
    return np.max(c.real)  # real part.
```

import numpy as np

# -------------------------------------------------------
# Helper routines – all FFT based, linear memory usage.
# -------------------------------------------------------
def _conv_max(a: np.ndarray, b: np.ndarray) -> float:
    """Maximum value of the linear convolution a * b (full length)."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(fa * fb, n=nfft)[:m]
    return float(np.max(c.real))


def _fast_score(a: np.ndarray) -> float:
    """Exact AC1 bound used by the evaluator (scale‑invariant)."""
    n = a.shape[0]
    max_conv = _conv_max(a, a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_conv / (s * s)

    # 1) Inverse‑square‑root family a_i = 1 / sqrt(i + off)
    # -------------------------------------------------------
    offs = np.geomspace(0.01, 50.0, num=30)   # wide, log‑spaced range
    for off in offs:
        a = 1.0 / np.sqrt(i + off)
        score = _fast_score(a)
        if score < best_score:
            best_seq = a
            best_score = score

    # -------------------------------------------------------
    # 2) General power‑law a_i = (i + off)^{-p}
    #    Grid of exponent p and offset off.
    # -------------------------------------------------------
    p_vals = np.linspace(0.40, 0.60, 21)      # focus around 0.5
    off_vals = np.geomspace(0.01, 10.0, 21)  # from tiny to moderate
    for p in p_vals:
        for off in off_vals:
            a = (i + off) ** (-p)
            score = _fast_score(a)
            if score < best_score:
                best_seq = a
                best_score = score

    # -------------------------------------------------------
    # 3) Iterative “balancing” – cheap matrix‑free analogue of the LP step.
    # -------------------------------------------------------
    f = best_seq.copy()
    alpha = 0.20
    max_iter = 30
    # shift fractions to try (as integer offsets)
    shifts = [0.05, 0.10, 0.20, 0.30, 0.40, 0.50]

    for _ in range(max_iter):
        M_f = _conv_max(f, f)
        improved = False

        # candidate pool: reversed and several shifted copies
        cand_pool = [f[::-1]]
        for frac in shifts:
            s = int(n * frac)
            if s <= 0 or s >= n:
                continue
            # right shift (zeros at start)
            b = np.empty_like(f)
            b[:s] = 0.0
            b[s:] = f[:n - s]
            cand_pool.append(b)
            # left shift (zeros at end)
            b2 = np.empty_like(f)
            b2[:n - s] = f[s:]
            b2[n - s:] = 0.0
            cand_pool.append(b2)

        for b in cand_pool:
            M_fb = _conv_max(f, b)
            if M_fb <= 0.0:
                continue
            scale = M_f / M_fb
            g = b * scale
            f_new = (1.0 - alpha) * f + alpha * g
            f_new = np.clip(f_new, 0.0, 1000.0)
            score = _fast_score(f_new)
            if score < best_score:
                best_seq = f_new.copy()
                best_score = score
                f = f_new
                improved = True
                # accept first improvement and continue the outer loop
                break
        if not improved:
            break
        # gradually reduce step size
        alpha = max(alpha * 0.90, 0.01)

    # -------------------------------------------------------
    # 4) Tiny random multiplicative tweaks around the current best.
    # -------------------------------------------------------
    for _ in range(300):
        perturb = np.exp(rng.normal(0.0, 0.015, size=n))   # ~1.5 % log‑normal jitter
        cand = best_seq * perturb
        cand = np.clip(cand, 0.0, 1000.0)
        score = _fast_score(cand)
        if score < best_score:
            best_seq = cand
            best_score = score

    # -------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    # -------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
