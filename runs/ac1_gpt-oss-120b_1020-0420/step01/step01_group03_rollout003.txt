import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Catalan‐type sequence a_i = C(2i,i)/4**i (root‑constant)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    # a_i = a_{i-1} * (2*i-1) / (2*i)
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    # safety clipping (required by the evaluator)
    return np.clip(seq, 0.0, 1000.0)


def _max_convolution(a: np.ndarray) -> float:
    """Full convolution a*a using FFT and return its maximum value."""
    n = a.shape[0]
    # length for FFT (next power of two >= 2*n-1)
    m = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a
    reduced AC1 upper bound relative to the baseline root‑constant
    construction.  The routine starts from the root‑constant or the
    provided parent `height_sequence_1` and performs a cheap hill‑climbing
    that moves mass from early (large) coefficients to late (small)
    coefficients, lowering the peak of the self‑convolution while keeping
    the total sum unchanged.
    """
    import numpy as np

    # ---------------------------------------------------------------
    # 1. initialise the base sequence
    # ---------------------------------------------------------------
    try:
        # parent sequence may already be a good starting point
        base = np.array(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    except Exception:
        # fallback: generate a fresh root‑constant of modest size
        n = 256
        base = _root_constant_seq(n)

    base = np.clip(base, 0.0, 1000.0)

    # ---------------------------------------------------------------
    # 2. helper to evaluate the current score (no call to evaluate_sequence)
    # ---------------------------------------------------------------
    sum_a = float(base.sum())
    max_b = _max_convolution(base)
    best_score = 2.0 * n * max_b / (sum_a * sum_a)
    best_seq = base.copy()

    # ---------------------------------------------------------------
    # 3. hill‑climbing parameters
    # ---------------------------------------------------------------
    rng = np.random.RandomState(seed)
    max_iters = 3000                     # a few thousand cheap iterations
    step_factor = 0.08                   # up‑to 8 % of the source coefficient
    early_end = max(1, n // 3)           # region with relatively large entries
    late_start = max(1, 2 * n // 3)      # region with relatively small entries

    # ---------------------------------------------------------------
    # 4. main optimisation loop
    # ---------------------------------------------------------------
    for _ in range(max_iters):
        # pick an early index i (large) and a late index j (small)
        i = rng.randint(0, early_end)
        j = rng.randint(late_start, n)

        if base[i] <= 1e-12:
            continue

        # propose moving a fraction of a_i to a_j
        delta = base[i] * step_factor * rng.rand()
        # respect upper bound 1000 for the target entry
        if base[j] + delta > 1000.0:
            delta = 1000.0 - base[j]
            if delta <= 0.0:
                continue
        # respect non‑negativity for the source entry
        if base[i] - delta < 0.0:
            delta = base[i]

        # apply the tentative move
        base[i] -= delta
        base[j] += delta

        # evaluate the new max of the self‑convolution
        new_max = _max_convolution(base)
        new_score = 2.0 * n * new_max / (sum_a * sum_a)

        # accept if improvement; otherwise revert
        if new_score < best_score and np.isfinite(new_score):
            best_score = new_score
            best_seq = base.copy()
        else:
            # revert
            base[i] += delta
            base[j] -= delta

    # ---------------------------------------------------------------
    # 5. final safety check with the official evaluator
    # ---------------------------------------------------------------
    try:
        # convert to list as required by the harness
        candidate = best_seq.tolist()
        # validate – the evaluator returns np.inf on invalid input
        val = float(evaluate_sequence(candidate))
        if not np.isfinite(val):
            raise ValueError
    except Exception:
        # any unexpected problem – fall back to the simple root‑constant
        candidate = _root_constant_seq(len(best_seq)).tolist()

    return candidate
