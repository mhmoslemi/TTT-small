def compute_score(vec):
    # vec normalized to sum 1
    # compute convolution via FFT
    n = vec.size
    n_fft = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(vec, n=n_fft)
    conv = np.fft.irfft(fft_a * fft_a, n=n_fft)
    # conv length n_fft; relevant part first 2*n-1
    max_b = conv[:2*n-1].max()
    # sum is 1 (but compute anyway)
    sum_a = vec.sum()
    # Avoid division by 0 (should not happen)
    if sum_a == 0:
        return float('inf')
    return 2 * n * max_b / (sum_a * sum_a)

import numpy as np

def _fft_score(vec):
    n = vec.size
    n_fft = 1 << ((2*n - 1).bit_length())
    # real FFT
    fft_a = np.fft.rfft(vec, n=n_fft)
    conv = np.fft.irfft(fft_a * fft_a, n=n_fft)
    max_b = conv[:2*n-1].max()
    sum_a = vec.sum()
    if sum_a == 0:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def _normalize(vec):
    total = vec.sum()
    if total <= 0:
        return vec
    return vec / total

import numpy as np

def _fft_score(vec):
    # Ensure non-negative, but we assume already.
    n = vec.size
    # Determine FFT length as next power of two >= 2n-1
    n_fft = 1 << ((2 * n - 1).bit_length())
    # Real-valued FFT
    fft_a = np.fft.rfft(vec, n=n_fft)
    conv = np.fft.irfft(fft_a * fft_a, n=n_fft)
    # Linear convolution part
    max_b = conv[:2 * n - 1].max()
    sum_a = vec.sum()
    if sum_a == 0.0:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

    if total <= 0.0:

import numpy as np

def _fft_score(vec: np.ndarray) -> float:
    """
    Compute the AC1 upper bound for a non‑negative sequence `vec`
    using an FFT based linear convolution. The result is exactly the
    quantity 2 * n * max(conv) / (sum(vec) ** 2).
    """
    n = vec.size
    # next power of two >= 2n‑1  (linear convolution via zero‑padding)
    n_fft = 1 << ((2 * n - 1).bit_length())
    # real FFT → element‑wise product → inverse FFT gives linear convolution
    fft_a = np.fft.rfft(vec, n=n_fft)
    conv = np.fft.irfft(fft_a * fft_a, n=n_fft)
    max_b = conv[: 2 * n - 1].max()
    sum_a = vec.sum()
    if sum_a == 0.0:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)

    return vec / total

    for beta in np.linspace(5e-4, 4e-3, 9):
        cand = np.exp(-beta * np.arange(n, dtype=np.float64))
        cand = _normalize(cand)
        sc = _fft_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # Hann (raised‑cosine) window
    hann = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    cand = _normalize(hann)
    sc = _fft_score(cand)
    if sc < best_score:
        best, best_score = cand, sc

    # Triangular window (peak in the middle)
    tri = 1.0 - np.abs(np.linspace(-1.0, 1.0, n))
    cand = _normalize(tri)
    sc = _fft_score(cand)
    if sc < best_score:
        best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 3.  Simulated‑annealing‑style hill climbing
    # ------------------------------------------------------------------
    max_iter = 1500
    T0 = 0.02                         # temperature (relative to best_score)

    for it in range(max_iter):
        # linear cooling schedule
        T = T0 * (1.0 - it / max_iter)

        cand = best.copy()

        # --- multiplicative block perturbation --------------------------------
        start = rng.integers(0, n)
        max_block = max(1, n // 200)               # ≤0.5 % of length
        blk_len = rng.integers(1, max_block + 1)
        stop = min(n, start + blk_len)
        noise = np.exp(rng.normal(0.0, 0.22, size=stop - start))
        cand[start:stop] *= noise

        # --- occasional mild smoothing ----------------------------------------
        if rng.random() < 0.08:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            cand = np.convolve(cand, kernel, mode="same")

        cand = np.clip(cand, 0.0, 1e3)
        cand = _normalize(cand)

        sc = _fft_score(cand)

        # Metropolis acceptance
        if sc < best_score or rng.random() < np.exp(-(sc - best_score) / (T * best_score + 1e-12)):
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 4.  Random restarts with short local refinements
    # ------------------------------------------------------------------
    for _ in range(4):
        cand = _normalize(rng.random(n))
        sc = _fft_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

        for __ in range(300):
            cand2 = cand.copy()
            start = rng.integers(0, n)
            max_block = max(1, n // 250)
            blk_len = rng.integers(1, max_block + 1)
            stop = min(n, start + blk_len)
            noise = np.exp(rng.normal(0.0, 0.18, size=stop - start))
            cand2[start:stop] *= noise
            cand2 = np.clip(cand2, 0.0, 1e3)
            cand2 = _normalize(cand2)
            sc2 = _fft_score(cand2)
            if sc2 < sc:
                cand, sc = cand2, sc2

        if sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 5.  Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
