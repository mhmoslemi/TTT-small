import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a))**2 using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a via FFT (real output)."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    return np.fft.irfft(fa * fa, n=nfft)[:m]


# ----------------------------------------------------------
# Candidate generation utilities.
# ----------------------------------------------------------
def _update_best(seq: np.ndarray, best_seq: np.ndarray,
                 best_score: float) -> tuple:
    """Clip, normalise, score and possibly replace the current best."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:
        return best_seq, best_score
    seq = seq / total                     # scale – shape only matters
    sc = _fast_score(seq)
    if sc < best_score:
        return seq, sc
    return best_seq, best_score


def _refine(seq: np.ndarray, score: float,
            max_iter: int = 150) -> tuple:
    """Simple gradient‑descent on the max‑convolution term."""
    n = seq.shape[0]
    best = seq.copy()
    best_score = score
    step = 0.05                           # initial step magnitude
    idx = np.arange(n)
    for _ in range(max_iter):
        conv = _conv_full(best)
        kmax = int(np.argmax(conv))
        # gradient of conv[kmax] w.r.t. best (ignoring the max operator)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[mask] = best[j[mask]]
        same = mask & (j == idx)
        grad[same] = 2.0 * best[same]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        cand = best - step * grad / norm
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0:
            cand = cand / s                 # re‑normalise
        sc = _fast_score(cand)
        if sc < best_score:
            best, best_score = cand, sc
            step *= 1.2
        else:
            step *= 0.7
        if step < 1e-12:
            break
    return best, best_score

    ramp = (i + 1.0) / (n)                       # increasing from ~0 to 1
    best_seq, best_score = _update_best(ramp, best_seq, best_score)
    best_seq, best_score = _update_best(ramp[::-1], best_seq, best_score)

    # ---- 5) Gradient‑descent refinement ----------------------------------
    best_seq, best_score = _refine(best_seq, best_score, max_iter=150)

    # ---- 6) Small random multiplicative tweaks -------------------------
    for _ in range(500):
        perturb = np.exp(rng.normal(0.0, 0.025, size=n))
        cand = best_seq * perturb
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0:
            cand = cand / s
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # Return as plain Python list (the evaluator will clip again if needed)
    return [float(x) for x in best_seq.tolist()]
