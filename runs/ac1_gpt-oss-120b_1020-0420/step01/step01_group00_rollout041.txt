import numpy as np

# ----------------------------------------------------------------------
# fast evaluation of the AC1 bound (exact, uses FFT)
# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Exact AC1 upper bound for a non‑negative vector (scale invariant)."""
    n = vec.size
    if n == 0:
        return float("inf")
    m = 1 << ((2 * n - 1).bit_length())          # next power of two >= 2n‑1
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = float(conv.max())
    s = float(vec.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)
