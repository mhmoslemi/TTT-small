def _fast_score(a):
    n = a.shape[0]
    m = 2*n -1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa*fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S*S)
```

# ------------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound using FFT.
# ------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)^2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float("inf")
    return 2.0 * n * M / (S * S)
