def propose_candidate(seed=42, num_steps=5000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = parent.size
        else:
            n = 256
    else:
        n = 256
    # Build Catalan sequence
    def catalan_seq(m):
        a = np.empty(m, dtype=np.float64)
        a[0] = 1.0
        for i in range(m-1):
            a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0 * (i+1)*(i+1))
        return a

    cand1 = catalan_seq(n)
    cand1 = np.clip(cand1, 0.0, 1000.0)
    # Another candidate: parent sequence (if exists)
    if 'height_sequence_1' in globals():
        cand2 = np.clip(parent.astype(np.float64), 0.0, 1000.0)  # copy
    else:
        cand2 = cand1  # fallback
    # Evaluate both using our _score function
    def bound_of(seq):
        conv = np.convolve(seq, seq)
        M = conv.max()
        S = seq.sum()
        return 2.0 * len(seq) * M / (S * S)

    bound1 = bound_of(cand1)
    bound2 = bound_of(cand2)
    if bound2 < bound1:
        a = cand2.copy()
        best_bound = bound2
    else:
        a = cand1.copy()
        best_bound = bound1

    n = a.size
    sum_a = a.sum()
    b = np.convolve(a, a)
    # Deterministic early shrink & tail reallocation
    prefix_ratios = [0.05, 0.10, 0.15, 0.20, 0.30]
    shrink_factors = np.linspace(0.2, 0.9, 15)  # from 20% to 90%
    for r in prefix_ratios:
        k = max(1, int(n * r))
        early = a[:k]
        tail = a[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()
            # scale tail uniformly
            g = 1.0 + delta / tail_sum
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            cur_bound = bound_of(cand)
            if cur_bound < best_bound:
                a = cand
                b = np.convolve(a, a)
                sum_a = a.sum()
                best_bound = cur_bound

    # Hill climbing
    steps = kwargs.get('num_steps', num_steps)
    # Optionally reduce step size adaptively
    for step in range(steps):
        # pick src from early region (top part) and dst from later region
        # choose src proportionally to a (large value)
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # move up to 15% of source entry
        max_move = a[src] * 0.15
        if max_move <= 0.0:
            continue
        delta = rng.uniform(0.01 * a[src], max_move)  # at least small fraction
        # compute candidate bound via incremental update
        cand_a = a.copy()
        cand_b = b.copy()
        # shift updates using original a
        # subtract src shift
        cand_b[src:src+n] -= 2.0 * delta * a
        # add dst shift
        cand_b[dst:dst+n] += 2.0 * delta * a
        # self terms
        cand_b[2*src] += delta*delta
        cand_b[2*dst] += delta*delta
        # cross term
        cand_b[src+dst] -= 2.0 * delta*delta
        # update a
        cand_a[src] -= delta
        cand_a[dst] += delta
        # enforce nonnegativity and clipping
        if cand_a[src] < -1e-12:

        cur_bound = 2.0 * n * M_new / (sum_a * sum_a)
        if cur_bound < best_bound * (1 - 1e-12): # strictly better
            a = cand_a
            b = cand_b
            best_bound = cur_bound
    # Return final best as list
    return [float(v) for v in a]

import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _catalan_seq(m: int) -> np.ndarray:
    """Central‐binomial (Catalan‑like) positive sequence."""
    a = np.empty(m, dtype=np.float64)
    a[0] = 1.0
    for i in range(m - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


def _bound(seq: np.ndarray) -> float:
    """Exact bound used by the evaluator (scale‑invariant)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = np.convolve(seq, seq)
    m = conv.max()
    return 2.0 * seq.size * m / (s * s)


def _apply_move(a: np.ndarray, b: np.ndarray, src: int, dst: int, delta: float):
    """
    Return (new_a, new_b) after moving `delta` mass from src to dst.
    `a` and `b` are the current sequence and its full convolution.
    The update is performed in O(n) using the analytical formulas.
    """
    n = a.shape[0]
    new_a = a.copy()
    new_b = b.copy()

    # shift contributions (using the *old* a)
    new_b[src:src + n] -= 2.0 * delta * a
    new_b[dst:dst + n] += 2.0 * delta * a

    # self‑terms and cross‑term
    new_b[2 * src] += delta * delta
    new_b[2 * dst] += delta * delta
    new_b[src + dst] -= 2.0 * delta * delta

    # update the sequence itself
    new_a[src] -= delta
    new_a[dst] += delta
    return new_a, new_b

    """
    Return a list of non‑negative heights with a low AC1 bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Problem size
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

        if parent.ndim == 1 and parent.size > 0:

    # --------------------------------------------------------
    cand_cat = _catalan_seq(n)
    cand_cat = np.clip(cand_cat, 0.0, 1000.0)
    best_a = cand_cat
    best_score = _bound(best_a)

    if "height_sequence_1" in globals():
        cand_par = np.clip(parent.astype(np.float64), 0.0, 1000.0)
        score_par = _bound(cand_par)
        if score_par < best_score:
            best_a = cand_par
            best_score = score_par

    # --------------------------------------------------------
    # Deterministic reallocation (shrink early prefix, inflate tail)
    # --------------------------------------------------------
    prefix_ratios = [0.05, 0.10, 0.15, 0.20, 0.30]
    shrink_factors = np.linspace(0.20, 0.80, 13)   # 20 % … 80 % in steps of 5 %
    improved = True

            k = max(1, int(n * r))               # size of early block

            early_sum = early.sum()
            tail_sum = tail.sum()
            if tail_sum == 0.0:

            for f in shrink_factors:
                new_early = early * f
                delta = early_sum - new_early.sum()
                g = 1.0 + delta / tail_sum               # uniform inflation factor
                new_tail = tail * g
                cand = np.concatenate([new_early, new_tail])
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _bound(cand)
                if sc < best_score:

                    best_score = sc
                    improved = True
        # recompute convolution only when an improvement was found

    # relatively large entry to a relatively small one.

    a = best_a.copy()

    # --------------------------------------------------------
    return [float(v) for v in a]
