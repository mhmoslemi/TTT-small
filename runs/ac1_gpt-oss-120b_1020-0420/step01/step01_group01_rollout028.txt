def _norm_score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    conv = np.convolve(seq, seq)
    M = conv.max()
    if M <= 0:
        return np.inf
    seq_norm = seq / np.sqrt(M)
    s = seq_norm.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * len(seq) * 1.0 / (s*s))

import numpy as np

# ------------------------------------------------------------
# Helper: same score as the harness uses
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution (FFT‑based for large n)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size – use the provided parent if present.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = base.size
        else:
            base = None
            n = 256
    else:
        base = None
        n = 256

    if base is None:
        # fall back to the classic Catalan construction
        base = _catalan_seq(n)

    base = np.clip(base, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Keep the best candidate seen so far (start from the provided one)
    # ------------------------------------------------------------------
    best = base.copy()
    best_score = _score(best)

    # ------------------------------------------------------------
    # Deterministic two‑parameter sweep:
    #   – shrink an early prefix by factor f (0 < f <= 1)
    #   – optionally inflate the remaining tail by factor g (g >= 1)
    # The sweep explores many cut positions but limits the total
    # number of evaluated candidates to stay well within the time budget.
    # ------------------------------------------------------------
    # cut positions (as integer indices) – roughly logarithmic spacing
    cuts = np.unique(
        np.concatenate(
            (
                np.linspace(1, n - 1, num=80, dtype=int),                # linear grid
                np.logspace(0, np.log10(max(1, n - 1)), num=80, dtype=int)  # log grid
            )
        )
    )
    cuts = cuts[(cuts > 0) & (cuts < n)]

    # early‑shrink factors
    f_vals = np.linspace(0.02, 0.98, 25)   # from 2 % up to 98 %
    # tail‑inflate factors (optional)
    g_vals = np.linspace(1.0, 3.0, 21)    # up to three‑times

    # --- only early shrink -------------------------------------------------
    for cut in cuts:
        early = base[:cut]
        tail = base[cut:]
        for f in f_vals:
            cand = np.concatenate([early * f, tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    # --- only tail inflation -----------------------------------------------
    for cut in cuts:
        early = base[:cut]
        tail = base[cut:]
        for g in g_vals:
            cand = np.concatenate([early, tail * g])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    # --- combined early shrink & tail inflate (coarse grid) -----------------
    # only a subset of cuts is examined more thoroughly to keep work modest.
    coarse_cuts = cuts[::max(1, len(cuts) // 12)]   # ~12 representative cuts
    for cut in coarse_cuts:
        early = base[:cut]
        tail = base[cut:]
        for f in f_vals[::4]:           # coarser step
            for g in g_vals[::4]:
                cand = np.concatenate([early * f, tail * g])
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc

    # ------------------------------------------------------------
    # Stochastic refinement: random mass transfers between two indices.
    # Accept only if the AC1 bound improves.
    # ------------------------------------------------------------
    steps = kwargs.get("num_steps", num_steps)
    for _ in range(steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # move a random fraction (1‑% … 30‑%) of src to dst
        delta = best[src] * rng.uniform(0.01, 0.30)
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # ------------------------------------------------------------
    return [float(v) for v in best]
