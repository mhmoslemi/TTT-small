def propose_candidate(seed=42, max_iters=300, restarts=5, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n and parent vector if exists
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
            base_vec = parent.copy()
        else:
            raise NameError
    except NameError:
        n = 256
        base_vec = None

    # Define function to initialize candidate vector
    def init_vec():
        if base_vec is not None and rng.random() < 0.5:
            # start from base, but maybe perturb a little
            v = base_vec.copy()
            # small random perturb *0.2 factor
            v *= np.exp(rng.normal(0.0, 0.05, size=n))
            v = np.clip(v, 0.0, 1000.0)
            # normalize sum to sum_target
            s = v.sum()
            if s > 0:
                v *= (base_vec.sum() / s)
            return v
        else:
            # random positive values, e.g., exponential
            v = rng.exponential(scale=1.0, size=n)
            v = np.clip(v, 0.0, 1000.0)
            # bring sum to a target e.g., 1.0 or base sum:
            v *= (1.0 / v.sum())
            return v

    # target sum for scaling: Let's set sum_target = 1.0 (any)
    sum_target = 1.0

    best_vec = None
    best_score = float('inf')
    # fallback uniform candidate
    uniform = np.ones(n, dtype=np.float64) * (sum_target / n)
    uniform_score = compute_bound(uniform)
    best_vec = uniform
    best_score = uniform_score

    # Evaluate base if existed
    if base_vec is not None:
        base_norm = base_vec.copy()
        base_norm *= (sum_target / base_norm.sum())
        s = compute_bound(base_norm)
        if s < best_score:
            best_vec = base_norm
            best_score = s

    # main loops
    for r in range(restarts):
        a = init_vec()
        # scale to sum_target
        sum_a = a.sum()
        if sum_a > 0:
            a *= (sum_target / sum_a)
        # gradient descent
        step = 0.1
        for it in range(max_iters):

            max_b = conv.max()
            k_max = int(conv.argmax())
            sum_a = a.sum()

            end = min(n, k_max + 1)
            idx = np.arange(start, end, dtype=int)

            # gradient for bound (ignoring prefactor)
            d = g - (2.0 * max_b / sum_a) * a

            # compute new bound

            if new_score < best_score:
                best_score = new_score
                best_vec = a_new.copy()
                # accept and maybe increase step
                a = a_new
                step = min(step * 1.1, 1.0)
            else:
                # reject

                if step < 1e-8:

    # final fallback: optionally perform local random perturbation as in previous program to try improvements

import numpy as np

def _full_convolution_self(a):
    """Return full convolution a*a using FFT. a is 1‑D float array."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def _bound(a):
    """Exact bound = 2 n * max(b) / (sum a)^2."""
    n = a.shape[0]
    conv = _full_convolution_self(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a == 0:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def _gradient(a, k_max, max_b):
    """Derivative of the bound w.r.t a (ignoring the common prefactor)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    start = max(0, k_max - (n - 1))
    end = min(n, k_max + 1)
    if start < end:
        idx = np.arange(start, end, dtype=int)
        g[idx] = 2.0 * a[k_max - idx]
    sum_a = a.sum()
    if sum_a == 0:
        return g
    return g - (2.0 * max_b / sum_a) * a

def propose_candidate(seed=42, max_iters=200, restarts=3, **kwargs):
    """Return a list of non‑negative coefficients with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
        base_vec = parent.copy()
    except NameError:
        n = 256
        base_vec = None

    # ----- target sum (scale is irrelevant for the bound) -------------
    sum_target = 1.0

    # ----- initialise best candidate ------------------------------------
    # uniform fallback (always feasible)
    best_vec = np.full(n, sum_target / n, dtype=np.float64)
    best_score = _bound(best_vec)

    if base_vec is not None:
        tmp = base_vec.copy()
        tmp *= sum_target / tmp.sum()
        sc = _bound(tmp)
        if sc < best_score:
            best_vec = tmp
            best_score = sc

    # ----- random‑initialised gradient descent runs --------------------
    for _ in range(restarts):
        # random start: exponential distribution, then renorm
        a = rng.exponential(scale=1.0, size=n)
        a *= sum_target / a.sum()

        # optionally blend with the base vector on half of the restarts
        if base_vec is not None and rng.random() < 0.4:
            mix = rng.random()
            a = mix * a + (1 - mix) * (base_vec * sum_target / base_vec.sum())

        step = 0.1
        for _ in range(max_iters):
            conv = _full_convolution_self(a)
            max_b = conv.max()
            k_max = int(conv.argmax())
            d = _gradient(a, k_max, max_b)

            # propose update
            a_new = a - step * d
            np.maximum(a_new, 0.0, out=a_new)
            # renormalise to keep the fixed sum
            s = a_new.sum()
            if s == 0.0:
                break
            a_new *= (sum_target / s)

            new_score = _bound(a_new)

            # accept if improvement, otherwise shrink step
            if new_score < best_score:
                best_vec = a_new.copy()
                best_score = new_score
                a = a_new
                step = min(step * 1.1, 1.0)
            else:
                step = step * 0.5
                if step < 1e-8:
                    break  # step too small, give up this run

    # ----- final lightweight hill‑climb (random local perturbations) --
    for _ in range(50):
        width = rng.integers(1, max(2, n // 10))
        start = rng.integers(0, n - width + 1)
        cand = best_vec.copy()
        factor = np.exp(rng.normal(0.0, 0.05, size=width))

        s = cand.sum()
        if s == 0.0:
            continue
        cand *= (sum_target / s)
        sc = _bound(cand)
        if sc < best_score:
            best_vec = cand
            best_score = sc

    # return plain Python list
    return [float(v) for v in best_vec.tolist()]
