    n = a.shape[0]

    conv = np.fft.irfft(fft * fft, m)[:length]

    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

# ------------------------------------------------------------
# Half‑power series (coefficients of (1‑z)^(-1/2))
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return the first ``length`` terms of (1‑z)^(-1/2)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Score evaluation (same objective as the harness)
# ------------------------------------------------------------
def _fft_len_for(n: int) -> int:
    """next power‑of‑two length sufficient for full convolution of size n."""
    length = 2 * n - 1
    return 1 << (length.bit_length())


def compute_score(a: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector a."""
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    n = a.shape[0]
    m = _fft_len_for(n)
    # convolution via FFT (full length)
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)


# ------------------------------------------------------------
# Helper: a single block perturbation (fallback diversification)
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block of ``base`` by a log‑normal factor."""
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))   # mild perturbation
    cand = base.copy()
    cand[start : start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 bound
    is (hopefully) lower than the reference 1.57086.
    """
    rng = np.random.default_rng(seed)

    # ------- initialise -------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        a = height_sequence_1.astype(np.float64).copy()
        np.clip(a, 0.0, 1000.0, out=a)
    else:
        # fallback length – the parent used 256 when no sequence was present
        n = 256
        a = generate_half_power_series(n)
        np.clip(a, 0.0, 1000.0, out=a)

    n = a.shape[0]
    m_fft = _fft_len_for(n)

    # ------- best so far ------------------------------------------------
    best = a.copy()
    best_score = compute_score(best)

    # ------- gradient‑guided stochastic descent -------------------------
    steps = 12000                     # total iterations
    max_step = 0.45                   # initial transfer fraction
    min_step = 0.015                  # final transfer fraction

    cur = best.copy()
    cur_score = best_score

    # pre‑allocate index array for speed
    all_idx = np.arange(n, dtype=np.int64)

    for it in range(steps):
        # current full convolution and index of its maximum
        fa = np.fft.rfft(cur, m_fft)
        conv_full = np.fft.irfft(fa * fa, m_fft)[: 2 * n - 1]
        idx_max = int(np.argmax(conv_full))

        # partners that actually appear in the max‑index convolution
        partners = idx_max - all_idx
        mask = (partners >= 0) & (partners < n)
        if not mask.any():
            # degenerate – pick a random i
            i = rng.integers(0, n)
            partner = -1
        else:
            # weight proportional to the contribution a[i]*a[partner]
            contrib = cur[all_idx[mask]] * cur[partners[mask]]
            weights = np.zeros(n, dtype=np.float64)
            weights[all_idx[mask]] = contrib
            total_w = weights.sum()
            if total_w == 0.0:
                i = rng.integers(0, n)
            else:
                probs = weights / total_w
                i = rng.choice(n, p=probs)
            partner = idx_max - i  # guaranteed to be in range because i came from mask

        # choose a destination index far in the second half (helps keep
        # the new mass away from the current maximum region)
        if n >= 2:
            k = rng.integers(n // 2, n)
            while k == i or k == partner:
                k = rng.integers(n // 2, n)
        else:
            k = 0

        # linear schedule for the transfer magnitude
        step_frac = max_step - (max_step - min_step) * (it / steps)

        if partner == i:
            # self‑pair case: move mass from i to k
            max_ex = cur[i] * step_frac
            cap = 1000.0 - cur[k]
            ex = min(max_ex, cap)
            if ex <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= ex
            cand[k] += ex
        else:
            # double‑pair case: take the same amount from i and its partner,
            # dump twice that amount into k.
            max_ex = min(cur[i], cur[partner]) * step_frac
            cap = (1000.0 - cur[k]) / 2.0
            ex = min(max_ex, cap)
            if ex <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= ex
            cand[partner] -= ex
            cand[k] += 2.0 * ex

        # ensure feasibility (clipping already guarantees bounds)
        # compute new score
        cand_score = compute_score(cand)

        if np.isfinite(cand_score) and cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

    # ------- occasional random block tweaks (optional safety net) ------
    for _ in range(200):
        cand = random_block_perturbation(best, rng)
        cand_score = compute_score(cand)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score

    # Return as a plain Python list – the harness expects this format.
    return [float(x) for x in best.tolist()]
