def simplex_projection(v):
    # ensures sum(v) == 1, v >= 0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def full_convolution(a):
    n = a.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    return conv
```

def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the classical root‑of‑constant sequence
# ----------------------------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    """
    a_i = C(2i,i)/4**i  (i=0,…,length‑1)  –  the sequence whose
    (a*a)[k] = 1 for 0 ≤ k < length.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Convolution (FFT, full mode)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, m)
    conv = np.fft.irfft(fx * fx, m)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (sum == 1, non‑negative)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, sum w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond].max() - 1               # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 upper
    bound is (hopefully) below that of the reference root‑constant sequence.
    The routine starts from the reference shape, normalises it to sum‑one
    and then performs a cheap projected gradient descent on the maximum
    convolution value while staying on the simplex.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine length and initial vector
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is supplied by the harness when a parent exists
        n = int(len(height_sequence_1))
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        # fallback: moderate size if no parent is present
        n = 2048
        a = _root_constant_seq(n)

    # Clamp, then normalise to a probability vector (sum == 1)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # Keep the best point seen so far
    best_a = a.copy()
    best_M = _full_convolution(best_a).max()

    # Target max‑convolution derived from the desired AC1 bound
    target_ratio = 1.50274349
    target_M = target_ratio / (2.0 * n)

    max_iters = 2000            # modest budget – easily stays linear‑memory
    step0 = 0.5                 # initial step size (will decay)

    for it in range(max_iters):
        conv = _full_convolution(a)
        M = conv.max()

        # Record improvement
        if M < best_M:
            best_M = M
            best_a = a.copy()
            # early stop if we beat the target
            if best_M <= target_M:
                break

        # Determine the index attaining the maximum (any one is fine)
        k = int(np.argmax(conv))

        # Gradient of the max‑convolution w.r.t. a (2*a_{k‑i} where defined)
        g = np.zeros_like(a)
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        if start <= end:
            idx = np.arange(start, end + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]

        # Simple decaying step size
        step = step0 / (it + 1.0)

        # Gradient step + tiny random jitter to avoid stagnation
        a_candidate = a - step * g
        a_candidate += (np.random.rand(n) - 0.5) * step * 0.01

        # Enforce non‑negativity, then project back onto the simplex
        a_candidate = np.clip(a_candidate, 0.0, None)
        total = a_candidate.sum()
        if total == 0.0:
            a_candidate = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a_candidate = a_candidate / total

        a = a_candidate

    # ------------------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # ------------------------------------------------------------------
    candidate = best_a

    # Final safety check – if something went awry, fall back to a simple root‑seq
    try:
        score = float(evaluate_sequence(candidate.tolist()))
        if not np.isfinite(score):
            raise ValueError
    except Exception:
        fallback = _root_constant_seq(n)
        fallback = fallback / fallback.sum()
        candidate = fallback

    return candidate.tolist()
