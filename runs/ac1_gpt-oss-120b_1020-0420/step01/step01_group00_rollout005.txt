import numpy as np

def _fast_score(vec: np.ndarray) -> float:
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _gradient(a: np.ndarray, conv: np.ndarray, max_c: float) -> np.ndarray:
    n = a.size
    # indices where conv equals max_c (within tolerance)
    tol = max_c * 1e-12 + 1e-16
    max_idx = np.flatnonzero(conv >= max_c - tol)
    grad_c = np.zeros_like(a)
    for k in max_idx:
        idx1 = k - np.arange(n)
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad_c[mask1] += a[idx1[mask1]]
        idx2 = np.arange(n) - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad_c[mask2] += a[idx2[mask2]]

    # remove the component that changes sum (project onto sum‑zero subspace)
    grad -= grad.mean()
    return grad

def _fast_score(vec: np.ndarray) -> float:
    """Fast AC1 upper bound for a non‑negative vector (sum normalisation not required)."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()

        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _gradient(a: np.ndarray, conv: np.ndarray, max_c: float) -> np.ndarray:
    """Sub‑gradient of the AC1 bound w.r.t. a (projected to sum‑zero)."""
