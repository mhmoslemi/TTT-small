import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Root‑of‑constant sequence a_i = C(2i,i)/4**i (capped to 0…1000)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _score(arr: np.ndarray) -> float:
    """Exact AC1 score (identical to evaluate_sequence for a valid array)."""
    if arr.size == 0:
        return float("inf")
    s = float(arr.sum())
    if s < 0.01:
        return float("inf")
    conv = np.convolve(arr, arr)
    m = float(conv.max())
    return float(2.0 * arr.size * m / (s * s))


def _gradient(a: np.ndarray, total: float, max_val: float, kstar: int) -> np.ndarray:
    """Gradient of the AC1 score w.r.t. a (subgradient for the max)."""
    n = a.size
    idx = kstar - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    a_k = np.zeros_like(a)
    a_k[valid] = a[idx[valid]]
    # ds/da = (4n / S^3) * (S * a_{k*-i} - M)
    return (4.0 * n / (total ** 3)) * (a_k * total - max_val)


def propose_candidate(seed: int = 42, **kwargs):
    # ---- initialise ---------------------------------------------------------
    src = globals().get("height_sequence_1")
    if isinstance(src, (list, tuple, np.ndarray)):
        try:
            a = np.array(src, dtype=np.float64).copy()
        except Exception:
            a = _root_constant_seq(256)
    else:
        a = _root_constant_seq(256)

    a = np.clip(a, 0.0, 1000.0)
    n = a.size

    # ---- baseline score -----------------------------------------------------
    best = a.copy()
    best_score = _score(best)

    # ---- simple gradient‑descent (subgradient) -------------------------------
    max_iters = 30
    step = 0.1
    for _ in range(max_iters):
        S = float(best.sum())
        if S < 0.01:
            break
        conv = np.convolve(best, best)
        M = float(conv.max())
        k_star = int(conv.argmax())

        grad = _gradient(best, S, M, k_star)

        # try a step; shrink if no improvement
        improved = False
        cur_step = step
        for _ in range(6):
            cand = best - cur_step * grad
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = _score(cand)
            if cand_score < best_score - 1e-12:
                best = cand
                best_score = cand_score
                step = min(cur_step * 1.2, 1.0)
                improved = True
                break
            cur_step *= 0.5
        if not improved:
            break
        # early exit if update is tiny
        if np.linalg.norm(grad) * step < 1e-7:
            break

    # ---- safety fallback ----------------------------------------------------
    if not np.isfinite(best_score):
        best = _root_constant_seq(best.size)
    # return as plain list (the harness expects a list)
    return best.tolist()
