import numpy as np

def generate_half_power_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def random_block_perturbation(base, rng, width_frac=0.05, std=0.07):
    n = base.size
    max_width = max(1, int(n * width_frac))
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, std)))
    cand = base.copy()
    cand[start:start+width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine length n
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):
        n = int(height_sequence_1.size)
    else:
        n = 256

    # Base - half-power series
    base = generate_half_power_series(n)
    np.clip(base, 0.0, 1000.0, out=base)

    # Ensure sum not too small
    sum_target = base.sum()
    if sum_target < 0.01:
        base = np.full(n, 0.01/n)
        sum_target = base.sum()

    # Evaluate base
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Mixing parameters
    shift_fracs = [0.2, 0.33, 0.5]   # fractions of n
    t_vals = [0.05, 0.1, 0.15, 0.2, 0.25]

    for frac in shift_fracs:
        d = max(1, int(frac * n))
        if d >= n:
            continue
        # shifted version (right shift)
        g = np.concatenate((np.zeros(d, dtype=np.float64), base[:n - d]))
        for t in t_vals:
            cand = (1.0 - t) * base + t * g
            # renormalize sum to original
            scale = sum_target / cand.sum()
            cand *= scale
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score

    # Simple stochastic refinement
    num_steps = 200
    for _ in range(num_steps):
        cand = random_block_perturbation(best, rng)
        # renormalize to keep comparable sum (optional)
        # we can keep as is; evaluate will handle sum
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# Helper: generate the truncated series of (1 - z)^(-1/2)
# a_0 = 1,  a_i = a_{i-1} * (2*i - 1) / (2*i)
# This yields a decreasing non‑negative step function whose
# self‑convolution equals 1 for the first n entries.
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Small random block perturbation (keeps O(n) memory)
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    if n == 1:
        return base.copy()
    # block width between 1 and roughly n/20
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))      # mild log‑normal factor
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

                cand[i] -= delta
                if dest_type == "mirror":
                    j = n - 1 - i               # far‑end symmetric position
                else:  # "center"
                    j = n // 2
                cand[j] += delta
            # No scaling needed – total mass is unchanged.
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score

    # -----------------------------------------------------------------
    # 3. Light stochastic refinement (few steps to avoid heavy cost)
    # -----------------------------------------------------------------
    for _ in range(30):
        cand = random_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list as required by the harness
    return [float(x) for x in best]
```
