            best_vec = cand.copy()
            best_score = cand_score

    # ------------------------------------------------------------------
    # Final sanity check – ensure the returned vector is valid.
    # ------------------------------------------------------------------
    final_score = _ac1_score(best_vec)
    if not np.isfinite(final_score):
        # Fallback to a safe uniform vector (guaranteed finite).
        best_vec = uniform.copy()

    # Convert to plain Python floats for the harness.
    return [float(v) for v in best_vec]
