def build_block_matrix(f, block_bounds):
    n = f.size
    m = len(block_bounds)
    rows = [] # list of row indices for non-zero entries
    cols = [] # column indices
    data = [] # values
    # We'll compute each column (block) as vector a = np.convolve(f, mask) / block_len
    # full length = 2*n -1
    for j, (start, end) in enumerate(block_bounds):
        mask_len = end - start
        mask = np.zeros(n, dtype=np.float64)
        mask[start:end] = 1.0
        a = np.convolve(f, mask)  # length 2n-1
        # divide by mask_len
        a = a / mask_len
        # For constraints: a <= M
        # We'll accumulate entries
        rows.extend(range(len(a)))   # each row index from 0 to 2n-2
        cols.extend([j]*len(a))
        data.extend(a.tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, m))
    return A

def build_A(f, block_bounds):

import numpy as np

try:
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

def _score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    m = conv.max()
    n = seq.size
    return float(2.0 * n * m / (s * s))

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0*(i+1)*(i+1))
    return a

def _build_blocks(n, m):
    bounds = []
    start = 0
    for j in range(m):
        end = (j+1)*n // m
        if j == m-1:
            end = n
        bounds.append((start, end))
        start = end
    return bounds

def _build_A(f, block_bounds):
    n = f.size
    m = len(block_bounds)
    rows = []
    cols = []
    data = []
    full_len = 2*n - 1
    for col, (st, en) in enumerate(block_bounds):
        blen = en - st
        if blen == 0:
            continue
        mask = np.zeros(n, dtype=np.float64)
        mask[st:en] = 1.0
        a = np.convolve(f, mask)  # length full_len
        a = a / blen
        rows.extend(range(full_len))
        cols.extend([col]*full_len)
        data.extend(a.tolist())
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, m))

def propose_candidate(seed: int = 42, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # Initial candidates
    cat = _catalan_seq(n)
    cat = np.clip(cat, 0.0, 1000.0)
    # Use parent if it gives a better initial score
    best = cat
    best_score = _score(best)
    if 'height_sequence_1' in globals():
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        sc = _score(par)
        if sc < best_score:
            best = par
            best_score = sc

    # Parameters
    block_cnt = min(150, max(20, n // 50))   # between 20 and 150 blocks
    block_bounds = _build_blocks(n, block_cnt)

    f = best.copy()
    # Main iterative improvement using an LP with block variables
    for outer in range(6):
        # compute max convolution for current f
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Build linear constraints A * v <= M
        if SCIPY_AVAILABLE:
            try:
                A = _build_A(f, block_bounds)
                b_ub = np.full(2*n-1, M, dtype=np.float64)
                c = -np.ones(A.shape[1], dtype=np.float64)   # maximize sum of masses
                res = linprog(c, A_ub=A, b_ub=b_ub,
                              bounds=[(0, None)]*A.shape[1],
                              method='highs', options={'presolve': True})
                if not res.success:
                    # fallback to random refinement if LP fails
                    break
                v = res.x
                # construct g from block masses
                g = np.empty_like(f)
                for j, (st, en) in enumerate(block_bounds):
                    blen = en - st
                    if blen > 0:
                        g[st:en] = v[j] / blen
                g = np.clip(g, 0.0, 1000.0)
                # mixing step – try a few mixing coefficients
                improved = False
                for t in (0.2, 0.4, 0.6):
                    cand = (1 - t) * f + t * g
                    cand = np.clip(cand, 0.0, 1000.0)
                    sc = _score(cand)
                    if sc < best_score:
                        best = cand
                        best_score = sc
                        improved = True
                if improved:
                    f = best.copy()
                else:
                    # still move a little towards g to allow future progress
                    f = (0.9) * f + (0.1) * g
            except Exception:
                # any unexpected error – exit the LP loop
                break
        else:
            # SciPy not available, exit loop
            break

    # Small stochastic refinement – try many random transfers
    steps = kwargs.get('num_steps', 2000)
    for _ in range(steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # avoid moving from a near-zero entry
        if best[src] < 1e-12:
            continue
        # move a random fraction up to 30% of source value
        frac = rng.uniform(0.01, 0.30)
        delta = best[src] * frac
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # Final safety clipping and conversion to plain list
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

import numpy as np

# ------------------------------------------------------------
# Optional SciPy imports (used for a compact LP improvement)
# ------------------------------------------------------------
try:
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False

# ------------------------------------------------------------
# Score function – identical to the evaluator used by the harness
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    m = conv.max()
    n = seq.size
    return float(2.0 * n * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like sequence a_i = C(2i,i)/4^i  (the classic starting point)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Build block structure for the linear‑programming relaxation
# ------------------------------------------------------------
def _make_blocks(n: int, nb: int):
    """return list of (start, end) pairs partitioning 0..n."""
    bounds = []
    start = 0
    for i in range(nb):
        end = (i + 1) * n // nb
        if i == nb - 1:
            end = n
        bounds.append((start, end))
        start = end
    return bounds


def _build_A(f: np.ndarray, blocks):
    """Sparse matrix A where A @ v <= M encodes convolution bound."""
    n = f.size
    full_len = 2 * n - 1
    rows, cols, data = [], [], []
    for col, (st, en) in enumerate(blocks):
        blen = en - st
        if blen == 0:
            continue
        mask = np.zeros(n, dtype=np.float64)
        mask[st:en] = 1.0
        a = np.convolve(f, mask)  # length 2n-1
        a = a / blen               # per unit total mass in the block
        rows.extend(range(full_len))
        cols.extend([col] * full_len)
        data.extend(a.tolist())
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, len(blocks)))

    #    inflate the tail.  This explores many (prefix, shrink, tail‑scale)

    # --------------------------------------------------------
    prefix_ratios = np.linspace(0.02, 0.5, 10)          # 2 % … 50 % of the vector
    shrink_factors = np.linspace(0.1, 0.9, 9)          # 10 % … 90 % of original height

    # --------------------------------------------------------
    steps = kwargs.get("num_steps", 2500)
    for _ in range(steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst or best[src] < 1e-12:
            continue
        # move up to 30 % of the source entry
        delta = best[src] * rng.uniform(0.01, 0.30)
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------
    # Return a plain Python list as required by the harness
    # --------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
