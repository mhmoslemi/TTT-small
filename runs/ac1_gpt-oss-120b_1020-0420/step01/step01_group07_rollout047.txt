import numpy as np

# ------------------------------------------------------------
# Convolution peak (max of full self‑convolution) via FFT
# ------------------------------------------------------------
def _conv_max(mask: np.ndarray) -> float:
    """
    Return max_{k} (mask * mask)[k] for a real 1‑D mask.
    Uses FFT; works for any length.
    """
    n = mask.size
    L = 2 * n - 1
    # real‑FFT of the zero‑padded mask
    f = np.fft.rfft(mask, n=L)
    # element‑wise product gives FFT of the convolution
    conv = np.fft.irfft(f * f, n=L)
    # numerical noise may produce tiny negatives – ignore them
    return float(conv.max())


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iters: int = 4000, **kwargs):
    """
    Return a list of non‑negative heights (length = height_sequence_1
    if present, otherwise 256) that (hopefully) yields a low AC1 bound.
    A quick simulated‑annealing on a binary mask is performed; the
    best vector found (or a simple fallback) is returned.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Problem size
    # ----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
    except NameError:
        n = 256
        parent = None

    # ----------------------------------------------------------------
    # Baseline: all‑ones vector (guaranteed feasible)
    # ----------------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # ----------------------------------------------------------------
    # Optional: use parent if it improves the baseline
    # ----------------------------------------------------------------
    if parent is not None:
        cand = np.clip(parent, 0.0, 1000.0)
        if cand.sum() >= 0.01:
            s = float(evaluate_sequence(cand.tolist()))
            if s < best_score:
                best_vec = cand
                best_score = s

    # ----------------------------------------------------------------
    # Simulated‑annealing on a binary mask
    # ----------------------------------------------------------------
    # Target cardinality – about a quarter of the length works well
    target_m = max(2, int(0.25 * n))

    # Initialise a random mask of that size
    mask = np.zeros(n, dtype=np.float64)
    init_sel = rng.choice(n, size=target_m, replace=False)
    mask[init_sel] = 1.0
    m = target_m

    # Initial score for the mask
    cur_max = _conv_max(mask)
    cur_score = 2.0 * n * cur_max / (m * m)

    if cur_score < best_score:
        best_vec = mask.copy()
        best_score = cur_score

    # Annealing schedule (linear cooling)
    T_start = 1.0
    T_end = 1e-4

    for it in range(max_iters):
        # temperature
        T = T_start * (1.0 - it / max_iters) + T_end * (it / max_iters)

        # propose a swap that keeps the cardinality fixed
        sel_idx = int(rng.choice(np.flatnonzero(mask)))          # currently 1
        unsel_idx = int(rng.choice(np.flatnonzero(1.0 - mask))) # currently 0

        # perform swap
        mask[sel_idx] = 0.0
        mask[unsel_idx] = 1.0

        new_max = _conv_max(mask)
        new_score = 2.0 * n * new_max / (m * m)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability exp(-(Δ)/T); guard against underflow
            delta = new_score - cur_score
            if T > 0.0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_score = new_score
            cur_max = new_max
            if cur_score < best_score:
                best_vec = mask.copy()
                best_score = cur_score
        else:
            # revert swap
            mask[unsel_idx] = 0.0
            mask[sel_idx] = 1.0

        # occasional random restart if progress stalls
        if (it + 1) % 1000 == 0 and rng.random() < 0.10:
            mask[:] = 0.0
            new_sel = rng.choice(n, size=target_m, replace=False)
            mask[new_sel] = 1.0

    # ----------------------------------------------------------------
    # Final safety: ensure feasibility and clip to limits
    # ----------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # The evaluator also clips internally, but we keep the vector tidy.
    return [float(x) for x in best_vec]
