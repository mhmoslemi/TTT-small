def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine n
    global height_sequence_1
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, np.ndarray):
        n = int(height_sequence_1.size)
        base = height_sequence_1.astype(np.float64).copy()
    else:
        n = 256
        base = generate_half_power_series(n)

    # Clip
    np.clip(base, 0.0, 1000.0, out=base)

    # Evaluate base
    best = base
    best_score = float(evaluate_sequence(best.tolist()))

    # parameters for perturbation
    max_block = max(1, n // 20)

    for step in range(num_steps):
        cand = best.copy()
        # Choose action
        if rng.random() < 0.7:
            # early shrink
            width_e = int(rng.integers(1, max_block + 1))
            start_e = int(rng.integers(0, max(1, n // 2 - width_e + 1)))
            factor_e = float(np.exp(rng.normal(-0.05, 0.02)))  # < 1
            cand[start_e:start_e+width_e] *= factor_e

            # tail grow
            width_t = int(rng.integers(1, max_block + 1))
            start_t = int(rng.integers(n // 2, n - width_t + 1))
            factor_t = float(np.exp(rng.normal(0.05, 0.02)))   # > 1
            cand[start_t:start_t+width_t] *= factor_t
        else:
            # simple block perturbation
            width = int(rng.integers(1, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand[start:start+width] *= factor

        np.clip(cand, 0.0, 1000.0, out=cand)

        # compute score
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(x) for x in best]

import numpy as np
from scipy import stats  # only for log‑normal sampling, optional

# ------------------------------------------------------------
# Helper: (1 - z)^(-1/2) series – the classic baseline.
# ------------------------------------------------------------
def half_power_series(length: int) -> np.ndarray:
    """Generate a_i = binom(2i,i)/4**i for i=0..length‑1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Simple two‑block perturbation:
#   – shrink a small early block (to lower the interior peak)
#   – enlarge a larger tail block (to raise the total sum)
# ------------------------------------------------------------
def two_block_perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    cand = base.copy()

    # ----- early shrink (first quarter of the sequence) -----
    max_early = max(1, n // 100)               # width up to ~1 % of n
    early_w = int(rng.integers(1, max_early + 1))
    early_start = int(rng.integers(0, max(1, n // 4 - early_w + 1)))
    # factor < 1  (log‑normal centred below 1)
    factor_e = float(np.exp(rng.normal(-0.10, 0.025)))
    cand[early_start:early_start + early_w] *= factor_e

    # ----- tail enlarge (second half of the sequence) -----
    min_tail = max_early
    max_tail = max(1, n // 5)                  # up to ~20 % of n
    tail_w = int(rng.integers(min_tail, max_tail + 1))
    tail_start = int(rng.integers(n // 2, n - tail_w + 1))
    # factor > 1  (log‑normal centred above 1)
    factor_t = float(np.exp(rng.normal(0.25, 0.05)))
    cand[tail_start:tail_start + tail_w] *= factor_t

    # safety
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
