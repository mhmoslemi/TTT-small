            np.clip(cand, 0.0, 1000.0, out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                if best_score < target:
                    return [float(x) for x in best]

    # ------------------------------------------------------------------ #
    # 2) Light stochastic hill climbing
    # ------------------------------------------------------------------ #
    max_steps = 2000  # modest number to keep runtime bounded
    for step in range(max_steps):
        if best_score < target:
            break

        op = rng.integers(0, 4)   # choose move type
        cand = best.copy()

        if op == 0:
            # scale a random tail block (log‑normal factor)
            min_len = max(1, n // 200)   # 0.5 % of n
            max_len = max(2, n // 10)    # 10 % of n
            width = int(rng.integers(min_len, max_len + 1))
            start = n - width
            factor = float(np.exp(rng.normal(0.0, 0.35)))  # ≈ log‑normal
            cand[start:] *= factor
        elif op == 1:
            # small positive bump at a random position
            idx = int(rng.integers(0, n))
            bump = max(0.0, rng.normal(0.0, 0.05))
            cand[idx] += bump
        elif op == 2:
            # swap two entries (pure permutation)
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # slight global scaling (preserves relative shape)
            factor = 1.0 + rng.normal(0.0, 0.01)
            factor = max(factor, 1e-3)
            cand *= factor

        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(x) for x in best]
