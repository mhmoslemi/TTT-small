def propose_candidate(seed=42, num_steps=8000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        n = 256
        parent = None

    # Analytic base
    base = _coefficients_binom_half(n)
    base = np.clip(base, 0.0, 1000.0)

    # Choose initial candidate
    candidates = [base]
    if parent is not None:
        candidates.append(parent)
    best = candidates[0].copy()
    best_score = float(evaluate_sequence(best.tolist()))
    for cand in candidates[1:]:
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best = cand.copy()
            best_score = s

    target = 1.50274349

    # Parametric scaling search
    idx_frac = np.arange(n) / n
    for p in (1, 2, 3):
        factor = idx_frac ** p
        for alpha in np.linspace(0.0, 5.0, 51):
            cand = best * (1.0 + alpha * factor)
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
                if best_score < target:
                    return [float(x) for x in best]

    # Optional LP refinement if cvxpy available
    if HAVE_CVX:
        max_iter_lp = 6
        t = 0.3
        for it in range(max_iter_lp):
            conv = np.convolve(best, best)
            max_b = float(conv.max())
            # select constraints: indices where conv near max
            thresh = max_b * 0.92
            ks = np.where(conv >= thresh)[0]
            # limit number of constraints
            if ks.size > 20:
                ks = ks[np.linspace(0, ks.size - 1, 20, dtype=int)]
            # Build sparse matrix
            A = _build_constraint_matrix(best, ks)
            # Solve LP
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= max_b,
                           g <= 1000.0]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            try:
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=2000, verbose=False)
            except Exception:
                break
            if g.value is None:
                break
            g_opt = np.maximum(0.0, np.asarray(g.value, dtype=np.float64))
            # convex combine
            cand = (1 - t) * best + t * g_opt
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
                if best_score < target:
                    break
            else:
                # if no improvement, reduce step size
                t *= 0.5
                if t < 1e-3:
                    break

    # Simple hill climbing fallback if still above target and steps remain
    steps = 0
    max_steps = max(num_steps - 2000, 0)
    while steps < max_steps and best_score > target:
        # Random block scaling as before
        start = int(rng.integers(0, n))
        width = int(rng.integers(1, max(2, n // 50)))
        factor = np.exp(rng.normal(0.0, 0.2))
        cand = best.copy()
        cand[start:start+width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = float(evaluate_sequence(cand.tolist()))
        steps += 1
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    return [float(x) for x in best]

def _build_constraint_matrix(f, ks):
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(ks):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    return sp.csr_matrix((data, (rows, cols)), shape=(len(ks), n))

# Optional heavy libraries
try:
    import cvxpy as cp
    import scipy.sparse as sp
    _HAVE_CVX = True
except Exception:  # pragma: no cover
    _HAVE_CVX = False


def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2, i.e. C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _build_constraint_matrix(f: np.ndarray, ks: np.ndarray) -> sp.csr_matrix:
    """Build sparse matrix A such that (A @ g)_k = sum_i f[i] * g[k-i] for k in ks."""
    n = f.size
    rows, cols, data = [], [], []
    for row, k in enumerate(ks):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # j runs over g‑indices, i = k‑j runs over f‑indices
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(row)
            cols.append(j)
            data.append(f[i])
    return sp.csr_matrix((data, (rows, cols)), shape=(len(ks), n))
