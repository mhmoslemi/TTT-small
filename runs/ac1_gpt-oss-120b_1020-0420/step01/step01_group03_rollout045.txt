import numpy as np

# ----------------------------------------------------------------------
# Helper: root‑of‑constant sequence (fallback / initialisation)
# ----------------------------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Helper: safe evaluation of a candidate sequence
# ----------------------------------------------------------------------
def _eval(seq: np.ndarray) -> float:
    """Return the AC1 score for a numpy array (list conversion required)."""
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:
        return float('inf')


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that
    improves on the classic root‑constant construction.
    The routine tries a few deterministic re‑weightings and a short
    hill‑climb; the best found sequence is returned.
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Determine length and initialise from the supplied parent if any
    # --------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # safety: enforce evaluator's clipping and positivity
        np.clip(base, 0.0, 1000.0, out=base)
    except Exception:
        n = 256
        base = _root_constant_seq(n)

    # --------------------------------------------------------------
    # Baseline score
    # --------------------------------------------------------------
    best_seq = base.copy()
    best_score = _eval(best_seq)

    # --------------------------------------------------------------
    # 1) Linear tilt: lift the tail, depress the head
    #    factor_i = 1 + s * ((2*i)/(n-1) - 1),  s >= 0

    for s in np.linspace(0.0, 0.9, 10):

        cand = base * factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _eval(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
        # early exit if target already beaten
        if best_score <= 1.50274349:
            return best_seq.tolist()

    # --------------------------------------------------------------
    # 2) Block re‑weighting: shrink an early block, enlarge the remaining
    #    block while (approximately) keeping the total sum unchanged.
    # --------------------------------------------------------------
    total_sum = float(np.sum(base))
    for block_frac in (0.3, 0.5, 0.7):
        cut = int(block_frac * n)
        if cut == 0 or cut == n:
            continue
        early_sum = float(np.sum(base[:cut]))
        tail_sum = total_sum - early_sum
        if tail_sum <= 0:
            continue
        # sweep a scaling factor for the early part
        for alpha in np.linspace(0.5, 0.99, 15):
            beta = (total_sum - alpha * early_sum) / tail_sum

            cand = base.copy()
            cand[:cut] *= alpha
            cand[cut:] *= beta
            np.clip(cand, 0.0, 1000.0, out=cand)

            if sc < best_score:

            if best_score <= 1.50274349:
