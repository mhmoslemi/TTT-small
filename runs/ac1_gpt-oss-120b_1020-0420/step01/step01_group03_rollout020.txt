def compute_score(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length()) # next power of 2 >= 2n-1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    # truncate to length 2*n-1
    conv = conv[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    # compute conv via FFT and return (score, max_val, max_idx)
    n = a.shape[0]

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = binom(2i,i)/4**i for 0 <= i < length."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    # safety clipping (problem bounds)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _next_pow_two(x: int) -> int:
    """Small helper: next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_max_score(a: np.ndarray, fft_len: int):
    """
    Compute the AC1 upper‑bound score using an FFT‑based convolution.
    Returns (score, max_value, max_index).
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv[: 2 * n - 1]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return np.inf, max_val, max_idx
    score = 2.0 * n * max_val / (sum_a * sum_a)
    return score, max_val, max_idx
