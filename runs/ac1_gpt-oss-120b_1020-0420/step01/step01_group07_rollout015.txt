def binary_mask_search(rng, n, m_target, max_allowed, max_iters):
    selected = set(rng.choice(n, size=m_target, replace=False))
    # compute conv using binary mask
    mask = np.zeros(n, dtype=np.int8)
    mask[list(selected)] = 1
    max_c = int(np.convolve(mask, mask).max())
    best_selected = selected.copy()
    best_bound = 2*n*max_c/(len(selected)**2)
    for it in range(max_iters):
        if max_c <= max_allowed:
            break
        # find sums attaining max_c
        conv = np.convolve(mask, mask)
        ks = np.where(conv == max_c)[0]
        # pick a random sum k
        k = int(rng.choice(ks))
        # find participants
        participants = [i for i in selected if (k - i) in selected]
        # choose one to remove
        if participants:
            remove = int(rng.choice(participants))
        else:
            remove = int(rng.choice(list(selected)))
        # remove
        selected.remove(remove)
        mask[remove]=0
        # optionally add a new index to keep size around target
        # choose random candidate not in selected
        add = int(rng.choice(np.setdiff1d(np.arange(n), np.array(list(selected)), assume_unique=True)))
        # update mask temporarily for evaluation
        mask[add]=1
        new_selected = selected | {add}
        # compute new max_c
        conv_new = np.convolve(mask, mask)
        new_max_c = int(conv_new.max())
        # If new_max_c <= max_c (or bound improves), keep addition; else revert
        if new_max_c <= max_c:
            selected.add(add)
            max_c = new_max_c
            # update best if bound better
            bound = 2.0*n*max_c/(len(selected)**2)
            if bound < best_bound:
                best_bound = bound
                best_selected = selected.copy()
        else:
            # revert add
            mask[add] = 0
    # After loop, output best_selected
    return best_selected
```

            if participants:
                rem = int(rng.choice(participants))
            else:
                rem = int(rng.choice(list(selected)))
            mask[rem] = 0
            selected.remove(rem)

            # attempt a cheap swap (add a new index) that does not increase the bound
            swapped = False
            for _a in range(3):
                cand = int(rng.integers(n))
                if cand in selected:
                    continue
                mask[cand] = 1
                new_score = _bound(mask, n)
                if new_score <= cur_score:
                    selected.add(cand)
                    cur_score = new_score
                    swapped = True
                    break
                mask[cand] = 0
            if not swapped:
                # no beneficial addition – just recompute the bound after deletion
                cur_score = _bound(mask, n)

            if cur_score < best_score:
                best_score = cur_score
                best_mask = mask.astype(np.float64)

        # after the hill‑climb try to enlarge the mask while staying under TARGET
        while cur_score <= TARGET and len(selected) < n:
            cand = int(rng.integers(n))
            if cand in selected:
                continue
            mask[cand] = 1
            new_score = _bound(mask, n)
            if new_score <= TARGET:
                selected.add(cand)
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_mask = mask.astype(np.float64)
            else:
                mask[cand] = 0
                break

        if best_score <= TARGET:
            break

    # ----------------------------------------------------
    # Power‑law fallback (cheap sweep) if needed
    # ----------------------------------------------------
    if best_score > TARGET:
        base = np.arange(1, n + 1, dtype=np.float64)
        for p in np.linspace(0.45, 0.80, 71):
            cand = base ** (-p)
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_score:
                best_score = score
                best_mask = cand

    return best_mask.tolist()
