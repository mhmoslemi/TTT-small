    n = v.size
    if n == 0:

            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, n - width + 1))
            block = cur.copy()
            factor = np.exp(rng.normal(0.0, 0.12))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, np.inf)
            block = block / block.sum()
            block_score = _fast_score(block)
            if block_score < cur_score:
                cur = block
                cur_score = block_score
                if cur_score < best_score:
                    best = block.copy()
                    best_score = cur_score

    # ------------------------------------------------------------------
    # final sanity check using the official evaluator; fallback if needed
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = np.full_like(best, 1.0 / n)

    return [float(v) for v in best]
