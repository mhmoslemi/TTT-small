def _coefficients_binom_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0*i - 1.0) / (2.0*i)
    return coeffs

import numpy as np

def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2 : C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _score(vec: np.ndarray) -> float:
    """Exact objective used by the evaluator (avoids list conversion)."""
    if np.any(vec < 0) or np.any(np.isnan(vec)) or np.any(np.isinf(vec)):
        return np.inf
    s = vec.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(vec, vec)          # full convolution
    return float(2.0 * vec.size * b.max() / (s * s))


def propose_candidate(seed: int = 42, num_steps: int = 10000, **kwargs) -> list:
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine starts from the classic (1‑z)^‑1/2 coefficients and
    performs a specialised hill‑climbing that transfers mass from early
    (large) entries to many late (tiny) entries.  The move pattern keeps
    the total sum constant, reduces the peak of the self‑convolution,
    and therefore lowers the evaluated bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # 1) problem size and initial sequences
    # ------------------------------------------------------------------ #
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
    else:
        n = 256
        parent = None

    base = np.clip(_coefficients_binom_half(n), 0.0, 1000.0)

    # ------------------------------------------------------------------ #
    # 2) initialise best candidate (choose the better of base / parent)
    # ------------------------------------------------------------------ #
    best = base.copy()
    best_score = _score(best)
    if parent is not None:
        parent_score = _score(parent)
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ------------------------------------------------------------------ #
    # 3) region definitions for directed moves
    # ------------------------------------------------------------------ #
    early_cut = max(1, int(n * 0.12))            # first ~12 % indices
    late_start = max(early_cut + 1, int(n * 0.80))   # last ~20 % indices
    early_idx = np.arange(early_cut, dtype=np.int32)
    late_idx  = np.arange(late_start, n, dtype=np.int32)

    # ------------------------------------------------------------------ #
    # 4) hill‑climbing loop
    # ------------------------------------------------------------------ #
    cur = best.copy()            # start proposals from the current best
    target = 1.50274349

    for step in range(num_steps):
        if best_score <= target:
            break

        op = rng.integers(0, 4)  # 0: mass move, 1: block scale, 2: swap, 3: bump

        if op == 0:
            # ---------------------------------------------------------- #
            # Move a fraction of an early entry to many late entries.
            # ---------------------------------------------------------- #
            i = int(rng.choice(early_idx))
            donor_val = cur[i]
            if donor_val <= 0.0:
                continue
            frac = float(rng.uniform(0.02, 0.30))       # up to 30 % of donor
            delta = donor_val * frac                   # total mass to relocate
            # number of receivers (spread the mass)
            max_recv = min(300, len(late_idx))
            m = int(rng.integers(5, max_recv + 1))
            receivers = rng.choice(late_idx, size=m, replace=False)
            cand = cur.copy()
            cand[i] -= delta
            cand[receivers] += delta / m
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif op == 1:
            # ---------------------------------------------------------- #
            # Scale a random modest block (log‑normal factor).
            # ---------------------------------------------------------- #
            start = int(rng.integers(0, n))
            max_len = max(1, n // 10)                 # block at most 10 % of length
            length = int(rng.integers(1, max_len + 1))
            end = min(n, start + length)
            factor = float(np.exp(rng.normal(0.0, 0.12)))  # modest perturbation
            cand = cur.copy()
            cand[start:end] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif op == 2:
            # ---------------------------------------------------------- #
            # Swap an early entry with a late one.
            # ---------------------------------------------------------- #
            i = int(rng.choice(early_idx))
            j = int(rng.choice(late_idx))
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]

        else:
            # ---------------------------------------------------------- #
            # Small positive bump on a random coordinate.
            # ---------------------------------------------------------- #
            idx = int(rng.integers(0, n))
            bump = max(0.0, float(rng.normal(0.0, 0.04)))
            cand = cur.copy()
            cand[idx] += bump
            np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            cur = cand          # continue from the improved point

    # ------------------------------------------------------------------ #
    # 5) final lightweight refinement: tiny mixing with the analytical base
    # ------------------------------------------------------------------ #
    for t in (0.0, 0.02, 0.05, 0.10):
        mix = (1.0 - t) * best + t * base
        np.clip(mix, 0.0, 1000.0, out=mix)
        sc = _score(mix)
        if np.isfinite(sc) and sc < best_score:
            best = mix
            best_score = sc

    return [float(x) for x in best]
