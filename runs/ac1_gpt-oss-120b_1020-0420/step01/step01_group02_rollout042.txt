def propose_candidate(seed=42, num_iters=200, step_alpha=0.1, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        base = generate_half_power_series(n)
    # Ensure nonnegative (already)
    # Normalize sum to 1
    a = base.copy()
    sum_a = a.sum()
    if sum_a < 1e-12:
        a = np.full_like(a, 1.0/a.size)
        sum_a = a.sum()
    a /= sum_a

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    # Iterate
    for it in range(num_iters):
        # compute convolution
        b = np.convolve(a, a)
        max_b = b.max()
        m = b.argmax()
        # compute gradient component g_i = 2*a_{m-i}
        # we can compute: create array g of zeros length n.
        g = np.zeros_like(a)
        # indices i where 0<=m-i<n => i in [m-(n-1), m]
        i_start = max(0, m-(len(a)-1))
        i_end = min(len(a)-1, m)
        # For i in [i_start,i_end], set g[i] = 2*a[m-i]
        idx_i = np.arange(i_start, i_end+1)
        idx_j = m - idx_i
        g[idx_i] = 2.0 * a[idx_j]
        # Determine step magnitude: we reduce a where g is large
        # step = step_alpha * g * (a.mean())
        # Or better: compute reduction proportion: delta = step_alpha * (g / g.max())
        if g.max() > 0:
            reduction = step_alpha * (g / g.max())
            # subtract but not below 0
            a_new = a * (1 - reduction)
            # ensure non-negative
            a_new = np.maximum(a_new, 0.0)
        else:
            a_new = a.copy()
        # Renormalize to keep sum 1
        sum_new = a_new.sum()
        if sum_new > 0:
            a_new /= sum_new
        else:
            # fallback: random perturbation
            a_new = a.copy()
        # Evaluate
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
        # Update a for next iteration (use a_new).
        a = a_new

    # Return best as list
    return [float(x) for x in best_a]

def subgradient_step(a, eta):
    # compute convolution and max shift index m
    b = np.convolve(a, a)
    max_b = b.max()
    m = b.argmax()
    # gradient = 2 * reversed a shifted by m
    n = len(a)
    grad = np.zeros_like(a)
    i_start = max(0, m - (n - 1))
    i_end = min(n - 1, m)
    idx_i = np.arange(i_start, i_end + 1)
    idx_j = m - idx_i
    grad[idx_i] = 2.0 * a[idx_j]  # consistent with deriv
    # take step opposite direction
    a_new = a - eta * grad
    # project onto nonnegative
    a_new = np.maximum(a_new, 0.0)
    # renormalize sum to 1
    s = a_new.sum()
    if s < 1e-12:
        a_new = a / a.sum()  # fallback
    else:
        a_new /= s
    return a_new, max_b
```

def generate_half_power_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def fft_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n - 1]
    return conv

def random_block_perturbation(base, rng):
    n = base.size
    if n <= 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    # Normalise to unit L1 norm (scale invariance of the bound)
    total = base.sum()
    if total <= 0.0:
        base = np.ones_like(base) / base.size
    else:
        base = base / total

    a = base.copy()
    n = a.size

    # Helper to compute bound (sum == 1)
    def bound_from_seq(seq):
        conv = fft_convolve(seq)
        maxb = conv.max()
        return 2.0 * n * maxb

    best = a.copy()
    best_score = bound_from_seq(best)

    for it in range(num_iters):
        # Occasionally perform a random block perturbation for exploration
        if rng.random() < 0.15:
            cand = random_block_perturbation(a, rng)
            cand = cand / cand.sum()
        else:
            # Gradient step aimed at reducing the maximal convolution entry
            conv = fft_convolve(a)
            maxb = conv.max()
            m = int(conv.argmax())

            # Compute sub‑gradient of max(b) w.r.t. a
            grad = np.zeros_like(a)
            i_start = max(0, m - (n - 1))
            i_end = min(n - 1, m)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            idx_j = m - idx_i
            grad[idx_i] = 2.0 * a[idx_j]

            # Determine a stepsize that keeps a non‑negative
            pos = grad > 0
            if not np.any(pos):
                # nothing to change
                cand = a.copy()
            else:
                ratios = np.full_like(a, np.inf, dtype=np.float64)
                ratios[pos] = a[pos] / grad[pos]
                max_eta = ratios[pos].min()
                # Schedule decreasing steps
                step_factor = 0.9 * (1.0 - it / num_iters)
                eta = step_factor * max_eta
                cand = a - eta * grad
                # Numerical safety
                cand = np.maximum(cand, 0.0)

            # Renormalise to keep L1 = 1
            s = cand.sum()
            if s > 0.0:
                cand /= s
            else:
                cand = a.copy()

        # Evaluate bound
        score = bound_from_seq(cand)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand.copy()

        # Accept the move for the next iteration (simple hill‑climb)
        a = cand

    # Return as plain Python list
    return [float(x) for x in best]

import numpy as np

# --------------------------------------------------------------------
# Generate the (1 - z)^(-1/2) power series (the classic baseline)
# --------------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return the first ``length`` terms of (1 - z)^(-1/2) series."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# --------------------------------------------------------------------
# FFT based full convolution (linear time, low memory)
# --------------------------------------------------------------------
def fft_convolve(x: np.ndarray) -> np.ndarray:
    """Return np.convolve(x, x) using real FFT (no extra memory)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------
# Evaluate the AC1 bound for a sequence whose L1 norm is 1.
# (The evaluator uses sum^2 in the denominator; with sum == 1 the
# bound simplifies to 2 * n * max(convolution).)
# --------------------------------------------------------------------
def bound_of_sequence(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = fft_convolve(seq)
    maxb = conv.max()
    return 2.0 * n * maxb


# --------------------------------------------------------------------
# Main entry point required by the harness
# --------------------------------------------------------------------
def propose_candidate(seed: int = 42, num_iters: int = 2000, **kwargs):
    """
    Return a list of non‑negative step‑function heights whose AC1 bound
    (as computed by evaluate_sequence) is as low as possible.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Determine length and starting vector (parent or analytical)
    # ----------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size == 0:
            parent = None
    else:
        parent = None

    if parent is not None:
        # normalise parent to L1 = 1
        if parent.sum() == 0.0:
            parent = None
        else:
            a0 = np.clip(parent, 0.0, 1000.0)
            a0 = a0 / a0.sum()
    else:
        a0 = None

    n = parent.shape[0] if parent is not None else 256
    # analytical baseline
    analytic = generate_half_power_series(n)
    analytic = analytic / analytic.sum()

    # pick the better of the two as the starting point
    if a0 is not None:
        if bound_of_sequence(a0) < bound_of_sequence(analytic):
            a = a0.copy()
        else:
            a = analytic.copy()
    else:
        a = analytic.copy()

    # ----------------------------------------------------------------
    # Keep the best candidate seen so far
    # ----------------------------------------------------------------
    best = a.copy()
    best_score = bound_of_sequence(best)

    # ----------------------------------------------------------------
    # Iterative improvement: move mass from the leading part toward the tail
    # by adding the sub‑gradient of the maximal convolution entry.
    # The step size shrinks linearly to keep the search stable.
    # ----------------------------------------------------------------
    for it in range(num_iters):
        # convolution and index of the maximal entry
        conv = fft_convolve(a)
        max_b = conv.max()
        m = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        n_len = a.shape[0]
        i_start = max(0, m - (n_len - 1))
        i_end = min(n_len - 1, m)
        idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
        idx_j = m - idx_i
        grad[idx_i] = 2.0 * a[idx_j]

        # Choose a modest step that never makes a negative.
        # Scale by the smallest non‑zero entry to keep the update tiny.
        pos = a > 0
        if not np.any(pos):
            break
        min_nonzero = a[pos].min()
        grad_max = grad.max()
        if grad_max <= 0.0:
            break

        # Linear schedule decreasing from 1 to 0
        step_factor = (num_iters - it) / num_iters
        eta = step_factor * min_nonzero / (grad_max + 1e-12)

        # Update, re‑normalise to keep L1 = 1
        cand = a + eta * grad
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            cand = a.copy()
        else:
            cand /= cand_sum

        # Evaluate bound
        score = bound_of_sequence(cand)

        # Keep the best ever seen
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand.copy()

        # Use the new candidate for the next iteration (hill‑climb)
        a = cand

    # ----------------------------------------------------------------
    # Return the best sequence as a plain Python list of floats.
    # ----------------------------------------------------------------
    return [float(x) for x in best]
