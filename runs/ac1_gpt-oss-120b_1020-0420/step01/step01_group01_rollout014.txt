# ------------------------------------------------------------
# Catalan‑like base sequence (used when no parent is supplied)
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Fast AC1 bound using FFT (identical to evaluate_sequence)
# ------------------------------------------------------------
def _score_fft(seq: np.ndarray, L: int) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))²  –‑ FFT based, linear memory."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))
