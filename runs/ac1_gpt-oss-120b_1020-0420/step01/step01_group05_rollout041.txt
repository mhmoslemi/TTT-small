import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound using FFT.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1                     # length of full convolution
    nfft = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                     # pointwise product → convolution in time domain
    c = np.fft.irfft(fc, n=nfft)[:m]  # keep only the needed part
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

    """
    rng = np.random.default_rng(seed)

    return [float(x) for x in best_seq.tolist()]
