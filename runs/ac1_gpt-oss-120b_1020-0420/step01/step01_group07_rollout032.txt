Define function fast_score(a):

    n = len(a)
    n_conv = 2*n - 1
    n_fft = 1 << ((n_conv-1).bit_length())  # smallest power of 2 >= n_conv

    fa = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(fa * fa, n_fft)[:n_conv]
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)

def refine_candidate(a, max_iter=300, tol=1e-6):
    best_score = fast_score(a)
    for it in range(max_iter):
        n = a.shape[0]
        # conv, max_idx
        n_conv = 2*n - 1
        n_fft = 1 << ((n_conv-1).bit_length())
        fa = np.fft.rfft(a, n_fft)
        conv = np.fft.irfft(fa * fa, n_fft)[:n_conv]
        max_idx = int(np.argmax(conv))
        max_c = conv[max_idx]
        sum_a = a.sum()
        # Compute gradient g
        idx = np.arange(n)
        k_minus_idx = max_idx - idx
        mask = (k_minus_idx >= 0) & (k_minus_idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[k_minus_idx[mask]]
        grad = (2.0 * n / (sum_a * sum_a)) * g - (4.0 * n * max_c / (sum_a**3)) * np.ones_like(a)
        # Determine step size
        # Use adaptive step: start step size = 0.01 * sum_a / n
        step = 0.01 * (sum_a / n + 1e-12)
        # Try improving step
        improved = False
        for _ in range(10):
            a_new = a - step * grad
            # project
            np.clip(a_new, 0.0, 1000.0, out=a_new)
            new_score = fast_score(a_new)
            if new_score < best_score - 1e-12:
                a = a_new
                best_score = new_score
                improved = True
                step *= 1.2  # try larger step next time
                break
            else:
                step *= 0.5
        if not improved:
            # maybe random perturbation to escape
            # random index
            i = np.random.randint(0, n)
            delta = (np.random.rand() - 0.5) * step * 10
            a_new = a.copy()
            a_new[i] = max(0.0, min(1000.0, a_new[i] + delta))
            new_score = fast_score(a_new)
            if new_score < best_score:
                a = a_new
                best_score = new_score
            else:
                # break if no improvement for several iterations
                pass
        # termination condition if improvement tiny
        if step < 1e-12:
            break
    return a

import numpy as np

# ------------------------------------------------------------
# Fast AC1 bound calculator (uses FFT for O(n log n) work)
# ------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector a."""
    n = a.shape[0]
    n_conv = 2 * n - 1
    # next power of two for efficient FFT
    n_fft = 1 << ((n_conv - 1).bit_length())
    fa = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(fa * fa, n_fft)[:n_conv]
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a == 0.0:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)


# ------------------------------------------------------------
# Pair‑count for a binary mask (ordered pairs with equal sum)
# ------------------------------------------------------------
def pair_counts(selected: set, n: int) -> np.ndarray:
    """
    selected : set of indices (0 … n‑1) where the coefficient is 1
    returns   : np.ndarray of length 2*n‑1 containing ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt


# ------------------------------------------------------------
# Gradient‑based refinement for a real‑valued candidate
# ------------------------------------------------------------
def refine_candidate(a_init: np.ndarray,
                     rng: np.random.Generator,
                     max_iter: int = 250) -> np.ndarray:
    """Run a simple projected gradient descent to lower the bound."""
    a = a_init.copy()
    n = a.shape[0]
    n_conv = 2 * n - 1
    n_fft = 1 << ((n_conv - 1).bit_length())

    best = a.copy()
    best_score = fast_score(best)

    for _ in range(max_iter):
        # full convolution via FFT
        fa = np.fft.rfft(best, n_fft)
        conv = np.fft.irfft(fa * fa, n_fft)[:n_conv]
        max_idx = int(np.argmax(conv))
        max_c = conv[max_idx]
        sum_a = best.sum()
        if sum_a == 0.0:
            break

        # gradient of the maximal term
        idx = np.arange(n)
        k_minus = max_idx - idx
        mask = (k_minus >= 0) & (k_minus < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[k_minus[mask]]

        # full gradient of the objective
        grad = (2.0 * n / (sum_a * sum_a)) * g - \
               (4.0 * n * max_c / (sum_a ** 3)) * np.ones_like(best)

        # simple back‑tracking line search
        step = 0.01 * (sum_a / n + 1e-12)
        improved = False
        for _ in range(10):
            cand = best - step * grad
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = fast_score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                step *= 1.2
                improved = True
                break
            step *= 0.5
        if not improved:
            # random jitter to escape plateaus
            i = rng.integers(0, n)
            jitter = (rng.random() - 0.5) * step * 12.0
            cand = best.copy()
            cand[i] = max(0.0, min(1000.0, cand[i] + jitter))
            sc = fast_score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
        # termination when step becomes negligible
        if step < 1e-12:
            break

    return best
