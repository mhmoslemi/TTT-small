Define pairwise_transfer:

def pairwise_transfer(base, rng, scale):
    cand = base.copy()
    n = cand.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    # Choose amount based on scale * max(cand)
    max_val = cand.max()
    delta = float(scale * max_val * rng.random())
    # ensure we do not make j negative
    if cand[j] < delta:
        delta = cand[j]  # take all.
    cand[i] += delta
    cand[j] -= delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def compute_score(a):
    n = len(a)
    # compute convolution using FFT
    n2 = 2*n - 1
    nfft = 1 << ((n2 - 1).bit_length())
    fft_a = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(fft_a * fft_a, nfft)[:n2]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

import numpy as np

# ------------------------------------------------------------
# Basic half‑power series ( (1‑z)^(-1/2) coefficients )
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

# Three cheap perturbation operators (all O(n) memory)
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.1)))  # can be <1 or >1
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def tail_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Heavy scaling of a block located in the last 30 % of the array."""
    n = base.size
    if n == 1:
        return base.copy()

    else:
        n = 1024
        base = generate_half_power_series(n)
        np.clip(base, 0.0, 1000.0, out=base)

    best = base.copy()
    best_score = compute_score(best)

    # ----------------------------------------------------------------
    # Simple hill‑climbing with three inexpensive mutation operators
    # ----------------------------------------------------------------
    for _ in range(num_steps):
        r = rng.random()
        if r < 0.5:
            cand = random_block_perturbation(best, rng)
        elif r < 0.8:
            cand = tail_perturbation(best, rng)
        else:
            cand = pairwise_transfer(best, rng, scale=0.2)

        cand_score = compute_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

    # ----------------------------------------------------------------
    # Final safety check with the official evaluator; fall back if needed
    # ----------------------------------------------------------------
    final = [float(x) for x in best]
    val = float(evaluate_sequence(final))
    if not np.isfinite(val):
        fallback = generate_half_power_series(n)
        np.clip(fallback, 0.0, 1000.0, out=fallback)
        final = [float(x) for x in fallback]

    return final
