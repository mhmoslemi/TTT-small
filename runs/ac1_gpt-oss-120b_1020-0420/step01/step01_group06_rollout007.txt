def propose_candidate(seed=42, **kwargs):
    # try parent height_sequence_1 if valid
    if 'height_sequence_1' in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                # evaluate parent
                best = np.clip(arr, 0.0, 1000.0)
                best_score = float(evaluate_sequence(best.tolist()))
            else:
                raise ValueError
        except Exception:
            best = None
            best_score = np.inf
    else:
        best = None
        best_score = np.inf

    # fallback uniform length 256
    def uniform_candidate(m):
        return np.full(m, 1.0, dtype=np.float64)

    n = best.size if best is not None else 256
    cand = uniform_candidate(n)
    score = float(evaluate_sequence(cand.tolist()))
    if score < best_score:
        best = cand
        best_score = score

    rng = np.random.default_rng(seed)
    # try a few power laws
    indices = np.arange(1, n+1, dtype=np.float64)
    for a in np.linspace(0.01, 1.0, 20):
        w = indices**(-a)
        w = w / w.sum()
        sc = float(evaluate_sequence(w.tolist()))
        if sc < best_score:
            best = w
            best_score = sc

    # try random perturbations of best
    for _ in range(100):
        if best is None:
            cur = uniform_candidate(n)
        else:
            cur = best.copy()
            # pick block
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n//50))
            stop = min(n, start+width)
            cur[start:stop] *= np.exp(rng.normal(0, 0.1, size=stop-start))
            total = cur.sum()
            if total > 0:
                cur /= total
        sc = float(evaluate_sequence(cur.tolist()))
        if sc < best_score:
            best = cur
            best_score = sc

    # Return best as list
    return [float(v) for v in best]

import numpy as np

# --------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# --------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    """Return the AC1 bound for a numpy vector."""
    # evaluate_sequence expects a plain Python list
    return float(evaluate_sequence(vec.tolist()))


def _uniform(length: int) -> np.ndarray:
    """Uniform vector of given length (any positive constant works)."""
    return np.full(length, 1.0, dtype=np.float64)


def _power_law(length: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-alpha), normalised to sum = 1."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    w = np.power(idx, -alpha)
    total = w.sum()
    if total == 0.0:
        return _uniform(length)
    w /= total
    return w


def _random_block_perturb(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation on a random contiguous block."""
    n = x.size
    start = rng.integers(0, n)
    width = rng.integers(1, max(2, n // 50))
    stop = min(n, start + width)
    y = x.copy()
    noise = np.exp(rng.normal(0.0, 0.1, size=stop - start))
    y[start:stop] *= noise
    s = y.sum()
    if s > 0.0:
        y /= s
    return y

# Main entry point required by the harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound.  The strategy is simple:
      * Use the provided parent `height_sequence_1` if it is valid.
      * Compare it to a uniform baseline (objective ≈ 2).
      * Explore a coarse family of power‑law profiles.
      * Perform a short stochastic local search around the current best.
    The best candidate found is returned as a plain Python list.
    """
    # ------------------------------------------------------------------
    # 1.  Initialise from parent if possible
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    if "height_sequence_1" in globals():
        try:
            parent = np.asarray(height_sequence_1, dtype=np.float64)
            if (
                parent.ndim == 1
                and parent.size > 0
                and np.isfinite(parent).all()
            ):
                parent = np.clip(parent, 0.0, 1000.0)
                sc = _score(parent)
                if np.isfinite(sc):
                    best = parent
                    best_score = sc
        except Exception:
            # any problem – ignore the parent
            pass

    # ------------------------------------------------------------------
    # 2.  Uniform baseline (theoretical lower bound is 2)
    # ------------------------------------------------------------------
    n = best.size if best is not None else 256
    uniform = _uniform(n)
    sc = _score(uniform)
    if np.isfinite(sc) and sc < best_score:
        best = uniform
        best_score = sc

    # ------------------------------------------------------------------
    # 3.  Scan a modest power‑law family
    # ------------------------------------------------------------------
    alphas = np.linspace(0.05, 0.95, 19)  # 0.05 … 0.95
    for a in alphas:
        cand = _power_law(n, a)
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 4.  Short stochastic local search around the current best
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for _ in range(120):
        if best is None:
            cur = _uniform(n)
        else:
            cur = _random_block_perturb(best, rng)
        sc = _score(cur)
        if np.isfinite(sc) and sc < best_score:
            best = cur
            best_score = sc

    # ------------------------------------------------------------------
    # 5.  Ensure a valid non‑empty fallback (should never happen)
    # ------------------------------------------------------------------
    if best is None:
        best = _uniform(n)

    # Return as plain Python list of floats
    return [float(v) for v in best]
