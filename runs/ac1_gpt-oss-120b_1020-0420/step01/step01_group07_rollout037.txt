def _pair_counts(selected, n):
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2*i] += 1
        for b in range(a+1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2
    return cnt

import numpy as np

# ----------------------------------------------------------------------
# Helper: ordered‑pair counts for a binary mask (sparse representation)
# ----------------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices (0 … n‑1) where the coefficient is non‑zero (1)
    Returns   : np.ndarray of length 2*n‑1 with ordered‑pair counts.
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt


def _bound_from_cnt(cnt, m, n):
    """
    Fast AC1 bound for a binary vector whose 1‑entries are described by `cnt`.
    m : number of non‑zero entries
    n : length of the full vector
    """
    max_c = int(cnt.max())
    return 2.0 * n * max_c / (m * m)

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step‑function (binary mask) that minimises the
    AC1 upper bound defined by `evaluate_sequence`.  The routine combines a
    greedy pair‑count optimiser with a cheap power‑law fallback.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initialise a fallback candidate
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
            best_vec = np.clip(parent, 0.0, 1000.0)
            best_score = float(evaluate_sequence(best_vec.tolist()))
        else:
            raise Exception()
    except Exception:
        n = 256
        best_vec = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_vec.tolist()))

    TARGET = 1.50274349            # desired bound
    TRIALS = 10                    # independent random starts
    MAX_ITER = 1000                # swaps per trial
    MAX_ADDS = 500                 # greedy additions per trial

    # ------------------------------------------------------------------
    # Binary‑mask optimiser (matrix‑free, linear memory)
    # ------------------------------------------------------------------
    for _ in range(TRIALS):
        # start with a moderately sized random subset
        m0 = max(80, int(0.06 * n))          # about 6 % of the indices
        m0 = min(m0, n)
        selected = set(rng.choice(n, size=m0, replace=False))
        cnt = _pair_counts(selected, n)
        m = len(selected)

        # iterative swaps targeting the current maximum count
        for __ in range(MAX_ITER):
            max_c = int(cnt.max())
            bound = 2.0 * n * max_c / (m * m)
            if bound <= TARGET:
                break

            # pick a sum where the maximum occurs
            ks = np.flatnonzero(cnt == max_c)
            k = int(rng.choice(ks))

            # indices that participate in that sum
            involved = [i for i in selected if (k - i) in selected]
            if involved:
                i_remove = int(rng.choice(involved))
            else:
                i_remove = int(rng.choice(list(selected)))

            # random index to insert (ensure it is not already selected)
            i_add = int(rng.integers(0, n))
            while i_add in selected:
                i_add = int(rng.integers(0, n))

            # simulate the swap on a copy of the count array
            cnt2 = cnt.copy()

            # remove i_remove
            for t in selected:
                if t == i_remove:
                    cnt2[2 * i_remove] -= 1
                else:
                    cnt2[i_remove + t] -= 2

            # add i_add

                cnt2[i_add + t] += 2
