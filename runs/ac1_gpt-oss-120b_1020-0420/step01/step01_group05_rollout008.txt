```
def _conv_max_and_index(a):
    n = a.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    k = int(np.argmax(c))
    return float(c[k]), k
```

def _b_for_max(a, k):

    b = np.empty_like(a)
    if k < n:

import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the AC1 bound (scale‑invariant)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using an FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------------------
# Helper: full convolution and position of its maximum
# ----------------------------------------------------------------------
def _conv_max_info(a: np.ndarray):
    """Return (max value, index of max) of conv(a,a)."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    idx = int(np.argmax(c))
    return float(c[idx]), idx


# ----------------------------------------------------------------------
# Helper: gradient of the objective for a given active index k
# ----------------------------------------------------------------------
def _gradient_step(a: np.ndarray, M: float, k: int, eta: float) -> np.ndarray:
    """Projected gradient step for the bound (scale invariant)."""
    n = a.shape[0]
    S = float(np.sum(a))
    if S == 0.0:
        return a.copy()

    coeff = 4.0 * n / (S * S)
    const = 4.0 * n * M / (S * S * S)

    # build b_i = a_{k-i} where defined, else 0
    b = np.empty_like(a)
    if k < n:
        b[:k + 1] = a[k::-1]
        if k + 1 < n:
            b[k + 1:] = 0.0
    else:
        start = k - (n - 1)
        b[:start] = 0.0
        b[start:] = a[start:][::-1]

    grad = coeff * b - const
    new_a = a - eta * grad
    new_a = np.maximum(new_a, 0.0)

    # re‑normalise to keep the scale (objective is scale invariant)
    total = float(np.sum(new_a))
    if total > 0.0:
        new_a /= total
    else:
        new_a = np.ones_like(a) / n

    # enforce monotone increase (optimal by the rearrangement inequality)
    new_a.sort()                     # ascending → smallest at start, largest at end
    return new_a
