import numpy as np

# ------------------------------------------------------------
# Convolution (FFT‑based, linear)
# ------------------------------------------------------------
def _conv(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution seq * seq using FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())       # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv_full = np.fft.irfft(fft_seq * fft_seq, n=size)
    return conv_full[: 2 * n - 1]


# ------------------------------------------------------------
# Score identical to the evaluator (but faster)
# ------------------------------------------------------------
def _score_seq(seq: np.ndarray) -> float:
    """2 * n * max(conv) / (sum**2) with clipping like evaluate_sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    m = _conv(seq).max()
    return float(2.0 * seq.shape[0] * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence – exact optimum for M=1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

            n = parent.size
        else:

    else:
        n = 256

    # ------------------------------------------------------------------

    # sequence (both are valid candidates).

    cand_catalan = _catalan_seq(n)
    best = cand_catalan

    if "height_sequence_1" in globals():

        p_score = _score_seq(parent_seq)
        if p_score < best_score:
            best = parent_seq.copy()
            best_score = p_score

    # ------------------------------------------------------------------

    # scaling (or uniformly raising) the remaining tail.  Scan a grid
    # of prefix lengths and shrink factors.
    # ------------------------------------------------------------------
    prefix_ratios = np.linspace(0.02, 0.50, 10)      # 2 % … 50 % of n
    shrink_factors = np.linspace(0.0, 1.0, 51)       # from 0 % to 100 % in 2 % steps

    for r in prefix_ratios:
        k = max(1, int(round(r * n)))               # size of early block
        if k >= n:
            continue
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue

        for f in shrink_factors:
            # ---- multiplicative tail scaling ---------------------------------
            new_early = early * f
            delta = early_sum - new_early.sum()          # mass removed from early block
            if delta < 0:
                continue
            # scale tail uniformly to keep total mass unchanged
            g = 1.0 + delta / tail_sum
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)

                best_score = sc

                tail_sum = tail.sum()

                    tail_sum = tail.sum()

        sc = _score_seq(cand)

            best_score = sc

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
