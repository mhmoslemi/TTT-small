import numpy as np

# ----------------------------------------------------------------------
# Helper: fast exact AC1 score (no FFT needed – the max of the convolution
# for non‑negative vectors is always the sum of squares).
# ----------------------------------------------------------------------
def _ac1_score(vec: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    max_c = np.dot(vec, vec)          # = sum_i v_i^2   (max of full convolution)
    return float(2.0 * n * max_c / (s * s))
