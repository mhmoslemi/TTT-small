        # overall gradient of the AC1 upper bound
        coeff = 4.0 * n / (sum_a ** 2)
        g = coeff * gM - (4.0 * n * M / (sum_a ** 3)) * np.ones_like(a)

        # projected gradient step
        eta = step0 / np.sqrt(it + 1.0)
        a_new = a - eta * g
        a_new = np.clip(a_new, 0.0, 1000.0)
        total = a_new.sum()
        if total == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / total

        # occasional jitter to escape flat regions
        if (it + 1) % 120 == 0:
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            a_new = a_new * noise
            a_new = a_new / a_new.sum()

        a = a_new

        # cheap score using the already computed M
        sc = 2.0 * n * M / (sum_a ** 2)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()

    # ------------------------------------------------------------------
    # 6.  Light random block perturbations as a final polish.
    # ------------------------------------------------------------------
    for _ in range(200):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 100))
        stop = min(n, start + width)

        cand = best_vec.copy()
        noise = np.exp(rng.normal(0.0, 0.05, size=stop - start))
        cand[start:stop] *= noise
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total > 0.0:
            cand = cand / total
            sc = _score(cand)
            if sc < best_score:
                best_vec = cand
                best_score = sc

    # ------------------------------------------------------------------
    # 7.  Return the best candidate (as a plain Python list).
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        # absolute safety net – uniform distribution
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
