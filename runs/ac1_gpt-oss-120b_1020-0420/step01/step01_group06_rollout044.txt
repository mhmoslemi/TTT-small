def local_optimize(a, init_step=0.1, max_iter=500, patience=20):
    step = init_step
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2 * n * np.convolve(a, a).max()
    no_improve = 0
    for i in range(max_iter):
        conv = np.convolve(a, a)
        max_b = conv.max()
        kmax = conv.argmax()
        # gradient
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2 * a[idx[mask]]
        # step update
        cand = a - step * grad
        cand = np.clip(cand, 0.0, None)
        total = cand.sum()
        if total > 0:
            cand /= total
        # compute score (fast)
        cand_conv = np.convolve(cand, cand)
        cand_score = 2 * n * cand_conv.max()
        if cand_score < best_score - 1e-12:
            best_a = cand.copy()
            best_score = cand_score
            a = cand
            no_improve = 0
        else:
            # maybe reject but still continue descent
            a = cand
            no_improve += 1
            if no_improve >= patience:
                step *= 0.5
                no_improve = 0
                # restrict step not too small
                if step < 1e-6: break
    return best_a, best_score

import numpy as np

def _normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative, renormalise to sum 1 (fallback to uniform)."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        v = np.ones_like(v, dtype=np.float64)
        s = v.sum()
    return v / s

def _conv_max(v: np.ndarray):
    """Return (max value, index of max) of the linear convolution v*v."""
    c = np.convolve(v, v)
    return c.max(), int(c.argmax())

def _fast_score(v: np.ndarray, n: int) -> float:
    """Score = 2·n·max(v*v) for a vector already normalised to sum 1."""
    max_b, _ = _conv_max(v)
    return 2.0 * n * max_b

def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 upper bound
    is hopefully below the target 1.50274349.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # 1.  Choose a starting vector (parent if present, else a simple fallback)
    # ------------------------------------------------------------------ #
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and np.isfinite(parent).all():
            parent = _normalize(parent)
        else:
            parent = None
    else:
        parent = None

    if parent is None:
        # a generic fallback – 256 elements of a simple decreasing shape
        n_fallback = 256
        i = np.arange(n_fallback, dtype=np.float64) + 1.0
        parent = np.power(i, -0.5)
        parent = _normalize(parent)

    n = int(parent.size)
    best = parent.copy()
    best_score = _fast_score(best, n)

    # ------------------------------------------------------------------ #
    # 2.  Helper that evaluates a candidate (fast, no external evaluator)
    # ------------------------------------------------------------------ #
    def _try(candidate: np.ndarray):
        """Update global best if candidate improves the score."""
        nonlocal best, best_score
        sc = _fast_score(candidate, n)
        if np.isfinite(sc) and sc < best_score:
            best = candidate.copy()
            best_score = sc
            return True
        return False

    # ------------------------------------------------------------------ #
    # 3.  Scan parametrised families
    # ------------------------------------------------------------------ #
    indices = np.arange(1, n + 1, dtype=np.float64)

    # --- Power‑law family ------------------------------------------------
    alphas = np.linspace(0.05, 1.5, 71)          # coarse grid
    best_alpha = None
    for a in alphas:
        w = np.power(indices, -a)
        w = _normalize(w)
        if _try(w):
            best_alpha = a

    # fine‑grained search near the best exponent, if any
    if best_alpha is not None:
        lo = max(0.05, best_alpha - 0.03)
        hi = min(1.5, best_alpha + 0.03)
        fine = np.linspace(lo, hi, 41)
        for a in fine:
            w = np.power(indices, -a)
            w = _normalize(w)
            _try(w)

    # --- Exponential decay ----------------------------------------------
    betas = np.linspace(5e-5, 0.02, 60)
    for b in betas:
        w = np.exp(-b * np.arange(n, dtype=np.float64))
        w = _normalize(w)
        _try(w)

    # --- Stretched exponentials (random sampling) -----------------------
    for _ in range(80):
        gamma = rng.uniform(0.5, 3.0)               # power on the index
        beta = rng.uniform(5e-5, 0.01)             # decay rate
        i = np.arange(n, dtype=np.float64)
        w = np.exp(-beta * np.power(i, gamma))
        if w.sum() == 0.0:
            continue
        w = _normalize(w)
        _try(w)

    # ------------------------------------------------------------------ #
    # 4.  Gradient‑descent refinement on the current best candidate
    # ------------------------------------------------------------------ #
    a = best.copy()
    step = 0.12
    max_iter = 500
    patience = 30
    no_improve = 0

    for it in range(max_iter):
        # Convolution and location of the active (max) index
        c = np.convolve(a, a)
        max_b = c.max()
        kmax = int(c.argmax())

        # Sub‑gradient of the active convolution term
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # Descent step
        cand = a - step * grad
        cand = np.clip(cand, 0.0, None)
        cand = _normalize(cand)

        sc = _fast_score(cand, n)
        if sc + 1e-12 < best_score:
            best = cand.copy()
            best_score = sc
            a = cand
            no_improve = 0
        else:
            a = cand

            if no_improve >= patience:

                no_improve = 0
                if step < 1e-7:

    # ------------------------------------------------------------------ #

    final = [float(v) for v in best]
    return final
