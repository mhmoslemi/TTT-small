import numpy as np

# ------------------------------------------------------------
# Helper: count ordered pairs (i,j) with i+j = k for a binary set
# ------------------------------------------------------------
def _pair_counts(selected_set, n):
    """
    selected_set : set of indices (0 … n-1) where the coefficient is 1
    returns      : np.ndarray of length 2*n-1 containing ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    sel = list(selected_set)
    L = len(sel)
    for a in range(L):
        i = sel[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            s = i + sel[b]
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt

        n = 256

        except Exception:

    # --------------------------------------------------------------------

    for _ in range(num_starts):
        # initial size – a modest fraction of n, never below 80
        m = max(80, int(0.04 * n))

    if final_score < best_score:
        best_score = final_score
    return [float(x) for x in best_vec.tolist()]
