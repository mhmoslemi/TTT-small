def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by the supplied `evaluate_sequence`.
    The routine explores several compact parametric families and then
    performs cheap stochastic refinements, all using a matrix‑free FFT
    implementation of the convolution bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # parent sequence, if the harness supplies one
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            parent = np.clip(tmp, 0.0, 1000.0)

    alphas = np.linspace(0.001, 0.30, 150)          # finer near 0
    for a in alphas:
        w = np.power(idx + 1.0, -a)
        w /= w.sum()
        sc = fast_score(w)
        if sc < best_score:
            best, best_score = w, sc

    # ------------------------------------------------------------------
    # 2. Exponential decay a_i ∝ exp(‑β·i)
    # ------------------------------------------------------------------
    betas = np.concatenate((
        np.linspace(1e-4, 5e-4, 60),
        np.linspace(5e-4, 2e-3, 60),
        np.linspace(2e-3, 8e-3, 60),
        np.linspace(8e-3, 3e-2, 40),
    ))
    for b in betas:
        w = np.exp(-b * idx)
        w /= w.sum()
        sc = fast_score(w)
        if sc < best_score:
            best, best_score = w, sc

    # ------------------------------------------------------------------
    # 3. Front‑loaded linear‑power family a_i ∝ (n‑i)^p
    # ------------------------------------------------------------------
    rev = n - idx                                 # n, n‑1, … , 1
    ps = np.linspace(0.0, 5.0, 70)
    for p in ps:
        w = np.power(rev, p)
        s = w.sum()
        if s == 0.0:
            continue
        w /= s
        sc = fast_score(w)
        if sc < best_score:
            best, best_score = w, sc

    # ------------------------------------------------------------------
    # 4. Mixed power‑exponential a_i ∝ (i+1)^{-α}·exp(‑β·i)
    # ------------------------------------------------------------------
    alphas2 = np.linspace(0.001, 0.25, 20)
    betas2 = np.linspace(1e-4, 6e-3, 20)
    for a in alphas2:
        pow_part = np.power(idx + 1.0, -a)
        for b in betas2:
            w = pow_part * np.exp(-b * idx)
            s = w.sum()
            if s == 0.0:
                continue
            w /= s
            sc = fast_score(w)
            if sc < best_score:
                best, best_score = w, sc

    # ------------------------------------------------------------------
    # 5. Random Dirichlet samples with varied concentration
    # ------------------------------------------------------------------
    for _ in range(500):
        conc = rng.uniform(0.05, 5.0)            # concentration parameter
        g = rng.gamma(conc, 1.0, size=n)
        w = g / g.sum()
        sc = fast_score(w)
        if sc < best_score:
            best, best_score = w, sc

    # ------------------------------------------------------------------
    # 6. Local stochastic refinements around the current best
    # ------------------------------------------------------------------
    start_time = time.time()
    for _ in range(2000):
        if time.time() - start_time > 5.0:      # safety guard
            break
        i0 = rng.integers(0, n)
        block_len = rng.integers(1, max(2, n // 20) + 1)
        i1 = min(n, i0 + block_len)
        cand = best.copy()
        # multiplicative log‑normal noise on a contiguous block
        noise = np.exp(rng.normal(0.0, 0.07, size=i1 - i0))
        cand[i0:i1] *= noise
        cand = np.clip(cand, 0.0, None)
        total = cand.sum()
        if total > 0.0:
            cand /= total
        sc = fast_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
