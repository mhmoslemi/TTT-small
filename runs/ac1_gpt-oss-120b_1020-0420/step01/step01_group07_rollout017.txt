def _pair_counts_from_set(selected, n):
    size = 2*n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = np.array(sorted(selected), dtype=np.int32)
    L = idx.size
    for a in range(L):
        i = idx[a]
        cnt[2*i] += 1
        for b in range(a+1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2
    return cnt

def _bound_from_cnt(cnt, m, n):
    if m == 0:
        return float('inf')
    max_c = int(cnt.max())
    return 2.0 * n * max_c / (m * m)
```

import numpy as np

# ------------------------------------------------------------
# Helper: pair‑count for a binary mask (ordered pairs)
# ------------------------------------------------------------
def _pair_counts_from_set(selected, n):
    """
    selected : set of indices where the coefficient is 1
    returns   : np.ndarray of length 2*n-1 containing ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = np.array(list(selected), dtype=np.int32)
    L = idx.size
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt


def _bound_from_cnt(cnt, m, n):
    """Exact AC1 bound for binary mask (sum = m)."""
    if m == 0:
        return float("inf")
    max_c = int(cnt.max())
    return 2.0 * n * max_c / (m * m)


def _mask_to_vector(selected, n):
    """Convert index set to a float vector of 0/1."""
    v = np.zeros(n, dtype=np.float64)
    if selected:
        v[list(selected)] = 1.0
    return v
