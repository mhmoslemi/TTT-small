def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def propose_candidate(seed: int = 42, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    parent = globals().get('height_sequence_1')
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.size)
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_half_power_series(n)

    # safety clamp
    np.clip(a, 0.0, 1000.0, out=a)

    # ensure minimum total mass
    total = float(a.sum())
    if total < 0.01:
        a += (0.01 - total) / n

    # initial convolution and score
    b = np.convolve(a, a)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    max_iter = 10000
    patience = 2000
    no_improve = 0
    step = 0.2
    decay = 0.9995

    # main transfer loop
    for it in range(max_iter):
        # current max convolution index
        k_max = int(np.argmax(b))
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)

        # choose source index i1 with largest contribution to the peak
        cand = np.arange(i_low, i_high + 1)
        prod = a[cand] * a[k_max - cand]
        i1 = int(cand[np.argmax(prod)])

        # choose a target index outside the active interval
        attempts = 0
        while True:
            i2 = int(rng.integers(0, n))
            if i2 != i1 and not (i_low <= i2 <= i_high):
                break
            attempts += 1
            if attempts > 10:
                # give up and skip this iteration
                i2 = None
                break
        if i2 is None:
            continue

        # decide how much mass to move
        max_transfer = step * a[i1]
        max_transfer = min(max_transfer, 1000.0 - a[i2])  # respect upper bound
        if max_transfer <= 0.0:
            continue
        amount = max_transfer

        delta1 = -amount
        delta2 = amount

        # backup state
        b_old = b.copy()
        a_i1_old = a[i1]
        a_i2_old = a[i2]

        # linear updates using the old a values
        b[i1:i1 + n] += 2.0 * delta1 * a
        b[2 * i1] += delta1 * delta1
        b[i2:i2 + n] += 2.0 * delta2 * a
        b[2 * i2] += delta2 * delta2
        b[i1 + i2] += 2.0 * delta1 * delta2

        # apply the mass transfer
        a[i1] = a_i1_old + delta1
        a[i2] = a_i2_old + delta2

        # evaluate new bound (sum stays the same)
        new_max = b.max()
        new_score = 2.0 * n * new_max / (a.sum() * a.sum())

            best_score = new_score

            a[i2] = a_i2_old
            b = b_old
            no_improve += 1

        # decay step size
        step *= decay
        if no_improve >= patience:
            break

import numpy as np

# ------------------------------------------------------------
# Half‑power series ( (1‑z)^(-1/2) ), used as a solid baseline.
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """a_i = binom(2i,i)/4**i, i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Choose problem size and initialise a starting sequence.
    # Prefer the supplied parent if it exists, otherwise fall back
    # to the half‑power series.
    # --------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.size)
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_half_power_series(n)

    np.clip(a, 0.0, 1000.0, out=a)

    # ensure a non‑trivial total mass
    total = float(a.sum())
    if total < 0.01:
        a += (0.01 - total) / n

    # current full autocorrelation and best score so far
    b = np.convolve(a, a)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------
    # Optimisation: move a fraction of mass from a large early
    # coefficient to a far‑tail coefficient.  This tends to
    # reduce the peak of the convolution while keeping the sum
    # (and thus the denominator of the bound) unchanged.
    # Occasionally we also try a single‑index perturbation.
    # --------------------------------------------------------
    max_iter = 10000          # overall budget
    patience = 2000           # stop if no improvement for this many steps
    no_improve = 0
    step = 0.20               # initial transfer fraction
    decay = 0.9995            # geometric decay of the step size

    for it in range(max_iter):
        # occasionally perform a simple single‑index mutation
        if rng.random() < 0.07:
            i = int(rng.integers(0, n))
            factor = 1.0 + rng.normal(0.0, step)
            new_val = a[i] * factor
            new_val = max(0.0, min(1000.0, new_val))
            delta = new_val - a[i]
            if abs(delta) < 1e-12:
                continue

            b_old = b.copy()
            a_old = a[i]

            # incremental update of the convolution
            b[i:i + n] += 2.0 * delta * a
            b[2 * i] += delta * delta
            a[i] = new_val

            new_score = 2.0 * n * b.max() / (a.sum() * a.sum())
            if new_score < best_score:
                best = a.copy()
                best_score = new_score
                no_improve = 0
            else:
                a[i] = a_old
                b = b_old
                no_improve += 1
            step *= decay
            if no_improve >= patience:
                break
            continue

        # ----------------------------------------------------
        # Transfer move: choose a source index that contributes
        # most to the current maximal convolution value and a
        # target index outside that active region.
        # ----------------------------------------------------
        k_max = int(np.argmax(b))
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)

        # source index with largest product a[i] * a[k_max-i]
        candidates = np.arange(i_low, i_high + 1)
        prod = a[candidates] * a[k_max - candidates]
        i1 = int(candidates[np.argmax(prod)])

        # pick a target far from the active window
        attempts = 0
        while True:
            i2 = int(rng.integers(0, n))
            if i2 != i1 and not (i_low <= i2 <= i_high):
                break
            attempts += 1
            if attempts > 10:
                i2 = None
                break
        if i2 is None:
            continue

        # decide how much mass to shift (respect the 0‑1000 bounds)
        max_transfer = step * a[i1]
        max_transfer = min(max_transfer, 1000.0 - a[i2])
        if max_transfer <= 0.0:
            continue
        amount = max_transfer
        delta1 = -amount
        delta2 = amount

        # backup current state
        b_old = b.copy()
        a_i1_old = a[i1]
        a_i2_old = a[i2]

        # linear updates using the *old* a values (still in a)
        b[i1:i1 + n] += 2.0 * delta1 * a
        b[2 * i1] += delta1 * delta1
        b[i2:i2 + n] += 2.0 * delta2 * a
        b[2 * i2] += delta2 * delta2
        b[i1 + i2] += 2.0 * delta1 * delta2

        # apply the mass transfer
        a[i1] = a_i1_old + delta1
        a[i2] = a_i2_old + delta2

        # evaluate the new bound (sum unchanged, but recompute for safety)
        new_max = b.max()
        new_score = 2.0 * n * new_max / (a.sum() * a.sum())

        if new_score < best_score:
            best = a.copy()
            best_score = new_score
            no_improve = 0
        else:
            # revert all changes
            a[i1] = a_i1_old
            a[i2] = a_i2_old
            b = b_old
            no_improve += 1

        step *= decay
        if no_improve >= patience:
            break

    # -----------------------------------------------------------------
    # Final verification with the provided evaluator; fall back to the
    # last valid state if something went wrong.
    # -----------------------------------------------------------------
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = a  # fallback to the last state we held
    return [float(x) for x in best]
