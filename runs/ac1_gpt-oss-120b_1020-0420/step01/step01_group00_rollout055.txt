import numpy as np

# ----- fast convolution max (FFT, matrix‑free) -----
def _conv_max(vec: np.ndarray, fft_len: int):
    """Return (max value, index) of the linear convolution vec*vec."""
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * vec.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def propose_candidate(seed: int = 42, num_steps: int = 15000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 bound.
    Returns a list of coefficients (the best found).
    """
    rng = np.random.default_rng(seed)

    # ----- obtain a parent length (height_sequence_1) if it exists -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        # fall back to a modest size when no parent is available
        n = 256
        parent = np.full(n, 1.0 / n, dtype=np.float64)

    # ----- FFT length (next power of two for 2n‑1 points) -----
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ----- several deterministic starting points (parent + a few random seeds) -----
    init_vectors = []

    # 1) parent (clipped)
    init_vectors.append(np.clip(parent, 0.0, 1000.0))

    # 2) uniform
    init_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) random exponential
    init_vectors.append(rng.exponential(scale=1.0, size=n))

    # 4) random log‑normal
    init_vectors.append(np.exp(rng.normal(loc=0.0, scale=1.0, size=n)))

    best_vec = None
    best_score = np.inf

    # ----- optimization loop for each seed -----
    for seed_vec in init_vectors:
        cur = np.clip(seed_vec, 0.0, 1000.0)
        s = cur.sum()
        if s < 1e-12:
            cur[:] = 1.0
            s = cur.sum()
        cur = cur / s

        # small jitter to avoid the trivial uniform stationary point
        cur = cur * np.exp(rng.normal(0.0, 0.04, size=n))
        cur = np.clip(cur, 0.0, 1000.0)
        cur = cur / cur.sum()

        # initial convolution data
        cur_max, cur_kmax = _conv_max(cur, fft_len)
        cur_score = 2.0 * n * cur_max      # sum(cur) == 1, so score = 2·n·max(conv)

        # best found in this run
        best_local = cur.copy()
        best_local_score = cur_score

        # momentum buffer
        vel = np.zeros_like(cur)

        # schedule parameters
        init_eta = 0.27               # starting step length
        init_T = 0.12                 # starting temperature
        beta = 0.9                    # momentum decay

        for step in range(num_steps):
            # ------- sub‑gradient of the active convolution peak -------
            idx = cur_kmax - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            b = np.zeros_like(cur)

            # ------- momentum update -------
            vel = beta * vel + (1.0 - beta) * grad

            # ------- annealed step size & temperature -------
            eta = init_eta * (1.0 - step / num_steps)
            T = init_T * (1.0 - step / num_steps)

            # ------- candidate proposal -------
            cand = cur - eta * vel

                    best_local_score = cur_score

    return [float(v) for v in best_vec]
