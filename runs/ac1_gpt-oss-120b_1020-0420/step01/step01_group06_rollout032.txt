import numpy as np

def full_convolution(a):
    n = a.shape[0]
    size = 2*n - 1
    m = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, m)
    B = A * A
    conv = np.fft.irfft(B, m)[:size]
    return conv

def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:
        return np.inf
    # normalized? evaluator does not require normalization
    # but we keep sum as is; compute bound formula using current total
    b = full_convolution(a)
    max_b = b.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (total * total))

def project_simplex(v):
    # projection onto simplex {x>=0, sum x = 1}
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all zero, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        if sc < best_score:

            best_score = sc

        w = np.exp(-b * np.arange(n, dtype=np.float64))
        w_sum = w.sum()

        for b in betas:
            w = pw * np.exp(-b * np.arange(n, dtype=np.float64))
            w_sum = w.sum()
            if w_sum == 0:
                continue
            w /= w_sum
            sc = _score(w)
            if sc < best_score:
                best = w
                best_score = sc

    for it in range(4000):

        a = rng.uniform(0.0, 1.5)
        b = rng.uniform(0.0, 5e-3)

        w_sum = w.sum()

            continue
        w /= w_sum

        # small multiplicative noise

            cand *= noise
        # keep within bounds and renormalise
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total == 0:

        cand /= total
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ---- projected gradient descent on the max index ----
    # perform a few steps to polish
    for it in range(200):

        k_max = int(np.argmax(b))
        # compute gradient of the active convolution term
        grad = np.zeros_like(best)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1)
        j = k_max - i_range
        grad[i_range] = best[j]
        # double contribution on diagonal
        if k_max % 2 == 0:
            mid = k_max // 2
            if i_start <= mid <= i_end:

        # step size schedule
        eta = 0.5 / (it + 1)
        cand = best - eta * grad
        cand = np.maximum(cand, 0.0)
        # re‑normalize to sum = 1
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # Return as plain python list (float)
    return [float(v) for v in best]

import numpy as np

# -------------------------------------------------------------
# Helper: linear convolution via FFT (exact for real vectors)
# -------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.size
    size = 2 * n - 1
    m = 1 << (size - 1).bit_length()          # next power‑of‑two
    A = np.fft.rfft(a, m)
    B = A * A
    conv = np.fft.irfft(B, m)[:size]
    return conv


# -------------------------------------------------------------
# Compute the AC1 upper bound exactly as the evaluator does
# (except the evaluator also clips, we mimic that here)
# -------------------------------------------------------------
def _score_vec(a: np.ndarray) -> float:
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:
        return np.inf
    conv = _full_convolution(a)
    max_b = conv.max()
    n = a.size
    return float(2.0 * n * max_b / (total * total))


# -------------------------------------------------------------
# Project a vector onto the probability simplex {x>=0, sum=1}
# -------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1.0) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
