import numpy as np

# ------------------------------------------------------------
# Scoring utility – identical to the supplied evaluator
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    # direct convolution (NumPy internally chooses fast path)
    conv = np.convolve(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ------------------------------------------------------------
# Fast full convolution via FFT (real‑valued)
# ------------------------------------------------------------
def _full_conv(seq: np.ndarray) -> np.ndarray:
    n = seq.size
    m = 2 * n - 1
    # rfft pads automatically to length m
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)
    return conv

def propose_candidate(seed: int = 42, max_iters: int = 20000, **kwargs):
    """
    Return a list of non‑negative heights attempting to minimise the
    AC1 upper bound.  The routine starts from the provided parent
    sequence (if any) and applies a focused hill‑climbing search that
    repeatedly reduces the contribution to the currently maximal
    convolution entry.
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # Determine the problem size – use the provided parent if
    # available; otherwise fall back to a uniform start.
    # ---------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        ):
            n = parent.size
            cur = parent.copy()
        else:
            n = 256
            cur = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    cur = np.clip(cur, 0.0, 1000.0)
    best_seq = cur
    best_score = _score(best_seq)
    conv = _full_conv(best_seq)

    # ---------------------------------------------------------
    # Parameters
    # ---------------------------------------------------------
    target_score = 1.50274349          # desired upper bound
    max_iters = kwargs.get("num_steps", max_iters)

    # counter for stagnation – occasional random perturbation
    stagnation = 0
    for it in range(max_iters):
        # -----------------------------------------------------
        # locate the shift with the current maximal convolution
        # -----------------------------------------------------
        k_max = int(np.argmax(conv))

        # -----------------------------------------------------
        # partner values for this shift: a_{k_max-i} (zero outside)
        # -----------------------------------------------------
        idx = np.arange(n)
        partner_idx = k_max - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        partner_vals = np.zeros(n, dtype=best_seq.dtype)
        partner_vals[mask] = best_seq[partner_idx[mask]]

        # product contributions a_i * a_{k_max-i}
        contributions = best_seq * partner_vals

        if best_seq[src] <= 0.0:
