def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2, a_i = C(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs

def _max_convolution(a: np.ndarray) -> float:
    """Return max of the full linear convolution a*a using FFT."""
    n = a.size
    size = 2 * n - 1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return float(np.max(conv))

    # ------------------------------------------------------------------
    base = _coefficients_binom_half(n)
    target_sum = math.sqrt(2.0 * n)
    base *= target_sum / base.sum()
    np.clip(base, 0.0, 1000.0, out=base)

    try:
        best = base.copy()
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best = base.copy()
        best_score = np.inf

    # possibly replace by the provided parent if it is better
    if parent is not None and parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # ------------------------------------------------------------------
    # Optional LP refinement (few iterations).  If cvxpy is unavailable
    # or the problem is too large the block is silently skipped.
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp

        f = best.copy()
        M = _max_convolution(f)                       # current max convolution
        t = 0.5                                        # mixing weight
        for _ in range(3):                            # at most three LP steps
            g = cp.Variable(n, nonneg=True)
            conv_fg = cp.conv(cp.Constant(f), g)      # linear in g
            prob = cp.Problem(cp.Maximize(cp.sum(g)),
                               [conv_fg <= M])
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=3000, feastol=1e-7, abstol=1e-7,
                       reltol=1e-7, verbose=False)
            if g.value is None:
                break
            g_val = np.clip(np.asarray(g.value, dtype=np.float64).ravel(),
                            0.0, 1000.0)
            s = g_val.sum()
            if s <= 0.0:
                break
            g_scaled = g_val * (target_sum / s)       # renormalise sum
            f_new = (1.0 - t) * f + t * g_scaled
            np.clip(f_new, 0.0, 1000.0, out=f_new)
            # evaluate the mixed candidate
            new_score = float(evaluate_sequence(f_new.tolist()))
            if np.isfinite(new_score) and new_score < best_score:
                best = f_new.copy()
                best_score = new_score
                f = f_new
                M = _max_convolution(f)               # new max conv
            else:
                break                                 # no progress
    except Exception:
        pass                                            # LP not used

    # ------------------------------------------------------------------
    # Fast stochastic hill‑climbing (FFT‑based evaluation).
    # ------------------------------------------------------------------
    target = 1.50274349
    steps = 8000
    n = best.shape[0]

    for _ in range(steps):
        if best_score < target:
            break

        cand = best.copy()
        op = rng.integers(0, 5)

        if op == 0:                                   # swap two entries
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        elif op == 1:                                 # scale a random block
            min_len = max(1, n // 100)
            max_len = max(2, n // 4)
            width = int(rng.integers(min_len, max_len + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.20)))
            cand[start:start + width] *= factor

        elif op == 2:                                 # boost a tail block
            start = int(rng.uniform(0.75 * n, n - 1))
            width = int(rng.integers(1, max(2, n - start)))
            factor = float(np.exp(rng.normal(0.0, 0.30)))
            cand[start:start + width] *= factor

        elif op == 3:                                 # add a small bump
            idx = int(rng.integers(0, n))
            bump = max(0.0, rng.normal(0.0, 0.05))
            cand[idx] += bump

        else:                                         # shuffle a modest slice
            slice_len = int(rng.integers(2, min(100, n // 2)))
            start = int(rng.integers(0, n - slice_len + 1))
            sub = cand[start:start + slice_len]
            rng.shuffle(sub)
            cand[start:start + slice_len] = sub

        np.clip(cand, 0.0, 1000.0, out=cand)

        # cheap FFT‑based bound estimate
        conv_max = _max_convolution(cand)
        S = cand.sum()
        fast_score = 2.0 * n * conv_max / (S * S)

        if fast_score + 1e-7 < best_score:
            # verify with the exact evaluator
            try:
                true_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue
            if np.isfinite(true_score) and true_score < best_score:
                best = cand
                best_score = true_score

    # ------------------------------------------------------------------
    # final safety: ensure a non‑empty, valid sequence is returned.
    # ------------------------------------------------------------------
    if best_score == np.inf or best.size == 0:
        best = np.full(n, 0.01)
    return [float(x) for x in best]
