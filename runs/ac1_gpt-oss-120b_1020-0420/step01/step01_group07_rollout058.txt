                break
    return a, best_score

# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Returns a list of non‑negative coefficients (length = len(height_sequence_1)
    if available, otherwise 256) whose AC1 bound is as low as possible.
    The routine first attempts a few sparse linear‑programming updates
    (matrix‑free convolution), then finishes with a small hill‑climb.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size and an initial feasible vector
    # --------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.size
        init = np.clip(init, 0.0, 1000.0)
        if init.sum() == 0:
            init = np.full(n, 1.0 / n)
        else:
            init = init / init.sum()
    except NameError:
        n = 256
        init = np.full(n, 1.0 / n)

    # --------------------------------------------------------
    # Baseline score
    # --------------------------------------------------------
    best_a = init.copy()
    best_score = _score(best_a)
    TARGET = 1.50274349

    # --------------------------------------------------------
    # Try a few matrix‑free LP refinements (convex combination)
    # --------------------------------------------------------
    try:
        import cvxpy as cp

        # small mixing factor (reduced if no progress)
        mix_factors = [0.5, 0.25, 0.125, 0.0625]

        a = best_a.copy()
        for outer in range(5):                     # at most five LP passes
            conv = np.convolve(a, a)
            M = conv.max()
            # ---- LP: maximise sum(g)  s.t.  conv(a,g) ≤ M,  g ≥ 0 ----
            f_param = cp.Parameter(n, nonneg=True)
            f_param.value = a
            g = cp.Variable(n, nonneg=True)
            conv_expr = cp.conv(f_param, g)         # length 2n‑1
            prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_expr <= M])
            try:
                prob.solve(solver='ECOS', warm_start=False, max_iters=4000)
            except Exception:                      # fall back to a different solver
                prob.solve(solver='SCS', max_iters=8000, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break
            g_opt = g.value
            if g_opt is None:
                break
            g_opt = np.maximum(g_opt, 0.0)
            sum_g = g_opt.sum()
            if sum_g == 0:
                break
            g_norm = g_opt / sum_g                     # same ℓ1‑norm as a
            improved = False
            cur_score = _score(a)
            # ---- try mixing a and g_norm with decreasing t ----
            for t in mix_factors:
                a_new = (1.0 - t) * a + t * g_norm     # ℓ1‑norm stays 1
                sc = _score(a_new)
                if np.isfinite(sc) and sc < cur_score:
                    a = a_new
                    cur_score = sc
                    improved = True
                    break
            if not improved:
                break
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            if best_score <= TARGET:
                break
    except Exception:
        # any problem with cvxpy – fall back silently
        pass

    # --------------------------------------------------------
    # Final polishing with cheap hill‑climb if needed
    # --------------------------------------------------------
    if best_score > TARGET:
        best_a, best_score = _hill_climb(best_a, rng, best_score,
                                         TARGET, max_iters=3000)

    # --------------------------------------------------------
    # Safety sanitisation and return
    # --------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() == 0:
        best_a = np.full(n, 1.0 / n)
    best_a /= best_a.sum()
    return [float(x) for x in best_a.tolist()]
