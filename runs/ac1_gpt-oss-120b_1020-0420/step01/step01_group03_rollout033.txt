import numpy as np

# -------------------------------------------------------------
# Helper: generate the classic root‑constant sequence (a_i = C(2i,i)/4**i)
# -------------------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


# -------------------------------------------------------------
# Convolution via FFT (linear convolution, length = 2*n-1)
# -------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    return conv[: 2 * n - 1]


# -------------------------------------------------------------
# Compute the exact score used by the evaluator (no I/O)
# Returns: (score, max_conv, sum_seq, conv_array)
# -------------------------------------------------------------
def _score_of(seq: np.ndarray):
    n = seq.shape[0]
    conv = _conv_fft(seq)
    max_conv = np.max(conv)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return float("inf"), max_conv, sum_seq, conv
    score = 2.0 * n * max_conv / (sum_seq * sum_seq)
    return float(score), max_conv, sum_seq, conv
