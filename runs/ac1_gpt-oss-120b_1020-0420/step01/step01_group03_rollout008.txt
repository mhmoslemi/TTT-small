import numpy as np

# ---------- utilities ----------
def _root_constant_seq(length: int) -> np.ndarray:
    """
    a_i = binom(2i,i) / 4**i  (recurrence a_i = a_{i-1} * (2i-1)/(2i))
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _conv_score(arr: np.ndarray) -> float:
    """
    Wrapper around the supplied evaluate_sequence.  Returns np.inf on any error.
    """
    try:
        return float(evaluate_sequence(arr.tolist()))
    except Exception:
        return float('inf')


# ---------- main entry ----------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function that improves the AC1 bound.
    The construction starts from the classic root‑constant sequence and
    searches a few simple linear combinations with shifted (or reversed)
    copies of itself.  All candidates keep the ℓ1‑norm of the original
    sequence, so only the maximal convolution value changes.
    """
    np.random.seed(seed)

    # Determine the length – use the provided parent if present, otherwise a modest fallback.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # Base sequence (root‑constant).  This already gives ~1.57 score.
    base = _root_constant_seq(n)
    best_seq = base.copy()
    best_score = _conv_score(best_seq)

    # Guard against pathological parent (sum too tiny).
    if np.sum(best_seq) < 0.01 or not np.isfinite(best_score):
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_score = _conv_score(best_seq)

    sum_base = float(np.sum(best_seq))

    # ---- explore shifted copies -------------------------------------------------
    # Fractions of the length to shift by (roughly ¼,⅓,½,2/3,3/4).
    shift_fracs = [0.25, 0.33, 0.5, 0.66, 0.75]
    # Mixing weights (convex combination).  The ℓ1‑norm stays constant for any t.
    mix_weights = [0.2, 0.4, 0.6, 0.8]

    for frac in shift_fracs:
        D = int(n * frac)
        if D <= 0 or D >= n:
            continue
        # Right‑shift with zero padding.
        shifted = np.concatenate((np.zeros(D, dtype=np.float64), best_seq[: n - D]))
        sum_shifted = float(np.sum(shifted))
        if sum_shifted < 1e-12:
            continue
        # Scale the shifted copy so that its total mass equals that of the base.
        scale = sum_base / sum_shifted
        shifted_scaled = shifted * scale
        for t in mix_weights:
            cand = (1.0 - t) * best_seq + t * shifted_scaled
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _conv_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    # ---- explore reversed copy --------------------------------------------------
    rev = best_seq[::-1]                # exact reversal, same ℓ1‑norm
    for t in mix_weights:
        cand = (1.0 - t) * best_seq + t * rev
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _conv_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # ---- final safety check ----------------------------------------------------
    final_score = _conv_score(best_seq)
    if not np.isfinite(final_score):
        # fallback to a simple uniform sequence that is always valid
        best_seq = np.full(n, 1.0, dtype=np.float64)

    return best_seq.tolist()
