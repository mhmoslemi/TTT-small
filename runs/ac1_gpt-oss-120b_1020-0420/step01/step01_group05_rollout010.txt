import numpy as np

# ----------------------------------------------------------
# Fast evaluation of the AC1 bound using FFT (matrix‑free)
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a, a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                                 # point‑wise product
    c = np.fft.irfft(fc, n=nfft)[:m]             # full convolution
    M = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _conv_max_idx(a: np.ndarray, nfft: int):
    """Return (max value, index) of full convolution a*a."""
    n = a.shape[0]
    m = 2 * n - 1
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    idx = int(np.argmax(c))
    return float(c[idx]), idx
