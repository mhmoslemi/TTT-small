def _gradient_step(a, lr):
    conv = np.convolve(a, a)
    kmax = conv.argmax()
    M = conv[kmax]
    S = a.sum()
    # Compute gradient of M
    idx = kmax - np.arange(len(a))
    mask = (idx >= 0) & (idx < len(a))
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    # gradient of objective
    grad = (2.0 * len(a) / (S**2)) * g - (4.0 * len(a) * M / (S**3)) * a
    # update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0:
        # restart 
        a_new = np.ones_like(a) / len(a)
    else:
        a_new /= a_new.sum()  # renormalize to sum=1
    return a_new, kmax, M
```

import numpy as np

def _objective(vec: np.ndarray) -> float:
    """Fast computation of the AC1 upper bound for a vector."""
    s = vec.sum()
    if s == 0.0:
        return float('inf')
    conv = np.convolve(vec, vec)
    m = conv.max()
    return 2.0 * vec.size * m / (s * s)


def _gradient_step(a: np.ndarray, lr: float) -> tuple:
    """
    Perform one gradient‐descent step on the AC1 bound.
    Returns (new_vector, k_max, M) where k_max is the index of the
    maximal convolution value and M that value.
    """
    n = a.size
    conv = np.convolve(a, a)
    k = int(conv.argmax())
    M = conv[k]
    S = a.sum()
    # gradient of the maximal convolution value
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    # full gradient of the objective
    grad = (2.0 * n / (S * S)) * g - (4.0 * n * M / (S * S * S)) * a
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    tot = a_new.sum()
    if tot == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new = a_new / tot
    return a_new, k, M
