        exp_shape = exp_shape / exp_shape.sum()

    best_score = np.inf

        if score < best_score:

def _fast_score(vec):
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

import numpy as np

def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative vector (sum normalisation optional)."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def propose_candidate(seed: int = 42, num_steps: int = 12000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises the AC1 upper bound.
    Returns a list of floats.

        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = None

    n = base.size if base is not None else 256
    idx_arr = np.arange(n)

    # ----- deterministic initial candidates -----
    starts = []

    # uniform
    starts.append(np.full(n, 1.0 / n))

    # cosine edge shape: high at ends, low near centre

    cos_shape = cos_shape / cos_shape.sum()

        exp_shape = np.exp(-gamma * idx_arr) + np.exp(-gamma * (n - 1 - idx_arr))
        exp_shape = exp_shape / exp_shape.sum()
        starts.append(exp_shape)

    # a slightly perturbed parent, if present
    if base is not None:
