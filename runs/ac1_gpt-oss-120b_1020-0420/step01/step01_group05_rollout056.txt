import numpy as np

# ----------------------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound.
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """2 * n * max(conv(a, a)) / (sum(a) ** 2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(fa * fa, nfft)[:m]
    M = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)
