def propose_candidate(seed=42, **kwargs):
    import numpy as np
    # define fast_score
    def fast_score(a):
        n = a.shape[0]
        m = 2*n - 1
        nfft = 1 << ((m - 1).bit_length())
        fa = np.fft.rfft(a, n=nfft)
        c = np.fft.irfft(fa*fa, n=nfft)[:m]
        M = c.max()
        S = a.sum()
        if S < 0.01:
            return float('inf')
        return 2.0 * n * M / (S * S)

import numpy as np

def _fast_score(a: np.ndarray) -> float:
    """Fast computation of 2*n*max(conv(a,a))/(sum(a)^2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1                     # length of full convolution
    nfft = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = c.max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Acquire a fallback parent sequence if available.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = None

    if parent is None:
        n = 256
        best_seq = np.ones(n, dtype=np.float64)
    else:
        n = parent.size
        best_seq = np.clip(parent, 0.0, 1000.0)

    best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # Index array for analytic families.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1) Inverse‑square‑root families (forward and reversed).
    # ------------------------------------------------------------------
    offsets = np.geomspace(1e-3, 1e2, num=20)
    for off in offsets:
        a = 1.0 / np.sqrt(i + off)
        a = np.clip(a, 0.0, 1000.0)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s

        a_rev = 1.0 / np.sqrt((n - 1 - i) + off)
        a_rev = np.clip(a_rev, 0.0, 1000.0)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 2) Power‑law families (forward and reversed) over a wide exponent range.
    # ------------------------------------------------------------------
    powers = np.linspace(0.20, 0.98, 40)
    for p in powers:
        a = (i + 1.0) ** (-p)
        a = np.clip(a, 0.0, 1000.0)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s

        a_rev = (n - i) ** (-p)           # reversed version
        a_rev = np.clip(a_rev, 0.0, 1000.0)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 3) Multiplicative combination of power‑law and inverse‑sqrt.
    # ------------------------------------------------------------------
    for p in np.linspace(0.40, 0.90, 5):
        for off in np.geomspace(0.1, 10.0, 5):
            a = ((i + 1.0) ** (-p)) / np.sqrt(i + off)
            a = np.clip(a, 0.0, 1000.0)
            s = _fast_score(a)
            if s < best_score:
                best_seq, best_score = a, s

            a_rev = ((n - i) ** (-p)) / np.sqrt((n - 1 - i) + off)
            a_rev = np.clip(a_rev, 0.0, 1000.0)
            s = _fast_score(a_rev)
            if s < best_score:
                best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 4) Smooth the current best with a few moving‑average kernels.
    # ------------------------------------------------------------------
    for k in (3, 7, 15, 31, 61, 101, 201):
        kernel = np.ones(k, dtype=np.float64) / k
        a = np.convolve(best_seq, kernel, mode='same')
        a = np.clip(a, 0.0, 1000.0)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s

    # ------------------------------------------------------------------
    # 5) Stochastic multiplicative refinement (many small jitter steps).
    # ------------------------------------------------------------------
    total_iters = 5000
    sigma_start = 0.5
    sigma_end = 0.001
    target = 1.50274349  # desired upper bound

    for it in range(total_iters):
        # exponential decay of jitter magnitude
        sigma = sigma_start * (sigma_end / sigma_start) ** (it / total_iters)
        perturb = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_seq * perturb
        cand = np.clip(cand, 0.0, 1000.0)
        s = _fast_score(cand)
        if s < best_score:
            best_seq, best_score = cand, s
            if best_score < target:
                break

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq.tolist()]
