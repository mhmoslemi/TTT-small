import numpy as np

# ------------------------------------------------------------
# Helper: generate the truncated series of (1 - z)^(-1/2)
#   a_0 = 1,  a_{i} = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Helper: reorder values so that the smallest entry sits at index 0.
# This often reduces the peak of the autocorrelation.
# ------------------------------------------------------------
def initial_reorder(vals: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(vals)                     # ascending
    n = sorted_vals.size
    out = np.empty_like(sorted_vals)
    lo, hi, pos = 0, n - 1, 0
    while lo <= hi:
        out[pos] = sorted_vals[lo]
        pos += 1
        lo += 1
        if lo <= hi:
            out[pos] = sorted_vals[hi]
            pos += 1
            hi -= 1
    return out


# ------------------------------------------------------------
# Fast AC1 score (identical to evaluate_sequence but uses FFT)
# ------------------------------------------------------------
def compute_score(arr: np.ndarray) -> float:
    n = arr.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a <= 0.0:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]
            no_improve += 1

        # occasional reshuffle to escape plateaus
        if no_improve >= patience:
            rng.shuffle(a)
            no_improve = 0

    # --------------------------------------------------------
    # Final validation using the official evaluator (exact match)
    # --------------------------------------------------------
    final_candidate = best.tolist()
    # The harness will call evaluate_sequence itself; we just return.
    return final_candidate
```
