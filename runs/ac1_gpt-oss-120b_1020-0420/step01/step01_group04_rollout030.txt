import numpy as np

def _coefficients_binom_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs

def _ac1_bound(arr):
    s = np.sum(arr)
    if s < 0.01:
        return np.inf
    b = np.convolve(arr, arr)
    return float(2 * arr.shape[0] * np.max(b) / (s * s))

def propose_candidate(seed=42, num_steps=15000, **kwargs):

    # Use provided parent if available, otherwise generate analytic start.
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = kwargs.get('n', 256)
        base = _coefficients_binom_half(n)
    base = np.clip(base, 0.0, 1000.0)
    best = base.copy()
    best_score = _ac1_bound(best)

    # Target bound for early exit
    target = 1.50274349

    n = best.shape[0]

    # Main search loop
    for step in range(num_steps):
        if best_score <= target:
            break

        # Pre‑compute data of the current best (used by several moves)
        b_best = np.convolve(best, best)
        k0 = int(np.argmax(b_best))

        # Choose a mutation type
        op = rng.integers(0, 5)
        cand = best.copy()

        if op == 0:
            # Gradient‑guided scaling: increase entries that contribute little
            # to the largest convolution entry.
            g = np.zeros(n, dtype=np.float64)
            lo = max(0, k0 - (n - 1))
            hi = min(n - 1, k0)
            idx = np.arange(lo, hi + 1)
            partner = k0 - idx
            g[idx] = best[idx] + best[partner]

            # Pick a percentile of the smallest contributors.
            mask = g > 0
            if np.any(mask):
                thresh = np.quantile(g[mask], 0.25)
                low_idx = np.where(g <= thresh)[0]
            else:
                low_idx = np.arange(n)

            if low_idx.size:
                cnt = max(1, n // 2000)               # ~2‑3 indices for our size
                chosen = rng.choice(low_idx,
                                    size=min(cnt, low_idx.size),
                                    replace=False)
                factor = 1.0 + rng.uniform(0.0, 0.4, size=chosen.shape[0])
                cand[chosen] *= factor

        elif op == 1:
            # Randomly scale a contiguous block.
            min_len = max(1, n // 200)          # 0.5 %‑5 % of n
            max_len = max(2, n // 20)
            width = int(rng.integers(min_len, max_len + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = 1.0 + rng.uniform(0.0, 0.5)
            cand[start:start + width] *= factor
