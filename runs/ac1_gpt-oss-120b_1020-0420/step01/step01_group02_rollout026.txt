def propose_candidate(seed=42, num_steps=3000, **kwargs):
   rng = np.random.default_rng(seed)
   # Determine base:
   if 'height_sequence_1' in globals() and isinstance(height_sequence_1, np.ndarray) and height_sequence_1.ndim == 1 and height_sequence_1.size > 0:
       base = height_sequence_1.astype(np.float64).copy()
   else:
       n = 256
       base = generate_half_power_series(n)
   n = base.size

   # ensure non-neg & clip
   np.clip(base, 0.0, 1000.0, out=base)

   best = base
   best_score = float(evaluate_sequence(best.tolist()))
   if not np.isfinite(best_score):
       best_score = float('inf')

   # iteration parameters
   max_width = max(1, n // 10)  # block width up to n/10
   # initial sigma for factor log-normal
   sigma_inc = 0.12
   sigma_a0 = 0.02

   for step in range(num_steps):
       cand = best.copy()
       # random block
       width = int(rng.integers(1, max_width + 1))
       start = int(rng.integers(0, n - width + 1))
       inc_factor = float(np.exp(rng.normal(0.0, sigma_inc)))  # >0
       cand[start:start+width] *= inc_factor

       # modify a0 with reduction bias
       if rng.random() < 0.5:
           a0_factor = float(np.exp(rng.normal(-0.5*sigma_a0, sigma_a0)))  # mean <1
           cand[0] *= a0_factor

       # Clip
       np.clip(cand, 0.0, 1000.0, out=cand)

       # evaluate
       score = float(evaluate_sequence(cand.tolist()))
       if np.isfinite(score) and score < best_score:
           best = cand
           best_score = score

       # optional cooling
       sigma_inc = max(0.005, sigma_inc * 0.9995)  # slowly decrease
       sigma_a0 = max(0.001, sigma_a0 * 0.9995)

   # final check: ensure best is valid
   if not np.isfinite(best_score):
       # fallback to base
       best = base.copy()
   return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# Helper: truncated series of (1 - z)^(-1/2)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return a length‑`length` array with a_i = binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 4000, **kwargs):
    """
    Return a list of non‑negative step‑function heights aiming to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine starting vector (use the provided parent if any)
    # --------------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        base = height_sequence_1.astype(np.float64).copy()
    else:
        # a modest deterministic fallback
        base = generate_half_power_series(256)

    # enforce evaluator’s safety limits
    np.clip(base, 0.0, 1000.0, out=base)

    n = base.size
    best = base
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = float("inf")

    # --------------------------------------------------------
    # Parameters for the stochastic refinement
    # --------------------------------------------------------
    max_width = max(1, n // 12)            # block size up to ≈ n/12
    sigma_block = 0.18                    # log‑normal spread for block scaling
    sigma_a0 = 0.08                       # spread for a_0 scaling (biased down)
    a0_bias = -0.04                       # negative mean -> typical reduction

    target = 1.50274349

    for step in range(num_steps):
        cand = best.copy()

        # --- perturb a_0 (usually a slight reduction) ----------
        if rng.random() < 0.7:  # most steps touch a_0
            factor_a0 = float(np.exp(rng.normal(a0_bias, sigma_a0)))
            cand[0] *= factor_a0

        # --- first random contiguous block -----------------------
        w1 = int(rng.integers(1, max_width + 1))
        s1 = int(rng.integers(0, n - w1 + 1))
        factor1 = float(np.exp(rng.normal(0.0, sigma_block)))
        cand[s1:s1 + w1] *= factor1

        # --- optional second block (adds extra diversity) -------
        if rng.random() < 0.4:  # occasional second perturbation
            w2 = int(rng.integers(1, max_width + 1))
            s2 = int(rng.integers(0, n - w2 + 1))
            factor2 = float(np.exp(rng.normal(0.0, sigma_block)))
            cand[s2:s2 + w2] *= factor2

        # Safety clipping
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Evaluate candidate
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        # Accept only strict improvements
        if np.isfinite(score) and score < best_score - 1e-12:
            best = cand
            best_score = score
            if best_score < target:
                break

        # (optional) slowly shrink the perturbation magnitude
        sigma_block = max(0.005, sigma_block * 0.9995)
        sigma_a0 = max(0.001, sigma_a0 * 0.9995)

    # ----------------------------------------------------------------
    # Fallback – if something went wrong return the deterministic base
    # ----------------------------------------------------------------
    try:
        _ = evaluate_sequence(best.tolist())
    except Exception:
        best = base

    return [float(x) for x in best]
