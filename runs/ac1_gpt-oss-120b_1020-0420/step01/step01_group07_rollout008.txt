def compute_score(a):
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    m = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    max_c = conv[:2*n-1].max()
    return 2.0 * n * max_c / (sum_a * sum_a)

import numpy as np

# ------------------------------------------------------------
# Helper: compute the AC1 bound for a given sequence (no external calls)
# ------------------------------------------------------------
def _conv_max(seq: np.ndarray, fft_len: int) -> float:
    """Linear convolution via FFT (zero‑padded) and return the maximal entry."""
    ft = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(ft * ft, n=fft_len)
    # first 2*n-1 entries correspond to the linear convolution
    return float(conv[: 2 * seq.shape[0] - 1].max())


def _ac1_score(seq: np.ndarray, fft_len: int) -> float:
    """Return the exact score used by `evaluate_sequence`."""
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    max_c = _conv_max(seq, fft_len)
    n = seq.shape[0]
    return 2.0 * n * max_c / (s * s)

    fft_len = 1 << ((2 * n - 1).bit_length())
