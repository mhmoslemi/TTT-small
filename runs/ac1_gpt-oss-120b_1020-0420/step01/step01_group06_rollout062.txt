def _score_vec(vec):
    """Return the AC1 upper bound for a NumPy vector."""
    return float(evaluate_sequence(vec.tolist()))

def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a non‑negative step‑function (list of heights) whose
    AC1 upper bound is minimized.  The routine tries a few
    analytically‑motivated families (power‑law, exponential,
    combined) and then performs limited local refinements.
    """
    # ------------------------------------------------------------------
    # 1.  Base sequence (parent if supplied, else uniform fallback)
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            parent = np.clip(tmp, 0.0, 1000.0)

    if parent is None:
        n = 256
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size

    best = parent.copy()
    best_score = _score_vec(best)

    # ------------------------------------------------------------------
    # 2.  Parameter grids – size adapted to problem dimension
    # ------------------------------------------------------------------
    if n > 1000:
        n_alpha = 12
        n_beta  = 12
        n_comb  = 12
        n_local = 60
        n_swap  = 60
    else:
        n_alpha = 30
        n_beta  = 30
        n_comb  = 30
        n_local = 200
        n_swap  = 200

    # ------------------------------------------------------------------
    # 3.  Power‑law family a_i ∝ (i+1)^{-α}
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    alphas = np.linspace(0.05, 2.0, n_alpha)
    for a in alphas:
        w = np.power(idx, -a)
        w /= w.sum()
        sc = _score_vec(w)
        if np.isfinite(sc) and sc < best_score:
            best = w.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 4.  Reversed power‑law a_i ∝ (n-i)^{-α}
    # ------------------------------------------------------------------
    rev_idx = np.arange(n, 0, -1, dtype=np.float64)
    for a in alphas:
        w = np.power(rev_idx, -a)
        w /= w.sum()
        sc = _score_vec(w)
        if np.isfinite(sc) and sc < best_score:
            best = w.copy()
            best_score = sc

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    betas = np.logspace(np.log10(5e-5), np.log10(0.02), n_beta)

        s = w.sum()
        if s == 0.0:

        w /= s
        sc = _score_vec(w)
        if np.isfinite(sc) and sc < best_score:
            best = w.copy()
            best_score = sc

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Choose a modest subset of parameters for speed.
    comb_alphas = np.linspace(0.1, 1.2, 6)

    for a in comb_alphas:
