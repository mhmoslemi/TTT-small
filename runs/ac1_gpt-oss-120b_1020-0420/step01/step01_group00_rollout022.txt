import numpy as np

def _prepare(seq):
    """Convert to 1‑D float array, enforce bounds, and avoid degenerate sum."""
    arr = np.asarray(seq, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.sum() < 0.01:
        arr = np.ones_like(arr)
    return arr

def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    The routine uses a tiny stochastic search seeded by the provided parent
    ``height_sequence_1`` (if available)."""
    rng = np.random.default_rng(seed)

    # ----- initialise from the supplied parent or fall back to a uniform vector -----
    try:
        cur = _prepare(height_sequence_1)
    except Exception:
        cur = _prepare(np.ones(256, dtype=np.float64))

    n = cur.size

    # ----- evaluate the starting candidate -----
    try:
        best_score = evaluate_sequence(cur.tolist())
    except Exception:                # defensive – should never happen
        best_score = np.inf
    best_seq = cur.copy()

    # ----- simple stochastic refinement (log‑normal multiplicative noise) -----
    iters = 250                     # modest budget, linear memory
    sigma = 0.07                    # magnitude of the perturbation
    for _ in range(iters):

        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:

        try:

        except Exception:

            best_score = cand_score

    except Exception:
        final_score = np.inf

    if not np.isfinite(final_score):
        fallback = np.full(n, 1.0, dtype=np.float64)   # any positive, bounded vector works
