    cur = cur / cur.sum()
    # apply small random perturbation to break symmetry
    cur = cur * np.exp(rng.normal(0.0, 0.05, size=n))
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur / cur.sum()
    # precompute FFT length
    m = 1 << ((2*n - 1).bit_length())
    best = cur.copy()
    best_score = _fast_score(cur)
    cur_score = best_score
    # schedule parameters
    init_eta = 0.2
    init_T = 0.15
    # tolerance to accept multiple max indices
    tol = 1e-12
    for step in range(num_steps):
        # FFT convolution
        fa = np.fft.rfft(cur, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
        max_c = conv.max()
        # Find indices achieving max (within tolerance)
        max_idx = np.where(conv >= max_c - tol)[0]  # array
        # compute gradient as sum over each max index
        grad = np.zeros_like(cur)
        for k in max_idx:
            # for i where 0 <= k-i < n:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # i runs from lo to hi inclusive
            i = np.arange(lo, hi+1)
            j = k - i
            grad[i] += 2.0 * cur[j]   # add contribution 2*a_j
        # normalize gradient to avoid huge steps? Maybe scale.
        # step size schedule
        eta = init_eta * (1.0 - step / num_steps)
        T = init_T * (1.0 - step / num_steps)
        # gradient descent step
        cand = cur - eta * grad
        # ensure nonnegative
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = cur.copy()
            s = cand.sum()
        cand = cand / s
        cand_score = _fast_score(cand)
        accept = False
        if cand_score < cur_score:
            accept = True

def run_search(start_vec, steps, rng, init_eta=0.2, init_T=0.15):
    cur = start_vec.copy()
    n = cur.size
    m = 1 << ((2*n - 1).bit_length())
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score
    tol = 1e-12
    for step in range(steps):
        fa = np.fft.rfft(cur, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
        max_c = conv.max()
        max_idx = np.where(conv >= max_c - tol)[0]
        grad = np.zeros_like(cur)
        # loop
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi+1, dtype=np.int64)
            j = k - i
            grad[i] += 2.0 * cur[j]
        eta = init_eta * (1.0 - step / steps)
        T = init_T * (1.0 - step / steps)
        cand = cur - eta * grad
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = cur.copy()
            s = cand.sum()
        cand /= s
        cand_score = _fast_score(cand)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        # occasional perturb
        if (step + 1) % 500 == 0 and rng.random() < 0.15:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

import numpy as np

# ----- fast AC1 upper bound (same as the reference implementation) -----
def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----- core search routine (gradient descent on the max‑convolution) -----
def _run_search(start_vec: np.ndarray, steps: int, rng: np.random.Generator,
                init_eta: float = 0.20, init_T: float = 0.15) -> tuple[np.ndarray, float]:
    cur = start_vec.copy()
    n = cur.size
    m = 1 << ((2 * n - 1).bit_length())
    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score
    tol = 1e-12

    for step in range(steps):
        # convolution via FFT
        fa = np.fft.rfft(cur, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]

        max_c = conv.max()
        max_idx = np.where(conv >= max_c - tol)[0]   # all indices that tie for max

        # sub‑gradient of the maximal convolution entry
        grad = np.zeros_like(cur)
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=np.int64)
            j = k - i
            grad[i] += 2.0 * cur[j]

        # annealed step size
        eta = init_eta * (1.0 - step / steps)
        T = init_T * (1.0 - step / steps)

        cand = cur - eta * grad
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = cur.copy()
            s = cand.sum()
        cand /= s

        cand_score = _fast_score(cand)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # occasional random block perturbation (helps escape local minima)
        if (step + 1) % 500 == 0 and rng.random() < 0.15:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, n - width + 1))
            block = cur.copy()
            factor = np.exp(rng.normal(0.0, 0.12))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block /= block.sum()
            block_score = _fast_score(block)
            if block_score < cur_score:
                cur = block
                cur_score = block_score
                if block_score < best_score:
                    best = block.copy()
                    best_score = block_score

    return best, best_score
