import numpy as np
def _fast_score(vec):
    n=vec.size
    m=1 << ((2*n - 1).bit_length())
    fa=np.fft.rfft(vec, n=m)
    conv=np.fft.irfft(fa*fa, n=m)[:2*n-1]
    max_c=conv.max()
    s=vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

    best_score = _fast_score(cur)
    cur_score = best_score

    # schedule
    init_eta = 0.12
    init_T = 0.12

    # momentum buffer
    momentum = np.zeros_like(cur)
    beta_mom = 0.9

    for step in range(num_steps):
        # compute convolution via FFT
        fa = np.fft.rfft(cur, n=m_fft)
        conv = np.fft.irfft(fa * fa, n=m_fft)[: 2 * n - 1]
        max_c = conv.max()

        k = min(top_k, conv.size)
        if k == conv.size:
            top_idx = np.arange(conv.size)
        else:

            part = np.argpartition(conv, -k)
            top_idx = part[-k:]
        # accumulate gradient contributions
        grad_acc = np.zeros_like(cur)
        for kk in top_idx:
            idx = kk - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            # add cur[idx] where mask true
            grad_acc[mask] += cur[idx[mask]]
        # zero-mean gradient to preserve sum
        grad = grad_acc - grad_acc.mean()
        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        # step size schedule
        eta = init_eta * (1.0 - step / num_steps)
        T = init_T * (1.0 - step / num_steps)

            s = cand.sum()

import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation of the AC1 upper bound (identical to the official one)
# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
