import numpy as np

# ------------------------------------------------------------
# Helper: ordered‑pair counts for a binary mask (sparse O(m^2))
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices (0 … n-1) where the coefficient is non‑zero
    returns   : np.ndarray of length 2*n‑1 with ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i) pair
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt
