Implementation of simplex projection:

```python
def simplex_projection(v):
    # v: np.ndarray, not necessarily sum 1.
    # Projects onto {x >=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last True
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np

def _fast_score(a):
    n = a.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # fallback uniform
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    # start best
    best_seq = parent.copy()
    best_score = _fast_score(best_seq)
    # uniform fallback
    uniform = np.ones(n, dtype=np.float64)
    s = _fast_score(uniform)
    if s < best_score:
        best_seq = uniform
        best_score = s
    # analytic families
    i = np.arange(n, dtype=np.float64)
    # offsets and exponents grid
    offsets = np.linspace(0.1, 10.0, 20)   # 20 values
    exponents = np.linspace(0.1, 1.3, 13) # 13 values
    alphas = [0.0, 0.001, 0.005, 0.01, 0.02]
    for p in exponents:
        for off in offsets:
            base = (i + off) ** (-p)
            for alpha in alphas:
                a = base * np.exp(-alpha * i)
                # no need to normalize
                s = _fast_score(a)
                if s < best_score:
                    best_seq = a.copy()
                    best_score = s
                # reversed
                a_rev = a[::-1]
                s = _fast_score(a_rev)
                if s < best_score:
                    best_seq = a_rev.copy()
                    best_score = s
    # exponential-only family (p=0)
    for alpha in alphas[1:]:  # skip 0 (already covered)
        a = np.exp(-alpha * i)
        s = _fast_score(a)
        if s < best_score:
            best_seq = a.copy()
            best_score = s
        a_rev = a[::-1]
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq = a_rev.copy()
            best_score = s
    # random jitter refinement
    max_jitter_iter = 2000
    max_sigma = 0.05
    for it in range(max_jitter_iter):
        sigma = max_sigma * (1.0 - it / max_jitter_iter)
        perturb = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_seq * perturb
        cand = np.clip(cand, 0.0, 1000.0)
        s = _fast_score(cand)
        if s < best_score:
            best_seq = cand
            best_score = s
    # ensure sum >= 0.01, if not scale up
    if np.sum(best_seq) < 0.01:
        best_seq = best_seq * (0.01 / np.sum(best_seq) + 1e-6)
    return [float(x) for x in best_seq.tolist()]

def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a))**2 using FFT."""
    n = a.shape[0]
    m = 2 * n - 1                     # length of full convolution
    nfft = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                     # a * a in frequency domain

    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)
