import numpy as np

# ------------------------------------------------------------
# Helper: ordered‑pair count for a binary mask (set of indices)
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices (0 … n-1) where the coefficient is 1
    returns   : np.ndarray of length 2*n-1 with ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                      # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                      # (i,j) and (j,i)
    return cnt

    """
    Search for a non‑negative step‑function whose AC1 bound
    (as defined by `evaluate_sequence`) is as low as possible.
    The routine combines a fast greedy removal on binary masks,
    a modest power‑law sweep, and a tiny hill‑climb.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine sequence length (use the provided parent if any)
    # --------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0:
            n = int(parent_arr.size)
        else:
            n = 256
    except NameError:
        n = 256

    # -----------------------------------------------------------------
    # Fallback candidate: uniform vector (guaranteed feasible & simple)
    # -----------------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # -----------------------------------------------------------------
    # Also consider using the supplied parent as a starting point
    # -----------------------------------------------------------------
    try:
        parent_vec = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_vec.shape == (n,):
            s = float(evaluate_sequence(parent_vec.tolist()))
            if np.isfinite(s) and s < best_score:
                best_vec = parent_vec.copy()
                best_score = s
    except NameError:
        pass

    # -----------------------------------------------------------------
    # Binary‑mask construction via removal (targets small max pair count)
    # -----------------------------------------------------------------
    thresholds = (6, 5, 4)                # target maximum ordered‑pair count
    attempts_per = 10                     # random restarts per threshold
    INIT_M = min(350, n)                  # start size of random subset
    MIN_M   = 30                          # we never drop below this

    for max_c_target in thresholds:
        for _ in range(attempts_per):
            # random initial subset
            selected = set(rng.choice(n, size=INIT_M, replace=False))
            cnt = _pair_counts(selected, n)
            max_c = int(cnt.max())

            # ----------------------------------------------------
            # Remove indices until the max count respects the target
            # ----------------------------------------------------
            while max_c > max_c_target and len(selected) > MIN_M:
                ks = np.where(cnt == max_c)[0]
                k = int(rng.choice(ks))
                # indices that participate in this sum
                involved = [i for i in selected if (k - i) in selected]
                if involved:
                    rem = int(rng.choice(involved))
                else:
                    rem = int(rng.choice(list(selected)))
                selected.remove(rem)
                cnt = _pair_counts(selected, n)
                max_c = int(cnt.max())

            m = len(selected)
            if m == 0:
                continue

            # inexpensive bound for a binary vector
            cand_est = 2.0 * n * max_c / (m * m)

            # evaluate only if promising
            if cand_est < best_score * 0.99:
                cand_vec = np.zeros(n, dtype=np.float64)
                cand_vec[list(selected)] = 1.0
                s = float(evaluate_sequence(cand_vec.tolist()))
                if np.isfinite(s) and s < best_score:
                    best_vec = cand_vec
                    best_score = s
                    # early exit if already under the target bound
                    if best_score <= 1.45:
                        return best_vec.tolist()

    # ------------------------------------------------------------
    # Power‑law sweep (continuous shapes, often produce low bounds)
    # ------------------------------------------------------------
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.20, 1.50, 66):           # 0.20 … 1.50
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best_vec = cand
            best_score = s
            if best_score <= 1.45:
                return best_vec.tolist()

    # ------------------------------------------------------------
    # Tiny stochastic hill‑climb around the best candidate so far
    # ------------------------------------------------------------
    for _ in range(200):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        cand = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.05)))   # ~5 % perturbation
        cand[start:stop] *= factor

        if rng.random() < 0.15:                       # occasional smoothing
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best_vec = cand
            best_score = s
            if best_score <= 1.45:
                break

    return best_vec.tolist()
