import numpy as np

def propose_candidate(seed: int = 42, **kwargs):
    # Determine parent
    if 'height_sequence_1' in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all():
            parent = np.clip(parent_arr, 0.0, 1000.0)
        else:
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    # helper function
    def _score(arr):
        return float(evaluate_sequence(arr.tolist()))
    
    # Normalize parent to have sum 1 (avoid scaling issues)
    if parent.sum() == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = parent.copy()
        a = a / a.sum()
    best = a.copy()
    best_score = _score(best)

    # 1. Power law families
    indices = np.arange(1, n+1, dtype=np.float64)  # 1..n
    alphas = np.linspace(0.05, 1.20, 80)  # broaden range
    for alpha in alphas:
        w = np.power(indices, -alpha)
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w = w / w_sum
        sc = _score(w)
        if sc < best_score:
            best = w
            best_score = sc

    # 2. Sine shape
    # a_i = sin(pi*(i+0.5)/n)
    t = np.arange(n, dtype=np.float64)
    sin_vec = np.sin(np.pi * (t + 0.5) / n)
    sin_vec = sin_vec / sin_vec.sum()
    sc = _score(sin_vec)
    if sc < best_score:
        best = sin_vec
        best_score = sc

    # 3. Linear ramp (a_i ∝ i+1)
    ramp = (t + 1.0)
    ramp = ramp / ramp.sum()
    sc = _score(ramp)
    if sc < best_score:
        best = ramp
        best_score = sc

    # 4. Random perturbations (block noise) like previous
    rng = np.random.default_rng(seed)
    cand = best.copy()
    for _ in range(300):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 200))
        stop = min(n, start + width)
        # multiplicative log-normal noise
        noise = np.exp(rng.normal(0.0, 0.04, size=stop - start))
        cand[start:stop] *= noise
        cand = np.clip(cand, 0.0, 1000.0)
        # renorm
        total = cand.sum()
        if total > 0:
            cand = cand / total
        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # 5. Gradient descent style reduction of max convolution
    # keep a copy of best for refinement
    a = best.copy()
    fixed_sum = a.sum() if a.sum() > 0 else 1.0
    gamma = 0.5   # step size
    for _ in range(60):
        # convolution
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))
        maxb = conv[k]
        # indices contributing
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        # avoid division by zero
        if maxb == 0:
            break
        factor = 1.0 - gamma * (contrib / maxb)
        # ensure nonnegative
        factor = np.maximum(factor, 0.0)
        a[i_range] = a[i_range] * factor
        # clip and renorm
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s > 0:
            a = a * (fixed_sum / s)
        else:
            break
        sc = _score(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc
        # optional: reduce gamma slowly
        gamma *= 0.97

    # Return best as list
    return [float(v) for v in best]

import numpy as np

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 upper bound
    (as computed by the pre‑imported evaluate_sequence) is as low as possible.
    The routine starts from the provided parent (if any), scans a few parametric
    families, performs stochastic block perturbations and a simple gradient‑like
    refinement that reduces the maximal convolution entry.
    """
    # ----------------------------------------------------------------------
    # 0.  Obtain a parent vector (fallback to uniform ones if unavailable)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            parent = np.clip(tmp, 0.0, 1000.0)
        else:
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # ----------------------------------------------------------------------
    # 1.  Utility to evaluate a candidate (numpy array -> float score)
    # ----------------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        return float(evaluate_sequence(vec.tolist()))

    # ----------------------------------------------------------------------
    # 2.  Normalise the starting vector (scale invariant, sum=1 keeps numbers tiny)
    # ----------------------------------------------------------------------
    if parent.sum() == 0.0:
        base = np.ones(n, dtype=np.float64) / n
    else:
        base = parent.astype(np.float64)
        base = base / base.sum()

    best = base.copy()
    best_score = _score(best)

    # ----------------------------------------------------------------------
    # 3.  Scan a power‑law family a_i ∝ (i+1)^{-α}
    # ----------------------------------------------------------------------
    indices = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    alphas = np.linspace(0.05, 1.20, 80)                    # broader than before
    for a in alphas:
        w = np.power(indices, -a)
        s = w.sum()
        if s == 0.0:
            continue
        w = w / s
        sc = _score(w)
        if sc < best_score:
            best = w
            best_score = sc

    # ----------------------------------------------------------------------
    # 4.  Two deterministic smooth families
    # ----------------------------------------------------------------------
    t = np.arange(n, dtype=np.float64)

    # 4a – shifted sine (positive on (0,π))
    sin_vec = np.sin(np.pi * (t + 0.5) / n)
    sin_vec = sin_vec / sin_vec.sum()
    sc = _score(sin_vec)
    if sc < best_score:
        best = sin_vec
        best_score = sc

    # 4b – linear ramp a_i ∝ i+1
    ramp = t + 1.0
    ramp = ramp / ramp.sum()
    sc = _score(ramp)
    if sc < best_score:
        best = ramp
        best_score = sc

    # ----------------------------------------------------------------------
    # 5.  Light stochastic block perturbations (exploration)
    # ----------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    cand = best.copy()
    for _ in range(300):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 200))
        stop = min(n, start + width)

        # multiplicative log‑normal noise on the chosen block
        noise = np.exp(rng.normal(0.0, 0.04, size=stop - start))
        cand[start:stop] *= noise

        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total > 0.0:
            cand = cand / total
        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ----------------------------------------------------------------------
    # 6.  Gradient‑like reduction of the maximal convolution entry
    # ----------------------------------------------------------------------
    a = best.copy()
    fixed_sum = a.sum() if a.sum() > 0 else 1.0
    gamma = 0.5                     # initial step size
    for _ in range(60):
        conv = np.convolve(a, a)               # full convolution
        k = int(np.argmax(conv))               # index of maximal entry
        maxb = conv[k]
        if maxb <= 0.0:
            break                               # degenerate case

        # All i such that 0 ≤ i < n and 0 ≤ k‑i < n
        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range

        contrib = a[i_range] * a[j_range]       # contribution of each i
        factor = 1.0 - gamma * (contrib / maxb) # shrink proportionally
        factor = np.maximum(factor, 0.0)        # stay non‑negative
        a[i_range] = a[i_range] * factor

        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s <= 0.0:
            break
        a = a * (fixed_sum / s)                # renorm to keep scale

        sc = _score(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc

        gamma *= 0.97                           # slowly decay step size

    # ----------------------------------------------------------------------
    # 7.  Final fine‑tuning with a few more random block moves
    # ----------------------------------------------------------------------
    cand = best.copy()
    for _ in range(150):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 250))
        stop = min(n, start + width)

        noise = np.exp(rng.normal(0.0, 0.03, size=stop - start))
        cand[start:stop] *= noise

        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total > 0.0:
            cand = cand / total
        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ----------------------------------------------------------------------
    # 8.  Return the best candidate found as a plain Python list
    # ----------------------------------------------------------------------
    return [float(v) for v in best]
