    start = int(n * start_frac)

# Helper: exact score (same as evaluator but fast)

def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv(seq,seq))/sum(seq)^2 after clipping."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    n = seq.size
    # FFT based convolution (full length 2n-1)
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[: 2 * n - 1]
    m = conv.max()
    return float(2.0 * n * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i   for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Produce a candidate by shrinking a region and scaling the rest.
# ------------------------------------------------------------
def _shrink_region(seq: np.ndarray, start_frac: float, end_frac: float, factor: float) -> np.ndarray:
    """Shrink entries in [start_frac, end_frac) by *factor* and renormalise sum."""
    n = seq.size
    start = int(n * start_frac)
    end = int(n * end_frac)
    if start >= end:
        return None
    mid = seq[start:end]
    sum_mid = mid.sum()
    new_mid = mid * factor
    delta = sum_mid - new_mid.sum()          # mass to redistribute
    other_sum = seq.sum() - sum_mid
    if other_sum <= 0.0:
        return None
    g = 1.0 + delta / other_sum              # uniform scaling factor for the rest
    new_seq = seq.copy()
    new_seq[start:end] = new_mid
    if start > 0:
        new_seq[:start] *= g
    if end < n:
        new_seq[end:] *= g
    return new_seq


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size and an initial candidate.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
            init = parent.copy()
        else:
            n = 256
            init = _catalan_seq(n)
    else:
        n = 256
        init = _catalan_seq(n)

    # If the parent is worse than the Catalan baseline, replace it.
    catalan = _catalan_seq(n)
    best = init
    best_score = _score(best)
    cat_score = _score(catalan)
    if cat_score < best_score:
        best = catalan
        best_score = cat_score

    # --------------------------------------------------------
    # Deterministic annealing: shrink selected regions.
    # --------------------------------------------------------
    regions = [
        (0.0, 0.3),   # early‑to‑mid part
        (0.3, 0.7),   # central band
        (0.0, 0.2),   # very early prefix
        (0.2, 0.5),   # left‑middle
        (0.5, 0.8),   # right‑middle
    ]
    factors = np.linspace(0.5, 0.99, 11)   # shrink to 50 % … 99 %
    for _ in range(4):                    # a few passes
        improved = False
        for fr, to in regions:
            for f in factors:
                cand = _shrink_region(best, fr, to, f)
                if cand is None:
                    continue
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc
                    improved = True
        if not improved:
            break

    # --------------------------------------------------------
    # Stochastic refinement – mass transfers and tiny additions.
    # --------------------------------------------------------
    steps = kwargs.get("num_steps", 1500)
    n = best.size
    for _ in range(steps):
        # 30 % probability: pure addition in the far tail
        if rng.random() < 0.30:
            idx = rng.integers(int(0.8 * n), n)
            delta = best.sum() * rng.uniform(0.0005, 0.003)   # 0.05 %–0.3 % of total mass
            cand = best.copy()
            cand[idx] = min(cand[idx] + delta, 1000.0)
        else:
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst or best[src] <= 0.0:
                continue
            delta = best[src] * rng.uniform(0.01, 0.10)        # move 1 %–10 % of source
            cand = best.copy()
            cand[src] -= delta
            cand[dst] = min(cand[dst] + delta, 1000.0)

        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------
    # Return as a plain Python list (required by the harness).
    # --------------------------------------------------------
    return [float(v) for v in best]
