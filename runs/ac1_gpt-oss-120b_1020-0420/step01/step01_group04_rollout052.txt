import numpy as np

def _coefficients_binom_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs

def _random_block_scale(arr, rng):
    n = arr.size
    min_len = max(1, n // 200)  # about 0.5% of n
    max_len = max(2, n // 8)    # up to ~12.5% of n
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # log‑normal factor
    new = arr.copy()
    new[start:start+width] *= factor
    np.clip(new, 0.0, 1000.0, out=new)
    return new

import numpy as np

# ------------------------------------------------------------
# Helper functions (FFT‑based convolution, scoring, moves)
# ------------------------------------------------------------

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of a with itself using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n                     # zero‑pad length >= 2n‑1
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)
    return conv[: 2 * n - 1]


def _score_of(a: np.ndarray) -> float:
    """
    Exact AC1 bound (identical to evaluate_sequence) for a non‑negative array.
    """
    S = a.sum()
    if S < 0.01:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (S * S))


def _random_block_scale(arr: np.ndarray, rng: np.random.Generator):
    """Scale a random contiguous block (log‑normal factor) and score."""
    n = arr.size
    min_len = max(1, n // 200)          # ~0.5 % of n
    max_len = max(2, n // 8)            # up to ~12.5 % of n
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # modest log‑normal perturbation
    cand = arr.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand, _score_of(cand)


def _gradient_swap(arr: np.ndarray, rng: np.random.Generator):
    """
    Move a small amount of mass from a high‑gradient index to a low‑gradient one.
    Returns (new_array, new_score) or (None, inf) if no move is possible.
    """
    n = arr.size
    conv = _full_convolution(arr)
    max_b = conv.max()
    m = int(conv.argmax())
    S = arr.sum()
    if S < 0.01:
        return None, np.inf

    idx = np.arange(n)
    partner = m - idx
    mask = (partner >= 0) & (partner < n)

    # gradient g_i = a_i + a_{m-i} (if partner inside range)
    g = arr.copy()
    g[mask] += arr[partner[mask]]

    thresh = 2.0 * max_b / S
    high = np.where(g > thresh)[0]
    low = np.where(g < thresh)[0]

    # source must have positive mass
    high_pos = high[arr[high] > 0.0]
    if high_pos.size == 0 or low.size == 0:
        return None, np.inf

    i = int(rng.choice(high_pos))
    j = int(rng.choice(low))

    # transfer a modest fraction of a[i] to a[j]
    frac = float(rng.uniform(0.01, 0.30))          # 1 % – 30 % of source
    delta = min(arr[i] * frac, arr[i])
    if delta <= 0.0:
        return None, np.inf

    cand = arr.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand, _score_of(cand)

            break

        # pick a move
        if rng.integers(0, 2) == 0:
            cand, cand_score = _gradient_swap(best, rng)
            if cand is None:
                # gradient swap failed – fall back to block move
                cand, cand_score = _random_block_scale(best, rng)
        else:
            cand, cand_score = _random_block_scale(best, rng)

        if not np.isfinite(cand_score):
            continue

        if cand_score < best_score:
            best = cand
            best_score = cand_score
            stagnation = 0
        else:
            stagnation += 1

        # simple restart if stuck for a while
        if stagnation >= 2000:
            # perturb the original base a little and restart
            best, best_score = _random_block_scale(base, rng)
            stagnation = 0

    # -------------------------------------------------------------
    # Return the best candidate as a plain Python list of floats
    # -------------------------------------------------------------
    return [float(x) for x in best]
```
