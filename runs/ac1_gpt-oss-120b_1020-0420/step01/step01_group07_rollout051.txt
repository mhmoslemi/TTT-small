# ------------------------------------------------------------
# Fast AC1 bound using FFT (linear convolution via zero‑padding)
# ------------------------------------------------------------
def _fast_ac1_bound(a: np.ndarray) -> float:
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << ((size - 1).bit_length())
    buf = np.zeros(fft_len, dtype=np.float64)
    buf[:n] = a
    F = np.fft.rfft(buf)
    conv = np.fft.irfft(F * F, n=fft_len)[:size]
    max_b = conv.max()
    s = a.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)

        cand[start:start + width] *= factor
        # occasional symmetric smoothing
        if rng.random() < 0.1:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        fast = _fast_ac1_bound(cand)
        if fast < best_fast - 1e-8:
            score = _exact_score(cand)
            if score < best_score:
                best_vec, best_score, best_fast = cand, score, fast
                if best_score <= 1.50274349:
                    break

    return [float(x) for x in best_vec]
