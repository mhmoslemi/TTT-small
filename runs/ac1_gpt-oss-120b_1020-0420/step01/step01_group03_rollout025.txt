def project_simplex(v, s):
    # ensure s > 0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _project_to_simplex(v, s):
    # v: numpy array
    # s: desired sum (positive)
    # returns w >=0, sum(w)=s, closest to v in Euclidean norm
    # algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0

import numpy as np

def _project_to_simplex(v, s):
    v = v.copy()
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _relocate_mass(a, g, lr, tail_start):
    a_temp = a - lr * g
    a_temp = np.maximum(a_temp, 0.0)
    removed = a.sum() - a_temp.sum()

        add = removed / tail_len
        a_temp[tail_start:] += add
    return np.clip(a_temp, 0.0, 1000.0)

import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the classic root‑constant sequence (fallback)
# ----------------------------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Helper: move removed mass to the trailing part of the vector
# ----------------------------------------------------------------------
def _relocate_mass(a: np.ndarray, g: np.ndarray, lr: float, tail_start: int) -> np.ndarray:
    """One gradient step: decrease a by lr*g, keep non‑negative,
    then put the removed ℓ1‑mass uniformly onto the tail region."""
    a_tmp = a - lr * g
    a_tmp = np.maximum(a_tmp, 0.0)                # non‑negative
    removed = a.sum() - a_tmp.sum()               # how much ℓ1‑mass we lost
    if removed > 0.0:
        tail_len = a.shape[0] - tail_start
        if tail_len > 0:
            add = removed / tail_len
            a_tmp[tail_start:] += add
    return np.clip(a_tmp, 0.0, 1000.0)

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):

    it exists, then performs a modest number of gradient‑type
    updates that shift mass from early indices – which dominate
    the convolution peak – to the trailing third of the vector.
    The best sequence seen during the run is returned.
    """
    # ------------------------------------------------------------------
    # Initialise --------------------------------------------------------
    # ------------------------------------------------------------------
    try:
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    except Exception:                     # no parent supplied
        a = _root_constant_seq(256)

    n = a.shape[0]

    # ------------------------------------------------------------------
    # Parameters --------------------------------------------------------
    # ------------------------------------------------------------------
    lr = 0.20                              # step size for gradient‑like move
    max_iters = 150                        # limited to keep runtime modest
    tail_start = n * 2 // 3                # last third of indices receive mass

    # keep the best candidate seen
    best_seq = a.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = float('inf')

    # ------------------------------------------------------------------
    # Iterative mass‑relocation loop ------------------------------------
    # ------------------------------------------------------------------
    for _ in range(max_iters):
        # full convolution and current worst peak
        b = np.convolve(a, a)
        M = b.max()

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:

            g[idx] = a[k - idx]

        # one update: pull mass from the coefficients that raise the peak,

        a_new = _relocate_mass(a, g, lr, tail_start)
