import numpy as np

# ------------------------------------------------------------
# Helper: generate the truncated series of (1 - z)^(-1/2)
# a_0 = 1, a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return an array of size ``length`` with entries binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

    """
    Return (max_value, index_of_max) of the full convolution a*a
    using an FFT of length ``fft_len`` (must be a power of two ≥ 2*len(a)-1).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.shape[0] - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 200, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Initialise from the provided parent if it exists,
    # otherwise fall back to the analytic half‑power series.
    # --------------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        base = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    else:
        n_default = 256
        base = generate_half_power_series(n_default)

    n = base.size
    np.clip(base, 0.0, 1000.0, out=base)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = best.copy()

    # FFT length: smallest power of two ≥ 2*n-1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Parameters for the hill‑climbing schedule
    # --------------------------------------------------------
    max_delta = 0.04   # initial reduction factor
    min_delta = 0.005  # final reduction factor
    no_improve_counter = 0

    for step in range(num_steps):
        # ---- convolution max and its index (FFT, O(n log n)) ----
        _, idx_max = conv_max_fft(cur, fft_len)

        # ---- contributions to that max (pairwise products) ----
        i_start = max(0, idx_max - (n - 1))
        i_end = min(n - 1, idx_max)
        i_range = np.arange(i_start, i_end + 1)
        j_range = idx_max - i_range
        contribs = cur[i_range] * cur[j_range]

        # ---- pick the most influential pair ----
        local_max = int(np.argmax(contribs))
        i_max = int(i_range[local_max])
        j_max = int(j_range[local_max])

        # ---- annealed reduction factor ----
        delta = max_delta - (max_delta - min_delta) * (step / (num_steps - 1))

        cand = cur.copy()

        # ---- reduce the two entries slightly ----
        red_i = cand[i_max] * delta
        cand[i_max] -= red_i
        total_red = red_i
        if j_max != i_max:
            red_j = cand[j_max] * delta
            cand[j_max] -= red_j
            total_red += red_j

        # ---- relocate the removed mass to a random location ----
        recv = int(rng.integers(0, n))
        while recv == i_max or recv == j_max:
            recv = int(rng.integers(0, n))
        cand[recv] += total_red

        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            cur = cand
            no_improve_counter = 0
            continue  # successful improvement, go to next iteration

        # ---- no improvement: occasionally apply a random block perturbation ----
        no_improve_counter += 1
        if no_improve_counter >= 30:
            block_width = max(1, n // 20)
            width = int(rng.integers(1, block_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            trial = cur.copy()
            trial[start : start + width] *= factor
            np.clip(trial, 0.0, 1000.0, out=trial)
            trial_score = float(evaluate_sequence(trial.tolist()))
            if np.isfinite(trial_score) and trial_score < best_score:
                best = trial
                best_score = trial_score
                cur = trial
                no_improve_counter = 0
            else:
                no_improve_counter = 0  # reset even if not accepted

    # --------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # --------------------------------------------------------
    return [float(x) for x in best]
