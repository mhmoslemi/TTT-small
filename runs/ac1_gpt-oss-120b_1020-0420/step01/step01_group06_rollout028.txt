def evaluate_sequence(sequence):
    if not isinstance(sequence, list): return inf
    if not sequence: return inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int,float)): return inf
        if np.isnan(x) or np.isinf(x): return inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0,x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01: return inf
    return float(2 * n * max_b / (sum_a**2))

def gradient_descent(a, steps, eta0):
    a = np.maximum(a, 0.0)  # ensure nonneg
    a = a / a.sum()
    best_local = a.copy()

import numpy as np

# -------------------------------------------------------------
#  Convolution (max value + its index) – FFT based, O(n log n)
# -------------------------------------------------------------
def _conv_max_idx(a: np.ndarray):
    """Return (max convolution value, index) of a*a."""
    n = a.shape[0]
    size = 2 * n - 1
    # real FFT – works for any size
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    idx = int(np.argmax(conv))
    M = float(conv[idx])
    return M, idx


# -------------------------------------------------------------
#  Gradient of the AC1 upper bound (sub‑gradient)
# -------------------------------------------------------------
def _gradient(a: np.ndarray):
    """
    Compute sub‑gradient of the objective 2*n*max(a*a)/(sum a)^2.
    Returns (grad, max_value, sum_a, max_index).
    """
    n = a.shape[0]
    S = a.sum()
    if S == 0:
        # degenerate case – zero gradient
        return np.zeros_like(a), 0.0, 0.0, 0
    M, idx = _conv_max_idx(a)

    # constants (the expression is scale‑invariant, but we keep them
    # explicit for clarity)
    const = (4.0 * n * M) / (S ** 3)
    coef = (2.0 * n) / (S ** 2)

    # b_i = a_{idx-i} when the index is in range, else 0
    b = np.zeros_like(a)
    lo = max(0, idx - (n - 1))
    hi = min(n - 1, idx)
    if lo <= hi:
        i_range = np.arange(lo, hi + 1)
        b[lo:hi + 1] = a[idx - i_range]

    grad = coef * b - const
    return grad, M, S, idx


# -------------------------------------------------------------
#  Simple projected gradient descent on the feasible set
# -------------------------------------------------------------
def _descend(init: np.ndarray, steps: int = 400, eta0: float = 0.02):
    """
    Starting from `init` (any non‑negative vector), run a few
    projected gradient steps.  The vector is kept non‑negative and
    re‑normalised to unit L1 norm after each step.
    Returns the final vector and its fast AC1 score (2*n*max).
    """
    a = np.maximum(init, 0.0)
    if a.sum() == 0:
        a = np.ones_like(a)
    a = a / a.sum()
    n = a.shape[0]

    for t in range(steps):
        eta = eta0 * (0.99 ** t)
        g, _, _, _ = _gradient(a)
        a = a - eta * g
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0:
            a = np.ones_like(a) / n
        else:
            a = a / total

    M, _ = _conv_max_idx(a)
    fast_score = 2.0 * n * M / (a.sum() ** 2)  # sum is 1 → 2*n*M
    return a, fast_score

    Generate a non‑negative step‑function (list of floats) that strives
    to minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    """

    # ---------------------------------------------------------
    #  Determine length and a sensible fallback parent
    # ---------------------------------------------------------
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            n = tmp.size
            parent = np.clip(tmp, 0.0, 1000.0)
        else:

            parent = np.ones(n, dtype=np.float64)
    else:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Normalise parent to unit L1 norm (scale does not matter)
    if parent.sum() == 0:
        parent = np.ones_like(parent)
    parent = parent / parent.sum()

    # ---------------------------------------------------------
    #  Initial best candidate (fallback)
    # ---------------------------------------------------------
    best_vec = parent.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # ---------------------------------------------------------
    #  Helper to possibly improve the current best
    # ---------------------------------------------------------
    def _maybe_update(vec):
        nonlocal best_vec, best_score
        sc = float(evaluate_sequence(vec.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_vec = vec.copy()

    # ---------------------------------------------------------

    # ---------------------------------------------------------
    seeds = [parent]

    # Power‑law family a_i ∝ (i+1)^{-α}

    # ---------------------------------------------------------

        best_vec = np.ones(n, dtype=np.float64) / n
    return [float(v) for v in best_vec]
