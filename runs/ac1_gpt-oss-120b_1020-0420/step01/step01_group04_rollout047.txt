import numpy as np
import cvxpy as cp
from numpy.fft import fft, ifft

def _coefficients_binom_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0*i - 1.0) / (2.0*i)
    return coeffs

def _initial_sequence():
    # use parent if available else analytic
    if 'height_sequence_1' in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 4806
        seq = _coefficients_binom_half(n)
    # clip for safety
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq

def _max_convolution(seq):
    # compute full convolution using FFT for speed
    # length 2n-1
    # Use real FFT?
    f = np.fft.rfft(seq, n*2-1)
    # Actually we need convolution not correlation; using FFT product
    conv = np.fft.irfft(f * f, n*2-1)
    # Ensure numerical errors
    conv = np.where(conv < 0, 0, conv)
    return np.max(conv)

def _solve_lp(f, M):
    n = f.shape[0]
    g = cp.Variable(n)
    # constraints
    conv_expr = cp.conv(f, g)  # length 2n-1
    constraints = [
        g >= 0,
        g <= 1000,
        conv_expr <= M
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, max_iter=500, warm_start=False, verbose=False)
    if g.value is None:
        # fallback: return f unchanged
        return f.copy()
    g_val = np.array(g.value, dtype=np.float64)
    # numerical safety: clip
    np.clip(g_val, 0.0, 1000.0, out=g_val)
    return g_val

def propose_candidate(seed=42, max_iters=10, **kwargs):
    rng = np.random.default_rng(seed)
    seq = _initial_sequence()
    # compute initial best and score
    best = seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # fallback parent checked as best also if exists
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    target = 1.50274349

    f = best.copy()
    for it in range(max_iters):
        M = _max_convolution(f)
        # solve LP to maximize sum under cross-convolution bound
        g = _solve_lp(f, M)
        # evaluate candidate g
        try:
            score_g = float(evaluate_sequence(g.tolist()))
        except Exception:
            score_g = np.inf
        # also evaluate convex combination
        t = 0.5
        candidate = (1 - t) * f + t * g
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        try:
            score_c = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            score_c = np.inf
        # select best among f, g, candidate
        if score_g < best_score:
            best = g.copy()
            best_score = score_g
        if score_c < best_score:
            best = candidate.copy()
            best_score = score_c
        # update f to the better of candidate and g for next iteration
        if score_c < score_g:
            f = candidate.copy()
        else:
            f = g.copy()
        if best_score < target:
            break

    # final fallback: ensure nonempty and valid
    if best_score == np.inf or best.size == 0:
        # fallback simple uniform sequence that yields metric 2 (still finite)
        n = seq.shape[0]
        uniform = np.full(n, 1.0)
        best = uniform

def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2 :  C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _initial_sequence() -> np.ndarray:
    """
    Return the starting sequence.
    Prefer the provided parent `height_sequence_1`; otherwise use the
    analytic (1‑z)^‑1/2 coefficients.
    """
    if 'height_sequence_1' in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 4806
        seq = _coefficients_binom_half(n)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _max_convolution(seq: np.ndarray) -> float:
    """Fast full convolution max using FFT (real‑valued input)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())           # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    conv = conv[: 2 * n - 1]                         # keep the required part
    # numerical noise may produce tiny negatives
    conv[conv < 0] = 0.0
    return float(np.max(conv))


def _solve_lp(fixed: np.ndarray, bound: float) -> np.ndarray:
    """
    Linear programme:
        maximise   sum(g)
        subject to g >= 0,
                   g <= 1000,
                   conv(fixed, g) <= bound   (element‑wise)
    Returns the optimiser's vector; if the solver fails, returns a copy of
    ``fixed``.
    """
    if not _CVX_AVAILABLE:                     # pragma: no cover
        return fixed.copy()

    n = fixed.shape[0]
    g = cp.Variable(n)
    # linear convolution of a constant vector with the variable
    conv_expr = cp.conv(fixed, g)             # length 2n‑1
    constraints = [
        g >= 0,
        g <= 1000,
        conv_expr <= bound,
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, verbose=False, max_iters=2000)
    except Exception:                         # pragma: no cover
        return fixed.copy()

    if g.value is None:
        return fixed.copy()
    g_val = np.asarray(g.value, dtype=np.float64)
    np.clip(g_val, 0.0, 1000.0, out=g_val)
    return g_val


def _random_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """A cheap stochastic perturbation that conserves non‑negativity."""
    n = seq.size
    cand = seq.copy()
    op = rng.integers(0, 5)
    if op == 0:                     # swap two entries
        i, j = rng.integers(0, n, size=2)
        cand[i], cand[j] = cand[j], cand[i]
    elif op == 1:                   # scale a random block
        min_len = max(1, n // 200)          # ≈0.5 % of n
        max_len = max(2, n // 10)           # ≈10 % of n
        width = int(rng.integers(min_len, max_len + 1))
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.14)))   # mild log‑normal
        cand[start:start + width] *= factor
    elif op == 2:                   # transfer a small amount between two positions
        i, j = rng.integers(0, n, size=2)

            cand[i] -= delta
            cand[j] += delta
    elif op == 3:                   # add a tiny positive bump
