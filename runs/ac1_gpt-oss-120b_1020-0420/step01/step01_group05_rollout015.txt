def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1

    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

        s = _fast_score(cand)
        if s < best_score:
            best_seq, best_score = cand, s
            # also examine its reversal
            cand_rev = cand[::-1]
            s_rev = _fast_score(cand_rev)
            if s_rev < best_score:
                best_seq, best_score = cand_rev, s_rev
        # anneal sigma (never go below a small floor)
        sigma = max(0.005, sigma * 0.9995)

    # ------------------------------------------------------------------
    # Final sanity check and return.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_seq.tolist()]
