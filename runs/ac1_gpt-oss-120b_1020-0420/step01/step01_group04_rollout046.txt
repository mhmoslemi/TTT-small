import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2: C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _make_block_candidate(base, rng):
    """Multiply a random contiguous block of `base` by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    min_len = max(1, n // 100)          # 1 % of n
    max_len = max(2, n // 4)            # up to 25 % of n
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.25)))  # moderate perturbation
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _mass_transfer_step(seq, rng):
    """
    Reduce the contribution to the current maximal convolution index by
    moving a small fraction of mass from a heavy index to a random light one.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_idx = int(np.argmax(conv))

    # indices participating in the maximal entry
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    i_inds = np.arange(lo, hi + 1, dtype=np.int64)
    j_inds = max_idx - i_inds

    # contributions a[i]*a[j]
    contrib = seq[i_inds] * seq[j_inds]
    top_k = min(5, contrib.size)
    # pick one of the largest contributors
    idxs = np.argpartition(-contrib, top_k - 1)[:top_k]
    sel = rng.choice(idxs)
    i = i_inds[sel]
    j = j_inds[sel]

    # choose which side to reduce
    red = i if rng.random() < 0.5 else j

    # choose a target index different from the reduced one
    tar = rng.integers(0, n)
    while tar == red:
        tar = rng.integers(0, n)

    # amount to move: a modest fraction of the reduced entry
    frac = rng.uniform(0.02, 0.07)          # 2 %–7 %
    amt = seq[red] * frac
    if amt <= 0.0:
        return seq.copy()                  # nothing to do

    new_seq = seq.copy()
    new_seq[red] -= amt
    new_seq[tar] += amt
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq
