def refine_with_gradient(a0, max_iter=800, base_lr=0.5):

    best_local = a.copy()
    best_local_score = score_vec(a)

        # locate indices with max value
        max_val = conv.max()
        max_idxs = np.flatnonzero(conv == max_val)
        # pick one randomly among maxes
        idx = rng.choice(max_idxs)
        # transform idx to shift distance <= n-1
        if idx > n-1:
            shift = 2*n-2 - idx
        else:
            shift = idx
        # compute gradient vector for this shift
        if shift == 0:
            g = 2 * a
        else:
            g = np.zeros_like(a)
            # contributions
            g[:n-shift] += a[shift:]
            g[shift:] += a[:n-shift]
        # center gradient to zero mean
        g_centered = g - g.mean()
        # learning rate schedule: decay
        lr = base_lr * (1.0 - it / max_iter)
        # multiplicative update
        factor = np.exp(-lr * g_centered)
        a = a * factor
        # renormalize
        total = a.sum()
        if total <= 0:
            a = np.ones_like(a) / n
            total = 1.0
        else:
            a = a / total

        # occasional random perturbation
        if (it & 31) == 0:  # every 32 steps
            # choose random block
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 100))
            stop = min(n, start + width)
            noise = np.exp(rng.normal(0.0, 0.03, size=stop - start))
            a[start:stop] *= noise
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        sc = score_vec(a)
        if sc < best_local_score:
            best_local = a.copy()
            best_local_score = sc
        # also keep global best
        if sc < best_score:
            best = a.copy()
            best_score = sc
        # optional early stop
        if it % 100 == 0 and best_local_score <= best_score * 0.99:
            # we already improved substantially
            pass

    return best.copy()
