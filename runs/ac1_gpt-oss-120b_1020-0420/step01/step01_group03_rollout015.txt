        # convolution via FFT for speed
        conv = np.fft.ifft(np.fft.fft(a, n*2) * np.fft.fft(a, n*2)).real
        # conv length 2n-1 (additional zero), we need first 2n-1 elements
        conv = conv[:2*n-1]

        k = int(conv.argmax())
        # compute gradient: for i where 0 <= i < n and 0 <= k-i < n:
        i_lo = max(0, k-(n-1))
        i_hi = min(k, n-1)
        # gradient vector

        # Perform gradient descent step
        a_new = a - alpha * grad

        # Clip to nonnegative
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Scale to maintain original sum (or increase sum?)
        sum_old = a.sum()
        sum_new = a_new.sum()
        if sum_new > 0:
            scale = sum_old / sum_new
            a_new *= scale
        else:
            break

        # Additional small epsilon for numerical stability
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Compute score (optional check)
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            # fallback break if evaluation fails
            break

        # Keep best
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        # Early stop if improvement not large
