        p = rng.uniform(0.3, 1.5)
        alpha = rng.uniform(0.0, 0.0015)
        idx = np.arange(n, dtype=np.float64)
        cand = (idx + 1.0) ** (-p) * np.exp(-alpha * idx)
        np.clip(cand, 0.0, 1000.0, out=cand)
        if cand.sum() < 0.01:
            cand *= 0.01 / cand.sum()
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

    # ------------------------------------------------------------
    # Gradient‑descent–style refinement (matrix‑free, O(n) memory)
    # ------------------------------------------------------------
    max_gd_iters = 400
    step_factor = 0.12          # relative step size (as fraction of mean value)
    no_improve = 0
    for it in range(max_gd_iters):
        a = best
        sum_a = a.sum()
        if sum_a == 0.0:
            break

        # full convolution and location of the maximal entry
        conv = np.convolve(a, a)
        max_b = conv.max()
        L = int(conv.argmax())

        # vector a_rev where a_rev[i] = a[L-i] if index in range, else 0
        idx = L - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        a_rev = np.zeros_like(a)
        a_rev[mask] = a[idx[mask]]

        # gradient of the AC1 bound
        c1 = 2.0 * n / (sum_a * sum_a)
        c2 = 4.0 * n * max_b / (sum_a * sum_a * sum_a)
        grad = c1 * a_rev - c2

        # propose a step opposite to the gradient
        avg = sum_a / n
        step = step_factor * avg
        cand = a - step * grad
        np.clip(cand, 0.0, 1000.0, out=cand)

        # renormalise the sum (scaling does not affect the objective)
        s = cand.sum()
        if s > 0.0:
            cand *= (sum_a / s)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 12:          # stagnation – shrink step
                step_factor *= 0.55
                no_improve = 0
                if step_factor < 1e-6:
                    break

    # ------------------------------------------------------------
    # Stochastic block perturbations – fine‑grained local search
    # ------------------------------------------------------------
    num_steps = 2000
    max_width = max(1, n // 20)
    for step_id in range(num_steps):
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        # variance decays linearly
        sigma = 0.30 * (1.0 - step_id / num_steps)
        factor = float(np.exp(rng.normal(0.0, sigma)))
        cand = best.copy()
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        # keep the total mass (optional – not required for the bound)
        s = cand.sum()
        if s > 0.0:
            cand *= (best.sum() / s)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------
    # Return a plain Python list as required by the harness
    # ------------------------------------------------------------
    return [float(x) for x in best]
