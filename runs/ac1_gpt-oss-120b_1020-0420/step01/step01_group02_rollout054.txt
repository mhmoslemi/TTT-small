    A = np.fft.rfft(a, n=size)

def full_convolution(a):
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m).bit_length())
    # FFT
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)
    return conv[:m]

import numpy as np

# ------------------------------------------------------------
# Generate the (1 - z)^(-1/2) coefficient series.
# a_0 = 1, a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Linear convolution via FFT (linear, not circular).
# Returns the full convolution array of length 2*n-1.
# ------------------------------------------------------------
def full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (m.bit_length())
    X = np.fft.rfft(x, n=size)
    Y = X * X                     # convolution in frequency domain (real * real)
    conv = np.fft.irfft(Y, n=size)
    return conv[:m]               # keep only the linear part


# ------------------------------------------------------------
# Greedy solver for: max sum(g)  s.t. f * g <= M (elementwise)
# f is fixed, M = max(f * f).  Returns g0 (not normalized).
# ------------------------------------------------------------
def greedy_max_sum_g(f: np.ndarray, M: float) -> np.ndarray:
    n = f.shape[0]
    conv = np.zeros(2 * n - 1, dtype=np.float64)   # current f * g values
    g = np.zeros_like(f)

    # indices sorted by increasing f_i – cheaper variables first
    order = np.argsort(f, kind='mergesort')
    for i in order:
        fi = f[i]
        if fi <= 0.0:
            continue
        lo = i
        hi = i + n - 1
        residual = M - conv[lo:hi + 1]
        # smallest slack among the constraints that involve g_i
        min_res = residual.min()
        if min_res <= 0.0:
            continue
        delta = min_res / fi
        if delta <= 0.0:
            continue
        g[i] = delta
        conv[lo:hi + 1] += fi * delta
    return g

        n = 256
        a = generate_half_power_series(n)

    # Normalise to unit sum (scale‑invariant for the objective)
    total = a.sum()
    if total <= 0.0:
        a = np.full_like(a, 1.0 / a.size, dtype=np.float64)
    else:
        a = a / total

    n = a.size
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------
    # Iteratively improve using the LP‑style update described in the statement
    # --------------------------------------------------------
    for outer in range(max_outer_iters):
        # current autocorrelation and its maximum
        conv = full_convolution(a)
        M = conv.max()

        # solve the linear program (greedy analogue)
        g0 = greedy_max_sum_g(a, M)
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # rescale to keep the same L1‑norm (here = 1)
        g = g0 / sum_g0

        # enforce evaluator bounds before mixing
        np.clip(g, 0.0, 1000.0, out=g)
        sg = g.sum()
        if sg > 0.0:
            g /= sg

        # mixing coefficient – decreasing schedule
        t = max(0.01, 0.2 / (outer + 1))

        cand = (1.0 - t) * a + t * g
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = cand.sum()
        if sc > 0.0:
            cand /= sc

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand                     # continue from the improved point
        # optional early stop if we are already close to the theoretical optimum
        if best_score <= 1.01:
            break

    # --------------------------------------------------------
    # Light stochastic fine‑tuning (block perturbations)
    # --------------------------------------------------------
    for _ in range(random_perturb_steps):
        cand = best.copy()
        if n == 1:
            width = 1
            start = 0
        else:
            max_width = max(1, n // 20)
            width = int(rng.integers(1, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.07)))
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        sm = cand.sum()
        if sm > 0.0:
            cand /= sm
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

    # --------------------------------------------------------
    # Return as plain Python list of floats
    # --------------------------------------------------------
    return [float(v) for v in best]
