import numpy as np

# ------------------------------------------------------------
# Utility: fast full convolution via FFT (real input)
# ------------------------------------------------------------
def _full_convolve(a: np.ndarray) -> np.ndarray:
    """
    Return the full convolution a*a using an FFT of length
    next power of two >= 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ------------------------------------------------------------
# Compute the AC1 upper bound for a (non‑negative) vector.
# The scaling of ``a`` does not affect the bound, so the routine
# works with any positive sum (the routine works also with sum=1).
# ------------------------------------------------------------
def _bound(a: np.ndarray) -> float:
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    b = _full_convolve(a)
    max_b = b.max()
    return float(2.0 * a.shape[0] * max_b / (s * s))


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 500, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    AC1 upper bound defined by `evaluate_sequence`.  The routine
    uses a cheap parametric sweep (power‑law), a few random restarts,
    and a simple projected gradient descent that reduces the peak of
    the self‑convolution.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Determine length ``n`` (use the parent construction if present)
    # ----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
            init_vec = parent.copy()
        else:
            n = 256
            init_vec = np.ones(n, dtype=np.float64)
    except NameError:
        n = 256
        init_vec = np.ones(n, dtype=np.float64)

    # ----------------------------------------------------------------
    # Normalise the initial vector (scaling does not influence the bound)
    # ----------------------------------------------------------------
    init_vec = np.clip(init_vec, 0.0, 1000.0)
    if init_vec.sum() == 0.0:
        init_vec[:] = 1.0
    init_vec = init_vec / init_vec.sum()

    # ----------------------------------------------------------------
    # Keep the best candidate seen so far (fallback uniform vector)
    # ----------------------------------------------------------------
    best_vec = init_vec.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))
    TARGET = 1.50274349

    # ----------------------------------------------------------------
    # 1) Power‑law sweep (covers the analytically known optimum ~0.5)
    # ----------------------------------------------------------------
    p_grid = np.linspace(0.30, 0.80, 51)          # 0.30 … 0.80 in steps of 0.01
    for p in p_grid:
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a = a / a.sum()
        sc = _bound(a)                           # fast bound, no clipping needed
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            if best_score <= TARGET:
                return [float(x) for x in best_vec]

    # ----------------------------------------------------------------
    # 2) Random restarts – a few different normalised seeds
    # ----------------------------------------------------------------
    random_seeds = 3
    for _ in range(random_seeds):
        a = rng.random(n, dtype=np.float64)
        a = a / a.sum()
        sc = _bound(a)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            if best_score <= TARGET:
                return [float(x) for x in best_vec]

    # ----------------------------------------------------------------
    # 3) Simple projected gradient descent on the max‑convolution
    #    (started from the current best vector)
    # ----------------------------------------------------------------
    max_iter = 200
    a = best_vec.copy()                     # start point for the descent
    for it in range(max_iter):
        # current bound and best‑so‑far update
        sc = _bound(a)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            if best_score <= TARGET:
                return [float(x) for x in best_vec]

        # convolution and index of the maximal entry
        b = _full_convolve(a)
        k_max = int(b.argmax())

        # sub‑gradient of the maximal quadratic form:
        # g_i = 2 * a_{k_max - i}  when the index is valid
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # decreasing step size
        lr = 0.5 / (it + 1.0)

        # projected gradient step
        a = a - lr * g
        a = np.maximum(a, 0.0)

        # re‑normalise to keep the scale fixed (sum = 1)
        s = a.sum()
        if s == 0.0:                         # degenerate case → reset to uniform
            a = np.ones(n, dtype=np.float64) / n
            s = 1.0
        a = a / s

    # ----------------------------------------------------------------
    # 4) Light stochastic polishing (window scaling + occasional mirroring)
    # ----------------------------------------------------------------
    for _ in range(num_steps):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.05)))   # typical 5 % tweak
        cand[start:stop] *= factor

        if rng.random() < 0.20:                       # occasional symmetrisation
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_vec = cand
            best_score = sc
            if best_score <= TARGET:
                break

    # ----------------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # ----------------------------------------------------------------
    return [float(x) for x in best_vec]
