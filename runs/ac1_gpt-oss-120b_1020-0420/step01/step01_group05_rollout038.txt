import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the AC1 bound using FFT.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using FFT (matrix‑free)."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    return c


# ----------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound.

    The routine builds a few analytic families (inverse‑square‑root,
    power‑law, exponential) and keeps the best one according to the fast
    score.  A short deterministic gradient‑like refinement (based on the
    derivative of the bound) is then applied to push the sum up while
    keeping the maximal convolution essentially unchanged.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Parent fallback (if height_sequence_1 is available at runtime).
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Determine length n and initialise a safe fallback vector.
    # ------------------------------------------------------------------
    if parent is not None:
        n = parent.size
        parent = np.clip(parent, 0.0, 1000.0)
    else:
        # use a modest length when no parent is present
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best_seq = parent.copy()
    best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # 1) Inverse‑square‑root family a_i = 1 / sqrt(i + offset)
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    offsets = np.linspace(0.1, 4.0, 30)
    for off in offsets:
        a = 1.0 / np.sqrt(i + off)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = a[::-1]
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 2) Power‑law family a_i = (i+1)^{-p}
    # ------------------------------------------------------------------
    powers = np.linspace(0.35, 0.65, 13)
    for p in powers:
        a = (i + 1) ** (-p)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = a[::-1]
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 3) Simple exponential decay a_i = exp(-alpha * i)
    # ------------------------------------------------------------------
    alphas = np.linspace(0.0005, 0.005, 10)
    for a0 in alphas:
        a = np.exp(-a0 * i)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s
        a_rev = a[::-1]
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # Gradient‑like refinement: increase entries that have a small
    # counterpart in the currently maximal convolution index.
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    max_iters = 200
    for it in range(max_iters):
        # convolution of seq with itself
        conv = _conv_self(seq)
        M = np.max(conv)
        k0 = int(np.argmax(conv))
        S = np.sum(seq)
        if S < 0.01:
            break
        # partner values a_{k0-i} (zero where out of range)
        idx = k0 - np.arange(n)
        partner = np.empty_like(seq)
        partner.fill(0.0)
        mask = (idx >= 0) & (idx < n)
        partner[mask] = seq[idx[mask]]

        # target increase: indices whose partner is below the
        # critical value 2M/S raise the bound most efficiently.
        thresh = 2.0 * M / S
        diff = np.maximum(0.0, thresh - partner)

        # step size slowly decays; early steps are larger.
        step_factor = 0.02 if it < 100 else 0.005
        delta = diff * step_factor
        cand = seq + delta
        cand = np.clip(cand, 0.0, 1000.0)

        s = _fast_score(cand)
        if s < best_score:
            seq = cand
            best_seq = cand
            best_score = s
        # otherwise keep the old sequence (no acceptance)

    # ------------------------------------------------------------------
    # Final random jitter around the best candidate (tiny multiplicative
    # perturbations).  This can sometimes escape very flat plateaus.
    # ------------------------------------------------------------------
    for _ in range(150):
        pert = np.exp(rng.normal(0.0, 0.01, size=n))
        cand = best_seq * pert
        cand = np.clip(cand, 0.0, 1000.0)
        s = _fast_score(cand)
        if s < best_score:
            best_seq = cand
            best_score = s

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
