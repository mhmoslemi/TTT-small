import numpy as np

def _conv_max_fft(vec: np.ndarray, nfft: int) -> float:
    """Linear convolution max using zero‑padded FFT."""
    # rfft of padded vector
    f = np.fft.rfft(vec, n=nfft)
    # convolution by element‑wise product (real ⇒ real)
    conv = np.fft.irfft(f * f, n=nfft)
    # first 2*len(vec)-1 entries contain the linear convolution
    return float(conv[: 2 * len(vec) - 1].max())

def _score_vec(vec: np.ndarray, nfft: int) -> float:
    """Exact objective value (scaling‑invariant)."""
    total = vec.sum()
    if total < 0.01 or np.any(vec < 0):
        return float('inf')
    max_c = _conv_max_fft(vec, nfft)
    return float(2.0 * len(vec) * max_c / (total * total))

def _normalize(vec: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and scale to unit ℓ¹ norm."""
    v = np.maximum(vec, 0.0)
    s = v.sum()
    if s == 0.0:
        # uniform fallback (should never happen for our generations)
        return np.full_like(v, 1.0 / v.size)
    return v / s

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that improves the
    AC1 upper bound.  The routine scans several compact parametric families,
    uses a few random perturbations and simple mixing with a possible parent
    sequence, keeping the best candidate found.
    """
    # ------------------------------------------------------------------
    # 1.  Parent (if supplied) – fall back to a uniform vector otherwise.
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():

        if (
            arr.ndim == 1

            parent = np.clip(arr, 0.0, 1000.0)

    if parent is None:
        # generic safe fallback – uniform of modest length
        parent = np.ones(256, dtype=np.float64) / 256.0

    n = parent.size

    nfft = 1 << ((2 * n - 1).bit_length())
