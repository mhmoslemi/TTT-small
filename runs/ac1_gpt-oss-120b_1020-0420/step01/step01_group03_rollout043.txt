def propose_candidate(seed=42, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    try:
        n = int(len(height_sequence_1))
        base = _root_constant_seq(n)
    except Exception:
        n = 256
        base = _root_constant_seq(n)

    def score_of(arr):
        conv = np.convolve(arr, arr)
        max_b = conv.max()
        s = arr.sum()
        if s < 0.01:
            return np.inf
        return 2 * n * max_b / (s * s)

    best = base.copy()
    best_score = score_of(best)

    start_time = time.time()
    max_time = 4.0   # seconds

    while time.time() - start_time < max_time:
        # random segment
        s = np.random.randint(0, n)
        # length distribution: exponential mean 0.1 n maybe
        max_len = n - s
        l = int(np.random.exponential(scale=n/10))
        if l < 1:
            l = 1
        if l > max_len:
            l = max_len
        # scale factor
        t = 0.5 + np.random.rand() * 1.0  # between 0.5 and 1.5
        cand = best.copy()
        cand[s:s+l] *= t
        # ensure bounds
        np.clip(cand, 0, 1000, out=cand)
        sc = score_of(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best.tolist()
