    """
    Evaluate a candidate vector using the provided evaluate_sequence.
    The vector is converted to a plain Python list as required.
    """
    return float(evaluate_sequence(vec.tolist()))

    """
    Produce a non‑negative step‑function (list of floats) whose AC1 upper bound
    is as low as possible under the given evaluator.

    Strategy:
      * Use the supplied `height_sequence_1` as a fallback parent if it is
        present and looks valid.
      * Initialise a uniform sequence (all entries = 1.0).  This is the
        theoretical minimiser of the bound under the non‑negativity
        constraints, because it minimises the ratio
        sum(a_i^2) / (sum(a_i))^2.
      * If the parent scores better, keep it.
      * Perform a short, cheap stochastic local search around the current best:
        multiplicative log‑normal perturbations (scale‑invariant) are tried and
        accepted only when they improve the score.
      * Return the best vector found as a plain Python list.
    """
    # ------------------------------------------------------------------
    # 1.  Retrieve a parent sequence if available
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            tmp.ndim == 1
            and tmp.size > 0
            and np.isfinite(tmp).all()
        ):
            parent = np.clip(tmp, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 2.  Determine the working length
    # ------------------------------------------------------------------
    n = parent.shape[0] if parent is not None else 256

    # ------------------------------------------------------------------
    # 3.  Uniform baseline (theoretical optimum under the constraints)
    # ------------------------------------------------------------------
    best = np.full(n, 1.0, dtype=np.float64)          # non‑negative, well‑within bounds
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 4.  Compare with the parent (if any)
    # ------------------------------------------------------------------
    if parent is not None:
        parent_score = _score(parent)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # 5.  Small stochastic refinement around the current best
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    # A modest number of trials keeps total time low (convolution is O(n^2)).
    for _ in range(300):
        # Log‑normal multiplicative noise (scale‑invariant)

    # ------------------------------------------------------------------
