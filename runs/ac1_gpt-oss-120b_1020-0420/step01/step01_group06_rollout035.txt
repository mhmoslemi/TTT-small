import numpy as np

def _array_score(arr: np.ndarray) -> float:
    """Helper that forwards a NumPy array to the provided evaluator."""
    return float(evaluate_sequence(arr.tolist()))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step function (list of floats) whose AC1 upper bound
    is as low as possible.  The routine starts from the supplied parent
    `height_sequence_1` (if present), explores several parametric families,
    draws random Dirichlet vectors, and then performs cheap local optimisation
    (pairwise mass transfers and occasional smoothing).  The best sequence
    discovered is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.   Load a parent sequence if the harness provides one.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
        else:
            parent = None
    else:
        parent = None

    # ------------------------------------------------------------------
    # 2.   Fallback: a uniform distribution of unit mass.
    # ------------------------------------------------------------------
    if parent is None:
        n = 256
        parent = np.ones(n, dtype=np.float64) / n            # sum == 1
    else:
        n = parent.size
        s = parent.sum()
        parent = np.ones(n, dtype=np.float64) / n if s == 0 else parent / s

    best = parent.copy()
    best_score = _array_score(best)

    indices = np.arange(1, n + 1, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3.   Power‑law family a_i ∝ (i+1)^{-α}
    # ------------------------------------------------------------------
    for alpha in np.linspace(0.1, 2.0, 80):
        w = np.power(indices, -alpha)
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        sc = _array_score(w)
        if sc < best_score:
            best, best_score = w, sc

    # ------------------------------------------------------------------
    # 4.   Exponential decay a_i ∝ exp(-β·i)
    # ------------------------------------------------------------------
    for beta in np.linspace(0.0, 0.02, 80)[1:]:          # skip β = 0 (uniform)
        w = np.exp(-beta * np.arange(n, dtype=np.float64))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        sc = _array_score(w)
        if sc < best_score:
            best, best_score = w, sc

    # ------------------------------------------------------------------
    # 5.   Combined power‑law × exponential (coarse grid)
    # ------------------------------------------------------------------
    for alpha in np.linspace(0.2, 1.5, 12):

            w_sum = w.sum()
            if w_sum == 0.0:
