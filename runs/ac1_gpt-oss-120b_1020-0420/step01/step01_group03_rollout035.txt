def compute_greedy(f, M):
    n = len(f)
    caps = np.full(n, M, dtype=np.float64)  # only for k=0..n-1
    g = np.zeros_like(f)
    for i in reversed(range(n)):
        # compute ub
        # Extract f_slice = f[0:n-i] (since k-i from 0..n-1-i)
        f_slice = f[:n - i]  # length n-i, corresponds to f[k-i] for k=i..n-1
        caps_slice = caps[i:]  # length n-i
        # compute ub candidates
        # Avoid division by zero via mask
        mask = f_slice > 0
        ub_candidates = np.full_like(caps_slice, np.inf)
        ub_candidates[mask] = caps_slice[mask] / f_slice[mask]
        ub = ub_candidates.min()
        if ub <= 0:
            g[i] = 0.0
            continue
        # Use a safety factor to avoid floating error
        ub = min(ub, M)  # limit by M maybe
        g[i] = ub
        # Update caps: caps[i:] -= f_slice * ub
        caps[i:] -= f_slice * ub
        # Ensure caps non-negative (could become small negative due to float)
        caps[i:] = np.maximum(caps[i:], 0.0)
    return g

def _root_constant_seq(length):
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i-1] * (2*i - 1) / (2*i)
    return np.clip(seq, 0.0, 1000.0)

def _greedy_g(f, M):
    n = f.shape[0]
    caps = np.full(n, M, dtype=np.float64)
    g = np.empty_like(f)

            ub = M
        g[i] = ub
        caps[i:] -= f_slice * ub

import numpy as np

# ------------------------------------------------------------
# Helper: root‑constant sequence (fallback if no parent present)
# ------------------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i = 0..length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


# ------------------------------------------------------------
# Fast max‑convolution using FFT (only the maximum is needed)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] using an FFT‑based convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)            # real inverse FFT
    # keep only the first 2n‑1 valid entries
    conv = conv[: 2 * n - 1]
    return float(conv.max())

    # ---- normalise so that the convolution upper bound is exactly 1 ----
    max_conv = _max_convolution(f)
    if max_conv > 0:
        f = f / np.sqrt(max_conv)

    # ---- linear tilt (additive) ------------------------------------------------
    # offset ranges linearly from –1 (first entry) to +1 (last entry)
    offset = (np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)

    best_vec = f
    best_sum = f.sum()
    best_alpha = 0.0
    max_alpha = float(f[0])               # α cannot exceed the first coefficient

    # coarse sweep
    for alpha in np.linspace(0.0, max_alpha, 15):
        cand = f + alpha * offset
        cand = np.maximum(cand, 0.0)
        max_c = _max_convolution(cand)
        if max_c <= 0:
            continue
        cand = cand / np.sqrt(max_c)
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > best_sum:
            best_sum, best_vec, best_alpha = s, cand, alpha

    # fine sweep centred on the best coarse α
    lo = max(0.0, best_alpha - max_alpha / 15)
    hi = min(max_alpha, best_alpha + max_alpha / 15)
    for alpha in np.linspace(lo, hi, 15):
        cand = f + alpha * offset
        cand = np.maximum(cand, 0.0)
        max_c = _max_convolution(cand)
        if max_c <= 0:
            continue
        cand = cand / np.sqrt(max_c)
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > best_sum:
            best_sum, best_vec = s, cand

    # ---- multiplicative tilt (scale) ----------------------------------------
    best_mul_vec = best_vec
    best_mul_sum = best_sum
    best_beta = 0.0
    max_beta = 0.9                         # keep factors positive (1‑β >= 0)

    for beta in np.linspace(0.0, max_beta, 15):
        factor = 1.0 + beta * offset       # <1 early, >1 later
        cand = f * factor
        max_c = _max_convolution(cand)
        if max_c <= 0:
            continue
        cand = cand / np.sqrt(max_c)
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > best_mul_sum:
            best_mul_sum, best_mul_vec, best_beta = s, cand, beta

    # fine refinement around the best multiplicative β
    lo = max(0.0, best_beta - max_beta / 15)
    hi = min(max_beta, best_beta + max_beta / 15)
    for beta in np.linspace(lo, hi, 15):
        factor = 1.0 + beta * offset
        cand = f * factor
        max_c = _max_convolution(cand)
        if max_c <= 0:
            continue
        cand = cand / np.sqrt(max_c)
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > best_mul_sum:
            best_mul_sum, best_mul_vec = s, cand

    # choose the better of additive / multiplicative searches
    final_vec = best_mul_vec if best_mul_sum > best_sum else best_vec

    # ---- final sanity check using the exact evaluator ------------------------
    try:
        score = float(evaluate_sequence(final_vec.tolist()))
        if not np.isfinite(score):
            raise ValueError
    except Exception:
        # fallback to the (already normalised) root‑constant sequence
        final_vec = f

    return final_vec.tolist()
