def compute_score(a):
    # a is np.ndarray float.
    # Ensure nonnegative & bounded?? but we keep.
    # Compute linear conv using FFT size L = 1<<ceil_pow2(2*n-1)
    L = int(1 << ((2*n - 1).bit_length()))
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[:2*n-1]
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (sum_a * sum_a)

def _pair_counts(selected, n):
    """Return ordered pair counts for a binary mask."""
    size = 2 * n - 1

        cnt[2 * i] += 1
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2
    return cnt
