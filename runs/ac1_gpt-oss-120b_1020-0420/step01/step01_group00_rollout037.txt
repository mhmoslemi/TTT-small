import numpy as np

# ----- fast convolution (real FFT, length = next power of two) -----
def _conv_self(v: np.ndarray, m: int) -> np.ndarray:
    """Return the (2*n-1)-length self‑convolution of v using FFT of length m."""
    fv = np.fft.rfft(v, n=m)
    conv_full = np.fft.irfft(fv * fv, n=m)
    return conv_full[: 2 * v.size - 1]

# ----- fast AC1 upper‑bound for a vector assumed to have sum>0 -----
def _fast_score(v: np.ndarray, n: int, m: int) -> float:
    """2*n*max(conv)/sum(v)^2 using the same FFT length m as _conv_self."""
    if v.size == 0:
        return np.inf
    s = v.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_self(v, m)
    return float(2.0 * n * conv.max() / (s * s))

# ----- main candidate proposal function -----
def propose_candidate(seed: int = 42, num_steps: int = 20000, **kwargs):

    cur = cur / cur.sum()                     # normalise to sum = 1

    n = cur.size
    # FFT length (next power of two) – same for all convolutions

    # initial convolution, score and best record
