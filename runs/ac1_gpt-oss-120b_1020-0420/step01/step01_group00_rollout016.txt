       # v is numpy array
       n = v.shape[0]
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u) - 1
       ind = np.arange(1, n+1)
       cond = u - cssv / ind > 0
       rho = ind[cond][-1]
       theta = cssv[cond][-1] / rho
       w = np.maximum(v - theta, 0)
       return w
   ```

        n = v.size
        if n == 0:
            return np.inf
        m = 1 << ((2*n - 1).bit_length())
        fa = np.fft.rfft(v, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
        max_c = conv.max()
        s = v.sum()
        if s < 1e-12:
            return np.inf
        return float(2.0 * n * max_c / (s * s))

import numpy as np

def _fast_score(v):
    n = v.size
    if n == 0:
        return np.inf
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(v, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    max_c = conv.max()
    s = v.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _proj_simplex(v):
    # project onto simplex {x>=0, sum x = 1}
    # reference: Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

# ----- fast AC1 upper bound (FFT based) -----
def _fast_score(v: np.ndarray) -> float:
    n = v.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(v, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = v.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----- projection onto the probability simplex {x>=0, sum x = 1} -----
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
