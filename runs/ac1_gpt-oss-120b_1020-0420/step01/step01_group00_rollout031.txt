import numpy as np

# ---------- small utilities ----------
def _next_pow2(x: int) -> int:
    """Return the next power of two >= x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _fast_score(vec: np.ndarray) -> float:
    """Fast AC1 bound (scale‑invariant) using FFT."""
    n = vec.size
    if n == 0:
        return np.inf
    N = _next_pow2(2 * n - 1)
    f = np.fft.rfft(vec, n=N)
    conv = np.fft.irfft(f * f, n=N)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
