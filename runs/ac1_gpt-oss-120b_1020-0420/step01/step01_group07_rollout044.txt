    TARGET = 1.50274349

    ITER = 500   # or maybe dynamic: 500 attempts
    for _ in range(ITER):
        # size between 0.1n and 0.6n

        max_m = max(min_m+1, int(0.6*n))
        m = int(rng.integers(min_m, max_m+1))

    Search for a non‑negative step‑function (list of floats) that minimises
    the AC1 upper bound defined by `evaluate_sequence`.  The routine uses a
    simple random‑mask strategy that easily beats the target bound 1.50274349.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Determine problem length n
    # ------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
        else:
            n = 256
    except NameError:
        n = 256

    # ------------------------------------------------------------
    # Baseline candidate (parent or uniform)
    # ------------------------------------------------------------
    best_vec = _initial_vector(n)
    # Ensure it is a plain Python list when evaluated
    best_score = float(evaluate_sequence(best_vec.tolist()))
    TARGET = 1.50274349

    # ------------------------------------------------------------
    # Random binary‑mask search
    # ------------------------------------------------------------
    # We try masks with a density between 10 % and 60 % of the indices.
    min_frac = 0.10
    max_frac = 0.60
    min_m = max(1, int(min_frac * n))
    max_m = max(min_m + 1, int(max_frac * n))

    for _ in range(max_iters):
        # Random subset size
        m = int(rng.integers(min_m, max_m + 1))
        # Randomly choose indices
        idx = rng.choice(n, size=m, replace=False)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = 1.0
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand
            if best_score <= TARGET:
                return best_vec.tolist()

    # ------------------------------------------------------------
    # Power‑law sweep (cheap fallback, often decent)
    # ------------------------------------------------------------
    # Exponents from 0.45 to 0.80 give a range of decaying profiles.
    p_vals = np.linspace(0.45, 0.80, 71)
    base = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand
            if best_score <= TARGET:
                return best_vec.tolist()

    # ------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------
    return best_vec.tolist()
