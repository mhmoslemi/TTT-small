import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound defined by `evaluate_sequence`.

    Strategy:
        * Use a linearly increasing sequence a_i = (i+1) * (1000 / n)
          which gives the closed‑form bound 4·(n+2)/(3·(n+1)) ≈ 4/3 for n≥7.
        * If the parent construction `height_sequence_1` exists, evaluate it
          and keep the better of the two.
        * Fallback to a uniform vector of ones if anything goes wrong.
    """
    # ------------------------------------------------------------
    # Determine problem dimension
    # ------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
        else:
            n = 256
    except NameError:
        n = 256

    # ------------------------------------------------------------
    # Primary candidate: linear increase, max capped at 1000
    # ------------------------------------------------------------
    scale = 1000.0 / n                     # ensures final entry equals 1000
    cand = scale * np.arange(1, n + 1, dtype=np.float64)
    cand = np.clip(cand, 0.0, 1000.0)

    # ------------------------------------------------------------
    # Score primary candidate
    # ------------------------------------------------------------
    try:
        best_score = float(evaluate_sequence(cand.tolist()))
    except Exception:                     # very unlikely, but safe
        best_score = float('inf')

    best_vec = cand

    # ------------------------------------------------------------
    # Optional: compare with provided parent construction
    # ------------------------------------------------------------
    try:
        parent_vec = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent_vec.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best_vec = parent_vec
            best_score = parent_score
    except Exception:
        pass  # ignore if parent is not usable

    # ------------------------------------------------------------
    # Fallback: uniform ones (guaranteed finite)
    # ------------------------------------------------------------
    if not np.isfinite(best_score):
        fallback = np.ones(n, dtype=np.float64)
        fallback = np.clip(fallback, 0.0, 1000.0)
        best_vec = fallback

    return best_vec.tolist()
