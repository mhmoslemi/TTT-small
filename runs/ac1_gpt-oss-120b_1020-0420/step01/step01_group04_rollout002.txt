def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals(): n = height_seq.shape[0] else: n = 256
    base = _coefficients_binom_half(n)
    base = np.clip(base, 0, 1000)
    # fallback parent
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    target = 1.50274349
    num_steps = kwargs.get('num_steps', 30000)
    # define block move function
    def block_scale(seq):
        cand = seq.copy()
        n = len(seq)
        # choose block with bias towards tail
        if rng.random() < 0.7:
            start = int(rng.integers(int(0.5*n), n))
        else:
            start = int(rng.integers(0, n))
        max_len = n - start
        if max_len <= 0:
            return cand
        min_len = max(1, max_len // 20)  # 5% min
        width = int(rng.integers(min_len, max_len+1))
        factor = float(np.exp(rng.normal(0.0, 0.12)))  # moderate scaling
        cand[start:start+width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    # small bump move
    def single_bump(seq):
        cand = seq.copy()
        idx = int(rng.integers(0, n))
        bump = max(0.0, rng.normal(0.0, 0.05))
        cand[idx] += bump
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    no_improve_counter = 0
    for step in range(num_steps):
        # choose move:
        op = rng.integers(0, 3)
        if op == 0:
            cand = block_scale(best)
        elif op == 1:
            cand = single_bump(best)
        else:
            # swap two entries
            cand = best.copy()
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            no_improve_counter = 0
        else:
            no_improve_counter += 1
        if best_score < target:
            break
        # occasional perturbation if stuck
        if no_improve_counter >= 2000:
            # random global scaling of tail
            cand = best.copy()
            start = int(rng.integers(int(0.5*n), n))
            factor = float(np.exp(rng.normal(0.0, 0.30)))
            cand[start:] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            # evaluate
            try:
                score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                score = np.inf
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
            no_improve_counter = 0

    # final sanity check: ensure valid and finite
    final_score = float(evaluate_sequence(best.tolist()))
    # As a fallback, if best_score is still inf (should not), return base
    if not np.isfinite(final_score):
        best = base.copy()
    return [float(x) for x in best]
