import numpy as np

# ---------- helper utilities (top‑level) ----------
def _fft_conv(a: np.ndarray, nfft: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length nfft)."""
    # rfft works on real data and returns N/2+1 complex numbers
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)
    return conv  # length nfft (zero‑padded tail)


def _fast_score(a: np.ndarray, n: int, nfft: int) -> float:
    """Fast version of the AC1 upper bound (scale‑invariant)."""
    total = a.sum()
    if total <= 0.0:
        return float('inf')
    conv = _fft_conv(a, nfft)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (total * total)


def _normalise(a: np.ndarray) -> np.ndarray:
    """Shift to non‑negative, clip to the allowed range and renormalise to sum 1."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= s
    return a


# ---------- main entry point ----------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function sequence (Python list) whose
    AC1 upper bound is as low as possible.  The algorithm combines
    a short parametric scan with a long random hill‑climbing loop,
    all using an O(n log n) FFT based convolution.
    """
    rng = np.random.default_rng(seed)

    # --- 1. obtain a parent / fallback sequence --------------------
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            parent = np.clip(tmp, 0.0, 1000.0)

    if parent is None:
        # generic safe fallback (length 256)
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    # normalise to unit sum (scale does not affect the bound)
    a = _normalise(parent.copy())

    # FFT size – next power of two >= 2n‑1
    nfft = 1 << ((2 * n - 1).bit_length())

    # --- 2. keep the best candidate seen so far -------------------
    best = a.copy()
    best_score = _fast_score(best, n, nfft)

    # --- 3. parametric families (power‑law & exponential) ----------
    indices = np.arange(1, n + 1, dtype=np.float64)

    # power‑law : a_i ∝ (i)^{-α}
    alphas = np.linspace(0.15, 1.00, 71)  # coarse grid
    for alpha in alphas:
        w = np.power(indices, -alpha)
        w = _normalise(w)
        sc = _fast_score(w, n, nfft)
        if sc < best_score:
            best, best_score = w, sc

    # fine‑tune around the best α (if any)
    # we reuse the last good α; if none was better than parent we pick middle
    last_alpha = alpha if 'alpha' in locals() else 0.5
    lo = max(0.01, last_alpha - 0.03)
    hi = min(1.20, last_alpha + 0.03)
    fine_alphas = np.linspace(lo, hi, 51)
    for alpha in fine_alphas:
        w = np.power(indices, -alpha)
        w = _normalise(w)
        sc = _fast_score(w, n, nfft)
        if sc < best_score:
            best, best_score = w, sc

    # exponential decay : a_i ∝ exp(-β·i)
    betas = np.linspace(1e-5, 8e-3, 60)
    for beta in betas:
        w = np.exp(-beta * np.arange(n, dtype=np.float64))
        w = _normalise(w)
        sc = _fast_score(w, n, nfft)
        if sc < best_score:
            best, best_score = w, sc

    # --- 4. hill‑climbing / random local search --------------------
    # configuration
    max_iters = min(8000, 3 * n)          # enough iterations, still fast
    block_max_width = max(2, n // 30)     # maximal block size for a mutation
    sigma_noise = 0.12                    # log‑normal std‑dev for block scaling
    swap_prob = 0.3                       # probability to perform a swap instead of scaling

    cur = best.copy()
    cur_score = best_score
    no_improve = 0

    for it in range(max_iters):
        # ----- choose mutation type
        if rng.random() < swap_prob:
            # --- swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # --- multiplicative log‑normal noise on a contiguous block
            start = rng.integers(0, n)
            width = rng.integers(1, block_max_width + 1)
            stop = min(n, start + width)
            cand = cur.copy()
            noise = np.exp(rng.normal(0.0, sigma_noise, size=stop - start))
            cand[start:stop] *= noise

        # renormalise (preserves scale‑invariance)
        cand = _normalise(cand)

        # evaluate quickly
        sc = _fast_score(cand, n, nfft)

        # accept if strictly better (hill‑climb)
        if sc < cur_score:
            cur, cur_score = cand, sc
            no_improve = 0
            if sc < best_score:
                best, best_score = cand, sc
        else:
            no_improve += 1

        # occasional aggressive shake‑up if stuck
        if no_improve >= 500:
            # re‑seed from the best with a fresh random block
            start = rng.integers(0, n)
            width = rng.integers(1, block_max_width + 1)
            stop = min(n, start + width)
            cur = best.copy()
            cur[start:stop] *= np.exp(rng.normal(0.0, 0.5, size=stop - start))
            cur = _normalise(cur)
            cur_score = _fast_score(cur, n, nfft)
            no_improve = 0

    # --- 5. final safety check (evaluate with authoritative function) ---
    # The harness will call evaluate_sequence on the returned list,
    # but we double‑check that the sequence is valid for ourselves.
    final = _normalise(best.copy())
    # guarantee Python list of floats
    return [float(v) for v in final]
