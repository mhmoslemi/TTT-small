    selected = []

import numpy as np

# ------------------------------------------------------------
# Helper: evaluate the AC1 bound locally (same formula as the harness)
# ------------------------------------------------------------
def _ac1_score(seq: np.ndarray) -> float:
    """Fast local computation of the AC1 bound."""
    if seq.size == 0:
        return np.inf
    # the harness clips values to [0,1000] – we keep the same range here
    np.clip(seq, 0.0, 1000.0, out=seq)
    b = np.convolve(seq, seq)
    max_b = b.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * seq.size * max_b / (sum_a * sum_a)

# the number of unordered pairs that share the same index ≤ max_pairs.
# This guarantees that the largest convolution coefficient stays
# close to 1, while the total sum can exceed the half‑power series.
# ------------------------------------------------------------
def _build_candidate(n: int, rng: np.random.Generator,
                     v: float = 0.45, max_pairs: int = 2) -> np.ndarray:
    """
    Return a length‑n non‑negative step function.
    All non‑zero entries have height ``v``.
    The set of indices is built greedily so that the number of
    unordered pairs (i,j) with i<j and i+j = k never exceeds ``max_pairs``.
    """
    # pair‑count array for all possible convolution indices 0 … 2n‑2
    pair_counts = np.zeros(2 * n, dtype=np.int16)

    selected = []                     # indices that have been kept
    # random order (skip index 0 – we keep it zero to avoid a0^2)
    candidates = rng.permutation(np.arange(1, n, dtype=np.int32))

    for i in candidates:
        # check pair‑count constraints against already selected indices
        if selected:
            sums = i + np.array(selected, dtype=np.int32)
            if np.any(pair_counts[sums] >= max_pairs):
                continue
        # accept i
        selected.append(int(i))
        # update pair counts for the new unordered pairs (i, j) with j already in selected
        if len(selected) > 1:
            prior = np.array(selected[:-1], dtype=np.int32)   # all except the newly added i
            pair_counts[i + prior] += 1

    # build the array: height v at the selected positions
    a = np.zeros(n, dtype=np.float64)
    a[selected] = v
    # a tiny value at position 0 gives a non‑zero sum without hurting the bound
    a[0] = 0.01
    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 200, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.  The routine first tries a
    fast combinatorial construction (many 0.45‑height spikes with
    at most two unordered pairs per sum).  If that does not beat
    the target it falls back to a simple stochastic refinement
    starting from the parent sequence (if available) or from the
    half‑power series.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem length and initial sequence (parent if present)
    # ------------------------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        n = int(height_sequence_1.size)
    else:
        n = 256

    # ------------------------------------------------------------------
    # 1) Attempt the combinatorial “pair‑limited” construction
    # ------------------------------------------------------------------
    TARGET_SCORE = 1.50274349
    best = None
    best_score = np.inf

    # try a few random attempts with slightly different parameters
    for attempt in range(8):
        # height of the spikes – 0.45 works well, but we also probe 0.4
        v = 0.45 if attempt % 2 == 0 else 0.4
        cand = _build_candidate(n, rng, v=v, max_pairs=2)
        score = _ac1_score(cand.copy())
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
        if best_score < TARGET_SCORE:
            break

    # ------------------------------------------------------------------
    # 2) If the construction is not good enough, fall back to a
    #    stochastic refinement of the parent / half‑power series.
    # ------------------------------------------------------------------
    if best_score >= TARGET_SCORE:
        # base sequence: parent if available, else the classic half‑power series
        def _half_power(length: int) -> np.ndarray:
            a = np.empty(length, dtype=np.float64)
            a[0] = 1.0
            for i in range(1, length):
                a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
            return a

        if (
            "height_sequence_1" in globals()
            and isinstance(height_sequence_1, np.ndarray)
            and height_sequence_1.ndim == 1
            and height_sequence_1.size > 0
        ):
            base = height_sequence_1.astype(np.float64).copy()
        else:
            base = _half_power(n)

        np.clip(base, 0.0, 1000.0, out=base)
        best = base
        best_score = _ac1_score(best.copy())

        # simple stochastic hill‑climbing: try to add mass on the right side
        max_val = 0.99          # we never let a coefficient exceed this
        for _ in range(num_steps * 10):   # more iterations than the default
            # propose an increase at a random position in the right half
            j = rng.integers(n // 2, n)
            if best[j] >= max_val:
                continue
            # amount to add – up to the remaining headroom, scaled down a bit
            delta = (max_val - best[j]) * rng.random() * 0.5
            if delta <= 1e-12:
                continue
            cand = best.copy()
            cand[j] += delta
            # optional tiny reduction of an early entry to keep the bound in check
            i = rng.integers(0, n // 10)
            red = min(cand[i] * 0.1, delta * 0.5)   # remove at most 10 % of the early value
            cand[i] -= red
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = _ac1_score(cand.copy())
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                if best_score < TARGET_SCORE:
                    break

    # ------------------------------------------------------------------
    # Final safeguard – ensure the returned list is valid for the harness
    # ------------------------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    # the evaluator also clips, but we keep the same range explicitly
    final = final.tolist()
    return final
