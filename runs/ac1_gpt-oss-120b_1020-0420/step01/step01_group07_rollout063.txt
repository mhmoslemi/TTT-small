def propose_candidate(seed=42, **kwargs):
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
    except NameError:
        n = 256
    # baseline
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))
    # build sidon set
    max_heavy = 70  # limit
    selected = []
    used = set()
    for i in range(n):
        conflict = False
        # check self sum
        if i + i in used:
            conflict = True
        else:
            for j in selected:
                if i + j in used:
                    conflict = True
                    break
        if not conflict:
            # add i to set
            for j in selected:
                used.add(i + j)  # unordered sum
                # not needed to add j+i separately
            used.add(i + i)
            selected.append(i)
            if len(selected) >= max_heavy:
                break
    heavy_mask = np.zeros(n, dtype=bool)
    heavy_mask[selected] = True
    heavy_weight = 1.0
    # epsilon sweep
    eps_vals = np.concatenate((
        np.linspace(0.005, 0.05, 10),
        np.linspace(0.05, 0.2, 10)
    ))
    for eps in eps_vals:
        cand = np.full(n, eps, dtype=np.float64)
        cand[heavy_mask] = heavy_weight
        # clip just in case
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand.copy()
            if best_score <= 1.50274349:
                # early exit
                return [float(x) for x in best_vec]
    # optional local random perturbation: random weight adjustments of heavies
    rng = np.random.default_rng(seed)
    for _ in range(200):
        # pick random heavy index to toggle (remove) or change weight
        if rng.random() < 0.5 and selected:
            # turn off a heavy index (set to eps)
            idx = int(rng.choice(selected))

            # random scaling of whole vector

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand.copy()
            if best_score <= 1.50274349:

import numpy as np

# ------------------------------------------------------------
# Build a (large) Sidon set: all unordered sums i+j are distinct.
# ------------------------------------------------------------
def _build_sidon_set(n, max_size=70):
    selected = []
    used_sums = set()
    for i in range(n):
        # check whether adding i would create a duplicate unordered sum
        conflict = (i + i) in used_sums
        if not conflict:
            for j in selected:
                if (i + j) in used_sums:
                    conflict = True
                    break
        if conflict:
            continue
        # accept i
        for j in selected:
            used_sums.add(i + j)       # unordered sum, same as j+i
        used_sums.add(i + i)
        selected.append(i)
        if len(selected) >= max_size:
            break
    return selected
