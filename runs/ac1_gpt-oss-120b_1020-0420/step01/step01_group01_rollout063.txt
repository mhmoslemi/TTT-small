import numpy as np

# ------------------------------------------------------------
# Helper: exact same scoring as the provided evaluator
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Compute 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution (FFT for large n)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 5000, **kwargs):
    """
    Return a list of non‑negative heights with a low AC1 bound.
    The algorithm starts from the Catalan‑like sequence and performs
    targeted mass‑transfer moves that reduce the maximal convolution
    value while preserving (or modestly increasing) the total sum.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size – use the provided parent if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # ------------------------------------------------------------------
    # Initialise with the Catalan‑like sequence (exactly satisfies
    # a*a = 1 for the first n convolution entries).
    # ------------------------------------------------------------------
    best = _catalan_seq(n)
    best = np.clip(best, 0.0, 1000.0)
    best_score = _score(best)

    # ------------------------------------------------------------------
    # Deterministic re‑allocation (same idea as the previous program):
    # try a few prefix‑shrink / tail‑inflate transformations.
    # ------------------------------------------------------------------
    prefix_ratios = [0.05, 0.10, 0.20]
    shrink_factors = np.linspace(0.5, 1.0, 11)

    for r in prefix_ratios:
        k = max(1, int(n * r))
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()
            g = 1.0 + delta / tail_sum
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                early = best[:k]
                tail = best[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()

    # ------------------------------------------------------------------
    # Targeted refinement:
    # repeatedly move a fraction of the mass from the index that
    # contributes most to the current maximal convolution entry to an
    # index that contributes little.  Accept only improvements.
    # ------------------------------------------------------------------
    max_iter = kwargs.get("num_steps", num_steps)
    no_improve = 0
    for _ in range(max_iter):
        conv = np.convolve(best, best)
        M = conv.max()
        k_max = int(conv.argmax())

                dst = int(rng.choice(others)) if others.size else (src + 1) % n

        # amount to move – a modest fraction of the source entry
        frac = rng.uniform(0.01, 0.15)
        delta = best[src] * frac
        if delta <= 0.0:

        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
