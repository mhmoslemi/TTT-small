import numpy as np

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0 * (i+1)*(i+1))
    return a

def _score_arr(arr, n):
    # arr: numpy 1d
    # compute score as in evaluator, but avoid clipping
    s = arr.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(arr, arr)
    m = conv.max()
    return 2.0 * n * m / (s * s)

import numpy as np

# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Fast bound computation (same formula as the evaluator)
# ------------------------------------------------------------
def _fast_score(seq: np.ndarray, n: int) -> float:
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    m = conv.max()
    return float(2.0 * n * m / (s * s))

    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # --------------------------------------------------------
    # Keep the best sequence seen so far.
    # --------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # ---- parent ------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.clip(parent, 0.0, 1000.0)
        sc = _fast_score(cand, n)
        best_seq = cand
        best_score = float(sc)

    # ---- Catalan base -----------------------------------------
    base = _catalan_seq(n)
    base = np.clip(base, 0.0, 1000.0)
    sc = _fast_score(base, n)
    if sc < best_score:
        best_seq = base
        best_score = float(sc)

    # --------------------------------------------------------
    # Prepare data for a cheap uniform‑offset search.
    # The candidate has the form   a_i = base_i + eps .
    # --------------------------------------------------------
    ones = np.ones(n, dtype=np.float64)
    convb = np.convolve(base, base)          # base * base
    convb1 = np.convolve(base, ones)         # base * 1  (same as 1 * base)
    conv11 = np.convolve(ones, ones)         # 1 * 1   (triangular)
    base_sum = base.sum()

    # Search eps >= 0.  The optimum lies near a few‑percent offset.
    eps_vals = np.concatenate([
        np.linspace(0.0, 0.05, 51),          # 0.000 … 0.050 step 0.001
        np.linspace(0.05, 0.20, 31)[1:]      # 0.050 … 0.200 step 0.005
    ])
    eps_vals = np.unique(eps_vals)

    for eps in eps_vals:
        # C(eps) = max_k (convb + 2*eps*convb1 + eps^2*conv11)_k
        C = np.max(convb + 2.0 * eps * convb1 + (eps * eps) * conv11)
        S = base_sum + n * eps
        if S < 0.01:
            continue
        sc = 2.0 * n * C / (S * S)
        if sc < best_score:
            best_score = float(sc)
            best_seq = base + eps

    # --------------------------------------------------------
    # Small stochastic refinement around the current best.
    # --------------------------------------------------------
    steps = kwargs.get("num_steps", 500)
    seq = best_seq.copy()
    seq_score = best_score

    for _ in range(steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # move a modest fraction of mass from src to dst
        delta = seq[src] * rng.uniform(0.01, 0.05)
        if delta <= 0.0:
            continue
        cand = seq.copy()
        cand[src] -= delta
        cand[dst] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _fast_score(cand, n)
        if sc < seq_score:
            seq = cand
            seq_score = sc

    # --------------------------------------------------------
    # Return the better of the refined sequence and the previous best.
    # --------------------------------------------------------
    final_seq = seq if seq_score < best_score else best_seq
    return [float(v) for v in final_seq]
