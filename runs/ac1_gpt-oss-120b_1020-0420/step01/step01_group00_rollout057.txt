def compute_gradient(a, lags):
    n = len(a)
    grad = np.zeros_like(a)
    for d in lags:
        if d >= 0:
            # grad[i] += a[i-d] for i in [d, n)
            grad[d:] += a[:n-d]
            # grad[i] += a[i+d] for i in [0, n-d)
            grad[:n-d] += a[d:]
        else:
            dd = -d
            # lag negative means conv at -d (a[i+d] a[i]) same as positive due symmetry
            # Use same as d>=0 but swapped.
            # For negative d, contributions same as positive d
            grad[:n-dd] += a[dd:]  # i < n-dd: a[i+dd]
            grad[dd:] += a[:n-dd]  # i>=dd: a[i-dd]
    return grad

import numpy as np

# ---------- fast AC1 upper bound (uses FFT) ----------
def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ---------- gradient of selected convolution lags ----------
def _grad_from_lags(a: np.ndarray, lag_idxs):
    """Gradient of sum of selected convolution entries w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in lag_idxs:                     # k is convolution index (0 .. 2n‑2)
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        # Loop over all valid i, j = k‑i
        for i in range(i0, i1 + 1):
            j = k - i
            grad[i] += 2.0 * a[j]          # derivative of a_i*a_j + a_j*a_i
    return grad

    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        fallback = np.full_like(best, 1.0)
        fallback /= fallback.sum()
        best = fallback

    return [float(v) for v in best]
