def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2 : C_i = comb(2i,i)/4^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function heights sequence that (hopefully)
    improves on the AC1 upper bound.  The search starts from the provided
    parent `height_sequence_1` when present, otherwise from the analytic
    (1‑z)^‑1/2 coefficients.  A lightweight hill‑climbing loop performs
    targeted “tail bumps” and occasional mass transfers from the front to
    the tail, maintaining the convolution maximum close to its initial
    value while raising the total sum.
    """
    rng = np.random.default_rng(seed)

    # --- initialise from parent or analytic baseline -----------------
    a = None
    if "height_sequence_1" in globals():
        try:
            a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        except Exception:
            a = None
    if a is None or a.size == 0:
        n = 256
        a = _coefficients_binom_half(n)
    else:
        n = int(a.shape[0])

    # ensure a has at least minimal total mass
    if a.sum() < 0.01:
        a[0] = max(a[0], 0.01)

    # --- pre‑compute convolution, sum and score ----------------------
    conv = np.convolve(a, a)                 # full convolution, length 2n‑1
    S = float(a.sum())
    M = float(conv.max())
    best = a.copy()
    best_score = float(2 * n * M / (S * S))

    # --- search hyper‑parameters ------------------------------------

    max_iters = 20000

    min_step = 0.01          # minimum fraction at the end of the run
    tail_start = n // 2      # indices >= this are considered “tail”
    front_end = n // 3       # indices < this are “front”

    # --- main optimisation loop --------------------------------------
    for it in range(max_iters):
        if best_score <= target:
            break

        # slowly shrink the allowed step size
        step_factor = max_step - (max_step - min_step) * (it / max_iters)

        # decide which mutation to try

        if r < 0.60:                     # tail bump (most frequent)
