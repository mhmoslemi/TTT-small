def fast_conv_max(a):
    n = a.shape[0]
    m = n*2 - 1
    size = 1 << (m-1).bit_length()   # next power of 2
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    # conv length = size
    # take first m elements
    max_val = np.max(conv[:m])
    return max_val
```

All values are real. Since a entries are non-negative, conv values should be >=0.

Now compute score:

```
def fast_score(a):
    m = fast_conv_max(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2 * len(a) * m / (s * s)
```

    if 'height_sequence_1' in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)

    try:
        parent_score = fast_score(base_seq)
        if parent_score < best_score:
            best = base_seq.copy()
            best_score = parent_score
    except Exception:
        pass
    # main loop
    max_steps = kwargs.get('num_steps', 3000)
    target = 1.50274349

    late_start = n - max(1, n // 8)
    # temperature schedule for simulated annealing acceptance
    T0 = 0.1
    for step in range(max_steps):
        if best_score <= target:
            break
        # define temperature decreasing
        T = T0 * (1 - step / max_steps)
        # pick move
        move_type = rng.integers(0, 5)
        cand = best.copy()
        if move_type == 0:

            i = rng.integers(0, early_end)
            j = rng.integers(late_start, n)

                cand[i] -= delta
                cand[j] += delta

            # scale a late block up modestly
            start = rng.integers(late_start, n)
            max_len = n - start
            if max_len <= 0:

            width = rng.integers(1, max_len + 1)
            factor = np.exp(rng.normal(0.0, 0.12))  # mean 1, small std dev

def fast_score(a):
    n = a.shape[0]
    # compute convolution using FFT
    m = 2*n - 1
    size = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    max_val = np.max(conv[:m])
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_val / (s * s)
```

# ----------------------------------------------------------------------
# Fast evaluation of the AC‑1 bound using FFT (linear rather than O(n²))
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the score 2·n·max(conv(seq,seq)) / (sum(seq)²) using FFT."""
    n = seq.shape[0]
    if n == 0:
        return float("inf")
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)
    max_val = np.max(conv[:m])
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Analytic coefficients of (1‑z)^‑1/2  :  c_i = (2i‑1)!!/(2i)!!
# ----------------------------------------------------------------------
def _analytic_coeffs(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a

def propose_candidate(seed=42, num_steps=5000, **kwargs):
    """Return a non‑negative step‑function that aims to minimise the AC‑1 bound."""
    rng = np.random.default_rng(seed)

    # ----- initialise ---------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = int(base.shape[0])
    else:
        n = 256
        base = _analytic_coeffs(n).copy()
        base = np.clip(base, 0.0, 1000.0)

    # analytic start point (known max‑convolution = 1)
    cur = _analytic_coeffs(n)
    cur = np.clip(cur, 0.0, 1000.0)

    best = cur.copy()
    best_score = _fast_score(best)

    # consider the supplied parent as a possible fallback
    try:
        parent_score = _fast_score(base)
        if parent_score < best_score:
            best = base.copy()
            best_score = parent_score
    except Exception:
        pass

    # ----- search parameters --------------------------------------------
    target = 1.50274349
    max_steps = int(num_steps)

    early_end = max(1, n // 20)                # first ~5 % (large values)
    tail_start = max(early_end + 1, (n * 3) // 4)   # last ~25 % (tiny values)

    # ----- main optimisation loop ---------------------------------------
    for step in range(max_steps):
        if best_score <= target:
            break

        cand = best.copy()
        move = rng.integers(0, 6)

        if move == 0:
            # mass transfer early → tail (reduces the peak)
            i = rng.integers(0, early_end)
            j = rng.integers(tail_start, n)
            if cand[i] > 0.0:
                frac = rng.uniform(0.02, 0.20)   # move 2‑20 % of the early mass
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta

        elif move == 1:
            # shrink a random early block (helps lower the peak)
            start = rng.integers(0, early_end)
            width = rng.integers(1, early_end - start + 1)
            factor = np.exp(rng.normal(-0.10, 0.05))   # < 1
            cand[start:start + width] *= factor

        elif move == 2:
            # add a tiny constant to a tail block (raises the sum gently)
            start = rng.integers(tail_start, n)
            max_len = n - start
            width = rng.integers(1, max_len + 1)
            eps = rng.uniform(0.001, 0.008)
            cand[start:start + width] += eps

        elif move == 3:
            # modestly enlarge a tail block (increases sum, limited M‑rise)
            start = rng.integers(tail_start, n)
            max_len = n - start
            width = rng.integers(1, max_len + 1)
            factor = 1.0 + rng.uniform(0.01, 0.05)    # 1‑5 % rise
            cand[start:start + width] *= factor

        elif move == 4:
            # swap a random early entry with a tail entry
            i = rng.integers(0, early_end)
            j = rng.integers(tail_start, n)
            cand[i], cand[j] = cand[j], cand[i]

        else:
            # tiny positive bump at a random position
            idx = rng.integers(0, n)
            bump = max(0.0, rng.normal(0.0, 0.01))
            cand[idx] += bump

        np.clip(cand, 0.0, 1000.0, out=cand)
        score = _fast_score(cand)

        # Accept only improvements (the landscape is monotone enough for our purpose)
        if score < best_score:
            best = cand
            best_score = score

    # final validation with the official evaluator (ensures compatibility)
    try:
        final_score = float(evaluate_sequence(best.tolist()))
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
    except Exception:
        pass

    return [float(v) for v in best]
