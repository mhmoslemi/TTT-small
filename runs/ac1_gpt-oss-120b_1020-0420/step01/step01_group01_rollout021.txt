import numpy as np

# ------------------------------------------------------------
# Fast (FFT) convolution of a real sequence with itself
# ------------------------------------------------------------
def _conv(seq: np.ndarray) -> np.ndarray:
    """Return full self‑convolution using a power‑of‑two FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_a = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


# ------------------------------------------------------------
# Score that matches the provided evaluator (lower is better)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = _conv(seq)
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        # a_{i+1} = a_i * (2i+2)(2i+1) / (4 (i+1)^2)
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative heights whose AC1 bound is as small as possible.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size – use the provided parent if available.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        ):
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # --------------------------------------------------------
    # Initialise with the Catalan‑like sequence (the classic baseline)
    # --------------------------------------------------------
    seq = _catalan_seq(n).copy()
    best_seq = seq.copy()
    best_score = _score(best_seq)

    # --------------------------------------------------------
    # If a parent is present and better, start from it.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        if parent.shape == (n,) and np.isfinite(parent).all():
            parent_score = _score(parent)
            if parent_score < best_score:
                seq = parent.copy()
                best_seq = seq.copy()
                best_score = parent_score

    # --------------------------------------------------------
    # Deterministic “peak‑flattening” phase.
    # Transfer a fraction of the mass that contributes most to the current
    # maximal autocorrelation towards a location that currently carries the
    # smallest value.  The total sum stays unchanged, while the peak tends to
    # shrink.
    # --------------------------------------------------------
    target = 1.50274349
    det_iters = kwargs.get("num_steps", 2000) // 2
    idx = np.arange(n, dtype=np.int64)

    for _ in range(det_iters):
        conv = _conv(seq)
        cur_score = 2.0 * n * conv.max() / (seq.sum() ** 2)
        if cur_score < target:
            best_seq = seq.copy()
            best_score = cur_score
            break

        k_max = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=seq.dtype)
        contrib[valid] = seq[idx[valid]] * seq[j[valid]]

        src = int(np.argmax(contrib))
        if seq[src] <= 0.0:
            break                     # nothing left to move

        dst = int(np.argmin(seq))
        if src == dst:
            dst = (src + n // 2) % n   # pick a far away place

        # move a modest fraction of the source mass
        frac = 0.05                     # 5 % per step
        delta = frac * seq[src]
        seq[src] -= delta
        seq[dst] += delta

        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            if best_score < target:
                break

    # --------------------------------------------------------
    # Stochastic fine‑tuning phase (small random reallocations).
    # --------------------------------------------------------
    rand_iters = kwargs.get("num_steps", 2000) - det_iters
    for _ in range(rand_iters):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst or seq[src] <= 0.0:
            continue
        delta = seq[src] * rng.uniform(0.01, 0.20)   # up to 20 % move
        seq[src] -= delta
        seq[dst] += delta
        seq = np.clip(seq, 0.0, 1000.0)

        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            if best_score < target:
                break

    # --------------------------------------------------------
    # Return the best candidate found (as a plain Python list).
    # --------------------------------------------------------
    return [float(v) for v in best_seq]
