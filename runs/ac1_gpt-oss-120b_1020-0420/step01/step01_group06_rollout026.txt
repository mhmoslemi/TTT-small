def compute_ac1(seq):
    arr = np.asarray(seq, dtype=np.float64)
    S = arr.sum()
    if S < 0.01:
        return float('inf')
    n = arr.size
    # Use FFT to compute convolution
    # Next power of two length
    m = 1 << ((2*n - 1).bit_length())
    # Compute FFT
    fft_a = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]  # Use real FFT
    max_b = conv.max()
    return 2 * n * max_b / (S * S)

import numpy as np

# ----------------------------------------------------------------------
# Helper: convolution by FFT (fast, linear in n log n)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.size
    # next power of two for efficient FFT
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv_full = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv_full.real


# ----------------------------------------------------------------------
# Helper: AC‑1 upper bound (same formula as the evaluator but fast)
# ----------------------------------------------------------------------
def _ac1_score(a: np.ndarray) -> float:
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_fft(a)
    return float(2.0 * a.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
#          { x >= 0, sum x = 1 }
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    # Duchi et al. (2008) O(n log n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
