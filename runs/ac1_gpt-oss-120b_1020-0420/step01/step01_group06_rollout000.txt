def compute_score(vec, n):

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------
def _self_convolution(v, fft_len):
    """
    Compute the full convolution of v with itself using a real FFT of length fft_len.
    Returns an array of length 2*len(v)-1.
    """
    ft = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(ft * ft, n=fft_len)[: 2 * v.size - 1]
    return conv


def _score_from_conv(conv, total, n):
    """
    Given a convolution array, total sum of the original sequence,
    and its length, return the AC1 upper‑bound score.
    """
    if total < 0.01:
        return float("inf")
    return 2.0 * n * conv.max() / (total * total)


def compute_score(v):
    """
    Compute the exact score for a candidate vector v.
    """
    v = np.clip(v, 0.0, None)
    total = v.sum()
    n = v.size
    fft_len = 1 << ((2 * n - 2).bit_length())
    conv = _self_convolution(v, fft_len)
    return _score_from_conv(conv, total, n)


def gradient_max_conv(v, max_idx):
    """
    Sub‑gradient of max(conv(v,v)) w.r.t. v, assuming the maximal index is max_idx.
    For the convolution of a vector with itself the gradient simplifies to
    2 * v[ max_idx - i ] for each component i that lies inside the support.
    """
    n = v.size
    shifts = max_idx - np.arange(n)
    mask = (shifts >= 0) & (shifts < n)
    grad = np.zeros_like(v)
    grad[mask] = 2.0 * v[shifts[mask]]
    return grad
