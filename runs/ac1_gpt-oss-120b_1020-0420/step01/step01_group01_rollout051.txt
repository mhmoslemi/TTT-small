import numpy as np

# ----------------------------------------------------------------------
# Score helper – identical to the harness evaluator (except it works on
# NumPy arrays directly and avoids repeated list conversion).
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)            # full convolution, O(n log n) via FFT
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ----------------------------------------------------------------------
# Catalan‑like (central binomial) sequence.
# a_i = C(2i,i) / 4^i
# ----------------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 1000, grad_steps: int = 500, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – use the provided parent if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all():
            n = parent_arr.size
            parent_seq = parent_arr.copy()
        else:
            n = 256
            parent_seq = None
    else:
        n = 256
        parent_seq = None

    # --------------------------------------------------------------
    # Initial candidates: Catalan and (if present) the parent.
    # --------------------------------------------------------------
    cat_seq = _catalan_seq(n)
    best = cat_seq
    best_score = _score(best)

    if parent_seq is not None:
        sc = _score(parent_seq)
        if sc < best_score:
            best = parent_seq
            best_score = sc

    # Keep the total sum (scale‑invariant) for later normalisation.
    sum_target = float(best.sum())

    # --------------------------------------------------------------
    # Helper to test a new candidate and keep the best one.
    # --------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray):
        nonlocal best, best_score
        cand = np.clip(candidate, 0.0, 1000.0)
        s = cand.sum()
        if s > 0:
            cand = cand * (sum_target / s)          # re‑scale to the original sum
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            return True
        return False

    # --------------------------------------------------------------
    # 1️⃣  Deterministic re‑weightings (linear, exponential, prefix‑shrink).
    # --------------------------------------------------------------
    bases = [cat_seq]
    if parent_seq is not None:
        bases.append(parent_seq)

    idx = np.arange(n, dtype=np.float64)

    for base in bases:
        # ---- linear tilt ----
        for alpha in np.linspace(0.0, 5.0, 11):          # 0 … 5
            w = 1.0 + alpha * idx / (n - 1)
            _maybe_update(base * w)

        # ---- exponential tilt ----
        for beta in np.linspace(0.0, 2.0, 11):           # 0 … 2
            w = np.exp(beta * idx / (n - 1))
            _maybe_update(base * w)

        # ---- early‑prefix shrink + uniform tail inflate ----
        prefix_ratios = np.linspace(0.05, 0.40, 8)       # 5 % … 40 %
        shrink_factors = np.linspace(0.20, 1.00, 9)      # 20 % … 100 %
        for r in prefix_ratios:
            k = max(1, int(n * r))
            early = base[:k]
            tail = base[k:]
            early_sum = early.sum()
            tail_sum = tail.sum()
            if tail_sum == 0.0:
                continue
            for f in shrink_factors:
                new_early = early * f
                delta = early_sum - new_early.sum()
                g = 1.0 + delta / tail_sum
                new_tail = tail * g
                cand = np.concatenate([new_early, new_tail])
                _maybe_update(cand)

    # --------------------------------------------------------------
    # 2️⃣  Random local mass transfers (hill‑climbing).
    # --------------------------------------------------------------
    random_steps = kwargs.get("num_steps", num_steps)
    for _ in range(random_steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # move a random fraction of the source entry
        frac = rng.uniform(0.001, 0.30)
        delta = best[src] * frac
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        # re‑scale in case clipping altered the total mass
        _maybe_update(cand)

    # --------------------------------------------------------------
    # 3️⃣  Gradient‑descent style refinement on the presently worst
    #     convolution entry.
    # --------------------------------------------------------------
    step = 0.01                     # initial absolute step size
    max_grad_steps = kwargs.get("grad_steps", grad_steps)
    for _ in range(max_grad_steps):
        conv = np.convolve(best, best)
        k = int(conv.argmax())                     # index of the current maximum
        # Build the gradient of the max‑conv term: grad_i = 2 * best_{k-i}
        rev = np.zeros_like(best)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_end >= i_start:
            idx_range = np.arange(i_start, i_end + 1)
            rev[idx_range] = best[k - idx_range]
        grad = 2.0 * rev
        cand = best - step * grad
        _maybe_update(cand)

        # adapt step size
        if _score(cand) < best_score:
            step *= 1.20
        else:
            step *= 0.70
        if step < 1e-12:
            break

    # --------------------------------------------------------------
    # Final sanity – clip once more and restore the target sum.
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    s = best.sum()
    if s > 0:
        best = best * (sum_target / s)

    # Return a plain Python list as required by the harness.
    return [float(v) for v in best]
