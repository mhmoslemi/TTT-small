import numpy as np

def _fast_score(vec: np.ndarray) -> float:
    """Fast AC1 upper bound for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def propose_candidate(seed: int = 42, num_steps: int = 12000,
                     init_eta: float = 0.12, init_T: float = 0.15,
                     beta: float = 0.9, eps: float = 1e-3,
                     top_k: int = 5, block_interval: int = 400,
                     block_prob: float = 0.2) -> list:
    """
    Search for a non‑negative step‑function minimizing the AC1 bound.
    Returns a list of coefficients.
    """
    rng = np.random.default_rng(seed)

    # ----- initialise from provided parent or fallback uniform -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    cur = np.clip(parent, 0.0, 1000.0)
    if cur.sum() < 1e-12:
        cur[:] = 1.0
    cur = cur / cur.sum()                     # normalise to sum = 1

    # small random perturbation to escape degenerate stationary points
    cur = cur * np.exp(rng.normal(0.0, 0.03, size=cur.shape))
    cur = np.clip(cur, 0.0, 1000.0)
    cur = cur / cur.sum()

    n = cur.size
    best = cur.copy()
    best_score = _fast_score(best)
    cur_score = best_score
    v = np.zeros_like(cur)                    # momentum term
    m = 1 << ((2 * n - 1).bit_length())       # FFT length (next power of two)

    idx_range = np.arange(n)                  # reusable index array

    for step in range(num_steps):
        # ----- convolution and peak detection -----
        fa = np.fft.rfft(cur, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
        max_c = conv.max()

        # pick a peak index among the top_k values (adds stochasticity)
        if top_k >= conv.size:
            peak_idx = rng.integers(0, conv.size)
        else:
            top_idxs = np.argpartition(-conv, top_k)[:top_k]
            peak_idx = rng.choice(top_idxs)

        # ----- sub‑gradient from the chosen peak -----
        shift = peak_idx - idx_range
        mask = (shift >= 0) & (shift < n)
        b = np.zeros_like(cur)
        b[mask] = cur[shift[mask]]
        grad = 2.0 * (b - b.mean())            # direction reducing the peak

        # ----- momentum update -----
        v = beta * v + (1.0 - beta) * grad

        # ----- simulated‑annealing step -----
        eta = init_eta * (1.0 - step / num_steps)
        T = init_T * (1.0 - step / num_steps)
        cand = cur - eta * v
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = cur.copy()
            s = cand.sum()
        cand = cand / s

        cand_score = _fast_score(cand)
        accept = False
        if cand_score < cur_score:
            accept = True
        elif T > 0 and rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-16)):
            accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # ----- occasional random block perturbation (global exploration) -----
        if (step + 1) % block_interval == 0 and rng.random() < block_prob:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            block = cur.copy()
            factor = np.exp(rng.normal(0.0, 0.12))
            block[start:start + width] *= factor
            block = np.clip(block, 0.0, 1000.0)
            block = block / block.sum()
            block_score = _fast_score(block)
            if block_score < cur_score:
                cur = block
                cur_score = block_score
                if block_score < best_score:
                    best = block.copy()
                    best_score = block_score

    # ----- final validation via the official evaluator -----
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        fallback = np.full_like(best, 1.0)
        fallback = fallback / fallback.sum()
        best = fallback

    return [float(v) for v in best]
