import numpy as np
import scipy.optimize
import scipy.sparse as sp

def _root_constant_seq(length):
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i-1] * (2*i - 1) / (2*i)
    return np.clip(seq, 0.0, 1000.0)

def _conv_max(seq):
    # full convolution
    conv = np.convolve(seq, seq)
    return np.max(conv)

def _build_conv_matrix(f):
    n = f.shape[0]
    # Build a sparse matrix where each row k corresponds to conv with shift k.
    # Row k uses f_i for column j such that i + j = k.
    # This is equivalent to placing f on diagonals.
    data = []
    offsets = []
    for i in range(n):
        data.append(np.full(n, f[i], dtype=np.float64))
        offsets.append(i)
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    return A

def _solve_max_sum_lp(f, M):
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b_ub = np.full(2*n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                     method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success:
        return None
    return res.x

def _evaluate_score(seq):
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:
        return float('inf')

def propose_candidate(seed=42, **kwargs):
    # Pick a modest size to keep LP tractable
    target_len = 800
    try:
        parent = height_sequence_1
        n = min(len(parent), target_len)
        f = np.array(parent[:n], dtype=np.float64)
    except Exception:
        n = target_len
        f = _root_constant_seq(n)
    # Ensure nonnegativity and bounds
    f = np.clip(f, 0.0, 1000.0)
    # Normalize sum to sqrt(2*n)
    target_sum = np.sqrt(2.0 * n)
    if np.sum(f) == 0:
        f = _root_constant_seq(n)
    scale = target_sum / np.sum(f)
    f = f * scale
    # Compute baseline score
    best_seq = f.copy()
    best_score = _evaluate_score(best_seq)
    # Compute current max convolution bound M
    M = _conv_max(f)
    # Solve LP to increase L1 norm under same M
    b = _solve_max_sum_lp(f, M)
    if b is not None:
        # normalize b to same target sum
        sum_b = np.sum(b)
        if sum_b > 0:
            g = b * (target_sum / sum_b)
            # Clip to feasible range for evaluator
            g = np.clip(g, 0.0, 1000.0)
            # Evaluate score of g
            g_score = _evaluate_score(g)
            if np.isfinite(g_score) and g_score < best_score:
                best_seq = g
                best_score = g_score
            else:
                # Try convex blend between f and g
                t = 0.5
                blended = (1 - t) * f + t * g
                blended_score = _evaluate_score(blended)
                if np.isfinite(blended_score) and blended_score < best_score:
                    best_seq = blended
                    best_score = blended_score
    # As final safety, if best_score is not finite, fallback to uniform positive vector
    if not np.isfinite(best_score):
        best_seq = np.full(n, max(1.0, target_sum / n), dtype=np.float64)
    # Return as list
    return best_seq.tolist()

import numpy as np

def _root_constant_seq(length):
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i-1] * (2*i - 1) / (2*i)
    return np.clip(seq, 0.0, 1000.0)

def _conv_max(seq):
    # full convolution; returns max value
    return np.max(np.convolve(seq, seq))

def _build_ub_matrix(f):
    n = f.shape[0]
    # Build sparse matrix for conv constraints: A_ub @ b <= M
    import scipy.sparse as sp
    # Each column j contributes f[0] on row j, f[1] on row j+1, ... f[n-1] on row j+n-1
    # Equivalent to placing f on diagonals shifted by column index.
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(data, offsets, shape=(2*n - 1, n), format='csr')
    return A

def _linprog_max_sum(f, M):
    try:
        import scipy.optimize as opt
    except Exception:
        return None
    A_ub = _build_ub_matrix(f)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)
    bounds = [(0, None)] * f.shape[0]
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs',

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the full self‑convolution of seq."""
    return float(np.max(np.convolve(seq, seq)))


def _linprog_max_sum(f: np.ndarray, M: float):
    """
    Solve   max sum(b)   s.t.   (f * b)[k] <= M  (all k),   b >= 0.
    Returns the optimal b as a NumPy array or None on failure.
    """
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception:
        return None

    n = f.shape[0]
    # Build a sparse matrix representing the convolution constraints.
    # Row k has entry f[i] in column j whenever i + j == k.
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A_ub = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                # maximize sum(b)  → minimize -sum(b)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={})
    except Exception:
        return None
    if not res.success:
        return None
    return res.x
