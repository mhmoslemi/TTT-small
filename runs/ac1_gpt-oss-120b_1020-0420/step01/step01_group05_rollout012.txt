def _grad_B(a):
    n = a.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    # choose first index of max
    k = int(np.argmax(c))
    i = np.arange(n, dtype=np.int64)
    idx = k - i
    mask = (idx >= 0) & (idx < n)
    gradM = np.zeros_like(a)
    # Use 2*a[idx] for interior values
    gradM[mask] = 2.0 * a[idx[mask]]
    # Optionally adjust interior vs border: for idx at edges, may have duplication count 1, but we ignore.
    S = np.sum(a)

import numpy as np

# ----------------------------------------------------------
# Fast evaluation of the AC1 bound using FFT.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Compute 2 * n * max(conv(a, a)) / (sum(a)**2) efficiently."""
    n = a.shape[0]
    m = 2 * n - 1                     # length of full convolution
    nfft = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                      # a * a in frequency domain
    c = np.fft.irfft(fc, n=nfft)[:m]  # real convolution, trimmed
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------
# Gradient (subgradient) of the bound with respect to a.
# ----------------------------------------------------------
def _grad_B(a: np.ndarray):
    """Return gradient of the bound, the current max value M and sum S."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    k = int(np.argmax(c))                     # index where the max occurs

    # gradient of the max autocorrelation term
    i = np.arange(n, dtype=np.int64)
    idx = k - i                               # index of partner element
    mask = (idx >= 0) & (idx < n)
    gradM = np.zeros_like(a)
    # each interior pair contributes twice; we use 2*a[idx] for all valid pairs
    gradM[mask] = 2.0 * a[idx[mask]]

    S = np.sum(a)
    coeff1 = 2.0 * n / (S * S)
    coeff2 = 4.0 * n * M / (S * S * S)
    grad = coeff1 * gradM - coeff2 * np.ones_like(a)
    return grad, M, S
