def _coeff_binom_half(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1,n):
        coeffs[i] = coeffs[i-1] * (2.0*i - 1.0) / (2.0*i)
    return coeffs

def _conv_max(seq):
    n = seq.size
    m = 2*n - 1
    N = 1 << (m-1).bit_length()
