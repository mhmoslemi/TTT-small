def build_sidon_set(n, target_m=None, attempts=50):
    best_set = set()
    for _ in range(attempts):
        order = rng.permutation(n) (shuffling)
        selected = set()
        sum_used = np.zeros(2*n-1, dtype=bool)
        for i in order:
            # Quick check early break if (?) not needed
            # compute sums with current selected
            ok = True
            # check self sum first
            if sum_used[2*i]:
                ok = False
            else:
                for j in selected:
                    s = i + j
                    if sum_used[s]:
                        ok = False
                        break
            if ok:
                # add
                selected.add(i)
                # mark sums
                sum_used[2*i] = True
                for j in selected:
                    sum_used[i + j] = True
                # note we double-mark i+j for both (i,j) and (j,i) but that doesn't matter cause boolean.
        if len(selected) > len(best_set):
            best_set = selected
            # optionally early break if reached target_m
            if target_m is not None and len(best_set) >= target_m:
                break
    return best_set

import numpy as np

# ------------------------------------------------------------
# Helper: count ordered pair sums for a binary mask (used only for
# exact bound checking – cheap for the small masks we generate)
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices (0 … n-1) where the coefficient is 1
    returns   : np.ndarray of length 2*n-1 with ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt


def _bound_from_selected(selected, n):
    """
    Fast upper‑bound for a binary vector whose ones are at `selected`.
    """
    m = len(selected)
    if m == 0:
        return float("inf")
    cnt = _pair_counts(selected, n)
    max_c = int(cnt.max())
    return float(2.0 * n * max_c / (m * m))

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Search for a non‑negative step‑function (binary mask) whose
    AC1 upper bound (as defined by `evaluate_sequence`) is as low as
    possible.  The search builds large Sidon‑type sets (all pair‑sums
    distinct) using many random permutations and stops as soon as a
    bound below the target is found.  If this fails we fall back to a
    simple power‑law sweep and a tiny hill‑climb, guaranteeing a valid
    answer.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Determine the problem length
    # ----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
        else:
            n = 256
    except NameError:
        n = 256

    # ----------------------------------------------------------------
    # Simple uniform fallback (guaranteed feasible, bound = 2)
    # ----------------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # ----------------------------------------------------------------
    # Sidon‑set construction (binary mask, ordered‑pair collisions ≤2)
    # ----------------------------------------------------------------
    TARGET = 1.50274349                     # desired bound
    # minimal support size guaranteeing bound ≤ TARGET if max_c = 2
    min_m = int(np.ceil(np.sqrt(4.0 * n / TARGET)))
    min_m = max(min_m, 80)                  # never go below a reasonable size
    min_m = min(min_m, n)                   # cannot exceed length

    BEST_ATTEMPTS = 400                     # number of random shuffles

    best_selected = set()

    for _ in range(BEST_ATTEMPTS):
        order = rng.permutation(n)
        selected = set()
        used_sums = set()                   # sums i+j that are already taken

        for i in order:
            s_self = 2 * int(i)
            if s_self in used_sums:
                continue
            conflict = False
            for j in selected:
                if (i + j) in used_sums:
                    conflict = True
                    break
            if conflict:
                continue

            # add i to the set
            selected.add(i)
            used_sums.add(s_self)
            # mark all new sums i + j (including j = i)
            for j in selected:
                used_sums.add(i + j)

            if len(selected) >= min_m:
                # we already have enough elements – stop this trial
                break

        # quick bound estimate (max_c is 2 for a true Sidon set)
        m = len(selected)
        if m < min_m:
            continue
        bound_est = 2.0 * n * 2.0 / (m * m)        # = 4n / m^2
        if bound_est >= best_score:
            continue

        # exact evaluation (still cheap – only one convolution)
        vec = np.zeros(n, dtype=np.float64)
        vec[list(selected)] = 1.0
        score = float(evaluate_sequence(vec.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = vec
            best_selected = selected
            if best_score <= TARGET:
                return [float(x) for x in best_vec]

    # ----------------------------------------------------------------
    # Power‑law sweep (fallback, cheap and often decent)
    # ----------------------------------------------------------------
    p_vals = np.linspace(0.45, 0.80, 71)   # 0.45 … 0.80 in steps of 0.005
    base = np.arange(1, n + 1, dtype=np.float64)   # 1 … n
    for p in p_vals:
        cand = base ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand
            if best_score <= TARGET:
                return [float(x) for x in best_vec]

    # ----------------------------------------------------------------
    # Tiny local‑perturbation hill‑climb (optional polishing)
    # ----------------------------------------------------------------
    for _ in range(num_steps):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.05)))   # ≤ 5 % change
        cand[start:stop] *= factor

        if rng.random() < 0.20:                        # occasional smoothing
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_vec = cand
            best_score = score
            if best_score <= TARGET:
                return [float(x) for x in best_vec]

    # ----------------------------------------------------------------
    # Return the best candidate found
    # ----------------------------------------------------------------
    return [float(x) for x in best_vec]
