import numpy as np

# ------------------------------------------------------------
# Pair‑count utility for binary masks (ordered pairs)
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices (0 … n‑1) where the coefficient is 1
    returns   : np.ndarray of length 2*n-1 with ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = np.fromiter(selected, dtype=np.int32, count=len(selected))
    L = idx.size
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt

# ------------------------------------------------------------
# Fast bound for a binary mask
# ------------------------------------------------------------
def _bound_from_selected(selected, n):
    m = len(selected)
    if m == 0:
        return float("inf")
    cnt = _pair_counts(selected, n)
    max_c = int(cnt.max())
    return float(2.0 * n * max_c / (m * m))

# ------------------------------------------------------------

    k = int(np.argmax(b))
    M = b[k]
    s = vec.sum()
    if s <= 0.0:
        return vec
    # build rev = a_{k-i} (zero outside the array)
    rev = np.zeros_like(vec)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        rev[idx] = vec[k - idx]
    # direction = ∇max(b) - (max(b)/sum) * 1
    direction = rev - (M / s) * np.ones_like(vec)
    # explicit step (learning rate incorporates the analytical factor)
    vec = vec - lr * direction
    # feasibility projection
    np.clip(vec, 0.0, 1000.0, out=vec)
    # avoid an all‑zero vector (evaluate_sequence rejects sum < 0.01)
    if vec.sum() < 1e-8:
        vec[:] = 1.0 / n
    return vec

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed=42, num_steps=30, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) lowers the
    AC1 bound defined by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Determine problem dimension
    # ----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            n = int(parent.size)
        else:
            n = 256
    except NameError:
        n = 256

    # ----------------------------------------------------------------
    # Simple uniform fallback (guaranteed feasible)
    # ----------------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # ----------------------------------------------------------------
    # Optional use of the provided parent as a starting candidate
    # ----------------------------------------------------------------
    try:
        cand = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
        if cand.sum() >= 0.01:
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_vec = cand.copy()
                best_score = sc
    except Exception:
        pass

    # ----------------------------------------------------------------
    # Binary‑mask construction (random starts + greedy removal)
    # ----------------------------------------------------------------
    TARGET = 1.50274349               # desired bound
    INIT_M = min(600, n)              # start with a relatively large subset
    MIN_M  = 20                       # we never drop below this size
    TRIES  = 50                       # independent random starts

    for _ in range(TRIES):
        selected = set(rng.choice(n, size=INIT_M, replace=False))
        while True:
            m = len(selected)
            if m < MIN_M:
                break
            # bound for the current binary mask (exact)
            bound = _bound_from_selected(selected, n)
            if bound <= TARGET:
                break
            # locate a sum that attains the current maximum count
            cnt = _pair_counts(selected, n)
            max_c = int(cnt.max())
            ks = np.where(cnt == max_c)[0]
            k = int(rng.choice(ks))
            # indices that participate in this sum
            involved = [i for i in selected if (k - i) in selected]
            if involved:
                remove = int(rng.choice(involved))
            else:
                remove = int(rng.choice(list(selected)))
            selected.remove(remove)

        # evaluate the binary candidate (exact bound = bound)
        if not selected:
            continue
        cand_vec = np.zeros(n, dtype=np.float64)
        cand_vec[list(selected)] = 1.0
        sc = float(evaluate_sequence(cand_vec.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_vec = cand_vec
            best_score = sc
            if best_score <= TARGET:
                return [float(x) for x in best_vec]

    # ----------------------------------------------------------------
    # Gradient‑descent refinement starting from the current best
    # ----------------------------------------------------------------
    a = best_vec.copy()
    # adapt learning rate to the scale of the problem
    lr = 0.7
    for _ in range(num_steps):
        a = _grad_step(a, n, lr)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_vec = a.copy()
            best_score = sc
            if best_score <= TARGET:
                break

    # ----------------------------------------------------------------
    # Small random perturbations (hill‑climbing)
    # ----------------------------------------------------------------
    for _ in range(200):
        width = int(rng.integers(1, max(2, n // 30)))
        start = int(rng.integers(0, n - width + 1))
        cand = best_vec.copy()
        factor = float(np.exp(rng.normal(0.0, 0.03)))  # ≈ 3 % multiplicative tweak
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_vec = cand
            best_score = sc

    return [float(x) for x in best_vec]
