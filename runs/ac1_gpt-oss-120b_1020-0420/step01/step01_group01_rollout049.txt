    rows = []

    data = []

import numpy as np

# optional SciPy imports – if unavailable we simply skip the LP stage
try:
    import scipy.sparse as _sp
    import scipy.optimize as _opt
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Build a sparse matrix for constraints (f * g)_k <= M  for
# the selected convolution indices ks.
# ------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, ks) -> "_sp.csr_matrix":
    n = f.size
    rows = []
    cols = []
    data = []
    for r, k in enumerate(ks):
        # i runs over indices where f[i] participates in (f * g)[k]
        i_start = 0 if k < n else k - (n - 1)
        i_end = n - 1 if k < n else 2 * n - 2 - k
        # i_start .. i_end inclusive
        for i in range(i_start, i_end + 1):
            j = k - i          # index of g
            rows.append(r)
            cols.append(j)
            data.append(f[i])
    return _sp.csr_matrix((data, (rows, cols)), shape=(len(ks), n))


# ------------------------------------------------------------
# Perform one LP‑based improvement of the current sequence.
# Returns a new sequence g (scaled to the same total sum) or
# None if the LP fails.
# ------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, target_sum: float, top_k: int) -> np.ndarray | None:
    if not _SCIPY_OK:
        return None
    # choose the top_k convolution indices (largest values)
    conv_full = np.convolve(f, f)
    if top_k >= conv_full.size:
        ks = np.arange(conv_full.size)
    else:
        # argpartition is faster than full sort
        part = np.argpartition(-conv_full, top_k)[:top_k]
        ks = part[np.argsort(-conv_full[part])]  # keep as ordered list
    A = _build_constraint_matrix(f, ks)
    b = np.full(A.shape[0], M, dtype=np.float64)

    c = -np.ones(f.size, dtype=np.float64)        # maximise sum(g)
    bounds = [(0.0, None) for _ in range(f.size)]

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception:                               # pragma: no cover
        return None
    if not res.success:
        return None
    g0 = res.x
    s = g0.sum()
    if s <= 0.0:
        return None
    g = g0 * (target_sum / s)
    return g


# ------------------------------------------------------------
# Simple product‑based mass transfer: move a small fraction from
# indices that contribute heavily to the central autocorrelation
# to indices that contribute little.
# ------------------------------------------------------------
def _product_rebalance(seq: np.ndarray, steps: int = 500) -> np.ndarray:
    n = seq.size
    best = seq.copy()
    best_score = evaluate_sequence(best.tolist())
    for _ in range(steps):
        prod = best * best[::-1]                     # a_i * a_{n-1-i}
        i_high = int(np.argmax(prod))
        i_low = int(np.argmin(prod))
        if i_high == i_low:
            break
        delta = best[i_high] * 0.01                 # move 1 % of the source mass
        if delta <= 0.0:
            break
        cand = best.copy()
        cand[i_high] -= delta
        cand[i_low] += delta
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best = cand
            best_score = sc
    return best

# Small stochastic hill‑climbing refinement.
# ------------------------------------------------------------
def _stochastic_refine(seq: np.ndarray, rng: np.random.Generator,
                       steps: int = 2000) -> np.ndarray:
    n = seq.size
    best = seq.copy()
    best_score = evaluate_sequence(best.tolist())
    for _ in range(steps):
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:

        # move a tiny fraction (0.1‑0.5 %) of the source entry

        if delta <= 0.0:

        cand[src] -= delta
        cand[dst] += delta
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best = cand
            best_score = sc
    return best
