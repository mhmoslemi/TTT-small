import numpy as np

# ------------------------------------------------------------
# Helper: generate the truncated series of (1 - z)^(-1/2)
#   a_0 = 1,  a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Fast full convolution using a real FFT (length = next power of 2)
# ------------------------------------------------------------
def full_convolution(x: np.ndarray) -> np.ndarray:
    m = 2 * x.size - 1
    # next power of two ≥ m
    N = 1 << ((m - 1).bit_length())
    fx = np.fft.rfft(x, N)
    conv = np.fft.irfft(fx * fx, N)[:m]
    return conv

def propose_candidate(seed: int = 42, **kwargs):
    """
    Returns a list of non‑negative step‑function heights that (heuristically)
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine length of the problem
    # --------------------------------------------------------
    n = 256
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):
        n = int(height_sequence_1.shape[0])

    cand_hp = generate_half_power_series(n)
    cand_hp = np.clip(cand_hp, 0.0, 1000.0)
    conv_hp = np.convolve(cand_hp, cand_hp)
    best_a = cand_hp.copy()
    best_score = 2.0 * n * conv_hp.max() / (cand_hp.sum() ** 2)

    if "height_sequence_1" in globals() and isinstance(height_sequence_1, np.ndarray):

        if par_score < best_score:

            best_score = par_score

    # --------------------------------------------------------

    a = best_a.copy()
