import numpy as np

# ------------------------------------------------------------
# Half‑power series (1‑z)^(-1/2) – the classical baseline
# ------------------------------------------------------------
def half_power_series(length: int) -> np.ndarray:
    """Return the first ``length`` coefficients of (1‑z)^(-1/2)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Helper: evaluate a candidate (wrapper for the supplied evaluator)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    """Exact score using the pre‑imported ``evaluate_sequence``."""
    return float(evaluate_sequence(vec.tolist()))


# ------------------------------------------------------------
# Deterministic scaling of a prefix of the vector.
# The prefix is shrunk by ``factor`` (<1) and the saved mass is
# redistributed proportionally over the remaining entries.
# ------------------------------------------------------------
def _scale_prefix(vec: np.ndarray, prefix_len: int, factor: float) -> np.ndarray:
    if prefix_len <= 0 or prefix_len >= vec.size:
        return vec.copy()
    early = vec[:prefix_len] * factor
    late = vec[prefix_len:].copy()
    lost = np.sum(vec[:prefix_len]) - np.sum(early)          # mass to give back
    late_sum = np.sum(late)
    if late_sum > 0:
        late += (late / late_sum) * lost                       # proportional share
    else:                                            # pathological – put everything in first slot
        late[0] += lost
    new = np.concatenate([early, late])
    np.clip(new, 0.0, 1000.0, out=new)
    return new


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iters: int = 1500, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Initialise from the supplied parent if it exists,
    # otherwise fall back to the half‑power series.
    # --------------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)
        and height_sequence_1.ndim == 1
        and height_sequence_1.size > 0
    ):
        a = height_sequence_1.astype(np.float64).copy()
    else:
        n = 1024
        a = half_power_series(n)

    np.clip(a, 0.0, 1000.0, out=a)          # safety clamp
    n = a.size

    best = a.copy()
    best_score = _score(best)

    # --------------------------------------------------------
    # 1. Deterministic exploration: shrink a prefix and
    #    redistribute the saved mass.  A few split points and
    #    shrink factors are tried; any improvement is kept.
    # --------------------------------------------------------
    split_fractions = (0.20, 0.30, 0.40, 0.50, 0.60, 0.70, 0.80)
    factors = np.linspace(0.80, 0.99, 11)   # 0.99 is almost the identity
    for frac in split_fractions:
        p = int(frac * n)
        for f in factors:
            cand = _scale_prefix(best, p, f)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    # --------------------------------------------------------
    # 2. Stochastic fine‑tuning: multiplicative perturbations
    #    with a global re‑normalisation (keeps the total sum)
    #    and a slowly decaying step size.
    # --------------------------------------------------------
    cur = best.copy()
    for it in range(max_iters):
        # decreasing log‑normal spread
        sigma = 0.08 * (0.95 ** (it // 200))
        log_factor = rng.normal(0.0, sigma, size=n)
        factor = np.exp(log_factor)
        cand = cur * factor

        # re‑scale to keep the L1 sum unchanged (scaling does not affect the ratio)
        sum_cur = np.sum(cur)
        sum_cand = np.sum(cand)
        if sum_cand == 0:
            continue
        cand *= (sum_cur / sum_cand)

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            cur = cand
            best_score = sc
            if best_score < 1.50274349:            # early exit if target reached
                break

    return [float(x) for x in best]
