    j = max_idx - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(v)
    g[mask] = 2.0 * v[j[mask]]

import numpy as np

_TARGET = 1.50274349  # target AC1 upper bound


def _objective(vec: np.ndarray) -> float:
    """Exact same quantity that evaluate_sequence returns (up to clipping)."""
    s = vec.sum()
    if s <= 0.0:
        return np.inf
    conv = np.convolve(vec, vec)
    return float(2.0 * len(vec) * conv.max() / (s * s))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound
    is hoped to be below the target.  The algorithm starts from several
    simple families, picks the best, then performs a cheap projected
    gradient descent on the maximal convolution entry with occasional
    stochastic jitter.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Determine length and optional parent vector
    # --------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            parent = np.clip(arr, 0.0, 1000.0)

    n = parent.size if parent is not None else 256

    # --------------------------------------------------------------
    # 2. Build a small collection of initial candidates
    # --------------------------------------------------------------
    cand_list = []

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    cand_list.append(uniform)

    # exponential decay
    beta = 0.001
    exp_vec = np.exp(-beta * np.arange(n, dtype=np.float64))
    exp_vec /= exp_vec.sum()
    cand_list.append(exp_vec)

    # power‑law decay
    alpha = 0.5
    pow_vec = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
    pow_vec /= pow_vec.sum()
    cand_list.append(pow_vec)

    # random Dirichlet‑like vector
    rand_vec = rng.random(n) + 1e-6
    rand_vec /= rand_vec.sum()
    cand_list.append(rand_vec)

    # parent, if available
    if parent is not None:
        p = parent.copy()
        total = p.sum()
        if total > 0.0:
            p /= total
        else:
            p = uniform.copy()
        cand_list.append(p)

    # --------------------------------------------------------------
    # 3. Pick the best starting point according to the objective
    # --------------------------------------------------------------
    best_init = None
    best_score = np.inf
    for vec in cand_list:
        sc = _objective(vec)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_init = vec.copy()

    # --------------------------------------------------------------
    # 4. Local optimisation – projected gradient descent on the max
    #    convolution entry, with occasional jitter.
    # --------------------------------------------------------------
    # we will also try a couple of the next‑best seeds to diversify
    sorted_cands = sorted(cand_list, key=_objective)[:3]
    starts = [best_init] + [v.copy() for v in sorted_cands[1:]]

    global_best = best_init.copy()
    global_best_score = best_score

    max_iters = 250  # budget per start

    for a in starts:
        for it in range(max_iters):
            # convolution and location of its maximum
            conv = np.convolve(a, a)
            k_max = int(conv.argmax())

            # sub‑gradient of the maximal convolution entry
            idx = np.arange(n)
            j = k_max - idx

            grad = np.zeros_like(a)

            # normalise step size so that lr * max(grad) = 0.5
            maxg = grad.max()
            lr = 0.5 / maxg if maxg > 0 else 0.0
