import numpy as np

def compute_score(arr: np.ndarray, n: int) -> float:
    """Exact objective used by the evaluator (no clipping needed, arr>=0)."""
    conv = np.convolve(arr, arr)
    maxb = conv.max()
    s = arr.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * maxb / (s * s)


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound defined by evaluate_sequence.
    The routine uses a cheap global scan (power laws, exponentials) and a
    simple multiplicative gradient descent on the dominant convolution peak.
    """
    # ------------------------------------------------------------------
    # 1.  Parent / fallback initialisation
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            tmp.ndim == 1
            and tmp.size > 0
            and np.isfinite(tmp).all()
        ):
            parent = np.clip(tmp, 0.0, 1000.0)

    if parent is None:
        # generic fallback (length 256, uniform)
        n = 256
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size

    # normalise to sum = 1 for internal optimisation (scale‑invariant)
    a = parent.astype(np.float64).copy()
    total = a.sum()
    if total == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= total

    best = a.copy()
    best_score = compute_score(best, n)

    # ------------------------------------------------------------------
    # 2.  Scan simple parametric families (power‑law, exponential)
    # ------------------------------------------------------------------
    indices = np.arange(1, n + 1, dtype=np.float64)

    # power‑law coarse grid
    alphas = np.linspace(0.20, 0.80, 61)          # step ≈ 0.01
    best_alpha = None
    for alpha in alphas:
        w = np.power(indices, -alpha)
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        sc = compute_score(w, n)
        if sc < best_score:
            best = w.copy()
            best_score = sc
            best_alpha = alpha

    # power‑law fine grid around the best coarse α
    if best_alpha is not None:
        lo = max(0.01, best_alpha - 0.02)
        hi = min(0.99, best_alpha + 0.02)
        fine_alphas = np.linspace(lo, hi, 41)
        for alpha in fine_alphas:
            w = np.power(indices, -alpha)
            w_sum = w.sum()
            if w_sum == 0.0:
                continue
            w /= w_sum
            sc = compute_score(w, n)
            if sc < best_score:
                best = w.copy()
                best_score = sc
                best_alpha = alpha

    betas = np.linspace(1e-4, 5e-3, 40)
    for beta in betas:
        w = np.exp(-beta * np.arange(n, dtype=np.float64))
        w_sum = w.sum()
        if w_sum == 0.0:

        w /= w_sum
        sc = compute_score(w, n)
        if sc < best_score:
            best = w.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
