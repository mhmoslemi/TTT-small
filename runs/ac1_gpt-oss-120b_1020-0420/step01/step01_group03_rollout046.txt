def step_update(a, step):
    n = len(a)
    b = full_conv(a)
    k = int(b.argmax())
    # compute range
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end+1)
    g = 2 * a[k - idx]    # shape matches idx
    # compute raw delta
    delta_raw = step * g
    # cap to a[idx] to avoid negative
    delta_capped = np.minimum(delta_raw, a[idx])
    # Ensure not reducing too much; maybe also limit total reduction
    # If total reduction zero, break
    if np.all(delta_capped == 0):
        return a, False
    # update early indices
    a_new = a.copy()
    a_new[idx] -= delta_capped
    total_removed = delta_capped.sum()
    # distribute to later half
    mid = n // 2
    later_len = n - mid
    if later_len > 0:
        addition = total_removed / later_len
        a_new[mid:] += addition
        # Clips to bounds
        np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new, True

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for 0 <= i < length."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    # evaluator clamps to [0,1000]; do the same early.
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full discrete convolution a*a using FFT (real inputs)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv
