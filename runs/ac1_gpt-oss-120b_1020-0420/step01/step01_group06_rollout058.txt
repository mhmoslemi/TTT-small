import numpy as np
# scipy may be unavailable – the code works without it
try:
    from scipy.optimize import minimize_scalar
except Exception:  # pragma: no cover
    minimize_scalar = None

def _analytic_score(r: float, n: int) -> float:
    """Exact AC1 bound for the geometric sequence a_i = r**i (scaled to sum=1)."""
    if r <= 0.0:
        return np.inf
    if r >= 1.0:
        # limit case r → 1 gives the uniform sequence, bound = 2
        return 2.0
    # geometric sum S = sum_i r**i
    log_r = np.log(r)
    r_pow_n = np.exp(log_r * n)            # r**n
    S = (1.0 - r_pow_n) / (1.0 - r)        # exact sum
    # maximum of the convolution a*a (full mode)
    if r <= 0.5:
        max_val = 1.0                      # k=0 is the maximiser
    else:
        # continuous maximiser k* ≈ -1/ln r - 1
        k_star = int(np.floor(-1.0 / log_r - 1.0))
        if k_star < 0:
            k_star = 0
        if k_star >= n:
            k_star = n - 1
        # evaluate at k_star and the next index (if any)
        r_pow_k = np.exp(log_r * k_star)   # r**k_star
        max_val = (k_star + 1) * r_pow_k
        if k_star + 1 < n:
            r_pow_k1 = r_pow_k * r         # r**(k_star+1)
            cand = (k_star + 2) * r_pow_k1
            if cand > max_val:
                max_val = cand
    # AC1 bound
    return float(2.0 * n * max_val / (S * S))
