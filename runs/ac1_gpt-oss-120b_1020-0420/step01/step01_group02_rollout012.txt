import numpy as np

def next_pow2(x):
    return 1 << (x-1).bit_length()

def max_convolution(a):
    n = a.shape[0]
    size = next_pow2(2*n-1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    conv = conv[:2*n-1]
    return np.max(conv)

def ac1_score(a):
    n = a.shape[0]
    S = a.sum()
    if S < 0.01:
        return np.inf
    max_c = max_convolution(a)
    return 2.0 * n * max_c / (S * S)

def generate_half_power_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def scale_block(arr, start, width, factor):
    b = arr.copy()
    # factor can be >0
    b[start:start+width] *= factor
    np.clip(b, 0.0, 1000.0, out=b)
    return b

def propose_candidate(seed: int = 42, **kwargs):
    # Determine n
    n = 256
    if 'height_sequence_1' in globals():
        obj = globals()['height_sequence_1']
        if isinstance(obj, np.ndarray) and obj.ndim == 1 and obj.size > 0:
            n = int(obj.size)
    rng = np.random.default_rng(seed)

    # Base candidates
    bases = []
    # 1. Use provided height_sequence_1 if available
    if 'height_sequence_1' in globals():
        hs = globals()['height_sequence_1']
        if isinstance(hs, np.ndarray) and hs.ndim == 1 and hs.size == n:
            arr = hs.astype(np.float64).copy()
            np.clip(arr, 0.0, 1000.0, out=arr)
            bases.append(arr)
    # 2. Half power series
    hp = generate_half_power_series(n)
    np.clip(hp, 0.0, 1000.0, out=hp)
    bases.append(hp)

    best = None
    best_score = np.inf
    for b in bases:
        sc = ac1_score(b)
        if np.isfinite(sc) and sc < best_score:
            best = b
            best_score = sc

    # Grid search over block scaling
    widths = [max(1, n // k) for k in (2000, 1000, 500, 200, 100, 50, 20, 10)]
    # Ensure unique and sorted
    widths = sorted(set(widths))
    up_factors = [1.1, 1.2, 1.5, 2.0, 3.0]
    down_factors = [0.9, 0.8, 0.7, 0.5, 0.3]

    for width in widths:
        # Early up-scaling
        for f in up_factors:
            cand = scale_block(best, 0, width, f)
            sc = ac1_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
        # Late down-scaling
        for f in down_factors:
            start = n - width
            cand = scale_block(best, start, width, f)
            sc = ac1_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
        # Combined up early & down late
        for fu in up_factors:
            for fd in down_factors:
                cand = best.copy()
                cand = scale_block(cand, 0, width, fu)
                cand = scale_block(cand, n - width, width, fd)
                sc = ac1_score(cand)
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc

    # Random block perturbations as a final refinement
    for _ in range(200):
        # randomly choose block start and width
        max_width = max(1, n // 15)
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.12)))  # larger variance
        cand = best.copy()
        cand[start:start+width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = ac1_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # Return as plain Python list
    return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# Utility: smallest power of two >= x
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    if x <= 1:
        return 1
    return 1 << ((x - 1).bit_length())


# ------------------------------------------------------------
# Fast max of a full convolution (a * a) using FFT
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())


# ------------------------------------------------------------
# AC1 upper bound for a given non‑negative sequence
# ------------------------------------------------------------
def _ac1_score(a: np.ndarray) -> float:
    n = a.shape[0]
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    max_c = _max_convolution(a)
    return 2.0 * n * max_c / (S * S)


# ------------------------------------------------------------
# Half‑power series a_i = binom(2i,i)/4**i   ( (1‑z)^(-1/2) )
# ------------------------------------------------------------
def _half_power_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Multiply a contiguous block by a factor and clip to [0,1000]
# ------------------------------------------------------------
def _scale_block(arr: np.ndarray, start: int, width: int, factor: float) -> np.ndarray:
    b = arr.copy()
    if width > 0:
        b[start : start + width] *= factor
        np.clip(b, 0.0, 1000.0, out=b)
    return b


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------
    n = 256
    if "height_sequence_1" in globals():
        hs = globals()["height_sequence_1"]
        if isinstance(hs, np.ndarray) and hs.ndim == 1 and hs.size > 0:
            n = int(hs.size)

    # --------------------------------------------------------
    # Build initial bases (parent if available, half‑power series)
    # --------------------------------------------------------
    bases = []

    if "height_sequence_1" in globals():
        hs = globals()["height_sequence_1"]
        if isinstance(hs, np.ndarray) and hs.ndim == 1 and hs.size == n:
            arr = hs.astype(np.float64).copy()
            np.clip(arr, 0.0, 1000.0, out=arr)
            bases.append(arr)

    hp = _half_power_series(n)
    np.clip(hp, 0.0, 1000.0, out=hp)
    bases.append(hp)

    # --------------------------------------------------------
    # Choose the best among the simple bases
    # --------------------------------------------------------
    best = None
    best_score = np.inf
    for b in bases:
        sc = _ac1_score(b)
        if np.isfinite(sc) and sc < best_score:
            best = b
            best_score = sc

    # --------------------------------------------------------
    # Grid search: scale early block up, late block down,
    # and combined up/down. Also try a middle block down.
    # --------------------------------------------------------
    # width candidates (roughly 0.05% .. 10% of n)
    width_candidates = sorted(
        set(
            max(1, n // div)
            for div in (2000, 1000, 500, 200, 100, 50, 20, 10, 5, 4, 3, 2)
        )
    )
    up_factors = [1.1, 1.2, 1.5, 2.0, 3.0]
    down_factors = [0.9, 0.8, 0.7, 0.5, 0.3]

    # ----- early block (skip index 0 to avoid huge a0² growth) -----
    start_up = 1 if n > 1 else 0
    for w in width_candidates:
        w_up = min(w, n - start_up)
        if w_up <= 0:
            continue
        for f in up_factors:
            cand = _scale_block(best, start_up, w_up, f)

            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc

    for w in width_candidates:

            cand = _scale_block(best, start_down, w, f)
            sc = _ac1_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc

        start_down = max(0, n - w)

            continue
        for fu in up_factors:
            for fd in down_factors:
                cand = best.copy()
                cand = _scale_block(cand, start_up, w_up, fu)
                cand = _scale_block(cand, start_down, w, fd)
                sc = _ac1_score(cand)
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc

    # ----- middle block down‑scale (target central convolution) -----
    mid_center = n // 2
    for w in width_candidates:

        for f in down_factors:

            sc = _ac1_score(cand)
            if np.isfinite(sc) and sc < best_score:

                best_score = sc

    for _ in range(200):

        cand = best.copy()
        cand[s : s + w] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _ac1_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------
    # Return the best candidate as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
