import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def _root_constant_seq(length: int) -> np.ndarray:
    """
    Generate the first ``length`` terms of the sequence
    a_i = C(2i,i)/4**i, which satisfies (a * a)[k] == 1
    for 0 <= k < length (in the infinite limit).
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _eval(seq: np.ndarray) -> float:
    """
    Safely evaluate a candidate using the provided evaluator.
    Returns np.inf on any error.
    """
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:
        return float('inf')


def _build_toeplitz_upper(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (n x n) upper‑triangular Toeplitz matrix T such that
    (T @ g)[k] = sum_{i=0}^k f[i] * g[k-i]  for 0 <= k < n.
    This matrix supplies the constraints (f * g)[k] <= M for k = 0..n-1.
    """
    n = f.shape[0]
    # offsets are 0, -1, -2, ..., -(n-1)
    offsets = -np.arange(n)
    # each diagonal d (offset = -d) is constant f[d] and has length n-d
    data = [np.full(n - d, f[d], dtype=np.float64) for d in range(n)]
    return sp.diags(data, offsets, shape=(n, n), format='csr')


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1
    score improves on the classic root‑constant sequence.
    The method builds a modest‑size problem, solves a linear program
    that maximizes the total mass while keeping the convolution with
    the current sequence bounded, mixes the result with the current
    sequence and rescales to keep the convolution maximum at 1.
    """
    # ------------------------------------------------------------------
    # 1. Base sequence (root‑constant or provided parent)
    # ------------------------------------------------------------------
    try:
        # parent may be present; use it for a quick fallback later
        full_parent = np.array(height_sequence_1, dtype=np.float64).copy()
        parent_score = _eval(full_parent)
    except Exception:
        full_parent = None
        parent_score = float('inf')

    # work with a manageable length for the LP (≤1500)
    n_target = 1500
    if full_parent is not None and len(full_parent) >= n_target:
        base = full_parent[:n_target].copy()
    else:
        base = _root_constant_seq(n_target)

    base = np.clip(base, 0.0, 1000.0)
    best_seq = base
    best_score = _eval(best_seq)

    # ------------------------------------------------------------------
    # 2. Iterative LP‑based improvement
    # ------------------------------------------------------------------
    seq = base
    for _ in range(3):                     # a few refinement rounds
        # Build constraints for (seq * g)[k] <= M  (k = 0..len(seq)-1)
        A = _build_toeplitz_upper(seq)
        M = np.max(np.convolve(seq, seq))   # current maximal self‑convolution
        if not np.isfinite(M) or M <= 0.0:
            break
        b = np.full(seq.shape[0], M, dtype=np.float64)
        c = -np.ones(seq.shape[0], dtype=np.float64)   # maximise sum(g) == minimise -sum(g)
        bounds = [(0.0, None)] * seq.shape[0]

        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        g = res.x

        # ------------------------------------------------------------------
        # 3. Mix and renormalise
        # ------------------------------------------------------------------
        t = 0.5                               # mixing parameter
        mixed = (1.0 - t) * seq + t * g
        mixed = np.clip(mixed, 0.0, 1000.0)

        # Rescale so that the maximal self‑convolution becomes 1 again
        conv_mixed = np.convolve(mixed, mixed)
        M_mixed = np.max(conv_mixed)
        if not np.isfinite(M_mixed) or M_mixed <= 0.0:
            break
        scale = 1.0 / np.sqrt(M_mixed)
        new_seq = mixed * scale
        new_seq = np.clip(new_seq, 0.0, 1000.0)

        new_score = _eval(new_seq)
        if np.isfinite(new_score) and new_score < best_score:
            best_seq = new_seq
            best_score = new_score
            seq = new_seq                     # continue iterating from improved point
        else:
            break                              # no further progress

    # ------------------------------------------------------------------
    # 4. Choose the best among the improvements and the original parent
    # ------------------------------------------------------------------
    if full_parent is not None and np.isfinite(parent_score):
        # parent may be longer; evaluate its score directly
        if parent_score < best_score:
            return full_parent.tolist()

    # ensure the returned sequence is a non‑empty list of finite numbers
    if not np.isfinite(best_score) or best_seq.size == 0:
        # fallback to a safe uniform sequence
        fallback_len = 256
        best_seq = np.full(fallback_len, 1.0, dtype=np.float64)

    return best_seq.tolist()
