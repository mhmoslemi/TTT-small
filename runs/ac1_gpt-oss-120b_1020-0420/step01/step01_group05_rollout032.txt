import numpy as np

def _fast_score(a):
    n = a.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa*fa
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0*n*M/(S*S)
```

import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the objective using FFT.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1                       # length of linear convolution
    nfft = 1 << ((m - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                         # pointwise product → convolution in time domain
    c = np.fft.irfft(fc, n=nfft)[:m]     # linear convolution (first m entries)
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)
