import numpy as np

def _score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))

def generate_catalan(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*(i)+2)*(2*(i)+1)) / (4.0 * (i+1)*(i+1))
    return a

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size >0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # initialize with Catalan
    cur = generate_catalan(n)
    cur = np.clip(cur, 0.0, 1000.0)
    best = cur.copy()
    best_score = _score(best)

    # deterministic prefix shrink (like before) many combos
    # Let's try a few random ratios and shrink factors
    for _ in range(20):
        r = rng.uniform(0.01, 0.4)  # proportion for early block
        k = max(1, int(n * r))
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0:
            continue
        # shrink factor less than 1
        f = rng.uniform(0.3, 0.9)
        new_early = early * f
        delta = early_sum - new_early.sum()
        # distribute uniformly to tail
        g = 1.0 + delta / tail_sum
        new_tail = tail * g
        cand = np.concatenate([new_early, new_tail])
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    cur = best.copy()
    # hill climbing targeting max conv
    steps = kwargs.get("num_steps", 2000)
    for step in range(steps):
        # compute conv and find max index
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        k_max = int(conv.argmax())
        # define source region
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        # choose a source index that contributes significantly
        # pick with probability proportional to a_i*a_{k-i}
        src_candidates = np.arange(i_low, i_high+1)
        # compute contributions
        contrib = cur[src_candidates] * cur[k_max - src_candidates]
        if contrib.sum() == 0:
            src = rng.integers(i_low, i_high+1)
        else:
            probs = contrib / contrib.sum()
            src = rng.choice(src_candidates, p=probs)
        # avoid zero source
        if cur[src] < 1e-12:
            continue
        # choose destination outside source region
        if k_max < n - 1:
            # source region near start; choose dst in right side
            dst_low = k_max + 1
            dst = rng.integers(dst_low, n)
        else:
            # source region near end; choose dst in left side
            dst_high = i_low - 1
            if dst_high < 0:
                continue
            dst = rng.integers(0, dst_high+1)
        # move a fraction
        delta = cur[src] * rng.uniform(0.02, 0.15)
        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            cur = cand
        else:
            # with small prob accept (simulated annealing)
            if rng.random() < 0.001:
                cur = cand

    return [float(v) for v in best]

import numpy as np

# ----------------------------------------------------------------------
# Score function – identical to the evaluator's computation
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """2 * n * max(convolution) / (sum(seq) ** 2)  (non‑negative, clipped)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution, O(n log n)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Catalan‑like (central binomial) sequence a_i = C(2i,i) / 4^i
# ----------------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative heights with a low AC1 bound.
    The algorithm starts from the Catalan‑like sequence,
    performs a randomised prefix‑shrink / tail‑inflate search,
    then refines the sequence by targeted mass transfer
    (always moving mass away from the current maximal convolution index).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size – use the provided parent if present.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 256
    else:
        n = 256

    # ------------------------------------------------------------------
    # Initial candidate: Catalan‑like sequence (exactly gives max‑conv = 1)
    # ------------------------------------------------------------------
    cur = _catalan_seq(n)
    cur = np.clip(cur, 0.0, 1000.0)
    best = cur.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 1) Randomised prefix‑shrink / tail‑inflate exploration
    # ------------------------------------------------------------------
    for _ in range(800):                      # a few hundred random attempts
        r = rng.uniform(0.01, 0.30)            # size of the early block (fraction)
        k = max(1, int(n * r))
        f = rng.uniform(0.20, 0.95)            # shrink factor for the early block
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue
        new_early = early * f
        delta = early_sum - new_early.sum()    # mass removed from the prefix
        g = 1.0 + delta / tail_sum             # uniform inflation of the tail
        new_tail = tail * g
        cand = np.concatenate([new_early, new_tail])
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            cur = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 2) Targeted hill‑climbing: move mass from indices that
    #    heavily contribute to the current maximal convolution value
    #    to indices that do not participate in that convolution.
    # ------------------------------------------------------------------
    max_steps = kwargs.get("num_steps", 3500)
    no_improve = 0
    for step in range(max_steps):
        conv = np.convolve(cur, cur)
        M = conv.max()
        # choose an index where the maximum is attained (random tie‑break)
        max_idxs = np.flatnonzero(conv == M)
        k = int(rng.choice(max_idxs))

        # source index range that actually contributes to this M
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        if i_low == 0 and i_high == n - 1:
            # the whole array participates – pick a random k instead
            # (otherwise we could not find a destination outside the range)
            k = int(rng.integers(0, 2 * n - 1))
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)

        src_candidates = np.arange(i_low, i_high + 1)
        # contribution of each candidate to the maximal entry
        contrib = cur[src_candidates] * cur[k - src_candidates]
        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            src = int(rng.choice(src_candidates, p=probs))
        else:
            src = int(rng.integers(i_low, i_high + 1))

        if cur[src] <= 1e-12:
            continue

        # choose a destination outside the active set of the maximal index
        if k < n - 1:
            dst_low = k + 1
            if dst_low >= n:
                continue
            dst = int(rng.integers(dst_low, n))
        else:
            dst_high = i_low - 1
            if dst_high < 0:
                continue
            dst = int(rng.integers(0, dst_high + 1))

        # move a modest fraction of mass
        delta = cur[src] * rng.uniform(0.02, 0.15)
        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score(cand)
        if sc < best_score:
            best = cand
            cur = cand
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional acceptance to escape plateaus
            if rng.random() < 0.001:
                cur = cand
                no_improve = 0

        # early termination if stalled
        if no_improve >= 1000:
            break

        # occasional random swap (preserves total mass)
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                cand2 = cur.copy()
                cand2[i], cand2[j] = cand2[j], cand2[i]
                sc2 = _score(cand2)
                if sc2 < best_score:
                    best = cand2
                    cur = cand2
                    best_score = sc2
                    no_improve = 0

    # ------------------------------------------------------------------
    # Return a plain Python list, as required by the harness.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
