import numpy as np

# ------------------------------------------------------------
# Helper: compute ordered‑pair counts for a binary mask
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices (0 … n‑1) where the coefficient is 1
    returns   : np.ndarray of length 2*n‑1 containing ordered‑pair counts
    """
    mask = np.zeros(n, dtype=np.int8)
    mask[list(selected)] = 1
    # convolution of a 0/1 mask with itself yields exactly the ordered‑pair counts
    cnt = np.convolve(mask, mask)
    return cnt.astype(np.int64)

def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step‑function (binary mask) that minimises

    TARGET = 1.50274349          # target bound

    init_m = max(150, int(INIT_FRAC * n))
