import numpy as np

def fast_score(vec: np.ndarray) -> float:
    """
    Compute the AC1 upper bound 2*n*max(a*a)/(sum(a)**2) using FFT.
    This matches evaluate_sequence but is much faster for many candidates.
    """
    n = vec.shape[0]
    if n == 0:
        return np.inf
    total = vec.sum()
    if total <= 0.0:
        return np.inf
    # linear convolution via zero‑padded FFT
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    conv = conv[: 2 * n - 1]
    max_conv = conv.max()
    return 2.0 * n * max_conv / (total * total)


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The search uses a parametric family a_i ∝ (i+1)^(-α)·exp(-β·i) and a
    short stochastic refinement.  A provided `height_sequence_1` is used as
    a fallback parent if it exists and passes basic sanity checks.
    """
    # ------------------------------------------------------------------
    # 1.  Parent (fallback) construction
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            parent = np.clip(tmp, 0.0, 1000.0)

    if parent is None:
        parent = np.ones(256, dtype=np.float64)

    n = parent.shape[0]

    # ------------------------------------------------------------------
    # 2.  Baseline scores
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    parent_score = best_score

    # ------------------------------------------------------------------
    # 3.  Parameter grid (α,β) for the family (i+1)^(-α)·exp(-β·i)
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)          # 0‑based indices
    alphas = np.linspace(0.0, 2.0, 31)           # power‑law exponent
    betas = np.array([0.0, 1e-4, 2e-4, 3e-4, 4e-4, 5e-4,
                      1e-3, 2e-3, 3e-3, 4e-3, 5e-3])  # exponential rate

    for a in alphas:
        pow_term = np.power(i + 1.0, -a)        # (i+1)^(-α)
        for b in betas:
            if b == 0.0:
                w = pow_term
            else:
                w = pow_term * np.exp(-b * i)   # combine with exp decay
            total = w.sum()
            if total <= 0.0:
                continue
            w_norm = w / total
            sc = fast_score(w_norm)
            if sc < best_score:
                best = w_norm.copy()
                best_score = sc

    # ------------------------------------------------------------------
    # 4.  Light stochastic refinement (block perturbations)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for _ in range(200):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 200))
        stop = min(n, start + width)

        cand = best.copy()
        # multiplicative log‑normal noise on a contiguous block
        noise = np.exp(rng.normal(0.0, 0.15, size=stop - start))
        cand[start:stop] *= noise

        if total > 0.0:
            cand /= total

        sc = fast_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best.tolist()))

        best = parent

    # ------------------------------------------------------------------
    # 6.  Return the candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
