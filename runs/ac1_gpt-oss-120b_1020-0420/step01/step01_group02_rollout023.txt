def pair_transfer(seq, rng):
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return seq.copy()
    val_i = seq[i]
    if val_i <= 0.0:
        return seq.copy()
    # Transfer fraction up to 30%
    frac = float(rng.uniform(0.0, 0.3))
    delta = frac * val_i
    cand = seq.copy()
    cand[i] = val_i - delta
    cand[j] = cand[j] + delta
    np.clip(cand, 0.0, 1000.0, out=cand)

import numpy as np

# ------------------------------------------------------------
# Helper: generate the truncated series of (1 - z)^(-1/2)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return the first ``length`` coefficients of (1 - z)^(-1/2)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Simple random block perturbation (keeps O(n) memory)
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block of ``base`` by a log‑normal factor."""
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # modest log‑normal perturbation
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Early‑late block perturbation:
#   shrink an early block and renormalise the later part to keep total sum.
# ------------------------------------------------------------
def early_late_perturbation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale down a short early prefix and scale up the remainder to preserve sum."""
    n = seq.size
    # early block length: between 1 and n//10 (inclusive)
    max_L = max(1, n // 10)
    L = int(rng.integers(1, max_L + 1))

    # reduction factor for the early block (keep it below 1)
    f = float(rng.uniform(0.55, 0.95))

    early = seq[:L].copy()
    late = seq[L:].copy()

    early_new = early * f
    sum_before = seq.sum()
    sum_early_new = early_new.sum()
    sum_late = late.sum()

    # renormalise the later block to keep the total sum unchanged
    if sum_late > 0.0:
        g = (sum_before - sum_early_new) / sum_late
        # keep the scaling moderate – excessive scaling would blow up max‑convolution
        if g > 5.0:
            g = 5.0
        elif g < 0.2:
            g = 0.2
        late_new = late * g
    else:
        late_new = late.copy()

    cand = np.concatenate([early_new, late_new])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Pairwise transfer: move a fraction of mass from one index to another.
# ------------------------------------------------------------
def pair_transfer(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return seq.copy()
    src_val = seq[i]
    if src_val <= 0.0:
        return seq.copy()

    # transfer up to 30 % of the source mass
    frac = float(rng.uniform(0.0, 0.30))
    delta = frac * src_val
    cand = seq.copy()
    cand[i] = src_val - delta
    cand[j] = cand[j] + delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

        hs = globals()["height_sequence_1"]
        if isinstance(hs, np.ndarray) and hs.ndim == 1 and hs.size > 0:
            n = int(hs.size)
            base = hs.astype(np.float64).copy()
        else:
            base = generate_half_power_series(n)
    else:
        base = generate_half_power_series(n)

    np.clip(base, 0.0, 1000.0, out=base)

    # --------------------------------------------------------
    # Keep the best candidate seen so far
    # --------------------------------------------------------
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------
    # Stochastic improvement loop
    # --------------------------------------------------------
    max_steps = 5000                     # enough iterations for a modest search
    target_score = 1.50274349            # early‑exit threshold

    for step in range(max_steps):
        r = rng.random()
        if r < 0.35:
            cand = early_late_perturbation(best, rng)
        elif r < 0.70:
            cand = random_block_perturbation(best, rng)
        else:
            cand = pair_transfer(best, rng)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            if best_score < target_score:
                break   # we have reached the required bound

    # --------------------------------------------------------
    # Return a plain Python list of floats (the harness expects this)
    # --------------------------------------------------------
    return [float(x) for x in best]
