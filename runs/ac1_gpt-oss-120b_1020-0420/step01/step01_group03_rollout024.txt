def max_step_size(a, d):
    # a: current coefficients; d: descent direction (positive changes)
    # Return maximum α such that a + α*d in [0, 1000].
    # constraints:
    # if d_i > 0: α <= (1000 - a_i) / d_i
    # if d_i < 0: α <= -a_i / d_i  (since d_i negative)
    # if d_i == 0: no bound.
    pos_mask = d > 0
    neg_mask = d < 0
    alpha_pos = np.min((1000.0 - a[pos_mask]) / d[pos_mask]) if np.any(pos_mask) else np.inf
    alpha_neg = np.min((-a[neg_mask]) / d[neg_mask]) if np.any(neg_mask) else np.inf
    return min(alpha_pos, alpha_neg)

import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    # safety clipping matches evaluator limits
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real inputs)."""
    n = a.shape[0]
    size = 2 * n - 1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv


def _gradient_M(a: np.ndarray, k: int) -> np.ndarray:
    """Derivative of max convolution value M w.r.t. a, given argmax index k."""
    n = a.shape[0]
    idx = np.arange(n)
    other = k - idx
    mask = (other >= 0) & (other < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[other[mask]]
    return grad


def _max_step(a: np.ndarray, d: np.ndarray) -> float:
    """Largest step size α so that a + α*d stays inside [0,1000]."""
    pos = d > 0.0
    neg = d < 0.0
    if np.any(pos):
        alpha_pos = np.min((1000.0 - a[pos]) / d[pos])
    else:
        alpha_pos = np.inf
    if np.any(neg):
        alpha_neg = np.min((-a[neg]) / d[neg])
    else:
        alpha_neg = np.inf
    return min(alpha_pos, alpha_neg)
