def convolve_fft(x, y):
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1) - 1).bit_length()
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n+m-1]

import numpy as np

# ------------------------------------------------------------
# Helper: generate (1 - z)^(-1/2) series as a fallback initializer.
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    """Return the first `length` coefficients of (1 - z)^(-1/2)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Small random block perturbation (always O(n) memory).
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block of `base` by a log‑normal factor."""
    n = base.size
    if n <= 1:
        return base.copy()
    max_width = max(1, n // 10)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.1)))  # moderate variation
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Convolution using FFT (matrix‑free, linear memory).
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: next power of two >= x."""
    return 1 << ((x - 1).bit_length())


def convolve_fft_fixed(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Return full convolution of `x` and `y` using FFT of given `size`."""
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: x.size + y.size - 1]

def propose_candidate(seed: int = 42, num_iters: int = 500, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size and initialise a feasible candidate.
    # -----------------------------------------------------------------
    if (
        "height_sequence_1" in globals()
        and isinstance(height_sequence_1, np.ndarray)

        n = 1024

    np.clip(base, 0.0, 1000.0, out=base)

    # Ensure a non‑trivial sum (the evaluator rejects sums < 0.01).
    if base.sum() < 0.01:
        base *= 0.01 / base.sum()

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # -----------------------------------------------------------------
    # Pre‑compute FFT sizes (constant throughout the search).
    # -----------------------------------------------------------------
    size_a = _next_pow2(2 * n - 1)          # for a * a
    size_w = _next_pow2(3 * n - 2)          # for indicator * a_rev

    no_improve_counter = 0
    max_no_improve = 15  # after this many idle steps we try a random block

    for _ in range(num_iters):

        src = int(np.argmax(w))

        if src == tgt:

        delta = 0.02 * best[src]          # move 2 % of the source mass

        else:

            cand = best.copy()
            cand[src] -= delta
            cand[tgt] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)

        # ----- evaluate candidate -----
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
