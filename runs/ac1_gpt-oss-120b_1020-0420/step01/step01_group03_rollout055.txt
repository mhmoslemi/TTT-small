import numpy as np

def _root_constant_seq(length: int) -> np.ndarray:
    """
    Generates the first ``length`` terms of the sequence
    a_i = C(2i,i)/4**i.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step function that (heuristically) improves
    the AC1 upper bound.  The routine starts from the classic root‑constant
    sequence, performs a few gradient‑style updates that move mass from the
    early large coefficients to the tail, and keeps the best result.
    If a parent construction ``height_sequence_1`` is present it is used as
    an additional candidate.
    """
    # ----- determine length -------------------------------------------------
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256  # safe fallback length

    # ----- base candidates --------------------------------------------------
    root_seq = _root_constant_seq(n)

    def _score(arr):
        """Return the evaluator score or np.inf on failure."""
        try:
            s = float(evaluate_sequence(arr.tolist()))
            return s if np.isfinite(s) else np.inf
        except Exception:
            return np.inf

    best_seq = root_seq.copy()
    best_val = _score(best_seq)

    # try the parent if it exists
    try:
        parent_seq = np.array(height_sequence_1, dtype=np.float64)
        parent_val = _score(parent_seq)
        if parent_val < best_val:
            best_seq = parent_seq.copy()
            best_val = parent_val
    except Exception:
        pass

    # ----- gradient‑type refinement -----------------------------------------
    a = best_seq.copy()
    max_iters = 5
    bound_initial = 0.5          # maximum absolute change per element each step

    for it in range(max_iters):
        # full convolution via FFT (linear time)
        L = 2 * n
        fft_a = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fft_a * fft_a, n=L)[:2 * n - 1]

        M = float(conv.max())
        idx = int(conv.argmax())
        sum_a = float(a.sum())

        # direction = M for all indices, corrected where a contributes to the max
        direction = np.full(n, M, dtype=np.float64)
        j_start = max(0, idx - (n - 1))
        j_end = min(idx, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rev_idx = idx - js                    # indices inside a
            direction[js] = M - a[rev_idx] * sum_a

        max_change = float(np.max(np.abs(direction)))
        if max_change == 0.0:
            break

        # bound the step size so that no entry moves more than ``bound_initial``
        bound = bound_initial / (2 ** it)        # shrink bound a little each iteration
        lr = bound / max_change
        a_new = a + lr * direction
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        new_val = _score(a_new)
        if new_val < best_val:
            a = a_new
            best_seq = a_new.copy()
            best_val = new_val
        else:
            # no improvement – stop early
            break

    # ----- final safety net --------------------------------------------------
    if not np.isfinite(best_val):
        best_seq = np.full(n, 1.0, dtype=np.float64)

    return best_seq.tolist()
