def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)
    # Determine n and parent
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = None
        else:
            parent = np.clip(parent, 0.0, 1000.0)
    else:
        parent = None
    if parent is None:
        n = 256
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size
    # baseline candidate: uniform
    best = np.ones(n, dtype=np.float64)
    best /= best.sum()
    best_score = compute_score(best)

    # fallback: try parent normalized
    if parent is not None:
        cand = parent.copy()
        total = cand.sum()
        if total > 0:
            cand /= total
            sc = compute_score(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    # grid search over power-exponential families
    indices = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.0, 2.0, 41)  # 0 to 2 step ~0.05
    betas = np.linspace(0.0, 0.01, 21) # 0 to .01 step .0005
    for a in alphas:

        if a == 0.0:
            pow_part = np.ones(n, dtype=np.float64)
        else:
            pow_part = np.power(indices + 1.0, -a)
        for b in betas:
            if b == 0.0:
                exp_part = np.ones(n, dtype=np.float64)
            else:
                exp_part = np.exp(-b * indices)
            w = pow_part * exp_part
            s = w.sum()
            if s == 0.0:
                continue
            w /= s
            sc = compute_score(w)
            if sc < best_score:
                best = w
                best_score = sc

# ----------------------------------------------------------------------
# Helper utilities (matrix‑free, linear memory)

def _max_convolution(a: np.ndarray) -> float:
    """
    Linear‑time convolution via FFT (zero‑padded) and return the maximum entry.
    """
    n = a.shape[0]
    # size = next power of two >= 2*n-1  (ensures linear convolution)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # first (2*n-1) entries are the linear convolution
    return float(np.max(conv[: 2 * n - 1]))
