def _build_lp_matrix(f, tail_start):
    n = f.shape[0]
    m = n - tail_start
    num_shifts = 2*n - 1
    rows = []
    cols = []
    data = []
    for k in range(num_shifts):
        i_start = max(0, k - (n - 1))
