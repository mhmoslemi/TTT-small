import numpy as np

# ------------------------------------------------------------
# Helper: fast replica of the provided evaluator
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Compute the AC1 bound for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Generate a power‑law step sequence a_i = (i+1)^{-alpha}
# ------------------------------------------------------------
def _power_law_seq(n: int, alpha: float) -> np.ndarray:
    """Return a length‑n non‑negative vector with power‑law decay."""
    idx = np.arange(1, n + 1, dtype=np.float64)   # 1‑based indices
    return idx ** (-alpha)
