def approx_score(a):
    n = a.size
    m = 2*n - 1
    nfft = 1 << (m-1).bit_length()  # next power of two
    # maybe use rfft: len = nfft//2+1
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[:m]
    max_conv = conv.max()
    sum_a = a.sum()
    # avoid division by zero
    if sum_a == 0:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)

def approx_score(vec):
    # Return approximate score
    a = vec.astype(np.float64)
    # Normalize sum to 1
    s = a.sum()
    if s == 0:
        return np.inf
    a = a / s
    n = a.size
    m = 2*n - 1
    nfft = 1 << (m - 1).bit_length()
    # Real FFT
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[:m]
    max_conv = conv.max()
    # Score
    return 2.0 * n * max_conv

import numpy as np

# The evaluator is provided in the execution environment
# from it we will obtain the exact AC1 bound for a candidate.
# We also implement a fast, matrix‑free approximate scorer based on FFT
# (scale‑invariant, suitable for many cheap evaluations).

def _fft_score(vec: np.ndarray) -> float:
    """Return the AC1 bound using FFT (vec is assumed non‑negative)."""
    a = vec.astype(np.float64, copy=False)
    s = a.sum()
    if s == 0.0:
        return np.inf
    a = a / s                     # normalise: sum == 1  → denominator disappears
    n = a.size
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    max_conv = conv.max()
    return 2.0 * n * max_conv          # denominator = 1 after normalisation

def propose_candidate(seed: int = 42, **kwargs):

    Search for a non‑negative step‑function (list of floats) whose
    AC1 bound is as low as possible.  The routine works in three phases:
      1.  Scan a rich family of analytic shapes (exponential, power‑law,
          their product, simple linear taper, sine‑based windows).
      2.  Refine the best shape with a finer grid and a short local‑search.
      3.  Verify the final best with the exact evaluator and return it.
    The result is guaranteed to be a non‑empty list of finite numbers.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Base sequence (fallback = height_sequence_1 if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            base = np.clip(base, 0.0, 1000.0)
        else:
            base = np.ones(256, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    n = base.size
    # Normalise once – the exact score is scale invariant, so any scaling works.
    best = base / base.sum()
    best_score = _exact_score(best)

    # ------------------------------------------------------------------
    # 2.  Helpers
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)                    # 0 … n‑1

    def try_candidate(arr: np.ndarray) -> None:
        """Update global best if arr (already normalised) improves the score."""
        nonlocal best, best_score
        sc = _fft_score(arr)
        if sc < best_score:
            best = arr.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 3.  Exponential family – dense log‑grid for the decay rate
    # ------------------------------------------------------------------
    betas = np.logspace(np.log10(1e-5), np.log10(5e-2), num=200)
    for b in betas:
        w = np.exp(-b * idx)
        w = w / w.sum()
        try_candidate(w)

    # ------------------------------------------------------------------
    # 4.  Power‑law family
    # ------------------------------------------------------------------
    alphas = np.linspace(0.05, 3.0, num=150)
    for a in alphas:
        w = (idx + 1.0) ** (-a)
        w = w / w.sum()
        try_candidate(w)

    # ------------------------------------------------------------------
    # 5.  Product of power‑law and exponential (grid sampling)
    # ------------------------------------------------------------------
    for _ in range(120):
        a = rng.uniform(0.05, 3.0)
        b = rng.uniform(1e-5, 5e-2)
        w = ((idx + 1.0) ** (-a)) * np.exp(-b * idx)
        w = w / w.sum()
        try_candidate(w)

    # ------------------------------------------------------------------
    # 6.  Linear taper (including the flat case)
    # ------------------------------------------------------------------
    gammas = np.linspace(0.0, 2.0 / (n - 1), num=120)
    for g in gammas:
        w = 1.0 - g * idx
        w = np.clip(w, 0.0, None)
        if w.sum() == 0.0:
            continue
        w = w / w.sum()
        try_candidate(w)

    # ------------------------------------------------------------------
    # 7.  Sine‑based windows (Hann‑like, raised‑cosine, etc.)
    # ------------------------------------------------------------------
    for p in (1, 2, 3, 4):
        w = np.sin(np.pi * (idx + 0.5) / (2.0 * n)) ** p
        w = w / w.sum()
        try_candidate(w)

    # ------------------------------------------------------------------
    # 8.  Local refinement around the best exponential candidate
    # ------------------------------------------------------------------
    # Identify a tight neighbourhood of the best decay rate seen so far
    # (the last successful beta from the exponential scan)
    # If no exponential improved over the parent, use a modest default.
    def _estimate_best_beta():
        # crude extraction: the best from the exponential scan is usually
        # the beta that gave the current best score and was tried in step 3.
        # We re‑scan the beta grid with the current best shape kept,
        # looking for the one that minimises the FFT score.
        # This tiny inner loop is cheap.
        best_local_beta = None
        best_local_score = np.inf
        for b in betas:

            w = w / w.sum()
            sc = _fft_score(w)
            if sc < best_local_score:
                best_local_score = sc
                best_local_beta = b
        return best_local_beta if best_local_beta is not None else 1e-3

    est_beta = _estimate_best_beta()
    fine_betas = np.linspace(max(1e-6, est_beta * 0.6),
                             min(5e-2, est_beta * 1.4), 80)
    for b in fine_betas:
        w = np.exp(-b * idx)
        w = w / w.sum()
        try_candidate(w)

    # ------------------------------------------------------------------
    # 9.  Small random block perturbations around the current best
    # ------------------------------------------------------------------
    for _ in range(150):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 250))
        stop = min(n, start + width)

        cand = best.copy()
        # multiplicative log‑normal perturbation on a contiguous block
        noise = np.exp(rng.normal(0.0, 0.04, size=stop - start))
        cand[start:stop] *= noise
        cand = cand / cand.sum()
        try_candidate(cand)

    # ------------------------------------------------------------------
    # 10.  Final exact verification on the best candidate and a few neighbours
    # ------------------------------------------------------------------
    best_exact = _exact_score(best)
    # if a neighbour beats it, adopt it
    for _ in range(12):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 250))
        stop = min(n, start + width)

        cand = best.copy()
        cand[start:stop] *= np.exp(rng.normal(0.0, 0.03, size=stop - start))
        cand = cand / cand.sum()
        sc = _exact_score(cand)
        if sc < best_exact:
            best = cand.copy()
            best_exact = sc

    # ------------------------------------------------------------------
    # Return the final candidate as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
