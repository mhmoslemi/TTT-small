import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _root_constant_seq(length):
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i-1] * (2*i - 1) / (2*i)
    return np.clip(seq, 0.0, 1000.0)

def _build_convolution_constraints(f):
    n = len(f)
    rows = 2*n - 1
    data = []
    indices = []
    indptr = np.empty(rows + 1, dtype=np.int32)
    ptr = 0
    for k in range(rows):
        j_start = max(0, k - (n-1))
        j_end = min(k, n-1)
        indptr[k] = ptr
        # number of entries
        for j in range(j_start, j_end + 1):
            data.append(f[j])
            indices.append(k - j)  # column index
            ptr += 1
    indptr[rows] = ptr
    return csr_matrix((data, indices, indptr), shape=(rows, n))

def _blend_and_bound(f, g, M, max_iter=10):
    # try blending with decreasing t until max conv <= M (or small increase)
    t = 0.5
    for _ in range(max_iter):
        f_new = (1 - t) * f + t * g
        # compute max convolution
        conv = np.convolve(f_new, f_new, mode='full')
        new_M = float(conv.max())
        if new_M <= M * (1 + 1e-8):
            return f_new, new_M
        t *= 0.5
    # fallback: return original f
    return f, M

def propose_candidate(seed: int = 42, **kwargs):

    try:

    # Pick a moderate size for tractable LP; cap at 1200 for safety

    # initial sequence f
    f = _root_constant_seq(n)
    # compute its self-convolution max
    conv_f = np.convolve(f, f, mode='full')
    M = float(conv_f.max())
    best_seq = f.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = float('inf')
    # iteration parameters
    max_iters = 5
    for it in range(max_iters):
        # Build convolution constraints f * g <= M
        A_ub = _build_convolution_constraints(f)
        b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
        bounds = [(0.0, 1000.0) for _ in range(n)]
        # solve LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if not res.success:

        g = np.clip(res.x, 0.0, 1000.0)

        # evaluate new candidate

        except Exception:

        # early exit if already below target

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for i=0..length-1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0

        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)

def _build_convolution_constraints(f: np.ndarray) -> csr_matrix:
    """Construct sparse matrix A defining (f * g)[k] <= M for all k."""
    n = f.shape[0]
    rows = 2 * n - 1
    data = []
    indices = []
    indptr = np.empty(rows + 1, dtype=np.int32)

    return csr_matrix((data, indices, indptr), shape=(rows, n))
