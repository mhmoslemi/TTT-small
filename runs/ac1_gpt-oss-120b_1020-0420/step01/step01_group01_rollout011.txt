       n = f.shape[0]
       b = cp.Variable(n)
       constraints = [b >= 0, b <= 1000]
       conv = cp.conv(cp.Parameter(shape=f.shape, value=f), b)
       constraints.append(conv <= M)
       objective = cp.Maximize(cp.sum(b))
       prob = cp.Problem(objective, constraints)
       try:

       except Exception:
           return None
       if b.value is None:
           return None
       return np.maximum(0, np.minimum(1000, b.value))

import numpy as np
try:
    import cvxpy as cp
    _CVX_AVAILABLE = True
except Exception:
    _CVX_AVAILABLE = False

def _conv_max(seq):
    # Use FFT for speed
    n = seq.size
    size = 1 << (2*n - 1).bit_length()
    fft_a = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    return conv.max()

def _score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    m = _conv_max(seq)
    return float(2.0 * seq.size * m / (s * s))

def _solve_lp(f, M):
    n = f.shape[0]
    b = cp.Variable(n)
    constraints = [b >= 0, b <= 1000]
    # linear convolution constraint
    conv_expr = cp.conv(f, b)
    constraints.append(conv_expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=10000, verbose=False)
    except Exception:
        return None
    if b.value is None:
        return None
    b_val = np.asarray(b.value, dtype=np.float64)
    # enforce bounds manually in case of small violations
    b_val = np.clip(b_val, 0.0, 1000.0)
    return b_val

def propose_candidate(seed: int = 42, max_iter: int = 20, **kwargs):
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals():
        f = np.asarray(height_sequence_1, dtype=np.float64)
        f = np.clip(f, 0.0, 1000.0)
    else:
        n = kwargs.get('n', 256)
        # simple baseline: decreasing 1/sqrt(i+1)
        idx = np.arange(n, dtype=np.float64)
        f = 1.0 / np.sqrt(idx + 1)
        f = np.clip(f, 0.0, 1000.0)

    n = f.size
    best = f.copy()
    best_score = _score(best)

    target = 1.50274349

    for it in range(max_iter):
        # compute M
        M = _conv_max(best)
        # Try solve LP
        b = None
        if _CVX_AVAILABLE:
            b = _solve_lp(best, M)
        if b is None:
            # fallback: simple deterministic mass shift (like previous program)
            # choose a random suffix length and allocate mass uniformly while respecting M
            # compute prefix sums of best
            prefix = np.cumsum(best)
            # try a few suffix lengths
            for L in [int(0.01*n), int(0.02*n), int(0.05*n), int(0.1*n)]:
                if L <= 0 or L > n:
                    continue
                # sum of first L elements of best (largest values)
                S = prefix[L-1]
                # maximum H allowed for constant tail to keep conv <= M:
                H = M / S if S>0 else 0.0
                if H <= 0:
                    continue
                # construct candidate: keep best unchanged in first n-L entries, set last L entries to H
                cand = np.concatenate([best[:n-L], np.full(L, H)])
                cand = np.clip(cand, 0.0, 1000.0)
                cand_score = _score(cand)
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score
            # continue to next iter
            continue

        # mixing factor t, small to avoid raising M
        # Compute self-conv bound of b
        M_b = _conv_max(b)
        # Choose t adaptively: if M_b <= M then can use larger t
        if M_b <= M:
            t = 0.5
        else:
            # limit t so that increase in max <= 1%
            # approx factor increase = 1 + t**2 * (M_b / M - 1)
            # solve for t: t**2 * (c-1) <= 0.01 => t <= sqrt(0.01/(c-1))
            c = M_b / M
            t = min(0.5, (0.01 / max(c-1, 1e-12))**0.5)
            t = max(t, 0.01)  # ensure minimum progress
        cand = (1 - t) * best + t * b
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        if best_score <= target:
            break

    return [float(v) for v in best]

import numpy as np

# ------------------------------------------------------------
# Optional CVXPY import – used for a sparse linear program.
# ------------------------------------------------------------
try:
    import cvxpy as cp
    _CVX_PRESENT = True
except Exception:  # pragma: no cover
    _CVX_PRESENT = False

# ------------------------------------------------------------
# Utility: fast max of full convolution using FFT.
# ------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    """Return max(conv(seq,seq)) using an FFT‑based convolution."""
    n = seq.size
    # length of full convolution
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (conv_len - 1).bit_length()
    # real FFT, multiply, inverse FFT
    fft_a = np.fft.rfft(seq, size)
    conv_full = np.fft.irfft(fft_a * fft_a, size)[:conv_len]
    return float(conv_full.max())

# ------------------------------------------------------------
# Scoring function – identical to the evaluator's computation.
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    m = _conv_max(seq)
    return float(2.0 * seq.size * m / (s * s))

# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence – used as a fallback.
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

# ------------------------------------------------------------
# Linear program: maximise sum(b) with non‑negative b and
#   convolution(f,b) <= M  (elementwise).
# ------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    if not _CVX_PRESENT:          # pragma: no cover
        return None
    n = f.shape[0]
    b = cp.Variable(n)
    constraints = [
        b >= 0,
        b <= 1000,
        cp.conv(f, b) <= M
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=10000, verbose=False)
    except Exception:            # pragma: no cover
        return None
    if b.value is None:          # pragma: no cover
        return None
    bv = np.asarray(b.value, dtype=np.float64)
    return np.clip(bv, 0.0, 1000.0)

# ------------------------------------------------------------
# Core improvement routine – a few LP solves followed by
# small convex‑combination steps.
# ------------------------------------------------------------
def _improve_sequence(seq: np.ndarray, target: float) -> np.ndarray:
    best = seq.copy()
    best_sc = _score(best)
    cur = best.copy()
    cur_sc = best_sc
    # parameters controlling the search
    max_lp_iters = 3               # number of LP invocations
    mix_steps_per_lp = 80          # convex‑combination steps after each LP
    mix_factor = 0.12              # default mixing weight
    for _ in range(max_lp_iters):
        M = _conv_max(cur)
        g = _solve_lp(cur, M)
        if g is None:
            break
        # repeated mixing with the same g
        for _ in range(mix_steps_per_lp):
            cand = (1.0 - mix_factor) * cur + mix_factor * g
            cand = np.clip(cand, 0.0, 1000.0)
            cand_sc = _score(cand)
            # keep globally best
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
                if best_sc <= target:
                    return best
            # accept only non‑worsening candidates for the next step
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
            else:
                # if no progress, shrink the mixing factor slightly
                mix_factor = max(0.01, mix_factor * 0.7)
                break
    return best

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
