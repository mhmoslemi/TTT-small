# ---------------------------------------------------------------
# Helper: uniform vector of given length with unit L1 norm
# ---------------------------------------------------------------
def _uniform(length: int) -> np.ndarray:
    """Return a length‑`length` non‑negative vector with sum = 1."""
    if length <= 0:
        return np.array([1.0])
    vec = np.full(length, 1.0 / length, dtype=np.float64)
    return np.clip(vec, 0.0, 1000.0)


# ---------------------------------------------------------------

def propose_candidate(seed: int = 42, num_trials: int = 1024, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that
    minimizes the AC1 upper bound as evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # Determine length from the supplied parent if present, otherwise fallback.
    try:

        n = int(parent.size)
        if n == 0:
            raise ValueError
    except Exception:
        n = 256

    # Start from the optimal (in theory) uniform distribution.
    base = _uniform(n)
    best_vec = base.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # Simple stochastic refinement – inexpensive and matrix‑free.
    for _ in range(num_trials):
