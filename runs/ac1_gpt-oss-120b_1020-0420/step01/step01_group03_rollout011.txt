import numpy as np
import math

def _root_constant_seq(length: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4**i for 0 <= i < length."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims
    to minimise the AC1 bound used by `evaluate_sequence`.
    The method:
      1. Use the parent construction `height_sequence_1` if present,
         otherwise fall back to the root‑constant sequence.
      2. Solve the linear programme
            maximise   sum(b)
            subject to f * b <= M   (convolution, first n entries)
                       b >= 0
         where f is the current sequence and M = max(f*f).
      3. Scale the optimiser's solution so that its sum equals sqrt(2n).
         The resulting maximal convolution equals sqrt(2n)/sum(b_opt),
         which is the achieved score.
      4. If any step fails, fall back to a simple valid sequence.
    """
    # ------------------------------------------------------------------
    # 1. Base sequence f
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
        f = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 256
        f = _root_constant_seq(n)

    # Ensure values are within the evaluator's bounds.
    f = np.clip(f, 0.0, 1000.0)

    # Upper bound M = max of the self‑convolution (first n entries are enough)
    conv_ff = np.convolve(f, f)
    M = float(np.max(conv_ff[:n]))

    # ------------------------------------------------------------------
    # 2. Linear programme to maximise the sum of b under f*b <= M
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp

        # Parameter for the fixed sequence f
        f_param = cp.Parameter(n, value=f)

        # Decision variables
        b = cp.Variable(n, nonneg=True)

        # Linear convolution (full length = 2n-1)
        conv_fb = cp.conv(f_param, b)

        # Constraints: first n entries of the convolution must not exceed M
        constraints = [conv_fb[:n] <= M]

        # Objective: maximise the total sum of b
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        # Solve (ECOS works well for pure LPs)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)

        if b.value is None:
            raise RuntimeError("LP solver failed to produce a solution")

        b_opt = np.array(b.value, dtype=np.float64)
        b_opt = np.maximum(b_opt, 0.0)          # eliminate tiny negatives
        b_opt = np.clip(b_opt, 0.0, 1000.0)      # respect hard caps

        sum_b = b_opt.sum()
        if sum_b <= 0.0:
