def convolve_max(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    # real FFT
    fft_a = np.fft.rfft(a, n=size)
    b = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
    max_b = b.max()
    return max_b, b

import numpy as np

# -------------------------------------------------
# Helper utilities (top‑level, no closures)
# -------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    """Classic a_i = C(2i,i)/4**i sequence (non‑negative)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(seq, 0.0, 1000.0)


def _max_convolution_fft(arr: np.ndarray) -> float:
    """
    Returns max(arr * arr) using an FFT based convolution.
    Works for any length; the convolution size chosen is a power of two.
    """
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fft = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[: 2 * n - 1]
    return float(conv.max())


def _transform_seq(seq: np.ndarray, split: int, lam: float) -> np.ndarray:
    """
    Scale the first ``split`` entries by ``lam`` (0 < lam ≤ 1) and
    redistribute the removed mass uniformly over the remaining entries.
    The total L1‑norm stays unchanged.
    """
    early = seq[:split] * lam
    early_orig_sum = float(seq[:split].sum())
    early_sum = float(early.sum())
    delta = early_orig_sum - early_sum          # mass taken from early part
    late_len = seq.shape[0] - split
    if late_len > 0:
        add = delta / late_len
        late = seq[split:] + add
    else:
        late = seq[split:]
    return np.concatenate([early, late])
