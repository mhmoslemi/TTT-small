def _score(seq: np.ndarray) -> float:
    """Calculate 2*n*max(conv(seq,seq))/sum(seq)^2 with clipping."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)               # full convolution
    m = conv.max()
    n = seq.size
    return float(2.0 * n * m / (s * s))

# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence: a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------
    # Determine problem size – use the provided parent if available.
    # --------------------------------------------------
    if "height_sequence_1" in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0:
            n = parent_arr.size
        else:
            n = 256
    else:
        n = 256

    # --------------------------------------------------
    # Initial candidate: Catalan‑like sequence (normalized by clipping)
    # --------------------------------------------------
    best = _catalan_seq(n)
    best = np.clip(best, 0.0, 1000.0)
    best_score = _score(best)

    # --------------------------------------------------
    # If a parent sequence exists and is better, start from it.
    # --------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(parent_arr, 0.0, 1000.0)
        sc = _score(parent)
        if sc < best_score:
            best = parent.copy()
            best_score = sc

    # --------------------------------------------------
    # Aggressive deterministic re‑allocation.
    # Shrink an early prefix (various sizes) by a wide range of factors
    # and compensate by uniformly inflating the remaining tail.
    # --------------------------------------------------
    prefix_ratios = np.linspace(0.01, 0.50, 10)   # 1 % … 50 % of the vector
    shrink_factors = np.linspace(0.01, 1.0, 100)  # shrink from 1 % to 100 %
    for r in prefix_ratios:
        k = max(1, int(n * r))        # size of the early block
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum <= 0.0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()
            if delta <= 0.0:
                continue
            g = 1.0 + delta / tail_sum
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                # refresh early / tail for further refinements
                early = best[:k]
                tail = best[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()

    # --------------------------------------------------
    # Directed stochastic refinement:
    #   Move a small amount of mass from an index that contributes most
    #   to the current worst convolution value to an index that contributes
    #   least. Accept only if the AC1 bound improves.
    # --------------------------------------------------
    steps = kwargs.get("num_steps", num_steps)
    for _ in range(steps):
        conv = np.convolve(best, best)
        k_max = int(conv.argmax())            # shift with maximal convolution
        # contributions of each index to this shift
        idx = np.arange(n, dtype=np.int64)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = best[idx[mask]] * best[partner[mask]]

        # source: one of the largest contributors
        src_candidates = np.where(contrib == contrib.max())[0]
        src = int(rng.choice(src_candidates))
        # destination: one of the smallest contributors (including zero)
        dst_candidates = np.where(contrib == contrib.min())[0]
        dst = int(rng.choice(dst_candidates))
        if src == dst or best[src] <= 0.0:
            continue

        # move a small fraction (up to 5 %) of the source entry
        delta = best[src] * rng.uniform(0.001, 0.05)
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------
    # Return a plain Python list as required by the harness.
    # --------------------------------------------------
    return [float(v) for v in best]
