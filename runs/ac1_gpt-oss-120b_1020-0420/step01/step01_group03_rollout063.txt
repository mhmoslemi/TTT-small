    """

import numpy as np

# ----------------------------------------------------------------------
# Helper: classic root‑constant (a_i = C(2i,i)/4**i) sequence.
# ----------------------------------------------------------------------
def _root_constant_seq(length: int) -> np.ndarray:
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq  # values already lie in [0,1] ⊂ [0,1000]


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step function whose AC1 bound is low.

    The construction starts from a feasible sequence f (the root‑constant
    sequence).  A secondary sequence g is built greedily so that the
    convolution f*g never exceeds the current maximal autocorrelation M.
    Afterwards we mix f and g with several mixing weights and keep the
    best candidate according to ``evaluate_sequence``.
    """
    # --------------------------------------------------------------
    # 1. Obtain length and base sequence f
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` is supplied by the harness when a parent
        # construction exists.  Use its length and initialise from the
        # classic root‑constant sequence – it is a good universal start.
        n = int(len(height_sequence_1))
        f = _root_constant_seq(n)
    except Exception:
        n = 256
        f = _root_constant_seq(n)

    # --------------------------------------------------------------
    # 2. Compute the current maximal convolution value M = max(f*f)
    # --------------------------------------------------------------
    conv_f = np.convolve(f, f)
    M = float(conv_f.max())

    # --------------------------------------------------------------
    # 3. Greedy construction of g  (maximise sum(g) s.t. f*g ≤ M)
    # --------------------------------------------------------------
    # Working convolution of f with the partially built g.
    c = np.zeros(2 * n - 1, dtype=np.float64)
    g = np.zeros(n, dtype=np.float64)

    # Small epsilon to avoid division by zero – f[i] is never zero for the
    # root‑constant sequence but we keep it safe for any possible parent.
    eps = 1e-20
    f_pos = f + eps  # ensures positivity for division

    for i in range(n - 1, -1, -1):
        # Slice of the convolution that will be affected by g[i]
        slice_c = c[i:i + n]                     # length n view
        # Allowed increase for each lag d (where d = 0..n-1)
        # bound[d] = (M - c[i+d]) / f[d]
        bounds = (M - slice_c) / f_pos
        # The feasible increase is the most restrictive bound.
        delta = float(bounds.min())
        if delta <= 0:
            delta = 0.0
        # Respect the evaluator's hard limit of 1000 per entry.
        if delta > 1000.0:
            delta = 1000.0
        g[i] = delta
        if delta != 0.0:
            # Update the convolution c with the new contribution.
            slice_c += f * delta

    # --------------------------------------------------------------
    # 4. Mix f and g with several mixing coefficients and keep the best.
    # --------------------------------------------------------------
    best_seq = f
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = np.inf

    # Mixing weights (including the original f with t=0).
    for t in (0.0, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3):
        cand = (1.0 - t) * f + t * g
        # Evaluate – ``evaluate_sequence`` clips values to [0,1000] internally.
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand

    # --------------------------------------------------------------
    # 5. Safety net – if something went wrong return a simple fallback.
    # --------------------------------------------------------------
    if not np.isfinite(best_score):
        # Uniform fallback (still respects the evaluator's constraints).
        best_seq = np.full(n, 1.0, dtype=np.float64)

    return best_seq.tolist()
