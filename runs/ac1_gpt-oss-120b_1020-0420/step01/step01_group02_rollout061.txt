import numpy as np

# ------------------------------------------------------------
# Half‑power (1‑z)^(-1/2) series – a compact baseline
#   a_0 = 1, a_i = a_{i-1} * (2*i-1)/(2*i)
# ------------------------------------------------------------
def half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Apply a simple “early‑block scaling”:
#   scale the first K entries by factor s (<1) and
#   redistribute the removed mass uniformly over the rest.
# ------------------------------------------------------------
def scale_early_block(vec: np.ndarray, K: int, s: float) -> np.ndarray:
    """Return a new array with the first K entries scaled by s."""
    n = vec.size
    if K <= 0 or K >= n:
        return vec.copy()
    out = vec.copy()
    donor_sum = float(np.sum(out[:K]))
    out[:K] *= s
    removed = donor_sum - float(np.sum(out[:K]))
    # distribute removed mass over the tail (indices K … n‑1)
    inc = removed / float(n - K)
    out[K:] += inc
    np.clip(out, 0.0, 1000.0, out=out)
    return out

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a list of non‑negative step‑function heights that aim
    to minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ----- determine problem length ---------------------------------
    n = 256
    if "height_sequence_1" in globals():
        hs = globals()["height_sequence_1"]
        if isinstance(hs, np.ndarray) and hs.ndim == 1 and hs.size > 0:
            n = int(hs.size)

    # ----- initialise with a deterministic baseline -----------------
    base = half_power_series(n)
    np.clip(base, 0.0, 1000.0, out=base)

    # start from the parent if it exists and looks reasonable
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    if "height_sequence_1" in globals():
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        np.clip(parent, 0.0, 1000.0, out=parent)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ----- deterministic early‑block scalings -----------------------
    # try a few different block sizes and scaling factors
    K_vals = [max(1, n // 20), max(2, n // 10), max(5, n // 5)]
    s_vals = [0.4, 0.5, 0.6, 0.7]
    for K in K_vals:
        for s in s_vals:
            cand = scale_early_block(best, K, s)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score

    # ----- simple stochastic hill‑climbing -------------------------
    # move a little mass from an early index to a late one.
    # accept only improvements (greedy descent).
    max_iters = 1500
    for _ in range(max_iters):
        # choose donor from the first 20 % and receiver from the last 20 %
        donor = int(rng.integers(0, max(1, n // 5)))
        receiver = int(rng.integers(4 * n // 5, n))
        if donor == receiver or best[donor] <= 0.0:
            continue
        # move up to 8 % of the donor's current value
        delta = float(best[donor] * rng.random() * 0.08)
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ----- final return ---------------------------------------------
    return [float(x) for x in best]
