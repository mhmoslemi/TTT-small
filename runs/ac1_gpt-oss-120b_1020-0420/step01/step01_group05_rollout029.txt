def fast_score(a):
    n = a.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:

import numpy as np

# ----------------------------------------------------------
# Fast, matrix‑free evaluation of the objective.
# ----------------------------------------------------------
def _fast_score(arr: np.ndarray) -> float:
    """Return 2 * n * max(conv(arr,arr)) / (sum(arr)**2) using FFT."""
    n = arr.shape[0]
    # Length of the full linear convolution
    m = 2 * n - 1
    # Next power‑of‑two for an efficient FFT
    nfft = 1 << ((m - 1).bit_length())
    # FFT of the array (zero‑padded)
    fa = np.fft.rfft(arr, n=nfft)
    # Point‑wise product = convolution in time domain
    conv_full = np.fft.irfft(fa * fa, n=nfft)[:m]
    M = float(np.max(conv_full))
    S = float(np.sum(arr))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

# Main entry point required by the harness.
# ----------------------------------------------------------

    """Return a non‑negative step‑function (list of floats) with a low AC1 bound.

    The construction is deliberately simple: a constant (uniform) sequence is
    optimal for the objective 2·n·max(conv)/sum² (it attains the theoretical
    lower bound 2).  If a parent sequence ``height_sequence_1`` is present,
    it is evaluated as a fallback candidate – the better of the two is returned.
    """
    # ------------------------------------------------------------------
    # Determine the length to use (parent length if available, else a safe default).
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
    except Exception:
        parent_arr = None
        n = 256  # default length when no parent is supplied

    # ------------------------------------------------------------------
    # Candidate 1: uniform sequence (theoretical optimum for the given score).
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0, dtype=np.float64)            # any constant >0 works
    uniform = np.clip(uniform, 0.0, 1000.0)                # respect evaluator bounds
    best_seq = uniform
    best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # Candidate 2: use the supplied parent if it is valid and better.
    # ------------------------------------------------------------------
    if parent_arr is not None:

        if np.sum(parent) >= 0.01:                         # must satisfy evaluator's sum check
            parent_score = _fast_score(parent)
            if parent_score < best_score:

                best_score = parent_score

    # Return a plain Python list as required by the evaluator.
    # ------------------------------------------------------------------
