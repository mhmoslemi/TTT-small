def _coefficients_binom_half(n):
    """First n coefficients of (1‑z)^‑1/2 (C_i = comb(2i,i)/4^i)."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _eval_seq(seq):
    """Wrapper for the provided evaluator – expects a NumPy array."""
    return float(evaluate_sequence(seq.tolist()))

    """Return a non‑negative step‑function (list of floats) with a low AC1 bound.

    The routine starts from either the supplied parent `height_sequence_1`
    (if present) or from the analytic (1‑z)^‑1/2 coefficients.
    A lightweight simulated‑annealing / hill‑climbing loop then tries to
    flatten the autocorrelation peak while keeping the total sum large.
    """
    rng = np.random.default_rng(seed)

    # ---- Determine problem size and base candidates -----------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.shape[0]
    else:
        parent = None
        n = 256

    analytic = np.clip(_coefficients_binom_half(n), 0.0, 1000.0)

    best = None

            sc = _eval_seq(cand)

            continue
        if np.isfinite(sc) and sc < best_score:

            best_score = sc
