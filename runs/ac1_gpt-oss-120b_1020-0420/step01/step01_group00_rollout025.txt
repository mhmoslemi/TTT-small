```
import numpy as np

def _fast_score(a):
    n = a.size
    if n == 0: return np.inf
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:L]
    return float(2.0 * n * conv.max())
```

Note: since sum normalized to 1, we don't need sum; but to be safe use actual sum? But current scoring defined: 2n*max_c/(sum^2). If we set sum to 1, we can compute simply 2n * max_c. However we will keep general for safety: compute sum_a = a.sum(), return 2*n*conv.max()/(sum_a**2). Use that.

def _fast_score(vec: np.ndarray) -> float:

    n = vec.size

    if s < 1e-12:
        return np.inf

    m = 1 << ((L - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:L]
    max_c = conv.max()

def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector."""
    n = vec.size
    if n == 0:
        return np.inf
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())          # next power of two
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:L]
    max_c = conv.max()
    return float(2.0 * n * max_c / (s * s))
