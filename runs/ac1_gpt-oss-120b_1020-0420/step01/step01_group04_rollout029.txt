import numpy as np

def _coefficients_binom_half(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^‑1/2: C_i = comb(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound.

    The routine starts from the supplied parent (if any) and from the analytic
    half‑binomial coefficients.  It then performs a modest stochastic
    optimisation using three move types:
      * a gradient‑guided mass transfer between indices that contribute most
        and least to the current convolution peak,
      * a random log‑normal scaling of a contiguous block,
      * a tiny bump on a tail block or a simple element swap.
    The best sequence encountered is returned.
    """
    rng = np.random.default_rng(seed)

    # ---- determine problem size and initial vectors -----------------
    if "height_sequence_1" in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = base.shape[0]
    else:
        n = 256
        base = np.clip(_coefficients_binom_half(n), 0.0, 1000.0)

    # another analytic start – useful if the parent is poor
    alt = np.clip(_coefficients_binom_half(n), 0.0, 1000.0)

    # evaluate both starters
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    alt_score = float(evaluate_sequence(alt.tolist()))
    if alt_score < best_score:
        best = alt.copy()
        best_score = alt_score

    max_steps = 8000
    target = 1.50274349

    for step in range(max_steps):
        if best_score <= target:
            break

        cand = best.copy()
        move = rng.integers(0, 4)          # 0‑grad,1‑block,2‑tail bump,3‑swap

        if move == 0:
            # ---- gradient‑guided transfer -----------------
            conv = np.convolve(cand, cand)          # full convolution
            peak = int(np.argmax(conv))

            i = np.arange(n)
            partner = peak - i
            mask = (partner >= 0) & (partner < n)

            grad = np.zeros(n, dtype=np.float64)
            vi = i[mask]
            pi = partner[mask]
            grad[vi] = cand[vi] + cand[pi]
            same = vi == pi
            if same.any():
                grad[vi[same]] = 2.0 * cand[vi[same]]

            # donor = large‑gradient & positive entry
            donors = np.where(cand > 0)[0]
            if donors.size == 0:
                continue
            top_k = max(1, donors.size // 20)
            dvals = grad[donors]
            donor_candidates = donors[np.argpartition(dvals, -top_k)[-top_k:]]
            donor = int(rng.choice(donor_candidates))

            # receiver = small‑gradient index
            bot_k = max(1, n // 20)
            receiver_candidates = np.argpartition(grad, bot_k)[:bot_k]
            receiver = int(rng.choice(receiver_candidates))

            if donor != receiver:
                max_transfer = cand[donor] * 0.5
                if max_transfer > 0.0:
                    frac = (max_steps - step) / max_steps
                    delta = float(rng.random() * max_transfer * frac)
                    cand[donor] -= delta
                    cand[receiver] += delta

        elif move == 1:
            # ---- random block log‑normal scaling ----------
            min_len = max(1, n // 100)
            max_len = max(2, n // 4)
            width = int(rng.integers(min_len, max_len + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif move == 2:
            # ---- tiny bump on a tail piece ---------------
            start = int(rng.integers(int(n * 0.75), n))
            width = int(rng.integers(1, max(2, n - start)))
            bump = max(0.0, rng.normal(0.0, 0.07))
            cand[start:start + width] += bump
            np.clip(cand, 0.0, 1000.0, out=cand)

        else:
            # ---- simple swap -----------------------------
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # ---- evaluate ------------------------------------------------
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Ensure a valid (non‑empty) return
    if best.size == 0:
        return [1.0]
    return [float(v) for v in best]
