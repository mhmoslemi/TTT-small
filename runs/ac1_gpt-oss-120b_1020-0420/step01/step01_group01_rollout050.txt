import numpy as np

def _catalan_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0]=1.0
    for i in range(n-1):
        a[i+1] = a[i] * ((2*i+2)*(2*i+1)) / (4.0*(i+1)*(i+1))
    return a

def _max_conv(seq):
    n=seq.size
    m = 2*n - 1
    size = 1 << ((m - 1).bit_length())
    fft_seq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_seq * fft_seq, size)[:m]
    return conv.max()
    
def _score_seq(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    m = _max_conv(seq)
    return float(2.0 * seq.size * m / (s * s))

import numpy as np

# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a


# ------------------------------------------------------------
# Fast max of the full (linear) convolution via FFT.
# ------------------------------------------------------------
def _max_convolution(seq: np.ndarray) -> float:
    n = seq.size
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_seq * fft_seq, size)[:m]
    return float(conv.max())


# ------------------------------------------------------------
# Score identical to the evaluator (except clipping already done)
# ------------------------------------------------------------
def _score_seq(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    m = _max_convolution(seq)
    return float(2.0 * seq.size * m / (s * s))
