import numpy as np

# ------------------------------------------------------------
# Score function – matches the evaluator used by the harness.
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv)/sum(seq)^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # linear convolution, O(n log n)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence.
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a

            best = cand
            best_score = sc

    # --------------------------------------------------------
    # Final deterministic pass (same family as the first) – helps
    # escape any small local plateau left after stochastic phase.
    # --------------------------------------------------------
    for r in prefix_ratios:
        k = max(1, int(n * r))
        early = best[:k]
        tail = best[k:]
        early_sum = early.sum()
        tail_sum = tail.sum()
        if tail_sum == 0.0:
            continue
        for f in shrink_factors:
            new_early = early * f
            delta = early_sum - new_early.sum()
            g = 1.0 + delta / tail_sum
            new_tail = tail * g
            cand = np.concatenate([new_early, new_tail])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                early = best[:k]
                tail = best[k:]
                early_sum = early.sum()
                tail_sum = tail.sum()

    # --------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # --------------------------------------------------------
    return [float(v) for v in best]
```
