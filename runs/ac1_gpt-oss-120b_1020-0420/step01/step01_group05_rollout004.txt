def _conv_full(a):
    n = a.shape[0]
    m = 2*n-1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa*fa, n=nfft)[:m]
    return c

def _fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if n == 0:
        return float('inf')
    conv = _conv_full(a)
    M = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))

# Fast convolution (full) via FFT – matrix‑free.

def _conv_full(a):

    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    return np.fft.irfft(fa * fa, n=nfft)[:m]
