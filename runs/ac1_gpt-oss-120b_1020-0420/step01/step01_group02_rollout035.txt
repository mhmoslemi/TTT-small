def symmetric_early_tail_redistribute(base, rng):
   n = len(base)
   max_width = max(1, n // 50)   # early block width
   w = int(rng.integers(1, max_width + 1))
   factor = float(rng.uniform(0.2, 0.8))  # reduction factor
   cand = base.copy()
   # compute original sum for early and tail
   early_old = cand[:w].sum()
   tail_old = cand[-w:].sum()
   cand[:w] *= factor
   cand[-w:] *= factor
   early_new = cand[:w].sum()
   tail_new = cand[-w:].sum()
   delta = (early_old - early_new) + (tail_old - tail_new)  # sum reduction
   # distribute delta uniformly across middle part
   mid_len = n - 2*w
   if mid_len > 0:
       cand[w:n-w] += delta / mid_len
   np.clip(cand, 0.0, 1000.0, out=cand)
   return cand

import numpy as np

# ------------------------------------------------------------
# Generate the (1 - z)^(-1/2) half‑power series.
# a_0 = 1, a_i = a_{i-1} * (2*i - 1) / (2*i)
# ------------------------------------------------------------
def generate_half_power_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ------------------------------------------------------------
# Simple random block perturbation (multiplicative log‑normal).
# ------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.15)))  # slightly larger spread
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Symmetric reduction of the first and last blocks and redistribution
# of the removed mass uniformly over the middle part.
# This tends to lower the peak of the self‑convolution while keeping the
# total L1‑mass almost unchanged, which improves the AC1 bound.
# ------------------------------------------------------------
def symmetric_early_tail_redistribute(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    # size of the edge blocks (1 … n//50)
    max_width = max(1, n // 50)
    w = int(rng.integers(1, max_width + 1))
    # reduction factor (<1) – we shrink both edge blocks
    factor = float(rng.uniform(0.2, 0.8))
    cand = base.copy()

    # sums before modification
    early_sum_before = cand[:w].sum()
    tail_sum_before = cand[-w:].sum()

    # shrink both edges
    cand[:w] *= factor
    cand[-w:] *= factor

    # sums after modification
    early_sum_after = cand[:w].sum()
    tail_sum_after = cand[-w:].sum()

    # total removed mass
    delta = (early_sum_before - early_sum_after) + (tail_sum_before - tail_sum_after)

    # distribute uniformly over the middle part
    middle_len = n - 2 * w
    if middle_len > 0:
        cand[w:n - w] += delta / middle_len

    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# ------------------------------------------------------------

    """
    Return a list of non‑negative step‑function heights that aims to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # Determine the problem size (use the provided parent if present)
    n = 256
    if (
        "height_sequence_1" in globals()

    # Baseline: exact half‑power series (constant self‑convolution = 1)
    base = generate_half_power_series(n)
    np.clip(base, 0.0, 1000.0, out=base)

    # Keep the best candidate seen so far
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # First deterministic attempt with symmetric reduction

    if np.isfinite(score0) and score0 < best_score:
