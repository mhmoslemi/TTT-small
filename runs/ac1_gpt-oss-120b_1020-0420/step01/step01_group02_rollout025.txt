import numpy as np

def random_block_perturbation(base, rng):
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand = base.copy()
    cand[start:start+width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def power_series(length, p):
    # a_i = (i+1)^{-p}
    i = np.arange(1, length+1, dtype=np.float64)

    max_b = b.max()
    sum_a = a.sum()

import numpy as np

# -------------------------------------------------------------
# Helper: a simple power‑law step function a_i = (i+1)^{-p}
# -------------------------------------------------------------
def power_series(length: int, p: float) -> np.ndarray:
    """Return a non‑negative vector of size ``length`` with entries (i+1)^{-p}."""
    i = np.arange(1, length + 1, dtype=np.float64)
    return np.power(i, -p)


# -------------------------------------------------------------
# Small random block perturbation (keeps O(n) memory)
# -------------------------------------------------------------
def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block of ``base`` by a log‑normal factor."""
    n = base.size
    if n == 1:
        return base.copy()
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))   # mild perturbation
    cand = base.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# -------------------------------------------------------------
# Compute the AC1 bound directly (avoids the expensive evaluator)
# -------------------------------------------------------------
def compute_ac1_score(a: np.ndarray, n: int, fft_len: int):
    """Return the AC1 bound and the full convolution of a with itself."""
    A = np.fft.rfft(a, fft_len)
    b = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf, b
    score = 2.0 * n * max_b / (sum_a * sum_a)
    return float(score), b

# -------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that aims
    to minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # Determine problem size – use the provided parent if present.
    # ---------------------------------------------------------
    if "height_sequence_1" in globals():
        arr = height_sequence_1
        if isinstance(arr, np.ndarray) and arr.ndim == 1 and arr.size > 0:
            n = int(arr.size)
            base_seq = arr.astype(np.float64)
        else:
            n = 256
            base_seq = power_series(n, 0.5)
    else:
        n = 256
        base_seq = power_series(n, 0.5)

    # ---------------------------------------------------------
    # Scan a few power‑law exponents and keep the best shape.
    # ---------------------------------------------------------
    best = None
    best_score = np.inf
    fft_len = 1 << ((2 * n - 1).bit_length())

    for p in np.arange(0.55, 0.81, 0.03):
        a = power_series(n, float(p))
        score, _ = compute_ac1_score(a, n, fft_len)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()

    # Fallback – half‑power series if the scan failed.
    if best is None:
        best = base_seq.copy()
        best_score, _ = compute_ac1_score(best, n, fft_len)

    # ---------------------------------------------------------
    # Local refinement: exponential‑gradient steps + occasional jitter.
    # ---------------------------------------------------------
    a = best.copy()
    target_sum = a.sum()
    steps = ([0.2] * 5 + [0.1] * 10 + [0.05] * 30 +
             [0.02] * 60 + [0.01] * 120)          # decreasing step sizes
    no_improve = 0

    for step in steps:
        # current convolution and its maximal index
        _, b = compute_ac1_score(a, n, fft_len)
        k_max = int(np.argmax(b))

        # gradient‑like multiplier: rev[i] = a[k_max-i] where defined
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        rev = np.zeros_like(a)
        rev[mask] = a[j[mask]]

        # exponential‑gradient candidate
        cand = a * np.exp(-step * rev)
        s = cand.sum()
        if s > 0:
            cand *= target_sum / s               # keep L1 norm (scaling does not affect the bound)
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score, _ = compute_ac1_score(cand, n, fft_len)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand.copy()
            no_improve = 0
            continue

        cand2 = random_block_perturbation(a, rng)
        s2 = cand2.sum()
        if s2 > 0:
            cand2 *= target_sum / s2
        np.clip(cand2, 0.0, 1000.0, out=cand2)

        cand2_score, _ = compute_ac1_score(cand2, n, fft_len)
        if np.isfinite(cand2_score) and cand2_score < best_score:

            no_improve = 0

    # ---------------------------------------------------------
    # Return plain Python list as required by the harness.
    # ---------------------------------------------------------
    return [float(x) for x in best]
