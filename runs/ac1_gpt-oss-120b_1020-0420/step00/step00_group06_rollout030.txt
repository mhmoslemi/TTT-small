import numpy as np

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # get parent sequence if exists
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size
    # compute fft length
    fft_len = 1 << ((2 * n - 1).bit_length())

    def compute_score(arr):
        sum_a = arr.sum()
        if sum_a < 0.01:
            return np.inf
        # FFT-based convolution
        f = np.fft.rfft(arr, n=fft_len)
        c = np.fft.irfft(f * f, n=fft_len)[:2 * n - 1]
        max_b = c.max()
        return 2.0 * n * max_b / (sum_a * sum_a)

    # initial best = parent
    best = parent.copy()
    best_score = compute_score(best)

    # function to generate sparse random candidate
    def random_sparse():
        # target number of nonzeros roughly sqrt(2n) to have chance <2
        target = int(np.sqrt(2 * n)) + rng.integers(-5, 6)
        target = max(10, min(n, target))
        positions = rng.choice(n, size=target, replace=False)
        vec = np.zeros(n, dtype=np.float64)
        # assign height 1.0 (any scale)
        vec[positions] = 1.0
        return vec

    # Evaluate a few random sparse seeds as possible starting points
    for _ in range(20):
        cand = random_sparse()
        sc = compute_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # main local search
    for step in range(num_steps):
        # compute convolution of current best to find peak
        f = np.fft.rfft(best, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[:2 * n - 1]

def propose_candidate(seed=42, num_steps=1500, **kwargs):
    rng = np.random.default_rng(seed)
    n = height_sequence_1.size if defined else 256
    # Start with random positive vector
    a = rng.random(n) + 0.01  # ensure positivity
    a = np.clip(a, 0.0, 1000.0)
    # Normalize sum to 1 for scale invariance
    a = a / a.sum()
    # Keep reference sum for scaling (1)
    best = a.copy()
    best_score = compute_score(best)

    # Precompute FFT length
    fft_len = 1 << ((2*n - 1).bit_length())

    for step in range(num_steps):
        # compute convolution via FFT
        f = np.fft.rfft(best, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[:2*n-1]
        k = int(np.argmax(conv))

        # compute gradient g
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(best)
        g[valid] = best[partner[valid]]

        # step size
        alpha = 0.1 / (np.max(g) + 1e-12)  # ensure small changes
        cand = best - alpha * g
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Renormalize to sum 1
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        sc = compute_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

        # Occasionally random perturb (to escape)
        if step % 100 == 0:
            # random block perturb similar to earlier
            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))
            stop = start + width
            pert = best.copy()
            pert[start:stop] *= np.exp(rng.normal(0.0, 0.1))
            np.clip(pert, 0.0, 1000.0, out=pert)
            pert = pert / pert.sum()
            scp = compute_score(pert)
            if scp < best_score:
                best = pert
                best_score = scp

import numpy as np

# ------------------------------------------------------------
# Helper: fast score (the same objective that evaluate_sequence uses)
# ------------------------------------------------------------
def _score(vec, fft_len):
    """Return 2*n*max(conv)/sum(vec)**2 for a non‑negative vector."""
    s = vec.sum()
    if s < 0.01:
        return np.inf
    # convolution via FFT (real‑to‑real convolution)
    f = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[:2 * vec.size - 1]
    return 2.0 * vec.size * conv.max() / (s * s)


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed=42, num_steps=4000, **kwargs):
    """
    Return a list of non‑negative heights.
    The routine starts from a few random initialisations and performs
    a cheap stochastic hill‑climb that tries to lower the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Base sequence (fallback to a simple uniform vector if the parent is
    # missing or malformed).  The parent is available as `height_sequence_1`.
    # ----------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    n = base.size

    # FFT length for all convolutions (next power of two >= 2*n-1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------------
    # Initialise the current best candidate.
    # ----------------------------------------------------------------
    best = base.copy()
    best_score = _score(best, fft_len)

    # ----------------------------------------------------------------
    # Helper to generate a random sparse start (Dirichlet with small alpha).
    # ----------------------------------------------------------------
    def _random_start():
        # Gamma shape <1 yields a sparse distribution; we normalise to sum=1.
        vec = rng.gamma(0.5, 1.0, size=n)
        vec = np.clip(vec, 0.0, None)
        if vec.sum() == 0.0:
            vec = np.ones(n, dtype=np.float64)
        return vec / vec.sum()

    # ----------------------------------------------------------------
    # Main optimisation loop.  We perform a few independent restarts;
    # each restart carries out a simple mass‑transfer hill‑climb.
    # ----------------------------------------------------------------
    restarts = 3
    steps_per_restart = max(1, num_steps // restarts)

    for _ in range(restarts):
        cur = _random_start()
        cur_score = _score(cur, fft_len)

        # Keep the globally best solution seen so far.
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score

        for _ in range(steps_per_restart):
            # --------------------------------------------------------
            # Choose a source index weighted by its current mass.
            # --------------------------------------------------------
            probs = cur / cur.sum()
            src = int(rng.choice(n, p=probs))

            # --------------------------------------------------------
            # Choose a different destination index uniformly.
            # --------------------------------------------------------
            dst = int(rng.integers(0, n - 1))
            while dst == src:
                dst = int(rng.integers(0, n - 1))

            # --------------------------------------------------------
            # Transfer a random fraction (up to 50%) of the source mass.
            # --------------------------------------------------------
            delta = cur[src] * rng.random() * 0.5
            if delta <= 0.0:
                continue

            cand = cur.copy()
            cand[src] -= delta
            cand[dst] += delta

            # Numerical safety (non‑negativity and clipping to the allowed range)
            np.clip(cand, 0.0, 1000.0, out=cand)

            cand_score = _score(cand, fft_len)

            # Accept improving moves; also occasionally accept a worse move
            # to escape shallow local minima (simulated‑annealing style).
            if cand_score < cur_score or rng.random() < 0.001:
                cur, cur_score = cand, cand_score
                if cand_score < best_score:
                    best, best_score = cand.copy(), cand_score

        # End of one restart ------------------------------------------------

    # ------------------------------------------------------------
    # Final safety: ensure the vector respects the evaluator's limits.
    # ------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:                     # never happen, but guard anyway
        best[0] = max(best[0], 0.01)

    # Return as plain Python list of floats as required by the harness.
    return [float(x) for x in best]
