def propose_candidate(seed=42, num_steps=1500, **kwargs):
    rng = np.random.default_rng(seed)
    # get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    # keep best
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size
    # candidate0 = best (like deterministic interleaved sorted)
    # Try deterministic initial improvements:
    sorted_vals = np.sort(best)
    # interleaving small large
    interleaved = np.empty_like(best)
    left = 0
    right = n-1
    toggle = True
    idx = 0
    while idx < n:
        if toggle:
            interleaved[idx] = sorted_vals[right]
            right -= 1
        else:
            interleaved[idx] = sorted_vals[left]
            left += 1
        toggle = not toggle
        idx += 1
    # Evaluate interleaved
    inter_score = float(evaluate_sequence(interleaved.tolist()))
    if np.isfinite(inter_score) and inter_score < best_score:
        best = interleaved
        best_score = inter_score

    # Now iterative swap guided by convolution peak.
    for step in range(num_steps):
        # compute convolution of current best
        conv = np.convolve(best, best)
        # find index of max
        k_max = int(np.argmax(conv))
        left_idx = max(0, k_max - (n - 1))
        right_idx = min(n - 1, k_max)
        # pick i inside region
        i = int(rng.integers(left_idx, right_idx + 1))
        # pick j outside region or anywhere else
        # To avoid always picking same region, we pick j randomly not equal i
        j = int(rng.integers(0, n))
        while j == i:
            j = int(rng.integers(0, n))
        candidate = best.copy()
        candidate[i], candidate[j] = candidate[j], candidate[i]
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            # possibly continue; also can adjust region.
        # Occasionally try random shuffle candidate to escape local minima
        if step % 100 == 0 and step > 0:
            # generate random shuffle
            cand2 = best.copy()
            rng.shuffle(cand2)
            s2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(s2) and s2 < best_score:
                best = cand2
                best_score = s2

import numpy as np

# Helper: create a deterministic interleaved ordering of the values.
def interleaved_sorted(seq):
    """Return a copy of *seq* with values sorted and interleaved (large, small, …)."""
    vals = np.sort(seq)                     # ascending
    n = vals.size
    out = np.empty_like(vals)
    left, right = 0, n - 1
    idx = 0
    take_large = True
    while left <= right:
        if take_large:
            out[idx] = vals[right]
            right -= 1
        else:
            out[idx] = vals[left]
            left += 1
        take_large = not take_large
        idx += 1
    return out


def propose_candidate(seed=42, num_steps=800, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    if (parent.ndim != 1 or parent.size == 0
            or not np.isfinite(parent).all()):
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # ------------------------------------------------------------------
    # Initial best candidate and its score.
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Deterministic heuristic: interleaved sorted values.
    # ------------------------------------------------------------------
    cand = interleaved_sorted(best)
    cand_score = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(cand_score) and cand_score < best_score:
        best = cand

    # ------------------------------------------------------------------
    # Deterministic heuristic: simple reversal.
    # ------------------------------------------------------------------
    rev = best[::-1]
    rev_score = float(evaluate_sequence(rev.tolist()))
    if np.isfinite(rev_score) and rev_score < best_score:
        best = rev
        best_score = rev_score

    # ------------------------------------------------------------------
    # Main iterative improvement loop.
    #   * At each step we locate the convolution peak.
    #   * Swap the element that contributes most to this peak with a
    #     low‑valued element elsewhere.
    #   * Occasionally try a random shuffle and a small multiplicative
    #     perturbation to escape local minima.
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # Convolution of the current best candidate.
        conv = np.convolve(best, best)          # length 2n-1
        k_max = int(np.argmax(conv))            # index of the maximal entry

        # Indices that overlap for this shift.
        left_idx = max(0, k_max - (n - 1))
        right_idx = min(n - 1, k_max)

        # Compute per‑index contributions a[i] * a[k_max-i].
        i_range = np.arange(left_idx, right_idx + 1)
        partner_idx = k_max - i_range
        contributions = best[i_range] * best[partner_idx]

        # Index that gives the largest contribution to the peak.
        i_max = int(i_range[np.argmax(contributions)])

        # Choose a low‑valued index to swap with.
        median_val = np.median(best)
        low_idxs = np.where(best <= median_val)[0]

        if low_idxs.size > 1:

                while j == i_max:
                    j = int(rng.integers(0, n))
            else:
                j = int(rng.choice(low_candidates, size=1))
        else:
            j = int(rng.integers(0, n))
            while j == i_max:
                j = int(rng.integers(0, n))

        candidate[i_max], candidate[j] = candidate[j], candidate[i_max]

                best_score = s_pert

    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
