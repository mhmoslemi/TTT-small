def propose_candidate(seed=42, num_steps=5000, **kwargs):
    rng = np.random.default_rng(seed)
    n = height_sequence_1.shape[0] if 'height_sequence_1' in globals() else 256
    # base = compute base coefficients
    coeffs = compute_coeffs(n)
    # Anti-pair ordering
    coeffs_desc = np.sort(coeffs)[::-1]  # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    for i in range(half):
        a[i] = coeffs_desc[i]
        a[n-1-i] = coeffs_desc[-(i+1)]
    if n % 2 == 1:
        a[half] = coeffs_desc[half]
    a = np.clip(a, 0.0, 1000.0)
    best = a
    best_score = float(evaluate_sequence(best.tolist()))
    # fallback parent
    fallback = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    fallback_score = float(evaluate_sequence(fallback.tolist()))
    if fallback_score < best_score:
        best = fallback
        best_score = fallback_score
    # Search loop
    for step in range(num_steps):
        candidate = best.copy()
        # pick operation
        op = rng.integers(0, 3)
        if op == 0:
            # swap two indices
            i, j = rng.integers(0, n, size=2)
            candidate[i], candidate[j] = candidate[j], candidate[i]
        elif op == 1:
            # block scaling
            width = max(1, n // rng.integers(20, 50))  # width between n/50 and n/20
            start = rng.integers(0, n - width + 1)
            factor = float(np.exp(rng.normal(0.0, 0.05)))
            candidate[start:start+width] *= factor
        else:
            # local noise perturb
            idx = rng.integers(0, n)
            delta = float(rng.normal(0.0, 0.02))
            candidate[idx] = max(0.0, candidate[idx] + delta)
        # clip bounds
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    # Return best as list
    return [float(x) for x in best]
