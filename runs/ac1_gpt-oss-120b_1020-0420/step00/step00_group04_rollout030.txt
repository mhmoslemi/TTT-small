import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate a candidate sequence (list or 1‑D array) and return the
# scalar score used by the harness.
# ----------------------------------------------------------------------
def _score(seq):
    """Return the AC1 upper‑bound for the given sequence."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Shape generators – they return a NumPy array of length n with non‑negative
# entries (already clipped to the allowed range).
# ----------------------------------------------------------------------
def _exp_shape(y, n):
    """Exponential decay a_i = exp(- (y/n) * i).  y>0."""
    lam = y / float(n)
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    return np.clip(a, 0.0, 1000.0)


def _power_shape(p, n):
    """Power‑law decay a_i = (i+1)^(-p).  p>=0."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return np.clip(a, 0.0, 1000.0)


def _linear_shape(s, n):
    """Linear decay a_i = max(0, 1 - s * i/(n-1)).  s>=0."""
    if n == 1:
        a = np.ones(1, dtype=np.float64)
    else:
        i = np.arange(n, dtype=np.float64)
        a = 1.0 - s * (i / float(n - 1))
        a = np.maximum(a, 0.0)
    return np.clip(a, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Simple block perturbation (same idea as the reference program).
# ----------------------------------------------------------------------
def _make_block_perturbation(base, rng):
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improve
    on the default bound.  The search explores a few parametric families
    (exponential, power‑law, linear) and then performs a short
    block‑perturbation hill‑climb.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the base length.  If the runtime supplies `height_sequence_1`
    # use it; otherwise fall back to a modest uniform vector.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)

    # Ensure a valid start vector (non‑negative, clipped to limits)
    best = np.clip(parent, 0.0, 1000.0)
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 1) Exhaustive sweep over a few exponential shapes (golden‑section‑like).
    # ------------------------------------------------------------------
    # Search interval for the scaled decay parameter y = λ·n
    y_low, y_high = 0.2, 5.0
    phi = (np.sqrt(5) - 1) / 2          # inverse golden ratio
    for _ in range(20):                 # ~40 evaluations
        c = y_high - phi * (y_high - y_low)
        d = y_low + phi * (y_high - y_low)
        cand_c = _exp_shape(c, n)
        cand_d = _exp_shape(d, n)
        score_c = _score(cand_c)
        score_d = _score(cand_d)
        if score_c < best_score:
            best, best_score = cand_c, score_c
        if score_d < best_score:
            best, best_score = cand_d, score_d
        if score_c < score_d:
            y_high = d
        else:
            y_low = c

    # ------------------------------------------------------------------
    # 2) Simple grid over power‑law exponents.
    # ------------------------------------------------------------------
    for p in np.linspace(0.0, 2.0, 15):
        cand = _power_shape(p, n)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 3) Simple grid over linear‑decay slopes.
    # ------------------------------------------------------------------
    for s in np.linspace(0.0, 3.0, 12):
        cand = _linear_shape(s, n)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # 4) Short stochastic hill‑climb based on block perturbations.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = _make_block_perturbation(best, rng)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # Return a plain Python list as required by the harness.
    return [float(v) for v in best]
