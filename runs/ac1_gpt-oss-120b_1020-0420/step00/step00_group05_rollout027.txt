import numpy as np

def compute_score(seq):
    return float(evaluate_sequence(seq.tolist()))

def make_block_perturbation(base, rng, sigma=0.08):
    cand = base.copy()
    n = cand.size
    # Choose block width between n//200 and n//12
    lo = max(1, n // 200)
    hi = max(lo+1, n // 12)
    width = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, sigma))
    cand[start:stop] *= factor
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length n
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = parent_arr.size
        parent = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)
    best = parent.copy()
    best_score = compute_score(best)
    # Uniform fallback
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform_score = compute_score(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score
    # Local search
    for _ in range(num_steps):
        cand = make_block_perturbation(best, rng)
        score = compute_score(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    # Ensure we produce a finite, non-empty list
    if not np.isfinite(best_score):
        # Fallback to uniform
        best = uniform
    return [float(x) for x in best]

import numpy as np

def _score(seq: np.ndarray) -> float:
    """Exact AC1 score for a NumPy 1‑D array."""
    # evaluate_sequence expects a Python list
    return float(evaluate_sequence(seq.tolist()))


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator,
                             sigma: float = 0.08) -> np.ndarray:
    """Return a single O(n) perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width between n//200 and n//12 (inclusive)
    lo = max(1, n // 200)
    hi = max(lo + 1, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    factor = np.exp(rng.normal(0.0, sigma))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps remove bias)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    # clip to allowed range
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, num_steps: int = 800, **kwargs) -> list:
    """
    Return a non‑empty list of non‑negative heights.
    Uses a simple block‑perturbation hill‑climb starting from the provided
    parent (``height_sequence_1``) or, if unavailable, from a uniform vector.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem dimension and initialise a parent sequence
    # ----------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = parent_arr.size
        parent = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    best = parent.copy()
    best_score = _score(best)

    # ----------------------------------------------------------------------
    # Uniform fallback – it attains the theoretical lower bound 2.0
    # ----------------------------------------------------------------------
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # ----------------------------------------------------------------------
    # Simple local search: repeatedly perturb the current best
    # ----------------------------------------------------------------------
    for _ in range(num_steps):
        cand = _make_block_perturbation(best, rng)
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ----------------------------------------------------------------------
    # Defensive fallback – ensure we always return a valid finite candidate
    # ----------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = uniform

    # Convert to plain Python floats for the harness
    return [float(v) for v in best]
