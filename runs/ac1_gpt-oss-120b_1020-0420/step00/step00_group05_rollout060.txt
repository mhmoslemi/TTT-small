import numpy as np

def propose_candidate(seed=42, num_trials=10, num_steps=100, **kwargs):
    """
    Generate a non‑negative step‑function that aims to minimise the AC1 upper bound.
    The routine starts from a few random dense candidates and refines the best
    one with cheap block perturbations. All operations use O(n) auxiliary memory.
    """
    rng = np.random.default_rng(seed)

    # Length of the sequence – normally supplied as height_sequence_1.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:                     # fallback if the variable is missing
        n = 256

    # Parent fallback (never used unless all random tries fail)
    parent = (
        np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if "height_sequence_1" in globals()
        else np.full(n, 1000.0 / n)
    )

    best = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Random dense initialisation – we try a handful of independent draws.
    # ------------------------------------------------------------------
    for _ in range(num_trials):
        cand = rng.random(n) * 1000.0                      # uniform in [0,1000)
        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best = cand

    # If, for some unexpected reason, none of the random candidates are
    # acceptable, initialise from the parent sequence.
    if best is None or not np.isfinite(best_score):
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 2) Light local search – block perturbations on the current best.
    #    Each step modifies a contiguous block by a log‑normal factor,
    #    optionally mirrors the whole array, and keeps the best result.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        candidate = best.copy()

        # Choose a random block width (at least one element, at most n/12)
        min_w = max(1, n // 200)
        max_w = max(2, n // 12)
        width = int(rng.integers(min_w, max_w))

        # Random start index so that the block fits completely.
        start = int(rng.integers(0, n - width + 1))
        stop = start + width

        # Log‑normal scaling of the selected block.
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        candidate[start:stop] *= factor

        # With probability 0.35 symmetrise the sequence (helps reduce peaks).
        if rng.random() < 0.35:
            candidate = 0.5 * (candidate + candidate[::-1])

        # Enforce the evaluator’s hard bounds in‑place.
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # Evaluate the perturbed candidate.
        score = evaluate_sequence(candidate.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best = candidate

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
