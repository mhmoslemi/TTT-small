import numpy as np

def propose_candidate(seed=42, num_iters=500, **kwargs):
    """
    Generate a non‑negative step‑function height sequence that strives to
    minimise the AC1 upper bound used by `evaluate_sequence`.
    The method combines a small parametric search (power‑law shapes) with a
    lightweight gradient‑descent style refinement that only touches O(n)
    memory per iteration.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (use the provided parent if available)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 256                     # a harmless fallback length
        base = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1️⃣  Coarse parametric search: a_i = 1 / (i + 1 + offset) ** exp
    #    (exp≈0.5 gives a near optimal “flat” autocorrelation)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    best_vec = None
    best_score = np.inf

    exponents = [0.4, 0.45, 0.5, 0.55, 0.6]               # around the theoretical optimum
    max_offset = max(1, n // 5)                           # up to ~20 % of the length
    offsets = np.linspace(0.0, max_offset, num=21)        # evenly spaced offsets

    for exp in exponents:
        for off in offsets:
            # power‑law candidate
            a = 1.0 / np.power(idx + 1.0 + off, exp)

            # avoid the degenerate all‑zero case
            if np.all(a == 0):
                continue

            # ensure the sum is not too tiny (scaling does not affect the score)
            s = a.sum()
            if s < 0.01:
                a *= (0.01 / s)

            # evaluate
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_vec = a.copy()

    # ------------------------------------------------------------------
    # If the parametric sweep failed for any reason, fall back to a simple
    # uniform vector (the previous baseline).
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2️⃣  Local refinement: minimise the maximal convolution entry.
    #     The objective is invariant under scaling, so we keep the total
    #     sum constant (sum_target) to stay away from the “sum < 0.01”
    #     guard.
    # ------------------------------------------------------------------
    candidate = best_vec.copy()
    sum_target = max(candidate.sum(), 1e-12)   # protect against division by zero

    for _ in range(num_iters):
        conv = np.convolve(candidate, candidate)
        max_idx = int(conv.argmax())

        # gradient of the maximal convolution entry w.r.t. each a_i:
        i = np.arange(n, dtype=np.int64)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(candidate)
        grad[mask] = 2.0 * candidate[j[mask]]

        # modest step in the opposite direction of the gradient
        step = 0.02 * grad
        cand2 = candidate - step
        cand2 = np.clip(cand2, 0.0, 1000.0)

        # re‑scale to keep the total sum (exact scale does not matter)
        s2 = cand2.sum()
        if s2 > 0.0:
            cand2 = cand2 * (sum_target / s2)

        score2 = float(evaluate_sequence(cand2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_vec = cand2.copy()
            candidate = cand2
            continue

        # occasional random block perturbation to escape shallow plateaus
        if rng.random() < 0.02:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            scale = float(np.exp(rng.normal(0.0, 0.12)))
            cand3 = candidate.copy()
            cand3[start:start + width] *= scale
            cand3 = np.clip(cand3, 0.0, 1000.0)
            s3 = cand3.sum()
            if s3 > 0.0:
                cand3 = cand3 * (sum_target / s3)
            score3 = float(evaluate_sequence(cand3.tolist()))
            if np.isfinite(score3) and score3 < best_score:
                best_score = score3
                best_vec = cand3.copy()
                candidate = cand3

    # ------------------------------------------------------------------
    # Final sanity check – if something went terribly wrong, fall back to
    # the original parent (always a valid non‑negative sequence).
    # ------------------------------------------------------------------
    if best_vec is None or not np.isfinite(best_score):
        best_vec = np.clip(base, 0.0, 1000.0)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_vec]
