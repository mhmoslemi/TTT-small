import numpy as np

def make_block_perturbation(seq, rng):
    """Scale a random contiguous block of `seq` multiplicatively."""
    n = seq.size
    # block length between ~n/200 and ~n/12
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    seq[start:start + width] *= factor
    return seq


def make_pairwise_transfer(seq, rng):
    """Move mass from an index to its mirror index (i, n-1-i)."""
    n = seq.size
    i = int(rng.integers(0, n // 2))
    j = n - 1 - i
    max_transfer = min(seq[i], 1000.0 - seq[j])
    if max_transfer > 0.0:
        delta = rng.random() * max_transfer
        seq[i] -= delta
        seq[j] += delta
    return seq


def make_swap(seq, rng):
    """Swap two random positions."""
    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)
    seq[i], seq[j] = seq[j], seq[i]
    return seq


def make_noise(seq, rng):
    """Add small Gaussian noise to a random subset of entries."""
    n = seq.size
    # affect roughly 2 % of the entries
    subset = rng.choice(n, size=max(1, n // 50), replace=False)
    noise = rng.normal(0.0, 5.0, size=subset.shape)
    seq[subset] += noise
    return seq
