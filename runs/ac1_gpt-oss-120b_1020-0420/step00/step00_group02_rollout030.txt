    base = np.asarray(height_sequence_1, dtype=np.float64)
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # baseline maybe uniform as fallback
    n = best.size

    # try multiple strategies
    for _ in range(max_iter):
        # random choice of move type
        move_type = rng.integers(0,3)  # 0: block perturbation, 1: smoothing, 2: rearrangement
        if move_type == 0:
            # block perturbation similar to previous but maybe bigger
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            candidate = best.copy()
            scale = float(np.exp(rng.normal(0.0, 0.12)))
            candidate[start:stop] *= scale
            if rng.random() < 0.35:
                candidate = 0.5 * (candidate + candidate[::-1])
        elif move_type == 1:
            # smoothing with random radius
            radius = int(rng.integers(1, max(2, n // 50)))  # radius at least 1
            ksize = 2 * radius + 1
            kernel = np.ones(ksize, dtype=np.float64)
            kernel /= kernel.sum()

        else:
            # rearrangement: shuffle or balanced placement
            # use a random permutation or deterministic alternating large values
            if rng.random() < 0.5:
                candidate = rng.permutation(best)
            else:
                # sort descending and place extremes alternately
                sorted_vals = np.sort(best)[::-1]
                idxs = np.empty_like(best, dtype=int)
                left = 0
                right = n - 1
                toggle = True

import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the exact AC1 upper bound for a candidate sequence.
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the value computed by the supplied evaluate_sequence."""
    # evaluate_sequence expects a plain Python list of scalars
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Helper: create an interleaved descending arrangement.
#   Largest, smallest, second‑largest, second‑smallest, ...
# ----------------------------------------------------------------------
def _interleaved_desc(values: np.ndarray) -> np.ndarray:
    """Return a new array where the sorted values are placed alternating
    from the left and right ends."""
    n = values.size
    out = np.empty_like(values)
    left, right = 0, n - 1
    toggle = True
    for v in values:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients whose AC1 upper bound
    (as defined by evaluate_sequence) is as small as possible.
    """
    # ------------------------------------------------------------------
    # 1.  Obtain the parent construction if it exists.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:                     # fallback when no parent is provided
        base = np.ones(256, dtype=np.float64)

    # safety clipping – the evaluator also clips, but we do it up‑front
    base = np.clip(base, 0.0, 1000.0)

    # sanity check – a completely empty or non‑finite parent never happens,
    # but we keep a robust fallback.
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    n = base.size
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 2.  Initialise the best candidate with the parent (or uniform fallback).
    # ------------------------------------------------------------------
    best = base.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 3.  Deterministic interleaved descending ordering – often a huge win.
    # ------------------------------------------------------------------
    desc_vals = np.sort(best)[::-1]           # descending sort
    cand = _interleaved_desc(desc_vals)
    cand_score = _score(cand)
    if cand_score < best_score:
        best, best_score = cand, cand_score

    # ------------------------------------------------------------------
    # 4.  A handful of random permutations – cheap, high‑variance moves.
    # ------------------------------------------------------------------
    for _ in range(200):
        cand = rng.permutation(best)
        cand_score = _score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score

    # ------------------------------------------------------------------
    # 5.  Simple stochastic hill‑climbing (swap‑based moves).
    # ------------------------------------------------------------------
    max_iters = int(kwargs.get("max_iters", 1200))
    no_improve = 0
    for it in range(max_iters):
        cand = best.copy()

        # perform a few random swaps (1‑3 swaps per iteration)
        for _ in range(rng.integers(1, 4)):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            cand[i], cand[j] = cand[j], cand[i]

        # occasional tiny block scaling (acts like a local perturbation)
        if rng.random() < 0.07:
            width = int(rng.integers(1, max(2, n // 100)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            scale = float(np.exp(rng.normal(0.0, 0.07)))
            cand[start:start + width] *= scale

        # enforce the hard limits before scoring
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _score(cand)
        if cand_score < best_score:

            fresh_score = _score(fresh)
            if fresh_score < best_score:

            no_improve = 0

    return [float(x) for x in best]
