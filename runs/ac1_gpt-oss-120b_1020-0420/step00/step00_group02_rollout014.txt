import numpy as np

# ----------------------------------------------------------------------
# Helper to obtain a reliable length for the construction.
# Uses the provided parent `height_sequence_1` when it exists and is valid,
# otherwise falls back to a modest default length.
# ----------------------------------------------------------------------
def _get_length():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        ):
            return int(parent.shape[0])
    except Exception:
        pass
    # Fallback length – any positive integer works.
    return 256


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# Returns a list of non‑negative coefficients that yields a low AC1 bound.
# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    # Length of the sequence we have to output.
    n = _get_length()

    # Upper bound for individual heights (the evaluator also clamps).
    C = 1000.0

    best_seq = None
    best_score = np.inf

    # Try a handful of geometric decay rates.
    # The decay parameter x = k / n (k = 1 … 10) gives r = 1‑x.
    # Theory predicts the optimum near k = 2 for large n,
    # but we evaluate a small set and keep the best.
    for k in range(1, 11):
        x = float(k) / n
        if not (0.0 < x < 1.0):
            continue
        r = 1.0 - x

        # a_i = C * r^i, i = 0 … n‑1
        seq = C * np.power(r, np.arange(n, dtype=np.float64))
        seq = np.clip(seq, 0.0, 1000.0)

        # Exact evaluation of the objective.
        try:
            score = float(evaluate_sequence(seq.tolist()))
        except Exception:
            continue

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq

    # ------------------------------------------------------------------
    # Fallback: uniform positive sequence (guaranteed to be valid).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Return a plain Python list of floats as required.
    return [float(v) for v in best_seq]
