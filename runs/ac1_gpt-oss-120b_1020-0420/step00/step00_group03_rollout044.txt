import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function that is optimal for the AC1 bound.
    The bound 2·n·max(a*a)/(sum(a)^2) achieves its global minimum (=2)
    when all entries are equal.  This routine therefore builds a uniform
    sequence of the same length as the provided `height_sequence_1`
    (fallback length 256) and returns it.
    """
    # Try to use the supplied parent construction.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        # Fallback: a modest size if the parent is unavailable.
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # Uniform vector – any constant value yields the same bound (2),
    # we pick 1.0 for simplicity.
    candidate = np.full(n, 1.0, dtype=np.float64)

    # Safety clipping to the evaluator's allowed range.
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Convert to plain Python floats as required by the evaluator.
    return candidate.tolist()
