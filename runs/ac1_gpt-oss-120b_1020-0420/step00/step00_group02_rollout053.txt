    # exponential search

    cs = np.logspace(np.log10(1e-5), np.log10(1e-2), num=num_candidates)
    for c in cs:
        seq = np.exp(-c * np.arange(n))
        s = np.sqrt(2.0 * n) / seq.sum()
        seq = seq * s
        # Clip for safety
        np.clip(seq, 0.0, 1000.0, out=seq)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
    # optional local refinement using block perturbations around best_seq
    for _ in range(num_local):
        # generate block perturbation like earlier
        width = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, max(1, n - width + 1))
        candidate = best_seq.copy()

        factor = np.exp(rng.normal(0.0, 0.12))
        candidate[start:start+width] *= factor
        if rng.random() < 0.35:

import numpy as np

def propose_candidate(seed=42, num_c=60, num_local=200, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound evaluated by the supplied ``evaluate_sequence`` function.

    Strategy:
    1. Try a family of exponentially decaying sequences a_i = exp(-c*i) and
       renormalize them to have sum sqrt(2*n).  The convolution peak of such
       sequences can be significantly lower than that of a constant sequence.
    2. Keep the best among those candidates.
    3. Perform a few cheap block‑perturbation local‑search steps around the
       best candidate (similar to the reference program) to possibly improve
       further.
    4. Return the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # Base length – use the provided construction if possible.
    # ------------------------------------------------------------------ #
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:                        # fallback if missing / bad
        base = np.ones(256, dtype=np.float64)

    n = int(base.size)

    # ------------------------------------------------------------------ #
    # Helper: evaluate a candidate and keep the best one.
    # ------------------------------------------------------------------ #
    def eval_and_update(seq, best_seq, best_score):
        # seq is a NumPy array; convert to list only for the evaluator.
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            return seq.copy(), score
        return best_seq, best_score

    # ------------------------------------------------------------------ #
    # Start with a simple uniform sequence (the baseline constant case).
    # ------------------------------------------------------------------ #
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform *= np.sqrt(2.0 * n) / uniform.sum()      # normalise to sqrt(2n)
    best_seq, best_score = eval_and_update(uniform, uniform, float('inf'))

    # ------------------------------------------------------------------ #
    # Search over exponential decay rates c > 0.
    # ------------------------------------------------------------------ #
    # Log‑space grid from 1e-5 to 1e-2 (covers the region where the optimum
    # is expected for n≈ few thousand).
    c_values = np.logspace(-5, -2, num=num_c, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)  # reused for all candidates
    sqrt2n = np.sqrt(2.0 * n)

    for c in c_values:
        seq = np.exp(-c * idx)            # a_i = exp(-c*i)
        seq *= sqrt2n / seq.sum()         # renormalise to sum sqrt(2n)
        np.clip(seq, 0.0, 1000.0, out=seq)
        best_seq, best_score = eval_and_update(seq, best_seq, best_score)

    # Local refinement: cheap block perturbations around the current best.
    # ------------------------------------------------------------------ #
    for _ in range(num_local):
        # Random block width between n/200 and n/12 (at least 1 element)
        min_w = max(1, n // 200)
        max_w = max(2, n // 12)

        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        cand = best_seq.copy()
        # Multiplicative jitter on the chosen block
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor

        # Optional mirroring to keep the shape roughly symmetric
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])

        # Renormalise and clip
        if cand.sum() == 0.0:
            # very unlikely, but protect against division by zero
            cand = uniform.copy()
        else:
            cand *= sqrt2n / cand.sum()
        np.clip(cand, 0.0, 1000.0, out=cand)

        best_seq, best_score = eval_and_update(cand, best_seq, best_score)

    # ------------------------------------------------------------------ #
    # Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------ #
    return [float(v) for v in best_seq]
