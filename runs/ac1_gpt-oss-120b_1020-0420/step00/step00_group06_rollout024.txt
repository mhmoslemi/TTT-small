def solve_lp(f, M, ks):
    n = f.size
    g = cp.Variable(n)
    constraints = [g >= 0, g <= 1000]
    f_np = np.asarray(f, dtype=np.float64)
    for k in ks:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        if low > high:
            continue
        idx_i = np.arange(low, high + 1, dtype=int)
        idx_j = k - idx_i
        # cvxpy indexing
        expr = cp.sum(cp.multiply(f_np[idx_i], g[idx_j]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status not in (cp.settings.OPTIMAL, cp.settings.OPTIMAL_INACCURATE):
        raise RuntimeError(f"LP not solved: {prob.status}")
    return g.value

import numpy as np

# Optional linear programming solver
try:
    import cvxpy as cp
    _has_cvxpy = True
except Exception:  # pragma: no cover
    _has_cvxpy = False


def _convolution_max(seq: np.ndarray) -> float:
    """Return max value of full convolution of seq with itself."""
    conv = np.convolve(seq, seq)
    return float(conv.max()), conv


def _solve_lp(f: np.ndarray, M: float, active_ks):
    """
    Solve   max  sum(g)
    s.t.    g >= 0, g <= 1000,
           (f * g)[k] <= M   for k in active_ks
    Returns the optimal g as a NumPy array.
    """
    n = f.shape[0]
    g = cp.Variable(n)
    constraints = [g >= 0, g <= 1000]

    f_const = np.asarray(f, dtype=np.float64)

    for k in active_ks:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo > hi:
            continue
        idx_i = np.arange(lo, hi + 1, dtype=int)
        idx_j = k - idx_i
        # linear expression sum_i f[i] * g[j]
        expr = cp.sum(cp.multiply(f_const[idx_i], g[idx_j]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status not in (cp.settings.OPTIMAL, cp.settings.OPTIMAL_INACCURATE):
        raise RuntimeError(f"LP not solved, status={prob.status}")
    return np.asarray(g.value, dtype=np.float64)


def _improve_via_lp(initial: np.ndarray, max_outer=8, max_cuts=12):
    """
    Iteratively improve the sequence using the cutting‑plane LP described
    in the problem statement. Returns a sequence with lower AC1 bound
    (or the original if no improvement is found).
    """
    f = initial.copy()
    best_score = float(evaluate_sequence(f.tolist()))
    n = f.shape[0]

    for outer in range(max_outer):
        M, conv_f = _convolution_max(f)
        # active constraints: indices where the maximum is attained
        active = list(np.where(conv_f >= M - 1e-12)[0])

        # solve LP with cutting‑plane augmentation
        for _ in range(max_cuts):
            try:
                g = _solve_lp(f, M, active)
            except Exception:
                return f, best_score
            conv_fg = np.convolve(f, g)
            viol = np.where(conv_fg > M + 1e-8)[0]
            if viol.size == 0:
                break
            # add newly violated indices to the active set
            active = list(set(active) | set(viol.tolist()))

        # evaluate candidate g
        g_score = float(evaluate_sequence(g.tolist()))
        if np.isfinite(g_score) and g_score < best_score:
            f, best_score = g, g_score
            continue

        # try convex combinations of f and g
        improved = False
        for alpha in (0.5, 0.25, 0.75, 0.1, 0.9):
            cand = (1.0 - alpha) * f + alpha * g
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(cand_score) and cand_score < best_score:
                f, best_score = cand, cand_score
                improved = True
                break
        if not improved:
            break  # no progress

    return f, best_score
