def _convolution_max(arr, n, m):
    """Return max coefficient of the polynomial square of ``arr``."""
    # rfft of length m (power‑of‑two ≥ 2n‑1)
    fa = np.fft.rfft(arr, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return np.max(conv)


def _score(arr, n, m):

    if s < 0.01:

    max_c = _convolution_max(arr, n, m)
    return 2.0 * n * max_c / (s * s)

    for _ in range(max_iters):
        # recompute convolution and locate the current peak index
        fa = np.fft.rfft(a, m)
        conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
        t_max = int(np.argmax(conv))

        # contributions of individual entries to this peak
        i_start = max(0, t_max - (n - 1))
        i_end = min(n - 1, t_max)
        idx = np.arange(i_start, i_end + 1)
        jdx = t_max - idx
        contrib = a[idx] * a[jdx]

        # choose source (large contribution) and target (small contribution)
        src = idx[contrib.argmax()]
        tgt = idx[contrib.argmin()]
        if src == tgt:
            # fallback to a random pair if the extremes coincide
            src, tgt = rng.integers(0, n, size=2)
            if src == tgt:
                continue

        # maximal admissible transfer respecting bounds
        max_transfer = min(a[src], 1000.0 - a[tgt])
        if max_transfer <= 0.0:
            # try a random admissible pair
            src, tgt = rng.integers(0, n, size=2)
            max_transfer = min(a[src], 1000.0 - a[tgt])
            if max_transfer <= 0.0:
                continue

        # propose a modest transfer
        delta = max_transfer * step_factor * rng.random()
        if delta <= 0.0:
            continue

        a[src] -= delta
        a[tgt] += delta

        new_score = _score(a, n, m)
        if np.isfinite(new_score) and new_score < cur_score:
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best = a.copy()
                best_score = new_score
        else:
            # reject – revert the change
            a[src] += delta
            a[tgt] -= delta
            no_improve += 1

        # occasional random restart if we stagnate
        if no_improve >= restart_patience:
            a = rng.random(n) * 1000.0
            a = np.clip(a, 0.0, 1000.0)
            if np.sum(a) < 0.01:
                a[0] = 0.01
            cur_score = _score(a, n, m)
            no_improve = 0
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score

    # final safety clipping
    best = np.clip(best, 0.0, 1000.0)
    if np.sum(best) < 0.01:
        best[0] = max(best[0], 0.01)

    return [float(x) for x in best]
