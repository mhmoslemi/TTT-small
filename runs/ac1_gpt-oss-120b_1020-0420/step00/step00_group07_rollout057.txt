import numpy as np

def _initial_length():
    """
    Return the length of the problem instance.
    If height_sequence_1 is present and valid, use its length.
    Otherwise fall back to a modest default.
    """
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
            raise ValueError
        return arr.size
    except Exception:
        # safe fallback length
        return 256


def _triangular_candidate(n):
    """
    Produce the decreasing linear (triangular) sequence:
        a_i ∝ (n - i)   for i = 0 .. n-1
    Normalised so that the sum equals 1 (well above the 0.01 cutoff).
    """
    values = np.arange(n, 0, -1, dtype=np.float64)          # n, n-1, ..., 1
    total = values.sum()
    return values / total


def _dirichlet_candidate(rng, n, alpha=0.5):
    """
    Sample a random non‑negative vector from a Dirichlet distribution.
    Small alpha favours sparsity; the vector is normalised to unit sum.
    """
    # Gamma sampling is the standard way to generate Dirichlet variates.
    g = rng.gamma(alpha, 1.0, size=n)
    # Guard against an all‑zero draw (extremely unlikely for alpha>0)
    if g.sum() == 0.0:
        g = np.ones_like(g)
    return g / g.sum()


def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative heights that minimise the AC1 upper bound.
    The construction mixes a known analytical candidate (triangular) with a
    few random Dirichlet draws, keeping the best found.
    """
    rng = np.random.default_rng(seed)

    n = _initial_length()

    # start with the analytical triangular sequence (score ≈ 4/3)
    best = _triangular_candidate(n)
    best_score = float(evaluate_sequence(best.tolist()))

    # a small set of random Dirichlet candidates may occasionally beat it
    # (they are cheap to evaluate and provide a safe fallback)
    for _ in range(15):
        cand = _dirichlet_candidate(rng, n, alpha=0.5)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ensure the result respects the evaluator's clipping limits
    best = np.clip(best, 0.0, 1000.0)

    return [float(v) for v in best]
