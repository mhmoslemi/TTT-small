import numpy as np

# --------------------- Helper utilities ---------------------

def build_sidon_set(n):
    """
    Greedy construction of a Sidon set S ⊆ {0,…,n-1}
    (all pairwise sums i+j are distinct, including i=j).
    Returns a list of indices.
    """
    used = set()
    S = []
    for pos in range(n):
        # check self-sum 2*pos
        if 2 * pos in used:
            continue
        conflict = False
        for p in S:
            if (pos + p) in used:
                conflict = True
                break
        if conflict:
            continue
        # accept pos
        for p in S:
            used.add(pos + p)
        used.add(2 * pos)
        S.append(pos)
    return S


def compute_small_amplitude(n, k_large, target_score,
                            max_amp=1000.0, safety=1.5):
    """
    Given n, number of large spikes k_large (each = max_amp),
    compute a constant small amplitude that guarantees a score ≤ target_score
    under the optimistic bound max_b ≤ max_amp**2 * safety.
    """
    max_b_est = max_amp * max_amp * safety
    needed_sum = np.sqrt(2.0 * n * max_b_est / target_score)
    remaining = max(1, n - k_large)
    small = (needed_sum - max_amp * k_large) / remaining
    if small < 0.0:
        small = 0.0
    # add a modest margin
    small *= 1.05
    return float(min(max_amp, max(0.0, small)))


# --------------------- Main entry point ---------------------

def propose_candidate(seed=42, **kwargs):
    """
    Construct a step-function sequence that aims to beat the AC1 upper bound.
    Uses a Sidon set of large spikes (value 1000) and a uniform small value
    for the remaining entries. Falls back to random search if the constructed
    candidate does not meet the target.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 4498
        else:
            n = base.size
    except Exception:
        n = 4498

    # --------- Build Sidon set of large entries ----------
    large_indices = build_sidon_set(n)
    k_large = len(large_indices)

    # Target bound we want to beat
    TARGET = 1.50274349

    # Compute a uniform small amplitude that should satisfy the bound
    small_val = compute_small_amplitude(n, k_large, TARGET,
                                        max_amp=1000.0, safety=1.5)

    # Create the candidate sequence
    seq = np.full(n, small_val, dtype=np.float64)
    if k_large:
        seq[large_indices] = 1000.0
    seq = np.clip(seq, 0.0, 1000.0)

    # Evaluate the constructed candidate
    score = float(evaluate_sequence(seq.tolist()))

    # If the construction is not good enough, try a few random refinements
    if not np.isfinite(score) or score > TARGET:
        best_seq = seq
        best_score = score if np.isfinite(score) else np.inf
        # limited random attempts (kept small for runtime)
        for _ in range(20):
            cand = np.full(n, small_val, dtype=np.float64)
            # randomise a subset of large positions (keep count k_large)
            if k_large:
                chosen = rng.choice(n, size=k_large, replace=False)
                cand[chosen] = 1000.0
            else:
                # if Sidon set empty, just random uniform
                cand = rng.uniform(0.0, 1000.0, n)
            cand = np.clip(cand, 0.0, 1000.0)
            s = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(s) and s < best_score:
                best_seq, best_score = cand, s
                if best_score <= TARGET:
                    break
        return [float(x) for x in best_seq]

    # Successful construction
    return [float(x) for x in seq]
