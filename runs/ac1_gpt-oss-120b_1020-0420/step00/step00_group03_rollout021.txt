```python
import numpy as np

def make_block_perturbation(base, rng):
    candidate = np.array(base, copy=True)
    n = candidate.size
    # width between n//200 and n//12
    min_width = max(1, n//200)
    max_width = max(2, n//12)
    width = int(rng.integers(min_width, max_width+1))  # inclusive?
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Apply multiplicative noise to block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate
```

```python
def spectral_perturbation(base, rng, lowpass_alpha=0.3):
    n = base.shape[0]
    # Real FFT
    coeffs = np.fft.rfft(base)
    freqs = np.arange(coeffs.shape[0])  # includes DC index 0
    max_freq = coeffs.shape[0] - 1
    # Low-pass factor: higher frequencies damped with exponential or power law
    # Determine a random damping strength for this candidate
    alpha = rng.uniform(0.0, lowpass_alpha)  # damping amount
    power = rng.uniform(1.0, 2.5)  # shape
    damp = 1.0 - alpha * (freqs / max_freq) ** power
    # Add random jitter to each factor within a small range
    jitter = rng.uniform(0.95, 1.05, size=coeffs.shape)
    factor = damp * jitter
    # Ensure DC factor is >=0 (DC shouldn't be damped too much)
    factor[0] = max(0.1, factor[0])  # keep positive
    # Apply factor
    new_coeffs = coeffs * factor
    # Inverse real FFT
    candidate = np.fft.irfft(new_coeffs, n=n)
    # Post-process: enforce nonnegativity, clip
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    # optionally add small noise to avoid being stuck at zero due to rounding
    if rng.random() < 0.05:
        candidate += rng.normal(0.0, 0.001, size=n)
        np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate
```

    # If solution not optimal, bail
    if g.value is None:

    g_val = np.array(g.value).flatten()

    # Normalization: keep sum same after mixing:
    # new_f = (1-t)*current + t*g_val
    new_f = (1 - t) * current + t * g_val
    new_f = np.clip(new_f, 0.0, 1000.0)
    return new_f

import numpy as np

# optional import of cvxpy – if unavailable we simply skip LP steps
try:
    import cvxpy as cp
    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.array(base, copy=True, dtype=np.float64)
    n = cand.size
    # choose a random block width, at least 1 and at most n//12
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal noise on the block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring to keep some symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _lp_step(current, rng, top_k=8, mix=0.1):
    """
    One linear‑programming based improvement step.
    Returns a new candidate array or None on failure.
    """
    if not _CVX_AVAILABLE:
        return None
    n = current.size
    # current convolution and its maximum
    conv = np.convolve(current, current)
    M = float(conv.max())
    # pick the `top_k` largest convolution indices (they are the active constraints)
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    # build the constraint matrix A (shape len(idxs) × n)
    A = np.zeros((len(idxs), n), dtype=np.float64)
    for r, k in enumerate(idxs):
        # coefficient of g[t] is current[k - t] when 0 <= k - t < n
        t_start = max(0, k - (n - 1))
        t_end = min(k, n - 1)
        t_range = np.arange(t_start, t_end + 1, dtype=int)
        A[r, t_range] = current[k - t_range]
    # linear program: maximise sum(g)  s.t.  g >= 0,  A @ g <= M
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    new = (1.0 - mix) * current + mix * g_val
