import numpy as np

# ------------------------------------------------------------
# Helper: compute the AC1 upper bound for a given array.
# ------------------------------------------------------------
def _ac1_score(arr: np.ndarray) -> float:
    """Return the score used by evaluate_sequence (exactly the same formula)."""
    s = arr.sum()
    if s < 1e-12:
        return float("inf")
    # full convolution (ordered pairs)
    conv = np.convolve(arr, arr)
    max_conv = conv.max()
    n = arr.shape[0]
    return float(2.0 * n * max_conv / (s * s))

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function height sequence whose AC1 upper bound
    is as small as possible.  The function never returns an empty list and
    respects all evaluator constraints.
    """
    rng = np.random.default_rng(seed)

    # length of the problem – the parent construction is always present
    n = len(np.asarray(height_sequence_1, dtype=np.float64))

    # --------------------------------------------------------
    # 1.  Build a collection of parametric candidates.
    # --------------------------------------------------------
    candidates = []

    # Uniform baseline (known score = 2)
    candidates.append(np.ones(n, dtype=np.float64))

    # Exponential families: a_i = r**i
    r_vals = np.linspace(0.90, 0.99995, 22)            # 22 different decays
    idx = np.arange(n, dtype=np.float64)
    for r in r_vals:
        cand = r ** idx
        candidates.append(cand)

    # Power‑law decreasing families: a_i = (n-i)**p
    p_vals_pow = np.linspace(0.2, 4.0, 15)            # gradually steeper tails
    rev_idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n-1, ..., 1
    for p in p_vals_pow:
        cand = rev_idx ** p
        candidates.append(cand)

    # Combined power‑exponential families:
    # a_i = (i+1)^p * r^i   (i from 0)
    p_grid = np.linspace(-2.0, 2.0, 9)                # both decreasing / increasing
    r_grid = np.linspace(0.92, 0.9999, 12)            # moderate to very mild decay

        for r in r_grid:

    best_arr = None
