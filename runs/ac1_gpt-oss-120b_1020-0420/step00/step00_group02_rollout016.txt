    if not isinstance(sequence, list):
        return np.inf
    if not sequence:

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimize
    the AC1 upper bound evaluated by `evaluate_sequence`.

    Strategy:
    * If the global `height_sequence_1` is available, compare it with a uniform
      sequence having the same total mass (its mean value).
    * Also consider a simple fallback uniform sequence of length 256.
    * Return the sequence with the lowest evaluated score.
    """
    # Random generator is unused but kept for signature compatibility.
    _ = np.random.default_rng(seed)

    # Helper to safely evaluate a candidate and return a float score.
    def _score(seq):
        try:
            return float(evaluate_sequence(seq.tolist()))
        except Exception:
            return float("inf")

    # Prepare fallback candidate (length 256, all ones).
    fallback_len = 256
    fallback = np.full(fallback_len, 1.0, dtype=np.float64)
    best = fallback
    best_score = _score(fallback)

    # Attempt to use the provided `height_sequence_1`.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        # No valid parent available – return the fallback.
        return [float(x) for x in best]

    n = base.size

    # Candidate 1: the original parent (clipped by the evaluator anyway).
    parent_score = _score(base)
    if parent_score < best_score:
        best = base
        best_score = parent_score

    # Candidate 2: uniform sequence with the same mean as the parent.
    avg = float(np.mean(base))
    # Clip to the admissible range [0, 1000].
    avg = min(max(avg, 0.0), 1000.0)
    uniform = np.full(n, avg, dtype=np.float64)
    # Guard against the sum‑too‑small rule.
    if np.sum(uniform) < 0.01:
        uniform = np.full(n, 1.0, dtype=np.float64)

    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # Return the best candidate as a plain Python list of floats.
    return [float(x) for x in best]
