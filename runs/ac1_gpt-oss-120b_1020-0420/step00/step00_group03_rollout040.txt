def propose_candidate(seed=42, num_steps=5000, **kwargs):
    rng = np.random.default_rng(seed)
    # initialize
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
        a = base.copy()
    except Exception:
        n = 256
        a = np.ones(n, dtype=np.float64)
    # ensure sum >0
    if a.sum() < 0.01:
        a += 1.0
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # optionally normalize sum to 1 for scaling invariance (not needed)
    # greedy descent
    for step in range(num_steps):
        conv = np.convolve(a, a)
        kmax = int(conv.argmax())
        # compute contributions
        i = np.arange(n)
        j = kmax - i
        mask = (j >= 0) & (j < n)
        products = np.zeros_like(a)
        products[mask] = a[i[mask]] * a[j[mask]]
        # find donor index with largest product
        donor = int(products.argmax())
        partner = kmax - donor
        if not (0 <= partner < n):
            continue
        # decide amount to move
        max_delta = a[donor]
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5  # up to half of current value
        if delta <= 0:
            continue
        # choose recipient different from donor and partner
        # Prefer location with small value
        candidates = np.where(a < a[donor])[0]
        if candidates.size == 0:
            continue
        recipient = int(rng.choice(candidates))
        # apply move
        a[donor] -= delta
        a[recipient] += delta
        # assure nonnegative
        if a[donor] < 0:
            a[donor] = 0.0
        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
        else:
            # revert move
            a[donor] += delta
            a[recipient] -= delta
        # occasional random perturbation to escape
        if step % 100 == 0 and rng.random() < 0.05:
            # random Dirichlet restart
            rand = rng.random(n)
            rand = rand / rand.sum() * a.sum()
            a = np.clip(rand, 0.0, 1000.0)
    return [float(x) for x in best]

def solve_lp(f, M, ks):
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # for each i in range(i_start, i_end+1)
        # coefficient f[i] at column j = k - i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(ks), n))
    b_vec = np.full(len(ks), M)
    c = -np.ones(n)  # maximize sum
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    if not res.success:
        return None
    b_opt = res.x
    return b_opt
```

import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp


def _score(arr: np.ndarray) -> float:
    """Exact score matching evaluate_sequence (no clipping needed, arr already clipped)."""
    n = arr.shape[0]
    maxb = np.convolve(arr, arr).max()
    s = arr.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * maxb / (s * s))


def _block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a copy of a with a random block scaled and optional mirroring."""
    n = a.size
    cand = a.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs) -> list:
    """
    Search for a non‑negative step function that lowers the AC1 bound.
    The routine works with only O(n) auxiliary memory and uses a simple
    greedy move together with occasional random block perturbations.
    """
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------------
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
    except Exception:
        # fallback: a modest uniform vector
        base = np.ones(256, dtype=np.float64)

    a = base.copy()
    if a.sum() < 0.01:
        a += 1.0                                   # guarantee a valid sum

    best = a.copy()
    best_score = _score(best)
    n = a.shape[0]

    # ----- main search loop -------------------------------------------------
    for step in range(num_steps):
        # 60 % of the time try a targeted greedy move,
        # the remainder use a random block perturbation.
        if rng.random() < 0.6:
            # --- greedy reduction of the current worst convolution entry ---
            conv_full = np.convolve(a, a)
            k_max = int(conv_full.argmax())

            idx = np.arange(n, dtype=np.int64)
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)

            # product contributions for the active shift
            contrib = np.zeros_like(a)
            contrib[mask] = a[idx[mask]] * a[partner[mask]]

            donor = int(contrib.argmax())
            mate = k_max - donor
            if not (0 <= mate < n):
                continue
            if a[donor] <= 0.0:
                continue

            # move a modest amount from the donor towards a small‑value entry
            delta = a[donor] * rng.random() * 0.30      # up to 30 % of donor
            tiny = np.where(a <= a[donor] * 0.90)[0]    # candidates with smaller value
            if tiny.size == 0:
                continue
            recipient = int(rng.choice(tiny))

            a[donor] -= delta
            a[recipient] += delta
            a = np.clip(a, 0.0, 1000.0)

            cur_score = _score(a)
            if np.isfinite(cur_score) and cur_score < best_score:
                best = a.copy()
                best_score = cur_score
            else:
                # revert the move if it did not improve
                a[donor] += delta
                a[recipient] -= delta
        else:
            # --- random block perturbation (exploratory) --------------------
            cand = _block_perturb(a, rng)
            cur_score = _score(cand)
            if np.isfinite(cur_score) and cur_score < best_score:
                a = cand
                best = a.copy()
                best_score = cur_score
            # otherwise keep the current a unchanged

        if step % 250 == 0 and rng.random() < 0.05:
            rnd = rng.random(n)
            rnd = rnd / rnd.sum() * a.sum()
            a = np.clip(rnd, 0.0, 1000.0)
