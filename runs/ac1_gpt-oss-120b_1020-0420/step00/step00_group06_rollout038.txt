def _conv_max(a):
    """Return the maximum value of the full convolution a*a."""
    return np.convolve(a, a).max()


def _gradient_at_max(a, max_idx):
    """Gradient of the max‑convolution entry (subgradient) w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    # indices i where a[i] participates in the max entry
    idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
    idx_j = max_idx - idx_i
    grad[idx_i] = a[idx_j]
    return grad
