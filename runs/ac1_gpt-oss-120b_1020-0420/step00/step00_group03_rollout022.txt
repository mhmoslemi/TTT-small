        for j in range(i, len(indices)):
            s = indices[i] + indices[j]
            if s in sums:

    for _ in range(max_tries):
        indices = []
        sums = set()
        # random permutation of candidate indices
        order = rng.permutation(n)

            # skip if we already have k_target
            if len(indices) >= k_target:
                break

            if i + i in sums:
                conflict = True
            else:
                for j in indices:
                    if i + j in sums:
                        conflict = True
                        break
            if not conflict:
                # add
                for j in indices:
                    sums.add(i + j)
                sums.add(i + i)
                indices.append(i)
        if len(indices) >= k_target:
            return indices[:k_target]
    # if fail, return what we got (maybe less)
    return indices

def greedy_sidon(n, rng, attempts=200, target_k=None):
    best_set = []
    for _ in range(attempts):
        order = rng.permutation(n)
        current = []
        sums = set()
        for x in order:
            conflict = False
            # quick early break: if len(current) is large enough needed to stop
            for y in current:
                s = x + y
                if s in sums:
                    conflict = True
                    break
            if conflict:
                continue
            if x + x in sums:
                continue
            # add
            for y in current:
                sums.add(x + y)
            sums.add(x + x)
            current.append(x)
            # early exit if target_k reached
            if target_k is not None and len(current) >= target_k:
                break
        if len(current) > len(best_set):
            best_set = current
            if target_k is not None and len(best_set) >= target_k:
                break
    return best_set

    sums=set()

        # check if i can be added
        if any((i+j) in sums for j in indices) or (i+i) in sums:
            continue
        # add i
        for j in indices:
            sums.add(i+j)
        sums.add(i+i)
        indices.append(i)

def _is_sidon(indices):
    """Check that all pairwise sums (including i+i) are distinct."""
    seen = set()
    for i, a in enumerate(indices):
        s = a + a
        if s in seen:
            return False
        seen.add(s)
        for b in indices[:i]:
            s = a + b
            if s in seen:
                return False
            seen.add(s)
    return True

    """
