def compute_score(a):
    n = a.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf, max_c
    return 2.0 * n * max_c / (s*s), max_c

import numpy as np

# ------- helper to compute the AC1 score (fast, uses FFT) -------
def _fast_score(vec: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative vector (sum normalized not required)."""
    n = vec.size
    if n == 0:
        return np.inf
    # length for convolution via FFT (next power of two for speed)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(vec, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_c = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))
