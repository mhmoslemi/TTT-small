import numpy as np

def _contributions(seq: np.ndarray, k_max: int) -> np.ndarray:
    """Return per-index contribution to the convolution value at shift k_max."""
    n = seq.size
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[idx[mask]] * seq[j[mask]]
    return contrib
