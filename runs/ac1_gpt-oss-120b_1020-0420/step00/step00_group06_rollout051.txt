import numpy as np

def compute_score(arr):
    # arr should be numpy 1-D of float64 in bounds
    n = arr.shape[0]
    if arr.sum() < 0.01:
        return np.inf
    # FFT convolution
    size = 1 << ((2*n - 1).bit_length())
    # Compute real FFT
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_b = conv.max()
    return float(2 * n * max_b / (arr.sum() ** 2))

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # get parent if defined.
    try:
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
    except Exception:
        parent_arr = np.full(256, 1.0, dtype=np.float64)
    best = parent_arr.copy()
    best_score = compute_score(best)
    n = best.size

    # evaluation of simple shapes
    # shape functions
    for exponent in (-2.0, -1.5, -1.0, -0.5, -0.2, 0.2, 0.5, 1.0, 2.0):
        base = (np.arange(n, dtype=np.float64) + 1.0) ** exponent
        # normalize max to 1000
        base = base / base.max() * 1000.0
        score = compute_score(base)
        if score < best_score:
            best = base
            best_score = score

        # reversed
        rev = base[::-1]
        score = compute_score(rev)
        if score < best_score:
            best = rev
            best_score = score

    # Random mixture loop
    for step in range(num_steps):
        # propose new candidate by random perturbations of best
        candidate = best.copy()
        # Choose random operation
        op = rng.integers(0, 4)
        if op == 0:
            # block scaling
            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            candidate[start:start+width] *= factor
        elif op == 1:
            # random shuffle of a subsegment
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width)
            sub = candidate[start:start+width]
            rng.shuffle(sub)
            candidate[start:start+width] = sub
        elif op == 2:
            # generate fresh random decreasing array
            vals = rng.random(n)  # uniform [0,1)
            vals = -np.log(vals + 1e-12)  # exponential
            vals = np.sort(vals)[::-1]  # descending
            vals = vals / vals.max() * 1000.0
            candidate = vals
        else:
            # generate random power-law descending
            vals = rng.random(n) ** rng.uniform(0.2, 5.0)
            vals = np.sort(vals)[::-1]
            vals = vals / vals.max() * 1000.0
            candidate = vals

        # occasional symmetrization
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])

        # Clip
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        score = compute_score(candidate)
        if score < best_score:
            best = candidate
            best_score = score

    # final fallback ensure non-empty and valid
    if best_score == np.inf or best.size == 0:
        best = np.ones(256, dtype=np.float64)
    # Return as plain list
    return [float(x) for x in best]

import numpy as np

def _score(arr: np.ndarray) -> float:
    """Return the AC1 upper bound for a 1‑D array of heights."""
    # Clip to the allowed interval (in‑place to avoid extra copies).
    np.clip(arr, 0.0, 1000.0, out=arr)
    n = arr.shape[0]
    total = arr.sum()
    if total < 0.01:
        return np.inf
    # Length for a power‑of‑two FFT (fast convolution).
    size = 1 << ((2 * n - 1).bit_length())
    # Convolution via FFT (real → real).
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2 * n * max_b / (total * total))

    n = parent.size
