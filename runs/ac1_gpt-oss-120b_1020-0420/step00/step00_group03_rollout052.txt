import numpy as np
import math

# ------------------------------------------------------------
# Helper: build a sparse 0/1 pattern with bounded convolution peaks.
# ------------------------------------------------------------
def _build_sparse_pattern(length, target_bound, max_contrib, rng):
    """
    Greedy random construction of a set S ⊂ {0,…,length-1}
    such that for every possible sum s = i+j (i,j∈S) the total
    ordered contribution count does not exceed max_contrib.
    Returns the list of selected indices (may be empty if construction fails).
    """
    # Minimum number of non‑zeros needed to beat the target bound
    # (using the worst case max_b = max_contrib·1²):
    #   target_bound ≥ 2·length·max_contrib / k²   →   k ≥ sqrt(2·length·max_contrib/target)
    needed = int(math.ceil(math.sqrt(2.0 * length * max_contrib / target_bound))) + 5

    contrib = {}                # sum → ordered contribution count
    selected = []               # list of chosen indices
    selected_set = set()        # fast membership test
    attempts = 0
    max_attempts = length * 20   # generous limit; resets on every successful addition

    while len(selected) < needed and attempts < max_attempts:
        idx = int(rng.integers(0, length))
        if idx in selected_set:
            attempts += 1
            continue

        # Self‑pair contribution (i+i) adds 1
        if contrib.get(idx + idx, 0) + 1 > max_contrib:
            attempts += 1
            continue

        # Check cross contributions with already selected indices
        ok = True
        for j in selected:
            s = idx + j
            if contrib.get(s, 0) + 2 > max_contrib:   # each unordered pair adds 2 ordered terms
                ok = False
                break
        if not ok:
            attempts += 1
            continue

        # Accept the index: update contribution counters
        for j in selected:
            s = idx + j
            contrib[s] = contrib.get(s, 0) + 2
        contrib[idx + idx] = contrib.get(idx + idx, 0) + 1

        selected.append(idx)
        selected_set.add(idx)
        attempts = 0      # reset counter after a successful insertion

    return selected

# ------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Generates a non‑negative step‑function (list of floats) that aims
    to minimise the AC1 bound evaluated by `evaluate_sequence`.
    The algorithm builds a sparse 0/1 pattern with bounded convolution
    peaks; if the bound is not achieved, the best fallback (the provided
    parent construction) is returned.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Parent sequence – used as a safe fallback (already clipped to [0,1000])
    # ----------------------------------------------------------------
    parent = np.clip(
        np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
    )
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ----------------------------------------------------------------
    # Target bound  (the lower the better)
    # ----------------------------------------------------------------
    TARGET = 1.50274349

    # Try a few increasingly permissive contribution limits
    for max_contrib in (3, 4, 5):
        indices = _build_sparse_pattern(len(parent), TARGET, max_contrib, rng)
        if not indices:
            continue

        # Build the candidate sequence (height = 1 for selected positions)
        cand = np.zeros(len(parent), dtype=np.float64)
        cand[indices] = 1.0

        # Exact evaluation – cheap because the vector is very sparse
        score = float(evaluate_sequence(cand.tolist()))

        # Keep the best seen candidate
        if score < best_score:
            best_seq = cand
            best_score = score

        # If we already beat the target, return immediately
        if score <= TARGET:
            return cand.tolist()

    # ----------------------------------------------------------------
    # If none of the sparse constructions succeeded, fall back to the
    # best candidate found (parent or a partially built sparse vector).
    # ----------------------------------------------------------------
    return [float(x) for x in best_seq]
