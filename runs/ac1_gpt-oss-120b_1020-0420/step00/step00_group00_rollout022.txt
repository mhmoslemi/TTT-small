import numpy as np

# ------------------------------------------------------------
# Helper: fast full convolution via FFT (real input, power‑of‑2 length)
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # length for zero‑padding >= 2n‑1, next power of two
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv[: 2 * n - 1]

# Helper: generate a random sparse non‑negative sequence of length n,
#           normalised to sum == 1

def _random_sparse(rng: np.random.Generator, n: int) -> np.ndarray:
    # choose a random density between 1 % and 20 %
    p = rng.uniform(0.01, 0.20)
    k = max(2, int(p * n))

    w = rng.random(k)
    w /= w.sum()
    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = w
    return seq
