import numpy as np

def propose_candidate(seed=42, **kwargs):
    """Return a simple uniform sequence as a robust fallback.
       Uses height_sequence_1 for length if available.
    """
    rng = np.random.default_rng(seed)  # Not used but for determinism
    # Determine length
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        if seq.ndim != 1 or seq.size == 0:
            raise ValueError
        n = seq.size
    except Exception:
        n = 256
    # Uniform constant (ensuring within bounds)
    const = 1.0
    best = np.full(n, const, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    # Try using the provided parent after clipping
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n:
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent
                best_score = parent_score
    except Exception:
        pass
    # Simple local refinement: occasional small random multiplicative updates.
    for _ in range(500):
        # pick a random block
        width = max(1, n // 200)
        start = rng.integers(0, n - width + 1)
        scale = np.exp(rng.normal(0.0, 0.03))
        trial = best.copy()
        trial[start:start+width] *= scale
        trial = np.clip(trial, 0.0, 1000.0)
        score = float(evaluate_sequence(trial.tolist()))
        if np.isfinite(score) and score < best_score:
            best = trial
            best_score = score
    return [float(x) for x in best]

import numpy as np

def _fast_score(arr: np.ndarray) -> float:
    """
    Fast O(n) surrogate for the evaluator: 2*n*sum(ai^2)/(sum(ai)^2).
    Returns np.inf for invalid (near‑zero) sums.
    """
    s = float(np.sum(arr))
    if s < 0.01:
        return np.inf
    n = arr.size
    return 2.0 * n * float(np.dot(arr, arr)) / (s * s)


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function height list.
    Uses only O(n) auxiliary memory and a cheap surrogate objective,
    falling back to the exact evaluator only a few times.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Determine the length to use (from the supplied construction)
    # ------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = int(base.size)
    except Exception:
        # Generic fallback when the global is missing or malformed
        n = 256

    # ------------------------------------------------------------
    # Initialise candidates
    # ------------------------------------------------------------
    # Uniform vector – the theoretical optimum for the surrogate objective
    uniform = np.full(n, 1.0, dtype=np.float64)

    # Clip parent (if available) into the admissible box
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.shape[0] != n:
            raise ValueError
    except Exception:
        parent = None

    # Choose the best of the simple starters according to the cheap surrogate
    best = uniform
    best_score = _fast_score(best)
    if parent is not None:
        p_score = _fast_score(parent)
        if p_score < best_score:
            best = parent
            best_score = p_score

    # ------------------------------------------------------------
    # Random local search (many cheap evaluations)
    # ------------------------------------------------------------
    for _ in range(5000):
        # Randomly decide the perturbation type
        if rng.random() < 0.7:
            # Block multiplicative perturbation
            width = max(1, n // 200)               # ~1‑5 % of the length
            start = rng.integers(0, n - width + 1)  # inclusive start
            scale = np.exp(rng.normal(0.0, 0.04))   # modest log‑normal factor
            trial = best.copy()
            trial[start:start + width] *= scale
        else:

            scale = np.exp(rng.normal(0.0, 0.03))
            trial = best * scale

        # Enforce box constraints
        np.clip(trial, 0.0, 1000.0, out=trial)

        # Cheap surrogate evaluation
        score = _fast_score(trial)

        if np.isfinite(val) and val < best_val:

            best_val = val

    # Return a plain Python list of floats
    return [float(x) for x in best_final]
