def _convolve_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a*a using a real FFT (O(n log n), O(n) memory)."""
    fa = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(fa * fa, fft_len)
