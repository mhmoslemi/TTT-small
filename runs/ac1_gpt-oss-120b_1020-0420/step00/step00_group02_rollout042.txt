def propose_candidate(seed=42, num_iters=200, step_size=0.1, **kwargs):
    rng = np.random.default_rng(seed)

    # evaluate initial score
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    n = a.size

    for it in range(num_iters):
        # compute convolution
        conv = np.convolve(a, a)
        max_b = np.max(conv)
        # fast break if max_b == 0
        if max_b == 0:
            break
        # find first index of max (or any)
        k_max = int(np.argmax(conv))
        # compute partner array
        # partner[i] = a[k_max - i] if within bounds else 0
        # Use slicing:
        partner = np.zeros_like(a)
        # Compute overlapping range
        # i from max(0, k_max-(n-1)) to min(n-1, k_max)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        # For i in that range, partner[i] = a[k_max - i]
        # vectorized: generate indices
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[k_max - idx]
        # compute target = max_b / np.sum(a)   # M / S
        sum_a = np.sum(a)
        if sum_a == 0:
            break

        # gradient direction = partner - target
        grad = partner - target
        # update with step size
        a = a - step_size * grad
        # ensure nonnegative and clip bounds
        a = np.clip(a, 0.0, 1000.0)

        # compute score
        score = float(evaluate_sequence(a.tolist()))

            best_seq = a.copy()

    seq = np.arange(n, 0, -1, dtype=np.float64) * scale
    # Safety clipping (in case of rounding issues).
    seq = np.clip(seq, 0.0, max_height)

    # Evaluate the initial candidate; keep it if valid.
    best_seq = seq
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = float('inf')

    # Optional tiny local refinements (does not hurt correctness).
    rng = np.random.default_rng(seed)
    for _ in range(200):
        # Randomly perturb a single element by up to ±5 % of its current value.
        idx = rng.integers(0, n)
        perturb = (rng.random() - 0.5) * 0.10 * best_seq[idx]
        cand = best_seq.copy()
        cand[idx] = np.clip(cand[idx] + perturb, 0.0, max_height)
        try:
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_seq = cand
                best_score = score
        except Exception:
            continue

    # Final safety: ensure a non‑empty, finite, non‑negative sequence.
    if not (isinstance(best_seq, np.ndarray) and best_seq.ndim == 1 and best_seq.size > 0):
        best_seq = np.ones(n, dtype=np.float64)

    # Return as a plain Python list of floats.
    return [float(x) for x in best_seq]
