def compute_score(a):
    n = a.shape[0]
    sum_a = np.sum(a)
    if sum_a < 1e-12: return np.inf
    # Convolution using FFT
    size = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]  # real result
    maxb = np.max(conv_full)
    return float(2 * n * maxb / (sum_a*sum_a))

# ----------------------------------------------------------------------
# Helper: convolution of a vector with itself (full linear convolution)
# ----------------------------------------------------------------------
def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a*a (full convolution) using FFT – O(n log n) memory O(n)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())       # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv
