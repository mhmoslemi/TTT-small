    n = arr.shape[0]
    L = 2 * n - 1
    F = np.fft.rfft(arr, n=L)

    rng = np.random.default_rng(seed)
    parent = globals().get('height_sequence_1')
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and np.isfinite(parent).all():

    temperature = 1.0
    for it in range(num_iterations):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        if delta == 0:

        cur[j] += delta
        new_score = compute_score(cur)
        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score

            cur[i] += delta
            cur[j] -= delta
        # cooling

        if (it+1) % 500 == 0 and rng.random() < 0.05:

    n = a.shape[0]
    # compute convolution via FFT
    L = 2*n - 1
    # use rfft of length L

# Helper: compute the AC1 upper bound and the index of the
# maximal entry of the self‑convolution using an FFT (O(n log n))
# ------------------------------------------------------------
def _score_and_max(arr: np.ndarray):
    """Return (score, max_index) for a non‑negative array."""
    n = arr.shape[0]
    L = 2 * n - 1                     # full convolution length
    # linear convolution via FFT
    F = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(F * F, n=L)
    max_idx = int(conv.argmax())

    s = arr.sum()
    if s < 0.01:                      # invalid (should never happen)
        return np.inf, max_idx
    score = float(2.0 * n * max_val / (s * s))
    return score, max_idx

    # ------------------------------------------------------------------
    best = cur.copy()
    best_score, cur_max_idx = _score_and_max(best)

    # ------------------------------------------------------------------
    # Simulated‑annealing loop
    # ------------------------------------------------------------------
    temperature = 1.0
    for it in range(num_iter):
        # --- current score and the index where the convolution peaks
        cur_score, cur_max_idx = _score_and_max(cur)

        # --- indices that actually contribute to the current maximal entry
        lo = max(0, cur_max_idx - (n - 1))
        hi = min(n - 1, cur_max_idx)          # inclusive

        # choose a source index inside the active window
        src = rng.integers(lo, hi + 1)

        # choose a destination index (anywhere else, just avoid src)
        dst = rng.integers(0, n)
        while dst == src:
            dst = rng.integers(0, n)

        # feasible amount that can be transferred while respecting bounds
        max_delta = min(cur[src], 1000.0 - cur[dst])
        if max_delta <= 0.0:
            continue
        # transfer a random fraction (up‑to 50 % of the feasible amount)
        delta = max_delta * rng.random() * 0.5
        if delta <= 0.0:
            continue

        # perform the move
        cur[src] -= delta
        cur[dst] += delta

        # evaluate new candidate
        new_score, new_max_idx = _score_and_max(cur)

        # acceptance criterion (hill‑climbing + annealing)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            cur_max_idx = new_max_idx
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
        else:
            # revert the move
            cur[src] += delta
            cur[dst] -= delta

        # cooling schedule
        temperature *= 0.9999

        # occasional gentle perturbation of the best solution to escape plateaus
        if (it + 1) % 2000 == 0:
            cur = best.copy()
            # apply a modest block scaling
            block_w = max(1, n // rng.integers(30, 120))
            start = rng.integers(0, n - block_w + 1)
            factor = np.exp(rng.normal(0.0, 0.1))
            cur[start:start + block_w] *= factor
            cur = np.clip(cur, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Fallback safety: ensure a non‑empty, finite result
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.ones(max(256, n), dtype=np.float64)

    # Convert to plain Python list as required by the harness
    return [float(v) for v in best]
