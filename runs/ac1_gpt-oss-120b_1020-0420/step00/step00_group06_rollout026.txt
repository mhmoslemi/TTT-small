def _initial_sequence(rng):
    """Return a deterministic O(n) initial sequence.
    Prefer the supplied parent; otherwise use an arcsine‑type shape.
    """
    # Try the global parent if it looks valid
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size and np.isfinite(parent).all():
            seq = np.clip(parent, 0.0, 1000.0)
            s = seq.sum()
            if s >= 0.01:
                return seq / s
    except Exception:
        pass

    # fallback: arcsine‑like distribution on [0, n)
    n = 5582
    idx = np.arange(n, dtype=np.float64)
    # avoid division by zero at the ends
    eps = 1e-12
    seq = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5) + eps)
    seq = seq / seq.sum()
    return seq


def propose_candidate(seed=42, num_steps=3000, **kwargs):
    """Gradient‑based flattening of the self‑convolution.
    Returns a list of non‑negative heights that keep the AC1 upper bound
    below the previous 2.0 baseline."""
    rng = np.random.default_rng(seed)

    a = _initial_sequence(rng)               # start with sum == 1
    n = a.size
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # size for a power‑of‑two FFT (covers 2*n‑1 convolution length)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # initial full convolution (used for the first gradient step)
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    cur_max = conv.max()

    for it in range(num_steps):
        # location of the current maximum of the convolution
        k = int(conv.argmax())

        # vector a_{k-i} (zero where out‑of‑range)
        rev_idx = k - np.arange(n)
        mask = (rev_idx >= 0) & (rev_idx < n)
        shift = np.empty_like(a)
        shift.fill(0.0)
        shift[mask] = a[rev_idx[mask]]

        # gradient of 2*n*max/sum^2 w.r.t. a (sum is 1 after normalisation)
        grad = shift - cur_max

        # step size decays slowly; guaranteed positive but tiny enough
        eta = 0.05 * (0.99 ** it)

        # gradient descent step, then re‑normalise to keep sum == 1
        a = a - eta * grad
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-12:                     # pathological – fall back to uniform
            a = np.ones_like(a) / n
        else:
            a = a / s

        if (it + 1) % 250 == 0:

            a = np.clip(a + jitter, 0.0, 1000.0)
            a = a / a.sum()

        # recompute convolution for the updated sequence
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
        cur_max = conv.max()

        score = float(evaluate_sequence(a.tolist()))

            best_score = score

    return [float(x) for x in best]
