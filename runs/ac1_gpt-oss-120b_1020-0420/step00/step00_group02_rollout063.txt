            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                current = cand
                improved = True
                break

        if not improved:
            # occasional random block perturbation as explorer
            block = max(2, n // 250)
            start = rng.integers(0, n - block + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            cand = current.copy()
            cand[start:start + block] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                current = cand

        # early exit if we already beat the target
        if best_score < 1.5028:
            break

    # guarantee a valid return
    if best_score == float('inf') or best.size == 0:
        best = fallback

    return [float(v) for v in best]
