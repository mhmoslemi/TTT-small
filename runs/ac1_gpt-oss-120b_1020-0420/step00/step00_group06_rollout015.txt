import numpy as np

# ----------------------------------------------------------------------
# Helper: generate candidate shapes
# ----------------------------------------------------------------------
def exponential_shape(n, eps, beta=1.0):
    """a_i = exp(-eps * i**beta), i = 0..n-1."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-eps * np.power(i, beta))


def power_decay_shape(n, p):
    """a_i = (1 - i/(n-1))**p, monotone decreasing to 0."""
    if n == 1:
        return np.ones(1, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    return np.power(1.0 - i / (n - 1.0), p, dtype=np.float64)


# ----------------------------------------------------------------------
# Small random block perturbation (keeps O(n) memory)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a copy of ``base`` with a random block scaled multiplicatively."""
    cand = base.copy()
    n = cand.size
    # block width proportional to sqrt(n) but at least 1 element
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal perturbation
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mild symmetrisation (helps sometimes)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Transfer a small amount of mass from a high‑contributing index to a
# low‑contributing one (guided by the current max‑convolution index).
# ----------------------------------------------------------------------
def mass_transfer_step(seq, rng):
    """Move a fraction of mass from the larger side of the peak to the smaller."""
    n = seq.size
    # convolution of the current sequence (numpy uses FFT for these sizes)
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))

    # valid i such that both i and j = k_max-i lie in [0, n)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low >= i_high:
        # degenerate case – fall back to a tiny random perturbation
        return make_block_perturbation(seq, rng)

    i = int(rng.integers(i_low, i_high + 1))
    j = k_max - i
    if i == j:                     # same index – pick another pair
        return make_block_perturbation(seq, rng)

    # Decide direction: move from larger to smaller entry
    if seq[i] > seq[j]:
        src, dst = i, j
    else:
        src, dst = j, i

    if seq[src] <= 0.0:
        return make_block_perturbation(seq, rng)

    # Transfer up to 10 % of the source entry (but not more than the entry itself)
    delta = rng.random() * min(seq[src] * 0.10, seq[src])
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    # keep within the allowed range
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq
