import numpy as np

def _evaluate(seq):
    """Helper that safely calls the pre‑imported evaluator."""
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound
    improves on the default construction.

    The algorithm:
      1. Start from the provided parent sequence ``height_sequence_1``.
      2. Scan a family of geometric sequences a_i = r**i (r close to 1)
         and keep the best one.
      3. Perform a cheap local search that transfers a small amount of mass
         between two positions while preserving the total sum.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise from the parent construction (guaranteed to exist).
    # ------------------------------------------------------------------
    parent = np.clip(
        np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
    )
    if (
        parent.ndim != 1
        or parent.size == 0
        or not np.isfinite(parent).all()
    ):
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = _evaluate(best)

    # ------------------------------------------------------------------
    # 2. Scan simple geometric families a_i = r^i.
    #    r is taken in a narrow interval just below 1 where the bound
    #    tends to be smaller than 2.
    # ------------------------------------------------------------------
    # 200 values give a fine enough resolution while staying cheap.
    r_vals = np.linspace(0.998, 0.9999, 200)
    powers = np.arange(n, dtype=np.float64)  # reused for all r
    for r in r_vals:
        seq = np.power(r, powers)               # a_i = r**i
        # values are already in [0,1], thus within the allowed range.
        score = _evaluate(seq)
        if np.isfinite(score) and score < best_score:
            best = seq
            best_score = score

    # ------------------------------------------------------------------
    # 3. Simple local search: move a small amount of weight from i to j.
    #    The move keeps the total sum unchanged, which is irrelevant
    #    for the scale‑invariant bound but helps stability.
    # ------------------------------------------------------------------
    sigma = 0.05                     # relative std. dev. of the transferred mass
    for _ in range(num_steps):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # Propose a change proportional to the current value at i.
        delta = rng.normal(0.0, sigma) * best[i]
        new_i = best[i] + delta
        # Enforce box constraints.
        new_i = max(0.0, min(1000.0, new_i))
        diff = new_i - best[i]
        new_j = best[j] - diff
        if not (0.0 <= new_j <= 1000.0):
            continue

        cand = best.copy()
        cand[i] = new_i
        cand[j] = new_j
        if np.sum(cand) < 0.01:          # sanity check (should never happen)
            continue

        score = _evaluate(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # slowly tighten the step size after a successful move
            sigma = max(0.001, sigma * 0.99)

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
