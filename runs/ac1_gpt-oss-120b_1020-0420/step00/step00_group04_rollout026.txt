def generate_sidonic_sequence(n, L):
    rng = default_rng
    # Greedy random: start with empty list.
    indices = []
    sums = set()
    attempts = 0
    while len(indices) < L and attempts < 10_000:
        cand = rng.integers(0, n)
        # ensure not duplicate index
        if cand in indices: 
            attempts += 1; continue
        # compute new sums with existing indices
        new_sums = [cand + i for i in indices] + [cand + cand]
        # also sums with potential reversed? Since i+j = j+i same; we already consider unordered.
        # Since we want all ordered sums to be unique, duplicates for reversed combos appear anyway.
        # Ensure not overlapping with existing sums.
        conflict = False
        for s in new_sums:
            if s in sums:
                conflict = True
                break
        if conflict:
            attempts += 1
            continue
        # accept
        indices.append(cand)
        for s in new_sums:
            sums.add(s)
        attempts = 0  # reset attempts after successful addition
    return sorted(indices)

def generate_sidonic_set(n, L, rng):
    indices = []
    sums = set()
    attempts_total = 0
    # We'll try random candidates repeatedly until we have L or exceed attempts limit.
    while len(indices) < L and attempts_total < L*5000:
        cand = rng.integers(0, n)
        if cand in indices:
            attempts_total += 1
            continue
        # consider new sums
        conflict = False
        # compute sums with all existing indices (including self sum)
        for i in indices:
            s = cand + i
            if s in sums:
                conflict = True
                break
        if conflict:
            attempts_total += 1
            continue
        # check self sum
        if cand + cand in sums:
            attempts_total += 1
            continue
        # No conflict, add
        indices.append(cand)
        sums.add(cand + cand)
        for i in indices[:-1]:
            sums.add(cand + i)

import numpy as np

def compute_score(arr):
    arr = np.asarray(arr, dtype=np.float64)
    n = arr.size
    conv = np.convolve(arr, arr)
    max_val = np.max(conv)
    sum_val = np.sum(arr)
    if sum_val < 0.01:
        return np.inf
    return float(2.0 * n * max_val / (sum_val ** 2))

def generate_sidonic_set(n, L_target, rng):
    indices = []
    sums = set()
    attempts = 0
    max_attempts = L_target * 1000
    while len(indices) < L_target and attempts < max_attempts:
        cand = rng.integers(0, n)
        if cand in indices:
            attempts += 1
            continue
        conflict = False
        # check self sum
        if (cand + cand) in sums:
            attempts += 1
            continue
        # check sums with existing indices
        for i in indices:
            s = cand + i
            if s in sums:
                conflict = True
                break
        if conflict:
            attempts += 1
            continue
        # accept
        indices.append(cand)
        sums.add(cand + cand)
        for i in indices[:-1]:
            sums.add(cand + i)
        attempts = 0
    return np.array(sorted(indices), dtype=int)

import numpy as np

# ------------------------------------------------------------
# Helper: exact score (mirrors evaluate_sequence but without the
# security checks).  Used only for internal selection.
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the AC1 upper‑bound score for a 1‑D array."""
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.size
    # convolution (full)
    conv = np.convolve(seq, seq)
    max_conv = np.max(conv)
    total = np.sum(seq)
    if total < 0.01:
        return float("inf")
    return float(2.0 * n * max_conv / (total ** 2))


# ------------------------------------------------------------
# Build a sequence whose large values sit opposite tiny values.
# This reduces the central convolution term (i + (n‑1‑i))
# ------------------------------------------------------------
def _anti_correlated_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a non‑negative sequence of length n.
    Random values are drawn, sorted, then placed so that the
    largest half occupies the left side of the array while the
    smallest half occupies the right side – the opposite halves
    are thus paired in the central convolution.
    """
    # draw a positive distribution with some variance
    values = rng.exponential(scale=1.0, size=n)
    # ensure bounds required by the evaluator
    values = np.clip(values, 0.0, 1000.0)

    # sort to obtain ordering
    vals_sorted = np.sort(values)               # ascending
    half = n // 2

    if n % 2 == 0:
        # left side: largest half (descending)
        left = vals_sorted[-half:][::-1]
        # right side: smallest half (ascending)
        right = vals_sorted[:half]
        seq = np.concatenate([left, right])
    else:
        left = vals_sorted[-half:][::-1]                 # largest half
        right = vals_sorted[:half]                       # smallest half
        middle = np.array([vals_sorted[half]])           # median element
        seq = np.concatenate([left, middle, right])

    return seq

        pert[start:stop] *= factor
        pert = np.clip(pert, 0.0, 1000.0)
        val = _score(pert)
        if val < best_val:
            best_seq, best_val = pert, val

    # ------------------------------------------------------------------
    # Final safety: ensure the sequence satisfies the evaluator’s
    # minimal requirements (non‑negative, finite, sum≥0.01).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if np.sum(best_seq) < 0.01:
        # inject a tiny mass if something went wrong
        best_seq[0] = max(best_seq[0], 0.01)

    return [float(x) for x in best_seq]
