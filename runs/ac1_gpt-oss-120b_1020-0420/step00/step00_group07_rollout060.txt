def gradient_descent_step(a, m1, m2, eta):
    # compute FFT of a (size m1)
    fa = np.fft.rfft(a, n=m1)
    conv = np.fft.irfft(fa * fa, n=m1)[:2*a.size - 1]
    mu = conv.mean()
    c = conv - mu
    # compute FFTs for gradient
    fc = np.fft.rfft(c, n=m2)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=m2)
    corr = np.fft.irfft(fc * fa_rev, n=m2)
    g = 4.0 * corr[a.size - 1 : a.size - 1 + a.size]  # n entries
    # multiplicative update
    a_new = a * np.exp(-eta * g)
    # ensure positivity and renormalize
    sum_a = a_new.sum()
    if sum_a > 0:
        a_new /= sum_a
    else:
        a_new = np.ones_like(a) / a.size
    return a_new, conv.max()

import numpy as np

# ------------------------------------------------------------
# Helper: compute full linear convolution using FFT (real)
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a (linear convolution) using real FFT of given size."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * a.size - 1]


# ------------------------------------------------------------
# Gradient‑descent step that reduces the L2‑error of the
# convolution against the flat target (mean of the convolution).
# ------------------------------------------------------------
def _gd_step(a: np.ndarray, m_conv: int, m_corr: int, eta: float) -> tuple:
    """One multiplicative gradient‑descent step.
    Returns (new_a, max_convolution_value)."""
    n = a.size

    # 1) convolution a*a
    fa = np.fft.rfft(a, n=m_conv)
    conv = np.fft.irfft(fa * fa, n=m_conv)[: 2 * n - 1]

    # target mean of the convolution (sum(a)=1 => mean = 1/(2n-1))
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                     # deviation from flat target

    # 2) correlation of diff with a (g_i = 4 * Σ_k diff_k * a_{k-i})
    fd = np.fft.rfft(diff, n=m_corr)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=m_corr)
    corr = np.fft.irfft(fd * fa_rev, n=m_corr)
    grad = 4.0 * corr[n - 1 : n - 1 + n]      # length‑n gradient

    # 3) multiplicative update (keeps positivity)
    new_a = a * np.exp(-eta * grad)

    # 4) renormalize to keep sum = 1 (scale‑invariant objective)
    s = new_a.sum()
    if s > 0.0:
        new_a /= s
    else:
        new_a = np.full_like(new_a, 1.0 / n)

    max_b = conv.max()
    return new_a, max_b
