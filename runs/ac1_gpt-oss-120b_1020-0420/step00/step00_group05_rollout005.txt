import numpy as np

# --------- helper functions (top‑level) -------------------------------------

def _fft_size_for_length(n: int) -> int:
    """
    Return the smallest power of two >= 2*n-1, suitable for a linear convolution.
    """
    return 1 << ((2 * n - 1).bit_length())


def max_convolution_fft(arr: np.ndarray) -> float:
    """
    Compute max value of the full convolution arr * arr using a real FFT.
    """
    n = arr.shape[0]
    sz = _fft_size_for_length(n)
    # real FFT of the padded array
    f = np.fft.rfft(arr, sz)
    conv = np.fft.irfft(f * f, sz)
    # keep only the first 2*n-1 entries (full linear convolution)
    conv = conv[: 2 * n - 1]
    return float(np.max(conv))


def fast_score(arr: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluate_sequence().
    Returns np.inf for degenerate inputs.
    """
    s = float(np.sum(arr))
    if s < 0.01:
        return np.inf
    n = arr.shape[0]
    maxb = max_convolution_fft(arr)
    return 2.0 * n * maxb / (s * s)


# --------- main entry point ------------------------------------------------

def propose_candidate(seed=42, num_steps=1200, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) whose AC1 upper
    bound (according to evaluate_sequence) is as small as possible.
    The algorithm mixes cheap parametric families with a fast block‑perturbation
    local search that uses a FFT‑based surrogate of the objective.
    """
    rng = np.random.default_rng(seed)

    # ----- determine length -------------------------------------------------
    try:
        # height_sequence_1 is provided by the harness
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(base_arr.shape[0])
    except Exception:
        # fallback if the variable is missing or malformed
        n = 256
        base_arr = np.full(n, 1.0, dtype=np.float64)

    # ----- initialise a reasonable starting point ---------------------------
    if base_arr.ndim == 1 and np.isfinite(base_arr).all():
        parent = np.clip(base_arr, 0.0, 1000.0)
        if np.sum(parent) >= 0.01:
            best = parent.copy()
        else:
            best = np.full(n, 1.0, dtype=np.float64)
    else:
        best = np.full(n, 1.0, dtype=np.float64)

    best_score = fast_score(best)

    # ----- explore a small library of parametric shapes --------------------
    for _ in range(300):
        typ = rng.integers(0, 5)
        if typ == 0:                     # uniform constant
            const = float(rng.uniform(0.1, 5.0))
            cand = np.full(n, const, dtype=np.float64)

        elif typ == 1:                   # exponential decay
            rate = float(rng.uniform(0.0005, 0.02))
            i = np.arange(n, dtype=np.float64)
            cand = np.exp(-rate * i)

        elif typ == 2:                   # geometric series q**i
            q = float(rng.uniform(0.9, 0.9999))
            i = np.arange(n, dtype=np.float64)
            cand = q ** i

        elif typ == 3:                   # linear ramp (high -> low)
            high = float(rng.uniform(0.5, 5.0))
            low = float(rng.uniform(0.0, high))
            cand = np.linspace(high, low, n, dtype=np.float64)

        else:                            # piecewise constant (2‑5 segments)
            k = int(rng.integers(2, 6))
            cuts = np.sort(rng.integers(1, n, size=k - 1))
            seg_sizes = np.diff(np.concatenate(([0], cuts, [n])))
            vals = rng.uniform(0.0, 5.0, size=k)
            cand = np.repeat(vals, seg_sizes)[:n].astype(np.float64)

        cand = np.clip(cand, 0.0, 1000.0)
        sc = fast_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ----- local search: block perturbations --------------------------------
    for _ in range(num_steps):
        # choose a random contiguous block
        block_min = max(1, n // 200)
        block_max = max(2, n // 12)
        width = int(rng.integers(block_min, block_max))
        start = int(rng.integers(0, n - width + 1))
        stop = start + width

        cand = best.copy()
        # multiplicative log‑normal perturbation of the block
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor

        # occasional symmetric averaging (mirroring)
        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        # occasional tiny additive noise on the whole vector
        if rng.random() < 0.20:
            cand += rng.normal(0.0, 0.015, size=n)

        cand = np.clip(cand, 0.0, 1000.0)
        sc = fast_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ----- optional final re‑ordering (descending) -------------------------
    if rng.random() < 0.5:
        cand = np.sort(best)[::-1]          # make it non‑increasing
        sc = fast_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ----- final safety check with the official evaluator ------------------
    try:
        official = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(official):
            raise ValueError
    except Exception:
        # If something went wrong, fall back to a simple uniform vector
        best = np.full(n, 1.0, dtype=np.float64)

    # Return a pure Python list of floats
    return [float(x) for x in best]
