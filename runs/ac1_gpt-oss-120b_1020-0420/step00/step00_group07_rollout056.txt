import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a simple deterministic candidate that provably beats the
    constant‑vector baseline.  The construction uses a linear ramp
    a_i ∝ i+1 (or scaled to respect the 0–1000 clip), which yields
    the analytical bound

        2·n·max(conv(a,a)) / (sum(a))² = (4/3)·(n+2)/(n+1) < 1.5

    for any length n ≥ 2.  The ramp is generated on the fly, uses
    only O(n) memory, and respects the evaluator’s clipping limits.
    """
    # Length of the provided parent construction, if any.
    try:
        n = int(np.shape(height_sequence_1)[0])
        if n <= 0:
            raise Exception
    except Exception:
        # Fallback length when the parent array is missing or invalid.
        n = 256

    # Scale so that the largest entry does not exceed the evaluator's cap (1000).
    scale = 1000.0 / n
    if scale > 1.0:
        scale = 1.0

    # Linear ramp: a_i = (i+1) * scale   (i = 0 … n‑1)
    seq = np.arange(1, n + 1, dtype=np.float64) * scale
    seq = np.clip(seq, 0.0, 1000.0)

    return seq.tolist()
