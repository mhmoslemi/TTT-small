def compute_score(a):
    s = np.sum(a)
    if s < 0.01: return np.inf
    # compute convolution using FFT length 2*n-1 padded to next power of 2.
    n = a.size
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    max_b = np.max(conv)
    return float(2 * n * max_b / (s*s))

   def score(arr):
        s = arr.sum()
        if s < 0.01:

        size = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(arr, n=size)
        conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
        max_b = conv.max()

   best = base.copy()

def generate_sidon_set(n, target_m, rng):
    idx = []
    sums = set()
    attempts = 0
    max_attempts = n * 10
    while len(idx) < target_m and attempts < max_attempts:
        cand = rng.integers(0, n)
        # quick reject if already present
        if cand in idx:
            attempts += 1
            continue
        # compute candidate self sum
        new_self = cand + cand
        if new_self in sums:
            attempts += 1
            continue
        conflict = False
        for i in idx:
            s = cand + i
            if s in sums:
                conflict = True
                break
        if conflict:
            attempts += 1
            continue
        # add
        for i in idx:
            sums.add(cand + i)
        sums.add(new_self)
        idx.append(cand)
        attempts = 0  # reset attempts after success
    return np.array(idx, dtype=int)
