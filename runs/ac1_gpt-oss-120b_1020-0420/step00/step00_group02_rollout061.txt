import numpy as np

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)

    # get base sequence
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception()
    except Exception:
        base = np.ones(256, dtype=np.float64)

    # start from random perturbation of base to avoid uniform shape
    current = base.copy()
    # randomize: multiply by exp(noise) to create variability
    current *= np.exp(rng.normal(0.0, 0.2, size=current.shape))
    # clip
    np.clip(current, 0.0, 1000.0, out=current)

    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # store best known

    for step in range(num_steps):
        # compute convolution only once per step
        conv = np.convolve(current, current)
        k = int(conv.argmax())
        i_start = max(0, k - (len(current) - 1))
        i_end = min(len(current) - 1, k)

        # decide operation: scaling largest in region (70%) or swap (30%)
        if rng.random() < 0.7:
            # scale down the largest element in the offending region
            region = slice(i_start, i_end + 1)
            sub = current[region]
            if sub.size == 0:
                continue
            max_idx = int(np.argmax(sub)) + i_start
            factor = np.exp(rng.normal(-0.12, 0.08))  # bias to shrink
            candidate = current.copy()
            candidate[max_idx] *= factor
        else:
            # random swap of two positions
            i, j = rng.integers(0, len(current), size=2)
            if i == j:
                continue
            candidate = current.copy()
            candidate[i], candidate[j] = candidate[j], candidate[i]

        # enforce bounds
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            current = candidate
            continue

        # occasionally accept slight worsening to escape local minima
        if rng.random() < 0.01:
            current = candidate

        # occasionally perform a global random perturbation
        if step % 200 == 0 and step > 0:
            perturb = np.exp(rng.normal(0.0, 0.25, size=current.shape))
            candidate = current * perturb
            np.clip(candidate, 0.0, 1000.0, out=candidate)
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score
                current = candidate

    # ensure we have a non-empty finite list
    if best_score == float('inf') or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]

def generate_sidon_positions(n, rng, max_trials=5):
    positions = []
    sums = set()
    indices = np.arange(n)
    # We'll attempt multiple shuffled passes to get as many positions as possible.
    for _ in range(max_trials):
        rng.shuffle(indices)
        for idx in indices:
            # skip if already in positions
            if idx in positions:
                continue
            # test sums uniqueness
            if (idx + idx) in sums:
                continue
            ok = True
            for p in positions:
                if (idx + p) in sums:
                    ok = False
                    break
            if ok:
                positions.append(idx)
                # add sums with existing (including itself)
                for p in positions:
                    sums.add(idx + p)
        # stop if we reached cap maybe sqrt(2 n)
        if len(positions) >= int(np.sqrt(2 * n)):
            break
    return positions

import numpy as np

def _sidon_positions(length, rng, max_attempts=3):
    """
    Greedy construction of a set of indices whose pairwise sums are all distinct.
    Returns a list of positions (integers) suitable for placing high spikes.
    """
    best = []
    for _ in range(max_attempts):
        positions = []
        sums = set()
        order = rng.permutation(length)
        for idx in order:
            i = int(idx)
            # check that i+i and i+existing are new
            if (i + i) in sums:
                continue
            ok = True
            for p in positions:
                if (i + p) in sums:
                    ok = False
                    break
            if not ok:
                continue
            # accept
            positions.append(i)
            # update all new sums (including i+i)
            for p in positions:
                sums.add(i + p)
        if len(positions) > len(best):
            best = positions
        # early stop if we already hit the known sqrt bound
        if len(best) >= int(np.sqrt(length)) + 5:
            break
    return best

def propose_candidate(seed=42, num_steps=800, **kwargs):
    """
    Construct a low‑autocorrelation step function.
    A base level (≈8) is set everywhere, and a small Sidon set of positions
    receives the maximal height 1000.  This guarantees that any convolution
    offset contains at most one large product, yielding a very small
    AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Obtain a valid parent length (fallback to a modest size if needed)
    # ------------------------------------------------------------------
    try:
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception()
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)

    # ------------------------------------------------------------------
    # 2.  Build the Sidon‑set based spike pattern
    # ------------------------------------------------------------------
    spike_val = 1000.0          # maximal allowed height
    base_val = 8.0              # low uniform level (must satisfy n*base^2 < spike^2)

    # generate positions where the spike will sit
    spike_positions = _sidon_positions(n, rng)

    # start from the low uniform level
    candidate = np.full(n, base_val, dtype=np.float64)
    if spike_positions:
        candidate[spike_positions] = spike_val

    # ------------------------------------------------------------------
    # 3.  Evaluate the constructed candidate and keep the better of
    #     (a) the Sidon‑based construction and
    #     (b) the original parent sequence.
    # ------------------------------------------------------------------
    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))

    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent
        best_score = parent_score

    # ------------------------------------------------------------------
    # 4.  Light local search: try a few random swaps or small reductions
    #     of the biggest spike to possibly improve the bound further.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # random modification choice
        if rng.random() < 0.6 and spike_positions:
            # shrink the largest spike slightly
            idx = spike_positions[rng.integers(len(spike_positions))]
            factor = np.exp(rng.normal(-0.15, 0.05))  # <1 on average
            cand = best.copy()
            cand[idx] = np.clip(cand[idx] * factor, 0.0, 1000.0)
        else:

            i, j = rng.integers(0, n, size=2)
            if i == j:

            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]

            best_score = sc

    # 5.  Safety fallback: if everything failed, return a simple uniform
    # ------------------------------------------------------------------
    if best_score == float('inf') or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
