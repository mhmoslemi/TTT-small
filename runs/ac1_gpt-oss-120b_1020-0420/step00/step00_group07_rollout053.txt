    best = None
    best_score = np.inf
    for cand in candidates:
        try:
            sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    if best is None:
        # Should never happen, but keep a safe fallback.
        best = uniform.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Simple stochastic local search: move a small amount of mass
    # from one index to another while keeping the total sum unchanged.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Transfer up to 10 % of the donor's value.
        delta = best[i] * rng.random() * 0.1
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # Enforce the problem’s hard bounds.
        np.clip(cand, 0.0, 1000.0, out=cand)

        try:
            sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # Return a plain Python list as required by the harness.
    return best.tolist()
