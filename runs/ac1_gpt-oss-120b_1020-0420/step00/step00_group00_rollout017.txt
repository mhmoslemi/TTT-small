import numpy as np

# The evaluation function is provided by the harness.
# from evaluator import evaluate_sequence   # (already available)

# ----------------------------------------------------------------------
# Helper to build a candidate consisting of a few high spikes (value 1000)
# and zero elsewhere.
def build_spike_candidate(length: int, num_spikes: int, rng: np.random.Generator) -> np.ndarray:
    """Return a length‑`length` array with `num_spikes` entries set to 1000."""
    cand = np.zeros(length, dtype=np.float64)
    # choose distinct positions for the spikes
    idx = rng.choice(length, size=num_spikes, replace=False)
    cand[idx] = 1000.0
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, num_trials: int = 250, **kwargs) -> list:
    """
    Generate a non‑negative step‑function that aims to minimise the
    AC1 upper bound evaluated by ``evaluate_sequence``.

    The strategy is to place a handful of large spikes (value 1000) at
    random positions, keeping the number of spikes modest so that the
    maximal convolution value stays low.  Random trials are performed
    until a satisfactory score is found or the budget is exhausted.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the problem.  ``height_sequence_1`` is
    # supplied by the harness; fall back to a modest size if absent.
    try:
        seq_src = np.asarray(height_sequence_1, dtype=np.float64)
        if seq_src.ndim != 1 or seq_src.size == 0 or not np.isfinite(seq_src).all():
            raise ValueError
        n = seq_src.size
        parent = np.clip(seq_src, 0.0, 1000.0)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Start with the given parent as a valid fallback candidate.
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Target that we would like to beat (the published benchmark).
    TARGET_SCORE = 1.50274349

    # Choose a reasonable number of spikes.
    # Theory suggests ~sqrt(2n) spikes give a good trade‑off.
    # We allow a small random variation around that value.
    base_spikes = int(np.sqrt(2.0 * n))
    min_spikes = max(1, base_spikes - 10)
    max_spikes = base_spikes + 40

    # ------------------------------------------------------------------
    for _ in range(num_trials):
        # Randomly decide how many spikes this trial will have.
        m = int(rng.integers(min_spikes, max_spikes + 1))
        # Build the candidate.
        cand = build_spike_candidate(n, m, rng)

        # Evaluate precisely using the provided evaluator.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score
            # Early exit if we already beat the benchmark.
            if best_score <= TARGET_SCORE:
                break

    # As a final safeguard, if for any reason we did not obtain a
    # finite score, fall back to the original parent.
    if not np.isfinite(best_score):
        best_seq = parent.copy()

    return [float(v) for v in best_seq]
