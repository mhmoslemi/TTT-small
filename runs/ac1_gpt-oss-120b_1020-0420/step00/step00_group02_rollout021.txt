import numpy as np

def make_block_perturbation(base, rng):
    """Generate a modest O(n) block perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width between n/200 and n/12 (inclusive)
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal tweak
    factor = float(np.exp(rng.normal(0.0, 0.05)))
    cand[start:stop] *= factor
    # occasional symmetric smoothing
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
