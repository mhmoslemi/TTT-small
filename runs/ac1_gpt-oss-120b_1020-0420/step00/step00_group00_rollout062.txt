import numpy as np

def generate_power_law(n, p, offset=0.0):
    # idx from offset+1 to offset+n
    idx = np.arange(1 + offset, n + offset + 1, dtype=np.float64)
    # avoid division by zero; offset ensures idx>0
    return idx ** (-p)

def propose_candidate(seed=42, num_steps=5000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim ==1 and parent_arr.size>0:
            n = parent_arr.size
        else:
            n = 256
    except NameError:
        n = 256
    # initialize best as uniform
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    # Search over power-law exponents
    # Use a range of p between 0.3 and 0.8 (inclusive)
    p_values = np.linspace(0.3, 0.8, 51)  # 51 values
    for p in p_values:
        candidate = generate_power_law(n, p)
        # Clip to ensure bounds
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    # Additional random local perturbations to escape local minima
    for _ in range(num_steps):
        # Pick a random segment width proportional to n
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        # Apply a multiplicative perturbation drawn log-normal
        factor = float(np.exp(rng.normal(0.0, 0.05)))  # small perturbation
        candidate = best.copy()
        candidate[start:stop] *= factor
        # occasionally smooth by averaging with reversed
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# Helper: count ordered pairs (i,j) with i+j = k for a binary mask
# ------------------------------------------------------------
def _pair_counts(selected, n):
    """
    selected : set of indices (0 … n-1) where the coefficient is 1
    returns   : np.ndarray of length 2*n-1 containing ordered‑pair counts
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    idx = list(selected)
    L = len(idx)
    for a in range(L):
        i = idx[a]
        cnt[2 * i] += 1                     # (i,i)
        for b in range(a + 1, L):
            j = idx[b]
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt


def _bound_from_selected(selected, n):
    """
    Fast bound for a binary vector whose ones are at `selected`.
    """
    m = len(selected)
    if m == 0:
        return float("inf")
    cnt = _pair_counts(selected, n)
    max_c = int(cnt.max())
    # 2*n*max_c / (sum a_i)^2 ;  sum a_i = m because each entry is 1
    return float(2.0 * n * max_c / (m * m))
