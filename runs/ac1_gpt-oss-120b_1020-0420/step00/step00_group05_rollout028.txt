import numpy as np

# The evaluator `evaluate_sequence` is provided by the harness.


def _gaussian_candidate(length: int, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative Gaussian‑shaped sequence of a given length.

    The values are optionally scaled by a small random factor to avoid exact symmetry.
    """
    centre = (length - 1) / 2.0
    idx = np.arange(length, dtype=np.float64)
    # Gaussian (unnormalised); values lie in (0, 1].
    vals = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    # Random mild scaling does not affect the objective (it cancels out),
    # but breaks perfect symmetry which can slightly improve the bound.
    scale = rng.uniform(0.8, 1.2)
    vals *= scale
    return np.clip(vals, 0.0, 1000.0)


def propose_candidate(seed: int = 42, **kwargs):
    """Return a coefficient list whose AC1 upper bound is < 2.

    The routine builds a wide Gaussian‑shaped step function and
    refines the width by a modest random search.  It falls back to the
    supplied parent `height_sequence_1` if that is available.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the working length `n` and obtain a parent sequence if
    # the harness supplied one.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise NameError
        n = int(parent.size)
    except Exception:
        # No parent provided – use a modest default length.
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Basic search parameters.
    # ------------------------------------------------------------------
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Search width range: the Gaussian must be wide enough to flatten
    # the convolution but not so wide as to become almost constant.
    sigma_low = max(1.0, n / 12.0)      # ≈ n/12  (still gives improvement)
    sigma_high = max(3.0, n / 2.0)      # ≈ n/2   (approaches the lower bound)

    # Number of random sigma trials – cheap enough for the typical size.
    trials = 300

    for _ in range(trials):
        sigma = rng.uniform(sigma_low, sigma_high)
        cand = _gaussian_candidate(n, sigma, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

    if best_score == np.inf:
        # An extremely unlikely situation – return a simple uniform vector.
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Convert to plain Python list of floats as required by the contract.
    return [float(v) for v in best_seq]
