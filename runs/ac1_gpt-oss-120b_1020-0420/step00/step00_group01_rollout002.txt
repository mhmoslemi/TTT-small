import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound given by `evaluate_sequence`.  The method builds a uniform
    sequence – the theoretical minimizer of the bound for non‑negative data –
    and then optionally accepts a tiny random perturbation only if it improves
    the score.
    """
    # ------------------------------------------------------------
    # 1.  Obtain a base vector (the provided parent if possible)
    # ------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        # Fallback: a short default vector
        base = np.ones(256, dtype=np.float64)

    # Clip to the allowed range [0, 1000]
    base = np.clip(base, 0.0, 1000.0)
    n = base.size

    # ------------------------------------------------------------
    # 2.  Build the uniform candidate (the optimal shape)
    # ------------------------------------------------------------
    avg = np.mean(base)                     # average after clipping
    candidate = np.full(n, avg, dtype=np.float64)
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Verify the candidate (should be finite); otherwise fall back to ones.
    init_score = evaluate_sequence(candidate.tolist())
    if not np.isfinite(init_score):
        candidate = np.ones(n, dtype=np.float64)
        candidate = np.clip(candidate, 0.0, 1000.0)

    # ------------------------------------------------------------
    # 3.  Optional tiny stochastic search – keep improvements only.
    # ------------------------------------------------------------
    rng = np.random.default_rng(seed)
    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))

    # A modest number of very small perturbations; the bound is tight at
    # the uniform vector, so most changes will be rejected.
    for _ in range(200):
        # Scale perturbation relative to the typical magnitude.
        scale = max(1e-8, avg * 0.01)
        pert = rng.normal(0.0, scale, n)
        trial = best + pert
        trial = np.clip(trial, 0.0, 1000.0)

        trial_score = float(evaluate_sequence(trial.tolist()))
        if np.isfinite(trial_score) and trial_score < best_score:
            best = trial
            best_score = trial_score

    # ------------------------------------------------------------
    # 4.  Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------
    return best.tolist()
