def _evaluate(arr):
    """Helper to evaluate a candidate list with the provided evaluator."""

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound used by ``evaluate_sequence``.
    The routine builds several exponential‑decay candidates and
    performs a few cheap local perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Obtain a base length and a fallback parent sequence.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
        base = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # If the supplied construction is unavailable or corrupted,
        # fall back to a modest uniform vector.
        n = 256
        base = np.ones(n, dtype=np.float64)

    best = base.copy()
    best_score = _evaluate(best)

    # ------------------------------------------------------------------
    # 2. Scan a family of exponential decays a_i = exp(-C * i / n).
    #    The constant C controls the decay; the optimal region lies
    #    near C≈1.8–2.0 for typical lengths.
    # ------------------------------------------------------------------
    # Choose a reasonably dense grid but keep total evaluations modest.
    C_grid = np.linspace(0.5, 4.0,  thirty) if (thirty := 30) else np.linspace(0.5, 4.0, 30)
    idx = np.arange(n, dtype=np.float64) / n  # i / n ∈ [0,1)
    for C in C_grid:
        # exponential decay : a_i = exp(-C * i / n)
        cand = np.exp(-C * idx)
        # Clip to the allowed range (although values are already ≤1).
        cand = np.clip(cand, 0.0, 1000.0)
        score = _evaluate(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # 3. Small stochastic block perturbations around the current best.
    #    This can escape the single‑parameter family if a modest local
    #    improvement exists.
    # ------------------------------------------------------------------
    num_blocks = 80  # modest number to keep total runtime low
    for _ in range(num_blocks):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = best.copy()
        # multiplicative log‑normal perturbation on the chosen block
        factor = float(np.exp(rng.normal(0.0, 0.07)))

        score = _evaluate(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # 4. Return the best candidate as a plain list of Python floats.
    # ------------------------------------------------------------------
