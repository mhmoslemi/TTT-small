import numpy as np

def _get_parent():
    """Return a valid clipped 1‑D numpy array to use as the initial parent."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    # enforce bounds that evaluate_sequence will also enforce
    parent = np.clip(parent, 0.0, 1000.0)
    return parent

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound evaluated by `evaluate_sequence`.  The algorithm simply
    compares the supplied parent with its uniform (constant) version, which
    is known to minimize the bound for non‑negative sequences.
    """
    rng = np.random.default_rng(seed)  # kept for possible extensions
    parent = _get_parent()
    n = parent.size

    # Uniform candidate – the optimal shape under the bound (any constant works)
    avg = float(parent.mean())
    # Guard against the degenerate zero‑sum case (evaluate_sequence rejects sum<0.01)
    if avg * n < 0.01:
        avg = 0.01 / n
    uniform = np.full(n, avg, dtype=np.float64)

    # Evaluate both candidates
    best = parent
    best_score = float(evaluate_sequence(best.tolist()))
    uniform_score = float(evaluate_sequence(uniform.tolist()))

    # Choose the better (lower) score if finite; fall back to uniform on tie
    if np.isfinite(uniform_score) and (np.isnan(best_score) or uniform_score < best_score):
        best = uniform
        best_score = uniform_score

    # As a tiny safety net, if for any reason both scores are invalid, return a small constant.
    if not np.isfinite(best_score):
        best = np.full(n, 1.0, dtype=np.float64)

    # Return as a plain Python list of floats
    return [float(x) for x in best]
