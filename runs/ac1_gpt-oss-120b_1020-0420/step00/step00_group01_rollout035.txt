import numpy as np

def propose_candidate(seed=42, num_trials=100, **kwargs):
    """Return a non‑negative sequence that (empirically) lowers the AC1 bound.

    The routine draws several independent random vectors of the same length as
    the provided ``height_sequence_1`` (or 256 as a safe fallback), evaluates
    them with the supplied ``evaluate_sequence`` and keeps the best one.
    All values lie in [0, 1], which satisfies the evaluator’s clipping bounds.
    """
    rng = np.random.default_rng(seed)

    # Determine the length to use – prefer the parent length if available.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = base.size
    except Exception:
        n = 256               # safe fallback length

    best_seq = None
    best_score = float('inf')

    # Sample a few random candidates and keep the best.
    for _ in range(num_trials):
        seq = rng.random(n)               # uniform values in [0, 1)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq
            best_score = score

    # Very unlikely that no finite candidate was found; fall back to a flat one.
    if best_seq is None:
        best_seq = np.full(n, 0.5, dtype=np.float64)

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_seq]
