import numpy as np

# ----------------------------------------------------------------------
# Helper: construct a sparse index set S ⊂ {0,…,n‑1}
# such that for every possible sum s = i+j (i,j∈S) the ordered‑pair
# count does not exceed `max_per_sum`.  The function returns the list of
# indices or None if it could not reach the requested size.
# ----------------------------------------------------------------------
def build_sparse_indices(n, target_k, max_per_sum, rng):
    # counts for each possible sum 0 … 2n‑2 (ordered pairs)
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    indices = []
    idx_set = set()

    # total number of attempts before giving up on this run
    max_trials = target_k * 500
    trials = 0
    while len(indices) < target_k and trials < max_trials:
        i = int(rng.integers(0, n))
        if i in idx_set:
            trials += 1
            continue

        # self‑pair (i,i) would increase count by 1
        s_self = i + i
        if sum_counts[s_self] + 1 > max_per_sum:
            trials += 1
            continue

        # check all pairs with already chosen indices (each adds 2 ordered pairs)
        if indices:
            existing = np.array(indices, dtype=np.int32)
            s = i + existing
            if np.any(sum_counts[s] + 2 > max_per_sum):
                trials += 1
                continue

        # --------------------------------------------------------------
        # Accept the index – update bookkeeping
        # --------------------------------------------------------------
        indices.append(i)
        idx_set.add(i)
        sum_counts[s_self] += 1
        if indices[:-1]:            # there were previous indices
            sum_counts[i + existing] += 2
        trials = 0                 # reset counter after a successful addition

    return indices if len(indices) == target_k else None

# Produce a candidate sequence using the sparse‑index construction.
# ----------------------------------------------------------------------
def generate_sparse_candidate(n, rng):
    # target: enough non‑zero entries to push the bound below 1.5.
    # With max_per_sum = 4 the theoretical score is 2*n*4/k^2.
    # k≈230 yields ≈1.05, comfortably better than the target.
    target_k = 230
    max_per_sum = 4
    best_seq = None
    best_score = np.inf

    # try a few independent attempts; keep the best one found
    for _ in range(6):
        idx = build_sparse_indices(n, target_k, max_per_sum, rng)
        if idx is None:
            continue
        seq = np.zeros(n, dtype=np.float64)
        seq[idx] = 1.0                     # unit weight for every chosen position
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq
            best_score = score
            # early exit if we already beat the target comfortably
            if best_score < 1.5:
                break

    return best_seq, best_score
