import numpy as np

def propose_candidate(seed: int = 42, **kwargs):
    """
    Returns a non‑negative step‑function sequence that (almost) minimizes the
    AC1 upper bound given by ``evaluate_sequence``.
    The construction is simple: a uniform vector of length *n* with sum
    sqrt(2n).  This achieves the theoretical lower bound of the objective
    (score = 2).  If a valid parent ``height_sequence_1`` is present, its
    score is compared and the better one is returned.
    """
    # --------------------------------------------------------------------
    # Determine the length n to use.
    # --------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        valid_parent = (
            parent_arr.ndim == 1
            and parent_arr.size > 0
            and np.isfinite(parent_arr).all()
        )
    except NameError:
        parent_arr = None
        valid_parent = False

    if valid_parent:
        n = parent_arr.size
    else:
        # fallback length – any positive integer works; choose a modest size.
        n = 256

    # --------------------------------------------------------------------
    # Uniform candidate (theoretical optimum for non‑negative vectors).
    # --------------------------------------------------------------------
    uniform_val = np.sqrt(2.0 * n) / n
    uniform = np.full(n, uniform_val, dtype=np.float64)

    # Compute its score once.
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------------
    # If a parent exists and is better, adopt it.
    # --------------------------------------------------------------------
    if valid_parent:
        parent = np.clip(parent_arr, 0.0, 1000.0)
        # Ensure a non‑trivial sum (the evaluator rejects sum < 0.01).
        if parent.sum() >= 0.01:
            parent_score = float(evaluate_sequence(parent.tolist()))
            if parent_score < best_score:
                best = parent
                best_score = parent_score

    # --------------------------------------------------------------------
    # Return the best candidate as a plain Python list of floats.
    # --------------------------------------------------------------------
    return [float(x) for x in best]
