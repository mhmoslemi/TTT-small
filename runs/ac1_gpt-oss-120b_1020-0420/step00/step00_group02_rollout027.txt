import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function whose AC1 bound is close to 4/3.
    The construction is a linear ramp from 0 to 1000 (or as high as allowed)
    over the length of the provided parent sequence `height_sequence_1`.
    If that variable is unavailable or invalid, a default length of 256 is used.
    """
    # ----------------------------------------------------------------------
    # Determine the length of the problem instance.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        ):
            n = parent.size
        else:
            raise ValueError
    except Exception:
        # Fallback length when the parent array is missing or malformed.
        n = 256

    # ----------------------------------------------------------------------
    # Build a linear ramp a_i = α * i, clipped to [0, 1000].
    # ----------------------------------------------------------------------
    if n > 1:
        scale = 1000.0 / (n - 1)          # α so that the last entry is exactly 1000
        ramp = scale * np.arange(n, dtype=np.float64)
    else:
        ramp = np.array([0.0], dtype=np.float64)

    ramp = np.clip(ramp, 0.0, 1000.0)

    # ----------------------------------------------------------------------
    # Evaluate the ramp; keep a uniform fallback just in case.
    # ----------------------------------------------------------------------
    try:
        ramp_score = float(evaluate_sequence(ramp.tolist()))
    except Exception:
        ramp_score = float('inf')

    # Uniform sequence (all entries equal) is a guaranteed valid candidate.
    uniform = np.full(n, 1000.0 / n, dtype=np.float64)
    uniform = np.clip(uniform, 0.0, 1000.0)
    try:

    except Exception:
        uniform_score = float('inf')

    best = ramp if ramp_score <= uniform_score else uniform

    # ----------------------------------------------------------------------
    # Return the best sequence as a plain Python list of floats.
    # ----------------------------------------------------------------------
