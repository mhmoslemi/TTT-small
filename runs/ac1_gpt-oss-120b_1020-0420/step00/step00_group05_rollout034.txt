def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # Initialize parent
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        # fallback: uniform random
        base = rng.uniform(0.1, 10.0, size=kwargs.get('length', 256))
    # Validate base
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)
    # Clip
    base = np.clip(base, 0.0, 1000.0)
    # If sum is zero, set small uniform
    if base.sum() < 0.01:
        base = np.full_like(base, 1.0)
    current = base.copy()
    current_score = float(evaluate_sequence(current.tolist()))
    best = current.copy()
    best_score = current_score

    n = current.size
    # parameters
    max_steps = int(num_steps)
    # Temperature schedule
    T0 = best_score * 0.5 if best_score>0 else 0.1
    Tmin = 1e-6
    sigma_start = 0.3
    sigma_end = 0.01

    for step in range(max_steps):
        # temperature linear decay
        T = T0 * (1 - step / max_steps) + Tmin
        # sigma linear decay
        sigma = sigma_start * (1 - step / max_steps) + sigma_end

        # propose candidate
        move_type = rng.random()
        cand = current.copy()
        if move_type < 0.6:
            # block scaling
            width_low = max(1, n // 200)
            width_high = max(2, n // 12)
            start = rng.integers(0, n - width_low + 1)
            width = rng.integers(width_low, width_high)
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:stop] *= factor
        elif move_type < 0.8:
            # global scaling: could induce clipping
            factor = float(np.exp(rng.normal(0.0, sigma * 0.5)))
            cand *= factor
