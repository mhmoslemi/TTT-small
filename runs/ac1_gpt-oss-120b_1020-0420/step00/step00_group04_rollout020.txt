def generate_sidon_set(n, target_size):
    selected = []
    sums = set()
    for i in range(n):
        ok = True
        for j in selected:
            s = i + j
            if s in sums:
                ok = False
                break
        if not ok:
            continue
        # also need to check self sum i+i
        if (i+i) in sums:
            continue
        # if passes, add
        selected.append(i)
        for j in selected: # include i with all (including itself)
            sums.add(i + j)
        if len(selected) >= target_size:
            break
    return selected

import numpy as np


def _generate_sidon_set(n, target_size):
    """
    Greedy construction of a Sidon (B2) set:
    all unordered sums i+j (including i=i) are distinct.
    Returns a list of up to target_size indices in [0, n).
    """
    selected = []
    sums = set()
    for i in range(n):
        # check whether any new sum would collide
        if (i + i) in sums:
            continue
        collision = False
        for j in selected:
            if (i + j) in sums:
                collision = True
                break
        if collision:
            continue
        # i is admissible
        selected.append(i)
        for j in selected:          # now selected already contains i
            sums.add(i + j)
        if len(selected) >= target_size:
            break
    return selected


def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step sequence with a low AC1 bound."""
    # length of the problem instance (provided as height_sequence_1)
    try:

        n = base.size
    except Exception:                # fallback if the array is missing
        n = 256

    # parameters
    max_val = 1000.0                         # upper box bound
    # epsilon chosen so that n*epsilon^2 <= 2*max_val^2  (keeps background low)
    eps = int(np.floor(max_val * np.sqrt(2.0) / np.sqrt(float(n))))
    eps = max(1, eps)                        # avoid all‑zero vector
    eps = float(eps)

    # try to place O(sqrt(n)) large entries while keeping all pairwise sums distinct
    target_m = int(np.sqrt(n))               # desired number of large spikes
