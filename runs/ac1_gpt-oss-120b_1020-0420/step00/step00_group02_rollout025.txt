    # parameters
    max_iter = 2000
    step_size = 0.05  # relative to target_sum
    for it in range(max_iter):
        # compute convolution via FFT
        # using real FFT for speed
        fa = np.fft.rfft(a, n*2)  # zero-pad to length 2n
        conv = np.fft.irfft(fa * fa, n*2)[:2*n-1]  # linear convolution length 2n-1
        max_idx = int(np.argmax(conv))

        # compute gradient for max_idx
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[i[mask]] += a[j[mask]]
        # gradient descent step
        a_new = a - step_size * grad
        # enforce nonnegativity
        a_new = np.clip(a_new, 0.0, 1000.0)
        # renormalize sum to target_sum
        sum_new = a_new.sum()
        if sum_new <= 0:
            # fallback: keep old
            continue
        a_new *= target_sum / sum_new
        # evaluate
        score = float(evaluate_sequence(a_new.tolist()))

            best_score = score
            a = a_new

import numpy as np

# ------------------------------------------------------------
# Helper: linear convolution via FFT (O(n log n), linear memory)
# ------------------------------------------------------------
def linear_convolution(x, y):
    n = x.size
    L = 1 << (int(np.ceil(np.log2(2 * n - 1))))  # next power of two
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[: 2 * n - 1]
    return conv


def propose_candidate(seed=42, **kwargs):
    """
    Produce a non‑negative step‑function height sequence that improves
    on the trivial bound 2.0 for the supplied AC1 upper bound.

    The method:

        self‑convolution).

        maximal convolution entry, projecting back onto the feasible set
        (non‑negative, sum = sqrt(2·n), values ≤1000).
      * occasionally inject small random noise to escape plateaus.
    """
