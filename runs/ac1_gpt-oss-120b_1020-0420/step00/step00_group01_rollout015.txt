import numpy as np

def _arcsine_normalized(length: int) -> np.ndarray:
    """
    Return a non‑negative step function of the given length whose discrete
    distribution approximates the arcsine law.  The vector is normalized so
    that its entries sum to one; scaling does not affect the objective.
    """
    i = np.arange(length, dtype=np.float64)
    # (i+½)(N‑½‑i) is always positive for 0≤i<N, avoiding division by zero.
    denom = (i + 0.5) * (length - 0.5 - i)
    # Guard against any possible underflow (should not happen for reasonable N).
    denom = np.maximum(denom, 1e-12)
    w = 1.0 / np.sqrt(denom)
    total = w.sum()
    if total == 0.0 or not np.isfinite(total):
        w = np.ones(length, dtype=np.float64)
        total = w.sum()
    w /= total                     # now sum(w) == 1
    return np.clip(w, 0.0, 1000.0)  # respect the evaluator's bound
