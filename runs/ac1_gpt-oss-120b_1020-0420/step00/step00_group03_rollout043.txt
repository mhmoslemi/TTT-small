def make_block_perturbation(base, rng):
    candidate = base.copy()

    start = rng.integers(0, n - width + 1)

    # clip
    np.clip(candidate, 0.0, 1000.0, out=candidate)

import numpy as np

def _make_block(base, rng):
    """Return a modest O(n)‑memory perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width roughly between 1 and n/20 (covers diverse scales)
    width = int(rng.integers(1, max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal tweak
    factor = float(np.exp(rng.normal(0.0, 0.18)))
    cand[start:stop] *= factor
    # keep within evaluator's hard bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, num_steps=600, **kwargs):
    """
    Search starting from a 1/√ shape.
    Returns a list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)

    # length of the given construction
    n = int(len(height_sequence_1))

    # initial guess: a_i = 1 / sqrt(i+1)  (good AC1 behaviour)
    base = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    np.clip(base, 0.0, 1000.0, out=base)

    # validate parent – fallback to uniform if something is wrong
    if n == 0 or not np.isfinite(base).all():
        base = np.ones(n, dtype=np.float64)

    best = base
    best_score = float(evaluate_sequence(best.tolist()))

    # main local‑search loop (memory‑bounded, exact evaluation)
    for _ in range(num_steps):
        cand = _make_block(best, rng)

        # occasional global refresh: small symmetric averaging
        if rng.random() < 0.07:
            cand = 0.5 * (cand + cand[::-1])

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # final safeguard – if something went wrong, fall back to the parent
    if not np.isfinite(best_score):
        fallback = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        best = fallback if fallback.size else np.ones(256, dtype=np.float64)

    return [float(v) for v in best]
