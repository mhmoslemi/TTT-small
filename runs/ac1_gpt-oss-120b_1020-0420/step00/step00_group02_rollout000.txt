    """
    c = np.sqrt(2.0 * n) / n
    seq = np.full(n, c, dtype=np.float64)
    # Enforce the evaluator's hard limits.
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a cheap O(n) perturbation of ``base``:
    - pick a random contiguous block,
    - multiply it by a log‑normal factor,
    - optionally symmetrize,
    - renormalise the sum to sqrt(2*n),
    - clip to the allowed range.
    """
    cand = base.copy()
    n = cand.size
    # block width between 1% and 25% of the length, at least 1.
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    # log‑normal multiplicative perturbation
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor

    # optional gentle symmetry enforcement (helps keep autocorrelation flat)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])

    # renormalise to maintain the same total mass (sqrt(2*n))
    total = cand.sum()
    if total > 0.0:
        scale = np.sqrt(2.0 * n) / total
        cand *= scale

    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, num_steps: int = 200, **kwargs):
    """
    Return a list of non‑negative heights that minimises the AC1 upper bound.
    The routine starts from a uniform sequence (the provable optimum) and
    performs a modest stochastic local search while never worsening the best
    known score.  The search uses only O(n) auxiliary memory.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and initialise a uniform baseline.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = None

    if (
        parent is None
        or parent.ndim != 1
        or parent.size == 0
        or not np.isfinite(parent).all()
    ):
        n = 256
    else:
        n = int(parent.size)

    best = _uniform_sequence(n)
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # Stochastic block‑wise refinement (never accepts a worsening move).
    # ----------------------------------------------------------------------
    for _ in range(num_steps):
        cand = _make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ----------------------------------------------------------------------
    # Guarantee a valid return even if something went wrong.
    # ----------------------------------------------------------------------
    if not np.isfinite(best_score) or best_score >= np.inf:
        # fallback to the uniform sequence
        best = _uniform_sequence(n)

    # Convert to plain Python floats for the harness.
    return [float(v) for v in best]
