# the objective ourselves to avoid the O(n²) cost of np.convolve.

def _initial_sequence():
    """Return a non‑negative starting vector."""
    try:
        # height_sequence_1 is provided at runtime if a parent exists.
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        seq = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # Fallback: a modest constant vector.
        seq = np.ones(256, dtype=np.float64)
    # Normalize to keep the sum moderate (the objective is scale‑invariant).
    total = seq.sum()
    if total < 1e-12:
        seq[:] = 1.0
        total = seq.sum()
    seq = seq / total
    return seq


def _compute_score(a, conv, n):
    """Return the AC1 upper bound for a given convolution array."""
    max_b = conv.max()
    s = a.sum()
    # Guard against the tiny‑sum rule inside the official evaluator.
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))
