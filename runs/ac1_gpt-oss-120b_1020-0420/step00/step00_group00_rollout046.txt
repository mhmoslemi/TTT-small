    for _ in range(num_steps):
        # Random block width (at most n//10, at least 1)
        width = max(1, rng.integers(1, max(2, n // 10)))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.05))      # mild multiplicative change
        cand = best.copy()
        cand[start:start + width] *= factor
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
