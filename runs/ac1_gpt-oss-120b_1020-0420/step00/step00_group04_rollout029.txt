def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = height_sequence_1.size if defined else fallback n.

    best = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # candidate generators
    def gen_uniform():
        seq = np.full(n, 1.0)
        return seq

    def gen_exponential(r):
        seq = r ** np.arange(n, dtype=np.float64)
        return seq

    def gen_power(p):
        seq = (np.arange(1, n+1, dtype=np.float64) ** (-p))
        return seq

    # Evaluate a set of predetermined parameters
    # Exponential r from 0.5 to 0.9995 on log scale
    for r in np.linspace(0.5, 0.99995, 20):
        candidate = gen_exponential(r)
        # clip to bounds
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if score < best_score and np.isfinite(score):
            best = candidate
            best_score = score

    # power law
    for p in np.linspace(0.0, 3.0, 20):
        candidate = gen_power(p)
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if score < best_score and np.isfinite(score):
            best = candidate
            best_score = score

    # random Dirichlet draws
    for _ in range(200):
        # Dirichlet with concentration 0.5 maybe produce sparser?
        conc = rng.uniform(0.2, 2.0)
        raw = rng.gamma(conc, 1.0, size=n)
        candidate = raw / raw.sum()
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if score < best_score and np.isfinite(score):
            best = candidate
            best_score = score

    # local perturbations around best
    def block_perturb(arr):
        # similar to earlier but preserve sum? Not needed
        n = arr.size
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        out = arr.copy()
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        out[start:stop] *= factor
        if rng.random() < 0.35:
            out = 0.5 * (out + out[::-1])
        out = np.clip(out, 0.0, 1000.0)
        return out

    for _ in range(500):
        cand = block_perturb(best)
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score and np.isfinite(score):
            best = cand
            best_score = score

    # return as list
    return [float(x) for x in best]

    except NameError:

    best = np.full(n, 1.0, dtype=np.float64)

    if np.isfinite(best_score):

        parent = np.clip(base, 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if parent_score < best_score:
            best = parent
            best_score = parent_score

    # function to update best
    def consider(candidate):
        nonlocal best, best_score
        # clip
        cand = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    for u in np.linspace(0.1, 4.0, 30):

        seq = seq / seq.sum()
        consider(seq)

    # Power family (single exponent)
    for p in np.linspace(-2.0, 2.0, 30):

        # if p < 0, large values at start; safe
        # ensure no inf or nan

        consider(seq)

    tot = seq.sum()

def _sparse_candidate(rng, n, p):
    """Random sparse support with density p, values from Exp(1)."""
    mask = rng.random(n) < p
    if mask.sum() < 2:
        return None
    vals = rng.exponential(1.0, size=mask.sum())
    seq = np.zeros(n, dtype=np.float64)
    seq[mask] = vals
    return _normalize(seq)


def _block_perturb(arr, rng):
    """Multiplicative block perturbation with optional mirroring."""
    n = arr.size
    width = int(rng.integers(max(1, n // 300), max(2, n // 15)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    out = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.05)))
    out[start:stop] *= factor
    if rng.random() < 0.25:
        out = 0.5 * (out + out[::-1])
    tot = out.sum()
    if tot > 0:
        out = out / tot
    return np.clip(out, 0.0, 1000.0)


def _targeted_step(arr, rng):
    """Move a small fraction from the index that contributes most to the
    current peak of the auto‑convolution."""
    n = arr.size
    b = np.convolve(arr, arr)
    k = int(np.argmax(b))
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_vals = k - i_vals
    contrib = arr[i_vals] * arr[j_vals]
    if contrib.size == 0:
        return arr
    idx_max = i_vals[np.argmax(contrib)]
    if arr[idx_max] <= 0:
        return arr
    delta = arr[idx_max] * 0.1
    idx_other = rng.integers(0, n)
    while idx_other == idx_max:
        idx_other = rng.integers(0, n)
    out = arr.copy()
    out[idx_max] -= delta
    out[idx_other] += delta
    out = out / out.sum()
    return np.clip(out, 0.0, 1000.0)


def _mass_transfer_step(arr, rng):
    """Transfer a random fraction of mass from one index to another."""
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    while j == i:
        j = rng.integers(0, n)
    if arr[i] <= 0:
        return arr
    frac = rng.uniform(0.01, 0.15)
    delta = arr[i] * frac
    out = arr.copy()
    out[i] -= delta
    out[j] += delta
    return out

        base = np.ones(256, dtype=np.float64)

    # -------------------------------------------------------------------------
    best = _normalize(np.full(n, 1.0, dtype=np.float64))
    best_score = _score(best)

    # Try the supplied parent as an alternative starting point
    parent = np.clip(base, 0.0, 1000.0)
    parent_score = _score(parent)
    if parent_score < best_score and np.isfinite(parent_score):
        best = _normalize(parent)
        best_score = parent_score

    # -------------------------------------------------------------------------
    # 1. Structured families (exponential, power‑law, two‑parameter power)
    # -------------------------------------------------------------------------
    for u in np.linspace(0.1, 4.0, 30):
        cand = _exp_candidate(n, u)
        sc = _score(cand)
        if sc < best_score and np.isfinite(sc):
            best, best_score = cand, sc

    for p in np.linspace(-2.0, 2.0, 30):
        cand = _pow_candidate(n, p)
        sc = _score(cand)
        if sc < best_score and np.isfinite(sc):
            best, best_score = cand, sc

    alphas = np.linspace(-2.0, 2.0, 7)
    betas = np.linspace(-2.0, 2.0, 7)
    for a in alphas:
        for b in betas:

                continue
            sc = _score(cand)
            if sc < best_score and np.isfinite(sc):

    # -------------------------------------------------------------------------
    # 2. Randomised candidates – Dirichlet draws and sparse masks
    # -------------------------------------------------------------------------
    for _ in range(300):

            if cand is None:

            if sc < best_score and np.isfinite(sc):
                best, best_score = cand, sc

    # 3. Local refinement – block perturbations
    # -------------------------------------------------------------------------

    # -------------------------------------------------------------------------

    # -------------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # -------------------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    # Return a plain Python list of floats as required by the harness
    return [float(x) for x in best]
