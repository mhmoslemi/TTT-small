def _score(seq):
    """Helper to evaluate a candidate safely."""
    try:
        return float(evaluate_sequence(seq))
    except Exception:
        return float('inf')

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    The routine prefers a uniform sequence (which is provably optimal
    under the given constraints) but falls back to the supplied parent
    ``height_sequence_1`` when it yields a better score.
    """
    # Initialise RNG (not used in the current simple strategy)
    _ = np.random.default_rng(seed)

    # Try to use the provided parent sequence; ensure it is a 1‑D finite vector.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)

        parent = np.clip(parent, 0.0, 1000.0)

    n = parent.size

    # Uniform candidate – the mathematically optimal shape for the ratio.
    uniform = np.full(n, 1.0, dtype=np.float64)
