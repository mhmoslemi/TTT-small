import numpy as np

def _linear_ramp(length: int) -> np.ndarray:
    """
    Construct a linear ramp a_i = (i+1) * (1000 / length)  (0‑based index).
    The ramp is non‑negative, respects the 0‑1000 bound and has the
    analytical AC1‑upper‑bound  ≈ 4/3·(length+2)/(length+1) < 1.5
    for any length >= 2.
    """
    # scale so that the last entry is exactly 1000 (or as close as float permits)
    scale = 1000.0 / length
    # values from 1*scale up to length*scale
    return np.arange(1, length + 1, dtype=np.float64) * scale
