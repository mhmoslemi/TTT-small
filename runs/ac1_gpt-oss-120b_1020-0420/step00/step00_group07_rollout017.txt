import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the classic 1/sqrt(i) profile (i starts at 1)
# ----------------------------------------------------------------------
def initial_profile(length: int) -> np.ndarray:
    """Return a non‑negative vector of given length with a_i = 1/sqrt(i)."""
    # i+1 to avoid division by zero
    idx = np.arange(length, dtype=np.float64) + 1.0
    a = 1.0 / np.sqrt(idx)
    return np.clip(a, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Simple block‑wise random perturbation (as in the reference program)
# ----------------------------------------------------------------------
def perturb_block(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = vec.size
    # block width between n/200 and n/12 (inclusive)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    out = vec.copy()
    # multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    out[start:stop] *= factor
    # optional mirroring (helps symmetry)
    if rng.random() < 0.35:
        out = 0.5 * (out + out[::-1])
    return np.clip(out, 0.0, 1000.0)
