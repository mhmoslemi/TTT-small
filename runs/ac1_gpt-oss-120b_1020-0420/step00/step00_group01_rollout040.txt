import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        { w | w_i >= 0, sum w_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, num_iters=800, lr_init=0.2, beta=6.0, **kwargs):
    """Search for a non‑negative step sequence whose AC1 bound is low."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Prepare an initial vector.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        parent = np.clip(base, 0.0, 1000.0)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Edge‑heavy deterministic start (more mass near the ends)
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(idx + 1.0) + 1.0 / np.sqrt(n - idx)
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # make it a probability vector

    # Keep the best found candidate (fallback to the given parent)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Pre‑compute FFT length (enough for all needed convolutions).
    # ------------------------------------------------------------------
    L = _next_pow2(3 * n - 2)        # >= 2n-1 + n-1 = 3n-2
    offset = 2 * n - 2               # start index for gradient extraction

    # ------------------------------------------------------------------
    # Main optimisation loop (smooth‑max gradient descent on the
    # self‑convolution distribution).
    # ------------------------------------------------------------------
    lr = lr_init
    for it in range(num_iters):
        # ----- convolution a*a ------------------------------------------------
        A = np.fft.rfft(a, n=L)
        c_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]     # length 2n‑1
        max_c = c_full.max()
        score = 2.0 * n * max_c
        # keep the best seen so far
        if score < best_score:
            best = a.copy()
            best_score = score

        # ----- smooth approximation of the max (softmax) -----------------------
        # w is a probability vector over the 2n‑1 convolution entries
        w_raw = np.exp(beta * (c_full - max_c))
        w = w_raw / w_raw.sum()

        # ----- gradient of the smooth max w.r.t. a ----------------------------
        w_rev = w[::-1]                              # reverse for convolution
        W_rev = np.fft.rfft(w_rev, n=L)
        conv_rev = np.fft.irfft(A * W_rev, n=L)      # length L
        g = 2.0 * conv_rev[offset : offset + n]      # gradient (shape n)

        # ----- gradient step and projection onto the simplex ------------------
        v = a - lr * g
        a = _project_simplex(v)

        # ----- slowly decay the learning rate --------------------------------
        lr *= 0.9995

        # ----- occasional random perturbation to escape tiny plateaus ---------
        if (it + 1) % 100 == 0:
            perturb = rng.normal(0.0, 0.002, size=n)
            a = _project_simplex(a + perturb)

    # ------------------------------------------------------------------
    # Final sanity check – make sure the candidate is valid for the evaluator.
    # ------------------------------------------------------------------
    candidate = best
    # Scaling does not affect the score; keep values comfortably below 1000.
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Verify with the official evaluator; fall back to parent if something went wrong.
    final_score = float(evaluate_sequence(candidate.tolist()))
    if not np.isfinite(final_score):
        candidate = parent

    return [float(x) for x in candidate]
