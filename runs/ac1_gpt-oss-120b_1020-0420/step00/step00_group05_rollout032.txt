import numpy as np

# ----------------------------------------------------------------------
# Helper: a simple local perturbation (fallback when no useful pair is found)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return an O(n)-memory non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # modify a random contiguous block
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps to explore the space)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    Memory‑bounded heuristic that directly attacks the maximum convolution entry.
    Starts from the supplied ``height_sequence_1`` (if present), then iteratively

    """

            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

                not np.isfinite(parent).all()):

    else:

    n = parent.size

    best_score = float(evaluate_sequence(best.tolist()))
    # keep a working copy that we move around
    cur = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # iterative improvement
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # full convolution (O(n log n) via FFT inside NumPy)
        conv = np.convolve(cur, cur)
        max_idx = int(np.argmax(conv))

        # indices (i, j) that actually contribute to this maximal entry
        i_low = max(0, max_idx - (n - 1))
