    seq = scale * indices                                   # values from scale to 1000

    # Safety clipping (should already be within [0,1000])
    seq = np.clip(seq, 0.0, max_val)

    # Fallback if something went wrong (should not happen)
    if seq.sum() < 0.01:
        seq = np.full(n, 0.1, dtype=np.float64)

    # Return as plain Python list of floats.
    return [float(x) for x in seq]
