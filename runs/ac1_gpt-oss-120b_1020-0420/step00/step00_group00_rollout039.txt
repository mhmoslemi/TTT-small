import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function sequence that yields a low
    AC1 upper bound.  The construction places a fraction of entries at the
    maximum allowed value (1000) and fills the remaining positions with a
    small constant (10).  High entries are randomly scattered so that the
    convolution peak is dominated by a relatively small number of
    high‑high pair products, while the total sum is kept large, giving a
    score well below the target bound.
    """
    rng = np.random.default_rng(seed)

    # Determine the length; fall back to a modest size if the parent is missing.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    if n == 0:
        n = 256

    # Parameters
    high_val = 1000.0          # maximum allowed height
    low_val = 10.0             # modest filler height
    density = 0.20             # proportion of high entries (tunable)

    # Number of high entries – at least two to avoid degenerate cases.
    p = max(2, int(n * density))
    if p >= n:
        p = n - 1

    # Randomly choose distinct positions for the high values.
    high_pos = rng.choice(n, size=p, replace=False)

    # Build the candidate array.
    candidate = np.full(n, low_val, dtype=np.float64)
    candidate[high_pos] = high_val

    # Ensure the bounds required by the evaluator.
    candidate = np.clip(candidate, 0.0, 1000.0)

    return candidate.tolist()
