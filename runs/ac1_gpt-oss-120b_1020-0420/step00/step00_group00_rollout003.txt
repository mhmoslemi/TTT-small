    #    For any length n, a constant vector gives the lower bound 2.0.
    # ----------------------------------------------------------------------
    uniform = np.full_like(parent, 1.0, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 3. Evaluate both candidates.
    # ----------------------------------------------------------------------
    parent_score = float(evaluate_sequence(parent.tolist()))
    uniform_score = float(evaluate_sequence(uniform.tolist()))

    # Pick the better of the two as the current best.
    if uniform_score < parent_score:
        best = uniform.copy()
        best_score = uniform_score
    else:
        best = parent.copy()
        best_score = parent_score

    # ----------------------------------------------------------------------
    # 4. Simple stochastic local search (O(n) memory, cheap evaluations).
    #    It cannot beat the constant bound 2.0, but it safeguards against
    #    pathological inputs where the constant vector is not the best
    #    due to clipping at 1000.
    # ----------------------------------------------------------------------
    n = best.size
    for _ in range(num_steps):
        # Choose a random contiguous block and scale it multiplicatively.
        # Block size is between 1% and 5% of n (at least 1 element).
        blk_len = rng.integers(1, max(2, n // 20))
        start = rng.integers(0, n - blk_len + 1)
        factor = np.exp(rng.normal(0.0, 0.08))  # mild perturbation
        cand = best.copy()
        cand[start:start + blk_len] *= factor
        # Clip to the allowed range.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Exact evaluation.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ----------------------------------------------------------------------
    # 5. Return a plain Python list of floats, as required.
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
```
