import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound evaluated by the provided `evaluate_sequence` function.

    The theoretical minimum of the bound is 2, attained by any uniform
    non‑zero sequence.  We therefore construct a uniform vector of the same
    length as the given parent sequence `height_sequence_1` (if available) or
    fall back to a modest length.  The values are clipped to the allowed
    interval [0, 1000] and returned as a plain Python list.
    """
    # ------------------------------------------------------------------
    # Determine the problem dimension n.
    # ------------------------------------------------------------------
    try:
        # Use the supplied parent sequence if it looks valid.
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
    except Exception:
        # Fallback: a short uniform sequence.
        base = np.ones(256, dtype=np.float64)

    n = base.size

    # ------------------------------------------------------------------
    # Construct a uniform candidate.  Any constant >0 gives the same score,
    # because the bound 2*n*max_b/(sum_a**2) simplifies to 2 for uniform vectors.
    # Use the maximal allowed value to stay well clear of the minimal-sum guard.
    # ------------------------------------------------------------------
    uniform_value = 1.0          # any positive number works; 1.0 keeps numbers small
    candidate = np.full(n, uniform_value, dtype=np.float64)

    # Ensure the candidate obeys the problem's clipping rules.
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Optional sanity check – if for any reason the evaluator reports an
    # invalid score, fall back to the (clipped) parent vector.
    try:
        score = evaluate_sequence(candidate.tolist())
        if not np.isfinite(score):
            raise RuntimeError
    except Exception:
        candidate = np.clip(base, 0.0, 1000.0)

    # Return a plain Python list as required by the harness.
    return [float(x) for x in candidate]
