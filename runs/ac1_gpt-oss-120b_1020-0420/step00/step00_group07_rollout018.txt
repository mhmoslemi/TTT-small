import numpy as np

def solve_alpha_opt():
    """Solve e^α = 2α + 1 for the optimal α (≈1.256)."""
    alpha = 1.0
    for _ in range(20):
        f = np.exp(alpha) - (2.0 * alpha + 1.0)
        df = np.exp(alpha) - 2.0
        if df == 0.0:
            break
        alpha -= f / df
    return alpha

def generate_candidate(n, alpha):
    """Return an exponential‑decay vector a_i = r^i with r = exp(-α/n)."""
    r = np.exp(-alpha / float(n))
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx)
    # Clip for safety (values are naturally ≤1)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq

def make_block_perturbation(base, rng):
    """Small O(n) random modification of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width between ≈0.5 % and ≈5 % of the length
    lo = max(1, n // 200)
    hi = max(2, n // 20)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.3:               # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
