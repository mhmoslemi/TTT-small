def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # Base candidate: choose best geometric among a set
    best = None

    # candidate from geometric ratios
    for c in [0.5, 1.0, 2.0, 3.0, 5.0]:
        r = np.exp(-c / n)

        if np.isfinite(score) and score < best_score:

    # also consider reversed geometric (mirror)
    rev = cand[::-1]

def propose_candidate(seed=42, num_steps=5000, **kwargs):
    rng = np.random.default_rng(seed)
    # determine n
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if base_arr.ndim != 1 or base_arr.size == 0 or not np.isfinite(base_arr).all():
            n = 256
            base_arr = np.ones(n, dtype=np.float64)
    except NameError:
        n = 256
        base_arr = np.ones(n, dtype=np.float64)
    n = base_arr.size
    # Generate base candidates
    best = base_arr.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Evaluate V-shaped exponentials
    for c in [1.0, 2.0, 2.5, 3.0, 3.5, 4.0, 5.0]:
        r = 1.0 - c / n
        if r <= 0:
            continue
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        cand = r ** d.astype(np.float64)
        # ensure within bounds
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    # Also evaluate pure geometric (non V) for comparison
    for c in [0.5, 1.0, 2.0, 3.0, 5.0]:
        r = np.exp(-c / n)
        cand = r ** np.arange(n, dtype=np.float64)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Local refinement loop
    for _ in range(num_steps):
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(x) for x in best]

import numpy as np

def make_block_perturbation(base, rng):
    """Return a non‑negative O(n) block perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width: between 1/200 and 1/12 of the length
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal perturbation on the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional mirroring – harmless for the search
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _make_plateau_exponential(n, K, c):
    """
    Construct a sequence of length ``n``:
    - first ``K`` entries are 1.0,
    - the remaining ``n-K`` entries form a decreasing exponential
      a[i] = r**i   where r = exp(-c/(n-K)).
    """
    if K < 0:
        K = 0
    if K > n:
        K = n
    L = n - K
    if L == 0:                     # degenerate case: pure plateau
        return np.ones(n, dtype=np.float64)
    r = np.exp(-c / L)
    tail = r ** np.arange(L, dtype=np.float64)   # r**0 == 1
    if K == 0:
        seq = tail
    else:
        seq = np.concatenate([np.ones(K, dtype=np.float64), tail])
    return np.clip(seq, 0.0, 1000.0)
