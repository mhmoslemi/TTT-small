import numpy as np

def _construct_sidonic_positions(n, rng, trials=20):
    """Return a list of positions forming a Sidon set (distinct pair sums) with
    maximal size found in `trials` random greedy attempts."""
    best = []
    # theoretical upper bound on size of Sidon set in [0, n-1] is roughly sqrt(2*n)
    bound = int(np.sqrt(2 * n)) + 2
    for _ in range(trials):
        positions = []
        used = set()
        order = rng.permutation(n)
        for p in order:
            p = int(p)
            # check if sum with itself collides
            if (p + p) in used:
                continue
            coll = False
            for q in positions:
                if (p + q) in used:
                    coll = True
                    break
            if coll:
                continue
            # accept p
            for q in positions:
                used.add(p + q)
            used.add(p + p)
            positions.append(p)
            if len(positions) >= bound:
                # already hit theoretical bound, stop early
                break
        if len(positions) > len(best):
            best = positions
            if len(best) >= bound:
                break
    return best

def propose_candidate(seed=42, **kwargs):
    """Return a step‐function height sequence that minimizes the AC1 bound.
    It builds a Sidon set of indices so that the convolution maximum is
    1 (after unit weighting) while the sum is as large as possible."""
    rng = np.random.default_rng(seed)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256
    # search for a large Sidon set
    positions = _construct_sidonic_positions(n, rng, trials=30)
    # If we failed to find any (should not happen), fall back to uniform ones
    if not positions:
        candidate = np.ones(n, dtype=np.float64)
    else:
        candidate = np.zeros(n, dtype=np.float64)
        # unit weight; any positive value works, 1.0 keeps numbers small
        candidate[positions] = 1.0
    # Clip to the allowed range (though values already within limits)
    candidate = np.clip(candidate, 0.0, 1000.0)
    # Ensure a non‑zero sum – the construction already guarantees this,

    """Find a set of indices with all pairwise sums distinct (a Sidon set).
    Performs a few random greedy passes and keeps the largest set found.
    """
    best = []

    bound = int(np.sqrt(2 * n)) + 2

    if not positions:                         # safety fallback
        candidate = np.ones(n, dtype=np.float64)
    else:
        candidate = np.zeros(n, dtype=np.float64)
        candidate[positions] = 1.0            # any positive weight works

    # Enforce the evaluator's clipping bounds.
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Ensure the sum constraint is satisfied (should already be true).
    if candidate.sum() < 0.01:
        candidate = np.ones(n, dtype=np.float64)

    return candidate.tolist()
