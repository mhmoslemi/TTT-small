    The resulting score is 4*(n+2)/(3*(n+1)) ≈ 1.33 for n≈7000,

    """
    rng = np.random.default_rng(seed)   # not used, kept for signature compatibility

    # Try to use the supplied parent sequence; fall back to a small default.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)

        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():

    n = int(parent.size)

    # Construct the decreasing triangle (weights sum to 1).
    idx = np.arange(n, dtype=np.float64)
    tri = 2.0 * (n - idx) / (n * (n + 1))
