import numpy as np

def propose_candidate(seed=42, num_steps=0, **kwargs):
    """
    Return a simple uniform step-function that attains the theoretical lower
    bound 2 for the AC1 upper‑bound score.

    The routine prefers the provided `height_sequence_1` (if it exists) to
    preserve the original length, but replaces its values by the uniform
    average so that the convolution maximum equals the sum of squares and the
    ratio 2 · n·max_b/∑a² becomes exactly 2.  If the parent sequence is invalid
    or its sum is too small, a short fallback uniform vector is used.
    """
    # Try to obtain the parent sequence supplied by the harness.
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = arr
    except Exception:
        parent = None

    # Fallback if no usable parent is available.
    if parent is None:
        n = 256                     # arbitrary modest length
        avg = 1.0                    # any positive constant works
        uniform = np.full(n, avg, dtype=np.float64)
        return uniform.tolist()

    n = parent.size
    total = float(parent.sum())

    # Guard against a zero‑sum parent (which would make the score infinite).
    if total < 0.01:
        avg = 1.0
    else:
        avg = total / n

    # Clip to the evaluator’s admissible range.
    avg = max(0.0, min(1000.0, avg))
    uniform = np.full(n, avg, dtype=np.float64)
    uniform = np.clip(uniform, 0.0, 1000.0)

    return uniform.tolist()
