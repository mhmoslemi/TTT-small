    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))
    # Do a bounded number of trials; each uses O(n) memory.
    for _ in range(200):
        # Generate a random block scaling factor.
        width = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, max(1, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.05))
        trial = best.copy()
        trial[start:start+width] *= factor
        # Clip to allowed range.
        np.clip(trial, 0.0, 1000.0, out=trial)
        score = float(evaluate_sequence(trial.tolist()))
        if np.isfinite(score) and score < best_score:
            best = trial
            best_score = score

    return [float(x) for x in best]
