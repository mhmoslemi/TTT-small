import numpy as np
import math

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _eval_candidate(arr: np.ndarray) -> float:
    """
    Evaluate a candidate sequence using the provided evaluator.
    Returns np.inf on any invalidity.
    """
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.sum() < 0.01:
        return math.inf
    try:
        return float(evaluate_sequence(arr.tolist()))
    except Exception:
        return math.inf


def _interleaved_permutation(n: int) -> np.ndarray:
    """
    Produce indices [0, n-1, 1, n-2, 2, n-3, ...] for interleaving.
    """
    idx = np.arange(n)
    perm = np.where(idx % 2 == 0, idx // 2, n - idx // 2 - 1)
    return perm.astype(np.int64)


def _exponential_decay(n: int, c: float) -> np.ndarray:
    """
    Generate a decreasing exponential a_i = exp(-c * i / n).
    """
    return np.exp(-c * np.arange(n, dtype=np.float64) / n)


def _block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Small O(n)‑memory non‑negative perturbation of ``seq`` (same as the
    previous baseline implementation).
    """
    n = seq.size
    cand = seq.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    lowers the AC1 bound.
    """

    except Exception:

        base_arr = np.ones(n, dtype=np.float64)
    else:
        n = base_arr.size

    # ------------------------------------------------------------------
    # Initialise best candidate with the parent (fallback uniform if needed)
    # ------------------------------------------------------------------
    best = base_arr.copy()
    best_score = _eval_candidate(best)

    # ------------------------------------------------------------------
    # Helper to update the current best when a lower score is found
    # ------------------------------------------------------------------
    def _maybe_update(candidate):
        nonlocal best, best_score
        score = _eval_candidate(candidate)
        if score < best_score:
            best = candidate
            best_score = score

    # ------------------------------------------------------------------
    _maybe_update(np.ones(n, dtype=np.float64))

    # ------------------------------------------------------------------
