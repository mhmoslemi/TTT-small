import numpy as np

def conv_max_fft(seq):
    n = seq.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    conv = conv[:2*n-1]
    return float(conv.max())

def compute_ac1_score(seq):
    s = seq.sum()
    if s < 0.01:
        return np.inf
    maxb = conv_max_fft(seq)
    return 2.0 * seq.size * maxb / (s * s)

import numpy as np

# ---- Fast AC1 upper‑bound evaluation (FFT based) -----------------
def _conv_max_fft(seq: np.ndarray) -> float:
    """Return the maximum entry of the full convolution seq * seq
    using a power‑of‑two FFT (linear memory, O(n log n) time)."""
    n = seq.shape[0]
    # size of the zero‑padded FFT (next power of two ≥ 2n‑1)
    m = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    conv = conv[: 2 * n - 1]                # keep the true convolution length
    # numerical noise can make tiny negatives; clamp to 0.
    return float(conv.max() if conv.max() > 0 else 0.0)


def _ac1_score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound (same formula as evaluate_sequence) using the
    fast convolution above."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    maxb = _conv_max_fft(seq)
    return 2.0 * seq.size * maxb / (s * s)


# ---- Small random perturbation of a candidate --------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Generate a non‑negative O(n)‑memory perturbation of `base`."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (inclusive)
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    w = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = start + w
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional mirroring
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ---- Main entry point required by the harness --------------------
def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """
    Return a list of non‑negative step‑function heights that (heuristically)
    minimises the AC1 upper bound.  The search starts from the sorted version
    of the provided parent construction `height_sequence_1` and performs cheap
    local perturbations evaluated with an FFT‑based bound.
    """
    rng = np.random.default_rng(seed)

    # Load the parent construction if it exists; otherwise fall back.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)

    # Initial candidate: sorted ascending (large‑small pairing reduces correlation)
    best = np.sort(parent)
    best_score = _ac1_score(best)

    # Simple intensive local search – keep improvements only.
    for _ in range(num_steps):
        cand = _make_block_perturbation(best, rng)
        score = _ac1_score(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list of floats for the evaluator.
    return [float(v) for v in best]
