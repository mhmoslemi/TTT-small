import numpy as np

# The evaluator `evaluate_sequence` is provided in the execution environment.


def _base_sequence(n):
    """Return the classic 1/sqrt(i) step‑function (i from 1 to n)."""
    # a_i = 1 / sqrt(i)  (large at the front, slowly decaying)
    return 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))


def _score(arr):
    """Convenient wrapper for the supplied evaluator."""
    return float(evaluate_sequence(arr.tolist()))


def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step function whose AC1 upper bound is as low as possible.
    The routine starts from the 1/sqrt(i) shape, tries a handful of simple
    tail‑inflation families and finishes with a short local perturbation phase.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1.  Gather basic data and a simple fallback (the supplied parent)
    # ----------------------------------------------------------------------
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        # something is wrong with the provided parent – fall back to a uniform vector
        parent = np.ones(256, dtype=np.float64)
    n = int(parent.size)

    best_vec = parent.copy()
    best_score = _score(best_vec)

    # ----------------------------------------------------------------------
    # 2.  Core analytic family:  a_i = (1/sqrt(i+1)) * factor,
    #     where a suffix of the vector is multiplied by a constant gamma.
    # ----------------------------------------------------------------------
    base = _base_sequence(n)

    # consider the pure analytic shape as a candidate
    candidate = np.clip(base.copy(), 0.0, 1000.0)
    sc = _score(candidate)
    if sc < best_score:
        best_vec, best_score = candidate, sc

    # parameter grids
    fractions = [0.5, 0.6, 0.7, 0.8, 0.9, 0.95]           # where the tail starts
    gammas = [1, 2, 3, 5, 8, 10, 15, 20, 30, 50, 100, 200, 500, 1000]

    for frac in fractions:
        start = int(frac * n)                      # first index that is scaled
        for gamma in gammas:
            cand = base.copy()
            cand[start:] *= gamma                   # amplify the tail
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_vec, best_score = cand, sc

    # ----------------------------------------------------------------------
    # 3.  Tiny local refinement – random block perturbations around the best.
    # ----------------------------------------------------------------------
    def _perturb(vec):
        """Return a modestly perturbed copy of `vec`."""
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        out = vec.copy()
        out[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        if rng.random() < 0.2:                     # occasional gentle symmetrisation
            out = 0.5 * (out + out[::-1])
        return np.clip(out, 0.0, 1000.0)

    cur_vec = best_vec.copy()
    cur_score = best_score
    for _ in range(200):                           # modest number of refinements
        cand = _perturb(cur_vec)
        sc = _score(cand)
        if np.isfinite(sc) and sc < cur_score:
            cur_vec, cur_score = cand, sc
            if sc < best_score:
                best_vec, best_score = cand, sc

    # ----------------------------------------------------------------------
    # 4.  Return the best discovered vector as a plain Python list.
    # ----------------------------------------------------------------------
    return [float(x) for x in best_vec]
