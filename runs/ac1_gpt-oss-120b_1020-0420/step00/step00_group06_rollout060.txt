import numpy as np

def propose_candidate(seed=42, num_steps=200, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256
    # Function to evaluate sequence
    def eval_seq(seq):
        return float(evaluate_sequence(seq.tolist()))
    best_seq = None
    best_score = np.inf

    # try a few alpha values
    for alpha in (0.8, 1.0, 1.2):
        # geometric base
        r = np.exp(-alpha / n)
        base = r ** np.arange(n, dtype=np.float64)
        # ensure within bounds
        base = np.clip(base, 0.0, 1000.0)
        seq = base.copy()
        score = eval_seq(seq)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
        # local search
        for _ in range(num_steps):
            conv = np.convolve(seq, seq)
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            products = seq[i_range] * seq[j_range]
            # index with largest product
            idx = i_range[np.argmax(products)]
            partner = j_range[np.argmax(products)]
            # pick a low index to receive mass
            low_idx = int(rng.integers(n))
            # ensure we pick a low value
            attempts = 0
            while seq[low_idx] > seq[idx] * 0.5 and attempts < 5:
                low_idx = int(rng.integers(n))
                attempts += 1
            delta = min(seq[idx] * 0.05, (1000.0 - seq[low_idx]) * 0.05)
            if delta <= 0:
                break
            cand = seq.copy()
            cand[idx] -= delta
            cand[low_idx] += delta
            # optionally also adjust partner if distinct
            if partner != idx:
                delta2 = min(seq[partner] * 0.05, (1000.0 - cand[low_idx]) * 0.05)
                if delta2 > 0:
                    cand[partner] -= delta2
                    cand[low_idx] += delta2
            cand = np.clip(cand, 0.0, 1000.0)
            new_score = eval_seq(cand)
            if np.isfinite(new_score) and new_score < best_score:
                best_score = new_score
                best_seq = cand.copy()
                seq = cand  # continue search from improved sequence
            # else keep seq unchanged
    # Fallback to uniform if nothing found
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)
    return [float(x) for x in best_seq]
