import numpy as np

def propose_candidate(seed=42, num_steps=500, **kwargs):
    rng = np.random.default_rng(seed)
    # attempt to retrieve base array
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.full(256, 1.0, dtype=np.float64)
    n = base.size
    # Compute uniform candidate based on sum of base
    tot = float(base.sum())
    # Clip total per maximum value constraint
    uniform_val = max(0.0, min(1000.0, tot / n if n>0 else 1.0))
    candidate = np.full(n, uniform_val, dtype=np.float64)
    # evaluate uniform; keep as best
    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))
    # optionally try some random perturbations to possibly reduce due to rounding
    for _ in range(num_steps):
        # generate random perturbation: add small normal noise, clip
        pert = best + rng.normal(0.0, 0.01, size=n)
        np.clip(pert, 0.0, 1000.0, out=pert)
        # re-normalize sum to original sum to keep comparable
        s = pert.sum()
        if s < 0.01:
            continue
        factor = tot / s
        pert *= factor
        np.clip(pert, 0.0, 1000.0, out=pert)
        score = float(evaluate_sequence(pert.tolist()))
        if np.isfinite(score) and score < best_score:
            best = pert
            best_score = score
    return [float(x) for x in best]
