import numpy as np
import math

def _score(seq):
    """Helper to safely evaluate a candidate sequence."""
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:
        return float('inf')

def propose_candidate(seed=42, **kwargs):
    """Return a sequence of non‑negative heights with a low AC1 upper bound."""
    # Base parent (fallback) – ensure it is a 1‑D finite array.
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        parent = np.full(256, 1.0, dtype=np.float64)

    n = parent.size
    best_seq = parent.copy()
    best_score = _score(best_seq)

    # Try a few uniform candidates (different constants may give tiny rounding benefits).
    const_vals = [
        0.1, 0.2, 0.5, 1.0,
        math.sqrt(2.0), math.pi,
        10.0, 100.0, 500.0, 1000.0
    ]

    for c in const_vals:
        cand = np.full(n, float(c), dtype=np.float64)
        sc = _score(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc

    # If nothing valid was found (unlikely), fall back to a simple uniform array.
    if not np.isfinite(best_score):
        best_seq = np.full(256, 1.0, dtype=np.float64)

    # Return as a plain Python list of floats.
    return [float(x) for x in best_seq]
