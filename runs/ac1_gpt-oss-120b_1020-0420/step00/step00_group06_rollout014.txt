import numpy as np


def make_block_perturbation(base, rng):
    """Create a cheap O(n) random perturbation of a non‑negative vector."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # a random block width, never larger than the whole vector
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation of the block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetric smoothing (helps a bit on some runs)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, num_r_steps=80, local_steps=200, **kwargs):
    """
    Build a geometric‑decay sequence a_i = r**i (0 < r < 1) and
    optionally refine it with cheap block perturbations.
    Returns a list of floats suitable for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # obtain a valid length – use the supplied parent if possible
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    n = int(base.size)
    idx = np.arange(n, dtype=np.float64)          # shared index array

    best = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Coarse grid search over decay factors r = 1 - delta
    #    delta is searched logarithmically from 1e-5 up to ~0.3.
    # ------------------------------------------------------------------
    deltas = np.logspace(-5, -0.5, num=num_r_steps)  # 10**[-5 … -0.5]
    for delta in deltas:
        if delta >= 1.0:
            continue
        r = 1.0 - delta
        if r <= 0.0:
            continue
        log_r = np.log(r)
        cand = np.exp(idx * log_r)                  # a_i = r**i
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand

    # ------------------------------------------------------------------
    # 2) Refine around the best delta with a finer linear sweep.
    #    This often captures the narrow minimum near r ≈ 1.
    # ------------------------------------------------------------------
    if best is not None:
        # locate the delta that gave the current best
        best_delta = 1.0 - (np.log(best[1]) / np.log(best[0]) if best[0] > 0 else 0.0)
        # define a narrow window (±50 % of the current delta, bounded away from 0)
        low = max(best_delta * 0.5, 1e-7)
        high = best_delta * 1.5
        fine = np.linspace(low, high, num=30)
        for delta in fine:
            if delta >= 1.0:
                continue
            r = 1.0 - delta
            if r <= 0.0:
                continue
            log_r = np.log(r)
            cand = np.exp(idx * log_r)
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = cand

    # ------------------------------------------------------------------
    # 3) If no geometric candidate survived, fall back to a uniform vector.
    # ------------------------------------------------------------------
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 4) Local random improvements (few cheap steps, keeps O(n) memory).
    # ------------------------------------------------------------------
    for _ in range(local_steps):
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return as a plain Python list of floats.
    return [float(x) for x in best]
