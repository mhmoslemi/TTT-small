def build_sparse_sequence(n, target_K, C_allowed, rng):
    max_sum = 2 * n - 1
    sum_counts = np.zeros(max_sum, dtype=np.int16)  # ordered contributions count
    indices = []
    attempts = 0
    max_attempts = target_K * 1000   # arbitrary
    while len(indices) < target_K and attempts < max_attempts:
        idx = int(rng.integers(0, n))
        if idx in indices:
            attempts += 1
            continue
        # check constraints
        ok = True
        # self
        if sum_counts[2*idx] + 1 > C_allowed:
            ok = False
        else:
            # check pair sums
            for i in indices:
                s = i + idx
                if sum_counts[s] + 2 > C_allowed:
                    ok = False
                    break
        if ok:
            # update counts
            sum_counts[2*idx] += 1
            for i in indices:
                s = i + idx
                sum_counts[s] += 2
            indices.append(idx)
        attempts += 1
    if len(indices) < 2:  # need at least two for meaningful
        return None
    # construct candidate array
    a = np.zeros(n, dtype=np.float64)
    a[indices] = 1000.0
    # optional shuffle to random order
    return a

import numpy as np

# ------------------------------------------------------------
# Helper: construct a sparse non‑negative step function.
# ------------------------------------------------------------
def build_sparse_sequence(n, target_K, C_allowed, rng):
    """
    Build a length‑n vector with values 0 or 1000.
    At most C_allowed ordered contributions (i.e. a_i*a_j terms)
    are allowed for any convolution index k = i + j.
    Returns the vector (np.ndarray) or None on failure.
    """
    max_sum = 2 * n - 1               # possible values of i+j
    sum_counts = np.zeros(max_sum, dtype=np.int8)   # ordered contributions per k
    used = np.zeros(n, dtype=bool)    # which indices are already taken
    indices = []                       # chosen positions (int)

    attempts = 0
    # generous limit – enough to explore without endless loops
    max_attempts = target_K * 2000

    while len(indices) < target_K and attempts < max_attempts:
        idx = int(rng.integers(0, n))
        attempts += 1
        if used[idx]:
            continue

        # self‑pair contribution (i = j = idx)
        if sum_counts[2 * idx] + 1 > C_allowed:
            continue

        # check all new ordered pairs (idx with each existing i)
        ok = True
        for i in indices:
            s = i + idx
            if sum_counts[s] + 2 > C_allowed:   # both (i,idx) and (idx,i)
                ok = False
                break
        if not ok:
            continue

        # Accept the index: update counters
        for i in indices:
            sum_counts[i + idx] += 2
        sum_counts[2 * idx] += 1
        used[idx] = True
        indices.append(idx)

    if len(indices) < 2:            # need at least two non‑zeros to be meaningful
        return None

    a = np.zeros(n, dtype=np.float64)
    a[indices] = 1000.0
    return a
