def make_block_perturbation(base, rng):
    """Scale a random contiguous block (log‑normal factor) and possibly zero it."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width between 0.5% and 5% of the length
    lo = max(1, n // 200)
    hi = max(2, n // 20)
    width = int(rng.integers(lo, hi))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.3)))
    cand[start:stop] *= factor
    # with some probability wipe the block out (creates sparsity)
    if rng.random() < 0.12:
        cand[start:stop] = 0.0
    # optional small symmetric averaging
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])
    return cand


def make_center_edge_transfer(base, rng):
    """Move a random amount from a centre element to the two ends."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # pick a centre index (biased towards the middle)
    centre = int(rng.integers(n // 3, 2 * n // 3))
    amount = cand[centre] * rng.uniform(0.05, 0.25)  # up to 25 % of its value
    if amount > cand[centre]:
        amount = cand[centre]
    cand[centre] -= amount
    # distribute the mass to both ends (preserving total sum)
    cand[0] = min(1000.0, cand[0] + amount * 0.5)
    cand[-1] = min(1000.0, cand[-1] + amount * 0.5)
    return cand


def make_random_shape(n, rng):
    """Create a fresh candidate from a random parametric shape."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    rel = (i - centre) / centre            # -1 … +1

    typ = rng.integers(0, 5)
    if typ == 0:                           # absolute linear (U‑shaped)
        arr = np.abs(rel)
    elif typ == 1:                         # quadratic U
        arr = rel ** 2
    elif typ == 2:                         # power of absolute value
        p = rng.uniform(0.2, 3.0)
        arr = np.abs(rel) ** p
    elif typ == 3:                         # exponential decay from centre
        lam = rng.uniform(0.001, 0.02)
        arr = np.exp(-lam * np.abs(i - centre))
    else:                                  # completely random positive
        arr = rng.random(n)

    # make sure we have a non‑zero signal and give a modest scale
    arr += 1e-5
    scale = rng.uniform(0.5, 5.0)
    arr *= scale
    return arr

    """Deterministic U‑shaped candidate (distance from centre)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    dist = np.abs(i - centre) + 1e-4
    return dist
