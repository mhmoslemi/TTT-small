import numpy as np

def _generate_sidon_set(n, target, rng):
    """
    Greedy construction of a Sidon set (all pairwise sums distinct) of
    approximately ``target`` elements in {0,…,n‑1}.
    Returns a list of indices.
    """
    perm = rng.permutation(n)
    used = np.zeros(2 * n + 1, dtype=bool)          # used[s] == True  ⇔  sum s already taken
    sidon = []
    for i in perm:
        # i+i must be free as well
        if used[i + i]:
            continue
        conflict = False
        for j in sidon:
            if used[i + j]:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        for j in sidon:
            used[i + j] = True
        used[i + i] = True
        sidon.append(i)
        if len(sidon) >= target:
            break
    return sidon


def propose_candidate(seed=42, **kwargs):
    """
    Construct a step‑function sequence that attains an AC1 upper bound
    appreciably lower than 2.0 (target < 1.5027...).  The algorithm builds a
    Sidon set of indices (so that each sum i+j appears at most once) and
    fills the remaining positions with a small constant ``delta``.  This
    spreads the pairwise products uniformly, keeping the maximal convolution
    entry low while the total sum stays large.
    """
    rng = np.random.default_rng(seed)

    # Length of the target vector – use the supplied parent if present.
    try:
        n = len(height_sequence_1)          # provided by the harness
    except NameError:
        # Fallback length when the parent is unavailable.
        n = 256
        height_sequence_1 = np.ones(n, dtype=np.float64)

    # Desired size of a Sidon set (theoretical maximum ≈ sqrt(2n)).
    target_size = int(np.sqrt(2 * n)) + 1

    # Try a few random orders and keep the longest Sidon set found.
    best_set = []
    best_len = -1
    for _ in range(5):
        cand = _generate_sidon_set(n, target_size, rng)
        if len(cand) > best_len:
            best_set = cand
            best_len = len(cand)
            if best_len >= target_size:
                break

    # Base small value – √(2/n) makes the small‑small contribution ≈2.
    delta_base = np.sqrt(2.0 / n)

    # Search a few multipliers for the small value to improve the score.
    mults = [0.8, 1.0, 1.2, 1.5, 2.0]
    best_seq = None
