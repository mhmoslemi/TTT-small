    #    uniform, yielding a theoretical bound near 1.0.
    # --------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    np.clip(w, 0.0, 1000.0, out=w)               # respect the evaluator limits

    # Normalisation is not required for the score (scale‑invariant),
    # but we keep the vector non‑zero.
    if w.sum() < 1e-12:                          # degenerate fallback
        w = np.full(n, 1.0 / n)

    best = w.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------
    # 2. Very light local search – perturb a random block multiplicatively.
    #    The search is O(n) per iteration and respects the memory limits.
    # --------------------------------------------------------------
    num_steps = kwargs.get("num_steps", 200)

    for _ in range(num_steps):
        cand = best.copy()

        # random block width O(n) but far smaller than n
        width = int(rng.integers(max(1, n // 200), max(2, n // 30)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        # multiplicative log‑normal perturbation
        factor = float(np.exp(rng.normal(0.0, 0.06)))
        cand[start:stop] *= factor

        # occasional symmetrisation (helps when the distribution is already
        # close to symmetric)
        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list of floats as required.
    return [float(v) for v in best]
