import numpy as np

# fast evaluation of the AC1 upper bound (identical to the evaluator's formula)
def _score_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    # linear convolution a*a via FFT (length 2n-1)
    fa = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fa * fa, n=2 * n - 1)
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a small AC1 bound.

    The routine starts from the provided `height_sequence_1` (if valid) and
    performs a bounded‑memory stochastic local search that mixes random
    block perturbations, pairwise transfers and a gradient‑style update that
    reduces the peak of the self‑convolution.
    """
    rng = np.random.default_rng(seed)

    # ---- initialise -----------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    a = np.clip(base, 0.0, 1000.0)               # enforce hard bounds
    n = a.shape[0]

    # keep a best‑so‑far copy
    best = a.copy()
    best_score = _score_fft(best)

    # ---- auxiliary helpers ---------------------------------------------------------
    def _block_perturb(x: np.ndarray) -> np.ndarray:
        """Multiply a random block by a log‑normal factor; optional symmetrisation."""
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        out = x.copy()
        factor = float(np.exp(rng.normal(0.0, 0.15)))   # moderate variance
        out[start:start + width] *= factor
        if rng.random() < 0.2:                         # occasional symmetrisation
            out = 0.5 * (out + out[::-1])
        return np.clip(out, 0.0, 1000.0)

    def _pairwise_transfer(x: np.ndarray) -> np.ndarray:
        """Move a small amount from one index to another, preserving total mass."""
        i, j = rng.choice(n, size=2, replace=False)
        amount = x[i] * rng.random() * 0.1               # up to 10 % of source
        out = x.copy()
        out[i] = max(0.0, out[i] - amount)
        out[j] = min(1000.0, out[j] + amount)
        return out

    def _gradient_step(x: np.ndarray, step_frac: float) -> np.ndarray:
        """Reduce the contributions to the current maximal convolution index."""
        # convolution via FFT (fast, O(n log n))
        fa = np.fft.rfft(x, n=2 * n - 1)
        conv = np.fft.irfft(fa * fa, n=2 * n - 1)
        k_max = int(np.argmax(conv))

        # construct the reverse‑shifted copy that participates in the max index
        rev = np.zeros_like(x)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)              # inclusive range
        rev[idx] = x[k_max - idx]

        # gradient of the peak with respect to each coefficient
        grad = x + rev

        # move opposite to the gradient, centered around its mean
        delta = -step_frac * (grad - grad.mean())
        out = x + delta
        out = np.clip(out, 0.0, 1000.0)

        # renormalise to keep the total sum (scale‑invariant objective)
        s = out.sum()
        if s > 0.0:
            out *= (x.sum() / s)
        return out

    # ---- main optimisation loop ----------------------------------------------------
    for it in range(num_steps):
        # Choose a perturbation strategy
        op = rng.integers(0, 4)

        if op == 0:
            cand = _block_perturb(best)
        elif op == 1:
            cand = _pairwise_transfer(best)
        elif op == 2:
            # learning rate linearly decays from ~0.2 to ~0.01
            lr = 0.01 + 0.19 * (1.0 - it / num_steps)
            cand = _gradient_step(best, lr)
        else:
            # pure symmetrisation (often helpful for the AC1 problem)
            cand = 0.5 * (best + best[::-1])
            cand = np.clip(cand, 0.0, 1000.0)

        sc = _score_fft(cand)

        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

        # occasional random restart if stuck for long stretches
        if (it + 1) % 500 == 0 and it > 0 and best_score > 1.9:
            # inject a fresh perturbation of the original parent
            best = _block_perturb(a)
            best_score = _score_fft(best)

    # final safety: ensure non‑negative, finite and bounded values
    final = np.clip(best, 0.0, 1000.0)
    final = np.where(np.isfinite(final), final, 0.0)

    return [float(v) for v in final]
