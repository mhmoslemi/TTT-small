    n = len(height_sequence_1)

    a = np.full(n, 1.0, dtype=np.float64)  # sum = n

    for it in range(num_iters):
        # compute conv
        b = np.convolve(best, best)
        k = int(np.argmax(b))
        # find i range for pairs
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # random pick among these to adjust
        # compute contributions array
        i_vals = np.arange(i_start, i_end+1)
        j_vals = k - i_vals
        contrib = best[i_vals] * best[j_vals]
        # pick index with largest contribution
        max_idx = int(np.argmax(contrib))
        i = i_vals[max_idx]
        j = j_vals[max_idx]
        # choose partner index to increase: pick random l not in {i,j}
        l = int(rng.integers(0, n))
        while l == i or l == j:
            l = int(rng.integers(0, n))

        candidate[l] += delta
        candidate = np.clip(candidate, 0.0, 1000.0)
        # normalize maybe not needed because sum changes zero

# The evaluator `evaluate_sequence` is pre‑imported in the execution environment.
# `height_sequence_1` (if present) gives the desired length of the construction.

def _build_antisymmetric_sequence(n: int, rng: np.random.Generator,
                                 amplitude: float = 0.9, base: float = 1.0) -> np.ndarray:
    """
    Construct a non‑negative sequence a = base + w where
      * w is antisymmetric: w[i] = -w[n-1-i]
      * w has zero mean and entries in {+amplitude, -amplitude}
      * the signs of w in the first half are shuffled to keep prefix sums small.
    The resulting total sum is exactly n*base, guaranteeing a large denominator
    while the central convolution term is reduced by the antisymmetry.
    """
    half = n // 2                     # size of the left (and right) half
    # ensure an equal number of +1 and -1 in the left half
    pos = half // 2
    neg = half - pos
    signs = np.empty(half, dtype=np.int8)
    signs[:pos] = 1
    signs[pos:] = -1
    rng.shuffle(signs)

    w = np.empty(n, dtype=np.float64)
    w[:half] = amplitude * signs.astype(np.float64)

    if n % 2 == 1:                    # odd length – a central zero entry
        mid = half
        w[mid] = 0.0
        w[mid + 1:] = -w[:half][::-1]
    else:                             # even length – pure antisymmetry
        w[half:] = -w[:half][::-1]

    a = base + w                       # a[i] ∈ [base‑amplitude, base+amplitude]
    a = np.clip(a, 0.0, 1000.0)        # final safety clamp
    return a
