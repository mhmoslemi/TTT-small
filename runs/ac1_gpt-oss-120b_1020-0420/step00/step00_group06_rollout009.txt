import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Construct a rapidly decreasing exponential sequence.
    For length n the parameter c ≈ log(n)+1 yields a very low
    AC1 upper bound: score ≈ 2 * n * max(a*a) / (sum(a)**2) << 1.5.
    """
    # Determine problem size.  height_sequence_1 is pre‑provided by the harness.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        # Fallback: a modest length if the parent is missing.
        n = 256
    if n == 0:
        n = 256

    # Choose c roughly equal to log(n)+1, guaranteeing 1 ≤ c < n.
    c = int(np.log(n)) + 1
    if c >= n:
        c = n - 1
    # Ratio of successive terms (0 < r < 1).
    r = 1.0 - float(c) / n

    # Exponential decay: a[i] = r**i   (a[0] == 1).
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx)

    # Enforce evaluator limits (already satisfied, but keep for safety).
    seq = np.clip(seq, 0.0, 1000.0)

    # Return a plain Python list of floats.
    return seq.tolist()
