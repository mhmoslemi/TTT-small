def _binary_candidate(rng: np.random.Generator, n: int, prob: float) -> np.ndarray:
    """Generate a non‑negative binary (0/1) candidate of length n."""
    # Using a uniform random mask; scaling does not affect the score.
    return rng.random(n) < prob

def propose_candidate(seed=42, num_trials=25, **kwargs):
    """
    Return a list of non‑negative step‑function heights that (heuristically)
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    The routine tries several random binary patterns with different densities
    and keeps the best one found.  If all attempts fail, it falls back to a
    safe parent construction.
    """

    # Determine the length of the required sequence.

        n = int(len(height_sequence_1))

        n = 256

    # Probabilities to try for binary occupancy; they span a range that

    probs = [0.25, 0.33, 0.40, 0.45, 0.50]

    best_seq = None
    best_score = np.inf

    for t in range(num_trials):
        prob = probs[t % len(probs)]
        cand_bool = _binary_candidate(rng, n, prob)
        # Convert to float array (0.0 / 1.0)
        cand = cand_bool.astype(np.float64)

        # Ensure a non‑trivial sum (the evaluator rejects near‑zero sums).

        score = float(evaluate_sequence(cand.tolist()))

            best_score = score
            best_seq = cand
            # Early exit if we already beat the target bound.
            if best_score <= 1.50274349:

    # ------------------------------------------------------------------

    if best_seq is None or not np.isfinite(best_score):
