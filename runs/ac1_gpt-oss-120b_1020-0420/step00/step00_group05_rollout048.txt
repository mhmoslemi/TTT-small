import numpy as np

def make_block_perturbation(base, rng):
    """Generate a single O(n)-memory non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.shape[0]

    # choose a block width between n/200 and n/12 (inclusive)
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width

    # multiplicative log‑normal scaling of the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor

    # occasional mirroring of the whole sequence
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])

    # enforce evaluator's bounds [0, 1000]
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

        n = int(height_sequence_1.shape[0])
    except Exception:
