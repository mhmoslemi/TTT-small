def compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return np.inf
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2 * seq.size * max_b / (sum_a * sum_a))

def compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = np.sum(seq)
    if sum_a < 0.01:

    # convolution
    conv = np.convolve(seq, seq)

    n = seq.size

import numpy as np

# ------------------------------------------------------------
# Helper: compute the same score as the provided evaluator
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # full convolution, O(n log n) via FFT
    m = conv.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Catalan‑like (central binomial) sequence:
#   a_i = C(2i,i) / 4^i  for i = 0 .. n‑1
# ------------------------------------------------------------
def _catalan_seq(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        # a_{i+1} = a_i * (2i+2)(2i+1) / (4 (i+1)^2)
        a[i + 1] = a[i] * ((2 * i + 2) * (2 * i + 1)) / (4.0 * (i + 1) * (i + 1))
    return a
