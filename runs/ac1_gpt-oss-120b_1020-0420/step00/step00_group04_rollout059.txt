        cand = np.ones(n, dtype=np.float64)
        cand_sum = n
    cand = cand / cand_sum   # sum=1

    # compute initial score
    best = cand.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # For fallback use parent (height_sequence_1) also evaluate
    parent = np.clip(base, 0.0, 1000.0)
    if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
        parent = parent / parent.sum()
        cur_parent_score = float(evaluate_sequence(parent.tolist()))
        if cur_parent_score < best_score:
            best = parent
            best_score = cur_parent_score

    # parameters
    step_frac = 0.1
    decay = 0.995

    for it in range(num_iters):
        # compute convolution and its max index
        # using FFT for speed
        seq = best
        nlen = seq.size
        # convolution via FFT
        size = 1 << ((2*nlen - 1).bit_length())
        fft_seq = np.fft.rfft(seq, n=size)
        conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:2*nlen-1]
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # gradient for this lag
        g = np.zeros(nlen, dtype=np.float64)
        L = max_idx
        i_start = max(0, L - (nlen - 1))
        i_end = min(nlen - 1, L)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = L - i_vals
        g[i_vals] = seq[j_vals]   # gradient contributions

        # if gradient is all zeros (unlikely), random perturbation
        if np.all(g == 0):
            # random perturbation
            i = rng.integers(0, nlen)
            delta = step_frac * seq[i]
            candidate = seq.copy()
            candidate[i] = max(0.0, candidate[i] - delta)
            # add to random other index
            j = rng.integers(0, nlen)
            candidate[j] = min(1000.0, candidate[j] + delta)
            cand_norm = candidate / max(candidate.sum(), 1e-12)
            score = float(evaluate_sequence(cand_norm.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand_norm
                best_score = score
            continue

        # identify high and low gradient indices
        # high: largest g_i, low: smallest (including zeros)
        # take top_k and bottom_k
        top_k = max(1, nlen // 200)  # about 1% of n
        # get sorted indices
        high_inds = np.argpartition(-g, top_k-1)[:top_k]
        low_inds = np.argpartition(g, top_k-1)[:top_k]

        # choose random high and low
        i_high = rng.choice(high_inds)
        i_low = rng.choice(low_inds)
        if i_high == i_low:
            continue
        # amount to transfer, proportional to high value
        delta = step_frac * seq[i_high]
        if delta <= 0:
            continue
        candidate = seq.copy()
        candidate[i_high] -= delta
        candidate[i_low] += delta

        # ensure bounds
        candidate = np.clip(candidate, 0.0, 1000.0)
        # renormalize to keep sum 1 (scale)
        s = candidate.sum()
        if s <= 0:
            continue
        candidate = candidate / s

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            # optionally increase step size or keep same
        # decay step_frac
        step_frac *= decay
        if step_frac < 1e-5:
            step_frac = 1e-5

    # final fallback ensure non-empty and finite
    if best_score == float('inf') or best.size == 0:
        best = np.ones(256, dtype=np.float64)
    return [float(x) for x in best]
