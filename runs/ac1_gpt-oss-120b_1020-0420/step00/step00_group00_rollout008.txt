    # 1. Load and sanitise the parent sequence (if available)
    # --------------------------------------------------------------------
    try:

            raise ValueError
    except Exception:
        # Fallback to a modest length if the parent is missing/invalid.
        parent = np.ones(256, dtype=np.float64)

    # Clip to the evaluator's admissible range.
    parent = np.clip(parent, 0.0, 1000.0)

    # --------------------------------------------------------------------
    # 2. Initialise the best candidate with the parent.
    # --------------------------------------------------------------------
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------------
    # 3. Compare against a uniform sequence (theoretical lower bound of 2).
    # --------------------------------------------------------------------
    n = best.shape[0]
    uniform = np.full(n, 1.0, dtype=np.float64)  # any constant >0 works
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # --------------------------------------------------------------------

    for _ in range(200):
