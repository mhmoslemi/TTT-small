import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _initial_power_law(n: int) -> np.ndarray:
    """Power‑law  a_i = (i+1)^(-0.5)  normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** -0.5
    a /= a.sum()
    return a


def _active_constraints(conv: np.ndarray, max_val: float, rel_tol: float = 0.005):
    """
    Indices of convolution entries that are within `rel_tol` of the maximum.
    These are used as the active set of linear constraints.
    """
    return np.where(conv >= max_val * (1.0 - rel_tol))[0]


def _build_constraint_matrix(a: np.ndarray, act_idx: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A where each row corresponds to one active convolution
    index k and contains the coefficients a_i for the linear constraint
    sum_i a_i * g_{k-i} <= max_b.
    """
    n = a.size
    rows, cols, data = [], [], []
    for row, k in enumerate(act_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs from i_start .. i_end, partner index j = k - i
        for i in range(i_start, i_end + 1):
            rows.append(row)
            cols.append(k - i)
            data.append(a[i])
    return sp.coo_matrix((data, (rows, cols)), shape=(len(act_idx), n)).tocsr()

    """

    # Length of the problem (use the provided parent if possible)

        n = base.size
    except Exception:

    # ----------------------------------------------------------------------
    # 1.  Initialise with the ½‑power‑law (approximately optimal analytically)
    # ----------------------------------------------------------------------
    candidate = _initial_power_law(n)

    best_score = float(evaluate_sequence(candidate.tolist()))

    # ----------------------------------------------------------------------
    # 2.  Iterative improvement using a sparse linear programme
    # ----------------------------------------------------------------------
    for outer in range(5):                     # a handful of outer passes
        conv = np.convolve(candidate, candidate)   # full convolution
        max_b = float(conv.max())

        # pick the (near) active constraints – a small subset of rows
        act = _active_constraints(conv, max_b, rel_tol=0.005)
        if act.size == 0:

        # build the sparse A_ub matrix for the LP

        # linear programme:
