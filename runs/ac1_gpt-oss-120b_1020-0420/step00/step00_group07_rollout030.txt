import numpy as np

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """Simple local search for a low AC1 upper bound.

    The baseline is a constant (step) sequence which yields the theoretical
    minimum bound of 2.  Random block perturbations are tried and any improvement
    is kept.  If the evaluation ever fails, the best valid candidate is returned.
    """
    rng = np.random.default_rng(seed)

    # Determine the length from the provided parent if possible.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256  # fallback length

    # Baseline: uniform sequence (any constant works, 1.0 is convenient).
    best = np.full(n, 1.0, dtype=np.float64)
    best = np.clip(best, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # Local search: perturb a random contiguous block each iteration.
    for _ in range(num_steps):
        candidate = best.copy()

        # Pick a random block; keep it reasonably small to stay O(n).
        start = rng.integers(0, n)
        max_width = max(2, n // 200)          # at most a few percent of n
        width = rng.integers(1, max_width + 1)
        stop = min(n, start + width)

        # Apply a log‑normal multiplicative perturbation.
        factor = np.exp(rng.normal(0.0, 0.15))
        candidate[start:stop] *= factor
        candidate = np.clip(candidate, 0.0, 1000.0)

        # Very small sums are rejected by the evaluator.
        if candidate.sum() < 0.01:
            continue

        # Evaluate the candidate using the exact scorer.
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    # Return the best sequence found (as a plain Python list of floats).
    return [float(x) for x in best]
