def compute_bound(seq):
    a = np.asarray(seq, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return np.inf
    # Clip for safety?
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    max_b = b.max()
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Initialize from parent
    if 'height_sequence_1' in globals():

    # Optionally add small random perturbation to break symmetry
    a = a + 0.01 * rng.random(a.shape) * a
    a = np.clip(a, 0.0, 1000.0)
    # Normalize to sum 1 (scaling invariant)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_bound(best)

    # Number of iterations
    max_iter = 2000
    no_improve = 0
    for it in range(max_iter):
        # Compute convolution and find max index
        conv = np.convolve(a, a)
        k_max = int(conv.argmax())
        # Determine partner indices range for k_max
        i_start = max(0, k_max - (len(a) - 1))
        i_end = min(len(a) - 1, k_max)
        # Compute contributions of each component to the max
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = k_max - idx
        contrib = a[idx] * a[partner]
        i_max_local = int(idx[contrib.argmax()])
        j_max = k_max - i_max_local
        # Choose indices to receive mass
        # Pick two indices with smallest values not equal to i_max_local or j_max
        candidates = np.where(a < a.mean())[0]
        if candidates.size < 2:
            candidates = np.arange(len(a))
        i_recv = int(rng.choice(candidates))
        j_recv = int(rng.choice(candidates))
        # Ensure distinct and not same as donors
        while i_recv == i_max_local or i_recv == j_max:
            i_recv = int(rng.integers(len(a)))
        while j_recv == i_max_local or j_recv == j_max or j_recv == i_recv:
            j_recv = int(rng.integers(len(a)))
        # Amount to transfer: fraction of smaller donor
        donor_amount = min(a[i_max_local], a[j_max])
        eps = donor_amount * rng.uniform(0.05, 0.2)  # 5-20%
        if eps <= 0:
            # fallback random perturb
            candidate = a * np.exp(rng.normal(0.0, 0.02, size=a.shape))
            candidate = np.clip(candidate, 0.0, 1000.0)
        else:
            candidate = a.copy()
            candidate[i_max_local] -= eps
            candidate[j_max] -= eps
            candidate[i_recv] += eps / 2.0
            candidate[j_recv] += eps / 2.0
            # Clip to maintain bounds
            candidate = np.clip(candidate, 0.0, 1000.0)
            # Renormalize to sum 1 (scale invariance)
            total = candidate.sum()
            if total > 0:

        score = compute_bound(candidate)
        if score < best_score:
            best = candidate
            best_score = score
            a = candidate  # continue from improved
            no_improve = 0
        else:

import numpy as np

def _bound(seq: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative sequence."""
    a = np.clip(seq, 0.0, 1000.0)
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)               # full convolution, length 2n‑1
    return float(2.0 * n * b.max() / (s * s))


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform start with a tiny random jitter (breaks symmetry)."""
    a = np.ones(n, dtype=np.float64) / n
    # a small relative noise – enough to move off the exact uniform fixed‑point
    jitter = 0.01 * rng.random(n) * a
    a = np.clip(a + jitter, 0.0, 1000.0)
    a /= a.sum()
    return a
