import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function sequence that minimizes the
    AC1 bound evaluated by the supplied `evaluate_sequence` function.
    The method starts from a family of symmetric Gaussian‑shaped candidates,
    selects the best among them and then performs a short bounded random
    local search for possible improvements.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Base sequence (fallback) – the provided parent construction.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        # If the parent is unavailable or malformed, fall back to a uniform vector.
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 1) Scan a set of symmetric Gaussian profiles.
    # ------------------------------------------------------------------
    n = best.size
    idx = np.arange(n)
    center = (n - 1) / 2.0

    # sigma values ranging from n/8 up to n/2 (inclusive) – 12 samples.
    sigma_vals = np.linspace(n / 8.0, n / 2.0, num=12)

    for sigma in sigma_vals:
        # Gaussian shape, amplitude 1 at the centre.
        diff = (idx - center) / sigma
        cand = np.exp(-0.5 * diff * diff)

        # Enforce the problem's box constraints.
        cand = np.clip(cand, 0.0, 1000.0)

        # The ratio is scale‑invariant, but the evaluator rejects
        # sequences with sum < 0.01.
        if cand.sum() < 0.01:
            continue

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # 2) Light‑weight random local search around the best candidate.
    #    This is deliberately limited to keep runtime modest.
    # ------------------------------------------------------------------
    # Parameters for block perturbations – similar to the reference program.
    min_width = max(1, n // 200)
    max_width = max(2, n // 12) + 1            # upper bound for integers (exclusive)

    for _ in range(200):
        cand = best.copy()

        width = int(rng.integers(min_width, max_width))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        # multiplicative Gaussian perturbation of a contiguous block
        factor = float(np.exp(rng.normal(0.0, 0.07)))
        cand[start:stop] *= factor

        # occasional symmetrisation – helps keep the shape smooth
        if rng.random() < 0.3:

        cand = np.clip(cand, 0.0, 1000.0)

        if cand.sum() < 0.01:
            continue

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return the best found sequence as a plain Python list of floats.
    return best.astype(float).tolist()
