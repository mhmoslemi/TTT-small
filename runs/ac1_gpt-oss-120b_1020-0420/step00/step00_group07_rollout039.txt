        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

    # also treat randomly generated candidate from base uniform random.
    for _ in range(num_iters // 2):

        score = float(evaluate_sequence(cand.tolist()))

import numpy as np

# Lower bound we aim to beat (used only for early stopping)
_TARGET_SCORE = 1.50274349


def propose_candidate(seed=42, num_trials=2000, **kwargs):
    """
    Return a list of non‑negative coefficients that (heuristically) minimise the
    AC1 upper bound evaluated by the supplied ``evaluate_sequence`` function.
    The search builds on the known near‑optimal shape a_i = 1/sqrt(i+1) and
    explores random permutations of that shape.
    """
    rng = np.random.default_rng(seed)

    # Length of the construction – use the provided parent if available.
    try:
        n = int(height_sequence_1.shape[0])
        parent = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    except Exception:          # fallback for safety, should not happen.
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # -----------------------------------------------------------------
    # Baseline candidates (guaranteed valid fall‑backs)
    # -----------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)          # uniform 1.0
    best_score = float(evaluate_sequence(best.tolist()))

    # The given parent construction – keep it if it is better.
    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # A deterministic decreasing shape: a_i = 1 / sqrt(i+1)
    idx = np.arange(1, n + 1, dtype=np.float64)
    base_shape = 1.0 / np.sqrt(idx)
    base_shape = np.clip(base_shape, 0.0, 1000.0)
    shape_score = float(evaluate_sequence(base_shape.tolist()))
    if np.isfinite(shape_score) and shape_score < best_score:
        best = base_shape.copy()
        best_score = shape_score

    # -----------------------------------------------------------------
    # Randomised search – permute the decreasing shape.
    # -----------------------------------------------------------------
    for _ in range(num_trials):
        cand = base_shape.copy()
        rng.shuffle(cand)                     # random permutation
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            if best_score <= _TARGET_SCORE:
                break

    # -----------------------------------------------------------------
    # Additional random exploration (uniform values) – cheap fallback.
    # -----------------------------------------------------------------
    extra = max(1, num_trials // 10)
    for _ in range(extra):
        cand = rng.random(n)                  # values in [0, 1)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            if best_score <= _TARGET_SCORE:
                break

    # Return a plain Python list of floats, as required by the harness.
    return [float(x) for x in best]
