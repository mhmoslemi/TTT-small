def build_sid_set(n, m, rng):
    positions = []
    sums = set()
    max_attempts = m * 1000
    attempts = 0
    while len(positions) < m and attempts < max_attempts:
        attempts += 1
        pos = rng.integers(0, n)
        if pos in positions:
            continue
        # check new sums that would conflict
        conflict = False
        # self sum
        if pos + pos in sums:
            conflict = True
        else:
            for p in positions:
                if pos + p in sums:
                    conflict = True
                    break
        if conflict:
            continue
        # If ok, add
        positions.append(int(pos))
        # add sums
        sums.add(pos + pos)
        for p in positions[:-1]:  # previous positions
            sums.add(pos + p)
    if len(positions) < m:
        return None
    return positions
