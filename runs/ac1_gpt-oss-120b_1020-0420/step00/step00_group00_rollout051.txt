def propose_candidate(seed=42, **kwargs):
    # Determine length
    try:
        n = int(len(height_sequence_1))
        if n <= 0:
            raise Exception
    except Exception:
        n = 256

    candidate = np.linspace(1000.0, 0.0, n, dtype=np.float64)
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Evaluate candidate and fallback parent
    try:
        cand_score = float(evaluate_sequence(candidate.tolist()))

        cand_score = float('inf')

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent_arr.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:

            best_score = parent_score
    except Exception:
        pass

            if np.isfinite(parent_score) and parent_score < best_score:
                best_seq = parent_clipped
                best_score = parent_score
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # Return a plain Python list of floats as required by the harness.
    # ----------------------------------------------------------------------
    return [float(v) for v in best_seq]
