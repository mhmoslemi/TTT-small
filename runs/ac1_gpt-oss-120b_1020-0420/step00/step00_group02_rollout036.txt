    n = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
    except NameError:
        parent = None

    if n is None:
        # fallback default length maybe 1024
        n = 1024

    best_seq = None
    best_score = np.inf

    # Helper to evaluate and update best
    def consider(seq):
        nonlocal best_seq, best_score
        # Clip and ensure bounds (the evaluator does it anyway, but we may clip)
        seq = np.clip(seq, 0.0, 1000.0)
        # Evaluate
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
    # candidate 1: parent if valid
    if parent is not None:
        consider(parent)

    # candidate 2: linear ramp
    lin = np.linspace(0.0, 1000.0, n)
    consider(lin)

    # candidate 3: quadratic ramp
    t = np.linspace(0.0, 1.0, n)
    quad = (t**2) * 1000.0
    consider(quad)

    # candidate 4: cubic ramp
    cube = (t**3) * 1000.0
    consider(cube)

    # candidate 5: exponential ramp (smooth)
    exp = (1 - np.exp(-5 * t)) / (1 - np.exp(-5)) * 1000.0
    consider(exp)

    # candidate 6: sqrt ramp (concave)
    sqrt = np.sqrt(t) * 1000.0
    consider(sqrt)

    # candidate 7: sin shape normalized to max 1000
    sin = np.sin(np.pi * t) * 1000.0
    consider(sin)

    # candidate 8: random uniform
    rand = rng.random(n) * 1000.0
    consider(rand)

    # candidate 9: random block perturbation of best (few attempts)
    # local optimization
    for _ in range(5):
        if best_seq is None:
            break
        # random small block perturbation as earlier strategy but step limited
        width = max(1, n // 100)  # 1% width
        start = rng.integers(0, n - width + 1)
        stop = start + width
        pert = best_seq.copy()
        factor = np.exp(rng.normal(0.0, 0.1))
        pert[start:stop] *= factor
        consider(pert)

    # Fallback: if best_seq still None (should not happen), produce uniform small positive
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) * 1.0
    return [float(x) for x in best_seq]
