import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective using FFT (O(n log n) time)
# ----------------------------------------------------------------------
def _objective(arr: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(arr, arr)) / (sum(arr) ** 2) using
    an FFT based convolution.  The function mirrors the behaviour of
    the supplied ``evaluate_sequence`` (which clips values to
    [0,1000] and enforces a minimal sum), but works directly on the
    given array to avoid the cost of converting to a list many times.
    """
    arr = np.clip(arr, 0.0, 1000.0)          # safety – mirrors the evaluator
    n = arr.size
    s = arr.sum()
    if s < 0.01:
        return np.inf

    # size = next power of two >= 2*n-1  (FFT length)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, num_iter: int = 2000, **kwargs):
    """
    Return a list of non‑negative coefficients (length equal to the
    supplied ``height_sequence_1``) that attempts to minimise the
    AC1 upper bound.

    Strategy
    --------
    * Start from the uniform (all‑equal) sequence – it attains the
      theoretical lower bound of 2.0 for the objective.
    * If the runtime provides ``height_sequence_1`` we also try the
      normalised parent as a candidate.
    * Perform a fast stochastic search: draw many random probability
      vectors (Dirichlet‑type samples), evaluate their objective with the
      FFT helper, and keep the best one.
    * The search is limited to O(n) auxiliary memory and runs in
      sub‑second time for the typical size (≈ 3800).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the required sequence.
    # ``height_sequence_1`` is present at run time; fall back to 256.
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])      # type: ignore[name-defined]
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Baseline: the uniform sequence (sum == 1, each entry = 1/n).
    # ------------------------------------------------------------------
    best = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = _objective(best)

    # ------------------------------------------------------------------
    # Optional: try the provided parent (if it exists) after normalising.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            total = parent.sum()
            if total > 0.0:
                cand = parent / total                     # keep the shape, only scale
                score = _objective(cand)
                if np.isfinite(score) and score < best_score:
                    best = cand
                    best_score = score

    # ------------------------------------------------------------------
    # Random search – a mix of Dirichlet and log‑normal draws.
    # ------------------------------------------------------------------
    for _ in range(num_iter):
        # 80 % Dirichlet‑like (uniform gamma) draws, 20 % log‑normal draws.
        if rng.random() < 0.8:
            raw = rng.random(n)                # iid U(0,1)
        else:
            raw = np.exp(rng.normal(0.0, 1.0, size=n))  # log‑normal tail

        # Normalise to a probability vector (sum == 1).  This keeps the
        # objective scale‑invariant.
        cand = raw / raw.sum()
        score = _objective(cand)

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # Early exit if we already beat the target benchmark.
            if best_score <= 1.50274349:
                break

    # ------------------------------------------------------------------
    # Convert to the plain Python list required by the harness.
    # All values are already non‑negative and far below the 1000‑limit.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
