import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a compactly generated non‑negative step‑function that aims to
    minimise the AC1 upper bound used by `evaluate_sequence`.

    The construction follows a discrete arcsine law:
        a_i ∝ 1 / sqrt( (i+0.5) * (N-0.5-i) )
    for i = 0,…,N‑1, where N is the length of the provided
    `height_sequence_1`.  This shape makes the self‑convolution of the
    normalised distribution close to uniform, driving the objective
    value towards its theoretical minimum (~1).

    The resulting values are already within the allowed range
    [0, 1000]; they are returned as a plain Python list.
    """
    # Length of the required sequence (taken from the runtime parent).
    N = int(height_sequence_1.shape[0])

    # Discrete arcsine density (unnormalised).
    idx = np.arange(N, dtype=np.float64)
    denom = np.sqrt((idx + 0.5) * (N - 0.5 - idx))
    heights = 1.0 / denom                     # shape ≈ arcsine pdf on [0,N)

    # Safety fallback: if anything went wrong, fall back to a uniform vector.
    if not np.isfinite(heights).all() or heights.sum() < 0.01:
        heights = np.ones(N, dtype=np.float64)

    # Convert to the expected plain list of floats.
    return heights.tolist()
