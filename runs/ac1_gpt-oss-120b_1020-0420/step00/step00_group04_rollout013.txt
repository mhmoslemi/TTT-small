import numpy as np

# ----------------------------------------------------------------------
# Helper: construct a Sidon (B2) set of indices in {0,…,n-1}
# ----------------------------------------------------------------------
def _sidon_set(n: int):
    """
    Greedy construction of a B2 (Sidon) set:
    all pairwise sums i+j (including i=j) are distinct.
    Returns a list of selected indices.
    """
    pos = []
    used_sums = set()
    for cand in range(n):
        # self‑sum must be unused
        if cand + cand in used_sums:
            continue
        # check sums with already chosen positions
        conflict = False
        for p in pos:
            if cand + p in used_sums:
                conflict = True
                break
        if conflict:
            continue
        # accept this position
        for p in pos:
            used_sums.add(cand + p)
        used_sums.add(cand + cand)
        pos.append(cand)
    return pos

# Main entry point expected by the harness

def propose_candidate(seed=42, **kwargs):
    """
    Build a deterministic low‑autocorrelation candidate.
    The construction uses a Sidon set of “large” entries and a
    uniform background of small entries.  This yields a provably
    small value of the AC1 bound without any dense matrices.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Parent sequence (fallback) – already present as height_sequence_1
    # ------------------------------------------------------------------
    try:
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)   # generic fallback

    n = parent.size

    # ------------------------------------------------------------------
    # Build Sidon set and the two‑level step function
    # ------------------------------------------------------------------
    sidon = _sidon_set(n)
    L = len(sidon)

    if L == 0:                      # extremely degenerate case
        candidate = parent.copy()
    else:
        large_val = 1.0                     # arbitrary positive scale
        eps = large_val / max(L, 1)         # ensures large‑small ≈ large‑large
        candidate = np.full(n, eps, dtype=np.float64)
        candidate[sidon] = large_val
        candidate = np.clip(candidate, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Evaluate parent and the Sidon‑based candidate; keep the better one
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    cand_score = float(evaluate_sequence(candidate.tolist()))
    if np.isfinite(cand_score) and cand_score < best_score:
        best = candidate
        best_score = cand_score

    # ------------------------------------------------------------------
    # Light local refinement: occasional block perturbations
    # (kept tiny to stay within the low‑autocorrelation regime)
    # ------------------------------------------------------------------
    for _ in range(200):
        pert = best.copy()
        # choose a random contiguous block
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        factor = float(np.exp(rng.normal(0.0, 0.05)))   # modest log‑normal tweak
        pert[start:stop] *= factor

        # occasional gentle symmetrisation (helps occasionally)
        if rng.random() < 0.2:
            pert = 0.5 * (pert + pert[::-1])

        pert = np.clip(pert, 0.0, 1000.0)
        sc = float(evaluate_sequence(pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = pert
            best_score = sc

    # ------------------------------------------------------------------
    # Return a plain Python list of floats as required by the harness
    # ------------------------------------------------------------------
    return [float(x) for x in best]
