        if base_len == 0:
            base_len = 256
    except Exception:
        # Fallback length if the parent is not defined.
        base_len = 256

    # Any positive constant works; 1.0 is simple and respects the clipping limits.
    uniform_seq = np.full(base_len, 1.0, dtype=np.float64)

    # Convert to a plain Python list of floats for the evaluator.
    return [float(v) for v in uniform_seq]
