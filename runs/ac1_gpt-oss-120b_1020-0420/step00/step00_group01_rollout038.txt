def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # get parent length
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            raise Exception()
    except Exception:
        init = np.ones(256, dtype=np.float64)
    n = init.size

    # Normalize init to sum=1 probability vector
    p = init.copy()
    p = np.clip(p, 0.0, 1000.0)
    total = p.sum()
    if total < 1e-12:
        p = np.ones(n, dtype=np.float64) / n
    else:
        p = p / total

    # ensure p sum = 1.0 exactly maybe adjust small diff
    p = p / p.sum()

    # Precompute FFT length
    L = 1 << ((2*n - 1).bit_length())
    # compute current score
    def conv_max(p_vec):
        F = np.fft.rfft(p_vec, n=L)
        prod = F * F
        conv = np.fft.irfft(prod, n=L)
        # take first 2*n-1 entries
        conv = conv[:2*n-1]
        return conv.max()
    max_conv = conv_max(p)
    best_p = p.copy()
    best_val = max_conv
    # number of iterations
    num_iters = 2000
    for _ in range(num_iters):
        # propose a move
        i, j = rng.choice(n, size=2, replace=False)
        # amount to transfer
        # sample delta as a fraction of p[i]; maybe up to 10% of its value
        if p[i] == 0:
            continue
        frac = rng.uniform(0.0, 0.2)  # up to 20% of p[i]
        delta = p[i] * frac
        # Make move
        p[i] -= delta
        p[j] += delta
        # compute new score
        new_max = conv_max(p)
        if new_max < best_val - 1e-12:
            # accept and update best
            best_val = new_max
            best_p = p.copy()
        else:
            # reject move (revert)
            p[i] += delta
            p[j] -= delta
    # Build final sequence a_i = best_p * scale
    # choose scale such that max value <= 1000, prefer to scale up to use full range
    max_allowed = 1000.0
    scale = max_allowed / best_p.max()
    a = best_p * scale
    # Clip to ensure within [0,1000]
    a = np.clip(a, 0.0, max_allowed)
    # Return list
    return [float(x) for x in a]

def project_simplex(v):
    # Projects onto the probability simplex (sum to 1, nonnegative)
    # Implementation from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=42, num_iters=4000, **kwargs):

    eta0 = 0.5
    for t in range(num_iters):

import numpy as np

def project_simplex(v):
    # Project onto probability simplex (nonnegative, sum=1)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries lead to zero after projection; set to uniform.
        # This case rarely occurs; fallback to uniform.
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

import numpy as np


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Fallback: uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_max(p, L):

    F = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(F * F, n=L)
    conv = conv[: 2 * len(p) - 1]
    return float(conv.max())
