import numpy as np

def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a single O(n)-memory non‑negative perturbation of ``base``.
    The perturbation multiplies a random contiguous block by a log‑normal factor
    and optionally mirrors the whole array.
    """
    candidate = base.copy()
    n = candidate.size
    # Choose a block width between n/200 and n/12 (at least 1)
    low = max(1, n // 200)
    high = max(2, n // 12) + 1          # high is exclusive for rng.integers
    width = int(rng.integers(low, high))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # Optional symmetric smoothing
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def _initial_sequence() -> np.ndarray:
    """Return a valid starting sequence: uniform of length derived from parent or 256."""
    default_len = 256
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        # Ensure values are finite before using length
        if np.isfinite(parent).all():
            n = int(parent.size)
            # Uniform value 1.0 is safe for any length
            return np.full(n, 1.0, dtype=np.float64)
    # Fallback to a modest default length
    return np.full(default_len, 1.0, dtype=np.float64)


def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a list of non‑negative coefficients that (approximately) minimizes
    the AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # Start from a uniform baseline (optimal for the objective)
    best = _initial_sequence()
    best_score = float(evaluate_sequence(best.tolist()))

    # If a parent sequence exists, consider it as an alternative start.
    parent_arr = globals().get("height_sequence_1")
    if isinstance(parent_arr, np.ndarray) and parent_arr.ndim == 1 and parent_arr.size > 0:
        if np.isfinite(parent_arr).all():
            parent_clipped = np.clip(parent_arr, 0.0, 1000.0)
            parent_score = float(evaluate_sequence(parent_clipped.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent_clipped
                best_score = parent_score

    # Simple local search: random block perturbations with greedy acceptance.
    for _ in range(num_steps):
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
