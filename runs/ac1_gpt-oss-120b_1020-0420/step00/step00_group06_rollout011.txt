import numpy as np

def _uniform_candidate(length: int) -> np.ndarray:
    """Return a length‑n nonnegative uniform vector (value 1.0)."""
    # Value 1.0 gives sum=n, sum of squares=n, hence the optimal bound 2.
    return np.ones(length, dtype=np.float64)


def propose_candidate(seed=42, **kwargs):
    """
    Return a nonnegative step‑function (list of floats) that minimizes the
    AC1 upper bound under the evaluator's constraints.
    """
    # Determine the length from the provided parent sequence if possible.
    try:
        # ``height_sequence_1`` is supplied by the harness when a parent exists.
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise Exception
        n = parent_arr.size
    except Exception:
        # Fallback: use a modest default length.
        n = 256

    # Build the uniform candidate (value 1.0 everywhere).
    candidate = _uniform_candidate(n)

    # Verify that the candidate is valid for the evaluator.
    # (Should always be true, but we guard against unexpected issues.)
    score = evaluate_sequence(candidate.tolist())
    if not np.isfinite(score):

        candidate = np.full(n, 0.01, dtype=np.float64)

    # Return as a plain Python list of floats.
