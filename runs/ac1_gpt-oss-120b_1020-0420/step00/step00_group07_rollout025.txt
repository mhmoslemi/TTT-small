def perturb(candidate, rng):
   # Choose one block or modify random coordinates
   new = candidate.copy()
   # Determine number of indices to perturb: block size
   block_len = rng.integers(1, max(2, n // 50))
   start = rng.integers(0, n - block_len + 1)
   # Multiply block by random factor lognormal
   factor = np.exp(rng.normal(0, 0.1))
   new[start:start+block_len] *= factor
   # Optional small global smoothing by averaging with reversed
   if rng.random() < 0.1:
       new = 0.5 * (new + new[::-1])
   # Ensure within bounds
   new = np.clip(new, 0.0, 1000.0)
   return new

def compute_metric(arr):
    # compute convolution via FFT
    n = arr.size
    # next power of two length for FFT
    size = int(2**np.ceil(np.log2(2*n - 1)))
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
