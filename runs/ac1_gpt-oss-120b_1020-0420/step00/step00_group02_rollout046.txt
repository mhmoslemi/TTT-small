import numpy as np

def _candidate_score(seq):
    """Return the AC1 upper‑bound score for a NumPy array `seq`."""
    # evaluate_sequence is provided by the harness.
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimizes the
    AC1 upper bound.  The routine works with the global `height_sequence_1`
    when it exists, otherwise falls back to a uniform sequence.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------
    # 1.  Determine the problem dimension.
    # --------------------------------------------------------------------
    try:
        base = height_sequence_1          # global input, may be absent
    except NameError:
        base = None

    if (
        isinstance(base, np.ndarray)
        and base.ndim == 1
        and base.size > 0
        and np.isfinite(base).all()
    ):
        n = int(base.shape[0])
        parent = np.clip(base.astype(np.float64), 0.0, 1000.0)
    else:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # --------------------------------------------------------------------
    # 2.  Build a small portfolio of simple candidates.
    # --------------------------------------------------------------------
    candidates = []

    # (a)   the parent itself (if it is already decent)
    candidates.append(parent)

    # (b)   a fully uniform sequence – this is provably optimal for the
    #       score 2·n·‖a‖₂² / (‖a‖₁)².
    uniform = np.full(n, 1.0, dtype=np.float64)
    candidates.append(uniform)

    # (c)   symmetrised version of the parent (often reduces the max
    #       convolution at non‑zero lags).
    if parent.shape == uniform.shape:
        sym = 0.5 * (parent + parent[::-1])
        candidates.append(np.clip(sym, 0.0, 1000.0))

    # (d)   a few cheap random block perturbations of the best candidate
    #       seen so far – this keeps the search membrane alive without
    #       blowing up memory or time.
    #       (We limit the number to a small constant.)
    for _ in range(4):
        # start from the current best (will be updated later)
        start = uniform if len(candidates) == 0 else candidates[-1]
        cand = start.copy()
        width = max(1, int(rng.integers(max(1, n // 200), max(2, n // 12))))
        start_idx = int(rng.integers(0, max(1, n - width + 1)))
        stop_idx = min(n, start_idx + width)
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start_idx:stop_idx] *= factor
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        candidates.append(np.clip(cand, 0.0, 1000.0))

    # --------------------------------------------------------------------
    # 3.  Evaluate all candidates and keep the best.
    # --------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for cand in candidates:
        score = _candidate_score(cand)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand

    # --------------------------------------------------------------------
    # 4.  Safety fallback – should everything fail (unlikely).
    # --------------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best_seq]
