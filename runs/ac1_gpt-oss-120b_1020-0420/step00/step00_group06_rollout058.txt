import numpy as np

# the evaluator is pre‑imported in the harness
# evaluate_sequence(sequence: list) -> float

def _triangular_seq(n: int) -> np.ndarray:
    """
    Return a non‑negative linear decreasing sequence of length n whose sum is 1.
    This is the extremal solution of a_i = A - D*i with a_{n-1}=0.
    """
    if n < 2:
        # trivial fallback
        return np.ones(max(1, n), dtype=np.float64) / max(1, n)

    # D = 2/(n*(n-1))  makes a_{n-1}=0  and satisfies the sum=1 condition.
    D = 2.0 / (n * (n - 1))
    A = 1.0 / n + D * (n - 1) / 2.0   #  = 2/n
    idx = np.arange(n, dtype=np.float64)
    seq = A - D * idx
    # Numerical noise may produce tiny negative values at the end
    np.clip(seq, 0.0, 1000.0, out=seq)
    # Normalise explicitly to avoid tiny drift from rounding
    total = seq.sum()
    if total > 0:
        seq /= total   # keeps sum exactly 1.0 (scale‑invariant for the score)
    return seq


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative coefficients that beats the uniform AC1 bound.
    The routine compares the supplied parent (height_sequence_1) with a
    triangular decreasing construction and returns the better one.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Acquire parent if present and sensible
    # ------------------------------------------------------------------
    parent_valid = False
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_valid = True
        else:
            parent_valid = False
    except Exception:                     # height_sequence_1 not defined
        parent_valid = False

    if not parent_valid:
        # harmless uniform fallback
        parent = np.ones(256, dtype=np.float64) / 256.0

    # ------------------------------------------------------------------
    # 2. Build the triangular candidate of the same length
    # ------------------------------------------------------------------
    n = int(parent.shape[0])
    tri = _triangular_seq(n)

    # ------------------------------------------------------------------
    # 3. Evaluate both candidates using the exact evaluator
    # ------------------------------------------------------------------
    # The evaluator already clips values to [0,1000] and rescales sum,
    # so we can feed the raw lists directly.
    score_parent = float(evaluate_sequence(parent.tolist()))
    score_tri = float(evaluate_sequence(tri.tolist()))

    # ------------------------------------------------------------------
    # 4. Return the better (lower) score; keep a fallback in case
    #    both evaluations fail (they should not).
    # ------------------------------------------------------------------
    if np.isfinite(score_tri) and (not np.isfinite(score_parent) or score_tri < score_parent):
        best = tri
    else:
        best = parent

    # Ensure the final list complies with the contract (non‑empty, finite)
    if best.size == 0:
        best = np.ones(1, dtype=np.float64)

    return [float(x) for x in best]
