def make_block_perturbation(base, rng):
    candidate = np.array(base, copy=True)
    n = candidate.size
    min_width = max(1, n // 200)
    max_width = max(2, n // 12)
    width = int(rng.integers(min_width, max_width + 1))
    start = int(rng.integers(0, n - width + 1))

    candidate[start:start+width] *= factor
    if rng.random() < 0.35:

import numpy as np

def make_block_perturbation(base, rng):
    """Return an O(n)‑memory non‑negative perturbation of ``base``."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width between n/200 and n/12 (inclusive)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    # multiplicative log‑normal perturbation
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:start + width] *= factor
    # optional mirroring to keep symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
