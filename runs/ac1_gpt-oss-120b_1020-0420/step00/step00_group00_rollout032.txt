import numpy as np

def _fast_score(arr: np.ndarray, n: int, n_fft: int) -> float:
    """
    Compute 2*n*max(conv(arr,arr)) / (sum(arr)^2) using FFT.
    arr must be 1‑D, non‑negative and already clipped to [0,1000].
    """
    # convolution via FFT (real)
    A = np.fft.rfft(arr, n_fft)
    C = np.fft.irfft(A * A, n_fft)[: 2 * n - 1]
    max_b = float(C.max())
    sum_a = float(arr.sum())
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 42, num_iters: int = 800, **kwargs):
    """Return a list of coefficients with a low AC1 upper‑bound score."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Obtain an initial multiset of values.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    # Validate / clip
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)
    base = np.clip(base, 0.0, 1000.0)

    n = base.size
    # FFT length – next power of two ≥ 2n‑1
    n_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Start from a random permutation of the values.
    # ------------------------------------------------------------------
    current = rng.permutation(base)
    current_score = _fast_score(current, n, n_fft)

    best = current.copy()
    best_score = current_score

    # ------------------------------------------------------------------
    # 3. Main improvement loop: repeatedly try to lessen the
    #    coefficient that attains the current maximum.
    # ------------------------------------------------------------------
    for _ in range(num_iters):
        # Convolution of the current candidate (FFT once)
        A = np.fft.rfft(current, n_fft)
        C = np.fft.irfft(A * A, n_fft)[: 2 * n - 1]
        k = int(np.argmax(C))                     # lag with maximal value

        # Indices that actually contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_idx = k - i_idx

        # Contribution of each pair (i, k-i)
        contrib = current[i_idx] * current[j_idx]

        # Choose a pair with the largest and a pair with the smallest product
        idx_max = i_idx[np.argmax(contrib)]
        idx_min = i_idx[np.argmin(contrib)]

        # Their mirrored partners
        j_max = k - idx_max
        j_min = k - idx_min

        # Build a candidate by swapping these four positions (if distinct)
        cand = current.copy()
        cand[idx_max], cand[idx_min] = current[idx_min], current[idx_max]
        if j_max != j_min:
            cand[j_max], cand[j_min] = current[j_min], current[j_max]

        cand_score = _fast_score(cand, n, n_fft)

        # Record global best
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # Accept improvement locally
        if cand_score < current_score:
            current = cand
            current_score = cand_score
        else:
