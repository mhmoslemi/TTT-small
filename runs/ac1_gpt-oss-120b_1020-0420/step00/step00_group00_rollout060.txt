def fast_score(seq):
    n = len(seq)
    fft_len = 2*n - 1
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
```

def fast_score(seq):
    n = seq.shape[0]
    L = 2 * n - 1
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

# fast evaluation of the objective (identical to evaluate_sequence but O(n log n))
def _fast_score(seq: np.ndarray) -> float:
    n = seq.shape[0]
    L = 2 * n - 1                     # length of full convolution
    fa = np.fft.rfft(seq, n=L)        # real FFT of zero‑padded sequence
    conv = np.fft.irfft(fa * fa, n=L) # convolution via point‑wise product
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


# greedy construction of a Sidon (B2) set – all pairwise sums are distinct
def _random_sidon_positions(n: int, rng: np.random.Generator) -> list:
    order = np.arange(n, dtype=np.int64)
    rng.shuffle(order)
    pos = []
    used = set()                      # stores already taken sums i+j
    for x in order:
        # self‑sum must be free
        if (x + x) in used:
            continue
        # check all sums with previously chosen positions
        conflict = False
        for y in pos:
            if (x + y) in used:
                conflict = True
                break
        if conflict:
            continue
        # accept x – record its sums
        for y in pos:
            used.add(x + y)
        used.add(x + x)
        pos.append(int(x))
    return pos


# a tiny local‑search perturbation (kept from the reference program)
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    # random block width
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal noise on the block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring to keep a degree of symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, *,
                     max_sidon_trials: int = 120,
                     max_local_steps: int = 800,
                     **kwargs) -> list:
    """
    Search for a step‑function that lowers the AC1 upper bound.
    Strategy:
      * generate many random Sidon (B2) sets – binary support where all
        pairwise sums are distinct, guaranteeing a tiny convolution peak;
      * add a tiny uniform background (eps) to boost the denominator;
      * keep the best candidate seen;
      * if time permits, perform a few local block perturbations starting
        from the best candidate.
    Returns a list of floats suitable for evaluate_sequence.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # baseline (fallback) – the provided parent construction
    # ------------------------------------------------------------------
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                     0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    n = best_seq.size
    eps = 0.01                       # tiny uniform offset (helps the denominator)

    # ------------------------------------------------------------------
    # 1) Random Sidon‑set based candidates
    # ------------------------------------------------------------------
    for _ in range(max_sidon_trials):
        pos = _random_sidon_positions(n, rng)
        if len(pos) < 2:            # need at least one pair to get M > 0
            continue

        # start from a flat background and raise the Sidon positions
        seq = np.full(n, eps, dtype=np.float64)
        seq[pos] = 1.0 + eps
        np.clip(seq, 0.0, 1000.0, out=seq)

        sc = _fast_score(seq)
        if np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = float(sc)
            # early exit if we already beat the target bound
            if best_score <= 1.50274349:
                break

    # ------------------------------------------------------------------
    # 2) Light local refinement (block perturbations) – cheap fallback
    # ------------------------------------------------------------------
    # We only run this if we have not hit the target yet.
    if best_score > 1.50274349:
        seq = best_seq.copy()
        for _ in range(max_local_steps):
            cand = _make_block_perturbation(seq, rng)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                seq = cand
                best_score = sc
                best_seq = cand
                if best_score <= 1.50274349:
                    break

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
