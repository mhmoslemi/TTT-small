def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1)
    target_bound = 1.5  # can set to slightly lower for safety
    delta = target_bound * np.e / (2 * n)   # approximate delta
    r = 1.0 - delta
    # Generate geometric sequence
    idx = np.arange(n, dtype=np.float64)
    base = np.power(r, idx)
    # Ensure non-negative
    base = np.clip(base, 0.0, 1000.0)
    # Scale to sum sqrt(2n)
    desired_sum = np.sqrt(2.0 * n)
    current_sum = base.sum()
    if current_sum == 0:
        # fallback uniform
        candidate = np.full(n, desired_sum / n, dtype=np.float64)
    else:
        scale = desired_sum / current_sum
        candidate = base * scale
    # Clip final bounds
    candidate = np.clip(candidate, 0.0, 1000.0)
    # Evaluate and return list
    return [float(x) for x in candidate]

def propose_candidate(seed=42, max_refine=12, **kwargs):
    """
    Construct a geometric‑decay sequence whose AC1 bound is close to the
    theoretical optimum (≈1.5).  The construction uses the relation

        bound ≈ (2 n δ) / e   with   δ = 1 – r,

    where r is the geometric ratio.  By choosing δ ≈ bound·e/(2 n) we obtain
    a sequence whose bound is around the desired target.
    """
    rng = np.random.default_rng(seed)

    # length of the problem instance
    n = int(height_sequence_1.shape[0])
    # the bound we would like to achieve (the target from the statement)
    target_bound = 1.50274349
    # desired total L1‑norm (the original normalisation, any positive sum works)
    desired_sum = np.sqrt(2.0 * n)

    # initial guess for delta from the analytical relation bound ≈ 2 n δ/e
    delta = target_bound * np.e / (2.0 * n)
    # keep delta safely away from the regime where 1/δ exceeds n
    delta_min = 0.9 / n
    delta_max = 0.5          # never let delta be huge (r would become negative)

    best_seq = None
    best_score = np.inf

    for _ in range(max_refine):
        # geometric ratio (must stay in (0,1])
        r = max(1.0 - delta, 0.0)

        # generate the raw geometric sequence a_i = r^i
        idx = np.arange(n, dtype=np.float64)
        seq = np.power(r, idx, dtype=np.float64)

        # safety clipping (values are already in [0,1], but keep within the
        # evaluator's allowed range)
        seq = np.clip(seq, 0.0, 1000.0)

        # normalise to the required L1‑norm (this scaling does NOT change the
        # bound, it only satisfies the evaluator's sum>0.01 check)
        total = seq.sum()
        if total == 0.0:
            # fall back to a uniform positive vector
            seq = np.full(n, desired_sum / n, dtype=np.float64)
        else:
            seq = seq * (desired_sum / total)

        # final clipping against the hard limits
        seq = np.clip(seq, 0.0, 1000.0)

        # evaluate the candidate
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

        # if we already beat the target, stop early
