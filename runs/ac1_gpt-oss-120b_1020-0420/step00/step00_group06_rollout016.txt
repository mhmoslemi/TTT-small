import numpy as np


def _scale_to_bounds(arr):
    """Scale array so that its maximum does not exceed 1000 (actually <1 for convenience)."""
    max_val = np.max(arr)
    if max_val == 0:
        return arr
    # scaling factor <1 keeps values within the evaluator's clipping range
    factor = 0.999 / max_val
    return arr * factor


def propose_candidate(seed=42, num_steps=300, **kwargs):
    """
    Return a list of non‑negative coefficients that (empirically) improve on the
    baseline AC1 upper bound.  The approach combines a short parametric search
    over simple shapes (uniform, exponentials, triangle) with a few gradient‑
    descent steps that directly reduce the peak of the self‑convolution.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine length and a fallback parent (height_sequence_1 is supplied at
    # runtime).  If it is missing or malformed, use a default uniform vector.
    # ----------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = parent.size
    else:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Helper to evaluate a candidate and keep the best one seen so far.
    # ----------------------------------------------------------------------
    def _maybe_update(candidate, current_best, best_score):
        """Return (new_best, new_score) possibly updated with candidate."""
        cand_list = candidate.tolist()
        sc = float(evaluate_sequence(cand_list))
        if np.isfinite(sc) and sc < best_score:
            return candidate.copy(), sc
        return current_best, best_score

    # ----------------------------------------------------------------------
    # Start with a simple uniform candidate (score = 2 exactly).
    # ----------------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)
    best = _scale_to_bounds(best)
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # Parametric families to probe quickly.
    # ----------------------------------------------------------------------
    indices = np.arange(n, dtype=np.float64)

    # 1) Uniform (already stored)
    # 2) Exponential decays a_i = alpha**i  for several alphas.
    #    Choose t = x/n with x in a modest interval [0.5, 2.0]
    xs = np.linspace(0.5, 2.0, 20)          # x = n * t  (controls the decay)
    for x in xs:
        t = x / n
        alpha = 1.0 - t
        cand = np.power(alpha, indices)
        cand = _scale_to_bounds(cand)
        best, best_score = _maybe_update(cand, best, best_score)

    # 3) Reverse exponential (increasing).
    for x in xs:
        t = x / n
        alpha = 1.0 - t
        cand = np.power(alpha, n - 1 - indices)
        cand = _scale_to_bounds(cand)
        best, best_score = _maybe_update(cand, best, best_score)

    # 4) Symmetric triangular shape.
    tri = np.minimum(indices + 1, n - indices)
    tri = _scale_to_bounds(tri)
    best, best_score = _maybe_update(tri, best, best_score)

    # ----------------------------------------------------------------------
    # Local refinement: directly decrease the peak of the convolution.
    # The gradient of the maximal convolution entry c_k w.r.t. a_i is
    #   d c_k / d a_i = 2 * a_{k-i}   (if 0 <= k-i < n, else 0)
    # We take a small step opposite to this gradient, renormalise, and keep
    # improvements.  The process is kept cheap (O(n) per step) and limited
    # to a modest number of iterations.
    # ----------------------------------------------------------------------
    a = best.copy()
    target_sum = a.sum()
    eta = 0.5               # initial step size (will shrink if no progress)

    for _ in range(num_steps):
        # Full convolution and location of its maximum.
        conv = np.convolve(a, a)            # length 2n-1
        k_max = int(np.argmax(conv))

        # Gradient of the maximal entry.
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # Gradient step with projection to the feasible box.
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Renormalise to preserve the (irrelevant) total sum.
        s = a_new.sum()
        if s == 0.0:
            a_new = a.copy()          # avoid degenerate zero vector
        else:
            a_new *= target_sum / s    # keep the original scale

        # Evaluate and possibly accept.
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_new.copy()
            best_score = score
            a = a_new
            eta = min(eta * 1.1, 1.0)   # accelerate when things improve
        else:
            # No improvement – shrink step and keep the current a.
            eta *= 0.6
            if eta < 1e-6:
                break

    # ----------------------------------------------------------------------
    # Return the best sequence as a plain Python list of floats.
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
