def smooth_sequence(seq, rng):
    # use 3-point kernel
    kernel = np.array([0.25, 0.5, 0.25], dtype=float)
    candidate = np.convolve(seq, kernel, mode='same')
    # renormalize sum to original
    total = candidate.sum()
    if total > 0:
        candidate *= seq.sum() / total
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def transfer_mass(seq, rng):

import numpy as np

# --------------------------------------------------------------
# Helper perturbations (all top‑level, no closures or lambdas)
# --------------------------------------------------------------

def block_perturbation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block and optionally symmetrize."""
    n = seq.size
    # block width between ~n/200 and ~n/12
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand = seq.copy()
    cand[start:start + width] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def smooth_sequence(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a tiny 3‑point moving‑average and renormalise the sum."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    cand = np.convolve(seq, kernel, mode='same')
    total = cand.sum()
    if total > 0.0:
        cand *= seq.sum() / total
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def transfer_mass(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = seq.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return seq
    amount = seq[i] * rng.random() * 0.5  # up to 50 % of source entry
    cand = seq.copy()
    cand[i] -= amount
    cand[j] += amount
    return cand
