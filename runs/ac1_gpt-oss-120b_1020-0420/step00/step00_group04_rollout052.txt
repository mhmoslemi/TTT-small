import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a uniform non‑negative step function of the same length as the
    provided ``height_sequence_1``.  For any non‑negative sequence the AC1
    upper bound evaluated by ``evaluate_sequence`` is
        2 * n * (sum a_i^2) / (sum a_i)^2 .
    This expression is minimized when all entries are equal, giving the
    theoretical minimum value 2.  The returned uniform vector therefore
    is at least as good as any sequence produced by local random search.
    """

    # Determine the length of the construction.
    # ``height_sequence_1`` is supplied by the harness; fall back to a
    # reasonable default if it is missing or malformed.
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = None

        n = 256                     # a safe default size
    else:
        n = int(base.size)

    # Build a uniform candidate.  The actual magnitude does not affect the
    # score because the ratio (∑a_i^2)/(∑a_i)^2 is scale‑invariant.  Using 1.0
    # keeps the numbers well within the [0, 1000] clipping range.
    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
