import numpy as np

# =============================================================================
# Helper: generate a simple block perturbation (O(n) memory)
# =============================================================================
def block_perturb(base, rng):
    """Scale a random contiguous block of ``base``."""
    cand = base.copy()
    n = cand.size
    # block length between n/200 and n/12 (at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    # optional mirroring to keep the shape balanced
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # enforce box constraints
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, num_steps=1200, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (hopefully)
    beats the baseline AC1 bound.

    The routine starts from a flat (uniform) sequence of the same length as
    ``height_sequence_1`` and performs a cheap stochastic local search that
    only ever uses O(n) auxiliary memory.
    """
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------------------
    # Initial candidate: uniform heights (value 1.0).  Any constant works;
    # the evaluator is scale‑invariant, this choice yields the theoretical
    # bound 2.0.
    # -------------------------------------------------------------------------
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 256
    else:
        n = parent.size
    best = np.full(n, 1.0, dtype=np.float64)          # uniform start
    best_score = float(evaluate_sequence(best.tolist()))  # should be 2.0

    # -------------------------------------------------------------------------
    # Stochastic search – each iteration touches O(n) data and evaluates the
    # exact objective via ``evaluate_sequence``.
    # -------------------------------------------------------------------------
    for _ in range(num_steps):
        cand = block_perturb(best, rng)

        # occasional global re‑balancing (Dirichlet draw keeps the sum)
        if rng.random() < 0.07:
            total = cand.sum()
            # Dirichlet with all ones gives a random point on the simplex
            proportions = rng.dirichlet(np.ones(n))
            cand = proportions * total

        # evaluate; keep the better candidate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # -------------------------------------------------------------------------
    # Fallback: if no improvement occurred, the uniform vector is still a
    # valid solution (score = 2.0).  Return the best-found sequence.
    # -------------------------------------------------------------------------
    return [float(x) for x in best]
