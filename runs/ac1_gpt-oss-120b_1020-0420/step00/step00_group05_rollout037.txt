def evaluate_convolution(seq):
    return float(evaluate_sequence(seq.tolist()))

def make_exponential_candidate(n, r):
    indices = np.arange(n, dtype=np.float64)
    return np.exp(indices * np.log(r))

def perturb_swap(seq, rng):
    n = seq.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return seq
    new_seq = seq.copy()
    new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
    return new_seq

def perturb_block_scale(seq, rng):
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    new_seq = seq.copy()
    new_seq[start:stop] *= factor
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    return new_seq

def perturb_jitter(seq, rng):
    n = seq.size
    mask = rng.random(n) < 0.05
    jitter = np.exp(rng.normal(0.0, 0.05, size=mask.sum()))
    new_seq = seq.copy()
    new_seq[mask] *= jitter
    return np.clip(new_seq, 0.0, 1000.0)

def greedy_sidon_set(n):
    chosen = []
    sums = set()
    for i in range(n):
        # Check if adding i would create duplicate sum
        conflict = False

            sums.add(2*i)

import numpy as np

# -------------------------------------------------------------
# Helper functions (all top‑level, no closures)

def greedy_sidon_set(length: int):
    """Return a list of indices forming a Sidon set (no repeated sums i+j)."""
    chosen = []
    used = set()                     # sums that already appear
    for i in range(length):
        # self‑sum conflict?
        if 2 * i in used:
            continue
        conflict = False
        for j in chosen:
            if i + j in used:
                conflict = True
                break
        if not conflict:
            # add i and record its sums
            for j in chosen:
                used.add(i + j)
            used.add(2 * i)
            chosen.append(i)
    return chosen


def make_exponential_candidate(length: int, r: float):
    """Monotone decreasing exponential a_i = r**i (r close to 1)."""
    idx = np.arange(length, dtype=np.float64)
    return np.exp(idx * np.log(r))


def perturb_swap(seq: np.ndarray, rng: np.random.Generator):
    """Swap two random entries."""
    n = seq.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return seq
    cand = seq.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def perturb_block_scale(seq: np.ndarray, rng: np.random.Generator):
    """Scale a random block by a log‑normal factor."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    w = int(rng.integers(lo, hi))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = start + w
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand = seq.copy()
    cand[start:stop] *= factor
    return np.clip(cand, 0.0, 1000.0)


def perturb_jitter(seq: np.ndarray, rng: np.random.Generator):
    """Multiply a small random subset by exp(N(0,0.05))."""
    n = seq.size
    mask = rng.random(n) < 0.05
    jitter = np.exp(rng.normal(0.0, 0.05, size=mask.sum()))
    cand = seq.copy()
    cand[mask] *= jitter
    return np.clip(cand, 0.0, 1000.0)
