   ```python
   if np.isinf(best_score) or best_score < 0:
       best_sequence = np.ones(6121)
       best_score = evaluate_sequence(best_sequence.tolist())

   # Auto-configuration based on initial score
   def clamp(val, low, high):
       return max(low, min(high, val))

   # Main optimization loop
   for _ in range(N_POS):
       # SPSA gradient estimation (gradient of the positive part)
       perturb = np.random.uniform(-PERTRUB_SCALE, PERTRUB_SCALE, 6121)
       candidate_plus = np.clip(candidate + perturb, 0, 1000)
       score_plus = evaluate_sequence(candidate_plus.tolist())
       candidate_minus = np.clip(candidate - perturb, 0, 1000)
       score_minus = evaluate_sequence(candidate_minus.tolist())

       # Finite check
       if np.isinf(score_plus) or np.isinf(score_minus):
           # Resample if gradient estimation becomes unstable
           candidate = make_block_perturbation(best_sequence)
           continue

       grad_estimate = (score_plus - score_minus) / (2 * PERTRUB_SCALE)

       # Gradient direction with regularization
       grad_direction = -np.sign(grad_estimate) if np.abs(grad_estimate) > 1e-4 else None

       # Gradient descent candidate without line search
       if grad_direction is not None:
           grad_alpha = 0.01  # adaptive step size with regularization
           trial_sequence = candidate + grad_alpha * grad_direction
           trial_sequence = np.array([clamp(x, 0, 1000) for x in trial_sequence])
           trial_score = evaluate_sequence(trial_sequence.tolist())
           # Heuristic line-search verification
           if trial_score < best_score - TOL_SCORE:
               candidate = trial_sequence.copy()
               if trial_score < best_score:
                   best_sequence = candidate.copy()
                   best_score = trial_score

       # Simulated annealing component for escape from local minima
       if np.random.random() < 0.05 and np.random.random() < 0.8:
           perturbed_seq = make_block_perturbation(best_sequence)
           candidate_score = evaluate_sequence(perturbed_seq.tolist())
           if not np.isinf(candidate_score) and candidate_score < best_score - TOL_SCORE:
               candidate = perturbed_seq.copy()
               best_sequence = candidate.copy()
               best_score = candidate_score
               print("Improvement from perturbation:", best_score - trial_score)

       # Extensive line search if gradient is trusted
       if np.random.random() < 0.2:  # sparse line search
           best_alpha = 1.0
           best_found_value = float('inf')
           for alpha in ALPHA_SCALE:
               trial_seq = candidate + alpha * (-grad_direction) if grad_direction is not None else \
                           candidate + alpha * np.random.uniform(-1, 1, 6121)
               trial_seq = np.array([clamp(x, 0, 1000) for x in trial_seq])
               eval_count += 1
               trial_score = evaluate_sequence(trial_seq.tolist())
               if np.isfinite(trial_score) and trial_score < best_found_value:
                   best_alpha = alpha
                   best_found_value = trial_score
           if best_found_value < best_score:
               candidate = trial_seq.copy()
               if best_found_value < best_score:
                   best_sequence = candidate.copy()
                   best_score = best_found_value

   return best_sequence.tolist()
   ```

4. Search Choices and Variations
   The algorithm incorporates three mechanisms:
   - Stochastic gradient descent with regularization
   - Simulated annealing with gradient acceptance
   - Trajectory-wise adaptive line search

   Key parameters:
   - `PERTRUB_SCALE`: Controls gradient estimation precision (tradeoff between noise and accuracy)
   - `ALPHA_SCALE`: Defines the range of line searches to explore
   - Temperature control: Adaptive perturbation scale based on recent progress

   When to trigger specific behaviors:
   - Always: Line search and gradient descent updates
   - 5%: Annealing/perturbation verification
   - 20%: Line search with adaptive steps
   - 85%: Standard gradient descent moves
