def propose_candidate(seed=42, **kwargs):
    # Use numpy RNG optional but not needed.
    n = int(np.asarray(height_sequence_1).size)
    if n == 0:
        # fallback small vector
        candidate = np.ones(1, dtype=np.float64)
        return candidate.tolist()
    # consider uniform candidate
    uniform = np.ones(n, dtype=np.float64)
    # also fallback to parent if valid
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    # evaluate both
    # ensure not empty
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))
    # evaluate parent
    try:
        p_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(p_score) and p_score < best_score:
            best = parent
            best_score = p_score
    except Exception:
        pass
    return [float(x) for x in best]

import numpy as np

def exponential_candidate(length: int) -> np.ndarray:
    """
    Return a decreasing exponential sequence a_i = r**i with r chosen so that
    the central convolution value equals the endpoint value (both 1 after
    scaling). This yields a very low AC1 upper bound.
    """
    if length <= 1:
        # Trivial case: uniform non‑zero entry
        return np.ones(length, dtype=np.float64)

    # r = n^{-1/(n-1)}  ensures n * r^{n-1} = 1
    r = float(length) ** (-1.0 / (length - 1))
    # generate r**i for i = 0 .. n-1
    indices = np.arange(length, dtype=np.float64)
    return np.power(r, indices)
