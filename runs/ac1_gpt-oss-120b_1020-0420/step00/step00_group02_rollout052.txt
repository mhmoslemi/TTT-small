import numpy as np

def _clip_and_fix(arr: np.ndarray) -> np.ndarray:
    """Clamp to [0,1000] and avoid degenerate zero‐sum vectors."""
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size == 0 or np.sum(arr) < 0.01:
        arr = np.ones_like(arr, dtype=np.float64)
    return arr


def _approx_score(seq: np.ndarray, n: int) -> float:
    """Fast surrogate for the true score: 2*n*sum(seq^2)/sum(seq)^2."""
    s1 = np.sum(seq)
    s2 = np.dot(seq, seq)
    if s1 == 0.0:
        return float("inf")
    return float(2.0 * n * s2 / (s1 * s1))


def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs) -> list:
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    Uses a cheap surrogate objective for many iterations and falls back
    to the exact evaluator only when an improvement is observed.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    #  Initialise from the supplied parent or from a simple uniform vector.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    base = _clip_and_fix(base)
    n = base.size

    best_seq = base.copy()
    best_approx = _approx_score(best_seq, n)
    # one exact evaluation for a reliable starting score
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    #  Main search loop – cheap surrogate evaluation for speed.
    # ------------------------------------------------------------------
    for step in range(num_steps):
        cand = best_seq.copy()

        # ---- 1.  Replace a random block by its mean (smooths variance) ----
        block_len = int(rng.integers(2, max(3, n // 50)))
        start = int(rng.integers(0, n - block_len + 1))
        # sliding‑window mean via cumulative sum (O(n))
        csum = np.cumsum(np.concatenate(([0.0], cand)))
        block_mean = (csum[start + block_len] - csum[start]) / block_len
        # tiny Gaussian noise to keep the search stochastic
        cand[start:start + block_len] = block_mean + rng.normal(0.0, 1e-3, block_len)

        # ---- 2.  Occasionally scale a small random block -------------------
        if rng.random() < 0.5:
            mul_len = int(rng.integers(1, max(2, n // 100)))
            mul_start = int(rng.integers(0, n - mul_len + 1))
            factor = float(np.exp(rng.normal(0.0, 0.02)))
            cand[mul_start:mul_start + mul_len] *= factor

        # ---- 3.  Enforce bounds -------------------------------------------
        cand = np.clip(cand, 0.0, 1000.0)

        # ---- 4.  Cheap surrogate evaluation --------------------------------
        cand_approx = _approx_score(cand, n)

        # accept only if the surrogate predicts improvement
        if cand_approx < best_approx - 1e-12:
            cand_score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(cand_score) and cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                best_approx = cand_approx

    # ------------------------------------------------------------------
    #  Fallback – if the search never succeeded, fall back to uniform.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        uniform_val = np.sqrt(2.0 * n) / n
        best_seq = np.full(n, uniform_val, dtype=np.float64)

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_seq]
