def spectral_damp(base, rng, damp_factor_range=(0.8,0.95), num_freqs=8):
    a = base.copy()
    n = a.size
    # compute rfft
    freq = np.fft.rfft(a, n=n)
    # Choose random freq indices (excluding 0 and maybe Nyquist)
    # Frequency count: len(freq) = n//2 + 1
    freq_len = freq.shape[0]
    # Choose K = rng.integers(4, 12) maybe
    K = int(rng.integers(4, 12))
    if K > freq_len - 2:
        K = freq_len - 2
    # Randomly select K distinct indices from 1 .. freq_len-2 (exclude DC and Nyquist)
    idxs = rng.choice(np.arange(1, freq_len-1), size=K, replace=False)
    # For each index, apply damping factor
    factors = rng.uniform(damp_factor_range[0], damp_factor_range[1], size=K)
    freq[idxs] *= factors
    # Inverse FFT
    new_a = np.fft.irfft(freq, n=n)
    # Clip to [0,1000] and ensure real positivity
    new_a = np.clip(new_a.real, 0.0, 1000.0)

    # Normalize sum to original sum (keeping same total mass)
    sum_original = base.sum()
    sum_new = new_a.sum()
    if sum_new > 0:
        new_a *= sum_original / sum_new
    return new_a

def _get_parent():
    """

    # safe fallback

def _power_sequence(length, exponent):
    """

    seq = np.power(pos, exponent) * 1000.0
    # Numerical safety – clip to the allowed range.
    return np.clip(seq, 0.0, 1000.0)
