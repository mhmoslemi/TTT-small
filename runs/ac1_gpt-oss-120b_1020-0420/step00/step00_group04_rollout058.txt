import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the AC1 upper bound using a fast FFT convolution.
# ----------------------------------------------------------------------
def _fft_convolution_max(a: np.ndarray, fft_len: int) -> float:
    """Return max value of the full convolution a*a using FFT (real input)."""
    ft = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(ft * ft, fft_len)[: 2 * a.size - 1]
    return float(conv.max())
