    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    nfft = 1 << ((2 * n - 1).bit_length())          # power‑of‑2 for FFT

    # ------------------------------------------------------------------
    # Initial best candidate – start from the parent and a few random seeds.
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    best_fast = _fast_score(best, n, nfft)

    for _ in range(10):
        cand = rng.gamma(shape=0.3, scale=1.0, size=n)
        cand = np.clip(cand, 0.0, 1000.0)
        cand_fast = _fast_score(cand, n, nfft)
        if cand_fast < best_fast:
            cand_exact = float(evaluate_sequence(cand.tolist()))
            if cand_exact < best_score:
                best = cand
                best_score = cand_exact
                best_fast = cand_fast

    # ------------------------------------------------------------------
    # Local refinement: repeatedly move a small amount of mass from positions
    # that contribute heavily to the current maximum convolution value to a
    # position that does not participate in that maximum.
    # ------------------------------------------------------------------
    no_improve = 0
    for step in range(num_steps):
        max_b, conv = _fft_convolution(best, n, nfft)
        idx_max = int(np.argmax(conv))

        # indices that actually contribute to this peak
        start = max(0, idx_max - (n - 1))
        end = min(idx_max, n - 1)
        if start > end:
            continue
        idxs = np.arange(start, end + 1, dtype=int)
        ok = (best[idxs] > 0) & (best[idx_max - idxs] > 0)
        contrib = idxs[ok]
        if contrib.size == 0:                     # fall back to a tiny random tweak
            cand = best.copy()
            factor = float(np.exp(rng.normal(0.0, 0.05)))
            block = int(rng.integers(max(1, n // 300), max(2, n // 30)))
            pos = int(rng.integers(0, n - block + 1))
            cand[pos : pos + block] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
        else:
            i = int(rng.choice(contrib))
            j = idx_max - i
            # small amount to relocate
            eps = min(best[i], best[j]) * rng.random() * 0.15
            if eps <= 0:
                continue
            # destination far from the active region
            for _ in range(10):
                dest = int(rng.integers(0, n))
                if not (start <= dest <= end):
                    break
            total_removed = eps if i == j else 2 * eps
            cand = best.copy()
            cand[i] -= eps
            if i != j:
                cand[j] -= eps
            cand[dest] += total_removed
            cand = np.clip(cand, 0.0, 1000.0)

        cand_fast = _fast_score(cand, n, nfft)
        # accept only if fast score improves; verify with exact evaluator
        if cand_fast < best_fast:
            cand_exact = float(evaluate_sequence(cand.tolist()))
            if cand_exact < best_score:
                best = cand
                best_score = cand_exact
                best_fast = cand_fast
                no_improve = 0
                continue
        no_improve += 1

        # periodic exact re‑evaluation of the current best (guards against drift)
        if step % 250 == 0:
            exact_now = float(evaluate_sequence(best.tolist()))
            if exact_now < best_score:
                best_score = exact_now
                best_fast = _fast_score(best, n, nfft)

        # if we get stuck, replace the current best with a fresh random candidate
        if no_improve > 500:
            cand = rng.gamma(shape=0.3, scale=1.0, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand_fast = _fast_score(cand, n, nfft)
            cand_exact = float(evaluate_sequence(cand.tolist()))
            if cand_exact < best_score:
                best = cand
                best_score = cand_exact
                best_fast = cand_fast
            no_improve = 0

    # ------------------------------------------------------------------
    # Return a plain Python list of floats – the fallback always yields a valid sequence.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
