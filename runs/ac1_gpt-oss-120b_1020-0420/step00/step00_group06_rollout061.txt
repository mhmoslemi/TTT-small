import numpy as np

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)  # seed might not be needed but keep for fallback.

    # Attempt to use existing height_sequence_1
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # Determine block length m such that m > n/3 and 2*m < n
    m = max(1, n // 3 + 1)
    # If 2*m >= n, shrink m
    if 2 * m >= n:
        m = (n - 1) // 2
        if m < 1:
            m = 1

    candidate = np.zeros(n, dtype=np.float64)
    candidate[:m] = 1.0
    candidate[-m:] = 1.0

    # Verify score; fallback to parent if something wrong
    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))

    # For safety, also evaluate parent (in case it's better)
    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent
        best_score = parent_score

    # Optional local refinement: try random perturbations for a few steps.
    # But keep within O(n) memory.

    for _ in range(200):
        # Randomly flip a small block: multiply by random factor >0
        width = int(rng.integers(1, max(2, n // 100)))  # small width
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.05)))
        new_candidate = best.copy()
        new_candidate[start:start+width] *= factor
        new_candidate = np.clip(new_candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(new_candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = new_candidate
            best_score = score

    return [float(x) for x in best]

import numpy as np

def _build_sidon_positions(n, block_len, max_blocks=None):
    """
    Greedy construction of a Sidon (B2) set of start positions.
    Each block occupies [p, p+block_len-1] and blocks do not overlap.
    The set guarantees that all sums p_i + p_j are distinct,
    which limits the maximum convolution to 2*block_len.
    """
    positions = []
    sum_set = set()                     # stores all p_i + p_j seen so far
    start = 0
    # rough upper bound on how many blocks we could possibly fit
    if max_blocks is None:
        max_blocks = (n + 1) // (block_len + 1)
    while start <= n - block_len:
        # enforce non‑overlap with the previous block (gap of at least 1)
        if positions and start < positions[-1] + block_len + 1:
            start = positions[-1] + block_len + 1
            continue
        # Sidon condition: all new sums must be new
        conflict = False
        for p in positions:
            if (start + p) in sum_set:
                conflict = True
                break
        if not conflict and (start + start) not in sum_set:
            # accept this start
            positions.append(start)
            # update the sum set with all new sums involving this start
            for p in positions:
                sum_set.add(start + p)
            if len(positions) >= max_blocks:
                break
        start += 1
    return positions

def propose_candidate(seed=42, **kwargs):
    """
    Returns a list of non‑negative heights that aim to minimise the
    AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------
    # 1. Obtain a fallback parent (the provided height_sequence_1 if valid)
    # --------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best_seq = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # --------------------------------------------------------------------
    # 2. Try Sidon‑block constructions for several block lengths.
    #    The pattern consists of many small blocks (value 1) spaced so that
    #    all pairwise sums of start positions are distinct.
    #    This guarantees max(conv) = 2*block_len, while the sum grows with
    #    the number of blocks, leading to a low objective.
    # --------------------------------------------------------------------
    for block_len in (5, 8, 10, 12, 15, 20, 25, 30, 35, 40):
        positions = _build_sidon_positions(n, block_len)
        if not positions:
            continue
        candidate = np.zeros(n, dtype=np.float64)
        for p in positions:
            candidate[p:p + block_len] = 1.0
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = candidate
            best_score = score

    # --------------------------------------------------------------------
    # 3. Simple random fine‑tuning (few cheap perturbations) to possibly
    #    improve the best candidate found.
    # --------------------------------------------------------------------
    for _ in range(200):
        if best_seq is None:
            break
        # pick a random block and a random multiplicative factor
        width = int(rng.integers(1, max(2, n // 300)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.07)))
        trial = best_seq.copy()
        trial[start:start + width] *= factor
        trial = np.clip(trial, 0.0, 1000.0)
        # renormalise to keep the total sum unchanged (scale invariance)
        s = trial.sum()
        if s > 0.0:
            trial *= (best_seq.sum() / s)
        score = float(evaluate_sequence(trial.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = trial
            best_score = score

    # --------------------------------------------------------------------
    # 4. Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------------
    return [float(x) for x in best_seq]
