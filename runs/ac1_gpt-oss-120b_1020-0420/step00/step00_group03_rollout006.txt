import numpy as np

def propose_candidate(seed=42, num_steps=1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that minimises the
    AC1 upper bound used by ``evaluate_sequence``.

    Theory:
    For a sequence a of length n the bound is
        B = 2 * n * max(convolve(a, a)) / (sum(a))**2 .
    Because the zero‑shift entry of the convolution equals sum_i a_i**2,
    the inequality
        sum_i a_i**2 >= (sum_i a_i)**2 / n
    (Cauchy–Schwarz) gives B >= 2, with equality iff all entries are equal.
    Hence any uniform sequence attains the theoretical optimum B = 2.

    This implementation therefore returns a uniform sequence of the same
    length as the supplied ``height_sequence_1`` (or a default length 256)
    using a constant value inside the allowed range [0, 1000].
    A tiny sanity check with ``evaluate_sequence`` ensures the result is
    accepted; if it were rejected we fall back to a simple all‑ones vector.
    """
    # RNG kept for API compatibility – not needed for the uniform solution.
    _ = np.random.default_rng(seed)

    # Determine the desired length.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(base.size)
        if n == 0:
            n = 256
    except Exception:
        n = 256

    # Uniform constant – any value within [0,1000] works.
    const = 1.0
    candidate = np.full(n, const, dtype=np.float64)
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Validate with the provided evaluator; fallback to ones on failure.
    try:
        sc = evaluate_sequence(candidate.tolist())
        if not np.isfinite(sc):
            candidate = np.ones(n, dtype=np.float64)
    except Exception:
        candidate = np.ones(n, dtype=np.float64)

    # Return a plain Python list of floats as required.
    return [float(x) for x in candidate]
