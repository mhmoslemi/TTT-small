def simplex_projection(v):
    # v is array of length n.
    # This algorithm sorts v to find threshold.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_score(arr):
    a = np.asarray(arr, dtype=np.float64)
    if not np.isfinite(a).all():
        return np.inf
    sum_a = a.sum()
    if sum_a <= 0:
        return np.inf
    fa = np.fft.rfft(a, n=2*n)
    c = np.fft.irfft(fa * fa, n=2*n)
    max_c = c.max()
    return 2.0 * n * max_c / (sum_a * sum_a)
```

import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the AC1 upper bound using FFT (O(n log n))
# ----------------------------------------------------------------------
def _fast_score(arr: np.ndarray, n: int) -> float:
    """Return the value 2*n*max(conv(arr,arr))/(sum(arr)**2)."""
    a = np.asarray(arr, dtype=np.float64)
    if a.size == 0 or not np.isfinite(a).all():
        return np.inf
    s = a.sum()
    if s <= 0.0:
        return np.inf
    # linear convolution via FFT (length 2*n-1)
    fa = np.fft.rfft(a, n=2 * n)
    conv = np.fft.irfft(fa * fa, n=2 * n)[: 2 * n - 1]
    m = conv.max()
    return 2.0 * n * m / (s * s)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, num_iters: int = 5000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improve on the
    previous best AC1 bound.  The routine performs a cheap stochastic search
    using Dirichlet‑distributed samples, a few deterministic permutations and a
    lightweight local refinement.  All work is O(n) or O(n log n) in memory
    and time.
    """
    rng = np.random.default_rng(seed)

    # the parent construction is available as `height_sequence_1`
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 256
        best_vec = np.ones(n, dtype=np.float64) / n
    else:
        n = parent.size
        # start from a simple uniform candidate
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_score = _fast_score(best_vec, n)

    # a few deterministic baselines (they are cheap)
    # 1) descending linear (already a simple shape)
    lin_desc = np.arange(n, 0, -1, dtype=np.float64)
    lin_desc /= lin_desc.sum()
    s = _fast_score(lin_desc, n)
    if s < best_score:
        best_vec, best_score = lin_desc, s

    # 2) ascending linear (mirror of above)
    lin_asc = lin_desc[::-1]
    s = _fast_score(lin_asc, n)
    if s < best_score:
        best_vec, best_score = lin_asc, s

    # ------------------------------------------------------------------
    # Stochastic search: Dirichlet samples and a couple of cheap permutations
    # ------------------------------------------------------------------
    # a stride that is coprime with n – it spreads large entries uniformly.
    stride = 1049  # gcd(4806,1049)=1, works for the given length
    perm_indices = (np.arange(n) * stride) % n

    for _ in range(num_iters):

        if sc < best_score:
            best_vec, best_score = shuffled, sc

        # sorted descending then spread using the fixed stride
        desc = np.sort(vec)[::-1]
        spread = np.empty_like(desc)
        spread[perm_indices] = desc
        sc = _fast_score(spread, n)
        if sc < best_score:
            best_vec, best_score = spread, sc

        # early exit if we already beat the target bound
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Light local refinement: move a small mass from one coordinate to another
    # ------------------------------------------------------------------
    # The refinement works only if it yields an improvement; it stays on the
    # simplex (sum unchanged) and respects non‑negativity.
    local_steps = 2000
    for _ in range(local_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j or best_vec[i] <= 0.0:
            continue
        # move at most 10 % of the source mass
        delta = best_vec[i] * rng.uniform(0.0, 0.1)
        new_vec = best_vec.copy()
        new_vec[i] -= delta
        new_vec[j] += delta
        sc = _fast_score(new_vec, n)
        if sc < best_score:
            best_vec, best_score = new_vec, sc

    # ------------------------------------------------------------------
    # Final safety: clip to the allowed numeric range and return as list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    # Guarantee a non‑zero sum (the optimizer never creates a zero‑sum vector)
    if best_vec.sum() < 0.01:
        best_vec = np.full_like(best_vec, 0.01 / n)

    return best_vec.tolist()
