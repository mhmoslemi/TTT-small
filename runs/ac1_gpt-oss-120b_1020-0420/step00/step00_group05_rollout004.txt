    """Create a simple uniform non‑negative step function."""
    # Use a value of 1.0; any constant gives the same bound (2.0)
    return np.full(length, 1.0, dtype=np.float64)


def _perturb(candidate: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Generate a small block‑wise perturbation of ``candidate``."""
    n = candidate.size
    # block width between n//200 and n//12 (inclusive)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.05)))  # modest scaling
    out = candidate.copy()

    if rng.random() < 0.3:
        out = 0.5 * (out + out[::-1])
    np.clip(out, 0.0, 1000.0, out=out)
    return out
