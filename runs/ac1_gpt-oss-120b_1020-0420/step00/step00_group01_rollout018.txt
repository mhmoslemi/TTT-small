import numpy as np

def _perturb_block(seq: np.ndarray, rng: np.random.Generator, n: int) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    # block width between n//200 and n//12 (inclusive)
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand = seq.copy()
    cand[start:start + width] *= factor
    if rng.random() < 0.35:                     # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _perturb_point(seq: np.ndarray, rng: np.random.Generator, n: int) -> np.ndarray:
    """Replace a single entry by a fresh log‑normal sample."""
    cand = seq.copy()
    idx = int(rng.integers(0, n))
    cand[idx] = max(0.0, min(1000.0, float(np.exp(rng.normal(0.0, 0.5)))))
    return cand


def propose_candidate(seed: int = 42, num_steps: int = 1000, **kwargs) -> list:
    """
    Return a list of non‑negative heights that (approximately) minimise
    the AC1 upper bound as evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise from the supplied parent, falling back to a uniform array
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)   # simple fallback

    base = np.clip(base, 0.0, 1000.0)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # ------------------------------------------------------------------
    # simple stochastic local search (memory O(n), no dense matrices)
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # choose a perturbation type
        if rng.random() < 0.7:
            cand = _perturb_block(best, rng, n)
        else:
            cand = _perturb_point(best, rng, n)

        # occasional random restart to keep diversity
        if rng.random() < 0.01:
            cand = rng.uniform(0.0, 1000.0, size=n)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # return a plain Python list of floats as required by the contract
    # ------------------------------------------------------------------
    return [float(v) for v in best]
