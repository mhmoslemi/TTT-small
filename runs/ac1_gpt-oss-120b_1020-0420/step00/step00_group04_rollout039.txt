import numpy as np

# evaluate_sequence is provided by the harness.


def _convolution(a: np.ndarray) -> np.ndarray:
    """Return full convolution a*a (numpy array)."""
    return np.convolve(a, a)


def _max_convolution_info(a: np.ndarray):
    """Return (max_value, max_index, contribution_array, i_range, partner_range)."""
    c = _convolution(a)
    max_idx = int(np.argmax(c))
    max_val = float(c[max_idx])

    n = a.size
    # indices that contribute to this maximal entry
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    partner = max_idx - i_range
    contrib = a[i_range] * a[partner]
    return max_val, max_idx, contrib, i_range, partner

    """
    Search for a non‑negative step function that reduces the AC1 upper bound.
    The algorithm starts from a uniform vector (which yields score 2) and
    performs a simple simulated‑annealing style hill‑climb that continually
    moves a small amount of mass away from the pair of indices that currently
    dominate the maximal convolution entry.
    """

    # Length of the provided parent sequence (or fallback length).
    try:

    except Exception:

        n = 256
        base = np.ones(n, dtype=np.float64)

    a = base.copy()
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Temperature schedule for occasional uphill moves.
    T0 = 1.0

    # Counters to trigger a random restart if progress stalls.
    stagnant = 0
    restart_limit = 300

    for it in range(max_iters):
        # Current convolution information.
        max_val, max_k, contrib, i_range, partner = _max_convolution_info(a)

        # Pick the pair (i, j) inside the maximal sum that contributes the most.
        pair_idx = int(np.argmax(contrib))
        i = int(i_range[pair_idx])
        j = int(partner[pair_idx])

        # Choose a random third index to receive some mass.
        r = int(rng.integers(0, n))
        while r == i or r == j:
            r = int(rng.integers(0, n))

        # Move a modest fraction of the larger of a[i], a[j] to the random index.
        # This keeps the total sum essentially unchanged (scaling does not affect
        # the objective, but preserving sum simplifies the search).
        source = i if a[i] >= a[j] else j
        move_frac = 0.07                      # 7 % move each step
        delta = a[source] * move_frac
        if delta <= 0.0:
            # nothing to move – restart with a fresh random vector.
            a = rng.random(n) * 2.0
            a = np.clip(a, 0.0, 1000.0)
            continue

        a_candidate[source] -= delta
        a_candidate[r] += delta
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
