def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Produce a non‑negative step‑function that (approximately) minimizes the
    AC1 bound 2*n*max(conv(f,f))/(sum f)^2.
    The algorithm starts from the provided `height_sequence_1` if present,
    builds a uniform distribution with the same total mass (which attains the
    theoretical minimum 2), then performs a simple mass‑transfer smoothing
    on the original sequence to possibly improve the score further.
    """
    rng = np.random.default_rng(seed)

    # ---- obtain a sensible starting vector ----
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    n = base.size
    total = base.sum()

    if total < 0.01:                     # guarantee a valid denominator
        base = np.full(n, (0.01 / n) + 1.0, dtype=np.float64)
        total = base.sum()

    # ---- uniform candidate (theoretical optimum) ----
    uniform = np.full(n, total / n, dtype=np.float64)
    uniform = np.clip(uniform, 0.0, 1000.0)

    best = uniform.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ---- simple smoothing via mass transfer ----
    cur = base.copy()
    for step in range(num_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
