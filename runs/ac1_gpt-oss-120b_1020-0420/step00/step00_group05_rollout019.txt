def compute_score(seq):
    # seq is numpy array 1d of floats, already nonnegative and finite.
    n = seq.size
    if n == 0: return np.inf
    sum_a = np.sum(seq)
    if sum_a < 0.01: return np.inf

def compute_score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound for a non‑negative sequence."""
    n = seq.size
    if n == 0:
        return float('inf')
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a * sum_a))

    # over a uniform fallback – this is cheap and safe.
    # --------------------------------------------------------------
    uniform = np.full(n, np.mean(candidate), dtype=np.float64)
    if compute_score(uniform) < compute_score(candidate):
        candidate = uniform

    # --------------------------------------------------------------
    # Perform a tiny random‑walk refinement to possibly lower the bound
    # further without heavy computation.
    # --------------------------------------------------------------
    best = candidate
    best_score = compute_score(best)

    # Parameters for the refinement (very modest to stay fast)
    width_low = max(1, n // 200)
    width_high = max(2, n // 12)
    sigma = 0.12

    for _ in range(500):                     # a few hundred cheap attempts
        cand = best.copy()
        # 60 % of the time use a block scaling, otherwise move a random chunk.
        if rng.random() < 0.6:
            width = int(rng.integers(width_low, width_high + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:stop] *= factor
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            # Transfer a small random fraction of mass from i to j.
            delta = float(rng.random()) * cand[i] * 0.3
            cand[i] -= delta
            cand[j] += delta

        # Optional mild mirroring – helps break residual symmetries.
        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        sc = compute_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
