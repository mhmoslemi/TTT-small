import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _generate_sidon_set(length: int, rng: np.random.Generator) -> list:
    """
    Greedy construction of a Sidon set (all pairwise sums are distinct).
    Returns a list of indices.  The size typically grows as ~sqrt(length).
    """
    indices = np.arange(length)
    rng.shuffle(indices)                     # deterministic shuffle via seed
    chosen = []                              # the Sidon set
    used_sums = set()                        # sums i+j already present

    for idx in indices:
        idx = int(idx)
        # Check if self‑sum collides
        if idx + idx in used_sums:
            continue
        # Check collisions with existing members
        collision = False
        for j in chosen:
            if idx + j in used_sums:
                collision = True
                break
        if collision:
            continue

        # No collision – add the element and all its new sums
        for j in chosen:
            used_sums.add(idx + j)
        used_sums.add(idx + idx)
        chosen.append(idx)

    return chosen


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs) -> list:
    """
    Produce a step‑function height sequence that aims at a low AC1 upper bound.
    The construction uses a large constant block placed on a Sidon set of
    indices and a smaller constant on the remaining positions.  This limits the
    maximum convolution value while keeping the total sum large, driving the
    objective 2·n·max(b)/(sum(a)^2) below the target 1.5.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256                     # safe fallback length
        else:
            n = int(base.size)
    except NameError:
        n = 256                         # when the global is not defined

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    BIG_VAL = 1000.0                    # maximal allowed height
    # Choose the smallest epsilon that guarantees
    # epsilon^2 * n <= 2 * BIG_VAL^2  (so max_b is driven by the big block)
    eps_limit = np.sqrt(2.0) * BIG_VAL / np.sqrt(n)
    epsilon = float(np.floor(eps_limit))
    epsilon = max(0.01, min(1000.0, epsilon))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try a few random Sidon constructions and keep the best result
    # ------------------------------------------------------------------
    for _ in range(5):
        sidon = _generate_sidon_set(n, rng)

        seq = np.full(n, epsilon, dtype=np.float64)
        if sidon:
            seq[sidon] = BIG_VAL
        np.clip(seq, 0.0, 1000.0, out=seq)      # safety, though values already respect limits

        try:
            score = float(evaluate_sequence(seq.tolist()))
        except Exception:
            score = float("inf")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # Fallback – simple uniform sequence (guaranteed valid)
    # ------------------------------------------------------------------
    if best_seq is None or not np.isfinite(best_score):
        fallback = np.full(n, 1.0, dtype=np.float64)
        fallback = np.clip(fallback, 0.0, 1000.0)
        return fallback.tolist()

    # Return the best candidate found
    return [float(v) for v in best_seq.tolist()]
