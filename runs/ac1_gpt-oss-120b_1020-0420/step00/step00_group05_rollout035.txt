import numpy as np

def propose_candidate(seed=42, num_steps=500, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 4498
    # Initialize uniform
    base = np.full(n, 1.0, dtype=np.float64)
    # Evaluate baseline
    try:
        best = base.copy()
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best = base
        best_score = float('inf')
    # Simple random mass-transfer search
    for _ in range(num_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Choose delta limited to a fraction of source value
        amt = best[i]
        if amt <= 0.0:
            continue
        delta = rng.random() * 0.5 * amt
        candidate = best.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # Clip to allowed range
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        try:
            score = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:

    """
    Return a non‑negative step‑function (list of floats) that minimizes
    the AC1 upper bound evaluated by the supplied `evaluate_sequence`.

    Strategy:
      * Start from the uniform distribution (all entries equal).  For a
        fixed L1 norm this distribution is known to give a very low
        max‑convolution value.
      * Perform a cheap stochastic mass‑transfer search that preserves the
        total sum.  Each move transfers a small amount of weight from one
        coordinate to another and accepts the move only if the AC1 score
        improves.
      * Keep the best candidate seen; the uniform vector is a safe fallback.
    """
    rng = np.random.default_rng(seed)
    n = _initial_length()

    # Uniform start: any constant works because the score is scale‑invariant.
    best = np.full(n, 1.0, dtype=np.float64)
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = float("inf")

    # Simple mass‑transfer hill‑climb (keeps the L1 norm fixed).
    for _ in range(num_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        src_val = best[i]
        if src_val <= 0.0:
            continue
        # Transfer at most half of the source value (random fraction).
        delta = rng.random() * 0.5 * src_val
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # Enforce the evaluator's bounds.
        np.clip(cand, 0.0, 1000.0, out=cand)

            score = float(evaluate_sequence(cand.tolist()))
        except Exception:

            best = cand
            best_score = score

    # Ensure a valid, finite result; fall back to uniform if needed.
    if not np.isfinite(best_score):
