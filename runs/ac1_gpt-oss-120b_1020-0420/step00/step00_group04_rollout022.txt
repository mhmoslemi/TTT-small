"""
Compact optimizer for the AC1 upper bound.

The objective evaluated by the provided `evaluate_sequence` is
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

For non‑negative sequences the maximum of the full convolution is always
the sum of squares (zero lag).  By Cauchy–Schwarz the ratio
    sum_i a_i^2 / (sum_i a_i)^2
is minimized when all entries are equal, giving the theoretical lower bound
of 2.  This implementation therefore returns a uniform vector of the same
length as the supplied `height_sequence_1` (or a safe fallback length) and
adds a tiny stochastic refinement loop – it never hurts, and the uniform
vector is already optimal.

The function respects the required signature and works with the harness's
`evaluate_sequence` without allocating more than O(n) additional memory.
"""

import numpy as np

def _valid_parent():
    """Return a valid 1‑D NumPy array from the global `height_sequence_1`."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        return None
    if arr.ndim != 1 or arr.size == 0:
        return None
    if not np.isfinite(arr).all():
        return None
    # Clamp to the evaluator's admissible range.
    return np.clip(arr, 0.0, 1000.0)

def _uniform_candidate(length, value=1.0):
    """Create a uniform non‑negative candidate of given length."""
    return np.full(length, float(value), dtype=np.float64)

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Return a list of heights that (near)‑optimally minimizes the AC1 bound.

    Parameters
    ----------
    seed : int, optional
        Random seed for the tiny stochastic refinement loop.
    num_steps : int, optional
        Number of random perturbation attempts (default 200).
    **kwargs : dict
        Ignored – retained for compatibility with the harness.

    Returns
    -------
    list[float]
        A non‑empty list of finite, non‑negative coefficients.
    """
    rng = np.random.default_rng(seed)

    # Determine problem dimension.
    parent = _valid_parent()
    n = parent.size if parent is not None else 256

    # Start from the optimal uniform vector (value 1.0 is arbitrary).
    best = _uniform_candidate(n, 1.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # Tiny stochastic refinement – cannot beat the uniform optimum,
    # but safeguards against pathological input where the uniform vector
    # is not yet the best seen (e.g., if the evaluator's clipping altered it).
    for _ in range(num_steps):
        # Random block scaling.
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = float(np.exp(rng.normal(0.0, 0.07)))  # modest perturbation
        cand = best.copy()
        cand[start:stop] *= factor

        # Optional gentle symmetrisation.
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # Early exit when we have reached the theoretical optimum.
            if best_score <= 2.0 + 1e-12:
                break

    # Convert to plain Python list of floats as required.
    return [float(x) for x in best]
