def solve_lp(f, M):
    n = f.shape[0]
    # Build A
    # To save memory, we can re-use a precomputed diagonal index if needed but not necessary.
    offsets = np.arange(n)
    # Using list of data arrays
    data = [np.full(n, float(val)) for val in f]  # list of arrays of length n
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except cp.error.SolverError:
        return None, None
    if g.value is None:
        return None, None
    g_val = np.array(g.value).flatten()
    return g_val, prob.value

import numpy as np
import scipy.sparse as sp

# ----- optional CVXPY support -------------------------------------------------
try:
    import cvxpy as cp
    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    cp = None
    _HAS_CVXPY = False
# -----------------------------------------------------------------------------


def _toeplitz_conv_matrix(f_vec: np.ndarray) -> sp.csr_matrix:
    """
    Return the (2n-1) × n sparse matrix A such that  A @ g = f * g
    (linear convolution, zero‑padding outside the support).
    Each column i contains the value f[i] on the diagonal offset i.
    """
    n = f_vec.shape[0]
    # Every diagonal has length n and constant value f[i]
    data = [np.full(n, float(val), dtype=np.float64) for val in f_vec]
    offsets = np.arange(n, dtype=np.int64)
    return sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")


def _solve_lp_max_sum(f_vec: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max 1ᵀg   s.t.   (f * g)_k ≤ M   ∀k,   g ≥ 0.
    Returns the optimal g as a dense ndarray, or None if the solver fails.
    """
    if not _HAS_CVXPY:
        return None
    n = f_vec.shape[0]
    A = _toeplitz_conv_matrix(f_vec)          # sparse (2n-1) × n
    g = cp.Variable(n, nonneg=True)           # variable to optimise
    constraints = [A @ g <= M]                # element‑wise upper bound
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=15000)
    except Exception:                         # pragma: no cover
        return None
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64).ravel()


def propose_candidate(seed: int = 42, num_iters: int = 8, mix: float = 0.5, **kwargs) -> list:
    """
    Produce a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 bound used by `evaluate_sequence`.

    The routine starts from the supplied `height_sequence_1` (if present),
    then repeatedly solves a small linear program to enlarge the sum of the
    vector while keeping the convolution with the current iterate bounded.
    The new vector is mixed with the old one; the best result seen is kept.
    """
    rng = np.random.default_rng(seed)

    # ------- initialise -------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:                                 # pragma: no cover
        parent = np.ones(256, dtype=np.float64)

    f = parent.copy()
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------- main iterative improvement ---------------------------------------
    for _ in range(num_iters):
        # current maximal convolution value
        conv = np.convolve(f, f)
        M = float(conv.max())

        # linear programme to enlarge the sum under the same peak bound
        g = _solve_lp_max_sum(f, M)
        if g is None or not np.isfinite(g).all():
            break

        sum_f = float(f.sum())
        sum_g = float(g.sum())
        if sum_g <= 1e-12:
            break

        # rescale g to keep the total mass unchanged, then mix
        g = g * (sum_f / sum_g)
        f_new = (1.0 - mix) * f + mix * g
        f_new = np.clip(f_new, 0.0, 1000.0)

        # evaluate candidate
        score = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f_new.copy()
            best_score = score

        f = f_new

    # ------- safety fallback --------------------------------------------------
    if not np.isfinite(best_score) or best_score == float('inf'):
        # a simple uniform vector always yields bound 2.0
        n = max(1, parent.size)
        best = np.full(n, 1.0, dtype=np.float64)

    return [float(v) for v in best]
