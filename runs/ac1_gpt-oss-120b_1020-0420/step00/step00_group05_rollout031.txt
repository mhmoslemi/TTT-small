    seq = np.clip(seq, 0.0, 1000.0)
    cur_sum = seq.sum()
    if cur_sum <= 0:
        return np.full(seq.shape, target_sum/len(seq))
    seq = seq * (target_sum / cur_sum)
    return seq

def propose_candidate(seed=42, max_iter=30, top_k=20, step_frac=0.2, **kwargs):

    # get parent or default
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            raise Exception
    except Exception:
        init = np.ones(256, dtype=np.float64)
    init = np.clip(init, 0.0, 1000.0)
    n = init.size
    # ensure sum not too small
    sum_init = init.sum()
    if sum_init < 0.01:
        init = np.full(n, 1.0)
        sum_init = n
    f = init.copy()
    # store best
    best_seq = f.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    target_sum = sum_init

    for iter_idx in range(max_iter):
        # compute convolution
        conv = np.convolve(f, f)
        M = conv.max()
        # pick top offsets
        idx_desc = np.argsort(-conv)[:top_k]
        # build sparse matrix
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(idx_desc):
            start = max(0, k - (n-1))
            end = min(k, n-1)
            # iterate over j indices
            # vectorized approach: j = np.arange(start, end+1); col = k - j; coeff = f[j]
            j = np.arange(start, end+1)
            rows.extend([row_idx]*j.size)
            cols.extend((k - j).tolist())
            data.extend(f[j].tolist())
        A = sp.csr_matrix((data, (rows, cols)), shape=(top_k, n))
        b = np.full(top_k, M)

        # linear program: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        except Exception:

        if res is not None and res.success:
            g = res.x
            # enforce max bound per entry
            max_val = g.max()
            if max_val > 1000.0:
                g = g * (1000.0 / max_val)
            # rescale to preserve sum
            sum_g = g.sum()
            if sum_g > 0:
                g = g * (target_sum / sum_g)
            # mix
            t = step_frac
            f_new = (1.0 - t) * f + t * g
        else:
            # fallback: random block perturbation
            width_low = max(1, n // 200)
            width_high = max(2, n // 12)
            start = rng.integers(0, n - width_low + 1)

            f_new[start:stop] *= factor
            if rng.random() < 0.35:

            # clip and rescale
            f_new = np.clip(f_new, 0.0, 1000.0)
            cur_sum = f_new.sum()
            if cur_sum > 0:

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def _clip_and_rescale(arr: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and renormalise to the desired ℓ₁ sum."""
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s <= 0:
        return np.full_like(arr, target_sum / arr.size)
    return arr * (target_sum / s)


def _build_constraint_matrix(f: np.ndarray, offsets: np.ndarray) -> sp.csr_matrix:
    """Return sparse matrix A where A[k,i] = f[i] if i participates in (f*g)[offsets[k]]."""
    n = f.size
    rows, cols, data = [], [], []
    for row, k in enumerate(offsets):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        j = np.arange(start, end + 1, dtype=int)
        rows.extend([row] * j.size)
        cols.extend((k - j).tolist())
        data.extend(f[j].tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(len(offsets), n))

    # ---------- optimisation loop ----------
    for _ in range(max_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        # pick offsets with the largest convolution values
        top_idxs = np.argsort(-conv)[:top_k]

        # build linear constraints: (f * g)[offset] ≤ M
        A = _build_constraint_matrix(f, top_idxs)
        b = np.full(top_idxs.shape, M, dtype=np.float64)

        # linear programme: maximise 1ᵀg  ⇔ minimise -1ᵀg
        c = -np.ones(n, dtype=np.float64)
        try:
            lp_res = linprog(c, A_ub=A, b_ub=b,
                             bounds=[(0, None)] * n,
                             method='highs')
        except Exception:        # safety net for any solver failure
            lp_res = None

        if lp_res is not None and lp_res.success:
            g = lp_res.x
            # enforce upper bound 1000
            mx = g.max()
            if mx > 1000.0:
                g = g * (1000.0 / mx)
            # keep the ℓ₁ norm unchanged
            g = _clip_and_rescale(g, target_sum)
            # convex update
            f_new = (1.0 - step_frac) * f + step_frac * g
        else:
            # fall‑back: simple stochastic block perturbation
            width_low = max(1, n // 200)
            width_high = max(2, n // 12)
            start = int(rng.integers(0, n - width_low + 1))
            width = int(rng.integers(width_low, width_high))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            f_new = f.copy()
            f_new[start:stop] *= factor
            if rng.random() < 0.35:
                f_new = 0.5 * (f_new + f_new[::-1])
            f_new = _clip_and_rescale(f_new, target_sum)

        # final clipping & renorm to preserve sum
        f_new = _clip_and_rescale(f_new, target_sum)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = f_new.copy()

        f = f_new  # continue from the latest point

    # return as plain Python list of floats
    return [float(x) for x in best_seq]
