import numpy as np

def _full_convolution(x):
    n = x.size
    nfft = 1 << ((2*n - 1).bit_length())
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx*fx, n=nfft)[:2*n-1]
    return conv

def propose_candidate(seed=42, num_steps=500, **kwargs):
    rng = np.random.default_rng(seed)

    # obtain base sequence
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    # ensure within bounds
    base = np.clip(base, 0.0, 1000.0)
    # fall back if invalid
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    # start from a random permutation to break symmetry
    candidate = base.copy()
    rng.shuffle(candidate)

    n = candidate.size
    total = candidate.sum()
    if total < 0.01:
        candidate[:] = 1.0
        total = candidate.sum()

    # initial best using exact evaluator
    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # precompute FFT size
    nfft = 1 << ((2*n - 1).bit_length())

    for _ in range(num_steps):
        # current convolution
        fx = np.fft.rfft(candidate, n=nfft)
        conv = np.fft.irfft(fx * fx, n=nfft)[:2*n - 1]

        # index of maximal convolution value
        k_max = int(conv.argmax())
        # contributions of each entry to that index
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        contributions = np.zeros_like(candidate)
        if mask.any():
            contributions[mask] = candidate[idx[mask]] * candidate[j[mask]]

        # pick a location with largest contribution
        i_max = int(contributions.argmax())
        # pick a location with smallest contribution (avoid i_max)
        # using argmin and handling equality
        i_min = int(contributions.argmin())
        if i_min == i_max:
            # choose a random different index
            i_min = int(rng.integers(0, n - 1))
            while i_min == i_max:
                i_min = int(rng.integers(0, n - 1))

        # attempt swap
        candidate[i_max], candidate[i_min] = candidate[i_min], candidate[i_max]

        # recompute convolution and approximate score
        fx2 = np.fft.rfft(candidate, n=nfft)
        conv2 = np.fft.irfft(fx2 * fx2, n=nfft)[:2*n - 1]
        max_val2 = float(conv2.max())
        approx_score2 = 2.0 * n * max_val2 / (total * total)

        # accept if improvement
        if approx_score2 < best_score:
            best = candidate.copy()
            best_score = approx_score2
        else:
            # revert swap
            candidate[i_max], candidate[i_min] = candidate[i_min], candidate[i_max]

    # final fallback: ensure result passes evaluator
    # (the best sequence is already finite and within bounds)
    result = [float(v) for v in best]
    if not result:
        result = [1.0]
    return result

        base = np.ones(256, dtype=np.float64)

    # Safety clipping and validation.
    base = np.clip(base, 0.0, 1000.0)
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    # Random permutation to break any hidden symmetry of the parent.
    candidate = base.copy()
    rng.shuffle(candidate)

    n = candidate.size
    total = candidate.sum()
    if total < 0.01:                     # avoid degenerate zero‑sum case
        candidate[:] = 1.0
        total = candidate.sum()

    # ------------------------------------------------------------------
    # 2) Record the current best (exactly evaluated) candidate.
    # ------------------------------------------------------------------
    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))   # exact score

    # Pre‑compute the FFT length (constant for a given n).
    nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 3) Greedy local improvement: repeatedly swap a large‑contributing
    #    entry with a small‑contributing entry for the worst convolution
    #    index.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # current convolution and the index attaining its maximum
        fx = np.fft.rfft(candidate, n=nfft)
        conv = np.fft.irfft(fx * fx, n=nfft)[: 2 * n - 1]
        k_max = int(conv.argmax())

        # contribution of each entry to the value at k_max
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        contributions = np.zeros_like(candidate)
        if mask.any():
            contributions[mask] = candidate[idx[mask]] * candidate[j[mask]]

        # indices with largest and smallest contributions (different)
        i_max = int(contributions.argmax())
        i_min = int(contributions.argmin())
        if i_min == i_max:
            i_min = int(rng.integers(0, n - 1))
            while i_min == i_max:
                i_min = int(rng.integers(0, n - 1))

        # propose the swap
        candidate[i_max], candidate[i_min] = candidate[i_min], candidate[i_max]

        # compute the approximate new score (exact evaluation is too costly)
        fx2 = np.fft.rfft(candidate, n=nfft)
        conv2 = np.fft.irfft(fx2 * fx2, n=nfft)[: 2 * n - 1]
        max_val2 = float(conv2.max())
        approx_score2 = 2.0 * n * max_val2 / (total * total)

        # accept if the approximation improves the best known score
        if approx_score2 < best_score:
            best = candidate.copy()
            best_score = approx_score2
        else:
            # revert the unsuccessful swap
            candidate[i_max], candidate[i_min] = candidate[i_min], candidate[i_max]

    # ------------------------------------------------------------------
    # 4) Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    result = [float(v) for v in best]
    if not result:               # should never happen, but keep contract safe
        result = [1.0]
    return result
