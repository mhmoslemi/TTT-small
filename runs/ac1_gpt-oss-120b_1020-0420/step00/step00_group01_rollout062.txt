def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1)
    # Initialize as exponential
    alpha = 1.0 / n
    a = np.exp(-alpha * np.arange(n))
    # Normalize? not needed.
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    for step in range(num_steps):
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        # compute product contributions for each i where both indices in range
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        # indices i_start..i_end inclusive
        idx = np.arange(i_start, i_end + 1)
        partner = max_idx - idx
        contributions = a[idx] * a[partner]
        # select top fraction
        thresh = np.percentile(contributions, 80)  # top 20%
        mask = contributions > thresh
        if mask.any():
            # reduce these a values
            reduce_factor = 0.95  # 5% reduction
            delta = a[idx[mask]] * (1 - reduce_factor)
            a[idx[mask]] *= reduce_factor
            # redistribute the removed mass uniformly
            total_removed = delta.sum()
            a += total_removed / n
        else:
            # fallback random perturb
            a = make_block_perturbation(a, rng)  # reuse earlier function perhaps
        # clip
        np.clip(a, 0.0, 1000.0, out=a)
        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
        # optionally occasional random perturb to escape
        if rng.random() < 0.05:
            a = make_block_perturbation(best, rng)
    return [float(x) for x in best]

import numpy as np

def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # choose a block width between n//200 and n//12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring to keep some symmetry
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, num_steps=1200, **kwargs):
    """Return a non‑negative step‑function that (heuristically) lowers the AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Obtain a valid starting vector (parent or fall‑back)
    # ------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
            parent = tmp.copy()

    # length to work with
    n = parent.size if parent is not None else 6121

    # ------------------------------------------------------------------
    # 2.  Initialise a few exponential candidates and keep the best
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf

    # consider the parent as a possible fallback
    if parent is not None:
        sc = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(sc):
            best = parent.copy()
            best_score = sc

    idx = np.arange(n, dtype=np.float64)

    # multiples of 1/n to try (0.5, 1.0, 1.5, 2.0)
    for mult in (0.5, 1.0, 1.5, 2.0):
        alpha = mult / n
        cand = np.exp(-alpha * idx)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 3.  Refine the best candidate by attacking the current worst shift
    # ------------------------------------------------------------------
    # work on a copy of the current best
    a = best.copy()

    for _ in range(num_steps):
        conv = np.convolve(a, a)
        max_k = int(np.argmax(conv))

        # indices i that participate in this maximal convolution term
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        i_idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_idx = max_k - i_idx

        # contribution of each pair (i, j)
        contrib = a[i_idx] * a[j_idx]

        # reduce the top‑contributing entries a little
        if contrib.size > 0:
            thr = np.percentile(contrib, 85)          # keep the ~15 % largest
            mask = contrib > thr
            if mask.any():
                red_factor = 0.90                      # 10 % reduction
                # amount removed (only for diagnostics, not needed for update)
                # total_removed = (a[i_idx[mask]] * (1 - red_factor)).sum()
                a[i_idx[mask]] *= red_factor

        # occasional random shake‑up to avoid getting stuck
        if rng.random() < 0.02:
            a = make_block_perturbation(a, rng)

        # keep values in the allowed range
        np.clip(a, 0.0, 1000.0, out=a)

        # evaluate and keep the best ever seen
        sc = float(evaluate_sequence(a.tolist()))

            best_score = sc

    # ------------------------------------------------------------------
    # 4.  Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best is None or best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
