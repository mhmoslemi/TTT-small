import numpy as np


def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """
    Search for a non‑negative step function that minimizes the AC1 upper bound
    evaluated by the supplied ``evaluate_sequence`` function.

    The algorithm starts from a feasible vector (the provided ``height_sequence_1``
    if present, otherwise a uniform vector) and performs a simple greedy
    redistribution of mass: in each iteration it identifies the convolution index
    with the largest value and moves a small amount of mass from the entry that
    contributes most to that index to the smallest entry overall.
    The total sum of the vector is preserved, keeping the evaluation
    scale‑invariant.  The best candidate encountered is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # Initialise a feasible candidate (use the supplied construction if it
    # exists and looks sane).
    # ------------------------------------------------------------------ #
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        a = np.clip(base, 0.0, 1000.0)
    except Exception:
        a = np.ones(256, dtype=np.float64)

    n = a.size

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------ #
    # Main greedy search loop.
    #
    # In each iteration:
    #   1. Compute the full linear convolution a*a.
    #   2. Locate the index k where the convolution attains its maximum.
    #   3. Among the indices i that participate in that diagonal (i.e. i+j=k),
    #      pick the one with the largest current height (i1).
    #   4. Pick the overall smallest height (i2) as a sink.
    #   5. Transfer a modest amount of mass from i1 to i2 and keep the change
    #      only if it improves the objective.
    # ------------------------------------------------------------------ #
    for _ in range(num_steps):
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # range of i such that i + (k_max - i) = k_max and both indices are valid
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)

        if i_start > i_end:
            continue

        # index with the largest value in the active range
        sub = a[i_start : i_end + 1]
        if sub.size == 0:
            continue
        i1 = i_start + int(np.argmax(sub))

        # partner index for i1 on the same diagonal (useful for intuition only)
        _ = k_max - i1  # not needed further, kept for clarity

        # global index with the smallest value (the sink)
        i2 = int(np.argmin(a))
        if i2 == i1:
            # pick the second‑smallest if we hit the same index
            sorted_idx = np.argsort(a)
            i2 = int(sorted_idx[1]) if n > 1 else i1

        # move a limited amount of mass from the donor to the sink
        delta = min(a[i1] * 0.5, 0.1)      # step size cap prevents overshooting
        if delta <= 0.0:
            continue

        a[i1] -= delta
        a[i2] += delta
        a = np.clip(a, 0.0, 1000.0)         # respect evaluator’s clipping rules

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
        else:
            # revert the move – it did not improve the objective
            a[i1] += delta
            a[i2] -= delta

    # ------------------------------------------------------------------ #
    # Fallback: if somehow no finite candidate was ever found, return a
    # simple uniform vector (guaranteed to give score 2).
    # ------------------------------------------------------------------ #
    if best_score == float("inf"):
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
