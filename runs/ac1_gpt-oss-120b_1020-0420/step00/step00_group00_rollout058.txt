def peak_reduction(arr, rng):
    a = arr.astype(np.float64, copy=True)
    n = a.size
    conv = np.convolve(a, a)
    max_idx = int(conv.argmax())
    # Determine i range that contributes:
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    i_vals = np.arange(i_start, i_end + 1)
    j_vals = max_idx - i_vals
    # compute product
    prod = a[i_vals] * a[j_vals]
    # find index with max prod
    idx = int(prod.argmax())
    i = int(i_vals[idx])
    j = int(j_vals[idx])
    # choose reduction factor
    alpha = rng.uniform(0.05, 0.15)  # small proportion
    # subtract from i and j
    delta_i = a[i] * alpha
    a[i] = max(a[i] - delta_i, 0.0)
    total_removed = delta_i
    if i != j:
        delta_j = a[j] * alpha
        a[j] = max(a[j] - delta_j, 0.0)
        total_removed += delta_j
    # add removed mass to random other index(es)
    # choose at most two target indices to distribute
    # probability to split or single.
    if total_removed > 0:
        # Choose target indices distinct from i,j
        # For simplicity, add to one random index
        target = int(rng.integers(0, n))
        while target == i or target == j:
            target = int(rng.integers(0, n))
        a[target] = min(a[target] + total_removed, MAX_VAL)
    return a

def block_perturb(arr, rng):
    a = arr.copy()
    n = a.size
    # choose block width relative to n
    min_width = max(1, n // 200)
    max_width = max(2, n // 12)
    width = int(rng.integers(min_width, max_width + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = np.exp(rng.normal(0.0, 0.1))
    a[start:stop] *= scale
    if rng.random() < 0.35:
        a = 0.5 * (a + a[::-1])
    np.clip(a, 0.0, MAX_VAL, out=a)
    return a
```

def propose_candidate(seed=42, num_steps=2000, **kwargs):

        parent = np.asarray(height_sequence_1, dtype=np.float64)

    best = parent.copy()

    # Ensure best_score finite; if not, set best to uniform
    if not np.isfinite(best_score):
         best = np.ones(256, dtype=np.float64)
         best_score = float(evaluate_sequence(best.tolist()))
    for step in range(num_steps):
        # choose mutation type
        if rng.random() < 0.7:

            noise = rng.normal(0.0, 0.02, size=candidate.shape)

            np.clip(candidate, 0.0, 1000.0, out=candidate)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    return [float(x) for x in best]

def generate_sidon_set(n, k_target, rng):
    # Randomized greedy.
    indices = []
    sum_set = set()
    attempts = 0
    while len(indices) < k_target and attempts < k_target * 20:
        cand = int(rng.integers(0, n))
        if cand in indices:
            attempts += 1
            continue
        ok = True
        # compute sums
        cand_sums = [cand + idx for idx in indices] + [2 * cand]
        for s in cand_sums:
            if s in sum_set:
                ok = False
                break
        if ok:
            indices.append(cand)
            for s in cand_sums:
                sum_set.add(s)
        attempts += 1
    return np.array(indices, dtype=int)
```

import numpy as np
from scipy.optimize import linprog
from scipy import sparse

MAX_VAL = 1000.0
TARGET_SCORE = 1.50274349          # target for early stopping (optional)
MAX_ITER = 8                        # LP / blending iterations
MAX_CONS = 30                       # max active convolution constraints per LP
BLEND_T = 0.15                      # blending factor (1‑t)*f + t*g


def _conv_full(x: np.ndarray) -> np.ndarray:
    """Full convolution x*x (numpy uses direct algorithm for modest sizes)."""
    return np.convolve(x, x)


def _active_indices(conv: np.ndarray, max_cons: int) -> np.ndarray:
    """Return up to ``max_cons`` indices of the largest entries of ``conv``."""
    if conv.size <= max_cons:
        return np.arange(conv.size, dtype=int)
    # argpartition gives unordered top‑k indices; we keep them as‑is
    top = np.argpartition(-conv, max_cons - 1)[:max_cons]
    return top


def _build_constraint_matrix(f: np.ndarray, idxs: np.ndarray) -> sparse.csr_matrix:
    """
    Build a (len(idxs) × n) sparse matrix A where
    (A @ g)[k] = (f * g)[idxs[k]] = Σ_i f[i] * g[idxs[k] - i],
    using the convention that out‑of‑range indices are ignored.
    """
    n = f.size
    rows = []
    cols = []
    data = []
    for row, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals                     # corresponding g‑indices
        f_vals = f[i_vals]
        rows.extend([row] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(f_vals.tolist())
    return sparse.csr_matrix((data, (rows, cols)), shape=(len(idxs), n), dtype=np.float64)


def _lp_step(f: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """
    One linear‑programming improvement step.
    Returns the new candidate vector (already blended with ``f``) or ``None`` on failure.
    """
    n = f.size
    conv = _conv_full(f)
    M = conv.max()
    # choose the most binding constraints (largest conv entries)
    act_idx = _active_indices(conv, MAX_CONS)
    A = _build_constraint_matrix(f, act_idx)
    b = np.full(A.shape[0], M, dtype=np.float64)

    # maximize sum(g)  ->  minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, MAX_VAL) for _ in range(n)]

    # solve LP (highs handles sparse A_ub)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    g0 = res.x
    s_f = f.sum()
    s_g = g0.sum()
    if s_g == 0.0:
        return None
    # rescale g to have the same total mass as f (scaling does not affect the bound)
    scale = s_f / s_g
    g = g0 * scale
    # blending
    new_f = (1.0 - BLEND_T) * f + BLEND_T * g
    np.clip(new_f, 0.0, MAX_VAL, out=new_f)
    return new_f


def _block_perturb(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Simple block‑multiplicative perturbation (fallback move).
    """
    a = x.copy()
    n = a.size
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.10))
    a[start:stop] *= factor
    if rng.random() < 0.35:
        a = 0.5 * (a + a[::-1])
    np.clip(a, 0.0, MAX_VAL, out=a)
    return a


def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑empty list of coefficients whose AC1 bound is (hopefully) below 2.
    The algorithm starts from the supplied parent (``height_sequence_1`` if present)
    and performs a small number of LP‑based smoothing steps,
    falling back to cheap block perturbations when the LP fails.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise from the parent construction if it exists and looks sane
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    np.clip(parent, 0.0, MAX_VAL, out=parent)

    # make a working copy and keep the best solution seen so far
    f = parent.copy()
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        # very unlikely, but ensure we have a feasible start
        f = np.ones(256, dtype=np.float64)
        best = f.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    n = f.size

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    for _ in range(MAX_ITER):
        # try an LP‑based smoothing step
        new_f = _lp_step(f, rng)
        if new_f is None:
            # LP failed – fall back to a cheap random move
            new_f = _block_perturb(f, rng)

        # evaluate the candidate
        score = float(evaluate_sequence(new_f.tolist()))
        if np.isfinite(score) and score < best_score:
            best = new_f.copy()
            best_score = score
            # early exit if we have already beaten the target
            if best_score < TARGET_SCORE:
                break
        # keep the current vector for the next iteration (even if not improved)
        f = new_f

    # ------------------------------------------------------------------
    # final sanity check – guarantee a valid, non‑empty result
    # ------------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(v) for v in best]
