    n = parent.size

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Ensure best is valid.

    # Define shape functions
    def exp_decay(r):
        i = np.arange(n)
        return np.power(r, i)

    def sym_center(r):
        # center index (float)
        center = (n - 1) / 2.0
        i = np.arange(n)
        return np.power(r, np.abs(i - center))

    def exp_min(r):
        i = np.arange(n)
        dist = np.minimum(i, n-1-i)
        return np.power(r, dist)

    shape_funcs = [exp_decay, sym_center, exp_min]

    # r list
    r_list = np.concatenate([
        np.linspace(0.6, 0.99, 30, endpoint=False),
        np.linspace(0.99, 0.9999, 30, endpoint=False),
        np.linspace(0.9999, 0.99999, 10)
    ])

    for func in shape_funcs:
        for r in r_list:
            seq = func(r)
            # Clip to bounds
            seq = np.clip(seq, 0.0, 1000.0)
            # Ensure sum >= 0.01
            if seq.sum() < 0.01:
                continue
            # Evaluate
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score

    # Additional random perturbations around best to further improve
    # Perform a few local tweaks using block perturbations
    def block_perturb(seq):
        candidate = seq.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = np.exp(rng.normal(0.0, 0.12))
        candidate[start:stop] *= factor
        # random symmetry
        if rng.random() < 0.35:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0.0, 1000.0)
        return candidate

    # apply some iterations of local search starting from best
    for _ in range(300):

        score = float(evaluate_sequence(cand.tolist()))

def fast_score(seq):
    n = seq.size
    # compute convolution max
    size = 1 << (2*n-1).bit_length()
    # rfft length for real input yields size//2 + 1 points
    f = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f * f, size)
    max_val = conv[:2*n-1].max()
    sum_val = seq.sum()
    if sum_val < 0.01:
        return np.inf
    return float(2 * n * max_val / (sum_val ** 2))

import numpy as np

# -------------------------------------------------------------
# Helper utilities (all top‑level, no closures or lambdas)
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the full self‑convolution of ``seq`` (FFT, O(n log n))."""
    n = seq.size
    size = _next_pow2(2 * n - 1)
    # real FFT, element‑wise product, inverse transform
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)
    # keep only the first 2*n-1 entries (full convolution length)
    return float(conv[: 2 * n - 1].max())


def _fast_score(seq: np.ndarray) -> float:
    """Score 2·n·max(conv)/sum^2 – cheap version using FFT."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    maxv = _conv_max(seq)
    return float(2.0 * seq.size * maxv / (s * s))


def _exact_score(seq: np.ndarray) -> float:
    """Score using the provided evaluator (the ground‑truth)."""
    # evaluate_sequence is provided by the harness
    return float(evaluate_sequence(seq.tolist()))


# -------------------------------------------------------------
# Shape generators – all use O(n) memory
# -------------------------------------------------------------
def _exp_decay(r: float, n: int) -> np.ndarray:
    """a_i = r**i, i = 0..n‑1  (monotone decay)."""
    i = np.arange(n, dtype=np.float64)
    return np.power(r, i)


def _sym_center(r: float, n: int) -> np.ndarray:
    """Symmetric exponential peaked in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    return np.power(r, np.abs(i - centre))


def _exp_min(r: float, n: int) -> np.ndarray:
    """Peak at the ends, decay towards the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    return np.power(r, dist)


# -------------------------------------------------------------
# Perturbation utilities
# -------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random O(n) block perturbation (as in the reference program)."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def make_targeted_perturbation(base: np.ndarray,
                               conv_max_idx: int,
                               rng: np.random.Generator) -> np.ndarray:
    """Bias the block location toward indices that contribute to the current worst lag."""
    n = base.size
    # indices that actually appear in the convolution term with lag = conv_max_idx
    lo = max(0, conv_max_idx - (n - 1))
    hi = min(n - 1, conv_max_idx)

    # pick a random i inside that interval and perturb i and its partner j = k-i
    i = int(rng.integers(lo, hi + 1))
    j = conv_max_idx - i
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.10)))  # moderate log‑normal move
    cand[i] *= factor
    cand[j] *= factor
    # occasional symmetry to keep things balanced
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def propose_candidate(seed: int = 42,
                     num_shape_steps: int = 120,
                     num_local_steps: int = 800,
                     **kwargs):
    """Return a list of non‑negative heights that (heuristically) lowers the AC1 bound."""
    rng = np.random.default_rng(seed)

    # ---- obtain parent sequence (height_sequence_1 may or may not exist) ----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    parent = np.clip(parent, 0.0, 1000.0)

    # ---- fallback baseline (uniform) ----
    best_seq = parent.copy()
    best_score = _exact_score(best_seq)

    # ---------------------------------------------------------
    # 1) Coarse shape exploration (exponential families)
    # ---------------------------------------------------------
    # R values: coarse to fine, emphasising the region (1‑r)≈1/n
    r_vals = np.concatenate((

        np.linspace(0.85, 0.95, 8, endpoint=False),
        np.linspace(0.95, 0.985, 9, endpoint=False),
        np.linspace(0.985, 0.997, 7, endpoint=False),
        np.linspace(0.997, 0.9995, 7, endpoint=False),

    shape_fns = [_exp_decay, _sym_center, _exp_min]

    for fn in shape_fns:

            seq = np.clip(seq, 0.0, 1000.0)
            if seq.sum() < 0.01:
                continue
            # cheap screening
            scr = _fast_score(seq)
            if scr < best_score:
                exact = _exact_score(seq)
                if np.isfinite(exact) and exact < best_score:
                    best_seq = seq
                    best_score = exact

    # ---------------------------------------------------------

    # ---------------------------------------------------------
    # Pre‑compute current worst‑lag index for targeted moves
    # (re‑computed each iteration because the sequence changes)
    for step in range(num_local_steps):
        # recompute convolution only when needed (cheap via FFT)
        conv_full = np.convolve(best_seq, best_seq, mode='full')
        worst_idx = int(np.argmax(conv_full))

        # Choose perturbation type
        if rng.random() < 0.35:                     # targeted move ~35 %

        else:                                        # random block move ~65 %
            cand = make_block_perturbation(best_seq, rng)

        # cheap pre‑screen
        scr = _fast_score(cand)
        if scr >= best_score:
            continue

        # full evaluation
        exact = _exact_score(cand)
        if np.isfinite(exact) and exact < best_score:
            best_seq = cand
            best_score = exact

    # ---------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ---------------------------------------------------------
    return [float(x) for x in best_seq]
