    objective) and performs a cheap stochastic perturbation loop, keeping any
    improvement that may arise from floating‑point quirks.
    """
    rng = np.random.default_rng(seed)

    n = _initial_length()
    best = _uniform_candidate(n, value=1.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # Very cheap stochastic refinement – normally uniform already attains the
    # theoretical minimum (objective = 2).  The loop keeps any accidental
    # improvement due to round‑off while staying O(n) in memory.
    for _ in range(num_steps):
        cand = best.copy()

        # Choose two distinct positions.
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Apply a random multiplicative perturbation that roughly conserves
        # the total sum but stays within bounds.
        factor = np.exp(rng.normal(0.0, 0.1))
        cand[i] *= factor
        cand[j] /= factor

        # Enforce the problem’s hard bounds.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Evaluate the candidate.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list as required.
    return [float(x) for x in best]
