def conv_max(a):
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A*A, m)[:2*n-1]
    return C.max()
```

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # initial seq: uniform (maybe random)

    best_score = float(evaluate_sequence(best.tolist()))
    # Precompute FFT size
    m_fft = 1 << ((2*n-1).bit_length())
    
    def compute_score(arr):
        # Use FFT for fast max, but also compute exact via evaluate for acceptance
        # We can compute max via FFT approx (exact due to floating rounding) but acceptable
        A = np.fft.rfft(arr, m_fft)
        conv = np.fft.irfft(A*A, m_fft)[:2*n-1]
        maxb = conv.max()
        s = arr.sum()
        if s < 0.01:
            return np.inf
        return 2 * n * maxb / (s * s)
    
    # Track best
    for step in range(num_iter):

        if rng.random() < 0.5:
            # multiplicative block perturbation as previous but with log-normal
            width = int(rng.integers(1, max(2, n // 10)))  # block size up to n/10

            factor = float(np.exp(rng.normal(0.0, 0.2)))
            candidate = best.copy()
            candidate[start:start+width] *= factor
        else:
            # swap two random indices
            i, j = rng.integers(0, n, size=2)
            candidate = best.copy()
            candidate[i], candidate[j] = candidate[j], candidate[i]
        # clip

        if score_est < best_score:
            # verify with evaluator
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score

def propose_candidate(seed=42, max_iters=3000, **kwargs):

    # get length n

        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    except Exception:
        n = 256
    # precompute FFT length
    fft_len = 1 << ((2*n-1).bit_length())
    # function to compute max convolution
    def max_conv(v):
        A = np.fft.rfft(v, fft_len)
        conv = np.fft.irfft(A*A, fft_len)[:2*n-1]
        return conv.max()
    # initial candidates: uniform and exponential family
    best_vec = np.ones(n, dtype=np.float64) / n
    best_score = 2 * n * max_conv(best_vec)   # sum =1
    # exponential scan
    for r in np.linspace(0.99, 0.99995, 30):

        vec_sum = vec.sum()
        if vec_sum == 0:

        vec = vec / vec_sum
        sc = 2 * n * max_conv(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
    # local search
    cur = best_vec.copy()
    cur_score = best_score
    for it in range(max_iters):
        # propose modification
        cand = cur.copy()
        # choose operation
        op = rng.random()
        if op < 0.4:

            width = int(rng.integers(1, max(2, n // 20)))  # up to n/20

import numpy as np

# ------------------------------------------------------------
# Helper: fast maximum of the full linear convolution a*a
# ------------------------------------------------------------
def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return the maximum value of np.convolve(a, a) using an FFT of length fft_len."""
    # real FFT, squared magnitude gives convolution in frequency domain
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)
    # the linear convolution has length 2*len(a)-1, take the relevant prefix
    conv = conv[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iters: int = 4000, **kwargs):
    """Search for a non‑negative step‑function that lowers the AC1 upper bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size (use the supplied parent if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256
        else:
            n = base.size
    else:
        n = 256

    # ------------------------------------------------------------
    # FFT parameters (next power of two >= 2*n-1)
    # ------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Initialise a candidate – we start from a short exponential family
    # ------------------------------------------------------------
    best_vec = np.ones(n, dtype=np.float64) / n          # uniform (sum = 1)
    best_score = 2.0 * n * max_convolution(best_vec, fft_len)

    # Scan a few exponentials, keep the best one as a starting point
    for r in np.linspace(0.99, 0.99995, 21):
        exp_vec = np.power(r, np.arange(n, dtype=np.float64))
        s = exp_vec.sum()
        if s == 0.0:
            continue
        exp_vec /= s                                   # normalise to sum = 1
        sc = 2.0 * n * max_convolution(exp_vec, fft_len)
        if sc < best_score:
            best_score = sc
            best_vec = exp_vec.copy()

    # ------------------------------------------------------------
    # Local optimisation: gradient descent on the current worst shift,
    # plus occasional random block perturbations to escape plateaus.
    # ------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_score = best_score

    for it in range(max_iters):
        # full convolution, its maximum and the index where it occurs
        A = np.fft.rfft(cur_vec, fft_len)
        conv_full = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
        max_idx = int(conv_full.argmax())
        max_val = float(conv_full[max_idx])

        # --------------------------------------------------------
        # Gradient of the peak (for shift = max_idx):  ∂b_k/∂a_i = 2 * a_{k-i}
        # --------------------------------------------------------
        grad = np.zeros_like(cur_vec)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        if lo <= hi:
            i_vals = np.arange(lo, hi + 1, dtype=int)
            j_vals = max_idx - i_vals
            grad[i_vals] = 2.0 * cur_vec[j_vals]

        # Step size slowly decreasing; keep it modest to stay in the feasible region
        eta = 0.05 / (1.0 + it * 0.001)

        # Gradient step, projection onto the non‑negative orthant and re‑normalisation
        cand = cur_vec - eta * grad
        np.clip(cand, 0.0, None, out=cand)
        total = cand.sum()
        if total == 0.0:
            cand = cur_vec.copy()          # fallback – should almost never happen
        else:
            cand /= total                    # keep the sum equal to 1 (scaling‑invariant)

        # Evaluate the new candidate quickly
        sc = 2.0 * n * max_convolution(cand, fft_len)

        # Accept improvement
        if sc < cur_score:
            cur_vec = cand
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()
            continue

        # Occasionally try a random block perturbation (simulated‑annealing style)
        if rng.random() < 0.07:
            width = int(rng.integers(1, max(2, n // 80)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand2 = cur_vec.copy()
            cand2[start : start + width] *= factor
            np.clip(cand2, 0.0, None, out=cand2)
            tot2 = cand2.sum()
            if tot2 > 0.0:
                cand2 /= tot2
                sc2 = 2.0 * n * max_convolution(cand2, fft_len)
                # Accept if it improves the current solution
                if sc2 < cur_score:
                    cur_vec = cand2
                    cur_score = sc2
                    if sc2 < best_score:
                        best_score = sc2
                        best_vec = cand2.copy()

    # ------------------------------------------------------------------
    # Final safety checks and conversion to the required list format
    # ------------------------------------------------------------------
    final = np.clip(best_vec, 0.0, 1000.0)
    if final.sum() < 0.01:                     # ensure the evaluator does not reject
        final = np.ones_like(final) * (0.01 / final.size)

    return [float(v) for v in final.tolist()]
