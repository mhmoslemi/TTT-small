        max_idx = int(np.argmax(conv))
        # compute overlapping range
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)

        overlapping_vals = best_seq[i_start:i_end+1]

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j: continue

            # ensure bounds etc...
            # ...
        else:
            local_i = i_start + int(np.argmax(overlapping_vals))
            j = max_idx - local_i
            # choose recipient index outside overlapping region (or random)
            # find a candidate index outside region randomly
            candidates = list(set(range(n)) - set(range(i_start, i_end+1)))
            if candidates:
                rec = rng.choice(candidates)
            else:
                rec = rng.integers(0, n)
            # compute epsilon
            eps = min(best_seq[local_i], best_seq[j]) * 0.2
            if eps <= 0: continue
            # propose new sequence
            cand = best_seq.copy()
            cand[local_i] -= eps
            cand[j] -= eps
            cand[rec] += 2*eps
            # clip to [0,1000] after adjustment
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_seq = cand
                best_score = score
                continue
            # else maybe try moving only from one index
            # ...
        # optionally random perturb occasional
        if it % 50 == 0:
            # occasional random block perturb similar to earlier
            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))

def propose_candidate(seed=42, max_iters=1000, **kwargs):
    rng = np.random.default_rng(seed)
    # starting sequence: maybe use height_sequence_1 if provided else uniform
    base = np.asarray(height_sequence_1, dtype=np.float64)
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)
    seq = np.clip(base, 0.0, 1000.0)

    n = seq.size
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    for it in range(max_iters):
        # compute conv and max index
        conv = np.convolve(best_seq, best_seq)
        max_idx = int(np.argmax(conv))
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        # if overlap size small, random move
        if i_end < i_start:
            # no overlap (shouldn't happen)
            continue
        # compute contributions
        # vector of indices
        idxs = np.arange(i_start, i_end + 1)
        partners = max_idx - idxs
        contrib = best_seq[idxs] * best_seq[partners]
        # choose i with max contribution
        max_pos = int(idxs[np.argmax(contrib)])
        max_partner = int(partners[np.argmax(contrib)])

        # find position with minimal value not in the pair
        # create mask to avoid max and partner
        mask = np.ones(n, dtype=bool)
        mask[max_pos] = False
        mask[max_partner] = False
        # Get index of minimal value among masked positions
        if np.any(mask):
            min_pos = int(np.argmin(best_seq * mask))  # this multiplies by mask (False=>0)
        else:
            min_pos = rng.integers(0, n)

        # epsilon scale
        if max_pos == max_partner:
            eps = best_seq[max_pos] * 0.1
            if eps <= 0:
                continue
            cand = best_seq.copy()
            cand[max_pos] -= eps
            cand[min_pos] += eps
        else:
            eps = min(best_seq[max_pos], best_seq[max_partner]) * 0.1
            if eps <= 0:
                continue
            cand = best_seq.copy()
            cand[max_pos] -= eps
            cand[max_partner] -= eps
            cand[min_pos] += 2*eps

        # clip

import numpy as np

def propose_candidate(seed=42, max_iters=400, **kwargs):
    """
    Search for a non‑negative step‑function that lowers the AC1 upper bound.
    The algorithm starts from the provided parent (height_sequence_1) and
    repeatedly reduces the largest coefficient of the self‑convolution by
    moving a small amount of mass from the most contributing pair to a
    low‑value location.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # initialise candidate (fallback to uniform if parent is missing)
    # ------------------------------------------------------------------ #
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)

    seq = np.clip(base, 0.0, 1000.0)
    n = seq.size

    best = seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------ #
    # main optimisation loop
    # ------------------------------------------------------------------ #
    for it in range(max_iters):
        # compute full convolution and locate its maximal entry
        conv = np.convolve(best, best)
        k_max = int(np.argmax(conv))

        # overlapping index interval for this k_max
        i0 = max(0, k_max - (n - 1))
        i1 = min(n - 1, k_max)
        if i1 < i0:
            continue  # should not happen, but stay safe

        idx = np.arange(i0, i1 + 1)
        partner = k_max - idx
        contrib = best[idx] * best[partner]

        # pair (i, j) that contributes most to the current peak
        max_loc = int(idx[np.argmax(contrib)])
        max_part = int(partner[np.argmax(contrib)])

        # pick a low‑value position that is not involved in the peak
        mask = np.ones(n, dtype=bool)
        mask[max_loc] = False
        mask[max_part] = False
        low_candidates = np.where(mask)[0]
        if low_candidates.size:
            low_idx = int(low_candidates[np.argmin(best[low_candidates])])
        else:
            low_idx = int(rng.integers(0, n))

        # amount of mass to relocate (10 % of the smaller of the two)
        if max_loc == max_part:
            eps = best[max_loc] * 0.10
            if eps <= 0.0:
                continue
            cand = best.copy()
            cand[max_loc] -= eps
            cand[low_idx] += eps
        else:
            eps = min(best[max_loc], best[max_part]) * 0.10
            if eps <= 0.0:
                continue
            cand = best.copy()
            cand[max_loc] -= eps
            cand[max_part] -= eps
            cand[low_idx] += 2.0 * eps

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best = cand
            best_score = s
            continue

        # occasional random block perturbation for exploration
        if it % 20 == 0:
            w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            st = int(rng.integers(0, max(1, n - w + 1)))
            en = min(n, st + w)
            cand2 = best.copy()
            factor = float(np.exp(rng.normal(0.0, 0.08)))
            cand2[st:en] *= factor
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            s2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(s2) and s2 < best_score:
                best = cand2
                best_score = s2

    return [float(v) for v in best]
