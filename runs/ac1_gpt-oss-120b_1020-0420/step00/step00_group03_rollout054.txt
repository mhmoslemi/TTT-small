def propose_candidate(seed=42, num_r=5, num_iter=200, **kwargs):
    rng = np.random.default_rng(seed)
    n = height_sequence_1.shape[0]
    # start with parent fallback default uniform maybe
    # generate initial candidates using exponential shape
    best_seq = None
    best_score = np.inf
    # try several r values
    for _ in range(num_r):
        # sample r close to 1: take u uniformly in [0.9,0.9999] maybe
        # but we want r near 1. Use rng.uniform(0.999,0.9999)
        r = rng.uniform(0.998, 0.9999)
        seq = np.exp(np.log(r) * np.arange(n)) * 1000   # a0=1000
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
    # if not found valid, fallback to uniform
    if best_seq is None:
        best_seq = np.full(n, 1000.0)  # though too large? but uniform gives 2.
        best_score = float(evaluate_sequence(best_seq.tolist()))
    # local search
    for _ in range(num_iter):
        # pick random block to perturb, similar to original make_block_perturbation
        candidate = best_seq.copy()
        # choose a block width
        width = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, max(1, n - width + 1))
        stop = min(n, start + width)
        # perturb by multiplicative log-normal factor
        factor = np.exp(rng.normal(0.0, 0.05))
        candidate[start:stop] *= factor
        # small random jitter to whole vector
        jitter = np.exp(rng.normal(0.0, 0.01, size=n))
        candidate *= jitter
        # enforce bounds
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate.copy()
    # final candidate: ensure sum>0.01; already true.
    return best_seq.tolist()
