def _initial_sequence(seed: int, n: int) -> np.ndarray:
    """Return a non‑negative starting vector of length n."""
    rng = np.random.default_rng(seed)
    # try the provided parent

        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size == n
            and np.isfinite(parent).all()
        ):
            a = np.clip(parent, 0.0, 1000.0)
        else:
            raise ValueError
    except Exception:
        a = np.ones(n, dtype=np.float64)
    # break perfect symmetry with a tiny random perturbation
    a += rng.normal(scale=0.01, size=n)
    a = np.clip(a, 0.0, 1000.0)
    # normalise the ℓ₁‑norm to √(2n) (the evaluator is homogeneous)
    target_sum = np.sqrt(2.0 * n)
    s = a.sum()
    if s > 0.0:
        a *= target_sum / s
    else:
        a = np.full(n, target_sum / n, dtype=np.float64)
    return a


def propose_candidate(seed: int = 42, max_steps: int = 2000, delta_factor: float = 0.20, **kwargs):
    """
    Heuristic that repeatedly removes a small amount of mass from the
    most offending convolution diagonal and redistributes it elsewhere.
    The result is kept non‑negative, bounded and renormalised after each
    move. The best candidate seen is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    n = len(np.asarray(height_sequence_1, dtype=np.float64))
    a = _initial_sequence(seed, n)

    # best solution seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # parameters controlling stagnation handling
    stagnation = 0

        k = int(np.argmax(conv))
        M = conv[k]

        end = min(n - 1, k)

        found = False
        for _ in range(5):

            j = k - i
            if i != j and a[i] > 0.0 and a[j] > 0.0:
                found = True
                break
        if not found:

            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
