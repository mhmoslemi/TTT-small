    # Determine length n.

        base = np.asarray(height_sequence_1, dtype=np.float64)

            raise ValueError
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)
    else:
        n = base.size

    # Compute uniform candidate.
    uniform = np.full(n, 1.0, dtype=np.float64)

    # Evaluate scores.
    best = None
    best_score = np.inf

    for candidate in (base, uniform):
        # Ensure candidate is finite etc.
        # Clip to [0,1000].
        cand = np.clip(candidate, 0.0, 1000.0)
        # Convert to list for evaluator
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand

    # If best still None (unlikely), set uniform.
    if best is None:
        best = uniform

    return [float(x) for x in best]
