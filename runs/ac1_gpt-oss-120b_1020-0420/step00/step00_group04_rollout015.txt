def propose_candidate(seed=42, num_steps=500, **kwargs):
    """Return a uniform nonnegative sequence of length matching height_sequence_1 or fallback."""
    # Determine length from existing parent if available
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = None
    if base is None or base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        n = 256
    else:
        n = base.size

    # Uniform constant (value 1.0 gives objective 2)
    candidate = np.full(n, 1.0, dtype=np.float64)

    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))

    # Simple random perturbations with acceptance if improve (unlikely)
    rng = np.random.default_rng(seed)
    for _ in range(num_steps):
        # perturb a random block multiplicatively
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        proposal = best.copy()
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        proposal[start:start+width] *= factor
        # optional reflection
        if rng.random() < 0.35:
            proposal = 0.5 * (proposal + proposal[::-1])
        # Clip into allowed range
        np.clip(proposal, 0.0, 1000.0, out=proposal)
        score = float(evaluate_sequence(proposal.tolist()))
        if np.isfinite(score) and score < best_score:
            best = proposal
            best_score = score

    # Return best list
    return [float(x) for x in best]

import numpy as np

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function sequence that (up to floating‑point
    noise) is optimal for the built‑in AC1 bound.
    The sequence length matches the provided ``height_sequence_1`` when
    available; otherwise a safe fallback length of 256 is used.
    """
    # ------------------------------------------------------------------
    # Determine target length -------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = None
    if (
        base is None
        or base.ndim != 1
        or base.size == 0
        or not np.isfinite(base).all()
    ):
        n = 256
    else:
        n = int(base.size)

    # ------------------------------------------------------------------
    # Start from a constant (the analytically optimal shape for this bound) --
    candidate = np.full(n, 1.0, dtype=np.float64)

    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Small stochastic hill‑climb – only accepts genuine improvements -----
    rng = np.random.default_rng(seed)
    for _ in range(num_steps):
        # pick a random contiguous block
        min_w = max(1, n // 200)
        max_w = max(2, n // 12)
        width = int(rng.integers(min_w, max_w))
        start = int(rng.integers(0, max(1, n - width + 1)))

        proposal = best.copy()
        # multiplicative perturbation of the block
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        proposal[start:start + width] *= factor

        # occasional mirroring (provides a different direction)
        if rng.random() < 0.35:
            proposal = 0.5 * (proposal + proposal[::-1])

        # enforce the problem’s hard bounds
        np.clip(proposal, 0.0, 1000.0, out=proposal)

        score = float(evaluate_sequence(proposal.tolist()))
        if np.isfinite(score) and score < best_score:
            best = proposal
            best_score = score

    # ------------------------------------------------------------------
    # Return the best candidate discovered as a plain Python list -----------
    return [float(x) for x in best]
