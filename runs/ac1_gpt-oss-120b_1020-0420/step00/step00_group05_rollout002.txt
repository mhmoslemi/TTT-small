import numpy as np

# ------------------------------------------------------------
# Helper: a cheap O(n) perturbation of a sequence.
# ------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative O(n) perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # choose block width between n/200 and n/12 (inclusive)
    w_lo = max(1, n // 200)
    w_hi = max(2, n // 12)
    width = int(rng.integers(w_lo, w_hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation on the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # optional symmetrisation
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce bounds required by the evaluator
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 200, **kwargs):
    """
    Return a list of non‑negative coefficients that (approximately) minimize the
    AC1 upper bound.  The routine prefers a uniform sequence (which is optimal)
    and then performs a brief local random search for any accidental improvement.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Obtain an initial parent sequence; use the provided construction if
    # available, otherwise fall back to a modest sized uniform array.
    # -----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # fallback length chosen to stay well within typical limits
        parent = np.full(256, 1.0, dtype=np.float64)

    n = parent.size

    # -----------------------------------------------------------------
    # Candidate 1: the (possibly clipped) parent itself.
    # Candidate 2: a uniform sequence (the proven optimal shape).
    # -----------------------------------------------------------------
    best = parent
    best_score = float(evaluate_sequence(best.tolist()))

    uniform = np.full(n, 1.0, dtype=np.float64)          # any positive constant works
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if np.isfinite(uniform_score) and uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # -----------------------------------------------------------------
    # Brief stochastic local search around the current best.
    # Each iteration uses only O(n) extra memory.
    # -----------------------------------------------------------------
    for _ in range(num_steps):
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # -----------------------------------------------------------------
    # Return a plain Python list as required by the harness.
    # -----------------------------------------------------------------
    return [float(x) for x in best]
