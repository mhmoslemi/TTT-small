    if f.ndim != 1 or f.size == 0 or not np.isfinite(f).all():
        f = np.ones(256, dtype=np.float64)
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # iterative LP improvement
    for iter_idx in range(5):
        # compute convolution maximum
        b = np.convolve(f, f)
        M = float(b.max())
        # pick constraints near max
        # scaling: top fraction
        threshold = M * 0.9
        ind_max = np.where(b >= threshold)[0]
        # ensure we have at most, say, 50 constraints
        if len(ind_max) > 50:
            # randomly sample 50 of them
            ind_max = rng.choice(ind_max, size=50, replace=False)
        if len(ind_max) == 0:
            break
        # Build sparse matrix
        from scipy import sparse
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(ind_max):
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            rows.extend([row_idx]*len(i_range))
            cols.extend(j_range.tolist())
            data.extend(f[i_range].tolist())
        A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(len(ind_max), n))
        # Solve LP
        c = -np.ones(n)  # minimize -sum(g) => maximize sum(g)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=np.full(len(ind_max), M), bounds=bounds, method='highs')
        except Exception:
            res = None
        if res is None or not res.success:
            # fallback random perturbation
            candidate = f * np.exp(rng.normal(0.0, 0.1, size=n))
            candidate = np.clip(candidate, 0.0, 1000.0)
        else:
            g = res.x
            # Rescale g to same sum as f
            sum_f = f.sum()
            sum_g = g.sum()
            if sum_g > 0:
                g = g * (sum_f / sum_g)
            else:
                g = f.copy()
            # small mixing
            t = 0.5
            candidate = (1 - t) * f + t * g
        # evaluate and keep improvement
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            f = candidate  # continue using improved f
        else:
            # if not improvement maybe try random perturbation
            f = candidate
    return [float(x) for x in best]

    rows = []
    cols = []
    data = []

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        rows.extend([row_idx] * i.size)
        cols.extend(j.tolist())
        data.extend(f[i].tolist())
    return sparse.csr_matrix((data, (rows, cols)), shape=(len(ks), n))

def lp_step(f, M, rng, max_constraints=15):
    # Choose up to max_constraints indices where convolution is maximal
    b = np.convolve(f, f)
    # Use argsort to get top values
    if len(b) <= max_constraints:
        ks = np.arange(len(b))
    else:
        # select indices with largest values
        ks = np.argpartition(b, -max_constraints)[-max_constraints:]
    ks = np.sort(ks)
    A = build_constraint_matrix(f, ks)
    c = -np.ones(f.shape[0])  # minimize negative sum => maximize sum
    bounds = [(0, None)] * f.shape[0]
    rhs = np.full(len(ks), M, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method='highs')
    except Exception:
        return None
    if not res.success:
        return None
    g = res.x
    # Scale g to have same L1 norm as f
    sum_f = float(f.sum())
    sum_g = float(g.sum())
    if sum_g > 0:
        g = g * (sum_f / sum_g)
    else:
        g = f.copy()
    return g

def _build_constraint_matrix(f: np.ndarray, ks: np.ndarray) -> "sparse.csr_matrix":
    """Construct A_ub for constraints (f * g)_k <= M, k in ks."""
    n = f.shape[0]
    rows, cols, data = [], [], []
    for row_idx, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        j = k - i
        rows.extend([row_idx] * i.size)
        cols.extend(j.tolist())
        data.extend(f[i].tolist())
    return sparse.csr_matrix((data, (rows, cols)), shape=(len(ks), n))


def _lp_step(f: np.ndarray, M: float, rng: np.random.Generator,
             max_constraints: int = 12) -> np.ndarray | None:
    """One LP iteration: given f and current bound M, produce a new g."""
    # identify the most critical convolution indices
    full_conv = np.convolve(f, f)
    if full_conv.size <= max_constraints:
        ks = np.arange(full_conv.size)
    else:
        ks = np.argpartition(full_conv, -max_constraints)[-max_constraints:]
    ks = np.sort(ks)

    A = _build_constraint_matrix(f, ks)
    c = -np.ones(f.shape[0], dtype=np.float64)          # min -sum(g)  == max sum(g)
    bounds = [(0.0, None)] * f.shape[0]
    rhs = np.full(len(ks), M, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method='highs')
    except Exception:                                   # pragma: no cover

        return None

    sum_f = float(f.sum())
    sum_g = float(g.sum())

        g *= sum_f / sum_g      # rescale to keep total mass unchanged
    else:
        g = f.copy()
    return g
