Define function flatten_step(a, eta=0.1):
    conv = np.convolve(a, a)
    k = conv.argmax()
    n = len(a)
    i_start = max(0, k-(n-1))
    i_end = min(k, n-1)
    idx = np.arange(i_start, i_end+1)
    partners = k - idx
    grad = np.zeros_like(a)
    grad[idx] = a[partners]  # gradient for each i in support is a_partner
    # for simplicity treat gradient only for i in support
    a_new = a - eta * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    # renorm to keep sum as prior
    sum_old = a.sum()
    sum_new = a_new.sum()
    if sum_new > 0:
        a_new = a_new * (sum_old / sum_new)
    return a_new
