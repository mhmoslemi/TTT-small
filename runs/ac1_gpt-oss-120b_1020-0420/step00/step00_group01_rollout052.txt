import numpy as np

# The evaluator `evaluate_sequence` is provided by the harness.

def _build_mask(rng: np.random.Generator, max_allowed: int) -> np.ndarray:
    """
    Construct a binary mask `a` of length n such that the full convolution
    of `a` with itself never exceeds `max_allowed`.  The construction is
    greedy: a random ordering of indices is examined and an index is kept
    iff adding it respects the bound.
    """
    n = len(height_sequence_1)          # length of the problem instance
    sum_counts = np.zeros(2 * n - 1, dtype=np.int32)   # ordered pair counts
    mask = np.zeros(n, dtype=np.uint8)                 # final 0/1 mask
    order = np.arange(n, dtype=np.int32)
    rng.shuffle(order)                                 # random permutation

    selected = []                                      # list of kept indices
    for i in order:
        # check the self‑pair contribution (shift 2*i)
        if sum_counts[2 * i] + 1 > max_allowed:
            continue

        # check contributions with already selected indices
        ok = True
        for j in selected:
            if sum_counts[i + j] + 2 > max_allowed:
                ok = False
                break
        if not ok:
            continue

        # all checks passed – keep i
        mask[i] = 1
        sum_counts[2 * i] += 1               # (i,i)
        for j in selected:
            s = i + j
            sum_counts[s] += 2                # (i,j) and (j,i)
        selected.append(i)

    return mask.astype(np.float64)            # convert to float for the evaluator


def propose_candidate(seed: int = 42, num_trials: int = 5, **kwargs):
    """
    Search for a non‑negative step‑function (binary mask) whose
    AC1 upper bound is low.  Multiple greedy constructions are tried
    with different random seeds; the best finite candidate is returned.
    """
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1)

    # Fallback: a simple uniform vector (guaranteed to be valid)
    fallback = np.ones(n, dtype=np.float64)
    best_seq = fallback.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Try a few greedy masks with a modest convolution bound.
    # The bound 10 is large enough to admit a reasonably sized set.
    max_allowed = 10

        score = float(evaluate_sequence(cand.tolist()))
