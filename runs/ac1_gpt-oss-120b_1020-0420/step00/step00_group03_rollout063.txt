import numpy as np

def _conv_max(arr: np.ndarray) -> float:
    """
    Compute the maximum entry of the linear convolution of `arr` with itself
    using an FFT (O(n log n)) implementation.
    """
    n = arr.size
    # size for a power‑of‑two FFT that can hold the full convolution
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fft * fft, size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray) -> float:
    """
    Return the AC1 upper‑bound score for `arr`.
    """
    s = float(arr.sum())
    if s < 0.01:
        return float("inf")
    max_b = _conv_max(arr)
    n = arr.size
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound
    is as small as possible.  The routine starts from the supplied
    `height_sequence_1` (if present), tries a few handcrafted constructions
    and then performs a lightweight simulated‑annealing style local search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Base candidate – the provided construction (if any)
    # ------------------------------------------------------------------
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:                     # fall back to a flat vector
        n = kwargs.get("length", 2535)
        base = np.full(n, 1.0, dtype=np.float64)

    best = base.copy()
    best_score = _score(best)

    n = best.size

    # ------------------------------------------------------------------
    # 2.  Hand‑crafted two‑level families (λ + d·sign)
    # ------------------------------------------------------------------
    for _ in range(150):
        # choose average level λ and deviation ratio r = d/λ
        lam = rng.uniform(0.2, 5.0)                     # average value
        r = rng.uniform(0.3, 0.9)                       # 0 < r < 1  (ensures non‑negativity)
        d = lam * r

        # random signs (+1 / -1) with zero mean
        signs = rng.choice([-1, 1], size=n)
        cand = lam + d * signs
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 3.  Simple mass‑transfer local search
    #    (move a small amount from a larger entry to a smaller one)
    # ------------------------------------------------------------------
    max_no_improve = 300
    no_improve = 0
    while no_improve < max_no_improve:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a fraction of the larger entry to the smaller one
        if best[i] > best[j]:
            donor, recv = i, j
        else:
            donor, recv = j, i
        if best[donor] <= 1e-12:
            continue
        delta = rng.uniform(0.0, 0.05) * best[donor]   # up to 5 % of the donor
        cand = best.copy()
        cand[donor] -= delta
        cand[recv] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc
            no_improve = 0
        else:
            no_improve += 1

    # ------------------------------------------------------------------
    # 4.  Return the best discovered candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
