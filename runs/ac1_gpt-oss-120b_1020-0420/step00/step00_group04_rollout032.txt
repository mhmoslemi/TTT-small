import numpy as np

def _symmetric_exponential(length: int, r: float, max_val: float = 1000.0) -> np.ndarray:
    """Return a length‑`length` non‑negative step function.
    The sequence is symmetric, decreasing exponentially from both ends:
        a[i] = max_val * r ** min(i, length-1-i)
    """
    idx = np.arange(length, dtype=np.float64)
    dist = np.minimum(idx, length - 1 - idx)
    # r may be exactly 1.0 for safety – avoid 0**0 by using where
    seq = max_val * np.exp(np.log(r) * dist)
    return np.clip(seq, 0.0, max_val)
