import numpy as np

def make_block_perturbation(base, rng):
    """Create a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.array(base, copy=True)
    n = cand.size
    # width of the perturbed block (between n/200 and n/12)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional slight symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def powerlaw_sequence(length, exponent):
    """
    Return a non‑negative sequence a_i = (i+1)^{-exponent}.
    Values are <= 1, so they respect the 0‑1000 bound automatically.
    """
    idx = np.arange(1, length + 1, dtype=np.float64)
    seq = np.power(idx, -exponent)
    return np.clip(seq, 0.0, 1000.0)
