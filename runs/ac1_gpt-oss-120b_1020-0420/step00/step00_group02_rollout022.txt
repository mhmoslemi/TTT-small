def conv_max(a):
    n = a.size
    size = 1 << (2*n-1).bit_length()  # next power of two >= 2n-1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    # take first 2n-1 values
    max_val = conv[:2*n-1].max()
    return max_val
```

def generate_candidate(method):
    if method == 'uniform':
        a = np.full(n, target_sum / n)
    elif method == 'exponential':
        a = rng.exponential(scale=1.0, size=n)
        a = np.clip(a, 0.0, 1000.0)
    elif method == 'decreasing_exp':
        alpha = rng.uniform(0.0, 5.0 / n)  # choose moderate
        i = np.arange(n)
        a = np.exp(-alpha * i)
    elif method == 'increasing_exp':
        alpha = rng.uniform(0.0, 5.0 / n)
        i = np.arange(n)
        a = np.exp(alpha * i)  # may overflow large, clip to 1000
    elif method == 'sorted_random':
        a = rng.random(n)  # uniform 0-1
        # optionally sort descending to create decreasing sequence
        a = np.sort(a)[::-1]
    else:
        a = rng.random(n) * 0.1
    # Normalize sum to target_sum
    total = a.sum()
    if total == 0:
        a = np.full(n, target_sum / n)
    else:
        a *= (target_sum / total)
    # Clip after scaling maybe needed
    np.clip(a, 0.0, 1000.0, out=a)
    # Renorm if clipping changed sum
    cur_sum = a.sum()
    if cur_sum == 0 or cur_sum == 0:
        a = np.full(n, target_sum / n)
        cur_sum = a.sum()
    a *= (target_sum / cur_sum)
    return a
```

    if n not in _fft_len_cache:
        _fft_len_cache[n] = _next_pow2(2 * n - 1)
    return _fft_len_cache[n]


def _max_convolution(a: np.ndarray) -> float:
    """Return max coefficient of a * a using an FFT‑based convolution."""
    n = a.size
    size = _fft_len(n)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())


def _normalize(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale to the given sum and enforce the [0,1000] box."""
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, target_sum / a.size)
        s = a.sum()
    a = a * (target_sum / s)
    np.clip(a, 0.0, 1000.0, out=a)
    # re‑scale if clipping altered the total mass
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, target_sum / a.size)
    else:
        a = a * (target_sum / s)
    return a


def _score_from_max(max_val: float, a_sum: float, n: int) -> float:
    """Exact score corresponding to a given max convolution value."""
    return 2.0 * n * max_val / (a_sum * a_sum)

# Main candidate proposal
# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """Return a height sequence with a reduced AC1 upper bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) obtain a base vector (fallback to uniform if needed)
    # ------------------------------------------------------------------
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        if raw.ndim != 1 or raw.size == 0 or not np.isfinite(raw).all():
            raw = None
    except Exception:  # height_sequence_1 may be absent
        raw = None

    if raw is None:
        n = 256  # a modest fallback length
        a = np.full(n, 1.0, dtype=np.float64)
    else:
        n = int(raw.size)
        a = np.clip(raw, 0.0, 1000.0)

    target_sum = np.sqrt(2.0 * n)
    a = _normalize(a, target_sum)

    # ------------------------------------------------------------------
    # 2) keep the best vector seen so far
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = _score_from_max(_max_convolution(best), best.sum(), n)

    # ------------------------------------------------------------------
    # 3) random exploration (cheap, many draws)
    # ------------------------------------------------------------------
    # a few constructive families + pure random draws
    families = ["uniform", "exponential", "desc_sorted", "asc_sorted", "exp_decay", "linear"]
    for _ in range(400):
        kind = rng.choice(families)
        if kind == "uniform":
            cand = np.full(n, target_sum / n, dtype=np.float64)
        elif kind == "exponential":
            cand = rng.exponential(scale=1.0, size=n)
        elif kind == "desc_sorted":
            cand = np.sort(rng.random(n))[::-1]
        elif kind == "asc_sorted":
            cand = np.sort(rng.random(n))
        elif kind == "exp_decay":
            # a_i = exp(-alpha * i) , alpha chosen randomly
            alpha = rng.uniform(0.0, 5.0 / n)
            idx = np.arange(n, dtype=np.float64)
            cand = np.exp(-alpha * idx)
        else:  # linear ramp
            slope = rng.uniform(-1.0, 1.0)
            idx = np.arange(n, dtype=np.float64)
            cand = 0.5 + slope * (idx - n / 2) / n  # centred around 0.5
            cand = np.clip(cand, 0.0, None)

        cand = _normalize(cand, target_sum)
        max_val = _max_convolution(cand)
        sc = _score_from_max(max_val, cand.sum(), n)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 4) deterministic local refinement on the best candidate
    # ------------------------------------------------------------------
    a = best.copy()
    # reduction factor linearly decays from 0.15 to 0.02
    max_iters = 250
    for it in range(max_iters):
        conv = np.fft.irfft(
            np.fft.rfft(a, _fft_len(n)) * np.fft.rfft(a, _fft_len(n)), _fft_len(n)
        )[: 2 * n - 1]

        k_max = int(conv.argmax())
        # index range of contributing pairs (i + j = k_max)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j_range = k_max - i_range
        products = a[i_range] * a[j_range]
        idx = int(products.argmax())
        i = int(i_range[idx])
        j = int(j_range[idx])

        # how much to pull out of the offending pair
        red = 0.15 * (1.0 - it / max_iters) + 0.02
        take_i = a[i] * red
        a[i] -= take_i
        taken = take_i
        if j != i:
            take_j = a[j] * red
            a[j] -= take_j
            taken += take_j

        # redistribute the removed mass uniformly (keeps the total sum fixed)
        a += taken / n

        # enforce the explicit [0,1000] box and renormalise to the target sum
        np.clip(a, 0.0, 1000.0, out=a)
        a = _normalize(a, target_sum)

        # evaluate new candidate
        max_val = _max_convolution(a)
        sc = _score_from_max(max_val, a.sum(), n)
        if sc < best_score:
            best = a.copy()
            best_score = sc

        # early exit if we reach a respectable bound
        if best_score <= 1.6:
            break

    # ------------------------------------------------------------------
    # 5) final sanity check – fall back to a uniform vector if something went wrong
    # ------------------------------------------------------------------
    if best_score == float("inf") or best.size == 0:
        best = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best]
