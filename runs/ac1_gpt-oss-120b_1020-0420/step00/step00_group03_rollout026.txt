def generate_sidon_set(n, max_trials=10):
    best_set=[]
    for t in range(max_trials):
        indices = list(range(n))
        rng.shuffle(indices)
        cur_set=[]
        sums=set()
        for i in indices:
            conflict=False
            # compute sums with current set
            for j in cur_set:
                if (i+j) in sums:
                    conflict=True
                    break
            if conflict:
                continue
            if (2*i) in sums:
                continue
            # add
            cur_set.append(i)
            for j in cur_set[:-1]:
                sums.add(i+j)
            sums.add(2*i)
        if len(cur_set)>len(best_set):
            best_set=cur_set
    return best_set

def generate_sidon_set(n, target_size=None, rng=None):
    best_set = []
    attempts = 30
    for _ in range(attempts):
        # random order
        order = np.arange(n)
        rng.shuffle(order)
        cur_set = []
        sumset = set()
        for i in order:
            conflict = False
            # check i+i
            if (2*i) in sumset:
                continue
            for j in cur_set:
                if (i + j) in sumset:
                    conflict = True
                    break
            if conflict:
                continue
            # add
            cur_set.append(i)
            for j in cur_set[:-1]:
                sumset.add(i + j)
            sumset.add(2*i)
            # stop if reached target
            if target_size and len(cur_set) >= target_size:
                break
        if len(cur_set) > len(best_set):
            best_set = cur_set
    return best_set

import numpy as np

def _generate_sidon_set(n, target_size, rng):
    """Greedy construction of a Sidon (B2) set.
    Returns a NumPy array of distinct indices whose pairwise sums are unique."""
    best = np.empty(0, dtype=int)
    attempts = 30
    for _ in range(attempts):
        order = np.arange(n, dtype=int)
        rng.shuffle(order)
        cur = []
        sums = set()
        for i in order:
            # self‑sum collision?
            if (2 * i) in sums:
                continue
            # check sums with already chosen indices
            conflict = False
            for j in cur:
                if (i + j) in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # accept i
            cur.append(i)
            # register new sums
            for j in cur[:-1]:
                sums.add(i + j)
            sums.add(2 * i)
            if len(cur) >= target_size:
                break
        if len(cur) > len(best):
            best = np.array(cur, dtype=int)
            if len(best) >= target_size:
                break
    return best

def propose_candidate(seed=42, num_steps=1000, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 upper bound.

    The method builds a Sidon set of “large’’ entries (value 1000) and fills
    the remaining positions with a uniform small value.  It searches over a
    modest grid of filler values and several random Sidon constructions,
    keeping the best candidate according to `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Base sequence (fallback if height_sequence_1 is missing or invalid)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    n = int(base.size)

    # ------------------------------------------------------------------
    # 2. Initial best = parent (clipped) – always a valid fallback
    parent = np.clip(base, 0.0, 1000.0)
    best_candidate = parent.copy()
    best_score = float(evaluate_sequence(best_candidate.tolist()))

    # ------------------------------------------------------------------
    # 3. Parameters for the Sidon construction
    #   Target number of "large" entries ≈ sqrt(n)
    target_large = max(1, int(np.sqrt(n)))
    filler_grid = np.linspace(0.0, 15.0, 31)           # 0.0 … 15.0 in 0.5 steps

    # Number of independent Sidon attempts – keep within the budget
    sidon_attempts = max(1, num_steps // 30)

    for _ in range(sidon_attempts):
        large_idx = _generate_sidon_set(n, target_large, rng)
        if large_idx.size == 0:
            continue

        # Pre‑compute mask for the small positions (used repeatedly)
        mask_small = np.ones(n, dtype=bool)
        mask_small[large_idx] = False

        for filler in filler_grid:
            cand = np.empty(n, dtype=np.float64)
            cand[mask_small] = filler
            cand[large_idx] = 1000.0   # maximal allowed height

            if np.isfinite(score) and score < best_score:
