        pos_mask = (partner * S - M) > 0

def pair_transfer_step(f, rng, step_frac=0.2):
    n = f.size
    conv = np.convolve(f, f, mode='full')
    k = int(conv.argmax())
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    contribution = np.zeros(n, dtype=np.float64)
    contribution[mask] = f[idx[mask]] * f[j[mask]]
    # source high contribution
    src = int(np.argmax(contribution))
    src_j = int(k - src)
    if not (0 <= src_j < n):
        # fallback random
        return None
    # target low contribution
    # pick low indices where contribution is low.
    low_mask = contribution < contribution[src] * 0.1  # or percentile
    low_candidates = np.where(low_mask)[0]
    if low_candidates.size == 0:
        low_candidates = np.arange(n)
    tgt = int(rng.choice(low_candidates))
    tgt_j = int(k - tgt)

    # ensure both src and src_j have sufficient mass
    amount = step_frac * min(f[src], f[src_j])
    if amount <= 0:

    candidate = f.copy()
    candidate[src] -= amount
    candidate[src_j] -= amount
    candidate[tgt] += amount
    if 0 <= tgt_j < n:
        candidate[tgt_j] += amount
    # clip
    candidate = np.clip(candidate, 0.0, 1000.0)
    return candidate

def propose_candidate(seed=42, num_iters=1500, **kwargs):
    rng = np.random.default_rng(seed)
    # Obtain parent or fallback uniform
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent_arr = np.ones(256, dtype=np.float64)
    if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
        parent_arr = np.ones(256, dtype=np.float64)
    parent_arr = np.clip(parent_arr, 0.0, 1000.0)
    n = parent_arr.size

    # initialize best
    best = parent_arr.copy()
    best_score = _score(best)
    # If best invalid, fallback uniform
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best_score = _score(best)

    # Try a few deterministic patterns
    # pattern 1: exponential decay
    exp_seq = np.exp(-np.arange(n) / (n / 5.0))
    exp_seq = np.clip(exp_seq, 0.0, 1000.0)
    s = _score(exp_seq)
    if s < best_score:
        best = exp_seq.copy()
        best_score = s

    # pattern 2: triangular shape
    tri_seq = np.concatenate([np.linspace(1, 0, (n+1)//2), np.linspace(0, 1, n//2)])
    tri_seq = np.clip(tri_seq, 0.0, 1000.0)
    s = _score(tri_seq)
    if s < best_score:
        best = tri_seq.copy()
        best_score = s

    # pattern 3: random shuffled version of parent
    shuffled = rng.permutation(parent_arr)
    s = _score(shuffled)
    if s < best_score:
        best = shuffled.copy()
        best_score = s

    f = best.copy()

    # Parameters
    max_iters = int(num_iters)
    step_frac = 0.3
    decay = 0.995
    swap_prob = 0.4
    block_prob = 0.2  # remaining prob goes to pair transfer

    for it in range(max_iters):
        # decide move type
        r = rng.random()
        if r < swap_prob:
            # swap two positions
            i, j = rng.integers(0, n, size=2)
            while j == i:
                j = rng.integers(0, n)
            cand = f.copy()
            cand[i], cand[j] = f[j], f[i]  # swap
        elif r < swap_prob + block_prob:
            # random block perturbation
            cand = make_block_perturbation(f, rng)
        else:
            # pair transfer step
            cand = pair_transfer_step(f, rng, step_frac)
            if cand is None:
                # fallback random perturb
                cand = make_block_perturbation(f, rng)

        # clip in case
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc
            f = cand.copy()
        else:
            # optionally, keep f unchanged to stay at best
            pass

        # annealing: reduce step size
        step_frac *= decay

    # Ensure returned list nonempty and finite
    result = [float(x) for x in best]
    if not result:
        result = [1.0] * n
    return result

# ----------------------------------------------------------------------
# Pair‑transfer move – moves mass from a high‑contributing pair to a
# low‑contributing pair for the current maximal convolution index.
# ----------------------------------------------------------------------
def pair_transfer_step(f, rng, frac):
    """
    Perform one mass‑transfer step.
    Returns a new candidate array or None if a suitable transfer could not be built.
    """
    n = f.size
    conv = np.convolve(f, f, mode='full')
    k = int(conv.argmax())                         # index of the current peak
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)                      # indices that actually participate
    if not mask.any():
        return None

    # contribution of each position to the peak convolution value
    contrib = np.zeros(n, dtype=np.float64)
    contrib[mask] = f[idx[mask]] * f[j[mask]]

    # source = pair with largest contribution
    src = int(np.argmax(contrib))
    src_j = int(k - src)
    if not (0 <= src_j < n):
        return None

    # low‑contribution candidates that also have a valid partner
    low_mask = mask & (contrib <= np.percentile(contrib[mask], 10))
    low_candidates = np.where(low_mask)[0]
    if low_candidates.size == 0:                   # fallback to any valid index
        low_candidates = np.where(mask)[0]
    tgt = int(rng.choice(low_candidates))
    tgt_j = int(k - tgt)
    if not (0 <= tgt_j < n):
        return None

    # amount to move (a fraction of the smaller entry of the source pair)
    amount = frac * min(f[src], f[src_j])
    if amount <= 0.0:
        return None

    cand = f.copy()
    cand[src] -= amount
    cand[src_j] -= amount
    cand[tgt] += amount
    cand[tgt_j] += amount
    return np.clip(cand, 0.0, 1000.0)

# current maximal convolution value.
# ----------------------------------------------------------------------
def mix_constant_step(f, rng, t):
    """
    Return (1‑t)·f + t·g where g is a constant vector.
    The constant value is chosen so that f * g never exceeds the current peak.
    """
    n = f.size
    conv = np.convolve(f, f, mode='full')
    M = float(conv.max())
    S = float(f.sum())
    if S <= 0.0:
        return None
    alpha = M / S                     # constant value that makes f·g exactly M
    # respect the allowed range
    if alpha > 1000.0:
        alpha = 1000.0
    g = np.full(n, alpha, dtype=np.float64)
    cand = (1.0 - t) * f + t * g
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_iters=1500, **kwargs):
    """
    Return a list of non‑negative step‑function heights.
    The routine combines several cheap O(n) moves (swap, block perturb,
    pair‑transfer, constant‑mix) while always keeping the best‑found
    candidate.  Memory usage stays linear in the sequence length.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Acquire the parent construction if it exists; otherwise fall back.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:               # height_sequence_1 not defined
        parent_arr = np.ones(256, dtype=np.float64)

    if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
        parent_arr = np.ones(256, dtype=np.float64)

    parent_arr = np.clip(parent_arr, 0.0, 1000.0)
    n = parent_arr.size

    # ------------------------------------------------------------------
    # Initialise best candidate with a few deterministic tries.
    # ------------------------------------------------------------------
    best = parent_arr.copy()
    best_score = _score(best)

    # exponential‐decay shape (scale chosen loosely)
    exp_seq = np.exp(-np.arange(n) / (n / 5.0))
    exp_seq = np.clip(exp_seq, 0.0, 1000.0)
    s = _score(exp_seq)
    if s < best_score:
        best, best_score = exp_seq.copy(), s

    # simple triangular ramp
    tri = np.concatenate([np.linspace(1.0, 0.0, (n + 1) // 2),
                          np.linspace(0.0, 1.0, n // 2)])
    tri = np.clip(tri, 0.0, 1000.0)
    s = _score(tri)
    if s < best_score:
        best, best_score = tri.copy(), s

    # random shuffle of the parent (sometimes breaks hidden structure)
    shuf = rng.permutation(parent_arr)
    s = _score(shuf)
    if s < best_score:
        best, best_score = shuf.copy(), s

    # ------------------------------------------------------------------
    # Core stochastic local‑search loop.
    # ------------------------------------------------------------------
    f = best.copy()                     # current base for proposing moves
    step_frac = 0.30                    # fraction used in pair‑transfer
    t_mix = 0.10                        # mixing strength for constant‑mix
    decay_frac = 0.995                  # geometric decay of step_frac
    decay_t = 0.995                     # decay of mixing strength
    swap_prob = 0.40
    block_prob = 0.20                   # remaining 0.40 → pair‑transfer

    max_iters = int(num_iters)
    for _ in range(max_iters):
        r = rng.random()
        cand = None

        if r < swap_prob:                      # --- random swap ---
            i, j = rng.integers(0, n, size=2)
            while j == i:
                j = rng.integers(0, n)
            cand = f.copy()
            cand[i], cand[j] = f[j], f[i]

        elif r < swap_prob + block_prob:       # --- block perturbation ---
            cand = make_block_perturbation(f, rng)

        elif r < swap_prob + block_prob + 0.30:  # --- pair transfer ---
            cand = pair_transfer_step(f, rng, step_frac)

        else:                                   # --- constant mix ---
            cand = mix_constant_step(f, rng, t_mix)

        if cand is None:
            continue

        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)

        # keep the globally best candidate
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc
            f = best.copy()                     # reset base to the new best

        # optionally, keep progressing from the latest candidate even if not better
        # (helps the stochastic walk); here we stay on the last successful move
        # to avoid drifting away from good region.
        # Comment out the following line if a purely elitist walk is desired.
        # else:
        #     f = cand

        # anneal step sizes
        step_frac *= decay_frac
        t_mix *= decay_t

    # ------------------------------------------------------------------
    # Final safety fallback – never return an empty or non‑finite list.
    # ------------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
