import numpy as np

# ------------------------------------------------------------
# Helper: a simple block perturbation (fallback/random move)
# ------------------------------------------------------------
def _block_perturbation(seq, rng):
    """Return a copy of ``seq`` with one random block scaled multiplicatively."""
    cand = seq.copy()
    n = cand.size
    # block width between 1% and 8% of length, at least 1 element
    min_w = max(1, n // 100)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.10)))  # log‑normal tweak
    cand[start:stop] *= factor
    # optional slight symmetrisation
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

        lo = max(0, max_idx - (n - 1))

        # -------------------------------------------------------------

            i_receiver = idx_range[int(np.argmin(products))]
