import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """
    Return a simple O(n) memory non‑negative perturbation of ``base``.
    Randomly selects a contiguous block and rescales it multiplicatively.
    """
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block size between n/200 and n/12 (inclusive, at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative noise (log‑normal)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional slight symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # enforce bounds required by the evaluator
    return np.clip(cand, 0.0, 1000.0)


def gen_geometric(n, t):
    """Geometric (exponential) decreasing shape: a_i = 1000 * exp(-t*i)."""
    return 1000.0 * np.exp(-t * np.arange(n, dtype=np.float64))


def gen_linear(n):
    """Linear ramp from 1000 down to 0."""
    return np.linspace(1000.0, 0.0, n, dtype=np.float64)


def gen_powerlaw(n, p):
    """Power‑law decay: a_i = 1000 / (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = 1000.0 * (idx ** (-p))
    return np.clip(seq, 0.0, 1000.0)

# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=300, **kwargs):
    """
    Produce a non‑negative height sequence whose AC1 upper bound is
    (hopefully) better than the naïve uniform construction.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length n and a reasonable fallback parent vector.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        # fallback: a modest uniform vector (size 256 if the real length is unknown)
        n = 256
        parent_arr = np.full(n, 1000.0, dtype=np.float64)
    else:
        n = parent_arr.size

    # ------------------------------------------------------------------
    # Initialise the best candidate (fallback is the provided parent).
    # ------------------------------------------------------------------
    best = parent_arr.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # A modest portfolio of deterministic shapes to seed the search.
    # ------------------------------------------------------------------
    seed_candidates = []

    # 1. Uniform maximal vector (all 1000) – gives the baseline 2.0 score.
    seed_candidates.append(np.full(n, 1000.0, dtype=np.float64))

    # 2. Geometric families (t roughly between 1/n and a few times that).
    for t in np.linspace(0.00015, 0.0010, 20):
        seq = gen_geometric(n, t)
        seed_candidates.append(seq)

    # 3. Linear decreasing ramp.
    seed_candidates.append(gen_linear(n))

    # 4. Power‑law decays for a variety of exponents.
    for p in np.linspace(0.0, 1.5, 12):
        seed_candidates.append(gen_powerlaw(n, p))

    # 5. A randomly shuffled version of the parent (to break possible patterns).
    shuffled = np.array(parent_arr, copy=True)
    rng.shuffle(shuffled)
    seed_candidates.append(shuffled)

    # ------------------------------------------------------------------
    # Evaluate all deterministic seeds and keep the best one.
    # ------------------------------------------------------------------
    for cand in seed_candidates:
        seq = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:

    for _ in range(num_steps):
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score

    # Return a plain Python list of floats (the harness expects this).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
