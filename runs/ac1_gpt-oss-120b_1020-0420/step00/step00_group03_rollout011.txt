import numpy as np

def propose_candidate(seed=42, num_iter=200, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) aiming at a low AC1 bound.
    The method starts from the provided parent sequence (height_sequence_1),
    repeatedly averages each entry with its two neighbours (periodic wrap‑around)
    which drives the vector towards a uniform distribution – the known optimal
    shape for the bound 2·n·max(b)/sum(a)².
    """
    # ----------------------------------------------------------------------
    # 1. Acquire parent sequence; fall back to a simple constant if needed.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        # enforce evaluator's hard limits
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # No valid parent – use a modest constant length.
        parent = np.full(256, 1.0, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 2. Diffusion / smoothing toward uniformity.
    #    The operation (a + roll(a,1) + roll(a,-1)) / 3 preserves the total sum.
    # ----------------------------------------------------------------------
    a = parent.copy()
    for _ in range(num_iter):
        a = (a + np.roll(a, 1) + np.roll(a, -1)) / 3.0

    # ----------------------------------------------------------------------
    # 3. Clip to admissible range.
    # ----------------------------------------------------------------------
    a = np.clip(a, 0.0, 1000.0)

    # ----------------------------------------------------------------------
    # 4. Validate – if for any reason the evaluator fails, return a safe fallback.
    # ----------------------------------------------------------------------
    try:
        score = evaluate_sequence(a.tolist())
        if not np.isfinite(score):
            raise ValueError
    except Exception:
        a = np.full(a.shape, 1.0, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 5. Return as plain Python list of floats.
    # ----------------------------------------------------------------------
    return [float(x) for x in a]
