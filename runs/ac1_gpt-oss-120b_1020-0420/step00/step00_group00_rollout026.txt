import numpy as np

# Helper: obtain an initial parent sequence (fallback if needed)

def _initial_parent():
    """Return a valid non‑negative sequence based on height_sequence_1
    or a simple uniform fallback."""
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        if seq.ndim == 1 and seq.size > 0 and np.isfinite(seq).all():
            return np.clip(seq, 0.0, 1000.0)
    except Exception:
        pass
    # Fallback: modest uniform vector
    return np.ones(256, dtype=np.float64)

# Helper: generate a random candidate with a few large spikes and a low baseline
# ------------------------------------------------------------
def _make_candidate(rng, n):
    """
    Build a candidate:
    - m large spikes (value B) at random distinct positions,
    - all other entries set to a small baseline e.
    Parameters are sampled randomly to explore the space.
    """
    # number of spikes – keep it moderate to limit pair collisions
    m = int(rng.integers(40, min(n, 200)))          # 40 … 199 spikes
    # large spike height (bounded by 1000)
    B = float(rng.uniform(300.0, 1000.0))           # 300 … 1000
    # low baseline (helps the denominator while contributing little to max)
    e = float(rng.uniform(0.5, 12.0))               # 0.5 … 12
    # distinct random positions for spikes
    positions = rng.choice(n, size=m, replace=False)
    # construct the array
    cand = np.full(n, e, dtype=np.float64)
    cand[positions] = B
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _make_block_perturbation(base, rng):
    """O(n) perturbation that scales a random block and optionally mirrors."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))

    if rng.random() < 0.35:

    return cand

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """

    rng = np.random.default_rng(seed)
