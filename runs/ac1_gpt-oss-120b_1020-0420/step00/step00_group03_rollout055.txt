def gradient_descent(a, n_steps=2000, step=0.01, rng=None):
    best = a.copy()
    best_score = float('inf')
    for t in range(n_steps):
        b = np.convolve(a, a)
        idx = int(np.argmax(b))
        max_b = b[idx]
        # compute loss = max_b / (sum(a)**2) (ignoring constant factor)
        loss = max_b / (np.sum(a)**2)
        if loss < best_score:
            best_score = loss
            best = a.copy()
        # gradient
        g = np.zeros_like(a)
        # compute slice boundaries
        start_i = max(0, idx - (len(a) - 1))
        end_i = min(len(a) - 1, idx)
        # vectorized assignment
        i_vals = np.arange(start_i, end_i+1)
        j_vals = idx - i_vals
        g[i_vals] = 2.0 * a[j_vals]
        # gradient descent step
        a = a - step * g
        # ensure nonnegative
        a = np.clip(a, 0.0, 1000.0)
        # optional renormalize sum to 1 to avoid scaling drift
        # but scaling doesn't affect loss, but helps avoid underflow
        s = np.sum(a)
        if s > 0:
            a = a / s   # normalize to sum=1 for numeric stability
        else:
            a = np.ones_like(a) / len(a)
        # optionally random perturbation occasionally to escape local minima
        if rng.random() < 0.01:
            # random multiplicative perturbations on some entries
            scale = np.exp(rng.normal(0, 0.1, size=a.shape))
            a = a * scale
            a = np.clip(a, 0.0, 1000.0)
            a = a / np.sum(a)
    # after loop, scale best to some sum; but scale doesn't matter for evaluator, we can leave as is.
    # we may multiply by a constant to bring sum to something > 0.01 (already) and within max < 1000.
    return best

def build_sidon_set(n, rng, max_trials=40):
    best_set=[]
    best_len=0
    for t in range(max_trials):
        order = rng.permutation(n)
        cur=[]
        sums=set()
        for idx in order:
            idx=int(idx)
            # check uniqueness of sums
            dup=False
            # self sum
            s=idx+idx
            if s in sums:
                dup=True
            else:
                for p in cur:
                    if idx + p in sums:
                        dup=True
                        break
            if not dup:
                # add sums
                for p in cur:
                    sums.add(idx+p)
                sums.add(s)
                cur.append(idx)
                # early stop if reached theoretical max (e.g., 100)
                if len(cur)>=100:
                    break
        if len(cur)>best_len:
            best_len=len(cur)
            best_set=cur
        if best_len>=100:
            break
    return best_set

import numpy as np

def build_sidon_set(n, rng, max_trials=200):
    best_set = []
    best_len = 0
    for _ in range(max_trials):
        order = rng.permutation(n)
        cur = []
        sums = set()
        for idx in order:
            idx = int(idx)
            # self sum
            s = idx + idx
            if s in sums:
                continue
            conflict = False
            for p in cur:
                if idx + p in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # add new
            for p in cur:
                sums.add(idx + p)
            sums.add(s)
            cur.append(idx)
            if len(cur) > best_len:
                best_len = len(cur)
                best_set = cur.copy()
            # early break if we reach some large size we are happy with
            if best_len >= 100:
                break
        # optional early stop
        if best_len >= 100:
            break
    return best_set

import numpy as np

# ----------------------------------------------------------------------
# Helper: construct a (near‑maximal) Sidon set (all pairwise sums distinct)
# ----------------------------------------------------------------------
def build_sidon_set(n, rng, max_trials=200):
    """
    Return a list of indices 0<=i<n forming a Sidon set.
    The routine tries several random orders and keeps the largest set found.
    """
    best_set = []
    best_len = 0

    for _ in range(max_trials):
        order = rng.permutation(n)                # random order of candidates
        cur = []                                   # current Sidon set
        sums = set()                               # occupied sums i+j

        for idx in order:
            idx = int(idx)
            # check whether any new sum would clash
            s_self = idx + idx
            if s_self in sums:
                continue

            conflict = False
            for p in cur:
                if idx + p in sums:
                    conflict = True
                    break
            if conflict:
                continue

            # accept idx – add all new sums
            for p in cur:
                sums.add(idx + p)
            sums.add(s_self)
            cur.append(idx)

            # keep the best set seen so far
            if len(cur) > best_len:
                best_len = len(cur)
                best_set = cur.copy()

        # stop early if we already reached a comfortable size
        if best_len >= 100:        # theoretical upper bound for this n is ~100
            break

    return best_set

def propose_candidate(seed=42, **kwargs):
    """
    Returns a list of non‑negative heights that (heuristically) minimise
    the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – fall back to a modest size if the parent
    # array is not present.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Build a Sidon set – positions for the relatively large spikes.
    # ------------------------------------------------------------------
    sidon = build_sidon_set(n, rng, max_trials=400)

    # If we could not find any Sidon positions (should not happen),
    # simply return a uniform vector – its score is exactly 2.0.
    if not sidon:
        return [1.0] * n

    # ------------------------------------------------------------------
    # Parameter grid: (spike height H, background level eps).
    # The idea is:
    #   • positions in `sidon` get the larger value H,
    #   • every other index gets the small background value eps.
    # This keeps the maximal convolution term essentially bounded by
    #   max(2*H**2, 2*len(sidon)*H*eps + n*eps**2)
    # while the total sum is increased by the background, lowering the
    # score 2*n*max_b/(sum**2).
    # ------------------------------------------------------------------
    param_grid = [
        (10.0, 0.10),
        ( 8.0, 0.12),
        (12.0, 0.08),
        ( 6.0, 0.15),
        ( 5.0, 0.18),
        ( 9.0, 0.09),
        ( 7.0, 0.14),
    ]

    best_score = float("inf")
    best_candidate = None

    for H, eps in param_grid:
        a = np.full(n, eps, dtype=np.float64)   # background
        a[sidon] = H                             # spikes on Sidon positions
        a = np.clip(a, 0.0, 1000.0)               # obey evaluator limits

        # Exact evaluation of the objective (cheap – just a few calls)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            continue

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_candidate = a.copy()

        # Early exit if we are already comfortably below the target
        if best_score < 1.5:
            break

    # ------------------------------------------------------------------
    # Safety net: if no candidate passed (very unlikely), fall back to
    # a simple uniform vector.
    # ------------------------------------------------------------------
    if best_candidate is None:
        best_candidate = np.ones(n, dtype=np.float64)

    # Return a plain Python list of floats as required by the harness
    return [float(x) for x in best_candidate]
