import numpy as np

def _constant_candidate(length, value=1.0):
    """Return a non‑negative constant sequence of given length."""
    return np.full(length, float(value), dtype=np.float64)


def _block_perturbation(base, rng):
    """Create an O(n) memory perturbation of ``base``."""
    cand = base.copy()
    n = cand.size

    # Random block width and position
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width

    # Random multiplicative factor
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor

    # Optional mirroring for extra diversity
    if rng.random() < 0.35:
        cand[:] = 0.5 * (cand + cand[::-1])

    # Enforce the evaluator's clipping range
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function sequence that (approximately) minimizes
    the AC1 upper bound evaluated by ``evaluate_sequence``.

    The routine starts from a constant sequence (which attains the theoretical
    lower bound 2.0 for the objective) and performs a limited number of
    inexpensive block‑perturbation steps, keeping any improvement.
    """
    rng = np.random.default_rng(seed)

    # Determine the length of the target sequence.
    try:
        n = int(height_sequence_1.shape[0])  # provided by the harness
    except Exception:
        n = 256  # safe fallback

    # Initialise with a constant sequence – this already yields the

    best_score = float(evaluate_sequence(best.tolist()))

        if np.isfinite(score) and score < best_score:
