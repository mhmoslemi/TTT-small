def build_sidon_set(n):
    used = np.zeros(2*n, dtype=bool)
    sidon = []
    for i in range(n):
        # Check if adding i possible
        ok = True
        # self-sum
        if used[i+i]:
            continue
        for p in sidon:
            if used[i + p]:
                ok = False
                break
        if not ok:
            continue
        # If ok, add
        for p in sidon:
            used[i + p] = True
            used[p + i] = True  # same index; but used[i+p] is symmetric
        used[i + i] = True
        sidon.append(i)

        if len(sidon) >= int(np.sqrt(2*n)) + 5:

import numpy as np

# ----------------------------------------------------------------------
# Build a Sidon (B₂) set: all pairwise sums i+j are distinct.
# ----------------------------------------------------------------------
def _build_sidon_set(n: int) -> np.ndarray:
    used = np.zeros(2 * n, dtype=bool)          # used[s] = True  ⇔  sum s already taken
    sidon = []                                   # list of chosen indices
    for i in range(n):
        # the sum i+i must be free
        if used[i + i]:
            continue
        # all mixed sums i+p must be free
        clash = False
        for p in sidon:
            if used[i + p]:
                clash = True
                break
        if clash:
            continue
        # i can be added – mark its sums
        for p in sidon:
            used[i + p] = True                   # i+p == p+i
        used[i + i] = True
        sidon.append(i)
    return np.array(sidon, dtype=int)
