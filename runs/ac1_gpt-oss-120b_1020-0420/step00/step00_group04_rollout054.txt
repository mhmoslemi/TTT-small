import numpy as np

def _greedy_high_indices(n: int, max_sum_count: int = 2):
    """
    Greedy construction of a set of indices with the property that
    any sum i+j (i<=j) occurs at most `max_sum_count` times.
    Returns the list of selected indices.
    """
    high = []
    sum_counts = {}                     # sum -> how many pairs produce this sum
    for i in range(n):
        # Check self‑pair i+i
        s_self = i + i
        if sum_counts.get(s_self, 0) >= max_sum_count:
            continue
        # Check all pairs with existing high indices
        ok = True
        for j in high:
            s = i + j
            if sum_counts.get(s, 0) >= max_sum_count:
                ok = False
                break
        if not ok:
            continue
        # Accept i
        high.append(i)
        # Update counts: self‑pair
        sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
        # Update counts with previous high indices
        for j in high[:-1]:
            s = i + j
            sum_counts[s] = sum_counts.get(s, 0) + 1
    return high
