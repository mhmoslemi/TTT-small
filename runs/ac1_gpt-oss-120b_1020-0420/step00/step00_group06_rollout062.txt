import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: bring a vector to the required scale and bounds.
# ------------------------------------------------------------
def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and scale so that sum == target_sum.
    A second clipping+scaling pass guards against occasional overflow."""
    vec = np.clip(vec, 0.0, 1000.0)
    s = vec.sum()
    if s == 0.0:
        # fallback to a tiny uniform vector
        vec = np.full_like(vec, target_sum / vec.size)
        return vec
    vec = vec * (target_sum / s)

    # second pass – very rare but safe
    over = vec > 1000.0
    if over.any():
        vec[over] = 1000.0
        s = vec.sum()
        if s > 0.0:
            vec = vec * (target_sum / s)
    return vec

def propose_candidate(seed: int = 42, **kwargs):
    """
    Search for a non‑negative step function that minimizes the
    AC1 upper bound (the evaluator returns the bound directly).

    The routine starts from the supplied ``height_sequence_1`` if it
    exists, otherwise from a uniform sequence.  It then performs a
    small number of LP‑guided updates combined with occasional
    random block perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # 1.  Initialise the base sequence.
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        valid_parent = (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        )
    except Exception:  # height_sequence_1 may be undefined
        valid_parent = False

    if valid_parent:
        seq = np.clip(parent, 0.0, 1000.0)
    else:
        # a safe fallback: 256‑element uniform vector
        seq = np.full(256, 1.0, dtype=np.float64)

    n = seq.size
    target_sum = np.sqrt(2.0 * n)

    # normalise to the required total sum
    seq = _normalize(seq, target_sum)

    best = seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------
    # 2.  Main optimisation loop.
    # --------------------------------------------------------
    max_iters = 200          # total outer iterations
    blend = 0.3              # mixing factor (1‑t)*f + t*g
    top_k = 12               # how many largest convolution indices to keep as constraints

    for it in range(max_iters):
        # current convolution and its peak
        conv = np.convolve(seq, seq)
        cur_max = float(conv.max())
        # indices of the largest peaks (used as LP constraints)
        peak_idx = np.argsort(-conv)[:top_k]

        # -----------------------------------------------------------------
        # Build the linear constraints A @ g <= cur_max   (one row per peak)
        # -----------------------------------------------------------------
        A = np.zeros((top_k, n), dtype=np.float64)
        for r, k in enumerate(peak_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_vals = k - i_vals                    # partner indices
            A[r, j_vals] = seq[i_vals]             # coefficient = f_i

        # -----------------------------------------------------------
        # Solve the LP: maximise sum(g)  subject to  A g <= cur_max
        # -----------------------------------------------------------
        c = -np.ones(n, dtype=np.float64)          # linprog minimises; -sum(g) => maximise sum(g)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        lp_res = linprog(
            c,
            A_ub=A,
            b_ub=np.full(top_k, cur_max),
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if lp_res.success:
            g = lp_res.x
            # re‑scale g to the target sum (the LP only guarantees
            # the bound on the selected convolutions)
            g = _normalize(g, target_sum)

            candidate = (1.0 - blend) * seq + blend * g
