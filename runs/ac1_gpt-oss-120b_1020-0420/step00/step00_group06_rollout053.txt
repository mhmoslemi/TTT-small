import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative coefficients that (empirically) lowers the
    AC1 bound.  The routine evaluates a handful of simple parametric families
    (exponential, linear, power‑law) and a few random Dirichlet draws, keeping
    the best candidate according to the provided ``evaluate_sequence`` function.
    """
    # ----------------------------------------------------------------------
    # Determine the problem size (the length of the given parent sequence)
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            n = 256
        else:
            n = parent.size
    except NameError:
        n = 256                      # safety fallback when the variable is absent

    # ----------------------------------------------------------------------
    # Storage for the best candidate found so far
    # ----------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Helper to evaluate a candidate and possibly promote it to “best”
    # ----------------------------------------------------------------------
    def maybe_update(vec):
        nonlocal best_seq, best_score
        lst = [float(v) for v in vec.tolist()]          # list of python floats
        score = float(evaluate_sequence(lst))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = lst

    # ----------------------------------------------------------------------
    # 1. Uniform baseline (sum normalised to 1)
    # ----------------------------------------------------------------------
    uniform = np.ones(n, dtype=np.float64)
    uniform /= uniform.sum()
    maybe_update(uniform)

    # ----------------------------------------------------------------------
    # 2. Exponential decays a_i = r^i  (r in (0,1))
    # ----------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    r_vals = [
        0.7, 0.8, 0.85, 0.9, 0.92, 0.94, 0.95, 0.96,
        0.97, 0.975, 0.98, 0.985, 0.99, 0.992, 0.994,
        0.996, 0.997, 0.998, 0.9985, 0.9988, 0.999,
        0.9992, 0.9995, 0.9997, 0.9999
    ]
    for r in r_vals:
        if not (0.0 < r < 1.0):
            continue
        base = np.power(r, idx)
        s = base.sum()
        if s == 0.0:
            continue
        cand = base / s
        maybe_update(cand)

    # ----------------------------------------------------------------------
    # 3. Linear decreasing shape a_i ∝ (n-i)
    # ----------------------------------------------------------------------
    linear = np.arange(n, 0, -1, dtype=np.float64)
    maybe_update(linear / linear.sum())

    # ----------------------------------------------------------------------
    # 4. Power‑law shapes a_i ∝ 1/(i+1)^p
    # ----------------------------------------------------------------------
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        base = 1.0 / np.power(idx + 1.0, p)
        s = base.sum()
        if s == 0.0:
            continue
        maybe_update(base / s)

    # ----------------------------------------------------------------------
    # 5. Random Dirichlet (Gamma) draws – a cheap way to explore the simplex
    # ----------------------------------------------------------------------
    for _ in range(15):
        vec = rng.gamma(shape=1.0, scale=1.0, size=n)
        total = vec.sum()
        if total == 0.0:
            continue
        maybe_update(vec / total)

    # ----------------------------------------------------------------------
    # Fallback – should never be reached because the uniform candidate always
    # yields a finite score, but we keep it for maximum safety.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * n

    return best_seq
