import numpy as np

# evaluate_sequence is provided by the harness


def _initial_candidate(n, rng):
    """
    Generate a smooth decreasing candidate based on an exponential profile.
    The decay parameter is chosen so that the theoretical bound is close to
    the best known value (~1.8) and provides a good starting point for the
    stochastic search.
    """
    # decay rate: alpha = 2 / n  (corresponds to c = 2 in the analysis)
    alpha = 2.0 / max(1, n)
    cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
    return cand


def _perturb(candidate, rng):
    """
    Produce a new candidate from ``candidate``.
    Two simple mutation operators are used:
      * a random permutation (breaks large‑value alignment)
      * multiplicative noise on a random contiguous block
    The result is clipped to the allowed range [0, 1000].
    """
    n = candidate.size
    if rng.random() < 0.5:
        # Shuffle the whole vector – cheap but often reduces peak autocorrelation
        new = rng.permutation(candidate)
    else:
        # Random block perturbation
        new = candidate.copy()
        # block width between 1 and n//10 (inclusive)
        width = int(rng.integers(1, max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        # multiplicative log‑normal factor
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        new[start:stop] *= factor
    # enforce the hard limits required by the evaluator
    return np.clip(new, 0.0, 1000.0)
