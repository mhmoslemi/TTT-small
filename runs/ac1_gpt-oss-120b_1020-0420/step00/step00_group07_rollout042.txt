def compute_bound(seq):
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    n = seq.size
    # compute convolution using FFT
    m = 1 << ((2*n -1).bit_length())
    fa = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    # conv length >= 2n-1, consider first 2n-1 entries
    max_b = np.max(conv[:2*n-1])
    return 2.0 * n * max_b / (s * s)

import numpy as np

def _bound_fft(seq: np.ndarray) -> float:
    """Fast computation of the AC1 upper bound for a candidate sequence."""
    s = float(np.sum(seq))
    if s < 0.01:
        return np.inf
    n = seq.size
    # length for convolution via FFT (next power of two ≥ 2n‑1)
    m = 1 << ((2 * n - 1).bit_length())
    # convolution = ifft( fft(seq)^2 )
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)
    max_b = float(np.max(conv[: 2 * n - 1]))
    return 2.0 * n * max_b / (s * s)

    The routine builds a power‑law profile a_i = (i+1)^{-α} (α≈0.5), scans a
    small grid of exponents, and then performs a cheap random‑swap local
    search using the FFT‑based bound evaluator.  The final candidate is
    verified with the official `evaluate_sequence` before returning.
    """
    rng = np.random.default_rng(seed)

    # Length of the construction – use the provided parent if available.
    try:
        n = int(np.asarray(height_sequence_1).size)

    # ------------------------------------------------------------------

    best_seq = None

    for alpha in np.linspace(0.40, 0.60, 9):          # 0.40,0.45,…,0.60
        seq = np.power(np.arange(n) + 1.0, -alpha)

        score = _bound_fft(seq)
