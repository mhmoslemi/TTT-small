def max_convolution(seq):
    n = seq.shape[0]
    m = 1 << ((2*n-1).bit_length())
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)[:2*n-1]
    return np.max(conv.real)

def propose_candidate(seed=42, **kwargs):
    """
    Generates a simple decreasing linear sequence.
    The shape a_i = (n - i) / n (i = 0 .. n-1) yields a provably low
    AC1 upper bound (≈4/3 for large n).  All values are already within the

    """

    seq = (n - np.arange(n, dtype=np.float64)) / n
