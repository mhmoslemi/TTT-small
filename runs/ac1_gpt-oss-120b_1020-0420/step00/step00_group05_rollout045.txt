def uniform_sequence(length, value=1.0):
    """Return a constant vector of the given length."""
    return np.full(length, float(value), dtype=np.float64)


def perturb_block(base, rng):
    """Generate an O(n) random perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)

    start = int(rng.integers(0, n - width + 1))

    np.clip(cand, 0.0, 1000.0, out=cand)

    """

    The routine starts from the uniform sequence (the theoretical optimum,
    achieving a bound of exactly 2) and performs a tiny local search as a

    # Try to use the supplied parent; otherwise fall back to a simple constant.
    try:
