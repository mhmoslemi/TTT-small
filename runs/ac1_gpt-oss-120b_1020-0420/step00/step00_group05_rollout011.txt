import numpy as np

def _initial_uniform(length):
    """Return a positive uniform vector of given length."""
    # value 1.0 is arbitrary (scale‑invariant); clipping ensures validity.
    return np.full(length, 1.0, dtype=np.float64)

def _make_block_perturbation(base, rng):
    """Generate an O(n) perturbation of ``base``."""
    n = base.size
    # Choose a block size between n/200 and n/12 (inclusive).
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width

    candidate = base.copy()
    # Scale the selected block by a log‑normal factor (small variance).
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    candidate[start:stop] *= factor

    # Occasionally symmetrize to explore structured patterns.
    if rng.random() < 0.2:
        candidate = 0.5 * (candidate + candidate[::-1])

    # Enforce the evaluator's hard bounds.
    return np.clip(candidate, 0.0, 1000.0)

def propose_candidate(seed=42, num_steps=1500, **kwargs):
    """
    Return a non‑negative step‑function sequence attempting to minimise
    the AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the supplied construction if it exists and is valid.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim != 1
            or parent.size == 0
            or not np.isfinite(parent).all()
        ):
            parent = None
    else:
        parent = None

    # If the parent is unusable fall back to a modest constant vector.
    if parent is None:
        n = 256
        best = _initial_uniform(n)
    else:
        n = parent.size
        # Start from a uniform vector – this is optimal for the score
        # among all non‑negative sequences (theoretical lower bound = 2).
        best = _initial_uniform(n)

    best = np.clip(best, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # ---------------------------------------------------------------
    # Simple stochastic local search – perturbation with acceptance.
    # ---------------------------------------------------------------
    for _ in range(num_steps):
        cand = _make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
