import numpy as np

# The evaluator `evaluate_sequence` is provided by the harness.
# --------------------------------------------------------------------
def _initial_geometric(n: int) -> np.ndarray:
    """
    Return a decreasing geometric sequence a_i = r**i, with r chosen
    (r = exp(-1.256/n)) so that the initial AC1 upper bound is close to
    the theoretical optimum for pure exponentials.
    """
    # Parameter x = a*n ≈ 1.256 gives the smallest bound for pure exponentials.
    # a = x / n, r = exp(-a)
    a = 1.256 / n
    r = np.exp(-a)
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(i * np.log(r))
    # enforce the evaluator's hard clipping limits
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative coefficients whose AC1 upper bound
    (as evaluated by `evaluate_sequence`) is as small as possible.
    The method starts from a simple exponential shape and then performs
    a greedy hill‑climbing search that moves mass away from the most
    harmful indices for the current maximal convolution entry.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Determine problem size (length of the sequence)
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        # fallback length if the parent construction is absent
        n = 256

    # ----------------------------------------------------------------
    # Initial candidate – a carefully tuned exponential
    seq = _initial_geometric(n)

    # Keep best found sequence and its score
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ----------------------------------------------------------------
    # Greedy local search
    max_iters = 1500          # total number of attempted moves
    move_fraction = 0.10      # fraction of the source value to relocate

    for _ in range(max_iters):
        # Full convolution and index of the current maximum entry
        conv = np.convolve(seq, seq)
        k_max = int(np.argmax(conv))

        # Contributions of each element to that maximal entry
        idx = np.arange(n, dtype=np.int64)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)

        # contributions[i] = seq[i] * seq[partner[i]]  if partner in range else 0
        contributions = np.zeros_like(seq)
        if valid.any():
            vi = idx[valid]
            pj = partner[valid]
            contributions[vi] = seq[vi] * seq[pj]

        # Source index: the one with the largest contribution to the peak
        if not valid.any():
            # Should not happen, but protect against pathological cases
            break
        i_src = int(np.argmax(contributions))

        # Candidate destination indices: those that do *not* influence the peak
        zero_mask = ~valid
        if zero_mask.any():
            dest_choices = np.flatnonzero(zero_mask)
        else:
            # If every index participates, pick the smallest‑contributing one
            # (excluding the source itself)
            mask = np.arange(n) != i_src
            dest_choices = np.flatnonzero(mask)

        i_dst = int(rng.choice(dest_choices))

        # Amount to move: a fraction of the source's current value
        delta = seq[i_src] * move_fraction
        # Do not exceed the upper clipping bound at the destination
        max_add = 1000.0 - seq[i_dst]
        if max_add < delta:
            delta = max_add
        if delta <= 0.0:
            continue  # nothing to move

        # Apply the move
        seq[i_src] -= delta
        seq[i_dst] += delta
        np.clip(seq, 0.0, 1000.0, out=seq)

        # Evaluate the new sequence
        score = float(evaluate_sequence(seq.tolist()))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
        else:
            # Revert the move
            seq[i_src] += delta
            seq[i_dst] -= delta
            np.clip(seq, 0.0, 1000.0, out=seq)

    # ----------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    return best_seq.tolist()
