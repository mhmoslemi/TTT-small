import numpy as np

def _candidate_uniform(length):
    """Return a uniform non‑negative vector of the given length."""
    return np.full(length, 1.0, dtype=np.float64)

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑empty list of non‑negative floats that (approximately) minimizes
    the AC1 upper bound evaluated by `evaluate_sequence`.

    The strategy is to compare a simple uniform vector with the provided
    `height_sequence_1` (if any) and return the better one.  This guarantees a
    finite, valid result and never exceeds O(n) auxiliary memory.
    """
    # Length of the problem – use the supplied parent if possible.
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
        n = int(parent_arr.shape[0])
    except Exception:
        # Fallback length when no parent is available.
        n = 256
        parent_arr = None

    # Candidate A: uniform vector.
    uniform_vec = _candidate_uniform(n)
    try:
        score_uniform = float(evaluate_sequence(uniform_arr.tolist()))
    except Exception:
        # In the unlikely event `evaluate_sequence` fails, give it a huge score.
        score_uniform = float('inf')

    # Candidate B: the parent sequence (if usable).
    if parent_arr is not None:
        try:
            score_parent = float(evaluate_sequence(parent_arr.tolist()))
        except Exception:
            score_parent = float('inf')
    else:
        score_parent = float('inf')

    # Choose the better candidate.
    if score_uniform <= score_parent:
        best_vec = uniform_vec
    else:
        best_vec = parent_arr

    # Ensure the returned list respects the evaluator’s constraints.
    # (The evaluator clips negatives and >1000, but we keep values safe.)
    return [float(x) for x in best_vec]
