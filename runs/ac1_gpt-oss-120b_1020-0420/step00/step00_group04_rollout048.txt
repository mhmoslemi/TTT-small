    n = base_seq.size
    # Uniform baseline
    uniform_val = 1.0
    best = np.full(n, uniform_val, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    # Simple local exploration (optional)
    for _ in range(num_steps):
        # random perturbation: small Gaussian noise then clip
        candidate = best + rng.normal(0.0, 0.1, size=n)
        candidate = np.clip(candidate, 0.0, 1000.0)
        # ensure sum not too small
        if candidate.sum() < 0.01:
            continue
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    return [float(x) for x in best]

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a simple uniform non‑negative sequence.
    If the runtime provides the parent array `height_sequence_1`,
    its length is used; otherwise a default length of 256 is taken.
    The returned list is guaranteed to be valid for the evaluator.
    """
    # Determine target length (fallback to 256 if needed)
    try:
        n = int(len(height_sequence_1))
        if n <= 0:
            raise Exception
    except Exception:
        n = 256

    # Uniform value – any positive constant gives the same score (2.0)
    uniform_value = 1.0

    # Build the sequence (O(n) memory)
    seq = np.full(n, uniform_value, dtype=np.float64)

    # Ensure the values respect the evaluator's clipping bounds
    seq = np.clip(seq, 0.0, 1000.0)

    # Return as a plain Python list of floats
    return [float(x) for x in seq]
