    prod = fv * fv
    conv = np.fft.irfft(prod, fft_len)[: 2 * v.size - 1]
    return conv


def _score(v, n, fft_len):
    """Exact AC1 score for a vector (no clipping, sum must be >= 0.01)."""
    s = v.sum()
    if s < 0.01:
        return np.inf
    conv = _conv_self(v, fft_len)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_iters=500, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) lower the AC1 bound.
    The function uses a simple hill‑climbing search that preserves the total
    mass and evaluates candidates with the fast FFT‑based convolution.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and a sensible fallback initialization
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = None

    n = parent.size if parent is not None else 256
    fft_len = _fft_len_for(n)

    # ------------------------------------------------------------------
    # Starting point: uniform distribution (sum = 1) – this gives the
    # theoretical lower bound of 2.0 for the AC1 metric.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # Choose the better of the supplied parent and the uniform vector
    best = uniform.copy()
    best_score = _score(best, n, fft_len)

    if parent is not None:
        parent_clipped = np.clip(parent, 0.0, 1000.0)
        parent_score = _score(parent_clipped, n, fft_len)
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent_clipped.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # Simple hill‑climbing moves: transfer a random fraction of mass from
    # one index to another while keeping the total sum unchanged.
    # ------------------------------------------------------------------
    def perturb(vec):
        """Return a new vector with a small mass transfer between two positions."""
        cand = vec.copy()
        # pick a source index with non‑zero mass
        src_candidates = np.nonzero(cand)[0]
        if src_candidates.size == 0:
            return cand
        i = int(rng.choice(src_candidates))
        # pick a distinct destination index
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # transfer a random fraction (up to 30 %) of the source mass
        frac = rng.random() * 0.3
        delta = cand[i] * frac
        if delta == 0.0:
            return cand
        cand[i] -= delta
        cand[j] += delta
        return cand

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    no_improve = 0
    for _ in range(num_iters):
        candidate = perturb(best)
        # keep values inside the allowed interval
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        cand_score = _score(candidate, n, fft_len)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = candidate
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
            # occasional random restart to escape tiny plateaus
            if no_improve >= 100:
                # restart from a fresh Dirichlet draw (keeps total mass = 1)
                rand = rng.random(n)
                rand /= rand.sum()
                best = rand
                best_score = _score(best, n, fft_len)
                no_improve = 0

    # ------------------------------------------------------------------
    # Final safety handling – ensure non‑negativity and the minimal sum.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.full(n, 0.01 / n, dtype=np.float64)

    # Return a plain Python list of floats as required by the harness.
    return [float(x) for x in best]
