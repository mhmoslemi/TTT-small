Implement function compute_score(a):
   n = len(a)
   # compute convolution via FFT
   N = 1 << ((2*n - 1).bit_length())
   fft_a = np.fft.rfft(a, n=N)
   b = np.fft.irfft(fft_a * fft_a, n=N)[:2*n-1]
   max_b = b.max()
   sum_a = a.sum()
   return 2 * n * max_b / (sum_a * sum_a)

import numpy as np

def _make_geometric(c, n):
    """
    Produce a non‑negative step sequence of length ``n`` whose entries follow
    a geometric law with decay parameter ``c``.
    The sequence is normalised to have sum 1, which is scale‑invariant for
    the evaluator.
    """
    # r = exp(-c/n)  ----  r**n = exp(-c)
    r = np.exp(-c / n)
    r_n = np.exp(-c)
    # normalising constant so that sum_i a_i == 1
    scale = (1.0 - r) / (1.0 - r_n)
    a = scale * np.power(r, np.arange(n, dtype=np.float64))
    # guard against tiny numerical drift
    total = a.sum()
    if total > 0.0:
        a /= total
    return a


def propose_candidate(seed=42, **kwargs):
    """
    Return a candidate coefficient list that (empirically) improves on the
    naive uniform construction.  The function builds a family of geometric
    sequences and keeps the one achieving the lowest AC1 upper bound
    according to ``evaluate_sequence``.
    """
    # Determine the problem size from the supplied parent construction.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:                 # fallback if the variable is absent
        n = 256

    # Candidate decay parameters to test – they span the region where the
    # bound is known to improve and include a generous out‑lier.
    decay_params = [0.5, 1.0, 2.0, 2.5, 3.0, 5.0, 10.0]

    best_seq = None
    best_score = float("inf")

    for c in decay_params:
        seq = _make_geometric(c, n)

        # Clamp to the evaluator's admissible range.
        seq = np.clip(seq, 0.0, 1000.0)

        # Evaluate – the evaluator returns ``inf`` on invalid data.
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq

    # If for any reason none of the candidates were valid, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_seq]
