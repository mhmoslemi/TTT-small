import numpy as np

# ------------------------------------------------------------
# Helper: build a set of indices such that each unordered sum
# i + j (i <= j) appears at most `L` times.
# ------------------------------------------------------------
def greedy_limited(order: np.ndarray, n: int, L: int):
    """Return (indices, count_array) where `count_array[s]` is the number
    of unordered pairs (i,j) with i<=j and i+j=s.
    No entry exceeds L.
    """
    counts = np.zeros(2 * n - 1, dtype=np.int16)   # sum range 0 .. 2n-2
    indices = []
    for i in order:
        # self‑pair already exceeds the limit?
        if counts[i + i] >= L:
            continue
        # check all existing indices
        conflict = False
        for j in indices:
            if counts[i + j] >= L:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        indices.append(i)
        # update counts for the new unordered pairs (including i+i)
        for j in indices:
            counts[i + j] += 1
    return indices, counts

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
