    # Here we can start with decreasing shape to guide.
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)   # start

    # Ensure sum>0.01.
    if a.sum() < 0.01:
        a = np.ones(n, dtype=np.float64) * 0.01

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Subgradient descent parameters
    max_iter = 2000
    step0 = 0.5
    for it in range(max_iter):
        # compute convolution
        b = np.convolve(a, a)
        M = b.max()
        # index of max
        k = int(b.argmax())
        # compute gradient of M w.r.t a
        grad_M = np.zeros_like(a)
        # overlapping indices: i from max(0, k-(n-1)) to min(k, n-1)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # For i in range(i_start, i_end+1):
        #   j = k - i
        #   grad_M[i] = a[j]
        # This loop is O(n) but we can vectorize:
        i_range = np.arange(i_start, i_end + 1)
        j = k - i_range
        grad_M[i_range] = a[j]

        # Compute gradient of objective f = 2n*M/(S^2)
        S = a.sum()
        grad = (2 * n / (S**2)) * (grad_M - (2 * M / S) * a)

        # Step size schedule
        step = step0 / np.sqrt(it + 1)
        a = a - step * grad
        # Project onto bounds
        a = np.clip(a, 0.0, 1000.0)

        # Evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score

        # Optionally early stop if improvement negligible
        if it % 100 == 0 and best_score < 1.6:
            break

    return best.tolist()
