def uniform_candidate(n, value=1.0):

def block_perturbation(arr, rng, scale_std=0.05):
    """Randomly scale a contiguous block of `arr` by a lognormal factor."""
    n = arr.size
    # Choose block size as a fraction of n
    block_div = int(rng.integers(10, 100))  # denominator between 10 and 99
    width = max(1, n // block_div)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, scale_std)))
    out = arr.copy()
    out[start:stop] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out

import numpy as np

def uniform_candidate(n, value=1.0):
    """Return a uniform non‑negative array of length *n*."""
    return np.full(n, float(value), dtype=np.float64)


def block_perturbation(arr, rng, scale_std=0.08):
    """Randomly scale a contiguous block of ``arr`` by a log‑normal factor."""
    n = arr.size
    # Choose a divisor to control block size (between 10 and 99)
    div = int(rng.integers(10, 100))
    width = max(1, n // div)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, scale_std)))
    out = arr.copy()
    out[start:stop] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out
