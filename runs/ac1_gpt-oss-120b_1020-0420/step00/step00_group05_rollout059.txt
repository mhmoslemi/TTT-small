def construct_sidon(n, target):
    rng = np.random.default_rng(seed)
    S = []
    sumset = set()
    # Try random numbers from 0..n-1 inclusive
    attempts = 0
    max_attempts = 1000000
    while len(S) < target and attempts < max_attempts:
        candidate = rng.integers(0, n)
        # compute the sums if we add candidate
        conflict = False
        # each existing s in S produce sum candidate + s; also candidate+candidate
        if (candidate + candidate) in sumset:
            conflict = True
        else:
            for s in S:
                if (candidate + s) in sumset:
                    conflict = True
                    break
        if not conflict:
            # accept
            for s in S:

import numpy as np

def _conv_self(x):
    """Full convolution of a 1‑D array with itself using FFT (O(n log n))."""
    n = x.shape[0]
    # length needed for linear convolution
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)
    return conv[: 2 * n - 1]


def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Heuristic optimisation that moves mass from indices contributing most
    to the current maximal convolution entry toward indices contributing the
    least.  The routine works with the runtime array ``height_sequence_1`` if
    present; otherwise it falls back to a uniform unit vector.
    """
    rng = np.random.default_rng(seed)

    # initialise from the provided construction if it exists
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                     # pragma: no cover
        base = np.ones(256, dtype=np.float64)

    # enforce the evaluator's clipping rules beforehand
    a = np.clip(base, 0.0, 1000.0)
    n = a.size
    total = float(a.sum())

        a = np.full(n, 1.0 / n, dtype=np.float64)
        total = 1.0

    # keep the original sum (scaling does not change the score)
    target_sum = total

    # current best solution
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # current convolution of the working candidate
    conv = _conv_self(best_seq)

    for step in range(num_steps):

        k_max = int(np.argmax(conv))

        if i_start > i_end:

        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
