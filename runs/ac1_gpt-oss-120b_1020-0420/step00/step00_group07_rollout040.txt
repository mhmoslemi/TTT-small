        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    return [float(x) for x in best]

def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except NameError:
        n = 6733
        parent = np.ones(n, dtype=np.float64)
    target_sum = np.sqrt(2.0 * n)
    best = None
    best_score = np.inf

    # include parent as candidate
    parent_candidate = np.clip(parent, 0.0, 1000.0)
    # rescale parent to target sum
    sum_parent = parent_candidate.sum()
    if sum_parent > 0:
        parent_candidate = parent_candidate * (target_sum / sum_parent)
    p_score = float(evaluate_sequence(parent_candidate.tolist()))
    if np.isfinite(p_score):
        best = parent_candidate
        best_score = p_score

    cs = [1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0]  # candidate c values
    for c in cs:
        if c >= n:
            continue
        log_q = np.log1p(-c / n)  # log(1 - c/n)
        if not np.isfinite(log_q):
            continue
        q_pow_n = np.exp(n * log_q)  # (1 - c/n)^n
        denom = 1.0 - q_pow_n
        if denom <= 0:
            continue
        alpha = target_sum * (c / n) / denom
        # compute a = alpha * q**i = alpha * exp(i * log_q)
        i_arr = np.arange(n, dtype=np.float64)
        a = alpha * np.exp(i_arr * log_q)
        # ensure within bounds
        a = np.clip(a, 0.0, 1000.0)
        # renormalize to target sum to correct rounding errors
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a * (target_sum / a_sum)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
    # local refinement: small random perturbations around best
    if best is not None:
        best_arr = best.copy()
        for _ in range(200):
            # jitter a bit
            jitter = rng.normal(0.0, best_arr.mean() * 0.05, size=n)
            cand = best_arr + jitter
            cand = np.clip(cand, 0.0, 1000.0)
            cand_sum = cand.sum()
            if cand_sum == 0:
                continue
            cand = cand * (target_sum / cand_sum)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_arr = cand.copy()
        best = best_arr
    else:
        # fallback to uniform if everything failed

    return [float(x) for x in best]

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Construct a decreasing exponential step sequence whose AC1 upper bound
    (as calculated by the provided evaluate_sequence) is below the target
    1.50274349.  The construction uses a closed‑form geometric series,
    rescales to the required L¹ norm sqrt(2n) and keeps all values in the
    allowable range [0,1000].  The best candidate among a few parameter
    choices (and the supplied parent, if any) is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain the problem size.  ``height_sequence_1`` may be present in the
    # execution environment; otherwise we fall back to a default length.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except Exception:            # pragma: no cover
        n = 6733                  # default length when no parent is given
        parent = np.ones(n, dtype=np.float64)

    # Required L1 norm (the evaluator cancels this factor in the score)
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate and keep the best one seen so far.
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf

    def consider(arr):
        nonlocal best, best_score
        # Clip to the evaluator's admissible range and renormalise.
        arr = np.clip(arr, 0.0, 1000.0)
        s = arr.sum()
        if s == 0.0:
            return
        arr = arr * (target_sum / s)
        score = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = arr.copy()

    # --------------------------------------------------------------
    # 1.  Parent (if it exists) as a baseline candidate.
    # --------------------------------------------------------------
    consider(parent)

    # --------------------------------------------------------------
    # 2.  Exponential (geometric) families a_i = α·qⁱ.
    #    We try several values of the decay parameter c where
    #          q = 1 - c/n   (0 < c < n)
    #    and α is chosen so that Σ a_i = target_sum.
    # --------------------------------------------------------------
    # candidate decay constants (theoretically optimal ~2)
    c_values = [0.8, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0]
    i_arr = np.arange(n, dtype=np.float64)

    for c in c_values:
        if c <= 0.0 or c >= n:
            continue
        # log_q = log(1 - c/n)   – stable even when c≪n
        log_q = np.log1p(-c / n)
        if not np.isfinite(log_q):
            continue
        # q^n = exp(n·log_q);  denominator = 1 - q^n  (>0)
        q_pow_n = np.exp(n * log_q)
        denom = 1.0 - q_pow_n
        if denom <= 0.0:
            continue
        # α that forces the L¹ norm to be target_sum
        alpha = target_sum * (c / n) / denom
        # Build the exponential vector
        a = alpha * np.exp(i_arr * log_q)
        consider(a)

    # --------------------------------------------------------------
    # 3.  Small local random perturbations around the best found so
    #     far – a cheap way to possibly improve the score a bit.
    # --------------------------------------------------------------
    if best is not None:
        best_arr = best.copy()
        # Scale of jitter relative to the typical entry size.
        jitter_scale = max(best_arr.mean(), 1e-8) * 0.15
        for _ in range(30):
            jitter = rng.normal(0.0, jitter_scale, size=n)
            cand = best_arr + jitter
            consider(cand)

    # --------------------------------------------------------------
    # 4.  Fallback: a uniform vector (guaranteed to be feasible) if
    #    everything else failed.
    # --------------------------------------------------------------
    if best is None or best_score == np.inf:
        uniform = np.full(n, target_sum / n, dtype=np.float64)
        best = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))

    return [float(x) for x in best]
