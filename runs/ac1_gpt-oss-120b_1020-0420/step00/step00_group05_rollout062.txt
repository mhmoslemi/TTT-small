import numpy as np
from scipy.optimize import linprog

def propose_candidate(seed=42, num_iter=12, blend=0.5, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that aims to minimise
    the AC1 upper bound used by `evaluate_sequence`.

    The method starts from a decreasing exponential profile (empirically good
    for this problem) and iteratively refines it by solving a small linear
    programme that pushes mass into positions that contribute little to the
    current worst convolution value.  The loop stops early if the target bound
    is reached.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # Determine the length of the construction.
    # Prefer the supplied `height_sequence_1` if it exists and looks valid.
    # ------------------------------------------------------------------ #
    parent = globals().get("height_sequence_1")
    if (
        isinstance(parent, np.ndarray)
        and parent.ndim == 1
        and parent.size > 0
        and np.isfinite(parent).all()
    ):
        n = int(parent.size)
    else:
        n = 256                     # safe fallback length

    # ------------------------------------------------------------------ #
    # Initial candidate – decreasing exponential (τ≈n/1.2 gives a good start).
    # ------------------------------------------------------------------ #
    tau = n / 1.2
    a = np.exp(-np.arange(n, dtype=np.float64) / tau)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------ #
    # Main refinement loop: solve tiny LPs that keep the worst convolution
    # value (M) unchanged while maximising the L1‑mass of the new vector.
    # ------------------------------------------------------------------ #
    for _ in range(num_iter):
        # Full convolution and the current worst value.
        conv = np.convolve(a, a)
        M = conv.max()

        # Choose a few shift indices that are responsible for M.
        # We take the three largest entries (by value) of the convolution.
        top_k = min(3, conv.size)
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
        shifts = np.unique(idxs)

        rows = []
        for k in shifts:
            # Build one row of the constraint matrix:
            #   sum_i a_i * g_{k-i} <= M   (g is the variable vector)
            start = max(0, k - (n - 1))
            end = min(n, k + 1)
            if start >= end:
                continue
            # coefficients that multiply the relevant g‑components
            coeff = a[k - np.arange(start, end, dtype=int)]
            row = np.zeros(n, dtype=np.float64)
            row[start:end] = coeff
            rows.append(row)

        if rows:
            A_ub = np.vstack(rows)
            b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
            # Bounds: 0 ≤ g_i ≤ 1000  (the evaluator also clips to these limits)
            bounds = [(0.0, 1000.0) for _ in range(n)]
            # Objective: maximise Σ g_i  →  minimise −Σ g_i
            c = -np.ones(n, dtype=np.float64)

            try:
                res = linprog(
                    c,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True},
                )
                if res.success:
                    g = res.x
                    # Blend a fraction of the LP solution into the current vector.
                    a = (1.0 - blend) * a + blend * g
                else:
                    # If the LP fails, fall back to a tiny random perturbation.
                    a = a * np.exp(rng.normal(0.0, 0.08, size=n))
            except Exception:
                # Any unexpected error also triggers a small random move.
                a = a * np.exp(rng.normal(0.0, 0.08, size=n))
        else:
            # No constraints (should not happen) – apply a small random change.
            a = a * np.exp(rng.normal(0.0, 0.08, size=n))

        # Clip to the range required by the evaluator.
        a = np.clip(a, 0.0, 1000.0)

        # Evaluate the new candidate.
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()

        # If we have already beaten the target bound, stop early.
        if best_score <= 1.50274349:
            break

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
