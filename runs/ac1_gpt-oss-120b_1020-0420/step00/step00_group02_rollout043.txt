    cur = np.clip(base, 0.0, 1000.0)
    n = cur.size

    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # For safety clip sum>0
    if np.sum(best) < 0.01:
        best[0] = 0.01
        best_score = float(evaluate_sequence(best.tolist()))

    for it in range(num_iters):
        # recompute conv only for current best (maybe better to recompute each iteration)
        conv = np.convolve(best, best)
        kmax = int(conv.argmax())

        if i_low > i_high:

                i = rng.integers(i_low, i_high + 1)
                j = kmax - i
                if best[i] > 0 and best[j] > 0:
                    # compute delta up to a fraction

                    if max_delta < 1e-8:

                    cand[j] -= delta
                    cand[p] += delta

import numpy as np

def _convolve_full(a: np.ndarray) -> np.ndarray:
    """Return full convolution a*a using FFT (faster than np.convolve for large n)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    fc = fa * fa
    c = np.fft.irfft(fc, n=size)[: 2 * n - 1]
    return c


def _max_conv_info(seq: np.ndarray):
    """Return (max value, index of a max element) of the full convolution."""
    conv = _convolve_full(seq)
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _random_pair_for_index(k, n, rng):
    """Choose a random pair (i, j) with i+j=k and 0≤i,j<n."""
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    i = int(rng.integers(i_min, i_max + 1))
    j = k - i
    return i, j
