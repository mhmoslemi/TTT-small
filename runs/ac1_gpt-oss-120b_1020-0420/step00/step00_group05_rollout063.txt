def compute_score(arr):
    n = arr.size
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(arr, arr)
    max_b = conv.max()
    return 2 * n * max_b / (sum_a * sum_a)
