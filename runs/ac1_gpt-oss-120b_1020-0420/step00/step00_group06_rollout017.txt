import numpy as np

def propose_candidate(seed=42, **kwargs):
    """Return a uniform sequence that minimizes the AC1 upper bound.

    The objective 2 * n * max(conv) / sum^2 is minimized when all entries are equal.
    This function builds such a uniform sequence using the parent `height_sequence_1`
    if it exists; otherwise it falls back to a small constant length.
    """
    # Try to obtain the provided parent sequence.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        # Clamp to allowed range.
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # Default length if parent missing or invalid.
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)
    total = float(parent.sum())

    # Ensure the total sum satisfies the evaluator's lower bound.
    if total < 0.01:
        # Use the smallest uniform positive sequence that meets the sum constraint.
        const_val = 0.01 / n
    else:
        const_val = total / n

    # Clamp constant to the allowed interval just in case of numerical issues.
    const_val = max(0.0, min(1000.0, const_val))

    candidate = np.full(n, const_val, dtype=np.float64)

    # Return as plain Python list of floats.
    return [float(x) for x in candidate]
