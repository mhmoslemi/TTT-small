        i_range = np.arange(left, right + 1, dtype=np.int64)
        j_range = idx_max - i_range
        products = cur[i_range] * cur[j_range]
        # pick index with largest product
        local_max_idx = int(np.argmax(products))
        i1 = i_range[local_max_idx]
        i2 = j_range[local_max_idx]
        # reduction factor random
        red_factor = rng.uniform(0.02, 0.12)  # 2% to 12%
        # compute new values
        new_i1 = cur[i1] * (1.0 - red_factor)
        new_i2 = cur[i2] * (1.0 - red_factor)
        delta = (cur[i1] + cur[i2]) - (new_i1 + new_i2)
        # choose two smallest indices (excluding i1,i2)
        # obtain sorted indices ascending
        sorted_idx = np.argsort(cur)
        inc = []
        for idx in sorted_idx:
            if idx != i1 and idx != i2:
                inc.append(idx)
                if len(inc) == 2:
                    break
        if len(inc) < 2:
            # fallback random different indices
            while len(inc) < 2:
                cand = rng.integers(0, n)
                if cand not in (i1, i2) and cand not in inc:
                    inc.append(cand)
        inc_amount = delta / 2.0
        cand = cur.copy()
        cand[i1] = new_i1
        cand[i2] = new_i2
        cand[inc[0]] += inc_amount
        cand[inc[1]] += inc_amount
        # clip values to valid range
        np.clip(cand, 0.0, 1000.0, out=cand)
        # optional re-normalize sum to 1 (preserve scale)
        # scaling does not affect score, but ensures stability
        s = np.sum(cand)
        if s > 0:
            cand = cand / s
        # evaluate
        sc = compute_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc
            cur = cand
            cur_score = sc
        else:
            # simulated annealing acceptance
            temp = max(0.0, 1.0 - step / num_steps)
            if sc < cur_score or rng.random() < np.exp((cur_score - sc) / (temp + 1e-12)):
                cur = cand
                cur_score = sc
        # occasional random perturbation to avoid stagnation
        if step % 200 == 0 and rng.random() < 0.1:
            # random small perturbation of a block
            width = max(1, n // rng.integers(100, 300))
            start = rng.integers(0, max(1, n - width + 1))
            stop = start + width
            # multiply block by random factor
            factor = np.exp(rng.normal(0.0, 0.03))
            cand2 = cur.copy()
            cand2[start:stop] *= factor
            # clip and renormalize
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            cand2 = cand2 / np.sum(cand2)
            sc2 = compute_score(cand2)
            if np.isfinite(sc2) and sc2 < best_score:
                best = cand2.copy()
                best_score = sc2
                cur = cand2
                cur_score = sc2
    # ensure final return is non-empty and finite
    if best_score == float('inf') or np.isnan(best_score):
        # fallback to uniform if needed
        best = np.full(n, 1.0 / n)
    return [float(x) for x in best]

import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the exact score used by the harness.
# ----------------------------------------------------------------------
def _score(arr: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/sum(a)^2 (same as evaluate_sequence)."""
    n = arr.size
    conv = np.convolve(arr, arr)
    max_b = np.max(conv)
    s = np.sum(arr)
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Core optimizer – projected sub‑gradient descent with a tiny stochastic
# component.  Works in O(n) memory and O(num_steps·n) time.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 2000, **_: dict) -> list:
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound 2·n·max(conv)/sum².

    The algorithm starts from a normalised version of the supplied
    `height_sequence_1` (or a uniform fallback) and repeatedly
    1. finds the index j where the convolution attains its maximum,
    2. identifies the pair (i, j‑i) contributing most to that maximum,
    3. moves a small fraction of mass from that pair to two currently
       smallest entries,
    4. projects back onto the probability simplex (scales so that the sum
       is 1 – this does not change the objective because it is scale‑invariant).

    An occasional random block perturbation is injected to escape shallow
    local minima.  The best vector seen is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise a normalised non‑negative vector.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:                 # when the harness does not provide it
        base = None

    if base is None or base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        # simple uniform fallback
        n = 256
        cur = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur = np.clip(base, 0.0, 1000.0)
        total = np.sum(cur)
        if total < 1e-12:
            cur = np.full(cur.shape, 1.0 / cur.size, dtype=np.float64)
        else:
            cur = cur / total                     # normalise to sum = 1

    n = cur.size
    best = cur.copy()
    best_score = _score(best)
    cur_score = best_score

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # ----- 1.  convolution and location of the current maximum -----
        conv = np.convolve(cur, cur)
        jmax = int(np.argmax(conv))

        # ----- 2.  pair (i, jmax-i) with the largest contribution -----
        left = max(0, jmax - (n - 1))
        right = min(jmax, n - 1)
        i_vals = np.arange(left, right + 1, dtype=np.int64)
        k_vals = jmax - i_vals
        contributions = cur[i_vals] * cur[k_vals]
        idx_local = int(np.argmax(contributions))
        i1 = i_vals[idx_local]
        i2 = k_vals[idx_local]

        # ----- 3.  move a small fraction of mass away from the pair -----
        # random reduction factor (2 % – 12 %)
        red = rng.uniform(0.02, 0.12)
        new_i1 = cur[i1] * (1.0 - red)
        new_i2 = cur[i2] * (1.0 - red)
        removed = (cur[i1] + cur[i2]) - (new_i1 + new_i2)

        # two smallest entries (different from i1,i2) receive the mass
        asc = np.argsort(cur)
        inc_idxs = []
        for idx in asc:
            if idx != i1 and idx != i2:
                inc_idxs.append(idx)
                if len(inc_idxs) == 2:
                    break
        # (fallback – should never happen)
        while len(inc_idxs) < 2:
            cand = rng.integers(0, n)
            if cand not in (i1, i2) and cand not in inc_idxs:
                inc_idxs.append(cand)

        inc_amount = removed / 2.0
        cand = cur.copy()
        cand[i1] = new_i1
        cand[i2] = new_i2
        cand[inc_idxs[0]] += inc_amount
        cand[inc_idxs[1]] += inc_amount

        # ----- 4.  projection onto the simplex (scale to sum = 1) -----
        # clipping first keeps the vector non‑negative; scaling does not
        # affect the objective because it is homogeneous.
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = np.sum(cand)
        if s > 0.0:
            cand = cand / s

        cand_score = _score(cand)

        # ----- 5.  accept if improvement, otherwise occasional SA move -----
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            cur = cand
            cur_score = cand_score
        else:
            # simple simulated‑annealing schedule
            temperature = max(0.0, 1.0 - step / num_steps)
            if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / (temperature + 1e-12)):
                cur = cand
                cur_score = cand_score

        # ----- 6.  occasional random block perturbation (exploration) -----
        if (step % 250 == 0) and (rng.random() < 0.08):
            width = max(1, n // rng.integers(120, 240))
            start = rng.integers(0, max(1, n - width + 1))
            stop = start + width
            factor = np.exp(rng.normal(0.0, 0.04))
            tmp = cur.copy()
            tmp[start:stop] *= factor
            np.clip(tmp, 0.0, 1000.0, out=tmp)
            tmp = tmp / np.sum(tmp)
            tmp_score = _score(tmp)
            if np.isfinite(tmp_score) and tmp_score < best_score:
                best = tmp.copy()
                best_score = tmp_score
                cur = tmp
                cur_score = tmp_score

    # ------------------------------------------------------------------
    # Fallback safety: if something went wrong, return a simple uniform vector.
    # ------------------------------------------------------------------
    if best_score == float("inf") or not np.isfinite(best_score):
        best = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best]
