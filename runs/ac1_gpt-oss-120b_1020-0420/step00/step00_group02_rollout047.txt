def geometric_candidate(n, delta):
    # delta small positive
    r = 1.0 - delta
    # compute exponent as arange
    idx = np.arange(n, dtype=np.float64)
    cand = np.exp(idx * np.log(r))
    # scale to max 1000
    cand = cand * (1000.0 / cand.max())
    return np.clip(cand, 0.0, 1000.0)

import numpy as np


def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # pick a block width between n/200 and n/12 (inclusive)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation on the block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring to keep some symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
