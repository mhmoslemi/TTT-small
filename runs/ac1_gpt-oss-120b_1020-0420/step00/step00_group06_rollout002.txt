import numpy as np

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Return a candidate sequence of non‑negative heights that (approximately)
    minimizes the AC1 upper bound used by `evaluate_sequence`.

    The strategy:
    1. Start from the provided `height_sequence_1` (or a simple fallback).
    2. Compare it with the uniform distribution having the same total mass;
       uniform is optimal for the bound, giving a score of exactly 2.
    3. Perform a cheap stochastic re‑balancing local search starting from the
       better of the two, keeping the best‑found sequence.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Load a valid parent sequence.
    # ------------------------------------------------------------------
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)

    best_arr = parent.copy()
    best_score = float(evaluate_sequence(best_arr.tolist()))

    n = parent.size

    # ------------------------------------------------------------------
    # 2. Uniform distribution with the same total mass.
    # ------------------------------------------------------------------
    total = float(best_arr.sum())
    if total < 1e-12:               # guard against a zero sum (should not happen)
        total = 1.0
    uniform = np.full(n, total / n, dtype=np.float64)
    uniform = np.clip(uniform, 0.0, 1000.0)

    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best_arr = uniform
        best_score = uniform_score

    # ------------------------------------------------------------------
    # 3. Simple mass‑rebalancing stochastic search.
    # ------------------------------------------------------------------
    # Move a small amount of mass from one index to another while keeping
    # the total sum constant.  This never violates the non‑negative bound.
    for _ in range(num_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Choose a delta up to 20 % of the donor's current height.
        donor_val = best_arr[i]
        if donor_val <= 0.0:
            continue
        delta = rng.random() * 0.2 * donor_val

        candidate = best_arr.copy()
        candidate[i] -= delta
        candidate[j] += delta

        # Enforce the per‑element bounds.
        candidate[i] = max(0.0, min(1000.0, candidate[i]))
        candidate[j] = max(0.0, min(1000.0, candidate[j]))
        # (All other entries already satisfy the bounds.)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_arr = candidate
            best_score = score

    # ------------------------------------------------------------------
    # 4. Final safeguard: ensure a non‑trivial total mass.
    # ------------------------------------------------------------------
    if best_arr.sum() < 0.01:
        best_arr[0] = max(best_arr[0], 0.01)

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_arr]
