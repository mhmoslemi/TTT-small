import numpy as np

# ----------------------------------------------------------------------
# Helper: produce a power‑law shaped candidate (decreasing if reverse=True)
# ----------------------------------------------------------------------
def _gen_power_candidate(n, beta, reverse=False):
    """Return a non‑negative step sequence of length n.

    The shape is proportional to (idx+1)^beta (or (n-idx)^beta if reverse).
    The result is normalised to sum 1 and then scaled to keep values modest.
    """
    idx = np.arange(n, dtype=np.float64)
    if reverse:
        vals = (n - idx) ** beta
    else:
        vals = (idx + 1) ** beta
    total = vals.sum()
    if total == 0.0:
        return np.zeros(n, dtype=np.float64)
    vals = vals / total                     # sum == 1
    vals = np.clip(vals * 100.0, 0.0, 1000.0)  # arbitrary safe scale
    return vals


# ----------------------------------------------------------------------
# Helper: exponential decay candidate (decreasing if reverse=True)
# ----------------------------------------------------------------------
def _gen_exp_candidate(n, rate, reverse=False):
    """Exponential shape: exp(-rate*idx) (or reversed)."""
    idx = np.arange(n, dtype=np.float64)
    if reverse:
        vals = np.exp(-rate * idx)
    else:
        vals = np.exp(-rate * (n - 1 - idx))
    total = vals.sum()
    if total == 0.0:
        return np.zeros(n, dtype=np.float64)
    vals = vals / total
    vals = np.clip(vals * 100.0, 0.0, 1000.0)
    return vals


# ----------------------------------------------------------------------
# Small local perturbation that keeps memory O(n)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Multiply a random contiguous block by a log‑normal factor."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block size roughly between n/200 and n/12
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional mirroring to keep symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑empty list of heights that (hopefully) beats the
    baseline bound 2.0.  The routine first explores a few
    parametric families (power‑law and exponential shapes) and then
    performs a short memory‑bounded local search.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the problem size; fall back to a small safe length.
    # --------------------------------------------------------------
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0 or not np.isfinite(parent_raw).all():
            raise ValueError
        n = int(parent_raw.size)
    except Exception:
        n = 256
        parent_raw = np.ones(n, dtype=np.float64)

    # --------------------------------------------------------------
    # Start from the provided parent (or the safe fallback)
    # --------------------------------------------------------------
    parent = np.clip(parent_raw, 0.0, 1000.0)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------
    # 1) Parameterised search (few dozen cheap evaluations)
    # --------------------------------------------------------------
    # power‑law (decreasing)  –  beta from 0 (uniform) up to 4
    betas = np.linspace(0.0, 4.0, 21)          # 21 values
    for beta in betas:
        cand = _gen_power_candidate(n, beta, reverse=True)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score

    # exponential decay (decreasing) – rate from 0 to 0.02
    rates = np.linspace(0.0, 0.02, 9)          # 9 values

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score

    local_steps = 400

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # --------------------------------------------------------------
    # Return the final candidate as a plain Python list
    # --------------------------------------------------------------
    return [float(v) for v in best]
