import numpy as np

def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑2 FFT (real input)."""
    n = a.size
    N = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[: 2 * n - 1]
    return conv


def _score(seq: np.ndarray) -> float:
    """Exact score matching ``evaluate_sequence`` (assumes clipping already)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    b = _fft_convolve(seq)
    max_b = float(b.max())
    return float(2.0 * seq.size * max_b / (s * s))

            red_j = cand[j_red] * 0.1
            cand[i_red] = max(0.0, cand[i_red] - red_i)
            cand[j_red] = max(0.0, cand[j_red] - red_j)
            total_red = red_i + red_j

        # re‑inject the taken mass at a random location (respecting the cap)
        rand_idx = rng.integers(0, n)
        while rand_idx == i_red or rand_idx == j_red:
            rand_idx = rng.integers(0, n)
        cand[rand_idx] = min(1000.0, cand[rand_idx] + total_red)

        cand_score = _score(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score

    # ---------- final fallback -------------------------------------------------
    if not np.isfinite(best_score):
        best = np.full_like(parent, 1.0)
    return [float(x) for x in best]
