    The search space is the family a_i = (i + α)^{-p} with

    is applied afterwards.
    """

    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)

            raise ValueError
        n = base.size
    except Exception:
        # Fallback length if the parent is unavailable
        n = 256

    # ------------------------------------------------------------------
    # Helper for evaluating a candidate (already handles clipping)
    # ------------------------------------------------------------------
    def _score(arr):
        arr = np.clip(arr, 0.0, 1000.0)
        return float(evaluate_sequence(arr.tolist()))

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Grid search over exponent p and additive shift α
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    p_vals = np.linspace(0.5, 0.78, 30)                  # 30 values
