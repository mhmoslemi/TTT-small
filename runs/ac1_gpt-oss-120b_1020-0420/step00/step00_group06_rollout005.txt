def propose_candidate(seed=42, num_steps=200, block_scale=0.12, **kwargs):
    """
    Return a non‑negative step‑function sequence that (approximately) minimizes
    the AC1 upper bound defined by ``evaluate_sequence``.

    The algorithm starts from a uniform vector (the provably optimal shape for
    the given objective) and performs a tiny amount of random block‑wise
    multiplicative perturbation to guarantee a valid fallback in case the

    rng = np.random.default_rng(seed)

    # ----- 1️⃣  Initial uniform candidate (optimal w.r.t the bound) ----------
    best = _initial_candidate()
    best_score = float(evaluate_sequence(best.tolist()))

    n = best.size

        return [1.0]

    # ----- 2️⃣  Light local search (strictly optional) -----------------------
    # The search never creates a candidate worse than the current best.
