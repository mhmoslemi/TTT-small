# The evaluator `evaluate_sequence` is provided by the harness.

def _score(seq):
    """Return the AC1 upper‑bound score for a 1‑D sequence."""
    # Convert to Python list as required by the evaluator.
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1
    upper‑bound is (hopefully) below the baseline 2.0.

    The algorithm starts from a uniform vector and performs a cheap
    hill‑climbing that repeatedly lowers the largest convolution entry
    while keeping the total sum (and thus the denominator of the score)
    large.  All auxiliary storage is O(n).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Length of the problem (fallback to a modest size if the parent is
    # missing for any reason).
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])          # parent provided by harness
    except Exception:                                # pragma: no cover
        n = 256

    # ------------------------------------------------------------------
    # Initial candidate: a uniform non‑zero vector.
    # ------------------------------------------------------------------
    a = np.ones(n, dtype=np.float64)                  # uniform height = 1.0
    best = a.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # Helper to perform a cheap random perturbation (Gaussian multiplicative
    # noise) and keep the best candidate seen so far.
    # ------------------------------------------------------------------
    def _random_perturb(vec, sigma):
        """Multiply every entry by exp(N(0, sigma)) and clip to the allowed range."""
        pert = np.exp(rng.normal(0.0, sigma, size=vec.shape))
        cand = np.clip(vec * pert, 0.0, 1000.0)
        return cand

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    sigma = 0.12                      # initial noise level for random moves
    for step in range(num_steps):
        # --------------------------------------------------------------
        # Occasionally try a global random perturbation to escape plateaus.
        # --------------------------------------------------------------
        if rng.random() < 0.12:
            cand = _random_perturb(a, sigma)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                a = cand
                best = cand.copy()
                best_score = sc
            # Reduce the noise gradually – a simple annealing schedule.
            sigma *= 0.998
            continue

        # --------------------------------------------------------------
        # Targeted move: identify the index of the largest convolution entry,
        # then move a small amount of mass away from the pair(s) that create it.
        # --------------------------------------------------------------
        conv = np.convolve(a, a)                     # length 2n‑1
        k_max = int(np.argmax(conv))                 # shift with maximal value

        # indices i that contribute to this shift (i + j == k_max)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        i_range = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_range = k_max - i_range

        # contributions a[i] * a[j] (both non‑negative)
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()
        if total_contrib == 0.0:                     # nothing to improve – fall back
            cand = _random_perturb(a, sigma * 0.5)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                a = cand
                best = cand.copy()
                best_score = sc
            continue

        # pick a single index among the contributors, weighted by its
        # contribution to the maximal entry.
        prob = contrib / total_contrib
        i_pick = rng.choice(i_range, p=prob)

        # Reduce that entry a[i_pick] a little and give the removed mass
        # to a random, non‑identical location.
        reduction_factor = 0.90                     # 10 % reduction
        old_val = a[i_pick]
        new_val = old_val * reduction_factor
        delta = old_val - new_val                    # mass to relocate

        a[i_pick] = new_val
        # choose a different index to receive the mass
        other = rng.integers(0, n)
        while other == i_pick:
            other = rng.integers(0, n)
        a[other] = min(1000.0, a[other] + delta)

        # --------------------------------------------------------------
        # Evaluate the new candidate.
        # --------------------------------------------------------------
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc
        else:
            # revert the change – it was not beneficial
            a[i_pick] = old_val
            a[other] = a[other] - delta

    # ------------------------------------------------------------------
    # Fallback: if for some reason we never found a finite score,
    # return a simple uniform vector (guaranteed to give score = 2.0).
    # ------------------------------------------------------------------
    if best_score == float('inf'):
        best = np.ones(n, dtype=np.float64)
        best_score = _score(best)

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
