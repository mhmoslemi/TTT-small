import numpy as np

def make_block_perturbation(base, rng):
    """Create an O(n)-memory nonnegative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size

    # Choose a block width between ~0.5% and ~5% of the length.
    min_w = max(1, n // 200)
    max_w = max(2, n // 20)
    if max_w < min_w:
        max_w = min_w
    width = int(rng.integers(min_w, max_w + 1))

    # Random block start.
    start = int(rng.integers(0, n - width + 1))
    stop = start + width

    # Multiplicative log‑normal perturbation.
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor

    # Optional mild symmetry to keep structure.
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])

    # Enforce the problem bounds.
    return np.clip(cand, 0.0, 1000.0)
