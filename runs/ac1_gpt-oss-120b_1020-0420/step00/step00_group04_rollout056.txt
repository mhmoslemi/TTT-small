def interleaved_sequence(arr):
    arr_sorted = np.sort(arr)  # ascending.
    # We'll produce sequence where we take from high end then low end alternately.
    high = arr_sorted.size - 1
    low = 0
    out = np.empty_like(arr_sorted)
    i = 0
    while low <= high:
        out[i] = arr_sorted[high]
        i += 1
        high -= 1
        if low <= high:
            out[i] = arr_sorted[low]
            i += 1
            low += 1
    return out

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that aim to
    minimise the evaluator's AC1 upper bound.
    The method searches over simple power‑law families a_i = (i+beta)^(-gamma)
    (with normalisation) and then performs a modest local random refinement.
    """
    # Determine problem size from the provided parent array, if any.
    n = 256  # a safe fallback length
    try:
        # height_sequence_1 is supplied by the harness when available.
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = int(parent.size)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) Global parametrised search (gamma, beta) over a simple power law.
    # ------------------------------------------------------------------
    # Initialise with a uniform candidate (guaranteed feasible).
    best = np.ones(n, dtype=np.float64) / n
    best_score = float(evaluate_sequence(best.tolist()))

    # Parameter grids (few values – cheap to evaluate).
    gamma_vals = [0.55, 0.60, 0.65, 0.70, 0.75]
    # beta ranging from 1e-2 to 1e1 (log‑spaced).
    beta_vals = np.logspace(-2, 1, num=20)

    idx = np.arange(n, dtype=np.float64)        # indices 0 … n‑1
    for gamma in gamma_vals:
        inv_gamma = -gamma                      # power used later
        for beta in beta_vals:
            seq = (idx + beta) ** inv_gamma      # a_i = (i+beta)^(-gamma)
            total = seq.sum()
            if total <= 0.0:
                continue                         # malformed – skip
            seq /= total                         # normalise L1‑norm to 1
            # The evaluator caps values to [0,1000]; our numbers are far below.
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score

    # --------------------------------------------------------------
    # 2) Light local random refinement starting from the best found.
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)

    # Number of refinement steps – modest to keep run‑time low.
    for _ in range(800):
        cand = best.copy()

        # Pick a random contiguous block and scale it multiplicatively.
        # Block width is O(n) but never larger than the array.
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        factor = float(np.exp(rng.normal(0.0, 0.07)))   # small log‑normal perturbation
        cand[start:stop] *= factor

        # Keep entries non‑negative and within the evaluator's limits.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Re‑normalise – the objective is scale‑invariant, this also avoids
        # the sum‑being too small (the evaluator requires sum ≥ 0.01).
        s = cand.sum()
        if s > 0.0:
            cand /= s

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # --------------------------------------------------------------
    # 3) Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------
    # Ensure the result is non‑empty and finite.
    if best.size == 0 or not np.isfinite(best).all():
        # Fallback to a simple uniform sequence if something went wrong.
        best = np.ones(256, dtype=np.float64) / 256.0

    return [float(x) for x in best]
