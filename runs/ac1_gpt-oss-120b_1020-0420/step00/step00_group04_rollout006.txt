import numpy as np

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as computed by ``evaluate_sequence``.

    The theoretical lower bound for non‑negative sequences is 2, achieved by
    any constant sequence.  This implementation starts from a constant
    sequence of ones (which already attains the bound) and performs a very
    light random block‑perturbation search that only accepts strictly better
    candidates, guaranteeing the final score is never worse than 2.
    """
    rng = np.random.default_rng(seed)

    # Determine the length of the desired sequence.
    # Use the provided parent ``height_sequence_1`` when available,
    # otherwise fall back to a modest default length.
    try:
        n = int(height_sequence_1.shape[0])
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)  # never used, just for safety

    # Start from a constant sequence – this already gives the optimal score 2.
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # Light local search: try random multiplicative tweaks on a random block.
    # Accept only strictly improving candidates, and stop early if we hit the
    # known lower bound (within floating‑point tolerance).
    for _ in range(num_steps):
        # Block width scales with the length, but stays O(n).
        max_width = max(1, n // 5)
        width = rng.integers(1, max_width + 1)
        start = rng.integers(0, n - width + 1)
        stop = start + width

        # Small multiplicative perturbation.
        factor = np.exp(rng.normal(0.0, 0.08))
        candidate = best.copy()
        candidate[start:stop] *= factor

        # Occasionally reflect to keep symmetry (optional).
        if rng.random() < 0.1:
            candidate = 0.5 * (candidate + candidate[::-1])

        # Enforce the problem's box constraints.
        candidate = np.clip(candidate, 0.0, 1000.0)

        # Evaluate and accept if better.
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score - 1e-12:
            best = candidate
            best_score = score
            if best_score <= 2.000001:   # reached the provable optimum
                break

    return best.tolist()
