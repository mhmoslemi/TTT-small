def compute_score(a):
    n = a.size
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, nfft)
    fc = fa * fa
    c = np.fft.irfft(fc, nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

import numpy as np

# ----------------------------------------------------------
# Fast (matrix‑free) evaluation of the objective.
# ----------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.shape[0]
    # length of the full convolution
    m = 2 * n - 1
    # next power of two for efficient FFT
    nfft = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fc = fa * fa                     # pointwise product → convolution in time domain
    c = np.fft.irfft(fc, n=nfft)[:m]
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

# Main entry point required by the harness.
# ----------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound.

    The routine builds several compact analytic families (inverse‑square‑root,
    power laws, their reversals) and keeps the best one according to the fast
    score.  A short random refinement around the current best is performed as
    a final touch.  The parent construction `height_sequence_1` is used as a
    fallback candidate.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Base length and a safe parent fallback.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    parent = np.clip(parent, 0.0, 1000.0)

    # start with a uniform vector – it yields the classic bound 2.0
    best_seq = np.ones(n, dtype=np.float64)
    best_score = _fast_score(best_seq)

    # also try the supplied parent (might already be decent)
    if np.sum(parent) >= 0.01:
        s = _fast_score(parent)
        if s < best_score:
            best_seq = parent.copy()
            best_score = s

    # ------------------------------------------------------------------
    # 1) Inverse‑square‑root family  a_i = 1 / sqrt(i + offset)
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    offsets = np.linspace(0.1, 4.0, 30)          # 30 trial offsets
    for off in offsets:
        a = 1.0 / np.sqrt(i + off)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s

        # its reversal (decreasing instead of increasing)
        a_rev = 1.0 / np.sqrt((n - 1 - i) + off)
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 2) Power‑law family a_i = (i+1)^{-p}
    # ------------------------------------------------------------------
    powers = np.linspace(0.35, 0.65, 13)        # around the optimal 0.5
    for p in powers:
        a = (i + 1) ** (-p)
        s = _fast_score(a)
        if s < best_score:
            best_seq, best_score = a, s

        a_rev = (n - i) ** (-p)                # same law reflected
        s = _fast_score(a_rev)
        if s < best_score:
            best_seq, best_score = a_rev, s

    # ------------------------------------------------------------------
    # 3) Tiny random multiplicative tweaks around the current best.
    # ------------------------------------------------------------------
    for _ in range(200):
        perturb = np.exp(rng.normal(0.0, 0.02, size=n))   # ~2 % jitter
        cand = best_seq * perturb
        cand = np.clip(cand, 0.0, 1000.0)
        s = _fast_score(cand)
        if s < best_score:
            best_seq, best_score = cand, s

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
