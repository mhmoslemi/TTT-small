    Generate a non‑negative step‑function heights sequence that minimizes the
    AC1 upper bound evaluated by `evaluate_sequence`.  The construction uses a

    score is far below the target bound.
    """

    # If not, fall back to a modest default length.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            n = 256
        else:
            n = parent.size
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Target L1 norm (the evaluator normalises by the square of this sum).
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build an exponentially decreasing vector: a_i = c * r**i  (i = 0 .. n-1)
    # with 0 < r < 1.  Using r = 0.5 yields a very steep decay, making
    # every autocorrelation term tiny.
    # ------------------------------------------------------------------
    r = 0.5

    # Powers r**i, underflow to 0 for large i – this is fine.
    idx = np.arange(n, dtype=np.float64)
    powers = np.power(r, idx, dtype=np.float64)

    # Sum of the geometric series: (1 - r**n) / (1 - r)
    denom = 1.0 - np.power(r, n, dtype=np.float64)
    if denom == 0.0:                    # degenerate case (practically never hit)
        coeff = target_sum
    else:

    seq = np.clip(seq, 0.0, 1000.0)

    cur_sum = seq.sum()

    seq = np.clip(seq, 0.0, 1000.0)

    # Return as a plain Python list of floats.
    return [float(x) for x in seq]
