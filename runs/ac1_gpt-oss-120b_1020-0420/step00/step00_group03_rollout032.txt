def conv_full(p):
    fft_p = np.fft.rfft(p, n=size)
    conv = np.fft.irfft(fft_p * fft_p, n=size)
    conv = conv[:2*n-1]
    conv[conv < 0] = 0.0
    return conv
```

import numpy as np
import time

def full_convolution(arr):
    """Linear convolution using FFT (size = next power of two). Returns non‑negative real array."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # smallest power of two ≥ 2n‑1
    fft_arr = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fft_arr * fft_arr, n=size)
    conv = conv[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)              # guard against tiny negative rounding errors
    return conv
