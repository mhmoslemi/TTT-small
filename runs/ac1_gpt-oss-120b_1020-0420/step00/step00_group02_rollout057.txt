import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a modest block-wise random perturbation (fallback)
# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional mirroring for symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the evaluator’s bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
