import numpy as np

# The evaluator `evaluate_sequence` is pre‑imported by the harness.

def evaluate_score(arr):
    """Return the AC1 upper‑bound score for a NumPy 1‑D array."""
    return float(evaluate_sequence(arr.tolist()))


def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound.  Uses a very small memory footprint and
    falls back to a constant sequence which attains the theoretical optimum
    score of 2.0.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the provided parent sequence, falling back to a simple
    # uniform vector if the parent is missing or malformed.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:          # height_sequence_1 may be undefined
        parent = None

    if (parent is None or parent.ndim != 1 or parent.size == 0
            or not np.isfinite(parent).all()):
        parent = np.ones(256, dtype=np.float64)

    # Ensure values are within the allowed range.
    parent = np.clip(parent, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Keep track of the best candidate seen so far.
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = evaluate_score(best)

    # ------------------------------------------------------------------
    # A trivial constant candidate (the theoretical optimum, score = 2.0).
    # ------------------------------------------------------------------
    const_candidate = np.full_like(best, fill_value=1.0)
    const_score = evaluate_score(const_candidate)

    if np.isfinite(const_score) and const_score < best_score:
        best = const_candidate
        best_score = const_score

    # ------------------------------------------------------------------
    # Light local search: small random block perturbations around the best.
    # Each perturbation uses O(n) memory and calls the exact evaluator.
    # ------------------------------------------------------------------
    n = best.size
    for _ in range(num_steps):
        # Choose a random block width (between n/200 and n/12) and location.
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        # Apply a multiplicative perturbation drawn from a log‑normal.
        factor = float(np.exp(rng.normal(0.0, 0.05)))  # modest change
        cand = best.copy()
        cand[start:stop] *= factor

        # Occasionally symmetrise the candidate (helps reduce peaks).
        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        # Enforce the admissible range.
        cand = np.clip(cand, 0.0, 1000.0)

        score = evaluate_score(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
