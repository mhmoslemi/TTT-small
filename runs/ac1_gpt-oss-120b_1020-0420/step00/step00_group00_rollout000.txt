import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a constant (uniform) sequence of length matching the provided
    ``height_sequence_1`` if available, otherwise a modest fallback length.
    A uniform sequence minimizes the AC1 upper bound analytically, achieving
    the theoretical optimum of 2.0.
    """
    # Determine the target length.
    try:
        # height_sequence_1 is supplied by the harness when a parent exists.
        n = int(height_sequence_1.shape[0])
        # Ensure it is a one‑dimensional vector.
        if height_sequence_1.ndim != 1 or n == 0:
            raise ValueError

        # Fallback length if the parent vector is missing or malformed.
        n = 256

    # Uniform value – any positive constant works (clipped to [0,1000]).
    # Using 1.0 keeps numbers modest and avoids overflow.
    uniform_value = 1.0
    candidate = np.full(n, uniform_value, dtype=np.float64)

    # Enforce the evaluator's clipping rules (redundant but safe).
