def project_simplex(v):
    # based on algorithm from Duchi et al.
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step‑function (height sequence) whose
    AC1 upper bound 2 * n * max(conv) / (sum**2) is as small as possible.
    The function tries a variety of parametrised families and a few
    random Dirichlet draws, keeping the best candidate found.
    """
    rng = np.random.default_rng(seed)

    # length of the problem instance
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    # --- initial uniform candidate ------------------------------------------------
    best = np.ones(n, dtype=np.float64) / n
    best_score = float(evaluate_sequence(best.tolist()))

    # helper to test a new array and keep the best one
    def consider(arr):
        nonlocal best, best_score
        # normalise to a probability vector (sum == 1) – scale does not affect score
        s = arr.sum()
        if s == 0:
            return
        arr = arr / s
        arr = np.clip(arr, 0.0, 1000.0)
        sc = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = arr.copy()
            best_score = sc

    # -------------------------------------------------------------------------------
    # 1. Exponential families  a_i = r^i
    for t in np.linspace(0.5, 5.0, 12):               # t ≈ n*(1‑r)
        r = 1.0 - t / n
        if r <= 0:
            continue
        arr = np.power(r, np.arange(n, dtype=np.float64))
        consider(arr)

    # -------------------------------------------------------------------------------
    # 2. Power‑law decreasing  a_i ∝ (i+1)^{-beta}
    for beta in np.linspace(0.1, 5.0, 30):
        idx = np.arange(1, n + 1, dtype=np.float64)
        arr = 1.0 / np.power(idx, beta)
        consider(arr)

    # -------------------------------------------------------------------------------
    # 3. Polynomial (1‑x)^beta  with x = i/(n‑1)
    for beta in np.linspace(0.1, 5.0, 30):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        arr = np.power(1.0 - x, beta)
        consider(arr)

    # -------------------------------------------------------------------------------
    # 4. Stretched exponentials  a_i = exp( - (i/(n‑1))^gamma )
    for gamma in (0.5, 1.0, 2.0, 3.0):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        arr = np.exp(-np.power(x, gamma))
        consider(arr)

    # -------------------------------------------------------------------------------
    # 5. Random Dirichlet draws (uniform concentration)
    for _ in range(300):
        # Gamma(1,1) -> Dirichlet(1,…,1) (uniform on simplex)
        arr = rng.gamma(shape=1.0, scale=1.0, size=n)
        consider(arr)

    # -------------------------------------------------------------------------------
    # If nothing beats the uniform fallback (unlikely), best already holds it.
    return [float(v) for v in best]
