import numpy as np

# Helper that forwards the candidate to the provided evaluator.
def _score(arr):
    # `arr` is a NumPy 1‑D array of floats.
    # The harness has already imported `evaluate_sequence`.
    return float(evaluate_sequence(arr.tolist()))


def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) with a low
    autocorrelation upper bound.  The routine starts from a random
    0/1 pattern, performs cheap local flips and occasional full
    restarts, always keeping the best candidate seen.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the problem and a valid fallback parent.
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.size)          # length supplied by the harness
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64),
            0.0, 1000.0
        )
    except Exception:                           # very unlikely, but safe
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best = parent.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # First candidate: a dense random binary (0/1) pattern.
    # ------------------------------------------------------------------
    cand = (rng.random(n) < 0.5).astype(np.float64)   # 0‑1 vector
    sc = _score(cand)
    if np.isfinite(sc) and sc < best_score:
        best, best_score = cand, sc

    # ------------------------------------------------------------------
    # Local improvement loop.
    #   * Flip a small block of bits in the current best solution.
    #   * Occasionally replace the current best by a fresh random binary.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # --- small block flip -------------------------------------------------
        block_len = int(rng.integers(1, max(2, n // 300)))   # O(1)‑size block
        start = int(rng.integers(0, n - block_len + 1))
        new = best.copy()
        # binary flip: 0 → 1, 1 → 0
        new[start:start + block_len] = 1.0 - new[start:start + block_len]
        sc = _score(new)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = new, sc

        # --- occasional full restart -----------------------------------------
        if rng.random() < 0.03:
            rnd = (rng.random(n) < 0.5).astype(np.float64)
            sc = _score(rnd)
            if np.isfinite(sc) and sc < best_score:
                best, best_score = rnd, sc

    # Return the best vector as a plain Python list of floats.
    return [float(v) for v in best]
