def compute_score(a):
    n = a.shape[0]

    m = int(2 ** np.ceil(np.log2(2*n - 1)))

def targeted_transfer(a, rng, factor):
    n = a.shape[0]
    m = max_len (precomputed) # hold outside...
    fft = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
    max_idx = int(conv.argmax())
    start = max(0, max_idx - (n - 1))
    end = min(max_idx, n - 1)
    idx_range = np.arange(start, end + 1)
    counterpart = max_idx - idx_range
    prod = a[idx_range] * a[counterpart]
    # find index with max product
    max_pos = int(idx_range[np.argmax(prod)])
    max_partner = int(counterpart[np.argmax(prod)])
    # Transfer mass
    if a[max_pos] > 0:
        delta = a[max_pos] * factor
        a[max_pos] -= delta
        # random target not same
        target = rng.integers(0, n)
        a[target] += delta
    if max_partner != max_pos and a[max_partner] > 0:
        delta2 = a[max_partner] * factor
        a[max_partner] -= delta2
        target2 = rng.integers(0, n)
        a[target2] += delta2
    # Ensure nonnegatives
    np.clip(a, 0.0, 1000.0, out=a)
    # Renormalize sum to 1 to maintain scale?
    total = a.sum()
    if total > 0:
        a /= total
```

# Helper: fast convolution using FFT (exact for real numbers)
# ------------------------------------------------------------
def _conv(a: np.ndarray) -> np.ndarray:
    """Return full convolution a*a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact score used by evaluate_sequence (scale‑invariant)."""
    n = a.shape[0]
    conv = _conv(a)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ------------------------------------------------------------
# Targeted mass transfer that tries to reduce the current peak
# ------------------------------------------------------------
def _targeted_move(vec: np.ndarray, rng: np.random.Generator, factor: float) -> None:
    """
    Reduce the contribution that creates the current maximum convolution value.
    The function mutates ``vec`` in place.
    """
    n = vec.shape[0]
    conv = _conv(vec)
    k = int(conv.argmax())                     # index of the largest convolution entry
    # overlapping indices that contribute to this entry
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    rng_idx = np.arange(start, end + 1, dtype=np.int64)
    partner = k - rng_idx
    # product of the two sides of each contributing pair
    prod = vec[rng_idx] * vec[partner]
    # pick the pair with the largest product
    pos = int(rng_idx[np.argmax(prod)])
    partner_pos = int(partner[np.argmax(prod)])

    # move a small fraction of mass from the chosen positions to random locations
    if vec[pos] > 0.0:
        delta = vec[pos] * factor
        vec[pos] -= delta
        target = rng.integers(0, n)
        while target == pos:
            target = rng.integers(0, n)
        vec[target] += delta

    if partner_pos != pos and vec[partner_pos] > 0.0:
        delta = vec[partner_pos] * factor
        vec[partner_pos] -= delta
        target = rng.integers(0, n)
        while target == partner_pos:
            target = rng.integers(0, n)
        vec[target] += delta

    # keep within the allowed numeric box and renormalise to keep the sum constant
    np.clip(vec, 0.0, 1000.0, out=vec)
    total = vec.sum()
    if total > 0.0:
        vec /= total


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 5000, **kwargs):
    """
    Return a list of non‑negative floats that tries to minimise the AC1 bound.
    The routine starts from a uniform vector, repeatedly applies a light
    targeted mass‑transfer heuristic and retains the best candidate seen.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem (height_sequence_1 is supplied by the harness)
    n = int(np.asarray(height_sequence_1, dtype=np.float64).shape[0])

    # ---- initial vector -------------------------------------------------
    # uniform distribution (sum = 1) – this gives score = 2.0 and is a valid fallback
    cur = np.full(n, 1.0 / n, dtype=np.float64)

    best = cur.copy()
    best_score = _score(best)

    # schedule for the amount of mass moved (large at the start, small later)
    for step in range(num_steps):
        # decay factor – between about 0.01 and 0.15
        factor = 0.12 * (1.0 - step / num_steps) + 0.01

        # produce a candidate by a few targeted moves
        cand = cur.copy()
        n_moves = 1 + int(rng.integers(3))        # 1–3 moves per iteration
        for _ in range(n_moves):
            _targeted_move(cand, rng, factor)

        cand_score = _score(cand)

        # keep the best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            cur = cand.copy()                      # continue search from improvement
        else:
            # occasional uphill move to escape shallow plateaux
            if rng.random() < 0.02:
                cur = cand.copy()

        # early exit if we have already crossed the target bound
        if best_score < 1.50274349:
            break

    # Return as plain python list of floats (the harness will re‑evaluate it)
    return [float(v) for v in best]
