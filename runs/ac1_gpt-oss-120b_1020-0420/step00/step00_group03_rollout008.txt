import numpy as np

def _get_parent():
    """Return a validated 1‑D non‑negative array to start from."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
            raise ValueError
        # Clip to the allowed range just in case
        return np.clip(arr, 0.0, 1000.0)
    except Exception:
        # Fallback: a modest length of ones
        return np.ones(256, dtype=np.float64)


def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as defined by `evaluate_sequence`.  For non‑negative
    sequences the bound is minimized by a uniform distribution of the total
    mass, which yields the theoretical optimum of 2.  The function therefore
    constructs a uniform candidate, compares it with the supplied parent, and
    returns the better one.
    """
    rng = np.random.default_rng(seed)  # kept for compatibility; not used

    parent = _get_parent()
    n = parent.size

    # Total mass of the parent (clipped); enforce a tiny positive mass if needed.
    total = float(np.sum(parent))
    if total < 0.01:
        total = 0.01

    # Uniform height respecting the per‑element cap of 1000.
    uniform_h = total / n
    if uniform_h > 1000.0:
        uniform_h = 1000.0
        total = uniform_h * n  # adjust total for consistency

    uniform_candidate = np.full(n, uniform_h, dtype=np.float64)

    # Evaluate both candidates safely.
    best = uniform_candidate
    best_score = float(evaluate_sequence(best.tolist()))

    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent
        best_score = parent_score

    # If for any unforeseen reason the best score is not finite, fall back to a safe constant.
    if not np.isfinite(best_score):
        safe_val = max(0.01 / n, 0.1)
        best = np.full(n, safe_val, dtype=np.float64)

    return [float(x) for x in best]
