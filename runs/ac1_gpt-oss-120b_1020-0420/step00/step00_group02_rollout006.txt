import numpy as np

def _block_perturbation(seq, rng):
    """Return a non‑negative O(n) perturbation of ``seq``."""
    n = seq.size
    # Choose a block width that is a small fraction of n
    min_w = max(1, n // 300)
    max_w = max(2, n // 20)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width

    # Apply a multiplicative log‑normal factor on the block
    factor = float(np.exp(rng.normal(0.0, 0.1)))
    new_seq = seq.copy()
    new_seq[start:stop] *= factor

    # Optional mild symmetrisation (helps in some cases)
    if rng.random() < 0.3:
        new_seq = 0.5 * (new_seq + new_seq[::-1])

    # Clamp to the admissible range
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the AC1 upper bound as measured by ``evaluate_sequence``.
    The routine keeps O(n) auxiliary memory and uses the provided
    ``height_sequence_1`` only to obtain the problem size.
    """
    rng = np.random.default_rng(seed)

    # Determine problem dimension n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256
        else:
            n = int(base.size)
    except NameError:
        n = 256

    # Uniform vector (theoretical optimum, objective = 2)
    best = np.full(n, 1.0, dtype=np.float64)
    np.clip(best, 0.0, 1000.0, out=best)

    # Evaluate the starting point
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:   # safety net – should never happen
        best_score = float('inf')

    # Simple stochastic local search: keep any improvement
    for _ in range(num_steps):
        cand = _block_perturbation(best, rng)
        # Quick sanity check before costly evaluation
        if np.isfinite(cand).all() and cand.sum() >= 0.01:
            try:
                score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score

    # Fallback: if for any reason the search produced an invalid result,
    # return the uniform vector (guaranteed to be valid).
    if not (np.isfinite(best).all() and best.sum() >= 0.01):
        best = np.full(n, 1.0, dtype=np.float64)

    return [float(x) for x in best]
