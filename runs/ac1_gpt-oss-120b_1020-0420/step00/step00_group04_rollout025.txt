def fast_score(arr):
    n = arr.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(arr, L)
    conv = np.fft.irfft(fft_a * fft_a, L)[:2*n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(arr)
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

# ---------- helper: FFT based full convolution (memory O(n)) ----------
def _full_convolution(arr: np.ndarray) -> np.ndarray:
    """Return np.convolve(arr, arr) using an FFT (size is a power of two)."""
    n = arr.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft = np.fft.rfft(arr, L)
    prod = fft * fft
    conv = np.fft.irfft(prod, L)
    return conv[: 2 * n - 1]                     # same shape as np.convolve


# ---------- main entry point ----------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative heights (a step‑function) that tries to
    minimise 2·n·max(conv(a,a)) / (sum(a)²).  The routine works without any
    external files and keeps memory usage linear in the sequence length.
    """
    rng = np.random.default_rng(seed)

    # ---- initialise from the supplied parent (if any) -----------------
    try:
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
    except Exception:               # height_sequence_1 not defined
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    if parent.sum() < 0.01:          # respect the evaluator's sum‑lower‑bound
        parent = np.full_like(parent, 0.01 / parent.size)

    n = parent.size
    cur_a = parent.copy()
    cur_sum = cur_a.sum()
    cur_conv = _full_convolution(cur_a)          # O(n log n)
    cur_score = 2.0 * n * cur_conv.max() / (cur_sum * cur_sum)

    # ---- parameters --------------------------------------------------
    num_steps = kwargs.get("num_steps", 20000)   # total iterations
    patience = kwargs.get("patience", 5000)      # stop after many failures
    no_improve = 0
    target = 1.50274349                         # early‑exit threshold

    # ---- main search loop --------------------------------------------
    for _ in range(num_steps):
        # Choose move type: 70 % swaps, 30 % block perturbations
        if rng.random() < 0.7:
            # -------------- swap two positions -------------------------
            i, j = rng.integers(0, n, size=2)
