def propose_candidate(seed=42, **kwargs):
    """
    Return a valid non‑negative step‑function sequence.
    The implementation chooses the uniform sequence, which is provably

    score, otherwise the uniform sequence is returned.
    """
    # Determine length from the supplied parent if available.
    if 'height_sequence_1' in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.shape[0])
    else:
        parent_arr = None
        n = 256  # default length when no parent is present

    # Uniform candidate (all ones). Any positive constant works; 1.0 keeps values small.
    uniform = np.full(n, 1.0, dtype=np.float64)

    # Evaluate uniform candidate.
    try:
        best_score = float(evaluate_sequence(uniform.tolist()))
        best_candidate = uniform

    # If a parent sequence exists, evaluate it and keep it if better.
    if parent_arr is not None:

        parent = np.clip(parent_arr, 0.0, 1000.0)
