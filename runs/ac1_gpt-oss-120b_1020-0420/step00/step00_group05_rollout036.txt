def ac1_score(seq):
    n = seq.size
    fft_len = 2*n - 1
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a <= 0:  # avoid div by zero
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
