import numpy as np

def _eval_score(seq, n, nfft):
    total = np.sum(seq)
    if total < 0.01:
        return np.inf
    # convolution using FFT
    fa = np.fft.rfft(seq, nfft)
    conv = np.fft.irfft(fa * fa, nfft)[:2*n-1]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (total * total))

import numpy as np

def _score_seq(seq: np.ndarray, n: int, nfft: int) -> float:
    """
    Compute the AC1 upper bound for a given sequence using FFT‑based convolution.
    This matches the logic of evaluate_sequence but runs in O(n log n) time.
    """
    total = np.sum(seq)
    if total < 0.01:
        return np.inf
    # convolution via FFT (real signals)
    fa = np.fft.rfft(seq, nfft)
    conv = np.fft.irfft(fa * fa, nfft)[: 2 * n - 1]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (total * total))


def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Memory‑bounded stochastic optimisation that starts from a simple exponential
    profile and performs random block‑perturbations together with occasional
    re‑initialisations.  The best candidate found is returned as a Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Obtain the parent construction if it exists; otherwise fall back.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (

        parent = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    # power‑of‑two length for FFT convolution (≥ 2n‑1)
    nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2.  Build a few simple initial candidates and keep the best.
    # ------------------------------------------------------------------
    init_seqs = []

    # uniform
    init_seqs.append(np.ones(n, dtype=np.float64))

    # several exponential families with random decay rates
    for _ in range(4):
        lam = rng.uniform(0.5, 2.5)          # λ in the analysis above
        t = lam / n
        r = 1.0 - t
        seq = r ** np.arange(n, dtype=np.float64)
        init_seqs.append(seq)

    best_seq = None
    best_score = np.inf

    for seq in init_seqs:
        seq = np.clip(seq, 0.0, 1000.0)
        sc = _score_seq(seq, n, nfft)
        if np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = sc

    # If for some reason no initial candidate succeeded, fall back on the parent.
    if best_seq is None:
        best_seq = parent.copy()
        best_score = _score_seq(best_seq, n, nfft)

    # ------------------------------------------------------------------
    # 3.  Stochastic hill‑climbing (simulated‑annealing flavour).
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score

    for step in range(num_steps):
        # temperature decays linearly; small probability of a fresh re‑init.
        temperature = 0.05 * (1.0 - step / num_steps) + 0.001

        if rng.random() < 0.15:
            # fresh exponential re‑initialisation with a new λ
            lam = rng.uniform(0.5, 2.5)
            t = lam / n
            r = 1.0 - t
            cand = r ** np.arange(n, dtype=np.float64)
        else:
            # block perturbation (similar to the previous solution)
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand = current_seq.copy()
            cand[start : start + width] *= factor
            cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _score_seq(cand, n, nfft)

        # deterministic improvement or Metropolis acceptance
        if cand_score < current_score:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
        else:

            if rng.random() < np.exp(-(cand_score - current_score) / temperature):

                current_score = cand_score

    # ------------------------------------------------------------------

    if best_seq is None or not np.isfinite(best_score):

        if fallback.size == 0:
            fallback = np.ones(256, dtype=np.float64)
        best_seq = fallback

    return [float(x) for x in best_seq]
