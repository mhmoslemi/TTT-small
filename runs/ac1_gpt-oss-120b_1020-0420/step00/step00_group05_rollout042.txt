    # Parameter grid for exponent p
    best_seq = None

    # we try p values from 0.1 to 5 step

        seq = np.power(x, p)
        # Clip to [0,1000] scaling
        max_val = seq.max()
        if max_val > 0:
            seq = seq * (1000.0 / max_val)
        else:
            seq = np.zeros_like(seq)
        # Ensure non-negative and within range
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
    # fallback: uniform if best_seq None
    if best_seq is None:
        best_seq = np.full(n, 1000.0, dtype=np.float64)
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    return best_seq.tolist()

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Search for a step‐function that gives a low AC1 bound.
    Explores simple parametric families (exponential, power) and
    random permutations of them.
    Returns the best found sequence as a Python list.
    """
    # Length of the sequence – use the provided parent if present.
    try:
        n = int(height_sequence_1.shape[0])   # provided by the harness
    except Exception:
        n = 4498

    rng = np.random.default_rng(seed)

    # Helper: evaluate and keep the best sequence seen so far.
    best_seq = None
    best_score = np.inf

    def consider(seq):
        nonlocal best_seq, best_score
        # Ensure proper type and bounds before evaluation.
        seq = np.clip(seq, 0.0, 1000.0)
        # Convert to Python list only once for the evaluator.
        try:
            score = float(evaluate_sequence(seq.tolist()))
        except Exception:
            score = np.inf
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
        return

    # ------------------------------------------------------------------
    # 1. Uniform (all entries at the maximum) – gives the baseline 2.0.
    uniform = np.full(n, 1000.0, dtype=np.float64)
    consider(uniform)

    # 2. Original parent (if it existed and is valid) – a possible fallback.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            consider(parent)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Parametric families
    #    a) Exponential decay:   a_i = exp(-p * i / n)
    #    b) Power decay:        a_i = ((n - i) / n) ** p   (decreasing)
    #    c) Power increase:      a_i = ((i + 1) / n) ** p   (increasing)
    # ------------------------------------------------------------------
    p_values = np.linspace(0.0, 5.0, 21)   # 0, 0.25, … ,5.0

    for p in p_values:
        # ---- exponential decay -------------------------------------------------
        expo = np.exp(-p * np.arange(n, dtype=np.float64) / n)
        if expo.max() > 0:
            expo = expo * (1000.0 / expo.max())
        consider(expo)

        # try a few random permutations of the same multiset
        for _ in range(3):
            perm = expo.copy()
            rng.shuffle(perm)
            consider(perm)

        # ---- power decay (descending) -----------------------------------------
        x_desc = (n - np.arange(n, dtype=np.float64)) / n          # 1 → 1/n
        pow_desc = np.power(x_desc, p, where=x_desc > 0, out=np.zeros_like(x_desc))
        if pow_desc.max() > 0:
            pow_desc = pow_desc * (1000.0 / pow_desc.max())
        consider(pow_desc)
        for _ in range(3):
            perm = pow_desc.copy()
            rng.shuffle(perm)
            consider(perm)

        # ---- power increase (ascending) ----------------------------------------
        x_asc = (np.arange(n, dtype=np.float64) + 1) / n            # 1/n → 1
        pow_asc = np.power(x_asc, p, where=x_asc > 0, out=np.zeros_like(x_asc))
        if pow_asc.max() > 0:
            pow_asc = pow_asc * (1000.0 / pow_asc.max())
        consider(pow_asc)
        for _ in range(3):
            perm = pow_asc.copy()
            rng.shuffle(perm)
            consider(perm)

    # ------------------------------------------------------------------
    # 4. If nothing succeeded (should not happen), fall back to uniform.
    if best_seq is None:
        best_seq = uniform

    # Return as a plain Python list of floats.
    return best_seq.tolist()
