# The evaluator `evaluate_sequence` is provided by the harness.

def propose_candidate(seed=42, **kwargs):
    """Return a simple uniform step‐function.

    # Determine the length to use: prefer the supplied parent length.
    n = 256  # default fallback length
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)

        pass

    candidate = np.full(n, 1.0, dtype=np.float64)
