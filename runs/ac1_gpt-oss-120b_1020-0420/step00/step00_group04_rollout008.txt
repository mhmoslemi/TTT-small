def evaluate_sequence(sequence: list) -> float:
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:

    return float(2 * n * max_b / (sum_a**2))

# `height_sequence_1` is available in the global namespace when a parent
# construction exists.  We fall back to a small default if it is missing.

def propose_candidate(seed=42, **kwargs):
    """
    Return a simple constant step‑function.
    The constant sequence attains the theoretical optimum (score = 2.0)
    for the supplied AC1 upper‑bound evaluator.
    """

        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        if n == 0:

    # Any positive constant works; we use 1.0 for simplicity.
    candidate = np.full(n, 1.0, dtype=np.float64)
