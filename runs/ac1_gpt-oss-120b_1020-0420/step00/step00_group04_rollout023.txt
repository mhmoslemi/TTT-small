import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound evaluated by ``evaluate_sequence``.
    The implementation falls back to a uniform candidate (score = 2.0) and
    optionally refines it with cheap random block perturbations.
    """
    # ------------------------------------------------------------------
    # Determine problem size from the provided parent construction.
    # ------------------------------------------------------------------
    n = 256                     # default length if parent unavailable
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = arr
            n = arr.shape[0]
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Uniform candidate (guaranteed score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0, dtype=np.float64)
    # Ensure the sum is not below the evaluator's threshold.
    if uniform.sum() < 0.01:
        uniform[0] = 0.01

    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Try the original parent as a fallback if it is better.
    # ------------------------------------------------------------------
    if parent is not None:
        try:
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent
                best_score = parent_score
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Light-weight random block perturbation search.
    # This cannot beat the theoretical lower bound (2.0) but may recover
    # a valid solution if the uniform candidate is somehow rejected.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for _ in range(200):
        cand = best.copy()
        # Choose a random block width proportional to the length.
        width = max(
            1,
            int(rng.integers(max(1, n // 200), max(2, n // 12)))
        )
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        # Multiplicative perturbation of the block.
        factor = np.exp(rng.normal(0.0, 0.12))
        cand[start:stop] *= factor
        cand = np.clip(cand, 0.0, 1000.0)
        # Optional symmetrisation.
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        # Evaluate and accept if improved.
        try:
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
        except Exception:
            continue

    # Return a plain Python list of floats as required by the harness.
    return [float(x) for x in best]
