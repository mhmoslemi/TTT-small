import numpy as np

def uniform_sequence(length, value=1.0):
    """Return a 1‑D NumPy array of the given length filled with `value`."""
    return np.full(int(length), float(value), dtype=np.float64)

def propose_candidate(seed=42, **kwargs):
    """
    Produce a non‑negative step‑function sequence that minimizes the
    evaluator's bound.  The theoretical optimum under the given metric is
    achieved by any uniform sequence, so we return a constant‑value vector.
    If the runtime provides `height_sequence_1`, its length is used;
    otherwise a fallback length of 256 is employed.
    """
    # Determine the desired length.
    try:
        # height_sequence_1 is supplied by the harness (if present).
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
            raise ValueError
        n = int(arr.size)
    except Exception:
        # Fallback length when the parent sequence is unavailable or invalid.
        n = 256

    # Uniform constant (any positive value works; 1.0 keeps numbers modest).
    candidate = uniform_sequence(n, 1.0)

    # Return a plain Python list of floats as required.
    return [float(x) for x in candidate]
