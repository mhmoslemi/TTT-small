import numpy as np

def geometric_initial(n):
    r = 1.0 - 1.0 / max(2, n)
    return np.clip(r ** np.arange(n, dtype=np.float64), 0.0, 1000.0)

def block_perturb(base, rng):
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    # multiplicative log-normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:start+width] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def gradient_update(a, rng, lr=0.1):
    n = a.size
    # convolution via FFT
    m = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2 * n - 1]
    k_max = int(np.argmax(conv))
    max_val = conv[k_max]
    S = a.sum()
    if S < 1e-12:
        S = 1e-12
    # gradient of max component
    idx = np.arange(n)
    j = k_max - idx
    valid = (j >= 0) & (j < n)
    gradM = np.zeros_like(a)
    gradM[valid] = 2.0 * a[j[valid]]
    C = 2.0 * n
    grad = C * (gradM / (S * S) - (2.0 * max_val) / (S ** 3))
    # simple step with clipping
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def propose_candidate(seed=42, num_iters=1200, **kwargs):
    rng = np.random.default_rng(seed)
    # starting vector: either the provided parent or a geometric shape
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = None
    if parent is None or parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 256
        a = np.ones(n, dtype=np.float64)
    else:
        n = parent.size
        a = np.clip(parent, 0.0, 1000.0)
    # replace with a deterministic geometric seed that is known to be good
    geom = geometric_initial(n)
    # evaluate both and keep the better
    score_parent = float(evaluate_sequence(a.tolist()))
    score_geom = float(evaluate_sequence(geom.tolist()))
    if np.isfinite(score_geom) and score_geom < score_parent:
        best = geom.copy()
        best_score = score_geom
    else:
        best = a.copy()
        best_score = score_parent
    # main improvement loop
    for it in range(num_iters):
        if it % 7 == 0:
            # occasional gradient based move
            cand = gradient_update(best, rng, lr=0.3)
        else:
            cand = block_perturb(best, rng)
        # occasional symmetrisation (helps flatten convolution)

            # adapt learning rate if we used gradient
            if it % 7 == 0:

        # early exit if target met

    # final safety: ensure we have a valid vector

import numpy as np

# ------------------------------------------------------------
# Utility: fast evaluation of the AC1 upper bound (same as evaluate_sequence)
# ------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Return the exact score using FFT convolution (no extra imports)."""
    n = seq.size
    if n == 0:
        return float('inf')
    s = float(seq.sum())
    if s < 0.01:
        return float('inf')
    # length of the linear convolution
    conv_len = 2 * n - 1
    # next power‑of‑two for efficient FFT
    fft_len = 1 << (conv_len - 1).bit_length()
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[:conv_len]
    max_b = float(conv.max())
    return float(2 * n * max_b / (s * s))


# ------------------------------------------------------------
# Simple block perturbation (keeps O(n) memory)
# ------------------------------------------------------------
def block_perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:start + width] *= factor
    if rng.random() < 0.35:                     # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ------------------------------------------------------------
# One gradient‑like step that tries to lower the maximal convolution entry
# ------------------------------------------------------------
def gradient_step(a: np.ndarray, lr: float = 0.02) -> np.ndarray:
    n = a.size
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    f = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[:conv_len]
    k_max = int(np.argmax(conv))

    # sub‑gradient of the maximal convolution term
    idx = np.arange(n)
    j = k_max - idx
    valid = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[valid] = 2.0 * a[j[valid]]                 # ∂c_kmax / ∂a_i

    # take a small step opposite the gradient
    a_new = a - lr * grad

    # renormalise to keep the overall scale (scale does not affect the score)
    tot = a.sum()
    new_tot = a_new.sum()
    if new_tot > 0:
        a_new *= tot / new_tot

    return np.clip(a_new, 0.0, 1000.0)


# ------------------------------------------------------------
# A deterministic geometric start (r ≈ 1‑1/n) – known to give ~1.84
# ------------------------------------------------------------
def geometric_start(n: int) -> np.ndarray:
    r = 1.0 - 1.0 / max(2, n)
    return np.clip(r ** np.arange(n, dtype=np.float64), 0.0, 1000.0)
