import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself."""
    return np.convolve(seq, seq)


def _top_shifts(conv: np.ndarray, k: int) -> np.ndarray:
    """Indices of the k largest convolution entries (unsorted)."""
    if k >= conv.size:
        return np.arange(conv.size, dtype=np.int64)
    idx = np.argpartition(conv, -k)[-k:]
    return idx.astype(np.int64)


def _build_constraint_matrix(f: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Build the (m x n) matrix A for constraints  (f * g)[shift] <= M.
    Row r corresponds to shift = shifts[r]; entry A[r, j] = f[i] where i+j=shift.
    """
    n = f.size
    m = shifts.size
    A = np.zeros((m, n), dtype=np.float64)
    for r, k in enumerate(shifts):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        j = np.arange(start, end + 1, dtype=np.int64)
        i = k - j
        A[r, j] = f[i]
    return A


def _lp_step(f: np.ndarray, shifts: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximize   sum(g)   s.t.   (f * g)[shift] <= M   (for all given shifts),
    g >= 0.  Returns the optimal g or None if the LP fails.
    """
    A = _build_constraint_matrix(f, shifts)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)          # maximize sum(g)  <=> minimize -sum(g)
    bounds = [(0.0, None)] * f.shape[0]                 # g_j >= 0
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small O(n) random perturbation used as a fallback local search."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, num_lp_iters: int = 8,
                     lp_top_k: int = 10, mix_factor: float = 0.5,
                     num_local_steps: int = 300, **kwargs) -> list:
    """
    Generate a non‑negative step‑function (list of heights) that tries to minimise
    the AC1 upper bound used by ``evaluate_sequence``.
    The algorithm starts from the supplied ``height_sequence_1`` (if present)
    and performs a few linear‑programming mixing steps followed by a modest
    local‑search refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the provided parent construction if possible
    # ------------------------------------------------------------------
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)  # a safe fallback

    f = parent.copy()
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = f.size

    # --------------------------------------------------------------
    # Linear‑programming based mixing phase
    # --------------------------------------------------------------
    for _ in range(num_lp_iters):
        conv_ff = _convolve(f)
        M = float(conv_ff.max())
        # Choose a few shifts with the largest values (active set)
        active_shifts = _top_shifts(conv_ff, lp_top_k)

        g = _lp_step(f, active_shifts, M)
        if g is None:
            break

        # Mix the old vector with the LP solution
        new = (1.0 - mix_factor) * f + mix_factor * g
        np.clip(new, 0.0, 1000.0, out=new)

        score = float(evaluate_sequence(new.tolist()))
        if np.isfinite(score) and score < best_score:
            f = new
            best = new
            best_score = score
        else:
            # no improvement – keep the current f
            pass

    # --------------------------------------------------------------
    # Simple local‑search fallback (block perturbations)
    # --------------------------------------------------------------
    for _ in range(num_local_steps):
        cand = _make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Ensure we return a plain Python list of floats
    return [float(x) for x in best]
