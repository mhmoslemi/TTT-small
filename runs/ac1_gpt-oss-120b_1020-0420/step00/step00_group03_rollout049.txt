import numpy as np

def propose_candidate(seed=42, num_trials=200, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises the
    AC1 upper bound defined by ``evaluate_sequence``.

    The theoretical minimum of the bound for non‑negative sequences is 2,
    attained by a perfectly uniform sequence.  We start from that optimum and
    perform a tiny stochastic refinement; if a refinement ever yields a lower
    score we keep it, otherwise the uniform sequence is returned.
    """
    rng = np.random.default_rng(seed)

    # Length of the ambient problem (provided as a global variable).
    n = len(height_sequence_1)

    # Uniform candidate – analytically optimal (bound = 2).
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # A very light stochastic search: perturb the uniform vector multiplicatively.
    # Any improvement (unlikely) is kept; otherwise the uniform vector is final.
    for _ in range(num_trials):
        # Multiplicative log‑normal perturbation keeps positivity.
        perturb = np.exp(rng.normal(loc=0.0, scale=0.15, size=n))
        cand = best * perturb
        # Enforce the hard limits of the evaluator.
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list of floats (the harness expects this format).
    return [float(x) for x in best]
