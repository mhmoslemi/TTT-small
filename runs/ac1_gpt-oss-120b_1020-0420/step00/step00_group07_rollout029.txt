    """Smallest power‑of‑two length sufficient for a linear convolution of size n."""
    return 1 << ((2 * n - 1).bit_length())

    """
    Compute the linear convolution v*v via FFT and return its maximum entry.
    The routine works in O(n log n) time and uses only O(fft_len) auxiliary memory.
    """

    conv = np.fft.irfft(B, fft_len)[: 2 * v.size - 1]
    # Numerical noise can produce tiny negative values – clip them safely.
    return float(np.max(conv))
