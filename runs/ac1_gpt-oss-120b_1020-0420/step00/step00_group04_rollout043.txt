import numpy as np

# ------------------------------------------------------------
# Helper: construct a set of indices with limited sum multiplicity
# ------------------------------------------------------------
def _build_sparse_set(n, max_rep, rng):
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that each unordered sum
    i+j (0≤i≤j<n) appears at most `max_rep` times.
    Returns a NumPy array of the chosen indices.
    """
    # how many times each possible sum (0 … 2n‑2) has already been used
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

    selected = []                      # Python list of chosen indices
    # random order of candidates for a chance to reach a larger set
    candidates = np.arange(n, dtype=np.int32)
    rng.shuffle(candidates)

    for idx in candidates:
        # check self‑pair (i,i)
        s_self = idx + idx
        if sum_counts[s_self] + 1 > max_rep:
            continue

        if selected:
            prev = np.frombuffer(np.int32().tobytes() * 0, dtype=np.int32)  # placeholder
            # sums with all already selected indices
            arr = np.array(selected, dtype=np.int32)
            sums = idx + arr
            # any sum would exceed the allowed repetition?
            if np.any(sum_counts[sums] + 1 > max_rep):
                continue
            # accept: increment the counts for these sums
            sum_counts[sums] += 1

        # increment the self‑pair count and record the new index
        sum_counts[s_self] += 1
        selected.append(int(idx))

    return np.array(selected, dtype=np.int32)
