import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Construct a simple two‑block step function.
    The first and last ceil(n/3) entries are set to 1.0,
    all others are 0.0.  This pattern yields
        score = n / (2 * L)  ≤ 1.5
    for L = ceil(n/3), which beats the target bound.
    """
    # Use the provided parent length if available.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = None

    if base is None or base.ndim != 1 or base.size == 0:
        # Fallback to a small default vector.
        n = 256
    else:
        n = base.size

    # Length of each block; ensure it does not exceed half the array.
    L = max(1, (n + 2) // 3)          # ceil(n/3)
    if L > n // 2:                    # safety clamp for very small n
        L = n // 2

    candidate = np.zeros(n, dtype=np.float64)
    candidate[:L] = 1.0
    candidate[-L:] = 1.0

    # Validate the candidate using the evaluator; if something goes wrong,
    # fall back to a clipped version of the parent or a simple uniform vector.
    score = evaluate_sequence(candidate.tolist())
    if not np.isfinite(score):
        if base is not None and base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            fallback = np.clip(base, 0.0, 1000.0)
        else:
            fallback = np.ones(256, dtype=np.float64)
        return fallback.tolist()

    return candidate.tolist()
