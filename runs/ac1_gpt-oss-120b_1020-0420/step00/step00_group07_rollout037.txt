import numpy as np

def _uniform_sequence(length, value=1.0):
    """Create a constant (step) sequence of given length."""
    return np.full(length, float(value), dtype=np.float64)

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound evaluated by ``evaluate_sequence``.
    The algorithm starts from the uniform sequence (the theoretical optimum)
    and performs a tiny stochastic local search, keeping the best result.
    """
    rng = np.random.default_rng(seed)

    # Determine the required length n.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256                       # fallback length if the parent is missing

    # Start from the uniform sequence (optimal w.r.t the bound).
    best = _uniform_sequence(n, value=1.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # Simple block‑perturbation search; uniform remains optimal, but the loop
    # guarantees a valid fallback even if the initial guess were bad.
    for _ in range(num_steps):
        cand = best.copy()

        # Randomly pick a contiguous block and scale its heights.
        start = int(rng.integers(0, n))
        max_width = max(2, n // 10)
        width = int(rng.integers(1, max_width + 1))
        stop = min(n, start + width)
        factor = float(np.exp(rng.normal(0.0, 0.10)))
        cand[start:stop] *= factor

        # Occasionally enforce approximate symmetry (helps when the parent
        # has some structure, but harmless for a uniform start).
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])

        # Apply evaluator limits.
        candidate = np.clip(cand, 0.0, 1000.0)

        # Evaluate and keep the improvement.
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    # Return a plain Python list as required.
    return best.tolist()
