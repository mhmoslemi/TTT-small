def propose_candidate(seed=42, num_steps=5000, **kwargs):
    rng = np.random.default_rng(seed)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size
    m = 1 << ((2*n-1).bit_length())

    def get_max_shift_and_val(arr):
        fa = np.fft.fft(arr, m)
        conv = np.fft.ifft(fa * fa).real[:2*n-1]
        idx = int(np.argmax(conv))
        val = float(conv[idx])
        return idx, val

    # local search
    steps_without_improve = 0
    for step in range(num_steps):
        # compute max shift for best (or compute for random candidate)
        k_max, max_val = get_max_shift_and_val(best)
        # define overlapping region
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        # random reduction factor
        factor = float(rng.uniform(0.5, 0.95))
        candidate = best.copy()
        candidate[i_start:i_end+1] *= factor
        # optionally also add small random noise to other entries
        if rng.random() < 0.3:
            noise = rng.normal(0.0, 0.05, size=n)
            cand = candidate + noise
            candidate = np.clip(cand, 0.0, 1000.0)
        else:
            candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            steps_without_improve = 0
        else:
            steps_without_improve += 1
        # occasional random restart if stuck
        if steps_without_improve >= 200:
            # new random start: exponential shape with random c

            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score
            steps_without_improve = 0

def generate_sidon_set(n, target_m, rng):
    positions = []
    sums = set()
    attempts = 0
    max_attempts = target_m * 1000

    while len(positions) < target_m and attempts < max_attempts:
        p = int(rng.integers(0, n))
        # compute sums if added
        conflict = False
        new_sums = []
        for q in positions:
            s = p + q
            if s in sums:
                conflict = True
                break
            new_sums.append(s)
        # also check self sum
        s_self = p + p
        if s_self in sums:
            conflict = True
        if not conflict:
            positions.append(p)
            sums.add(s_self)
            for s in new_sums:
                sums.add(s)
        attempts += 1

    return positions

import numpy as np

def is_prime(num):
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num**0.5)
    for i in range(3, r+1, 2):
        if num % i == 0:
            return False
    return True

def largest_prime_leq(limit):
    for p in range(limit, 1, -1):
        if is_prime(p):
            return p
    return 2

def generate_sidon(p):
    positions = [i * p + (i * i) % p for i in range(p)]
    return positions

import numpy as np

def _is_prime(num: int) -> bool:
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num ** 0.5)
    for k in range(3, r + 1, 2):
        if num % k == 0:
            return False
    return True


def _largest_prime_leq(limit: int) -> int:
    for p in range(limit, 1, -1):
        if _is_prime(p):
            return p
    return 2  # fallback, should never happen for reasonable limits


def _sidon_positions(p: int) -> list:
    """Singer construction of a Sidon set of size p (p must be prime)."""
    return [i * p + (i * i) % p for i in range(p)]


def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a non‑negative step‑function that beats the uniform 2.0 bound.
    The construction mixes a tiny uniform background with a deterministic
    Sidon set of spikes, guaranteeing that each convolution lag receives
    at most one spike‑spike product.
    """
    rng = np.random.default_rng(seed)

    # base parent (fallback) – already available as height_sequence_1
    parent = np.clip(
        np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
    )
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    # tiny uniform background: epsilon = 1/√n gives baseline max_b ≈ 1
    eps = 1.0 / np.sqrt(float(n))
    cand = np.full(n, eps, dtype=np.float64)

    # Sidon set of spikes
    p = _largest_prime_leq(int(np.sqrt(n)))
    positions = _sidon_positions(p)
    # keep only positions that fit into the current length
    positions = [pos for pos in positions if pos < n]

    # set spike value to 1 (any positive value works, 1 keeps max_b small)
    for idx in positions:
        cand[idx] = 1.0

    # optional tiny random jitter (keeps the algorithm nondeterministic)
    if rng.random() < 0.3:
        jitter = rng.normal(0.0, eps * 0.1, size=n)
        cand = np.clip(cand + jitter, 0.0, 1000.0)

    # evaluate both candidate and parent; keep the better one
    try:
        cand_score = float(evaluate_sequence(cand.tolist()))
    except Exception:
        cand_score = np.inf

    try:
        parent_score = float(evaluate_sequence(parent.tolist()))
    except Exception:
        parent_score = np.inf

    best = cand if np.isfinite(cand_score) and cand_score < parent_score else parent

    return [float(x) for x in best]
