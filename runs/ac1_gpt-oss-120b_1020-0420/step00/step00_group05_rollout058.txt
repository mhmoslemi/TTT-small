import numpy as np

def _generate_sidon_set(n):
    """Greedy construction of a Sidon set (all pairwise sums distinct) in {0,…,n‑1}."""
    selected = []
    sums = set()
    for x in range(n):
        # quick reject if 2*x already used
        if 2 * x in sums:
            continue
        conflict = False
        for y in selected:
            if x + y in sums:
                conflict = True
                break
        if conflict:
            continue
        # accept x
        for y in selected:
            sums.add(x + y)
        sums.add(2 * x)
        selected.append(x)
    return selected

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the 2.0 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Length of the sequence – use the provided parent if it exists.
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256  # a safe default when the parent is missing

    # ------------------------------------------------------------------
    # Build a large Sidon set inside {0,…,n‑1}. Its size k governs the
    # theoretical lower bound 2‑k²/(2n).  For n≈4500 we typically obtain
    # k≈⌊√n⌋≈67, giving a bound close to 1.50.
    # ------------------------------------------------------------------
    sidon = _generate_sidon_set(n)
    k = len(sidon)

    # If the greedy construction failed to find any element (should not happen)
    # fall back to a constant sequence.
    if k == 0:
        best_seq = np.ones(n, dtype=np.float64)
        return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Choose heights.
    #   – a spike height H (any positive number, we use H=1).
    #   – a small background epsilon that is (close to) the optimal value:
    #       epsilon = H * (2 n - k²) / (k n)
    #   This choice minimizes the theoretical bound 2 - k²/(2n).
    # ------------------------------------------------------------------
    H = 1.0
    eps = max(1e-4, (2.0 * n - k * k) / (k * n) * H)

    seq = np.full(n, eps, dtype=np.float64)
    seq[sidon] = H
    seq = np.clip(seq, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Keep the best sequence seen so far.
    # ------------------------------------------------------------------
    best_seq = seq
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = best_seq.copy()

        # Random block scaling (mild perturbation)
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.08)))

        if rng.random() < 0.2:

        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    return [float(x) for x in best_seq]
