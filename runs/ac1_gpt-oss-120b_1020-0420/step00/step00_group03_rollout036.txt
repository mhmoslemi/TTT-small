def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]

def improve_candidate(a, steps=200, rng=None):
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    for _ in range(steps):
        # propose random swap between a high (large) index and low (small) index
        idx_large = rng.choice(np.where(best > np.mean(best))[0])
        idx_small = rng.choice(np.where(best < np.mean(best))[0])
        delta = min(best[idx_large], 0.1 * best[idx_large])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[idx_large] -= delta
        cand[idx_small] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            if best_score < 1.5:
                break
    return best, best_score

import math
import numpy as np

# evaluate_sequence is provided by the harness.


def _largest_prime_leq(m: int) -> int:
    """Return the largest prime number <= m (m >= 2)."""
    if m < 2:
        return 2
    for cand in range(m, 1, -1):
        if cand % 2 == 0:
            if cand == 2:
                return 2
            continue
        limit = int(math.isqrt(cand))
        is_prime = True
        for d in range(3, limit + 1, 2):
            if cand % d == 0:
                is_prime = False
                break
        if is_prime:
            return cand
    return 2


def _sidon_set(p: int) -> list:
    """
    Singer construction of a Sidon (B2) set of size p.
    All elements are < p**2, hence fit into any array of length >= p**2.
    """
    return [a + p * ((a * a) % p) for a in range(p)]

        j = rng.choice(small_idxs)

        # move a small amount of mass from i to j
        delta = min(best_vec[i], 0.08 * best_vec[i])  # at most 8% of the large entry
        if delta <= 0:
            continue
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        cur_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best_vec, best_score = cand, cur_score
            # early exit if we already beat the target comfortably
            if best_score < 1.45:
                break

    return [float(x) for x in best_vec]
