def propose_candidate(seed=42, attempts=20, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)
    # Determine n
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256
    # Compute upper bound on t (theoretical maximum)
    t_max = int((np.sqrt(16 * n - 7) - 1) // 2)
    best_set = []
    best_len = 0
    for _ in range(attempts):
        order = rng.permutation(n)
        used = np.zeros(2 * n - 1, dtype=bool)
        selected = []
        for idx in order:
            # Quick reject if self sum already used (should never happen if unique)
            s = idx + idx
            if used[s]:
                continue
            # Check collisions
            ok = True
            for j in selected:
                if used[idx + j]:
                    ok = False
                    break
            if not ok:
                continue
            # Accept
            selected.append(idx)
            # Mark all new sums
            for j in selected:
                used[idx + j] = True
            # (optional) check early termination
            if len(selected) == t_max:
                break
        if len(selected) > best_len:
            best_len = len(selected)
            best_set = selected
            if best_len == t_max:
                break
    # Build sequence
    seq = np.zeros(n, dtype=np.float64)
    if best_len == 0:
        # fallback uniform
        seq[:] = 1.0
    else:
        seq[best_set] = 1.0
    # Ensure we stay within bounds
    seq = np.clip(seq, 0.0, 1000.0)
    return [float(x) for x in seq]

import numpy as np

def propose_candidate(seed=42, attempts=20, **kwargs):
    """
    Generate a non‑negative step‑function that minimizes the AC1 upper bound.
    The construction builds a large Sidon set (all pairwise sums are distinct)
    and places unit height on those indices.  For a Sidon set of size t the
    convolution peak is 2 and the score equals 4·n / t², which can be well
    below the target bound when t ≈ √(4n).
    """
    rng = np.random.default_rng(seed)

    # length of the problem instance (provided as height_sequence_1)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256  # fallback length if the variable is missing

    # theoretical maximum size of a Sidon set in {0,…,n‑1}
    t_max = int((np.sqrt(16 * n - 7) - 1) // 2)

    best_set = []
    best_len = 0

    # try several random orderings and keep the largest Sidon set found
    for _ in range(attempts):
        order = rng.permutation(n)
        used = np.zeros(2 * n - 1, dtype=bool)   # marks occupied sums i+j
        cur = []                                 # current Sidon set

        for i in order:
            s = i + i
            if used[s]:
                continue
            conflict = False
            for j in cur:
                if used[i + j]:
                    conflict = True
                    break
            if conflict:
                continue

            # accept i
            cur.append(i)
            # mark all new sums (including the self‑sum)
            for j in cur:
                used[i + j] = True

            if len(cur) == t_max:   # reached the theoretical optimum
                break

        if len(cur) > best_len:
            best_len = len(cur)
            best_set = cur
            if best_len == t_max:
                break

    # Build the final sequence: unit height on the Sidon indices
    seq = np.zeros(n, dtype=np.float64)
    if best_len == 0:                # very unlikely, but keep a safe fallback
        seq[:] = 1.0
    else:
        seq[best_set] = 1.0
    seq = np.clip(seq, 0.0, 1000.0)

    return [float(x) for x in seq]
