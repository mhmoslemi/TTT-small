import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Construct a step‑function whose AC1 upper bound is well below 2.
    The construction is based on the sequence a_i = 1/sqrt(i+1) which
    gives a nearly flat self‑convolution, then a small greedy refinement
    adds a few low amplitudes at the tail to increase the total sum without
    raising the maximal convolution entry.
    """
    # Length of the problem (use the provided parent if present)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # ---- 1. Base sequence a_i = 1 / sqrt(i+1)  -------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    base = 1.0 / np.sqrt(idx)                     # values in (0,1]

    # Evaluate the starting score (for fallback handling)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Target that we try to beat
    TARGET = 1.50274349

    # ---- 2. Greedy refinement: add a small constant at a few positions ----
    #   Adding at the far right keeps the extra products tiny (a0*add ≤ add).
    #   Choose a value safely below sqrt(pi/2) ≈ 1.25 so that even the
    #   product of two added entries (2*add**2) does not exceed the original
    #   maximal convolution (~π).
    ADD_VAL = 1.2                                 # safe constant to add
    # indices ordered from the end (where the original entries are smallest)
    order = np.arange(n - 1, -1, -1, dtype=int)

    # Greedy pass – stop once we are under the target
    for pos in order:
        if best_score <= TARGET:
            break
        if best[pos] >= ADD_VAL:                 # nothing to improve here
            continue
        cand = best.copy()
        cand[pos] = ADD_VAL
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ---- 3. If still above the target, try a slightly larger addition -------
    if best_score > TARGET:
        for add_val in (1.5, 2.0):
            for pos in order:
                if best_score <= TARGET:
                    break
                if best[pos] >= add_val:
                    continue
                cand = best.copy()
                cand[pos] = add_val
                sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc
            if best_score <= TARGET:
                break

    # ---- 4. Final safety fallback -----------------------------------------
    # In the unlikely event that everything failed, use a simple uniform vector.
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best = np.clip(best, 0.0, 1000.0)

    # Return as a plain Python list of floats
    return [float(x) for x in best]
