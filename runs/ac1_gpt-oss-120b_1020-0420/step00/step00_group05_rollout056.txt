import math
import numpy as np

# ------------------------------------------------------------
# Helper: generate a mirror‑free random candidate.
# ------------------------------------------------------------
def _random_mirror_free(rng):
    """
    Return a candidate array of length len(height_sequence_1) with
    exactly one entry from each mirror pair (i, n-1-i) set to a constant
    amplitude a, the rest zero.  The amplitude a is chosen so that the
    L1‑norm equals sqrt(2*n) (the normalization used in the AC1 bound).
    """
    n = int(height_sequence_1.shape[0])
    half = n // 2                     # number of mirror pairs
    # amplitude forcing sum = sqrt(2*n)
    a = math.sqrt(2.0 * n) / half

    cand = np.zeros(n, dtype=np.float64)

    # pick uniformly which side of each pair receives the mass
    bits = rng.integers(0, 2, size=half)          # 0 → left, 1 → right
    left_idx = np.arange(half)
    right_idx = n - 1 - left_idx

    cand[left_idx] = np.where(bits == 0, a, 0.0)
    cand[right_idx] = np.where(bits == 1, a, 0.0)

    # If n is odd we leave the centre element zero (it would add a square term).
    return cand


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed=42, num_attempts=12, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose
    AC1 upper bound (evaluate_sequence) is as low as possible.
    The routine tries a few random mirror‑free constructions and keeps
    the best one; a uniform‑ones fallback guarantees a valid return.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Determine the length n from the provided parent (if any) or use 256.
    # -----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    n = parent.shape[0] if parent is not None else 256

    # -----------------------------------------------------------------
    # Fallback candidate – uniform ones (bound = 2).  This guarantees
    # a non‑empty, finite answer should the random search fail.
    # -----------------------------------------------------------------
    fallback = np.ones(n, dtype=np.float64)
    best_seq = fallback
    best_score = float(evaluate_sequence(fallback.tolist()))

    # -----------------------------------------------------------------
    # Random mirror‑free search.
    # -----------------------------------------------------------------
    for _ in range(num_attempts):
        cand = _random_mirror_free(rng)

        # Ensure the candidate has the correct length (in case height_sequence_1
        # was not defined and n differs from the internal construction).
        if cand.shape[0] != n:
            # Pad or truncate to length n
            if cand.shape[0] < n:
                pad = np.zeros(n - cand.shape[0], dtype=np.float64)
                cand = np.concatenate([cand, pad])
            else:
                cand = cand[:n]

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            # Early exit once we are clearly below the target.
            if best_score < 1.50274349:
                break

    # Return plain Python floats as required.
    return [float(v) for v in best_seq]
