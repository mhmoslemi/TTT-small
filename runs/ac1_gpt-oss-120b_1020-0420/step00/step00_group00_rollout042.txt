import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a height sequence that aims to lower the AC1 upper bound.
    The construction builds a (approximate) Sidon set of positions that receive
    the maximal allowed height (1000) and fills the remaining slots with a
    constant small value.  Several small values are tried and the best one
    (according to the supplied evaluate_sequence) is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the sequence (use height_sequence_1 if present)
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:                     # fallback length
        n = 256

    # ------------------------------------------------------------------
    # Generate a Sidon‑like set of indices.
    # For a Sidon set all sums i+j are distinct; this keeps the convolution
    # peak low.  We perform several random orderings and keep the largest set.
    # ------------------------------------------------------------------
    best_positions = []
    for _ in range(30):                    # few attempts – cheap and effective
        order = rng.permutation(n)
        pos = []
        sums = set()
        for i in order:
            # reject i if any i+existing position would repeat a previous sum
            conflict = False
            for p in pos:
                if i + p in sums:
                    conflict = True
                    break
            if conflict or (2 * i) in sums:
                continue
            # accept i
            for p in pos:
                sums.add(i + p)            # unordered pair i+p
            sums.add(2 * i)                # the diagonal pair i+i
            pos.append(i)
        if len(pos) > len(best_positions):
            best_positions = pos

    # ------------------------------------------------------------------
    # Candidate construction
    # ------------------------------------------------------------------
    large_val = 1000.0                     # maximal height for “big” entries
    small_vals = [10, 20, 30, 40, 50, 60, 80, 100, 150, 200, 250]
    best_score = float('inf')
    best_seq = None

    # Try each small value
    for sv in small_vals:
        seq = np.full(n, float(sv), dtype=np.float64)
        if best_positions:
            seq[best_positions] = large_val
        seq = np.clip(seq, 0.0, 1000.0)    # safety clamp
        try:
            score = float(evaluate_sequence(seq.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # Uniform baseline (value 1 everywhere → score exactly 2)
    uniform_seq = np.full(n, 1.0, dtype=np.float64)
    try:
        score = float(evaluate_sequence(uniform_seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = uniform_seq
    except Exception:
        pass

    # Parent fallback – keep it if it happens to be better
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).clip(0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = parent
    except Exception:
        pass

    # Absolute fallback (should never be needed)
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)

    # Return as plain Python list of floats
    return [float(x) for x in best_seq]
