        b = np.convolve(best, best)
        k = int(np.argmax(b))

        i_start = max(0, k - (n-1))

        i_max = idx_i[np.argmax(contrib)]

import numpy as np

# convolution of a vector with itself (matrix‑free, O(n log n))
def convolve_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    conv_full = np.fft.irfft(fx * fx, n=size)
    return conv_full[: 2 * n - 1]                    # length 2n‑1


# compute the AC1 upper bound without calling the external evaluator
def ac1_score(x: np.ndarray) -> float:
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float("inf")
    n = x.shape[0]
    mx = convolve_self(x).max()
    return float(2.0 * n * mx / (s * s))


def propose_candidate(seed: int = 42, num_iters: int = 400, **kwargs):
    """Search for a step‑function with a low AC1 bound.

    The routine starts from the supplied `height_sequence_1` (if present)
    or from a uniform vector.  It repeatedly identifies the convolution
    index with maximal value and transfers a fraction of the two most
    responsible coefficients to a random location, accepting the move
    only when it improves the bound.  Small random perturbations and
    occasional symmetrisation are also tried.
    """
    rng = np.random.default_rng(seed)

    # ---- initial candidate -------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        init = np.ones(256, dtype=np.float64)

    if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
        init = np.ones(256, dtype=np.float64)

    cand = np.clip(init, 0.0, 1000.0)
    n = cand.size

    # also keep a uniform fallback (might be better than the provided one)
    uniform = np.full(n, 1.0, dtype=np.float64)
    if ac1_score(uniform) < ac1_score(cand):
        cand = uniform.copy()

    best = cand.copy()
    best_score = ac1_score(best)

    # ---- main optimisation loop -------------------------------------------
    gamma0 = 0.12                     # initial transfer fraction
    for it in range(num_iters):
        # 1) locate the index with maximal self‑convolution
        conv = convolve_self(best)
        k = int(np.argmax(conv))

        # 2) find the pair (i , j) within this index that contributes most
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        prod = best[i_range] * best[j_range]
        max_pos = int(np.argmax(prod))
        i_max = i_range[max_pos]
        j_max = j_range[max_pos]

        # 3) transfer a fraction of the two coefficients to a random slot
        r = rng.integers(0, n)
        while r == i_max or r == j_max:
            r = rng.integers(0, n)

        gamma = gamma0 * (0.99 ** it)               # cooling schedule
        delta_i = best[i_max] * gamma
        delta_j = best[j_max] * gamma
        # protect from negativity
        delta_i = min(delta_i, best[i_max])
        delta_j = min(delta_j, best[j_max])

        cand = best.copy()
        cand[i_max] -= delta_i
        cand[j_max] -= delta_j
        cand[r] += delta_i + delta_j
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = ac1_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            continue

        # 4) occasional tiny random multiplicative perturbation
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.12))
            cand2 = best.copy()
            cand2[idx] = np.clip(cand2[idx] * factor, 0.0, 1000.0)
            sc2 = ac1_score(cand2)
            if np.isfinite(sc2) and sc2 < best_score:

        if rng.random() < 0.02:

            sc3 = ac1_score(cand3)
            if np.isfinite(sc3) and sc3 < best_score:
                best, best_score = cand3, sc3

    # -----------------------------------------------------------------------
    return [float(v) for v in best]
