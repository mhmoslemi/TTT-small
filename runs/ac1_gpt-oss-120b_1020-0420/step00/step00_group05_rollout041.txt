def build_support(n, target_m=200, max_unordered=2, max_attempts=100000):
    rng = np.random.default_rng(seed)
    support = []
    sum_counts = {}
    attempts = 0
    while len(support) < target_m and attempts < max_attempts:
        idx = rng.integers(0, n)
        if idx in support:
            attempts +=1
            continue
        # compute sums with existing support
        ok = True
        # check self sum
        s_self = idx*2
        c_self = sum_counts.get(s_self, 0)
        if c_self >= max_unordered:
            ok=False
        else:
            # check sums with each existing element
            for j in support:
                s = idx + j
                if sum_counts.get(s, 0) >= max_unordered:
                    ok=False
                    break
        if ok:
            # add idx
            support.append(idx)
            # update sum_counts for self
            sum_counts[s_self] = sum_counts.get(s_self,0) + 1
            for j in support[:-1]:
                s = idx + j
                sum_counts[s] = sum_counts.get(s,0) + 1
        attempts +=1
    return support

import numpy as np

# Target score we want to beat
TARGET_SCORE = 1.50274349

def _can_add(idx, support, sum_counts, max_unordered):
    """Check whether a new index can be added without breaking the
    unordered‑pair limit."""
    if sum_counts[idx * 2] >= max_unordered:
        return False
    if support:
        # sums with already chosen indices
        s = idx + np.array(support, dtype=np.int32)
        if np.any(sum_counts[s] >= max_unordered):
            return False
    return True


def _add_index(idx, support, sum_counts, used):
    """Insert idx into the support and update the sum‑count bookkeeping."""
    sum_counts[idx * 2] += 1          # the self‑pair (idx,idx)
    if support:
        s = idx + np.array(support, dtype=np.int32)
        sum_counts[s] += 1            # each unordered pair (idx, j)
    support.append(int(idx))
    used[idx] = True


def _build_binary_candidate(rng, n, max_unordered):
    """
    Greedy construction of a binary (0/1) sequence of length n.
    The set of positions with value 1 respects the constraint that each
    unordered pair (i,j) with i<=j contributes to a sum i+j at most
    max_unordered times.
    """
    sum_counts = np.zeros(2 * n + 1, dtype=np.int16)   # counts for each possible sum
    used = np.zeros(n, dtype=bool)                    # which positions are already taken
    support = []                                       # list of chosen indices

    # First pass: sweep a random permutation
    perm = rng.permutation(n)
    for idx in perm:
        if not used[idx] and _can_add(idx, support, sum_counts, max_unordered):
            _add_index(idx, support, sum_counts, used)

    # Second pass: keep trying random indices for a while
    attempts = 0
    max_extra = n * 5   # a few extra trials per element
    while attempts < max_extra:
        idx = rng.integers(0, n)
        if not used[idx] and _can_add(idx, support, sum_counts, max_unordered):
            _add_index(idx, support, sum_counts, used)
        attempts += 1

    # Build the actual sequence (values are 0.0 or 1.0)
    seq = np.zeros(n, dtype=np.float64)
    seq[support] = 1.0
    return seq


def propose_candidate(seed=42, **kwargs):
    """
    Returns a list of non‑negative coefficients whose AC1 upper bound
    (as computed by the provided evaluate_sequence) is as small as possible.
    The algorithm builds a large binary support set with a bounded
    number of representations for each sum, which keeps the self‑convolution
    peak low while the total ℓ1‑norm stays high.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem instance (height_sequence_1 is guaranteed to exist)
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256  # fallback length

    best_seq = None
    best_score = float('inf')

    # Try a few different limits on unordered‑pair multiplicities;
    # smaller limits give a smaller convolution peak.
    max_unordered_options = [2, 3]       # 2 → ordered peak ≤4, 3 → ≤6
    trials_per_option = 250

    for max_unordered in max_unordered_options:
        for _ in range(trials_per_option):
            candidate = _build_binary_candidate(rng, n, max_unordered)
            # Evaluate the exact score
            try:
                score = float(evaluate_sequence(candidate.tolist()))
            except Exception:

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = candidate.copy()
                # Early exit if we already beat the target bound
                if best_score <= TARGET_SCORE:
                    break
        if best_score <= TARGET_SCORE:

    # Fallback: if nothing reasonable was found, use the parent or a uniform vector

            best_seq = parent
        except Exception:
            best_seq = np.ones(n, dtype=np.float64)

    # Ensure the returned list consists of plain Python floats
    return [float(x) for x in best_seq]
