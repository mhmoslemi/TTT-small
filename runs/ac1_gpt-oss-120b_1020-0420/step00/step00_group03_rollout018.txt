import numpy as np

# The evaluator is provided in the execution environment.
# It is used directly; do not redeclare it.


def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound.

    The strategy:
      1. Use a family of smooth, analytically‑motivated sequences
         a_i = ((i+0.5)*(n-i-0.5))**(-alpha)  (discrete arcsine shape).
      2. Scan a few values of ``alpha`` and keep the best one.
      3. Perform a short, memory‑bounded local search (block perturbations)
         around the best candidate to squeeze any remaining improvement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # 1️⃣  Initialise from the supplied parent (fallback to a simple vector)
    # ------------------------------------------------------------------ #
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:                     # pragma: no cover
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    n = best.size
    # ------------------------------------------------------------------ #
    # 2️⃣  Scan the “arcsine” family a_i = ((i+0.5)*(n-i-0.5))**(-alpha)
    # ------------------------------------------------------------------ #
    idx = np.arange(n, dtype=np.float64) + 0.5          # i+0.5
    factor = idx * (n - idx)                           # (i+0.5)*(n-i-0.5)

    # a few alphas spaced linearly – they are cheap to evaluate
    alphas = np.linspace(0.25, 0.85, 13)

    for alpha in alphas:
        cand = factor ** (-alpha)               # shape described above
        cand = np.clip(cand, 0.0, 1000.0)       # respect evaluator limits
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------ #
    # 3️⃣  Light local search – block perturbations (same idea as the baseline)
    # ------------------------------------------------------------------ #
    def make_block_perturbation(base):
        """Return a cheap O(n) non‑negative perturbation of ``base``."""
        candidate = base.copy()
        # width of the perturbed block – proportional to n but never tiny
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        # multiplicative log‑normal tweak
        candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        # occasional mirroring to keep symmetry in the search space
        if rng.random() < 0.35:
            candidate = 0.5 * (candidate + candidate[::-1])
        return np.clip(candidate, 0.0, 1000.0)

    # a modest number of random steps – each call evaluates the exact bound
    for _ in range(100):
        cand = make_block_perturbation(best)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # Return as a plain Python list of floats (the harness expects this)
    return [float(v) for v in best]
