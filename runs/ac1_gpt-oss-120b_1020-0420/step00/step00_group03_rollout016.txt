def propose_candidate(seed=42, num_steps=5000, **kwargs):

    if not np.isfinite(best_score):
        a = np.ones(256, dtype=np.float64)
        best = a.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    n = a.size

    # Main loop
    for step in range(num_steps):
        # compute convolution for gradient
        conv = np.convolve(a, a)
        maxb = conv.max()
        k_max = int(conv.argmax())
        sum_a = a.sum()
        if sum_a < 0.01:
            break
        # gradient g
        g = np.zeros_like(a)
        # indices where k_max - i in [0, n-1]
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        # vectorized:
        idx = np.arange(i_min, i_max + 1, dtype=int)
        other = k_max - idx
        g[idx] = a[other]
        # double for i where idx == other
        if k_max % 2 == 0:
            mid = k_max // 2
            if 0 <= mid < n:
                g[mid] = 2.0 * a[mid]
        # threshold
        t = 2.0 * maxb / sum_a
        # find donors and sinks
        donors = np.where((g > t) & (a > 0.0))[0]
        sinks = np.where(g < t)[0]
        if donors.size == 0 or sinks.size == 0:
            # random perturbation to escape
            # Choose random direction: pick random index i, add small random jitter
            i = rng.integers(0, n)
            delta = a[i] * 0.1 * rng.standard_normal()
            a[i] = max(0.0, a[i] + delta)
            a = np.clip(a, 0.0, 1000.0)
            continue
        i_src = donors[np.argmax(g[donors] - t)]
        i_dst = sinks[np.argmin(g[sinks] - t)]
        # compute delta
        # Use fraction of donor
        frac = 0.3  # up to 30% transfer
        delta = min(a[i_src] * frac, a[i_src])  # ensure non-negative
        if delta <= 0:
            # fallback random
            i = rng.integers(0, n)
            delta = max(0.0, a[i] * 0.1)
            a[i] = max(0.0, a[i] - delta)
            continue
        a[i_src] -= delta
        a[i_dst] += delta
        # Clip again
        a = np.clip(a, 0.0, 1000.0)

        # Evaluate new candidate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
        # Optionally early stop if as low as target
        if best_score < 1.50274349:
            break

    return [float(x) for x in best]
