    x = 0

def _linear_decreasing(length: int) -> np.ndarray:
    """Return a non‑negative linear decreasing step function of given length.
    The first element is 1000, the last ≈ 1000/length, all values ≤1000."""
    # values proportional to (length - i)
    arr = np.arange(length, 0, -1, dtype=np.float64)  # length, length-1, ..., 1
    arr *= 1000.0 / float(length)
    return arr
