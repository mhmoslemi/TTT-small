def perturb(candidate, rng):
    if rng.random() < 0.5:
        # block scaling
        n = candidate.size
        width = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, max(1, n - width + 1))
        stop = start + width
        scale = float(np.exp(rng.normal(0.0, 0.1)))
        candidate[start:stop] *= scale
    else:
        # mass transfer
        n = candidate.size
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j and candidate[i] > 0:
            amount = candidate[i] * rng.uniform(0.0, 0.1) # up to 10%
            candidate[i] -= amount
            candidate[j] += amount
    # optionally enforce symmetry: candidate = (candidate + candidate[::-1])/2
    if rng.random() < 0.3:
        candidate = (candidate + candidate[::-1]) * 0.5
    # clip to bounds
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

        cand = best.copy()
        # random move: either block scaling, mass transfer, or gradient move
        move_type = rng.integers(3)
        if move_type == 0:
            # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            scale = float(np.exp(rng.normal(0.0, 0.08)))
            cand[start:stop] *= scale
        elif move_type == 1:
            # mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and cand[i] > 0:
                amount = cand[i] * rng.uniform(0.0, 0.1)
                cand[i] -= amount
                cand[j] += amount
        else:
            # targeted reduction based on max conv
            conv = np.convolve(cand, cand)
            k = int(conv.argmax())
            # compute contributions
            # compute best_rev = cand[::-1]
            # for each i, j = k - i, check bounds
            # we could compute contributions via vectorized:
            # i = np.arange(n)
            # j = k - i
            # mask = (j>=0)&(j<n)
            # contrib = np.zeros_like(cand)
            # contrib[mask] = cand[i[mask]] * cand[j[mask]]
            i_arr = np.arange(n)
            j_arr = k - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            contrib = np.zeros_like(cand)
            # use fancy indexing
            contrib[mask] = cand[i_arr[mask]] * cand[j_arr[mask]]
            # choose index with highest contribution
            idx = int(np.argmax(contrib))
            # transfer small amount from idx to a random other index
            if cand[idx] > 0:
                target = rng.integers(0, n)
                if target != idx:
                    amount = cand[idx] * rng.uniform(0.0, 0.08)
                    cand[idx] -= amount
                    cand[target] += amount
        # optional symmetrization

        # clip to bounds

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
