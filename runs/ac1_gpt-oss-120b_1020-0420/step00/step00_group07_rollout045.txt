import numpy as np

def _uniform_sequence(length: int) -> np.ndarray:
    """Return a non‑negative uniform step‑function of the given length."""
    # Any positive constant works; 1.0 is convenient.
    return np.full(length, 1.0, dtype=np.float64)


def propose_candidate(seed=42, **kwargs):
    """
    Return a candidate that (theoretically) minimizes the AC1 bound.
    The optimal shape for non‑negative coefficients is the uniform vector,
    giving the lower bound 2.0.  We fall back to the provided parent
    sequence if something goes wrong.
    """
    # Determine the working length from the supplied construction.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:
        # If the parent is unavailable, pick a modest default length.
        n = 256
        parent = None

    # Primary candidate: a uniform sequence.
    cand = _uniform_sequence(n)

    # Evaluate the primary candidate (protect against unexpected errors).
    try:
        cand_score = float(evaluate_sequence(cand.tolist()))
    except Exception:
        cand_score = float('inf')

    # Evaluate the parent (if it exists) to keep a safe fallback.
    parent_score = float('inf')
    parent_clipped = None
    if parent is not None:
        try:
            parent_clipped = np.clip(parent, 0.0, 1000.0)
            parent_score = float(evaluate_sequence(parent_clipped.tolist()))
        except Exception:
            parent_score = float('inf')

    # Choose the best finite candidate.
    if np.isfinite(parent_score) and parent_score < cand_score:

    else:
        best = cand

    # Return as a plain Python list of floats.
