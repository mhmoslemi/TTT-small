import numpy as np

def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a small multiplicative perturbation of ``base`` using O(n) memory."""
    candidate = np.copy(base)
    n = candidate.size
    # Choose a block width between 0.5% and ~8% of the length.
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Slight log‑normal scaling of the block.
    factor = float(np.exp(rng.normal(0.0, 0.05)))
    candidate[start:stop] *= factor
    # Occasionally enforce a mild symmetry (helps some shapes).
    if rng.random() < 0.2:
        candidate = 0.5 * (candidate + candidate[::-1])
    # Clip to the allowed range.
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate
