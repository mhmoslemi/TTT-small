import numpy as np

# -------------------------------------------------------------
# Helper: fast full convolution using FFT (real inputs only)
# -------------------------------------------------------------
def _full_convolve(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using an FFT that is linear‑time in n."""
    n = a.size
    # size for a linear convolution (next power of two for speed)
    size = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

# Core scoring function (wrapper around the supplied evaluator)
# -------------------------------------------------------------
def _score(candidate: np.ndarray) -> float:
    """Exact score via the hidden evaluate_sequence."""
    # the evaluator expects a Python list of floats
    return float(evaluate_sequence(candidate.tolist()))


# -------------------------------------------------------------
# Local search improvement routine
# -------------------------------------------------------------
def _try_improve(base: np.ndarray, rng: np.random.Generator,
                 steps: int = 3000, block_prob: float = 0.15) -> np.ndarray:
    """
    Perform a cheap stochastic hill‑climb on ``base``.
    * move mass between two positions;
    * occasionally apply a random block‑perturbation.
    The best candidate seen is returned.
    """
    n = base.size
    best = base.copy()
    best_score = _score(best)

    # pre‑compute a small constant for later scaling
    eps = 1e-12

    for _ in range(steps):
        if rng.random() < block_prob:
            # --- block perturbation (similar to the reference program) ---
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            candidate = best.copy()
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            candidate[start:stop] *= factor
            if rng.random() < 0.35:
                candidate = 0.5 * (candidate + candidate[::-1])
            candidate = np.clip(candidate, 0.0, 1000.0)
        else:
            # --- two‑point mass transfer (mass conserving) ---
            i, j = rng.choice(n, size=2, replace=False)
            # transfer a small fraction of the larger entry to the smaller
            if best[i] > best[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            # amount to move: up to 5 % of the source (but never makes it negative)
            max_move = max(eps, 0.05 * best[src])
            delta = rng.random() * max_move
            if delta <= eps:
                continue
            candidate = best.copy()
            candidate[src] -= delta
            candidate[dst] += delta
            candidate = np.clip(candidate, 0.0, 1000.0)

        # evaluate the proposal
        sc = _score(candidate)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = candidate
            best_score = sc

    return best

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑empty list of non‑negative heights that (hopefully)
    improves on the basic uniform construction.  The routine works
    with the supplied parent array ``height_sequence_1`` if it exists,
    otherwise falls back to a simple uniform initialisation.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Load parent (if available) and fall back to a safe uniform vector.
    # -----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        # safe fallback – 256 entries of value 1.0
        parent = np.ones(256, dtype=np.float64)

    # Normalise into the allowed range and ensure a positive sum
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 0.01:
        parent[:] = 1.0

    # --------------------------------------------------------------
    # Start from a uniform vector – it is already optimal for many
    # lengths, but we still run a short stochastic refinement.
    # --------------------------------------------------------------
    n = parent.size
    uniform = np.full(n, 1.0, dtype=np.float64)          # any constant works
    uniform = np.clip(uniform, 0.0, 1000.0)

    # blend the supplied parent with the uniform vector (helps when the
    # parent is already decent)
    start_vec = 0.5 * parent + 0.5 * uniform
    start_vec = np.clip(start_vec, 0.0, 1000.0)

    # -----------------------------------------------------------------
    # Run the local search; the number of steps can be tuned via kwargs.
    # -----------------------------------------------------------------
    max_steps = int(kwargs.get("num_steps", 4000))
    refined = _try_improve(start_vec, rng, steps=max_steps)

    # -----------------------------------------------------------------
    # Return a plain Python list – the evaluator expects that format.
    # -----------------------------------------------------------------
    return [float(x) for x in refined]
