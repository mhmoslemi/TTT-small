import numpy as np

def _evaluate(seq):
    """Wrapper for the pre‑imported evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    """
    # Determine the length of the sequence we are allowed to use.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256                     # fallback length if the parent is absent

    # ----- 1. Start from a simple constant sequence (the known optimal shape) -----
    best = np.full(n, 1.0, dtype=np.float64)   # any constant >0 works
    best_score = _evaluate(best)

    # ----- 2. Optionally keep the provided parent if it is already better ----------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n:
            parent_score = _evaluate(parent)
            if parent_score < best_score:
                best = parent
                best_score = parent_score
    except Exception:
        pass   # ``height_sequence_1`` is not defined – ignore

    # ----- 3. Light random perturbation search (still O(n) memory) --------------
    rng = np.random.default_rng(seed)
    num_steps = kwargs.get('num_steps', 500)   # can be tuned by the caller
    width_low = max(1, n // 200)
    width_high = max(2, n // 12)

    for _ in range(num_steps):
        cand = best.copy()
        # random block size and position
        width = int(rng.integers(width_low, width_high))
        start = int(rng.integers(0, n - width + 1))
        stop = start + width

        # multiplicative perturbation of the block
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor

        # occasional mirroring (keeps the sequence non‑negative)
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])

        # enforce the problem's [0,1000] bounds
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate and keep the best so far
        score = _evaluate(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list of floats as required by the harness.
    return [float(x) for x in best]
