```
def max_convolution_info(a):
    # compute convolution via FFT
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    k_max = int(np.argmax(conv))
    max_val = conv[k_max]
    # compute contributions
    i = np.arange(n)
    j = k_max - i
    valid = (j >= 0) & (j < n)
    contrib = np.empty_like(a)
    contrib.fill(0.0)
    contrib[valid] = a[valid] * a[j[valid]]
    # find i maximizing contrib
    i_max = int(np.argmax(contrib))
    j_max = int(k_max - i_max)
    return k_max, max_val, i_max, j_max
```

```
def max_convolution_info(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]

    max_b = float(conv[k_max])
```

import numpy as np

def _fft_conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using FFT (full length)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact score used by the evaluator (but computed fast)."""
    n = a.shape[0]
    conv = _fft_conv_self(a)
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _max_conv_info(a: np.ndarray):
    """Return (k_max, max_b, i_max, j_max) for the current sequence."""
    n = a.shape[0]
    conv = _fft_conv_self(a)
    k_max = int(np.argmax(conv))
    max_b = float(conv[k_max])

    # contributions of each index to the max coefficient
    i = np.arange(n, dtype=np.int64)
    j = k_max - i
    valid = (j >= 0) & (j < n)
    contrib = np.zeros_like(a)
    contrib[valid] = a[valid] * a[j[valid]]
    i_max = int(np.argmax(contrib))
    j_max = int(k_max - i_max)
    return k_max, max_b, i_max, j_max
