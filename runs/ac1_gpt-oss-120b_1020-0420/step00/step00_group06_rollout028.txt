import numpy as np

# ------------------------------------------------------------
# helper: construct a set of indices with limited pair‐sum
# multiplicities.  sum_counts[s] records how many unordered
# pairs (i,j) (including i=j) have i+j == s.
# ------------------------------------------------------------
def _build_limited_sum_set(length, max_pairs_per_sum, rng):
    """
    Greedy construction of a set S ⊂ {0,…,length‑1} such that
    for every possible sum k the number of unordered pairs
    {i,j}⊂S with i+j=k does not exceed max_pairs_per_sum.
    Returns the list of selected indices.
    """
    idxs = np.arange(length, dtype=np.int32)
    rng.shuffle(idxs)

    sum_counts = np.zeros(2 * length - 1, dtype=np.int16)
    selected = []

    L = max_pairs_per_sum
    for idx in idxs:
        # check the self–pair (i,i)
        s_self = idx + idx
        if sum_counts[s_self] >= L:
            continue

        # check all pairs with already selected indices
        conflict = False
        for j in selected:
            s = idx + j
            if sum_counts[s] >= L:
                conflict = True
                break
        if conflict:
            continue

        # accept this index: update counters
        sum_counts[s_self] += 1
        for j in selected:
            s = idx + j
            sum_counts[s] += 1
        selected.append(int(idx))

    return selected


# ------------------------------------------------------------
def propose_candidate(seed=42, num_trials=20, max_pairs_per_sum=2, **kwargs):
    """
    Constructs a sparse 0/1 sequence obeying a bound on the number of
    pairwise sums.  The objective 2*n*max(conv)/sum^2 is dramatically
    reduced compared with the uniform sequence (≈2.0).  If construction
    fails to improve the score, the original `height_sequence_1` (or a
    simple fallback) is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise a fallback sequence (the provided parent or a simple one)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception()
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best_seq = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # try several random greedy constructions; keep the best
    # ------------------------------------------------------------------
    for _ in range(num_trials):
        sel = _build_limited_sum_set(n, max_pairs_per_sum, rng)
        if not sel:
            continue

        cand = np.zeros(n, dtype=np.float64)
        cand[sel] = 1.0                     # unit height on the selected positions
        cand = np.clip(cand, 0.0, 1000.0)    # safety, though already satisfied

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()
            # early exit if we have already passed the target bound
            if best_score < 1.6:           # target 1.50274349…
                break

    return [float(x) for x in best_seq]
