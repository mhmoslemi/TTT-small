import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function sequence that minimizes the
    AC1 upper bound evaluated by the provided `evaluate_sequence` function.

    The strategy is to use simple analytically‑motivated shapes:
    decreasing power‑law profiles a_i = (1 - i/(n-1))^p (and their mirror).
    These shapes decouple large values from each other in the convolution,
    yielding max‑convolution values substantially smaller than those of a
    uniform sequence.  A small number of candidates are evaluated and the
    best one (including the supplied `height_sequence_1` fallback) is returned.
    """
    # ------------------------------------------------------------------
    # Determine problem size and obtain a reasonable fallback parent.
    # ------------------------------------------------------------------
    parent_raw = globals().get("height_sequence_1")
    if isinstance(parent_raw, np.ndarray) and parent_raw.ndim == 1:
        parent = np.clip(parent_raw.astype(np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        # safe deterministic fallback if the global is missing or malformed
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initialise best candidate with the parent (if it is valid).
    # ------------------------------------------------------------------
    try:
        best_score = float(evaluate_sequence(parent.tolist()))
    except Exception:
        best_score = float("inf")
    if not np.isfinite(best_score):
        best_score = float("inf")
    best_seq = parent.copy()

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate and replace the current best if better.
    # ------------------------------------------------------------------
    def consider(seq: np.ndarray):
        global best_seq, best_score
        # Values are already in the allowed range; clipping guarantees safety.
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq.copy()
            best_score = score

    # ------------------------------------------------------------------
    # Candidate families: decreasing and increasing power‑law shapes.
    # ------------------------------------------------------------------
    lin = np.linspace(0.0, 1.0, n, dtype=np.float64)  # 0 → 1
    exponents = [1, 2, 3, 4, 5, 6]                    # modest powers give good results

    # decreasing profiles  a_i = (1 - x)^p
    for p in exponents:
        seq = np.power(1.0 - lin, p)
        seq = np.clip(seq, 0.0, 1000.0)
        consider(seq)

    # increasing profiles  a_i = x^p  (mirror of the above)
    for p in exponents:
        seq = np.power(lin, p)
        seq = np.clip(seq, 0.0, 1000.0)
        consider(seq)

    # ------------------------------------------------------------------
    # Light random perturbations around the current best (local refinement).
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for _ in range(10):
        # multiplicative log‑normal noise; amplitude kept tiny to stay near optimum.
        factor = np.exp(rng.normal(0.0, 0.02, size=n))
        candidate = best_seq * factor
        candidate = np.clip(candidate, 0.0, 1000.0)
        consider(candidate)

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
