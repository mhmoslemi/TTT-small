import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound used by `evaluate_sequence`.
    The method builds a random 0/1 pattern (≈ half ones) and performs a
    lightweight local‑search by flipping single entries.
    """
    rng = np.random.default_rng(seed)

    # Determine length from the provided parent if present, else fall back.
    parent = globals().get('height_sequence_1')
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.size)
    else:
        n = 256                     # arbitrary safe fallback length

    # ----- initialise a random binary candidate (≈ half ones) -----
    # Using a probability of 0.5 gives a good balance between sum and
    # expected maximal convolution value.
    cand = (rng.random(n) < 0.5).astype(np.float64)

    # Evaluate the initial candidate.
    best = cand
    best_score = float(evaluate_sequence(best.tolist()))

    # Number of local‑search steps (configurable via kwargs).
    steps = int(kwargs.get('num_steps', 500))

    for _ in range(steps):
        # Propose a tiny mutation: flip a single entry.
        new = best.copy()
        idx = rng.integers(0, n)
        new[idx] = 1.0 - new[idx]          # 0 ↔ 1

        # Enforce the problem bounds (they are already satisfied but keep it safe).
        np.clip(new, 0.0, 1000.0, out=new)

        # Exact evaluation.
        score = float(evaluate_sequence(new.tolist()))
        # Accept only improving, finite proposals.
        if np.isfinite(score) and score < best_score:
            best = new
            best_score = score

    # If for some unexpected reason the best is invalid, fall back to a uniform vector.
    if not (np.isfinite(best_score) and best_score < np.inf):
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # Return a plain Python list of floats as required by the harness.
    return [float(x) for x in best]
