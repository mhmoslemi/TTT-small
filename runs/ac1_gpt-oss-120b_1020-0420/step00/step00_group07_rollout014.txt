import numpy as np

def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative O(n) perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # Choose a block width proportional to n
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Random multiplicative factor (log‑normal perturbation)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # Occasionally symmetrize to keep the search well‑behaved
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # Clip to the evaluator's admissible interval
    return np.clip(cand, 0.0, 1000.0)
