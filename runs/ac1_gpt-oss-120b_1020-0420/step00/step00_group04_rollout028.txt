def compute_max_convolution(a):
    n = a.shape[0]
    L = int(2**np.ceil(np.log2(2*n-1)))  # next power of two.
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    # conv length L, but only first 2n-1 elements are meaningful.
    # So take conv[:2*n-1]
    maxb = conv[:2*n-1].max()
    return maxb

import numpy as np

def _next_pow2(x: int) -> int:
    """Return the next power of two greater or equal to x."""
    return 1 << (x - 1).bit_length()


def _conv_max(arr: np.ndarray, L: int) -> float:
    """Fast convolution (full) max using FFT."""
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    # relevant part of full convolution
    return float(conv[: 2 * arr.size - 1].max())


def _score(arr: np.ndarray, L: int) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = arr.size
    maxb = _conv_max(arr, L)
    s = float(arr.sum())
    if s == 0.0:
        return np.inf
    return 2.0 * n * maxb / (s * s)
