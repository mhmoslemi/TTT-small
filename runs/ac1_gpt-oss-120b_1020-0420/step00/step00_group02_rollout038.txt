def flatten_step(a, conv, k_max, mass_fraction=0.02):
    # mass_fraction fraction of total sum to relocate each step
    n = len(a)
    i_lo = max(0, k_max - (n-1))
    i_hi = min(n-1, k_max)
    idx = np.arange(i_lo, i_hi+1)
    # partners:
    partner = k_max - idx
    # compute contributions product of a[idx] * a[partner] > 0
    # compute total mass in those indices based on a values themselves
    total_mass = np.sum(a[idx] + a[partner])  # maybe double count
    # we will take from these indices a mass = mass_fraction * total_mass
    # Actually we'll compute reduction for each idx proportionally to a[idx] and partner contributions.
    # Simplify: subtract from each idx and partner a small proportion of their value:
    reduction = mass_fraction * a[idx]
    a[idx] = np.maximum(a[idx] - reduction, 0.0)
    # same for partner positions:
    a[partner] = np.maximum(a[partner] - reduction, 0.0)  # correct? partner indices are same shape set, may overlap with idx automatically.
    # compute lost mass:
    lost = np.sum(reduction) * 2  # because reduced from both
    # distribute lost uniformly:
    a += lost / len(a)
    # clip
    np.clip(a, 0.0, 1000.0, out=a)
    return a

        a = np.clip(a, 0.0, 1000.0)
        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
    # Also try symmetric (double-sided)
    for p in p_vals:
        left = (np.arange(n, dtype=np.float64) + 1) ** (-p)
        right = (np.arange(n, dtype=np.float64)[::-1] + 1) ** (-p)
        a = left + right
        a = np.clip(a, 0.0, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()

    # Optionally refine using hill climbing
    # We'll perform limited steps
    steps = 2000
    for _ in range(steps):
        # propose perturbation: pick random index range
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 50))
        stop = min(n, start + width)
        # random scaling factor multiplicative
        factor = np.exp(rng.normal(0.0, 0.02))
        candidate = best.copy()
        candidate[start:stop] *= factor
        # maybe random global small Gaussian noise
        candidate += rng.normal(0.0, 0.01, size=n)
        # Clip
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    return [float(x) for x in best]

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound.

    The method builds a family of decreasing power‑law sequences,
    optionally truncates the largest entries, evaluates them, and
    refines the best result by cheap local stochastic moves.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))          # provided parent length
    except Exception:                           # fallback length
        n = 256

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate a numpy array (list conversion done inside)
    # ------------------------------------------------------------------
    def score_of(arr):
        """Return the evaluator score for a 1‑D float array."""
        s = float(evaluate_sequence(arr.tolist()))
        return s

    # ------------------------------------------------------------------
    # 1) Grid search over simple analytic families
    # ------------------------------------------------------------------
    # power‑law exponent grid (p = 0 → uniform, p ≈ 0.5 → near‑optimal)
    p_vals = np.linspace(0.05, 0.55, 16)           # 16 values
    # how many leading entries to zero‑out (to drop the biggest terms)
    trunc_fracs = [0.0, 0.001, 0.002, 0.005, 0.01] # fractions of n
    # also try a symmetric “double‑sided’’ version
    double_side = False

    for p in p_vals:
        # base decreasing sequence a_i = (i+1)^{-p}
        base = (np.arange(n, dtype=np.float64) + 1.0) ** (-p)

        for frac in trunc_fracs:
            m = int(frac * n)                     # number of leading zeros
            if m >= n:
                continue
            if m > 0:
                seq = np.empty_like(base)
                seq[:m] = 0.0
                seq[m:] = (np.arange(1, n - m + 1, dtype=np.float64) ** (-p))
            else:
                seq = base.copy()

            # try the plain version
            s = score_of(seq)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_seq = seq.copy()

            # try a symmetric version (left+right) – only once per (p, m)
            if not double_side:
                left = seq
                right = left[::-1]
                sym = left + right
                s2 = score_of(sym)
                if np.isfinite(s2) and s2 < best_score:
                    best_score = s2
                    best_seq = sym.copy()
        double_side = True   # avoid recomputing the same double side repeatedly

    # ------------------------------------------------------------------
    # 2) Fallback to a simple uniform sequence if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = score_of(best_seq)

    # ------------------------------------------------------------------
    # 3) Local stochastic refinement (cheap hill‑climbing)
    # ------------------------------------------------------------------
    # number of refinement steps – modest to stay within time limits
    refinement_steps = 2000
    for _ in range(refinement_steps):
        cand = best_seq.copy()

        # random window perturbation (multiplicative)
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 200))
        stop = min(n, start + width)
        factor = float(np.exp(rng.normal(0.0, 0.02)))  # small log‑normal tweak
        cand[start:stop] *= factor

        # tiny additive Gaussian noise everywhere
        cand += rng.normal(0.0, 0.001, size=n)

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate
        s = score_of(cand)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = cand

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
