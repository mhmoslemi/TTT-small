def max_self_convolution(seq):
    n = seq.size
    # Compute length for full convolution: 2*n - 1
    fft_len = 1 << (2 * n - 1).bit_length()  # Pad to power of two for speed.
    # Use real FFT.
    freq = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(freq * freq, n=fft_len)
    # conv length = fft_len; but only first (2*n-1) entries correspond to real convolution, possibly others are small due to zero padding.
    # Take first 2*n-1 elements.
    conv = conv[:2*n-1]
    max_val = np.max(conv)
    return max_val

import numpy as np

# fast AC1 upper‑bound computation (identical to the official evaluator)
def _fast_score(seq: np.ndarray, fft_len: int) -> float:
    # convolution of seq with itself via FFT (full length 2*len-1)
    freq = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(freq * freq, n=fft_len)[: 2 * seq.size - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * seq.size * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):
    """
    Returns a list of non‑negative coefficients (length taken from the
    runtime variable ``height_sequence_1``) whose AC1 bound is (hopefully)
    lower than the baseline 2.0.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Obtain a parent sequence (fallback if the runtime object is absent)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)          # very small fallback
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size

    # ------------------------------------------------------------------
    # 2.  Prepare a cheap evaluation environment (FFT length is a power of two)
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # ------------------------------------------------------------------
    # 3.  Initialise the best candidate
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = _fast_score(best, fft_len)

    # a few random exponential‑shaped starts (permuted to break symmetry)
    for _ in range(5):
        alpha = rng.uniform(5e-4, 3e-3)                   # tunable decay
        base = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand = rng.permutation(base)                     # random reorder
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _fast_score(cand, fft_len)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

    # ------------------------------------------------------------------
    # 4.  Hill‑climbing / simulated‑annealing style search
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # start from the current best
        new = best.copy()

        # choose one of several cheap perturbations
        op = rng.integers(0, 4)

        if op == 0:                         # multiplicative block perturbation
            width = rng.integers(max(1, n // 300), max(2, n // 12))
            start = rng.integers(0, n - width + 1)
