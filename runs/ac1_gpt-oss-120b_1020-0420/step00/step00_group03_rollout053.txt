import numpy as np

# the evaluator is available in the execution environment
# from the harness we get the parent sequence as a NumPy array `height_sequence_1`

def propose_candidate(seed=42, num_iters=500, step0=0.01, **kwargs):
    """
    Gradient‐descent search for a step function that lowers the AC1 bound.

    The search works on a normalized vector (sum == 1) because the bound is
    scale‑invariant.  The objective that is minimized is the L2 distance of the
    convolution to its average value, which pushes the convolution towards a
    flat distribution and thus reduces the maximum entry.
    """
    rng = np.random.default_rng(seed)

    # length of the problem (use the provided parent if possible)
    try:
        n = int(height_sequence_1.shape[0])
        if n == 0:
            raise ValueError
    except Exception:                     # fallback length if parent missing
        n = 256

    # start from a small random positive vector and renormalise (sum == 1)
    a = rng.random(n) + 1e-6
    a /= a.sum()

    # pre‑compute constants used many times
    avg = 1.0 / (2 * n - 1)               # average convolution entry for sum == 1
    start_idx = 2 * n - 2                 # index offset for the correlation slice

    # keep the best vector seen so far
    best = a.copy()
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max()     # score = 2 * n * max(conv)  (sum == 1)

    for it in range(num_iters):
        # ----- convolution and gradient of squared‑error objective -----
        conv = np.convolve(a, a)                     # full linear convolution
        diff = conv - avg                            # deviation from the average
        corr = np.correlate(a, diff, mode='full')    # correlation a ⋆ diff
        grad = 4.0 * corr[start_idx:start_idx + n]   # ∂/∂a of Σ (conv‑avg)²

        # ----- gradient step (step size decays slowly) -----
        step = step0 / (1.0 + it / 100.0)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, None)            # enforce non‑negativity
        s = a_new.sum()
        if s < 1e-12:                               # safety restart if everything vanished
            a_new = rng.random(n) + 1e-6
            s = a_new.sum()
        a_new /= s                                   # renormalise to keep sum == 1

        # ----- evaluate the new candidate (exact bound) -----
        conv_new = np.convolve(a_new, a_new)
        new_score = 2.0 * n * conv_new.max()

        # accept improvement (greedy) – keeps the walk moving forward
        if new_score < 2.0 * n * conv.max():
            a = a_new

        # keep the globally best vector
        if new_score < best_score:
            best = a_new.copy()
            best_score = new_score

        # occasional tiny random jitter to avoid tiny plateaus
        if (it + 1) % 50 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a /= a.sum()

    # final candidate – clip to the evaluator's admissible range
    candidate = [float(min(1000.0, max(0.0, v))) for v in best]

    # (optional) ensure the returned list is accepted
    # _ = evaluate_sequence(candidate)   # the harness will call it anyway

    return candidate
