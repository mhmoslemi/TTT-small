def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

def generate_sidon_set(n, max_val=1000, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    positions = []
    sums = set()
    attempts = 0
    max_attempts = n * 10
    while attempts < max_attempts and len(positions) < int(np.sqrt(2*n)):
        idx = rng.integers(0, n)
        # test adding idx
        conflict = False
        for p in positions:
            if (idx + p) in sums:
                conflict = True
                break
        if not conflict and (idx + idx) not in sums:
            # add
            for p in positions:
                sums.add(idx + p)
            sums.add(idx + idx)
            positions.append(idx)
        attempts += 1
    return positions
```

    # Parameters: C = max allowed height, alpha controls the decay rate.
    C = 1000.0                 # max height (respecting the 0‑1000 bound)
    alpha = 5.0                # decay factor – larger values give lower scores

    # Compute the common ratio r = 1 - alpha/n  (always positive for our n).
    r = 1.0 - alpha / n
    if r <= 0.0:               # safeguard against pathological n/alpha choices
        r = 0.0

    # Build the sequence: a_i = C * r**i  for i = 0 … n-1
    idx = np.arange(n, dtype=np.float64)
    seq = C * np.power(r, idx)

    # Enforce the evaluator's clipping limits exactly.
    seq = np.clip(seq, 0.0, 1000.0)

    # Convert to a plain Python list (required by evaluate_sequence).
    result = seq.tolist()

    # Verify the candidate; if something went wrong, fall back to the parent.
    try:
        score = float(evaluate_sequence(result))
        if np.isfinite(score):
            return result
    except Exception:
        pass

    # Fallback: use the original parent (or a simple uniform vector if needed).
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    return parent.tolist()
