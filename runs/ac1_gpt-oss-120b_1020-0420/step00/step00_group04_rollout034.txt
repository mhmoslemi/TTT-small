import numpy as np


def _eval(seq: np.ndarray) -> float:
    """Helper that forwards a NumPy array to the supplied evaluator."""
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the 1.5027 bound.

    The routine builds a simple geometric‑decay family, picks the best member,
    and then performs a lightweight local‑search that shifts mass from the
    indices that contribute most to the current worst convolution peak to those
    that contribute least.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem (use the provided parent if present)
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        parent = np.clip(base, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(n, dtype=np.float64)
    else:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # start with a valid fallback
    best_seq = parent.copy()
    best_score = _eval(best_seq)

    # -----------------------------------------------------------------
    # 1) Scan a simple geometric family  a_i = r**i  (r in (0,1]).
    # -----------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    for r in np.linspace(0.5, 0.9995, 30):
        cand = np.power(r, idx)           # a_0 = 1, a_{n-1}=r^{n-1}
        sc = _eval(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # -----------------------------------------------------------------
    # 2) Light local refinement.
    #    Move a small amount of mass from the index that most raises the
    #    current maximal convolution entry to the index that raises it the
    #    least.  Accept only improving moves (very occasional uphill steps).
    # -----------------------------------------------------------------
    seq = best_seq.copy()
    max_iter = 2000          # number of refinement steps
    step_factor = 0.05       # relative amount moved each step

    for _ in range(max_iter):
        conv = np.convolve(seq, seq)          # full convolution, length 2n‑1
        k = int(conv.argmax())                # shift where the peak occurs

        # partner[i] = seq[k‑i] for the active shift (0 where undefined)
        partner = np.zeros_like(seq)
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)                 # exclusive
        i_vals = np.arange(i_start, i_end, dtype=np.int64)
        partner[i_vals] = seq[k - i_vals]

        # contribution of each position to the active peak
        contrib = seq * partner
        i_high = int(np.argmax(contrib))      # most responsible index
        i_low = int(np.argmin(contrib))       # least responsible index
        if i_high == i_low:                   # degenerate – pick a neighbour
            i_low = (i_high + 1) % n

        # amount to transfer (never make the source negative)
        eps = max(step_factor * seq[i_high], 1e-8)
        eps = min(eps, seq[i_high])
        if eps <= 0.0:
            continue

        cand = seq.copy()
        cand[i_high] -= eps
        cand[i_low] += eps
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _eval(cand)
        if sc < best_score:                   # improvement – keep it
            seq = cand
            best_seq, best_score = cand, sc
            step_factor = max(0.01, step_factor * 0.995)   # slowly tighten
        else:
            # occasional uphill step to escape tiny plateaus
            if rng.random() < 0.001:
                seq = cand
                step_factor = max(0.01, step_factor * 0.995)

    # Return a plain Python list of finite floats
    return [float(v) for v in best_seq]
