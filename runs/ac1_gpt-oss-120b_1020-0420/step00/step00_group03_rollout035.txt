import numpy as np

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Generate a non‑negative step‑function (list of heights) that aims to
    minimise the AC1 upper bound used by ``evaluate_sequence``.
    The routine builds several analytically‑motivated candidates and then
    performs a cheap stochastic local search while preserving the exact
    score (no surrogate).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and a simple fallback.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is supplied by the harness, if present.
        base_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = int(base_seq.size)
        fallback = base_seq.copy()
    except Exception:                               # pragma: no cover
        n = 256
        fallback = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (needs a Python list for the scorer).
    # ------------------------------------------------------------------
    def score_vec(v):
        return float(evaluate_sequence(v.tolist()))

    # ------------------------------------------------------------------
    # 1️⃣  Initial candidates (closed‑form shapes).
    # ------------------------------------------------------------------
    # Uniform – gives the baseline value 2.0.
    best = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = score_vec(best)

    # Power‑law with exponent 0.5 (≈ 1/√i) – the analytical “inverse‑sqrt” shape.
    inv_sqrt = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    inv_sqrt /= inv_sqrt.sum()
    s = score_vec(inv_sqrt)
    if s < best_score:
        best, best_score = inv_sqrt, s

    # Symmetric “inverse‑sqrt‑product” – 1/√(i·(n‑i+1)).
    idx = np.arange(1, n + 1, dtype=np.float64)
    sym_prod = 1.0 / np.sqrt(idx * (n - idx + 1))
    sym_prod /= sym_prod.sum()
    s = score_vec(sym_prod)
    if s < best_score:
        best, best_score = sym_prod, s

    # ------------------------------------------------------------------
    # 2️⃣  Small grid search over power‑law exponents and additive shifts.
    # ------------------------------------------------------------------
    alphas = np.linspace(0.40, 0.60, 7)                # 0.40 … 0.60
    betas = np.array([0.0, 0.5, 1.0, 2.0, 5.0])        # additive offset
    for alpha in alphas:
        for beta in betas:
            w = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64) + beta, alpha)
            w /= w.sum()
            sc = score_vec(w)
            if sc < best_score:
                best, best_score = w, sc

    # ------------------------------------------------------------------
    # 3️⃣  Stochastic local improvement.
    #    Transfer a small mass between two random positions while keeping
    #    the total sum unchanged.
    # ------------------------------------------------------------------
    stagnation = 0
    for _ in range(num_steps):
        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # donate from the larger entry
        donor = i if best[i] > best[j] else j
        recv = j if donor == i else i
        if best[donor] <= 0.0:
            continue

        # move a random fraction (up to 20 %) of the donor's mass
        delta = best[donor] * rng.random() * 0.20
        cand = best.copy()
        cand[donor] -= delta
        cand[recv] += delta

        # safety clipping (kept well below the 1000‑limit)
        cand = np.clip(cand, 0.0, 1000.0)

        sc = score_vec(cand)
        if sc < best_score:
            best, best_score = cand, sc
            stagnation = 0
        else:
            stagnation += 1

        # occasional random restart from a fresh analytic shape
        if stagnation >= 300:
            a = rng.choice(alphas)
            b = rng.choice(betas)
            cand = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64) + b, a)
            cand /= cand.sum()
            sc = score_vec(cand)
            if sc < best_score:
                best, best_score = cand, sc
                stagnation = 0
            else:
                stagnation = 0   # give the main search a fresh start

    # 4️⃣  Fallback safety – should never be needed, but guard against
    #    accidental NaNs or infinities.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = fallback
        best_score = score_vec(best)

    # Return as a plain Python list of floats.
    return [float(x) for x in best]
