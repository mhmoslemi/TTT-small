import numpy as np

def evaluate(seq):
    return float(evaluate_sequence(seq))

def initial_sequence():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0:
            raise ValueError
    except Exception:
        arr = np.ones(256, dtype=np.float64)
    return np.clip(arr, 0.0, 1000.0)

def block_perturbation(seq, rng):
    n = seq.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    new_seq = seq.copy()
    factor = np.exp(rng.normal(0.0, 0.12))
    new_seq[start:stop] *= factor
    if rng.random() < 0.35:
        new_seq = 0.5 * (new_seq + new_seq[::-1])
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq

def apply_transform(seq, rng):
    n = seq.size
    ttype = rng.choice(['pow','lin','cos','mix'])
    if ttype == 'pow':
        p = rng.uniform(0.3, 2.0)
        # Avoid p=0 leading to all ones (makes constant). Use np.power; preserve zeros.
        new = np.power(seq, p)
    elif ttype == 'lin':
        b = rng.uniform(-0.9, 0.9)
        i = np.arange(n, dtype=np.float64)
        w = 1.0 + b * ((i - (n-1)/2) / ((n-1)/2))
        new = seq * w
    elif ttype == 'cos':
        b = rng.uniform(-0.9, 0.9)
        i = np.arange(n, dtype=np.float64)
        w = 1.0 + b * np.cos(2*np.pi * i / n)
        new = seq * w
    else:  # mix
        # Add small multiplicative noise
        noise = np.exp(rng.normal(0.0, 0.08, size=n))
        new = seq * noise
    np.clip(new, 0.0, 1000.0, out=new)
    return new

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    base = initial_sequence()
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # Phase 1: global transforms
    for _ in range(num_steps//2):
        cand = apply_transform(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Phase 2: block perturbations around best
    for _ in range(num_steps//2):
        cand = block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return as list of floats
    return [float(x) for x in best]
