def compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    b = np.convolve(seq, seq)
    max_b = b.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    n = seq.size
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    n = height_sequence_1.size if defined else 256
    # generate base exponential shape
    eps = 0.00044  # can be tuned
    alpha = 1.0 - eps
    idx = np.arange(n, dtype=np.float64)
    seq = 1000.0 * np.power(alpha, idx)
    seq = np.clip(seq, 0.0, 1000.0)
    best = seq.copy()
    best_score = compute_score(best)
    cur = seq.copy()
    cur_score = best_score

    for step in range(num_steps):
        # compute convolution using FFT for efficiency
        size = 1 << ((2*n - 1).bit_length())
        A = np.fft.rfft(cur, n=size)
        conv = np.fft.irfft(A * A, n=size)[:2*n-1]
        max_j = int(np.argmax(conv))
        max_val = conv[max_j]
        # Compute contributions for this max_j
        i_start = max(0, max_j - (n-1))
        i_end = min(n-1, max_j)
        i_idx = np.arange(i_start, i_end+1)
        partner = max_j - i_idx
        contrib = cur[i_idx] * cur[partner]
        # Get indices of top contribution(s)
        top_k = int(rng.integers(1, 4))  # choose 1-3 top entries
        # sort descending
        order = np.argsort(contrib)[::-1]
        improved = False
        for t in range(min(top_k, len(order))):
            i = i_idx[order[t]]
            k = int(max_j - i)
            if i == k:
                # single index case
                # reduce value
                old = cur[i]
                new_val = old * 0.9  # reduction factor
                delta = old - new_val
                candidate = cur.copy()
                candidate[i] = new_val
                if delta > 0:
                    candidate += delta / n
                np.clip(candidate, 0.0, 1000.0, out=candidate)
            else:
                old_i = cur[i]
                old_k = cur[k]
                new_i = old_i * 0.9
                new_k = old_k * 0.9
                delta = (old_i - new_i) + (old_k - new_k)
                candidate = cur.copy()
                candidate[i] = new_i
                candidate[k] = new_k
                if delta > 0:
                    candidate += delta / n
                np.clip(candidate, 0.0, 1000.0, out=candidate)
            cand_score = compute_score(candidate)
            if cand_score < cur_score:
                cur = candidate
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best = candidate.copy()
                    best_score = cand_score
                break  # accept first improvement
        if not improved:
            # random perturb to escape
            # random block scaling similar to previous approach
            width = int(rng.integers(max(1, n//200), max(2, n//12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            candidate = cur.copy()
            candidate[start:start+width] *= factor

import numpy as np

def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound score for a non‑negative sequence."""
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:
        return np.inf
    n = seq.size
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, num_steps: int = 500, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1
    upper bound is (hopefully) better than the naive uniform bound.
    The routine starts from an exponential profile and then performs a
    cheap greedy optimisation that repeatedly reduces the largest
    convolution entry while keeping the total sum (approximately)
    unchanged.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine length and a safe fallback initialisation.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    else:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Exponential start – this already beats the uniform 2.0 bound.
    # ------------------------------------------------------------------
    eps = 0.00044                     # tuning parameter (≈ target/2n·e)
    alpha = 1.0 - eps
    idx = np.arange(n, dtype=np.float64)
    seq = 1000.0 * np.power(alpha, idx)   # values stay in [0,1000]
    np.clip(seq, 0.0, 1000.0, out=seq)

    if seq.sum() < 0.01:               # extremely unlikely, but safe
        seq = np.full(n, 1.0, dtype=np.float64)

    best = seq.copy()
    best_score = _score(best)
    cur = seq.copy()
    cur_score = best_score

    # pre‑compute FFT size (next power of two) for fast convolutions
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 3. Greedy optimisation loop.
    #    At each step we locate the convolution peak, pick the pair
    #    of indices that contribute most to that peak and shrink them
    #    a little, redistributing the removed mass uniformly.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # -- convolution via FFT (O(n log n)) ------------------------
        A = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
        j_max = int(np.argmax(conv))

        # -- indices that lie on the anti‑diagonal j_max -------------
        i_start = max(0, j_max - (n - 1))
        i_end = min(n - 1, j_max)
        i_idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = j_max - i_idx
        contributions = cur[i_idx] * cur[partner]

        # -- try a few of the biggest contributors --------------------
        order = np.argsort(contributions)[::-1]      # descending
        improved = False

        for ord_i in order[:3]:                     # up to three attempts
            i = i_idx[ord_i]
            k = int(partner[ord_i])
            factor = 0.90                           # shrink factor

            cand = cur.copy()
            if i == k:                              # diagonal entry
                old = cand[i]
                new = old * factor
                delta = old - new
                cand[i] = new
            else:
                old_i, old_k = cand[i], cand[k]
                new_i, new_k = old_i * factor, old_k * factor
                delta = (old_i - new_i) + (old_k - new_k)
                cand[i], cand[k] = new_i, new_k

            if delta > 0.0:                         # mass redistribution
                cand += delta / n

            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _score(cand)

            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                improved = True
                if cand_score < best_score:
                    best, best_score = cand.copy(), cand_score
                break                               # accept first improvement

        # -- if no improvement, perform a small random perturbation --
        if not improved:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand = cur.copy()
            cand[start : start + width] *= factor
            if rng.random() < 0.35:                # occasional mirroring
                cand = 0.5 * (cand + cand[::-1])
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _score(cand)
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                if cand_score < best_score:
                    best, best_score = cand.copy(), cand_score

    # ------------------------------------------------------------------
    # 4. Return the best candidate found (as a plain Python list).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
