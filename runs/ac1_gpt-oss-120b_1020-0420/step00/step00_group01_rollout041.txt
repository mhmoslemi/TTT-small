    """
    Generate a deterministic candidate sequence using simple power-law ramps.
    Several exponents are tried and the one with the lowest AC1 upper bound
    (as measured by `evaluate_sequence`) is returned.
    """
    # Determine the length from the provided parent if possible.
    try:
        n = int(height_sequence_1.shape[0])
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        # Fallback length if the parent is not available.
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    best_arr = parent.copy()
    best_score = float(evaluate_sequence(best_arr.tolist()))

    # Try a few power‑law shapes a_i = (i/(n‑1))**p * 1000.
    # Exponents are small integers; higher exponents keep the maximum at the
    # last entry while spreading the mass increasingly towards the end.
    for p in (1, 2, 3, 4, 5):
        # Build the ramp in floating point to avoid integer overflow.
        idx = np.arange(n, dtype=np.float64)
        # i = 0 gives 0, i = n‑1 gives the maximum before scaling.
        raw = idx ** p
        # Scale so that the last element is exactly 1000 (the allowed max).
        if raw[-1] == 0:
            continue
        scale = 1000.0 / raw[-1]
        cand = raw * scale
        # Enforce the allowed range explicitly (clip handles any tiny rounding errors).
        cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate the candidate using the supplied evaluator.
        try:
            sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = cand

    # Ensure we always return a non‑empty, finite list.
    if best_arr.size == 0:
        best_arr = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_arr]
