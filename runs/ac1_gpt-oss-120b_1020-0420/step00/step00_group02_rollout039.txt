def generate_sidon_positions(n, m_target):
    positions = []
    sums = set()
    for i in range(n):
        if (i + i) in sums:
            continue
        conflict = False
        for j in positions:
            if (i + j) in sums:
                conflict = True
                break
        if not conflict:
            for j in positions:
                sums.add(i + j)
            sums.add(i + i)
            positions.append(i)
            if len(positions) >= m_target:
                break
    return positions

    The uniform sequence of ones achieves the theoretical optimum of 2.0.
