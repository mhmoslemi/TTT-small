```python
def simplex_projection(v, s=1.0):
    # v is np.ndarray, may have negative values.
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _max_convolution(seq: np.ndarray) -> float:
    """Return max value of the full linear convolution of seq with itself."""
    n = seq.size
    L = _next_pow2(2 * n - 1)
    ft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Exact objective used by the evaluator (but faster, no clipping)."""
    if seq.size == 0:
        return float("inf")
    maxb = _max_convolution(seq)
    s = float(seq.sum())
    if s <= 0.0:
        return float("inf")
    return 2.0 * seq.size * maxb / (s * s)

    # linear decreasing shapes
    for r in np.linspace(0.0, 0.9, 16):
        cand = _linear_seq(n, r)
        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # exponential decreasing shapes
    lam_max = 5.0 / n   # gives exp(-lam_max * (n-1)) ≈ e^{-5}
    for lam in np.linspace(0.0, lam_max, 16):
        cand = _exp_seq(n, lam)
        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # two‑block families (different block sizes and value ratios)
    for p in (0.25, 0.33, 0.5, 0.66, 0.75):
        m = max(1, int(round(p * n)))
        for ratio in np.linspace(1.0, 3.0, 9):
            cand = _two_block_seq(n, m, ratio)
            if cand is None:
                continue
            sc = _score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc

    # -----------------------------------------------------------------
    # 4. cheap random local refinements (mass transfer between entries)
    # -----------------------------------------------------------------
    refinements = 200
    for _ in range(refinements):
        a = best_seq.copy()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = a[i] * 0.1
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        a[i] -= delta
        a[j] += delta
        # no need to renormalise – transfer keeps total sum unchanged
        sc = _score(a)
        if sc < best_score:
            best_seq, best_score = a, sc

    # -----------------------------------------------------------------
    # 5. compare with the original parent (after normalising it)
    # -----------------------------------------------------------------
    if np.sum(parent) > 0.0:
        parent_norm = np.clip(parent, 0.0, 1000.0)
        parent_norm = parent_norm / parent_norm.sum()
        parent_score = _score(parent_norm)
        if parent_score < best_score:
            best_seq = parent_norm
            best_score = parent_score

    # -----------------------------------------------------------------
    # 6. final list of python floats – the harness will run the official evaluator
    # -----------------------------------------------------------------
    return [float(x) for x in best_seq]
