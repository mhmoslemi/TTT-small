    """
    Generate a linear ramp from 0 to 1000 (inclusive) with the given length.
    The values are already within the evaluator's allowed range.
    """
    return np.linspace(0.0, 1000.0, length, dtype=np.float64)


def propose_candidate(seed: int = 42, **kwargs):
