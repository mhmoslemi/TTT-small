import numpy as np

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that strives to
    minimize the AC1 upper bound evaluated by `evaluate_sequence`.
    The routine works with the provided `height_sequence_1` if available,
    otherwise falls back to a uniform 1‑vector.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and an initial feasible vector.
    # ------------------------------------------------------------------
    try:
        # Use the supplied parent sequence if it looks valid.
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        # Clip to the evaluator's admissible range.
        base = np.clip(base, 0.0, 1000.0)
        n = base.size
    except Exception:
        # Fallback: a modest length with all ones.
        n = 256
        base = np.full(n, 1.0, dtype=np.float64)

    # Start the search from the (clipped) base vector.
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Simple stochastic local search.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        candidate = best.copy()

        # 1) Block multiplicative perturbation (large‑scale change).
        width = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.05))  # modest log‑normal factor
        candidate[start:start + width] *= factor

        # 2) Small independent jitter on a random subset.
        subset_size = max(1, n // 100)
        idx = rng.choice(n, size=subset_size, replace=False)
        candidate[idx] += rng.normal(0.0, 0.01, size=subset_size)

        # Enforce the admissible bounds.
        candidate = np.clip(candidate, 0.0, 1000.0)

        # Evaluate and keep improvement.
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    # ------------------------------------------------------------------
    # Return the best-found sequence as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
