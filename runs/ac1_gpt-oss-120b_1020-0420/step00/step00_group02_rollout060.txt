def propose_candidate(seed=42, num_samples=300, **kwargs):
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # base sequence
    base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    # initial best: base
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # function to evaluate and possibly update best
    def try_candidate(cand):
        nonlocal best, best_score
        # enforce bounds and clip
        cand = np.clip(cand, 0.0, 1000.0)
        # ensure sum >0.01 to avoid inf; if too small, scale up
        s = cand.sum()
        if s < 0.01:
            cand = cand + (0.01 - s) / cand.size
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

    # random samples
    for i in range(num_samples):
        # decide sample type
        if i % 3 == 0:
            # random permutation of base
            cand = base[rng.permutation(base.size)]
        elif i % 3 == 1:
            # exponential random values
            cand = rng.exponential(scale=1.0, size=base.size)
            # scale to have similar sum as base to keep ratio invariant (optional)
            # just normalize to sum of base
            cand = cand * (base.sum() / cand.sum())
        else:
            # uniform random values across [0,1]
            cand = rng.random(base.size)
            cand = cand * (base.sum() / cand.sum())
        try_candidate(cand)

    # refinement: apply up to 50 local smoothing steps on best
    for _ in range(50):
        conv = np.convolve(best, best)
        kmax = int(np.argmax(conv))
        shift = kmax - (len(best) - 1)
        # indices participating
        idx = np.arange(len(best))
        mask = (idx - shift >= 0) & (idx - shift < len(best))
        # reduce those entries slightly
        reduction_factor = 0.95
        new_best = best.copy()
        new_best[mask] *= reduction_factor
        # redistribute removed mass uniformly
        removed = best[mask].sum() - new_best[mask].sum()
        # add back uniformly (or to random positions not in mask)
        # choose random positions for redistribution
        if removed > 0:
            # pick random indices (could be all)
            add_mask = ~mask
            if add_mask.any():
                # distribute proportionally to current values + epsilon to avoid zero
                total = new_best[add_mask].sum()
                if total == 0:
                    new_best[add_mask] = removed / add_mask.sum()
                else:
                    new_best[add_mask] += (new_best[add_mask] / total) * removed
            else:
                # all indices involved, just scale back uniformly
                new_best *= (best.sum() / new_best.sum())
        try_candidate(new_best)

    return [float(x) for x in best]

        # enforce bounds and avoid sum zero
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:

            cand = cand + (0.01 - cand.sum()) / cand.size
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

    # Uniform candidate
    uniform = np.ones(n, dtype=np.float64)
    try_update(uniform)

    # Compute optimal constant c solving e^c = 2c + 1
    def solve_c():
        c = 1.0
        for _ in range(30):
            f = np.exp(c) - (2.0 * c + 1.0)
            fp = np.exp(c) - 2.0
            c = c - f / fp
        return c
    c_opt = solve_c()
    # Grid around optimal
    for c in np.linspace(0.5, 3.0, 20):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        # exponential decreasing from start
        a = r ** np.arange(n, dtype=np.float64)
        try_update(a)
        # reverse order (decreasing from end)
        a_rev = a[::-1]
        try_update(a_rev)

    # Include exponential with solved c_opt as well
    r_opt = 1.0 - c_opt / n
    if r_opt > 0.0:
        a_opt = r_opt ** np.arange(n, dtype=np.float64)
        try_update(a_opt)
        try_update(a_opt[::-1])

    # If still no valid best, fallback to uniform
    if best is None:
        best = uniform.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    # Local random block perturbations to possibly improve
    for _ in range(num_steps):
        # block perturbation
        cand = best.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        # occasional mirroring
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])
        try_update(cand)

    Generate a non‑negative step‑function sequence that (heuristically) lowers
    the AC1 upper bound 2·n·max(b)/sum(a)^2.
    The routine builds a simple exponential shape (near‑optimal analytically)

        n = base.size

        n = 256

        nonlocal best, best_score
        cand = np.clip(candidate, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 5. If nothing succeeded (should not happen) fall back to uniform
    # ------------------------------------------------------------------
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 6. Light local search: random block scaling & occasional mirroring
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = best.copy()
        # random block width roughly between 0.5 % and 15 % of n
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        # modest multiplicative tweak
        factor = float(np.exp(rng.normal(0.0, 0.05)))
        cand[start:stop] *= factor
        # occasional symmetric averaging (helps flatten the shape)
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])
        try_update(cand)

    # ------------------------------------------------------------------
    # 7. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
