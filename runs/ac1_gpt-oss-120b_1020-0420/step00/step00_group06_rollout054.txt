import numpy as np

def make_block_perturbation(base, rng):
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # Choose block width fairly large but O(n)
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = start + w
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def _transfer(a, donor, receiver, frac, rng):
    """Transfer a fraction of donor's mass to receiver, return new array."""
    if donor == receiver or a[donor] <= 0.0:
        return a
    delta = float(a[donor] * frac * rng.random())
    # ensure delta not exceed donor
    delta = min(delta, a[donor])
    a_new = a.copy()
    a_new[donor] -= delta
    a_new[receiver] += delta
    return np.clip(a_new, 0.0, 1000.0)
