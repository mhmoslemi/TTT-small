import numpy as np

def make_block_perturbation(base, rng):
    """Return a random O(n) perturbation of ``base`` by scaling a random block."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block length ~ uniform between n//200 and n//12
    min_len = max(1, n // 200)
    max_len = max(2, n // 12)
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative factor from log-normal distribution
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrization
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # clip to allowed range
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

import numpy as np

def make_block_perturbation(base, rng):
    """Return an O(n) random perturbation of ``base``."""
    cand = np.array(base, copy=True)
    n = cand.size
    # choose a random block width
    min_len = max(1, n // 200)
    max_len = max(2, n // 12)
    width = int(rng.integers(min_len, max_len + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation of the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps keep a monotone shape)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # occasionally enforce a non‑increasing ordering
    if rng.random() < 0.20:
        cand = np.maximum.accumulate(cand[::-1])[::-1]
    # clip to the allowed interval
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

            raise Exception

        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate and keep it if it improves the best.
    def try_candidate(cand):
        nonlocal best, best_score

        if np.isfinite(score) and score < best_score:

            best_score = score
