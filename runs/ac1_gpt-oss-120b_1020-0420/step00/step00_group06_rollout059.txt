import numpy as np

def _build_sparse_set(n: int, rng: np.random.Generator, max_per_sum: int) -> np.ndarray:
    """Return indices of a sparse set where each unordered sum appears at most max_per_sum times.
    Uses a greedy randomized construction."""
    order = rng.permutation(n)
    # counts for unordered sums i + j (0 .. 2n-2)
    sum_counts = np.zeros(2 * n, dtype=np.int16)
    selected = []
    for i in order:
        si = int(i)
        # check self‑sum
        if sum_counts[si + si] >= max_per_sum:
            continue
        # check all sums with already selected indices
        ok = True
        for j in selected:
            if sum_counts[si + j] >= max_per_sum:
                ok = False
                break
        if not ok:
            continue
        # accept i: update counters
        for j in selected:
            sum_counts[si + j] += 1
        sum_counts[si + si] += 1
        selected.append(si)
    return np.array(selected, dtype=np.int32)

    The method builds a sparse index set with few repeated pair‑sums,
    which keeps the peak of the self‑convolution low while the L1 norm stays large.
    """
    rng = np.random.default_rng(seed)

    # Determine length from the provided parent if present
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256  # safe fallback length

    # fallback: a uniform vector (guarantees bound = 2)
    best_seq = np.ones(n, dtype=np.float64)

    # parameters for the sparse set search
    max_per_sum = 2            # each unordered sum may appear at most twice
    target_score = 1.50274349 # stop early if we beat the target
