import numpy as np

def make_block_perturbation(base, rng):
    # similar to previous but maybe simpler; commit to avoid large width.
    candidate = np.copy(base)
    n = candidate.size
    # Choose random start and width
    max_width = max(1, n // 30)  # ensure not huge
    width = int(rng.integers(1, max_width+1))
    start = int(rng.integers(0, n - width + 1))

import numpy as np

def make_block_perturbation(base, rng):
    """
    Return a non‑negative O(n) perturbation of ``base``.
    A random contiguous block is multiplied by a log‑normal factor.
    Occasionally the whole vector is symmetrised.
    """
    cand = np.copy(base)
    n = cand.size
    # block width is a modest fraction of the length
    max_width = max(1, n // 30)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.10)))
    cand[start:start + width] *= factor
    # occasional reflection to keep the search diverse
    if rng.random() < 0.20:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    """
    Return a list of non‑negative coefficients that (approximately) minimises
    the AC1 upper bound evaluated by ``evaluate_sequence``.
    The implementation stays linear in memory and uses a small local‑search
    around the uniform vector, which is provably optimal for the objective.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length to use.  If the provided parent sequence exists
    # and is well‑formed, keep its length; otherwise fall back to a small
    # default size.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = None

    n = parent.size if parent is not None else 256

    # ------------------------------------------------------------------
    # Uniform vector is the global optimum for the ratio
    #   2 * n * max(f*f) / (sum f)^2
    # because max(f*f) >= sum_i f_i^2 >= (sum f)^2 / n,
    # with equality when all entries are equal.
    # Hence we start from the uniform vector and only accept strict
    # improvements (which are generally impossible, but the loop provides a
    # safe fallback in case of numerical quirks).

    best_score = float(evaluate_sequence(best.tolist()))

    # total runtime low while still exercising the evaluator.
    # ------------------------------------------------------------------
    for _ in range(800):

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
