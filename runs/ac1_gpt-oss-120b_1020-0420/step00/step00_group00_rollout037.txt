import numpy as np
from math import sqrt

# evaluate_sequence is supplied by the harness.
# It returns a float score (inf for invalid sequences).

def _eval(seq):
    """Wrapper for evaluate_sequence that returns a float."""
    return float(evaluate_sequence(seq))

def _make_exponential(n, c, p):
    """
    Return a monotone decreasing sequence a_i = exp(-c * (i/(n-1))**p).
    The sequence is already non‑negative and bounded by 1.
    """
    if n == 1:
        return np.ones(1, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    frac = idx / (n - 1)               # values in [0,1]
    return np.exp(-c * np.power(frac, p))

def _make_block_perturbation(base, rng):
    """Apply a cheap local perturbation to a sequence."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12
    wmin = max(1, n // 200)
    wmax = max(2, n // 12)
    width = int(rng.integers(wmin, wmax + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative noise (log‑normal)
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:stop] *= factor
    # keep within allowed range
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    uniform bound.  The routine explores a small family of exponential
    profiles, mixes them with the provided parent, and finishes with a short
    local search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. initialise with the parent (if it exists) or with a uniform list.
    # ------------------------------------------------------------------
    try:

        parent = np.clip(parent_arr, 0.0, 1000.0)

        parent = np.ones(256, dtype=np.float64)            # fallback

    n = parent.size

    best_seq = parent.copy()
    best_score = _eval(best_seq.tolist())

    # ------------------------------------------------------------------
    # 2. explore a grid of exponential shapes a_i = exp(-c*(i/(n-1))**p)
    #    with several (c, p) pairs.
    # ------------------------------------------------------------------

            cand = np.clip(cand, 0.0, 1000.0)

            score = _eval(cand.tolist())
            if np.isfinite(score) and score < best_score:
