def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative O(n) perturbation of ``base``."""
    n = base.size
    # block width between n/200 and n/12 (at least 1)
    wmin = max(1, n // 200)
    wmax = max(2, n // 12)
    width = int(rng.integers(wmin, wmax + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:start + width] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    n = int(parent_arr.size)

    # Normalise to unit sum (scale‑invariant for the score)
    total = parent_arr.sum()
    if total <= 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = parent_arr / total

    # ----- parameters for the smooth‑max optimiser ------------------------
    total_iters = 300          # gradient steps (adjust for time budget)
    beta_start = 0.5
    beta_end = 6.0
    lr_initial = 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    size_fft = _next_pow2(3 * n - 2)          # enough for linear correlation
    offset = 2 * n - 2                        # index of zero‑shift in correlation

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    # ----- smooth‑max gradient descent ------------------------------------
    for it in range(1, total_iters + 1):
        beta = beta_start + (beta_end - beta_start) * (it - 1) / max(1, total_iters - 1)

        A = np.fft.rfft(a, size_fft)
        conv_full = np.fft.irfft(A * A, n=size_fft)
        conv = conv_full[: 2 * n - 1]

        # soft‑max weights on the convolution values
        max_c = conv.max()
        shifted = conv - max_c               # non‑positive, avoids overflow
        exp_term = np.exp(beta * shifted)
        w = exp_term / exp_term.sum()        # shape (2n‑1,)

        W = np.fft.rfft(w, size_fft)
        corr_full = np.fft.irfft(A * np.conj(W), n=size_fft)

        # linear correlation (first 3n‑2 entries are the non‑wrapped part)
        L = 3 * n - 2
        corr_line = corr_full[:L]
        grad = 2.0 * corr_line[offset: offset + n]   # gradient of smooth‑max

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr_initial * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # enforce constraints
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s <= 0.0:
            a = np.ones_like(a) / n
            s = 1.0
        a = a / s

        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score, best_a = sc, a.copy()

    # ----- cheap random local search as a safety net --------------------
    for _ in range(num_steps):
        cand = make_block_perturbation(best_a, rng)
        tot = cand.sum()
        if tot <= 0.0:
            continue
        cand = cand / tot
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score, best_a = sc, cand.copy()

    # Return as plain Python list
    return [float(x) for x in best_a]
