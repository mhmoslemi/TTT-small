def compute_score(seq):
    n = seq.shape[0]
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return np.inf
    # Compute convolution via FFT
    # Need conv length = 2n-1
    m = 1
    while m < 2*n - 1:
        m <<= 1
    # Use real FFT for speed
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]  # real convolution
    max_conv = np.max(conv)
    return 2.0 * n * max_conv / (sum_seq * sum_seq)

def compute_score(seq):
    n = len(seq)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return np.inf
    # next power of two for FFT
    m = 1
    while m < 2*n - 1:
        m <<= 1
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[:2*n-1]
    max_conv = np.max(conv)
    return 2.0 * n * max_conv / (sum_seq * sum_seq)

import numpy as np

def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound for a non‑negative sequence."""
    n = seq.size
    s = np.sum(seq)
    if s < 1e-12:
        return np.inf
    # FFT length: next power of two ≥ 2n‑1
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return 2.0 * n * max_conv / (s * s)


def _perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Produce a cheap O(n) non‑negative perturbation."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = rng.integers(lo, hi + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = seq.copy()
    out[start:start + width] *= factor
    if rng.random() < 0.35:                       # optional mirroring
        out = 0.5 * (out + out[::-1])
    np.clip(out, 0.0, 1000.0, out=out)
