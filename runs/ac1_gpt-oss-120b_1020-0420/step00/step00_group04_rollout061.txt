def propose_candidate(seed=42, num_iters=2500):
    rng = np.random.default_rng(seed)
    base = np.copy(height_sequence_1).astype(np.float64)
    # fallback: if invalid, set uniform.
    if base.ndim!=1 or base.size==0 or not np.isfinite(base).all():
        base = np.full(256, 1.0)
    # ensure nonneg and bounded
    base = np.clip(base, 0.0, 1000.0)
    # Normalize to have sum > 0
    if np.sum(base) < 0.01:
        base[:] = 1.0
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Early improvement: Could optionally start with uniform
    # Now local search focusing on max conv reduction
    n = base.size
    for it in range(num_iters):
        cand = best.copy()
        # compute conv and max index
        conv = np.convolve(cand, cand)
        max_idx = int(np.argmax(conv))
        # Determine indices involved in this max conv
        involved = np.arange(n)
        # mask of indices where (max_idx - i) in [0, n-1]
        mask = (max_idx - involved >= 0) & (max_idx - involved < n)
        idxs = involved[mask]
        # Reduce those entries a bit
        step = 0.05 * rng.random()  # between 0-0.05
        cand[idxs] *= (1.0 - step)
        # compute reduction in sum
        reduction = np.sum(best) - np.sum(cand)
        # distribute reduction to random positions not in idxs to keep sum constant
        other_idxs = involved[~mask]
        # if there are any other positions
        if other_idxs.size > 0 and reduction > 0:
            # allocate equally
            add_per = reduction / other_idxs.size
            cand[other_idxs] += add_per
        # clip
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            continue # keep improvement

        # else occasional random perturbation for escape
        if rng.random() < 0.1:
            # apply block perturbation like before
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand = best.copy()
            cand[start:stop] *= factor
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)
            score2 = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand
                best_score = score2
    return [float(x) for x in best]

import numpy as np

# ------------------------------------------------------------
# Helper: a cheap block perturbation (used for occasional jumps)
# ------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return one O(n)-memory non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # choose a block width between n//200 and n//12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps flatten the convolution)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    AC1 upper bound.  Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the supplied parent if it looks sane, otherwise use
    # a simple uniform start.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                     # pragma: no cover
        parent = None
    if (

    ):
        parent = np.full(256, 1.0, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    if np.sum(parent) < 0.01:              # enforce a minimal total mass
        parent[:] = 1.0

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # ------------------------------------------------------------
    # Pre‑compute FFT length (next power of two >= 2n‑1) for fast conv
    # ------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    for _ in range(num_iters):
        # ---- compute full convolution via FFT (O(n log n)) ----------
        fft_arr = np.fft.rfft(best, n=fft_len)
        conv = np.fft.irfft(fft_arr * fft_arr, n=fft_len)[: 2 * n - 1]

        # ---- locate a shift with maximal convolution -----------------
        k_max = int(np.argmax(conv))

        # ---- which indices actually contribute to that shift -------
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)          # boolean mask of contributors
        involved = i_vals[mask]

        # ---- propose a targeted reduction of the contributors -------
        cand = best.copy()

        # reduction factor (up to ~4 % shrinkage) – random per step
        reduction_factor = 1.0 - rng.uniform(0.0, 0.04)
        cand[involved] *= reduction_factor

        # ---- keep total mass roughly constant: redistribute the loss
        delta = best.sum() - cand.sum()
        if delta > 0.0:
            not_involved = i_vals[~mask]
            if not_involved.size > 0:
                add_per = delta / not_involved.size
                cand[not_involved] += add_per

        # enforce bounds required by the evaluator
        cand = np.clip(cand, 0.0, 1000.0)

        # ---- evaluate the new candidate -----------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score
            continue                     # keep the improvement; skip random move
