import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimize the AC1 upper bound used by `evaluate_sequence`.

    The construction is a linearly decreasing profile
        a_i = 2 * (n - i) / (n + 1)    (0 ≤ i < n)
    which has total sum exactly n and a known theoretical bound of
        4 (n+2) / (3 (n+1))  ≈ 1.333 for large n.
    Small random block perturbations are tried to see if a slight
    improvement is possible without breaking the bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Base candidate – linear decreasing shape (normalized sum = n)
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise ValueError
        # a_i = 2 * (n - i) / (n + 1)
        factor = 2.0 / (n + 1)
        idx = np.arange(n, dtype=np.float64)
        candidate = factor * (n - idx)
        candidate = np.clip(candidate, 0.0, 1000.0)
        best_score = float(evaluate_sequence(candidate.tolist()))
    except Exception:
        # Fallback: short uniform vector (always valid)
        n = 256
        candidate = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(candidate.tolist()))

    # ------------------------------------------------------------------
    # 2.  Light local search: random block scaling (keeps total sum ≈ n)
    # ------------------------------------------------------------------
    for _ in range(200):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        # multiplicative perturbation of the chosen block
        mult = float(np.exp(rng.normal(0.0, 0.05)))
        new_cand = candidate.copy()
        new_cand[start:stop] *= mult

        # enforce bounds and renormalise to keep the sum close to n
        new_cand = np.clip(new_cand, 0.0, 1000.0)
        s = new_cand.sum()
        if s > 0.0:
            new_cand *= (n / s)

        score = float(evaluate_sequence(new_cand.tolist()))
        if np.isfinite(score) and score < best_score:
            candidate = new_cand
            best_score = score

    # ------------------------------------------------------------------
    # 3.  Return the best found vector as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in candidate]
