def conv_score(a):
    n = a.size
    # Compute convolution via FFT
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a == 0:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # get parent/height_sequence_1
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    # Validate parent
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 256
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size

    target_sum = np.sqrt(2.0 * n)

    # Initialize best as uniform distribution
    best = np.full(n, target_sum / n, dtype=np.float64)
    best_score = compute_score(best)

    # Also evaluate original parent just in case it's better
    parent_score = compute_score(parent)
    if parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # function for mass transfer
    def propose_mass_transfer(arr):
        cand = arr.copy()
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * cand[i] * 0.2  # up to 20% of source
        cand[i] -= delta
        cand[j] += delta
        return cand

    # function for anti-symmetric random
    def propose_anti_random():
        vals = rng.random(n) + 1e-9
        # sort descending
        vals = np.sort(vals)[::-1]
        a = np.empty(n, dtype=np.float64)
        left = 0
        right = n - 1
        toggle = True
        for v in vals:
            if toggle:
                a[left] = v
                left += 1
            else:
                a[right] = v
                right -= 1
            toggle = not toggle
        # Normalize sum
        a_sum = a.sum()
        if a_sum > 0:
            a *= target_sum / a_sum
        return a

    # main loop
    for step in range(num_steps):
        if step % 20 == 0:
            cand = propose_anti_random()
        else:
            cand = propose_mass_transfer(best)

        # ensure nonnegative and clip
        np.clip(cand, 0.0, 1000.0, out=cand)

        # compute score
        score = compute_score(cand)
        if score < best_score:
            best = cand
            best_score = score
            # optionally break if sufficiently low
            if best_score < 1.6:
                # continue but we could break early; continue for more improvements.
                pass

    # final evaluation with evaluate_sequence to ensure valid output.
    final_score = float(evaluate_sequence(best.tolist()))
    # If somehow final score is not finite (shouldn't happen), fallback to uniform.
    if not np.isfinite(final_score):
        best = np.full(n, target_sum / n, dtype=np.float64)

    return [float(x) for x in best]

import numpy as np
import math

def _convolution_max(a):
    n = a.shape[0]
    nfft = 2 * n - 1
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:nfft]
    return conv.max()

def _score(a):
    s = a.sum()
    if s == 0.0:
        return np.inf
    maxb = _convolution_max(a)
    n = a.shape[0]
    return 2.0 * n * maxb / (s * s)

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)

    # Obtain initial parent if present
    try:
        height_sequence_1
    except NameError:
        height_sequence_1 = None

    if height_sequence_1 is None:
        n = 256
        parent = np.ones(n, dtype=np.float64)
    else:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    n = parent.size

    target_sum = math.sqrt(2.0 * n)

    # start from uniform distribution (guaranteed finite score)
    best = np.full(n, target_sum / n, dtype=np.float64)
    best_score = _score(best)

    # also consider the given parent as a possible start
    parent_score = _score(parent)
    if parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # helper to generate an anti‑symmetric random candidate
    def _random_alternating():
        vals = rng.random(n) + 1e-12
        vals.sort()
        vals = vals[::-1]               # descending
        out = np.empty(n, dtype=np.float64)
        lo, hi = 0, n - 1
        toggle = True
        for v in vals:
            if toggle:
                out[lo] = v
                lo += 1
            else:
                out[hi] = v
                hi -= 1
            toggle = not toggle
        # scale to the target L1 norm
        s = out.sum()
        if s > 0:
            out *= target_sum / s
        return out

    # helper for a simple mass‑transfer move that preserves the sum
    def _mass_transfer(x):
        y = x.copy()
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * y[i] * 0.25   # move at most 25 % of source
        y[i] -= delta
        y[j] += delta
        return y

    # main stochastic search
    for step in range(num_steps):
        # periodically re‑sample a fresh random pattern
        if step % 30 == 0:
            cand = _random_alternating()
        else:
            cand = _mass_transfer(best)

        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if sc < best_score:

            best_score = sc
            # early stop if we already beat the target comfortably
            if best_score < 1.6:
                # continue a few more iterations to maybe improve further
                pass

    # final safety check using the official evaluator
    final_val = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_val):

import math
import numpy as np

def _convolution_max(a: np.ndarray) -> float:
    """Return max value of the linear convolution a*a (exact, O(n log n))."""
    n = a.shape[0]
    size = 2 * n - 1
    # FFT‑based convolution (real input → real output)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:size]
    return float(conv.max())


def _sequence_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) / (sum a)^2."""
    s = a.sum()
    if s == 0.0:
        return np.inf
    max_val = _convolution_max(a)
    n = a.shape[0]
    return 2.0 * n * max_val / (s * s)


def _random_alternating(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a candidate where the sorted random values are placed
    alternately at the left‑most and right‑most free positions.
    This tends to keep large values far apart.
    """
    vals = rng.random(n) + 1e-12          # strict positivity
    vals.sort()
    vals = vals[::-1]                     # descending
    out = np.empty(n, dtype=np.float64)
    lo, hi = 0, n - 1
    toggle = True
    for v in vals:
        if toggle:
            out[lo] = v
            lo += 1
        else:
            out[hi] = v
            hi -= 1
        toggle = not toggle
    total = out.sum()
    if total > 0.0:
        out *= target_sum / total
    return out


def _mass_transfer(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a random fraction (up to 25 %) of mass from one coordinate
    to another, preserving the total L1 norm.
    """
    y = x.copy()
    n = y.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    delta = rng.random() * y[i] * 0.25
    y[i] -= delta
    y[j] += delta
    return y
