    candidate = np.asarray(base, dtype=np.float64).copy()

import numpy as np

def make_block_perturbation(base, rng):
    """Return a O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    if n == 0:
        return candidate
    # block width in [1, n//12] (at least 1)
    max_w = max(2, n // 12)
    w = int(rng.integers(1, max_w + 1))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = start + w
    # multiplicative log‑normal factor (small perturbation)
    factor = float(np.exp(rng.normal(0.0, 0.08)))
    candidate[start:stop] *= factor
    # occasional mirroring to encourage symmetry
    if rng.random() < 0.33:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce problem bounds
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Produce a non‑empty list of non‑negative heights that (empirically)
    minimizes the AC1 upper bound evaluated by `evaluate_sequence`.
    """
