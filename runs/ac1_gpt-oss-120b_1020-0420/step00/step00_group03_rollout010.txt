import numpy as np

# ------------------------------------------------------------
# Helper: fast construction of a sparse 0/1 sequence whose
# convolution maximum is kept small using a greedy insertion.
# ------------------------------------------------------------
def _build_sparse_candidate(n: int, m: int, rng: np.random.Generator,
                           trials_per_step: int = 30) -> np.ndarray:
    """
    Build a length-n vector with exactly m ones.
    The greedy procedure tries a few random positions at each step
    and keeps the one that yields the smallest possible new maximum
    of the convolution (i.e. of the ordered pair count).
    Returns the resulting vector (dtype=float64).
    """
    used = np.zeros(n, dtype=bool)          # marks selected indices
    counts = np.zeros(2 * n - 1, dtype=np.int32)  # ordered‑pair count per sum
    positions = []                          # list of selected indices
    cur_max = 0

    for _ in range(m):
        best_p = None
        best_max = None

        # try a few random candidates and keep the best
        for _ in range(trials_per_step):
            p = rng.integers(0, n)
            if used[p]:
                continue

            # max after adding p:
            # diagonal contribution
            new_max = cur_max
            diag = counts[2 * p] + 1
            if diag > new_max:
                new_max = diag

            if positions:
                # sums with existing positions
                existing = np.array(positions, dtype=np.int32)
                sum_idxs = p + existing
                # each ordered pair contributes 2
                max_with_existing = (counts[sum_idxs] + 2).max()
                if max_with_existing > new_max:
                    new_max = max_with_existing

            if best_max is None or new_max < best_max:
                best_max = new_max
                best_p = p
                if best_max == cur_max:   # cannot improve further
                    break

        # if no candidate was found in the random trials (rare), pick any free slot
        if best_p is None:
            free = np.flatnonzero(~used)
            if free.size == 0:
                break
            best_p = rng.choice(free)
            best_max = cur_max

        # commit the chosen position
        used[best_p] = True
        positions.append(best_p)

        # update counts
        counts[2 * best_p] += 1                     # (p,p) ordered pair
        if len(positions) > 1:
            prev = np.array(positions[:-1], dtype=np.int32)
            sums = best_p + prev
            counts[sums] += 2                        # (p,q) and (q,p)

        cur_max = max(cur_max, best_max)

    # construct the 0/1 vector
    vec = np.zeros(n, dtype=np.float64)
    if positions:
        vec[positions] = 1.0
    return vec
