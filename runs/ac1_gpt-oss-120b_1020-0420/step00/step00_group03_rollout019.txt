import numpy as np

def _uniform_seq(n):
    """Return a non‑negative uniform sequence of length n with sum √(2 n)."""
    total = np.sqrt(2.0 * n)
    return np.full(n, total / n, dtype=np.float64)

def _perturb_conserve_sum(arr, rng):
    """Transfer a random amount of mass between two positions, keeping the total sum."""
    n = arr.shape[0]
    out = arr.copy()
    i, j = rng.choice(n, size=2, replace=False)
    # random direction of transfer
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    # how much can be moved without breaking bounds?
    max_transfer = min(out[src], 1000.0 - out[dst])
    if max_transfer <= 0.0:
        return out
    delta = rng.random() * max_transfer
    out[src] -= delta
    out[dst] += delta
    return out

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Return a list of non‑negative coefficients that (approximately) minimize
    the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size.  Use the provided height_sequence_1 if present,
    # otherwise fall back to a modest length.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256

    # Start from the theoretically optimal uniform vector.
    best = _uniform_seq(n)
    best_score = float(evaluate_sequence(best.tolist()))

    # If a parent vector is available and better, adopt it.
    if parent is not None:
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:

            best_score = parent_score

    # Simple stochastic search that conserves the total sum.
    steps = int(kwargs.get('num_steps', num_steps))
    for _ in range(steps):
        cand = _perturb_conserve_sum(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
