import numpy as np

# The evaluator is provided in the environment
# from the harness: evaluate_sequence(sequence: list) -> float

def _linear_decreasing(n: int) -> np.ndarray:
    """
    Produce a simple non‑negative step function:
    a_0 = 1, a_{n-1} = 0, linearly decreasing.
    The exact scale does not matter (the score is scale‑invariant).
    """
    return np.linspace(1.0, 0.0, n, dtype=np.float64)


def propose_candidate(seed=42, num_steps=1000, **kwargs):
    """
    Return a candidate sequence of non‑negative heights.
    The construction is a linear decreasing profile, which yields a
    theoretical AC‑1 bound of 4/3 ≈ 1.333… for any length n.
    As a safety net we also evaluate the supplied parent
    `height_sequence_1` (if present) and return the better of the two.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Build candidate from the analytic linear profile
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])       # parent is available
    except Exception:
        # Fallback length if the parent does not exist
        n = 256

    candidate = _linear_decreasing(n)
    # Clip to the evaluator limits (0 … 1000) – already satisfied
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Evaluate it
    cand_score = float(evaluate_sequence(candidate.tolist()))

    # ------------------------------------------------------------------
    # Evaluate the provided parent as a secondary option
    # ------------------------------------------------------------------
    fallback = None
    fallback_score = np.inf
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
            fallback = parent
            fallback_score = float(evaluate_sequence(parent.tolist()))
    except Exception:
        pass

    # Choose the better one
    if fallback is not None and fallback_score < cand_score:
        best = fallback
    else:
        best = candidate

    # ------------------------------------------------------------------
    # Optional cheap local refinement: random block scaling.
    # This is cheap (O(n)) and can only improve the score.
    # ------------------------------------------------------------------
    best_score = float(evaluate_sequence(best.tolist()))
    for _ in range(min(num_steps, 200)):   # limited extra attempts
        # pick a random contiguous block and scale it a little
        width = rng.integers(1, max(2, n // 50))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.07))
        proposal = best.copy()
        proposal[start:start + width] *= factor
        proposal = np.clip(proposal, 0.0, 1000.0)

        score = float(evaluate_sequence(proposal.tolist()))
        if np.isfinite(score) and score < best_score:
            best = proposal
            best_score = score

    # Return as plain list of Python floats
    return [float(x) for x in best]
