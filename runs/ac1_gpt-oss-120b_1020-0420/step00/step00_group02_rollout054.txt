import numpy as np
import math

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function sequence that minimizes the
    AC1 upper bound.  The routine builds geometric‑decay candidates
    a_i = exp(-t * i) and selects the best according to `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1.  Determine a fallback parent sequence (height_sequence_1 if available)
    # ----------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                parent = cand
        except Exception:
            parent = None
    if parent is None:
        # generic sensible fallback – all ones, length 256
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best_seq = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ----------------------------------------------------------------------
    # 2.  Prepare geometric‑decay search
    # ----------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)          # i = 0,…,n‑1

    # Rough guess of t from the target bound (t ≈ target·e/(2n))
    target_bound = 1.50274349
    t_guess = (target_bound * math.e) / (2.0 * n)

    # Search budget – number of distinct evaluations (default 200)
    max_evals = int(kwargs.get("num_steps", 200))
    if max_evals < 10:
        max_evals = 10

    # First pass: explore multiplicative factors around the guess
    factors = np.geomspace(0.2, 5.0, num=max_evals // 2)
    t_vals = np.clip(t_guess * factors, 1e-7, 0.01)

    best_t = None
    for t in t_vals:
        seq = np.exp(-t * idx)               # geometric decreasing
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_t = t

    # ----------------------------------------------------------------------
    # 3.  Local refinement around the current best t (if any)
    # ----------------------------------------------------------------------
    remaining = max_evals - len(t_vals)
    for _ in range(remaining):
        if best_t is None:
            # random t when nothing found yet
            t = rng.uniform(1e-7, 0.01)
        else:
            # multiplicative log‑normal perturbation of the best t
            t = best_t * math.exp(rng.normal(0.0, 0.12))
        if not (1e-7 < t < 0.01):
            continue
        seq = np.exp(-t * idx)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_t = t

    # ----------------------------------------------------------------------
    # 4.  Return the best candidate as a plain list of floats
    # ----------------------------------------------------------------------
    return [float(x) for x in best_seq]
