import numpy as np

def propose_candidate(seed=42, **kwargs):
    """Return a uniform non‑negative step function that attains the theoretical optimum.

    The score 2·n·max(conv)/(sum^2) is minimized (value = 2) by any constant
    sequence.  If a parent sequence ``height_sequence_1`` is present we compare
    its score and keep the better one.
    """
    # Determine the length to use.
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        n = int(parent_arr.size)
    except NameError:
        parent_arr = None
        n = 256                     # fallback length if no parent is supplied

    # Uniform candidate (value 1.0, respects the [0,1000] bound)
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform = np.clip(uniform, 0.0, 1000.0)

    best_seq = uniform
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # If a parent is available, evaluate it and keep the better of the two.
    if parent_arr is not None and parent_arr.size > 0 and np.isfinite(parent_arr).all():
        parent_seq = np.clip(parent_arr, 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent_seq.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best_seq = parent_seq
            best_score = parent_score

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_seq]
