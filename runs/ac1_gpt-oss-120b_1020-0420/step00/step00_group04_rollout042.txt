def gradient_descent_optimize(initial, num_iters=200, step=0.01, rng=None):
   a = initial.copy()
   total = a.sum()
   for it in range(num_iters):
       # compute convolution via FFT for speed
       size = 1 << ((2 * len(a) - 1).bit_length())
       fft_a = np.fft.rfft(a, n=size)
       conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*len(a)-1]
       # find max index
       max_idx = int(np.argmax(conv))
       # compute gradient
       grad = np.zeros_like(a)
       # overlapped indices
       start_i = max(0, max_idx - (len(a)-1))
       end_i = min(len(a)-1, max_idx)
       if start_i <= end_i:
           i_range = np.arange(start_i, end_i+1)
           j = max_idx - i_range
           grad[start_i:end_i+1] = -2 * a[j]
       # update
       a = a + step * grad
       # ensure nonnegative
       a = np.clip(a, 0.0, 1000.0)
       # scale to keep sum constant
       s = a.sum()
       if s > 0:
           a = a * (total / s)
       # optional early stop if no improvement in max conv
   return a

Define function improve_by_gradient(a, iterations, step):
   for it range(iterations):
       conv = conv_fft(a)
       max_idx = int(conv.argmax())
       # compute gradient vector g_i = -2 * a[max_idx - i] where index in range
       g = np.zeros_like(a)

def linear_conv(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)

    return conv

import math
import numpy as np

# --------------------------------------------------------------
# Helper: linear convolution via FFT (O(n log n) memory O(n))
# --------------------------------------------------------------
def _linear_conv_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------
# Gradient‑descent step that tries to reduce the current maximum
# convolution entry while keeping the L1‑norm (sum) unchanged.
# --------------------------------------------------------------
def _reduce_max_conv(a: np.ndarray, iters: int = 200, step: float = 0.12) -> np.ndarray:
    """Iteratively lower max(conv) while preserving non‑negativity."""
    n = a.shape[0]
    target_sum = float(a.sum()) if a.size else 1.0

    for _ in range(iters):
        conv = _linear_conv_fft(a)
        k_max = int(np.argmax(conv))

        # gradient of the maximal entry w.r.t. a
        grad = np.zeros_like(a)
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        if start_i <= end_i:
            i = np.arange(start_i, end_i + 1, dtype=int)
            j = k_max - i
            grad[start_i : end_i + 1] = -2.0 * a[j]

        # take a small step, enforce bounds and renormalise the sum
        a = a + step * grad
        np.clip(a, 0.0, 1000.0, out=a)
        s = float(a.sum())
        if s > 0.0:
            a *= (target_sum / s)

        # slowly shrink step size – helps convergence
        step *= 0.99

    return a

# Simple block perturbation (kept from the previous solution)
# --------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand[:] = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
