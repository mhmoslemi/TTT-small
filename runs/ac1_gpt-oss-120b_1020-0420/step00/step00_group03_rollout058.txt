def geometric_seq(r, n):
    # generate r^i
    exponents = np.arange(n, dtype=np.float64)
    seq = np.power(r, exponents)
    return seq

def evaluate(seq):
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed=42, num_steps=1000, **kwargs):
    rng = np.random.default_rng(seed)
    # base parent fallback
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    best = parent.copy()
    best_score = evaluate(best)

    n = best.size

    # Search over geometric bases
    # eta from 1e-5 to 1e-1 logspace
    eta_vals = np.logspace(-5, -1, num=20)  # small epsilons
    for eta in eta_vals:

        base = geometric_seq(r, n)

        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

import numpy as np

# fast, matrix‑free evaluation of the AC1 bound (identical to the official one,
# but using FFT for speed).  It is used only inside the search; the harness
# will finally call `evaluate_sequence` on the returned list.
def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    # length for a linear convolution via zero‑padding (next power of two)
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_b = np.max(conv)
    return float(2 * n * max_b / (sum_a * sum_a))


def geometric_seq(r, n):
    """return a length‑n vector a_i = r**i (r in (0,1))."""
    return np.power(r, np.arange(n, dtype=np.float64))


def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 upper bound
    is as low as possible.  The function is deterministic for a given seed.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Obtain a valid fallback parent (the provided construction).
    # ------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = fast_score(best)

    # ------------------------------------------------------------------
    # 2.  Scan a family of geometric sequences a_i = r**i, r = exp(-eta).
    #    For each generated sequence we also apply a random permutation
    #    (the ordering matters for the convolution).
    # ------------------------------------------------------------------
    eta_vals = np.logspace(-5, -1, num=20)          # eta = -log(r)
    for eta in eta_vals:
        r = np.exp(-eta)                            # r in (0,1)
        base = geometric_seq(r, n)

        # Bring the largest entry just below 1000 (scale does not affect the bound)
        if base.max() > 0.0:
            base = base * (999.0 / base.max())

        # Random permutation of the entries
        perm = rng.permutation(n)
        cand = base[perm]

        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 3.  Simple stochastic local optimisation.
    #    Two kinds of moves are used:
    #      * swap two positions,
    #      * multiplicative perturbation of a random block (as in the previous baseline).
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        if rng.random() < 0.5:
            # ---- swap two coordinates -------------------------------------------------
            i, j = rng.choice(n, 2, replace=False)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # ---- block multiplicative perturbation ------------------------------------
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            cand = best.copy()
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:stop] *= factor
            if rng.random() < 0.35:               # optional mirror symmetry
                cand = 0.5 * (cand + cand[::-1])

        # enforce the hard constraints before scoring
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # 4.  Final safety check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(final):
            raise ValueError
    except Exception:
        best = parent.copy()   # fallback to a valid construction

    return [float(x) for x in best]
