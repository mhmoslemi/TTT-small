def compute_score(arr):
    n = arr.shape[0]
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    b = np.convolve(arr, arr)
    max_b = b.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

import numpy as np

def _score(seq: np.ndarray) -> float:
    """
    Compute the same quantity that evaluate_sequence returns
    (but without the security checks).  The sequence is assumed
    to be a one‑dimensional non‑negative NumPy array.
    """
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    # full convolution; length 2n-1
    conv = np.convolve(seq, seq)
    m = conv.max()
    return float(2.0 * n * m / (s * s))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Build a candidate sequence that aims to minimise the AC1 bound.
    The strategy is:
      1. Try a family of exponentially decaying sequences a_i = exp(-eps*i)
         for a range of eps values and keep the best.
      2. Refine the best candidate with cheap block‑scale and Gaussian
         perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Base sequence – use the provided parent if it looks sensible.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.shape[0]

    # ------------------------------------------------------------------
    # 2.  Initialise the best candidate with a simple uniform vector.
    # ------------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 3.  Scan an exponential family a_i = exp(-eps * i).
    # ------------------------------------------------------------------
    eps_vals = np.geomspace(1e-4, 5e-3, num=30)  # from 0.0001 to 0.005
    idx = np.arange(n, dtype=np.float64)

    for eps in eps_vals:
        cand = np.exp(-eps * idx)               # decaying exponential
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 4.  A symmetric exponential (peak in the centre) – optional.
    # ------------------------------------------------------------------
    for eps in eps_vals:
        dist = np.minimum(idx, n - 1 - idx)
        cand = np.exp(-eps * dist)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 5.  Local random refinements (cheap block scaling + Gaussian noise).
    # ------------------------------------------------------------------
    # number of refinement steps – keeps the total work modest.
    for _ in range(200):
        cand = best.copy()

        # block multiplication with a mild random factor
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.05)))  # close to 1
        cand[start:start + width] *= factor

        # small Gaussian jitter on all entries
        cand += rng.normal(0.0, 0.01, size=n)

        # enforce the hard constraints of the evaluator
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 6.  Return a plain Python list as required by the harness.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
