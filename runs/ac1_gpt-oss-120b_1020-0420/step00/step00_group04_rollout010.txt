import numpy as np

def generate_sidon_indices(n, max_k):
    used = np.zeros(2*n, dtype=bool)
    indices = []
    for i in range(n):
        # Check self sum
        if used[i+i]:
            continue
        ok = True
        for j in indices:
            if used[i+j]:
                ok = False
                break
        if not ok:
            continue
        # Accept i
        indices.append(i)
        # Mark sums with all existing indices, including i itself
        for j in indices:  # now includes i
            used[i+j] = True
        if len(indices) >= max_k:
            break
    return indices

def build_candidate(n, high, low, high_idx):
    cand = np.full(n, low, dtype=np.float64)
    cand[high_idx] = high
    return cand

def propose_candidate(seed=42, num_variants=10, **kwargs):
    rng = np.random.default_rng(seed)
    # Obtain parent if valid
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = None
    if parent is None or parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # Generate Sidon set up to sqrt(n)

    sidon_idx = generate_sidon_indices(n, target_k)

import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a Sidon (B2) set of indices up to a prescribed size.
# ----------------------------------------------------------------------
def generate_sidon_indices(n: int, max_k: int) -> list:
    """
    Greedy construction of a Sidon set.
    Returns a list of indices (0 <= i < n) such that all pairwise sums are distinct.
    The result size does not exceed max_k.
    """
    used = np.zeros(2 * n, dtype=bool)          # used[s] == True if sum s already taken
    idx = []
    for i in range(n):
        # the sum i+i must be unused
        if used[i + i]:
            continue
        conflict = False
        for j in idx:
            if used[i + j]:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        idx.append(i)
        # mark all new sums (including i+i)
        for j in idx:          # idx now contains i as well
            used[i + j] = True
        if len(idx) >= max_k:
            break
    return idx

    cand = np.full(n, low_val, dtype=np.float64)
    if high_idx:
        cand[high_idx] = high_val
    return cand
