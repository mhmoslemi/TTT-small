import numpy as np

def _initial_length():
    """Return the length of the problem sequence."""
    try:
        # height_sequence_1 is supplied by the harness when available
        seq = height_sequence_1
        return int(seq.shape[0])
    except Exception:  # NameError or others – fallback to a modest size
        return 256


def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound used by `evaluate_sequence`.

    The search builds a sparse “spike” pattern on top of a tiny constant
    background.  By keeping the spike density low the convolution peak stays
    small while the total sum grows, yielding a low score.
    """
    rng = np.random.default_rng(seed)

    # number of entries
    n = _initial_length()

    # ----- baseline candidate (small uniform background) -----------------
    eps_baseline = 0.01  # tiny constant background; ensures sum > 0.01
    best = np.full(n, eps_baseline, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # try the provided parent if it exists – it might already be decent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    except Exception:
        pass  # no parent available, ignore

    # search parameters
    target_score = 1.50274349          # early‑stop threshold
    max_steps = int(kwargs.get('num_steps', 2000))

    # ----- stochastic search ------------------------------------------------
    for _ in range(max_steps):
        # background level (small, keeps background contribution tiny)
        eps = float(rng.uniform(0.001, 0.02))

        # amplitude of the spikes – a modest value keeps the peak bounded
        spike_val = float(rng.uniform(0.5, 2.0))

        # number of spikes – we stay far below the full length
        # a typical useful range is 30…120; adapt to problem size
        max_spikes = min(300, n // 2)
        m = int(rng.integers(30, max_spikes + 1))

        # construct candidate
        cand = np.full(n, eps, dtype=np.float64)
        idx = rng.choice(n, size=m, replace=False)
        cand[idx] = spike_val

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            if best_score < target_score:
                break  # good enough – stop early

    # safe fallback (should never be needed)
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    # return a plain Python list of floats as required
    return [float(v) for v in best]
