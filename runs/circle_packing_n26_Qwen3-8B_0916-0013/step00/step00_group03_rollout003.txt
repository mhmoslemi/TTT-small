from scipy.optimize import minimize, Bounds
from scipy.optimize import LinearConstraint, NonlinearConstraint

    spacing = 0.3

    cols = 5
    centers = []

            x = col * spacing
            y = row * spacing * np.sqrt(3) / 2  # Hexagonal grid y-spacing
            centers.append([x, y])

    def objective(radii):

    # Constraints: distance between circles >= r1 + r2
    def constraint_dist(x, r):

        # Extract centers and radii
        n = len(r)
        x_centers = x[:2*n].reshape(n, 2)
        r_values = x[2*n:]

        for i in range(n):

                dx = x_centers[i, 0] - x_centers[j, 0]
                dy = x_centers[i, 1] - x_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r_values[i] + r_values[j] - 1e-8:
                    return dist - r_values[i] - r_values[j] + 1e-8
        return 0

    def constraint_boundary(x, r):
