import scipy.optimize as optimize

    radii_guess = []

            grid.append([x, y])
            radii_guess.append(0.1)

    radii = np.array(radii_guess)

        return -np.sum(radii)  # Minimize negative sum to maximize sum of radii

    def constraint_overlap(i, j, centers, radii):

        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx*dx + dy*dy)
        return dist - (radii[i] + radii[j]) >= -1e-8

    cons = []

    # Bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(num_circles)]

    # Run optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP'
