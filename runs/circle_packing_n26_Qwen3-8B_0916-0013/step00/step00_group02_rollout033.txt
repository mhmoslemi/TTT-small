import cvxpy as cp

def run_packing():
    # Initial grid packing with one extra circle
    n = 26
    grid_size = 5
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place 25 circles in a grid and one in the center
    for i in range(grid_size):
        for j in range(grid_size):
            x = i / (grid_size - 1)
            y = j / (grid_size - 1)
            centers[i * grid_size + j] = [x, y]
    
    # Center circle
    centers[-1] = [0.5, 0.5]

    # Optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Constraints
    constraints = []
    for i in range(n):
        # Ensure circle i is within the unit square
        constraints += [
            x[i, 0] - r[i] >= -1e-12,
            x[i, 0] + r[i] <= 1 + 1e-12,
            x[i, 1] - r[i] >= -1e-12,
            x[i, 1] + r[i] <= 1 + 1e-12,
        ]
    
    # Ensure circles do not overlap

        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0]) ** 2 + (x[i, 1] - x[j, 1]) ** 2)
            constraints += [dist >= r[i] + r[j] - 1e-12]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
