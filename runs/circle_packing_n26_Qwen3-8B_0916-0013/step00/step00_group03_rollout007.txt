import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    # Define initial grid of circle centers in a hexagonal packing
    n = 26
    grid = []
    for i in range(n):
        row = i // 2
        col = i % 2
        x = col * (1 / (math.sqrt(3) * 1.5))
        y = row * (1 / (math.sqrt(3) * 1.5)) + (col * 0.5) / (math.sqrt(3) * 1.5)
        if y > 1 - 1e-6:
            y = 1 - 1e-6
        grid.append((x, y))
    centers = np.array(grid)
    
    # Initial guess for radii: all radii equal, based on hexagonal packing density
    r0 = 1 / (2 * math.sqrt(3))  # Approximate radius for hexagonal packing
    radii = np.full(n, r0)
    
    # Objective function: maximize sum of radii, so we minimize -sum(radii)
    def objective(r):
        return -np.sum(r)
    
    # Constraints
    def constraint_center_inside(r):
        return np.sum((centers[:, 0] + r) <= 1.0 + 1e-6) + np.sum(
            (centers[:, 0] - r) >= -1e-6
        ) + np.sum((centers[:, 1] + r) <= 1.0 + 1e-6) + np.sum(
            (centers[:, 1] - r) >= -1e-6
        )
    
    def constraint_no_overlap(r):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                if dist < r[i] + r[j] - 1e-6:
                    return False
        return True
    
    # Prepare the bounds for radii
    bounds = [(0, 1.0) for _ in range(n)]
    
    # Optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda r: 1.0 - np.max(r)},  # Maximum radius <= 1.0
            {'type': 'ineq', 'fun': constraint_center_inside},  # Ensure all centers are within bounds
            {'type': 'ineq', 'fun': constraint_no_overlap},     # Ensure no circle overlaps
        ],
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Extract the optimized centers and radii
    centers_opt = centers
    radii_opt = result.x

    return centers_opt, radii_opt, -result.fun
