from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def generate_hexagonal_grid(n_circles, max_x, max_y, spacing):
    centers = []
    for i in range(n_circles):
        row = i // 3
        col = i % 3
        x = col * spacing + spacing / 2
        y = row * (spacing * np.sqrt(3)) / 2 + spacing / 2
        if x > max_x or y > max_y:
            break
        centers.append([x, y])
    return np.array(centers)

def cost_function(params, centers, n_circles, max_x, max_y):
    radii = params
    if np.any(radii < 0):
        return -np.inf
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > max_x + 1e-12 or y - r < -1e-12 or y + r > max_y + 1e-12:
            return -np.inf
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                return -np.inf
    return -np.sum(radii)

def run_packing():
