from scipy.optimize import minimize
from scipy.optimize import Bounds, LinearConstraint

def run_packing():
    # Initial grid-based configuration for 25 circles (5x5 grid)
    grid_size = 5
    spacing = 1.0 / (grid_size - 1)  # Spacing between grid points
    
    # Define grid centers
    centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i * spacing
            y = j * spacing
            centers.append([x, y])
    
    # Start with same radius for all circles; constraints will adjust it
    initial_radii = np.full(len(centers), 0.1)
    
    # Objective function
    def objective(radii):
        return -np.sum(radii)  # Maximize the sum of radii -> minimize the negative sum
    
    # Constraint: distance between any two circles must be at least sum of radii
    def constraint_distance(radii):
        constraints = []
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)
    
    # Create bounds: radii >= 0
    bounds = Bounds(lb=0.0, ub=np.inf)
    
    # Solve using optimization
    result = minimize(
        objective,
        initial_radii,
        constraints=LinearConstraint(
            constraint_distance,
            lb=-np.inf,
            ub=0.0
        ),
        bounds=bounds,
        method='SLSQP'
    )
    
    # Extract final solution
    radii = result.x
    centers = np.array(centers)

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
