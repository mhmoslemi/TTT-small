import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        r = radii[i]

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def initialize_hexagonal_grid(n, spacing=0.1):
    centers = []
    radii = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * spacing + spacing / 2
        y = row * spacing * np.sqrt(3) / 2 + spacing / 2

        else:
            x = col * spacing + spacing / 2 + spacing / 2
        centers.append([x, y])
        radii.append(0.05)
    return np.array(centers), np.array(radii)

def run_packing():

    centers, radii = initialize_hexagonal_grid(n)

        return -np.sum(radii_flat)  # Minimize negative sum to maximize sum

        radii_flat = x[2 * n:]
