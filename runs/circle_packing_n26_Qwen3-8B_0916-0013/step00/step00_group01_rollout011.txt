import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Define variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: non-overlapping and boundary conditions
    constraints = []
    for i in range(n):
        # Each circle must be within the unit square
        constraints.append(centers[i, 0] - radii[i] >= 0)
        constraints.append(centers[i, 0] + radii[i] <= 1)
        constraints.append(centers[i, 1] - radii[i] >= 0)
        constraints.append(centers[i, 1] + radii[i] <= 1)
        constraints.append(radii[i] >= 0)

    for i in range(n):
        for j in range(i + 1, n):
            # Non-overlapping constraint: distance between centers >= sum of radii
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)

    # Initial guess: place circles in a staggered hexagonal pattern
    initial_centers = np.zeros((n, 2))
    max_radii = np.zeros(n)

    # First layer (near the bottom)
    for i in range(13):

        y = 0.25
        initial_centers[i, 0] = x
        initial_centers[i, 1] = y
        max_radii[i] = 0.1

    # Second layer (staggered)
    for i in range(13, 26):

        initial_centers[i, 0] = x

        max_radii[i] = 0.1

    # Set initial guess
    centers_initial = initial_centers

    # Solve with initial guess
    prob.setup(quiet=True)
    prob.solve(solver=cp.OSQP, warm_start=True, initial_point=cp.Point({'centers': centers_initial, 'radii': radii_initial}))

    if not prob.status == "optimal":
