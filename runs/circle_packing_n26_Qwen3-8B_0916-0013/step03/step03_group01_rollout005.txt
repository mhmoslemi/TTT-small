import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():

    num_clusters = 16  # 4x4 clusters

    # Define 4x4 grid layout for cluster centers
    cluster_centers = []
    cluster_radii = []
    cluster_max_radii = []

    # For each cluster, we define a bounding box and a starting radius
    for i in range(4):

            # Define cluster position (centered in a grid of 4x4 clusters)
            x = i / 3 + 0.1  # Offset to avoid edge

            cluster_radii.append(0.05)  # Initial radius
            # Max radius possible is defined based on cluster size
            cluster_max_radii.append(0.2)

    # Convert to NumPy arrays
    centers = np.array(cluster_centers)
    radii = np.array(cluster_radii)

    # Objective function: maximize sum of radii (minimize negative sum)
    def objective(radii_vector):
        return -np.sum(radii_vector)

    # Bounds: each radius must be between 0 and max_radii[i]
    bounds = [(0, cluster_max_radii[i]) for i in range(num_clusters)]
