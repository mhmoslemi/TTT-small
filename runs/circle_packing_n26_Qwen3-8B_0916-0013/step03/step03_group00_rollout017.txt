import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26
    
    # Define grid positions using a hexagonal tiling strategy
    def generate_hexagonal_grid(num_circles):
        grid = []
        for i in range(num_circles):
            r = 0.25  # Spacing
            if i % 2 == 0:
                x = i * r
                y = 0.5
            else:
                x = (i - 1) * r + r / 2
                y = 0.5
            grid.append((x, y))
        return np.array(grid)

    # Generate initial hexagonal grid and then randomly perturb it to avoid overlaps
    centers = generate_hexagonal_grid(num_circles)
    perturbation = np.random.uniform(-0.05, 0.05, (num_circles, 2))
    centers += perturbation

    # Ensure centers are within [0, 1]^2
    centers = np.clip(centers, 0, 1)

    # Define objective function to maximize the sum of radii
    def objective(radii_vector):
        return -np.sum(radii_vector)

    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(num_circles)]

    # Define constraints
    constraints = []

    # Constraint: Circle i must not go outside the square
    for i in range(num_circles):
        def constraint_boundary(r):
            x, y = centers[i]
            return np.min([x - r[i], y - r[i], 1 - x - r[i], 1 - y - r[i]])
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary
        })

    # Constraint: Circle i and j must not overlap
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def constraint_overlap(r):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                return dist - (r[i] + r[j])
            constraints.append({
                'type': 'ineq',
                'fun': constraint_overlap
            })

    # Initial guess: small uniform radii
    initial_guess = np.full(num_circles, 0.05)

    # Optimize
    result = opt.minimize(

        bounds=bounds,
