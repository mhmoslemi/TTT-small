import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Define number of circles
    n_circles = 26

    # Initial guess: place circles in a hexagonal grid
    # First row: 5 circles wide
    # Second row: 5 circles wide
    # Third row: 5 circles wide
    # Fourth row: 6 circles wide (to fill the square)
    # Fifth row: 1 circle (center)
    # Total: 5 + 5 + 5 + 6 + 1 = 22, need 4 more

    # First, place the base structure
    initial_centers = []
    radii = []
    
    # First 4 rows: hexagonal packing
    for i in range(3):
        for j in range(5):
            x = j + (i % 2) * 0.5
            y = i * np.sqrt(3)/2
            initial_centers.append([x, y])
    
    # Add 6 circles in the 4th row
    for j in range(6):
        x = j + (0.5) * 0.5
        y = 3 * np.sqrt(3)/2
        initial_centers.append([x, y])
    
    # Add a central circle
    initial_centers.append([0.5, 3 * np.sqrt(3)/2 + 0.5])
    
    # Now add 4 more circles in the remaining space
    # Top right corner
    initial_centers.append([0.9, 0.9])
    # Bottom left corner
    initial_centers.append([0.1, 0.1])
    # Bottom right corner
    initial_centers.append([0.9, 0.1])
    # Top left corner
    initial_centers.append([0.1, 0.9])
    
    # Convert to numpy array
    initial_centers = np.array(initial_centers)
    radii = [0.0] * len(initial_centers)
    
    # Create a function to calculate the total sum of radii
    def objective(radii):
        return -np.sum(radii)  # We minimize the negative to maximize

    def constraints(radii):
        # Check for non-negativity
        if np.any(radii < 0):
            return False
        
        # Check for boundary constraints
        for i in range(len(radii)):
            x, y = initial_centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        
        # Check for non-overlapping
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        
        return True
    
    # Create bounds for radii
    bounds = [(0, 1) for _ in range(len(radii))]
    
    # Run optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={'type': 'constraint', 'fun': constraints, 'jac': None},
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 500}
    )
    
    # Extract the optimal radii
    optimal_radii = result.x
    optimal_centers = initial_centers
    
    # Return centers, radii, and total sum
    return (optimal_centers, optimal_radii, np.sum(optimal_radii))
