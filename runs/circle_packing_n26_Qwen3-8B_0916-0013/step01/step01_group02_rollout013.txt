import numpy as np
import itertools
import random

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def generate_initial_population(pop_size, n_circles):
    population = []
    for _ in range(pop_size):
        centers = np.random.rand(n_circles, 2)
        radii = np.random.rand(n_circles) * 0.5
        population.append((centers, radii))
    return population

def fitness(centers, radii):
    if not validate_packing(centers, radii)[0]:
        return 0.0
    return np.sum(radii)

def tournament_selection(population, tournament_size=3):
    selected = random.sample(population, tournament_size)
    best = max(selected, key=lambda x: fitness(x[0], x[1]))
    return best

def crossover(parent1, parent2):
    centers1, radii1 = parent1
    centers2, radii2 = parent2
    child_centers = np.zeros_like(centers1)
    child_radii = np.zeros_like(radii1)
    
    for i in range(centers1.shape[0]):
        if random.random() < 0.5:
            child_centers[i] = centers1[i]
            child_radii[i] = radii1[i]
        else:
            child_centers[i] = centers2[i]
            child_radii[i] = radii2[i]
    
    return (child_centers, child_radii)

def mutate(individual, mutation_rate=0.1, mutation_strength=0.1):
    centers, radii = individual
    for i in range(centers.shape[0]):
        if random.random() < mutation_rate:
            centers[i] += np.random.uniform(-mutation_strength, mutation_strength, 2)
            centers[i] = np.clip(centers[i], 0, 1)
        if random.random() < mutation_rate:
            radii[i] += np.random.uniform(-mutation_strength, mutation_strength)
            radii[i] = np.clip(radii[i], 0, 0.5)
    return (centers, radii)

def run_packing():
    n_circles = 26
    pop_size = 100
    generations = 100
    mutation_rate = 0.1
    mutation_strength = 0.1

    population = generate_initial_population(pop_size, n_circles)

    for _ in range(generations):
        new_population = []
        for _ in range(pop_size):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate, mutation_strength)
            new_population.append(child)
        population = new_population

    best_individual = max(population, key=lambda x: fitness(x[0], x[1]))
    best_centers, best_radii = best_individual
    best_sum = fitness(best_centers, best_radii)
    return best_centers, best_radii, best_sum
