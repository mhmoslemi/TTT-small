import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    # Clusters:
    # - First cluster (row 1): 7 circles
    # - Second cluster (row 2): 7 circles
    # - Third cluster (row 3): 6 circles
    # - Fourth cluster (row 4): 6 circles

    # Helper: Construct initial positions (hexagonal packing)
    def hexagonal_positions(n_rows, n_cols, x_offset, y_offset, r):
        positions = np.empty((n_rows * n_cols, 2))
        for i in range(n_rows):
            for j in range(n_cols):
                x = x_offset + j * (2 * r)
                y = y_offset + i * (np.sqrt(3) * r)
                positions[i * n_cols + j] = [x, y]
        return positions

    # Sample starting cluster configurations
    clusters = [
        {"row": 0, "cols": 7, "x0": 0.1, "y0": 0.2, "r0": 0.1},
        {"row": 1, "cols": 7, "x0": 0.1, "y0": 0.5, "r0": 0.1},
        {"row": 2, "cols": 6, "x0": 0.1, "y0": 0.8, "r0": 0.1},
        {"row": 3, "cols": 6, "x0": 0.1, "y0": 0.9, "r0": 0.1}
    ]

    # Build initial point positions
    points = []
    for cluster in clusters:
        cluster_points = hexagonal_positions(
            cluster["row"], cluster["cols"], cluster["x0"], cluster["y0"], cluster["r0"]
        )
        points.append(cluster_points)
    initial_positions = np.vstack(points)

    # Define the initial radii
    initial_radii = np.full(len(initial_positions), 0.1)

    # Objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize the sum

    # Define constraints to ensure no circle is outside the unit square
    def constraint1(x):
        return np.all((x[:, 0] - x_radii) >= 0) & np.all((x[:, 1] - x_radii) >= 0) & np.all(
            (1 - x[:, 0] - x_radii) >= 0
        ) & np.all((1 - x[:, 1] - x_radii) >= 0)

    # Define a constraint to ensure no two circles overlap
    def constraint2(x):
        r = x[1:]  # radii
        dists = cdist(x[:2].reshape(-1, 2), x[:2].reshape(-1, 2))
        return np.all(dists >= r[:, np.newaxis] + r[np.newaxis, :])

    # Combine all constraints
    def constraints(x):
        x_radii = x[1:]  # Assume x[0] is positions[:], x[1:] are radii
        positions = x[:2]
        return (
            np.all((positions[:, 0] - x_radii) >= 0)
            & np.all((positions[:, 1] - x_radii) >= 0)
            & np.all((1 - positions[:, 0] - x_radii) >= 0)
            & np.all((1 - positions[:, 1] - x_radii) >= 0)
        )

    # Run the optimization
    result = minimize(
        objective,
        np.concatenate((initial_positions.flatten(), initial_radii)),
        method="SLSQP",
        bounds=[(0, 1) for _ in range(2 * 26)] + [(0, 1) for _ in range(26)],
        constraints={"type": "ineq", "fun": constraints},
        options={"ftol": 1e-8, "disp": False},

    centers = result.x[:2 * 26].reshape(26, 2)
    radii = result.x[2 * 26:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
