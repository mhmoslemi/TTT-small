import scipy.optimize as optimize

def run_packing():

            centers.append([x, y])

    extra_positions = [(0.7, 0.7), (0.3, 0.3), (0.8, 0.3), (0.2, 0.7), (0.5, 0.5)]

    def inequality_constraints(radii):
