import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Start with a grid-like configuration for 26 circles
    # Define the number of circles and dimensions
    n = 26
    grid_rows = 5
    grid_cols = 6

    # Grid configuration
    grid_centers = []
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j / (grid_cols - 1) if grid_cols > 1 else 0.5
            y = i / (grid_rows - 1) if grid_rows > 1 else 0.5
            grid_centers.append([x, y])

    # Initial guess for radii - uniform scaling
    radii = np.ones(n) * 0.1

    # Define the cost function to minimize (negative sum of radii)
    def cost(radii):
        total_sum = -np.sum(radii)

        # Boundary constraint: circle must be within the unit square
        for i in range(n):
            x, y = grid_centers[i]
            if radii[i] < 0:
                return float('inf'), "Negative radius"
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return float('inf'), "Circle out of bounds"
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return float('inf'), "Circle out of bounds"

        # Compute pairwise distance between centers
        dist_matrix = cdist(grid_centers, grid_centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                    return float('inf'), "Circles overlap"

        return total_sum, "ok"

    # Bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(n)]

    # Optimization: minimize the negative of total radii
    res = minimize(
        cost,
        radii,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 200}
    )

    # Extract results
    if res.success:
        best_radii = res.x
        centers = np.array(grid_centers)
        total_sum = np.sum(best_radii)
        return (centers, best_radii, total_sum)
    else:
        raise ValueError(f"Optimization failed: {res.message}")
