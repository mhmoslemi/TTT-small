import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26

    def generate_hexagonal_centers():
        grid_x = np.linspace(0.1, 0.9, 5)
        grid_y = np.linspace(0.1, 0.9, 5)
        centers = []
        for i in range(5):
            for j in range(5):
                centers.append([grid_x[i], grid_y[j]])

        for i in range(5):
            if i % 2 == 0:
                centers.append([grid_x[i] - 0.1, grid_y[-1] - 0.1])
            else:
                centers.append([grid_x[i] + 0.1, grid_y[-1] - 0.1])

        centers = np.array(centers)
        unique_centers = []
        seen = set()
        for p in centers:
            p_tuple = tuple(p)
            if p_tuple not in seen:
                seen.add(p_tuple)
                unique_centers.append(p)
        unique_centers = np.array(unique_centers)
        return unique_centers[:26]

    def generate_square_grid_centers(n):
        size = int(np.ceil(np.sqrt(n)))
        centers = []
        for i in range(size):
            for j in range(size):
                if i * size + j < n:
                    x = i / (size - 1)
                    y = j / (size - 1)
                    centers.append([x, y])
        return np.array(centers)

    def generate_subdivided_square_centers(n):
        subdivisions = int(np.ceil(np.sqrt(n)))
        centers = []
        for i in range(subdivisions):
            for j in range(subdivisions):
                if i * subdivisions + j < n:
                    x = i / (subdivisions - 1)
                    y = j / (subdivisions - 1)
                    centers.append([x, y])
        return np.array(centers)

    def generate_corner_based_centers(n):
        corners = np.array([
            [0.1, 0.1],
            [0.9, 0.1],
            [0.9, 0.9],
            [0.1, 0.9]
        ])
        remaining = n - 4
        if remaining < 0:
            return corners[:n]
        grid_x = np.linspace(0.1, 0.9, 5)
        grid_y = np.linspace(0.1, 0.9, 5)
        centers = []
        for i in range(5):
            for j in range(5):
                if i * 5 + j < remaining:
                    centers.append([grid_x[i], grid_y[j]])
        centers = np.array(centers)
        unique_centers = []
        seen = set()
        for p in centers:
            p_tuple = tuple(p)
            if p_tuple not in seen:
                seen.add(p_tuple)
                unique_centers.append(p)
        unique_centers = np.array(unique_centers)
        full_centers = np.vstack([corners, unique_centers])
        full_centers = np.unique(full_centers, axis=0)
        return full_centers[:n]

    def run_optimization(initial_centers):
        centers = initial_centers
        n = centers.shape[0]
        bounds = [(0, 1) for _ in range(n)]

        def objective(radii_vector):
            return -np.sum(radii_vector)

        constraints = []

        def constraint_overlap(i, j):
            def fun(r):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                return dist - (r[i] + r[j])
            return {'type': 'ineq', 'fun': fun}

        def constraint_boundary(i):
            def fun(r):
                x, y = centers[i]
                r_val = r[i]
                return np.min([x - r_val, y - r_val, 1 - x - r_val, 1 - y - r_val])
            return {'type': 'ineq', 'fun': fun}

    return best_centers, best_radii, best_sum_radii
