import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def generate_hexagonal_centers_with_corners(num_circles):
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    centers = []
    for i in range(5):
        for j in range(5):
            centers.append([grid_x[i], grid_y[j]])

    for i in range(5):
        if i % 2 == 0:
            centers.append([grid_x[i] - 0.1, grid_y[-1] - 0.1])
        else:
            centers.append([grid_x[i] + 0.1, grid_y[-1] - 0.1])

    centers = np.array(centers)
    unique_centers = []
    seen = set()
    for p in centers:
        p_tuple = tuple(p)
        if p_tuple not in seen:
            seen.add(p_tuple)
            unique_centers.append(p)
    unique_centers = np.array(unique_centers)

    # Add corner circles
    corner_points = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]
    ])
    unique_centers = np.vstack((unique_centers, corner_points))

    # Select up to num_circles
    return unique_centers[:num_circles]

def run_optimization(initial_centers):
    centers = initial_centers
    n = centers.shape[0]
    bounds = [(0, 1) for _ in range(n)]

    def objective(radii_vector):
        return -np.sum(radii_vector)

    constraints = []

    def constraint_overlap(i, j):
        def fun(r):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            return dist - (r[i] + r[j])
        return {'type': 'ineq', 'fun': fun}

    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            r_val = r[i]
            return 1 - r_val - x - y  # ensures circle stays within square
        return {'type': 'ineq', 'fun': fun}

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(constraint_overlap(i, j))

    for i in range(n):
        constraints.append(constraint_boundary(i))

    initial_guess = np.full(n, 0.05)

    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 300, 'ftol': 1e-10, 'iprint': 0}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    radii = result.x
    return centers, radii

def run_packing():

    centers, radii = run_optimization(initial_centers)
    valid, msg = validate_packing(centers, radii)
    if not valid:

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
