import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():

    # Create multiple initial configurations to explore
    def generate_clustered_centers():

        # Create concentric rings around the center of the square
        ring_radius = np.linspace(0.1, 0.7, 10)

                y = r * np.sin(angle)
                centers.append([x, y])
        # Add random points to fill gaps
        for _ in range(26 - len(centers)):
            x = np.random.uniform(0, 1)
            y = np.random.uniform(0, 1)
            centers.append([x, y])
        centers = np.array(centers)
        return centers

    def generate_sparse_centers():
        # Create random sparse points in [0.1, 0.9]^2

        return np.array(list(zip(x, y)))

        generate_ring_centers(),
        generate_sparse_centers()
    ]

    best_centers = None
    best_radii = None

    for initial_centers in initial_centers_list:

            constraints = []

                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

                return {'type': 'ineq', 'fun': fun}
