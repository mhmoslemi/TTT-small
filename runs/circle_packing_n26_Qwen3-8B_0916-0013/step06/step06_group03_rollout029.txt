import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def generate_hexagonal_packing_with_boundaries(num_circles):
    # Start with a 6x4 hexagonal grid, as 6x4 = 24
    # Then add two circles at the corners
    grid_x = np.linspace(0.1, 0.9, 6)
    grid_y = np.linspace(0.1, 0.9, 4)
    centers = []
    for i in range(4):
        for j in range(6):
            centers.append([grid_x[j], grid_y[i]])

    centers.append([0.1, 0.1])
    centers.append([0.9, 0.1])
    centers.append([0.1, 0.9])
    centers.append([0.9, 0.9])

    # Remove duplicate centers
    unique_centers = []

    for p in centers:
        p_tuple = tuple(p)
        if p_tuple not in seen:
            seen.add(p_tuple)
            unique_centers.append(p)
    unique_centers = np.array(unique_centers)
    return unique_centers[:num_circles]

def optimize_packing(initial_centers):
    n = initial_centers.shape[0]
    bounds = [(0, 1) for _ in range(n)]

    def objective(radii_vector):
        return -np.sum(radii_vector)

    constraints = []

    def constraint_overlap(i, j):
        def fun(r):
            dist = np.sqrt(np.sum((initial_centers[i] - initial_centers[j]) ** 2))
            return dist - (r[i] + r[j])
        return {'type': 'ineq', 'fun': fun}

    def constraint_boundary(i):
        def fun(r):
            x, y = initial_centers[i]
            r_val = r[i]
            return np.min([x - r_val, y - r_val, 1 - x - r_val, 1 - y - r_val])
        return {'type': 'ineq', 'fun': fun}

            constraints.append(constraint_overlap(i, j))

        x0=initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'iprint': 0}

    radii = result.x
