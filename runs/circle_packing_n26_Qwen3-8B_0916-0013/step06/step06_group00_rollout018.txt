import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26

    def generate_hexagonal_centers():
        h_size = 7
        grid_x = np.linspace(0.1, 0.9, h_size)
        grid_y = np.linspace(0.1, 0.9, h_size)
        centers = []
        for i in range(h_size):
            for j in range(h_size):
                centers.append([grid_x[i], grid_y[j]])

        # Add offset rows for hexagonal packing
        for i in range(h_size):
            if i % 2 == 0:
                centers.append([grid_x[i] - 0.1, grid_y[-1] - 0.1])
            else:
                centers.append([grid_x[i] + 0.1, grid_y[-1] - 0.1])

        # Deduplicate and limit to 26
        centers = np.array(centers)
        unique_centers = []
        seen = set()
        for p in centers:
            p_tuple = tuple(p)
            if p_tuple not in seen:
                seen.add(p_tuple)
                unique_centers.append(p)
        unique_centers = np.array(unique_centers)
        return unique_centers[:26]

    def generate_random_centers():
        return np.random.rand(num_circles, 2)

    def generate_skewed_centers():
        centers = np.random.rand(num_circles, 2)
        centers[:, 0] *= 0.5
        centers[:, 1] *= 1.5
        return centers

    def run_optimization(initial_centers):
        centers = initial_centers
        n = centers.shape[0]
        bounds = [(0, 1) for _ in range(n * 2)]

        def objective(radii_vector):
            return -np.sum(radii_vector)

        constraints = []

        for i in range(n):
            def constraint_boundary(i):
                def fun(r):
                    x, y = centers[i]
                    r_val = r[i]
                    return np.min([x - r_val, y - r_val, 1 - x - r_val, 1 - y - r_val])
                return {'type': 'ineq', 'fun': fun}
            constraints.append(constraint_boundary(i))

        for i in range(n):
            for j in range(i + 1, n):
                def constraint_overlap(i, j):
                    def fun(r):
                        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                        return dist - (r[i] + r[j])
                    return {'type': 'ineq', 'fun': fun}
                constraints.append(constraint_overlap(i, j))

        initial_guess = np.full(n, 0.05)

        result = opt.minimize(
            fun=objective,
            x0=initial_guess,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
        )

        if not result.success:
            raise ValueError("Optimization failed")

        radii = result.x
        return centers, radii

    initial_centers_list = [
        generate_hexagonal_centers(),
        generate_random_centers(),
        generate_skewed_centers()
    ]

    best_sum_radii = -1
    best_centers = None
    best_radii = None

    for initial_centers in initial_centers_list:
        try:
            centers, radii = run_optimization(initial_centers)
            valid, msg = validate_packing(centers, radii)
            if not valid:
                raise ValueError(f"Validation failed: {msg}")
            sum_radii = np.sum(radii)
            if sum_radii > best_sum_radii:
                best_sum_radii = sum_radii
                best_centers = centers
                best_radii = radii
        except Exception as e:
            print(f"Optimization failed with error: {e}")

    return best_centers, best_radii, best_sum_radii
