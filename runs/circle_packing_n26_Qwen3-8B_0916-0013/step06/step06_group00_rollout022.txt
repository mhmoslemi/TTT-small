import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26

    def generate_hexagonal_centers():
        h_size = 5

                centers.append([grid_x[i] + 0.1, grid_y[-1] - 0.1])

        centers = np.array(centers)

        for p in centers:
            p_tuple = tuple(p)
            if p_tuple not in seen:
                seen.add(p_tuple)
                unique_centers.append(p)
        unique_centers = np.array(unique_centers)
        return unique_centers[:26]

        return np.random.rand(num_circles, 2) * 0.9 + 0.05

    def run_optimization(initial_centers):
        centers = initial_centers
        n = centers.shape[0]
        bounds = [(0, 1) for _ in range(n)]

        def objective(radii_vector):
            return -np.sum(radii_vector)

        constraints = []

        for i in range(n):
            def constraint_boundary(i):
                def fun(r):
                    x, y = centers[i]
                    r_val = r[i]
                    return np.min([x - r_val, y - r_val, 1 - x - r_val, 1 - y - r_val])
                return {'type': 'ineq', 'fun': fun}
