import cvxpy as cp

def run_packing():
    num_circles = 26
    # Define 26 guide points within the square to help with circle packing
    # Manually selected guide points for better distribution
    guide_points = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])

    # Extract the first 26 guide points
    guide_points = guide_points[:26]

    # Define problem variables
    centers = cp.Variable((num_circles, 2))
    radii = cp.Variable(num_circles, non_negative=True)

    # Define objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints:

    # 1. Each circle must be fully within the unit square
    constraints = [
        cp.Square(centers[i, 0] - radii[i]) <= 1 for i in range(num_circles),
        cp.Square(centers[i, 1] - radii[i]) <= 1 for i in range(num_circles),
        cp.Square(1 - centers[i, 0] - radii[i]) <= 1 for i in range(num_circles),
        cp.Square(1 - centers[i, 1] - radii[i]) <= 1 for i in range(num_circles)
    ]

    # 2. Circles must not overlap with each other
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraint = cp.sqrt(cp.sum_squares(centers[i] - centers[j])) >= radii[i] + radii[j]
            constraints.append(constraint)

    # 3. Ensure circles are near the guide points (optional soft constraint)
    for i in range(num_circles):
        constraint = cp.sqrt(cp.sum_squares(centers[i] - guide_points[i])) <= 0.5 + radii[i]
        constraints.append(constraint)

    # Create problem and solve
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.OSQP, eps_abs=1e-8, eps_rel=1e-8, verbose=False)

    if not problem.status == "optimal":
        raise ValueError(f"Optimization failed: {problem.status}")

    # Extract results
    centers = centers.value
    radii = radii.value
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
