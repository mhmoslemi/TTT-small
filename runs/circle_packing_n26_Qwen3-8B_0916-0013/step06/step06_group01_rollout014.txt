import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def generate_hexagonal_grid(n):
    """Generate a hexagonal (close-packed) grid of centers in a unit square."""
    max_row = int(np.ceil(np.sqrt(n)))
    max_col = int(np.ceil(np.sqrt(n)))
    centers = []
    for row in range(max_row):
        for col in range(max_col):
            if row * max_col + col < n:
                x = col / (max_col - 1)
                y = row / (max_row - 1)
                centers.append([x, y])
    # Improve spacing by adding offsets for better packing
    centers = np.array(centers)
    centers = np.vstack([centers, np.array([[0.5, 0.9], [0.5, 0.85], [0.5, 0.8], [0.5, 0.75], [0.5, 0.7]])])
    return centers[:26]

def run_optimization(initial_centers):
    centers = initial_centers
    n = centers.shape[0]
    bounds = [(0, 1) for _ in range(n)]

    def objective(radii_vector):
        return -np.sum(radii_vector)

        def fun(r):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            return dist - (r[i] + r[j])
        return {'type': 'ineq', 'fun': fun}

            x, y = centers[i]
            r_val = r[i]
