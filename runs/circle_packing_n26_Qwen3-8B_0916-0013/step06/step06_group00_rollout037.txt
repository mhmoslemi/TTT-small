import numpy as np
import scipy.optimize as opt
import cvxpy as cp

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26

    def generate_hexagonal_centers():
        h_size = 6
        grid_x = np.linspace(0.1, 0.9, h_size)
        grid_y = np.linspace(0.1, 0.9, h_size)
        centers = []
        for i in range(h_size):
            for j in range(h_size):
                centers.append([grid_x[i], grid_y[j]])
        for i in range(h_size):
            if i % 2 == 0:
                centers.append([grid_x[i] - 0.1, grid_y[-1] - 0.1])
            else:
                centers.append([grid_x[i] + 0.1, grid_y[-1] - 0.1])
        centers = np.array(centers)
        unique_centers = []
        seen = set()
        for p in centers:
            p_tuple = tuple(p)
            if p_tuple not in seen:
                seen.add(p_tuple)
                unique_centers.append(p)
        unique_centers = np.array(unique_centers)
        return unique_centers[:26]

    def generate_random_centers():
        return np.random.rand(num_circles, 2)

    def build_initial_centers():
        hex_centers = generate_hexagonal_centers()
        random_centers = generate_random_centers()
        cluster_centers = np.random.rand(num_circles, 2) * 0.5 + 0.5
        return [hex_centers, random_centers, cluster_centers]

    def optimize_packing(initial_centers):
        n = num_circles
        centers = initial_centers
        radii = cp.Variable(n)
        constraints = []
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r >= -1e-12)
            constraints.append(x + r <= 1 + 1e-12)
            constraints.append(y - r >= -1e-12)
            constraints.append(y + r <= 1 + 1e-12)
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(radii[i] + radii[j] <= dist + 1e-12)

        objective = cp.Maximize(cp.sum(radii))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.SCS, eps=1e-6, max_iter=1000)
        if not problem.success:
            raise ValueError("CVXPY optimization failed")
        return centers, radii.value

    initial_centers = build_initial_centers()
    best_sum_radii = -1
    best_centers = None
    best_radii = None

    for initial in initial_centers:

                best_centers = centers
                best_radii = radii
        except Exception as e:
            print(f"Optimization failed: {e}")

    return best_centers, best_radii, best_sum_radii
