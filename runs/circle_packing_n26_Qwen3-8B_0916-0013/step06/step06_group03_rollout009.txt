import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def generate_hexagonal_centers():
    num_rows = 6
    num_cols = 6
    points = []
    for row in range(num_rows):
        for col in range(num_cols):
            x = col / (num_cols - 1)
            y = row / (num_rows - 1)
            points.append([x, y])
    # Add additional points for corners and boundary alignment
    for row in range(num_rows):
        for col in range(num_cols):
            x = col / (num_cols - 1)
            y = row / (num_rows - 1)
            if row % 2 == 0:
                x_offset = 0.05
            else:
                x_offset = 0.05
            points.append([x + x_offset, y])
    points = np.array(points)
    # Deduplicate and keep only the first 26 points
    unique_points = np.unique(points, axis=0)
    return unique_points[:26]

def run_optimization(initial_centers):
    centers = initial_centers
    n = centers.shape[0]
    bounds = [(0, 1) for _ in range(n)]

    def objective(radii_vector):
        return -np.sum(radii_vector)

    constraints = []

    def constraint_overlap(i, j):
        def fun(r):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            return dist - (r[i] + r[j])
        return {'type': 'ineq', 'fun': fun}

    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            r_val = r[i]
            return np.min([x - r_val, y - r_val, 1 - x - r_val, 1 - y - r_val])
        return {'type': 'ineq', 'fun': fun}

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(constraint_overlap(i, j))

    for i in range(n):
        constraints.append(constraint_boundary(i))

    initial_guess = np.full(n, 0.05)

    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-10, 'iprint': 0}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    radii = result.x
    return centers, radii

def run_packing():
    num_circles = 26
    initial_centers_list = [
        generate_hexagonal_centers(),

    best_centers = None
    best_radii = None

                best_centers = centers
