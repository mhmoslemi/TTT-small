import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26

    def generate_hexagonal_centers():
        grid_x = np.linspace(0.1, 0.9, 5)
        grid_y = np.linspace(0.1, 0.9, 5)

        for p in centers:
            p_tuple = tuple(p)

        return np.random.rand(num_circles, 2)

        centers = []
        for i in range(num_circles):
            angle = 2 * np.pi * i / num_circles
            r = 0.4
            x = r * np.cos(angle) + 0.5
            y = r * np.sin(angle) + 0.5
            centers.append([x, y])
        centers = np.array(centers)
        return centers

    def run_optimization(initial_centers):
        centers = initial_centers
        n = centers.shape[0]
        bounds = [(0, 1) for _ in range(n)]

            return -np.sum(radii_vector)

        constraints = []

        def constraint_overlap(i, j):
            def fun(r):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                return dist - (r[i] + r[j])
            return {'type': 'ineq', 'fun': fun}
