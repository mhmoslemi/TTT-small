import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:

    return True, "ok"

def generate_hexagonal_grid(num_rows=6, num_cols=5):
    """Generate a hexagonal grid of circles inside [0,1]x[0,1], optimized for boundary and spacing."""
    # Generate a grid with maximum spacing
    horizontal_spacing = 1.0 / (num_cols + 1)
    vertical_spacing = (1.0 / (num_rows + 1)) * np.sqrt(3) / 2  # Hexagonal spacing
    centers = []
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * horizontal_spacing + horizontal_spacing / 2
            y = row * vertical_spacing + vertical_spacing / 2
            centers.append([x, y])
        # Add offset for staggered rows
        if row % 2 == 1:
            for col in range(num_cols):
                x = (col + 0.5) * horizontal_spacing + horizontal_spacing / 2
                y = row * vertical_spacing + vertical_spacing / 2
                centers.append([x, y])
    return np.array(centers[:26])

def run_packing():
    num_circles = 26
    initial_centers_list = [
        generate_hexagonal_grid(num_rows=6, num_cols=5),
        np.random.rand(num_circles, 2),
        np.random.rand(num_circles, 2) * 0.5 + 0.5,

    best_radii = None

            return -np.sum(radii_vector)

                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

            return {'type': 'ineq', 'fun': fun}

                x, y = centers[i]
                r_val = r[i]
                return np.min([x - r_val, y - r_val, 1 - x - r_val, 1 - y - r_val])
            return {'type': 'ineq', 'fun': fun}

            for j in range(i + 1, n):
                constraints.append(constraint_overlap(i, j))
        for i in range(n):
            constraints.append(constraint_boundary(i))

        bounds = [(0, 1) for _ in range(n)]
        initial_guess = np.full(n, 0.05)

        result = opt.minimize(
            fun=objective,
            x0=initial_guess,
            bounds=bounds,
            constraints=constraints,
