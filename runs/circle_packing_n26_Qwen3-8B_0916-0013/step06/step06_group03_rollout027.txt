import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():

        # Keep only the first 26 circles and deduplicate
        centers = np.array(centers)

        for p in centers:

                seen.add(p_tuple)
                unique_centers.append(p)
        unique_centers = np.array(unique_centers)
        return unique_centers[:26]

    # Run the optimization with improved constraints and multi-start
    def run_optimization(initial_centers):
        centers = initial_centers
        n = centers.shape[0]
        bounds = [(0, 1) for _ in range(n)]

            return -np.sum(radii_vector)

                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                return dist - (r[i] + r[j])
            return {'type': 'ineq', 'fun': fun}

        def constraint_boundary(i):
