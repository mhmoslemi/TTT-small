import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def generate_hexagonal_centers(n=26):
    # Generate hexagonal lattice inside [0, 1] x [0, 1]
    # Approximate spacing: 0.25
    row_offset = 0.25 * (np.sqrt(3) / 2)
    col_offset = 0.25

    centers = []
    col = 0
    row = 0
    max_col = 3
    max_row = 3

    for i in range(max_row + 1):
        for j in range(max_col + 1):
            centers.append([j * col_offset, i * row_offset])
    
    # Generate more columns but limit to n circles
    while len(centers) < n:
        # Extend outward in both x and y
        for dx in [-col_offset, 0, col_offset]:

                x = centers[-1][0] + dx
                y = centers[-1][1] + dy
                if 0 <= x <= 1 and 0 <= y <= 1:
                    centers.append([x, y])

    # Deduplicate and keep first n
    unique_centers = []
    seen = set()
    for p in centers:
        p_tuple = tuple(p)
        if p_tuple not in seen:
            seen.add(p_tuple)
            unique_centers.append(p)
    return np.array(unique_centers[:n])

def run_optimization(initial_centers):
    centers = initial_centers
    n = centers.shape[0]
    bounds = [(0, 1) for _ in range(n * 2)]

    def objective(radii_vector):
        return -np.sum(radii_vector)

    constraints = []

        def fun(r):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            return dist - (r[i] + r[j])
        return {'type': 'ineq', 'fun': fun}

    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            r_val = r[i]
            return np.min([x - r_val, y - r_val, 1 - x - r_val, 1 - y - r_val])
        return {'type': 'ineq', 'fun': fun}

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(constraint_overlap(i, j))

    for i in range(n):
        constraints.append(constraint_boundary(i))

    initial_guess = np.full(n, 0.05)

    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1500, 'ftol': 1e-10, 'iprint': 0}

        raise ValueError("Optimization failed")

    radii = result.x
    return centers, radii
