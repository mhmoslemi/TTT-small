import scipy.optimize as opt

def validate_packing(centers, radii):

        return False, "NaN values present"

            if dist < radii[i] + radii[j] - 1e-12:

    return True, "ok"

def run_packing():

    def hexagonal_grid(n, x_min=0.1, x_max=0.9, y_min=0.1, y_max=0.9):

            x = x_min + (col + 0.5) * (x_max - x_min) / (np.sqrt(3) - 0.5)
