import numpy as np
import scipy.optimize as opt
import random

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26
    best_sum_radii = -1
    best_centers = None
    best_radii = None

    def generate_random_centers():
        return np.random.rand(num_circles, 2)

    def generate_corner_centers():
        corners = [(0.1, 0.1), (0.1, 0.9), (0.9, 0.9), (0.9, 0.1)]
        centers = []
        for i in range(num_circles):
            if i % 4 == 0:
                centers.append(corners[i // 4])
            else:
                centers.append((0.5, 0.5))
        return np.array(centers)

    initial_center_configs = [
        generate_random_centers,
        generate_corner_centers
    ]

    for generate_centers in initial_center_configs:
        try:
            centers = generate_centers()
            bounds = [(0, 1) for _ in range(num_circles * 2)]
            initial_guess = np.random.rand(num_circles * 2) * 0.5

            def objective(radii_vector):
                return -np.sum(radii_vector)

            constraints = []

            def constraint_overlap(i, j):
                def fun(r):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    return dist - (r[i] + r[j])
                return {'type': 'ineq', 'fun': fun}

            def constraint_boundary(i):
                def fun(r):
                    x, y = centers[i]
                    r_val = r[i]
                    return np.min([x - r_val, y - r_val, 1 - x - r_val, 1 - y - r_val])
                return {'type': 'ineq', 'fun': fun}

            for i in range(num_circles):
                for j in range(i + 1, num_circles):
                    constraints.append(constraint_overlap(i, j))

            for i in range(num_circles):
                constraints.append(constraint_boundary(i))

            result = opt.minimize(
                fun=objective,
                x0=initial_guess,
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',
                options={'maxiter': 1000, 'ftol': 1e-8, 'iprint': 0}
            )

            radii = result.x

                best_centers = centers

            print(f"Optimization failed: {e}")

    return best_centers, best_radii, best_sum_radii
