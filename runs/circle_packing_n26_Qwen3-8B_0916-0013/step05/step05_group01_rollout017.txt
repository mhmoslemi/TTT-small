import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26

    def generate_hexagonal_centers():

        grid_y = np.linspace(0.1, 0.9, 5)
        centers = []
        for i in range(5):
            for j in range(5):
                centers.append([grid_x[i], grid_y[j]])
        # Add extra rows for hexagonal packing
        for i in range(5):
            if i % 2 == 0:
                centers.append([grid_x[i] - 0.1, grid_y[-1] - 0.1])
            else:
                centers.append([grid_x[i] + 0.1, grid_y[-1] - 0.1])
        centers = np.array(centers)
        unique_centers = np.unique(centers, axis=0)
        return unique_centers[:26]

    # Define helper constraints for optimization
    def get_overlap_constraints(centers):
        n = centers.shape[0]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                def constraint(r):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    return dist - (r[i] + r[j])
                constraints.append({'type': 'ineq', 'fun': constraint})
        return constraints

    def get_boundary_constraints(centers):
        n = centers.shape[0]
        constraints = []
        for i in range(n):
            def constraint(r):
                x, y = centers[i]
                r_val = r[i]
                return np.min([x - r_val, y - r_val, 1 - x - r_val, 1 - y - r_val])
            constraints.append({'type': 'ineq', 'fun': constraint})
        return constraints

        n = centers.shape[0]

        initial_guess = np.full(n, 0.05)

            fun=objective,
            x0=initial_guess,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 500, 'ftol': 1e-10, 'iprint': 0}
