import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26
    # Define hexagonal grid in the unit square, refined and near corners
    grid_size = 5
    radius_est = 0.11  # Based on finest packing assumption

    # Generate coordinates in a hexagonal grid
    def generate_hex_grid(num_points):
        coords = []
        for i in range(num_points):
            col = i % grid_size
            row = i // grid_size
            x = col * (1 / (grid_size - 1)) + (row % 2) * (1 / (2 * (grid_size - 1)))
            y = row * (1 / (grid_size - 1))
            coords.append([x, y])
        return np.array(coords)

    # Add extra circles near the corners
    corner_circles = [
        [0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]
    ]

    # Create a more refined hexagonal grid
    grid = generate_hex_grid(num_circles - len(corner_circles))
    centers = np.vstack([grid, corner_circles])

    # Ensure we only have 26 circles
    centers = centers[:26]

    # Define objective function to maximize the sum of radii
    def objective(radii_vector):
        return -np.sum(radii_vector)

    # Define variables for the optimization
    num_circles = centers.shape[0]
    bounds = [(0, 1) for _ in range(num_circles)]

    # Define constraints
    constraints = []

    # Constraint for non-overlap and boundary containment
    def constraint_overlap(i, j):
        def fun(r):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            return dist - (r[i] + r[j])
        return {'type': 'ineq', 'fun': fun}

    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            r_val = r[i]
            return x - r_val
        return {'type': 'ineq', 'fun': fun}

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append(constraint_overlap(i, j))

    for i in range(num_circles):
        constraints.append(constraint_boundary(i))

    # Initial guess: based on assumed packing efficiency
    initial_guess = np.full(num_circles, radius_est)

    # Perform optimization with enhanced settings
    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
