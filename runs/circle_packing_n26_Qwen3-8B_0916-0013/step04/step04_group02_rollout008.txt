import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26

    # Initialize centers and radii
    centers = np.random.rand(num_circles, 2)
    radii = np.full(num_circles, 0.05)

    # Define bounds for the optimization: each radius must be in [0, 1]
    bounds = [(0, 1) for _ in range(num_circles)]

    # Define constraints: non-overlapping and boundary conditions
    def constraint_overlap(i, j):
        return lambda r: np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (r[i] + r[j])

    def constraint_boundary(i):
        return lambda r: np.min([
            centers[i, 0] - r[i],
            centers[i, 1] - r[i],
            1 - centers[i, 0] - r[i],
            1 - centers[i, 1] - r[i],
        ])

    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': constraint_overlap(i, j)
            })

    for i in range(num_circles):
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary(i)
        })

    # Objective function: maximize the sum of the radii
    def objective(radii_vector):
        return -np.sum(radii_vector)

    # Optimization with improved settings
    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={
            'ftol': 1e-8,
            'maxiter': 1000,
            'disp': False
        }
    )

    radii = result.x
    centers = np.array(centers)

    # Validate the packing and raise an error if invalid
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
