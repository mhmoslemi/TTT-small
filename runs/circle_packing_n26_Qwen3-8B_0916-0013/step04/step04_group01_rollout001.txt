import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26
    spacing = 1.0 / np.sqrt(num_circles)
    
    centers = []
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = col * spacing
        y = row * spacing
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)
        centers.append([x, y])
    
    centers = np.array(centers)

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(i, j):
        return lambda r: np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (r[i] + r[j])

    def constraint_boundary(i):
        return lambda r: centers[i, 0] - r[i]
    
    def constraint_boundary_y(i):
        return lambda r: centers[i, 1] - r[i]
    
    def constraint_boundary_x_max(i):
        return lambda r: 1 - centers[i, 0] - r[i]
    
    def constraint_boundary_y_max(i):
        return lambda r: 1 - centers[i, 1] - r[i]

    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': constraint_overlap(i, j)
            })

    for i in range(num_circles):
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary(i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary_y(i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary_x_max(i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary_y_max(i)
        })

    bounds = [(0, 1) for _ in range(num_circles)]
    initial_guess = np.full(num_circles, 0.1)

    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-9, 'eps': 1e-8}
    )

    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")

    radii = result.x
    centers = np.array(centers)

    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
