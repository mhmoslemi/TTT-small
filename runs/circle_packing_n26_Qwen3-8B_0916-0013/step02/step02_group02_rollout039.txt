import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():

    # 16 circles: 4x4 hexagonal grid (approx)
    grid_centers = []
    for i in range(4):
        for j in range(4):
            x = i + 0.5 * j / 3
            y = j + 0.5 * i / 3
            grid_centers.append([x, y])

    # 10 boundary circles: 4 corners and 6 on edges
    boundary_centers = [
        [0.1, 0.1],   # bottom-left
        [0.9, 0.1],   # bottom-right
        [0.1, 0.9],   # top-left
        [0.9, 0.9],   # top-right

        [0.9, 0.5],   # right
        [0.25, 0.25], # bottom-left corner
        [0.75, 0.25], # bottom-right corner
    ]
    initial_centers = grid_centers + boundary_centers

    # Initial guess for radii: equal to the minimum distance to the boundary or adjacent circles
    initial_radii = []
    for c in initial_centers:
        # Minimum distance to boundary
        min_dist_to_boundary = min(c[0], 1 - c[0], c[1], 1 - c[1])
        # Minimum distance to other centers
        dists = [np.hypot(c[0] - other[0], c[1] - other[1]) for other in initial_centers]
        min_other_dist = min(dists)
        r = min(min_dist_to_boundary, min_other_dist / 2)
        initial_radii.append(r)

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii, so minimize negative sum

    def constraint_dist(x, y, r, i, j):
        return np.hypot(x[i][0] - x[j][0], x[i][1] - x[j][1]) - (r[i] + r[j])

    def constraint_boundary(x, r, i):
        return min(x[i][0] - r[i], x[i][1] - r[i], 1 - x[i][0] - r[i], 1 - x[i][1] - r[i])

    # Transform variables: x = [x1, y1, x2, y2, ..., x26, y26], r = [r1, r2, ..., r26]
    # Where x[0], x[1] is center of circle 1, x[2], x[3] is center of circle 2, etc.

    bounds = []

    # Initial guess
    x0 = []
    for center in initial_centers:

    x0.extend(initial_radii)

    # Optimization
    result = minimize(

        x0,
        bounds=bounds,
        constraints=[
            {"type": "ineq", "fun": constraint_dist, "args": (i, j) for i in range(26) for j in range(i + 1, 26)},

        method='SLSQP'
    )

    # Extract centers and radii
    centers = np.array([result.x[2 * i : 2 * i + 2] for i in range(26)])

    # Compute the sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
