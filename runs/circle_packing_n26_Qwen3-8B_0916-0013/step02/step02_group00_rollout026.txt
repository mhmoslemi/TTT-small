import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26
    # Create hexagonal packing layout
    centers = []
    radii = []
    
    # Calculate the hexagonal grid
    for row in range(5):
        for col in range(5):
            x = col / 4
            y = row / 4
            # For odd rows, offset by half a column width
            if row % 2 == 1:
                x += 0.5
            centers.append([x, y])
    
    # Add the 26th circle at the center
    centers.append([0.5, 0.5])
    
    # Convert to numpy array
    centers = np.array(centers)

    def objective(radii_vector):

    # Define constraints
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            # Constraint to ensure circles do not overlap
            def constraint_overlap(r, i=i, j=j):
                return np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (r[i] + r[j])
            constraints.append({
                'type': 'ineq',
                'fun': constraint_overlap
            })
    
    for i in range(num_circles):
        # Constraint to ensure circle stays within the unit square
        def constraint_boundary(r, i=i):
            x, y = centers[i]
            r_val = r[i]
            return np.min([
                x - r_val,
                y - r_val,
                1 - x - r_val,
                1 - y - r_val
            ])
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary
        })
    
    # Define bounds for radius values
    bounds = [(0, 1) for _ in range(num_circles)]

    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    radii = result.x
    centers = np.array(centers)
    
    # Validate the packing

    if not valid:
        raise ValueError(f"Validation failed: {msg}")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
