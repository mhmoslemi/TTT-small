from scipy.optimize import minimize

def run_packing():

    def initial_guess(num_circles):
        rows = int(np.sqrt(num_circles))
        cols = rows
        if rows * cols < num_circles:
            rows += 1
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                grid.append([x, y])
        return np.array(grid)

    centers = initial_guess(26)
    radii = np.ones(26) * 0.1  # Initial estimate of radius based on spacing

    def objective(x):
        # x is a flattened vector of [x1, y1, r1, x2, y2, r2, ...]
        x_vec = x[:2*26].reshape(26, 2)
        r_vec = x[2*26:]

        # Objective: maximize sum of radii => minimize negative of sum
        return -np.sum(r_vec)

    def constraint_1(x):
        # Geometric constraints: each circle must stay within [0,1]^2
        x_vec = x[:2*26].reshape(26, 2)
        r_vec = x[2*26:]
        for i in range(26):
            x, y = x_vec[i]
            r = r_vec[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return 1  # Constraint not satisfied
        return 0

    def constraint_2(x):
        # Pairwise distance constraint: distance between centers >= sum of radii
        x_vec = x[:2*26].reshape(26, 2)
        r_vec = x[2*26:]

                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < r_vec[i] + r_vec[j] - 1e-12:
                    return 1  # Constraint not satisfied
        return 0

    bounds = []
    for i in range(26):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # r
    x0 = np.concatenate([centers.ravel(), radii])

    # Optimization
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[
            {"type": "ineq", "fun": constraint_1},
            {"type": "ineq", "fun": constraint_2},
        ],
        method="L-BFGS-B",

    centers_opt = result.x[:2*26].reshape(26, 2)

    return centers_opt, radii_opt, sum_radii
