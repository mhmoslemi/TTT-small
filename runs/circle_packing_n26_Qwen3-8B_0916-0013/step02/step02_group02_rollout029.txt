import numpy as np
import scipy.optimize as optimize
import scipy.spatial as spatial
from typing import Tuple, List, Union

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns a packing of 26 circles in a unit square with maximum total radius.
    """
    # Initialize a basic grid packing as a starting point
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Define grid positions
    grid = np.array([[i, j] for i in range(6) for j in range(6)])
    centers[:36] = grid[:26]
    radii[:26] = 0.1
    
    # Optimize the radii while maintaining non-overlapping and boundary constraints
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii
    
    def constraints(radii):
        # Constraint: all circles must be within the unit square
        for i in range(26):
            r = radii[i]
            if not (0 - r <= 0 <= 1 + r) and (0 - r <= 0 <= 1 + r):
                return False
        return True
    
    bounds = [(0, 1) for _ in range(26)]
    
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
