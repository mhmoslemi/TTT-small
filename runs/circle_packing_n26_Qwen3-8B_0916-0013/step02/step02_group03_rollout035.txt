import scipy.optimize as optimize

def run_packing():
    # Initial configuration: 5 circles on top, 21 circles in hexagonal pattern
    # First 5 circles on top
    centers = []
    radii = []

    # Top row of 5 circles (along y=1)
    for i in range(5):
        x = i / 4.0
        r = 0.2
        centers.append([x, 1 - r])
        radii.append(r)

    # Middle rows of 21 circles in 3 rows of 7, 7, 7
    for row in range(3):
        for col in range(7):
            x = col / 6.0
            y = 0.5 + (row - 1) * np.sqrt(3)/3.0
            r = 0.15
            centers.append([x, y])
            radii.append(r)

    # Constraints
    def calc_constraints(centers, radii):
        n = len(centers)
        cons = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Boundary constraints
            cons.append((x - r, -1e-12))
            cons.append((x + r, 1 + 1e-12))
            cons.append((y - r, -1e-12))
            cons.append((y + r, 1 + 1e-12))
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                cons.append((dist - (radii[i] + radii[j]), -1e-12))
        return cons

    # Objective function: maximize sum of radii
    def objective(x):
        flat_x = x.reshape(-1, 2)
        return -np.sum(flat_x[:, 1])  # Minimize negative sum

    # Bounds: each position (x, y) must be within [0, 1] and radii must be non-negative
    bounds = []
    for i in range(26):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # radius

    flat_radii = []
    for x, y in centers:
        flat_centers.append(x)
        flat_centers.append(y)
    for r in radii:
        flat_radii.append(r)

    # Optimization
    result = optimize.minimize(
        fun=objective,
        x0=np.concatenate((flat_centers, flat_radii)),

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
