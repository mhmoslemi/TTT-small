from scipy.optimize import minimize, Bounds, LinearConstraint

def run_packing():

    grid_cols = 6
    spacing = 2 * avg_radius_guess

    centers = []
    for row in range(grid_rows):

            x = col * spacing

    def objective(radii):

        return np.array([r for r in radii])

    def boundary_box(radii):
