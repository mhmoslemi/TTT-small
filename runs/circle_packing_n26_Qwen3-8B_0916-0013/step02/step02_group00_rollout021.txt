import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26
    centers = []
    radii = []

    # Initial positions for the large circles in the center
    for i in range(4):
        for j in range(4):
            x = j / 3
            y = i / 3
            centers.append([x, y])

    # Add surrounding positions
    for i in range(4):
        for j in range(5):
            x = j / 4
            y = i / 3
            centers.append([x, y])

    # Add a final circle to reach 26 circles
    centers.append([0.5, 0.5])

    # Convert to numpy array
    centers = np.array(centers)

    # Initial guess for radii
    radii = np.full(num_circles, 0.05)

    # Objective to minimize (we use -sum to maximize)
    def objective(radii_vector):
        return -np.sum(radii_vector)

    # Bounds for each circle radius
    bounds = [(0, 1) for _ in range(num_circles)]

    # Constraints: non-overlapping with each other and within the unit square
    constraints = []

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            # Non-overlap constraint
            def constraint_overlap(r):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                return np.sqrt(dx**2 + dy**2) - (r[i] + r[j])
            constraints.append({
                'type': 'ineq',
                'fun': constraint_overlap
            })

    for i in range(num_circles):
        # Boundary constraint
        def constraint_boundary(r):
            x, y = centers[i]
            return np.min([
                x - r[i],
                y - r[i],
                1 - x - r[i],
                1 - y - r[i]
            ])
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary
        })

    # Initial guess: initial radii
    initial_guess = np.full(num_circles, 0.05)

    # Run the optimization
    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}

    if not valid:

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
