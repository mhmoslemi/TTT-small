import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    num_circles = 26
    centers = []
    # Generate a hexagonal-like pattern with some adjustments to fit into the square
    for i in range(num_circles):
        if i < 5:
            centers.append([i / 4, 0.0])
        elif i < 10:
            centers.append([(i - 5) / 4, 0.25])
        elif i < 15:
            centers.append([(i - 10) / 4, 0.5])
        elif i < 20:
            centers.append([(i - 15) / 4, 0.75])
        else:
            centers.append([(i - 20) / 4, 1.0])
    centers = np.array(centers)

    # Add a circle in the center to allow more densely packed configuration
    centers = np.vstack((centers, [0.5, 0.5]))

    # Initial radii as all 0.1 to allow for optimization
    radii = np.full(num_circles, 0.1)

    def objective(radii_vector):
        return -np.sum(radii_vector)

    # Constraints:
    # 1. Ensure each circle stays within the square
    # 2. Ensure no two circles overlap
    # 3. Bounds for each radius

    bounds = [(0, 1) for _ in range(num_circles)]

    constraints = []

    # Constraint for each circle to stay within the square
    for i in range(num_circles):
        def boundary_constraint(r):
            x, y = centers[i]
            r_i = r[i]
            return np.min([x - r_i, y - r_i, 1 - x - r_i, 1 - y - r_i])
        constraints.append({
            'type': 'ineq',
            'fun': boundary_constraint
        })

    # Constraint for each pair of circles to not overlap
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def overlap_constraint(r):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r_i = r[i]
                r_j = r[j]
                return np.sqrt((x1 - x2)**2 + (y1 - y2)**2) - (r_i + r_j)
            constraints.append({
                'type': 'ineq',
                'fun': overlap_constraint
            })

    # Optimize
    initial_guess = radii

    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract results
    radii = result.x
    centers = np.array(centers)

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
