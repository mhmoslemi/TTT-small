    w = np.exp(beta * (cr - cr.max()))

    return x[_sp['n']]

    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

    n = _sp['n']

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(120):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_c5(h, dx, beta):
    n = len(h)
    o = n - 1
    cr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (cr - cr.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    g = np.zeros(n)
    for j in range(n):
        k1lo, k1hi = max(-(n-1), j-(n-1)), min(n-1, j)
        if k1lo <= k1hi:
            ks = np.arange(k1lo, k1hi+1)
            g[j] += np.dot(w[o+ks], 1.0 - h[j-ks])
        k2lo, k2hi = max(-(n-1), -j), min(n-1, n-1-j)
        if k2lo <= k2hi:
            ks = np.arange(k2lo, k2hi+1)
            g[j] -= np.dot(w[o+ks], h[j+ks])
