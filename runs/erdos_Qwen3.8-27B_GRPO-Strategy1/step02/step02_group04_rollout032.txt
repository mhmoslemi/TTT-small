def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.49:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:

            idx = np.argsort(-h)
            for i in idx:

                diff -= rem

    return h

def gradient_descent(h, n, T_start, T_end, max_iter, lr_start, lr_end, momentum=0.9):
    h_curr = h.copy()
    v_mom = np.zeros(n)
    
    for it in range(max_iter):
        # Anneal temperature
        frac = it / max(max_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        val, grad = soft_max_value_and_grad(h_curr, T)
        
        # Update velocity
        v_mom = momentum * v_mom - lr * grad
        
        # Step
        h_new = h_curr + v_mom
        
        # Project
        h_new = project_constraint(h_new, n)
        
        # Check improvement

        else:
            # Backtrack
            v_mom *= 0.5
            h_new = h_curr + v_mom
            h_new = project_constraint(h_new, n)
    
    return h_curr

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.49:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13:
                    break
    return h

def project_feasible(y, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):

        x = np.clip(y - mid, 0.0, 1.0)

        ni = np.sum(interior)

def soft_max_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    shifted = np.clip(corr - cmax, -50.0 * T, 0.0)
    s = np.exp(shifted / T).sum()
    return (cmax + T * np.log(max(s, 1e-300))) * dx

def soft_max_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    cmax = corr.max()
    shifted = np.clip(corr - cmax, -50.0 * T, 0.0)
    p = np.exp(shifted / T)
    p /= p.sum()
    p_rev = p[::-1]
    term1 = np.correlate(v, p_rev, mode='full')[n-1 : 2*n-1]
    term2 = np.correlate(h, p, mode='full')[n-1 : 2*n-1]
    return dx * (term1 - term2)

def gradient_descent(h0, n, T_start, T_end, max_iter, lr, momentum=0.85, 
                     time_limit=None, t_start=None):
    h = h0.copy()
    v_mom = np.zeros(n)
    
    for it in range(max_iter):
        if time_limit is not None:
            el = time.time() - t_start
            if el > time_limit:
                break
        
        frac = it / max(max_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        cur_lr = lr * (1.0 - 0.8 * frac)
        
        grad = soft_max_gradient(h, T)

        h_new = h + v_mom
        h_new = project_feasible(h_new, n)
        
        # Simple acceptance: always accept (the projection maintains feasibility)
        h = h_new
    
    return h
