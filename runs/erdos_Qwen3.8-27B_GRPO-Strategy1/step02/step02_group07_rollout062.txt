        s = (corr - M) / T
        logS = np.log(np.sum(np.exp(s)))
        f = (M + T * logS) * dx
        w = np.exp(s)
        w /= w.sum()
    conv1 = np.real(np.fft.ifft(np.fft.fft(w) * np.fft.fft(h)))
    wf = np.empty(n)
    wf[0] = w[0]
    wf[1:] = w[:0:-1]
    conv2 = np.real(np.fft.ifft(np.fft.fft(wf) * np.fft.fft(h)))
    grad = dx * (1.0 - conv1 - conv2)
    return f, grad

def adam_phase(h, T0, T1, steps, lr, n, stall=5000):
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_f, best_h = np.inf, h.copy()
    last_improve = 0
    for t in range(1, steps + 1):
        fr = (t - 1) / max(steps - 1, 1)
        T = T0 * (T1 / T0) ** fr
        f, grad = corr_and_grad(h, T)
        if f < best_f - 1e-12:
            best_f, best_h = f, h.copy()
            last_improve = t
        if t - last_improve > stall and t > 1000:
            break
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = fix_sum(h, n)
    return best_h

def coord_descend(h, n, passes=4):
    bc = compute_c5(h)
    deltas = (-0.015, -0.008, -0.004, -0.002, 0.002, 0.004, 0.008, 0.015)
    for _ in range(passes):
        imp = False
        for j in range(n):
            hj = h[j]
            if hj < 5e-4 or hj > 1.0 - 5e-4:
                continue
            for dl in deltas:
                h[j] = min(max(hj + dl, 0.0), 1.0)
                d = 0.5 * n - np.sum(h)
                if abs(d) > 1e-10:
                    m2 = (h > 5e-4) & (h < 1 - 5e-4)
                    m2[j] = False
                    if m2.sum() > 0:
                        idx = np.where(m2)[0][0]
                        h[idx] = min(max(h[idx] + d, 0.0), 1.0)
                c = compute_c5(h)
                if c < bc - 2e-6:
                    bc = c
                    imp = True
                    break
                h[j] = hj
        if not imp:
            break
    return fix_sum(h, n)

def pair_descend(h, n, rng_obj, npair=400):
    bc = compute_c5(h)
    for _ in range(npair):
        j, l = rng_obj.choice(n, 2, replace=False)
        if h[j] < 5e-3 or h[j] > 1 - 5e-3 or h[l] < 5e-3 or h[l] > 1 - 5e-3:
            continue
        best_d = 0.0
        for d in (-0.03, -0.01, -0.004, 0.004, 0.01, 0.03):
            if h[j] + d < 5e-3 or h[j] + d > 1 - 5e-3:
                continue
            if h[l] - d < 5e-3 or h[l] - d > 1 - 5e-3:
                continue
            h[j], h[l] = h[j] + d, h[l] - d
            c = compute_c5(h)
            if c < bc - 2e-6:
                bc = c
                best_d = d
                break
            h[j], h[l] = h[j] - d, h[l] + d
    return fix_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h)) * 2.0 / len(h)
    x1 = np.arange(nn) * 2.0 / nn
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 13)

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = fix_sum(h, n)
    c5 = compute_c5(h)

    # Phase 1: Broad Adam exploration
    h = adam_phase(h, 0.05, 0.005, 40000, 0.025, n, stall=8000)
    c5 = compute_c5(h)

    if time.time() - t0 > budget_s - 400:
        return (fix_sum(h, n), compute_c5(h), n)

    # Phase 2: Mid-range refinement
    h = adam_phase(h, 0.005, 0.0005, 60000, 0.012, n, stall=8000)
    h = fix_sum(h, n)
    c5 = compute_c5(h)

    if time.time() - t0 > budget_s - 300:
        return (fix_sum(h, n), compute_c5(h), n)

    # Phase 3: Fine Adam
    h = adam_phase(h, 0.0005, 0.00005, 50000, 0.004, n, stall=6000)
    h = fix_sum(h, n)
    c5 = compute_c5(h)

    # Phase 4: Coordinate + pair polish
    if time.time() - t0 < budget_s - 150:
        h = coord_descend(h, n, 5)
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 120:
        h = pair_descend(h, n, rng, 800)
        c5 = compute_c5(h)

    # Phase 5: Diverse restarts with Adam refine
    for i in range(25):
        if time.time() - t0 > budget_s - 100:
            break
        mode = i % 5
        ht = h.copy()
        if mode == 0:
            ht += rng.randn(n) * 0.012
        elif mode == 1:
            idx = rng.choice(n, 15, replace=False)
            ht[idx] += rng.randn(15) * 0.08
        elif mode == 2:
            ht += rng.randn(n) * 0.006
        elif mode == 3:
            mask = (ht > 0.05) & (ht < 0.95)
            ht[mask] += rng.randn(np.sum(mask)) * 0.05
        else:
            idx = rng.choice(n, 8, replace=False)
            for ii in idx:
                lt = rng.choice(n)
                dv = rng.uniform(0.02, 0.08)
                if 0 < h[ii] < 1 and 0 < h[lt] < 1:
                    ht[ii] = min(max(ht[ii] + dv, 0), 1)
                    ht[lt] = min(max(ht[lt] - dv, 0), 1)
        ht = fix_sum(ht, n)
        ht = adam_phase(ht, 0.008, 0.0008, 15000, 0.015, n, stall=4000)
        ht = fix_sum(ht, n)
        ct = compute_c5(ht)
        if ct < c5:
            h, c5 = ht, ct
            if time.time() - t0 < budget_s - 80:
                h = coord_descend(h, n, 3)
                h = pair_descend(h, n, rng, 400)
                c5 = compute_c5(h)

    # Phase 6: Final fine Adam push
    if time.time() - t0 < budget_s - 40:
        h = adam_phase(h, 0.00008, 0.00001, 30000, 0.002, n, stall=5000)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 7: Upsample attempt
    if time.time() - t0 < budget_s - 50:
        n2 = 560
        h2 = upsample(h, n2)
        h2 = adam_phase(h2, 0.01, 0.0003, 30000, 0.01, n2, stall=6000)
        h2 = fix_sum(h2, n2)
        h2 = coord_descend(h2, n2, 3)
        c2 = compute_c5(h2)
        if c2 < c5:
            h, n, c5 = h2, n2, c2

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
