        else:

def anneal_binary(h, N, deadline, rng, T0=10.0, T1=0.1):

    h_b = (h > 0.5).astype(float)

    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    if len(ones) > N//2:
        excess = ones[len(ones)-N//2:]
        h_b[excess] = 0
        zeros = np.where(h_b == 0)[0]
    elif len(ones) < N//2:
        deficit = zeros[:N//2-len(ones)]
        h_b[deficit] = 1
        ones = np.where(h_b == 1)[0]
    
    best_h = h_b.copy()
    best_peak = float(np.max(compute_corr(h_b)))
    curr_peak = best_peak

    it = 0
    while time.time() < deadline and it < n_iter:
        frac = it / n_iter
        T = T0 * (T1/T0) ** frac
        
        # Random swap
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_b[i], h_b[j] = h_b[j], h_b[i]
        new_peak = float(np.max(compute_corr(h_b)))

            curr_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        it += 1
    
    # Convert back to continuous
    result = best_h.copy()
    return result, best_peak * dx

def adam_topk(h, N, target, deadline, k, lr0, decay):
    """Adam with top-k subgradient."""
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    dx = 2.0/N
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        true_c5 = float(np.max(corr)) * dx
        if true_c5 < best_c5:
            best_c5 = true_c5
            best_h = h.copy()
        
        # Top-k indices
        top_idx = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        inv_k = 1.0 / k
        for ai in top_idx:
            L = ai - center
            if L == 0:
                grad += inv_k * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += inv_k * (1.0 - h[:N-L])
                grad[:N-L] -= inv_k * h[L:]
            else:
                M = -L
                grad[:N-M] += inv_k * (1.0 - h[M:])
                grad[M:] -= inv_k * h[:N-M]
        
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1-b1**it)
        vh = v / (1-b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
    return best_h, best_c5

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_run(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 150 == 0:
            lr *= decay
    return best_h, best_peak * dx

def anneal_binary(h, N, deadline, rng):
    dx = 2.0 / N
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > N // 2:
        idx = np.where(h_b == 1)[0]
        h_b[idx[N // 2:]] = 0
    elif n_ones < N // 2:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:N // 2 - n_ones]] = 1
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur_pk = best_pk
    T0, T1 = 8.0, 0.005
    n_total = int((deadline - time.time()) * 60)
    n_total = max(n_total, 1000)
    for it in range(n_total):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / n_total)
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = h_b[j], h_b[i]
        new_pk = float(np.max(compute_corr(h_b)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.01 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
    return best, best_pk * dx
