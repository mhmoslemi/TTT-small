def binary_round(h, n):
    """Round h to binary, maintaining sum constraint."""
    thresh = 0.5
    h_bin = (h >= thresh).astype(float)
    # Adjust to get exactly n/2 ones
    while h_bin.sum() > n/2:
        # Flip the 1 with smallest h value
        ones = np.where(h_bin == 1)[0]
        if len(ones) == 0:
            break
        flip = ones[np.argmin(h[ones])]
        h_bin[flip] = 0
    while h_bin.sum() < n/2:
        # Flip the 0 with largest h value
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) == 0:
            break
        flip = zeros[np.argmax(h[zeros])]
        h_bin[flip] = 1
    return h_bin

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_erdos(h, T):
    n = len(h)
