def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        np.clip(h, 0.0, 1.0, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

def sgrad(h, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    m = cr.max()
    w = np.exp(beta * (cr - m))
    sw = w.sum()
    if sw < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= sw
    J = np.arange(n, dtype=float)
    S = np.arange(-(n - 1), n, dtype=float)

    H2 = np.where(valid2, h[np.clip(V2, 0, n - 1).astype(int)], 0.0)

        b = 10.0 + 90.0 * min(1.0, it / max(1, 0.4 * n_it))
        lr = 0.015 / (1.0 + 0.0008 * it)
        g = sgrad(h, b)
        v = 0.85 * v - lr * g
        ok = False
        for sh in (1.0, 0.5, 0.25, 0.1, 0.05):
            h2 = proj(h + sh * v, ts)
            c2 = compute_c5(h2, dx)
            if c2 < cm - 1e-14:
                h, cm = h2, c2
                if cm < bc:
                    best, bc = h2.copy(), cm
                    noimp = 0
                else:
                    noimp += 1
                ok = True
                break
        if not ok:
            v *= 0.5
            noimp += 1
            if noimp > 300:
                break
    return best, bc

def cdes(h0, dx, ts, dl, rng):
    n = len(h0)
    h = h0.copy()
    best = h0.copy()
    bc = compute_c5(h, dx)
    steps = (0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.003, -0.003, 0.03, -0.03, 0.001, -0.001)
    nd = 0
    while nd < 300 and time.time() < dl:
        imp = False
        for j in rng.permutation(n):
            if time.time() > dl:
                break
            sv = h[j]
            for d in steps:
                nv = sv + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[j] = nv
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < bc - 1e-15:
                    bc, h, best, imp = c, h, h.copy(), True
                    break
            else:
                h[j] = sv
                h = proj(h, ts)
        nd += 1
        if not imp:
            break
    return best, bc

def sa_search(h0, dx, ts, n_it, dl, rng):
    n = len(h0)
    h = h0.copy()
    best = h0.copy()
    bc = compute_c5(h, dx)
    c = bc
    t_start = c * 0.4
    t_end = c * 0.002
    for it in range(n_it):
        if it % 200 == 0 and time.time() > dl:
            break
        T = t_start + (t_end - t_start) * (it / max(1, n_it))
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        if rng.random() < 0.4:
            h[i], h[j] = h[j], h[i]
        else:
            d = rng.normal(0, 0.05)
            h[i] += d
            h[j] -= d
            h = proj(h, ts)
        cn = compute_c5(h, dx)
        if cn <= c or rng.random() < np.exp(-(cn - c) / max(T, 1e-15)):
            c = cn
            if cn < bc:
                bc, best = cn, h.copy()
        else:
            h = best.copy()
            c = bc
    return best, bc

def quick_proj(h, ts, changed_idx):
    """Quick projection when only one element was changed."""
    d = ts - h.sum()
    if abs(d) < 1e-15:
        return h
    # Find one other element to adjust
    if d > 0 and h[changed_idx] < 1.0:
        # Can increase changed_idx
        adj = min(d, 1.0 - h[changed_idx])
        h[changed_idx] += adj
        d -= adj
    if d < 0 and h[changed_idx] > 0.0:
        adj = max(d, h[changed_idx])
        h[changed_idx] += adj
        d -= adj
    # If still imbalanced, spread across all
    if abs(d) > 1e-15:
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += d / cnt
    np.clip(h, 0.0, 1.0, out=h)
    return h
