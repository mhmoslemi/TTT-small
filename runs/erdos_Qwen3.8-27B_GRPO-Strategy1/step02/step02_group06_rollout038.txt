def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        ni = int(np.sum(interior))
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0, 1)
    return h

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sh = np.clip((corr - cmax) / T, -500, 500)
    return (cmax + T * np.log(np.sum(np.exp(sh)))) * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sh = np.clip((corr - cmax) / T, -500, 500)
    es = np.exp(sh)
    w = es / np.sum(es)
    grad = np.zeros(n)
    sig = w > 1e-10
    if not np.any(sig):
        return cmax * dx, grad
    idxs = np.where(sig)[0]
    for j in idxs:
        lag = j - (n - 1)
        wj = w[j]
        if lag >= 0:
            i1 = np.arange(lag, n)
            grad[i1] += wj * (1.0 - h[i1 - lag])
            i2 = np.arange(0, n - lag)
            grad[i2] -= wj * h[i2 + lag]
        else:
            ll = -lag
            i1 = np.arange(0, n - ll)
            grad[i1] += wj * (1.0 - h[i1 + ll])
            i2 = np.arange(ll, n)
            grad[i2] -= wj * h[i2 - ll]
    grad *= dx
    obj = (cmax + T * np.log(np.sum(es))) * dx
    return obj, grad

    return lse_grad(h, T)[0]

def lse_grad_only(h, T):
    return lse_grad(h, T)[1]

        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10, 'iprint': 0}
    )

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1e-8 * 0)
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        ni = int(np.sum(interior))
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0, 1)
    return h

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sh = np.clip((corr - cmax) / T, -500, 500)
    return (cmax + T * np.log(np.sum(np.exp(sh)))) * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sh = np.clip((corr - cmax) / T, -500, 500)
    es = np.exp(sh)
    w = es / np.sum(es)
    grad = np.zeros(n)
    sig = w > 1e-10
    if not np.any(sig):
        return (cmax + T * np.log(np.sum(es))) * dx, grad
    idxs = np.where(sig)[0]
    for j in idxs:
        lag = int(j - (n - 1))
        wj = w[j]
        if lag >= 0:
            i1 = np.arange(lag, n)
            grad[i1] += wj * (1.0 - h[i1 - lag])
            i2 = np.arange(0, n - lag)
            grad[i2] -= wj * h[i2 + lag]
        else:
            ll = -lag
            i1 = np.arange(0, n - ll)
            grad[i1] += wj * (1.0 - h[i1 + ll])
            i2 = np.arange(ll, n)
            grad[i2] -= wj * h[i2 - ll]
    grad *= dx
    obj = (cmax + T * np.log(np.sum(es))) * dx
    return obj, grad

def f_obj(h, T):
    return lse_obj(h, T)

def f_grad(h, T):
    return lse_grad(h, T)[1]

def eq_constr(h, n):
    return np.sum(h) - 0.5 * n

def eq_constr_jac(h, n):
    return np.ones(n)

def slsqp_run(h0, n, T, max_iter):
    res = opt.minimize(
        f_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': eq_constr, 'jac': eq_constr_jac, 'args': (n,)}],
        options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0}
    )
    return proj(res.x, n)

def lbe_run(h0, n, T, max_iter):
    res = opt.minimize(
        f_obj, h0, args=(T,),
        jac=f_grad,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12, 'iprint': 0}
    )
    return proj(res.x, n)

def block_fn(x, h_tpl, s, e, T):
    hf = h_tpl.copy()
    hf[s:e] = x
    return lse_obj(hf, T)

def block_jac(x, h_tpl, s, e, T):
    hf = h_tpl.copy()
    hf[s:e] = x
    g = lse_grad(hf, T)[1]
    return g[s:e]

def upsample(h, new_n):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj(h_new, new_n)

def downsample(h, new_n):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj(h_new, new_n)

def active_grad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    k_star = np.argmax(corr)
    lag = int(k_star - (n - 1))
    grad = np.zeros(n)
    if lag >= 0:

        grad[i1] = 1.0 - h[i1 - lag]
        i2 = np.arange(0, n - lag)
        grad[i2] -= h[i2 + lag]
    else:

        grad[i1] = 1.0 - h[i1 + ll]
        i2 = np.arange(ll, n)
        grad[i2] -= h[i2 - ll]
    return grad * dx
