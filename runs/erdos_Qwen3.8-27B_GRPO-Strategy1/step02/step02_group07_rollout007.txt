import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-13:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
        h = np.clip(h, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-10:
        idx = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + d2, 0, 1)
    return h

def lse_smoothed(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    shifted = corr - c
    m2 = np.max(shifted)
    return (c + T * (m2 + np.log(np.exp(shifted - m2).sum()))) * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        w = np.zeros(L)
        w[np.argmax(corr)] = 1.0
    else:
        sh = corr - c
        m2 = np.max(sh)
        w = np.exp(sh - m2)
        w /= w.sum()
    ns = 1
    while ns < L:
        ns *= 2
    H = np.fft.fft(h, ns)
    W1 = np.zeros(ns, dtype=complex)
    for k in range(n):
        W1[k] = w[n - 1 + k]
        if k > 0:
            W1[ns - k] = w[n - 1 - k]
    W2 = np.zeros(ns, dtype=complex)
    for k in range(n):
        W2[k] = w[n - 1 - k]
        if k > 0:
            W2[ns - k] = w[n - 1 + k]
    A = np.real(np.fft.ifft(np.conj(W1) * H)[:n])
    B = np.real(np.fft.ifft(np.conj(W2) * H)[:n])
    return -(A + B) * dx

def proj_grad(h, n, lr, niters=200):
    stp = lr
    for _ in range(niters):
        T = max(stp * 0.01, 1e-5)
        g = lse_grad(h, T)
        hn = h - stp * g
        hn = fix_sum(hn, n)
        f_new = lse_smoothed(hn, T)
        f_old = lse_smoothed(h, T)
        if f_new > f_old:
            stp *= 0.6
            if stp < 1e-7:
                break
        else:
            h = hn
            stp = min(stp * 1.08, 0.5)
    return h

def coord_pass(h, n, npass=3):
    for _ in range(npass):
        changed = False
        for j in range(n):
            if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5:
                continue
            best = compute_c5(h)
            best_j = h[j]
            for d in (-0.08, -0.03, -0.008, 0.008, 0.03, 0.08):
                hv = h.copy()
                adj = -d / (n - 1)
                hv[:] += adj
                hv[j] = min(max(h[j] + d, 0.0), 1.0)
                hv = np.clip(hv, 0.0, 1.0)
                c = compute_c5(hv)
                if c < best - 1e-14:
                    best = c
                    best_j = h[j] + d
            if abs(best_j - h[j]) > 1e-6:
                h = best_j
                changed = True
        if not changed:
            break
    return fix_sum(h, n)

def pair_pass(h, n, rng, npairs=400):
    for _ in range(npairs):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5 or h[l] <= 1e-5 or h[l] >= 1.0 - 1e-5:
            continue
        best = compute_c5(h)
        bd = 0.0
        for d in (-0.06, -0.02, -0.006, 0.006, 0.02, 0.06):
            hj, hl = h[j] + d, h[l] - d
            if hj < 1e-5 or hj > 1.0 - 1e-5 or hl < 1e-5 or hl > 1.0 - 1e-5:
                continue
            hv = h.copy()
            hv[j], hv[l] = hj, hl
            c = compute_c5(hv)
            if c < best - 1e-13:
                best = c
                bd = d
        if bd != 0.0:
            h[j] = np.clip(h[j] + bd, 0, 1)
            h[l] = np.clip(h[l] - bd, 0, 1)
    return fix_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h), nn)

def downsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h), nn)

def topk_lse_obj(h, K, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:K]
    vals = corr[idx]
    mx = vals[0]
    return (mx + alpha * np.log(np.exp((vals - mx) / alpha).sum())) * dx

def slsqp_opt(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_smoothed, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5 * n}],
            options={'maxiter': iters, 'ftol': 1e-14})
        hn = fix_sum(r.x, n)
        return hn if compute_c5(hn) < c0 else h
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    t0 = time.time()
    h = np.array(initial_h_values, dtype=float).copy()

        mode = i % 5
        if mode == 0:

            if ct < c5:

                if time.time() - t0 < budget_s - 80:
                    h = coord_pass(h, n, 2)
                    h = pair_pass(h, n, rng, 200)
                    c5 = compute_c5(h)

        h = slsqp_opt(h, n, T, it)

    if time.time() - t0 < budget_s - 60:

    return (h, compute_c5(h), len(h))
