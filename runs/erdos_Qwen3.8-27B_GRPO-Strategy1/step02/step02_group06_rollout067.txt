import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        ni = int(np.sum(interior))
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0.0, 1.0)
    return h

def sm_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    val = dx * cm
    if T > 1e-15:
        lp = (corr - cm) / T
        lp -= np.max(lp)
        val = dx * (cm + T * np.log(np.sum(np.exp(lp))))
    return val

def sm_weights(corr, T):
    cm = np.max(corr)
    if T < 1e-15:
        p = np.zeros(len(corr))
        p[np.argmax(corr)] = 1.0
        return p
    lp = (corr - cm) / T
    lp -= np.max(lp)
    p = np.exp(lp)
    return p / np.sum(p)

def sm_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    p = sm_weights(corr, T)
    grad = np.zeros(n)
    for j in range(n):
        t1 = np.dot(p[n-1-j : 2*n-1-j], 1.0 - h)
        t2 = np.dot(p[j : j+n], h[::-1])
        grad[j] = dx * (t1 - t2)
    return grad

def lbfgs_obj(h, T, lam, n):
    val = sm_val(h, T)
    s = np.sum(h) - 0.5 * n
    return val + lam * s * s

def lbfgs_grad(h, T, lam, n):
    g = sm_grad(h, T)
    s = np.sum(h) - 0.5 * n
    g = g + 2.0 * lam * s
    return g

def do_lbfgs(h0, T, lam, max_iter, t_start, deadline):
    n = len(h0)
    res = opt.minimize(lbfgs_obj, h0, args=(T, lam, n),
                       jac=lbfgs_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': max_iter * 20})
    if time.time() > deadline:
        return h0
    return project_feasible(res.x, 0.5 * n)

def proj_grad(h0, T, max_iter, t_start, deadline):
    n = len(h0)
    target = 0.5 * n
    h = h0.copy()
    best = compute_c5(h)
    best_h = h.copy()
    alpha = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = sm_grad(h, T)
        # Compute max gradient component for scaling
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-12:
            break
        g = g / g_norm  # normalize
        # Line search
        for _ in range(20):
            h_try = project_feasible(h - alpha * g, target)
            c_try = compute_c5(h_try)
            if c_try < best or c_try < compute_c5(h):
                break
            alpha *= 0.5
        h = project_feasible(h - alpha * g, target)
        c = compute_c5(h)
        if c < best:
            best = c
            best_h = h.copy()
            alpha = min(alpha * 1.15, 0.2)
        else:
            alpha *= 0.7
    return best_h, best

def sa_refine(h0, n, t_start, deadline, seed, n_steps=500, T_init=0.001):
    rng = np.random.RandomState(seed)
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    T = T_init
    target = 0.5 * n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Random block mutation
        n_blk = rng.randint(2, 12)
        idxs = rng.choice(n, size=n_blk, replace=False)
        h_try = h_cur.copy()
        h_try[idxs] = np.clip(h_try[idxs] + rng.randn(n_blk) * 0.03, 0, 1)
        h_try = project_feasible(h_try, target)
        c_try = compute_c5(h_try)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T):
            h_cur = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        T *= 0.995
    return h_best, c_best

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, 0.5 * new_n)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        ni = int(np.sum(interior))
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0.0, 1.0)
    return h

def sm weights(corr, T):
    cm = np.max(corr)
    if T < 1e-15:
        p = np.zeros(len(corr))
        p[np.argmax(corr)] = 1.0
        return p
    lp = (corr - cm) / T
    lp -= np.max(lp)
    p = np.exp(lp)
    return p / np.sum(p)

def sm_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    p = sm_weights(corr, T)
    grad = np.zeros(n)
    rev1mh = (1.0 - h)[::-1]
    for m in range(n):
        t1 = np.dot(p[m:m+n], rev1mh)
        t2 = np.dot(p[n-1-m:2*n-1-m], h)
        grad[m] = dx * (t1 - t2)
    return grad

def lbfgs_obj(h, T, lam, n):
    n2 = int(n)
    dx = 2.0 / n2
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    val = dx * cm
    if T > 1e-15:
        lp = (corr - cm) / T
        lp -= np.max(lp)
        val = dx * (cm + T * np.log(np.sum(np.exp(lp))))
    s = np.sum(h) - 0.5 * n2
    return val + lam * s * s

def lbfgs_grad(h, T, lam, n):
    n2 = int(n)
    g = sm_grad(h, T)
    s = np.sum(h) - 0.5 * n2
    return g + 2.0 * lam * s

def do_lbfgs(h0, T, lam, max_iter, deadline):
    n = len(h0)
    res = opt.minimize(lbfgs_obj, h0, args=(T, lam, n),
                       jac=lbfgs_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': max_iter * 25})
    if time.time() > deadline:
        return h0
    return project_feasible(res.x, 0.5 * n)

def proj_grad(h0, T, max_iter, deadline):
    n = len(h0)
    target = 0.5 * n
    h = h0.copy()
    c_cur = compute_c5(h)
    best = c_cur
    best_h = h.copy()
    alpha = 0.04
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        g = sm_grad(h, T)
        gn = np.max(np.abs(g))
        if gn < 1e-13:
            break
        g = g / gn
        for _ in range(25):
            h_try = project_feasible(h - alpha * g, target)
            c_try = compute_c5(h_try)
            if c_try < c_cur:
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        h = project_feasible(h - alpha * g, target)
        c_cur = compute_c5(h)
        if c_cur < best:
            best = c_cur
            best_h = h.copy()
            alpha = min(alpha * 1.12, 0.15)
        else:
            alpha *= 0.75
    return best_h, best

def sa_refine(h0, t_start, deadline, seed, n_steps=500):
    n = len(h0)
    target = 0.5 * n
    rng = np.random.RandomState(seed)
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    T = 0.0008
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        n_blk = rng.randint(2, 10)
        idxs = rng.choice(n, size=n_blk, replace=False)
        h_try = h_cur.copy()
        h_try[idxs] = np.clip(h_try[idxs] + rng.randn(n_blk) * 0.025, 0, 1)
        h_try = project_feasible(h_try, target)
        c_try = compute_c5(h_try)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T):
            h_cur = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        T *= 0.996
    return h_best, c_best

def upsample_h(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, 0.5 * new_n)
