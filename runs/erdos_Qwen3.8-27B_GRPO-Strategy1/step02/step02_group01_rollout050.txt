    dx = 2.0 / n

                v += 1.0 - h[p + s]
            if 0 <= p - s < n:

    grad += 2.0 * penalty * d

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def grad_obj(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    m = float(np.max(corr))
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = n - 1 - k
        for p in range(n):
            v = 0.0
            ps = p + s
            if 0 <= ps < n:
                v += 1.0 - h[ps]
            pm = p - s
            if 0 <= pm < n:
                v -= h[pm]
            grad[p] += w[k] * v
    grad *= dx
    d = h.sum() - n / 2.0
    grad += 2.0 * penalty * d
    return grad

def obj_val(h, T, penalty):
    n = len(h)
    return soft_max_c5(h, T) + penalty * (h.sum() - n / 2.0) ** 2

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h *= target / tot
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgs_opt(h, T, penalty, maxiter):
    n = len(h)
    try:
        res = minimize(obj_val, h, jac=lambda x: grad_obj(x, T, penalty),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_step(h, T, lr, penalty):
    n = len(h)
    g = grad_obj(h, T, penalty)
    h_new = h - lr * g
    return project_sum(h_new, n)

def randomized_binary(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def block_pattern(n, rng):
    blocks = []
    pos = 0
    while pos < n:
        bl = max(1, int(rng.exponential(max(3, n // 8))))
        bl = min(bl, n - pos)
        blocks.append(bl)
        pos += bl
    b = np.zeros(n)
    pos = 0

        if bi % 2 == 0:
            val = 1.0 if rng.random() > 0.2 else 0.5
            b[pos:pos + bl] = val
        pos += bl
    return b
