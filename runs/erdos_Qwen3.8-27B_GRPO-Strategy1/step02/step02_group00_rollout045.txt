import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    s = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    val = s * dx
    if lam > 0:

    return val

    target = n * 0.5
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

        if room < 1e-14:
            return h
        h = h + (1.0 - h) * (diff / room)
    else:
        room = h.sum()
        if room < 1e-14:
            return h

    bh_h = best_h.copy()
    bh_c5 = best_c5
    for it in range(n_iter_bh):
        if elapsed() > budget_s * 0.50:
            break
        frac = it / n_iter_bh
        step = 0.18 * (1.0 - frac) + 0.03
        h_pert = bh_h + step * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        hc, c5v = try_optimize(h_pert, n, T_bh, 150, best_c5)
        if hc is None:
            hc, c5v = try_topk_optimize(h_pert, n, T_bh * 2, 5, 150, best_c5)
        if c5v is not None:
            bh_c5 = c5v
            bh_h = hc
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        else:
            # Metropolis: accept worse for exploration
            h_check = project_feasible(h_pert, n)
            c5_check = compute_c5(h_check)
            temp = 0.004 * (1.0 - frac) + 0.0002
            if c5_check < bh_c5 or np.random.rand() < np.exp(-(c5_check - bh_c5) / temp):
                bh_c5 = c5_check
                bh_h = h_check

    # Phase 3: Different resolutions
    for n_try in [60, 80, 100, 150]:
        if elapsed() > budget_s * 0.68:
            break
        x_old = (np.arange(n) + 0.5) * 2.0 / n
        x_new = (np.arange(n_try) + 0.5) * 2.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        for T in [50, 200, 800, 3000]:
            if elapsed() > budget_s * 0.68:
                break
            hc, c5v = try_optimize(h_interp.copy(), n_try, T, 300, best_c5)
            if hc is not None:
                best_c5 = c5v
                best_h = hc
                h_interp = hc
            hc, c5v = try_topk_optimize(h_interp.copy(), n_try, T * 2, 4, 150, best_c5)
            if hc is not None:
                best_c5 = c5v
                best_h = hc
                h_interp = hc

    # Phase 4: Structured perturbations
    n = len(best_h)
    struct_starts = []
    # Block shifts
    for shift in [n//6, n//5, n//4, n//3, n//2.7]:
        hs = np.roll(best_h, shift)
        struct_starts.append(project_feasible(hs, n))
    # Inversion
    struct_starts.append(project_feasible(1.0 - best_h, n))
    # Half-inversion
    hsemi = best_h.copy()
    hsemi[n//2:] = 1.0 - hsemi[n//2:]
    struct_starts.append(project_feasible(hsemi, n))
    # Random binary (balanced)
    for _ in range(3):
        hb = (np.random.rand(n) < 0.5).astype(float)
        struct_starts.append(project_feasible(hb, n))

    for h0 in struct_starts:
        if elapsed() > budget_s * 0.82:
            break
        for T in [100, 500, 2000]:
            if elapsed() > budget_s * 0.82:
                break
            hc, c5v = try_optimize(h0, n, T, 250, best_c5)
            if hc is not None:
                best_c5 = c5v
                best_h = hc
                h0 = hc

    # Phase 5: Fine coordinate descent
    n = len(best_h)
    for iteration in range(500):
        if elapsed() > budget_s * 0.94:
            break
        improved = False
        order = np.random.permutation(n)
        deltas = [0.005, 0.01, 0.02, 0.05, -0.005, -0.01, -0.02, -0.05]
        for j in order:
            for delta in deltas:
                h_try = best_h.copy()
                h_try[j] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
        if not improved and iteration > 30:
            break

    # Phase 6: Final ultra-high-T polish
    n = len(best_h)
    for T in [10000, 30000, 100000]:
        if elapsed() > budget_s * 0.99:
            break
        hc, c5v = try_optimize(best_h.copy(), n, T, 500, best_c5)
        if hc is not None:
            best_c5 = c5v
            best_h = hc
        hc, c5v = try_topk_optimize(best_h.copy(), n, 500, 3, 300, best_c5)
        if hc is not None:
            best_c5 = c5v
            best_h = hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
