    bc = compute_c5(h, dx)
    T = T0
    for i in range(n_iter):
        if dl and time.time() > dl:
            break
        T *= Tcool
        mt = rng.randint(5)
        ht = h.copy()
        if mt == 0:
            j = rng.randint(n)
            ht[j] += rng.normal(0, 0.06)
        elif mt == 1:
            j = rng.randint(max(1, n - 8))
            wt = rng.randint(2, 9)
            ht[j:j + wt] += rng.normal(0, 0.04, wt)
        elif mt == 2:
            i1, i2 = rng.randint(n), rng.randint(n)
            if i1 != i2:
                ht[i1], ht[i2] = ht[i2], ht[i1]
        elif mt == 3:
            mk = rng.rand(n) < 0.15
            ht[mk] += rng.normal(0, 0.05, max(1, mk.sum()))
        else:
            i1 = rng.randint(n)
            i2 = rng.randint(n)
            amt = rng.normal(0, 0.04)
            ht[i1] += amt
            ht[i2] -= amt
        ht = proj(ht, ts)
        ct = compute_c5(ht, dx)
        delta = ct - bc
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / max(T * dx, 1e-15))):
            h, bc = ht, ct
    return h, bc

def coord_desc(h0, dx, ts, n_round, dl, steps):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    for r in range(n_round):
        if dl and time.time() > dl:
            break
        imp = False
        for j in range(n):
            sv = h[j]
            for d in steps:
                nv = sv + d
                if nv < -1e-12 or nv > 1 + 1e-12:
                    continue
                h[j] = nv
                h2 = proj(h, ts)
                c = compute_c5(h2, dx)
                if c < bc - 1e-15:
                    h, bc = h2, c
                    imp = True
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def slsqp_local(h0, dx, ts, maxiter=120):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.004]])
    bnds = [(0, 1)] * n + [(0.0, float(n) * 2)]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 * dx + 0.02:
                return hr
    except Exception:
        pass
    return h0.copy()

def gen_starts(n, ts, rng, ref=None):
    out = []
    if ref is not None:
        if len(ref) == n:
            out.append(ref.copy())
        else:
            xi = np.linspace(0, 1, n, endpoint=False)
            xo = np.linspace(0, 1, len(ref), endpoint=False)
            out.append(np.interp(xi, xo, ref))
    out.append(np.full(n, ts / n))
    for _ in range(4):
        hv = np.zeros(n)
        p = rng.permutation(n)
        nt = int(ts)
        hv[p[:nt]] = 1.0
        fr = ts - nt
        if fr > 0.01:
            hv[p[nt]] = fr
        out.append(hv)
    for m in (1, 2, 3, 5, 8, 13):
        x = np.arange(n) / n
        out.append(0.5 + 0.35 * np.sin(2 * np.pi * m * x + rng.uniform(0, 6.28)))
    for _ in range(2):
        out.append(rng.uniform(0, 1, n))
    for per in (3, 5, 8):
        pat = np.zeros(per)
        pat[:per // 2] = 1.0
        hv = np.tile(pat, n // per + 1)[:n]
        out.append(hv)
    x = np.arange(n) / n
    ctr = rng.uniform(0.25, 0.75)
    out.append(np.clip(1.0 - 1.5 * ((x - ctr) / 0.35) ** 2, 0, 1))
    if n % 2 == 0:
        half = n // 2
        hv = rng.uniform(0.2, 0.8, half)
        hv = np.concatenate([hv, 1.0 - hv])
        out.append(hv)
    return [proj(s, ts) for s in out]

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5
    bh, bc = None, 1e9

    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None

    if hr is not None:
        n0 = len(hr)
        dx0 = 2.0 / n0
        ts0 = n0 / 2.0
        h_w = proj(hr, ts0)
        # Phase 1: Annealing for exploration
        h_a, c_a = anneal(h_w, dx0, ts0, 3000, deadline - 200, rng, 0.08, 0.9995)
        # Phase 2: Gradient descent
        h_g, c_g = grad_desc(h_a, dx0, ts0, 20, 300, 2000, deadline - 120, 0.04, 0.0008, rng)
        # Phase 3: SLSQP
        h_s = slsqp_local(h_g, dx0, ts0, 100)
        c_s = compute_c5(h_s, dx0)
        # Phase 4: More gradient with high beta
        h_g2, c_g2 = grad_desc(h_s, dx0, ts0, 200, 800, 1500, deadline - 50, 0.015, 0.001, rng)
        # Phase 5: SLSQP again
        h_s2 = slsqp_local(h_g2, dx0, ts0, 80)
        c_s2 = compute_c5(h_s2, dx0)
        # Phase 6: Fine coordinate descent
        h_f, c_f = coord_desc(h_s2, dx0, ts0, 30, deadline - 5,
                              (0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002))
        # Track best
        for hh, cc in [(h_a, c_a), (h_g, c_g), (h_s, c_s), (h_g2, c_g2), (h_s2, c_s2), (h_f, c_f)]:
            if cc < bc:
                bh, bc = hh.copy(), cc

    # Try other n values
    base_n = len(hr) if hr is not None else 50
    for dn in (1, -1, 2, -2, 3, 4, -3, 5, -4, 6, -5, 7, -6, 8):
        nt = base_n + dn
        if nt < 25 or nt > 85 or time.time() > deadline - 40:
            continue
        dxt = 2.0 / nt
        tst = nt / 2.0
        s_list = gen_starts(nt, tst, rng, bh)
        best_local = None
        for hs in s_list[:6]:
            if time.time() > deadline - 15:
                break
            h1, c1 = grad_desc(hs, dxt, tst, 15, 250, 600, deadline - 10, 0.03, 0.001, rng)
            h2 = slsqp_local(h1, dxt, tst, 60)
            c2 = compute_c5(h2, dxt)
            h3, c3 = coord_desc(h2, dxt, tst, 10, deadline - 5,
                                (0.015, -0.015, 0.008, -0.008, 0.004, -0.004))
            if best_local is None or c3 < best_local[1]:
                best_local = (h3, c3)
        if best_local is not None and best_local[1] < bc:
            bh, bc = best_local[0].copy(), best_local[1]

    # Final polish on best
    if bh is not None:
        nb = len(bh)
        dxb = 2.0 / nb
        tsb = nb / 2.0
        if time.time() < deadline - 25:
            hf = slsqp_local(bh, dxb, tsb, 80)
            cf = compute_c5(hf, dxb)
            if cf < bc:
                bh, bc = hf, cf
        if time.time() < deadline - 8:
            hd, cd = coord_desc(bh, dxb, tsb, 20, deadline - 2,
                                (0.008, -0.008, 0.004, -0.004, 0.002, -0.002, 0.001, -0.001))
            if cd < bc:
                bh, bc = hd, cd

    if bh is None:
        bh = proj(np.full(50, 0.5), 25.0)
        bc = compute_c5(bh, 2.0 / 50)

    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
