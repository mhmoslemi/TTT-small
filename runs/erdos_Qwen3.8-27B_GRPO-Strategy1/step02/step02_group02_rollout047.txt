import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_sm(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    sm = mx + np.log(np.sum(np.exp(s))) / t
    return sm * dx + lam * (np.sum(h) - n * 0.5) ** 2

def obj_topk(h, t, k, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx]
    mx = np.max(c)
    s = t * (c - mx)
    sm = mx + np.log(np.sum(np.exp(s))) / t
    return sm * dx + lam * (np.sum(h) - n * 0.5) ** 2

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        m = h < 0.999 if d > 0 else h > 0.001
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0, 1, out=h)
    return h

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def bin_push(h, n, sharp=12.0):
    return proj(0.5 * (1.0 + np.tanh(sharp * (h - 0.5))), n)

def grad_topk_corr(h, n, k_vals):
    """Gradient of topk correlation objective w.r.t. h."""
    grad = np.zeros(n)
    for k in k_vals:
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            j = np.arange(lo1, hi1 + 1)
            grad[j] += (1.0 - h[j - k + n - 1])
        lo2 = max(0, n - 1 - k)
        hi2 = min(n - 1, 2 * n - 2 - k)
        if lo2 <= hi2:
            j = np.arange(lo2, hi2 + 1)
            grad[j] -= h[j + k - n + 1]
    return grad

def grad_descent_polish(h, n, steps, lr_init):
    h = h.copy()
    corr_len = 2 * n - 1
    for step in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = max(3, corr_len // 10)
        top_idx = np.argsort(corr)[-top_k:]
        grad = grad_topk_corr(h, n, top_idx)
        lr = lr_init * (0.995 ** step)
        h = h - lr * grad / n
        h = proj(h, n)
    return h

def coord_search(h, n, max_rounds=300, rng=None):
    h = h.copy()
    dx = 2.0 / n
    for _ in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = np.max(corr) * dx
        top_idx = np.argsort(corr)[-4:]
        improved = False
        mid = (h > 0.15) & (h < 0.85)
        candidates = np.where(mid)[0]
        if len(candidates) == 0:
            candidates = np.where(h > 0.05 & (h < 0.95))[0]
        if len(candidates) == 0:
            break
        rng2 = np.random.RandomState(_ + 7)
        rng2.shuffle(candidates)
        for j in candidates[:40]:
            for delta in [0.04, -0.04, 0.08, -0.08]:
                h2 = h.copy()
                h2[j] = h[j] + delta
                if h2[j] < 0 or h2[j] > 1:
                    continue
                diff = -delta
                adj = (h2 < 0.95) if diff > 0 else (h2 > 0.05)
                adj[j] = False
                if not adj.any():
                    continue
                # Pick adjustment target that best reduces top lags
                best_k = -1
                best_c5 = np.inf
                adj_idx = np.where(adj)[0]
                for kk in adj_idx[:8]:
                    h3 = h2.copy()
                    h3[kk] += diff
                    if h3[kk] < 0 or h3[kk] > 1:
                        continue
                    c = np.max(np.correlate(h3, 1.0 - h3, mode='full')) * dx
                    if c < best_c5:
                        best_c5 = c
                        best_k = kk
                if best_k >= 0 and best_c5 < c5_cur - 1e-9:
                    h = h.copy()
                    h[j] += delta
                    h[best_k] += diff
                    h = proj(h, n)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h
