    # h is 0/1 array with exactly n/2 ones
    g = 1.0 - h
    corr = np.correlate(h, g, 'full')
    return np.max(corr) * (2.0 / len(h))

def binary_hill_climb(h, rng, n_iter, temperature=0.0):
    n = len(h)
    best_c5 = binary_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        # Choose a random one to flip to zero
        i = ones[rng.randint(len(ones))]
        # Choose a random zero to flip to one
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Try the flip
        h[i] = 0
        h[j] = 1
        new_c5 = binary_c5(h)
        
        if new_c5 < best_c5 or (temperature > 0 and rng.rand() < np.exp((best_c5 - new_c5) / temperature)):
            best_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Revert
            h[i] = 1
            h[j] = 0
    
    return h, best_c5

        T0, T1 = 0.05, 0.001
        n_iter = min(3000000, int(tleft() * 8000))
        cool = (T1 / T0) ** (1.0 / max(n_iter, 1))
        T = T0
        ones_l = np.where(h_sa == 1)[0].tolist()
        zeros_l = np.where(h_sa == 0)[0].tolist()
        
        for it in range(n_iter):
            ii = rng.randint(len(ones_l))
            jj = rng.randint(len(zeros_l))
            i = ones_l[ii]
            j = zeros_l[jj]
            h_sa[i] = 0.0
            h_sa[j] = 1.0
            c = float(np.max(np.correlate(h_sa, 1.0 - h_sa, 'full')))
            d = c - b
            if d < 0 or np.log(rng.rand()) < -d / T:
                b = c
                ones_l[ii] = j
                zeros_l[jj] = i
            else:
                h_sa[i] = 1.0
                h_sa[j] = 0.0
            T *= cool
        
        if b < best_bin:
            best_bin = b
            h_best_bin = h_sa.copy()
    
    # Convert best binary to c5
    c5_bin = best_bin * 2.0 / n
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_best_bin.copy()
    
    # Phase 2: Continuous optimization from best
    if tleft() > 100:
        h_cont = proj(best_h + rng.randn(n) * 0.01, n)
        # L-BFGS-B with penalty
        def obj_pen(h, t, lam):
            corr = np.correlate(h, 1.0 - h, 'full')
            mx = np.max(corr)
            s = t * (corr - mx)
            soft = mx + np.log(np.sum(np.exp(s))) / t
            pen = lam * (np.sum(h) - n / 2.0) ** 2
            return soft * (2.0 / n) + pen

        for t_val in [50, 200, 1000]:
            if tleft() < 20:
                break
            lam = 1000.0
            r = minimize(obj_pen, h_cont, args=(t_val, lam),
                        method='L-BFGS-B',
                        bounds=[(0,1)]*n,
                        options={'maxiter': 3000, 'ftol': 1e-14})
            hc = proj(np.clip(r.x, 0, 1), n)
            c5 = compute_c5(hc)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
                h_cont = hc.copy()
    
    return best_h, best_c5, n
