    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        np.clip(h, 0.0, 1.0, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break
        step = d / c
        # Don't overshoot
        if d > 0:
            step = min(step, 1.0 - h[m].max())
        else:
            step = max(step, -h[m].min())
        h[m] += step
    return h

def smooth_obj_and_grad(h, ts, beta, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr * dx
    scaled -= scaled.max()
    w = np.exp(scaled)
    sw = w.sum()
    w /= sw
    smax = (dx / beta) * (np.log(sw) + scaled.max())
    
    val = smax + lam * (h.sum() - ts) ** 2
    
    # Gradient
    g = np.zeros(n)
    for j in range(n):
        s_vals = np.arange(-(n-1), n)
        k_vals = s_vals + (n-1)
        v1 = (j + s_vals >= 0) & (j + s_vals < n)
        v2 = (j - s_vals >= 0) & (j - s_vals < n)
        g[j] = np.dot(w[k_vals[v1]], (1.0 - h[j + s_vals[v1]])) - \
               np.dot(w[k_vals[v2]], h[j - s_vals[v2]])
    g *= dx
    g += 2.0 * lam * (h.sum() - ts)
    
    return val, g
