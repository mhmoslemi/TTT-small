def soft_max_obj(h, T):

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    shifted = np.clip(shifted, -100.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def sum_con(x, n):
    return np.sum(x) - n * 0.5

def fix_sum(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12:
                    break
    return h

def resample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return fix_sum(h, new_n)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def do_slsqp(h, n, T, max_iter):
    try:
        h0 = fix_sum(h, n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
        res = opt.minimize(soft_max_obj, h0, args=(T,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-18})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def active_set_obj(x, T, top_idx, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_vals = corr[top_idx]
    if T < 1e-15:
        return float(np.max(top_vals) * dx)
    c_max = np.max(top_vals)
    shifted = top_vals - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def do_slsqp_active(h, n, T, top_idx, max_iter):
    try:
        h0 = fix_sum(h, n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
        res = opt.minimize(active_set_obj, h0, args=(T, top_idx, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-18})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def get_active_indices(h, n, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.argsort(corr)[-top_k:]

def subgradient_step(h, n, step_size):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    threshold = c_max - 0.03 * c_max
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    n_half = n - 1
    grad = np.zeros(n)
    for m in active:
        lag = m - n_half
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        for j in range(lo1, hi1):
            grad[j] += (1.0 - h[j - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        for j in range(lo2, hi2):
            grad[j] -= h[j + lag]
    grad /= len(active)
    grad -= np.mean(grad)
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, n)

def coord_local_search(h, n, n_sweeps=50, delta=0.02):
    h = h.copy()
    n_half = n - 1
    for sweep in range(n_sweeps):
        for j in range(n):
            best_val = h[j]
            best_c = compute_c5(h)
            for dv in [-delta, -delta*0.5, delta*0.5, delta]:
                h[j] = np.clip(h[j] + dv, 0.0, 1.0)
                h[j2] = h.copy()
                h[j2] = h
                c = compute_c5(fix_sum(h, n))
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = h[j]
            h[j] = best_val
        h = fix_sum(h, n)
        delta *= 0.7
    return h

    t_start = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()

        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best - 1e-15:

            n_best = n_cand
            c5_best = c5_cand
            return True
        return False

    # Phase 1: Multi-temperature SLSQP (coarse to fine)
    T_schedule = [0.04, 0.01, 0.003, 0.0008, 0.0003, 0.0001, 0.00004]
    for T in T_schedule:
