    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_step(h, n_lags_top=3, step_size=0.05):
    """Take a sub-gradient step on C5"""
    n = len(h)
    corr = get_correlations(h)
    # Get top lags
    top_idx = np.argsort(corr)[-n_lags_top:]
    # Compute sub-gradient for each top lag, average them
    sg = np.zeros(n)
    for k_idx in top_idx:
        # lag = k_idx - (n-1)
        lag = k_idx - (n - 1)
        for j in range(n):
            # h[j] in first arg: contributes (1 - h[j+lag]) if j+lag valid
            jp = j + lag
            jm = j - lag
            coeff = 0.0
            if 0 <= jp < n:
                coeff += (1.0 - h[jp])
            if 0 <= jm < n:
                coeff += (-h[jm])
            sg[j] += coeff
    sg /= n_lags_top
    # Step
    h_new = h - step_size * sg
    return project_feasible(h_new, n)

def coord_descent(h, n_rounds=10, grid_points=21):
    """Coordinate descent: for each j, find optimal h[j]"""
    n = len(h)
    x_grid = np.linspace(0, 1, grid_points)

        for j in range(n):

            best_val = old_val

            h = project_feasible(h, n)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

            total = room.sum()
            if total < 1e-14:

            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
