import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_corr_and_topk(h, N, k_active):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    # Get top-k indices
    topk_idx = np.argsort(corr)[::-1][:k_active]
    topk_vals = corr[topk_idx]
    return corr, true_c5, topk_idx, topk_vals

def compute_topk_grad(h, N, topk_idx, weights):
    """Compute weighted sum of gradients for given lags."""
    grad = np.zeros(N)
    center = N - 1
    for t, ai in enumerate(topk_idx):
        L = ai - center
        wi = weights[t]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            # h[i+L] * (1-h[i]) for i=0..N-1-L
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            # h[i] * (1-h[i+M]) for i=0..N-1-M
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad

def gradient_descent(h, N, deadline, k_active=15, tau=0.05, lr_init=0.05):
    """Top-k focused gradient descent with adaptive learning rate."""
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/N)
    dx = 2.0 / N
    target = N / 2.0
    lr = lr_init
    momentum = np.zeros(N)
    beta = 0.9
    it = 0
    
    while time.time() < deadline:
        corr, c5_val, topk_idx, topk_vals = get_corr_and_topk(h, N, k_active)
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Soft weights for top-k
        sc = topk_vals / tau
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
        
        grad = compute_topk_grad(h, N, topk_idx, w)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        direction = momentum
        
        # Adaptive step: try to reduce the max
        h_new = h - lr * direction
