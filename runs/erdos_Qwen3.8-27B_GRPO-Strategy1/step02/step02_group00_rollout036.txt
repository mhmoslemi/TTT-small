        if t > 1e-14:

        if v < bv - 1e-15:
            best = h_try
            bv = v
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.55
            if lr < 1e-6:
                break
    return best, bv

def targeted_local(h, n, max_iter=400):
    best = h.copy()
    bv = compute_c5(best)
    for it in range(max_iter):
        corr = np.correlate(best, 1.0 - best, mode='full')
        top_k = np.argsort(corr)[-6:]
        improved = False
        for k in top_k:
            s = k - (n - 1)
            if s == 0:
                continue
            sa = abs(s)
            if s > 0:
                conf = best[:n - sa] * (1.0 - best[sa:])
                j = int(np.argmax(conf))
                i1, i2 = j, j + sa
            else:
                conf = best[sa:] * (1.0 - best[:n - sa])
                j = int(np.argmax(conf))
                i1, i2 = j + sa, j
            for d in [0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35]:
                ht = best.copy()
                ht[i1] -= d
                ht[i2] += d
                ht = proj(ht, n)
                v = compute_c5(ht)
                if v < bv - 1e-14:
                    best = ht
                    bv = v
                    improved = True
                    break
        if not improved:
            for _ in range(10):
                i1 = np.random.randint(n)
                i2 = np.random.randint(n)
                if i1 == i2:
                    continue
                d = np.random.uniform(0.04, 0.2) * np.random.choice([1.0, -1.0])
                ht = best.copy()
                ht[i1] += d
                ht[i2] -= d
                ht = proj(ht, n)
                v = compute_c5(ht)
                if v < bv - 1e-14:
                    best = ht
                    bv = v
                    improved = True
                    break
        if not improved:
            break
    return best, bv

def basin_hop(h, n, T, n_iter, t_end):
    best = h.copy()
    bv = compute_c5(best)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        step = 0.1 * (1.0 - 0.7 * it / n_iter) + 0.02
        h_p = proj(best + step * np.random.randn(n), n)
        hc = do_slsqp(h_p, T, n, mi=200, ft=1e-12)
        v = compute_c5(hc)
        temp = 0.005 * (1.0 - it / n_iter) + 0.0003
        if v < bv or np.random.rand() < np.exp(-(v - bv) / temp):
            best = hc
            bv = v
    return best, bv

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T (20% budget)
    t1 = t0 + budget_s * 0.20
    for T in [25, 80, 250, 800, 2500, 7000]:
        if time.time() > t1:
            break
        hc = do_slsqp(best_h, T, n, mi=900, ft=1e-15)
        v = compute_c5(hc)
        if v < best_c5:
            best_c5 = v
            best_h = hc

    # Phase 2: Projected gradient steps (12% budget)
    t2 = t0 + budget_s * 0.32
    while time.time() < t2:
        hc, vc = proj_grad_steps(best_h, 400, n, nsteps=600, lr0=0.003)
        if vc < best_c5:
            best_c5 = vc
            best_h = hc

    # Phase 3: SLSQP + targeted local alternating (22% budget)
    t3 = t0 + budget_s * 0.54
    while time.time() < t3:
        hc = do_slsqp(best_h, 3000, n, mi=700, ft=1e-15)
        v = compute_c5(hc)
        if v < best_c5:
            best_c5 = v
            best_h = hc
        hc, vc = targeted_local(best_h, n, max_iter=350)
        if vc < best_c5:
            best_c5 = vc
            best_h = hc

    # Phase 4: Basin hopping for escape (15% budget)
    t4 = t0 + budget_s * 0.69
    bh_h, bh_c5 = basin_hop(best_h, n, 600, 50, t4)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = proj(bh_h, n)

    # Phase 5: Try lower n (15% budget)
    t5 = t0 + budget_s * 0.84
    x_old = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in [60, 80, 100]:
        if time.time() > t5:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = proj(hi, n_try)
        for T in [150, 800, 3500]:
            if time.time() > t5:
                break
            hc = do_slsqp(hi, T, n_try, mi=500)
            v = compute_c5(hc)
            hc, vc = targeted_local(hc, n_try, max_iter=250)
            if vc < best_c5:
                best_c5 = vc
                best_h = proj(hc, n_try)
                n = n_try

    # Phase 6: More targeted refinement + gradient (10% budget)
    t6 = t0 + budget_s * 0.94
    n_cur = len(best_h)
    while time.time() < t6:
        hc, vc = proj_grad_steps(best_h, 1000, n_cur, nsteps=400, lr0=0.002)
        if vc < best_c5:
            best_c5 = vc
            best_h = hc
        hc = do_slsqp(best_h, 10000, n_cur, mi=800, ft=1e-15)
        v = compute_c5(hc)
        if v < best_c5:
            best_c5 = v
            best_h = hc
        hc, vc = targeted_local(best_h, n_cur, max_iter=500)
        if vc < best_c5:
            best_c5 = vc
            best_h = hc

    # Final polish
    for T in [20000, 60000]:
        if time.time() > t0 + budget_s * 0.99:
            break
        hc = do_slsqp(best_h, T, len(best_h), mi=1200, ft=1e-16)
        v = compute_c5(hc)
        if v < best_c5:
            best_c5 = v
            best_h = hc

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
