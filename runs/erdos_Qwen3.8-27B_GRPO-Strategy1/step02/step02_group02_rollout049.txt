import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_obj_grad(h, t, topk, lam):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

            hk = lbfgs_step(hc, 280, t_val, 15, 1.5, maxiter=2000, ftol=1e-15, gtol=1e-13)
            c5 = compute_c5(hk)
            if c5 < bc:
                bc = c5
                h3 = hk
            hc = hk

    # Phase 7: Random restarts to escape local optima
    if tleft() > 80:
        for _ in range(4):
            if tleft() < 15:
                break
            scale = 0.01 + 0.01 * _
            h_pert = proj(h3 + rng.randn(280) * scale, 280)
            for t_val in [500, 2000]:
                if tleft() < 10:
                    break
                hk = lbfgs_step(h_pert, 280, t_val, 15, 1.5, maxiter=1500, gtol=1e-12)
                c5 = compute_c5(hk)
                if c5 < bc:
                    bc = c5
                    h3 = hk
                    h_pert = hk

    # Phase 8: Ultra-fine final polish
    if tleft() > 30:
        for t_val in [20000, 60000, 150000]:
            if tleft() < 10:
                break
            hk = lbfgs_step(h3, 280, t_val, 5, 1.0, maxiter=4000, ftol=1e-17, gtol=1e-14)
            c5 = compute_c5(hk)
            if c5 < bc:
                bc = c5
            h3 = hk
            hg = projected_grad_step(h3, t_val, 5, 1.0, lr=0.005, n_steps=15)
            c5g = compute_c5(hg)
            if c5g < bc:
                bc = c5g
                h3 = hg

    # Final exact evaluation
    h3 = proj(h3, 280)
    final_c5 = compute_c5(h3)
    return h3, final_c5, 280
