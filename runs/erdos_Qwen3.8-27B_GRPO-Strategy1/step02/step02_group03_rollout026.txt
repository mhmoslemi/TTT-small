    for m in (1, 2, 3):
        x = np.arange(n) / n
        out.append(proj(0.5 + 0.25 * np.sin(2 * np.pi * m * x), ts))
    return out

def opt_full(h0, dx, ts, rng, deadline):
    h = slsqp(h0, dx, ts, 250)
    c = compute_c5(h, dx)
    if time.time() < deadline - 25:
        h2, c2 = gdes(h, dx, ts, 2000, deadline - 12)
        if c2 < c:
            h, c = h2, c2
    if time.time() < deadline - 12:
        h3, c3 = pair_opt(h, dx, ts, rng, 400, deadline - 6)
        if c3 < c:
            h, c = h3, c3
    if time.time() < deadline - 8:
        h4 = slsqp(h, dx, ts, 200)
        c4 = compute_c5(h4, dx)
        if c4 < c:
            h, c = h4, c4
    if time.time() < deadline - 4:
        h5, c5 = pair_opt(h, dx, ts, rng, 200, deadline - 2)
        if c5 < c:
            h, c = h5, c5
    if time.time() < deadline - 1:
        h6 = slsqp(h, dx, ts, 100)
        c6 = compute_c5(h6, dx)
        if c6 < c:
            h, c = h6, c6
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3
    bh, bc = None, 1e9
    try:
        hr = np.array(initial_h_values, dtype=float)
    except NameError:
        hr = None
    if hr is not None:
        n0 = len(hr)
        dx0 = 2.0 / n0
        ts0 = n0 / 2.0
        h_start = proj(hr, ts0)
        bh, bc = opt_full(h_start, dx0, ts0, rng, deadline - 120)
    n_base = len(hr) if hr is not None else 50
    if bh is not None:
        for dn in (1, -1, 2, -2, 3, 4, -3, -4):
            nt = n_base + dn
            if nt < 20 or nt > 120 or time.time() > deadline - 60:
                continue
            dxt = 2.0 / nt
            tst = nt / 2.0
            starts = make_starts(nt, tst, rng, bh)
            for hs in starts[:3]:
                if time.time() > deadline - 20:
                    break
                hr2, cr2 = opt_full(hs, dxt, tst, rng, deadline - 10)
                if cr2 < bc:
                    bh, bc = hr2, cr2
    if bh is not None and time.time() < deadline - 80:
        nbest = len(bh)
        dxbest = 2.0 / nbest
        tsbest = nbest / 2.0
        for amp in (0.12, 0.06, 0.03, 0.015):
            if time.time() > deadline - 15:
                break
            hp = proj(bh + amp * rng.randn(nbest), tsbest)
            hr, cr = opt_full(hp, dxbest, tsbest, rng, deadline - 5)
            if cr < bc:
                bh, bc = hr, cr
    if bh is not None:
        nf = len(bh)
        dxf = 2.0 / nf
        tsf = nf / 2.0
        hf = slsqp(bh, dxf, tsf, 200)
        cf = compute_c5(hf, dxf)
        if cf < bc:
            bh, bc = hf, cf
        hf2, cf2 = pair_opt(bh, dxf, tsf, rng, 100, t0 + budget_s - 1)
        if cf2 < bc:
            bh, bc = hf2, cf2
    if bh is None:
        bh = proj(np.full(50, 0.5), 25.0)
        bc = compute_c5(bh, 0.04)
    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
