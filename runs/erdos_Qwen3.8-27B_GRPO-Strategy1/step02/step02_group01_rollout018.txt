    return float(np.max(corr) * dx)

def get_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

            if s < 1e-14: break

        else:
            room = h
            s = room.sum()
            if s < 1e-14: break
            h -= room * (-diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, tau):
    """Smooth approximation to max of correlations"""
    corr = get_corr_array(h)
    m = corr.max()
    # Log-sum-exp for numerical stability
    shifted = corr - m
    val = m + np.log(np.sum(np.exp(shifted / tau))) * tau
    return val

def smooth_obj_pen(h, tau, pen_coeff):
    """Smooth max objective with penalty for constraints"""
    n = len(h)
    obj = smooth_obj(h, tau)
    sum_pen = pen_coeff * (h.sum() - n/2.0)**2
    return obj + sum_pen

def slsqp_smooth(h, tau, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(smooth_obj, h, args=(tau,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += room * (diff / s)
        else:
            room = h.copy()
            s = room.sum()
            if s < 1e-14:
                break
            h -= room * (-diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, tau):
    corr = get_corr(h)
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / tau))
    return m + tau * np.log(s)

def objective_smooth(h, tau):
    return smooth_obj(h, tau)

def slsqp_opt(h, tau, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(objective_smooth, h, args=(tau,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(np.clip(res.x, 0, 1), n)
        return h_out
    except Exception:
        return h

def fourier_perturb(h, mode, amp, phase=0.0):
    n = len(h)
    x = np.arange(n)
    fp = amp * np.sin(2.0 * np.pi * mode * x / n + phase)
    return h + fp

def block_shift(h, start, length, delta):
    """Shift a block of values by delta, distributing to neighbors"""
    n = len(h)
    h_new = h.copy()
    end = min(start + length, n)
    blk = h_new[start:end]
    # Add delta proportionally to room
    room = 1.0 - h_new
    blk_room = room[start:end]
    total_room = blk_room.sum()
    if total_room < 1e-14:
        return h
    h_new[start:end] += blk_room * (delta / total_room)
    return h_new
