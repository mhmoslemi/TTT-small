def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    # W[j] = sum(w[j:j+n])
    cw = np.concatenate([[0], np.cumsum(w)])
    W = cw[n:] - cw[:len(cw)-n]  # W[j] = cw[j+n] - cw[j]
    # dot[j] = sum_m w[j+m] * h[m]
    # This is like a correlation of w and h
    dot = np.correlate(w, h, mode='valid')  # dot[j] = sum_m w[j+m]*h[2n-1-n-m]...

    # Actually np.correlate(a, b, 'valid') where len(a)>=len(b):
    # output[j] = sum_k a[j+k]*b[N-b_len+k] ... I'm getting confused.
    # Let me just do it with a loop for clarity and correctness.
    dots = np.zeros(n)
    for j in range(n):
        dots[j] = np.dot(w[j:j+n], h)

                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

        h_interp = pad_or_reduce(best_h, n_try)
        for trial in range(3):
            if elapsed() > budget_s * 0.85:
                break
            h0 = h_interp + np.random.uniform(-0.05, 0.05, n_try)
            h0 = project_feasible(h0, n_try)
            for T in [150, 600, 2000]:
                if elapsed() > budget_s * 0.85:
                    break
                hc = slsqp_run(h0, T, n_try, maxiter=500, ftol=1e-14)
                c5v, _ = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    n = n_try

    # Phase 7: Final targeted refinement
    n = len(best_h)
    if elapsed() < budget_s * 0.92:
        best_h, best_c5 = corr_targeted_moves(best_h, n, n_trials=300)
        best_h, best_c5 = binary_snap(best_h, n)
        # Nesterov polish
        hc = nesterov_smooth(best_h, T=2000, n=n, steps=2000, lr_base=0.003)
        c5v, _ = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 8: Final high-T SLSQP polish
    n = len(best_h)
    for T in [5000, 15000, 50000]:
        if elapsed() > budget_s * 0.98:
            break
        hc = slsqp_run(best_h, T, n, maxiter=800, ftol=1e-16)
        c5v, _ = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
