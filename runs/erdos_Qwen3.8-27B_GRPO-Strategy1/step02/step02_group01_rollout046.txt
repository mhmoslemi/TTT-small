For the binary local search:
- State: binary array h with sum = n/2
- Move: pick a 1-position i and a 0-position j, swap them (h[i]=0, h[j]=1)
- Objective: C₅ = max_k corr(h, 1-h, 'full')[k] * dx
- Efficiency: After a swap, only a few correlation values change. Specifically, changing h[i] affects corr at shifts involving position i.

For n=90, the full correlation has 179 values. A swap of positions i and j changes at most 2n correlation values (those involving i or j). So evaluation after a swap is O(n) not O(n²).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def project_binary(h, n):
    """Project to nearest binary vector with sum = n/2"""
    h = np.sort(h)[::-1]
    h[:n//2] = 1.0
    h[n//2:] = 0.0
    return h

def binary_local_search(h, max_rounds=50):
    """Local search on binary vectors using pair swaps"""
    n = len(h)
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        corr = corr_full(h)
        c5 = np.max(corr)
        
        for i in ones:
            for j in zeros:
                # Try swap: h[i]=0, h[j]=1
                h[i] = 0.0
                h[j] = 1.0
                new_corr = corr_full(h)
                new_c5 = np.max(new_corr)
                if new_c5 < c5:
                    c5 = new_c5
                    corr = new_corr
                    ones = ones[ones != i]
                    ones = np.append(ones, j)
                    zeros = zeros[zeros != j]
                    zeros = np.append(zeros, i)
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    
    return h, np.max(corr_full(h)) * (2.0/n)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h -= room * (-diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_polish(h, maxiter=300):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        # Smooth max
        T = 100.0
        return (m + np.log(np.sum(np.exp(T*(corr - m))))/T) * dx + 0.01*abs(x.sum() - n/2)
    
    bnds = [(0.0, 1.0)]*n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'gtol': 1e-6})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

    """Generate Thue-Morse sequence of length n (padded/truncated to n)"""
    tm = np.zeros(1)
    while len(tm) < n:
        tm = np.concatenate([tm, 1.0 - tm])
    return tm[:n]

    for _ in range(n_swaps):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = h[j], h[i]
    return h

    if rng is None:

    corr = corr_full(h)
    best_c5 = np.max(corr)
    
    for _ in range(max_passes):
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        
        # Try a random subset of swaps (for speed)
        num_trials = min(500, len(ones) * len(zeros))
        for _ in range(num_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            new_corr = corr_full(h)
            new_c5 = np.max(new_corr)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                corr = new_corr
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    
    return h, best_c5 * (2.0/n)

def basin_hopping_binary(h, n_iter, rng):
    """Basin hopping on binary space"""
    n = len(h)

    best_c5 = compute_c5(best_h)

        if c5_ls < best_c5:

            best_h = h_ls.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h -= room * (-diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:n // 2]] = 1.0
    return b

def thue_morse(n):
    tm = np.array([0.0])
    while len(tm) < n:
        tm = np.concatenate([tm, 1.0 - tm])
    return tm[:n]

def exact_local_search(h):
    n = len(h)
    h = h.copy()
    corr = corr_full(h)
    best_c5 = float(np.max(corr))
    improved = True
    while improved:
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                new_c5 = float(np.max(corr_full(h)))
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
    return h, best_c5 * (2.0 / n)

def rand_local_search(h, n_passes=25, n_trials=400, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    h = h.copy()
    corr = corr_full(h)
    best_c5 = float(np.max(corr))
    for _ in range(n_passes):
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        for _ in range(n_trials):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i], h[j] = 0.0, 1.0
            new_c5 = float(np.max(corr_full(h)))
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                corr = corr_full(h)
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, best_c5 * (2.0 / n)

def basin_hop_binary(h, n_iter, rng, temp0=0.02, temp_decay=0.95):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    temp = temp0
    for it in range(n_iter):
        k_s = rng.randint(2, 6)
        h_p = best_h.copy()
        ones = np.where(h_p > 0.5)[0]
        zeros = np.where(h_p < 0.5)[0]
        for _ in range(k_s):
            a = int(rng.choice(ones))
            b = int(rng.choice(zeros))
            h_p[a], h_p[b] = h_p[b], h_p[a]
            ones = np.where(h_p > 0.5)[0]
            zeros = np.where(h_p < 0.5)[0]
        h_ls, c5_ls = rand_local_search(h_p, n_passes=12, n_trials=300, rng=rng)
        delta = c5_ls - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-8)):
            best_c5 = c5_ls
            best_h = h_ls.copy()
        temp *= temp_decay
    return best_h, best_c5

def slsqp_polish(h, maxiter=300):
    n = len(h)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        T = 200.0
        shifted = T * (corr - m)
        logsum = m + np.log(np.sum(np.exp(shifted))) / T
        pen = 0.1 * abs(x.sum() - n / 2.0)
        return logsum * dx + pen

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': 1e-7})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def greedy_construct(n, rng):
    h = np.zeros(n)
    for k in range(n // 2):
        best_pos = -1
        best_val = np.inf
        candidates = np.where(h == 0)[0]
        for pos in candidates:
            h[pos] = 1.0
            val = np.max(corr_full(h))
            h[pos] = 0.0
            if val < best_val:
                best_val = val
                best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remain():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0

    best_n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Binarize parent and local search
    h_bin = binarize(best_h, best_n)
    if best_n <= 66:
        h_ls, c5_ls = exact_local_search(h_bin)
    else:
        h_ls, c5_ls = rand_local_search(h_bin, n_passes=30, n_trials=600, rng=rng)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
        best_n = len(h_ls)

    # Phase 2: Multi-resolution binary search + basin hopping
    resolutions = [60, 72, 80, 84, 90, 96, 108, 120]
    time_budget_res = budget_s * 0.50 / len(resolutions)

    for n_try in resolutions:
        if remain() < time_budget_res * 0.5:
            break

        # Interpolate from best
        if best_n != n_try:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
        else:
            h_interp = best_h.copy()

        h_b = binarize(h_interp, n_try)

        # Local search
        if n_try <= 66:
            h_ls2, c5_ls2 = exact_local_search(h_b)
        else:
            h_ls2, c5_ls2 = rand_local_search(h_b, n_passes=20, n_trials=400, rng=rng)
        cur_n = n_try
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
            best_n = cur_n

        # Basin hopping
        if remain() > time_budget_res:
            n_iter = max(3, int(time_budget_res * 0.6 / 2.0))
            h_bh, c5_bh = basin_hop_binary(h_ls2, n_iter, rng)
            if c5_bh < best_c5:
                best_c5 = c5_bh
                best_h = h_bh.copy()
                best_n = cur_n

    # Phase 3: Thue-Morse and structured starts
    if remain() > 60:
        for n_tm in [64, 128]:
            if remain() < 30:
                break
            tm = thue_morse(n_tm)
            if tm.sum() == n_tm / 2:
                h_tm_ls, c5_tm = rand_local_search(tm, n_passes=15, n_trials=300, rng=rng)
                if c5_tm < best_c5:
                    best_c5 = c5_tm
                    best_h = h_tm_ls.copy()
                    best_n = n_tm

    # Phase 4: Greedy constructions at various sizes
    if remain() > 60:
        for n_g in [60, 80, 100]:
            if remain() < 30:
                break
            h_g = greedy_construct(n_g, rng)
            if n_g <= 66:
                h_g_ls, c5_g = exact_local_search(h_g)
            else:
                h_g_ls, c5_g = rand_local_search(h_g, n_passes=20, n_trials=400, rng=rng)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g_ls.copy()
                best_n = n_g

    # Phase 5: Intensive basin hopping on best solution
    if remain() > 40:
        n_b = len(best_h)
        h_bf = binarize(best_h, n_b)
        n_iters = max(5, int(remain() * 0.25 / 2.5))
        h_fh, c5_fh = basin_hop_binary(h_bf, n_iters, rng, temp0=0.015)
        if c5_fh < best_c5:
            best_c5 = c5_fh
            best_h = h_fh.copy()
            best_n = n_b

        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p
            best_n = len(h_p)

    # Second polish attempt with different temperature
    if remain() > 5:
        h_p2 = slsqp_polish(best_h + rng.randn(len(best_h)) * 0.01, maxiter=200)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
