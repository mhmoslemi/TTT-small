At the optimum, Σλ_k = 1 (complementary slackness on t), and we're minimizing Σ_k λ_k corr_k(h) over h.

1. Initialize λ uniformly

   This is a QP! (corr_k is bilinear in h, but Σ_k λ_k corr_k(h) is a quadratic form in h)

import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def lse_grad(h, dx, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    o = n - 1
    if beta > 100:
        # Use top-5 only for stability
        idx = np.argpartition(cr, -5)[-5:]
        w = np.zeros(2*n-1)
        w[idx] = 1.0 / 5.0
    elif beta > 1:
        cr_max = cr.max()
        w = np.exp(beta * (cr - cr_max))
        s = w.sum()
        if s < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= s
    else:
        w = np.ones(2*n-1) / (2*n-1)
    g = np.zeros(n)
    for j in range(n):
        k1, k2 = max(-(n-1), -j), min(n-1, n-1-j)
        if k1 <= k2:
            g[j] += np.dot(w[o+k1:o+k2+1], 1.0 - h[j+k1:j+k2+1])
        k3, k4 = max(-(n-1), j-(n-1)), min(n-1, j)
        if k3 <= k4:
            g[j] -= np.dot(w[o+k3:o+k4+1], h[j-k3:j-k4+1])
    return g * dx

    best_h = h.copy()
    stall = 0
    it = 0
    max_it = 100000
    while it < max_it and time.time() < deadline:
        it += 1
        if phase == 1:
            frac = min(1.0, it / 3000.0)
            beta = 2.0 + 30.0 * frac
            lr = 0.015 / (1.0 + 0.0002 * it)
        else:
            frac = min(1.0, it / 1000.0)
            beta = 50.0 + 450.0 * frac
            lr = 0.005 / (1.0 + 0.001 * it)
        g = lse_grad(h, dx, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        step_dir = -g / gn
        improved = False
        for mult in (lr, lr*0.5, lr*0.25, lr*0.125, lr*0.0625):
            h2 = proj(h + mult * step_dir, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-14:
                h, bc = h2, c2
                best_h = h.copy()
                stall = 0
                improved = True
                break
            elif c2 <= bc + 1e-15:
                h = h2
                stall = 0
                improved = True
        if not improved:
            stall += 1
            if stall > 200:
                break
            if stall % 50 == 0:
                h = proj(h + 0.01 * rng.randn(n), ts)
                bc = compute_c5(h, dx)
    return best_h, compute_c5(best_h, dx)

def equalize(h0, dx, ts, deadline, rng):
    """Directly try to equalize the top correlations"""
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    o = n - 1
    for rnd in range(50):
        if time.time() > deadline:
            break
        cr = np.correlate(h, 1.0 - h, mode='full')
        mx = cr.max()
        # Find active lags (within 5% of max or top 8)
        tol = 0.05 * mx
        active = set(np.where(cr >= mx - tol)[0])
        active = set(list(active)[:12])
        # For each active lag, compute gradient and try to reduce it
        improved = False
        for ci in list(active):
            k = ci - o
            g_k = np.zeros(n)
            if k >= 0:
                for i in range(n - k):
                    g_k[i] += (1 - h[i+k])
                    g_k[i+k] -= h[i]
            else:
                kk = -k
                for i in range(n - kk):
                    g_k[i+kk] += (1 - h[i])
                    g_k[i] -= h[i+kk]
            # Try small step in -g_k direction
            for step in (0.002, 0.001, 0.0005):
                h2 = proj(h - step * g_k, ts)
                c2 = compute_c5(h2, dx)
                if c2 < bc - 1e-14:
                    h, bc = h2, c2
                    improved = True
                    break
        if not improved:
            # Try random pair swaps
            for _ in range(100):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                di = h[i] - h[j]
                h[i] -= di * 0.1
                h[j] += di * 0.1
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < bc:
                    bc = c
                    improved = True
                else:
                    h[i] += di * 0.1
                    h[j] -= di * 0.1
                    h = proj(h, ts)
            if not improved:
                break
    return h, bc

def snap_01(h0, dx, ts, rng, deadline):
    """Push values toward 0 or 1"""
    n = len(h0)

                    imp = True
                else:

                    h = proj(h, ts)

def slsqp_polish(h0, dx, ts, mit=200):
    n = len(h0)
    def obj(x):
        h = x[:n]
        cr = np.correlate(h, 1.0-h, mode='full')
        b = 80.0
        return float(np.logaddexp.reduce(b * cr) / b)
    def obj_g(x):
        h = x[:n]
        return lse_grad(h, dx, 80.0)
    def sum_c(x):
        return x[:n].sum() - ts
    bnds = [(0,1)] * n
    cons = [{'type': 'eq', 'fun': sum_c}]
    try:
        r = minimize(obj, h0.copy(), jac=obj_g, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-15})
        if r.x is not None:
            return proj(r.x[:n], ts)
    except:
        pass
    return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def lse_grad(h, dx, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    o = n - 1
    if beta > 100:
        topk = min(6, 2 * n - 1)
        idx = np.argpartition(cr, -topk)[-topk:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / topk
    elif beta > 1:
        cr_max = cr.max()
        w = np.exp(beta * (cr - cr_max))
        s = w.sum()
        w = w / s if s > 1e-30 else np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    g = np.zeros(n)
    for p in range(n):
        lag_lo = max(-(n - 1), p - (n - 1))
        lag_hi = min(n - 1, p)
        if lag_lo <= lag_hi:
            lags = np.arange(lag_lo, lag_hi + 1)
            g[p] += np.dot(w[o + lags], 1.0 - h[p - lags])
        lag2_lo = max(-(n - 1), -p)
        lag2_hi = min(n - 1, n - 1 - p)
        if lag2_lo <= lag2_hi:
            lags2 = np.arange(lag2_lo, lag2_hi + 1)
            g[p] -= np.dot(w[o + lags2], h[p + lags2])
    return g * dx

def grad_phase(h0, dx, ts, deadline, rng, mode):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    best_h = h.copy()
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if mode == 1:
            frac = min(1.0, it / 2000.0)
            beta = 2.0 + 28.0 * frac
            lr = 0.012 / (1.0 + 0.0003 * it)
        else:
            frac = min(1.0, it / 800.0)
            beta = 40.0 + 460.0 * frac
            lr = 0.004 / (1.0 + 0.002 * it)
        g = lse_grad(h, dx, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        sd = -g / gn
        improved = False
        for mult in (lr, lr * 0.5, lr * 0.25, lr * 0.125, lr * 0.0625, lr * 0.01):
            h2 = proj(h + mult * sd, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-14:
                h, bc = h2, c2
                best_h = h.copy()
                stall = 0
                improved = True
                break
            elif c2 <= bc + 1e-15:
                h = h2
                stall = 0
                improved = True
        if not improved:
            stall += 1
            if stall > 300:
                break
            if stall % 100 == 0:
                h = proj(h + 0.008 * rng.randn(n), ts)
                bc = compute_c5(h, dx)
    return best_h, compute_c5(best_h, dx)

def equalize(h0, dx, ts, deadline, rng):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    o = n - 1
    for rnd in range(200):
        if time.time() > deadline:
            break
        cr = np.correlate(h, 1.0 - h, mode='full')
        mx = cr.max()
        active = np.where(cr >= mx - 0.03 * mx)[0]
        if len(active) > 8:
            active = active[np.argsort(cr[active])[-8:]]
        improved = False
        for ci in active:
            lag = ci - o
            g_k = np.zeros(n)
            for p in range(n):
                if 0 <= p - lag < n:
                    g_k[p] += (1.0 - h[p - lag])
                if 0 <= p + lag < n:
                    g_k[p] -= h[p + lag]
            for step in (0.001, 0.0005, 0.0002, 0.0001):
                h2 = proj(h - step * g_k, ts)
                c2 = compute_c5(h2, dx)
                if c2 < bc - 1e-14:
                    h, bc = h2, c2
                    improved = True
                    break
        if not improved:
            for _ in range(50):
                i, j = int(rng.randint(n)), int(rng.randint(n))
                if i == j:
                    continue
                d = (h[i] - h[j]) * 0.05
                h[i] -= d
                h[j] += d
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < bc:
                    bc = c
                    improved = True
                else:
                    h[i] += d
                    h[j] -= d
                    h = proj(h, ts)
            if not improved:
                break
    return h, bc

def snap_01(h0, dx, ts, rng, deadline):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    for _ in range(80):
        if time.time() > deadline:
            break
        imp = False
        for j in rng.permutation(n):
            for target in (0.0, 1.0):
                if abs(h[j] - target) < 0.003:
                    continue
                old = h[j]
                h[j] = target
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c <= bc + 1e-12:
                    bc = min(bc, c)
                    imp = True
                else:
                    h[j] = old
                    h = proj(h, ts)
        if not imp:
            break
    return h, bc

def slsqp_polish(h0, dx, ts, mit=200):
    n = len(h0)
    def obj(x):
        h = x[:n]
        cr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.logaddexp.reduce(80.0 * cr) / 80.0)
    def obj_g(x):
        return lse_grad(x[:n], dx, 80.0)
    def sum_c(x):
        return x[:n].sum() - ts
    bnds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': sum_c}]
    try:
        r = minimize(obj, h0.copy(), jac=obj_g, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-15})
        if r.x is not None:
            return proj(r.x[:n], ts)
    except Exception:
        pass
    return h0.copy()
