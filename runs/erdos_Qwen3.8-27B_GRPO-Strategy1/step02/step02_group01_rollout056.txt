def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:

            h += room * (diff / total)

            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h -= room * ((-diff) / total)
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h -= room * ((-diff) / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, s):
    """Gradient of correlate(h, 1-h, 'full')[n-1+s] w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        m = n - s
        grad[:m] = 1.0
        grad[:m] -= h[s:]
        grad[s:] -= h[:m]
    else:
        t = -s
        m = n - t
        grad[t:] = 1.0
        grad[t:] -= h[:m]
        grad[:m] -= h[t:]
    return grad

def subgradient_optimize(h, n_iter=20000, n_active=5, alpha_init=0.05, time_limit=30):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
    alpha = alpha_init
    t_start = time.time()
    
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Find top n_active lags
        top_idx = np.argsort(corr)[::-1][:n_active]
        # Average gradient
        grad = np.zeros(n)
        for k in top_idx:
            s = int(k) - (n - 1)
            grad += corr_gradient(h, s)
        grad /= n_active
        
        # Backtracking line search
        for _ in range(20):
            h_new = h - alpha * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)[0]
            if c5_new < best_c5 or c5_new < compute_c5(h)[0]:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                alpha = min(alpha * 1.1, 0.5)
                break
            else:
                alpha *= 0.6
        else:
            if alpha < 1e-10:
                break
    
    return best_h, best_c5

def binary_local_search(h0, time_limit=20):
    n = len(h0)
    h = (h0 >= 0.5).astype(float)
    # Ensure exactly n//2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    target = n // 2
    while len(ones) > target:
        idx = np.random.choice(ones)
        h[idx] = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    while len(ones) < target:
        idx = np.random.choice(zeros)
        h[idx] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
    t_start = time.time()
    
    while time.time() - t_start < time_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = None
        best_pair = None
        best_new_c5 = best_c5
        
        # Sample a subset of pairs if too many
        pairs = list(zip(ones, zeros))
        np.random.shuffle(pairs)
        
        for i, j in pairs:
            if time.time() - t_start > time_limit:
                break
            h[i] = 0
            h[j] = 1
            c5 = compute_c5(h)[0]
            h[i] = 1
            h[j] = 0
            if c5 < best_new_c5:
                best_new_c5 = c5
                best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            best_c5 = best_new_c5
            best_h = h.copy()
            improved = True
    
    return best_h, best_c5

def slsqp_polish(h, maxiter=500):
    n = len(h)
    def objective(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * dx
    
    def sum_con(x):
        return x.sum() - n / 2.0
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
