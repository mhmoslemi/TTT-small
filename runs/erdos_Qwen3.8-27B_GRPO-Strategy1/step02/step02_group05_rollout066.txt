```python
    best_h = h.copy()

        tau = tau0 * (tau1 / tau0) ** frac
        sc = tau * corr
        sc = sc - sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws > 1e-30:
            w /= ws
        else:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        grad = fast_grad(h, w)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)

            best_h = h.copy()
        if it % n_decay_iters == 0:

        if lr < 1e-10:
```

```python
    lr = lr0

    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    while time.time() < deadline:
        frac = min(it / 5000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws > 1e-30:
            w /= ws
        else:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        grad = fast_grad(h, w)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
```

- If f(x1) > f(x2): new interval is [x1, b], x2 becomes the new x1, compute new x1
- If f(x1) < f(x2): new interval is [a, x2], x1 becomes the new x2, compute new x2

```python
def minimize_1d(corr, gi, old, k2i, N):
    """Minimize max_k corr_at_x_k over x in [0,1]."""
    # Grid for initial bracket
    xg = np.linspace(0, 1, 33)
    vals = np.zeros(33)
    for ii in range(33):
        x = xg[ii]
        v = np.max(corr + (x - old) * gi)
        if k2i >= 0:
            v = max(v, corr[k2i] + x*(1-x) - old*(1-old))
        vals[ii] = v
    idx = np.argmin(vals)
    a = xg[max(0, idx-1)]
    b = xg[min(32, idx+1)]

    phi = 0.6180339887
    x1 = b - phi*(b-a)
    x2 = a + phi*(b-a)
    f1 = _eval_g(x1, corr, gi, old, k2i)
    f2 = _eval_g(x2, corr, gi, old, k2i)

            x2 = b - phi*(b-a)

        else:

            x1 = b - phi*(b-a)
            f1 = _eval_g(x1, corr, gi, old, k2i)

    if k2i >= 0:
        v = max(v, corr[k2i] + x*(1-x) - old*(1-old))
```

```python
            x2 = a + (1-phi)*(b-a)
            # Eval x1
            tv = np.max(corr + (x1-old)*gi)
            if k2i >= 0:
                tv = max(tv, corr[k2i]+x1*(1-x1)-old*(1-old))
            f1 = tv
            # Eval x2
            tv = np.max(corr + (x2-old)*gi)
            if k2i >= 0:
                tv = max(tv, corr[k2i]+x2*(1-x2)-old*(1-old))
            f2 = tv
            
            for _ in range(25):
                if b-a < 1e-7: break
                if f1 > f2:
                    a = x1; x1 = x2; f1 = f2
                    x2 = a + (1-phi)*(b-a)
                    tv = np.max(corr + (x2-old)*gi)
                    if k2i >= 0:
                        tv = max(tv, corr[k2i]+x2*(1-x2)-old*(1-old))
                    f2 = tv
                else:
                    b = x2; x2 = x1; f2 = f1
                    x1 = b - phi*(b-a)
                    tv = np.max(corr + (x1-old)*gi)
                    if k2i >= 0:
                        tv = max(tv, corr[k2i]+x1*(1-x1)-old*(1-old))
                    f1 = tv
            
            x_cand = (a+b)/2.0
            
            # Apply
            h[i] = x_cand
            # Incremental corr update
            corr += (x_cand - old) * gi
            if k2i >= 0:
                corr[k2i] -= (x_cand-old)*gi[k2i]
                corr[k2i] += x_cand*(1-x_cand) - old*(1-old)
        
        h = project_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
def adam_topK(h, N, target, dx, deadline, lr0, K, decay, decay_iter):
    """Adam using top-K subgradient instead of soft-max."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    while time.time() < deadline:
        # Find top-K lags
        max_val = np.max(corr)
        # Get indices of top K
        topK_idx = np.argsort(corr)[-K:]
        active = np.zeros(2*N-1)
        active[topK_idx] = 1.0 / K
        
        grad = fast_grad(h, active)
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1-b1**it)
        vh = v / (1-b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % decay_iter == 0:
            lr *= decay
        if lr < 1e-10:
            break
    return best_h, best_c5
```

This is simpler and potentially more effective because it directly targets the active constraints.
