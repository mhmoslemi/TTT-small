        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(120)
        best_h[:60] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: SLSQP with gradual T (reliable baseline)
    for T in [20, 80, 200, 500, 1500, 4000]:
        if elapsed() > budget_s * 0.15:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: L-BFGS-B augmented Lagrangian at higher T
    for T in [500, 1500, 4000, 10000]:
        if elapsed() > budget_s * 0.30:
            break
        hc = lbfgsb_auglag(best_h, T, n, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Targeted worst-correlation perturbation + local opt
    n_pert_iters = int((budget_s * 0.55 - elapsed()) / max(1, (budget_s * 0.25 - 0)))
    n_pert_iters = max(30, min(100, n_pert_iters))
    for it in range(n_pert_iters):
        if elapsed() > budget_s * 0.55:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_k = np.argsort(corr)[-4:]
        step = 0.10 * (1.0 - 0.6 * it / n_pert_iters) + 0.02
        h_pert = best_h.copy()
        for k in top_k:
            shift = k - (n - 1)
            if shift >= 0:
                i1 = np.arange(0, n - shift)
                i2 = np.arange(shift, n)
                h_pert[i2] += step * np.random.rand(len(i2)) * 0.4
                h_pert[i1] -= step * np.random.rand(len(i1)) * 0.4
            else:
                s2 = -shift
                i1 = np.arange(0, n - s2)
                i2 = np.arange(s2, n)
                h_pert[i1] += step * np.random.rand(len(i1)) * 0.4
                h_pert[i2] -= step * np.random.rand(len(i2)) * 0.4
        h_pert = project_feasible(h_pert, n)
        T_use = 400 + 800 * np.random.rand()
        hc = slsqp_opt(h_pert, T_use, n, maxiter=250)
        c5v = compute_c5(hc)
        temp = 0.006 * (1.0 - it / n_pert_iters) + 0.0004
        if c5v < best_c5 or np.random.rand() < np.exp(-(c5v - best_c5) / temp):
            best_c5 = c5v
            best_h = hc

    # Phase 4: Random basin hopping with L-BFGS-B
    n_bh = int((budget_s * 0.75 - elapsed()) / max(1.0, (budget_s * 0.20)))
    n_bh = max(30, min(120, n_bh))
    for it in range(n_bh):
        if elapsed() > budget_s * 0.75:
            break
        step = 0.18 * (1.0 - it / n_bh) + 0.04
        h_pert = best_h + step * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        T_use = 200 + 1200 * np.random.rand()
        hc = lbfgsb_auglag(h_pert, T_use, n, maxiter=200)
        c5v = compute_c5(hc)
        temp = 0.005 * (1.0 - it / n_bh) + 0.0003
        if c5v < best_c5 or np.random.rand() < np.exp(-(c5v - best_c5) / temp):
            best_c5 = c5v
            best_h = hc

    # Phase 5: Diverse structured starts
    if elapsed() < budget_s * 0.88:
        starts = []
        for _ in range(3):
            hb = (np.random.rand(n) < 0.5).astype(float)
            starts.append(project_feasible(hb, n))
        for nb in [4, 6, 8]:
            bn = n // nb
            hb = np.zeros(n)
            for b in range(0, nb, 2):
                hb[b*bn:(b+1)*bn] = 1.0
            starts.append(project_feasible(hb, n))
        hp = np.tile(np.array([1,1,0,1,0,0]), n//6+1)[:n].astype(float)
        starts.append(project_feasible(hp, n))
        for h0 in starts:
            if elapsed() > budget_s * 0.88:
                break
            hc = slsqp_opt(h0, 500, n, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
            hc = slsqp_opt(hc, 3000, n, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 6: Coordinate-level fine tuning
    if elapsed() < budget_s * 0.96:
        for it in range(200):
            if elapsed() > budget_s * 0.96:
                break
            improved = False
            for j in np.random.choice(n, size=min(15, n), replace=False):
                for delta in [0.02, 0.05, 0.08, -0.02, -0.05, -0.08]:
                    h_try = best_h.copy()
                    h_try[j] += delta
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if not improved and it > 60:
                break

    # Phase 7: Final high-T SLSQP polish
    for T in [5000, 20000, 50000]:
        if elapsed() > budget_s * 0.99:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
