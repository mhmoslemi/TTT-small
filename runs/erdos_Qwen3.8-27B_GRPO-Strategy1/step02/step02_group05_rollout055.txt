def compute_c5(h, dx):

        if s > target:

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_grad(h, N, k_top, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    # Top k lags with exponential weighting
    top_idx = np.argpartition(corr, -k_top)[-k_top:]
    cmax = corr[top_idx].max()
    # Weight: exp(-beta * (cmax - corr))
    diffs = cmax - corr[top_idx]
    w = np.exp(-beta * diffs)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    
    grad = np.zeros(N)
    center = N - 1
    for ii in range(k_top):
        ai = int(top_idx[ii])
        wi = w[ii]
        if wi < 1e-8:
            continue
        L = ai - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_gradient(h, N, k_top, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_exact = float(np.max(corr)) * (2.0 / N)
    top_idx = np.argpartition(corr, -k_top)[-k_top:]
    cmax = float(corr[top_idx].max())
    w = np.exp(-beta * (cmax - corr[top_idx]))
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5_exact
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ii in range(k_top):
        ai = int(top_idx[ii])
        wi = w[ii]
        if wi < 1e-10:
            continue
        L = ai - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, c5_exact

    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / N)
    b1, b2, eps = 0.85, 0.999, 1e-8

    vq = np.zeros(N)

        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        it += 1

                      lr0=0.004, k_top=4, beta=30.0, decay=0.99, dinterval=60)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    # Phase 4: mirror descent refinement (8%)
    p4 = t0 + 0.75 * budget_s
    h4, c4 = mirror_opt(h3.copy(), N, target, p4,
                        lr0=0.006, k_top=5, beta=20.0, decay=0.99, dinterval=50)
    if c4 < best_c5:
        best_c5, best_h = c4, h4.copy()

    # Phase 5: coordinate polish (13%)
    p5 = t0 + 0.88 * budget_s
    h5, c5p = coord_polish(best_h.copy(), N, target, 10, p5)
    if c5p < best_c5:
        best_c5, best_h = c5p, h5.copy()

    # Phase 6: perturbed restart (10%)
    if time.time() < deadline - 60:
        h_pert = h0 + rng.randn(n0) * 0.04
        h_pert = project_sum(h_pert, target)
        p6a = deadline - 35
        h6, c6 = adam_opt(h_pert, N, target, p6a,
                          lr0=0.015, k_top=10, beta=8.0, decay=0.985, dinterval=40)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()
        h7, c7 = coord_polish(h6.copy(), N, target, 3, deadline - 5)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    # Final coordinate polish on best
    h_f, c_f = coord_polish(best_h.copy(), N, target, 4, deadline - 1)
    if c_f < best_c5:
        best_c5, best_h = c_f, h_f.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
