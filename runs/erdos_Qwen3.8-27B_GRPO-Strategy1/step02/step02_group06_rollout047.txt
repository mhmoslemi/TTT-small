                      options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return proj_constraint(res.x, n)

def do_block_sweep(h, n, block_size, T, max_iter, num_passes):
    h_curr = h.copy()
    num_blocks = n // block_size
    for p in range(num_passes):
        for b in range(num_blocks):
            start = b * block_size
            end = min(start + block_size, n)
            bs = end - start
            fixed_sum = n * 0.5 - (np.sum(h_curr) - np.sum(h_curr[start:end]))
            cons = [{'type': 'eq', 'fun': block_sum_fn, 'jac': block_sum_jac,
                     'args': (fixed_sum,)}]
            res = opt.minimize(block_obj, h_curr[start:end],
                              args=(h_curr, start, end, n, T),
                              method='SLSQP',
                              bounds=[(0.0, 1.0)] * bs,
                              constraints=cons,
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
            h_curr[start:end] = res.x
    return proj_constraint(h_curr, n)

def de_evolve(h_init, n, t_start, end_time, pop_size, F0, CR, seed):
    rng = np.random.RandomState(seed)
    pop = np.zeros((pop_size, n))
    pop[0] = h_init
    for i in range(1, pop_size):
        sigma = 0.025 * (1.0 + 0.5 * rng.rand())
        pop[i] = proj_constraint(np.clip(h_init + rng.randn(n) * sigma, 0, 1), n)
    fits = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    h_best = pop[np.argmin(fits)].copy()
    c_best = fits[h_best is h_best and np.argmin(fits)]
    c_best = float(np.min(fits))
    gen = 0
    while time.time() < end_time:
        gen += 1
        F_gen = max(0.2, F0 * (0.97 ** gen))
        for i in range(pop_size):
            others = [j for j in range(pop_size) if j != i]
            a, b, c = rng.choice(others, 3, replace=False)
            mutant = pop[a] + F_gen * (pop[b] - pop[c])
            cross_mask = rng.rand(n) < CR
            trial = np.where(cross_mask, mutant, pop[i])
            trial = proj_constraint(trial, n)
            trial_fit = compute_c5(trial)
            if trial_fit <= fits[i]:
                pop[i] = trial
                fits[i] = trial_fit
        best_idx = int(np.argmin(fits))
        if fits[best_idx] < c_best:
            c_best = float(fits[best_idx])
            h_best = pop[best_idx].copy()
    return h_best, c_best

def fine_search(h0, n, t_start, end_time, seed):
    rng = np.random.RandomState(seed)
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    sigma = 0.006
    while time.time() < end_time:
        n_blk = rng.randint(2, 9)
        start = rng.randint(0, max(1, n - n_blk))
        h_try = h_cur.copy()
        h_try[start:start + n_blk] += rng.randn(n_blk) * sigma
        h_try = proj_constraint(h_try, n)
        c_try = compute_c5(h_try)
        if c_try <= c_cur or rng.rand() < np.exp(-(c_try - c_cur) / (max(sigma, 1e-6) * 0.08)):
            h_cur = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        sigma *= 0.9997
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Block SLSQP sweeps - efficient local optimization
    t_phase1_end = t_start + 280
    for bs, T, mi, npass in [(8, 0.004, 60, 4), (12, 0.002, 80, 3), (16, 0.001, 100, 3)]:
        if time.time() > t_phase1_end - 30:
            break
        h_opt = do_block_sweep(h_best, n_best, bs, T, mi, npass)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 2: Differential evolution for global exploration
    t_phase2_end = min(t_start + budget_s * 0.55, time.time() + 250)
    if time.time() < t_phase2_end - 120:
        h_de, c_de = de_evolve(h_best, n_best, t_start, t_phase2_end, 24, 0.6, 0.35, seed + 12345)
        if c_de < c5_best:
            h_best = h_de
            c5_best = c_de

    # Phase 3: Full SLSQP with active-set objective
    if time.time() < t_start + budget_s - 200:
        for T, mi, fr in [(0.008, 800, 0.2), (0.003, 1200, 0.12), (0.001, 1500, 0.08)]:
            if time.time() > t_start + budget_s - 100:
                break
            h_opt = do_full_slsqp(h_best, n_best, T, mi, True, fr)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 4: More block sweeps at finer T
    if time.time() < t_start + budget_s - 150:
        for bs, T, mi, npass in [(8, 0.001, 80, 4), (16, 0.0005, 100, 3), (6, 0.0005, 60, 5)]:
            if time.time() > t_start + budget_s - 80:
                break
            h_opt = do_block_sweep(h_best, n_best, bs, T, mi, npass)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 5: Fine local search (coordinate-like)
    t_phase5_end = time.time() + 80
    if budget_s - (time.time() - t_start) > 120:
        h_fs, c_fs = fine_search(h_best, n_best, t_start, t_phase5_end, seed + 777)
        if c_fs < c5_best:
            h_best = h_fs
            c5_best = c_fs

    # Phase 6: Try adjacent resolutions
    if time.time() < t_start + budget_s - 80:
        for n_try in [n_best + 14, n_best - 14]:
            if n_try < 100 or time.time() > t_start + budget_s - 50:
                continue
            h_up = upsample(h_best, n_try)
            h_opt = do_block_sweep(h_up, n_try, 8, 0.002, 60, 3)
            h_opt2 = do_full_slsqp(h_opt, n_try, 0.002, 600, True, 0.1)
            c = compute_c5(h_opt2)
            if c < c5_best:
                h_best = h_opt2
                c5_best = c
                n_best = n_try

    # Phase 7: Final ultra-fine polish
    if time.time() < t_start + budget_s - 20:
        h_opt = do_full_slsqp(h_best, n_best, 0.0005, 2000, True, 0.06)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c
        h_opt2 = do_block_sweep(h_best, n_best, 6, 0.0003, 100, 4)
        c2 = compute_c5(h_opt2)
        if c2 < c5_best:
            h_best = h_opt2
            c5_best = c2

    h_best = proj_constraint(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
