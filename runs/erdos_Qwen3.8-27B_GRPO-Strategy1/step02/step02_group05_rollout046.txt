        tau = tau0 * (tau1 / tau0) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        true_c5 = float(np.max(corr)) * (2.0 / N)
        w = make_weights(corr, tau, topk)
        grad = compute_grad(h, w)
        if true_c5 < best_c5 - 1e-14:
            best_c5 = true_c5
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % period == 0:
            lr *= decay
            if lr < 5e-10:
                break
    return best_h, best_c5

def local_refine(h, N, target, dx, deadline):
    """Targeted local search: nudge coordinates in gradient direction."""
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        w = make_weights(corr, 500.0, topk=5)
        grad = compute_grad(h, w)
        # Find top movers
        steps = np.abs(grad)
        order = np.argsort(steps)[::-1][:40]
        improved = False
        for i in order:
            for step in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
                save = h[i]
                h[i] = save + step
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h  # revert
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    # Sizes to try
    sizes = []
    for s in [n0, 2 * n0, 572, 760, 920]:
        if s % 2 == 0 and s <= 1100 and s not in sizes:
            sizes.append(s)
    sizes.sort()

    # Build initial candidates
    candidates = []
    for N in sizes:
        if time.time() > deadline - 200:
            break
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        hx = np.interp(xN, x0, h0)
        hx = project_sum(hx, float(N) / 2.0)
        candidates.append((N, hx.copy()))
        # Slightly perturbed version
        hp = hx + rng.randn(N) * 0.05
        hp = project_sum(hp, float(N) / 2.0)
        candidates.append((N, hp))

    for N, h_init in candidates:
        if time.time() > deadline - 150:
            break
        target = float(N) / 2.0
        remain = deadline - time.time()
        if remain < 120:
            break

        # Phase 1: broad exploration
        p1_dead = time.time() + remain * 0.30
        h1, c1 = optimize(h_init.copy(), N, target, p1_dead,
                          tau0=2.0, tau1=40.0, lr0=0.035,
                          decay=0.985, period=40, topk=0)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

        # Phase 2: medium focus
        if time.time() < deadline - 100:
            p2_dead = time.time() + (deadline - time.time()) * 0.35
            h2, c2 = optimize(h1, N, target, p2_dead,
                              tau0=40.0, tau1=250.0, lr0=0.012,
                              decay=0.99, period=50, topk=15)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

        # Phase 3: tight focus on top lags
        if time.time() < deadline - 60:
            h3, c3 = optimize(h2, N, target, deadline - 50,
                              tau0=250.0, tau1=1200.0, lr0=0.004,
                              decay=0.995, period=60, topk=8)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()

    # Final local refinement on best solution
    Nb = len(best_h)
    if time.time() < deadline - 10:
        dx_b = 2.0 / Nb
        h_lr, c_lr = local_refine(best_h.copy(), Nb, Nb / 2.0, dx_b, deadline - 2)
        if c_lr < best_c5:
            best_c5 = c_lr
            best_h = h_lr

    # Final micro-polish
    if time.time() < deadline - 3:
        Nb = len(best_h)
        h_f, c_f = optimize(best_h.copy(), Nb, Nb / 2.0, deadline - 1,
                            tau0=800.0, tau1=3000.0, lr0=0.0015,
                            decay=0.997, period=100, topk=5)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, float(final_c5), len(best_h)
