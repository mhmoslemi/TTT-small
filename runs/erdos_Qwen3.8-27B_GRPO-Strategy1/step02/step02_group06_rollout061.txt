import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        ni = np.sum(interior)
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0.0, 1.0)
    return h

def sm_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(corr))
    shifted = (corr - cm) / T
    ex = np.exp(shifted)
    s_ex = ex.sum()
    w = ex / s_ex
    obj = (cm + T * np.log(s_ex)) * dx
    
    I = np.arange(n)[:, None]
    K = np.arange(2 * n - 1)[None, :]
    idx1 = I + K - n + 1
    idx2 = I - K + n - 1
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(v1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where(v2, h[np.clip(idx2, 0, n - 1)], 0.0)
    wr = w[None, :]
    grad = dx * ((wr * h1).sum(axis=1) - (wr * h2).sum(axis=1))
    return obj, grad

def pg_round(h_in, n, T, n_iters, t_start, budget_s, min_time=30):
    h = h_in.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    obj, grad = sm_obj_grad(h, T)
    lr = max(0.01, T * 8.0)
    
    for it in range(n_iters):
        if time.time() - t_start > budget_s - min_time:
            break
        h_new = proj_constraint(h - lr * grad, n)
        obj_new, grad_new = sm_obj_grad(h_new, T)
        if obj_new > obj + 1e-12:
            lr *= 0.7
            h_new = proj_constraint(h - lr * grad, n)
            obj_new, grad_new = sm_obj_grad(h_new, T)
            if obj_new > obj + 1e-12:
                lr *= 0.5
                continue

                lr *= 1.05
        h = h_new
        obj, grad = obj_new, grad_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def do_slsqp(h0, n, T, max_iter):
    def obj_fn(x):
        o, _ = sm_obj_grad(x, T)
        return o
    def grad_fn(x):
        _, g = sm_obj_grad(x, T)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    res = opt.minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0})
    return proj_constraint(res.x, n)

    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        ni = np.sum(interior)
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0.0, 1.0)
    return h

def sm_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(corr))
    shifted = (corr - cm) / T
    ex = np.exp(shifted)
    s_ex = ex.sum()
    w = ex / s_ex
    obj = (cm + T * np.log(s_ex)) * dx
    I = np.arange(n)[:, None]
    K = np.arange(2 * n - 1)[None, :]
    idx1 = I + K - n + 1
    idx2 = I - K + n - 1
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(v1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where(v2, h[np.clip(idx2, 0, n - 1)], 0.0)
    wr = w[None, :]
    grad = dx * ((wr * h1).sum(axis=1) - (wr * h2).sum(axis=1))
    return obj, grad

def pg_round(h_in, n, T, n_iters, t_start, budget_s, min_time=25, rng=None):
    h = h_in.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    obj, grad = sm_obj_grad(h, T)
    lr = max(0.01, T * 6.0)
    stall = 0
    for it in range(n_iters):
        if time.time() - t_start > budget_s - min_time:
            break
        h_new = proj_constraint(h - lr * grad, n)
        obj_new, grad_new = sm_obj_grad(h_new, T)
        if obj_new > obj + 1e-13:
            lr *= 0.6
            if lr < 0.001:
                h_new = proj_constraint(h - lr * grad, n)
                obj_new, grad_new = sm_obj_grad(h_new, T)
                if obj_new > obj + 1e-13:
                    lr = max(lr, 0.005)
                    if rng is not None:
                        h_new = proj_constraint(h + rng.randn(n) * 0.005, n)
                        obj_new, grad_new = sm_obj_grad(h_new, T)
                    else:
                        continue
        else:
            if lr < 2.0:
                lr *= 1.03
        h = h_new
        obj, grad = obj_new, grad_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 800 and rng is not None:
                h = proj_constraint(h + rng.randn(n) * 0.008, n)
                obj, grad = sm_obj_grad(h, T)
                stall = 0
    return best_h, best_c

def sum_eq(x, n):
    return np.sum(x) - n * 0.5

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_obj(x, T):
    o, _ = sm_obj_grad(x, T)
    return o

def slsqp_grad(x, T):
    _, g = sm_obj_grad(x, T)
    return g

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(slsqp_obj, h0, jac=slsqp_grad, args=(T,),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0})
    return proj_constraint(res.x, n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)
