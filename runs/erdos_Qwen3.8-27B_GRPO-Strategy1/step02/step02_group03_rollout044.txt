import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _obj_t(x):
    return x[_sp['n']]

def _eq_sum(x):
    return x[:_sp['n']].sum() - _sp['ts']

def _ineq_corr(x):
    n = _sp['n']
    h = x[:n]
    t = x[n]
    dx = _sp['dx']
    return t - np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(120):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def opt_slsqp(h0, maxiter=250):
    n = len(h0)
    ts, dx = n / 2.0, 2.0 / n
    _sp['n'], _sp['ts'], _sp['dx'] = n, ts, dx
    best_h, best_c = h0.copy(), compute_c5(h0)
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0 * float(n))]
    cons = [{'type': 'eq', 'fun': _eq_sum}, {'type': 'ineq', 'fun': _ineq_corr}]
    for attempt in range(4):
        try:
            t_init = best_c + 0.005 * (attempt + 1)
            x0 = np.concatenate([best_h.copy(), [t_init]])
            r = minimize(_obj_t, x0, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            if r.x is not None:
                h_new = proj(r.x[:n], ts)
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_h, best_c = h_new, c_new
        except Exception:
            pass
    return best_h, best_c

def coord_descend(h0, n_rounds=40, seed=0):
    n = len(h0)
    ts = n / 2.0
    rng = np.random.RandomState(seed)
    h = h0.copy()
    bc = compute_c5(h)
    steps = np.array([0.08, -0.08, 0.04, -0.04, 0.02, -0.02, 0.015, -0.015, 0.01, -0.01])
    for r in range(n_rounds):
        improved = False
        for j in rng.permutation(n):
            sv = h[j]
            found = False
            for s in steps:
                h[j] = sv + s
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                    continue
                h = proj(h, ts)
                c = compute_c5(h)
                if c < bc - 1e-15:
                    bc, improved, found = c, True, True
                    break
            if not found:
                h[j] = sv
                h = proj(h, ts)
        if not improved:
            break
    return h, bc

def swap_search(h0, n_trials=2000, seed=0):
    n = len(h0)
    ts = n / 2.0
    rng = np.random.RandomState(seed)
    h = h0.copy()
    bc = compute_c5(h)
    for _ in range(n_trials):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c < bc - 1e-15:
            bc = c
        else:
            h[i], h[j] = h[j], h[i]
    return h, bc

def softmobj(x, n, ts, beta):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = corr.max()
    return m + np.log(np.sum(np.exp(corr - m)))
