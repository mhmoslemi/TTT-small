import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    interior = (h > 1e-12) & (h < 1.0 - 1e-12)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
        # If clipping changed sum, do second pass
        s2 = np.sum(h)
        if abs(s2 - target) > 1e-12:
            diff2 = target - s2
            interior2 = (h > 1e-12) & (h < 1.0 - 1e-12)
            if np.sum(interior2) > 0:
                h[interior2] += diff2 / max(1, np.sum(interior2))
                h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            remaining = diff
            for i in idx:
                space = 1.0 - h[i]
                add = min(remaining, space)
                h[i] += add
                remaining -= add
                if remaining <= 1e-12: break
        elif diff < 0:
            idx = np.argsort(-h)
            remaining = -diff
            for i in idx:
                rem = min(remaining, h[i])
                h[i] -= rem
                remaining -= rem
                if remaining <= 1e-12: break
    return h

def subgradient_step(h, n, K, lr):
    """One subgradient step with top-K active shifts."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Top K shifts
    top_idx = np.argsort(corr)[-K:]
    # Compute subgradient
    grad = np.zeros(n)
    for k in top_idx:
        shift = k - (n - 1)
        p = np.arange(n)
        # (1 - h[p + shift]) where valid
        pp = p + shift
        v1 = (pp >= 0) & (pp < n)
        g1 = np.zeros(n)
        g1[v1] = 1.0 - h[pp[v1]]
        # -h[p - shift] where valid
        pm = p - shift
        v2 = (pm >= 0) & (pm < n)
        g2 = np.zeros(n)
        g2[v2] = -h[pm[v2]]
        grad += (g1 + g2)
    grad /= K
    # Nesterov momentum could be added externally
    # Projected step
    h_new = h - lr * grad
    h_new = fix_sum(h_new, n)
    return h_new, np.max(corr)

    h_curr = h.copy()
    best = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    velocity = np.zeros(n)
    mu = 0.9  # momentum
    
    for it in range(total_iters):
        # Adaptive K
        frac = it / max(1, total_iters - 1)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = 0.5 * n
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    cnt = int(np.sum(mask))
    if cnt > 0 and abs(diff) < 0.3 * cnt:
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    else:
        if diff > 0:
            idx = np.argsort(h)
            r = diff
            for i in idx:
                a = min(r, 1.0 - h[i])
                h[i] += a
                r -= a
                if r <= 1e-12:
                    break
        else:
            idx = np.argsort(-h)
            r = -diff
            for i in idx:
                a = min(r, h[i])
                h[i] -= a
                r -= a
                if r <= 1e-12:
                    break
    return h
