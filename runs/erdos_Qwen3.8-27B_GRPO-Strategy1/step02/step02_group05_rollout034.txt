        if 0.0 < best_x < 1.0:

                fx = float(np.max(out + (x - old) * b))
                if fx < best_f:
                    best_f, best_x = fx, x
            lo, hi2 = 0.0, 1.0
            for _ in range(12):
                m1 = lo + (hi2 - lo) / 3.0
                m2 = hi2 - (hi2 - lo) / 3.0
                f_m1 = float(np.max(out + (m1 - old) * b))
                f_m2 = float(np.max(out + (m2 - old) * b))
                if f_m1 < f_m2:
                    hi2 = m2
                else:
                    lo = m1
            best_x = (lo + hi2) / 2.0
            best_f = float(np.max(out + (best_x - old) * b))
        if best_x != old:
            out += (best_x - old) * b
            h[i] = best_x
    return h, out

def soft_grad(h, N, tau):
    center = N - 1
    out = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(out)) * (2.0 / N)
    sc = tau * out
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5
    w /= ws
    grad = np.zeros(N)
    thresh = 0.005
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def adam_phase(h, N, target, deadline, tau0, tau1, lr0, decay):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    lr = lr0
    total = max(int((deadline - time.time()) / 0.005), 10)
    while time.time() < deadline:
        frac = min(it / max(total * 0.5, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        grad, c5v = soft_grad(h, N, tau)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 30 == 0:
            lr *= decay
        if lr < 1e-10:
            break
    return best_h, best_c5

def binary_round(h, target):
    N = len(h)
    h_b = (h > 0.5).astype(float)
    ones = int(h_b.sum())
    need = int(target)
    if ones > need:
        idx = np.where(h_b == 1.0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(idx)
        for j in idx[:ones - need]:
            h_b[j] = 0.0
    elif ones < need:
        idx = np.where(h_b == 0.0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(idx)
        for j in idx[:need - ones]:
            h_b[j] = 1.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    target = N / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent on parent
    h = h0.copy()
    cd1_end = t0 + (budget_s - 5) * 0.35
    while time.time() < cd1_end:
        h, out = coord_sweep(h.copy(), target, cd1_end)
        h = project_sum(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Soft-gradient Adam refinement
    h_ref = best_h.copy()
    p2_end = time.time() + (deadline - time.time()) * 0.35
    h_ref, c2 = adam_phase(h_ref, N, target, p2_end,
                           tau0=50.0, tau1=400.0, lr0=0.005, decay=0.985)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h_ref.copy()

    # Phase 3: More coordinate descent on best
    h = best_h.copy()
    cd3_end = time.time() + (deadline - time.time()) * 0.4
    while time.time() < cd3_end:
        h, out = coord_sweep(h.copy(), target, cd3_end)
        h = project_sum(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Try binary rounding + coordinate descent
    if time.time() < deadline - 30:
        h_b = binary_round(best_h, target)
        c5_b = compute_c5(h_b)
        h_batch = h_b.copy()
        b4_end = time.time() + min(deadline - time.time() - 5, 60)
        while time.time() < b4_end:
            h_batch, out = coord_sweep(h_batch.copy(), target, b4_end)
            h_batch = project_sum(h_batch, target)
            c5 = compute_c5(h_batch)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_batch.copy()

    # Phase 5: Final Adam polish with high tau
    if time.time() < deadline - 5:
        h_f = best_h.copy()
        h_f, c_f = adam_phase(h_f, N, target, deadline - 1,
                              tau0=300.0, tau1=1200.0, lr0=0
