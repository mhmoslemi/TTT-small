def topk_grad(h, dx, k, beta=0):

    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    if beta > 0:
        w = np.exp(beta * (cr - cr.max()))
        s = w.sum()
        if s < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= s
    else:
        # top-k weighting
        idx = np.argsort(cr)[-k:]
        w = np.zeros(2*n-1)
        w[idx] = 1.0 / k
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        kl1 = max(-(n-1), 1-j)
        kh1 = min(n-1, j)
        if kl1 <= kh1:
            k_arr = np.arange(kl1, kh1+1)
            g[j] += np.dot(w[o+k_arr], 1.0 - h[j-k_arr])
        kl2 = max(-(n-1), -j)
        kh2 = min(n-1, n-1-j)
        if kl2 <= kh2:
            k_arr = np.arange(kl2, kh2+1)
            g[j] -= np.dot(w[o+k_arr], h[j+k_arr])
    return g * dx
