    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)

        h_new = proj_sum(res.x, n)
        if compute_c5(h_new) < compute_c5(h):

        else:

def proj_sum(h, n):
    target = n * 0.5
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = (-1.0, 0.0) if s > target else (0.0, 1.0)
    for _ in range(80):

        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
