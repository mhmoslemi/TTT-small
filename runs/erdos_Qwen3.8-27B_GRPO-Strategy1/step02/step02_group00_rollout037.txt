def smooth_grad_inner(corr_raw, h, n):
    """Return matrix M[j,k] = d(corr_k)/d(h_j) using weights from raw corr."""
    # We just return the unweighted gradient structure
    # Actually for the power method we need: grad = sum_k p*corr_k^(p-1) * d(corr_k)/d(h)
    # Let's compute this directly
    m = len(corr_raw)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        ki = k + n - 1
        wi = corr_raw[ki] ** (p - 1) if p > 1 else 1.0
        # We don't have p here... this approach won't work cleanly
    return grad

# Simpler: just use SLSQP with sum-of-powers for a phase
def power_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.mean(corr**p)

def power_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # d(mean(corr^p))/d(h_j) = p * mean(corr^(p-1) * d(corr)/d(h_j))
    # We need d(corr_k)/d(h_j) which is what smooth_grad computes (with T=1, w=1 for all k)
    # smooth_grad with T such that w is uniform: use T=0 limit → w[k]=1/(2n-1)
    # Actually just compute: grad_j = p * (1/(2n-1)) * sum_k corr_k^(p-1) * d(corr_k)/d(h_j)
    # = (p * dx/(2n-1)) * sum_k (corr_raw[k]*dx)^(p-1) * d(corr_raw[k])/d(h_j)
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    weights = (corr_raw * dx) ** (p - 1) * dx
    # Compute weighted sum of d(corr_k)/d(h_j)
    total_grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(-(n-1), n):
        wi = weights[k + n - 1]
        if abs(wi) < 1e-18:
            continue
        if k == 0:
            total_grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            total_grad[:n-k] += wi * omh[k:]
            total_grad[k:] -= wi * h[:n-k]
        else:
            jk = -k
            total_grad[jk:] += wi * omh[:n-jk]
            total_grad[:n-jk] -= wi * h[jk:]
    return p * total_grad / (2 * n - 1)

def peak_shave(h, n, top_k=5, steps=80):
    best = h.copy()
    best_c5 = compute_c5(best)
    for s in range(steps):
        corr = np.correlate(best, 1.0 - best, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for ki in top_idx:
            k_shift = ki - (n - 1)
            if k_shift >= 0:
                indices = np.arange(0, n - k_shift + 1)
            else:
                indices = np.arange(-k_shift, n)
            for idx in indices:
                for delta in [0.03, -0.03, 0.07, -0.07]:
                    h_try = best.copy()
                    h_try[idx] += delta
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(120)
        best_h[:60] = 1.0

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: SLSQP with analytical gradient, increasing T
    for T in [5, 15, 50, 150, 500, 1500, 5000]:
        if time.time() - t0 > budget_s * 0.28:
            break
        hc = slsqp_grad(best_h, T, n, maxiter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Peak shaving
    if time.time() - t0 < budget_s * 0.42:
        best_h, best_c5 = peak_shave(best_h, n, top_k=5, steps=100)

    # Phase 3: SLSQP polish after shaving
    if time.time() - t0 < budget_s * 0.55:
        for T in [3000, 10000, 30000]:
            if time.time() - t0 > budget_s * 0.55:
                break
            hc = slsqp_grad(best_h, T, n, maxiter=3000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 4: Coarse (n=60) optimization then upscale
    if time.time() - t0 < budget_s * 0.72:
        nc = 60
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, nc, endpoint=False)
        h_c = np.interp(x_new, x_old, best_h)
        h_c = project_feasible(h_c, nc)
        for T in [30, 300, 3000]:
            if time.time() - t0 > budget_s * 0.68:
                break
            hc = slsqp_grad(h_c, T, nc, maxiter=1500)
            h_up = np.interp(x_old, x_new, hc)
            h_up = project_feasible(h_up, n)
            c5u = compute_c5(h_up)
            if c5u < best_c5:
                best_c5 = c5u
                best_h = h_up
        if time.time() - t0 < budget_s * 0.72:
            hc = slsqp_grad(best_h, 8000, n, maxiter=2000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 5: Sum-of-powers optimization (p=20, 50)
    if time.time() - t0 < budget_s * 0.85:
        for p in [20, 50]:
            if time.time() - t0 > budget_s * 0.82:
                break
            bnds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
            try:
                res = minimize(power_obj, best_h, args=(p,),
                              jac=power_grad, method='SLSQP',
                              bounds=bnds, constraints=cons,
                              options={'maxiter': 1000, 'ftol': 1e-16})
                hc = project_feasible(np.clip(res.x, 0, 1), n)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 6: Multi-start structured
    if time.time() - t0 < budget_s * 0.93:
        ha = np.zeros(n)
        ha[::2] = 1.0
        starts = [project_feasible(ha, n)]
        for i in range(3):
            hr = (np.random.rand(n) < 0.5).astype(float)
            starts.append(project_feasible(hr, n))
        x = np.linspace(0, 2, n, endpoint=False)
        hs = (np.sin(3 * np.pi * x) + 1) / 2.0
        starts.append(project_feasible(hs, n))
        for h0 in starts:
            if time.time() - t0 > budget_s * 0.93:
                break
            hc = slsqp_grad(h0, 200, n, maxiter=1500)
            hc = slsqp_grad(hc, 5000, n, maxiter=1500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 7: Final extreme-T polish + peak shave
    for T in [50000, 150000]:
        if time.time() - t0 > budget_s * 0.985:
            break
        try:
            hc = slsqp_grad(best_h, T, n, maxiter=4000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    if time.time() - t0 < budget_s * 0.995:
        best_h, best_c5 = peak_shave(best_h, n, top_k=3, steps=40)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
