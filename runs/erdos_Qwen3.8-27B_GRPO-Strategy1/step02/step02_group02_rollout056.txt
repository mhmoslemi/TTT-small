import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.sum(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, t, topk, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    if topk and topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        ct = corr[idx]
    else:
        ct = corr
    mx = np.max(ct)
    s = t * (ct - mx)
    sm = (mx + np.log(np.sum(np.exp(s))) / t) * dx
    return sm + lam * (np.sum(h) - n / 2.0) ** 2

def grad_obj(h, t, topk, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    mask = np.ones(nc, dtype=bool)
    if topk and topk < nc:
        idx = np.argpartition(corr, -topk)[-topk:]
        mask[:] = False
        mask[idx] = True
    mx = np.max(corr[mask])
    w = np.zeros(nc)
    w[mask] = np.exp(t * (corr[mask] - mx))
    w /= np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(nc):
        if w[k] < 1e-15:
            continue
        s = n - 1 - k
        js1 = max(0, -s)
        je1 = min(n - 1, n - 1 - s)
        if js1 <= je1:
            grad[js1:je1 + 1] += w[k] * omh[js1 + s:je1 + s + 1]
        js2 = max(0, s)
        je2 = min(n - 1, n - 1 + s)
        if js2 <= je2:
            grad[js2:je2 + 1] -= w[k] * h[js2 - s:je2 - s + 1]
    grad *= dx
    grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    return grad

def lbfgs_opt(h, n, t, topk, lam, maxiter):
    bnds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_obj(x, t, topk, lam)
    def g(x):
        return grad_obj(x, t, topk, lam)
    r = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bnds,
                 options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-7})
    return proj_sum(np.clip(r.x, 0.0, 1.0), n)

def interp_up(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj_sum(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    if n % nd == 0:
        return proj_sum(h.reshape(nd, n // nd).mean(axis=1), nd)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj_sum(np.interp(xd, xs, h), nd)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        n0 = len(initial_h_values)
        bh = proj_sum(np.array(initial_h_values, dtype=float).copy(), n0)
    except Exception:
        n0 = 280
        bh = proj_sum(np.full(n0, 0.5) + rng.randn(n0) * 0.05, n0)
    bc = compute_c5(bh)

    # Phase 1: Coarse at n=70
    if tleft() > 350:
        h70 = interp_down(bh, 70)
        for tv in [30, 100, 400, 1500]:
            if tleft() < 25:
                break
            for lm in [100, 1000]:
                if tleft() < 15:
                    break
                try:
                    hc = lbfgs_opt(h70, 70, tv, None, lm, 400)
                    c5 = compute_c5(hc)
                    if c5 < bc:
                        bc = c5
                        bh = interp_up(hc, n0)
                    h70 = hc
                except Exception:
                    pass

    # Phase 2: Medium at n=140
    if tleft() > 280:
        h140 = interp_up(bh, 140)
        for tv in [60, 200, 700, 2500]:
            if tleft() < 20:
                break
            for lm in [500, 5000]:
                if tleft() < 12:
                    break
                try:
                    hc = lbfgs_opt(h140, 140, tv, None, lm, 600)
                    c5 = compute_c5(hc)
                    if c5 < bc:
                        bc = c5
                        bh = interp_up(hc, n0)
                    h140 = hc
                except Exception:
                    pass

    # Phase 3: Full resolution, progressive topk
    n = n0
    h = bh.copy()
    nc = 2 * n - 1
    schedule = [
        (max(5, nc // 6), 40, 1000, 1500),
        (max(4, nc // 10), 100, 3000, 2000),
        (max(3, nc // 15), 300, 8000, 2500),
        (max(2, nc // 25), 800, 15000, 3000),
        (2, 2000, 30000, 4000),

            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                bh = hc.copy()
            h = hc
        except Exception:
            pass

    # Phase 4: Binary restarts
    if tleft() > 70:
        hb = proj_sum((h > 0.5).astype(float), n)
        for tv in [200, 1000, 4000]:
            if tleft() < 20:
                break
            try:
                hc = lbfgs_opt(hb, n, tv, max(2, nc // 12), 8000, 2500)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
                    h = hc.copy()
                hb = proj_sum((hc > 0.5).astype(float), n)
            except Exception:
                pass

    # Phase 5: Perturbation restarts
    for _ in range(8):
        if tleft() < 18:
            break
        hp = proj_sum(h + rng.randn(n) * 0.008, n)
        try:

    if tleft() > 15:
        h = bh.copy()
        for tv in [50000, 150000, 500000]:
            if tleft() < 10:

                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    bh = hc.copy()
                h = hc
            except Exception:
                break

    return bh, compute_c5(bh), n
