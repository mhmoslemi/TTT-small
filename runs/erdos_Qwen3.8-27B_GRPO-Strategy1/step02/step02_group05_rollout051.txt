def gradient_local_search(h, N, target, dx, n_iters, deadline):
    """Local search guided by gradient."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Compute gradient
        center = N - 1
        # Focus on top 5 lags
        top5 = np.argpartition(corr, -5)[-5:]
