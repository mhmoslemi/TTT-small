    for T in [8000, 25000]:
        if elapsed() > budget_s * 0.98:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=1500, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
