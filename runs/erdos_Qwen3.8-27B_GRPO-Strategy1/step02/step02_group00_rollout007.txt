            best_c5, best_h = c5v, hc

    # Phase 2: Manual DE for global exploration
    _de_n_global = n
    de_end = budget_s * 0.45
    n_iter_de = 150
    if elapsed() < de_end:
        try:
            de_h, de_c5 = manual_de(n, 25, n_iter_de, seed, best_h,
                                    lambda: elapsed() > de_end)
            if de_c5 < best_c5:
                best_c5, best_h = de_c5, de_h
        except Exception:
            pass

    # Phase 3: SLSQP multi-T refinement
    for T in [200, 800, 3000, 10000]:
        if elapsed() > budget_s * 0.62:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=700)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 4: Structured basin hopping with varied perturbations
    n_bh = 50
    bh_end = budget_s * 0.80
    T_list = [80, 200, 600, 1500, 5000]
    for it in range(n_bh):
        if elapsed() > bh_end:
            break
        frac = it / max(n_bh, 1)
        step = 0.10 * (1.0 - 0.8 * frac) + 0.015
        h_pert = best_h + step * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        T_try = T_list[it % len(T_list)]
        hc = slsqp_opt(h_pert, T_try, n, maxiter=350)
        c5v = compute_c5(hc)
        temp = 0.006 * (1.0 - frac) + 0.0002
        if c5v < best_c5 or np.random.rand() < np.exp(-(c5v - best_c5) / temp):
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
            else:
                best_h = hc

    # Phase 5: Second DE round focusing near best
    if elapsed() < budget_s * 0.90:
        _de_n_global = n
        seed2 = seed + 1000
        rng2 = np.random.RandomState(seed2)
        # Create population centered near best with larger perturbations
        try:
            def de_near_fn():
                return elapsed() > budget_s * 0.90
            de_h2, de_c5_2 = manual_de(n, 20, 80, seed2, best_h, de_near_fn)
            if de_c5_2 < best_c5:
                best_c5, best_h = de_c5_2, de_h2
        except Exception:
            pass

    # Phase 6: Resample to different n values
    for n_try in [80, 100, 140, 160]:
        if elapsed() > budget_s * 0.95:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        for trial in range(2):
            if elapsed() > budget_s * 0.95:
                break
            if trial == 0:
                h0 = h_interp.copy()
            else:
                h0 = h_interp + np.random.uniform(-0.04, 0.04, n_try)
            h0 = project_feasible(h0, n_try)
            for T in [300, 1500]:
                if elapsed() > budget_s * 0.95:
                    break
                hc = slsqp_opt(h0, T, n_try, maxiter=400)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
                    n = n_try

    # Phase 7: Coordinate descent fine-tuning
    n = len(best_h)
    if elapsed() < budget_s * 0.985:
        for _ in range(80):
            if elapsed() > budget_s * 0.985:
                break
            improved = False
            idx = np.random.choice(n, size=min(25, n), replace=False)
            for j in idx:
                for delta in [0.01, 0.03, 0.05, -0.01, -0.03, -0.05]:
                    h_try = best_h.copy()
                    h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5, best_h = c5_try, h_try
                        improved = True
                        break
                if improved:
                    break

    # Final SLSQP polish
    if elapsed() < budget_s * 0.995:
        hc = slsqp_opt(best_h, 5000, len(best_h), maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
