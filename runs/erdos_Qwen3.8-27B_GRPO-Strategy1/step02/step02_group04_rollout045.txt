def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Adjust interior points
    interior = (h > 1e-6) & (h < 1.0 - 1e-6)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-12: break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12: break
    return h

def soft_max_obj(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    val = (c_max + T * np.log(max(s, 1e-300))) * dx
    s_sum = np.sum(h) - n * 0.5
    val += penalty * s_sum * s_sum
    return val

def soft_max_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    exps = np.exp(shifted / T)
    w = exps / np.sum(exps)
    
    grad = np.zeros(2 * n - 1)  # will reshape
    # Actually grad is w.r.t. h, so it's length n
    grad_h = np.zeros(n)
    
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:

        tau = k - (n - 1)
        i_lo = max(0, -tau)

        js = np.arange(i_lo, i_hi + 1)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idx = np.argsort(-h)
        for i in idx:
            rem = min(diff, h[i])
            h[i] -= rem
            diff -= rem
            if diff <= 1e-12:
                break
    return h

def _sm_obj(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    sh = np.clip(corr - cm, -80.0 * T, 0.0)
    s = np.sum(np.exp(sh / T))
    val = (cm + T * np.log(max(s, 1e-300))) * dx
    ss = np.sum(h) - n * 0.5
    return val + pen * ss * ss

def _sm_grad(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    sh = np.clip(corr - cm, -80.0 * T, 0.0)
    exps = np.exp(sh / T)
    ws = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = ws[k]
        if wk < 1e-13:
            continue
        tau = k - (n - 1)
        i_lo = max(0, -tau)
        i_hi = min(n - 1, n - 1 - tau)
        js = np.arange(i_lo, i_hi + 1)
        g = 1.0 - h[js + tau]
        vm = (js - tau >= 0) & (js - tau < n)
        if np.any(vm):
            g = g.copy()
            g[vm] -= h[js[vm] - tau]
        grad[i_lo:i_hi + 1] += wk * g
    grad *= dx
    ss = np.sum(h) - n * 0.5
    grad += 2.0 * pen * ss
    return grad

                                    'maxfun': max_iter * 4})
        if res.x is not None:

    except Exception:
        pass
    return h
