import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    if abs(np.sum(h) - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        return c_max * dx
    sh = (corr - c_max) / T
    sh -= np.max(sh)
    return (c_max + T * (np.max(sh) + np.log(np.sum(np.exp(sh))))) * dx

def lbfgsb_pen_obj(h, T, pen):
    return soft_max_val(h, T) + pen * (np.sum(h) - len(h) * 0.5) ** 2

def slsqp_obj(h, T):
    return soft_max_val(h, T)

def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5

def sum_eq_jac(h):
    return np.ones(len(h))

def do_lbfgsb(h0, T, pen, max_iter):
    n = len(h0)
    res = opt.minimize(lbfgsb_pen_obj, h0, args=(T, pen), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-9})
    return proj_constraint(res.x, n)

def do_slsqp(h0, T, max_iter):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = opt.minimize(slsqp_obj, h0, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return proj_constraint(res.x, n)

def fourier_perturb(h, rng, max_freq, amp):
    n = len(h)
    h_new = h.copy()
    k = rng.randint(1, max_freq + 1)
    phase = rng.uniform(0, 2 * np.pi)
    a = rng.uniform(-amp, amp)
    h_new += a * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    return proj_constraint(h_new, n)

def block_perturb(h, rng, n_blk, sigma):
    n = len(h)
    h_new = h.copy()
    idxs = rng.choice(n, size=min(n_blk, n), replace=False)
    h_new[idxs] += rng.randn(len(idxs)) * sigma
    return proj_constraint(h_new, n)

def targeted_reduce(h, rng):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    h_new = h.copy()
    n_reduce = max(3, n // 50)
    w = np.zeros(n)
    for j in range(n):
        j2 = j + lag
        if 0 <= j2 < n:
            w[j] = (1 - h[j]) * h[j2]
    top = np.argsort(-w)[:n_reduce]
    h_new[top] -= 0.01 * rng.rand(n_reduce)
    h_new = proj_constraint(h_new, n)
    return h_new

def upsample(h, new_n):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return proj_constraint(np.interp(x_new, x_old, h), new_n)

    # Phase 2: SLSQP intermediate

    # Phase 3: Structured perturb + L-BFGS-B cycle
