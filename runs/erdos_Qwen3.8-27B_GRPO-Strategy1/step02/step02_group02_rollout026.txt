import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(corr)*dx)

def proj(h, n):
    h = np.clip(h, 0, 1)
    tgt = n/2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0: break
        h[m] += d/cnt
        np.clip(h, 0, 1, out=h)
    return h

def make_binary(h, n):
    """Project to binary with exactly n/2 ones"""
    h_thr = (h > 0.5).astype(float)
    cnt = np.sum(h_thr)
    if cnt == n/2:
        return h_thr
    # Need to adjust
    h_sorted = np.sort(h)
    if cnt > n/2:
        # Too many ones, flip some 1->0 (those closest to 0.5)
        one_indices = np.where(h_thr==1)[0]
        h_one = h[one_indices]
        # Sort by h value ascending (closest to 0.5 first)
        order = np.argsort(h_one)
        flip = int(cnt - n/2)
        h_thr[one_indices[order[:flip]]] = 0
    else:
        # Too few ones, flip some 0->1 (those closest to 0.5)
        zero_indices = np.where(h_thr==0)[0]
        h_zero = h[zero_indices]
        order = np.argsort(-h_zero)
        flip = int(n/2 - cnt)
        h_thr[zero_indices[order[:flip]]] = 1
    return h_thr

def obj_smooth(h, t):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    mx = np.max(corr)
    s = t*(corr - mx)
    s = np.clip(s, -500, 500)
    return (mx + np.log(np.sum(np.exp(s)))/t) * dx

def compute_c5(h):

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-13:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def make_binary(h, n):
    hb = (h > 0.5).astype(float)
    cnt = int(np.sum(hb))
    target = n // 2
    if cnt > target:
        oi = np.where(hb == 1)[0]
        order = np.argsort(h[oi])
        hb[oi[order[:cnt - target]]] = 0.0
    elif cnt < target:
        zi = np.where(hb == 0)[0]
        order = np.argsort(-h[zi])
        hb[zi[order[:target - cnt]]] = 1.0
    return hb

    # Phase 5: Another round of binary SA with different kick
    if tleft() > 80:
        h_bin3 = make_binary(best_h, n)
        corr3 = np.correlate(h_bin3, 1.0 - h_bin3, mode='full').astype(np.float64)
        hc3, bm3 = binary_sa(h_bin3.copy(), corr3.copy(), rng, budget_s * 0.12)
        if bm3 * dx < best_c5:
            best_c5 = bm3 * dx
            best_h = hc3.copy()

    # Phase 6: Fine multi-resolution check at n=140
    if tleft() > 60:
        h140 = interp_down(best_h, 140)
        h140_bin = make_binary(h140, 140)
        corr140 = np.correlate(h140_bin, 1.0 - h140_bin, mode='full').astype(np.float64)
        h140b, bm140 = binary_sa(h140_bin.copy(), corr140.copy(), rng, 30)
        h140_up = interp_up(h140b, 280)
        if compute_c5(h140_up) < best_c5:
            best_h = h140_up
            best_c5 = compute_c5(best_h)
            try:
                h_new = slsqp_opt(h140_up, 280, 500, None, 2000, bnds, cons)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
            except Exception:
                pass

    # Phase 7: Final polish
    if tleft() > 15:
        h_fin = proj(best_h + rng.randn(n) * 0.01, n)
        for t_val in [5000, 20000]:
            if tleft() < 10:
                break
            try:
                h_new = slsqp_opt(h_fin, n, t_val, None, 5000, bnds, cons)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
                h_fin = h_new
            except Exception:
                pass

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, n
