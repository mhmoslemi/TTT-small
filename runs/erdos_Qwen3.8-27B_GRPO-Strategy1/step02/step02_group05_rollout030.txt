        tau = tau0 * (tau1 / tau0) ** frac
        if use_topk or tau > 500:
            grad, c5_val = topk_subgrad(h, N, topk)
        else:
            grad, c5_val = smax_grad(h, N, tau)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % decay_every == 0:
            lr *= decay
        if lr < 1e-10:
            break
    c5_f = compute_c5(h, dx)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h.copy()
    return best_h, best_c5

def binary_push(h, target):
    N = len(h)
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(N)
    n_ones = int(np.floor(target))
    h_b[idx[:n_ones]] = 1.0
    frac = target - n_ones
    if frac > 0.01:
        h_b[idx[n_ones]] = frac
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    sizes = sorted(set([s for s in [n0, 512, 768] if s % 2 == 0 and s <= 1000]))
    for N in sizes:
        if time.time() > deadline - 120:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)
        remain = deadline - time.time()
        if remain < 100:
            break
        # Phase 1: tau 30->200, moderate lr
        p1d = time.time() + remain * 0.30
        h1, c1 = adam_opt(h_init.copy(), N, target, dx, p1d,
                          tau0=30.0, tau1=200.0, lr0=0.025, decay=0.985)
        if c1 < best_c5:
            best_c5 = c1; best_h = h1.copy()
        if time.time() > deadline - 80:
            continue
        # Phase 2: tau 200->1000, smaller lr
        p2d = time.time() + (deadline - time.time()) * 0.35
        h2, c2 = adam_opt(h1.copy(), N, target, dx, p2d,
                          tau0=200.0, tau1=1000.0, lr0=0.01, decay=0.99)
        if c2 < best_c5:
            best_c5 = c2; best_h = h2.copy()
        if time.time() > deadline - 50:
            continue
        # Phase 3: top-k subgrad, very small lr
        p3d = time.time() + (deadline - time.time()) * 0.35
        h3, c3 = adam_opt(h2.copy(), N, target, dx, p3d,
                          tau0=1000.0, tau1=5000.0, lr0=0.004, decay=0.994,
                          use_topk=True, topk=10)
        if c3 < best_c5:
            best_c5 = c3; best_h = h3.copy()
        # Binary push + reoptimize
        if time.time() < deadline - 40:
            h_bp = 0.6 * h3 + 0.4 * binary_push(h3, target)
            h_bp = project_sum(h_bp, target)
            h4, c4 = adam_opt(h_bp, N, target, dx, deadline - 15,
                              tau0=50.0, tau1=500.0, lr0=0.015, decay=0.99)
            if c4 < best_c5:
                best_c5 = c4; best_h = h4.copy()
        # Random perturbation restart
        if time.time() < deadline - 30:
            h_r = best_h + rng.randn(len(best_h)) * 0.05
            h_r = project_sum(h_r, target)
            h5, c5 = adam_opt(h_r, N, target, dx, deadline - 8,
                              tau0=50.0, tau1=400.0, lr0=0.012, decay=0.99)
            if c5 < best_c5:
                best_c5 = c5; best_h = h5.copy()
    # Final polish at best N
    if time.time() < deadline - 4:
        Nb = len(best_h)
        h_f, c_f = adam_opt(best_h.copy(), Nb, Nb / 2.0, 2.0 / Nb,
                            deadline - 1, tau0=1000.0, tau1=5000.0,
                            lr0=0.002, decay=0.996, use_topk=True, topk=12)
        if c_f < best_c5:
            best_c5 = c_f; best_h = h_f
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
