        if bp is None:
            break
        h[bp[0]], h[bp[1]] = 0.0, 1.0
        c5 = bt
    return h, c5

def bin_sa(h, n, t0, budget_s, rng):
    h = (h > 0.5).astype(float)
    h = proj(h, n)
    c5 = compute_c5(h)
    T = 0.0005
    while time_left(t0, budget_s) > 60:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc = compute_c5(h)
        d = nc - c5
        if d < 0 or rng.rand() < np.exp(-d / T):
            c5 = nc
        else:
            h[i], h[j] = 1.0, 0.0
        T *= 0.998
        if T < 1e-7:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    for tv in [2000, 8000, 25000]:

        if time_left(t0, budget_s) < 30:
