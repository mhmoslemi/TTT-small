def compute_corr(h):

def grad_at_max(h):
    """Gradient of the max correlation component."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = int(np.argmax(corr))
    n = len(h)
    s = k - (n - 1)

    else:

        for i in range(t, n):
            grad[i] += 1.0 - h[i - t]
            grad[i - t] -= h[i]
    return grad, k

def project(h, n):

            if t < 1e-15: break

    h = h.astype(float)
    h[-1] = max(0.0, min(1.0, h[-1] + target - h.sum()))
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_grad(h, T):
    """Smoothed gradient using softmax over correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    corr_shifted = corr - corr.max()
    w = np.exp(T * corr_shifted)
    w /= w.sum()
    # Compute weighted gradient
    grad = np.zeros(n)
    for p in range(len(corr)):
        if w[p] < 1e-8:
            continue
        s = p - (n - 1)
        if s > 0:
            grad[:n-s] += w[p] * (1.0 - h[s:])
            grad[s:] -= w[p] * h[:n-s]
        elif s < 0:
            t = -s
            grad[t:] += w[p] * (1.0 - h[:n-t])
            grad[:n-t] -= w[p] * h[t:]
        else:
            grad += w[p] * (1.0 - 2.0 * h)
    return grad * (2.0 / n)  # Include dx factor

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15: break
            h = h + r * (d / t)
        else:
            r = h.copy()
            t = r.sum()
            if t < 1e-15: break
            h = h + r * (d / t)
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    d = target - h.sum()
    if abs(d) > 1e-15:
        h[~(h >= 1.0)] += d / max(1, (h < 1.0).sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_smooth(h, n_iter, T_vals, a0=0.5):
    """Subgradient with annealed temperature."""
    h = h.copy()
    n = len(h)
    best = h.copy()
    best_v = compute_c5(h)
    for it in range(n_iter):
        T = T_vals[min(it // max(1, n_iter // len(T_vals)), len(T_vals) - 1)]
        a = a0 / (1.0 + 0.05 * it)
        g = smooth_grad(h, T)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        h_new = h - a * g / gn * gn  # = h - a * g
        # Actually let's use a better step size
        h_new = h - a * g
        h_new = project(h_new, n)
        v = compute_c5(h_new)
        if v < best_v:
            best_v = v
            best = h_new.copy()
        h = h_new
    return best, best_v

def slsqp_polish(h, maxiter=800):
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr))
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(np.clip(res.x, 0, 1), n)
    except:
        return h
