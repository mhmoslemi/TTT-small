    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-8:
        return c_max * dx
    shifted = corr - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-8:
        # Gradient of max
        k_max = np.argmax(corr)
        m = k_max - (n - 1)
        grad = np.zeros(n)
        if m >= 0:
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] -= h[m:]
        else:
            mp = -m
            grad[:n-mp] += (1.0 - h[mp:])
            grad[mp:] -= h[:n-mp]
        return dx * grad
    shifted = corr - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    w = np.exp(shifted / T)
    w = w / np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:

        m = k - (n - 1)
        if m >= 0:
            grad[m:] += wk * omh[:n-m]
            grad[:n-m] -= wk * h[m:]
        else:
            mp = -m
            grad[:n-mp] += wk * omh[mp:]
            grad[mp:] -= wk * h[:n-mp]
    return dx * grad

def fix_sum(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        # More careful redistribution
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add_v = min(diff, space)
                h[i] += add_v
                diff -= add_v

        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12: break
    return h

    """L-BFGS-B optimization of the softmax-max objective with sum constraint via penalty"""
    n = len(h)
    target_sum = n * 0.5

        hc = np.clip(x, 0.0, 1.0)
        base = soft_max_obj(hc, T)
        # Penalty for sum constraint
        s = np.sum(x)
        penalty = 100.0 * (s - target_sum) ** 2

        g = soft_max_grad(hc, T)
        # Only where not clipped
        g[hc <= 0.0] = 0.0
        g[hc >= 1.0] = 0.0
        # Penalty gradient
        g += 200.0 * (np.sum(x) - target_sum) * np.ones(n)
        return g
    
    res = opt.minimize(objective, h.copy(), jac=gradient,
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8,
                                'maxfun': 5000})
    if hasattr(res, 'x') and res.x is not None:
        return fix_sum(res.x, n)
    return h

import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-8:
        return c_max * dx
    shifted = corr - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    omh = 1.0 - h
    if T < 1e-8:
        k_max = np.argmax(corr)
        m = k_max - (n - 1)
        grad = np.zeros(n)
        if m >= 0:
            grad[m:] += omh[:n - m]
            grad[:n - m] -= h[m:]
        else:
            mp = -m
            grad[:n - mp] += omh[mp:]
            grad[mp:] -= h[:n - mp]
        return dx * grad
    shifted = corr - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    w = np.exp(shifted / T)
    w = w / np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += wk * omh[:n - m]
            grad[:n - m] -= wk * h[m:]
        else:
            mp = -m
            grad[:n - mp] += wk * omh[mp:]
            grad[mp:] -= wk * h[:n - mp]
    return dx * grad

def fix_sum(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idx = np.argsort(-h)
        for i in idx:
            r = min(diff, h[i])
            h[i] -= r
            diff -= r
            if diff <= 1e-12:
                break
    return h

def do_slsqp(h, n, T, max_iter=800):
    def f(x):
        return soft_max_obj(x, T)
    def g(x):
        return soft_max_grad(x, T)
    def cfun(x):
        return np.sum(x) - n * 0.5
    def cgrad(x):
        return np.ones(n)
    try:
        res = opt.minimize(f, h.copy(), jac=g, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cgrad}],
                           options={'maxiter': max_iter, 'ftol': 1e-15})
        return fix_sum(res.x, n) if res.x is not None else h
    except Exception:
        return h

def do_lgbfgs(h, n, T, max_iter=5000):
    target = n * 0.5
    pen_coef = 200.0
    def f(x):
        xc = np.clip(x, 0.0, 1.0)
        return soft_max_obj(xc, T) + pen_coef * (np.sum(x) - target) ** 2
    def g(x):
        xc = np.clip(x, 0.0, 1.0)
        gr = soft_max_grad(xc, T)
        mask = ((xc > 1e-6) & (xc < 1.0 - 1e-6)).astype(float)
        gr *= mask
        gr += 2.0 * pen_coef * (np.sum(x) - target)
        return gr
    try:
        res = opt.minimize(f, h.copy(), jac=g, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'maxfun': 10000})
        return fix_sum(res.x, n) if res.x is not None else h
    except Exception:
        return h

        h_opt = do_slsqp(hm, n_best, 0.0002, max_iter=800)
        c5_opt = compute_c5(fix_sum(h_opt, n_best))
        if c5_opt < c5_best - 1e-14:
            h_best = fix_sum(h_opt, n_best)
            c5_best = c5_opt

    # Phase 9: Final top-K squeeze
    if budget_s - elapsed() > 50:
        h_ref = topk_refine(h_best, n_best, 4, 0.0002, max_iter=800)
        c5_ref = compute_c5(fix_sum(h_ref, n_best))
        if c5_ref < c5_best - 1e-14:
            h_best = fix_sum(h_ref, n_best)
            c5_best = c5_ref

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    return (h_best, compute_c5(h_best), len(h_best))
