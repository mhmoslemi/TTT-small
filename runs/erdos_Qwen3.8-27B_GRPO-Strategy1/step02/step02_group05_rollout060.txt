import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_weights(corr, tau):
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros_like(corr)
    return w / ws

def exact_grad_at_k(h, k, N):
    """Gradient of corr[k] w.r.t. h, where k is index in full correlation (0 to 2N-2)."""
    L = k - (N - 1)
    grad = np.zeros(N)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        grad[L:] += (1.0 - h[:N-L])
        grad[:N-L] -= h[L:]
    else:
        M = -L
        grad[:N-M] += (1.0 - h[M:])
        grad[M:] -= h[:N-M]
    return grad

def weighted_grad(h, w, N):
    """Compute sum_k w[k] * d(corr[k])/dh."""
    grad = np.zeros(N)
    center = N - 1
    for k in range(2*N-1):
        wk = w[k]
        if wk < 1e-6:

        L = k - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N-L])
            grad[:N-L] -= wk * h[L:]
        else:
            M = -L
            grad[:N-M] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:N-M]

    """Nesterov-accelerated projected gradient with soft-max."""
    best_h = h.copy()
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr0)) * dx
    y = h.copy()  # Nesterov lookahead point
    v = np.zeros(N)  # velocity
    lr = lr0
    it = 0
    stagnation = 0
    while time.time() < deadline:
        frac = min(it / 3000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac

        w = soft_max_weights(corr_y, tau)
        grad_y = weighted_grad(y, w, N)

        # x_{t+1} = project(y_t - lr * grad_y)

        y = x_new + nesterov_alpha * (x_new - h)

    # Get top K indices
    idx = np.argsort(corr)[-K:]
    w = np.zeros_like(corr)

    top_vals = corr[idx]

        grad[:N-L] -= h[L:]
    else:

        grad[:N-M] += (1.0 - h[M:])
        grad[M:] -= h[:N-M]

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, L, N):
    grad = np.zeros(N)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        grad[L:] += (1.0 - h[:N - L])
        grad[:N - L] -= h[L:]
    else:
        M = -L
        grad[:N - M] += (1.0 - h[M:])
        grad[M:] -= h[:N - M]
    return grad

def momentum_opt(h, N, target, dx, deadline, lr_start, lr_decay, topK, mom,
                 jitter_every=2000, jitter_mag=0.015, min_lr=1e-10):
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    v = np.zeros(N)
    lr = lr_start
    it = 0
    stall = 0
    center = N - 1
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-11:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > jitter_every:
            h = project_sum(best_h + np.random.randn(N) * jitter_mag, target)
            v = np.zeros(N)
            stall = 0
            lr = max(lr_start * 0.4, min_lr * 10)
            continue
        k_sel = min(topK, 2 * N - 1)
        top_idx = np.argsort(corr)[-k_sel:]
        top_vals = corr[top_idx]
        w_base = top_vals - top_vals.min() + 1e-8
        ws = w_base.sum()
        if ws < 1e-18:
            weights = np.ones(k_sel) / k_sel
        else:
            weights = w_base / ws
        grad = np.zeros(N)
        for i in range(k_sel):
            L = top_idx[i] - center
            wk = weights[i]
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wk * (1.0 - h[:N - L])
                grad[:N - L] -= wk * h[L:]
            else:
                M = -L
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
        v = mom * v - lr * grad
        h = project_sum(h + v, target)
        lr *= lr_decay
        if lr < min_lr:
            break
        it += 1
    return best_h, best_c5

def coordinate_polish(h, N, target, dx, deadline, max_rounds=200):
    best_h = h.copy()
    dx_val = dx
    for rnd in range(max_rounds):
