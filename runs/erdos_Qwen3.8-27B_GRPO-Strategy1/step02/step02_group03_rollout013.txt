def sgrad(h, dx, beta):

import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    n = _sp['n']
    return float(np.sum(x[:n]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def slsqp_solve(h0, dx, ts, maxiter=300):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def sgrad(h, dx, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (cr - cr.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        kl1, kh1 = max(-(n-1), j-(n-1)), min(n-1, j)
        if kl1 <= kh1:
            k = np.arange(kl1, kh1+1)
            g[j] += np.dot(w[o+k], 1.0 - h[j-k])
        kl2, kh2 = max(-(n-1), -j), min(n-1, n-1-j)
        if kl2 <= kh2:
            k = np.arange(kl2, kh2+1)
            g[j] -= np.dot(w[o+k], h[j+k])
    return g * dx

def gdes(h0, dx, ts, maxit, deadline):

    h, bh = h0.copy(), h0.copy()
    bc = compute_c5(h0, dx)
    st = 0
    for i in range(maxit):
        if time.time() > deadline:
            break
        b = 25.0 + 100.0 * min(1.0, i / max(1, 0.4 * maxit))
        lr = 0.02 / (1.0 + 0.004 * i)
        g = sgrad(h, dx, b)
        ok = False
        for sh in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            h2 = proj(h - lr * sh * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                bh, bc, h, st = h2.copy(), c2, h2, 0
                ok = True
                break
        if not ok:
            st += 1
            if st > 200:
                break
    return bh, bc

def coord_descent(h0, dx, ts, rnds, deadline, rng):
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0, dx)
    for r in range(rnds):
        if time.time() > deadline:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in (0.01, -0.01, 0.03, -0.03, 0.06, -0.06, 0.005, -0.005):
                h[j] = sv + d
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                    continue
                np.clip(h, 0, 1, out=h)
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < bc - 1e-15:
                    bc, imp = c, True
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def anneal(h0, dx, ts, maxit, deadline, rng):
    n = len(h0)
    h, bh = h0.copy(), h0.copy()
    bc = compute_c5(h0, dx)
    T = 0.04

            if c < bc:
                bh, bc = h.copy(), c
        else:
            h[j] = sv
            h = proj(h, ts)
    return bh, bc
