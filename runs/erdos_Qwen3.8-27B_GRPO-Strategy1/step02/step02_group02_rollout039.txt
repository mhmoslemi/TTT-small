```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_topk(h, corr, topk_idx, n):
    """Gradient of weighted top-k corr values"""
    center = n - 1
    topk_vals = corr[topk_idx]
    mx = topk_vals.max()
    tau = max(1.0, (mx - topk_vals.min()) / max(1, len(topk_idx) - 1) * 3)
    w = np.exp((topk_vals - mx) / tau)
    w /= w.sum()
    
    xn = np.arange(n)
    grad = np.zeros(n)
    for idx, wi in zip(topk_idx, w):
        if wi < 1e-6:
            continue
        L = idx - center
        if L > 0:
            grad[:n-L] += wi * (1.0 - h[L:n])
            grad[L:] -= wi * h[:n-L]
        elif L < 0:
            L2 = -L
            grad[L2:] += wi * (1.0 - h[:n-L2])
            grad[:n-L2] -= wi * h[L2:]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad

def binary_local_search(h, n, max_rounds, rng):
    """Greedy binary local search: try all swaps, keep best"""
    dx = 2.0 / n
    h = (h > 0.5).astype(float)
    h = project_feasible(h, n)
    _, corr = compute_c5_full(h)
    best_c = np.max(corr) * dx
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for rd in range(max_rounds):
        improved = False
        best_delta = 1e-10
        best_i, best_j = -1, -1
        center = n - 1
        L_arr = np.arange(-(n-1), n)
        
        for i in ones:
            for j in zeros:
                # Compute delta corr for swap i(1→0), j(0→1)
                il = i - L_arr
                ir = i + L_arr
                jl = j - L_arr
                jr = j + L_arr
                
                valid_il = (il >= 0) & (il < n)
                valid_ir = (ir >= 0) & (ir < n)
                valid_jl = (jl >= 0) & (jl < n)
                valid_jr = (jr >= 0) & (jr < n)
                
                h_il = np.where(valid_il, h[np.clip(il, 0, n-1)], 0.0)
                h_ir = np.where(valid_ir, h[np.clip(ir, 0, n-1)], 0.0)
                h_jl = np.where(valid_jl, h[np.clip(jl, 0, n-1)], 0.0)
                h_jr = np.where(valid_jr, h[np.clip(jr, 0, n-1)], 0.0)
                
                delta = h_il + h_ir - h_jl - h_jr
                new_corr = corr + delta
                new_max = np.max(new_corr)
                if new_max < best_c - best_delta:
                    best_delta = best_c - new_max
                    best_i, best_j = i, j
                    best_new_corr = new_corr
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = best_new_corr
            best_c = np.max(corr)
            # Update sets
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            improved = True
        
        if not improved:
            break
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_lag(h, L, n):
    """Gradient of corr at lag L w.r.t. h"""
    g = np.zeros(n)
    if L > 0:
        g[:n-L] += (1.0 - h[L:n])
        g[L:] -= h[:n-L]
    elif L < 0:
        L2 = -L
        g[L2:] += (1.0 - h[:n-L2])
        g[:n-L2] -= h[L2:]
    else:
        g = 1.0 - 2.0 * h
    return g

def custom_gd(h, n, t_start, t_end, max_iter, topk_init, topk_min, rng):
    """Custom gradient descent with adaptive top-k"""
    dx = 2.0 / n
    center = n - 1
    best = h.copy()
    best_c, best_corr = float(np.max(np.correlate(h, 1-h, mode='full')) * dx), None
    curr_corr = np.correlate(h, 1-h, mode='full')
    curr_c = float(np.max(curr_corr) * dx)
    
    eta = 0.01
    for it in range(max_iter):
        if t_start is not None and time.time() - t_start > t_end:
            break
        
        # Adaptive top-k
        n_corr = len(curr_corr)
        k = max(topk_min, int(topk_init * (1.0 - it / max(1, max_iter))))
        topk_idx = np.argpartition(curr_corr, -k)[-k:]
        
        # Gradient
        topk_vals = curr_corr[topk_idx]
        mx = topk_vals.max()
        tau = max(0.5, (mx - topk_vals.min()) * 0.5)
        w = np.exp((topk_vals - mx) / tau)
        w /= w.sum()
        
        grad = np.zeros(n)
        for idx, wi in zip(topk_idx, w):
            if wi < 1e-6:
                continue
            L = idx - center
            grad += wi * grad_at_lag(h, L, n)
        
        # Line search
        for _ in range(20):
            new_h = project_feasible(h - eta * grad, n)
            new_corr = np.correlate(new_h, 1-new_h, mode='full')
            new_c = float(np.max(new_corr) * dx)
            if new_c < curr_c:
                h = new_h
                curr_corr = new_corr
                curr_c = new_c
                if curr_c < best_c:
                    best_c = curr_c
                    best = h.copy()
                break
            eta *= 0.5
        
        # Occasionally push binary
        if it % 50 == 49:
            hb = (h > 0.5).astype(float)
            hb = project_feasible(hb, n)
            cb, corr_b = float(np.max(np.correlate(hb, 1-hb, mode='full')) * dx), None
            if cb < curr_c:
                h = hb
                curr_c = cb
                curr_corr = np.correlate(hb, 1-hb, mode='full')
    
    return best, best_c, h, curr_c
```

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(2000):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        c = np.sum(m)
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, L, n):
    g = np.zeros(n)
    if L > 0:
        g[:n-L] += 1.0 - h[L:n]
        g[L:] -= h[:n-L]
    elif L < 0:
        L2 = -L
        g[L2:] += 1.0 - h[:n-L2]
        g[:n-L2] -= h[L2:]

    h = project(h, n)

    best_c = float(np.max(corr) * dx)
    best_h = h.copy()
    
    while time.time() < t_limit:

        A = np.zeros((n, 2*n-1))

            pl = p - L

                    np.where((pr>=0)&(pr<n), 1.0-h[np.clip(pr,0,n-1)], 0.0))
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]

        best_i, best_j = -1, -1

        for i in ones:
