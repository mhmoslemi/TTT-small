```python
def project_feasible(h, n):
    """Apply rigorous projection to enforce 0<=h<=1 and sum(h)=n/2"""
    # Copy input to avoid modifications
    h_temp = np.clip(h.copy(), 0.0, 1.0)
    total_value = np.sum(h_temp)
    ideal_sum = n / 2.0
    # Simple iterative adjustment
    for iteration in range(50):
        diff = ideal_sum - total_value
        if abs(diff) < 1e-6:  # Successful convergence
            break
        # Adjust using weighted distribution
        if diff > 0:  # Need to add more 1's
            room = 1.0 - h_temp
            total_room = np.sum(room)
            if total_room > 1e-12:
                h_temp += room * (diff / total_room)
                total_value = ideal_sum
        else:  # Need to add more 0's
            room = h_temp
            total_room = np.sum(room)
            if total_room > 1e-12:
                h_temp -= room * (diff / total_room)
                total_value = ideal_sum
    return np.clip(h_temp, 0.0, 1.0)
```
