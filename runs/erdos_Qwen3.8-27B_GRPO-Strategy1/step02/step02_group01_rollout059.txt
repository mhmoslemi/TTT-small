def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_from_continuous(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    n = len(h)
    target = n // 2
    # Sort by value descending, pick top target
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def repair_binary(b, n):
    """Ensure exactly n/2 ones in binary array."""
    target = n // 2
    cnt = int(b.sum())
    if cnt > target:
        # Need to turn off (cnt - target) ones
        ones = np.where(b == 1)[0]
        to_flip = np.random.choice(ones, cnt - target, replace=False)
        b[to_flip] = 0
    elif cnt < target:
        zeros = np.where(b == 0)[0]
        to_flip = np.random.choice(zeros, target - cnt, replace=False)
        b[to_flip] = 1
    return b

def mutate_binary(b, n, rng):
    """Pair swap mutation: flip k ones to 0 and k zeros to 1."""
    b = b.copy()
    n_flip = rng.randint(1, 4)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    if len(ones) < n_flip or len(zeros) < n_flip:
        n_flip = min(len(ones), len(zeros))
        if n_flip == 0:
            return b
    flip_ones = rng.choice(ones, n_flip, replace=False)
    flip_zeros = rng.choice(zeros, n_flip, replace=False)
    b[flip_ones] = 0
    b[flip_zeros] = 1
    return b

def crossover_binary(b1, b2, n, rng):
    """Uniform crossover maintaining count."""
    mask = rng.random(n) > 0.5
    child = np.where(mask, b1, b2).astype(float)
    return repair_binary(child, n)

        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx

    except Exception:
        return h

def slsqp_erdos(h, maxiter=400):
    n = len(h)
    def erdos_obj(x):
        return x[-1]
    def erdos_eq(x):
        return x[:n].sum() - n / 2.0
    def erdos_ineq(x):
        hh = x[:n]
        t = x[-1]
        corr = np.correlate(hh, 1.0 - hh, mode='full')
        return t - corr
    
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr0)) + 0.02
    x0 = np.concatenate([h, [t0]])

    cons = [
        {'type': 'eq', 'fun': erdos_eq},
        {'type': 'ineq', 'fun': erdos_ineq}

    if time.time() - t0 < budget_s * 0.04:
        ht = cont_opt(bh, 400, 10.0, 150)
        ct = compute_c5(ht)
        if ct < bc:
            bc, bh = ct, ht

    # Phase 2: GA on binary at parent resolution (main global search)
    if time.time() - t0 < budget_s * 0.28:
        bb, bs = ga_search(bh, bn, budget_s * 0.24, rng)
        if bs < bc:
            bc, bh = bs, bb

    # Phase 3: Greedy local search (best pair swap)
    if time.time() - t0 < budget_s * 0.38:
        bl, sl = local_bin(bh, bn, maxit=100)
        if sl < bc:
            bc, bh = sl, bl

    # Phase 4: Continuous polish of best binary
    if time.time() - t0 < budget_s * 0.48:
        ht = cont_opt(bh, 1000, 30.0, 400)
        ct = compute_c5(ht)
        if ct < bc:
            bc, bh = ct, ht

    # Phase 5: Simulated annealing refinement + local search
    if time.time() - t0 < budget_s * 0.64:
        bb, bs = sa_search(to_binary(bh), bn, budget_s * 0.10, rng)
        bl, sl = local_bin(bb, bn, maxit=80)
        cand_s = min(sl, bs)
        cand_b = bl if sl <= bs else bb
        if cand_s < bc:
            bc, bh = cand_s, cand_b

    # Phase 6: Multi-resolution GA
    for nt in [80, 100, 72, 110, 66]:
        if time.time() - t0 > budget_s * 0.78:
            break
        xo = np.linspace(0, 2, bn)
        xn = np.linspace(0, 2, nt)
        hi = np.interp(xn, xo, bh)
        hi = project_feasible(hi, nt)
        ht = cont_opt(hi, 300, 5.0, 120)
        bb, bs = ga_search(ht, nt, budget_s * 0.05, rng)
        bl, sl = local_bin(bb, nt, maxit=40)
        cs = min(sl, bs)
        cb = bl if sl <= bs else bb
        if cs < bc:
            bc, bh, bn = cs, cb, nt

    # Phase 7: Second GA round from best (exploit)
    if time.time() - t0 < budget_s * 0.90:
        bb, bs = ga_search(bh, bn, budget_s * 0.07, rng)
        bl, sl = local_bin(bb, bn, maxit=100)
        if sl < bc:
            bc, bh = sl, bl
        elif bs < bc:
            bc, bh = bs, bb

    # Phase 8: Final continuous polish
    if time.time() - t0 < budget_s * 0.96:
        ht = cont_opt(bh, 5000, 100.0, 400)
        ct = compute_c5(ht)
        if ct < bc:
            bc, bh = ct, ht
        bl, sl = local_bin(to_binary(bh), bn, maxit=50)
        if sl < bc:
            bc, bh = sl, bl
        ht2 = cont_opt(bh, 3000, 50.0, 200)
        ct2 = compute_c5(ht2)
        if ct2 < bc:
            bc, bh = ct2, ht2

    bh = project_feasible(bh, len(bh))
    bc = compute_c5(bh)
    return bh, bc, len(bh)
