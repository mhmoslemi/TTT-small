    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = int(np.sum(interior))
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def build_coeff(h_old, n):
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n))
    omh = 1.0 - h_old
    for m in range(n_lags):
        k = m - (n - 1)
        if k >= 0:
            A[m, k:] = omh[:n - k]
        else:
            A[m, :n + k] = omh[-k:]
    return A

def solve_lp(h_old, n, solver_name='CLARABEL'):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    A = build_coeff(h_old, n) * dx
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n * 0.5,
        A @ h_var <= t_var
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver_name, verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
        except Exception:
            return h_old.copy(), float('inf')
    if h_var.value is not None:
        return np.clip(h_var.value, 0.0, 1.0), float(t_var.value) if t_var.value is not None else float('inf')
    return h_old.copy(), float('inf')

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0}
    )
    return proj_constraint(res.x, n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def sa_refine(h0, n, t_start, budget_s, max_steps, seed):
    rng = np.random.RandomState(seed)
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    T_sa = 0.0003
    for step in range(max_steps):
        if time.time() - t_start > budget_s - 30:
            break
        n_blk = rng.randint(2, 8)
        idxs = rng.choice(n, size=n_blk, replace=False)
        h_try = h_cur.copy()
        h_try[idxs] = np.clip(h_try[idxs] + rng.randn(n_blk) * 0.025, 0, 1)
        h_try = proj_constraint(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / max(T_sa, 1e-8)):
            h_cur = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        T_sa *= 0.995
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Alternating LP with adaptive damping
    alpha_schedule = [1.0, 0.85, 0.7, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1]
    h_lp = h_best.copy()
    for alpha in alpha_schedule:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 250:
            break
        n_iters = 25 if alpha > 0.3 else 15
        for it in range(n_iters):
            elapsed = time.time() - t_start
            if budget_s - elapsed < 250:
                break
            h_new, t_val = solve_lp(h_lp, n_best)
            h_candidate = (1.0 - alpha) * h_lp + alpha * h_new
            h_candidate = proj_constraint(h_candidate, n_best)
            c = compute_c5(h_candidate)
            if c < c5_best:
                h_best = h_candidate.copy()
                c5_best = c
            h_lp = h_candidate
            if t_val < 1e-10:
                break

    # Phase 2: SLSQP polish with decreasing T
    elapsed = time.time() - t_start
    if budget_s - elapsed > 120:
        for T in [0.01, 0.004, 0.0015, 0.0006, 0.0002]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 30:
                break
            h_opt = do_slsqp(h_best.copy(), n_best, T, 800)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Phase 3: Multi-start perturbations + LP
    elapsed = time.time() - t_start
    for i in range(6):
        elapsed = time.time() - t_start
        if budget_s - elapsed < 100:
            break
        rng = np.random.RandomState(seed + i * 104729 + 7)
        sigma = 0.008 + 0.006 * (i % 3)
        h_pert = np.clip(h_best + rng.randn(n_best) * sigma, 0, 1)
        h_pert = proj_constraint(h_pert, n_best)
        for alpha in [0.6, 0.3, 0.15]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 50:
                break
            h_new, _ = solve_lp(h_pert, n_best)
            h_pert = (1.0 - alpha) * h_pert + alpha * h_new
            h_pert = proj_constraint(h_pert, n_best)
        h_opt = do_slsqp(h_pert, n_best, 0.001, 500)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 4: Simulated annealing
    elapsed = time.time() - t_start
    if budget_s - elapsed > 120:
        h_sa, c_sa = sa_refine(h_best, n_best, t_start, budget_s, 500, seed + 555)
        if c_sa < c5_best:
            h_best = h_sa
            c5_best = c_sa

    # Phase 5: Try upsampled resolution
    elapsed = time.time() - t_start
    if budget_s - elapsed > 150:
        n_up = n_best + n_best // 4  # 1.25x
        n_up = n_up // 4 * 4  # keep divisible
        if n_up > n_best:
            h_up = upsample(h_best, n_up)
            for alpha in [0.7, 0.4, 0.2, 0.1]:
                elapsed = time.time() - t_start
                if budget_s - elapsed < 80:
                    break
                h_new, _ = solve_lp(h_up, n_up)
                h_up = (1.0 - alpha) * h_up + alpha * h_new
                h_up = proj_constraint(h_up, n_up)
            h_up = do_slsqp(h_up, n_up, 0.002, 600)
            c = compute_c5(h_up)
            if c < c5_best:
                h_best = h_up
                c5_best = c
                n_best = n_up

    # Phase 6: Final ultra-fine polish
    elapsed = time.time() - t_start
    if budget_s - elapsed > 40:
        for T in [0.0003, 0.0001, 0.00005]:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 20:
                break
            h_opt = do_slsqp(h_best.copy(), n_best, T, 2000)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    # Final LP squeeze
    elapsed = time.time() - t_start
    if budget_s - elapsed > 15:
        for alpha in [0.1, 0.05, 0.02]:
            h_new, _ = solve_lp(h_best, n_best)
            h_candidate = (1.0 - alpha) * h_best + alpha * h_new
            h_candidate = proj_constraint(h_candidate, n_best)
            c = compute_c5(h_candidate)
            if c < c5_best:
                h_best = h_candidate
                c5_best = c

    h_best = proj_constraint(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
