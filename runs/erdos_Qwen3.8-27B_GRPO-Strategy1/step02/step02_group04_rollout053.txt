import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    t = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - t) < 1e-12:
        return h
    if s < t:
        d = t - s
        for i in np.argsort(h):
            a = min(d, 1.0 - h[i])
            h[i] += a
            d -= a
            if d <= 1e-12: break
    else:
        d = s - t
        for i in np.argsort(-h):
            a = min(d, h[i])
            h[i] -= a
            d -= a
            if d <= 1e-12: break
    return np.clip(h, 0.0, 1.0)

def sm_obj(h, T):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    if T < 1e-15: return cm * dx
    s = np.sum(np.exp(np.clip(c - cm, -80*T, 0.0) / T))
    return (cm + T * np.log(max(s, 1e-300))) * dx

def sum_eq(x, n): return np.sum(x) - n * 0.5

def slsqp_opt(h, n, T, mi):
    try:
        r = opt.minimize(sm_obj, h.copy(), args=(T,),
            method='SLSQP', bounds=[(0.0,1.0)]*n,
            constraints=[{'type':'eq','fun':sum_eq,'args':(n,)}],
            options={'maxiter':mi, 'ftol':1e-16})
        if r.x is not None: return fix_sum(r.x, n)
    except: pass
    return h

def grad_sm(h, T):
    n = len(h); dx = 2.0/n
    C = np.correlate(h, 1.0-h, mode='full')
    cm = np.max(C)
    w = np.exp(np.clip(C-cm, -80*T, 0.0)/T)
    ws = np.sum(w)
    if ws < 1e-300: return np.zeros(n)
    w /= ws
    omh = 1.0 - h
    g = np.zeros(n)
    for j in range(n):
        dmax = n - j
        if dmax > 0: g[j] += np.dot(w[n-1:n-1+dmax], omh[j:j+dmax])
        dl = max(-(n-1), j-n+1); dh = min(n-1, j)
        if dl <= dh: g[j] -= np.dot(w[n-1+dl:n+dh+1], h[j-dh:j-dl+1])
    return g * dx / T

def proj_grad(h, n, T0, T1, niter):
    h = h.copy()
    for it in range(niter):
        T = T0 * (T1/T0) ** (it / max(niter-1, 1))
        ob = sm_obj(h, T)
        g = grad_sm(h, T)
        sn = np.linalg.norm(g) ** 2
        if sn < 1e-20: break
        step = T * 0.1 / (sn / n + 1e-10)
        for _ in range(20):
            h2 = fix_sum(h - step * g, n)
            if sm_obj(h2, T) < ob: break
            step *= 0.5
        h = h2
    return h

def bin_ls(h, n, max_passes=8, tlimit=30.0):
    hb = (h > 0.5).astype(float)
    s = np.sum(hb)
    if s != n*0.5:
        if s > n*0.5:
            for i in np.argsort(h[np.where(hb==1)[0]])[:int(s-n*0.5)]:
                hb[np.where(hb==1)[0][np.argsort(h[np.where(hb==1)[0]])][i]] = 0
        else:
            for i in np.argsort(-h[np.where(hb==0)[0]])[:int(n*0.5-s)]:
                hb[np.where(hb==0)[0][np.argsort(-h[np.where(hb==0)[0]])][i]] = 1
    best = compute_c5(hb)
    t0 = time.time()
    for p in range(max_passes):
        if time.time()-t0 > tlimit: break
        c = np.correlate(hb, 1.0-hb, mode='full').astype(float)
        ones = np.where(hb==1)[0]; zeros = np.where(hb==0)[0]
        d = np.arange(-(n-1), n)
        found = False
        for i in ones:
            ip = i+d; im = i-d
            rem = -(1.0-hb[np.clip(ip,0,n-1)])*(ip>=0)*(ip<n) + hb[np.clip(im,0,n-1)]*(im>=0)*(im<n)
            jp = zeros[None,:]+d; jm = zeros[None,:]-d
            add = (1.0-hb[np.clip(jp,0,n-1)])*(jp>=0)*(jp<n) - hb[np.clip(jm,0,n-1)]*(jm>=0)*(jm<n)
            deltas = add + rem[None,:]
            nmx = np.max(c[None,:] + deltas, axis=1)
            bi = np.argmin(nmx)
            if nmx[bi] * 2.0/n < best - 1e-12:
                hb[i] = 0; hb[zeros[bi]] = 1
                c += rem + add[bi]
                best = nmx[bi] * 2.0/n
                ones = ones[ones != i]
                zeros = np.delete(zeros, bi)
                found = True
                break
        if not found: break
    return hb

def upsample(h, nn):
    x0 = np.arange(len(h)) * 2.0/len(h)
    x1 = np.arange(nn) * 2.0/nn
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def downsample(h, nn):

    x1 = np.arange(nn) * 2.0/nn
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    t = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - t) < 1e-12:
        return h
    if s < t:
        d = t - s
        for i in np.argsort(h):
            a = min(d, 1.0 - h[i])
            h[i] += a; d -= a
            if d <= 1e-12: break
    else:
        d = s - t
        for i in np.argsort(-h):
            a = min(d, h[i])
            h[i] -= a; d -= a
            if d <= 1e-12: break
    return np.clip(h, 0.0, 1.0)

def sm_obj(h, T):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    if T < 1e-15: return cm * dx
    s = np.sum(np.exp(np.clip(c - cm, -80.0*T, 0.0) / T))
    return (cm + T * np.log(max(s, 1e-300))) * dx

def sum_eq(x, n):
    return np.sum(x) - n * 0.5

def slsqp_opt(h, n, T, mi):
    try:
        r = opt.minimize(sm_obj, h.copy(), args=(T,),
            method='SLSQP', bounds=[(0.0,1.0)]*n,
            constraints=[{'type':'eq','fun':sum_eq,'args':(n,)}],
            options={'maxiter':mi, 'ftol':1e-16})
        if r.x is not None: return fix_sum(r.x, n)
    except Exception:
        pass
    return h

def grad_raw(h, T):
    n = len(h)
    C = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(C)
    w = np.exp(np.clip(C - cm, -80.0*T, 0.0) / T)
    ws = np.sum(w)
    if ws < 1e-300: return np.zeros(n)
    w /= ws
    omh = 1.0 - h
    g = np.zeros(n)
    for j in range(n):
        dmax = n - j
        if dmax > 0:
            g[j] += np.dot(w[n-1:n-1+dmax], omh[j:j+dmax])
        dl = max(-(n-1), j-n+1)
        dh = min(n-1, j)
        if dl <= dh:
            g[j] -= np.dot(w[n-1+dl:n+dh+1], h[j-dh:j-dl+1])
    return g

def proj_grad(h, n, T, niter, step0):
    h = h.copy()
    ob = sm_obj(h, T)
    g = grad_raw(h, T)
    gn = np.linalg.norm(g) / np.sqrt(n) + 1e-12
    step = step0 / gn
    for it in range(niter):

        for bs in range(25):

            step *= 0.5
        if not accepted: break
        g = grad_raw(h, T)
        gn = np.linalg.norm(g) / np.sqrt(n) + 1e-12
        step = step0 / gn
    return h

def bin_ls(h, n, max_passes=8, tlimit=25.0):

    s1 = int(np.sum(hb))
