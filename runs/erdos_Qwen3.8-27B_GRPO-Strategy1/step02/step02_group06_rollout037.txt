import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def downsample_avg(h, new_n):
    old_n = len(h)
    h_new = np.zeros(new_n)
    for i in range(new_n):
        lo_idx = int(i * old_n / new_n)
        hi_idx = min(int((i + 1) * old_n / new_n), old_n)

    return proj_constraint(h_new, new_n)

    x = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + 0.5 * np.cos(np.pi * x)
    return proj_constraint(h, n)

def _sum_eq(x, n):
    return np.sum(x) - n * 0.5

def _sum_eq_jac(x, n):
    return np.ones(n)

def do_slsqp(h0, n, T, max_iter):
    res = opt.minimize(
        soft_max_obj, h0, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}],
        options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0}
    )
    return proj_constraint(res.x, n)

def pair_cd(h, n, T, n_sweeps, seed):
    rng = np.random.RandomState(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    h = h.copy()
    hc = h.copy()
    hd = h.copy()
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for j in order:
            k = (j + 1 + (j * 7) % 13) % n
            if k == j:
                k = (j + 1) % n
            s_pair = h[j] + h[k]
            lo = max(0.0, s_pair - 1.0)
            hi = min(1.0, s_pair)
            if hi - lo < 1e-10:
                continue
            a, b = lo, hi
            cv = b - gr * (b - a)
            dv = a + gr * (b - a)
            hc[:] = h; hc[j] = cv; hc[k] = s_pair - cv
            fc = soft_max_obj(hc, T)
            hd[:] = h; hd[j] = dv; hd[k] = s_pair - dv
            fd = soft_max_obj(hd, T)
            for _ in range(25):
                if fc < fd:
                    b = dv; dv = cv; fd = fc
                    cv = b - gr * (b - a)
                    hc[:] = h; hc[j] = cv; hc[k] = s_pair - cv
                    fc = soft_max_obj(hc, T)
                else:
                    a = cv; cv = dv; fc = fd
                    dv = a + gr * (b - a)
                    hd[:] = h; hd[j] = dv; hd[k] = s_pair - dv
                    fd = soft_max_obj(hd, T)
            best_t = (a + b) / 2.0
            h[j] = best_t
            h[k] = s_pair - best_t
    return proj_constraint(h, n)
