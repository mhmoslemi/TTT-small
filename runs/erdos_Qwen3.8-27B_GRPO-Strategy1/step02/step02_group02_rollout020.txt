import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    val = (mx + np.log(np.sum(np.exp(s))) / t) * dx
    # Add small regularization for smoothness
    reg = 1e-6 * np.sum(np.diff(h)**2)
    return val + reg

def topk_obj(h, t, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx]
    mx = np.max(c)
    s = t * (c - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def binary_proj(h, n):
    # Project to binary with exactly n/2 ones
    idx = np.argsort(-h)[:n//2]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def sa_binary(hb, n, t0, t1, n_iter, rng):
    """Simulated annealing on binary sequence with swap moves."""
    ones = np.where(hb == 1.0)[0]
    zeros = np.where(hb == 0.0)[0]
    best_hb = hb.copy()
    best_c5 = compute_c5(hb)
    curr_c5 = best_c5
    
    for it in range(n_iter):
        T = t0 * (t1 / t0) ** (it / n_iter)
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        # Perform swap
        hb[i], hb[j] = 0.0, 1.0
        new_c5 = compute_c5(hb)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c5 = new_c5
            ones = np.where(hb == 1.0)[0]
            zeros = np.where(hb == 0.0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_hb = hb.copy()
        else:
            # Reject, undo swap
            hb[i], hb[j] = 1.0, 0.0
    
    return best_hb, best_c5

def swap_search(hb, n, max_rounds=20):
    """Steepest descent: try all single swaps, take best improving."""
    best_c5 = compute_c5(hb)
    for _ in range(max_rounds):
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        best_delta = 0
        best_pair = None
        # Sample a subset for speed
        n_try = min(len(ones) * len(zeros), 5000)
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]  # need rng...
            j = zeros[rng.randint(len(zeros))]
            hb[i], hb[j] = 0.0, 1.0
            c5 = compute_c5(hb)
            delta = c5 - best_c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)
            hb[i], hb[j] = 1.0, 0.0
        if best_pair is None:
            break
        hb[best_pair[0]], hb[best_pair[1]] = 0.0, 1.0
        best_c5 += best_delta
    return hb, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-16:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def topk_obj(h, t, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx]
    mx = np.max(c)
    s = t * (c - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def lbb_obj(h, t, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    base = (mx + np.log(np.sum(np.exp(s))) / t) * dx
    pen = penalty * (np.sum(h) - n / 2.0) ** 2
    return base + pen

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def interp_down(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def binary_proj(h, n):
    idx = np.argsort(-h)[:n // 2]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def sa_binary(hb, n, T_start, T_end, n_iter, rng):
    best_hb = hb.copy()
    best_c5 = compute_c5(hb)
    curr_c5 = best_c5
    ones = np.where(hb == 1.0)[0]
    zeros = np.where(hb == 0.0)[0]
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (it / max(n_iter - 1, 1))
        i = ones[rng.randint(ones.size)]
        j = zeros[rng.randint(zeros.size)]
        hb[i], hb[j] = 0.0, 1.0
        new_c5 = compute_c5(hb)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_c5 = new_c5
            ones = np.where(hb == 1.0)[0]
            zeros = np.where(hb == 0.0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_hb = hb.copy()
        else:
            hb[i], hb[j] = 1.0, 0.0
    return best_hb, best_c5

def swap_descent(hb, n, rng, max_rounds=30, n_try=3000):
    best_c5 = compute_c5(hb)
    for _ in range(max_rounds):
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        best_delta = 1e-15
        best_pair = None
        for _ in range(n_try):
            i = ones[rng.randint(ones.size)]
            j = zeros[rng.randint(zeros.size)]
            if i == j:
                continue
            hb[i], hb[j] = 0.0, 1.0
            c5 = compute_c5(hb)
            delta = c5 - best_c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)
            hb[i], hb[j] = 1.0, 0.0
        if best_pair is None:
            break
        hb[best_pair[0]], hb[best_pair[1]] = 0.0, 1.0
        best_c5 += best_delta
    return hb, best_c5

def make_periodic(n, p, frac):
    h = np.zeros(n)
    for i in range(n):
        if (i % p) < int(p * frac):
            h[i] = 1.0
    return proj(h, n)

def make_blocks(n, nb):
    h = np.zeros(n)
    block_len = n // nb
    for b in range(nb):
        if b % 2 == 0:
            h[b * block_len:(b + 1) * block_len] = 1.0
    return proj(h, n)
