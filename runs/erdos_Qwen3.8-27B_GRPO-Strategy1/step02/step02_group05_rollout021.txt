def project_sum(h, target):

    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            hi = mid
        else:
            lo = mid

                    v2[N - 1] = b[N - 1] + m2 - m2 * m2
                    if float(np.max(v1)) < float(np.max(v2)):
                        hi = m2
                    else:
                        lo = m1
                    if hi - lo < 1e-13:
                        break
                xbest = (lo + hi) / 2.0
            if abs(xbest - curr[j]) > 1e-9:
                curr[j] = xbest
                improved = True
        curr = project_sum(curr, target)
        c5_val = compute_c5(curr, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = curr.copy()
        if not improved:
            break
    return best_h, best_c5

def adam_grad_opt(h, N, target, dx, deadline, tau0, tau1, lr_init, decay_period, decay_factor):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    lr = lr_init
    it = 0
    center = N - 1
    while time.time() < deadline:
        frac = min(it / 5000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros_like(corr)
            w[center] = 1.0
        else:
            w /= ws
        grad = np.zeros(N)
        grad += w[center] * (1.0 - 2.0 * h)
        for L in range(1, N):
            wLp = w[center + L]
            wLm = w[center - L]
            if abs(wLp) < 1e-10 and abs(wLm) < 1e-10:
                continue
            if wLp > 1e-10:
                grad[L:] += wLp * (1.0 - h[:N - L])
                grad[:N - L] -= wLp * h[L:]
            if wLm > 1e-10:
                grad[:N - L] += wLm * (1.0 - h[L:])
                grad[L:] -= wLm * h[:N - L]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % decay_period == 0:
            lr *= decay_factor
        if lr < 1e-10:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    N = n0
    dx = 2.0 / N
    target = N / 2.0
    # Phase 1: Adam from initial (30% budget)
    p1_dead = t0 + (budget_s - 5) * 0.30
    if time.time() < p1_dead - 5:
        h1, c1 = adam_grad_opt(h0.copy(), N, target, dx, p1_dead,
                               tau0=5.0, tau1=300.0, lr_init=0.02,
                               decay_period=60, decay_factor=0.985)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()
    # Phase 2: CD polish on best (50% budget)
    cd_dead = t0 + (budget_s - 5) * 0.82
    if time.time() < cd_dead - 5:
        h2, c2 = cd_polish(best_h.copy(), N, max_sweeps=500, deadline=cd_dead)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
    # Phase 3: try perturbation + Adam + short CD
    if time.time() < deadline - 80:
        h_r = best_h + rng.randn(N) * 0.1
        h_r = project_sum(h_r, target)
        p3_dead = deadline - 40
        h3, c3 = adam_grad_opt(h_r, N, target, dx, p3_dead,
                               tau0=1.0, tau1=200.0, lr_init=0.025,
                               decay_period=50, decay_factor=0.98)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
        if time.time() < deadline - 10:
            h4, c4 = cd_polish(best_h.copy(), N, max_sweeps=200, deadline=deadline - 2)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()
    # Final CD polish
    if time.time() < deadline - 3:
        h5, c5f = cd_polish(best_h.copy(), N, max_sweeps=100, deadline=deadline - 1)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = h5.copy()
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
