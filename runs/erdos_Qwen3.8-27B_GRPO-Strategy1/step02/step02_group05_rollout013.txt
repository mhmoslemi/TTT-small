import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5, corr
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-8
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5, corr

def exact_worst_grad(h, N, corr):
    center = N - 1
    worst_idx = int(np.argmax(corr))
    L = worst_idx - center
    grad = np.zeros(N)
    if L == 0:
        grad = 1.0 - 2.0 * h
    elif L > 0:
        grad[L:] += (1.0 - h[:N - L])
        grad[:N - L] -= h[L:]
    else:
        M = -L
        grad[:N - M] += (1.0 - h[M:])
        grad[M:] -= h[:N - M]
    return grad

def adam_run(h, N, target, dx, deadline, tau0, tau1, lr0, decay, decay_every=25):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 5000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        grad, c5_val, _ = smax_grad(h, N, tau)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % decay_every == 0:
            lr *= decay
        if lr < 1e-11:
            break
    return best_h, best_c5

def focus_worst(h, N, target, dx, n_steps, lr_init):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr_init
    for _ in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        grad = exact_worst_grad(h, N, corr)

        if gnorm < 1e-14:
            break
        h = h - lr * grad / gnorm
        h = project_sum(h, target)
        lr *= 0.997
    return best_h, best_c5
