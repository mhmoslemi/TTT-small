import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def logsumexp_obj(h, T, mu=100.0):
    n = len(h)
    dx = 2.0 / n
    h_c = np.clip(h, 0, 1)
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    val = smooth_max(corr, T) * dx
    penalty = mu * (h_c.sum() - n/2.0)**2
    return val + penalty

def pnorm_obj(h, p, mu=100.0):
    n = len(h)
    dx = 2.0 / n
    h_c = np.clip(h, 0, 1)
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    val = (np.sum(corr**p)) ** (1.0/p) * dx
    penalty = mu * (h_c.sum() - n/2.0)**2
    return val + penalty

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def swap_move(h, n, rng):
    """Swap-based move that preserves the sum."""
    i, j = rng.choice(n, 2, replace=False)
    delta = h[i] - h[j]
    if abs(delta) < 1e-10:
        return h
    h2 = h.copy()
    h2[i] = h[j]
    h2[j] = h[i]
    return h2

def nudge_move(h, n, rng, step=0.05):
    """Move mass from one bin to another."""
    i, j = rng.choice(n, 2, replace=False)
    d = step * (1 if h[i] > h[j] else -1)
    h2 = h.copy()
    h2[i] = np.clip(h[i] + d, 0, 1)
    h2[j] = np.clip(h[j] - d, 0, 1)
    # Fix sum
    s = h2.sum() - n/2.0
    if abs(s) > 1e-13:
        # Redistribute
        room = 1 - h2 if s < 0 else h2
        total = room.sum()
        if total > 1e-13:
            h2 -= room * (s / total)
        h2 = np.clip(h2, 0, 1)
    return h2

def lbbfgs_optimize(h, T, n, mu=100.0, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    res = minimize(logsumexp_obj, h, args=(T, mu),
                  method='L-BFGS-B', bounds=bnds,
                  options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                           'maxfun': 50000})
    return project_feasible(np.clip(res.x, 0, 1), n)

    h = np.zeros(n)

    elif pattern_type == 1:  # Alternating

    elif pattern_type == 7:  # Diagonal mod pattern
        h = (np.arange(n) % 3).astype(float) / 2.0
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def logsumexp_obj(h, T, mu):
    h_c = np.clip(h, 0.0, 1.0)
    n = len(h_c)
    dx = 2.0 / n
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    m = np.max(corr)
    smooth_max = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return smooth_max * dx + mu * (h_c.sum() - n / 2.0) ** 2

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-13:
            h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-13:
            h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def lbbfgs_opt(h, T, n, mu=100.0, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    res = minimize(logsumexp_obj, h, args=(T, mu), method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                         'gtol': 1e-9, 'maxfun': 80000})
    return project_feasible(np.clip(res.x, 0.0, 1.0), n)
