import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_val(h, T):
    corr = compute_corr(h)
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max
    s = np.sum(np.exp(np.clip((corr - c_max) / T, -50.0, 0.0)))
    return c_max + T * np.log(max(s, 1e-300))

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.35:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            d = diff
            for i in idx:
                add = min(d, 1.0 - h[i])
                h[i] += add
                d -= add
                if d < 1e-13:
                    break
        else:
            d = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(d, h[i])
                h[i] -= rem
                d -= rem
                if d < 1e-13:
                    break
    return h

def sensitivity_matrix(h):
    """Compute sens[j, m] = d(corr[m])/dh[j] for all j, m."""
    n = len(h)
    corr_len = 2 * n - 1
    sens = np.zeros((n, corr_len))
    for m in range(corr_len):
        s = m - (n - 1)
        for j in range(n):
            val = 0.0
            if 0 <= j - s < n:
                val += (1.0 - h[j - s])
            if 0 <= j + s < n:
                val -= h[j + s]
            sens[j, m] = val
    return sens

def binary_local_search(h, n, max_rounds=50, time_limit=120, t0=None):
    """Greedy binary local search with precomputed sensitivities."""
    if t0 is None:
        t0 = time.time()
    c5 = compute_c5(h)
    corr = compute_corr(h)
    for rnd in range(max_rounds):
        if time.time() - t0 > t0 + time_limit:
            break
        sens = sensitivity_matrix(h)
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() - t0 > t0 + time_limit:
                break
            hj = h[j]
            if hj < 0.01 or hj > 0.99:
                continue
            # Try pushing h[j] toward 0 or 1, compensated by another point
            target_dir = 1.0 if hj < 0.5 else -1.0
            delta = 0.5 - abs(hj - 0.5)
            if delta < 0.01:
                delta = 0.1
            # Find best k to compensate
            best_c5 = c5
            best_k = -1
            # Try a subset of candidates for speed
            candidates = np.where((target_dir > 0) != (h > 0.5))[0]
            candidates = candidates[candidates != j]
            if len(candidates) == 0:
                continue
            # Random subset for speed
            n_trial = min(40, len(candidates))
            trial_idx = np.random.choice(candidates, size=n_trial, replace=False)
            for k in trial_idx:
                dk = -target_dir * delta
                if h[k] + dk < -1e-10 or h[k] + dk > 1.0 + 1e-10:
                    dk = np.sign(dk) * min(abs(dk), 
                                           1.0 - h[k] if dk > 0 else h[k])
                    if abs(dk) < 1e-8:
                        continue
                    delta = -target_dir * dk
                new_corr = corr + delta * (sens[j] - sens[k])
                new_c5 = np.max(new_corr) * (2.0 / n)
                if new_c5 < best_c5 - 1e-10:
                    best_c5 = new_c5
                    best_k = k
            if best_k >= 0:
                h[j] += delta
                h[best_k] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = fix_sum(h, n)
                corr = compute_corr(h)
                c5 = compute_c5(h)
                improved = True
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = fix_sum(h, n)
    c5 = compute_c5(h)
    dx = 2.0 / n

    def elapsed():
        return time.time() - t0

    def try_update(hc):
        nonlocal h, c5
        hc = fix_sum(hc.copy(), n)
        c5h = compute_c5(hc)
        if c5h < c5 - 1e-14:
            h = hc.copy()
            c5 = c5h
            return True
        return False

    # Phase 1: L-BFGS-B with sum penalty (fast, many iterations)
    penalties = [(0.03, 200, 1.0), (0.01, 400, 10.0), (0.004, 600, 50.0),
                 (0.0015, 800, 200.0), (0.0005, 1000, 1000.0)]
    for T, iters, lam in penalties:
        if budget_s - elapsed() < 60:
            break
        def obj_lbfgs(x):
            return soft_max_val(x, T) * dx + lam * (np.sum(x) - n * 0.5) ** 2
        res = opt.minimize(obj_lbfgs, h.copy(), method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-10})
        if hasattr(res, 'x') and res.x is not None:
            try_update(res.x)

    # Phase 2: SLSQP with exact constraint (fewer iterations, more accurate)
    def sum_con(x):
        return np.sum(x) - n * 0.5

    for T in [0.002, 0.0008, 0.0003]:
        if budget_s - elapsed() < 45:
            break
        res = opt.minimize(soft_max_val, h.copy(), args=(T,),
                         method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': sum_con}],
                         options={'maxiter': 3000, 'ftol': 1e-16})
        if hasattr(res, 'x') and res.x is not None:
            try_update(res.x)

    # Phase 3: Binary local search (key innovation)
    if budget_s - elapsed() > 120:
        h_bs, c5_bs = binary_local_search(h.copy(), n, max_rounds=80,
                                          time_limit=min(180, budget_s - elapsed() - 80), t0=t0)
        try_update(h_bs)

    # Phase 4: Another round of SLSQP polish after binary search
    for T in [0.0005, 0.0002, 0.00008]:
        if budget_s - elapsed() < 35:
            break
        res = opt.minimize(soft_max_val, h.copy(), args=(T,),
                         method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': sum_con}],
                         options={'maxiter': 4000, 'ftol': 1e-17})
        if hasattr(res, 'x') and res.x is not None:
            try_update(res.x)

    # Phase 5: Structured perturbation + re-optimize
    perturb_scales = [0.04, 0.08, 0.12, 0.18]
    for ps in perturb_scales:
        if budget_s - elapsed() < 100:
            break
        # Smooth structured perturbation (not white noise)
        freq = np.random.randint(2, 8)
        phase = np.random.uniform(0, 2 * np.pi)
        x_grid = np.arange(n) * dx
        pert = ps * np.sin(2 * np.pi * freq * x_grid / 2.0 + phase)
        h_pert = np.clip(h + pert, 0.0, 1.0)
        h_pert = fix_sum(h_pert, n)
        res = opt.minimize(soft_max_val, h_pert, args=(0.001,),
                         method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': sum_con}],
                         options={'maxiter': 2000, 'ftol': 1e-16})
        if hasattr(res, 'x') and res.x is not None:
            try_update(res.x)

    # Phase 6: Second binary local search round
    if budget_s - elapsed() > 100:
        h_bs2, _ = binary_local_search(h.copy(), n, max_rounds=60,
                                       time_limit=min(150, budget_s - elapsed() - 60), t0=t0)
        try_update(h_bs2)

    # Phase 7: Final ultra-fine polish
    if budget_s - elapsed() > 30:
        res = opt.minimize(soft_max_val, h.copy(), args=(0.00005,),
                         method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': sum_con}],
                         options={'maxiter': 5000, 'ftol': 1e-17})
        if hasattr(res, 'x') and res.x is not None:
            try_update(res.x)

    # Final micro-perturbation
    if budget_s - elapsed() > 25:
        h_micro = h + np.random.randn(n) * 0.003
        h_micro = fix_sum(h_micro, n)
        res = opt.minimize(soft_max_val, h_micro, args=(0.0003,),
                         method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': sum_con}],
                         options={'maxiter': 3000, 'ftol': 1e-17})
        if hasattr(res, 'x') and res.x is not None:
            try_update(res.x)

    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, n)
    c5_final = compute_c5(h)
    return (h, c5_final, n)
