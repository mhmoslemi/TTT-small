def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    log_sum = m + np.log(np.sum(np.exp(T * (corr - m))))
    val = log_sum / T * dx
    val += lam * (h.sum() - n/2.0)**2
    return val

def smooth_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w_sum = w.sum()
    w /= w_sum
    log_sum = m + np.log(w_sum)
    val = log_sum / T * dx + lam * (h.sum() - n/2.0)**2
    
    # Gradient of log-sum-exp part
    # d/dh[j] log(Σ exp(T*corr_k)) / T = (1/T) * Σ_k w_k * T * d(corr_k)/d(h[j])
    # = Σ_k w_k * d(corr_k)/d(h[j])
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(-(n-1), n):
        k = s + (n-1)
        ws = w[k]
        if ws < 1e-15:

        # d(corr_k)/d(h[j]) includes (1-h[j+s]) part and -h[j-s] part

        if j_hi > j_lo:

        j_lo2 = max(0, s)

        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= ws * h[j_lo2-s:j_hi2-s]
    grad *= dx
    
    # Gradient of penalty
    grad += 2 * lam * (h.sum() - n/2.0)
    
    return val, grad

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_obj_grad, args=(T, lam),
                      method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_opt = np.clip(res.x, 0, 1)
        return project_feasible(h_opt, n)
    except Exception:
        return h

def sa_step(h, T, rng):
    n = len(h)
    # Robin Hood move: transfer δ from i to j
    i = rng.integers(n)
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    # Maximum allowed transfer
    max_d = min(h[i], 1.0 - h[j])
    min_d = max(-h[i], -(1.0 - h[j]))
    # Scale: use T to determine typical step size
    scale = min(max_d - min_d, 0.1 * T * 10)
    if scale < 1e-10:
        return h, False
    delta = rng.uniform(min_d, max_d)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, True

def multi_step_sa(h, T, n_steps, rng, best_h, best_c5):

            current_c5 = c5_new
            if current_c5 < best_c5:

                best_h = h.copy()
    return h, best_h, best_c5

    h[jp] = s - best_t

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:

            total = room.sum()
            if total < 1e-14:

            h += room * (diff / total)

            if total < 1e-14:

        val = (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

subgrad = argmax_k c_k (unit vector in the k-space)

But in h-space, the "gradient" direction is:
For the active k*, the direction to decrease c_{k*} is -∂c_{k*}/∂h.

A simple algorithm:
1. Find k* = argmax c_k
2. Compute g = ∂c_{k*}/∂h
3. Take a step: h_new = h - α * g (with projection to [0,1] and sum constraint)
4. Repeat

This is subgradient descent on the max. It's simpler but slower to converge than coordinate descent.

    """Paired coordinate descent."""
    n = len(h)
    current_c5 = compute_c5(h)
    indices = list(range(n))

        rng.shuffle(indices)
        improved = False

            j = indices[i]
            jp = indices[(i+1) % n]  # partner
