For all other lags, cor[p] as a function of h[i] is linear: cor[p] = A_p + D_p * h[i].

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def cd_optimize(h, N, max_passes=500, top_k=30, deadline=None):
    """Coordinate descent optimization."""
    target = N / 2.0
    h = project_sum(h.copy(), target)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = float(np.max(corr)) * (2.0 / N)
    best_h = h.copy()
    
    for pas in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        # Refresh active set
        active = np.argsort(corr)[-top_k:]
        active = active[np.argsort(corr[active])[::-1]]
        
        improved = False
        for i in range(N):
            old = h[i]
            # Compute D and A for active lags
            D = np.zeros(top_k)
            A = np.zeros(top_k)
            for j, p in enumerate(active):
                j2 = i + p - N + 1
                j1 = i - p + N - 1
                if p == N - 1:
                    # Skip zero lag for linear optimization
                    D[j] = 0.0
                    A[j] = corr[N-1]  # constant (approx)
                else:
                    d = 0.0
                    if 0 <= j2 < N:
                        d += (1.0 - h[j2])
                    if 0 <= j1 < N:
                        d -= h[j1]
                    D[j] = d
                    A[j] = corr[p] - old * d
            
            # Find best t
            cur_max = np.max(corr)
            best_t = old
            best_val = cur_max
            
            # Try boundary and kink points
            cands = [0.0, 1.0]
            for a in range(min(4, top_k)):
                for b in range(a+1, min(4, top_k)):
                    denom = D[a] - D[b]
                    if abs(denom) > 1e-8:
                        tk = (A[b] - A[a]) / denom
                        if 0.0 < tk < 1.0:
                            cands.append(tk)
            
            for t in cands:
                if abs(t - old) < 1e-8:
                    continue
                val = np.max(A + D * t)
                if val < best_val:
                    best_val = val
                    best_t = t
            
            if best_t != old:
                delta = best_t - old
                h[i] = best_t
                # Update corr
                for j, p in enumerate(active):
                    if p == N - 1:
                        corr[p] += best_t*(1-best_t) - old*(1-old)
                    else:
                        corr[p] += delta * D[j]
                # Also update corr for non-active lags (they also change)
                for p in range(2*N-1):
                    if p in active:  # already updated
                        continue
                    j2 = i + p - N + 1
                    j1 = i - p + N - 1
                    dc = 0.0
                    if 0 <= j2 < N:
                        dc += delta * (1.0 - h[j2])
                    if 0 <= j1 < N:
                        dc -= delta * h[j1]
                    corr[p] += dc
                improved = True
        
        # Project to sum constraint
        h = project_sum(h, target)
        if improved:
            # Recompute corr exactly after projection
            corr = np.correlate(h, 1.0 - h, mode='full')
        
        c5 = float(np.max(corr)) * (2.0 / N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

def adam_opt(h, N, deadline, tau0=1.0, tau1=200.0, lr0=0.02):
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        tau = tau0 * (tau1 / tau0) ** min(it / 1000.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(2 * N - 1)
            w[N - 1] = 1.0
        else:
            w /= ws
        grad = np.zeros(N)
        center = N - 1
        active_idx = np.where(w > 0.001)[0]
        for ai in active_idx:
            L = ai - center
            wi = w[ai]
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wi * (1.0 - h[:N - L])
                grad[:N - L] -= wi * h[L:]
            else:
                M = -L
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= 0.97
        if lr < 1e-10:
            break
    return best_h, best_c5
