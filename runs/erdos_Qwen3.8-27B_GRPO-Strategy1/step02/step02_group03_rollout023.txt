```python
def exact_coord_min(h, i):
    n = len(h)
    h[i] = 0.0
    base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    unit = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 0.0  # reset
    
    # Center: c[n-1](v) = base[n-1] + v - v²
    # Others: c[m](v) = base[m] + (unit[m]-base[m])*v
    
    A = unit - base  # linear coefficients
    # But for center, we need to adjust: the quadratic term
    # c[n-1](v) = base[n-1] + v - v² (not base[n-1] + A[n-1]*v)
    # Note: A[n-1] = unit[n-1] - base[n-1] should be 0 (since center(1)=center(0))
    
    # Critical points:
    # 1. v = 0, v = 1
    # 2. Intersections of linear functions: base[m] + A[m]*v = base[m'] + A[m']*v
    #    v = (base[m'] - base[m]) / (A[m] - A[m']) for A[m] ≠ A[m']
    # 3. Intersections of quadratic with linear: base[n-1] + v - v² = base[m] + A[m]*v
    #    v² - (1-A[m])*v + (base[m] - base[n-1]) = 0
    
    vs = [0.0, 1.0]
    
    for m in range(2*n-1):
        if m == n-1:
            continue
        # Intersection of line m with each other line
        for m2 in range(m+1, 2*n-1):
            if m2 == n-1:
                continue
            dA = A[m] - A[m2]
            if abs(dA) > 1e-12:
                v = (base[m2] - base[m]) / dA
                if 0.0 < v < 1.0:
                    vs.append(v)
        # Intersection of line m with quadratic
        # v - v² + base[n-1] = A[m]*v + base[m]
        # -v² + (1-A[m])*v + (base[n-1] - base[m]) = 0
        # v² - (1-A[m])*v - (base[n-1] - base[m]) = 0
        a_c = 1.0
        b_c = -(1.0 - A[m])
        c_c = -(base[n-1] - base[m])
        disc = b_c*b_c - 4*a_c*c_c
        if disc > 0:
            sq = np.sqrt(disc)
            v1 = (-b_c + sq) / (2*a_c)
            v2 = (-b_c - sq) / (2*a_c)
            for v in (v1, v2):
                if 0.0 < v < 1.0:
                    vs.append(v)
    
    # Also: vertex of quadratic: v = 0.5
    vs.append(0.5)
    
    # Evaluate max at all critical points
    best_v = h[i]  # current value as default
    best_c = np.inf
    for v in vs:
        v = max(0.0, min(1.0, v))
        # Compute max correlation
        c = base + A * v  # linear approximation for all
        c[n-1] = base[n-1] + v - v*v  # correct the center
        cmax = np.max(c)
        if cmax < best_c:
            best_c = cmax
            best_v = v
    
    h[i] = best_v
    return best_v
```

import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    n = _sp['n']
    return float(np.sum(x[:n]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0, 1, out=h)
    return h

def exact_coord(h, i):
    n = len(h)
    nc = n - 1
    old = h[i]
    h[i] = 0.0
    base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    unit = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    A = unit - base
    vs = [0.0, 1.0, old]
    for m in range(2 * n - 1):
        if m == nc:
            continue
        b_c = 1.0 - A[m]
        c_c = base[nc] - base[m]
        disc = b_c * b_c - 4.0 * c_c
        if disc > 0:
            sq = np.sqrt(disc)
            v1 = (b_c + sq) * 0.5
            v2 = (b_c - sq) * 0.5
            if 0.0 < v1 < 1.0:
                vs.append(v1)
            if 0.0 < v2 < 1.0:
                vs.append(v2)
    for m in range(2 * n - 1):
        if m == nc:
            continue
        bm = base[m]
        Am = A[m]
        for m2 in range(m + 1, 2 * n - 1):
            if m2 == nc:
                continue
            dA = Am - A[m2]
            if abs(dA) > 1e-14:
                v = (base[m2] - bm) / dA
                if 0.0 < v < 1.0:
                    vs.append(v)
    best_v = old
    best_c = 1e18

def coord_descend(h, dx, ts, max_sweeps=500, dtl=None, rng=None):
    n = len(h)
    bc = compute_c5(h, dx)
    for sw in range(max_sweeps):

        improved = False

        for i in order:

            h = proj(h, ts)
            c = compute_c5(h, dx)
            if c < bc - 1e-15:
                bc = c
                improved = True
        if not improved:
            break
    return h, bc

def slsqp_opt(h0, dx, ts, mit=150):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])

    try:

                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-13})
        if r.x is not None:

            if compute_c5(hr, dx) <= t0 + 0.01:

    except Exception:

    return h0.copy()
