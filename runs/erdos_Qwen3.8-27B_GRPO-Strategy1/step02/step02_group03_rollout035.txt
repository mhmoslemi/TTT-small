    dx = 2.0 / n
    cr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = cr.max()
    if beta * (cr.max() - m) > 700:  # overflow guard

    if h_init is not None:
        n = len(h_init)
        ts = n / 2.0
        h_init = proj(h_init, ts)
    else:
        n, ts = 50, 25.0
        h_init = proj(np.full(n, 0.5), ts)

    bh, bc = h_init.copy(), c5_val(h_init)

    # Phase 1: Main optimization
    if time.time() < dl - 200:
        hr, cr = multi_opt(bh, ts, min(dl - 400, t0 + 250), rng)
        if cr < bc:
            bh, bc = hr, cr

    # Phase 2: Try nearby n values
    n_base = len(bh)
    for dn in (-2, -1, 1, 2, 3, -3, 4, -4):
        nt = n_base + dn
        if nt < 30 or nt > 70 or time.time() > dl - 150:
            continue
        ts_t = nt / 2.0
        xp = np.linspace(0, 1, nt, endpoint=False)
        xr = np.linspace(0, 1, n_base, endpoint=False)
        hi = proj(np.interp(xp, xr, bh), ts_t)
        hr, cr = multi_opt(hi, ts_t, min(dl - 80, t0 + 130), rng)
        if cr < bc:
            bh, bc, n_base = hr, cr, nt

    # Phase 3: Restart with perturbations
    if time.time() < dl - 120:
        n = len(bh)
        ts = n / 2.0
        for amp in (0.15, 0.08, 0.04):
            if time.time() > dl - 80:
                break
            hp = proj(bh + amp * rng.randn(n), ts)
            hr, cr = multi_opt(hp, ts, min(dl - 50, t0 + 100), rng)
            if cr < bc:
                bh, bc = hr, cr

    # Phase 4: Binary-like structure search
    if time.time() < dl - 80:
        n = len(bh)
        ts = n / 2.0
        hb0 = np.where(bh > 0.5, 1.0, 0.0)
        hb0 = proj(hb0, ts)
        hr, cr = flip_opt(hb0, ts, min(dl - 40, t0 + 80), rng)
        if cr < bc:
            bh, bc = hr, cr
        hr, cr = slsqp_opt(hr, ts, 10, 100), c5_val(slsqp_opt(hr, ts, 10, 100))
        hr2 = slsqp_opt(hr, ts, 10, 150)
        cr2 = c5_val(hr2)
        if cr2 < bc:
            bh, bc = hr2, cr2

    # Phase 5: Aggressive coordinate descent + final SLSQP
    if bh is not None and time.time() < dl - 5:
        n = len(bh)
        ts = n / 2.0
        hr, cr = coord_opt(bh, ts, dl - 10, rng)
        if cr < bc:
            bh, bc = hr, cr
        hr2 = slsqp_opt(bh, ts, 15, 200)
        cr2 = c5_val(hr2)
        if cr2 < bc:
            bh, bc = hr2, cr2

    bh = proj(bh, len(bh) / 2.0)
    bc = c5_val(bh)
    return bh, bc, len(bh)
