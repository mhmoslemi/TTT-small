    if time.time() - t0 < budget_s - 280:

        c5 = compute_c5(h)

    # Phase 4: More SLSQP refinement
    for T, it in [(0.0001, 1500), (0.00003, 2000), (0.00001, 2500)]:
        if time.time() - t0 > budget_s - 200:
            break
        h = slsqp_lse(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 5: Coordinate and pair polishing
    if time.time() - t0 < budget_s - 150:
        h = coord_polish(h, n, 2)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 120:
        h = pair_polish(h, n, rng, 500)
        c5 = compute_c5(h)

    # Phase 6: Diverse restarts (fewer, more targeted)
    for i in range(12):
        if time.time() - t0 > budget_s - 100:
            break
        ht = h.copy()
        mode = i % 3
        if mode == 0:
            ht += rng.randn(n) * 0.015
        elif mode == 1:
            idx = rng.choice(n, 8, replace=False)
            ht[idx] += rng.randn(8) * 0.1
        else:
            ht += rng.randn(n) * 0.008
        ht = fix_sum(ht, n)
        for T in (0.005, 0.001, 0.0003):
            ht = slsqp_lse(ht, n, T, 500)
        ht = proj_grad_descent(ht, n, 3, 500, 0.004)
        ht = fix_sum(ht, n)
        ct = compute_c5(ht)
        if ct < c5:
            h, c5 = ht, ct
            if time.time() - t0 < budget_s - 60:
                h = coord_polish(h, n, 1)
                h = pair_polish(h, n, rng, 200)
                c5 = compute_c5(h)

    # Phase 7: Up to higher resolution if time permits
    if time.time() - t0 < budget_s - 60:
        n2 = 560
        h2 = upsample(h, n2)
        for T, it in [(0.005, 600), (0.001, 800), (0.0003, 1000)]:
            if time.time() - t0 > budget_s - 20:
                break
            h2 = slsqp_lse(h2, n2, T, it)
            h2 = fix_sum(h2, n2)
        h2 = proj_grad_descent(h2, n2, 3, 500, 0.003)
        h2 = fix_sum(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < c5:
            h, n, c5 = h2, n2, c5_2

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
