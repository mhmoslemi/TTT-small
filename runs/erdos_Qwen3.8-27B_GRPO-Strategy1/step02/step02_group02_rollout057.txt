```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_sweep(h, n, max_iter=2000):
    """Coordinate descent: for each i, find optimal h[i] given others fixed."""
    ncorr = 2 * n - 1
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > 5:  # safety
            break
        improved = False
        for i in range(n):
            h_old = h[i]
            # Set h[i] = 0 and h[i] = 1 to get A and B
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            A = c0
            B = c1 - c0
            h[i] = h_old
            
            # Minimize max_k(A[k] + B[k]*x) for x in [0,1]
            # Ternary search (convex piecewise linear)
            lo, hi = 0.0, 1.0
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(A + B * m1)
                f2 = np.max(A + B * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            h_new = (lo + hi) / 2.0
            if abs(h_new - h_old) > 1e-10:
                improved = True
            h[i] = h_new
        h = proj(h, n)
        if not improved:
            break
    return h

def softmax_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def softmax_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    e = np.exp(s)
    w = e / np.sum(e)  # softmax weights
    # gradient of corr wrt h
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if w[k_idx] < 1e-10:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            # corr at shift k: sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            idx_h = np.arange(k, n)
            idx_1h = np.arange(0, n - k)
            # d/dh[j] for j in idx_h: (1-h[j-k])
            grad[idx_h] += w[k_idx] * (1 - h[idx_h - k])
            # d/dh[j] for j in idx_1h: -h[j+k]
            grad[idx_1h] += w[k_idx] * (-h[idx_1h + k])
        elif k < 0:
            p = -k
            # corr at shift -p: sum_{i=p}^{n-1} h[i]*(1-h[i-p])
            idx_h = np.arange(p, n)
            idx_1h = np.arange(0, n - p)
            grad[idx_h] += w[k_idx] * (1 - h[idx_h - p])
            grad[idx_1h] += w[k_idx] * (-h[idx_1h + p])
    return dx * grad
```

Wait, I can optimize: instead of calling np.correlate twice per coordinate, I can compute the "base" correlation once and then update it incrementally.

If h changes only at position i (from old_val to new_val, δ = new_val - old_val):
- h[i] changes by δ
- (1-h)[i] changes by -δ

The correlation corr[k] = sum_n h[n+k_idx]*(1-h[n]) (in some index convention) changes by:
For each pair (n, n+k) where one of them is i:
- If h[i] is the "h" factor: change = δ * (1-h[partner])
- If h[i] is the "(1-h)" factor: change = -δ * h[partner]

For each i, there are at most 2n-2 partners (one for each nonzero lag). So the update is O(n) per coordinate, not O(n²)!

```python
def coord_sweep_fast(h, n, max_iter=1000):
    ncorr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for it in range(max_iter):
        changed = False
        for i in range(n):
            old = h[i]
            # Compute A and B using the current corr
            # corr[k] = A_k + B_k * h[i]
            # When h[i] = 0: corr_0[k] = corr[k] - B_k * old
            # When h[i] = 1: corr_1[k] = corr[k] + B_k * (1 - old)
            
            # B_k = coefficient of h[i] in corr[k]
            # This is (1-h[partner]) for lags where h[i] is the h-factor
            # and -h[partner] for lags where h[i] is the 1-h factor
            
            # For the full correlation indexed 0 to 2n-2:
            # corr[m] for m = n-1 corresponds to shift 0
            # corr[m] for m > n-1 corresponds to positive shifts
            # corr[m] for m < n-1 corresponds to negative shifts
            
            # For shift s = m - (n-1):
            # If s >= 0: corr = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            #   h[j] appears as h[i+s] when i+s=j (i=j-s): valid if 0<=j-s<=n-1-s, i.e., s<=j<=n-1
            #     contribution: (1-h[j-s])
            #   h[j] appears as h[i] (in 1-h[i]) when i=j: valid if 0<=j<=n-1-s
            #     contribution: -h[j+s]
            
            # If s < 0, let p = -s:
            #   corr = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
            #   h[j] appears as h[i] when i=j: valid if p<=j<=n-1
            #     contribution: (1-h[j-p])
            #   h[j] appears as h[i-p] when i-p=j (i=j+p): valid if p<=j+p<=n-1, i.e., 0<=j<=n-1-p
            #     contribution: -h[j+p]
            
            # So for coordinate i:
            B = np.zeros(ncorr)
            for m in range(ncorr):
                s = m - (n - 1)
                if s >= 0:
                    b = 0.0
                    if s <= i <= n - 1:
                        b += (1 - h[i - s])  # h[i] is h-factor, partner is i-s
                    if 0 <= i <= n - 1 - s:
                        b -= h[i + s]  # h[i] is (1-h) factor, partner is i+s
                    B[m] = b
                else:
                    p = -s
                    b = 0.0
                    if p <= i <= n - 1:
                        b += (1 - h[i - p])  # h[i] is h-factor, partner is i-p
                    if 0 <= i <= n - 1 - p:
                        b -= h[i + p]  # h[i] is (1-h) factor, partner is i+p
                    B[m] = b
            
            # A = corr - B * old
            # We want to minimize max_k (corr[k] - B[k]*old + B[k]*x) = max_k (corr[k] + B[k]*(x-old))
            # Let x' = x - old, then minimize max_k (corr[k] + B[k]*x') for x' in [-old, 1-old]
            
            # Ternary search
            lo, hi = -old, 1.0 - old
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(corr + B * m1)
                f2 = np.max(corr + B * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_new = (lo + hi) / 2.0 + old
            
            if abs(x_new - old) > 1e-10:
                changed = True
                # Update corr incrementally
                delta = x_new - old
                corr += B * delta
                h[i] = x_new
            # else: h[i] unchanged, B*0 = 0, no update needed
        
        h = proj(h, n)
        # Re-sync corr after projection (projection changes multiple values)
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        if not changed:
            break
    return h
```
