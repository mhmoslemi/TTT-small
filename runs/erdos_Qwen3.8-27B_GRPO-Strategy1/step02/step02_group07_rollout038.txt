import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-14:
        return c * dx
    s = dx * T
    return (c + T * np.log(np.sum(np.exp((corr - c) / s)))) * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-14:
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)
        g = np.zeros(n)
        lo, hi = max(0, lag), min(n-1, n-1+lag)
        if lo <= hi:
            g[lo:hi+1] += 1.0 - h[lo-lag:hi-lag+1]
        lo, hi = max(0, -lag), min(n-1, n-1-lag)
        if lo <= hi:
            g[lo:hi+1] -= h[lo+lag:hi+lag+1]
        return dx * g
    s = dx * T
    p = np.exp((corr - c) / s)
    p /= p.sum()
    g = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)
    for k in range(2*n-1):
        w = p[k]
        if w < 1e-16:
            continue
        lag = lags[k]
        lo, hi = max(0, lag), min(n-1, n-1+lag)
        if lo <= hi:
            g[lo:hi+1] += w * (1.0 - h[lo-lag:hi-lag+1])
        lo, hi = max(0, -lag), min(n-1, n-1-lag)
        if lo <= hi:
            g[lo:hi+1] -= w * h[lo+lag:hi+lag+1]
    return dx * g

def sum_eq(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = m.sum()
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-11:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def slsqp(h, n, T, iters):
    c0 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    try:
        r = optimize.minimize(
            lse_obj, h.copy(), args=(T,), jac=lse_grad,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-14})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def coord2(h, n, n_pass, rng):
    for _ in range(n_pass):
        order = rng.permutation(n)
        for j in order:
            if h[j] <= 5e-4 or h[j] >= 1.0 - 5e-4:
                continue
            l = rng.randint(n)
            while l == j or h[l] <= 5e-4 or h[l] >= 1.0 - 5e-4:
                l = rng.randint(n)
            best_c5 = compute_c5(h)
            best_d = 0.0
            for d in (-0.03, -0.01, -0.003, 0.003, 0.01, 0.03):
                hj, hl = h[j] + d, h[l] - d
                if hj < 1e-4 or hj > 1.0 - 1e-4 or hl < 1e-4 or hl > 1.0 - 1e-4:
                    continue
                hn = h.copy()
                hn[j], hn[l] = hj, hl
                c = compute_c5(hn)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_d = d
            if best_d != 0.0:
                h[j] += best_d
                h[l] -= best_d
    return fix_sum(h, n)

def smooth_perturb(h, n, rng, scale):
    x = np.arange(n)
    f = np.fft.rfft(h - 0.5)
    freqs = np.fft.rfftfreq(n)
    mask = 1.0 - np.exp(-2.0 * freqs * n * 0.3)
    f[1:] *= mask
    base = np.fft.irfft(f, n) + 0.5
    ht = base + rng.randn(n) * scale
    return fix_sum(ht, n)

def block_perturb(h, n, rng):
    ht = h.copy()
    n_blocks = rng.randint(2, 6)
    for _ in range(n_blocks):
        bstart = rng.randint(0, n - 20)
        blen = rng.randint(10, min(50, n - bstart))
        shift = rng.randint(-15, 16)
        seg = ht[bstart:bstart+blen].copy()
        ht[bstart:bstart+blen] = np.roll(seg, shift * rng.choice([-1, 1]))
    return fix_sum(ht, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)
