    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_topk(h, t, topk, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -topk)[-topk:]
    c = corr[idx]
    mx = np.max(c)
    s = np.clip(t * (c - mx), -50, 50)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def obj_smooth(h, t, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = np.clip(t * (corr - mx), -50, 50)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def constraint_fun(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def interp_resample(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def binary_local_search(h, n, rng, time_limit=60):
    t_start = time.time()
    h_bin = (h > 0.5).astype(float)
    h_bin = proj(h_bin, n)
    dx = 2.0 / n
    best_max = np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full'))
    stall_count = 0
    while time.time() - t_start < time_limit:
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        improved = False
        n_trials = 250
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_max = np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full'))
            if new_max < best_max - 1e-12:
                best_max = new_max
                improved = True
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            stall_count += 1
            nf = rng.randint(2, min(7, len(ones), len(zeros)))
            fo = rng.choice(ones, size=nf, replace=False)
            fz = rng.choice(zeros, size=nf, replace=False)
            h_bin[fo] = 0.0
            h_bin[fz] = 1.0
            new_max = np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full'))
            if new_max < best_max - 1e-12:
                best_max = new_max
                improved = True
                stall_count = 0
            else:
                h_bin[fo] = 1.0
                h_bin[fz] = 0.0
            if stall_count > 5:
                break
        else:
            stall_count = 0
    return h_bin, float(best_max * dx)

def slsqp_round(h, n, t_val, topk, maxiter, ftol, bnds, cons):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        r = minimize(obj_topk, h, args=(t_val, topk, n),
                    method='SLSQP', bounds=bnds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': ftol})
    else:
        r = minimize(obj_smooth, h, args=(t_val, n),
                    method='SLSQP', bounds=bnds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = proj(np.array(initial_h_values, dtype=float), 280)
    except Exception:
        h_init = proj(np.full(280, 0.5) + rng.randn(280) * 0.05, 280)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    n = 280
    bnds = [(0.0, 1.0)] * 280
    cons = {'type': 'eq', 'fun': constraint_fun}

    # Phase 1: SLSQP at n=280 with increasing temperatures
    h280 = best_h.copy()
    temps = [25, 80, 250, 800, 3000, 12000]
    for t_val in temps:
        if budget_s - (time.time() - t0) < 35:
            break
        topk = max(4, 559 // 8)
        try:
            hc = slsqp_round(h280, 280, t_val, topk, 5000, 1e-15, bnds, cons)
            c5 = compute_c5(hc)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
            h280 = hc
        except Exception:
            pass

    # Phase 2: Binary local search
    if budget_s - (time.time() - t0) > 90:
        tl = min(100, budget_s - (time.time() - t0) - 50)
        h_bin, c5_bin = binary_local_search(best_h, 280, rng, time_limit=tl)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Refine from best (possibly binary) solution
        for t_val in [200, 1000, 5000]:
            if budget_s - (time.time() - t0) < 20:
                break
            try:
                hc = slsqp_round(best_h, 280, t_val, None, 3000, 1e-15, bnds, cons)
                c5 = compute_c5(hc)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
            except Exception:
                pass

    # Phase 3: Diverse restarts - perturb and re-optimize
    if budget_s - (time.time() - t0) > 60:
        for attempt in range(3):
            if budget_s - (time.time() - t0) < 25:
                break
            scale = 0.04 - 0.01 * attempt
            h_pert = proj(best_h + rng.randn(280) * scale, 280)
            topk = max(4, 559 // 6)
            for t_val in [50, 500, 3000]:
                if budget_s - (time.time() - t0) < 12:
                    break
                try:
                    hc = slsqp_round(h_pert, 280, t_val, topk, 2000, 1e-14, bnds, cons)
                    c5 = compute_c5(hc)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hc.copy()
                        h_pert = hc
                except Exception:
                    pass

    # Phase 4: Explore different n values
    for nd in [200, 350]:
        if budget_s - (time.time() - t0) < 50:
            break
        h_nd = interp_resample(best_h, nd)
        bnds_nd = [(0.0, 1.0)] * nd
        for t_val in [80, 500, 2500]:
            if budget_s - (time.time() - t0) < 18:
                break
            try:
                hc = slsqp_round(h_nd, nd, t_val, None, 2000, 1e-14, bnds_nd, cons)
                c5 = compute_c5(hc)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
                    n = nd
                h_nd = hc
            except Exception:
                pass
        if budget_s - (time.time() - t0) > 25:
            h_b, c5_b = binary_local_search(best_h, len(best_h), rng, time_limit=15)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 5: Final binary polish + continuous refinement
    if budget_s - (time.time() - t0) > 25:
        h_b2, c5_b2 = binary_local_search(best_h, len(best_h), rng, time_limit=min(20, budget_s - (time.time() - t0) - 10))
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 6: Ultra-fine final polish
    if budget_s - (time.time() - t0) > 12:
        n_f = len(best_h)
        bnds_f = [(0.0, 1.0)] * n_f
        try:
            hc = slsqp_round(best_h, n_f, 80000, None, 8000, 1e-17, bnds_f, cons)
            c5 = compute_c5(hc)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    final_n = len(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, final_n
