def local_refine(h, N, target, dx, deadline, seed=99):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    rng = np.random.RandomState(seed)
    while time.time() < deadline:
        improved = False
        for scale in [0.015, 0.04, 0.008]:
            nh = project_sum(best_h + rng.randn(N) * scale, target)
            c5 = compute_c5(nh, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = nh.copy()
                improved = True
        nb = max(2, N // 12)
        idx = rng.choice(N, nb, replace=False)
        nh = best_h.copy()
        for i in idx:
            nh[i] = 0.0 if nh[i] < 0.5 else 1.0
        nh = project_sum(nh, target)
        c5 = compute_c5(nh, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = nh.copy()
            improved = True
        for _ in range(4):
            i = rng.randint(N)
            nh = best_h.copy()
            nh[i] = 1.0 - best_h[i]
            nh = project_sum(nh, target)
            c5 = compute_c5(nh, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = nh.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    p1_dead = t0 + budget_s * 0.38
    h1, c1 = optimize_adam(h0.copy(), n0, target0, dx0, p1_dead, seed)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    if time.time() < deadline - 50:
        p2_dead = time.time() + (deadline - time.time()) * 0.45
        h2, c2 = optimize_momentum(h1.copy(), n0, target0, dx0, p2_dead, seed + 1)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    if time.time() < deadline - 15:
        p3_dead = time.time() + (deadline - time.time()) * 0.4
        h3, c3 = local_refine(h2.copy(), n0, target0, dx0, p3_dead, seed + 2)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()

    for dN in [20, -20, 40, -40]:
        N_alt = n0 + dN
        if N_alt % 2 != 0 or N_alt < 200 or N_alt > 800:
            continue
        if time.time() > deadline - 40:
            break
        dx_a = 2.0 / N_alt
        tgt_a = N_alt / 2.0
        xN = np.linspace(0.0, 2.0, N_alt, endpoint=False)
        h_a = project_sum(np.interp(xN, x0, best_h), tgt_a)
        pa_dead = time.time() + (deadline - time.time()) * 0.35
        h_opt, c_opt = optimize_adam(h_a.copy(), N_alt, tgt_a, dx_a, pa_dead, seed + abs(dN))
        if time.time() < deadline - 5:
            h_opt, c_opt = local_refine(h_opt, N_alt, tgt_a, dx_a, deadline - 3, seed + abs(dN) + 1)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    Nb = len(best_h)
    if time.time() < deadline - 2:
        hf, cf = local_refine(best_h.copy(), Nb, Nb / 2.0, 2.0 / Nb, deadline - 1, seed + 500)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
