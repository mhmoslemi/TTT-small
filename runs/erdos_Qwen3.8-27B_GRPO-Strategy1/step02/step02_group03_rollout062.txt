    h = np.array(h, dtype=float).copy()

_P = {}

def _obj(x):
    n = _P['n']
    h = x[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sm = _P['sm']
    return (m + np.log(np.sum(np.exp(sm * (corr - m)))) / sm) * _P['dx']

def _eq(x):
    return float(np.sum(x[:_P['n']]) - _P['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):

        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

import numpy as np
from scipy.optimize import minimize
import time

_P = {}

def _obj(x):
    n = _P['n']
    h = x[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sm = _P['sm']
    return (m + np.log(np.sum(np.exp(sm * (corr - m)))) / sm) * _P['dx']

def _eq(x):
    return float(np.sum(x[:_P['n']]) - _P['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        np.clip(h, 0.0, 1.0, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    d = ts - h.sum()
    if abs(d) > 1e-10:
        idx = int(np.argmin(1.0 - h)) if d > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + d, 0.0, 1.0))
    return h

def slsqp_opt(h0, dx, ts, deadline=None):
    n = len(h0)
    _P['n'], _P['ts'], _P['dx'] = n, ts, dx
    best = h0.copy()
    best_c = compute_c5(h0, dx)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq}]
    for sm in (5.0, 15.0, 40.0, 100.0, 300.0):
        if deadline and time.time() > deadline - 3:
            break
        _P['sm'] = sm
        try:
            r = minimize(_obj, best, method='SLSQP', bounds=bnds,
                        constraints=cons, options={'maxiter': 250, 'ftol': 1e-15})
            if r.x is not None:
                hr = proj(r.x[:n], ts)
                hc = compute_c5(hr, dx)
                if hc < best_c:
                    best, best_c = hr, hc
        except Exception:
            pass
    return best, best_c

def grad_of_c5(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(beta * (corr - m))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    g = np.zeros(n)
    for j in range(n):
        for k in range(-(n - 1), n):
            idx = n - 1 + k
            if max(0, -k) <= j <= min(n - 1, n - 1 - k):
                g[j] += w[idx] * (1.0 - h[j + k])
            if k >= 0:
                if k <= j:
                    g[j] -= w[idx] * h[j - k]
            else:
                if j <= n - 1 + k:
                    g[j] -= w[idx] * h[j - k]
    return g * dx

def grad_descent(h0, dx, ts, niter, deadline):
    n = len(h0)
    h = h0.copy()
    best_h, best_c = h0.copy(), compute_c5(h0, dx)

    for i in range(niter):

        beta = 10.0 + 60.0 * min(1.0, i / max(1, int(0.25 * niter)))
        g = grad_of_c5(h, dx, beta)
        ok = False
        for sh in (1.0, 0.5, 0.25, 0.1, 0.05):
            h2 = proj(h - lr * sh * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < best_c - 1e-15:
                h, best_h, best_c = h2, h2.copy(), c2
                no_imp = 0
                ok = True
                break
        if not ok:
            no_imp += 1
            lr *= 0.92
            if no_imp > 200:
                break
        if i > 0 and i % 400 == 0:
            lr = max(lr, 0.04)
    return best_h, best_c

def make_starts(n, ts, rng, ref):
    out = [np.full(n, ts / n)]
    for _ in range(3):
        h = np.zeros(n)
        p = rng.permutation(n)
        h[p[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[p[int(ts)]] = fr
        out.append(h.copy())
    x = np.arange(n) / n
    for m in (1, 2, 3):
        out.append(0.5 + 0.35 * np.sin(2 * np.pi * m * x))
    p = max(2, n // 4)
    out.append(np.array([1.0 if (i % (2 * p)) < p else 0.0 for i in range(n)]))
    out.append(rng.uniform(0, 1, n))
    if ref is not None:
        if len(ref) == n:
            out.insert(0, ref.copy())
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, len(ref), endpoint=False), ref)
            out.insert(0, hi)
    return [proj(s, ts) for s in out]
