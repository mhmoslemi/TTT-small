            h_cur[i] = 0.0

                       np.where(valid_t2[i], h_cur[idx_t2_c[i]], 0.0))
            res = minimize_scalar(_obj_1d_lambda(corr_base, slope_i, T),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            new_val = float(res.x)
            if abs(new_val - old_val) > 1e-7:
                h_cur[i] = new_val
                improved = True
        h_cur = project_feasible(h_cur, n)
        if not improved:
            break
    return h_cur

def _obj_1d_lambda(corr_base, slope, T):
    def f(t):
        corr_t = corr_base + slope * t
        m = np.max(corr_t)
        return m + np.log(np.sum(np.exp(T * (corr_t - m)))) / T
    return f

def proj_grad_step(h, T, masks, n_steps=100):
    n = len(h)
    dx = 2.0 / n
    idx_t1_c, idx_t2_c, valid_t1, valid_t2 = masks
    h_cur = h.copy()
    f_cur = soft_max_val(np.correlate(h_cur, 1.0 - h_cur, mode='full'), T) * dx
    for s in range(n_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        m = np.max(corr)
        w = np.exp(T * (corr - m))
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            wk = w[k]
            for j in range(n):
                ia = j + k - n + 1
                if 0 <= ia < n:
                    grad[j] += wk * (1.0 - h_cur[ia])
                ib = j - k + n - 1
                if 0 <= ib < n:
                    grad[j] -= wk * h_cur[ib]
        grad *= dx
        alpha = 0.005
        found = False
        for bt in range(25):
            h_new = project_feasible(h_cur - alpha * grad, n)
            f_new = soft_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), T) * dx
            if f_new < f_cur - 1e-6 * alpha * np.dot(grad, h_cur - h_new):
                found = True
                break
            alpha *= 0.5
        if not found:
            break
        h_cur = h_new
        f_cur = f_new
    return h_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(90)
        best_h[:45] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    masks = build_slope_masks(best_n)

    # Phase 1: Coordinate descent warmup
    for T in [100, 500, 2000]:
        if elapsed() > budget_s * 0.08:
            break
        h_new = coord_descent(best_h, T, masks, n_rounds=8)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Basin hopping with coordinate descent
    n_cur = len(best_h)
    niter2 = int(budget_s * 0.35 / 4.5)
    for i in range(max(20, niter2)):
        if elapsed() > budget_s * 0.45:
            break
        scale = 0.03 + 0.03 * np.exp(-0.06 * i)
        h_pert = best_h + np.random.randn(n_cur) * scale
        h_pert = project_feasible(h_pert, n_cur)
        masks_cur = build_slope_masks(n_cur) if True else masks
        h_opt = coord_descent(h_pert, 1000, masks_cur, n_rounds=6)
        h_opt = coord_descent(h_opt, 5000, masks_cur, n_rounds=4)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            masks = build_slope_masks(n_cur)

    # Phase 3: Projection gradient refinement
    if elapsed() < budget_s * 0.55:
        for T in [3000, 10000]:
            if elapsed() > budget_s * 0.55:
                break
            h_new = proj_grad_step(best_h, T, masks, n_steps=80)
            h_new = coord_descent(h_new, 10000, masks, n_rounds=4)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Resolution exploration
    if elapsed() < budget_s * 0.75:
        for n_try in [80, 90, 100, 110, 120]:
            if elapsed() > budget_s * 0.75:
                break
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_res = np.interp(x_new, x_old, best_h)
            h_res = project_feasible(h_res, n_try)
            masks_res = build_slope_masks(n_try)
            h_opt = coord_descent(h_res, 2000, masks_res, n_rounds=6)
            h_opt = coord_descent(h_opt, 8000, masks_res, n_rounds=4)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try
                masks = masks_res

    # Phase 5: Intensive fine basin hopping
    n_cur = len(best_h)
    n_fine = max(15, int(budget_s * 0.20 / 4.0))
    for i in range(n_fine):
        if elapsed() > budget_s * 0.93:
            break
        scale = 0.008 + 0.012 * np.random.random()
        h_pert = best_h + np.random.randn(n_cur) * scale
        h_pert = project_feasible(h_pert, n_cur)
        h_opt = coord_descent(h_pert, 20000, masks, n_rounds=5)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Final polish
    if elapsed() < budget_s * 0.99:
        h_final = coord_descent(best_h, 50000, masks, n_rounds=6)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        h_fg = proj_grad_step(h_final, 20000, masks, n_steps=60)
        h_fg = coord_descent(h_fg, 50000, masks, n_rounds=4)
        c5_fg = compute_c5(h_fg)
        if c5_fg < best_c5:
            best_c5 = c5_fg
            best_h = h_fg

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
