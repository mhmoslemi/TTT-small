def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    w = np.exp((corr - c_max) / T)
    w /= np.sum(w)
    
    # w has indices 0..2n-2, where w[n-1+j] corresponds to shift j
    # term1[s] = sum_j w[n-1+j] * h[s+j] for valid indices
    # term2[s] = sum_j w[n-1+j] * h[s-j] for valid indices
    
    # Use FFT for cross-correlation
    size = 2 * n
    w_padded = np.zeros(size)
    # Place w_full: for j in [-(n-1), n-1], w_full[j] = w[n-1+j]
    # In circular array of size `size`, index for shift j is (j % size)
    # w_full[j] at position: for j >= 0: position j; for j < 0: position j + size
    # But we want size >= 2(n-1)+1 = 2n-1, so size = 2n is fine
    for j in range(-(n-1), n):
        idx = j % size
        w_padded[idx] = w[n-1+j]
    
    h_padded = np.zeros(size)
    h_padded[:n] = h

    # = ifft(conj(fft(w_padded)) * fft(h_padded))
    W = np.fft.fft(w_padded)
    H = np.fft.fft(h_padded)

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def lse_obj_grad(h, T, lam=0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    w = np.exp((corr - c_max) / T)
    w_sum = np.sum(w)
    obj = (c_max + T * np.log(w_sum)) * dx
    
    if lam > 0:
        s = np.sum(h) - 0.5 * n
        obj += lam * s * s
    
    # Gradient
    W = w / w_sum  # normalized weights, length 2n-1
    size = 2 * n
    w_full = np.zeros(size)
    for j in range(-(n-1), n):
        w_full[j % size] = W[n-1+j]
    
    h_pad = np.zeros(size)
    h_pad[:n] = h
    
    Wf = np.fft.fft(w_full)
    Hf = np.fft.fft(h_pad)
    
    A = np.real(np.fft.ifft(conj(Wf) * Hf))[:n]

    Wr = np.fft.fft(w_rev)
    B = np.real(np.fft.ifft(conj(Wr) * Hf))[:n]
    
    grad = dx * (1.0 - A - B)
    
    if lam > 0:

        grad += 2 * lam * s

```python
import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def lse_obj_grad(h, T, lam=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    w = np.exp((corr - c_max) / T)
    w_sum = np.sum(w)
    obj = (c_max + T * np.log(w_sum)) * dx
    if lam > 0:
        s = np.sum(h) - 0.5 * n
        obj += lam * s * s
    
    W = w / w_sum
    size = 2 * n
    w_full = np.zeros(size)
    half = n - 1
    for j in range(-half, n):
        w_full[j % size] = W[n-1+j]
    
    h_pad = np.zeros(size)
    h_pad[:n] = h
    Wf = np.fft.fft(w_full)
    Hf = np.fft.fft(h_pad)
    A = np.real(np.fft.ifft(conj(Hf) * Wf))[:n]
    
    w_rev = np.zeros(size)
    w_rev[0] = w_full[0]
    if size > 2:
        w_rev[1:] = w_full[size-1:0:-1]
    Wr = np.fft.fft(w_rev)
    B = np.real(np.fft.ifft(conj(Hf) * Wr))[:n]
    
    grad = dx * (1.0 - A - B)
    if lam > 0:
        s = np.sum(h) - 0.5 * n
        grad += 2.0 * lam * s
    return obj, grad

def lbfqp(h, n, T, lam, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        o, _ = lse_obj_grad(x, T, lam)
        return o
    def fprime(x):
        _, g = lse_obj_grad(x, T, lam)
        return g
    try:
        res = optimize.minimize(fun, h, jac=fprime, method='L-BFGS-B',
                                bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return res.x
    except Exception:
        return h

def upsample(h, nn):
    n = len(h)
    x0 = np.arange(n, dtype=float) * (2.0 / n)
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0) if False else np.interp(x1 % 2.0, x0, h), nn)
```

def lse_obj_grad_direct(h, T, lam=0.0):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')  # length 2n-1
    c_max = np.max(corr)
    w = np.exp((corr - c_max) / T)
    w_sum = np.sum(w)
    W = w / w_sum  # length 2n-1
    
    obj = (c_max + T * np.log(w_sum)) * dx
    if lam > 0:
        s = np.sum(h) - 0.5 * n
        obj += lam * s * s
    
    # For each i, grad[i] = dx * sum_s W[n-1+s] * [(1-h[i+s])*valid(i+s) - h[i-s]*valid(i-s)]
    # where valid(k) = 1 if 0<=k<n
    
    grad = np.zeros(n)
    for i in range(n):
        total = 0.0
        for s in range(-(n-1), n):
            Ws = W[n-1+s]
            if Ws < 1e-15:
                continue
            # First term: (1-h[i+s]) if 0<=i+s<n
            ip = i + s
            im = i - s
