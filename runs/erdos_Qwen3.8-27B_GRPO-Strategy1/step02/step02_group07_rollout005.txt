import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    k = np.sum(mask)
    if k > 0:
        h[mask] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-10:
        idxs = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        for i in idxs:
            if abs(d2) < 1e-12:
                break
            step = np.sign(d2) * min(abs(d2), abs(1.0 - h[i]) if d2 < 0 else abs(h[i]))
            h[i] += step
            d2 -= step
    return h

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def lse_obj(h, T, topk=0):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk > 0 and topk < len(corr):
        xc = np.sort(corr)[-topk:]
        return float(np.mean(xc) * dx)
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def slsqp_step(h, n, T, iters, topk=0):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(
            lambda x: lse_obj(x, T, topk), h,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-18})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def coord_search(h, n, n_pass=4):
    for _ in range(n_pass):
        improved = False
        for j in range(n):
            if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5:
                continue
            c0 = compute_c5(h)
            best_c = c0
            best_v = h[j]
            vals = np.array([h[j] - 0.15, h[j] - 0.08, h[j] - 0.03,
                             h[j] + 0.03, h[j] + 0.08, h[j] + 0.15,
                             0.05, 0.5, 0.95])
            vals = vals[(vals > 1e-4) & (vals < 1.0 - 1e-4) & (np.abs(vals - h[j]) > 1e-6)]
            for v in vals:
                delta = v - h[j]
                hn = h.copy()
                hn[j] = v
                others = np.where(np.arange(n) != j)[0]
                hn[others] = h[others] - delta / (n - 1)
                hn = np.clip(hn, 0.0, 1.0)
                hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
                    delta = v - h[j]
                    hn = h.copy()
                    hn[j] = v
                    others = np.where(np.arange(n) != j)[0]
                    hn[others] = h[others] - delta / (n - 1)
                    hn = np.clip(hn, 0.0, 1.0)
                    hn = fix_sum(hn, n)
            if best_c < c0 - 1e-14:
                delta = best_v - h[j]
                h[j] = best_v
                others = np.where(np.arange(n) != j)[0]
                h[others] = h[others] - delta / (n - 1)
                h = np.clip(h, 0.0, 1.0)
                h = fix_sum(h, n)
                improved = True
        if not improved:
            break
    return fix_sum(h, n)

def pair_search(h, n, rng_obj, n_pairs=800):
    for _ in range(n_pairs):
        j, l = rng_obj.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
            continue
        if h[l] <= 1e-4 or h[l] >= 1.0 - 1e-4:
            continue
        c0 = compute_c5(h)
        best_c = c0
        best_d = 0.0
        for d in (-0.06, -0.03, -0.01, 0.01, 0.03, 0.06):
            hj2, hl2 = h[j] + d, h[l] - d
            if hj2 < 1e-4 or hj2 > 1.0 - 1e-4 or hl2 < 1e-4 or hl2 > 1.0 - 1e-4:
                continue
            hn = h.copy()
            hn[j], hn[l] = hj2, hl2
            c = compute_c5(hn)
            if c < best_c - 1e-14:
                best_c = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return fix_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def downsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)
