        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h

    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:

        h = h + room * (diff / total)

    return h.sum() - len(h) / 2.0

    # Phase 5: Different n values
    for n_try in [80, 100, 140, 160]:
        if elapsed() > budget_s * 0.85:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        for trial in range(3):
            if elapsed() > budget_s * 0.85:
                break
            h_interp = np.interp(x_new, x_old, best_h)
            h0 = h_interp + np.random.uniform(-0.05, 0.05, n_try)
            h0 = project_feasible(h0, n_try)
            for T in [100, 500, 2000]:
                if elapsed() > budget_s * 0.85:
                    break
                hc = slsqp_run(h0, smooth_c5, (T,), n_try, maxiter=600, ftol=1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc

    # Phase 6: Intensive local search on best solution
    n_cur = len(best_h)
    if elapsed() < budget_s * 0.94:
        n_local = 200
        for it in range(n_local):
            if elapsed() > budget_s * 0.94:
                break
            # Pick 1-3 random coordinates and try small adjustments
            n_adj = np.random.randint(1, 4)
            coords = np.random.choice(n_cur, size=n_adj, replace=False)
            improved = False
            for j in coords:
                for delta in [0.02, 0.04, 0.06, -0.02, -0.04, -0.06, 0.01, -0.01]:
                    h_try = best_h.copy()
                    h_try[j] = np.clip(h_try[j] + delta, 0, 1)
                    h_try = project_feasible(h_try, n_cur)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if not improved and it > 100:
                break

    # Phase 7: Final high-T polish
    n_cur = len(best_h)
    for T in [5000, 15000, 50000]:
        if elapsed() > budget_s * 0.99:
            break
        hc = slsqp_run(best_h, smooth_c5, (T,), n_cur, maxiter=1500, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        # Also try focused
        hc = slsqp_run(best_h, focused_c5, (T, 5), n_cur, maxiter=800, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
