import scipy.optimize as opt
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def proj_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def sum_eq_fun(x, n):
    return np.sum(x) - n * 0.5


def sum_eq_jac(x, n):
    return np.ones(n)


def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-15:
        return cmax * dx
    return (cmax + T * np.log(np.sum(np.exp((corr - cmax) / T)))) * dx


def topk_obj(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    kk = min(k, len(corr))
    idx = np.argpartition(corr, -kk)[-kk:]
    top = corr[idx]
    cmax = np.max(top)
    if T < 1e-15:
        return cmax * dx
    return (cmax + T * np.log(np.sum(np.exp((top - cmax) / T)))) * dx

    idx = np.argpartition(corr, -kk)[-kk:]
    top = corr[idx]
    cmax = np.max(top)

        base = cmax * dx
    else:
        base = (cmax + T * np.log(np.sum(np.exp((top - cmax) / T)))) * dx
    s = np.sum(h)
    target = n * 0.5
    return base + pen * (s - target) ** 2


def slsqp_run(h0, n, func, args, max_it):
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(func, h0, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_it, 'ftol': 1e-15, 'iprint': 0})
    return proj_feasible(res.x, n)


def lgbg_run(h0, n, func, args, max_it):
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(func, h0, args=args, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_it, 'ftol': 1e-14, 'iprint': 0})
    return proj_feasible(res.x, n)


def upsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x0 = np.arange(n) * (2.0 / n)
    x1 = np.arange(new_n) * (2.0 / new_n)
    hn = np.interp(x1, x0, h)
    return proj_feasible(hn, new_n)

    x = np.arange(n) * (2.0 / n)
