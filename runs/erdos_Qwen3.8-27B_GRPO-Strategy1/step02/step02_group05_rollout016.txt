def compute_c5(h):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def topk_subgrad(h, N, K, sharp):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(corr.max()) * (2.0 / N)
    if K >= corr.size:

    else:
        idx = np.argpartition(-corr, K - 1)[:K]
    tv = corr[idx].astype(float)
    tv -= tv.min()
    w = np.exp(sharp * tv)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(tv) / len(tv)

        w /= ws
    center = N - 1
    grad = np.zeros(N)

            grad += wj * (1.0 - 2.0 * h)

            grad[:N - M] += wj * (1.0 - h[M:])
