    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        wk = p[k]
        if wk < 1e-16:
            continue
        m = k - (n - 1)
        if m >= 0:
            A[:n - m] += wk * (1.0 - h[m:])
            B[m:] -= wk * h[:n - m]
        else:
            A[-m:] += wk * (1.0 - h[:n + m])
            B[:n + m] -= wk * h[-m:]
    grad = dx * (1.0 - A - B)
    return obj, grad

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def slsqp_grad(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj_grad, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-14})
        hn = fix_sum(np.clip(r.x, 0, 1), n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def lbfgs_pen_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    p = np.exp((corr - cmax) / T)
    p /= p.sum()
    obj = dx * (cmax + T * np.log(np.sum(p))) + lam * (np.sum(h) - 0.5 * n) ** 2
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        wk = p[k]
        if wk < 1e-16:
            continue
        m = k - (n - 1)
        if m >= 0:
            A[:n - m] += wk * (1.0 - h[m:])
            B[m:] -= wk * h[:n - m]
        else:
            A[-m:] += wk * (1.0 - h[:n + m])
            B[:n + m] -= wk * h[-m:]
    grad = dx * (1.0 - A - B) + 2 * lam * (np.sum(h) - 0.5 * n)
    return obj, grad

def lbfgs_pen(h, T, lam, iters):
    n = len(h)
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lbfgs_pen_obj, h, args=(T, lam), jac=True,
            method='L-BFGS-B', bounds=[(0., 1.)] * n,
            options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-9})
        hn = fix_sum(np.clip(r.x, 0, 1), n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def coord_polish(h, n, n_pass=3):
    for _ in range(n_pass):
        improved = False
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            bv = h[j]
            bc = compute_c5(h)
            for d in (-0.04, -0.015, -0.004, 0.004, 0.015, 0.04):
                hv = bv + d
                if hv < 1e-5 or hv > 1.0 - 1e-5:
                    continue
                h[j] = hv
                c = compute_c5(h)
                h[j] = bv
                if c < bc - 1e-13:
                    bc = c
                    h[j] = hv
                    bv = hv
                    improved = True
        h = fix_sum(h, n)
        if not improved:
            break
    return h

def pair_polish(h, n, rng_obj, n_pairs=400):
    for _ in range(n_pairs):
        j, l = rng_obj.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
            continue
        if h[l] <= 1e-4 or h[l] >= 1.0 - 1e-4:
            continue
        bc = compute_c5(h)
        best_d = 0.0
        for d in (-0.03, -0.01, 0.01, 0.03):
            hj, hl = h[j] + d, h[l] - d
            if hj < 1e-4 or hj > 1.0 - 1e-4 or hl < 1e-4 or hl > 1.0 - 1e-4:
                continue
            h[j], h[l] = hj, hl
            c = compute_c5(h)
            h[j], h[l] = h[j] - d, h[l] + d
            if c < bc - 1e-13:
                bc = c
                best_d = d
                h[j], h[l] = hj, hl
        h = fix_sum(h, n)
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 7)
    t0 = time.time()
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    for T, it in [(0.06, 250), (0.025, 350), (0.01, 450), (0.004, 550),
                  (0.0015, 700), (0.0005, 900), (0.0002, 1100)]:
        if time.time() - t0 > budget_s - 420:
            break
        h = slsqp_grad(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    if time.time() - t0 < budget_s - 320:
        for T in (0.003, 0.001, 0.0004, 0.00015):
            if time.time() - t0 > budget_s - 300:
                break
            for lam in (500, 5000, 50000):
                h = lbfgs_pen(h, T, lam, 800)
                h = fix_sum(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    if time.time() - t0 < budget_s - 200:
        h = coord_polish(best_h, n, 3)
        h = fix_sum(h, n)
        h = pair_polish(h, n, rng, 500)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    n_restarts = 35
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 90:
            break
        ht = best_h.copy()
        mode = i % 5
        if mode == 0:
            ht += np.random.randn(n) * 0.012
        elif mode == 1:
            idx = rng.choice(n, 12, replace=False)
            ht[idx] += np.random.randn(12) * 0.09
        elif mode == 2:
            ht += np.random.randn(n) * 0.006
        elif mode == 3:
            s = rng.randint(0, n - 25)
            L = rng.randint(8, 25)
            ht[s:s+L] = 1.0 - ht[s:s+L]
        else:
            smooth = np.random.randn(n)
            for _ in range(3):
                smooth = np.convolve(smooth, [0.25, 0.5, 0.25], mode='same')
            ht += smooth * 0.02
        ht = fix_sum(ht, n)
        for T in (0.004, 0.001, 0.0003):
            ht = slsqp_grad(ht, n, T, 350)
            ht = fix_sum(ht, n)
        ht = coord_polish(ht, n, 2)
        ht = pair_polish(ht, n, rng, 200)
        ht = fix_sum(ht, n)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht.copy()
            h = ht.copy()

    h = best_h
    n = len(h)
    for T, it in [(0.00008, 1500), (0.00003, 2000), (0.00001, 2500)]:
        if time.time() - t0 > budget_s - 15:
            break
        h = slsqp_grad(h, n, T, it)
        h = fix_sum(h, n)
    h = coord_polish(h, n, 2)
    h = fix_sum(h, n)

    if time.time() - t0 < budget_s - 60:
        n2 = 400
        h2 = upsample(h, n2)
        for T, it in [(0.005, 300), (0.001, 500), (0.0003, 700)]:
            if time.time() - t0 > budget_s - 10:
                break
            h2 = slsqp_grad(h2, n2, T, it)
            h2 = fix_sum(h2, n2)
        if compute_c5(h2) < best_c5:
            h = h2

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
