    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    cmax = corr.max()

    exps = np.exp((corr - cmax) / T)

    B = np.correlate(h, w, 'full')[:n]

    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    mask = (h > 1e-8) & (h < 1.0 - 1e-8)

def slsqp_opt(h, n, T, iters):
    def obj(h_):
        return lse_obj(h_, T)
    def obj_grad(h_):
        return lse_grad(h_, T) * (2.0/n)
    def con(h_):
        return np.sum(h_) - 0.5 * n
    c0 = compute_c5(h)
    try:

            bounds=[(0., 1.)]*n,
            constraints=[{'type':'eq', 'fun':con}],
            options={'maxiter': iters, 'ftol': 1e-18})

            rn = fix_sum(r.x, n)
            if compute_c5(rn) < c0:

    except:

    return h
