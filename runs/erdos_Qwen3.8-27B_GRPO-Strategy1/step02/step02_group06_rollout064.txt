import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    t = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        m = (lo + hi) / 2
        if np.sum(np.clip(h + m, 0, 1)) > t:
            hi = m
        else:
            lo = m
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += d / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def sig(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -400, 400)))

def h2x(h):
    return np.log(np.clip(h, 1e-4, 1 - 1e-4) / np.clip(1 - h, 1e-4, 1 - 1e-4))

def smx(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    m = c.max()
    return (m + T * np.log(np.sum(np.exp((c - m) / T)))) * dx

def slsqp_h(h0, n, T, mi, extra=None):
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n * 0.5, 'jac': lambda h: np.ones(n)}]
    if extra:
        cons.extend(extra)
    r = opt.minimize(smx, h0, args=(T,), method='SLSQP',
                     bounds=[(0, 1)] * n, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-18, 'iprint': 0})
    return proj(r.x, n)

def slsqp_x(x0, n, T, mi):
    def fx(x):
        return smx(sig(x), T)
    cons = [{'type': 'eq', 'fun': lambda x: sig(x).sum() - n * 0.5}]

                     bounds=[(-20, 20)] * n, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-18, 'iprint': 0})
    return r.x

    n = len(h)
    corr = np.correlate(h, 1.0 - h, 'full')
    iv = corr * (2.0 / n)
    return np.argsort(iv)[::-1][:k]

def lag_cons(n, lags, ct):
    cs = []
    for li in lags:
        cs.append({'type': 'ineq',
                    'fun': lambda h, li=li: ct - np.correlate(h, 1.0 - h, 'full')[li] * (2.0 / len(h))})
    return cs
