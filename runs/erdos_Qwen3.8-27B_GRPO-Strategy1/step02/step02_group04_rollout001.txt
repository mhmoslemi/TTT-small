import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.35:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                sp = 1.0 - h[i]
                add = min(diff, sp)
                h[i] += add
                diff -= add
                if diff <= 1e-13:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-13:
                    break
    return h

def grad_shifts_vectorized(h, m_list):
    n = len(h)
    g = np.zeros(n)
    for m in m_list:
        if m == 0:
            g[:] += (1.0 - 2.0 * h)
        elif m > 0:
            g[m:] += (1.0 - h[:-m])
            g[:n-m] += (-h[m:])
        else:
            mm = -m
            g[:n-mm] += (1.0 - h[mm:])
            g[mm:] += (-h[:n-mm])
    return g

def proj_grad_descent(h, n, topk, max_iter, eta0, decay, momentum_beta):
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for it in range(max_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[-topk:]
        top_m = [int(idx - n + 1) for idx in top_idx]
        top_frac = corr[top_idx] / corr[top_idx]
        top_frac = np.exp(corr[top_idx] / (0.001 * n))
        top_frac /= top_frac.sum()
        g = np.zeros(n)
        for w, m in zip(top_frac, top_m):
            if m == 0:
                g += w * (1.0 - 2.0 * h_cur)
            elif m > 0:
                g[m:] += w * (1.0 - h_cur[:-m])
                g[:n-m] += w * (-h_cur[m:])
            else:
                mm = -m
                g[:n-mm] += w * (1.0 - h_cur[mm:])
                g[mm:] += w * (-h_cur[:n-mm])
        eta = eta0 / (1.0 + decay * it)
        v = momentum_beta * v + (1.0 - momentum_beta) * g
        h_new = h_cur - eta * v
        h_new = fix_sum(h_new, n)
        h_new_c5 = compute_c5(h_new)
        if h_new_c5 < c5:
            h_cur = h_new
        else:
            v *= 0.5
            h_cur = h_new
    return best_h, best_c5

def soft_max_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def do_slsqp(h, n, T, max_iter):
    try:
        res = opt.minimize(
            soft_max_obj, h.copy(), args=(T, n),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-16}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def active_shift_slsqp(h, n, T, k_active, max_iter):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k_active:]
    top_m = [int(idx - n + 1) for idx in top_idx]
    def topk_obj(x):
        g = np.zeros(n)
        for m in top_m:
            if m == 0:
                g += (1.0 - 2.0 * x)
            elif m > 0:
                g[m:] += (1.0 - x[:-m])
                g[:n-m] += (-x[m:])
            else:
                mm = -m
                g[:n-mm] += (1.0 - x[mm:])
                g[mm:] += (-x[:n-mm])
        c_top = []
        for m in top_m:
            if m == 0:
                c_top.append(float(np.sum(x * (1.0 - x))))
            elif m > 0:
                c_top.append(float(np.sum(x[m:] * (1.0 - x[:n-m]))))
            else:
                mm = -m
                c_top.append(float(np.sum(x[:n-mm] * (1.0 - x[mm:]))))
        c_arr = np.array(c_top)
        cm = np.max(c_arr)
        if T < 1e-14:
            return cm * (2.0 / n)
        sh = np.clip(c_arr - cm, -60.0 * T, 0.0)
        s = np.sum(np.exp(sh / T))
        return (cm + T * np.log(max(s, 1e-300))) * (2.0 / n)
    res = opt.minimize(
        topk_obj, h.copy(),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
        options={'maxiter': max_iter, 'ftol': 1e-15}
    )
    if hasattr(res, 'x') and res.x is not None:
        return fix_sum(res.x, n)
    return h
