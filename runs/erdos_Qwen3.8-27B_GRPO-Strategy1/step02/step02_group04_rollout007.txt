    h = h.copy()
    c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    improved = True
    swaps = 0
    while improved and swaps < max_swaps:
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        trials = min(len(ones)*len(zeros), 200)
        for _ in range(trials):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-12:
                c5 = c5_new
                improved = True
                ones[ones == i] = j
                zeros[zeros == j] = i
                break
            else:
                h[i], h[j] = h[j], h[i]
            swaps += 1
    return h, c5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)
