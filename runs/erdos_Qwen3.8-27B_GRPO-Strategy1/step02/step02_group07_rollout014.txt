def project_sum(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    target = 0.5 * n
    # Binary search for tau
    lo, hi = -1.0, 2.0
    for _ in range(60):
        tau = (lo + hi) / 2
        h_proj = np.clip(h - tau, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

        if time.time() - t_start > budget_s - 250:
            break
        h, c5 = pg_opt(h, n, T, iters, lr, tlim)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Peak shaving
    if time.time() - t_start < budget_s - 120:
        h = peak_shave(best_h, n, 10, 15, 0.006)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Coordinate refinement
    if time.time() - t_start < budget_s - 100:
        h = coord_refine(best_h, n, 2, 20)
        h = project(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Random restarts
    n_restart = 0
    while time.time() - t_start < budget_s - 80 and n_restart < 40:
        mode = n_restart % 6
        ht = best_h.copy()
        if mode == 0:
            ht += rng.randn(n) * 0.025
        elif mode == 1:
            idx = rng.choice(n, 15, replace=False)
            ht[idx] += rng.randn(15) * 0.12
        elif mode == 2:
            ht += rng.randn(n) * 0.01
        elif mode == 3:
            i1 = rng.randint(0, n // 2)
            i2 = rng.randint(n // 2, n)
            amt = rng.uniform(0.05, 0.2)
            ht[i1] = np.clip(ht[i1] + amt, 0, 1)
            ht[i2] = np.clip(ht[i2] - amt, 0, 1)
        elif mode == 4:
            kernel = np.ones(7) / 7.0
            ht += 0.04 * np.convolve(rng.randn(n), kernel, mode='same')
        else:
            perm = rng.permutation(n)
            ht[perm[:10]] += rng.uniform(-0.1, 0.1, 10)
            ht[perm[-10:]] -= rng.uniform(-0.1, 0.1, 10)
        ht = project(ht, n)
        ht, ct = pg_opt(ht, n, 0.01, 2000, 0.2, 15)
        ht, ct = pg_opt(ht, n, 0.002, 2000, 0.1, 15)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht.copy()
            if time.time() - t_start < budget_s - 40:
                best_h = peak_shave(best_h, n, 6, 8, 0.005)
                best_h = project(best_h, n)
                best_c5 = compute_c5(best_h)
                best_h = coord_refine(best_h, n, 1, 10)
                best_c5 = compute_c5(project(best_h, n))
        n_restart += 1

    # Phase 5: Try different resolutions
    for n_try in (140, 200, 360, 480):
        if time.time() - t_start > budget_s - 25:
            break
        if n_try == n:
            continue
        h_try = upsample(best_h, n_try)
        h_try, c5_try = pg_opt(h_try, n_try, 0.02, 3000, 0.4, 25)
        h_try, c5_try = pg_opt(h_try, n_try, 0.003, 3000, 0.15, 25)
        h_try = peak_shave(h_try, n_try, 6, 8, 0.005)
        h_try = project(h_try, n_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            n = n_try
            if time.time() - t_start < budget_s - 20:
                best_h, best_c5 = pg_opt(best_h, n, 0.001, 4000, 0.08, 20)
                best_h = peak_shave(best_h, n, 5, 6, 0.004)
                best_h = project(best_h, n)
                best_c5 = compute_c5(best_h)

    # Phase 6: Final fine optimization
    if time.time() - t_start < budget_s - 10:
        for T in (0.0003, 0.0001, 3e-5):
            if time.time() - t_start > budget_s - 5:
                break
            best_h, c5 = pg_opt(best_h, n, T, 6000, 0.04, 20)
            if c5 < best_c5:
                best_c5 = c5

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
