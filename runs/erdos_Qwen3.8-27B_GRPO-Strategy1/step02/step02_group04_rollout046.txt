import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def autocorr_linear(h):
    n = len(h)
    full = np.correlate(h, h, mode='full')
    return full[n - 1:]

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def neg_smooth_min_ac(h, T):
    a = autocorr_linear(h)
    m = float(np.min(a))
    if T < 1e-15:
        return -m
    shifted = -(a - m) / T
    shifted = np.clip(shifted, -50.0, 0.0)
    s = np.sum(np.exp(shifted))
    return float(T * np.log(s) - m)

def sum_con(x, n):
    return np.sum(x) - n * 0.5

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.4:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12:
                    break
    return h

def do_slsqp_c5(h, n, T, max_iter):
    try:
        res = opt.minimize(
            lambda x: soft_max_obj(x, T), h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: sum_con(x, n)}],
            options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def do_slsqp_ac(h, n, T, max_iter):
    try:
        res = opt.minimize(
            lambda x: neg_smooth_min_ac(x, T), h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: sum_con(x, n)}],
            options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def do_lbbfgs_ac(h, n, T, max_iter):
    lam = 50.0
    def obj(x):
        xc = np.clip(x, 0, 1)
        return neg_smooth_min_ac(xc, T) + lam * (np.sum(xc) - n * 0.5) ** 2
    try:
        res = opt.minimize(obj, h.copy(), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def active_lag_refine(h, n, T, max_iter, top_k=8):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        tv = c[top_idx]
        tm = np.max(tv)
        sh = np.clip(tv - tm, -80 * T, 0) / T
        return float((tm + T * np.log(np.sum(np.exp(sh)))) )
    try:
        res = opt.minimize(obj, h.copy(), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x: sum_con(x, n)}],
                           options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def pg_subgradient(h, n, n_iter, lr):
    hc = h.copy().astype(float)
    for _ in range(n_iter):
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        k_max = int(np.argmax(corr))
        lag = k_max - (n - 1)
        grad = np.ones(n)
        if lag >= 0:
            grad[:n - lag] -= hc[lag:]
            grad[lag:] -= hc[:n - lag]
        else:
            pad = -lag
            grad[pad:] -= hc[:n - pad]
            grad[:n - pad] -= hc[pad:]
        hc -= lr * grad
        hc = np.clip(hc, 0.0, 1.0)
        hc = fix_sum(hc, n)
    return hc

def upsample(h, new_n):
    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    return fix_sum(np.interp(xn, xo, h, period=2.0), new_n)

def downsample(h, new_n):
    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    return fix_sum(np.interp(xn, xo, h, period=2.0), new_n)
