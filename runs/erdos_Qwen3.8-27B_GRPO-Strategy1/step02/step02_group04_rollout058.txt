    def try_update(h_cand, n_cand):
        nonlocal h_best, n_best, c5_best
        h_cand = project_feasible(h_cand, n_cand)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best - 1e-15:
            h_best = h_cand
            n_best = n_cand
            c5_best = c5_cand
            return True
        return False

    # Phase 1: Subgradient descent (fast, many iterations)
    for lr in [0.008, 0.004, 0.002, 0.001, 0.0005]:
        if budget_s - elapsed() < 30:
            break
        h_sg = subgradient_descent(h_best, n_best, n_iter=4000, lr_init=lr, lr_decay=0.9997)
        try_update(h_sg, n_best)

    # Phase 2: Block coordinate descent (different search landscape)
    if budget_s - elapsed() > 120:
        h_bd = block_descend(h_best, n_best, block_size=10, n_sweeps=6, seed=seed)
        try_update(h_bd, n_best)
    if budget_s - elapsed() > 80:
        h_bd2 = block_descend(h_best, n_best, block_size=20, n_sweeps=4, seed=seed+1)
        try_update(h_bd2, n_best)

    # Phase 3: Low-resolution SLSQP (faster per iteration)
    if budget_s - elapsed() > 200:
        n_low = 120
        h_low = upsample_to(h_best, n_low)
        for T in [0.02, 0.005, 0.001, 0.0003]:
            if budget_s - elapsed() < 60:
                break
            try:
                res = opt.minimize(soft_max_obj, h_low.copy(), args=(T,),
                                   method='SLSQP', bounds=[(0.0, 1.0)] * n_low,
                                   constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n_low,)}],
                                   options={'maxiter': 3000, 'ftol': 1e-15})
                if hasattr(res, 'x') and res.x is not None:
                    h_low = project_feasible(res.x, n_low)
            except Exception:
                pass
        h_up = upsample_to(h_low, n_best)
        try_update(h_up, n_best)
        # Refine at high res after upsample
        if budget_s - elapsed() > 60:
            h_ref = subgradient_descent(h_best, n_best, 3000, 0.003, 0.9997)
            try_update(h_ref, n_best)

    # Phase 4: More subgradient rounds
    for lr in [0.001, 0.0005, 0.0002]:
        if budget_s - elapsed() < 40:
            break
        h_sg = subgradient_descent(h_best, n_best, n_iter=5000, lr_init=lr, lr_decay=0.9998)
        try_update(h_sg, n_best)

    # Phase 5: SLSQP polish at full resolution
    for T in [0.005, 0.001, 0.0003, 0.0001]:
        if budget_s - elapsed() < 50:
            break
        try:
            res = opt.minimize(soft_max_obj, h_best.copy(), args=(T,),
                               method='SLSQP', bounds=[(0.0, 1.0)] * n_best,
                               constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n_best,)}],
                               options={'maxiter': 2500, 'ftol': 1e-16})
            if hasattr(res, 'x') and res.x is not None:
                h_try = project_feasible(res.x, n_best)
                if compute_c5(h_try) < c5_best - 1e-15:
                    h_best = h_try
                    c5_best = compute_c5(h_best)
        except Exception:
            pass

    # Phase 6: Multi-start with structured perturbations
    for trial in range(4):
        if budget_s - elapsed() < 100:
            break
        # Structured perturbation: random phase shift in low-freq components
        h_pert = h_best.copy()
        x = np.arange(n_best) * (2.0 / n_best)
        phase = np.random.uniform(0, 2 * np.pi)
        amp = np.random.uniform(0.02, 0.06)
        for k in [1, 2, 3]:
            h_pert += amp / k * np.sin(2 * np.pi * k * x / 2.0 + phase * k)
        h_pert = project_feasible(h_pert, n_best)
        h_opt = subgradient_descent(h_pert, n_best, 3000, 0.003, 0.9997)
        if budget_s - elapsed() > 40:
            try:
                T = 0.001
                res = opt.minimize(soft_max_obj, h_opt.copy(), args=(T,),
                                   method='SLSQP', bounds=[(0.0, 1.0)] * n_best,
                                   constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n_best,)}],
                                   options={'maxiter': 1500, 'ftol': 1e-15})
                if hasattr(res, 'x') and res.x is not None:
                    h_opt = project_feasible(res.x, n_best)
            except Exception:
                pass
        try_update(h_opt, n_best)

    # Phase 7: Try n=240 and n=480
    for n_try in [240, 480]:
        if budget_s - elapsed() < 120:
            break
        h_init = upsample_to(h_best, n_try)
        h_opt = subgradient_descent(h_init, n_try, 3000, 0.004, 0.9997)
        try:
            res = opt.minimize(soft_max_obj, h_opt.copy(), args=(0.001,),
                               method='SLSQP', bounds=[(0.0, 1.0)] * n_try,
                               constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n_try,)}],
                               options={'maxiter': 2000, 'ftol': 1e-15})
            if hasattr(res, 'x') and res.x is not None:
                h_opt = project_feasible(res.x, n_try)
        except Exception:
            pass
        try_update(h_opt, n_try)

    # Phase 8: Final block descent + subgradient + SLSQP
    if budget_s - elapsed() > 60:
        h_bd = block_descend(h_best, n_best, block_size=8, n_sweeps=4, seed=seed+100)
        try_update(h_bd, n_best)
    if budget_s - elapsed() > 30:
        h_sg = subgradient_descent(h_best, n_best, 4000, 0.001, 0.9998)
        try_update(h_sg, n_best)
    if budget_s - elapsed() > 30:
        try:
            res = opt.minimize(soft_max_obj, h_best.copy(), args=(5e-5,),
                               method='SLSQP', bounds=[(0.0, 1.0)] * n_best,
                               constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n_best,)}],
                               options={'maxiter': 3000, 'ftol': 1e-16})
            if hasattr(res, 'x') and res.x is not None:
                h_try = project_feasible(res.x, n_best)
                if compute_c5(h_try) < c5_best - 1e-15:
                    h_best = h_try
                    c5_best = compute_c5(h_best)
        except Exception:
            pass

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
