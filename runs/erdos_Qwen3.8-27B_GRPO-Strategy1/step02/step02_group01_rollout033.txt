            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= h * ((-diff) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_erdos(h, maxiter=300):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.01
    
    def obj(x):
        return x[-1]
    def eq(x):
        return x[:n].sum() - n / 2.0
    def ineq(x):
        hh = x[:n]
        c = np.correlate(hh, 1.0 - hh, mode='full')
        return x[-1] - c
    
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': eq}, {'type': 'ineq', 'fun': ineq}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def sa_structured(h, n, n_iter, T_init, T_end, rng):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    for it in range(n_iter):
        T = T_init * (T_end / T_init) ** (it / max(n_iter - 1, 1))
        move = rng.randint(3)
        h_try = h_cur.copy()
        if move == 0:
            # Exchange: move mass from i to j
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(-0.15, 0.15)
            h_try[i] += delta
            h_try[j] -= delta
        elif move == 1:
            # Segment shift: add constant to a segment, compensate elsewhere
            a = rng.randint(0, n - 3)
            b = rng.randint(a + 3, n)
            delta = rng.uniform(-0.05, 0.05)
            seg_len = b - a
            h_try[a:b] += delta
            comp = -delta * seg_len / (n - seg_len)
            mask = np.ones(n, dtype=bool)
            mask[a:b] = False
            h_try[mask] += comp
        else:
            # Gaussian perturbation
            h_try += rng.randn(n) * 0.03
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur or rng.random() < np.exp(-(c5_try - c5_cur) / max(T, 1e-10)):
            h_cur = h_try
            c5_cur = c5_try
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
    return best_h, best_c5

def subgrad_refine(h, n, n_iter, lr_init, lr_end):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for it in range(n_iter):
        lr = lr_init * (lr_end / lr_init) ** (it / max(n_iter - 1, 1))
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                grad[i] += (1.0 - h_cur[j])
            j2 = i - s
            if 0 <= j2 < n:
                grad[i] -= h_cur[j2]
        grad -= grad.mean()
        h_try = project_feasible(h_cur - lr * grad * dx, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5

        grad = np.zeros(n)
        if s >= 0:

            grad[:n-s] += (1.0 - h_cur[s:])
            # indirect: i=s..n-1, grad[i] -= h[i-s]
            grad[s:] -= h_cur[:n-s]

            grad[sp:] += (1.0 - h_cur[:n-sp])
            # indirect: i=0..n-1-sp, grad[i] -= h[i+sp]... 
            # Wait, for s<0, indirect: j2 = i-s = i+sp, valid if 0<=i+sp<n, i.e., i=0..n-1-sp
            grad[:n-sp] -= h_cur[sp:]
        grad -= grad.mean()

Wait, but the integration is over [0,2] for both x and x+k, so the effective domain shrinks. Let me re-examine.

The integral ∫₀² h(x)(1-h(x+k))dx - but this is only defined when x+k is also in [0,2], i.e., x ∈ [-k, 2-k] (for k>0) or x ∈ [0, 2+k] (for k<0). The actual integration is over the intersection [0,2] ∩ [-k, 2-k] for k>0, which is [0, 2-k].

So for shift k: the integral is over [0, 2-|k|] (roughly), and it's ∫ h(x)(1-h(x+k))dx over that domain.

This is exactly what np.correlate computes with the 'full' mode - it handles the boundary effects.

Actually, I can compute the change in C5 more efficiently. When I change h[i] by δ (and h[j] by -δ), the change in corr[m] is:

This makes a swap evaluation O(n) instead of O(n²), and a full sweep O(n³) which for n=90 is about 730K operations. Manageable.

            best_h = h_try.copy()
            h_cur = h_try

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 1e-15:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= h * ((-diff) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_erdos(h, maxiter=300):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.01

    def obj(x):
        return x[-1]
    def eq(x):
        return x[:n].sum() - n / 2.0
    def ineq(x):
        hh = x[:n]
        c = np.correlate(hh, 1.0 - hh, mode='full')
        return x[-1] - c

    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': eq}, {'type': 'ineq', 'fun': ineq}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def smooth_slsqp(h, T, maxiter=300):
    n = len(h)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def eq(x):
        return x.sum() - n / 2.0

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_refine(h, n, n_iter, lr_init, lr_end, k_top=1):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for it in range(n_iter):
        lr = lr_init * (lr_end / lr_init) ** (it / max(n_iter - 1, 1))
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        idx = np.argsort(corr)[-k_top:]
        c_top = corr[idx]
        w = (c_top - c_top.min() + 0.05)
        w = w / w.sum()
        grad = np.zeros(n)
        for ii in range(k_top):
            s = idx[ii] - (n - 1)
            if s >= 0:
                grad[:n-s] += w[ii] * (1.0 - h_cur[s:])
                grad[s:] -= w[ii] * h_cur[:n-s]
            else:
                sp = -s
                grad[sp:] += w[ii] * (1.0 - h_cur[:n-sp])
                grad[:n-sp] -= w[ii] * h_cur[sp:]
        grad -= grad.mean()
        h_try = project_feasible(h_cur - lr * grad * dx, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5

def sa_local(h, n, n_iter, T_init, T_end, rng):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    for it in range(n_iter):
        T = T_init * (T_end / T_init) ** (it / max(n_iter - 1, 1))
        move = rng.randint(4)
        h_try = h_cur.copy()
        if move == 0:
            idx = rng.choice(n, 2, replace=False)
            delta = rng.uniform(-0.12, 0.12)
            h_try[idx[0]] += delta
            h_try[idx[1]] -= delta
        elif move == 1:
            a = rng.randint(0, n - 4)
            b = rng.randint(a + 4, n)
            delta = rng.uniform(-0.04, 0.04)
            h_try[a:b] += delta
            mask = np.ones(n, dtype=bool)
            mask[a:b] = False
            h_try[mask] += -delta * (b - a) / (n - b + a)
        elif move == 2:
            i, j = rng.choice(n, 2, replace=False)
            tmp = h_try[i]
            h_try[i] = h_try[j]
            h_try[j] = tmp
        else:
            h_try += rng.randn(n) * 0.025
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur or rng.random() < np.exp(-(c5_try - c5_cur) / max(T, 1e-10)):
            h_cur = h_try
            c5_cur = c5_try
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
    return best_h, best_c5

def random_block_init(n, rng):
    h = np.zeros(n)
    positions = rng.choice(n, n // 2, replace=False)
    h[positions] = 1.0
    h += rng.randn(n) * 0.03
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    # Phase 1: Smooth SLSQP warmup then hard SLSQP

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
