    # Soft max weights
    c_max = np.max(pos_corr)
    shifted = (pos_corr - c_max) / T
    weights = np.exp(np.clip(shifted, -50, 0))
    weights /= weights.sum()
    
    # Only use top K for efficiency

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    ni = np.sum(interior)
    if ni > 0 and abs(diff) < 0.3 * ni:
        h[interior] += diff / ni
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)
        else:
            idx = np.argsort(-h)
            diff = -diff
        for i in idx:
            add = min(diff, 1.0 - h[i]) if diff > 0 else -min(diff, h[i])
            if diff > 0:
                h[i] += add
                diff -= add
            else:
                h[i] -= add
                diff -= add
            if abs(diff) <= 1e-14:
                break
    return h

def soft_max_subgrad(h, T, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use only positive shifts: indices n-1 to 2n-2
    # By symmetry, this covers all unique shifts
    pos_corr = corr[n-1:]  # length n, shifts k=0,...,n-1
    c_max = np.max(pos_corr)
    # Soft max weights
    shifted = np.clip((pos_corr - c_max) / T, -50, 0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Focus on top K
    top_k = np.argsort(pos_corr)[-K:]
    top_w = weights[top_k]
    top_w /= top_w.sum()
    
    grad = np.zeros(n)
    for ki in range(len(top_k)):
        k = top_k[ki]
        w = top_w[ki]
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            # j in [0, k): -h[j+k]
            grad[:k] += w * (-h[k:])
            # j in [k, n-1-k]: (1-h[j-k]) - h[j+k]
            m = n - 2 * k  # number of middle elements
            if m > 0:
                grad[k:n-k] += w * ((1.0 - h[0:m]) - h[2*k:n])
            # j in [n-k, n-1]: (1-h[j-k])
            grad[n-k:] += w * (1.0 - h[n-2*k:n-k])
    return grad

def subgrad_descent(h, n, T0, T_end, lr0, lr_end, iters, K=20):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    beta = 0.85
    
    for it in range(iters):
        frac = it / max(iters - 1, 1)
        T = T0 * (T_end / T0) ** frac
        lr = lr0 * (lr_end / lr0) ** frac
        
        g = soft_max_subgrad(h, T, K)
        momentum = beta * momentum - lr * g
        h_new = h + momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h = fix_sum(h_new, n)
        
        if it % 100 == 0 or it == iters - 1:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, periodic=True)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, periodic=True)
    return fix_sum(h_new, new_n)

def slsqp_refine(h, n, T, max_iter=1000):
    """SLSQP with soft-max objective as fallback"""
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(corr - c_max, -80*T, 0)
        s = np.sum(np.exp(shifted / max(T, 1e-15)))
        return (c_max + T * np.log(max(s, 1e-300))) * (2.0/n)
    
    try:
        res = opt.minimize(obj, h.copy(), method='SLSQP',
                          bounds=[(0,1)]*n,
                          constraints=[{'type':'eq','fun':lambda x: np.sum(x)-n*0.5}],
                          options={'maxiter': max_iter, 'ftol': 1e-15})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except:
        pass
    return h
