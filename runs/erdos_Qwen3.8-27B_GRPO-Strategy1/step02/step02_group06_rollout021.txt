    return (h_best, c5_final, len(h_best))

    max_corr = np.max(corr_shifted)
    exp_corr = np.exp(corr_shifted - max_corr)
    sum_exp = np.sum(exp_corr)
    log_sum_exp = max_corr + np.log(sum_exp)

    # Add penalty
    s = np.sum(h)
    penalty = mu * (s - n*0.5)**2
    obj = obj_sm + penalty

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def smooth_max_obj_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_s = corr / T
    max_cs = np.max(corr_s)
    exp_cs = np.exp(corr_s - max_cs)
    sum_exp = np.sum(exp_cs)
    log_sum_exp = max_cs + np.log(sum_exp)
    obj_sm = T * log_sum_exp * dx
    s = np.sum(h)
    penalty = mu * (s - n * 0.5) ** 2
    obj = obj_sm + penalty
    w = exp_cs / sum_exp
    grad_sm = np.zeros(n)
    for j in range(2 * n - 1):
        delta = j - (n - 1)
        if delta >= 0:
            m_lo, m_hi = delta, n - delta
        else:
            m_lo, m_hi = 0, n + delta
        if m_lo >= m_hi:
            continue
        idx1 = np.arange(m_lo, m_hi) - delta
        idx2 = np.arange(m_lo, m_hi) + delta
        dc_dm = (1.0 - h[idx1]) - h[idx2]
        grad_sm[m_lo:m_hi] += w[j] * dc_dm
    grad_sm *= dx
    grad_pen = 2.0 * mu * (s - n * 0.5)
    return obj, grad_sm + grad_pen

def run_lbgfsb(h0, n, T, mu, max_iter):
    def f(h):
        return smooth_max_obj_grad(h, T, mu)[0]
    def g(h):
        return smooth_max_obj_grad(h, T, mu)[1]
    res = opt.minimize(f, h0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
    return proj_constraint(res.x, n)

def do_slsqp_sm(h0, n, T, mu, max_iter):
    def f(h):
        return smooth_max_obj_grad(h, T, mu)[0]
    def g(h):
        return smooth_max_obj_grad(h, T, mu)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x) * 0.5,
             'jac': lambda x: np.ones(len(x))}]
    res = opt.minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return proj_constraint(res.x, n)
