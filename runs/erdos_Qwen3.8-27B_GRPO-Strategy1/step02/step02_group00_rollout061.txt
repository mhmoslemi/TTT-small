    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()

        if total < 1e-13:
            return h
        h = h + room * (diff / total)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1:
        return m * dx
    shifted = T * (corr - m)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1:
        # Just gradient of max (subgradient at argmax)
        idx = np.argmax(corr)
        k = idx - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
        else:
            ka = -k
            grad[ka:] += (1.0 - h[:n-ka])
            grad[:n-ka] -= h[ka:]
        return grad * dx
    shifted = T * (corr - m)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-12:

        k = k_idx - (n - 1)
        if k >= 0:
            if n > k:

                grad[:n-k] -= w * h[k:]
        else:
            ka = -k
            if n > ka:
                grad[ka:] += w * (1.0 - h[:n-ka])
                grad[:n-ka] -= w * h[ka:]
    return grad * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, T, n, maxiter=500, ftol=1e-13):
    def obj(x):
        return soft_c5(x, T)
    def grad(x):
        return soft_c5_grad(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_grad(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_grad}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def proj_grad_step(h, grad, n, step_size):
    """Take a projected gradient step maintaining sum constraint."""
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1:
        return m * dx
    shifted = T * (corr - m)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def soft_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1:
        idx = np.argmax(corr)
        k = idx - (n - 1)
        grad = np.zeros(n)
    else:
        shifted = T * (corr - m)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        grad = np.zeros(n)
        for ki in range(len(corr)):
            w = weights[ki]
            if w < 1e-13:
                continue
            k = ki - (n - 1)
            if k >= 0:
                if n > k:
                    grad[k:] += w * (1.0 - h[:n-k])
                    grad[:n-k] -= w * h[k:]
            else:
                ka = -k
                if n > ka:
                    grad[ka:] += w * (1.0 - h[:n-ka])
                    grad[:n-ka] -= w * h[ka:]
    return grad * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def slsqp(h, T, n, maxiter=500, ftol=1e-13):
    def obj(x):
        return soft_c5(x, T)
    def jac(x):
        return soft_c5_grad(x, T)
    def con_f(x):
        return x.sum() - n / 2.0
    def con_j(x):
        return np.ones(n)
    res = minimize(obj, h, jac=jac, method='SLSQP',
                   bounds=[(0.0,1.0)]*n,
                   constraints=[{'type':'eq','fun':con_f,'jac':con_j}],
                   options={'maxiter':maxiter,'ftol':ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def interp_h(h, n_new):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def grad_descent_loop(h, T, n, max_steps=100, lr0=0.008):
    lr = lr0
    best = h.copy()
    bc = compute_c5(best)
    for _ in range(max_steps):
        g = soft_c5_grad(best, T)
        h_new = project_feasible(best - lr * g, n)
        c = compute_c5(h_new)
        if c < bc - 1e-14:
            bc = c
            best = h_new
            lr = min(lr * 1.03, 0.03)
        else:
            lr *= 0.6
            if lr < 5e-6:
                break
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60); best_h[:30] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup with gradient
    for T in [50, 300, 1500, 8000]:
        if elapsed() > budget_s * 0.07:
            break
        try:
            hc = slsqp(best_h, T, n, 600, 1e-14)
            v = compute_c5(hc)
            if v < best_c5:
                best_c5, best_h = v, hc
        except Exception:
            pass

    # Phase 2: Gradient descent + periodic SLSQP
    if elapsed() < budget_s * 0.20:
        h_g, c_g = grad_descent_loop(best_h, 500, n, 200, 0.01)
        if c_g < best_c5:
            best_c5, best_h = c_g, h_g
        else:
            h_g, c_g = grad_descent_loop(best_h, 2000, n, 100, 0.005)
            if c_g < best_c5:

    # Phase 3: High-T SLSQP
    for T in [500, 2000, 10000, 40000]:
        if elapsed() > budget_s * 0.35:

        try:
            hc = slsqp(best_h, T, n, 800, 1e-14)
            v = compute_c5(hc)
            if v < best_c5:

        except Exception:
            pass

    # Phase 4: Basin hopping with structured moves
    bh = best_h.copy()
    bh_c5 = best_c5
    for it in range(120):

            break

        hp = bh.copy()
        dec = 1.0 - 0.6 * it / 120
        if mt == 0:

        elif mt == 1:

        elif mt == 2:

            hp[ps+bl:ps+2*bl] = bh[ps:ps+bl]
        else:
            i, j = np.random.choice(n, 2, replace=False)
            hp[i], hp[j] = bh[j], bh[i]
        hp = project_feasible(hp, n)
        try:
            hp2, _ = grad_descent_loop(hp, 600, n, 30, 0.004)
            hc = slsqp(hp2, 800, n, 150, 1e-12)
            v = compute_c5(hc)
            temp = 0.003 * dec + 0.0002
            if v < bh_c5 or np.random.rand() < np.exp(-(v-bh_c5)/temp):
                bh_c5, bh = v, hc
                if v < best_c5:
                    best_c5, best_h = v, hc
        except Exception:
            pass

    # Phase 5: Targeted coordinate search on worst lags
    if elapsed() < budget_s * 0.84:

        for _ in range(400):

                break

            ki = int(np.argmax(corr))
            lag = ki - (n - 1)
            imp = False
            if 0 < lag < n:
                contrib = best_h[lag:] * (1.0 - best_h[:n-lag])
            elif -n < lag < 0:
                contrib = best_h[:n+lag] * (1.0 - best_h[-lag:])
            else:
                contrib = best_h * (1.0 - best_h)
            if len(contrib) > 5:
                top_j = np.argsort(contrib)[::-1][:10]
            else:
                top_j = np.random.choice(n, min(10, n), replace=False)
            for j in top_j:
                for d in [0.02, 0.05, 0.09, -0.02, -0.05, -0.09]:
                    ht = best_h.copy()
                    ht[j] = np.clip(ht[j]+d, 0, 1)
                    ht = project_feasible(ht, n)
                    v = compute_c5(ht)
                    if v < best_c5 - 1e-11:
                        best_c5, best_h = v, ht
                        imp = True; break
                if imp: break
            if not imp:
                for j in np.random.choice(n, min(6, n), replace=False):
                    for d in [0.04, 0.09, -0.04, -0.09]:
                        ht = best_h.copy()
                        ht[j] = np.clip(ht[j]+d, 0, 1)
                        ht = project_feasible(ht, n)
                        v = compute_c5(ht)
                        if v < best_c5 - 1e-11:
                            best_c5, best_h = v, ht
                            imp = True; break
                    if imp: break
            stall = 0 if imp else stall + 1
            if stall > 120:
                break

    # Phase 6: Lower-res exploration
    if elapsed() < budget_s * 0.94:
        for nl in [48, 60, 80]:
            if elapsed() > budget_s * 0.93:
                break
            hl = project_feasible(interp_h(best_h, nl), nl)
            for T in [400, 2500]:
                if elapsed() > budget_s * 0.93:
                    break
                try:
                    hc = slsqp(hl, T, nl, 400, 1e-14)

        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
