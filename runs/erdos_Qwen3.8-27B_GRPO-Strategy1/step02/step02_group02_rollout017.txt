                        r = minimize(pen_obj, st, args=(t_val, 5, lam), method='L-BFGS-B',
                                     bounds=bnds280, options={'maxiter': 1500, 'ftol': 1e-18})
                        hc = proj(np.clip(r.x, 0.0, 1.0), n)
                        c5 = compute_c5(hc)
                        if c5 < bc:
                            bc = c5
                            h280 = hc
                            bh = hc
                    except Exception:
                        pass

    # Phase 7: Final targeted polish - alternate top-3 and top-8
    h280 = bh if compute_c5(bh) < compute_c5(h280) else h280
    if tleft() > 20:
        for t_val in [3000, 10000, 40000]:
            if tleft() < 8:
                break
            for topk in [3, 7]:
                if tleft() < 5:
                    break
                try:
                    r = minimize(pen_obj, h280, args=(t_val, topk, 20000.0), method='L-BFGS-B',
                                 bounds=bnds280, options={'maxiter': 3000, 'ftol': 1e-20})
                    hc = proj(np.clip(r.x, 0.0, 1.0), n)
                    c5 = compute_c5(hc)
                    if c5 < bc:
                        bc = c5
                        h280 = hc
                except Exception:
                    pass

    # Phase 8: Ultra-fine SLSQP
    if tleft() > 10:
        try:
            r = minimize(smax_obj, h280, args=(50000, 3), method='SLSQP',
                         bounds=bnds280, constraints=cons_sl,
                         options={'maxiter': 5000, 'ftol': 1e-18})
            hc = proj(np.clip(r.x, 0.0, 1.0), n)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                h280 = hc
        except Exception:
            pass

    final_h = bh if compute_c5(bh) <= compute_c5(h280) else h280
    final_h = proj(final_h, n)
    return final_h, compute_c5(final_h), n
