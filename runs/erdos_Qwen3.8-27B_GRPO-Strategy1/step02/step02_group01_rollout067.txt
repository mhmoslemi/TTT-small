def lse_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    exps = np.exp(T * (corr - m))
    w = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-12:
            continue
        s = j - (n - 1)
        wt = w[j] * dx
        # h[p-s] term: p-s in [0,n-1]
        lo = max(0, s)
        hi = min(n, n + s)
        if hi > lo:
            grad[lo:hi] += wt * (1.0 - h[lo - s:hi - s])
        # -h[p+s] term: p+s in [0,n-1]
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wt * h[lo2 + s:hi2 + s]
    return grad

        if total_room > 1e-15:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_c5(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    if T < 1e6:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    else:
        val = m
    return val * dx

def lse_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    if T < 1e6:
        exps = np.exp(T * (corr - m))
        w = exps / np.sum(exps)
    else:
        w = np.zeros(2 * n - 1, dtype=float)
        w[np.argmax(corr)] = 1.0
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-14:
            continue
        s = j - (n - 1)
        wt = w[j] * dx
        lo = max(0, s)
        hi = min(n, n + s)
        if hi > lo:
            grad[lo:hi] += wt * (1.0 - x[lo - s:hi - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wt * x[lo2 + s:hi2 + s]
    return grad

def obj_lse_pen(x, T, lam):
    n = len(x)
    v = lse_c5(x, T)
    sv = x.sum() - n / 2.0
    return v + lam * sv * sv

    gv = lse_grad(x, T)

            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += h * (diff / total)

    return h
