import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exp_t = np.exp(T * (corr - m))
    w = exp_t / exp_t.sum()
    val = (m + np.log(exp_t.sum()) / T) * dx
    s = x.sum() - n / 2.0
    val += mu * s * s
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        mk = k - (n - 1)
        # (1-x[j-mk]) for valid j
        lo1 = max(0, mk)
        hi1 = min(n, n + mk)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 - mk:hi1 - mk])
        # -x[j+mk] for valid j
        lo2 = max(0, -mk)
        hi2 = min(n, n - mk)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * x[lo2 + mk:hi2 + mk]
    grad += 2.0 * mu * s
    return val, grad

def optimize_lbfgs(h, T, mu, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h, args=(T, mu), jac=True, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n_iter=10):
    n = len(h)
    for _ in range(n_iter):
        improved = False
        for j in range(n):

                        h2[idx] = np.clip(h2[idx] + diff, 0, 1)

def compute_c5(h):

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exp_t = np.exp(T * (corr - m))
    w = exp_t / exp_t.sum()
    val = (m + np.log(exp_t.sum()) / T) * dx
    s = x.sum() - n / 2.0
    val += mu * s * s
    grad = np.zeros(n)

        wk = w[k] * dx
        if wk < 1e-16:

        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Coordinate descent polish
    if elapsed() < budget_s * 0.94:
        h_cd = coord_descent(best_h, n_iter=6)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 5: Final L-BFGS-B polish
    if elapsed() < budget_s * 0.98:
        h_final = optimize_lbfgs(best_h, 5000, 5000.0, maxiter=600)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
