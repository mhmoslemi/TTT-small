                h = project_sum(h, target)
                stale = 0

    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sgrad(h, N, tau, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    nk = len(corr)
    if top_k < nk:
        idx = np.argsort(corr)[::-1][:top_k]
    else:
        idx = np.arange(nk)
        top_k = nk
    sc = tau * corr[idx]
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    center = N - 1
    grad = np.zeros(N)
    for k in range(top_k):
        wi = w[k]
        if wi < 1e-7:
            continue
        L = int(idx[k]) - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

def optimize(h, N, target, dx, deadline, tau0, tau1, top_k0, top_k1, lr0, decay, decay_period):

        tk = max(3, int(round(top_k0 + (top_k1 - top_k0) * frac)))
        grad, c5_val = sgrad(h, N, tau, tk)
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % decay_period == 0:
            lr *= decay
            if stale > decay_period * 4:
                h += np.random.randn(N) * 0.003

        if lr < 1e-10:

    return best_h, best_c5
