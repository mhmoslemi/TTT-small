def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

        else:
            hi = mid

    corr = np.correlate(h, 1.0 - h, mode='full')
    true_max = float(corr.max())
    sc = tau * corr - tau * corr.max()
    w = np.exp(sc)
    ws = w.sum()

            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])

    return grad, true_max

    dx = 2.0 / N
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

        (1.0, 20.0, 0.04, 0.995),
        (20.0, 80.0, 0.015, 0.997),
        (80.0, 300.0, 0.005, 0.999),
        (300.0, 800.0, 0.002, 0.9995),
    ]

        if time.time() > deadline - 5:
            break
        tau0, tau1, lr, decay = phase
        n_phase_iters = 3000
        for k in range(n_phase_iters):
            if time.time() > deadline - 5:
                break
            frac = min(k / max(n_phase_iters, 1), 1.0)
            tau = tau0 * (tau1 / tau0) ** frac
            inertia = (k - 1.0) / (k + 2.0) if k > 0 else 0.0
            y = h + inertia * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
            g, c5_val = smax_grad(y, N, tau)
            lr_k = lr * (decay ** (k // 100))
            h_new = y - lr_k * g
            h_new = project_sum(h_new, target)
            h_prev = h
            h = h_new
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return best_h, best_c5

def binary_local_search(h_cont, N, deadline, rng):
    """Local search on binary solutions with simulated annealing."""
    dx = 2.0 / N
    target = N // 2
    # Round to binary
    h_bin = (h_cont >= 0.5).astype(float)
    # Adjust to have exactly target ones
    n_ones = int(h_bin.sum())
    if n_ones > target:
        # Need to turn off (n_ones - target) ones
        ones_idx = np.where(h_bin == 1)[0]
        # Choose the ones closest to 0.5 (i.e., h_cont closest to 0.5)
        priorities = h_cont[ones_idx]  # higher = closer to 1, less likely to flip
        flip_off = ones_idx[np.argsort(priorities)[:n_ones - target]]
        h_bin[flip_off] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        priorities = 1.0 - h_cont[zeros_idx]  # lower h_cont = closer to 0, less likely to flip on
        flip_on = zeros_idx[np.argsort(priorities)[:target - n_ones]]
        h_bin[flip_on] = 1
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h, dx)
    T = 0.01  # initial temperature
    T_min = 0.0001
    
    it = 0
    while time.time() < deadline:
        # Try random 2-flip
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i] = 1
        h_bin[j] = 0
        new_c5 = compute_c5(h_bin, dx)
        delta = new_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 0
            h_bin[j] = 1
        T *= 0.999
        it += 1
        if T < T_min:
            T = 0.01  # reheat
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_max = float(corr.max())
    sc = tau * (corr - corr.max())
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        w = np.full(2*N-1, 1.0/(2*N-1))
    else:
        w /= ws
    grad = np.zeros(N)
    center = N - 1
    active = np.where(w > 0.002)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, true_max

def optimize_nesterov(h_init, N, target, deadline, phases=None):
    if phases is None:
        phases = [
            (1.0, 15.0, 0.05, 0.998, 2000),
            (15.0, 60.0, 0.02, 0.9985, 3000),
            (60.0, 250.0, 0.008, 0.999, 4000),
            (250.0, 800.0, 0.003, 0.9995, 3000),
        ]
    dx = 2.0 / N
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for tau0, tau1, lr0, decay, n_iters in phases:
        if time.time() > deadline - 3:
            break
        for k in range(n_iters):
            if time.time() > deadline - 3:
                break
            frac = k / max(n_iters - 1, 1)
            tau = tau0 * (tau1 / tau0) ** frac
            inertia = (k - 1.0) / (k + 2.0) if k >= 1 else 0.0
            y = h + inertia * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
            g, c5_val = smax_grad(y, N, tau)
            lr_k = lr0 * decay ** (k // 200)
            h_new = project_sum(y - lr_k * g, target)
            h_prev = h
            h = h_new
            if k % 50 == 0:
                c5_now = compute_c5(h, dx)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
    c5_final = compute_c5(h, dx)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def binary_search(h_init, N, deadline, rng, n_restarts=3):
    dx = 2.0 / N
    target = N // 2
    best_h = None
    best_c5 = float('inf')
    
    for restart in range(n_restarts):
        if time.time() > deadline - 2:
            break
        # Create binary from h_init (or random for restarts)
        if restart == 0:
            h_bin = (h_init >= 0.5).astype(float)
        else:
            h_bin = (h_init + rng.randn(N) * 0.3 >= 0.5).astype(float)
        
        # Fix to exactly target ones
        n_ones = int(h_bin.sum())
        if n_ones > target:
            ones = np.where(h_bin == 1)[0]
            flip = ones[np.argsort(h_init[ones])[:n_ones - target]]
            h_bin[flip] = 0
        elif n_ones < target:
            zeros = np.where(h_bin == 0)[0]
            flip = zeros[np.argsort(1 - h_init[zeros])[:target - n_ones]]
            h_bin[flip] = 1
        
        c5_cur = compute_c5(h_bin, dx)
        
        # Simulated annealing
        T = 0.005
        T_floor = 0.00005
        stall = 0
        for it in range(50000):
            if time.time() > deadline - 2:
                break
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h_bin[i], h_bin[j] = 1.0, 0.0
            c5_new = compute_c5(h_bin, dx)
            delta = c5_new - c5_cur
            if delta < 0:
                c5_cur = c5_new
                stall = 0
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_bin.copy()
            else:
                if rng.random() < np.exp(-delta / T):
                    c5_cur = c5_new
                    stall += 1
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
                    stall += 1
            T = max(T * 0.9995, T_floor)
            if stall > 2000:
                T = 0.003
                stall = 0
    
    return best_h, best_c5
