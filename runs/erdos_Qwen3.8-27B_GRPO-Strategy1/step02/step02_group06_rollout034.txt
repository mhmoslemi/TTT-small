import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_pen(h, T, lam, n_target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        sm = c_max
    else:
        sm = c_max + T * np.log(np.sum(np.exp((corr - c_max) / T)))
    return sm * dx + lam * (np.sum(h) - n_target)**2

def active_soft_max(h, T, k_top, n_target):
    """Soft-max weighted toward top-k correlations"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    # Weight: emphasize top correlations
    ranks = np.argsort(corr)[::-1]
    weights = np.ones(len(corr))
    weights[ranks[:k_top]] = 5.0  # 5x weight on top-k
    shifted = corr - c_max
    w_exp = weights * np.exp(shifted / T)
    return (c_max + T * np.log(np.sum(w_exp))) * dx

def proj_sum(h, n):
    """Project to [0,1] with sum = n/2"""
    h = np.clip(h, 0, 1)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Bisection
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0, 1))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo+hi)/2, 0, 1)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.where((h > 0.01) & (h < 0.99))[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0, 1)
    return h

    n = len(h0)
    res = opt.minimize(
        soft_max_pen, h0, args=(T, lam, n*0.5),
        method='L-BFGS-B',
        bounds=[(0,1)]*n,
        options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-15}
    )
    return proj_sum(res.x, n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, assume_sorted=True)
    return proj_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    # Block average
    blocks = []
    for i in range(new_n):
        start = int(i * old_n / new_n)
        end = int((i+1) * old_n / new_n)
        end = min(end, old_n)
        if end > start:
            blocks.append(np.mean(h[start:end]))
        else:
            blocks.append(h[start % old_n])
    h_new = np.array(blocks)
    return proj_sum(h_new, new_n)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_pen_val(h, T, lam, n_target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        sm = c_max
    else:
        sm = c_max + T * np.log(np.sum(np.exp((corr - c_max) / T)))
    return sm * dx + lam * (np.sum(h) - n_target) ** 2

def soft_max_pen_grad(h, T, lam, n_target):
    n = len(h)
    m = n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    exp_c = np.exp((corr - c_max) / T)
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        # First term: t in [max(0,k-n+1), min(n,k+1))
        t1_lo = max(0, k - n + 1)
        t1_hi = min(n, k + 1)
        if t1_hi > t1_lo:
            idx = np.arange(t1_lo, t1_hi)
            grad[idx] += wk * one_minus_h[idx - k + m]
        # Second term: t in [max(0,n-1-k), min(n,2*n-1-k))
        t2_lo = max(0, n - 1 - k)
        t2_hi = min(n, 2 * n - 1 - k)
        if t2_hi > t2_lo:
            idx = np.arange(t2_lo, t2_hi)
            grad[idx] -= wk * h[idx + k - m]
    
    # Penalty gradient
    grad += 2.0 * lam * (np.sum(h) - n_target)
    return (grad, soft_max_pen_val(h, T, lam, n_target))

def sm_grad(h, T, lam, ntarget):

    m = n - 1
    dx = 2.0 / n
