import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        ni = np.sum(interior)
        if ni > 0:
            h[interior] += diff / ni
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_for_lag(h, k_full):
    n = len(h)
    d = k_full - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        grad[d:] += (1.0 - h[:n - d])
        grad[:n - d] -= h[d:]
    else:
        m = -d
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def topk_subgrad_obj(h, k):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    k_full = np.argsort(-corr)[:k]
    grad = np.zeros(n)
    for kk in k_full:
        grad += subgradient_for_lag(h, kk)
    return float(np.max(corr) * dx), grad * dx / k

def subgradient_descent(h0, n, t_start, deadline, n_iter, topk, step0, decay):
    h = h0.copy()
    c_best = compute_c5(h)
    h_best = h.copy()
    step = step0
    patience = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        c_val, grad = topk_subgrad_obj(h, topk)
        h_new = project_to_feasible(h - step * grad, n)
        c_new = compute_c5(h_new)
        if c_new < c_best:
            c_best = c_new
            h_best = h_new.copy()
            h = h_new
            patience = 0
        else:
            patience += 1
            if patience > 200:
                step *= 0.9
                patience = 0
        step *= decay
        if step < 1e-6:
            break
    return h_best, c_best

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def do_slsqp(h0, n, T, max_iter):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(soft_max_obj, h0, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return project_to_feasible(res.x, n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_to_feasible(h_new, new_n)

def coord_descent(h0, n, t_start, deadline, seed, n_iter):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    c_cur = compute_c5(h)
    c_best = c_cur
    h_best = h.copy()
    step = 0.015
    for it in range(n_iter):
        if time.time() > deadline:
            break
        n_coord = rng.randint(1, 5)
        idxs = rng.choice(n, size=n_coord, replace=False)
        h_try = h.copy()
        h_try[idxs] = np.clip(h_try[idxs] + rng.randn(n_coord) * step, 0, 1)
        h_try = project_to_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < c_cur:
            h = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h.copy()
        else:
            step *= 0.997
            if step < 0.002:
                step = 0.015
    return h_best, c_best
