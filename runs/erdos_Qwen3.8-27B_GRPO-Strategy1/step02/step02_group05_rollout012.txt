import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h - mid, 0.0, 1.0)

def softmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N)
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-3
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def subgrad_topk(h, N, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    top_k = np.argpartition(-corr, min(K, len(corr)))[:K]
    grad = np.zeros(N)
    for ai in top_k:
        L = ai - center
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            grad[:N - M] += (1.0 - h[M:])
            grad[M:] -= h[:N - M]
    grad /= K
    return grad

    lr = lr0

    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / 2000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        grad = softmax_grad(h, N, tau)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 25 == 0:

            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()
        if it % 40 == 0:
            lr *= decay
        if lr < 1e-10:
            break
    c5f = compute_c5(h, dx)
    if c5f < best_c5:

        best_c5 = c5f
    return best_h, best_c5
