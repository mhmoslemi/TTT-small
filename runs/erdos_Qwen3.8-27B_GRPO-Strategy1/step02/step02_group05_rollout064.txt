import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def to_binary(h, N, rng):
    h = h.copy()
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    s = int(h.sum())
    target = N // 2
    if s > target:
        idx = np.where(h == 1.0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:s - target]]] = 0.0
    elif s < target:
        idx = np.where(h == 0.0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:target - s]]] = 1.0
    return h

def sa_search(h, N, rng, deadline, T0, Tend, max_iter=200000):
