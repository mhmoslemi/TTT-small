def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(100):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    e = np.exp(t * (corr - mx))
    return (mx + np.log(e.sum()) / t) * dx

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    e = np.exp(t * (corr - mx))
    w = e / e.sum()
    g = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        wk = w[k]
        if wk < 1e-16:
            continue
        if m >= 0:
            j1s, j1e = 0, n - m
            j2s, j2e = m, n

        if j2e > j2s:
            grad[j2s:j2e] -= wk * h[j2s - m:j2e - m]
    return grad * dx

    dx = 2.0 / n

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0

        if abs(d) < 1e-14:

        if cnt == 0:
            break
        h[mask] += d / cnt

def smooth_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    e = np.exp(t * (corr - mx))
    return (mx + np.log1p(e.sum() - 1.0) / t) * dx if e.sum() > 0 else mx * dx

def smooth_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    e = np.exp(t * (corr - mx))
    w = e / e.sum()
    g = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        wk = w[k]
        if wk < 1e-16:
            continue
        if m >= 0:
            j1s, j1e = 0, n - m
        else:
            j1s, j1e = -m, n
        if j1e > j1s:

        if m >= 0:

        else:

        if j2e > j2s:
            grad[j2s:j2e] -= wk * h[j2s - m:j2e - m]
    return grad * dx

        return smooth_obj(x, t) + penalty * (x.sum() - n / 2.0) ** 2

        return smooth_grad(x, t) + 2 * penalty * (x.sum() - n / 2.0)

                 bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return proj(np.clip(r.x, 0.0, 1.0), n)

    r = minimize(smooth_obj, h, args=(t,), jac=smooth_grad, method='SLSQP',

                c5 = compute_c5(hc)
                if c5 < bc:
                    bc, bh, best_h, best_n = c5, hc, hc.copy(), 280
                h_bin = binary_push(hc, 280)
            except Exception:
                pass
        # Also try direct binary evaluation
        c5b = compute_c5(h_bin)
        if c5b < bc:
            bc, best_h, best_n = c5b, h_bin.copy(), 280

    # Phase 6: Restart from perturbed best at different t
    if tleft() > 60:
        for _ in range(3):
            if tleft() < 18:
                break
            pert = bh + rng.randn(280) * 0.02
            pert = proj(pert, 280)
            for t_val in [200, 2000]:
                if tleft() < 15:
                    break
                try:
                    hc = slsqp_solve(pert, 280, t_val, 1000)
                    c5 = compute_c5(hc)
                    if c5 < bc:
                        bc, bh, best_h, best_n = c5, hc, hc.copy(), 280
                        pert = hc
                except Exception:
                    pass

    # Phase 7: Final ultra-fine polish
    if tleft() > 15:
        try:
            hc = slsqp_solve(bh, 280, 100000, 3000)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc, best_h, best_n = c5, hc.copy(), 280
        except Exception:
            pass

    # Ensure final solution is feasible
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
