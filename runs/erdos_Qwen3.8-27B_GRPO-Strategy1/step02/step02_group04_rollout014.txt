    best_val = min(val_lo, val_hi)

        h_micro = fix_sum(h_micro, n_best)
        h_m1 = slsqp_opt(h_micro, n_best, 0.0002, max_iter=4000)
        try_update(h_m1, n_best)
        h_m2 = coord_descent(h_best, n_best, max_sweeps=80)
        try_update(h_m2, n_best)

    # Final cooldown SLSQP
    if budget_s - elapsed() > 30:
        h_last = slsqp_opt(h_best, n_best, 0.00005, max_iter=6000)
        try_update(h_last, n_best)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
