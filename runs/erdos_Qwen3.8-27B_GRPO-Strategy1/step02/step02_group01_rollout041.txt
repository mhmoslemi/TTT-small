    if T < 1e-8:
        return m * dx
    return (m + float(np.log(np.sum(np.exp(T * (c - m))))) / T) * dx

                     bounds=b, constraints=c, options={'maxiter': mi, 'ftol': 1e-14})
        return project_feasible(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

                     bounds=bnds, constraints=cons, options={'maxiter': mi, 'ftol': 1e-12})

    for _ in range(100):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s < 1e-14: break
            h += r * (d / s)

            if s < 1e-14: break
            h += h * (d / s)
