    corr = np.correlate(h, 1.0 - h, mode='full')

    sc = tau * corr
    sc -= sc.max()
    w_sm = np.exp(sc)

    idx_top = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 10.0) / 40.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    grad = np.zeros(N)
    thresh = 0.003
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def optimize_loop(h, N, target, deadline, tau_range, tk_range, lr0, lr_decay, lr_period, mom):
    best_h = h.copy()
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 2500.0, 1.0)
        tau = tau_range[0] + (tau_range[1] - tau_range[0]) * frac
        topK = int(tk_range[0] + (tk_range[1] - tk_range[0]) * frac)
        grad, c5_val = hybrid_grad(h, N, tau, topK)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        v = mom * v + grad
        h_new = h - lr * v
        h_new = project_sum(h_new, target)
        h = h_new
        if it % lr_period == 0:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_c5

def polish_coords(h, N, target, deadline):
    """Try setting individual coordinates to 0, 1, or nudge to reduce C5."""
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = h[i]
            for trial in (0.0, 1.0, max(0.0, orig - 0.02), min(1.0, orig + 0.02)):
                if abs(trial - orig) < 1e-8:
                    continue
                h[i] = trial
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-10:
                    best_c5 = c5
                    best_h = h.copy()
                    improved_any = True
                    break
            h[i] = best_h[i]
            h = best_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_par)
    x_par = np.linspace(0.0, 2.0, n_par, endpoint=False)
    sizes = [380, 462, 572]
    time_per_size = (budget_s - 60) / len(sizes)
    for N in sizes:
        if time.time() > deadline - 40:
            break
        size_dead = min(time.time() + time_per_size, deadline - 20)
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_base = np.interp(xN, x_par, h_par)
        h_base = project_sum(h_base, target)
        n_attempts = 3
        per_attempt = (size_dead - time.time() - 5) / n_attempts
        if per_attempt < 25:
            n_attempts = max(1, int((size_dead - time.time() - 5) / 25))
        for at in range(n_attempts):
            if time.time() > size_dead - 10:
                break
            trial_dead = time.time() + per_attempt
            if trial_dead > size_dead - 3:
                trial_dead = size_dead - 3
            if at == 0:
                h_cur = h_base.copy()
            else:
                h_cur = h_base + rng.randn(N) * (0.04 + 0.02 * at)
                h_cur = project_sum(h_cur, target)
            t_rem = trial_dead - time.time()
            if t_rem < 20:
                break
            p1_t = t_rem * 0.45
            p2_t = t_rem * 0.30
            p3_t = t_rem * 0.20
            h_cur, c1 = optimize_loop(h_cur, N, target, time.time() + p1_t,
                                      (2.0, 40.0), (30, 10), 0.04, 0.96, 60, 0.85)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h_cur.copy()
            if time.time() < trial_dead - 5:
                h_cur, c2 = optimize_loop(h_cur, N, target, time.time() + p2_t,
                                          (40.0, 200.0), (12, 4), 0.01, 0.97, 80, 0.88)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_cur.copy()
            if time.time() < trial_dead - 2:
                h_cur, c3 = optimize_loop(h_cur, N, target, trial_dead,
                                          (200.0, 600.0), (4, 2), 0.003, 0.98, 100, 0.90)
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = h_cur.copy()
    if time.time() < deadline - 8:
        Nb = len(best_h)
        h_f, c_f = optimize_loop(best_h.copy(), Nb, Nb / 2.0, time.time() + 30,
                                 (300.0, 1000.0), (3, 2), 0.002, 0.985, 100, 0.90)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f
    if time.time() < deadline - 5:
        Nb2 = len(best_h)
        best_h, best_c5 = polish_coords(best_h, Nb2, Nb2 / 2.0, deadline - 2)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
