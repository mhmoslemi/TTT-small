```python
def lp_solve(h_star, ts):
    n = len(h_star)
    dx = 2.0 / n
    cr = np.correlate(h_star, 1.0 - h_star, mode='full')
    c_vals = cr * dx
    
    # Compute gradients dc_k/dh_j at h_star
    # c_k = sum_i h[i] * (1 - h[i+k-(n-1)]) * dx
    # dc_k/dh_j = dx * [(1-h[j+k-(n-1)]) if valid - h[j-k+(n-1)] if valid]
    
    gmat = np.zeros((2*n-1, n))
    o = n - 1
    for j in range(n):
        for k in range(2*n-1):
            idx1 = j + k - o
            if 0 <= idx1 < n:
                gmat[k, j] += (1.0 - h_star[idx1])
            idx2 = j - k + o
            if 0 <= idx2 < n:
                gmat[k, j] -= h_star[idx2]
    gmat *= dx
    
    # LP: min t
    # vars: [h_0,...,h_{n-1}, t]
    # c_vec = [0,...,0, 1]
    c_vec = np.zeros(n+1)
    c_vec[-1] = 1.0
    
    # Inequality: g_k^T h - t ≤ c_k* - g_k^T h*
    # A_ineq: [g_k, -1] · [h, t] ≤ c_k* - g_k^T h*
    A_ineq = np.hstack([gmat, -np.ones((2*n-1, 1))])
    b_ineq = c_vals - gmat @ h_star
    
    # Equality: sum(h) = ts
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [ts]
    
    bounds = [(0, 1)] * n + [(None, None)]  # t is free
    
    res = linprog(c_vec, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_sp = {}

def _lp_obj(x):
    return x[_sp['n']]

def compute_c5(h, dx):
    cr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(cr) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0, 1, out=h)
    return h

def grad_c(h, dx):
    """Compute all gradients dc_k/dh_j. Returns (2n-1, n) matrix."""
    n = len(h)
    o = n - 1
    gmat = np.zeros((2*n-1, n))
    for j in range(n):
        # For each k, dc_k/dh_j
        # First term: (1-h[j+k-o]) if 0<=j+k-o<n
        # Second term: -h[j-k+o] if 0<=j-k+o<n
        # First term valid for k such that 0<=j+k-o<n, i.e., o-j <= k < o-j+n
        k_lo1 = max(0, o - j)
        k_hi1 = min(2*n-2, o - j + n - 1)
        for k in range(k_lo1, k_hi1 + 1):
            gmat[k, j] += (1.0 - h[j + k - o])
        # Second term valid for k such that 0<=j-k+o<n, i.e., j+o-n+1 <= k <= j+o
        k_lo2 = max(0, j + o - n + 1)
        k_hi2 = min(2*n-2, j + o)
        for k in range(k_lo2, k_hi2 + 1):
            gmat[k, j] -= h[j - k + o]
    gmat *= dx
    return gmat

def lp_step(h_star, ts, max_iter=50):
    n = len(h_star)
    dx = 2.0 / n
    o = n - 1
    
    # Compute correlations and gradients
    cr = np.correlate(h_star, 1.0 - h_star, mode='full')
    c_vals = cr * dx
    gmat = grad_c(h_star, dx)
    
    # LP: min t s.t. gmat @ h - t <= c_vals - gmat @ h_star, sum(h)=ts, 0<=h<=1
    c_vec = np.zeros(n+1)
    c_vec[-1] = 1.0
    A_ineq = np.hstack([gmat, -np.ones((2*n-1, 1))])
    b_ineq = c_vals - gmat @ h_star
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([ts])
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_vec, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x[:n]

def lp_optimize(h0, ts, n_iter=30, alpha_start=1.0):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    bc = compute_c5(h, dx)
    
    for it in range(n_iter):
        h_lp = lp_step(h, ts)
        if h_lp is None:
            break
        # Check if at fixed point
        if np.max(np.abs(h_lp - h)) < 1e-10:
            break
        # Damped step
        alpha = alpha_start * (0.9 ** it)
        h_new = (1 - alpha) * h + alpha * h_lp
        h_new = proj(h_new, ts)
        c_new = compute_c5(h_new, dx)
        if c_new < bc:

        else:

            c_new = compute_c5(h_new, dx)
            if c_new < bc:

            else:

def soft_grad(h, dx, beta):
    """Softmax-weighted gradient for C5."""
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    m = cr.max()
    w = np.exp(beta * (cr - m))
    s = w.sum()

    else:
        w /= s
    return grad_c(h, dx) @ w

def softmax_descent(h0, dx, ts, n_iter=2000, beta_start=10, beta_end=200):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    st = 0
    for it in range(n_iter):
        b = beta_start + (beta_end - beta_start) * min(1.0, it / (0.7 * n_iter))
        lr = 0.05 / (1.0 + 0.005 * it)
        g = soft_grad(h, dx, b)
        ok = False
        for sh in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            h2 = proj(h - lr * sh * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                h, bc = h2, c2
                st = 0
                ok = True
                break
        if not ok:
            st += 1
            if st > 200:
                break
    return h, bc

def coord_descent(h0, dx, ts, n_pass=50):
    """Coordinate descent with mass-swapping to maintain sum."""
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    rng = np.random.RandomState(12345)
    for p in range(n_pass):
        improved = False
        for j in rng.permutation(n):
            sv = h[j]
            # Try different values for h[j]

                h2[j] = sv + dv

                c2 = compute_c5(h2, dx)
                if c2 < bc - 1e-15:

                    improved = True
                    break
        if not improved:

    return h, bc

def _sineq(x):

def slsqp_finish(h0, dx, ts, maxiter=200):

    return h0.copy()
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    n = _sp['n']
    return float(np.sum(x[:n]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_mat(h, dx):
    n = len(h)
    gmat = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        i_lo = max(0, r - n + 1)
        i_hi = min(n - 1, r)
        for i in range(i_lo, i_hi + 1):
            idx2 = i - r + n - 1
            gmat[r, i] += (1.0 - h[idx2])
            if 0 <= idx2 < n:
                gmat[r, idx2] -= h[i]
    gmat *= dx
    return gmat

def lp_step(h_star, ts):
    n = len(h_star)
    dx = 2.0 / n
    cr = np.correlate(h_star, 1.0 - h_star, mode='full')
    c_vals = cr * dx
    gmat = grad_mat(h_star, dx)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.hstack([gmat, -np.ones((2 * n - 1, 1))])
    b_ub = c_vals - gmat @ h_star
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([ts])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def lp_optimize(h0, ts, n_iter=50):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    bc = compute_c5(h, dx)
    for it in range(n_iter):
        h_lp = lp_step(h, ts)
        if h_lp is None:
            break
        if np.max(np.abs(h_lp - h)) < 1e-12:
            break
        for alpha in (1.0, 0.7, 0.5, 0.3, 0.1):
            h_new = (1 - alpha) * h + alpha * h_lp
            h_new = proj(h_new, ts)
            c_new = compute_c5(h_new, dx)
            if c_new < bc - 1e-15:
                h, bc = h_new, c_new
                break
    return h, bc

def soft_grad(h, dx, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    m = cr.max()
    w = np.exp(beta * (cr - m))
    s = w.sum()
    if s < 1e-30:
        w = np.ones_like(cr) / len(cr)
    else:
        w /= s
    gmat = grad_mat(h, dx)
    return gmat @ w

def softmax_descent(h0, dx, ts, n_iter=3000, beta_start=8, beta_end=200, dl=None):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    st = 0
    for it in range(n_iter):
        if dl and time.time() > dl:
            break
        b = beta_start + (beta_end - beta_start) * min(1.0, it / (0.6 * n_iter))
        lr = 0.04 / (1.0 + 0.003 * it)
        g = soft_grad(h, dx, b)
        ok = False
        for sh in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            h2 = proj(h - lr * sh * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                h, bc, st = h2, c2, 0
                ok = True
                break
        if not ok:
            st += 1
            if st > 300:
                break
    return h, bc

def coord_descent(h0, dx, ts, n_pass=80, dl=None):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    rng = np.random.RandomState(777)

                c2 = compute_c5(h2, dx)
                if c2 < bc - 1e-15:
                    h, bc = h2, c2
                    imp = True
                    break
        if not imp:
            break
    return h, bc

def slsqp_finish(h0, dx, ts, maxiter=200):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) <= t0 * dx + 0.01:
                return hr
    except Exception:
        pass
    return h0.copy()

def full_pipeline(h0, ts, dl, rng):

    dx = 2.0 / n
    h = proj(h0, ts)
```
