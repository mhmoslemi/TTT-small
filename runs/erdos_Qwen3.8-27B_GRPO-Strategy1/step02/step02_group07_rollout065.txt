    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-13:
        return h
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-11:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

    V = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.power(np.sum(V**p) + 1e-30, 1.0/p))

    S = np.sum(V**p)

    c = np.max(corr)
    if T < 1e-15:
        return c
    return c + T * np.log(np.sum(np.exp((corr - c)/T)) + 1e-30)

    return np.sum(h) - 0.5 * n

    try:
        r = optimize.minimize(obj_pen, h, args=(p, lam), jac=obj_pen_grad,

    return h

Actually, rotations don't change the problem (since the integral is over all shifts). Mirroring also doesn't change it. So those aren't useful.

What about a "dual" transformation? The problem has some self-duality: if we replace h by 1-h, the objective becomes max_k ∫(1-h)(h(x+k))dx which is the same as max_k ∫h(x)(1-h(x-k))dx = max_k ∫h(x)(1-h(x+k))dx (by reindexing). So the problem is invariant under h → 1-h! But we also need ∫h = 1, so ∫(1-h) = 2-1 = 1. Both are valid. This duality means we can't get a better solution by flipping.

What about h → h(x + a) (translation)? This also preserves the problem since the max is over all shifts.

OK so the only useful perturbations are asymmetric ones.

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-13:
        return h
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-11:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def power_obj(h, p):
    dx = 2.0 / len(h)
    V = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.power(np.sum(V**p) + 1e-30, 1.0 / p))

def power_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    V = corr * dx
    S = np.sum(V ** p)
    factor = S ** (1.0 / p - 1.0)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        wm = factor * V[m] ** (p - 1) * dx
        if abs(wm) < 1e-30:
            continue
        if s == 0:
            grad += wm * (1.0 - 2.0 * h)
        else:
            jlo = max(0, -s)
            jhi = min(n - 1, n - 1 - s)
            if jlo <= jhi:
                grad[jlo:jhi + 1] += wm * (1.0 - h[jlo + s:jhi + s + 1])
            jlo2 = max(0, s)
            jhi2 = min(n - 1, n - 1 + s)
            if jlo2 <= jhi2:
                grad[jlo2:jhi2 + 1] -= wm * h[jlo2 - s:jhi2 - s + 1]
    return grad

def obj_pen(h, p, lam):
    sv = np.sum(h) - 0.5 * len(h)
    return power_obj(h, p) + lam * sv * sv

def obj_pen_grad(h, p, lam):
    sv = np.sum(h) - 0.5 * len(h)
    g = power_grad(h, p)
    g += 2.0 * lam * sv
    return g

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.max(corr)
    if T < 1e-15:
        return c
    return c + T * np.log(np.sum(np.exp((corr - c) / T)) + 1e-30)

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def lbgfs_opt(h, n, p, lam, iters):
    c0 = obj_pen(h, p, lam)
    try:
        r = optimize.minimize(obj_pen, h, args=(p, lam), jac=obj_pen_grad,
                              method='L-BFGS-B', bounds=[(0., 1.0)] * n,
                              options={'maxiter': iters, 'ftol': 1e-22, 'gtol': 1e-15})
        hn = fix_sum(r.x, n)
        if obj_pen(hn, p, lam) < c0:
            return hn
    except Exception:
        pass
    return h

def slsqp_lse(h, n, T, iters):
    c0 = lse_obj(h, T)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
                              bounds=[(0., 1.0)] * n,
                              constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                              options={'maxiter': iters, 'ftol': 1e-16})
        hn = fix_sum(r.x, n)
        if lse_obj(hn, T) < c0:
            return hn
    except Exception:
        pass
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

DELTA_SCALES = np.array([-1.0, -0.5, -0.2, 0.2, 0.5, 1.0])
DELTA_PAIRS = np.array([-1.0, -0.4, -0.1, 0.1, 0.4, 1.0])

def coord_desc(h, n, eta):
    idx = np.arange(n)
    for j in range(n):
        if h[j] <= 1e-5 or h[j] >= 1.0 - 1e-5:
            continue
        best_c = compute_c5(h)
        best_d = 0.0
        for d in eta * DELTA_SCALES:
            hv = h[j] + d
            if hv < 1e-5 or hv > 1.0 - 1e-5:
                continue
            hn = h.copy()
            hn[j] = hv
            dsum = 0.5 * n - np.sum(hn)
            msk = (idx != j) & (hn > 1e-6) & (hn < 1.0 - 1e-6)
            cnt = msk.sum()
            if cnt == 0:
                continue
            hn[msk] -= dsum / cnt
            hn = np.clip(hn, 0.0, 1.0)
            c = compute_c5(hn)
            if c < best_c - 1e-15:
                best_c = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            dsum = 0.5 * n - np.sum(h)
            msk = (idx != j) & (h > 1e-6) & (h < 1.0 - 1e-6)
            if msk.sum() > 0:
                h[msk] -= dsum / msk.sum()
                h = np.clip(h, 0.0, 1.0)
    return fix_sum(h, n)

def pair_desc(h, n, rng_obj, n_pairs, eta):
    for _ in range(n_pairs):
        j, l = rng_obj.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4 or h[l] <= 1e-4 or h[l] >= 1.0 - 1e-4:
            continue
        best_c = compute_c5(h)
        best_d = 0.0
        for d in eta * DELTA_PAIRS:
            hj2, hl2 = h[j] + d, h[l] - d
            if hj2 < 1e-4 or hj2 > 1.0 - 1e-4 or hl2 < 1e-4 or hl2 > 1.0 - 1e-4:
                continue
            hn = h.copy()
            hn[j], hn[l] = hj2, hl2
            c = compute_c5(hn)
            if c < best_c - 1e-15:
                best_c = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return fix_sum(h, n)
