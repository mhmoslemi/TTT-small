def grad_step(h, n, step_size):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    shift = k_star - n + 1
    # gradient of corr[k_star] w.r.t. h
    grad = np.zeros(n)
    for j in range(n):
        # Term 1: i=j, contributes (1 - h[j+shift]) if 0 <= j+shift < n
        g1 = 0.0
        if 0 <= j + shift < n:
            g1 = 1.0 - h[j + shift]
        # Term 2: i+shift=j => i=j-shift, contributes -h[j-shift] if 0 <= j-shift < n
        g2 = 0.0
        if 0 <= j - shift < n:
            g2 = -h[j - shift]
        grad[j] = g1 + g2
    # Also need to account for the constraint sum(h) = n/2
    # Project gradient: grad_proj = grad - mean(grad) (for the sum constraint)
    # Actually for the constraint gh + b = 0 where g=[1,1,...,1], b=n/2:
    # Projected grad = grad - (sum(grad)/n) * ones
    grad_norm = grad - np.mean(grad)
    # Take step
    h_new = h - step_size * dx * grad_norm
    h_new = fix_sum(np.clip(h_new, 0, 1), n)
    return h_new

import numpy as np

import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-13:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    k = np.sum(mask)
    if k > 0:
        h[mask] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-11:
        mask2 = (h > 0.0) & (h < 1.0)
        k2 = np.sum(mask2)
        if k2 > 0:
            h[mask2] += d2 / k2
    return np.clip(h, 0.0, 1.0)

    c0 = compute_c5(h)

            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': maxiter, 'ftol': 1e-19, 'maxls': 80})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h
