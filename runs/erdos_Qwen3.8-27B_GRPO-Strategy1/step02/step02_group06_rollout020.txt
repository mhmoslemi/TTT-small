def compute_grad_fft(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    p = np.exp((corr - cmax) / T)
    p /= np.sum(p)
    
    # p has length 2n-1, p[k] corresponds to shift s = k-(n-1)
    # Build circular kernel: p_circ[m] for m in [0, n-1]
    # p_circ[m] = sum of p[k] where (k-(n-1)) ≡ m (mod n)
    # s = k-(n-1) ranges from -(n-1) to n-1
    # s mod n: for s in [0,n-1]: s mod n = s
    #           for s in [-(n-1),-1]: s mod n = s+n (ranges from 1 to n-1)
    
    p_circ = np.zeros(n)
    # s=0: k=n-1, contributes to m=0
    p_circ[0] += p[n-1]
    # s in [1, n-1]: k=n-1+s, contributes to m=s
    for s in range(1, n):
        p_circ[s] += p[n-1+s]
    # s in [-(n-1), -1]: k=n-1+s, contributes to m=s+n
    for s in range(-(n-1), 0):
        p_circ[s+n] += p[n-1+s]
    
    # α[j] = ∑_s p_s * h[(j-s)%n] = circular_cross_corr(h, p_circ)[j]
    # numpy: α = ifft(fft(h) * conj(fft(p_circ)))

    alpha = np.fft.irfft(H * np.conj(P), n)
    
    # β[j] = ∑_s p_s * h[(j+s)%n] = ∑_s p_{-s} * h[(j-s)%n]
    # p_neg[m] = p_circ[-m mod n] = p_circ[(n-m) % n]
    p_neg = np.zeros(n)
    p_neg[0] = p_circ[0]
    for m in range(1, n):
        p_neg[m] = p_circ[n-m]
    Pn = np.fft.rfft(p_neg, n)
    beta = np.fft.irfft(H * np.conj(Pn), n)
    
    return dx * (1.0 - alpha - beta)

    corr = np.correlate(hp, 1.0 - hp, mode='full')
    cmax = np.max(corr)
    exp_c = np.exp((corr - cmax) / T)
    w = exp_c / np.sum(exp_c)
    obj = (cmax + T * np.log(np.sum(exp_c))) * dx
    
    # Gradient
    g = dx * np.ones(n)
    for k in range(len(corr)):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        g -= w[k] * dx * np.roll(hp, s)
        g -= w[k] * dx * np.roll(hp, -s)
    return obj, g

def lbfgs(h0, n, T, max_iter=500):
    res = opt.minimize(obj_grad, h0, args=(T, n), method='L-BFGS-B',
                       bounds=[(0, 1)] * n, jac=True,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10, 'iprint': 0})
    return project(res.x, n)

    x_new = np.linspace(0, 2, new_n, endpoint=False)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-10:
        m = (h > 1e-8) & (h < 1.0 - 1e-8)
        c = np.sum(m)
        if c > 0:
            h[m] += d / c
            h = np.clip(h, 0, 1)
    return h

def obj_grad(h, T, n):
    hp = project(h, n)
    dx = 2.0 / n
    corr = np.correlate(hp, 1.0 - hp, mode='full')
    cmax = np.max(corr)
    ec = np.exp((corr - cmax) / T)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (np.sum(corr ** p) / len(corr)) ** (1.0 / p) * dx

def pnorm_grad(h, p, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    base = (np.sum(corr ** p) / len(corr)) ** (1.0 / p)
    g = np.zeros(n)
    norm_factor = base ** (p - 1) / len(corr)
    for k in range(len(corr)):
        ck = corr[k]
        if abs(ck) < 1e-14:
            continue
        coef = norm_factor * p * ck ** (p - 1) * dx
        s = k - (n - 1)
        g += coef * np.roll(h, s)
        g += coef * np.roll(h, -s)
    return g

    """Block coordinate refinement: optimize subsets of coordinates."""
