    target = 0.5 * n
    # Bisection on lambda: h_i = clip(g_i + lambda, 0, 1), find lambda s.t. sum(h) = target
    lo, hi = -max(np.abs(g)) - 2, max(np.abs(g)) + 2

        s = np.sum(h)

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = (corr - c_max) / T
    shifted -= np.max(shifted)  # numerical stability
    p = np.exp(shifted)
    p_sum = np.sum(p)
    p /= p_sum
    obj = dx * (c_max + T * (np.log(p_sum) + np.max(shifted)))
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-14:
            continue
        shift = m - (n - 1)
        # Term 1: (1-h[j-shift]) for valid j
        j_lo = max(0, shift)
        j_hi = min(n - 1, n - 1 + shift)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += p[m] * (1.0 - h[j_lo - shift:j_hi + 1 - shift])
        # Term 2: -h[j+shift] for valid j
        j_lo2 = max(0, -shift)
        j_hi2 = min(n - 1, n - 1 - shift)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= p[m] * h[j_lo2 + shift:j_hi2 + 1 + shift]
    grad *= dx
    return obj, grad

    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-12:
        return h
    m = (h > 1e-6) & (h < 1.0 - 1e-6)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-15:
        return cmax * dx
    s = (corr - cmax) / T
    smax = np.max(s)
    return (cmax + T * (smax + np.log(np.sum(np.exp(s - smax))))) * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    s = (corr - cmax) / T
    smax = np.max(s)
    p = np.exp(s - smax)
    p /= np.sum(p)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-15:
            continue
        sh = m - (n - 1)
        a = max(0, sh)
        b = min(n - 1, n - 1 + sh)
        if a <= b:
            grad[a:b+1] += pm * (1.0 - h[a-sh:b+1-sh])
        c = max(0, -sh)
        d = min(n - 1, n - 1 - sh)
        if c <= d:
            grad[c:d+1] -= pm * h[c+sh:d+1+sh]
    return grad * dx

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-13:
        return h
    msk = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(msk)
    if k > 0:
        h[msk] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        idx = np.where(msk)[0]
        if len(idx) > 0:
            h[idx[0]] = min(max(h[idx[0]] + d2, 0.0), 1.0)
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return np.sum(x) - 0.5 * n

def _obj_wrap(x, T):
    return lse_obj(x, T)

def _jac_wrap(x, T):
    return lse_grad(x, T)

def do_slsqp(h, n, T, max_iter):
    c0 = compute_c5(h)
    try:
        res = optimize.minimize(_obj_wrap, h.copy(), args=(T,), method='SLSQP',
                                jac=_jac_wrap, bounds=[(0., 1.)] * n,
                                constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (n,)}],
                                options={'maxiter': max_iter, 'ftol': 1e-16})
        hn = fix_sum(res.x, n)
        c1 = compute_c5(hn)
        if c1 <= c0 + 1e-15:
            return hn
        return h
    except:
        return h

def coord_polish(h, n, n_pass, deadline):
    deltas = (-0.05, -0.02, -0.008, -0.003, 0.003, 0.008, 0.02, 0.05)
    for p in range(n_pass):
        if time.time() > deadline:
            break
        for j in range(n):
            if time.time() > deadline:
                break
            hv = h[j]
            if hv < 2e-4 or hv > 1.0 - 2e-4:
                continue
            best_c = compute_c5(h)
            best_v = hv
            for delta in deltas:
                nv = hv + delta
                if nv < 1e-5 or nv > 1.0 - 1e-5:
                    continue
                hn = h.copy()
                hn[j] = nv
                diff = nv - hv
                if abs(diff) > 1e-14:
                    others = np.ones(n, dtype=bool)
                    others[j] = False
                    hn[others] -= diff / (n - 1)
                    hn = np.clip(hn, 0.0, 1.0)
                    hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = nv
            if best_v != hv:
                h = fix_sum(h, n)
                h[j] = best_v
                h = fix_sum(h, n)
    return fix_sum(h, n)

def pair_polish(h, n, rng_obj, n_pairs, deadline):
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        j, l = rng_obj.choice(n, 2, replace=False)
        if h[j] < 2e-4 or h[j] > 1.0 - 2e-4:
            continue
        if h[l] < 2e-4 or h[l] > 1.0 - 2e-4:
            continue
        best_c = compute_c5(h)
        best_d = 0.0
        for d in (-0.05, -0.02, -0.006, 0.006, 0.02, 0.05):
            hv2, lv2 = h[j] + d, h[l] - d
            if hv2 < 1e-4 or hv2 > 1.0 - 1e-4 or lv2 < 1e-4 or lv2 > 1.0 - 1e-4:
                continue
            hn = h.copy()
            hn[j], hn[l] = hv2, lv2
            c = compute_c5(hn)
            if c < best_c - 1e-14:
                best_c = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return fix_sum(h, n)

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

    for i in range(20):
