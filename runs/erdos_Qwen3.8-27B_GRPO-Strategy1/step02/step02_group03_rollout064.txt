import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0, 1, out=h)
    for _ in range(200):
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1 - 1e-15
        else:
            m = h > 1e-15
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0, 1, out=h)
    return h

def smooth_max_val(corr, beta):
    m = corr.max()
    s = beta * (corr - m)
    s = s - s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / beta

def smooth_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = beta * (corr - m)
    s = s - s.max()
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        # Term 1: (1-h[v1]) where v1 = i+k-n+1, valid i: max(0,n-1-k) <= i <= min(n-1, 2n-2-k)
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if lo1 <= hi1:
            v1 = np.arange(lo1, hi1 + 1) + k - n + 1
            grad[lo1:hi1+1] += wk * (1.0 - h[v1])
        # Term 2: -h[v2] where v2 = i-k+n-1, valid i: max(0,k-n+1) <= i <= min(n-1, k-1)
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, k - 1)
        if lo2 <= hi2:
            v2 = np.arange(lo2, hi2 + 1) - k + n - 1
            grad[lo2:hi2+1] -= wk * h[v2]
    return grad * dx

def anneal(h0, ts, betas, steps, tlim):
    h = h0.copy()
    bh = h.copy()
    bc = compute_c5(h)
    for beta in betas:
        for _ in range(steps):
            if time.time() > tlim:
                return bh, bc
            g = smooth_grad(h, beta)
            fn = smooth_obj(h, beta)
            lr = 0.1
            ok = False
            for _ in range(40):
                h2 = proj(h - lr * g, ts)
                fn2 = smooth_obj(h2, beta)
                if fn2 < fn - 1e-16:
                    h = h2
                    fn = fn2
                    c = compute_c5(h)
                    if c < bc:
                        bc, bh = c, h.copy()
                    ok = True
                    break
                lr *= 0.5
            if not ok:
                break
    return bh, bc

def slsqp_smooth(h0, ts, beta, mit):
    n = len(h0)
    def obj(x):
        return smooth_obj(x, beta)
    def obj_grad(x):
        return smooth_grad(x, beta)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts}]
    bnds = [(0, 1)] * n
    try:
        r = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': mit, 'ftol': 1e-14})
        hr = proj(r.x, ts)
        if compute_c5(hr) < compute_c5(h0) + 0.01:
            return hr
    except:
        pass
    return h0.copy()

def coord_desc(h0, ts, rnd, tlim, rng):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h)
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for r in range(rnd):
        if time.time() > tlim:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in deltas:
                h[j] = sv + d
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                    continue
                h = proj(h, ts)
                c = compute_c5(h)
                if c < bc:
                    bc, imp = c, True
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def make_starts(n, ts, rng, ref=None):
    out = [np.full(n, ts / n)]
    # Random binary
    for _ in range(2):
        h = np.zeros(n)
        p = rng.permutation(n)
        int_ts = int(ts)
        fr = ts - int_ts
        h[p[:int_ts]] = 1.0
        if fr > 0.01:
            h[p[int_ts]] = fr
        out.append(h.copy())
    # Sine patterns
    for m in (1, 2, 3, 5, 7):
        x = np.arange(n) / n
        out.append(0.5 + 0.35 * np.sin(2 * np.pi * m * x + rng.uniform(0, 6.28)))
    # Random
    for _ in range(2):
        out.append(rng.uniform(0, 1, n))
    # Periodic patterns
    for p in (2, 3, 5, 7):
        pp = max(2, n // p)
        half = max(1, pp // 2)
        out.append(np.array([1.0 if (i % pp) < half else 0.0 for i in range(n)]))
    # Interpolated reference
    if ref is not None and len(ref) != n:
        hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                       np.linspace(0, 1, len(ref), endpoint=False), ref)
        out.append(hi)
    elif ref is not None:
        out.append(ref.copy())
    return [proj(s, ts) for s in out]

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    t_end = t_start + budget_s - 3
    bh, bc = None, 1e9
    
    # Get initial values
    try:

    except NameError:

    h_init = proj(hr, ts0)
    c_init = compute_c5(h_init)
    bh, bc = h_init, c_init
    
    # Smooth max annealing at n0
    betas = [3, 5, 10, 20, 40, 80, 150, 300]
    bh, bc = anneal(bh, ts0, betas, 80, t_end - 120)
    
    # SLSQP with smooth objective
    if time.time() < t_end - 100:
        hs = slsqp_smooth(bh, ts0, 100, 200)
        cs = compute_c5(hs)
        if cs < bc:
            bh, bc = hs, cs
    
    # Coordinate descent
    if time.time() < t_end - 80:
        bh, bc = coord_desc(bh, ts0, 40, t_end - 60, rng)
    
    # Phase 2: Try other resolutions
    for dn in (0, 2, -2, 4, -4, 6, -6, 3, -3, 5, -5, 8, -8):
        nt = n0 + dn
        if nt < 25 or nt > 90 or time.time() > t_end - 60:
            continue
        dxt, tst = 2.0 / nt, nt / 2.0
        starts = make_starts(nt, tst, rng, bh)
        try_s = min(5, len(starts))
        for si in range(try_s):
            if time.time() > t_end - 40:
                break
            h_s = starts[si]
            # Quick anneal
            h_s, c_s = anneal(h_s, tst, [5, 15, 50, 150], 50, t_end - 30)
            if c_s < bc:
                bh, bc = h_s, c_s
            # SLSQP polish

                hp = slsqp_smooth(h_s, tst, 80, 150)

                if cp < bc:
                    bh, bc = hp, cp
    
    # Phase 3: Final refinement at best resolution
    if bh is not None and time.time() < t_end - 30:
        nb = len(bh)
        tsb = nb / 2.0
        # Fine anneal
        bh, bc = anneal(bh, tsb, [100, 200, 400], 200, t_end - 20)
        # SLSQP

            hp = slsqp_smooth(bh, tsb, 200, 200)
            cp = compute_c5(hp)
            if cp < bc:
                bh, bc = hp, cp
        # Very fine coordinate descent
        if time.time() < t_end - 5:
            bh, bc = coord_desc(bh, tsb, 50, t_end - 2, rng)
    
    if bh is None:
        bh = proj(np.full(50, 0.5), 25.0)
        bc = compute_c5(bh)
    
    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh)
    return bh, bc, len(bh)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _sm(corr, beta):
    m = corr.max()
    s = beta * (corr - m) - (beta * (corr - m)).max()
    return m + (0.0 + np.log(np.sum(np.exp(s)))) / beta

def sm_obj(h, beta):
    return _sm(np.correlate(h, 1.0 - h, mode='full'), beta)

def sm_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = beta * (corr - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += wk * (1.0 - h[lo1 + k - n + 1:hi1 + k - n + 2])
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, k - 1)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= wk * h[lo2 - k + n - 1:hi2 - k + n]
    return g * dx

def anneal(h0, ts, betas, steps, tlim):
    h, bh, bc = h0.copy(), h0.copy(), compute_c5(h0)
    for beta in betas:
        for _ in range(steps):
            if time.time() > tlim:
                return bh, bc
            g = sm_grad(h, beta)
            fn = sm_obj(h, beta)
            lr = 0.05
            ok = False
            for _ in range(50):
                h2 = proj(h - lr * g, ts)
                if sm_obj(h2, beta) < fn - 1e-16:
                    h = h2
                    c = compute_c5(h)
                    if c < bc:
                        bc, bh = c, h.copy()
                    fn = sm_obj(h, beta)
                    ok = True
                    break
                lr *= 0.5
            if not ok:
                break
    return bh, bc

def slsqp_s(h0, ts, beta, mit):
    n = len(h0)
    try:
        r = minimize(sm_obj, h0, args=(beta,), jac=lambda x, b: sm_grad(x, b),
                     method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': mit, 'ftol': 1e-15})
        hr = proj(r.x, ts)
        c = compute_c5(hr)
        if c < compute_c5(h0) + 0.01:
            return hr, c
    except Exception:
        pass
    return h0.copy(), compute_c5(h0)

def cdesc(h0, ts, rnd, tlim, rng):
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0)
    dets = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
    for _ in range(rnd):
        if time.time() > tlim:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in dets:
                h[j] = sv + d
                if not (0 <= h[j] <= 1):
                    continue
                h = proj(h, ts)
                c = compute_c5(h)
                if c < bc:
                    bc, imp = c, True
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def make_starts(n, ts, rng, ref):
    x = np.arange(n) / n
    out = [np.full(n, ts / n)]
    for _ in range(3):
        h = np.zeros(n)
        p = rng.permutation(n)
        h[p[:int(ts)]] = 1.0
        fr = ts - int(ts)
        if fr > 0.01:
            h[p[int(ts)]] = fr
        out.append(h.copy())
    for m in (1, 2, 3, 5, 7):
        out.append(0.5 + 0.35 * np.sin(2 * np.pi * m * x + rng.randn() * 0.5))
    out.append(rng.uniform(0, 1, n))
    for p in (3, 5, 7):
        pp = max(2, n // p)
        hh = max(1, pp // 2)
        out.append(np.array([1.0 if i % pp < hh else 0.0 for i in range(n)]))
    if ref is not None:
        if len(ref) == n:
            out.insert(0, ref.copy())
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, len(ref), endpoint=False), ref)
            out.insert(0, hi)
    return [proj(s, ts) for s in out]

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    te = t0 + budget_s - 3
    bh, bc = None, 1e9
    try:
        hr = np.array(initial_h_values, dtype=float)

    bh, bc = h_init.copy(), compute_c5(h_init)
    # Phase 1: anneal + SLSQP + cdesc at n0
    bh, bc = anneal(bh, ts0, [3, 5, 8, 15, 25, 50, 100, 200, 400], 60, te - 180)
    if time.time() < te - 150:
        hs, cs = slsqp_s(bh, ts0, 150, 300)
        if cs < bc:
            bh, bc = hs, cs
    if time.time() < te - 120:
        bh, bc = cdesc(bh, ts0, 40, te - 100, rng)
    # Phase 2: other resolutions
    for dn in (2, -2, 4, -4, 6, -6, 3, -3, 8, -8, 5, -5):
        nt = n0 + dn
        if nt < 25 or nt > 95 or time.time() > te - 50:
            break
        tst = nt / 2.0
        st = make_starts(nt, tst, rng, bh)
        for hs_ in st[:4]:
            if time.time() > te - 35:
                break
            h_s, c_s = anneal(hs_, tst, [8, 20, 60, 150], 50, te - 25)
            if c_s < bc:
                bh, bc = h_s, c_s
            if time.time() < te - 20:
                hp, cp = slsqp_s(h_s, tst, 100, 200)
                if cp < bc:
                    bh, bc = hp, cp
    # Phase 3: final refinement
    if bh is not None and time.time() < te - 40:
        nb = len(bh)
        tsb = nb / 2.0
        bh, bc = anneal(bh, tsb, [150, 300, 500, 800], 150, te - 25)
        if time.time() < te - 15:
            hp, cp = slsqp_s(bh, tsb, 400, 300)
            if cp < bc:
                bh, bc = hp, cp
        if time.time() < te - 5:

    if bh is None:
        bh = proj(np.full(50, 0.5), 25.0)
        bc = compute_c5(bh)
    bh = proj(bh, len(bh) / 2.0)
    bc = compute_c5(bh)
    return bh, bc, len(bh)

    s = beta * (corr - m)
    s -= s.max()
