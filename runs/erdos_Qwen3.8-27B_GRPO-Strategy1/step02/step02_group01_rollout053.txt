```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def erdos_ineq(x):
    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr  # element-wise, need all >= 0

def slsqp_erdos(h, maxiter=300):
    n = len(h)
    corr = get_corr_full(h)
    t0 = float(np.max(corr)) + 0.02
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': erdos_eq},
        {'type': 'ineq', 'fun': erdos_ineq}
    ]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_c5, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgradient_step(h, n, lr):
    """One subgradient step for the max correlation."""
    corr = get_corr_full(h)
    max_val = np.max(corr)
    # Weighted subgradient: weights proportional to exp(T*(corr - max))
    T = 200
    weights = np.exp(T * (corr - max_val))
    weights /= weights.sum()
    # Subgradient of max is convex combination of subgradients of active constraints
    # For corr[m] with lag = m - (n-1):
    # grad[j] = (1-h[j-lag])*I(0<=j-lag<n) - h[j+lag]*I(0<=j+lag<n)
    # We need to average over m with weights
    grad = np.zeros(n)
    idx = np.arange(n)
    for m, w in enumerate(weights):
        if w < 1e-10:
            continue
        lag = m - (n - 1)
        # (1-h[j-lag]) for valid j-lag
        v1 = (idx - lag) >= 0
        v2 = (idx - lag) < n
        valid = v1 & v2
        grad[valid] += w * (1.0 - h[idx[valid] - lag])
        # -h[j+lag] for valid j+lag
        v3 = (idx + lag) >= 0
        v4 = (idx + lag) < n
        valid2 = v3 & v4
        grad[valid2] -= w * h[idx[valid2] + lag]
    return h - lr * grad, grad

    h_cur = h.copy()
    vel = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    lr = lr0
    for it in range(n_iters):
        corr = get_corr_full(h_cur)
        max_val = np.max(corr)
        T = 200
        weights = np.exp(T * (corr - max_val))
        weights /= weights.sum()
        idx = np.arange(n)
        grad = np.zeros(n)
        for m in range(len(corr)):
            if weights[m] < 1e-10:
                continue
            lag = m - (n - 1)
            v1 = (idx - lag) >= 0
            v2 = (idx - lag) < n
            valid = v1 & v2
            grad[valid] += weights[m] * (1.0 - h_cur[idx[valid] - lag])
            v3 = (idx + lag) >= 0
            v4 = (idx + lag) < n
            valid2 = v3 & v4
            grad[valid2] -= weights[m] * h_cur[idx[valid2] + lag]
        vel = momentum * vel - lr * grad
        h_new = h_cur + vel
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
        lr *= decay
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def sum_con(x):
    return x.sum() - len(x) / 2.0

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    return x[:len(x)-1].sum() - (len(x)-1) / 2.0

def erdos_ineq(x):
    n = len(x) - 1
    return x[-1] - get_corr(x[:n])

def slsqp_erdos(h, maxiter=400):
    n = len(h)
    t0 = float(np.max(get_corr(h))) + 0.01
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': erdos_eq}, {'type': 'ineq', 'fun': erdos_ineq}]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def compute_subgrad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-12:
            continue
        lag = m - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            js = np.arange(lo1, hi1)
            grad[js] += w * (1.0 - h[js - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            js = np.arange(lo2, hi2)
            grad[js] -= w * h[js + lag]
    return grad

def subgradient_descent(h, n_iters=200, lr0=0.006, decay=0.997, momentum=0.88, T=300):
    n = len(h)
    h_cur = h.copy()
    vel = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    lr = lr0
    for it in range(n_iters):
        corr = get_corr(h_cur)
        mv = np.max(corr)
        weights = np.exp(T * (corr - mv))
        weights /= weights.sum()
        grad = compute_subgrad(h_cur, weights)
        vel = momentum * vel - lr * grad
        h_new = project_feasible(h_cur + vel, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
        lr *= decay
    return best_h, best_c5

def make_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_local_search(h_bin, max_passes=20, full=True):

    h_cur = h_bin.copy()
    best_c5 = compute_c5(h_cur)

                    h_cur[i] = 1.0
                    h_cur[j] = 0.0
                    if c < best_val - 1e-14:
                        best_val = c
                        best_pair = (i, j)
        else:
            np.random.shuffle(ones)
            np.random.shuffle(zeros)
            tried = 0
            for i in ones:

                    if tried >= 400:

                    h_cur[i] = 0.0
                    h_cur[j] = 1.0

                    h_cur[i] = 1.0

                    if c < best_val - 1e-14:

                    tried += 1
                if tried >= 400:

        if best_pair is not None:

            improved = True
            ones = np.where(h_cur == 1)[0].tolist()
            zeros = np.where(h_cur == 0)[0].tolist()
        if not improved:

    return h_cur, best_c5

def rotate_h(h, shift):
    n = len(h)
    h_rot = np.zeros(n)
    for i in range(n):
        h_rot[i] = h[(i - shift) % n]
    return h_rot
```
