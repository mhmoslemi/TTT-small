        i2 = js - k + n - 1

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:

                if abs(0.5 * n - np.sum(h)) < 1e-10:
                    break

def lse_grad_obj(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    s = (corr - cmax) / max(T, 1e-12)
    e = np.exp(np.clip(s, -500, 0))
    tot = np.sum(e)
    obj = (cmax + T * np.log(tot)) * dx
    dsum = np.sum(h) - 0.5 * n
    obj += mu * dsum * dsum
    w = e / tot
    grad = 2.0 * mu * dsum + dx * np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

        # 0<=j+lag<n => -lag<=j<n-lag

    x1 = np.arange(nn, dtype=float) * (2.0 / nn)

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    cnt = np.sum(m)
    if cnt > 0:
        h[m] += d / cnt
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                if abs(0.5 * n - np.sum(h)) < 1e-10:
                    break
    return h

def lse_fg(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    s = np.clip((corr - cm) / max(T, 1e-12), -500, 0)
    e = np.exp(s)
    te = np.sum(e)
    obj = (cm + T * np.log(te)) * dx
    ds = np.sum(h) - 0.5 * n
    obj += mu * ds * ds
    w = e / te
    grad = np.full(n, 2.0 * mu * ds)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - n + 1
        lo1 = max(0, -lag)
        hi1 = min(n - 1, n - 1 - lag)
        if lo1 <= hi1:
            js = np.arange(lo1, hi1 + 1)
            grad[js] += dx * wk * (1.0 - h[js + lag])
        lo2 = max(0, lag)
        hi2 = min(n - 1, n - 1 + lag)
        if lo2 <= hi2:
            js = np.arange(lo2, hi2 + 1)
            grad[js] -= dx * wk * h[js - lag]

                                       'gtol': 1e-12, 'maxls': 40})
        return fix_sum(r.x.copy(), n)

        return h

    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def cd_h(h, n, npass, dts):
    for _ in range(npass):
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            cur = compute_c5(h)
            bh = h
            for d in dts:
                hn = h.copy()
                hn[:] -= d / (n - 1)
                hn[j] += d
                np.clip(hn, 0.0, 1.0, out=hn)
                hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < cur - 1e-13:
                    bh, cur = hn.copy(), c
            h = bh
    return fix_sum(h, n)

def pp_h(h, n, rng, npairs):
    dts = np.array([-0.05, -0.02, -0.008, 0.008, 0.02, 0.05])

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-8) & (h < 1.0 - 1e-8)
    c = np.sum(m)
    if c > 0:
        h[m] += d / c
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in range(n):
            if 1e-6 < h[i] < 1.0 - 1e-6:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                if abs(0.5 * n - np.sum(h)) < 1e-10:
                    break
    return h

def lse_fg(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    s = np.clip((corr - cm) / max(T, 1e-12), -500, 0)
    e = np.exp(s)
    te = np.sum(e)
    obj = (cm + T * np.log(te)) * dx
    ds = np.sum(h) - 0.5 * n
    obj += mu * ds * ds
    w = e / te
    grad = np.full(n, 2.0 * mu * ds)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - n + 1
        lo1 = max(0, -lag)
        hi1 = min(n - 1, n - 1 - lag)
        if lo1 <= hi1:
            js = np.arange(lo1, hi1 + 1)
            grad[js] += dx * wk * (1.0 - h[js + lag])
        lo2 = max(0, lag)
        hi2 = min(n - 1, n - 1 + lag)
        if lo2 <= hi2:
            js = np.arange(lo2, hi2 + 1)
            grad[js] -= dx * wk * h[js - lag]
    return obj, grad

    n = len(h)
    try:
        r = optimize.minimize(lse_fg, h, args=(T, mu), jac=True,
                              method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': mi, 'ftol': 1e-18,
                                       'gtol': 1e-12, 'maxls': 40})
        return fix_sum(np.asarray(r.x), n)
    except Exception:
        return h

def up_h(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def cd_h(h, n, npass, dts):
    for _ in range(npass):
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            cur = compute_c5(h)
            bh = h
            for d in dts:
                hn = h.copy()
                hn[:] -= d / (n - 1)
                hn[j] += d
                np.clip(hn, 0.0, 1.0, out=hn)
                hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < cur - 1e-13:
                    bh, cur = hn.copy(), c
            h = bh
    return fix_sum(h, n)

def pp_h(h, n, rng, npairs, dts):
    for _ in range(npairs):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
            continue
        if h[l] <= 1e-4 or h[l] >= 1.0 - 1e-4:
            continue

        bd = 0.0

            a2, b2 = h[j] + d, h[l] - d
            if a2 < 1e-4 or a2 > 1.0 - 1e-4:
                continue
            if b2 < 1e-4 or b2 > 1.0 - 1e-4:

            hn = h.copy()
            hn[j], hn[l] = a2, b2
            c = compute_c5(hn)
            if c < cur - 1e-13:

            h[j] = min(max(h[j] + bd, 0.0), 1.0)
            h[l] = min(max(h[l] - bd, 0.0), 1.0)
    return fix_sum(h, n)
