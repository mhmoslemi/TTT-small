def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    # Adjust the most "interior" values
    interior = (h > 1e-6) & (h < 1.0 - 1e-6)
    if np.sum(interior) > 0:
        # Distribute evenly among interior points
        h[interior] += d / np.sum(interior)
        h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - 0.5*n) > 1e-10:
        # Fallback: adjust one point
        for i in np.argsort(np.abs(h - 0.5))[::-1]:
            if 0 < h[i] < 1:
                h[i] = clip_val(h[i] + d, 0, 1)
                break

def clip_val(x, lo, hi):
    return max(lo, min(hi, x))

def project_sum(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = 0.5 * n
    # Bisection on lambda
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    return h

            if j > 0:

        else:
            jj = -j
            g[jj:] += (1 - h[:n-jj])
            if jj > 0:

        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-17})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def snap_binary(h, n):
    order = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[order[:n // 2]] = 1.0
    return hb

def bit_flip_local(h, n, rng, n_rounds=10):
    h = h.copy()
    c0 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        lows = np.where(h < 0.4)[0]
        highs = np.where(h > 0.6)[0]
        if len(lows) < 2 or len(highs) < 2:
            break
        for _ in range(1500):
            i = lows[rng.randint(len(lows))]
            j = highs[rng.randint(len(highs))]
            d = min(1.0 - h[i], h[j], 0.6)
            if d < 0.1:
                continue
            h[i], h[j] = h[i] + d, h[j] - d
            c1 = compute_c5(h)
            if c1 < c0 - 1e-13:
                c0 = c1
                improved = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved:
            break
    return fix_sum(h, n)

def shift_search(h, n, max_shift=12):
    best_h, best_c = h.copy(), compute_c5(h)
    for s in range(1, max_shift + 1):
        for hs in (np.roll(h, s), np.roll(h, -s)):
            c = compute_c5(hs)
            if c < best_c - 1e-13:
                best_h, best_c = hs.copy(), c
    return best_h

def pair_fine_polish(h, n, rng, n_pairs=4000):
    h = h.copy()
    c0 = compute_c5(h)
    interior = np.where((h > 0.02) & (h < 0.98))[0]
    if len(interior) < 2:
        return h
    for _ in range(n_pairs):
        i = interior[rng.randint(len(interior))]
        j = interior[rng.randint(len(interior))]
        if i == j:
            continue
        for d in (0.025, 0.01, -0.01, -0.025):
            hi2, hj2 = h[i] + d, h[j] - d
            if not (0 <= hi2 <= 1 and 0 <= hj2 <= 1):
                continue
            h[i], h[j] = hi2, hj2
            c1 = compute_c5(h)
            if c1 < c0 - 1e-14:
                c0 = c1
                break
            h[i], h[j] = hi2 - d, hj2 + d
    return fix_sum(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 13)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = fix_sum(h, n)
    c5 = compute_c5(h)

    # Phase 1: SLSQP geometric cooling
    sched1 = [(0.04, 400), (0.015, 500), (0.006, 600), (0.002, 800), (0.0008, 1000), (0.0003, 1200)]
    for T, it in sched1:
        if time.time() - t0 > budget_s - 550:
            break
        h = slsqp_opt(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 2: Binary snap + bit-flip + SLSQP relaxation
    if time.time() - t0 < budget_s - 420:
        hb = snap_binary(h, n)
        hb = bit_flip_local(hb, n, rng, n_rounds=12)
        hb = slsqp_opt(hb, n, 0.01, 400)
        hb = slsqp_opt(hb, n, 0.003, 600)
        hb = slsqp_opt(hb, n, 0.001, 800)
        hb = fix_sum(hb, n)
        cb = compute_c5(hb)
        if cb < c5:
            h, c5 = hb, cb
            h = shift_search(h, n, 8)
            c5 = compute_c5(h)

    # Phase 3: Shift + fine SLSQP
    if time.time() - t0 < budget_s - 350:
        h = shift_search(h, n, 12)
        c5 = compute_c5(h)
    for T, it in [(0.0005, 1000), (0.00015, 1200), (0.00005, 1500)]:
        if time.time() - t0 > budget_s - 280:
            break
        h = slsqp_opt(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 4: Pair fine polish
    if time.time() - t0 < budget_s - 220:
        h = pair_fine_polish(h, n, rng, n_pairs=5000)
        c5 = compute_c5(h)

    # Phase 5: Diverse structural restarts
    best_h, best_c5 = h.copy(), c5
    for i in range(14):
        if time.time() - t0 > budget_s - 160:
            break
        ht = best_h.copy()
        mode = i % 5
        if mode == 0:
            ht = np.roll(ht, rng.randint(1, 12))
        elif mode == 1:
            ht += np.random.randn(n) * 0.012
        elif mode == 2:
            s = rng.randint(0, max(1, n - 25))
            ln = rng.randint(8, min(25, n - 8))
            ht[s:s+ln] = 1.0 - ht[s:s+ln]
        elif mode == 3:
            ht = snap_binary(ht, n)
            idx = rng.choice(n, 8, replace=False)
            ht[idx] += rng.uniform(-0.3, 0.3, 8)
        else:
            ht = 0.5 * ht + 0.5 * np.roll(ht, rng.randint(2, 6))
        ht = fix_sum(ht, n)
        ht = slsqp_opt(ht, n, 0.004, 350)
        ht = slsqp_opt(ht, n, 0.001, 500)
        ht = fix_sum(ht, n)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_h, best_c5 = ht, ct
            best_h = pair_fine_polish(best_h, n, rng, 2000)
            best_h = shift_search(best_h, n, 6)
            best_c5 = compute_c5(best_h)
    h, c5 = best_h, best_c5

    # Phase 6: Final SLSQP
    for T, it in [(0.00003, 2000), (0.00001, 2500)]:
        if time.time() - t0 > budget_s - 50:
            break
        h = slsqp_opt(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 7: Try alternative resolutions
    if time.time() - t0 < budget_s - 90:
        h2 = upsample(h, 400)
        for T, it in [(0.008, 400), (0.002, 500), (0.0005, 700)]:
            if time.time() - t0 > budget_s - 30:
                break
            h2 = slsqp_opt(h2, 400, T, it)
            h2 = fix_sum(h2, 400)
        c2 = compute_c5(h2)
        if c2 < c5:
            h, n, c5 = h2, 400, c2

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
