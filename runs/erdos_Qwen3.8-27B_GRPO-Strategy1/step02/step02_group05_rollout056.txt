```python
    wc_arr = np.zeros(L)
```

```python
        else:
```

Let me also try a different approach: **direct optimization of the max using a min-max game**. 

The problem is: minimize_h max_k f_k(h) where f_k(h) = sum_i h[i]*(1-h[i+k])*dx

This is a minimax problem. We can solve it as:
minimize_h max_{p ≥ 0, ||p||_1 = 1} sum_k p_k * f_k(h)

By von Neumann's minimax theorem (if the feasible set for h is compact and convex, and the p simplex is compact):
min_h max_p sum_k p_k f_k(h) = max_p min_h sum_k p_k f_k(h)

For a fixed p, min_h sum_k p_k f_k(h) is a quadratic program (since f_k is quadratic in h). We can solve this with projected gradient or even analyically in some cases.

But this is a game, and we'd need to iterate. In practice, we can use:
- Inner loop: optimize h for a fixed p (with momentum)
- Outer loop: update p towards the active lags (the ones achieving the max)

This is essentially what the soft-max does! So it's not fundamentally different.

```python
import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    """Project h onto {x : 0<=x<=1, sum(x)=target}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = float(h.sum())
        if abs(s - target) < 1e-12:
            break
        h -= (s - target) / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final bisection for exactness
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_direct(w, h, N):
    """Direct gradient computation (O(N * n_active))."""
    grad = np.zeros(N)
    center = N - 1
    thresh = 1e-6
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad

def nesterov_opt(h, N, target, deadline, tau0, tau1, lr0, min_lr, n_iter_max):
    """Nesterov-accelerated gradient with soft-max."""
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/N)
    dx = 2.0 / N
    n = N
    # Nesterov: maintain y = x + momentum
    x = h.copy()
    y = h.copy()
    t_val = 1.0
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        t_val += 1.0
        beta = (1.0 - t_val) / (1.0 + t_val + 2.0)  # Nesterov momentum
        tau = tau0 * (tau1 / tau0) ** (it / max(n_iter_max, 1))
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = y.copy()
        
        # Soft-max weights
        tc = tau * corr
        tc -= tc.max()
        w = np.exp(tc)
        w /= w.sum()
        
        grad = gradient_direct(w, y, N)
        
        # Nesterov update
        lr = lr0 / (1.0 + 0.01 * it)
        x_new = y - lr * grad
        y_new = x_new + beta * (x_new - x)
        x = x_new
        y = project_sum(y_new, target)
        
        if lr < min_lr:
            break
    return best_h, best_c5

def adam_opt(h, N, target, deadline, tau0, tau1, lr0, decay, decay_period=50):
    """Adam optimizer (same as before but with slight tweaks)."""
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/N)
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    x = h.copy()
    while time.time() < deadline:
        np_b1 = 1.0 - b1
        np_b2 = 1.0 - b2
        frac = min(it / 3000.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        corr = np.correlate(x, 1.0 - x, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = x.copy()
        tc = tau * corr
        tc -= tc.max()
        w = np.exp(tc)
        w /= w.sum()
        grad = gradient_direct(w, x, N)
        m = np_b1 * m + (1.0 - np_b1) * grad
        v = np_b2 * v + (1.0 - np_b2) * grad * grad
        it += 1
        b1_hat = 1.0 - b1**it
        b2_hat = 1.0 - b2**it
        x = x - lr * m/b1_hat / (np.sqrt(v/b2_hat) + eps)
        x = project_sum(x, target)
        if it % decay_period == 0:
            lr *= decay
            if lr < 1e-10:
                break
    return best_h, best_c5

def coordinate_descent(h, N, target, n_sweeps, deadline):
    """Coordinate descent polish."""
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    this_c5 = best_c5
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(N):
            if time.time() > deadline:
                break
            # Try a few values for h[j]
            old_val = h[j]
            # Candidates: current, 0, 1, and a few linear interpolations
            candidates = np.array([old_val, 0.0, 1.0,
                                   max(0, old_val - 0.1), min(1, old_val + 0.1),
                                   max(0, old_val - 0.05), min(1, old_val + 0.05)])
            best_local = old_val
            best_local_c5 = this_c5
            for cv in candidates:
                h[j] = cv
                # Adjust for sum constraint: redistribute difference to others
                diff = target - h.sum()
                if abs(diff) > 1e-10:
                    # Add diff/N to all other elements
                    others = np.ones(N)
                    others[j] = 0
                    n_others = others.sum()
                    h_adj = h + (diff / n_others) * others
                    h_adj = np.clip(h_adj, 0, 1)
                    # Re-project
                    h_adj = project_sum(h_adj, target)
                c5_t = compute_c5(h, dx) if abs(diff) < 1e-10 else compute_c5(h_adj, dx)
                if c5_t < best_local_c5 - 1e-12:
                    best_local_c5 = c5_t
                    best_local = cv
                    if abs(diff) < 1e-10:
```
