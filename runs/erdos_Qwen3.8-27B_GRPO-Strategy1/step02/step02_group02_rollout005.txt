import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, t, topk):
    """Compute smooth topk objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if topk and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        idx = np.arange(len(corr))
        c = corr
    
    mx = np.max(c)
    s = t * (c - mx)
    s -= np.max(s)
    exp_s = np.exp(s)
    log_sum = np.log(np.sum(exp_s))
    obj = (mx + log_sum / t) * dx
    w = exp_s / np.sum(exp_s)
    
    # Gradient

                grad[k:] += w[i] * (1.0 - h[:n - k])
                grad[:n - k] += w[i] * (-h[k:n])
    grad *= dx
    return obj, grad

def proj(h, n):
    h = h.copy()

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)

        c = corr[idx]

    ts = es.sum()
    obj = (mx + np.log(ts) / t) * dx

        else:
            k = -lag
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] -= wi * h[k:n]

    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    tgt = n / 2.0
    for _ in range(150):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_grad(h, t, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    if topk and topk < m:
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        idx = np.arange(m)
        c = corr
    mx = c.max()
    s = t * (c - mx)
    s -= s.max()
    es = np.exp(s)
    ts = es.sum()
    obj = (mx + np.log(ts) / t) * dx
    w = es / ts
    grad = np.zeros(n)
    for i in range(len(idx)):
        wi = w[i]
        if wi < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += wi * (1.0 - h[lag:n])
            if lag > 0:
                grad[lag:] -= wi * h[:n - lag]
        else:
            k = -lag
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] -= wi * h[k:n]
    grad *= dx
    return obj, grad

def obj_wrap(x, t, topk):
    o, _ = obj_grad(x, t, topk)
    return o

def jac_wrap(x, t, topk):
    _, g = obj_grad(x, t, topk)
    return g

def ceq_fun(x):
    return np.sum(x) - len(x) / 2.0

def ceq_jac(x):
    return np.ones(len(x))

def slsqp_call(h0, t, topk, maxiter, ftol, bnds):
    constr = {'type': 'eq', 'fun': ceq_fun, 'jac': ceq_jac}
    r = minimize(obj_wrap, h0, jac=jac_wrap, args=(t, topk),
                 method='SLSQP', bounds=bnds, constraints=constr,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), len(h0))

def binary_local_search(h_bin, n, max_iters, rng, target_corr):
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = corr.max()
    if best_c <= target_corr:
        return h
    stagnation = 0
    for it in range(max_iters):
        if best_c <= target_corr:
            return h
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        worst_idx = corr.argmax()
        lag = worst_idx - (n - 1)
        improved = False
        for _ in range(10):
            if lag >= 0:
                active = [i for i in ones if i + lag < n and h[i + lag] < 0.5]
            else:
                k = -lag
                active = [i for i in ones if i >= k and h[i - k] < 0.5]
            if not active:
                i = ones[rng.randint(len(ones))]
            else:
                i = active[rng.randint(len(active))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            nc = new_corr.max()
            if nc < best_c - 1e-10:
                corr = new_corr
                best_c = nc
                improved = True
                if best_c <= target_corr:
                    return h
            else:
                h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        if not improved:
            stagnation += 1
            if stagnation > 5:
                break
        else:
            stagnation = 0
    return h
