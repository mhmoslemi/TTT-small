def exact_coord_min(h, p):
    """Find the best value of h[p] to minimize C5, given all other coordinates fixed."""
    n = len(h)
    dx = 2.0 / n
    # Current correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute slope A_k for each k
    # m_k = k - (n-1)
    m = np.arange(len(corr)) - (n - 1)  # shape (2n-1,)
    
    # A_k = (1 - h[p - m_k]) - h[p + m_k], with 0 for out-of-range
    idx1 = p - m  # indices for h[p - m_k]
    idx2 = p + m  # indices for h[p + m_k]
    
    A = np.zeros(len(corr))
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    A[valid1] += (1.0 - h[idx1[valid1]])
    A[valid2] -= h[idx2[valid2]]
    
    # B_k = corr[k] - A_k * h[p]

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        interior = (h > 0) & (h < 1)
        if np.sum(interior) > 0:
            h[interior] += d2 / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def exact_coord_min(h, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_k = np.arange(len(corr)) - (n - 1)
    idx1 = p - m_k
    idx2 = p + m_k
    A = np.zeros(len(corr))
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    A[v1] += (1.0 - h[idx1[v1]])
    A[v2] -= h[idx2[v2]]
    B = corr - A * h[p]
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(50):
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = np.max(A * c1 + B)
        f2 = np.max(A * c2 + B)
        if f1 < f2:
            hi = c2
        else:
            lo = c1
    return (lo + hi) / 2.0

def coord_descend(h, n, max_pass=30, time_limit=None, t0=None):
    for pp in range(max_pass):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        improved = False
        for p in range(n):
            if time_limit is not None and time.time() - t0 > time_limit:
                break
            old = h[p]
            new_val = exact_coord_min(h, p)
            if abs(new_val - old) > 1e-9:
                h[p] = new_val
                improved = True
        h = fix_sum(h, n)
        if not improved:
            break
    return h

def slsqp_step(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-16})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)
