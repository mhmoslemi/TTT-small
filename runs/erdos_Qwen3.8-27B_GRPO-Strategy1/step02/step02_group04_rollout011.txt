```python
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Distribute evenly among interior points
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12:
                    break
    return h

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def do_slsqp(h, n, T, max_iter):
    try:
        res = opt.minimize(
            soft_max_obj, h.copy(), args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-16}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def binary_project(h, n):

    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(np.sum(h_bin))

        ones = np.where(h_bin == 1)[0]
        # Remove the ones with lowest h values
        order = np.argsort(h[ones])
        h_bin[ones[order[:n_ones - target]]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[order[:target - n_ones]]] = 1
    return h_bin
```

```python
def soft_max_grad(h, T):
    """Compute gradient of soft-max objective"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        # Hard max: gradient is just the gradient of the max element
        max_idx = np.argmax(corr)
        d = max_idx - n + 1  # shift
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j - d < n:
                grad[j] += (1 - h[j - d])
            if 0 <= j + d < n:
                grad[j] -= h[j + d]
        return grad * dx
    
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    exp_vals = np.exp(shifted / T)
    p = exp_vals / np.sum(exp_vals)  # probabilities over shifts
    
    grad = np.zeros(n)
    # For each shift d, compute contribution
    # d ranges from -(n-1) to (n-1), indexed as k = n-1+d
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        d = k - n + 1
        for j in range(n):
            jm = j - d  # m = j - d, check 0 <= m < n
            jp = j + d
            if 0 <= jm < n:
                grad[j] += p[k] * (1 - h[jm])
            if 0 <= jp < n:
                grad[j] -= p[k] * h[jp]
    
    return grad * dx
```

```python
        grad[lo2:hi2] -= h[max(0,d):n-d]
        return grad * dx
    
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    exp_vals = np.exp(shifted / T)
    p = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-12:
            continue
        d = k - n + 1
        if d >= 0:
            # (1-h[j-d]) for j=d,...,n-1 → h[j-d] for j-d=0,...,n-1-d
            grad[d:] += w * (1 - h[:n-d])
            # -h[j+d] for j=0,...,n-1-d → h[j+d] for j+d=d,...,n-1
            grad[:n-d] -= w * h[d:]
        else:
            dd = -d  # dd > 0
            # (1-h[j+dd]) for j=0,...,n-1-dd
            grad[:n-dd] += w * (1 - h[dd:])
            # -h[j-dd] for j=dd,...,n-1
            grad[dd:] -= w * h[:n-dd]
    
    return grad * dx
```

```python
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12: break
    else:
        diff = -diff
        idx = np.argsort(-h)
        for i in idx:
            rem = min(diff, h[i])
            h[i] -= rem
            diff -= rem
            if diff <= 1e-12: break
    return h

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    if T < 1e-15:
        return c_max * dx
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    s = float(np.sum(np.exp(shifted / T)))
    return (c_max + T * np.log(max(s, 1e-300))) * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    if T < 1e-15:
        max_idx = int(np.argmax(corr))
        d = max_idx - n + 1
        grad = np.zeros(n)
        if d >= 0:
            grad[d:] += (1.0 - h[:n - d])
            grad[:n - d] -= h[d:]
        else:
            dd = -d
            grad[:n - dd] += (1.0 - h[dd:])
            grad[dd:] -= h[:n - dd]
        return grad * dx
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    exp_vals = np.exp(shifted / T)
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-12: continue
        d = k - n + 1
        if d >= 0:
            grad[d:] += w * (1.0 - h[:n - d])
            grad[:n - d] -= w * h[d:]
        else:
            dd = -d
            grad[:n - dd] += w * (1.0 - h[dd:])
            grad[dd:] -= w * h[:n - dd]
    return grad * dx

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def do_slsqp(h, n, T, max_iter):
    try:
        res = opt.minimize(soft_max_obj, h.copy(), args=(T,),
                           method='SLSQP', bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                           options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def projected_grad_descent(h, n, T0, time_limit):
    t0 = time.time()

    best_h = h_cur.copy()

    alpha = 0.01
    stall = 0

                alpha = 0.01
                if T < 1e-9: break

def binary_project(h, n):
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        h_bin[ones[order[:n_ones - target]]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[order[:target - n_ones]]] = 1
    return h_bin

def binary_hill_climb(h_bin, n, max_rounds, trials, patience):
    rng = np.random.default_rng(12345)
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    no_imp = 0
    for r in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        ones_i = np.where(h == 1)[0]
        zeros_i = np.where(h == 0)[0]
        found = False
        for t in range(trials):
            i = ones_i[rng.integers(len(ones_i))]
            j = zeros_i[rng.integers(len(zeros_i))]
            h[i] = 0; h[j] = 1
            new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if new_max < cur_max:
                cur_max = new_max
                found = True
                no_imp = 0
            else:
                h[i] = 1; h[j] = 0
        if found and cur_max * (2.0 / n) < best_c5:
            best_c5 = cur_max * (2.0 / n)
        if not found:
            no_imp += 1
            if no_imp >= patience: break
    return h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)
```
