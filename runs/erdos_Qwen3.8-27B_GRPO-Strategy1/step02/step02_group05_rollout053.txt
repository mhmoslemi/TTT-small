    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    sizes = sorted(set(s for s in [n0, 2 * n0, 572, 760] if s % 2 == 0 and 40 <= s <= 1200))

    for N in sizes:
        if time.time() > deadline - 150:
            break
        dx = 2.0 / N
        target = float(N) / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)
        remain = deadline - time.time()
        if remain < 80:
            break

        # Phase 1: broad balancing (high k, low tau)
        p1_dead = time.time() + remain * 0.30
        h1, c1 = run_phase(h_init.copy(), N, target, dx, p1_dead,
                           topk0=60, topk1=25, tau0=3.0, tau1=20.0,
                           lr0=0.06, lr1=0.002, total_est=1500)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

        # Phase 2: medium focus
        p2_dead = time.time() + (deadline - time.time()) * 0.40
        if time.time() < deadline - 60:
            h2, c2 = run_phase(h1.copy(), N, target, dx, p2_dead,
                               topk0=20, topk1=8, tau0=30.0, tau1=120.0,
                               lr0=0.02, lr1=0.0005, total_est=1500)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
        else:
            h2 = h1

        # Phase 3: fine focus on worst lags
        if time.time() < deadline - 40:
            p3_dead = deadline - 30
            h3, c3 = run_phase(h2.copy(), N, target, dx, p3_dead,
                               topk0=8, topk1=3, tau0=100.0, tau1=500.0,
                               lr0=0.008, lr1=0.0001, total_est=1000)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()
            h2 = h3

        # Random restart if time allows
        if time.time() < deadline - 100 and remain > 300:
            h_r = h2 + rng.randn(N) * 0.06
            h_r = project_sum(h_r, target)
            p4_dead = deadline - 30
            h4, c4 = run_phase(h_r, N, target, dx, p4_dead,
                               topk0=30, topk1=5, tau0=10.0, tau1=200.0,
                               lr0=0.03, lr1=0.001, total_est=1200)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()

    # Coordinate descent polish on best solution
    Nb = len(best_h)
    dx_b = 2.0 / Nb
    if time.time() < deadline - 10:
        hf, cf = coord_polish(best_h.copy(), Nb, dx_b,
                              max_sweeps=40, deadline=deadline - 5)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    # Final gradient polish
    if time.time() < deadline - 3:
        cfg_dead = deadline - 1
        hf2, cf2 = run_phase(best_h.copy(), Nb, Nb / 2.0, dx_b, cfg_dead,
                             topk0=4, topk1=2, tau0=300.0, tau1=1500.0,
                             lr0=0.003, lr1=0.00005, total_est=500)
        if cf2 < best_c5:
            best_c5 = cf2
            best_h = hf2

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
