def project_box_sum(h, target_sum):

    h = np.clip(h, 0, 1).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum by diff
            # Prioritize elements with most room (1-h[i])
            room = 1.0 - h
            # Add diff distributed proportional to room
            total_room = room.sum()
            if total_room < 1e-14:
                break
            h += room * (diff / total_room)
        else:
            # Need to decrease sum by |diff|
            # Prioritize elements with most value (h[i])
            total_val = h.sum()
            if total_val < 1e-14:
                break
            h += h * (diff / total_val)  # diff is negative
        h = np.clip(h, 0, 1)
    return h

    w = exp_s / Z

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = T * (corr - m)
    exp_s = np.exp(shifted)
    Z = np.sum(exp_s)
    obj = (m + np.log(Z) / T) * dx
    w = exp_s / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        i1 = max(0, -off)
        i1e = min(n, n - off)
        if i1e > i1:
            grad[i1:i1e] += wk * (1.0 - h[i1+off:i1e+off])
        i2 = max(0, off)
        i2e = min(n, n + off)
        if i2e > i2:
            grad[i2:i2e] -= wk * h[i2-off:i2e-off]
    grad *= dx
    return obj, grad

def project_box_sum(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += room * (diff / tr)
        else:
            tv = h.sum()
            if tv < 1e-14:
                break
            h += h * (diff / tv)
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_grad_desc(h, T, max_iter=400, tol=1e-13):
    n = len(h)
    target = n / 2.0
    h = project_box_sum(h, target)
    best_obj = compute_c5(h)
    best_h = h.copy()
    obj, grad = smooth_obj_grad(h, T)
    for it in range(max_iter):
        gnorm = np.linalg.norm(grad)
        if gnorm < tol:
            break
        lr = min(2.0 / max(gnorm, 1e-10), 2.0 / (T * 0.3 + 1.0))
        accepted = False
        for _ in range(40):
            h_new = h - lr * grad
            h_new = project_box_sum(h_new, target)
            new_obj, new_grad = smooth_obj_grad(h_new, T)
            if new_obj < obj + 1e-4 * lr * (grad @ grad):
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            break
        h = h_new
        obj = new_obj
        grad = new_grad
        true_obj = compute_c5(h)
        if true_obj < best_obj:
            best_obj = true_obj
            best_h = h.copy()
    return best_h

def lbfgs_polish(h, T=8000, maxiter=600):
    n = len(h)
    target = n / 2.0
    lam = 200.0

    def penalty_obj(x):
        hc = np.clip(x, 0.0, 1.0)
        base, _ = smooth_obj_grad(hc, T)
        sp = lam * (x.sum() - target) ** 2
        bp = lam * (np.sum(np.maximum(0.0, x - 1.0)) + np.sum(np.maximum(0.0, -x)))
        return base + sp + bp

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalty_obj, h, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 30})
        return project_box_sum(np.clip(res.x, 0.0, 1.0), target)
    except Exception:
        return h

def erdos_eq_fun(x, n):
    return x[:n].sum() - n / 2.0

def erdos_ineq_fun(x, n):
    h_x = x[:n]
    t = x[-1]
    corr = compute_corr(h_x)
    return t - np.max(corr)

def slsqp_final(h, maxiter=500):
    n = len(h)
    corr = compute_corr(h)
    t0 = float(np.max(corr)) + 0.005
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': lambda x: erdos_eq_fun(x, n)},
        {'type': 'ineq', 'fun': lambda x: erdos_ineq_fun(x, n)}
    ]
    try:
        res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_box_sum(np.clip(res.x[:n], 0.0, 1.0), n / 2.0)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):

    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        best_h = np.zeros(90)
        best_h[:45] = 1.0
    best_n = len(best_h)
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient with T annealing
    for T in [30, 100, 500, 2000, 10000, 50000]:
        if elapsed() > budget_s * 0.20:
            break

        if c5_new < best_c5:

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Multi-resolution search
    resolutions = [60, 72, 80, 90, 100, 120]
    for n_try in resolutions:
        if elapsed() > budget_s * 0.62:
            break
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_base = np.interp(x_new, x_old, best_h)
            h_base = project_box_sum(h_base, n_try / 2.0)

        n_trials = 6 if n_try <= 90 else 4

            h_pert = project_box_sum(h_pert, n_try / 2.0)

            h_opt = lbfgs_polish(h_opt, T=8000, maxiter=400)
            c5_opt = compute_c5(h_opt)

                best_h = h_opt.copy()
                best_n = n_try
                h_base = h_opt.copy()

    # Phase 4: Diverse structured perturbations
    if elapsed() < budget_s * 0.75:
        for pattern_type in range(4):
            if elapsed() > budget_s * 0.75:
                break
            h_struct = best_h.copy()
            if pattern_type == 0:
                mask = np.arange(best_n) % 3 == 0
                h_struct[mask] = np.clip(h_struct[mask] + 0.15, 0, 1)
            elif pattern_type == 1:

            elif pattern_type == 2:

            else:
                h_struct = np.clip(1.0 - best_h + 0.3 * np.random.randn(best_n), 0, 1)
            h_struct = project_box_sum(h_struct, best_n / 2.0)
            h_opt = projected_grad_desc(h_struct, 1000, max_iter=300)
            h_opt = lbfgs_polish(h_opt, T=8000, maxiter=400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 5: Rotations
    if elapsed() < budget_s * 0.82:
        for shift in [1, 2, 5, 11]:
            if elapsed() > budget_s * 0.82:
                break
            h_rot = np.roll(best_h, shift)
            h_opt = projected_grad_desc(h_rot, 3000, max_iter=300)
            h_opt = lbfgs_polish(h_opt, T=10000, maxiter=300)

            best_h = h_sqp

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
