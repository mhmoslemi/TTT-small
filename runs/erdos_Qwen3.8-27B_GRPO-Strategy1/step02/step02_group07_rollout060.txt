    if n_active < 3 or n_active > len(corr):
        return do_slsqp(h, n, T, iters)
    active_idx = np.where(active_mask)[0]
    def obj_act(x):
        c = get_corr(x)
        sub = c[active_idx]
        mx = sub.max()
        if T < 1e-15:
            return mx
        s = np.clip((sub - mx) / T, -50, 0)
        return mx + T * np.log(np.exp(s).sum())
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(obj_act, h, method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_c, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-19})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def structured_restart(h, n, rng, mode):
    ht = h.copy()
    if mode == 0:
        ht += rng.randn(n) * 0.015
    elif mode == 1:
        idx = rng.choice(n, max(5, n // 30), replace=False)
        ht[idx] += rng.randn(len(idx)) * 0.08
    elif mode == 2:
        ht += rng.randn(n) * 0.005
    elif mode == 3:
        k = rng.randint(n // 4, n // 2)
        ht[k:] += rng.uniform(-0.02, 0.02)
    else:
        ht = h.copy()
        for _ in range(n // 20):
            j, l = rng.choice(n, 2, replace=False)
            d = rng.uniform(-0.05, 0.05)
            ht[j], ht[l] = np.clip(ht[j]+d,0,1), np.clip(ht[l]-d,0,1)
    return fix_sum(ht, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 31)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    c5 = compute_c5(h)

    # Phase 1: Progressive SLSQP with active-set focus
    sched1 = [(0.015, 400), (0.005, 500), (0.002, 600), (0.0008, 700)]
    for i, (T, it) in enumerate(sched1):
        if time.time() - t0 > budget_s - 400:
            break
        if i >= 2:
            h = active_slsqp(h, n, T, it)
        else:
            h = do_slsqp(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 2: Coordinate + pair polish
    if time.time() - t0 < budget_s - 300:
        h = coord_desc(h, n, 4, (-0.03, -0.01, -0.003, 0.003, 0.01, 0.03))
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 250:
        h = pair_desc(h, n, rng, 600)
        c5 = compute_c5(h)

    # Phase 3: Structured restarts
    best_h, best_c5 = h.copy(), c5
    for trial in range(20):
        if time.time() - t0 > budget_s - 150:
            break
        mode = trial % 5
        ht = structured_restart(best_h, n, rng, mode)
        ht = do_slsqp(ht, n, 0.004, 400)
        ht = do_slsqp(ht, n, 0.001, 500)
        ht = coord_desc(ht, n, 3, (-0.01, -0.003, 0.003, 0.01))
        ct = compute_c5(ht)
        if ct < best_c5:
            best_h, best_c5 = ht, ct
            h = ht
            c5 = ct
            if time.time() - t0 < budget_s - 120:
                h = pair_desc(h, n, rng, 300)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()

    # Phase 4: Deep SLSQP refinement on best
    h = best_h.copy()
    for T, it in [(0.0004, 1000), (0.0001, 1200), (0.00003, 1500)]:
        if time.time() - t0 > budget_s - 50:
            break
        h = active_slsqp(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 5: Ultra-fine coordinate descent
    h = best_h.copy()
    if time.time() - t0 < budget_s - 20:
        h = coord_desc(h, n, 8, (-0.01, -0.003, -0.0008, 0.0008, 0.003, 0.01))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    if time.time() - t0 < budget_s - 10:
        h = best_h.copy()
        h = coord_desc(h, n, 10, (-0.004, -0.001, -0.0003, 0.0003, 0.001, 0.004))
        h = pair_desc(h, n, rng, 400, (-0.01, -0.003, 0.003, 0.01))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    h = fix_sum(best_h, len(best_h))
    return (h, compute_c5(h), len(h))
