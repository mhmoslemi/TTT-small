from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    obj = (m + np.log(e.sum()) / T) * dx
    
    grad = np.zeros(n)
    for k in range(n):
        a = 0.0
        b = 0.0
        for idx in range(len(w)):
            s = idx - (n - 1)
            if 0 <= k + s < n:
                a += w[idx] * (1.0 - h[k + s])
            if 0 <= k - s < n:
                b += w[idx] * h[k - s]
        grad[k] = dx * (a - b)
    return obj, grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t < 1e-14:
            return h.copy()
        h = h + room * (diff / t)
    else:
        t = h.sum()
        if t < 1e-14:
            return h.copy()
        h = h + h * (diff / t)
    return np.clip(h, 0.0, 1.0)

def lbfgs_opt(h0, T, n, maxiter=200):
    def fun(x):
        return smooth_obj(x, T)
    def jac(x):
        return smooth_obj_grad(x, T)[1]
    res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                   bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return project(np.clip(res.x, 0, 1), n)

    res = minimize(lambda x: smooth_obj(x, T), h0, args=(T,),

                  options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)
