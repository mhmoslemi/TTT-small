import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    # Bisection
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    residual = target - np.sum(h)
    interior = (h > 1e-12) & (h < 1.0 - 1e-12)
    if np.sum(interior) > 0:
        h[interior] += residual / np.sum(interior)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_active(h, active_idx, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_idx:
        s = k - (n - 1)
        if s >= 0:
            t = np.arange(n - s)
            grad[t] += (1.0 - h[t + s])
            grad[t + s] += (-h[t])
        else:
            ss = -s
            t = np.arange(ss, n)
            grad[t] += (1.0 - h[t - ss])
            grad[t - ss] += (-h[t])
    grad *= dx / len(active_idx)
    return grad

def projected_step(h, grad, step, n):
    h_new = h - step * grad
    h_new = fix_sum(h_new, n)
    return h_new

def upsample(h, new_n):
    old_n = len(h)
    # Use periodic interpolation
    x_old = np.concatenate([np.arange(old_n) * (2.0/old_n), [2.0]])
    h_old_ext = np.concatenate([h, [h[0]]])
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h_old_ext)
    return fix_sum(h_new, new_n)

    """Projected gradient with momentum and adaptive step"""
    h = h.copy()
    momentum = np.zeros(n)
    beta = 0.7
    step = 0.05
    best_c5 = compute_c5(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(70):

        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += r / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_single_shift(h, k, n):
    """Gradient of correlation[k] w.r.t. h"""
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        t = np.arange(n - s)
        g[t] += (1.0 - h[t + s])
        g[t + s] -= h[t]
    else:
        ss = -s
        t = np.arange(ss, n)
        g[t] += (1.0 - h[t - ss])
        g[t - ss] -= h[t]
    return g

def grad_active(h, active, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        g += grad_single_shift(h, k, n)
    return g * dx / len(active)
