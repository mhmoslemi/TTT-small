def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_and_grad(h, T):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    w = np.exp(shifted / T)
    w /= np.sum(w)
    val = (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx
    
    # Gradient
    m = 2 * n - 1  # length of corr
    grad = np.zeros(n)
    for t in range(n):
        g = 0.0
        for mm in range(m):
            lag = mm - (n - 1)
            # Contribution 1: h[t] appears as h[j+lag] when j = t-lag
            if 0 <= t - lag < n:
                g += w[mm] * (1.0 - h[t - lag])
            # Contribution 2: h[t] appears as h[j] when j = t
            if 0 <= t + lag < n:
                g += w[mm] * (-h[t + lag])
        grad[t] = g * dx
    return val, grad

def proj_constraint(h, n):

    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0

            hi = mid
        else:

    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)

    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):
        if time.time() - t_start > budget_s - reserve_s:
            break
        val, grad = soft_max_and_grad(h, T)
        # Adaptive step size
        h_new = h - lr * grad
        h_new = proj_constraint(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= val:
            h = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return best_h, best_c

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return proj_constraint(h_new, new_n)

def slsqp_optimize(h0, n, T, max_iter):

        c_max = np.max(corr)

    res = opt.minimize(obj_grad, h0, method='SLSQP',

                       options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})

def bimodal_push(h, n, threshold=0.3):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    shifted = corr - c_max
    if T < 1e-6:
        w = np.zeros(m)
        w[np.argmax(corr)] = 1.0
        val = c_max * dx
    else:
        w = np.exp(shifted / T)
        w /= w.sum()
        val = (c_max + T * np.log(w.sum())) * dx
    t_arr = np.arange(n)
    m_arr = np.arange(m)
    lag = m_arr - (n - 1)
    t_col = t_arr[:, None]
    lag_row = lag[None, :]
    idx1 = t_col - lag_row
    idx2 = t_col + lag_row
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1_c = np.clip(idx1, 0, n - 1)
    idx2_c = np.clip(idx2, 0, n - 1)
    w_row = w[None, :]
    part1 = w_row * (1.0 - h[idx1_c]) * valid1
    part2 = w_row * h[idx2_c] * valid2
    grad = (part1 - part2).sum(axis=1) * dx
    return val, grad

def proj_constraint(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = int(np.sum(interior))
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def proj_grad_desc(h0, n, T, lr_init, max_iter, deadline):
    h = h0.copy()
    h_best = h.copy()
    c_best = compute_c5(h)
    lr = lr_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        val, grad = soft_max_grad(h, T)
        h_new = proj_constraint(h - lr * grad, n)
        c_new = compute_c5(h_new)
        if c_new < c_best:
            c_best = c_new
            h_best = h_new.copy()
            stall = 0
            lr = min(lr * 1.06, 0.4)
        else:
            stall += 1
            lr *= 0.72
            if lr < 1e-8 or stall > 60:
                break
        h = h_new
    return h_best, c_best

def _sum_obj(x, n):
    return np.sum(x) - n * 0.5

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_run(h0, n, T, max_iter):
    def obj(x):
        return soft_max_val(x, T)
    def objg(x):
        return soft_max_grad(x, T)
    cons = [{'type': 'eq', 'fun': _sum_obj, 'jac': _sum_jac, 'args': (n,)}]
    res = opt.minimize(objg, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
    return proj_constraint(res.x, n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return proj_constraint(np.interp(x_new, x_old, h), new_n)

def bimodal_push(h, n, thresh):
    h_new = h.copy()
    h_new[h_new >= 1.0 - thresh] = 1.0
    h_new[h_new <= thresh] = 0.0
    mid = (h_new > thresh) & (h_new < 1.0 - thresh)
    h_new[mid] = 0.5 * (1.0 + np.tanh(5.0 * (h_new[mid] - 0.5)))
    return proj_constraint(h_new, n)

    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Projected gradient at parent resolution (fast, many iters)
    for T in [0.04, 0.015, 0.005, 0.002, 0.0008]:
        deadline = t_start + budget_s - 300
        h_pg, c_pg = proj_grad_desc(h_best, n_best, T, 0.15, 800, deadline)
        if c_pg < c5_best:
            h_best, c5_best = h_pg, c_pg

    # Phase 2: SLSQP polish at parent resolution
    if time.time() < t_start + budget_s - 200:
        h_opt = slsqp_run(h_best, n_best, 0.003, 1200)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 3: Upsample and optimize at higher resolutions
    for n_try in [400, 448, 512]:
        if time.time() > t_start + budget_s - 300:
            break
        h_up = upsample(h_best, n_try)
        deadline = t_start + budget_s - 250
        h_pg, c_pg = proj_grad_desc(h_up, n_try, 0.015, 0.12, 500, deadline)
        if c_pg < c5_best:
            h_best, c5_best, n_best = h_pg, c_pg, n_try
        if time.time() < t_start + budget_s - 180:
            h_opt = slsqp_run(h_best, n_best, 0.004, 700)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best, n_best = h_opt, c, n_best
        if time.time() < t_start + budget_s - 130:
            h_pg, c_pg = proj_grad_desc(h_best, n_best, 0.002, 0.06, 600, t_start + budget_s - 120)
            if c_pg < c5_best:
                h_best, c5_best, n_best = h_pg, c_pg, n_best

    # Phase 4: Bimodal exploration at current best resolution
    if time.time() < t_start + budget_s - 150:
        for thresh in [0.12, 0.22, 0.35]:
            if time.time() > t_start + budget_s - 120:
                break
            h_bim = bimodal_push(h_best, n_best, thresh)
            deadline = t_start + budget_s - 100
            h_pg, c_pg = proj_grad_desc(h_bim, n_best, 0.005, 0.08, 400, deadline)
            if c_pg < c5_best:
                h_best, c5_best = h_pg, c_pg

    # Phase 5: Diverse multi-start
    rng = np.random.RandomState(seed)
    inits = []
    h_alt = np.tile([1.0, 0.0], n_best // 2 + 1)[:n_best]
    inits.append(proj_constraint(h_alt, n_best))
    h_rand = (rng.rand(n_best) < 0.5).astype(float)
    inits.append(proj_constraint(h_rand, n_best))
    h_sin = np.clip(0.5 + 0.35 * np.sin(2 * np.pi * 3.0 * np.arange(n_best) / n_best), 0, 1)
    inits.append(proj_constraint(h_sin, n_best))
    h_blk = np.zeros(n_best)
    h_blk[:n_best // 2] = 1.0
    inits.append(proj_constraint(h_blk, n_best))

    for h_init in inits:
        if time.time() > t_start + budget_s - 120:
            break
        h_mixed = proj_constraint(0.4 * h_best + 0.6 * h_init + rng.randn(n_best) * 0.02, n_best)
        deadline = t_start + budget_s - 80
        h_pg, c_pg = proj_grad_desc(h_mixed, n_best, 0.01, 0.1, 400, deadline)
        if c_pg < c5_best:
            h_best, c5_best = h_pg, c_pg

    # Phase 6: Perturbation sweep around current best
    for i in range(12):
        if time.time() > t_start + budget_s - 60:
            break
        rng2 = np.random.RandomState(seed + i * 7919 + 1)
        sigma = 0.008 + 0.012 * (i % 4)
        h_pert = proj_constraint(np.clip(h_best + rng2.randn(n_best) * sigma, 0, 1), n_best)
        T_p = [0.005, 0.002, 0.001, 0.003, 0.0008, 0.004][i % 6]
        deadline = t_start + budget_s - 30
        h_pg, c_pg = proj_grad_desc(h_pert, n_best, T_p, 0.06, 500, deadline)
        if c_pg < c5_best:
            h_best, c5_best = h_pg, c_pg

    # Phase 7: Final SLSQP polish
    if time.time() < t_start + budget_s - 30:
        h_opt = slsqp_run(h_best, n_best, 0.0008, 2500)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 8: Ultra-fine projected gradient
    if time.time() < t_start + budget_s - 10:
        deadline = t_start + budget_s - 5
        h_pg, c_pg = proj_grad_desc(h_best, n_best, 0.0003, 0.02, 800, deadline)
        if c_pg < c5_best:
            h_best, c5_best = h_pg, c_pg

    # Phase 9: Try slightly different resolution as final squeeze
    if time.time() < t_start + budget_s - 20:
        for n_alt in [360, 380]:
            if n_alt == n_best:
                continue
            if time.time() > t_start + budget_s - 15:
                break
            h_alt2 = upsample(h_best, n_alt)
            deadline = t_start + budget_s - 8
            h_pg, c_pg = proj_grad_desc(h_alt2, n_alt, 0.002, 0.05, 300, deadline)
            if c_pg < c5_best:
                h_best, c5_best, n_best = h_pg, c_pg, n_alt

    h_best = proj_constraint(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
