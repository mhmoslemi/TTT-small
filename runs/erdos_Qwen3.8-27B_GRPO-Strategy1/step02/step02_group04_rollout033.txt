    h = h.astype(float).copy()

    s = np.sum(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    cnt = int(np.sum(mask))
    if cnt > 0 and abs(diff) < cnt * 0.4:
        h[mask] += diff / cnt
    else:
        order = np.argsort(h) if diff > 0 else np.argsort(-h)
        rem = abs(diff)
        for i in order:
            if rem <= 1e-14:
                break
            if diff > 0:
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
            else:
                sub = min(rem, h[i])
                h[i] -= sub
                rem -= sub
    return np.clip(h, 0.0, 1.0)

    return np.sum(h) - n * 0.5

def soft_max_obj(h, T):

    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
