import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _obj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    return float(np.sum(x[:_sp['n']]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(120):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_weighted(h, top_k, sharp):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    mx = cr.max()
    w = np.exp(sharp * (cr - mx))
    if w.max() > 0:
        idx = np.argsort(cr)[-top_k:]
        mask = np.zeros(len(cr))
        mask[idx] = 1.0
        w *= mask
    s = w.sum()
    if s < 1e-30:
        return np.zeros(n)
    w = w / s
    o = n - 1
    j = np.arange(n)
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-14:
            continue
        k = ki - o
        vp = (j + k >= 0) & (j + k < n)
        vm = (j - k >= 0) & (j - k < n)
        g[vp] += wi * (1.0 - h[j[vp] + k])
        g[vm] -= wi * h[j[vm] - k]
    return g

def gd_opt(h0, dx, ts, dl, max_it=5000, base_lr=0.06, max_topk=7):
    h = h0.copy()
    bc = compute_c5(h, dx)
    stuck = 0
    for it in range(max_it):
        if time.time() > dl:
            break
        prog = min(1.0, it / 800.0)
        b = 8.0 + 50.0 * prog
        tk = max(2, max_topk - int(3 * prog))
        lr = base_lr / (1.0 + 0.004 * it)
        g = grad_weighted(h, tk, b)
        if np.linalg.norm(g) < 1e-14:
            break
        improved = False
        for sh in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            h2 = proj(h - lr * sh * g, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                h = h2
                bc = c2
                improved = True
                break
        if not improved:
            stuck += 1
            if stuck > 300:
                break
        else:
            stuck = 0
    return h, bc

def push_worst(h0, dx, ts, dl, rng):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    for it in range(3000):
        if time.time() > dl:
            break
        cr = np.correlate(h, 1.0 - h, mode='full')
        kmax = cr.argmax()
        k = kmax - (n - 1)
        j = np.arange(n)
        g = np.zeros(n)
        vp = (j + k >= 0) & (j + k < n)
        vm = (j - k >= 0) & (j - k < n)
        g[vp] += (1.0 - h[j[vp] + k])
        g[vm] -= h[j[vm] - k]
        g2 = grad_weighted(h, 3, 15.0)
        improved = False
        for sh in (0.015, 0.008, 0.004, 0.002, 0.001):
            h2 = proj(h - sh * (0.6 * g + 0.4 * g2), ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                h = h2
                bc = c2
                improved = True
                break
        if not improved and it % 100 == 0:
            h = proj(h + 0.03 * rng.randn(n), ts)
            if compute_c5(h, dx) > bc + 0.02:
                break
    return h, bc

def slsqp_opt(h0, dx, ts, mit=300):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.004]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_obj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': mit, 'ftol': 1e-14})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            c = compute_c5(hr, dx)
            if c <= compute_c5(h0, dx) + 0.003:
                return hr, c
    except Exception:
        pass
    return h0.copy(), compute_c5(h0, dx)

def binarize_push(h0, dx, ts, dl, rng):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    for it in range(2000):
        if time.time() > dl:
            break
        idx = int(rng.randint(n))
        v = h[idx]
        target = 1.0 if v >= 0.5 else 0.0
        for step in (0.08, 0.04, 0.02, 0.01):
            dv = np.sign(target - v) * step
            h2 = h.copy()
            h2[idx] = v + dv
            h2 = proj(h2, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                h = h2
                bc = c2
                break
    return h, bc

def coord_refine(h0, dx, ts, dl, rng):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    for rnd in range(30):
        if time.time() > dl:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in (0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005):
                h[j] = sv + d
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                    continue
                np.clip(h, 0, 1, out=h)
                h = proj(h, ts)
                c = compute_c5(h, dx)
                if c < bc - 1e-15:
                    bc = c
                    imp = True
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def full_pipeline(h0, dx, ts, dl, rng):
    h, c = h0.copy(), compute_c5(h0, dx)
    if time.time() < dl - 80:
        h, c = gd_opt(h, dx, ts, dl - 80)
    if time.time() < dl - 50:
        h, c = slsqp_opt(h, dx, ts, 250)
    if time.time() < dl - 35:
        h, c = push_worst(h, dx, ts, dl - 35, rng)
    if time.time() < dl - 25:
        h, c = gd_opt(h, dx, ts, dl - 25, base_lr=0.03, max_topk=5)
    if time.time() < dl - 15:
        h, c = binarize_push(h, dx, ts, dl - 15, rng)
    if time.time() < dl - 8:
        h, c = slsqp_opt(h, dx, ts, 150)
    if time.time() < dl - 3:
        h, c = coord_refine(h, dx, ts, dl - 2, rng)
    return h, c

def gen_starts(n, ts, rng):
    out = []
    x = np.arange(n)
    out.append(np.full(n, ts / n))
    h = np.zeros(n)
    p = rng.permutation(n)
    h[p[:int(ts)]] = 1.0
    fr = ts - int(ts)
    if fr > 0.05:
        h[p[int(ts)]] = fr
    out.append(h.copy())
    for m in (1, 2, 3, 5, 7):
        out.append(0.5 + 0.25 * np.sin(2 * np.pi * m * x / n))
    out.append(rng.uniform(0, 1, n))
    p3 = max(2, n // 4)
    out.append(np.array([1.0 if (i % (2 * p3)) < p3 else 0.0 for i in range(n)]))
    return [proj(s, ts) for s in out]
