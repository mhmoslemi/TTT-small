import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_corr(h, N, tau):
    """Gradient of soft-max C5 approximation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0/N)
    tc = tau * (corr - corr.max())
    w = np.exp(tc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    thresh = 0.005
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, true_c5

def adam_step(h, grad, m, v, it, lr, b1=0.9, b2=0.999, eps=1e-8):
    m = b1*m + (1-b1)*grad
    v = b2*v + (1-b2)*grad*grad
    mh = m / (1 - b1**it)
    vh = v / (1 - b2**it)
    h_new = h - lr * mh / (np.sqrt(vh) + eps)
    return h_new, m, v

def optimize_phase(h, N, target, tau0, tau1, lr0, lr1, n_steps, deadline):
    """Adam with linear annealing of tau and lr."""
    dx = 2.0/N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(n_steps - 1, 1)
        tau = tau0 + (tau1 - tau0) * frac
        lr = lr0 + (lr1 - lr0) * frac
        grad, c5v = grad_corr(h, N, tau)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        h, m, v = adam_step(h, grad, m, v, it+1, lr)
        h = project_sum(h, target)
    return best_h, best_c5

def coord_descend(h, N, target, n_passes=3, deadline=None):
    """Coordinate descent: try adjusting each h[i] to reduce max corr."""
    dx = 2.0/N
    best_c5 = compute_c5(h, dx)
    for _ in range(n_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for i in range(N):
            if deadline and time.time() > deadline:
                break
            orig = h[i]
            # Try a few values
            for val in [0.0, 1.0, (orig + 0.05) % 1.0, (orig - 0.05) % 1.0, orig + 0.1, orig - 0.1]:
                if val < 0 or val > 1:

                h[i] = val

                if abs(h.sum() - target) > 0.5:
                    continue
                c5v = compute_c5(h, dx)
                if c5v < best_c5:
                    best_c5 = c5v
                    improved = True
                else:
                    h[i] = orig
        # Project at end of pass
        h = project_sum(h, target)
        if not improved:
            break
    return h, compute_c5(h, 2.0/N)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smax(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5v = float(np.max(corr)) * (2.0 / N)
    tc = tau * (corr - corr.max())
    w = np.exp(tc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5v
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    active = np.where(w > 0.005)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, c5v

def grad_argmax(h, N):

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(n_steps - 1, 1)
        if use_argmax_after is not None and frac > use_argmax_after:
            grad, c5v = grad_argmax(h, N)
        else:
            tau = tau0 + (tau1 - tau0) * frac
            grad, c5v = grad_smax(h, N, tau)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        lr = lr0 + (lr1 - lr0) * frac
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**(it+1))
        vh = v / (1 - b2**(it+1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_c5

def coord_descend(h, N, target, n_passes=5, deadline=None):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    for p in range(n_passes):

            break
        any_improve = False
        rng = np.random.RandomState(42 + p)
        order = rng.permutation(N)

            orig = h[i]
            candidates = [0.0, 1.0, 0.5, orig + 0.1, orig - 0.1, orig + 0.05, orig - 0.05]

            best_c = best_c5

        h = project_sum(h, target)
        if not any_improve:
            break

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smax(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5v = float(np.max(corr)) * (2.0 / N)
    tc = tau * (corr - corr.max())
    w = np.exp(tc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5v
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    active = np.where(w > 0.004)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5v

def grad_topk(h, N, k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5v = float(np.max(corr)) * (2.0 / N)
    top_idx = np.argsort(corr)[-k:]
    center = N - 1
    grad = np.zeros(N)
    for ai in top_idx:
        L = ai - center
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            grad[:N - M] += (1.0 - h[M:])
            grad[M:] -= h[:N - M]
    grad /= k
    return grad, c5v

def adam_opt(h, N, target, tau0, tau1, lr0, lr1, n_steps, deadline,
             use_topk_frac=None, topk=5):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(n_steps - 1, 1)
        if use_topk_frac is not None and frac > use_topk_frac:
            grad, c5v = grad_topk(h, N, k=topk)
        else:
            tau = tau0 + (tau1 - tau0) * frac
            grad, c5v = grad_smax(h, N, tau)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        lr = lr0 * (lr1 / lr0) ** frac
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** (it + 1))
        vh = v / (1 - b2 ** (it + 1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_c5

def coord_descend(h, N, target, n_passes=5, deadline=None):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(77)
    for p in range(n_passes):

        any_improve = False
        order = rng.permutation(N)

            if deadline and time.time() > deadline:
                break
            orig = h[i]
            best_val = orig
            best_c = best_c5
            cands = [0.0, 1.0, 0.5, orig + 0.1, orig - 0.1, orig + 0.05, orig - 0.05]
            for cv in cands:

                    continue

                    best_c = c
                    best_val = cv
            if best_val != orig:
                h[i] = best_val
                best_c5 = best_c
                any_improve = True
            else:
                h[i] = orig
        h = project_sum(h, target)
        if not any_improve:
            break
    return h, compute_c5(h, dx)
