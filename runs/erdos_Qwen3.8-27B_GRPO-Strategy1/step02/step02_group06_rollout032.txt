        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, nn):
    n = len(h)
    x = np.linspace(0, 2, n, endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return proj_sum(np.interp(xn, x, h), nn)

def downsample(h, nn):
    n = len(h)
    idx = np.minimum((np.arange(nn) * n // nn).astype(int), n - 1)
    return proj_sum(h[idx], nn)

def obj_sm(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    base = cm * dx if T < 1e-14 else (cm + T * np.log(np.sum(np.exp((corr - cm) / T)))) * dx
    sdev = np.sum(h) - n * 0.5
    return base + mu * sdev * sdev

def grad_sm(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    grad = np.zeros(n)
    if T < 1e-14:
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        else:
            jl = -lag
            grad[:n - jl] += (1.0 - h[jl:])
            grad[jl:] -= h[:n - jl]
    else:
        sh = corr - cm
        exps = np.exp(sh / T)
        p = exps / exps.sum()
        lags = np.arange(2 * n - 1) - (n - 1)
        for m in range(len(p)):
            w = p[m]
            if w < 5e-14:
                continue
            lag = int(lags[m])
            if lag >= 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
            else:
                jl = -lag
                grad[:n - jl] += w * (1.0 - h[jl:])
                grad[jl:] -= w * h[:n - jl]
    sdev = np.sum(h) - n * 0.5
    grad += 2.0 * mu * sdev
    return grad * dx

def run_lbfgs(h0, n, T_list, mu=1e8, max_iter=500, t_start=None, budget_s=1100):
    h = proj_sum(h0.copy(), n)
    best_c = compute_c5(h)
    best_h = h.copy()
    for T in T_list:
        if t_start is not None and time.time() - t_start > budget_s - 80:
            break
        res = opt.minimize(
            obj_sm, h, args=(T, mu), jac=grad_sm,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12, 'iprint': 0}
        )
        h = proj_sum(np.clip(res.x, 0.0, 1.0), n)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def run_slsqp(h0, n, T, mu, max_iter=800):
    def obj(h):
        n2 = len(h)
        dx = 2.0 / n2
        corr = np.correlate(h, 1.0 - h, mode='full')
        cm = np.max(corr)
        return (cm + T * np.log(np.sum(np.exp((corr - cm) / T)))) * dx + mu * (np.sum(h) - n2 * 0.5) ** 2
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x) * 0.5, 'jac': lambda x: np.ones(len(x))}]
    res = opt.minimize(obj, h0.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0})
    return proj_sum(np.clip(res.x, 0.0, 1.0), n)

def sa_refine(h0, n, t_start, budget_s, max_steps=500, seed=0):
    rng = np.random.RandomState(seed)
    h_cur = h0.copy()
    c_cur = compute_c5(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    T_sa = 0.0003
    for step in range(max_steps):
        if time.time() - t_start > budget_s - 40:
            break
        nbr = rng.randint(1, 8)
        idxs = rng.choice(n, size=nbr, replace=False)
        h_try = h_cur.copy()
        h_try[idxs] = np.clip(h_try[idxs] + rng.randn(nbr) * 0.03, 0, 1)
        h_try = proj_sum(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T_sa):
            h_cur = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_cur.copy()
        T_sa *= 0.996
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    h_init = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h_init)
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    n_best = n_init

    # Phase 1: L-BFGS with soft-max at original resolution
    elapsed = time.time() - t_start
    if budget_s - elapsed > 200:
        T_sched = [0.08, 0.03, 0.01, 0.004, 0.0015, 0.0006]
        h_opt, c_opt = run_lbfgs(h_best, n_best, T_sched, mu=1e6, max_iter=400,
                                  t_start=t_start, budget_s=budget_s)
        if c_opt < c5_best:
            h_best, c5_best, n_best = h_opt, c_opt, n_opt

    # Phase 2: SLSQP refinement at low T
    elapsed = time.time() - t_start
    if budget_s - elapsed > 120:
        for T in [0.0005, 0.0002, 0.00005]:
            if time.time() - t_start > budget_s - 60:
                break
            h_opt = run_slsqp(h_best, n_best, T, mu=1e8, max_iter=600)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best, c5_best = h_opt, c

    # Phase 3: Multi-start perturbations
    elapsed = time.time() - t_start
    for i in range(10):
        if time.time() - t_start > budget_s - 80:
            break
        rng = np.random.RandomState(seed + i * 104729 + 7)
        sigma = 0.01 + 0.006 * (i % 3)
        h_p = proj_sum(np.clip(h_best + rng.randn(n_best) * sigma, 0, 1), n_best)
        T_p = [0.003, 0.001, 0.002, 0.0005][i % 4]
        h_opt, c_opt = run_lbfgs(h_p, n_best, [T_p, T_p * 0.3], mu=1e6, max_iter=300,
                                  t_start=t_start, budget_s=budget_s)
        if c_opt < c5_best:
            h_best, c5_best, n_best = h_opt, c_opt, n_best

    # Phase 4: Try alternative resolutions
    elapsed = time.time() - t_start
    for n_try in [224, 448, 392]:
        if n_try == n_best:
            continue
        if time.time() - t_start > budget_s - 100:
            break
        h_c = upsample(h_best, n_try) if n_try > n_best else downsample(h_best, n_try)
        h_opt, c_opt = run_lbfgs(h_c, n_try, [0.02, 0.006, 0.002], mu=1e6, max_iter=350,
                                  t_start=t_start, budget_s=budget_s)
        if c_opt < c5_best:
            h_best, c5_best, n_best = h_opt, c_opt, n_try

    # Phase 5: SLSQP + SA at best resolution
    elapsed = time.time() - t_start
    if budget_s - elapsed > 150:
        h_opt = run_slsqp(h_best, n_best, 0.0001, mu=1e8, max_iter=1000)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c
        if time.time() - t_start > 0 and budget_s - (time.time() - t_start) > 100:
            h_sa, c_sa = sa_refine(h_best, n_best, t_start, budget_s, 600, seed + 555)
            if c_sa < c5_best:
                h_best, c5_best = h_sa, c_sa

    # Phase 6: Final ultra-fine polish
    elapsed = time.time() - t_start
    if budget_s - elapsed > 40:
        h_opt = run_slsqp(h_best, n_best, 1e-6, mu=1e8, max_iter=1500)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best, c5_best = h_opt, c

    # Phase 7: One more L-BFGS at very low T
    elapsed = time.time() - t_start
    if budget_s - elapsed > 30:
        h_opt, c_opt = run_lbfgs(h_best, n_best, [1e-5, 5e-6], mu=1e7, max_iter=800,
                                  t_start=t_start, budget_s=budget_s)
        if c_opt < c5_best:
            h_best, c5_best, n_best = h_opt, c_opt, n_best

    h_best = proj_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
