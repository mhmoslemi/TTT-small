3. Outer: update mu_k proportional to corr[k] (or mu_k = 1 for k in argmax)
4. Repeat

This is a **dual decomposition** approach! The inner problem is smooth, so L-BFGS-B should work well.

    g = np.zeros(n)
    L = abs(k)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:

        g[:n - L] -= h[L:]

        g[L:] -= h[:n - L]
    return g

def soft_max_val(h, T):

    corr = get_corr(h)
    c_max = np.max(corr)
    if T < 1e-15:

    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * (2.0 / n)

    dx = 2.0 / n
    corr = get_corr(h)

    nf = 2 * n - 1

            g += mu[kk] * grad_at_lag(h, lag)

    g += 2.0 * lam * (np.sum(h) - n * 0.5)

def do_lbfgsb_dual(h0, mu, n, lam, max_iter, t_start, budget_s):

        return dual_inner_grad(x, mu, n, lam)

                       bounds=[(0, 1)] * n,
                       options={'maxiter': max_iter, 'gtol': 1e-7, 'ftol': 1e-12})

def do_slsqp_soft(h0, n, T, max_iter):
