For mode='full', k ranges from -(n-1) to (n-1), and the output is indexed 0 to 2n-2 where index m corresponds to k = m - (n-1).

∂c[m]/∂h[t]:
This appears when:
- t = n+j for some n, contributing (1-h[n]) = (1-h[t-j])
- t = n for some n, contributing -h[n+j] = -h[t+j]

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    mask = (h > 1e-6) & (h < 1.0 - 1e-6)
    k = int(np.sum(mask))
    if k > 0:
        h[mask] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = target - np.sum(h)
    if abs(d2) > 1e-12:
        interior = np.where((h > 1e-5) & (h < 1.0 - 1e-5))[0]
        if len(interior) > 0:
            h[interior[0]] = min(max(h[interior[0]] + d2, 0.0), 1.0)
    return h

def sum_con(x, n):
    return np.sum(x) - 0.5 * n

def sum_jac(x, n):
    return np.ones(n)

def lse_val(x, T):
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def lse_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
    else:
        w = np.exp((corr - c_max) / T)
        w /= np.sum(w)
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        j = m - (n - 1)
        n_lo = max(0, -j)
        n_hi = min(n - 1, n - 1 - j)
        if n_lo > n_hi:
            continue
        length = n_hi - n_lo + 1
        idx1 = n_lo + j
        g[idx1:idx1 + length] += wm * (1.0 - x[n_lo:n_hi + 1])
        g[n_lo:n_lo + length] += wm * (-x[idx1:idx1 + length])
    return g * dx

def lsfbb_val(x, T, lam, n):
    f = lse_val(x, T)
    s = np.sum(x) - 0.5 * n
    return f + lam * s * s

def lsfbb_grad(x, T, lam, n):
    g = lse_grad(x, T)
    s = np.sum(x) - 0.5 * n
    g += 2.0 * lam * s
    return g

def coord_descend(h, n, passes=3):
    for _ in range(passes):
        for j in range(n):
            v = h[j]
            if v <= 1e-5 or v >= 1.0 - 1e-5:
                continue
            c0 = compute_c5(h)
            for delta in (0.02, -0.02, 0.008, -0.008, 0.003, -0.003, 0.001, -0.001):
                nv = v + delta
                if nv < 1e-5 or nv > 1.0 - 1e-5:
                    continue
                hn = h.copy()
                hn[j] = nv
                adj = -delta / (n - 1)
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                sub = hn[mask]
                if np.min(sub) + adj < -1e-6 or np.max(sub) + adj > 1.0 + 1e-6:
                    continue
                hn[mask] += adj
                hn = np.clip(hn, 0.0, 1.0)
                hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < c0 - 1e-13:
                    h = hn
                    c0 = c
                    break
    return h

def pair_polish(h, n, rng, n_pairs=400):
    for _ in range(n_pairs):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
            continue
        if h[l] <= 1e-4 or h[l] >= 1.0 - 1e-4:
            continue
        c0 = compute_c5(h)
        for d in (0.03, -0.03, 0.01, -0.01, 0.004, -0.004, 0.001, -0.001):
            hj2, hl2 = h[j] + d, h[l] - d
            if hj2 < 1e-4 or hj2 > 1.0 - 1e-4 or hl2 < 1e-4 or hl2 > 1.0 - 1e-4:
                continue
            hn = h.copy()
            hn[j], hn[l] = hj2, hl2
            c = compute_c5(hn)
            if c < c0 - 1e-13:
                h = hn
                c0 = c
                break
    return fix_sum(h, n)

    h = np.array(initial_h_values, dtype=float).copy()

    sched1 = [(0.05, 5, 300), (0.02, 10, 400), (0.008, 20, 500),
              (0.003, 50, 600), (0.001, 100, 800), (0.0004, 200, 1000)]
    for T, lam, mx in sched1:
        if time.time() - t0 > budget_s - 500:
            break
        c0 = compute_c5(h)
        try:

                method='L-BFGS-B', bounds=[(0., 1.)] * n,
                options={'maxiter': mx, 'ftol': 1e-20, 'gtol': 1e-12})
            hn = fix_sum(r.x, n)
            if compute_c5(hn) < c5:
                h = hn
                c5 = compute_c5(h)
        except Exception:
            pass

    # Phase 2: SLSQP with analytical gradients (exact constraint enforcement)
    sched2 = [(0.001, 500), (0.0004, 700), (0.00015, 1000), (0.00005, 1500)]
    for T, mx in sched2:
        if time.time() - t0 > budget_s - 400:

        c0 = compute_c5(h)
        try:
            r = optimize.minimize(lse_val, h, args=(T,), jac=lse_grad, method='SLSQP',
                bounds=[(0., 1.)] * n,
                constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}],
                options={'maxiter': mx, 'ftol': 1e-18})
            hn = fix_sum(r.x, n)
            if compute_c5(hn) < c0:
                h = hn
                c5 = compute_c5(h)
        except Exception:
            pass

    # Phase 3: Coordinate and pair polishing
    if time.time() - t0 < budget_s - 250:
        h = fix_sum(coord_descend(h, n, 3), n)
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 230:

        c5 = compute_c5(h)

        mode = i % 6
        if mode == 0:

        elif mode == 2:
            s = rng.randint(1, 6)
            ht = np.roll(ht, s)
        elif mode == 3:

            ln = rng.randint(8, min(25, n - st))
            ht[st:st + ln] = 1.0 - ht[st:st + ln]
        elif mode == 4:
            ht += rng.randn(n) * 0.006
        else:
            idx = rng.choice(n, 8, replace=False)
            directions = rng.choice([-1, 1], 8)
            for k in range(8):

            c_prev = compute_c5(ht)
            try:
                r2 = optimize.minimize(lsfbb_val, ht, args=(T, lam, n), jac=lsfbb_grad,
                    method='L-BFGS-B', bounds=[(0., 1.)] * n,
                    options={'maxiter': mx, 'ftol': 1e-18})
                hn = fix_sum(r2.x, n)
                if compute_c5(hn) < c_prev:
                    ht = hn
            except Exception:
                pass
        ct = compute_c5(ht)
        if ct < c5:
            h, c5 = ht, ct
            if time.time() - t0 < budget_s - 60:
                h = fix_sum(coord_descend(h, n, 2), n)
                h = pair_polish(h, n, rng, 200)
                c5 = compute_c5(h)

    # Phase 5: Final SLSQP with very small T
    for T, mx in [(0.0001, 2000), (0.00003, 3000)]:
        if time.time() - t0 > budget_s - 15:
            break

                bounds=[(0., 1.)] * n,

                options={'maxiter': mx, 'ftol': 1e-18})

            if compute_c5(hn) < c0:
                h = hn
        except Exception:
            pass

    h = fix_sum(h, len(h))
    c5 = compute_c5(h)
    return (h, c5, len(h))
