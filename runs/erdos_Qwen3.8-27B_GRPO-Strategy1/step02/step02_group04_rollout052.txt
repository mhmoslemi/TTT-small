        res = opt.minimize(soft_max_obj, h.copy(), args=(T,),
            method='SLSQP', bounds=[(0,1)]*n,
            constraints=[{'type':'eq','fun':sum_con,'args':(n,)}],
            options={'maxiter':max_iter,'ftol':1e-18})
        return fix_sum(res.x, n) if res.x is not None else h
    except: return h

def subgrad_descent(h, n, n_steps, lr0, tol_frac):
    """Subgradient method with diminishing step size"""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    for step in range(n_steps):
        lr = lr0 / (1 + 0.01 * step)
        corr = np.correlate(h, 1-h, mode='full')
        c_max = corr.max()
        active = np.where(corr >= c_max - tol_frac * c_max)[0]
        # Compute subgradient
        grad = np.zeros(n)
        for j in active:
            s = abs(j - (n-1))
            if s == 0:
                grad += 1.0 - 2.0*h
            elif s < n:
                sign = 1 if j >= n-1 else -1
                if sign == 1:  # positive shift
                    grad[s:] += (1.0 - h[:n-s])
                    grad[:n-s] -= h[s:]
                else:  # negative shift
                    grad[:n-s] += (1.0 - h[s:])
                    grad[s:] -= h[:n-s]
        grad /= len(active)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = fix_sum(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = mask.sum()
    if n_int > 0 and abs(diff) < n_int * 0.45:
        h[mask] += diff / n_int
    else:
        if diff > 0:
            for i in np.argsort(h):
                add = min(diff, 1.0 - h[i]); h[i] += add; diff -= add
                if diff <= 1e-12: break
        else:
            diff = -diff
            for i in np.argsort(-h):
                rem = min(diff, h[i]); h[i] -= rem; diff -= rem
                if diff <= 1e-12: break
