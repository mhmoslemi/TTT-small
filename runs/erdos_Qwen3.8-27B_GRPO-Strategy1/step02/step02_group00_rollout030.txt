            hp = best_h + 0.08 * np.random.randn(n)
            starts.append(project_feasible(hp, n))

        for h0 in starts:
            if time.time() - t0 > budget_s * 0.78:
                break
            for T in [200, 700]:
                if time.time() - t0 > budget_s * 0.78:
                    break
                try:
                    hc = slsqp_optimize(h0, T, 8, n, maxiter=400)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                except Exception:
                    pass

    # Phase 5: Local search (swap + coordinate)
    if time.time() - t0 < budget_s * 0.90:
        best_h, best_c5 = swap_search(best_h, n, max_rounds=150)
    if time.time() - t0 < budget_s * 0.94:
        best_h, best_c5 = coordinate_search(best_h, n, max_rounds=80)
    if time.time() - t0 < budget_s * 0.96:
        best_h, best_c5 = swap_search(best_h, n, max_rounds=100)

    # Phase 6: Final high-T SLSQP polish
    n_cur = len(best_h)
    for T in [5000, 15000]:
        if time.time() - t0 > budget_s * 0.99:
            break
        try:
            hc = slsqp_optimize(best_h, T, 1, n_cur, maxiter=1000, full=True)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
