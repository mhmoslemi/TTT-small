def project(h, n):

    h = np.clip(h, 0, 1)

    if abs(d) < 1e-12:

def smooth_max_grad(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    c = np.max(g)
    
    if T < 1e-14:
        k_max = np.argmax(g)
        exp_g = np.zeros(2*n-1)
        exp_g[k_max] = 1.0
    else:

        s = np.sum(exp_g)
        exp_g /= s

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            cur_lr *= 0.97
            cur_lr = max(cur_lr, 0.01)
    return best_h, best_c5

def slsqp_smooth(h, n, T, iters):
    c0 = compute_c5(h)
    dx = 2.0 / n
    def lse_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cx = np.max(corr)
        if T < 1e-14:
            return cx * dx
        return (cx + T * np.log(np.sum(np.exp((corr - cx) / T)))) * dx
    def sum_con(x):
        return np.sum(x) - 0.5 * n
    try:
        r = optimize.minimize(lse_obj, h, method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con}],
            options={'maxiter': iters, 'ftol': 1e-13})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def pair_polish(h, n, rng, n_pairs=2000):
    best_c5 = compute_c5(h)
    deltas = (-0.04, -0.015, -0.005, -0.001, 0.001, 0.005, 0.015, 0.04)
    for _ in range(n_pairs):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
            continue
        if h[l] <= 1e-4 or h[l] >= 1.0 - 1e-4:
            continue
        for d in deltas:
            hn = h.copy()
            hn[j] += d
            hn[l] -= d
            if hn[j] < 1e-6 or hn[j] > 1.0 - 1e-6 or hn[l] < 1e-6 or hn[l] > 1.0 - 1e-6:
                continue
            c = compute_c5(hn)
            if c < best_c5 - 1e-14:
                best_c5 = c
                h = hn
                break
    return fix_sum(h, n), best_c5

def triple_polish(h, n, rng, n_triples=1000):
    best_c5 = compute_c5(h)
    for _ in range(n_triples):
        idx = rng.choice(n, 3, replace=False)
        if any(h[i] <= 1e-4 or h[i] >= 1.0 - 1e-4 for i in idx):
            continue
        for d in (-0.02, -0.008, 0.008, 0.02):
            hn = h.copy()
            hn[idx[0]] += d
            hn[idx[1]] -= d * 0.7
            hn[idx[2]] -= d * 0.3
            if np.any(hn < 1e-6) or np.any(hn > 1.0 - 1e-6):
                continue
            c = compute_c5(hn)
            if c < best_c5 - 1e-14:
                best_c5 = c
                h = hn
                break
    return fix_sum(h, n), best_c5

def upsample(h, nn):
    n = len(h)
    x0 = np.arange(n, dtype=float) * (2.0 / n)
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return project(np.interp(x1, x0, h, period=2.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = fix_sum(h, n)
    c5 = compute_c5(h)

    # Phase 1: PGD with broad cooling
    if time.time() - t0 < budget_s - 400:
        h, c5 = pgd(h, n, 0.03, 0.001, 40000, 1.5)

    # Phase 2: SLSQP refinement
    if time.time() - t0 < budget_s - 300:
        h = slsqp_smooth(h, n, 0.001, 600)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 3: PGD fine
    if time.time() - t0 < budget_s - 250:
        h, c5 = pgd(h, n, 0.001, 1e-5, 50000, 1.0)

    # Phase 4: SLSQP very fine
    if time.time() - t0 < budget_s - 200:
        h = slsqp_smooth(h, n, 1e-4, 800)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 5: Pair + triple polish
    if time.time() - t0 < budget_s - 120:
        h, c5 = pair_polish(h, n, rng, 3000)
        h, c5 = triple_polish(h, n, rng, 1500)

    # Phase 6: Multi-start with diverse perturbations
    for i in range(40):
        if time.time() - t0 > budget_s - 100:
            break
        ht = h.copy()
        mode = i % 5
        if mode == 0:
            ht += rng.randn(n) * 0.015
        elif mode == 1:
            idx = rng.choice(n, max(1, n // 20), replace=False)
            ht[idx] += rng.randn(len(idx)) * 0.1
        elif mode == 2:
            ht += rng.randn(n) * 0.005
        elif mode == 3:
            m = rng.choice(n, n // 5, replace=False)
            ht[m] += rng.uniform(-0.05, 0.05, len(m))
        else:
            phases = rng.randint(1, 6)
            omega = 2 * np.pi * phases / n
            ht += 0.02 * np.sin(omega * np.arange(n))
        ht = fix_sum(ht, n)
        ht, ct = pgd(ht, n, 0.005, 0.0002, 20000, 1.5)
        if ct < c5 - 1e-6:
            ht = slsqp_smooth(ht, n, 0.0005, 400)
            ht = fix_sum(ht, n)
            ct = compute_c5(ht)
            if ct < c5:
                h, c5 = ht, ct
                h, c5 = pair_polish(h, n, rng, 1000)

    # Phase 7: Final PGD + SLSQP
    if time.time() - t0 < budget_s - 60:
        h, c5 = pgd(h, n, 1e-4, 1e-6, 30000, 0.8)
    if time.time() - t0 < budget_s - 30:
        h = slsqp_smooth(h, n, 5e-5, 600)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 15:
        h, c5 = pair_polish(h, n, rng, 1000)

    # Phase 8: Higher resolution
    if time.time() - t0 < budget_s - 120:
        n2 = n * 2
        h2 = upsample(h, n2)
        h2, c5_2 = pgd(h2, n2, 0.003, 0.0002, 20000, 1.5)
        h2 = slsqp_smooth(h2, n2, 1e-4, 400)
        h2 = fix_sum(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < c5:
            h, n, c5 = h2, n2, c5_2
            if time.time() - t0 < budget_s - 30:
                h, c5 = pgd(h, n, 1e-4, 1e-6, 20000, 1.0)
                h, c5 = pair_polish(h, n, rng, 1000)

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
