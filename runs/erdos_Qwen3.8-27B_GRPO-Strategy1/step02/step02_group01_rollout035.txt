            if s < 1e-14: break

            if s < 1e-14: break
            h += h * (diff / s)

        idx = np.where(hb == 0)[0]

                       bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:

def micro_refine(h, n, eps=0.01):
    # Try to reduce max by slightly adjusting values at 0/1 boundaries
    dx = 2.0 / n
    corr = compute_corr(h)
    k_max = int(np.argmax(corr))
    c_best = float(np.max(corr)) * dx
    h_best = h.copy()
    
    # For the worst lag, try adjusting pairs
    # corr[k_max] involves terms h[i+k_max]*(1-h[i])
    # If h[i+k_max] is near 1 and h[i] is near 0, reducing h[i+k_max] slightly helps
    for _ in range(200):
        k_m = int(np.argmax(corr))
        # Find contributing positions
        # corr[k_m] ≈ sum over valid i of h[i + offset] * (1 - h[i])
        # For the full correlation, let's just try random small adjustments
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        h_new = h.copy()
        # Reduce a 1-ish value, increase a 0-ish value
        if h_new[i] > 0.5 and h_new[j] < 0.5:
            h_new[i] -= eps
            h_new[j] += eps
        elif h_new[j] > 0.5 and h_new[i] < 0.5:
            h_new[j] -= eps
            h_new[i] += eps
        else:
            continue
        h_new = np.clip(h_new, 0, 1)
        c_new = compute_c5(h_new)
        if c_new < c_best - 1e-10:
            c_best = c_new
            h_best = h_new.copy()
            h = h_new
            corr = compute_corr(h)
    
    return h_best, c_best

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += room * (diff / s)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += h * (diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    hb = (h >= 0.5).astype(float)
    diff = int(round(n / 2.0 - hb.sum()))
    if diff > 0:
        idx = np.where(hb == 0)[0]
        order = np.argsort(-h[idx])
        for k in range(min(diff, len(order))):
            hb[idx[order[k]]] = 1.0
    elif diff < 0:
        idx = np.where(hb == 1)[0]
        order = np.argsort(h[idx])
        for k in range(min(-diff, len(order))):
            hb[idx[order[k]]] = 0.0
    return hb
