Now for the 2D local search. The key idea: for each pair (i,j), we want to find δ that minimizes C₅ when h_i → h_i+δ, h_j → h_j-δ (maintaining sum). We can do this by:
1. Computing the current c for all lags
2. For a given δ, the new c_L changes by:
   Δc_L ≈ δ * [(1-h[i+L]-h[i-L]) - (1-h[j+L]-h[j-L])] (× dx)
   for the lags where both i and j are in the valid range.

But this linear approximation might not be accurate for large δ. Better to just evaluate c5 for a few δ values.

```python
import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    return float(np.sum(x[:_sp['n']]) - _sp['ts'])

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_lse(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (corr - corr.max()))
    sw = w.sum()
    if sw < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= sw
    o = n - 1
    g = np.zeros(n)
    for m in range(n):
        lo = max(0, -m)
        hi = min(n - 1, n - 1 - m)
        indices = np.arange(lo, hi + 1)
        L_vals = indices - m
        wm = w[o + L_vals]
        h_plus = np.where((m + L_vals >= 0) & (m + L_vals < n), h[np.clip(m + L_vals, 0, n - 1)], 0.0)
        h_minus = np.where((m - L_vals >= 0) & (m - L_vals < n), h[np.clip(m - L_vals, 0, n - 1)], 0.0)
        g[m] = np.dot(wm, 1.0 - h_plus - h_minus)
    return g * dx

def fista(h0, dx, ts, beta, iters, deadline):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    best = compute_c5(h, dx)
    best_h = h.copy()
    L = 4.0 * n
    t = 1.0
    for _ in range(iters):
        if time.time() > deadline:
            break
        y = h + ((t - 1) / (t + 2)) * (h - h_prev)
        y = np.clip(y, 0, 1)
        g = grad_lse(y, dx, beta)
        step = 1.0 / L
        fy_grad = np.dot(g, y)
        for _ in range(15):
            h_new = proj(y - step * g, ts)
            if np.dot(g, h_new - y) >= -0.5 * L * step * np.dot(h_new - y, h_new - y) + 1e-18:
                break
            step *= 0.5
        h_prev = h
        h = h_new
        t += 1.0
        c = compute_c5(h, dx)
        if c < best:
            best = c
            best_h = h.copy()
    return best_h, best

def slsqp_opt(h0, dx, ts, maxiter=500):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            c = compute_c5(hr, dx)
            if c <= t0 * dx + 0.01:
                return hr, c
    except Exception:
        pass
    return h0.copy(), compute_c5(h0, dx)

def pair_search(h0, dx, ts, deadline, rounds=5):
    n = len(h0)
    h = h0.copy()
    bc = compute_c5(h, dx)
    for r in range(rounds):
        if time.time() > deadline:
            break
        imp = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for d in (0.003, -0.003, 0.007, -0.007, 0.015, -0.015, 0.03, -0.03):
                    ni, nj = h[i] + d, h[j] - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c = compute_c5(h, dx)
                    if c < bc - 1e-15:
                        bc = c
                        imp = True
                    else:
                        h[i], h[j] = nj + (h[j] - nj) * 0, h[i] - (h[i] - ni) * 0
                        h[i], h[j] = h[j] + d, h[i] - d  # revert
                        h[i], h[j] = h[i], h[j]
                        # Actually just revert properly
                        h[i], h[j] = h[i] - (ni - h[i]) + (ni - h[i]), h[j]
                        h[i], h[j] = ni - d, nj + d
                        h = proj(h, ts)
        if not imp:
            break
    return h, bc

def make_starts(n, ts, rng, ref=None):

    h = np.zeros(n)

    h[perm[:int(ts)]] = 1.0
    fr = ts - int(ts)
    if fr > 0.01:
        h[perm[int(ts)]] = fr
    out.append(h)
    out.append(rng.uniform(0, 1, n))

            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, len(ref), endpoint=False), ref)
            out.insert(0, hi)
    return [proj(s, ts) for s in out]
```
