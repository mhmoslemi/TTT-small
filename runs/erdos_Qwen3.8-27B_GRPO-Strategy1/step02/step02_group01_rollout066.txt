import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def binarize_balanced(h, n):
    h = np.asarray(h, dtype=float)
    if len(h) != n:
        if len(h) == 0:
            h = np.random.rand(n)
        else:
            x_old = np.linspace(0, 2, len(h))

    k = n // 2
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:k]] = 1.0
    return result

def make_thue_morse(n):
    # Thue-Morse sequence
    t = 0
    seq = np.zeros(n)
    for i in range(n):
        # Count bits
        c = bin(i).count('1')
        seq[i] = 1.0 if c % 2 == 1 else 0.0
    # Balance to exactly n/2 ones
    return binarize_balanced(seq, n)

def make_block_pattern(n, frac):
    # Alternating blocks
    n1 = int(n * frac)
    n0 = n - n1
    pattern = np.zeros(n)
    pos = 0
    while pos < n:
        pattern[pos:pos+n1] = 1.0
        pos += n1
        pattern[pos:pos+n0] = 0.0
        pos += n0
    return binarize_balanced(pattern, n)

def local_search(h, t_max, rng, n_trials=2000):

            break

            h[j] = 1.0
            if c < best_c5_trial:
                best_c5_trial = c
                best_i, best_j = i, j

            h[best_i] = 1.0
            h[best_j] = 0.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if best_c5_trial < best_c5:
                best_c5 = best_c5_trial
                stall = 0
            else:
                stall += 1
        else:

        if stall > 500:
            break

def slsqp_polish(h, maxiter=200):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return float(np.max(np.correlate(x, 1-x, mode='full'))) + 100*abs(x.sum()-n/2)
    def eq(x):
        return x.sum() - n/2
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                      constraints=[{'type':'eq','fun':eq}],
                      options={'maxiter':maxiter,'ftol':1e-12})
        h_new = np.clip(res.x, 0, 1)
        # Enforce exact sum
        diff = n/2 - h_new.sum()
        if abs(diff) > 1e-10:
            room = (1-h_new) if diff > 0 else h_new.copy()
            total = room.sum()
            if total > 0:
                h_new += room * (diff/total)
                h_new = np.clip(h_new, 0, 1)
        return h_new
    except:
        return h

                      options={'maxiter':maxiter,'ftol':1e-14})
        h_new = np.clip(res.x, 0, 1)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - h_new
            else:
                room = h_new.copy()
            total = room.sum()
            if total > 1e-14:
                h_new = np.clip(h_new + room * (diff / total), 0, 1)
        return h_new
    except:
        return h

Let me be more precise. corr at lag τ (positive):
corr[τ] = ∑_{i=0}^{n-1-τ} h[i+τ] * (1-h[i])

To reduce this:
- Set h[i] from 0 to 1 where h[i+τ] = 1: removes a contribution of 1 (the (1,0) pair)
- Set h[j] from 1 to 0 where h[j+τ] = 0: removes a contribution of 0 (no change)
  Wait: if h[j+τ] = 0, the term h[j+τ]*(1-h[j]) = 0 regardless of h[j]. So changing h[j] doesn't affect this term.
  If h[j+τ] = 1, the term is 1*(1-h[j]) = 1 when h[j]=0, 0 when h[j]=1. So setting h[j] from 1 to 0 INCREASES the term by 1. Bad!
  
  So for the first sum (i as the "1-h" index):
  - Setting h[i] from 0 to 1: removes h[i+τ] from the sum (if h[i+τ]=1, decreases by 1)
  - Setting h[j] from 1 to 0: adds h[j+τ] to the sum (if h[j+τ]=1, increases by 1)

For the "h" index (i+τ):
  - Setting h[i+τ] from 1 to 0: removes (1-h[i]) from the sum (if h[i]=0, decreases by 1)
  - Setting h[j+τ] from 0 to 1: adds (1-h[j]) to the sum (if h[j]=1, increases by 1)

So for a swap (set i: 0→1, set j: 1→0), the change in corr[τ] is:
- From i in (1-h[i]) position: -h[i+τ] (was contributing h[i+τ], now 0)
  But also i might be in the h[i+τ] position for some other i': i'+τ = i, i' = i-τ
  If i-τ is valid: now h[i] contributes as h[i-τ+τ]*(1-h[i-τ]) = 1*(1-h[i-τ])
  So add (1-h[i-τ]) if i-τ is valid
- From j in (1-h[j]) position: was 0, now h[j+τ]
  So add h[j+τ]
  And j in h position: j-τ valid? h[j] was 1, now 0.
  The term h[j]*(1-h[j-τ]) was (1-h[j-τ]), now 0. So subtract (1-h[j-τ]) if j-τ valid.

Delta[τ] = -h[i+τ] + (1-h[i-τ])_valid + h[j+τ] - (1-h[j-τ])_valid

where (x)_valid means x if the index is in [0,n-1], else 0.

3. Among candidates, pick one that also doesn't increase other lags too much

def focused_local_search(h, t_max, rng):
    n = len(h)
    h = h.copy()
    dx = 2.0/n
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    t_start = time.time()
    
    while time.time() - t_start < t_max:
        corr = np.correlate(h, 1-h, mode='full')
        max_lag = int(np.argmax(corr))
        m = n - 1  # center
        
        # Relative lag
        tau = max_lag - m
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_delta_max = np.inf
        best_i, best_j = -1, -1
        
        # Try a subset of swaps
        n_candidates = min(len(zeros)*len(ones), 5000)
        tried = set()
        for _ in range(n_candidates):
            ii = rng.integers(len(zeros))
            jj = rng.integers(len(ones))
            i, j = zeros[ii], ones[jj]
            key = (i, j)
            if key in tried:
                continue
            tried.add(key)
            
            # Compute delta for all lags (O(n))
            delta = np.zeros(2*n-1)
            # i: 0->1
            # As (1-h) index: -h[i+k-m] for valid k
            # As h index: +(1-h[i-k+m]) for valid k
            # j: 1->0
            # As (1-h) index: +h[j+k-m]
            # As h index: -(1-h[j-k+m])
            
            for k in range(2*n-1):
                d = 0.0
                idx1 = i + k - m
                if 0 <= idx1 < n:
                    d -= h[idx1]
                idx2 = i - k + m
                if 0 <= idx2 < n:
                    d += (1.0 - h[idx2])
                idx3 = j + k - m
                if 0 <= idx3 < n:
                    d += h[idx3]
                idx4 = j - k + m
                if 0 <= idx4 < n:
                    d -= (1.0 - h[idx4])
                delta[k] = d
            
            new_max = float(np.max(corr + delta))
            if new_max < best_delta_max:
                best_delta_max = new_max
                best_i, best_j = i, j
        
        if best_i >= 0 and best_delta_max * dx < c5_cur:
            h[best_i] = 1.0
            h[best_j] = 0.0
            c5_cur = best_delta_max * dx
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
        else:
            break
    
    return h_best, c5_best

def precompute_shifts(h):
    n = len(h)
    m = n - 1
    L = 2*n - 1
    forward = np.zeros((n, L))
    backward = np.zeros((n, L))
    for p in range(n):

        # p+k-m in [0,n-1] => k in [m-p, m-p+n-1]

        k_start2 = max(0, p+1)
