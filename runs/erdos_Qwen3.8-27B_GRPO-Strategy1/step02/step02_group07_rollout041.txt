import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_c = corr - corr.max()
    exp_c = np.exp(corr_c / T)
    W = exp_c / exp_c.sum()
    # grad[j] = dx * sum_i W[i] * (1 - h[j+i-(n-1)]) for valid i
    grad = np.zeros(n)
    for j in range(n):
        lo = max(0, n-1-j)
        hi = min(2*n-2, 2*n-2-j)
        ws = W[lo:hi+1]
        idx = np.arange(lo-(n-1-j), hi-(n-1-j)+1)
        valid = (idx >= 0) & (idx < n)
        grad[j] = dx * np.sum(ws[valid] * (1.0 - h[idx[valid]]))
    return grad

def project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-10:
        return h
    # Add/subtract from interior points
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    k = np.sum(interior)
    if k > 0:
        h[interior] += d / k
        h = np.clip(h, 0.0, 1.0)
    # If still off, adjust individual points
    d = target - np.sum(h)
    if abs(d) > 1e-10:
        # Sort by h value, adjust from extremes
        for i in range(n):
            if d > 0 and h[i] < 1.0:
                adj = min(1.0 - h[i], d)
                h[i] += adj
                d -= adj
            elif d < 0 and h[i] > 0.0:
                adj = max(h[i], d)  # d is negative, so adj = h[i] + d if h[i]+d > 0
                adj = min(adj, h[i])
                h[i] += adj
                d -= adj
            if abs(d) < 1e-10:
                break
    return h

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def fista(h, n, T, lr, iters):
    """FISTA (Nesterov accelerated gradient) for log-sum-exp"""
    y = h.copy()
    zk = h.copy()
    t = 1.0
    for k in range(iters):
        grad = compute_lse_grad(y, T)
        # Gradient step
        y = y - lr * grad
        y = project(y, n)
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = y + (t - 1.0) / t_new * (y - zk)
        y = project(y, n)
        zk = zk.copy()
        t = t_new
    return zk

def slsqp_h(h, n, T, iters):
    """SLSQP with log-sum-exp"""
    def lse_obj_local(x):
        return lse_obj(x, T)
    def sum_con(x):
        return np.sum(x) - 0.5 * n
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj_local, h, method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con}],
            options={'maxiter': iters, 'ftol': 1e-14})
        hn = project(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except:
        pass
    return h

def fix_sum(h, n):
    return project(h, n)

def coord_polish(h, n, n_pass=2):
    for _ in range(n_pass):
        improved = False
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            best_c5 = compute_c5(h)
            best_h = h.copy()
            for delta in (-0.04, -0.01, -0.003, 0.003, 0.01, 0.04):
                hn = h.copy()
                hn[:] += -delta / (n - 1)
                hn[j] = min(max(h[j] + delta, 0.0), 1.0)
                hn = np.clip(hn, 0.0, 1.0)
                c = compute_c5(hn)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_h = hn
            if best_h is not h:
                h = best_h
                improved = True
        h = fix_sum(h, n)
        if not improved:

    return h

    # grad[j] = dx * sum_i W[i] * (1 - h[j+i-(n-1)]) for valid
    grad = np.zeros(n)
    for j in range(n):
        lo = max(0, n - 1 - j)
        hi = min(2 * n - 2, 2 * n - 2 - j)
        ws = W[lo:hi + 1]
        h_idx = np.arange(lo - (n - 1 - j), hi - (n - 1 - j) + 1)
        valid = (h_idx >= 0) & (h_idx < n)
        grad[j] = dx * np.sum(ws[valid] * (1.0 - h[h_idx[valid]]))
    return lse_val, grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-12:
        return h
    interior = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(interior)
    if k > 0:
        h[interior] += d / k
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-10:
        idx = np.argsort(h) if d > 0 else np.argsort(-h)
        for i in idx:
            if d > 0:
                adj = min(1.0 - h[i], d)
                h[i] += adj
                d -= adj
            else:
                adj = max(-h[i], d)
                h[i] += adj
                d -= adj
            if abs(d) < 1e-12:
                break
    return h

def fista_step(h, n, T, lr, iters):
    y = h.copy()
    z = h.copy()
    t = 1.0
    for k in range(iters):
        _, grad = lse_obj_and_grad(y, T)
        y_new = y - lr * grad
        y_new = project(y_new, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_mm = y_new + (t - 1.0) / t_new * (y_new - z)
        y_mm = project(y_mm, n)
        z = y_new
        y = y_mm
        t = t_new
    return z

def slsqp_opt(h, n, T, iters):
    def obj(x):
        return lse_obj_and_grad(x, T)[0]
    def con(x):
        return np.sum(x) - 0.5 * n
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(obj, h, method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': con}],
            options={'maxiter': iters, 'ftol': 1e-14})
        hn = project(r.x, n)
        cn = compute_c5(hn)
        if cn < c0:
            return hn, cn
    except:
        pass
    return h, c0

def pair_polish(h, n, rng, n_pairs=400):
    for _ in range(n_pairs):
        if n < 3:
            break
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-3 or h[j] >= 1.0 - 1e-3:
            continue
        if h[l] <= 1e-3 or h[l] >= 1.0 - 1e-3:
            continue
        c0 = compute_c5(h)
        best_d = 0.0
        for d in (-0.03, -0.01, -0.003, 0.003, 0.01, 0.03):
            hj, hl = h[j] + d, h[l] - d
            if hj < 1e-3 or hj > 1.0 - 1e-3 or hl < 1e-3 or hl > 1.0 - 1e-3:
                continue
            h[j], h[l] = hj, hl
            c = compute_c5(h)
            h[j], h[l] = hj - d, hl + d  # restore
            if c < c0 - 1e-13:
                best_d = d
                c0 = c
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return h

def coord_polish(h, n, n_pass=2):

                if c < c0 - 1e-13:

                    best_h = hn

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-12:
        return h
    interior = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(interior)
    if k > 0:
        h[interior] += d / k
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-10:
        idx = np.argsort(h) if d > 0 else np.argsort(-h)
        for i in idx:
            if d > 0:
                adj = min(1.0 - h[i], d)
            else:
                adj = max(h[i] + d, 0.0) - h[i]
            h[i] += adj
            d -= adj
            if abs(d) < 1e-12:
                break
    return h

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

    def obj(x):
        return lse_obj(x, T)
    def con(x):
        return np.sum(x) - 0.5 * n
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(obj, h, method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': con}],
            options={'maxiter': iters, 'ftol': 1e-14})
        hn = project(r.x, n)
        cn = compute_c5(hn)
        if cn < c0:
            return hn, cn
    except:
        pass
    return h, c0

                                            method='bounded',
                                            options={'xatol': 1e-10, 'maxiter': 200})
            h[j] = res.x
        h = project(h, n)
    return h

    for _ in range(n_pairs):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-3 or h[j] >= 1.0 - 1e-3:
            continue
        if h[l] <= 1e-3 or h[l] >= 1.0 - 1e-3:
            continue
        c0 = compute_c5(h)
        best_d = 0.0
        for d in (-0.04, -0.015, -0.005, 0.005, 0.015, 0.04):
            hj, hl = h[j] + d, h[l] - d
            if hj < 1e-3 or hj > 1.0 - 1e-3 or hl < 1e-3 or hl > 1.0 - 1e-3:

            h[j], h[l] = hj, hl
            c = compute_c5(h)
            h[j], h[l] = hj - d, hl + d

            h[l] = min(max(h[l] - best_d, 0.0), 1.0)

        h_save = h.copy()
        h[j] = 0.0
        corr_0 = np.correlate(h, 1.0 - h, mode='full')
        h[j] = 1.0
        corr_1 = np.correlate(h, 1.0 - h, mode='full')
        h = h_save
        A = corr_1 - corr_0
        B = corr_0.copy()
        midnight = n - 1
        def obj_1d(v):
            vals = A * v + B
            vals[midnight] = B[midnight] + v - v * v
            return np.max(vals) * dx
        res = optimize.minimize_scalar(obj_1d, bounds=(0.0, 1.0),
            method='bounded', options={'xatol': 1e-10, 'maxiter': 250})
        h[j] = res.x
    return project(h, n)

def pair_polish(h, n, rng, n_pairs=300):
    for _ in range(n_pairs):
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 5e-4 or h[j] >= 1.0 - 5e-4:
            continue
        if h[l] <= 5e-4 or h[l] >= 1.0 - 5e-4:
            continue
        c0 = compute_c5(h)
        best_d = 0.0
        best_c = c0
        for d in (-0.04, -0.02, -0.008, -0.003, 0.003, 0.008, 0.02, 0.04):
            hj, hl = h[j] + d, h[l] - d
            if hj < 5e-4 or hj > 1.0 - 5e-4 or hl < 5e-4 or hl > 1.0 - 5e-4:
                continue
            h[j], h[l] = hj, hl
            c = compute_c5(h)
            h[j], h[l] = hj - d, hl + d
            if c < best_c:
                best_c = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return project(np.interp(x1, x0, h, period=2.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 97)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    c5 = compute_c5(h)

    # Phase 1: Multiple coordinate descent passes (exact 1D)
    for p in range(10):
        if time.time() - t0 > budget_s - 280:
            break
        h = coord_pass(h, n, rng, random_order=(p % 2 == 1))
        cn = compute_c5(h)
        if cn < c5:
            c5 = cn

    # Phase 2: SLSQP refinement
    for T in (0.03, 0.01, 0.004, 0.0015):
        if time.time() - t0 > budget_s - 180:
            break
        h, c5 = slsqp_opt(h, n, T, 350)
        h = project(h, n)
        c5 = compute_c5(h)

    # Phase 3: More coordinate descent after SLSQP
    if time.time() - t0 < budget_s - 100:
        h = coord_pass(h, n, rng)
        h = coord_pass(h, n, rng, random_order=True)
        c5 = compute_c5(h)

    # Phase 4: Pair polish
    if time.time() - t0 < budget_s - 70:
        h = pair_polish(h, n, rng, 500)
        h = project(h, n)
        c5 = compute_c5(h)

    # Phase 5: Random restarts with coord descent
    for i in range(18):
        if time.time() - t0 > budget_s - 50:
            break
        ht = h.copy()
        mode = i % 5
        if mode == 0:
            ht += rng.randn(n) * 0.018
        elif mode == 1:
            idx = rng.choice(n, 12, replace=False)
            ht[idx] += rng.randn(12) * 0.14
        elif mode == 2:
            ht += rng.randn(n) * 0.01
        elif mode == 3:
            ht = h[rng.permutation(n)]
        else:
            fft_h = np.fft.rfft(h)
            fft_h[1:-1] += rng.randn(len(fft_h) - 2) * 0.06
            ht = np.real(np.fft.irfft(fft_h, n))
        ht = project(ht, n)
        ht = coord_pass(ht, n, rng)
        ht = coord_pass(ht, n, rng, random_order=True)
        ct = compute_c5(ht)
        if ct < c5:
            h, c5 = ht.copy(), ct
            if time.time() - t0 < budget_s - 40:
                h, c5 = slsqp_opt(h, n, 0.003, 250)
                h = project(h, n)
                h = coord_pass(h, n, rng)
                c5 = compute_c5(h)

    # Phase 6: Final fine SLSQP + coord
    if time.time() - t0 < budget_s - 25:
        for T, it in ((0.001, 1500), (0.0004, 2000)):
            if time.time() - t0 > budget_s - 15:
                break
            h, c5 = slsqp_opt(h, n, T, it)
            h = project(h, n)
            c5 = compute_c5(h)
        h = coord_pass(h, n, rng)
        h = pair_polish(h, n, rng, 200)

    # Phase 7: Try higher resolution if time remains
    if time.time() - t0 < budget_s - 60:
        n2 = 420
        h2 = upsample(h, n2)
        h2 = coord_pass(h2, n2, rng)
        h2 = coord_pass(h2, n2, rng, random_order=True)
        h2, c5_2 = slsqp_opt(h2, n2, 0.005, 300)
        h2 = project(h2, n2)
        h2 = coord_pass(h2, n2, rng)
        c5_2 = compute_c5(h2)
        if c5_2 < c5:
            h, n, c5 = h2, n2, c5_2

    h = project(h, n)
    c5 = compute_c5(h)
    return (h, c5, n)
