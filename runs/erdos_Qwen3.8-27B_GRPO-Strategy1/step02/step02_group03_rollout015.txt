import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts, iters=200):
    h = np.array(h, dtype=float).copy()
    for _ in range(iters):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0 - 1e-10) if d > 0 else (h > 1e-10)

def smooth_max_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    a = beta * (corr - mx)
    am = a.max()
    ea = np.exp(a - am)
    sw = ea.sum()

    obj = mx + (am + np.log(sw)) / beta
    o = n - 1
    w0 = w[o]

    wn = w[:o]

        if wn[df - 1] > 1e-14:
            gh[df:] += wn[df - 1] * (1.0 - h[:n - df])
            gh[:n - df] -= wn[df - 1] * h[df:]
    return obj, gh

def subgrad_step(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = int(corr.argmax())
    o = n - 1
    lag = k - o
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] = 1.0 - h[lag:]
        g[lag:] = -h[:n - lag]
    else:
        ll = -lag
        g[ll:] = 1.0 - h[:n - ll]
        g[:n - ll] = -h[ll:]
    return g

def block_descend(h0, dx, ts, dl, rng, blk=3, max_rounds=2000):

    h = proj(h0.copy(), ts)

def sla_refine(h0, dx, ts, dl, rng, T0=0.005, cool=0.998, max_rounds=3000):
