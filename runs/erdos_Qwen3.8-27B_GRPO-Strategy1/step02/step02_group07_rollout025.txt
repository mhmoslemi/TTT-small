    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # softmax weights
    c_max = np.max(corr)
    exps = np.exp((corr - c_max) / T)
    w = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        # k is the lag, corr index is n-1+k

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def project_sum(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    target = 0.5 * n
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        hc = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hc) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def lse_grad(h, T):
    """Compute gradient of LSE-smoothed max correlation and the current C5."""
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    cmax = np.max(corr)
    e = np.exp((corr - cmax)/T)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[n-1+k]
        if wk < 1e-14:
            continue
        if k >= 0:
            grad[:n-k] += wk*(1.0 - h[k:])
            grad[k:]   -= wk*h[:n-k]
        else:
            p = -k
            grad[p:]   += wk*(1.0 - h[:n-p])
            grad[:n-p] -= wk*h[p:]
    return (dx/T)*grad, cmax*dx

def grad_loop(h, T, n_iter, lr0=0.15, patience=80):

def compute_c5(h):

    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def project_sum(h, n):
    target = 0.5*n
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(h-mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def lse_grad(h, T):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    cmax = np.max(corr)
    e = np.exp((corr-cmax)/T)
    w = e/e.sum()
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[n-1+k]
        if wk < 1e-14: continue
        if k >= 0:
            grad[:n-k] += wk*(1.0-h[k:])
            grad[k:] -= wk*h[:n-k]
        else:
            p = -k
            grad[p:] += wk*(1.0-h[:n-p])
            grad[:n-p] -= wk*h[p:]
    return (dx/T)*grad, cmax*dx

def grad_loop(h, T, n_iter, lr0=0.2, patience=80):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    stall = 0
    for _ in range(n_iter):
        g, _ = lse_grad(h, T)
        h_try = project_sum(h - lr*g, n)
        c_try = compute_c5(h_try)
        if c_try < best_c5 - 1e-13:
            best = h_try.copy()
            best_c5 = c_try
            h = h_try
            lr = min(lr*1.05, 0.5)
            stall = 0
        else:
            stall += 1
            lr *= 0.7
            if stall > patience: break
            if lr < 1e-10: break
    return best, best_c5

def coord_polish(h, n, npass=3):
    for _ in range(npass):
        for j in range(n):
            if h[j] < 1e-4 or h[j] > 1.0-1e-4: continue
            bc = compute_c5(h)
            bh = h
            for d in (-0.05, -0.015, 0.015, 0.05):
                hn = h.copy()
                hn[j] += d
                hn = project_sum(hn, n)
                c = compute_c5(hn)
                if c < bc - 1e-13:
                    bc, bh = c, hn
            h = bh
        h = project_sum(h, n)
    return h

    x1 = np.arange(nn)*2.0/nn
    return project_sum(np.interp(x1, x0, h, period=2.0), nn)
