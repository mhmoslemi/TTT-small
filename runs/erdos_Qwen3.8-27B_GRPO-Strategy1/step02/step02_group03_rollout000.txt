def c5_grad_softmax(h, dx, beta, ntop=None):

def slsqp_min(h0, dx, ts, maxit=300):
    n = len(h0)
    t0 = compute_c5(h0, dx) / dx
    _g = {'n': n, 'ts': ts}
    def obj(x):
        return x[n]

        return x[:n].sum() - _g['ts']

    x0 = np.concatenate([h0, [t0 + 0.005]])
