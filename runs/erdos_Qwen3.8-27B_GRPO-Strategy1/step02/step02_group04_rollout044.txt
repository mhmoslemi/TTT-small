import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_val(corr, T):
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max
    shifted = corr - c_max
    shifted = np.clip(shifted, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return c_max + T * np.log(max(s, 1e-300))

def slsqp_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return soft_max_val(corr, T) * dx

def slsqp_con(x, n):
    return np.sum(x) - n * 0.5

def topk_obj(x, T, n, topk):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    tv = corr[topk]
    return soft_max_val(tv, T) * dx

def fix_sum(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12:
                    break
    return h

def do_slsqp(h, n, T, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}]
    try:
        res = opt.minimize(slsqp_obj, h.copy(), args=(T, n),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None:
            h_new = fix_sum(res.x, n)
            if compute_c5(h_new) <= compute_c5(fix_sum(h, n)) + 1e-14:
                return h_new
    except Exception:
        pass
    return h

def do_topk(h, n, T, max_iter, top_k):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}]
    try:
        res = opt.minimize(topk_obj, h.copy(), args=(T, n, top_k),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None:
            h_new = fix_sum(res.x, n)
            if compute_c5(h_new) <= compute_c5(fix_sum(h, n)) + 1e-14:
                return h_new
    except Exception:
        pass
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n)
    x_new = np.arange(new_n) * old_n / float(new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * new_n / float(old_n)
    h_new = np.interp(np.arange(new_n), x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, new_n)
