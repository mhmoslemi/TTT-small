def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n, rng):
    """Convert to binary with exactly n//2 ones"""
    h = h.copy()
    target = n // 2
    # Sort by value, keep top n/2 as 1
    idx = np.argsort(h)[-target:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def sa_binary(h, n, start_T, end_T, n_steps, rng):
    """Simulated annealing on binary sequences"""
    h = h.copy()
    cost = compute_c5(h)
    best_cost = cost
    best_h = h.copy()
    
    one_idx = np.where(h == 1)[0]
    zero_idx = np.where(h == 0)[0]
    
    for step in range(n_steps):
        # Annealing schedule
        frac = step / max(1, n_steps - 1)
        T = start_T * (end_T / start_T) ** frac
        
        # Random swap
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, rng):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[-target:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def sa_binary(h, T0, T1, n_steps, rng):
    n = len(h)
    h = h.copy()
    cost = compute_c5(h)
    best_cost = cost
    best_h = h.copy()
    one_idx = np.where(h > 0.5)[0]
    zero_idx = np.where(h <= 0.5)[0]
    for step in range(n_steps):
        frac = step / max(1, n_steps - 1)
        T = T0 * (T1 / T0) ** frac
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]
        h[i], h[j] = 0.0, 1.0
        new_cost = compute_c5(h)
        delta = new_cost - cost
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cost = new_cost
            one_idx = np.where(h > 0.5)[0]
            zero_idx = np.where(h <= 0.5)[0]
            if cost < best_cost:
                best_cost = cost
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_cost

def guided_ls(h, n_steps, rng):
    n = len(h)
    h = h.copy()
    cost = compute_c5(h)
    best_cost = cost
    best_h = h.copy()
    one_idx = np.where(h > 0.5)[0]
    zero_idx = np.where(h <= 0.5)[0]
    for _ in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = np.argmax(corr)
        shift = k_worst - (n - 1)
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]
        # Bias: prefer swaps that affect the worst lag region
        alt_i = (i + shift) % n
        alt_j = (j + shift) % n
        if rng.random() < 0.3:
            i, j = alt_i, alt_j
            if h[i] <= 0.5 and h[j] > 0.5:
                i, j = j, i
            if h[i] <= 0.5:
                i = one_idx[rng.randint(len(one_idx))]
        h[i], h[j] = 0.0, 1.0
        new_cost = compute_c5(h)
        if new_cost <= cost + 1e-15:
            cost = new_cost
            one_idx = np.where(h > 0.5)[0]
            zero_idx = np.where(h <= 0.5)[0]
            if cost < best_cost:
                best_cost = cost
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_cost

def multi_start_sa(h_parent, n, T0, T1, n_steps, n_starts, rng):
    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)
    for s in range(n_starts):
        if s == 0:
            h_s = h_parent.copy()
        else:
            scale = 0.15 * (1.0 + 0.3 * s)
            h_s = h_parent + rng.randn(n) * scale
        h_s = make_binary(h_s, rng)
        h_res, c5_res = sa_binary(h_s, T0 * (1.0 + 0.05 * s), T1, n_steps, rng)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    # Initialize from parent or default
    if 'initial_h_values' in dir() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float).copy()
    else:
        init = np.zeros(90)
        init[::2] = 1.0

    best_h = make_binary(init, rng)
    best_n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SA at parent resolution (n=90 typically)
    n_p = best_n
    n_starts_p = max(2, int(rem() * 0.45 / 25))
    n_starts_p = min(n_starts_p, 8)
    steps_per = max(10000, int(rem() * 0.45 / 0.025 / n_starts_p / 3))
    steps_per = min(steps_per, 200000)
    best_h, best_c5 = multi_start_sa(best_h, n_p, 0.14, 0.0003, steps_per, n_starts_p, rng)
    best_n = n_p

    # Phase 2: Guided local search polish
    if rem() > 30:
        n_ls = min(60000, int(rem() * 0.2 / 0.025))
        h_ls, c5_ls = guided_ls(best_h, n_ls, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    # Phase 3: Try other resolutions
    for n_try in [80, 100, 72, 60, 120]:
        if rem() < 30:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_start = make_binary(h_interp, rng)
        # Random start too
        h_rand = np.zeros(n_try)
        h_rand[rng.choice(n_try, n_try // 2, replace=False)] = 1.0

        n_steps_r = max(5000, min(100000, int(rem() * 0.3 / 0.025 / 2)))
        h_r1, c5_r1 = sa_binary(h_start, 0.12, 0.0005, n_steps_r, rng)
        if rem() < 10:
            break
        h_r2, c5_r2 = sa_binary(h_rand, 0.14, 0.0005, n_steps_r, rng)

        if c5_r1 < best_c5:
            best_c5 = c5_r1
            best_h = h_r1
            best_n = n_try
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2
            best_n = n_try
        # Polish best of these
        if rem() > 10:
            h_better = h_r1 if c5_r1 < c5_r2 else h_r2
            c5_better = min(c5_r1, c5_r2)
            h_pol, c5_pol = guided_ls(h_better, min(20000, int(rem() * 20)), rng)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
                best_n = n_try

    # Phase 4: Final SA restarts on best solution
    n_f = len(best_h)
    n_final = max(1, int(rem() * 0.3 / 25 / 3))
    for fi in range(min(n_final, 5)):
        if rem() < 15:
            break
        h_start = best_h + rng.randn(n_f) * 0.08
        h_start = make_binary(h_start, rng)
        h_res, c5_res = sa_binary(h_start, 0.06, 0.0001, min(50000, int(rem() * 20)), rng)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    # Final enforcement of binary + exact constraint
    best_h = make_binary(best_h, rng)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
