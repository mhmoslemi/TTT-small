    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # For coordinate p, compute A[m] and D[m] for all m
    # A[m] * t + D[m] = c[m] when t = h[p]
    # A[m] = (1-h[p+s]) if 0<=p+s<n else 0 - h[p-s] if 0<=p-s<n else 0
    # D[m] = c[m] - A[m] * h[p]
    
    s_vals = np.arange(-(n-1), n)  # shifts
    # For shift s, m = s + n - 1
    valid_plus = (p + s_vals >= 0) & (p + s_vals < n)
    valid_minus = (p - s_vals >= 0) & (p - s_vals < n)
    
    h_plus = np.where(valid_plus, h[p + s_vals], 0.0)  # only valid where True
    h_minus = np.where(valid_minus, h[p - s_vals], 0.0)

    D = c - A * h[p]

def project_feasible(h, n):

                                  time_limit=t3 - elapsed(), t_start=t0)
            h_up = np.interp(x_fine, x_coarse, h_c)
            h_up = project_feasible(h_up, n)
            for sw in range(30):
                if elapsed() > t3:
                    break
                h_up = coord_sweep(h_up, np.arange(n),
                                   time_limit=t3 - elapsed(), t_start=t0)
            c5 = compute_c5(h_up)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_up.copy()

    # Phase 4: Perturbation + coordinate descent (20% time)
    t4 = budget_s * 0.95
    n_cur = len(best_h)
    h_bh = best_h.copy()
    for it in range(50):
        if elapsed() > t4:
            break
        step = 0.06 * (1.0 - 0.7 * it / 50.0) + 0.015
        h_pert = h_bh + step * np.random.randn(n_cur)
        h_pert = project_feasible(h_pert, n_cur)
        for sw in range(15):
            if elapsed() > t4:
                break
            h_pert = coord_sweep(h_pert, np.random.permutation(n_cur),
                                 time_limit=t4 - elapsed(), t_start=t0)
        c5 = compute_c5(h_pert)
        temp = 0.004 * (1.0 - it / 50.0) + 0.0003
        if c5 < best_c5 or (it < 40 and np.random.rand() < np.exp(-(c5 - best_c5) / temp)):
            h_bh = h_pert.copy()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pert.copy()

    # Phase 5: Final polish with alternating orders (buffer)
    n_cur = len(best_h)
    for sweep in range(300):
        if elapsed() > budget_s * 0.99:
            break
        if sweep % 4 == 0:
            order = np.random.permutation(n_cur)
        elif sweep % 4 == 1:
            order = np.arange(n_cur)
        elif sweep % 4 == 2:
            order = np.arange(n_cur - 1, -1, -1)
        else:
            order = np.random.permutation(n_cur)
        h_temp = coord_sweep(best_h.copy(), order)
        c5 = compute_c5(h_temp)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_temp.copy()

    # Try snapping
    h_snap = snap_and_fix(best_h, len(best_h))
    for sw in range(20):
        if elapsed() > budget_s * 0.995:
            break
        h_snap = coord_sweep(h_snap, np.random.permutation(len(h_snap)))
    c5_snap = compute_c5(h_snap)
    if c5_snap < best_c5:
        best_c5 = c5_snap
        best_h = h_snap

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
