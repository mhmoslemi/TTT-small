            method='SLSQP', bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_it, 'ftol': 1e-16})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def do_topk_opt(h, n, top_k, T, max_it=3000):
    try:
        res = opt.minimize(
            topk_soft_obj, h.copy(), args=(T, n, top_k),
            method='SLSQP', bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_it, 'ftol': 1e-15})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def do_target_push(h, n, top_k, target_corr, max_it=3000):
    try:
        res = opt.minimize(
            target_push_obj, h.copy(), args=(n, top_k, target_corr),
            method='SLSQP', bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_it, 'ftol': 1e-15})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    def try_update(h_c, n_c):
        nonlocal h_best, n_best, c5_best
        h_c = fix_sum(h_c, n_c)
        c5_c = compute_c5(h_c)
        if c5_c < c5_best - 1e-10:
            h_best = h_c.copy()
            n_best = n_c
            c5_best = c5_c
            return True
        return False

    # Phase 1: Temperature sweep SLSQP
    T_seq = [0.06, 0.025, 0.01, 0.004, 0.0015, 0.0006, 0.0002, 0.00008]
    for T in T_seq:
        if budget_s - elapsed() < 60:
            break
        h_opt = do_slsqp(h_best, n_best, T, max_it=5000)
        try_update(h_opt, n_best)

    # Phase 2: Iterative active-shift equalization (key improvement)
    for rnd in range(6):
        if budget_s - elapsed() < 100:
            break
        corr = get_corr(h_best)
        c_max = np.max(corr)
        # Identify active shifts (within 3% of max)
        thr = c_max - 0.03 * c_max
        active = np.where(corr >= thr)[0]
        # Always include top 6
        top6 = np.argsort(corr)[-6:]
        active = np.unique(np.concatenate([active, top6]))
        # Limit to top 10
        if len(active) > 10:
            active = np.argsort(corr[active])[-10:] + np.empty(0, dtype=int)
            # re-select properly
            active_sorted = active[np.argsort(corr[active])][-10:]
            active = active_sorted

        # Step A: top-k soft optimization
        h_a = do_topk_opt(h_best, n_best, active, T=0.001, max_it=3000)
        try_update(h_a, n_best)

        # Step B: target push (try to bring all below current max)
        target = c_max * 0.998
        h_b = do_target_push(h_best, n_best, active, target, max_it=3000)
        try_update(h_b, n_best)

        # Step C: low-T SLSQP polish
        h_c = do_slsqp(h_best, n_best, 0.0003, max_it=4000)
        try_update(h_c, n_best)

    # Phase 3: Multi-start exploration
    perturbations = [
        ('noise_s', 0.02), ('noise_m', 0.04), ('noise_l', 0.07),
        ('noise_xl', 0.10), ('freq', 0.03), ('edge', 0.06)
    ]
    for name, scale in perturbations:
        if budget_s - elapsed() < 120:
            break
        if name.startswith('noise'):
            h_p = h_best + np.random.randn(n_best) * scale
        elif name == 'freq':
            k_freq = np.random.randint(2, 8)
            h_p = h_best + scale * np.sin(2 * np.pi * k_freq * np.arange(n_best) / n_best)
        else:
            h_p = h_best.copy()
            edges = np.where(np.abs(np.diff(h_best, append=h_best[0])) > 0.4)[0]
            for e in edges[:6]:
                lo = max(0, e - 4)
                hi = min(n_best, e + 5)
                h_p[lo:hi] += np.random.randn(hi - lo) * scale
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = fix_sum(h_p, n_best)
        h_o1 = do_slsqp(h_p, n_best, 0.008, max_it=2000)
        h_o2 = do_slsqp(h_o1, n_best, 0.0008, max_it=3000)
        try_update(h_o2, n_best)

    # Phase 4: Resolution exploration
    for n_try in [400, 480, 540]:
        if budget_s - elapsed() < 130:
            break
        x_old = np.arange(n_best) * (2.0 / n_best)
        x_new = np.arange(n_try) * (2.0 / n_try)
        h_new = np.interp(x_new, x_old, h_best, period=2.0)
        h_new = fix_sum(h_new, n_try)
        for T in [0.01, 0.002, 0.0005]:
            if budget_s - elapsed() < 60:
                break
            h_opt = do_slsqp(h_new, n_try, T, max_it=3000)
            if try_update(h_opt, n_try):
                h_new = h_best.copy()
                x_new2 = np.arange(n_try) * (2.0 / n_best)

    # Phase 5: Final intensive polish
    for T in [0.00015, 0.00006, 0.00002]:
        if budget_s - elapsed() < 35:
            break
        h_opt = do_slsqp(h_best, n_best, T, max_it=6000)
        try_update(h_opt, n_best)

    # Phase 6: One more active balance at finest T
    if budget_s - elapsed() > 50:
        corr = get_corr(h_best)
        c_max = np.max(corr)
        active = np.argsort(corr)[-8:]
        h_ab = do_target_push(h_best, n_best, active, c_max * 0.997, max_it=4000)
        try_update(h_ab, n_best)
        h_fp = do_slsqp(h_best, n_best, 0.00005, max_it=5000)
        try_update(h_fp, n_best)

    # Final cleanup
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
