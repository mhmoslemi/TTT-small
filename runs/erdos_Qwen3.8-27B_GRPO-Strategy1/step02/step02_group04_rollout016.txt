            # grad[j] -= h[j+lag] for j in [0, n-1-lag]
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]

            # grad[j] -= h[j-l] for j in [l, n-1]
            g[:n-l] += (1.0 - h[l:])
            g[l:] -= h[:n-l]
        grad += g
    grad /= len(top_k)
    return corr, top_k, grad

def compute_c5(h):
    n = len(h)

    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Use bisection on shift
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2
        h_t = np.clip(h - mid, 0.0, 1.0)
        if h_t.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)

def corr_and_subgrad(h, n, K):

    corr = np.correlate(h, 1.0 - h, mode='full')

        if gnorm < 1e-15:

        grad = grad / gnorm
        # Nesterov-like momentum
        v = mom * v + lr * grad
        h_new = project_simplex(h - v, n)
        c5_new = compute_c5(h_new)
        h = h_new
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h.copy()
            patience = 0

            v *= 0.5

    return best_h, best_c5

def do_slsqp_polish(h, n, T, max_iter):
    """SLSQP with soft-max for final polish"""
    def soft_obj(x, T):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        if T < 1e-15:
            return float(c_max * 2.0 / n)
        shifted = np.clip(corr - c_max, -80*T, 0)
        s = np.sum(np.exp(shifted / T))
        return float((c_max + T * np.log(max(s, 1e-300))) * 2.0 / n)
    
    def sum_con(x):
        return x.sum() - n * 0.5
    
    try:
        res = opt.minimize(soft_obj, h.copy(), args=(T,),
                          method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': sum_con}],
                          options={'maxiter': max_iter, 'ftol': 1e-16})
        if hasattr(res, 'x') and res.x is not None:

    except:

    return h

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        h_t = np.clip(h - mid, 0.0, 1.0)
        if h_t.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def project_simplex(h, n):
    target = n * 0.5
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2
        h_t = np.clip(h - mid, 0.0, 1.0)
        if h_t.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def soft_max_obj_func(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return float(c_max * 2.0 / n)
    shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
    s = np.sum(np.exp(shifted / T))
    return float((c_max + T * np.log(max(s, 1e-300))) * 2.0 / n)

def sum_con_func(x, n):
    return x.sum() - n * 0.5

def slsqp_polish(h, n, T, max_iter):
    try:
        res = opt.minimize(
            soft_max_obj_func, h.copy(), args=(T, n),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con_func, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-16})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def upsample_func(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)
