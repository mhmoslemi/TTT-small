import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    interior = (h > 1e-9) & (h < 1.0 - 1e-9)
    n_int = int(np.sum(interior))
    if n_int > 0 and abs(diff) < n_int * 0.3:
        h[interior] += diff / n_int
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff < 1e-13:
                    break
        else:
            d = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(d, h[i])
                h[i] -= rem
                d -= rem
                if d < 1e-13:
                    break
    return np.clip(h, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def compute_grad(h, w):
    """Gradient of sum_k w[k]*c_k w.r.t. h, where c = correlate(h, 1-h, 'full')"""
    n = len(h)
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    j = np.arange(n)
    jm = j[:, None] - lags[None, :]  # j - lag, shape (n, 2n-1)
    jp = j[:, None] + lags[None, :]  # j + lag, shape (n, 2n-1)
    vm = (jm >= 0) & (jm < n)
    vp = (jp >= 0) & (jp < n)
    hm_idx = np.clip(jm, 0, n-1)
    hp_idx = np.clip(jp, 0, n-1)
    term1 = (w[None, :] * (1.0 - h[hm_idx]) * vm).sum(axis=1)
    term2 = (w[None, :] * h[hp_idx] * vp).sum(axis=1)
    return term1 - term2

def subgradient_optimize(h0, n, t_start, t_end, n_iter, T_init, T_end):
    """Projected subgradient with annealed temperature"""
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    for it in range(n_iter):
        elapsed = time.time() - t_start
        if t_end - elapsed < 10:
            break
        frac = it / max(n_iter - 1, 1)
        T = T_init * (T_end / T_init) ** frac
        m = np.max(corr)
        w = np.exp(np.clip(corr - m, -50.0 * T, 0.0) / T)
        w_sum = w.sum()

                t1 = a + (b - a) / 3.0
                t2 = b - (b - a) / 3.0
                f1 = np.max(c_fixed + t1 * g)
                f2 = np.max(c_fixed + t2 * g)
                if f1 < f2:
                    b = t2
                else:
                    a = t1
            t_opt = (a + b) / 2.0
            delta = t_opt - h[j]
            if abs(delta) > 1e-12:
                h[j] = t_opt
                corr += delta * g
    return fix_sum(h, n)

def slsqp_polish(h0, n, T, max_iter, t_start, t_end):
    if time.time() - t_start > t_end - 60:
        return h0
    try:
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(c)
            shifted = np.clip(c - m, -80.0 * T, 0.0)
            s = np.sum(np.exp(shifted / T))
            return (m + T * np.log(max(s, 1e-300))) * (2.0 / n)
        
        def obj_grad(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(c)
            w = np.exp(np.clip(c - m, -80.0 * T, 0.0) / T)
            w /= w.sum()
            g = compute_grad(x, w)
            return g * (2.0 / n)
        
        res = opt.minimize(obj, h0, jac=obj_grad, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n*0.5,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': max_iter, 'ftol': 1e-14})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    t_end = t_start + budget_s
    
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = fix_sum(h, n)
    c5_best = compute_c5(h)
    
    def elapsed():
        return time.time() - t_start
    
    def try_update(h_cand, n_cand):
        nonlocal h, n, c5_best
        h_cand = fix_sum(h_cand, n_cand)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best - 1e-15:
            h = h_cand
            n = n_cand
            c5_best = c5_cand
            return True
        return False
    
    # Phase 1: Subgradient optimization (annealed)
    if t_end - time.time() > 120:
        h1 = subgradient_optimize(h, n, t_start, t_end - 120, 20000, 0.05, 0.001)
        try_update(h1, n)
    
    # Phase 2: Coordinate descent (main workhorse)
    if t_end - time.time() > 200:
        h2 = coord_descent(h, n, 300, t_start, t_end - 200)
        try_update(h2, n)
    
    # Phase 3: CD + Subgradient interleaved
    if t_end - time.time() > 200:
        h3 = subgradient_optimize(h, n, t_start, t_end - 200, 10000, 0.01, 0.0005)
        try_update(h3, n)
        h3b = coord_descent(h, n, 200, t_start, t_end - 200)
        try_update(h3b, n)
    
    # Phase 4: SLSQP polish
    if t_end - time.time() > 120:
        h4 = slsqp_polish(h, n, 0.005, 500, t_start, t_end)
        try_update(h4, n)
        h4 = slsqp_polish(h, n, 0.001, 500, t_start, t_end)
        try_update(h4, n)
    
    # Phase 5: More CD
    if t_end - time.time() > 100:
        h5 = coord_descent(h, n, 500, t_start, t_end - 100)
        try_update(h5, n)
    
    # Phase 6: Multi-resolution
    for n_try in [240, 480]:
        if t_end - time.time() < 200:
            break
        if n_try > n:
            h_init = upsample(h, n_try)
        else:
            h_init = downsample(h, n_try)
        h_opt = coord_descent(h_init, n_try, 100, t_start, t_end)
        h_opt = subgradient_optimize(h_opt, n_try, t_start, t_end, 5000, 0.01, 0.001)
