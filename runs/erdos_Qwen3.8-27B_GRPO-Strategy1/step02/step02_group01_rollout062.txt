import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h, n):
    if len(h) != n:
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n)
        h = np.interp(x_new, x_old, h)
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def propose_move(h, n, k, rng):
    """Flip k pairs of bits (1->0 and 0->1)."""
    h_new = h.copy()
    for _ in range(k):
        ones = np.where(h_new == 1)[0]
        zeros = np.where(h_new == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_new[i] = 0.0
        h_new[j] = 1.0
    return h_new

def local_search(h, n, max_rounds=30, time_limit=None, t_start=None):
    """Steepest descent: try all flip pairs, take best improvement."""
    cur_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit and t_start and (time.time() - t_start) > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_val = cur_c5
        best_i, best_j = -1, -1
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                c5 = compute_c5(h)
                if c5 < best_val:
                    best_val = c5
                    best_i, best_j = i, j
                h[j] = 0
            h[i] = 1
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            cur_c5 = best_val
        else:
            break
    return h, cur_c5

def sa_optimize(h, n, n_iter, T0, T_decay, rng, time_limit=None, t_start=None):
    """Simulated annealing on binary space."""
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    T = T0
    for it in range(n_iter):
        if time_limit and t_start and (time.time() - t_start) > time_limit:
            break
        k = 1 + rng.integers(0, 3)  # 1, 2, or 3 pairs
        new_h = propose_move(cur_h, n, k, rng)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        T *= T_decay
    return best_h, best_c5

def slsqp_erdos(h, maxiter=300):
    n = len(h)
    def erdos_obj(x):
        return x[-1]
    def erdos_eq(x):
        return x[:n].sum() - n / 2.0
    def erdos_ineq(x):
        h_local = x[:n]
        t = x[-1]
        corr = np.correlate(h_local, 1.0 - h_local, mode='full')
        return t - np.max(corr)
    t0 = compute_c5(h) + 0.02
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': erdos_eq},
        {'type': 'ineq', 'fun': erdos_ineq}
    ]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return np.clip(res.x[:n], 0, 1)
    except Exception:
        return h

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:

        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-12:

            h += room * (diff / total)
        else:

            total = room.sum()
            if total < 1e-12:
                break
            h += room * (diff / total)

import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-12:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-12:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    return x[:-1].sum() - len(x[:-1]) / 2.0

def erdos_ineq(x):
    h_local = x[:-1]
    t = x[-1]
    corr = np.correlate(h_local, 1.0 - h_local, mode='full')
    return t - float(np.max(corr))

def slsqp_erdos(h, maxiter=400):
    n = len(h)
    t0 = compute_c5(h) + 0.03
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': erdos_eq},
        {'type': 'ineq', 'fun': erdos_ineq}
    ]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return np.clip(res.x[:n], 0, 1)
    except Exception:
        return h

            h_start = propose_move(best_h, n0, 5, rng)
        else:
            k_pert = min(10, 2 + r)
            h_start = propose_move(best_h, n0, k_pert, rng)
        T0_sa = 0.05 * (0.8 ** r)
        h_sa, c5_sa = sa_optimize(h_start, n0, sa_time,
                                   t0_global + elapsed() - sa_time, rng, T0=T0_sa)
        # Local search after SA
        h_ls, c5_ls = local_search(h_sa, n0, max_rounds=25,
                                    time_limit=budget_s * 0.02, t_start=t0_global)
        c5_candidate = min(c5_sa, c5_ls)
        h_candidate = h_ls if c5_ls <= c5_sa else h_sa
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()

    # Phase 3: First-improve search (different flavor)
    if elapsed() < budget_s * 0.72:
        h_fi, c5_fi = first_improve_search(best_h, n0, max_rounds=100,
                                            time_limit=budget_s * 0.04, t_start=t0_global)
        if c5_fi < best_c5:
            best_c5 = c5_fi
            best_h = h_fi

    # Phase 4: Try other resolutions
    for n_try in [80, 100, 120]:
        if elapsed() > budget_s * 0.85:
            break
        if n_try % 2 != 0:
            continue
        h_init = to_binary(best_h, n_try)
        h_ls, c5_ls = local_search(h_init, n_try, max_rounds=15,
                                    time_limit=budget_s * 0.015, t_start=t0_global)
        sa_time = min(budget_s * 0.03, max(5, (budget_s * 0.85 - elapsed()) / 3))
        h_sa, c5_sa = sa_optimize(h_ls, n_try, sa_time,
                                   t0_global + elapsed() - sa_time, rng, T0=0.04)
        h_ls2, c5_ls2 = local_search(h_sa, n_try, max_rounds=15,
                                      time_limit=budget_s * 0.015, t_start=t0_global)
        c5_cand = min(c5_ls, c5_sa, c5_ls2)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = (h_ls if c5_cand == c5_ls else (h_sa if c5_cand == c5_sa else h_ls2)).copy()

    # Phase 5: Directed local search targeting worst shifts
    if elapsed() < budget_s * 0.90:
        h_dir = best_h.copy()
        for _ in range(30):
            if time.time() - t0_global > budget_s * 0.90:
                break
            corr = np.correlate(h_dir, 1.0 - h_dir, mode='full')
            peak = np.max(corr)
            peak_idx = np.argmax(corr)
            d = peak_idx - (len(h_dir) - 1)
            improved = False
            ones = np.where(h_dir > 0.5)[0]
            zeros = np.where(h_dir < 0.5)[0]
            for i in ones:
                j_pos = i - d
                j_neg = i + d
                h_dir[i] = 0.0
                for j in zeros:
                    h_dir[j] = 1.0
                    c5_test = compute_c5(h_dir)
                    h_dir[j] = 0.0
                    if c5_test < peak * (2.0 / len(h_dir)) - 1e-12:
                        h_dir[j] = 1.0
                        c5_new = compute_c5(h_dir)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_dir.copy()
                        improved = True
                        break
                if improved:
                    break
                h_dir[i] = 1.0
            if not improved:
                break

    # Phase 6: Continuous polish
    if elapsed() < budget_s * 0.96:
        h_cont = slsqp_erdos(best_h, maxiter=500)
        h_cont = project_feasible(h_cont, len(h_cont))
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
