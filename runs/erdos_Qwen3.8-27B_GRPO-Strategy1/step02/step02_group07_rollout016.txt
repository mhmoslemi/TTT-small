    for _ in range(200):
        d = 0.5 * n - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        if not np.any(m):
            break

def project_feasible(h, n):
    h = h.copy()

def lse_value(h, T):
    c = corr_full(h)
    dx = 2.0 / len(h)
    cmax = np.max(c)
    if T < 1e-14:
        return cmax * dx
    s = np.sum(np.exp((c - cmax) / T))
    return (cmax + T * np.log(s)) * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = corr_full(h)
    cmax = np.max(c)
    ex = np.exp((c - cmax) / T)
    w = ex / ex.sum()
    gh = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    nh = n - 1
    for idx in nz:
        s = idx - nh
        wi = w[idx]
        if s >= 0:
            gh[:n-s] += wi * (1.0 - h[s:])
            gh[s:] -= wi * h[:n-s]
        else:
            ss = -s
            gh[ss:] += wi * (1.0 - h[:n-ss])
            gh[:n-ss] -= wi * h[ss:]
    return dx * gh

        elif mode == 3:
            ht = 1.0 - ht + rng.randn(n) * 0.01
        elif mode == 4:
            ht = ht[::-1].copy() + rng.randn(n) * 0.015
        else:
            noise_scale = 0.01 * (1 + i * 0.3)
            ht += rng.randn(n) * noise_scale
        ht = fix_sum(ht, n)
        for T in (0.015, 0.005, 0.0015):
            ht = lbfgsb_opt(ht, n, T, 1200)
            ht = fix_sum(ht, n)
        ct = compute_c5(ht)
        if ct < c5:
            h, c5 = ht, ct
            if time.time() < deadline - 40:
                h, c5 = subgrad_descent(h, n, 0.05, 400, gap=0.01)
                h = pair_descent(h, n, rng, 800)
                c5 = compute_c5(h)

    # Phase 6: Final SLSQP + subgradient
    for T, it in [(0.0003, 2000), (0.0001, 2500)]:
        if time.time() > deadline - 15:
            break
        h = slsqp_opt(h, n, T, it)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
    if time.time() < deadline - 10:
        h, c5 = subgrad_descent(h, n, 0.03, 300, gap=0.008)

    # Phase 7: Higher resolution attempt
    if time.time() < deadline - 50:
        n2 = 400
        h2 = upsample_h(h, n2)
        for T, it in [(0.01, 1500), (0.003, 2000), (0.001, 2500)]:
            if time.time() > deadline - 20:
                break
            h2 = lbfgsb_opt(h2, n2, T, it)
            h2 = fix_sum(h2, n2)
        h2 = pair_descent(h2, n2, rng, 1000)
        c2 = compute_c5(h2)
        if c2 < c5:
            h, n, c5 = h2, n2, c2
            c5 = compute_c5(h)

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
