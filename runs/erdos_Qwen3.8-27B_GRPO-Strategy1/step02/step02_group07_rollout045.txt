import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj(h, T, penalty):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        val = c * dx
    else:
        shifted = (corr - c) / T
        ms = np.max(shifted)
        exp_s = np.exp(shifted - ms)
        val = (c + T * (ms + np.log(np.sum(exp_s)))) * dx
    s = np.sum(h) - 0.5 * len(h)
    return val + penalty * s * s

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
        h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def lbfgsb(h, n, T, penalty, maxiter):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T, penalty), method='L-BFGS-B',
            bounds=[(0., 1.)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-13})
        hn = fix_sum(r.x, n)
        cn = compute_c5(hn)
        if cn < c0:
            return hn, cn
    except Exception:
        pass
    return h, c0

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)

def coord_cd(h, n, n_pass=2, step=0.004):
    for _ in range(n_pass):
        improved = False
        cur = compute_c5(h)
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            for d in (-step, -step * 0.3, step * 0.3, step):
                hn = h.copy()
                hn[j] = min(max(h[j] + d, 0.0), 1.0)
                hn = fix_sum(hn, n)
                c = compute_c5(hn)
                if c < cur - 1e-13:
                    cur = c
                    h[:] = hn
                    improved = True
                    break
        if not improved:
            break
    return h

def pair_cd(h, n, rng_obj, n_pairs=300):
    for _ in range(n_pairs):
        j, l = rng_obj.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4 or h[l] <= 1e-4 or h[l] >= 1.0 - 1e-4:
            continue

            if c < cur - 1e-13:

    return fix_sum(h, n)
