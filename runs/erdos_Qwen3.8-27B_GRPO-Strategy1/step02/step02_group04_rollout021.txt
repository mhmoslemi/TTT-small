    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

def fix_sum(h, n):
    target = n * 0.5
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Project onto {0 <= h <= 1, sum(h) = n/2} using bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_shifted = np.clip(h + mid, 0.0, 1.0)
        s_new = np.sum(h_shifted)
        if s_new > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def soft_max_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # Soft max
    c_max = np.max(corr)
    shifted = (corr - c_max) / T
    shifted = np.clip(shifted, -80.0, 0.0)
    exps = np.exp(shifted)
    Z = np.sum(exps)  # partition function
    obj = (c_max + T * np.log(Z)) * dx
    
    # Gradient: dObj/dh[i] = dx * sum_k w[k] * dcorr[k]/dh[i] / Z
    # where w[k] = exp(corr[k]/T)
    # dcorr[k]/dh[i] = (1 - h[(i+k) mod n]) - h[(i-k) mod n]
    # = (1 - h) shifted by k, minus h shifted by -k
    # 
    # sum_k w[k] * (1-h[(i+k) mod n]) = (w convolve (1-h))[i] = (w * (1-h))[i]
    # sum_k w[k] * h[(i-k) mod n] = (w * h)[i]  (convolution)

    # np.correlate(a, b, 'full')[i] = sum_j a[j] * conj(b[j-(i-len(b)+1)])
    # For same length, np.convolve is easier

    g2 = np.real(np.fft.ifft(W * hH))
    grad = dx * (g1 - g2)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def corr_full(h):
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

def grad_from_weights(w, h):
    n = len(h)
    dx = 2.0 / n
    w_rev = w[::-1]
    conv1 = np.convolve(1.0 - h, w, mode='full')
    conv2 = np.convolve(h, w_rev, mode='full')
    return (conv1[n-1:2*n-1] - conv2[n-1:2*n-1]) * dx

def soft_max_weights(corr, T):
    cmax = np.max(corr)
    shifted = (corr - cmax) / T
    shifted = np.clip(shifted, -80.0, 0.0)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    obj_val = cmax + T * np.log(max(Z, 1e-300))
    return exps / Z, obj_val

def topK_weights(corr, K):
    n_c = len(corr)
    w = np.zeros(n_c)
    idx = np.argsort(corr)[-K:]
    w[idx] = 1.0 / K
    return w, np.dot(w, corr)

def adam_opt(h, n, T, n_iters, lr, use_topK, t0, budget):
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    h_curr = h.copy()
    for it in range(n_iters):
        if time.time() - t0 > budget - 50:
            break
        corr = corr_full(h_curr)
        if use_topK > 0:
            w, obj_val = topK_weights(corr, use_topK)
            grad = grad_from_weights(w, h_curr)
        else:
            w, obj_val = soft_max_weights(corr, T)
            grad = grad_from_weights(w, h_curr)
        t = it + 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h_curr = fix_sum(h_curr - lr * m_hat / (np.sqrt(v_hat) + eps), n)
    return h_curr

def slsqp_objective(x, n):
    x2 = fix_sum(x, n)
    return compute_c5(x2)

def slsqp_constraint(x, n):
    return np.sum(x) - n * 0.5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)

    c5_best = compute_c5(h)
    best_h = h.copy()

    T_sched = [0.06, 0.025, 0.01, 0.004, 0.0015, 0.0006, 0.0002]
    for T in T_sched:

        n_it = 2000
        lr = 6.0 / n

        c = compute_c5(h_opt)
        if c < c5_best:

    # Phase 2: Multi-start perturbations

        for lr in [0.008, 0.003, 0.001]:

            h_try = np.interp(x_new, x_old, best_h, period=2.0)
