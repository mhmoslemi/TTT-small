import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_up(h, nd):
    n = len(h)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xd = (np.arange(nd) + 0.5) * 2.0 / nd
    return np.interp(xd, xs, h)

def binary_sa(h, n, T_start, T_end, n_iter, rng, kick_every=4000):
    h = h.copy().astype(float)
    idx_1 = np.where(h > 0.5)[0].copy()
    idx_0 = np.where(h <= 0.5)[0].copy()
    n1 = len(idx_1)
    nx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(c) * nx)
    best_h = h.copy()
    cur_c5 = best_c5
    for it in range(n_iter):
        if it > 0 and kick_every > 0 and it % kick_every == 0:
            for _ in range(4):
                ia = rng.randint(n1)
                ib = rng.randint(n1)
                h[idx_1[ia]], h[idx_0[ib]] = 0.0, 1.0
                idx_1[ia], idx_0[ib] = idx_0[ib], idx_1[ia]
            cur_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * nx)
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        ia = rng.randint(n1)
        ib = rng.randint(n1)
        a, b = idx_1[ia], idx_0[ib]
        h[a] = 0.0
        h[b] = 1.0
        c = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(c) * nx)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / (T * 0.001))):
            cur_c5 = new_c5
            idx_1[ia] = b
            idx_0[ib] = a
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
    return best_h, best_c5

def greedy_ls(h, n, max_passes, rng):
    h = h.copy().astype(float)
    idx_1 = np.where(h > 0.5)[0].copy()
    idx_0 = np.where(h <= 0.5)[0].copy()
    n1 = len(idx_1)
    nx = 2.0 / n
    cur_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * nx)
    for p in range(max_passes):
        improved = False
        for ia in rng.permutation(n1):
            for ib in rng.permutation(n1):
                a, b = idx_1[ia], idx_0[ib]
                if a == b:
                    continue
                h[a], h[b] = 0.0, 1.0
                new_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * nx)
                if new_c5 < cur_c5 - 1e-15:
                    cur_c5 = new_c5
                    idx_1[ia], idx_0[ib] = b, a
                    improved = True
                else:
                    h[a], h[b] = 1.0, 0.0
        if not improved:
            break
    return h, cur_c5

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))
