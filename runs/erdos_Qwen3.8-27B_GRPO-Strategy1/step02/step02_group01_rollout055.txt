def project_feasible(h, n):

def targeted_descent(h, max_iter=1000, lr_init=0.04):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = lr_init
    for it in range(max_iter):
        corr = np.correlate(best, 1.0 - best, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best = best.copy()
        # Top 5% of lags
        thresh = np.percentile(corr, 95)
        mask = corr >= thresh
        # Compute gradient: for each bottleneck lag k, 
        # d corr[k]/d h[j] contributes
        # Use vectorized approach
        # corr[k] = sum_j best[j] * (1 - best[j + k - (n-1)])  (for valid j)
        # = sum_j best[j] - best[j]*best[j+k-(n-1)]
        # d/d best[j] = 1 (from first term) - best[j+k-(n-1)] (from second)
        #                - best[j-1-(k-n+1)] ... (if j appears in second factor)
        # This is getting complex. Let me use a simpler approach:
        # Perturb h in direction that reduces high-correlation lags
        # For lag k (in full mode), offset = k - (n-1)
        # corr[k] = sum_{j: 0<=j<n, 0<=j+offset<n} best[j]*(1-best[j+offset])
        # If offset > 0: best[j] * (1-best[j+offset]) for j in [0, n-1-offset)
        # If offset < 0: best[j] * (1-best[j+offset]) for j in [-offset, n)
        
        # Simpler: just move h in the direction that reduces the max
        # Gradient of max(corr) w.r.t. h is the average gradient over bottleneck lags
        # For each k in bottleneck:
        #   d corr[k] / d h[j] = (1-h[j+offset_k]) if j+offset_k valid, plus
        #                        -h[j-offset_k] if j-offset_k valid (for the other term)
        
        # Let me just do a steepest descent with finite differences on a subset
        # Actually, let me use a simpler heuristic:
        # Identify positions where h=1 that contribute to many bottleneck lags
        # and positions where h=0 that could be changed to 1
        
        # Even simpler: use the smooth max gradient
        T = 50
        m = np.max(corr)
        weights = np.exp(T * (corr - m))
        weights /= weights.sum()
        
        # Gradient of soft_max(corr) w.r.t. h
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-10:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                # corr[k] = sum_{j=0}^{n-1-offset} h[j]*(1-h[j+offset])
                j_range = range(n - offset)
                for j in j_range:
                    grad[j] += w * (1.0 - best[j + offset])
                    grad[j + offset] += w * (-best[j])
            else:
                # corr[k] = sum_{j=-offset}^{n-1} h[j]*(1-h[j+offset])
                # = sum_{j=0}^{n-1+(-offset)} h[j+offset]*(1-h[j])
                # Let me re-derive: offset < 0, let d = -offset > 0
                # corr[k] = sum_{j=d}^{n-1} h[j]*(1-h[j-d])
                # = sum_{j=d}^{n-1} h[j] - h[j]*h[j-d]
                d = -offset
                for j in range(d, n):
                    grad[j] += w * (1.0 - best[j - d])
                    grad[j - d] += w * (-best[j])
        
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            new_h = best - lr * grad / gnorm
            new_h = project_feasible(new_h, n)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best = new_h
                lr = min(lr * 1.05, 0.1)
            else:
                lr *= 0.9
        if lr < 0.001:
            break
    return best, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(smooth_c5, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)

        return h

        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho

    # Phase 6: Targeted descent again (refined)
    if elapsed() < budget_s * 0.92:
        h_td2, c5_td2 = targeted_descent(best_h, max_iter=600, lr_init=0.02)
        c5_td2 = compute_c5(h_td2)
        if c5_td2 < best_c5:
            best_c5, best_h = c5_td2, h_td2

    # Phase 7: Final polish
    if elapsed() < budget_s * 0.97:
        hf = slsqp_erdos(best_h, maxiter=500)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        hs2 = slsqp_smooth(best_h, 5000, maxiter=300)
        cs2 = compute_c5(hs2)
        if cs2 < best_c5:
            best_c5, best_h = cs2, hs2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
