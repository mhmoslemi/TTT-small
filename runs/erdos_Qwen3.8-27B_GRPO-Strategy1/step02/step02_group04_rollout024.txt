def do_slsqp(h, n, T, max_iter):
    def obj(x):
        return soft_max_obj(x, T)
    def obj_grad(x):
        return soft_max_grad(x, T)
    def sum_con(x, nn):
        return np.sum(x) - nn * 0.5
    try:
        res = opt.minimize(
            obj, h.copy(), args=(),
            jac=obj_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-18}

            h_cd2, _ = coord_descent(h_best, n_best, max_rounds=10,
                                      t_limit=min(20, budget_s - elapsed() - 10))
            try_update(h_cd2, n_best)

    # Phase 7: Final polish - very low T SLSQP + coordinate descent
    if budget_s - elapsed() > 40:
        h_opt = do_slsqp(h_best, n_best, 0.00005, max_iter=3000)
        try_update(h_opt, n_best)
    if budget_s - elapsed() > 20:
        h_cd, _ = coord_descent(h_best, n_best, max_rounds=15,
                                 t_limit=min(15, budget_s - elapsed() - 5))
        try_update(h_cd, n_best)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
