import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0 and abs(diff) < n_int * 0.45:
        h[interior] += diff / n_int
    else:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                space = 1.0 - h[i]
                add = min(diff, space)
                h[i] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            diff = -diff
            idx = np.argsort(-h)
            for i in idx:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, T, alpha_vol=0.0, k_vol=5):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        obj = c_max * dx
        m = np.argmax(corr)
        weights = np.zeros(2 * n - 1)
        weights[m] = 1.0
    else:
        shifted = np.clip((corr - c_max) / T, -80.0, 0.0)
        exp_s = np.exp(shifted)
        s = np.sum(exp_s)
        weights = exp_s / s
        obj = (c_max + T * np.log(s + 1e-300)) * dx
    if alpha_vol > 1e-15:
        top_idx = np.argsort(corr)[-k_vol:]
        top_vals = corr[top_idx]
        vol = np.var(top_vals)
        obj += alpha_vol * vol * dx
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m] * dx
        if w < 1e-18:
            continue
        lag = m - n + 1
        if lag >= 0:
            L = n - lag
            grad[lag:] += w * omh[:L]
            grad[:L] -= w * h[lag:]
        else:
            L = n + lag
            grad[:L] += w * omh[-lag:]
            grad[-lag:] -= w * h[:L]
    if alpha_vol > 1e-15:
        top_idx = np.argsort(corr)[-k_vol:]
        top_vals = corr[top_idx]
        mean_tv = np.mean(top_vals)
        for mi in top_idx:
            tv = top_vals[top_idx.tolist().index(mi)]
            dl = (tv * 2.0 / k_vol - 2.0 * alpha_vol / k_vol) * (2.0 * (tv - mean_tv)) / k_vol
            dl *= alpha_vol * dx
            if abs(dl) < 1e-20:
                continue
            lag = mi - n + 1
            if lag >= 0:
                L2 = n - lag
                grad[lag:] += dl * omh[:L2]
                grad[:L2] -= dl * h[lag:]
            else:
                L2 = n + lag
                grad[:L2] += dl * omh[-lag:]
                grad[-lag:] -= dl * h[:L2]
    return obj, grad

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def slsqp_optimize(h, n, T, max_iter=2000, alpha_vol=0.0, k_vol=5):
    def obj(x):
        return lse_obj_grad(x, T, alpha_vol, k_vol)[0]
    def jac(x):
        return lse_obj_grad(x, T, alpha_vol, k_vol)[1]
    try:
        res = opt.minimize(
            obj, h.copy(), jac=jac,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: sum_con(x, n),
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0}
        )
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    return upsample(h, new_n)

def fourier_perturb(h, n, mode, amplitude):
    x = np.arange(n)
    pert = amplitude * np.sin(2.0 * np.pi * mode * x / n)
    h_p = np.clip(h + pert, 0.0, 1.0)
    return fix_sum(h_p, n)

def active_set_opt(h, n, T, top_k, max_iter=1500):
    corr = np.correlate(h, 1.0 - h, mode='full')
    active = np.argsort(corr)[-top_k:]
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        av = c[active]
        cm = np.max(av)
        if T < 1e-12:
            return float(cm) * (2.0 / n)
        sh = np.clip((av - cm) / T, -80.0, 0.0)
        return float((cm + T * np.log(np.sum(np.exp(sh)) + 1e-300))) * (2.0 / n)
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        av = c[active]
        cm = np.max(av)
        if T < 1e-12:
            mi = np.argmax(av)
            ws = np.zeros(top_k)
            ws[mi] = 1.0
        else:
            sh = np.clip((av - cm) / T, -80.0, 0.0)
            es = np.exp(sh)
            ws = es / np.sum(es)
        dx = 2.0 / n
        omh = 1.0 - x
        grad = np.zeros(n)
        for idx, mi in enumerate(active):
            w = ws[idx] * dx
            if w < 1e-18:
                continue
            lag = mi - n + 1
            if lag >= 0:
                L = n - lag
                grad[lag:] += w * omh[:L]
                grad[:L] -= w * x[lag:]
            else:
                L = n + lag
                grad[:L] += w * omh[-lag:]
                grad[-lag:] -= w * x[:L]
        return grad
    try:
        res = opt.minimize(obj, h.copy(), jac=jac, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x: sum_con(x, n),
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0})
        if hasattr(res, 'x') and res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h
