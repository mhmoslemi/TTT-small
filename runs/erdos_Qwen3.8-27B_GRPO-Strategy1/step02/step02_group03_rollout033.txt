import numpy as np
from scipy.optimize import minimize
import time

_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    return float(np.sum(x[:_sp['n']]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, float).copy()
    for _ in range(200):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0 - 1e-12) if d > 0 else (h > 1e-12)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def softgrad(h, dx, beta):
    n = len(h)
    cr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (cr - cr.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    o = n - 1
    g = np.zeros(n)
    for j in range(n):
        lo, hi = -j, n - 1 - j
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            g[j] += np.dot(w[o + ms], 1.0 - h[j + ms])
        lo2, hi2 = j - n + 1, min(n - 1, j)
        if lo2 <= hi2:
            ms = np.arange(lo2, hi2 + 1)
            g[j] -= np.dot(w[o + ms], h[j - ms])
    return g * dx

def adam_descend(h0, dx, ts, iters, dl):
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0, dx)
    m1, m2 = np.zeros(n), np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    for i in range(iters):
        if time.time() > dl:
            break
        beta = 8.0 + 60.0 * min(1.0, i / (0.5 * iters))
        g = softgrad(h, dx, beta)
        lr = 0.015 / (1.0 + 0.008 * i)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        m1h = m1 / (1 - b1 ** (i + 1))
        m2h = m2 / (1 - b2 ** (i + 1))
        step = lr * m1h / (np.sqrt(m2h) + eps)
        ok = False
        for sh in (1.0, 0.5, 0.25, 0.1):
            h2 = proj(h - sh * step, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc - 1e-15:
                h, bc, ok = h2, c2, True
                break
        if not ok and i % 50 == 0:
            step_small = lr * 0.1 * m1h / (np.sqrt(m2h) + eps)
            h2 = proj(h - step_small, ts)
            c2 = compute_c5(h2, dx)
            if c2 < bc:
                h, bc = h2, c2
    return h, bc

def slsqp_try(h0, dx, ts, t_off, maxit):
    n = len(h0)
    _sp['n'], _sp['ts'] = n, ts
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + t_off]])
    bnds = [(0, 1)] * n + [(0, float(n))]
    cons = [{'type': 'ineq', 'fun': _sineq}, {'type': 'eq', 'fun': _siq}]
    best = h0.copy()
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-15})
        if r.x is not None:
            hr = proj(r.x[:n], ts)
            if compute_c5(hr, dx) < compute_c5(h0, dx) + 0.01:
                best = hr
    except Exception:
        pass
    return best

def coord_desc(h0, dx, ts, rounds, dl, rng):
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0, dx)
    for r in range(rounds):
        if time.time() > dl:
            break
        imp = False
        for j in rng.permutation(n):
            sv = h[j]
            for d in (0.02, -0.02, 0.05, -0.05, 0.008, -0.008, 0.003, -0.003):
                h[j] = sv + d
                if h[j] < -1e-10 or h[j] > 1 + 1e-10:
                    continue
                h = proj(h, ts)
                if compute_c5(h, dx) < bc:
                    bc = compute_c5(h, dx)
                    imp = True
                    break
            else:
                h[j] = sv
        if not imp:
            break
    return h, bc

def make_starts(n, ts, rng, ref):
    out = []
    if ref is not None:
        if len(ref) == n:
            out.append(ref.copy())
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, len(ref), endpoint=False), ref)
            out.append(proj(hi, ts))
    out.append(proj(np.full(n, ts / n), ts))
    x = np.arange(n) / n
    for p in (max(2, n // 4), max(2, n // 3)):
        out.append(proj(np.array([1.0 if (i % (2 * p)) < p else 0.0 for i in range(n)]), ts))
    for m, a in ((1, 0.35), (2, 0.3), (3, 0.2)):
        out.append(proj(0.5 + a * np.sin(2 * np.pi * m * x + 0.5), ts))
    h = np.zeros(n)
    p = rng.permutation(n)
    h[p[:int(ts)]] = 1.0
    fr = ts - int(ts)
    if fr > 0.01:
        h[p[int(ts)]] = fr
    out.append(h.copy())
    out.append(rng.uniform(0, 1, n))
    return out

def opt_full(h0, dx, ts, dl, rng):
    h = slsqp_try(h0, dx, ts, 0.01, 350)
    c = compute_c5(h, dx)
    if time.time() < dl - 25:
        h2, c2 = adam_descend(h, dx, ts, 600, dl - 12)
        if c2 < c:
            h, c = h2, c2
    if time.time() < dl - 15:
        h3 = slsqp_try(h, dx, ts, 0.005, 250)
        c3 = compute_c5(h3, dx)
        if c3 < c:
            h, c = h3, c3
    if time.time() < dl - 10:
        h4, c4 = coord_desc(h, dx, ts, 15, dl - 6, rng)
        if c4 < c:
            h, c = h4, c4
    if time.time() < dl - 4:
        h5 = slsqp_try(h, dx, ts, 0.003, 200)
        c5 = compute_c5(h5, dx)
        if c5 < c:
            h, c = h5, c5
    return h, c
