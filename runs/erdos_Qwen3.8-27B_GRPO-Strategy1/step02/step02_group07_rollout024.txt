```python
def optimal_pair_move(h, j, l, top_idx, top_corr):
    """Find optimal d for pair move (h[j]+d, h[l]-d) using active lags."""
    n = len(h)
    dx = 2.0 / n
    # For each active lag k, corr_k is linear in d:
    # corr_k(d) = corr_k(0) + d * (grad_k[j] - grad_k[l])
    # where grad_k[j] = ∂corr_k/∂h[j], grad_k[l] = ∂corr_k/∂h[l]
    
    slopes = np.zeros(len(top_idx))
    for ii, k in enumerate(top_idx):
        m = k - (n-1)
        g_j = 0.0
        g_l = 0.0
        # grad w.r.t. h[j]: (1-h[j+m]) if 0≤j+m<n, minus h[j-m] if 0≤j-m<n
        j_m = j + m
        if 0 <= j_m < n:
            g_j += (1.0 - h[j_m])
        j_mm = j - m
        if 0 <= j_mm < n:
            g_j -= h[j_mm]
        # grad w.r.t. h[l]
        l_m = l + m
        if 0 <= l_m < n:
            g_l += (1.0 - h[l_m])
        l_mm = l - m
        if 0 <= l_mm < n:
            g_l -= h[l_mm]
        slopes[ii] = (g_j - g_l) * dx
    
    # We want to minimize max_i (top_corr[i] + d * slopes[i])
    # subject to: 0 ≤ h[j]+d ≤ 1, 0 ≤ h[l]-d ≤ 1
    d_lo = max(-h[j], h[l] - 1.0)
    d_hi = min(1.0 - h[j], h[l])
    
    if d_lo >= d_hi:
        return 0.0
    
    # The function f(d) = max_i (a_i + b_i * d) is convex piecewise linear
    # Minimum is at a breakpoint or boundary
    # Collect all breakpoints: d where a_i + b_i*d = a_j + b_j*d
    breakpoints = [d_lo, d_hi]
    for i in range(len(slopes)):
        for j2 in range(i+1, len(slopes)):
            if abs(slopes[i] - slopes[j2]) > 1e-15:
                d_bp = (top_corr[j2] - top_corr[i]) / (slopes[i] - slopes[j2])
                if d_lo < d_bp < d_hi:
                    breakpoints.append(d_bp)
    
    # Also add boundaries
    breakpoints = sorted(set([max(d_lo, min(d_hi, bp)) for bp in breakpoints]))
    
    # Evaluate f at each breakpoint
    best_d = 0.0
    best_val = compute_c5(h)  # current value (upper bound on max over active)
    
    for d in breakpoints:
        vals = top_corr + d * slopes
        f = np.max(vals)
        if f < best_val - 1e-14:
            best_val = f
            best_d = d
    
    # Also check midpoints between breakpoints for robustness
    if len(breakpoints) > 1:
        for i in range(len(breakpoints)-1):
            d_mid = (breakpoints[i] + breakpoints[i+1]) / 2
            vals = top_corr + d_mid * slopes
            f = np.max(vals)
            if f < best_val - 1e-14:
                best_val = f
                best_d = d_mid
    
    return best_d
```

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]

    c_max = corr[top_idx[0]]

            if 0 <= j - m < n:

    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    for _ in range(n_moves):
        if time.time() - t_start > time_limit:
            break
        j, l = rng.choice(n, 2, replace=False)
        if h[j] <= 1e-7 or h[j] >= 1.0 - 1e-7:
            continue
        if h[l] <= 1e-7 or h[l] >= 1.0 - 1e-7:
            continue
        slopes = pair_grads(h, j, l)
        d_lo = max(-h[j], h[l] - 1.0)
        d_hi = min(1.0 - h[j], h[l])
        if d_lo >= d_hi:
            continue
        for _ in range(45):
            m1 = d_lo + (d_hi - d_lo) * 0.3333
            m2 = d_hi - (d_hi - d_lo) * 0.3333
            if np.max(corr + m1 * slopes) < np.max(corr + m2 * slopes):
                d_hi = m2
            else:
                d_lo = m1
        d_opt = (d_lo + d_hi) / 2.0
        if abs(d_opt) < 1e-10:
            continue
        new_c5 = np.max(corr + d_opt * slopes) * dx
        if new_c5 < c5 - 1e-13:
            h[j] = min(max(h[j] + d_opt, 0.0), 1.0)
            h[l] = min(max(h[l] - d_opt, 0.0), 1.0)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(corr) * dx
    return h, c5

def systematic_pairs(h, n, time_limit):
    t_start = time.time()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    for gap in range(1, n):
        if time.time() - t_start > time_limit:
            break
        for j in range(n - gap):
            l = j + gap
            if h[j] <= 1e-7 or h[j] >= 1.0 - 1e-7:
                continue
            if h[l] <= 1e-7 or h[l] >= 1.0 - 1e-7:
                continue
            slopes = pair_grads(h, j, l)
            d_lo = max(-h[j], h[l] - 1.0)
            d_hi = min(1.0 - h[j], h[l])
            if d_lo >= d_hi:
                continue
            for _ in range(35):
                m1 = d_lo + (d_hi - d_lo) * 0.3333
                m2 = d_hi - (d_hi - d_lo) * 0.3333
                if np.max(corr + m1 * slopes) < np.max(corr + m2 * slopes):
                    d_hi = m2
                else:
                    d_lo = m1
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-10:
                new_c5 = np.max(corr + d_opt * slopes) * dx
                if new_c5 < c5 - 1e-13:
                    h[j] = min(max(h[j] + d_opt, 0.0), 1.0)
                    h[l] = min(max(h[l] - d_opt, 0.0), 1.0)
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    c5 = np.max(corr) * dx
    return h, c5

def direction_search(h, n, K):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argpartition(corr, -K)[-K:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    c_max = corr[top_idx[0]]
    T_w = 0.0005
    weights = np.exp((corr[top_idx] - c_max) / T_w)
    weights /= np.sum(weights)
    grad_avg = np.zeros(n)
    for ii, k in enumerate(top_idx):
        m = k - (n - 1)
        for j in range(n):
            val = 0.0
            if 0 <= j + m < n:
                val += (1.0 - h[j + m])
            if 0 <= j - m < n:
                val -= h[j - m]
            grad_avg[j] = weights[ii] * val
    grad_avg -= np.mean(grad_avg)
    c0 = np.max(corr) * dx
    best_h = h.copy()
    best_c = c0
    for a in [0.02, 0.01, 0.005, 0.002, 0.001, 0.03, 0.015]:
        hn = fix_sum(h - a * grad_avg, n)
        cn = compute_c5(hn)
        if cn < best_c:
            best_c = cn
            best_h = hn
    return best_h, best_c

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 13)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = fix_sum(h, n)
    c5 = compute_c5(h)

    # Phase 1: SLSQP with geometric cooling (top-K LSE)
    sched = [(0.05, 400), (0.02, 500), (0.008, 600), (0.003, 800),
             (0.001, 1000), (0.0004, 1200), (0.0001, 1500)]
    for T, it in sched:
        if time.time() - t0 > budget_s - 500:
            break
        h = slsqp_opt(h, n, T, it, K=50)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 2: Pair optimization (exact ternary search)
    rem = budget_s - (time.time() - t0)
    if rem > 350:
        h, c5 = pair_optimize(h, n, rng, 8000, min(180, rem * 0.4))
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 3: Direction search + SLSQP
    if time.time() - t0 < budget_s - 300:
        for _ in range(5):
            if time.time() - t0 > budget_s - 300:
                break
            h, c5 = direction_search(h, n, K=8)
            h = slsqp_opt(h, n, 0.001, 400, K=30)
            h = fix_sum(h, n)
            c5 = compute_c5(h)

    # Phase 4: More pair optimization
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        h, c5 = pair_optimize(h, n, rng, 12000, min(120, rem * 0.4))
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 5: Systematic pairs (narrow gaps)
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        h, c5 = systematic_pairs(h, n, min(80, rem * 0.4))
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 6: Final SLSQP + pair polish
    if time.time() - t0 < budget_s - 60:
        h = slsqp_opt(h, n, 0.00005, 2000, K=20)
        h = fix_sum(h, n)
        c5 = compute_c5(h)
    if time.time() - t0 < budget_s - 40:
        h, c5 = pair_optimize(h, n, rng, 5000, 40)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 7: Try higher resolution
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        for n2 in [400, 560]:
            if time.time() - t0 > budget_s - 60:
                break
            h2 = upsample(h, n2)
            h2 = slsqp_opt(h2, n2, 0.01, 400, K=50)
            h2 = slsqp_opt(h2, n2, 0.003, 600, K=30)
            h2 = slsqp_opt(h2, n2, 0.0005, 800, K=20)
            h2 = fix_sum(h2, n2)
            _, c5_2 = pair_optimize(h2, n2, rng, 4000, min(60, max(10, budget_s - (time.time() - t0) - 40)))
            h2 = fix_sum(h2, n2)
            c5_2 = compute_c5(h2)
            if c5_2 < c5:
                h, n, c5 = h2, n2, c5_2

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
