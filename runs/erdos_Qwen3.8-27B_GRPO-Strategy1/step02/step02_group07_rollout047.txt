import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_vals(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def obj_smooth(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    dx = 2.0 / len(h)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def obj_penalty(h, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    excess = np.maximum(0.0, corr - target)
    return float(np.sum(excess * excess))

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = int(np.sum(m))
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-10:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = min(max(h[idx] + d2, 0.0), 1.0)
    return h

def slsqp_opt(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(
            obj_smooth, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-18})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def penalty_opt(h, n, target, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(
            obj_penalty, h, args=(target,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-18})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def targeted_coord(h, n, n_pass=6):
    for p in range(n_pass):
        c_max = compute_c5(h)
        deltas = np.array([-0.06, -0.025, -0.008, 0.008, 0.025, 0.06])
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            best_c = c_max
            best_h = h.copy()
            for delta in deltas:
                hn = h.copy()
                hn[j] = h[j] + delta
                if hn[j] < 0 or hn[j] > 1:
                    continue
                adj = -delta / (n - 1)
                others = slice(0, n)
                hn[others] += adj
                hn[j] = h[j] + delta
                hn = np.clip(hn, 0.0, 1.0)
                c = compute_c5(hn)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = hn
            if best_c < c_max:
                h = best_h
                c_max = best_c
        h = fix_sum(h, n)
    return h

def pair_polish(h, n, rng_obj, n_pairs=400):
    for _ in range(n_pairs):
        j, l = rng_obj.choice(n, 2, replace=False)
        if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4 or h[l] <= 1e-4 or h[l] >= 1.0 - 1e-4:
            continue
        best_c5 = compute_c5(h)
        best_d = 0.0
        for d in (-0.03, -0.01, -0.003, 0.003, 0.01, 0.03):
            hj2, hl2 = h[j] + d, h[l] - d
            if hj2 < 1e-4 or hj2 > 1.0 - 1e-4 or hl2 < 1e-4 or hl2 > 1.0 - 1e-4:
                continue
            hn = h.copy()
            hn[j], hn[l] = hj2, hl2
            c = compute_c5(hn)
            if c < best_c5 - 1e-13:
                best_c5 = c
                best_d = d
        if best_d != 0.0:
            h[j] = min(max(h[j] + best_d, 0.0), 1.0)
            h[l] = min(max(h[l] - best_d, 0.0), 1.0)
    return fix_sum(h, n)
