    dx = 2.0 / n

def sla_c5(h, rng, n_iter=5000, T0=0.01, T1=0.0001):
    """Simulated annealing on binary h."""
    n = len(h)
    h = to_binary(h, n).astype(float)
    c5_cur = bin_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    best = h.copy()
    best_c5 = c5_cur
    
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / n_iter)
        # Swap a random 1 with a random 0
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = bin_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c5_cur = c5_new
            # Update sets
            if h[i] == 1.0:
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_c5

    zeros = np.where(h == 0.0)[0]
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(1, n_iter - 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = bin_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_c5

def lbb_opt(h0, n, t, topk, lam, maxiter=2000):
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_pen, h0, jac=lambda x: grad_pen(x, t, topk, lam),
                   method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'eps': 1e-8})
    return proj(np.clip(res.x, 0.0, 1.0), n)

def slsqp_opt(h0, n, t, topk, maxiter=2000, ftol=1e-15):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    res = minimize(lambda x: smooth_obj(x, t, topk), h0,
                   jac=lambda x: smooth_grad(x, t, topk),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    try:
        bh = proj(np.array(initial_h_values, dtype=float), 280)
    except Exception:
        bh = proj(np.full(280, 0.5) + rng.randn(280) * 0.05, 280)
    bc = compute_c5(bh)

    # Phase 1: Binary SA exploration at n=70
    if tleft() > 500:
        h70 = interp_down(bh, 70)
        for sa_i in range(5):
            if tleft() < 60:
                break
            h70_bin = to_binary2(h70, 70)
            try:
                hb, cb = sla_c5(h70_bin, rng, n_iter=10000, T0=0.02, T1=0.00003)
                h_up = interp_up(hb, 140)
                c5_up = compute_c5(h_up)
                if c5_up < bc:
                    bc = c5_up
                    bh = interp_up(h_up, 280)
                h70 = interp_down(bh, 70)
            except Exception:
                pass

    # Phase 2: L-BFGS-B refinement
    h280 = bh.copy()
    for t_val in [100, 300, 1000, 3000, 10000]:
        if tleft() < 30:
            break
        try:
            hc = lbb_opt(h280, 280, t_val, 80, lam=500, maxiter=1500)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                h280 = hc
        except Exception:
            pass

    # Phase 3: SLSQP with exact constraint
    for t_val in [500, 2000, 8000, 30000]:
        if tleft() < 25:
            break
        try:
            hc = slsqp_opt(h280, 280, t_val, 60, maxiter=2000)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                h280 = hc
        except Exception:
            pass

    # Phase 4: Multi-resolution cycles
    for cycle in range(3):
        if tleft() < 100:
            break
        h140 = interp_down(h280, 140)
        h70 = interp_down(h140, 70)
        try:
            hb = to_binary2(h70, 70)
            hb, cb = sla_c5(hb, rng, n_iter=6000, T0=0.01, T1=0.0001)
        except Exception:
            hb = to_binary2(h70, 70)
        h140 = interp_up(hb, 140)
        for t_val in [200, 800, 3000]:
            if tleft() < 15:
                break
            try:
                h140 = lbb_opt(h140, 140, t_val, 30, lam=300, maxiter=1000)
            except Exception:
                pass
        h280_new = interp_up(h140, 280)
        for t_val in [500, 2000, 8000]:
            if tleft() < 15:
                break
            try:
                hc = lbb_opt(h280_new, 280, t_val, 70, lam=1000, maxiter=1500)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    h280 = hc
                h280_new = hc
            except Exception:
                pass
        h280 = h280_new

    # Phase 5: Focused topk optimization
    if tleft() > 40:
        for tk in [10, 20, 35, 55]:
            if tleft() < 12:
                break
            try:
                hc = slsqp_opt(h280, 280, 5000, tk, maxiter=2500, ftol=1e-16)
                c5 = compute_c5(hc)
                if c5 < bc:
                    bc = c5
                    h280 = hc
            except Exception:
                pass

    # Phase 6: Final ultra-fine SLSQP polish
    for t_val in [50000, 200000, 1000000]:
        if tleft() < 10:
            break
        try:
            hc = slsqp_opt(h280, 280, t_val, 10, maxiter=5000, ftol=1e-17)
            c5 = compute_c5(hc)
            if c5 < bc:
                bc = c5
                h280 = hc
        except Exception:
            pass

    return h280, compute_c5(h280), 280
