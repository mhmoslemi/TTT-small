def project_onto_constraint(h, target_sum):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target_sum) < 1e-15:
        return h
    diff = target_sum - s
    # Distribute to interior points
    interior = (h > 1e-12) & (h < 1 - 1e-12)
    if interior.sum() > 0 and abs(diff) < interior.sum() * 0.49:
        h[interior] += diff / interior.sum()
        return np.clip(h, 0, 1)
    # More careful distribution
    if diff > 0:
        idx = np.argsort(h)[::-1]  # prefer points with room
        for i in idx:
            space = 1 - h[i]
            add = min(diff, space)
            h[i] += add
            diff -= add
            if diff < 1e-15: break
    else:
        diff = -diff
        idx = np.argsort(h)  # prefer points with mass
        for i in idx:
            rem = min(diff, h[i])
            h[i] -= rem
            diff -= rem
            if diff < 1e-15: break
    return np.clip(h, 0, 1)

def compute_c5(h):

    return float(np.max(corr) * dx)

def soft_max_obj_penalty(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        obj = c_max * dx
    else:
        shifted = np.clip(corr - c_max, -80.0 * T, 0.0)
        s = np.sum(np.exp(shifted / T))
        obj = (c_max + T * np.log(max(s, 1e-300))) * dx
    sum_viol = np.sum(h) - n * 0.5
    return obj + pen * sum_viol * sum_viol

def fix_sum(h, n):

    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

                    else:

def fix_sum(h, n):

    wsum = np.sum(w)
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    if top_k < 2 * n:
        thresh = np.sort(w)[-top_k] if top_k > 0 else 0
        w[w < thresh] = 0.0
        ws2 = np.sum(w)
        if ws2 > 1e-300:
            w /= ws2
