            total = room.sum()
            if total < 1e-14:

    n = len(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_max_c5, h, args=(T,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    n = len(x) - 1
    return x[:n].sum() - n / 2.0

    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr

def erdos_obj(x):
    return x[-1]

def slsqp_erdos(h, maxiter=400):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

    cons = [
        {'type': 'eq', 'fun': erdos_eq},
        {'type': 'ineq', 'fun': erdos_ineq}
    ]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        h_opt = np.clip(res.x[:n], 0, 1)
        return project_feasible(h_opt, n)
    except Exception:
        return h

    for _ in range(max_sweeps):
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        improved = False
        for i in ones:
            for j in zeros:
                cur[i], cur[j] = cur[j], cur[i]
                c5 = compute_c5(cur)

                    cur_c5 = c5
                    improved = True
                else:
                    cur[i], cur[j] = cur[j], cur[i]
        if not improved:
            break
    return cur, cur_c5
