import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, corr, topk, sharp):
    center = N - 1
    kk = min(topk, len(corr))
    idx = np.argpartition(corr, -kk)[-kk:]
    vals = corr[idx]
    w = np.exp(sharp * (vals - vals.max()))
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N)
    w /= ws
    grad = np.zeros(N)
    for i in range(kk):
        wi = w[i]
        if wi < 5e-5:
            continue
        L = idx[i] - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def nesterov_opt(h, N, target, dx, deadline, lr0, mom, sh0, sh1, tk0, tk1, rng):
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / 1200.0, 1.0)
        sh = sh0 * (sh1 / sh0) ** frac
        tk = max(4, int(tk0 + (tk1 - tk0) * frac))
        g = grad_topk(h, N, corr, tk, sh)
        c5 = float(corr.max()) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr = lr0 * (1.0 - 0.7 * frac)
        v = mom * v - lr * g
        h = h + v
        h = project_sum(h, target)
        if it > 0 and it % 400 == 0 and frac < 0.6:
            h += rng.randn(N) * 0.01 * (1.0 - frac)
            h = project_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        it += 1
    return best_h, best_c5

    best_h = h.copy()
    best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

            break
        for i in range(N):

                break
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            a = c1 - c0
            b = c0
            lo, hi2 = 0.0, 1.0

                else:

        c = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    return best_h, best_c5
