def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0

    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)  # for numerical stability
    return mx + np.log(np.sum(np.exp(s))) / t

    """Adam optimizer step"""
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = beta1 * m + (1 - beta1) * grads

    m_hat = m / (1 - beta1**(t_step + 1))
    v_hat = v / (1 - beta2**(t_step + 1))

        if time.time() - t_start > time_limit:
            break
        gu = compute_going_up(h)
        ones_arr = np.array(ones)
        best_cand = best_val
        best_i = -1
        best_j = -1
        for ii in range(len(zeros)):
            i = zeros[ii]
            base = old_corr + gu[i]
            cands = np.max(base[None, :] - gu[ones_arr][:, None], axis=1)
            j_idx = np.argmin(cands)
            if cands[j_idx] < best_cand:
                best_cand = cands[j_idx]
                best_i = i
                best_j = ones[j_idx]
        if best_i < 0 or best_cand >= best_val:
            no_improve += 1
            if no_improve > 3:
                break
            continue
        no_improve = 0
        h[best_i] = 1.0
        h[best_j] = 0.0
        zeros.remove(best_i)
        ones.remove(best_j)
        zeros.append(best_j)
        ones.append(best_i)
        old_corr = corr_all(h)
        best_val = np.max(old_corr)
        iterations += 1
    return h, best_val * dx

def lse_smoothed_obj(h, t):
    corr = corr_all(h)
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    return (mx + np.log(np.sum(np.exp(s))) / t)

def lse_grad(h, t):
    n = len(h)
    corr = corr_all(h)
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    w = np.exp(s) / np.sum(np.exp(s))
    i = np.arange(n)[:, None]
    d = np.arange(-(n-1), n)[None, :]
    im = i + d
    idm = i - d
    mask_im = (im >= 0) & (im < n)
    mask_idm = (idm >= 0) & (idm < n)
    im_safe = np.where(mask_im, im, 0)
    idm_safe = np.where(mask_idm, idm, 0)
    term1 = np.where(mask_im, 1.0 - h[im_safe], 0.0)
    term2 = np.where(mask_idm, h[idm_safe], 0.0)
    return (term1 - term2) @ w

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    n = 280
    try:
        bh = proj(np.array(initial_h_values, dtype=float), n)
    except Exception:
        bh = proj(np.full(n, 0.5) + rng.randn(n) * 0.1, n)
    bc = compute_c5(bh)
    h = bh.copy()

    # Phase 1: Quick continuous optimization with custom gradient (Adam-like)
    if tleft() > 300:
        m_acc = np.zeros(n)
        v_acc = np.zeros(n)
        lr = 0.02
        for it in range(2000):
            if tleft() < 30:
                break
            t_val = 50.0 + it * 2.0
            g = lse_grad(h, t_val)
            beta1, beta2, eps = 0.8, 0.999, 1e-8
            m_acc = beta1 * m_acc + (1 - beta1) * g
            v_acc = beta2 * v_acc + (1 - beta2) * g * g
            m_hat = m_acc / (1 - beta1 ** (it + 1))
            v_hat = v_acc / (1 - beta2 ** (it + 1))
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = proj(h_new, n)
            c = compute_c5(h)
            if c < bc:
                bc = c
                bh = h.copy()
        lr *= 0.5
        for it in range(1000):
            if tleft() < 20:
                break
            t_val = 500.0 + it * 5.0
            g = lse_grad(h, t_val)
            m_acc = 0.8 * m_acc + 0.2 * g
            v_acc = 0.999 * v_acc + 0.001 * g * g
            m_hat = m_acc / (1 - 0.8 ** 4001)
            v_hat = v_acc / (1 - 0.999 ** 4001)
            h = proj(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), n)
            c = compute_c5(h)
            if c < bc:
                bc = c
                bh = h.copy()

    # Phase 2: Binary local search (main work)
    if tleft() > 200:
        h_bin = (h > 0.5).astype(float)
        h_bin = proj(h_bin, n)
        h_l, c_l = binary_local_search(h_bin, n, 5000, max(0, tleft() - 150))
        if c_l < bc:
            bc = c_l
            bh = h_l.copy()
            h = h_l.copy()

    # Phase 3: Continuous refinement after binary search
    if tleft() > 100:
        h = bh.copy()
        for t_val in [200, 600, 2000, 6000]:
            if tleft() < 15:
                break
            m_acc = np.zeros(n)
            v_acc = np.zeros(n)
            lr2 = 0.005
            for it in range(500):
                if tleft() < 10:
                    break
                g = lse_grad(h, t_val + it * 0.5)
                m_acc = 0.85 * m_acc + 0.15 * g
                v_acc = 0.999 * v_acc + 0.001 * g * g
                m_hat = m_acc / (1 - 0.85 ** (it + 1))
                v_hat = v_acc / (1 - 0.999 ** (it + 1))
                h = proj(h - lr2 * m_hat / (np.sqrt(v_hat) + 1e-8), n)
                c = compute_c5(h)
                if c < bc:
                    bc = c
                    bh = h.copy()

    # Phase 4: Second binary local search pass
    if tleft() > 80:
        h_b2 = (bh > 0.45).astype(float)
        h_b2 = proj(h_b2, n)
        h_l2, c_l2 = binary_local_search(h_b2, n, 3000, max(0, tleft() - 60))
        if c_l2 < bc:
            bc = c_l2
            bh = h_l2.copy()

    # Phase 5: Final SLSQP polish
    if tleft() > 30:
        bnds = {n: [(0.0, 1.0)] * n}
        cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}

        def obj_fn(hx, t_val):
            c = corr_all(hx)
            mx = np.max(c)
            s = t_val * (c - mx)
            s -= np.max(s)
            return (mx + np.log(np.sum(np.exp(s))) / t_val) * (2.0 / n)

        for t_val in [3000, 10000]:
            if tleft() < 12:
                break
            try:
                r = minimize(obj_fn, bh, args=(t_val,), method='SLSQP',
                             bounds=bnds[n], constraints=cons,
                             options={'maxiter': 2000, 'ftol': 1e-16})
                hc = proj(np.clip(r.x, 0.0, 1.0), n)
                c5h = compute_c5(hc)
                if c5h < bc:
                    bc = c5h
                    bh = hc
            except Exception:
                pass

    # Final evaluation
    bh = proj(bh, n)
    final_c = compute_c5(bh)
    return bh, final_c, n
