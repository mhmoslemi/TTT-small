import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += room * (diff / s)
        else:
            room = h.copy()
            s = room.sum()
            if s < 1e-15:
                break
            h += room * (diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_h(h, n, top_k=20):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    idx = np.argsort(corr)[-top_k:]
    top_vals = corr[idx]
    w = top_vals - top_vals.min() + 1.0
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(idx)):
        if w[k] < 1e-12:
            continue
        j = idx[k]
        lag = j - (n - 1)
        lo = max(0, lag)
        hi = min(n - 1, (n - 1) + lag)
        if lo > hi:
            continue
        d = j - np.arange(lo, hi + 1)
        grad[lo:hi + 1] += w[k] * (1.0 - 2.0 * h[d])
    return grad

def subgrad_optimize(h, n, n_steps=500, lr_start=0.04, lr_end=0.002):
    h = h.copy()
    best_c = compute_c5(h)
    for step in range(n_steps):
        lr = lr_start * (lr_end / lr_start) ** (step / max(1, n_steps - 1))
        g = subgrad_h(h, n, top_k=25)
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new)
        if c_new <= best_c + 1e-14:
            h = h_new

            if c_new < best_c:
                h = h_new
                best_c = c_new
        if step % 100 == 99 and best_c < 0.3805:
            break
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=250):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def erdos_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def erdos_ineq(x):
    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr

def erdos_obj(x):
    return x[-1]

def slsqp_direct(h, maxiter=300):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.03

    cons = [
        {'type': 'eq', 'fun': erdos_eq},
