        h_opt = do_slsqp(h_opt, n_try, 0.003, 400)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c
            n_best = n_try

    # Phase 6: L-BFGS-B refinement at multiple T
    for T, mi in [(0.002, 800), (0.001, 1000), (0.0005, 1200)]:
        if time.time() - t_start > budget_s - 50:
            break
        h_opt = do_lbfgs(h_best.copy(), n_best, T, mi)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 7: Final SLSQP polish with very small T
    for T, mi in [(0.001, 1200), (0.0004, 1500), (0.0002, 2000)]:
        if time.time() - t_start > budget_s - 20:
            break
        h_opt = do_slsqp(h_best.copy(), n_best, T, mi)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c

    # Phase 8: One last targeted + polish cycle
    if time.time() - t_start < budget_s - 30:
        h_pert = targeted_perturb(h_best, n_best, rng, 0.01)
        h_opt = do_slsqp(h_pert, n_best, 0.0005, 1500)
        c = compute_c5(h_opt)
        if c < c5_best:
            h_best = h_opt
            c5_best = c
        if time.time() - t_start < budget_s - 15:
            h_opt = do_slsqp(h_best.copy(), n_best, 0.0001, 2000)
            c = compute_c5(h_opt)
            if c < c5_best:
                h_best = h_opt
                c5_best = c

    h_best = proj(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
