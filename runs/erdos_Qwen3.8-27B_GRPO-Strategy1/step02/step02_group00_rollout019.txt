    h = np.clip(h, 0.0, 1.0).copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h *= target / total
    return np.clip(h, 0.0, 1.0)

def slsqp_run(h, T, n, maxiter=600):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

def pair_anneal(h, n, c5, n_iter, t0, t1, rng):
    hc = h.copy()
    c5c = c5
    for it in range(n_iter):
        fr = (it + 1) / n_iter
        Tp = t0 * (t1 / t0) ** fr
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = rng.uniform(-0.25, 0.25)
        hi, hj = hc[i] + d, hc[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h_try = hc.copy()
        h_try[i] = hi
        h_try[j] = hj
        c5_try = compute_c5(h_try)
        if c5_try < c5c:
            hc, c5c = h_try, c5_try
        elif Tp > 1e-12 and rng.random() < np.exp(-(c5_try - c5c) / Tp):
            hc, c5c = h_try, c5_try
    return hc, c5c

def block_anneal(h, n, c5, n_iter, t0, t1, rng):
    hc = h.copy()
    c5c = c5
    for it in range(n_iter):
        fr = (it + 1) / n_iter
        Tp = t0 * (t1 / t0) ** fr
        idx = rng.choice(n, size=3, replace=False)
        i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
        d = rng.uniform(-0.12, 0.12)
        vals = (hc[i] + 2*d, hc[j] - d, hc[k] - d)
        if all(0 <= v <= 1 for v in vals):
            h_try = hc.copy()
            h_try[i], h_try[j], h_try[k] = vals
            c5_try = compute_c5(h_try)
            if c5_try < c5c:
                hc, c5c = h_try, c5_try
            elif Tp > 1e-12 and rng.random() < np.exp(-(c5_try - c5c) / Tp):
                hc, c5c = h_try, c5_try
    return hc, c5c

def syst_pair(h, n, c5, rounds=4):
    deltas = [0.004, 0.008, 0.015, 0.03, 0.06, 0.1]
    h_cur = h.copy()
    c5_cur = c5
    for rnd in range(rounds):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                oi, oj = h_cur[i], h_cur[j]
                for d in deltas:
                    for sgn in (1, -1):

                        hi, hj = oi + dd, oj - dd
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h_cur[i], h_cur[j] = hi, hj
                            c5_try = compute_c5(h_cur)

                                c5_cur = c5_try
                                improved = True
                            else:
                                h_cur[i], h_cur[j] = oi, oj
        if not improved:
            break
    return h_cur, c5_cur
