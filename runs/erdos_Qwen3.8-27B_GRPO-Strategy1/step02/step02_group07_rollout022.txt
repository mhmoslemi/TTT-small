    if T < 1e-15:

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-9) & (h < 1.0 - 1e-9)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-13:
        # redistribute
        for i in np.argsort(np.abs(d2 / h.astype(float) + 1)):
            if h[i] > 1e-6 and h[i] < 1.0 - 1e-6:
                h[i] = np.clip(h[i] + d2, 0, 1)
                break
    return h

def project_feasible(v, n):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=n/2}"""
    h = np.clip(v, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-15:
        return h
    # Scale interior points
    interior = (h > 0.01) & (h < 0.99)
    if np.sum(interior) > 0:
        h[interior] += d / np.sum(interior)
        h = np.clip(h, 0.0, 1.0)
    # Remaining: adjust one point
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + d2, 0, 1)
    return h

def subgrad_topk(h, n, topk=5):
    """Compute averaged sub-gradient from top-k correlation peaks"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k indices
    top_idx = np.argsort(corr)[-topk:]
    top_idx = np.sort(top_idx)
    g = np.zeros(n)
    for ki in top_idx:
        s = ki - (n - 1)  # shift value
        for j in range(n):
            term = 0.0
            jk = j + s
            if 0 <= jk < n:
                term += (1.0 - h[jk])
            jk2 = j - s
            if 0 <= jk2 < n:
                term -= h[jk2]
            g[j] += term
    g /= len(top_idx)
    return g

    # Current correlation values: corr[s] for s = -(n-1), ..., (n-1)
    # If h[j] changes by δ:
    # corr[s] changes by: δ*(1-h[j+s]) - δ*h[j-s] (with bounds)
    # = δ * slope_s where slope_s = (1-h[j+s])*[0≤j+s<n] - h[j-s]*[0≤j-s<n]

        slope = 0.0

        if 0 <= js < n:
            slope += (1.0 - h[js])
        js2 = j - s
        if 0 <= js2 < n:
            slope -= h[js2]
        slopes[s_offset] = slope

    # We want to minimize max_s (corr_current[s] + δ * slopes[s])
    # subject to -h[j] ≤ δ ≤ 1-h[j]

    if hi - lo < 1e-10:

    # Minimum is where subgradient contains 0
    # Subgradient at δ: {max of b_s for s achieving the max}
    # So we want: 0 ∈ [min{b_s : s in argmax}, max{b_s : s in argmax}]

        vals = corr_current + mid * slopes
        m = np.max(vals)

        sg_max = np.max(slopes[top_mask])
        if sg_min > 0:
            hi = mid
        elif sg_max < 0:
            lo = mid
        else:
            break  # 0 is in the subgradient
    
    best_delta = (lo + hi) / 2
    new_val = np.clip(h[j] + best_delta, 0.0, 1.0)
    return new_val

    return h

def block_descent(h, n, rng, block_size=4, max_iters=20000, time_cap=None, t0=None):
    h = h.copy()
    corr = get_corr(h)
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    for it in range(max_iters):
        if time_cap is not None and time.time() - t0 > time_cap:
            break
        j0 = rng.randint(0, n - block_size + 1)
        idxs = list(range(j0, j0 + block_size))
        if any(h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10 for i in idxs):
            continue
        comp_sl = np.zeros(2 * n - 1)
        for i in idxs:
            comp_sl += slopes_for_j(h, n, i)
        comp_sl /= block_size
        lo = -min(h[i] for i in idxs)
        hi = min(1.0 - h[i] for i in idxs)
        if hi - lo < 1e-12:
            continue
        best_d = bisect_convex(corr, comp_sl, lo, hi, 35)
        if abs(best_d) > 1e-11:
            for i in idxs:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
            corr = corr + best_d * comp_sl
            c5_v = np.max(corr) * dx
            if c5_v < best_c5 - 1e-13:
                best_c5 = c5_v
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 17)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    # Phase 1: SLSQP to escape local min
    for T, it in [(0.02, 150), (0.005, 200)]:
        if time.time() - t0 > budget_s - 750:
            break
        h = slsqp_run(h, n, T, it)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Fast coordinate descent
    cap = budget_s - 550
    h, c5 = coord_descent(h, n, 8000, cap, t0, seed)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 3: Pair descent
    cap = budget_s - 380
    h, c5 = pair_descent(h, n, rng, 80000, cap, t0)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 4: Coordinate descent again
    cap = budget_s - 250
    h, c5 = coord_descent(h, n, 6000, cap, t0, seed + 1)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 5: Block descent
    cap = budget_s - 180
    h, c5 = block_descent(h, n, rng, 3, 30000, cap, t0)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 6: SLSQP polish
    for T, it in [(0.001, 400), (0.0003, 600), (0.0001, 800)]:
        if time.time() - t0 > budget_s - 80:
            break
        h = slsqp_run(h, n, T, it)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 7: Final coordinate + pair polish
    cap = budget_s - 40
    h, c5 = coord_descent(h, n, 4000, cap, t0, seed + 2)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    if time.time() - t0 < budget_s - 20:
        h, c5 = pair_descent(h, n, rng, 50000, budget_s - 20, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 8: Higher resolution
    if time.time() - t0 < budget_s - 100:
        n2 = 400
        x0 = np.arange(n, dtype=float) * (2.0 / n)
        x1 = np.arange(n2, dtype=float) * (2.0 / n2)
        h2 = fix_sum(np.interp(x1, x0, best_h, period=2.0), n2)
        cap2 = budget_s - 30
        h2, c5_2 = coord_descent(h2, n2, 4000, cap2, t0, seed + 3)
        if c5_2 < best_c5:
            best_h, best_c5, n = h2, c5_2, n2

    best_h = fix_sum(best_h, len(best_h))
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, len(best_h))
