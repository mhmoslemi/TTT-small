            best_h = h_b.copy()

    # Phase C: fine polish
    if time.time() < deadline - 10:
        t_c = time.time() + (deadline - time.time()) * 0.3
        h_c, c_c = optimize_phase(best_h, n0, target0, dx0, t_c,
                                  3, 2, 0.002, 0.9995, 1e-7, rng)
        if c_c < best_c5:
            best_c5 = c_c
            best_h = h_c.copy()

    # Try N=572 for more resolution
    N2 = 572
    if time.time() < deadline - 30:
        target2 = N2 / 2
        dx2 = 2.0 / N2
        x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h2 = np.interp(x2, x0, best_h)
        h2 = project_sum(h2, target2)
        t_d = deadline - 3
        h_d, c_d = optimize_phase(h2, N2, target2, dx2, t_d,
                                  20, 3, 0.012, 0.997, 1e-5, rng, stochastic=True)
        if c_d < best_c5:
            best_c5 = c_d
            best_h = h_d.copy()

    # Random restart at parent size
    if time.time() < deadline - 25:
        h_r = best_h + rng.randn(n0) * 0.025
        h_r = project_sum(h_r, target0)
        h_r, c_r = optimize_phase(h_r, n0, target0, dx0, max(deadline - 2, time.time() + 20),
                                  15, 2, 0.008, 0.998, 1e-6, rng, stochastic=True)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()

    # Final micro-polish
    if time.time() < deadline - 3:
        Nb = len(best_h)
        dx_b = 2.0 / Nb
        h_f, c_f = optimize_phase(best_h.copy(), Nb, Nb / 2.0, dx_b, deadline - 1,
                                  2, 2, 0.001, 0.9998, 1e-8, rng)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
