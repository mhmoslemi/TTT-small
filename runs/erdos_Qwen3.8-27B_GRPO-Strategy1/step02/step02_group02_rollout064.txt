For a step function, this means: for each shift k, we're measuring how much of the "1" region of h overlaps with the "0" region of h shifted by k.

At the optimum, we want this to be as balanced as possible across all shifts. The constraint ∫h = 1 means h is 1 on a set of measure 1 (half the domain).

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)

        c = corr

    w = exp_c / exp_c.sum()

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
