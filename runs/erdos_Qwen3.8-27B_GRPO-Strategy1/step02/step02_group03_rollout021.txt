    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h < 1 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def corr_grad_cr(h, n):
    """Compute all correlations and their gradients stored compactly."""
    cr = np.correlate(h, 1.0 - h, mode='full')
    return cr

def lse_func(x, n, beta, ts, pen):
    h = x[:n]
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = cr.max()
    lse = (mx + np.log(np.exp(beta * (cr - mx)).sum())) / beta
    s = h.sum() - ts
    return lse + pen * s * s
