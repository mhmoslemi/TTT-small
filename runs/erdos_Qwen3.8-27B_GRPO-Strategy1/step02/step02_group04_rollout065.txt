def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    ni = int(np.sum(interior))
    if ni > 0 and abs(diff) < ni * 0.3:
        h[interior] += diff / ni
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff < 1e-12: break
    else:
        diff = -diff
        idx = np.argsort(-h)
        for i in idx:
            rem = min(diff, h[i])
            h[i] -= rem
            diff -= rem
            if diff < 1e-12: break
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, T, lambda_pen=100.0):
    """Returns (objective, gradient) for soft-max + penalty"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    dx = 2.0 / n
    
    if T > 1e-15:
        shifted = np.clip(corr - c_max, -100.0 * T, 0.0)
        exps = np.exp(shifted / T)
        s = np.sum(exps)
        if s < 1e-300:
            obj = c_max * dx
            weights = np.zeros_like(corr)
            weights[np.argmax(corr)] = 1.0
        else:
            obj = (c_max + T * np.log(s)) * dx
            weights = exps / s
    else:
        obj = c_max * dx
        n_active = np.sum(corr >= c_max - 1e-10)
        weights = np.zeros_like(corr)
        weights[corr >= c_max - 1e-10] = 1.0 / n_active
    
    # Penalty
    s_pen = np.sum(h) - n * 0.5
    obj += lambda_pen * s_pen * s_pen
    
    # Gradient of weighted correlation
    # W_m = weights[n-1+m] + weights[n-1-m] for m=1..n-1, W_0 = weights[n-1]
    W = np.zeros(n)
    W[0] = weights[n-1]
    for m in range(1, n):
        W[m] = weights[n-1+m] + weights[n-1-m]
    
    # g[j] = sum_{m=0}^{j} W[m]*(1-h[j-m]) - sum_{m=0}^{n-1-j} W[m]*h[j+m]
    g = np.zeros(n)
    omh = 1.0 - h
    for j in range(n):
        g[j] = np.dot(W[:j+1][::-1], omh[:j+1]) - np.dot(W[:n-j], h[j:])
    g *= dx
    
    # Add penalty gradient
    g += 2.0 * lambda_pen * s_pen
    
    return obj, g

def do_smooth_lbfgsb(h, n, T, max_iter=3000):
    h0 = h.copy()
    def obj_only(x):
        return smooth_obj_grad(x, n, T)[0]
    def obj_grad(x):
        return smooth_obj_grad(x, n, T)
    res = opt.minimize(obj_grad, h0, method='L-BFGS-B',
                       jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': max_iter * 3})
    if res.x is not None:
        return fix_sum(res.x, n)
    return h

def binary_project(h, threshold=0.5):
    """Push h towards binary, then fix sum"""
    n = len(h)
    h_bin = (h > threshold).astype(float)
    # Adjust to get correct sum
    target = n * 0.5
    # Need exactly target ones
    n_ones = int(target)
    if np.sum(h_bin) < n_ones:
        # Add more ones: pick smallest h values that are 0
        idx0 = np.where(h_bin == 0)[0]
        order = idx0[np.argsort(-h[idx0])]  # highest h first
        for i in order:
            if h_bin.sum() >= n_ones: break
            h_bin[i] = 1.0
    elif np.sum(h_bin) > n_ones:
        idx1 = np.where(h_bin == 1)[0]
        order = idx1[np.argsort(h[idx1])]  # lowest h first
        for i in order:
            if h_bin.sum() <= n_ones: break
            h_bin[i] = 0.0
    return fix_sum(h_bin, n)

def do_slsqp(h, n, T, max_iter=2000):

                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n*0.5}],
                          options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None:
            return fix_sum(res.x, n)
    except:
        pass
    return h

        if budget_s - elapsed() < 60:
            break
        try_upd(do_lbfgsb(h_best, n_best, T, 2500), n_best)

    # Phase 2: Binary project + re-optimize
    if budget_s - elapsed() > 100:
        hb = binary_project(h_best)
        for T in [0.005, 0.001, 0.0003]:
            h2 = do_lbfgsb(hb, n_best, T, 2000)
            if try_upd(h2, n_best):
                hb = h_best.copy()

    # Phase 3: SLSQP polish
    for T in [0.002, 0.0005, 0.00015]:
        if budget_s - elapsed() < 50:
            break
        try_upd(do_slsqp(h_best, n_best, T, 1500), n_best)

    # Phase 4: Subgradient descent (escape local optima)
    if budget_s - elapsed() > 150:
        h_sg = h_best.copy()
        lr = 0.02
        for it in range(2000):
            if budget_s - elapsed() < 60:
                break
            h_sg = subgradient_step(h_sg, n_best, lr)
            if it % 100 == 99:
                lr *= 0.85
                c5_sg = compute_c5(h_sg)
                if c5_sg < c5_best:
                    h_best, c5_best = h_sg.copy(), c5_sg
        try_upd(do_lbfgsb(h_sg, n_best, 0.001, 1500), n_best)

    # Phase 5: Structured multi-start
    for trial in range(6):
        if budget_s - elapsed() < 150:
            break
        h_p = h_best + np.random.randn(n_best) * 0.03
        h_p = np.clip(h_p, 0, 1)
        h_p = fix_sum(h_p, n_best)
        h_p = do_lbfgsb(h_p, n_best, 0.005, 1500)
        h_p = do_lbfgsb(h_p, n_best, 0.001, 1500)
        try_upd(h_p, n_best)
        # Also try shifted boundary perturbation
        if budget_s - elapsed() < 100:
            break
        h_s = h_best.copy()
        start = np.random.randint(0, n_best - 10)
        width = np.random.randint(5, 25)
        h_s[start:start + width] += 0.08 * np.sin(np.linspace(0, np.pi, width))
        h_s = np.clip(h_s, 0, 1)
        h_s = fix_sum(h_s, n_best)
        h_s = do_lbfgsb(h_s, n_best, 0.003, 1500)
        h_s = do_slsqp(h_s, n_best, 0.0005, 1000)
        try_upd(h_s, n_best)

    # Phase 6: Alternative resolutions
    for n_try in [200, 480, 600]:
        if budget_s - elapsed() < 200:
            break
        hi = upsample(h_best, n_try) if n_try > n_best else downsample(h_best, n_try)
        for T in [0.01, 0.003, 0.001]:
            if budget_s - elapsed() < 80:
                break
            hi = do_lbfgsb(hi, n_try, T, 2000)
        hi = do_slsqp(hi, n_try, 0.0005, 1500)
        if try_upd(hi, n_try):
            pass

    # Phase 7: Final polish
    for T in [0.0005, 0.0002, 0.00008]:
        if budget_s - elapsed() < 35:
            break
        try_upd(do_lbfgsb(h_best, n_best, T, 3000), n_best)
    if budget_s - elapsed() > 30:
        try_upd(do_slsqp(h_best, n_best, 0.0001, 2000), n_best)
    # Micro-perturbation + final
    if budget_s - elapsed() > 40:
        hm = h_best + np.random.randn(n_best) * 0.004
        hm = np.clip(hm, 0, 1)
        hm = fix_sum(hm, n_best)
        try_upd(do_lbfgsb(hm, n_best, 0.0003, 2000), n_best)
        if budget_s - elapsed() > 25:
            try_upd(do_slsqp(h_best, n_best, 0.00005, 1500), n_best)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    return (h_best, compute_c5(h_best), len(h_best))
