        m2 = hi - (hi-lo)/3
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2

Now for the coordinate descent polishing:

The key insight I derived earlier: for each coordinate i, the objective is piecewise-linear and convex in h[i] (when other coordinates are fixed). I can find the optimal value using ternary search.

where the indices are in the 'full' correlate convention.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h = h * (target / total)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        idx = np.where((h > 1e-10) & (h < 1-1e-10))[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, T, penalty, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exp_terms = np.exp(np.clip(T * (corr - m), -50, 50))
    smooth = m + np.log(np.sum(exp_terms)) / T
    n = len(x)
    return smooth * (2.0/n) + penalty * (x.sum() - target) ** 2

def lfbgsb_run(h, T, penalty, maxiter=300):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smooth_obj(x, T, penalty, target)
    
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

def slsqp_run(h, T, maxiter=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        exp_terms = np.exp(np.clip(T * (corr - m), -50, 50))
        return (m + np.log(np.sum(exp_terms)) / T) * dx
    
    def eq(x):
        return x.sum() - target
    
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

def coord_polish(h, n_iter=500):
    """Coordinate descent: try random perturbation of one coordinate."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    np.random.seed(12345)
    for it in range(n_iter):
        i = np.random.randint(n)
        old_val = h[i]
        # Try a few candidate values
        candidates = [0.0, 1.0, 0.5, old_val + 0.1, old_val - 0.1,
                      old_val + 0.05, old_val - 0.05, 1.0 - old_val]
        best_v = old_val
        best_c = compute_c5(h)
        for v in candidates:
            v = np.clip(v, 0, 1)
            h[i] = v
            h = project_feasible(h, n)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_v = v
                # Keep the projected version
            h[i] = old_val  # restore
        if best_v != old_val:
            h[i] = best_v
            h = project_feasible(h, n)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h = h * (target / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def _lfbg_obj(x, T, penalty, target, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exp_t = np.exp(np.minimum(T * (corr - m), 0.0))
    return (m + np.log(np.sum(exp_t)) / T) * dx + penalty * (x.sum() - target) ** 2

    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exp_t = np.exp(np.minimum(T * (corr - m), 0.0))
    return (m + np.log(np.sum(exp_t)) / T) * dx
