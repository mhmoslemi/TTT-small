```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(80):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        cnt = int(m.sum())
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def topk_loss(h, t, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argpartition(corr, -topk)[-topk:]
    c = corr[idx]
    mx = c.max()
    s = np.clip(t * (c - mx), -50, 50)
    return mx + np.log(np.sum(np.exp(s))) / t

def grad_lag(h, m):
    n = len(h)
    N = n - 1
    g = np.zeros(n)
    idx1 = np.arange(n) - m + N
    v1 = (idx1 >= 0) & (idx1 < n)
    if v1.any():
        g[v1] += (1.0 - h[idx1[v1]])
    idx2 = np.arange(n) + m - N
    v2 = (idx2 >= 0) & (idx2 < n)
    if v2.any():
        g[v2] -= h[idx2[v2]]
    return g

def grad_topk_loss(h, t, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cvals = corr * dx
    idx = np.argpartition(cvals, -topk)[-topk:]
    c = cvals[idx]
    mx = c.max()
    s = np.clip(t * (c - mx), -50, 50)
    w = np.exp(s) / np.sum(np.exp(s))
    g = np.zeros(n)
    for wt, m in zip(w, idx):
        g += wt * grad_lag(h, m)
    return g * dx

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def slsqp_run(h, n, obj_fn, args, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    r = minimize(obj_fn, h, args=args, method='SLSQP',
                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def pgd_optimize(h, n, t, topk, max_steps=20000, eta0=0.02):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    eta = eta0
    momentum_v = np.zeros(n)
    mom = 0.9
    for it in range(max_steps):
        g = grad_topk_loss(h, t, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        momentum_v = mom * momentum_v + (1 - mom) * g
        step = eta * momentum_v / (gn + 1e-15)
        h_new = h - step
        h_new = proj(h_new, n)
        c5 = compute_c5(h_new)[0]
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        eta = eta0 * (1.0 + it * 0.001) ** (-0.5)
    return best_h, best_c5

        if not improved:

        h_pgd, c5_pgd = pgd_optimize(bh, n, 5000, 2, max_steps=5000, eta0=0.005)
        if c5_pgd < bc5:
            bc5 = c5_pgd
            bh = h_pgd

    bh = proj(bh, n)
    final_c5 = compute_c5(bh)[0]
    if final_c5 < bc5:
        bc5 = final_c5
    return bh, bc5, n
```
