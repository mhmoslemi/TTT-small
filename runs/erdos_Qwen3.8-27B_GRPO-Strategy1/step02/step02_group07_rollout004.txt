        cm = np.max(tv)
        wts = np.exp((tv - cm) / T)
        wts /= np.sum(wts)
    grad = np.zeros(n)
    for ii, mi in enumerate(sel):
        w = wts[ii]
        if w < 1e-14:
            continue
        m = mi - (n - 1)
        lo1 = max(0, -m)
        hi1 = min(n, n - m)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 + m:hi1 + m])
        lo2 = max(0, m)
        hi2 = min(n, n + m)
        if lo2 < hi2:
            grad[lo2:hi2] += w * (-h[lo2 - m:hi2 - m])
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, n)

def subgrad_descent(h, n, max_iters, lr, T, k):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_lr = lr
    stall = 0
    for it in range(max_iters):
        h2 = subgrad_step(h, n, T, k, cur_lr)
        c2 = compute_c5(h2)
        if c2 < best_c5 - 1e-12:
            best_h = h2.copy()
            best_c5 = c2
            h = h2
            cur_lr = min(cur_lr * 1.02, lr * 2.0)
            stall = 0
        elif c2 < best_c5 + 1e-8:
            h = h2
            stall += 1
        else:
            cur_lr *= 0.85
            stall += 1
        if stall > 30:
            cur_lr *= 0.5
            stall = 10
        if cur_lr < 1e-10:
            break
    return best_h, best_c5

def coord_polish(h, n, n_pass=2):
    for _ in range(n_pass):
        best_c5 = compute_c5(h)
        for j in range(n):
            if h[j] <= 1e-4 or h[j] >= 1.0 - 1e-4:
                continue
            for delta in (-0.02, -0.005, 0.005, 0.02):
                hn = h.copy()
                hn[:] += -delta / (n - 1)
                hn[j] = min(max(h[j] + delta, 0.0), 1.0)
                hn = np.clip(hn, 0.0, 1.0)
                c = compute_c5(hn)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h = hn.copy()
        h = fix_sum(h, n)
    return h

def upsample(h, nn):
    x0 = np.arange(len(h), dtype=float) * (2.0 / len(h))
    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 13)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = fix_sum(h, n)
    c5 = compute_c5(h)

    # Phase 1: Fast subgradient descent (focus on top-2 to top-5 lags)
    for T, k, iters, lr in [(0.03, 5, 3000, 0.04), (0.01, 3, 3000, 0.03), (0.004, 2, 4000, 0.02)]:
        if time.time() - t0 > budget_s - 550:
            break
        h, c5_new = subgrad_descent(h, n, iters, lr, T, k)
        c5 = min(c5, c5_new)
        h = fix_sum(h, n)

    # Phase 2: SLSQP with top-k sharpening
    for T, k, iters in [(0.02, 5, 500), (0.008, 3, 700), (0.003, 2, 900), (0.0008, 1, 1200)]:
        if time.time() - t0 > budget_s - 400:
            break
        h = slsqp_topk(h, n, T, k, iters)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 3: Coordinate polish
    if time.time() - t0 < budget_s - 250:
        h = fix_sum(coord_polish(h, n, 2), n)
        c5 = compute_c5(h)

    # Phase 4: More subgradient (target single worst lag)
    for T, k, iters, lr in [(0.002, 2, 4000, 0.01), (0.0005, 1, 5000, 0.005)]:
        if time.time() - t0 > budget_s - 180:
            break
        h, c5_new = subgrad_descent(h, n, iters, lr, T, k)
        c5 = min(c5, c5_new)
        h = fix_sum(h, n)

    # Phase 5: Diverse restarts with structured perturbations
    for i in range(12):
        if time.time() - t0 > budget_s - 130:
            break
        ht = h.copy()
        mode = i % 4
        if mode == 0:
            ht += np.random.randn(n) * 0.012
        elif mode == 1:
            idx = np.random.choice(n, 6, replace=False)
            ht[idx] += np.random.randn(6) * 0.08
        elif mode == 2:
            # Perturb correlated with the worst lag structure
            corr = np.correlate(ht, 1.0 - ht, mode='full')
            mi = np.argmax(corr)
            m_shift = mi - (n - 1)
            ht += np.roll(ht, m_shift) * 0.01 + np.random.randn(n) * 0.005
        else:
            ht += np.random.randn(n) * 0.004
        ht = fix_sum(ht, n)
        for T, kk, iters in [(0.01, 3, 500), (0.003, 2, 700)]:
            ht = slsqp_topk(ht, n, T, kk, iters)
        ht = fix_sum(ht, n)
        ct = compute_c5(ht)
        if ct < c5:
            h, c5 = ht, ct
            if time.time() - t0 < budget_s - 80:
                h = fix_sum(coord_polish(h, n, 1), n)
                c5 = compute_c5(h)

    # Phase 6: Final SLSQP with very sharp objective
    for T, k, iters in [(0.0003, 1, 2000), (0.0001, 1, 2500)]:
        if time.time() - t0 > budget_s - 40:
            break
        h = slsqp_topk(h, n, T, k, iters)
        h = fix_sum(h, n)
        c5 = compute_c5(h)

    # Phase 7: Upscale and refine
    if time.time() - t0 < budget_s - 70:
        n2 = min(n * 2, 560)
        h2 = upsample(h, n2)
        for T, k, iters in [(0.02, 3, 400), (0.005, 2, 600), (0.001, 1, 800)]:
            if time.time() - t0 > budget_s - 20:
                break
            h2 = slsqp_topk(h2, n2, T, k, iters)
            h2 = fix_sum(h2, n2)
        c2 = compute_c5(h2)
        if c2 < c5:
            h, n, c5 = h2, n2, c2

    h = fix_sum(h, len(h))
    return (h, compute_c5(h), len(h))
