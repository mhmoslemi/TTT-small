# Global state for SLSQP callbacks
_sp = {}

def _sobj(x):
    return x[_sp['n']]

def _sineq(x):
    n = _sp['n']
    return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')

def _siq(x):
    return float(np.sum(x[:_sp['n']]) - _sp['ts'])

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14: break
        m = (h < 1-1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0: break
        h[m] += d/c
    return h

def sgrad(h, beta):
    n = len(h)
    cr = np.correlate(h, 1.0-h, mode='full')
    w = np.exp(beta*(cr - cr.max()))
    s = w.sum()
    if s < 1e-30: w = np.ones(2*n-1)/(2*n-1)
    else: w /= s
    o = n-1
    g = np.zeros(n)
    for j in range(n):
        lo1, hi1 = max(0, j), min(2*n-2, j+n-1)
        if lo1 <= hi1:
            ks = np.arange(lo1, hi1+1)
            g[j] += np.dot(w[ks], 1.0 - h[j-ks+o])
        lo2, hi2 = max(0, o-j), min(2*n-2, 2*n-2-j)
        if lo2 <= hi2:

            g[j] -= np.dot(w[ks], h[j+ks-o])
    return g

    st = 0

            lr *= 0.5

            st += 1
            if st > 100: break
    return bh, bc

    """Coordinate descent on pairs targeting worst shift"""
    n = len(h0)
    h, bc = h0.copy(), compute_c5(h0, dx)
    for r in range(rounds):

                    h2[j] -= d
                    h2[partner] += d
                    if h2[j] < -1e-10 or h2[partner] > 1+1e-10: continue
                    h2 = proj(h2, ts)
                    c2 = compute_c5(h2, dx)
                    if c2 < bc:
                        h, bc, imp = h2, c2, True
                        break
                if imp and c2 < bc - 0.001: break
        if not imp: break
    return h, bc

def sgrad(h, beta):
    n = len(h)
    cr = np.correlate(h, 1.0-h, mode='full')
    w = np.exp(beta*(cr - cr.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(2*n-1)/(2*n-1)
    else:
        w /= s
    o = n-1
    g = np.zeros(n)
    for j in range(n):
        # First sum: k from max(0,j) to min(2n-2, j+n-1)
        lo1, hi1 = max(0, j), min(2*n-2, j+n-1)
        if lo1 <= hi1:
            ks = np.arange(lo1, hi1+1)
            g[j] += np.dot(w[ks], 1.0 - h[j-ks+o])
        # Second sum: k from max(0, o-j) to min(2n-2, 2n-2-j)
        lo2, hi2 = max(0, o-j), min(2*n-2, 2n-2-j)
        if lo2 <= hi2:
            ks = np.arange(lo2, hi2+1)
            g[j] -= np.dot(w[ks], h[j+ks-o])
    return g
