import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_lags(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:2*n-1]  # lags 0 to n-1

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = compute_corr_lags(h)
    c_max = np.max(c)
    if T < 1e-15:
        return c_max * dx
    return (c_max + T * np.log(np.sum(np.exp((c - c_max) / T)))) * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = compute_corr_lags(h)
    c_max = np.max(c)
    w = np.exp((c - c_max) / T)
    w = w / np.sum(w)
    # gradient[j] = dx * (sum_{k=0}^{j} w[k]*(1-h[j-k]) - sum_{k=0}^{n-1-j} w[k]*h[j+k])

    # term1[j] = sum_{k=0}^{j} w[k] * (1-h[j-k])
    # = sum_{k=0}^{j} w[k] - sum_{k=0}^{j} w[k]*h[j-k]
    # = cumsum(w)[j] - (w * h)[j]  where (w*h) is convolution
    
    # Let me compute directly:
    # term1[j] = sum_{k=0}^{j} w[k] * (1-h[j-k])
    # This is the convolution (w_conv * (1-h))[j] where w_conv = w[0], w[1], ...
    # Actually term1[j] = sum_{k=0}^{j} w[k] * (1-h[j-k]) = (w * (1-h))[j] in linear convolution
    
    # term2[j] = sum_{k=0}^{n-1-j} w[k] * h[j+k]
    # = sum_{k=0}^{n-1-j} w[k] * h[j+k]
    # Let m = j+k: = sum_{m=j}^{n-1} w[m-j] * h[m]
    # This is like a cross-correlation...
    
    # For n=336, O(n^2) is fine:
    grad = np.zeros(n)
    for j in range(n):
        # term1: sum_{k=0}^{j} w[k]*(1-h[j-k])
        t1 = np.dot(w[:j+1], (1.0 - h[j::-1]))  # h[j], h[j-1], ..., h[0]

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    t = 0.5 * n
    if abs(h.sum() - t) < 1e-15:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(64):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0, 1).sum() > t:
            hi = m
        else:
            lo = m
    h = np.clip(h + 0.5 * (lo + hi), 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-13:
        msk = (h > 1e-12) & (h < 1 - 1e-12)
        if msk.any():
            h[msk] += d / msk.sum()
        h = np.clip(h, 0, 1)
    return h

def lags(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, 'full')[n - 1:]

def sm_obj(h, T):
    c = lags(h)
    cm = c.max()
    if T < 1e-16:
        return cm * 2.0 / len(h)
    return (cm + T * np.log(np.exp((c - cm) / T).sum())) * 2.0 / len(h)

def sm_grad(h, T):
    n = len(h)
    c = lags(h)
    cm = c.max()
    if T < 1e-16:
        w = np.zeros(n)
        w[c >= cm - 1e-12] = 1.0
        w /= w.sum()
    else:
        w = np.exp((c - cm) / T)
        w /= w.sum()
    t1 = np.convolve(w, 1.0 - h, 'full')[:n]
    t2 = np.convolve(w, h[::-1], 'full')[n - 1::-1]
    return 2.0 / n * (t1 - t2)

def act_grad(h, kl):
    n = len(h)
    g = np.zeros(n)
    for k in kl:
        if k > 0:
            g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    return 2.0 / n * g

def act_obj(h, kl):
    c = lags(h)
    return c[kl].sum()

    c_obj = sm_obj(h, T)
    for _ in range(iters):
        if time.time() - t0 > tmax:
            break

        h2 = proj(h - lr_s * g, n)
        c2 = sm_obj(h2, T)
        if c2 < c_obj:
            h, c_obj = h2, c2
        else:
            lr_s *= 0.8
            if lr_s < 1e-6:
                lr_s = lr
        c = compute_c5(h)
        if c < cb:
            cb, hb = c, h.copy()
    return hb, cb
