import numpy as np
import time
import scipy.optimize as opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = interior.sum()
    if n_int > 0 and abs(diff) < n_int * 0.49:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0:
            for i in np.argsort(h):
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            diff = -diff
            for i in np.argsort(-h):
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if diff <= 1e-12:
                    break
    return h

def make_binary(h, n, rng):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    target = n // 2
    ones = (h == 1.0).astype(int).sum()
    if ones > target:
        idx = np.where(h == 1.0)[0]
        rem = rng.choice(idx, ones - target, replace=False)
        h[rem] = 0.0
    elif ones < target:
        idx = np.where(h == 0.0)[0]
        add = rng.choice(idx, target - ones, replace=False)
        h[add] = 1.0
    return h

def binary_sa(h_init, n, time_budget, seed=42):
    rng = np.random.RandomState(seed)
    h = make_binary(h_init, n, rng)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    temp = 0.005
    while time.time() - t0 < time_budget:
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-15:
            best_h = h.copy()
            best_c5 = new_c5
            temp = max(temp * 0.95, 0.001)
        elif rng.uniform() < np.exp(-(new_c5 - best_c5) / max(temp, 1e-8)):
            temp *= 0.9997
        else:
            h[i] = 1.0
            h[j] = 0.0
            temp *= 0.9997
    return best_h, best_c5

def upsample_periodic(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, np.append(h, h[:1]), period=2.0)
    return fix_sum(h_new, new_n)

def slsqp_opt(h, n, T, max_iter):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = corr.max()
        shifted = np.clip(corr - cmax, -80.0 * T, 0.0)
        s = shifted.sum(np.exp(shifted / T))
        return (cmax + T * np.log(max(s, 1e-300))) * (2.0 / n)
    def con(x):
        return x.sum() - n * 0.5
    try:
        res = opt.minimize(obj, h.copy(), method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': con}],
                          options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def active_set_opt(h, n, top_k, T, max_iter):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        tv = c[top_idx]
        t = T
        return float(tv.max() + t * np.log(np.sum(np.exp((tv - tv.max()) / t)) / len(tv)))
    def con(x):
        return x.sum() - n * 0.5
    try:
        res = opt.minimize(obj, h.copy(), method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': con}],
                          options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def coordinate_descent(h, n, time_budget, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    step = 0.1
    while time.time() - t0 < time_budget:
        i = rng.randint(n)
        try_scalar = h[i] + step
        h[i] = min(1.0, max(0.0, try_scalar))
        h = fix_sum(h, n)
        c = compute_c5(h)
        if c > best_c5 + 1e-12:
            h[i] = try_scalar - step
            h = fix_sum(h, n)
        else:
            best_c5 = c
        if rng.uniform() < 0.5:
            step = -step
        if best_c5 < 0.3809:
            step *= 0.9
    return fix_sum(h, n)
