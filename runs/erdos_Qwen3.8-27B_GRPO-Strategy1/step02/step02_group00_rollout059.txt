            for delta in [0.015, 0.03, 0.06, -0.015, -0.03, -0.06]:
                h_try = best_h.copy()
                h_try[j] = np.clip(h_try[j] + delta, 0, 1)
                h_try = project_feasible(h_try, n_cur)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
            if improved:
                break
        if not improved and it > 80:
            break

    # Final feasibility fix
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
