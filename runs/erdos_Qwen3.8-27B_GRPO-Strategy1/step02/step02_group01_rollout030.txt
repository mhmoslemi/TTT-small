import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def slsqp_opt(h, T=None, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    if T is not None:
        obj = lambda x: smooth_c5(x, T)
    else:
        # Use penalty for the max
        def obj(x):
            c = compute_c5(x)
            pen = abs(x.sum() - n/2.0) * 100
            return c + pen
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:

But wait, in the inner loop I'm breaking and restarting when an improvement is found. That's actually a steepest-descent-like approach. Let me just do full passes without breaking:

Actually, the issue is that in a full pass, after making one change, the other positions change too. So the "greedy" approach is:
- For each pair (a,b), try the swap
- If it improves, keep it and move on (both a and b are now in the "other" group)
- If it doesn't, revert

This is a simple greedy sweep. Multiple sweeps until no improvement.

Let me also think about the **vectorized swap evaluation**. For a fixed swap (a: 1→0, b: 0→1), the change in correlation can be computed in O(n):

C_new[j] = C_old[j] + delta_a[j] + delta_b[j] + cross[j]

where:
- delta_a[j]: effect of h[a] going from 1 to 0
  = -(1 - h[a+j-(n-1)]) if 0 <= a+j-(n-1) < n, else 0
  (this is from the first factor h[i] with i=a)
  Plus: effect on the second factor (1-h[...])
  When i+j-(n-1) = a, i.e., j = a-i+(n-1): h[i] changes from h[i] to h[i] (no change in h[i])
  Wait, (1-h) at position a changes from 0 to 1. So for all j where i+j-(n-1)=a (i.e., i = a-j+(n-1)):
  The term h[i]*(1-h[a]) changes from h[i]*0 to h[i]*1, so delta = +h[a-j+(n-1)]
  (valid when 0 <= a-j+(n-1) < n)

- delta_b[j]: effect of h[b] going from 0 to 1
  = +1*(1-h[b+j-(n-1)]) if 0 <= b+j-(n-1) < n
  Plus: (1-h) at position b changes from 1 to 0. For j where i+j-(n-1)=b (i = b-j+(n-1)):
  The term h[i]*(1-h[b]) changes from h[i]*1 to h[i]*0, so delta = -h[b-j+(n-1)]

- cross[j]: second-order term (both a and b change simultaneously)
  This is non-zero only at specific lags.

This is getting complex. For n=90, let me just compute the full correlation each time. It's fast enough.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_direct(h, maxiter=300):
    """SLSQP with subgradient-like approach using max + barrier."""
    n = len(h)
    def obj(x):
        c = compute_c5(x)
        pen = max(0, abs(x.sum() - n/2.0) - 1e-10) * 1000
        return c + pen
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    n = len(h)
    n_t = n // 2
    h_b = (h > 0.5).astype(float)
    if h_b.sum() != n_t:
        idx = np.argsort(-h)
        h_b = np.zeros(n)
        h_b[idx[:n_t]] = 1.0
    best_c5 = compute_c5(h_b)
    for _ in range(max_passes):
        improved = False
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for a in ones:
            for b in zeros:
                h_b[a], h_b[b] = 0.0, 1.0
                c5 = compute_c5(h_b)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                    ones = np.where(h_b == 1)[0]
                    zeros = np.where(h_b == 0)[0]
                    break
                else:
                    h_b[a], h_b[b] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h_b, best_c5

def sa_binary(h, n, max_iter=80000, T_start=0.004, T_end=0.00005):
    h_c = (h > 0.5).astype(float)
    n_t = n // 2
    if h_c.sum() != n_t:

        h_c = np.zeros(n)
        h_c[idx[:n_t]] = 1.0

    c5_best = compute_c5(h_c)
    c5_cur = c5_best
    for it in range(max_iter):
        T = T_start * (T_end / T_start) ** (it / max_iter)
        ones = np.where(h_c == 1)[0]
        zeros = np.where(h_c == 0)[0]
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        h_c[a], h_c[b] = 0.0, 1.0
        c5_new = compute_c5(h_c)
        if c5_new <= c5_cur:
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_c.copy()
        elif np.random.rand() < np.exp(-(c5_new - c5_cur) / max(T, 1e-10)):
            c5_cur = c5_new
        else:
            h_c[a], h_c[b] = 1.0, 0.0
    return h_best, c5_best

def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

        if elapsed() > budget_s * 0.55:

        if n_res == best_n:

        else:

        # Binary greedy
        h_b, c5_b = binary_greedy(h_base, max_passes=40)
        if c5_b < best_c5:
            best_c5, best_h, best_n = c5_b, h_b, n_res
        
        # SA with multiple restarts
        n_sa = 3 if n_res <= 90 else 2
        for r in range(n_sa):
            if elapsed() > budget_s * 0.55:
                break
            h_pert = h_b + np.random.randn(n_res) * 0.15
            h_pert = project_feasible(h_pert, n_res)
            h_sa, c5_sa = sa_binary(h_pert, n_res, max_iter=40000, T_start=0.003, T_end=0.00005)
            if c5_sa < best_c5:
                best_c5, best_h, best_n = c5_sa, h_sa, n_res
                h_b = h_sa
    
    # Phase 3: SLSQP refinement at best resolution
    if elapsed() < budget_s * 0.75:
        for T in [200, 1000, 5000]:
            if elapsed() > budget_s * 0.75:
                break
            h_try = slsqp_smooth(best_h, T, maxiter=300)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try

        c5_try = compute_c5(h_try)
        if c5_try < best_c5:

    # Phase 4: Basin hopping at best resolution
    n_cur = best_n
    n_trials = min(20, max(5, int((budget_s * 0.75 - elapsed()) / 3)))
    for i in range(n_trials):
        if elapsed() > budget_s * 0.85:
            break
        scale = 0.08 * (1.0 - 0.03 * i)
        h_pert = best_h + np.random.randn(n_cur) * scale
        h_pert = project_feasible(h_pert, n_cur)
        # Binary snap + refine
        h_b, c5_b = binary_greedy(h_pert, max_passes=20)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b
        h_try = slsqp_smooth(h_pert, 1000, maxiter=200)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
    
    # Phase 5: Final SLSQP + binary polish
    if elapsed() < budget_s * 0.95:
        h_try = slsqp_smooth(best_h, 5000, maxiter=400)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        h_b, c5_b = binary_greedy(best_h, max_passes=50)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    h = np.clip(h, 0.0, 1.0).copy()

        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)

            if total < 1e-14:
                break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h
            total = room.sum()
            if total < 1e-14:
                break
            h = h + room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m) + 1e-10))) / T
    return float(val * dx)

def sum_eq(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_c5_obj, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_direct(h, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(compute_c5, h, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def binary_greedy(h, max_passes=30):
    n = len(h)
    n_t = n // 2
    h_b = (h > 0.5).astype(float)
    if int(np.sum(h_b)) != n_t:
        idx = np.argsort(-h)
        h_b = np.zeros(n)
        h_b[idx[:n_t]] = 1.0
    best_c5 = compute_c5(h_b)
    for _ in range(max_passes):
        improved = False
        ones = list(np.where(h_b == 1)[0])
        zeros = list(np.where(h_b == 0)[0])
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for a in ones:
            for b in zeros:
                h_b[a], h_b[b] = 0.0, 1.0
                c5 = compute_c5(h_b)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                    break
                else:
                    h_b[a], h_b[b] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h_b, best_c5

def sa_binary(h, n, max_iter=60000, T_start=0.004, T_end=0.00005):
    h_c = (h > 0.5).astype(float)
    n_t = n // 2
    if int(np.sum(h_c)) != n_t:
        idx = np.random.permutation(n)
        h_c = np.zeros(n)
        h_c[idx[:n_t]] = 1.0
    h_best = h_c.copy()
    c5_best = compute_c5(h_c)
    c5_cur = c5_best
    log_ratio = np.log(T_end / T_start) / max_iter
    for it in range(max_iter):
        T = T_start * np.exp(log_ratio * it)
        ones = np.where(h_c == 1)[0]
        zeros = np.where(h_c == 0)[0]
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        h_c[a], h_c[b] = 0.0, 1.0
        c5_new = compute_c5(h_c)
        if c5_new <= c5_cur:
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_c.copy()
        elif T > 1e-12 and np.random.rand() < np.exp(-(c5_new - c5_cur) / T):
            c5_cur = c5_new
        else:
            h_c[a], h_c[b] = 1.0, 0.0
    return h_best, c5_best

def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_feasible(h_old.copy(), n_new)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
