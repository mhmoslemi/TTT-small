import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, T, maxiter=500):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    def scons(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': scons}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
