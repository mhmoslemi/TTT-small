        return project_feasible(np.clip(res.x, 0, 1), n)

c[k] = sum_n a[n+k] * conj(v[n])

So c[k] = sum_n h[n+k] * (1-h[n]) for valid n and n+k.

The length is 2n-1, with k ranging from -(n-1) to (n-1), stored as indices 0 to 2n-2.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            s = np.sign(diff)
            step = min(abs(diff), abs(1.0 - h[i]) if diff > 0 else h[i])
            h[i] += s * step
            diff -= s * step
            if abs(diff) < 1e-14:
                break
    return h

def slsqp_smooth(h, T, maxiter=200):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx + 50.0 * abs(x.sum() - n / 2.0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_epeg(h, maxiter=300):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        t = x[-1]
        hv = x[:-1]
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        return t * dx
    def con_eq(x):
        return x[:-1].sum() - (len(x)-1) / 2.0
    def con_ineq_all(x):
        n2 = len(x) - 1
        hv = x[:-1]
        t = x[-1] / (2.0 / n2)
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        return t - corr
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': con_eq}]
    c0 = np.max(np.correlate(h, 1.0 - h, mode='full'))
    x0 = np.concatenate([h, [c0 + 0.05]])
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_feasible(np.clip(res.x[:n], 0, 1), n)
        return h_opt
    except Exception:
        return h

def targeted_cd(h, n_iters=2000):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - h, mode='full')
    peak_idx = np.argmax(corr)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for si, step in enumerate(steps):
        iters = max(400, n_iters // len(steps))
        for it in range(iters):
            improved = False
            if np.random.random() < 0.6:
                i = np.random.randint(n)
            else:
                corners = np.where((best > 0.05) & (best < 0.95))[0]
                if len(corners) > 0:
                    i = corners[np.random.randint(len(corners))]
                else:
                    i = np.random.randint(n)
            for d in [step, -step, 2*step, -2*step, 0.5*step, -0.5*step]:
                h_try = best.copy()
                h_try[i] = np.clip(best[i] + d, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if not improved and np.random.random() < 0.4:
                i, j = np.random.choice(n, 2, replace=False)
                for di in [step, -step]:
                    for dj in [step, -step]:
                        h_try = best.copy()
                        h_try[i] = np.clip(best[i] + di, 0.0, 1.0)
                        h_try[j] = np.clip(best[j] + dj, 0.0, 1.0)
                        h_try = project_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-12:
                            best = h_try
                            best_c5 = c5_try
                            improved = True
                            break
                    if improved:
                        break
    return best, best_c5

def sa_refine(h, n_iter=5000, T_start=0.02, T_end=0.0005):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    cur_c5 = best_c5
    for i in range(n_iter):
        T = T_start * (T_end / T_start) ** (i / max(n_iter - 1, 1))
        scale = 0.025 * (1.0 + 0.5 * T / T_start)
        h_try = cur + np.random.randn(n) * scale
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        d = c5_try - cur_c5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-12)):
            cur = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
    return best, best_c5

def golden_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n)
    h = ((x * phi) % 1.0 < 0.5).astype(float)
    return project_feasible(h, n)

def layered_init(n, frac=0.5):
    h = np.zeros(n)
    block = max(1, n // 8)
    for i in range(0, n, 2 * block):
        h[i:i+block] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = golden_init(90)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP warmup
    if remaining() > 80:
        for T in [300, 1500, 6000]:
            if remaining() < 30:
                break
            h_try = slsqp_smooth(best_h, T, maxiter=150)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try

    # Phase 2: Direct epigraph SLSQP
    if remaining() > 60:
        h_try = slsqp_epeg(best_h, maxiter=400)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 3: Targeted coordinate descent (main workhorse)
    if remaining() > 80:
        n_iters = min(4000, max(800, int(remaining() * 0.3)))
        best_h, best_c5 = targeted_cd(best_h, n_iters)

    # Phase 4: Multi-start with diverse initializations
    diverse_starts = []
    for n_try in [80, 90, 100, 84, 96, 108, 120]:
        if remaining() < 40:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        if n_try != best_n:
            h_interp = 0.7 * h_interp + 0.3 * golden_init(n_try)
            h_interp = project_feasible(h_interp, n_try)
        diverse_starts.append((n_try, h_interp))

    for n_try, h_start in diverse_starts:
        if remaining() < 50:
            break
        h_try = slsqp_smooth(h_start, 800, maxiter=200)
        h_try, c5_try = targeted_cd(h_try, n_iters=800)
        if c5_try < best_c5:
            best_c5, best_h, best_n = c5_try, h_try, n_try

    # Phase 5: Simulated annealing on best
    if remaining() > 60:
        n_sa = min(6000, max(1500, int(remaining() * 0.25)))
        best_h, best_c5 = sa_refine(best_h, n_iter=n_sa)
        best_h, best_c5 = targeted_cd(best_h, n_iters=1000)

    # Phase 6: Intensive final polish loop
    polish_rounds = 3
    for r in range(polish_rounds):
        if remaining() < 20:
            break
        h_try = slsqp_smooth(best_h, 10000, maxiter=300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        h_try, c5_try = targeted_cd(best_h, n_iters=max(500, int(remaining() * 0.15)))
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        if remaining() < 15:
            break
        h_try = slsqp_epeg(best_h, maxiter=250)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 7: Tiny final adjustments
    if remaining() > 10:
        n = len(best_h)
        for _ in range(500):
            if remaining() < 5:
                break
            i = np.random.randint(n)
            for d in [0.001, -0.001, 0.003, -0.003]:
                h_try = best_h.copy()
                h_try[i] = np.clip(best_h[i] + d, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5, best_h = c5_try, h_try
                    break

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
