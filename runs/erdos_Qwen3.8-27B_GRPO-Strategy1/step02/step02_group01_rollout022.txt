        diff = target - h.sum()
        if abs(diff) < 1e-14:

            h_left = np.concatenate([np.zeros(rn // 2), np.ones(rn // 2)]).astype(np.int8)
            # Use periodic as one individual, GA will explore
            pass

        h_ga, c5_ga = ga_binary(rn, 120, 3000, seed + 7 * rn, dl)
        if h_ga is not None:
            ga_results[rn] = (h_ga, c5_ga)
            if c5_ga < best_c5:
                best_c5 = c5_ga
                best_h = h_ga.copy()
                best_n = rn

    # Also try starting from parent rounded to binary
    if best_n > 0 and time.time() < hard_deadline - 120:
        h_parent_bin = (best_h >= 0.5).astype(np.int8)
        diff = best_n // 2 - int(h_parent_bin.sum())
        if diff > 0:
            z = np.where(h_parent_bin == 0)[0]
            if len(z) >= diff:
                h_parent_bin[np.random.choice(z, diff, replace=False)] = 1
        elif diff < 0:
            o = np.where(h_parent_bin == 1)[0]
            if len(o) >= -diff:
                h_parent_bin[np.random.choice(o, -diff, replace=False)] = 0
        h_ls, c5_ls = local_search_steepest(h_parent_bin, best_n, max_rounds=20,
                                            deadline=hard_deadline - 100)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.astype(np.float64)

    # SLSQP polish on best
    if time.time() < hard_deadline - 60:
        h_pol, c5_pol = polish_slsqp(best_h, best_n, hard_deadline - 20)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Final local search if we have time
    if time.time() < hard_deadline - 30:
        h_fin_bin = (best_h >= 0.5).astype(np.int8)
        diff = best_n // 2 - int(h_fin_bin.sum())
        if diff > 0:
            z = np.where(h_fin_bin == 0)[0]
            if len(z) >= diff:
                h_fin_bin[np.random.choice(z, diff, replace=False)] = 1
        elif diff < 0:
            o = np.where(h_fin_bin == 1)[0]
            if len(o) >= -diff:
                h_fin_bin[np.random.choice(o, -diff, replace=False)] = 0
        h_fin, c5_fin = local_search_steepest(h_fin_bin, best_n, max_rounds=10,
                                              deadline=hard_deadline - 10)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.astype(np.float64)

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
