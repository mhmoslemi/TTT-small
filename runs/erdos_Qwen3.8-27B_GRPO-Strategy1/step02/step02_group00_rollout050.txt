def subgradient_fast(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Soft weighting: give more weight to near-max shifts
    tol = max(0.01 * m, 1e-6)
    weights = np.exp((corr - m) / (tol / n))  # softer for larger n
    weights /= weights.sum()
    # Or just hard:
    # active = (corr >= m - tol).astype(float)
    # weights = active / active.sum()
    
    grad_h = np.zeros(n)
    for i in range(2*n - 1):
        if weights[i] < 1e-12:
            continue
        s = i - n + 1
        w = weights[i]
        # First term: (1-h[t-s]) for t in [max(s,0), min(n-1+s, n-1)]
        t_min1 = max(s, 0)
        t_max1 = min(n - 1 + s, n - 1)
        if t_min1 <= t_max1:

        if t_min2 <= t_max2:
            grad_h[t_min2:t_max2+1] -= w * h[t_min2 + s:t_max2 + s + 1]
    
    return grad_h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-15:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-15:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def subgradient_fast(h, soft_frac=0.02):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    tol = max(soft_frac * (corr.max() - corr.min() + 1e-10), 1e-8)
    weights = np.clip((corr - (m - tol)) / tol, 0.0, 1.0)
    ws = weights.sum()
    if ws < 1e-12:
        weights = np.zeros(2 * n - 1)
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= ws
    grad_h = np.zeros(n)
    for i in range(2 * n - 1):
        if weights[i] < 1e-12:
            continue
        s = i - n + 1
        w = weights[i]
        t1a = max(s, 0)
        t1b = min(n - 1 + s, n - 1)
        if t1a <= t1b:
            grad_h[t1a:t1b + 1] += w * (1.0 - h[t1a - s:t1b - s + 1])
        t2a = max(0, -s)
        t2b = min(n - 1, n - 1 - s)
        if t2a <= t2b:
            grad_h[t2a:t2b + 1] -= w * h[t2a + s:t2b + s + 1]
    return grad_h

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0.0, 1.0), n)

def subgrad_phase(h0, n, n_steps, t0, time_limit, lr_scale=0.4, decay=0.004):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for st in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        grad = subgradient_fast(h)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        alpha = lr_scale / (1.0 + decay * st)
        h = project_feasible(h - alpha * grad, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def basin_hop(h0, n, t0, time_limit, T_opt=400, n_iter=80):
    bh = h0.copy()
    bc5 = compute_c5(h0)
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        frac_rand = 0.65 - 0.40 * (it / max(n_iter, 1))
        step = 0.09 * (1.0 - it / n_iter) + 0.02
        h_p = bh + step * frac_rand * np.random.randn(n)
        if frac_rand < 0.99:
            g = subgradient_fast(bh)
            gn = np.linalg.norm(g) + 1e-12
            h_p -= (1.0 - frac_rand) * 0.15 * g / (gn / (n ** 0.5))
        h_p = project_feasible(h_p, n)
        try:
            hc = slsqp_opt(h_p, T_opt, n, maxiter=250, ftol=1e-12)
            c5v = compute_c5(hc)
            temp = 0.004 * (1.0 - it / n_iter) + 0.0003
            if c5v < bc5 or np.random.rand() < np.exp(-(c5v - bc5) / temp):
                bc5 = c5v
                bh = hc
        except Exception:
            pass
    return bh, bc5

def coord_refine(h0, n, t0, time_limit, n_rounds=80):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for rnd in range(n_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_max = int(np.argmax(corr))
        s = k_max - n + 1
        deltas = [0.02, 0.05, 0.08, -0.02, -0.05, -0.08]
        lo = max(0, -s)
        hi = min(n, n - s)
        for i in range(lo, hi):
            j = i + s
            d_j = i - s
            if j < 0 or j >= n or d_j < 0 or d_j >= n:
                continue
            for d in deltas:
                if not (0 <= best_h[i] + d <= 1 and 0 <= best_h[j] - d <= 1):
                    continue
                h_try = best_h.copy()
                h_try[i] += d
                h_try[j] -= d
                c5t = compute_c5(h_try)
                if c5t < best_c5 - 1e-13:
                    best_c5 = c5t
                    best_h = h_try
                    improved = True
                    break
            if improved:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(120)
        best_h[:60] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: SLSQP polish on parent
    for T in [40, 150, 600]:
        if time.time() - t0 > budget_s * 0.06:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=300, ftol=1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 2: Long subgradient descent
    best_h, best_c5 = subgrad_phase(best_h, n, 3000, t0, budget_s * 0.22, lr_scale=0.35, decay=0.003)

    # Phase 3: SLSQP after subgrad
    for T in [100, 400, 1500]:
        if time.time() - t0 > budget_s * 0.34:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=500, ftol=1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 4: Basin hopping with gradient-informed kicks
    bh, bc5 = basin_hop(best_h, n, t0, budget_s * 0.58, T_opt=500, n_iter=90)
    if bc5 < best_c5:
        best_c5, best_h = bc5, bh

    # Phase 5: Low-res optimization + upsample
    if time.time() - t0 < budget_s * 0.80:
        for n_low in [50, 70, 90]:
            if time.time() - t0 > budget_s * 0.80:
                break
            xo = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, n_low, endpoint=False)
            h_low = project_feasible(np.interp(xn, xo, best_h), n_low)
            for _ in range(3):
                if time.time() - t0 > budget_s * 0.80:
                    break
                h_try = project_feasible(h_low + 0.05 * np.random.randn(n_low), n_low)
                for T in [80, 400]:
                    if time.time() - t0 > budget_s * 0.80:
                        break
                    try:
                        hc = slsqp_opt(h_try, T, n_low, maxiter=300, ftol=1e-13)
                        x_up = np.linspace(0, 2, n, endpoint=False)
                        h_up = project_feasible(np.interp(x_up, xn, hc), n)
                        c5u = compute_c5(h_up)
                        if c5u < best_c5:
                            best_c5, best_h = c5u, h_up
                    except Exception:
                        pass

    # Phase 6: Second subgrad + SLSQP polish
    if time.time() - t0 < budget_s * 0.92:
        best_h, best_c5 = subgrad_phase(best_h, n, 1500, t0, budget_s * 0.07, lr_scale=0.25, decay=0.002)
        for T in [800, 4000]:
            if time.time() - t0 > budget_s * 0.97:
                break
            try:
                hc = slsqp_opt(best_h, T, n, maxiter=700, ftol=1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
            except Exception:
                pass

    # Phase 7: Coordinate refinement on worst shift
    if time.time() - t0 < budget_s * 0.99:
        best_h, best_c5 = coord_refine(best_h, n, t0, budget_s * 0.01, n_rounds=60)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
