import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lsq_val(h, beta):
    cr = np.correlate(h, 1.0 - h, mode='full')
    m = cr.max()
    return (m + np.log(np.sum(np.exp(beta * (cr - m))))) / beta

def proj(h, ts):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        np.clip(h, 0, 1, out=h)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = (h < 1.0 - 1e-12)
        else:
            m = (h > 1e-12)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

    n = 0
    ts = 0.0
    beta = 10.0

    return lsq_val(x[:_SP.n], _SP.beta)

            if compute_c5(hp, 2.0/n) <= compute_c5(h0, 2.0/n) + 0.01:
                return hp

        pass
    return h0.copy()
