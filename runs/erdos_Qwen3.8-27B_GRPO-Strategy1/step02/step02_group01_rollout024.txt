import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h = np.clip(h + (diff / total) * room, 0, 1)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h = np.clip(h + (diff / total) * room, 0, 1)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e10:
        return m * dx
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def eq_con(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=300):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def erdos_ineq(x):
    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr

def slsqp_erdos(h, maxiter=400):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.05
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': erdos_eq},
        {'type': 'ineq', 'fun': erdos_ineq}
    ]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        h_opt = np.clip(res.x[:n], 0, 1)
        return project_feasible(h_opt, n)
    except Exception:

def binary_swap_search(h_bin, n_iters=50000, rng=None):

    if rng is None:
        rng = np.random.RandomState(0)
    ones = list(np.where(h_bin == 1)[0])

        if elapsed() > budget_s * 0.06:

        h_try = slsqp_smooth(best_h, T, maxiter=300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Direct SLSQP (t-augmented)
    if elapsed() < budget_s * 0.12:
        h_try = slsqp_erdos(best_h, maxiter=500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: Binary local search (swap-based with simulated annealing)
    n_bin = best_n
    h_bin = binary_from_frac(best_h, n_bin)
    for trial in range(3):
        if elapsed() > budget_s * 0.50:
            break
        h_b0 = h_bin.copy()
        if trial > 0:
            # Perturb: flip 3-8 random pairs
            flips = rng.randint(3, 9)
            ones = np.where(h_b0 == 1)[0]
            zeros = np.where(h_b0 == 0)[0]
            for _ in range(flips):
                if len(ones) > 0 and len(zeros) > 0:
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    h_b0[a], h_b0[b] = 1 - h_b0[a], 1 - h_b0[b]
                    ones = np.where(h_b0 == 1)[0]
                    zeros = np.where(h_b0 == 0)[0]
        iters = 40000 if trial == 0 else 25000
        h_b_opt, c5_b = binary_swap_search(h_b0, n_iters=iters, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b_opt.astype(float).copy()
            h_bin = h_b_opt.copy()

    # Phase 4: Fractional refinement of best binary
    if elapsed() < budget_s * 0.55:
        h_frac, c5_frac = frac_refine(h_bin, best_n, maxiter=400)
        if c5_frac < best_c5:
            best_c5 = c5_frac
            best_h = h_frac

    # Phase 5: Multi-resolution exploration
    resolutions = [best_n, 72, 80, 84, 96, 108]
    for n_try in resolutions:
        if elapsed() > budget_s * 0.72:
            break
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_base = np.interp(x_new, x_old, best_h)
            h_base = project_feasible(h_base, n_try)
        # Binary search at this resolution
        h_b2 = binary_from_frac(h_base, n_try)
        h_b_opt2, c5_b2 = binary_swap_search(h_b2, n_iters=15000, rng=rng)
        if c5_b2 * (best_n / float(n_try)) < best_c5:
            # Convert back: c5 at this res
            pass
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b_opt2.astype(float).copy()
            best_n = n_try
        # Fractional
        if elapsed() < budget_s * 0.72:
            h_f2, c5_f2 = frac_refine(h_b_opt2, n_try, maxiter=300)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2
                best_n = n_try

    # Phase 6: Coordinate descent on full solution
    if elapsed() < budget_s * 0.85:
        h_cd, c5_cd = coord_descent(best_h, best_n, max_sweeps=50)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 7: Rotations + re-optimize
    if elapsed() < budget_s * 0.92:
        for shift in [1, 2, 3, 5, 7, 11]:
            if elapsed() > budget_s * 0.92:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_opt = slsqp_erdos(h_rot, maxiter=300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 8: Final intensive SLSQP
    if elapsed() < budget_s * 0.95:
        h_fin = slsqp_smooth(best_h, 50000, maxiter=500)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
        h_fin2 = slsqp_erdos(best_h, maxiter=600)
        c5_fin2 = compute_c5(h_fin2)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
