        starts.append(hh.copy())
    x = np.arange(n) / n
    for m, amp in [(1, 0.35), (2, 0.28), (3, 0.22), (5, 0.15)]:
        starts.append(np.clip(0.5 + amp * np.sin(2 * np.pi * m * x + 0.3), 0, 1))
    starts.append(np.clip(0.5 + 0.3 * np.cos(np.pi * x) + 0.15 * np.sin(3 * np.pi * x), 0, 1))
    for _ in range(2):
        starts.append(rng.uniform(0, 1, n))
    p = max(2, n // 4)
    starts.append(np.array([1.0 if (i % (2 * p)) < p else 0.0 for i in range(n)]))
    if ref is not None:
        if len(ref) == n:
            starts.insert(0, ref.copy())
        else:
            hi = np.interp(np.linspace(0, 1, n, endpoint=False),
                          np.linspace(0, 1, len(ref), endpoint=False), ref)
            starts.insert(0, np.clip(hi, 0, 1))
    return [project(s, ts) for s in starts]

def optimize_one(h0, dx, ts, deadline, rng):
    h = slsqp_solve(h0, dx, ts, 150)
    c = compute_c5(h, dx)
    if time.time() < deadline - 25:
        h2, c2 = grad_descent_momentum(h, dx, ts, 3000, deadline - 10, rng)
        if c2 < c:
            h, c = h2, c2
    if time.time() < deadline - 15:
        h3, c3 = coord_descent(h, dx, ts, 80, deadline - 8, rng)
        if c3 < c:
            h, c = h3, c3
    if time.time() < deadline - 8:
        h4 = slsqp_solve(h, dx, ts, 120)
        c4 = compute_c5(h4, dx)
        if c4 < c:
            h, c = h4, c4
    if time.time() < deadline - 4:
        h5, c5 = grad_descent_momentum(h, dx, ts, 2000, deadline - 2, rng)
        if c5 < c:
            h, c = h5, c5
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(50, 0.5)
    n0 = len(h_init)
    dx0 = 2.0 / n0
    ts0 = n0 / 2.0
    h_init = project(h_init, ts0)
    bh, bc = optimize_one(h_init, dx0, ts0, deadline - 80, rng)

    for dn in (-2, 2, -4, 4, -1, 1, -3, 3, 5, -5, 6, -6, 8, -8):
        nt = n0 + dn
        if nt < 24 or nt > 72 or time.time() > deadline - 50:
            continue
        dxt = 2.0 / nt
        tst = nt / 2.0
        hi = np.interp(np.linspace(0, 1, nt, endpoint=False),
                       np.linspace(0, 1, len(bh), endpoint=False), bh)
        hi = project(np.clip(hi, 0, 1), tst)
        h_r, c_r = optimize_one(hi, dxt, tst, deadline - 25, rng)
        if c_r < bc:
            bh, bc = h_r, c_r

    if time.time() < deadline - 50:
        nb, dxb, tsb = len(bh), 2.0 / len(bh), len(bh) / 2.0
        for amp in (0.12, 0.06, 0.03, 0.015):
            if time.time() > deadline - 25:
                break
            hp = project(bh + amp * rng.randn(nb), tsb)
            h_r, c_r = optimize_one(hp, dxb, tsb, deadline - 12, rng)
            if c_r < bc:
                bh, bc = h_r, c_r

    if time.time() < deadline - 50:
        nb, dxb, tsb = len(bh), 2.0 / len(bh), len(bh) / 2.0
        starts = make_starts(nb, tsb, rng, bh)
        for hs in starts[1:5]:
            if time.time() > deadline - 20:
                break
            h_r, c_r = optimize_one(hs, dxb, tsb, deadline - 10, rng)
            if c_r < bc:
                bh, bc = h_r, c_r

    if bh is not None:
        nb, dxb, tsb = len(bh), 2.0 / len(bh), len(bh) / 2.0
        hf = slsqp_solve(bh, dxb, tsb, 200)
        cf = compute_c5(hf, dxb)
        if cf < bc:
            bh, bc = hf, cf
        if time.time() < deadline - 3:
            hd, cd = grad_descent_momentum(bh, dxb, tsb, 3000, deadline - 1, rng)
            if cd < bc:
                bh, bc = hd, cd
        hd2, cd2 = coord_descent(bh, dxb, tsb, 30, deadline - 0.5, rng)
        if cd2 < bc:
            bh, bc = hd2, cd2

    bh = project(bh, len(bh) / 2.0)
    bc = compute_c5(bh, 2.0 / len(bh))
    return bh, bc, len(bh)
