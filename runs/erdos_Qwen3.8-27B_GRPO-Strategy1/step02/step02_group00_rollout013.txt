def project_box_sum(h, n):

    target = n / 2.0

        if s > target:

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_box_sum(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    out = np.clip(h - tau, 0.0, 1.0)
    # Fix tiny residual
    diff = target - out.sum()
    if abs(diff) > 1e-12:
        mask = (out > 1e-10) & (out < 1.0 - 1e-10)
        if mask.any():
            out[mask] += diff / mask.sum()
            out = np.clip(out, 0.0, 1.0)
    return out

def smooth_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def smooth_obj_grad(h, T):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    nz = np.where(w > 1e-14)[0]
    for k in nz:
        wk = w[k]
        a1 = max(0, n - 1 - k)
        b1 = min(n - 1, 2 * n - 2 - k)
        if b1 >= a1:
            grad[a1:b1 + 1] += wk * (1.0 - h[a1 + k - n + 1:b1 + k - n + 2])
        a2 = max(0, k - n + 1)
        b2 = min(n - 1, k)
        if b2 >= a2:
            grad[a2:b2 + 1] -= wk * h[a2 - k + n - 1:b2 - k + n]
    return grad

def slsqp_opt(h0, T, n, maxiter=400, ftol=1e-13):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h0, args=(T,), jac=smooth_obj_grad,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_box_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_box_sum(h0, n)

def proj_subgrad(h, n, n_iter=400, top_k=5, lr0=0.04):
    best_h, best_v = h.copy(), compute_c5(h)
    h = h.copy()
    for it in range(n_iter):
        lr = lr0 / (1.0 + 0.08 * it)
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in idx:
            a1 = max(0, n - 1 - k); b1 = min(n - 1, 2 * n - 2 - k)
            if b1 >= a1:
                grad[a1:b1+1] += 1.0 - h[a1+k-n+1:b1+k-n+2]
            a2 = max(0, k - n + 1); b2 = min(n - 1, k)
            if b2 >= a2:
                grad[a2:b2+1] -= h[a2-k+n-1:b2-k+n]
        h = project_box_sum(h - lr * grad, n)
        v = compute_c5(h)
        if v < best_v:
            best_v = v
            best_h = h.copy()
    return best_h, best_v

def coord_descent(h, n, passes=4, n_trials=7):
    h = h.copy()
    best_v = compute_c5(h)
    trial_vals = np.linspace(0.0, 1.0, n_trials)
    for _ in range(passes):
        perm = np.random.permutation(n)
        for i in perm:
            old = h[i]
            for tv in trial_vals:
                h[i] = tv
                hp = project_box_sum(h, n)
                v = compute_c5(hp)
                if v < best_v - 1e-13:
                    best_v = v
                    h = hp.copy()
                h[i] = old
    return h, best_v

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def el():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60); best_h[:30] = 1.0
    n = len(best_h)
    best_h = project_box_sum(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP gradient polish
    for T in [40, 150, 500, 1500, 5000]:
        if el() > budget_s * 0.12:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=300, ftol=1e-13)
        v = compute_c5(hc)
        if v < best_c5:
            best_c5, best_h = v, hc

    # Phase 2: Projected subgradient (multi top_k)
    if el() < budget_s * 0.30:
        for tk in [3, 5, 8, 12]:
            if el() > budget_s * 0.30:
                break
            hg, vg = proj_subgrad(best_h, n, n_iter=350, top_k=tk, lr0=0.03)
            if vg < best_c5:
                best_c5, best_h = vg, hg
        hc = slsqp_opt(best_h, 3000, n, maxiter=200, ftol=1e-13)
        v = compute_c5(hc)
        if v < best_c5:
            best_c5, best_h = v, hc

    # Phase 3: Coordinate descent
    if el() < budget_s * 0.48:
        hc, vg = coord_descent(best_h, n, passes=5, n_trials=9)
        if vg < best_c5:
            best_c5, best_h = vg, hc
        hc = slsqp_opt(best_h, 5000, n, maxiter=300, ftol=1e-14)
        v = compute_c5(hc)
        if v < best_c5:
            best_c5, best_h = v, hc

    # Phase 4: Diverse basin hopping
    if el() < budget_s * 0.72:
        bh_h, bh_v = best_h.copy(), best_c5
        n_iter_bh = 50
        for it in range(n_iter_bh):
            if el() > budget_s * 0.72:
                break
            frac = it / n_iter_bh
            step = 0.10 * (1.0 - 0.6 * frac) + 0.02
            rtype = np.random.rand()
            if rtype < 0.5:
                h_p = bh_h + step * np.random.randn(n)
            elif rtype < 0.75:
                h_p = bh_h.copy()
                i, j = np.random.choice(n, 2, replace=False)
                h_p[i], h_p[j] = bh_h[j], bh_h[i]
            else:
                h_p = bh_h.copy()
                mask = np.random.rand(n) < 0.15
                h_p[mask] = np.random.rand(mask.sum())
            h_p = project_box_sum(h_p, n)
            hc = slsqp_opt(h_p, 400, n, maxiter=200, ftol=1e-12)
            v = compute_c5(hc)
            temp = 0.004 * (1.0 - frac) + 0.0003
            if v < bh_v or np.random.rand() < np.exp(-(v - bh_v) / temp):
                bh_v, bh_h = v, hc
                if v < best_c5:
                    best_c5, best_h = v, hc

    # Phase 5: Multi-resolution (downsample + re-optimize, upsample)
    for n_try in [80, 100, 60]:
        if el() > budget_s * 0.87:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = project_box_sum(hi, n_try)
        for T in [200, 1000, 4000]:
            if el() > budget_s * 0.87:
                break
            hc = slsqp_opt(hi, T, n_try, maxiter=400, ftol=1e-13)
            v = compute_c5(hc)
            if v < best_c5:
                best_c5, best_h, n = v, hc, n_try
        if el() < budget_s * 0.87:
            hg, vg = proj_subgrad(hi, n_try, n_iter=200, top_k=5, lr0=0.03)
            if vg < best_c5:
                best_c5, best_h, n = vg, hg, n_try
            hc = slsqp_opt(hg, 5000, n_try, maxiter=300, ftol=1e-14)
            v = compute_c5(hc)
            if v < best_c5:
                best_c5, best_h, n = v, hc, n_try

    # Phase 6: Final SLSQP + subgradient polish
    for T in [8000, 25000]:
        if el() > budget_s * 0.97:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=500, ftol=1e-15)
        v = compute_c5(hc)
        if v < best_c5:
            best_c5, best_h = v, hc
    if el() < budget_s * 0.99:
        hg, vg = proj_subgrad(best_h, n, n_iter=200, top_k=3, lr0=0.015)
        if vg < best_c5:
            best_c5, best_h = vg, hg

    best_h = project_box_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
