def soft_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')  # length 2n-1
    # Softmax weights for the max
    Tc = T * corr
    Tc_max = np.max(Tc)
    exp_Tc = np.exp(Tc - Tc_max)
    w = exp_Tc / np.sum(exp_Tc)
    
    val = (np.max(corr) + np.log(np.sum(exp_Tc)) / T) * dx
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        idx = k + n - 1
        wk = w[idx]
        if wk < 1e-18:
            continue
        if k >= 0:
            # First term: i in [k, n-1], add wk*(1-h[i-k])
            i_s, i_e = k, n
            grad[i_s:i_e] += wk * (1.0 - h[0:i_e-k])
            # Second term: i in [0, n-1-k], subtract wk*h[i+k]
            i_s2, i_e2 = 0, n - k
            grad[i_s2:i_e2] -= wk * h[k:n]
        else:

            grad[i_s:i_e] += wk * (1.0 - h[l:n])

            i_s2, i_e2 = l, n
            grad[i_s2:i_e2] -= wk * h[0:n-l]
    
    grad *= dx
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

_T_g = 100.0
_mu_g = 100.0
_n_g = 120

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def soft_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    Tc = T * corr - T * np.max(corr)
    expT = np.exp(Tc)
    w = expT / np.sum(expT)
    val = (np.max(corr) + np.log(np.sum(expT)) / T) * dx
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        wk = w[idx]
        if wk < 1e-18:
            continue
        if k >= 0:
            grad[k:n] += wk * (1.0 - h[0:n - k])
            grad[0:n - k] -= wk * h[k:n]
        else:
            l = -k
            grad[0:n - l] += wk * (1.0 - h[l:n])
            grad[l:n] -= wk * h[0:n - l]
    return val, grad * dx

def _obj(h):
    return soft_c5(h, _T_g)

def _grad(h):
    _, g = soft_c5_grad(h, _T_g)
    return g

def _con_fun(h):
    return h.sum() - len(h) / 2.0

def _con_jac(h):
    return np.ones(len(h))

def _pen_obj(h):
    return soft_c5(h, _T_g) + _mu_g * (h.sum() - _n_g / 2.0) ** 2

def _pen_grad(h):
    _, g = soft_c5_grad(h, _T_g)
    g = g + 2.0 * _mu_g * (h.sum() - _n_g / 2.0)
    return g

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return h
        h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=800):
    global _T_g
    _T_g = T
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj, h, jac=_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(np.clip(res.x, 0, 1), n)

def lbfgs_opt(h, T, n, mu=100.0, maxiter=5000):
    global _T_g, _mu_g, _n_g
    _T_g = T
    _mu_g = mu
    _n_g = n
    res = minimize(_pen_obj, h, jac=_pen_grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return project_feasible(np.clip(res.x, 0, 1), n)

def t_schedule_opt(h, T_list, n, maxiter=600):
    for T in T_list:
        hc = slsqp_opt(h, T, n, maxiter=maxiter)
        h = hc
    return h
