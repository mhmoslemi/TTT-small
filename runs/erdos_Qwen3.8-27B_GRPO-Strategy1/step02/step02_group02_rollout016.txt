        if stuck > STUCK_RESTART:
            for _ in range(n // 10):
                a = rng.randint(n)
                b = rng.randint(n)
                if h[a] != h[b]:
                    h[a], h[b] = h[b], h[a]
            max_c = int(np.max(np.correlate(h, 1.0 - h, mode='full')))
            T = max(T, T_init * 0.5)
            stuck = 0
        T = max(T_min, T * cool)
    return best_h, best_c5

def hill_climb_binary(h_start, n, seconds, rng):
    h = h_start.copy()
    dx = 2.0 / n
    max_c = int(np.max(np.correlate(h, 1.0 - h, mode='full')))
    best_c5 = max_c * dx
    best_h = h.copy()
    t0 = time.time()
    while time.time() - t0 < seconds:
        improved = False
        zero_pos = np.where(h < 0.5)[0]
        one_pos = np.where(h > 0.5)[0]
        if len(zero_pos) == 0 or len(one_pos) == 0:
            break
        for _ in range(min(5000, len(zero_pos) * len(one_pos))):
            i = zero_pos[rng.randint(len(zero_pos))]
            j = one_pos[rng.randint(len(one_pos))]
            h[i] = 1.0
            h[j] = 0.0
            nc = int(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if nc < max_c:
                max_c = nc
                improved = True
                if max_c * dx < best_c5:
                    best_c5 = max_c * dx
                    best_h = h.copy()
                zero_pos = np.where(h < 0.5)[0]
                one_pos = np.where(h > 0.5)[0]
            else:
                h[i] = 0.0
                h[j] = 1.0
        if not improved:
            break
    return best_h, best_c5

def obj_max_tight(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def continuous_refine(h_init, n, seconds, rng):
    t0 = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
            'jac': lambda x: np.ones((1, n))}
    for t_val in [50, 200, 800, 3000, 10000]:
        if time.time() - t0 > seconds:
            break
        try:
            r = minimize(obj_max_tight, best_h, args=(t_val,), method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': 2000, 'ftol': 1e-15})
            hc = proj(np.clip(r.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t_start)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(280, 0.5) + rng.randn(280) * 0.03

    h280 = proj(h_init, 280)
    best_c5 = compute_c5(h280)
    best_h = h280.copy()
    best_n = 280

    # Binarize for SA
    h_bin = (h280 > 0.5).astype(float)
    h_bin = proj(h_bin, 280)
    bin_c5 = compute_c5(h_bin)
    if bin_c5 < best_c5:
        best_c5 = bin_c5
        best_h = h_bin.copy()

    # Phase 1: SA at n=280 (main exploration)
    sa_time = min(400, max(50, tleft() * 0.45))
    if tleft() > 60:
        h_sa, c5_sa = sa_binary(h_bin, 280, sa_time, rng, T_init=8.0, T_min=0.02, cool=0.9997)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 2: Hill climb to polish
    hc_time = min(150, max(20, tleft() * 0.2))
    if tleft() > 30:
        h_hc, c5_hc = hill_climb_binary(best_h, 280, hc_time, rng)
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc.copy()

    # Phase 3: Try n=420
    if tleft() > 80:
        h420 = proj(interp_to(best_h, 420), 420)
        h420_bin = (h420 > 0.5).astype(float)
        h420_bin = proj(h420_bin, 420)
        c420 = compute_c5(h420_bin)
        if c420 < best_c5:
            best_c5 = c420
            best_h = h420_bin.copy()
            best_n = 420
        if tleft() > 40:
            h420_sa, c5_420 = sa_binary(h420_bin, 420, min(60, tleft() * 0.4), rng, T_init=10.0, T_min=0.02, cool=0.9998)
            if c5_420 < best_c5:
                best_c5 = c5_420
                best_h = h420_sa.copy()
                best_n = 420

    # Phase 4: Try n=140 (coarser, different optimum)
    if tleft() > 60:
        h140 = proj(interp_to(best_h, 140), 140)
        h140_bin = (h140 > 0.5).astype(float)
        h140_bin = proj(h140_bin, 140)
        c140 = compute_c5(h140_bin)
        if c140 < best_c5:
            best_c5 = c140
            best_h = h140_bin.copy()
            best_n = 140
        if tleft() > 30:
            h140_sa, c5_140 = sa_binary(h140_bin, 140, min(40, tleft() * 0.4), rng, T_init=5.0, T_min=0.02, cool=0.9999)
            if c5_140 < best_c5:
                best_c5 = c5_140
                best_h = h140_sa.copy()
                best_n = 140

    # Phase 5: Continuous refinement at best_n
    if tleft() > 30:
        cref_time = min(120, tleft() * 0.35)
        h_ref, c5_ref = continuous_refine(best_h, best_n, cref_time, rng)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 6: Final hill climb
    if tleft() > 15:
        h_fin, c5_fin = hill_climb_binary(best_h, best_n, min(60, tleft() * 0.4), rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final check
    best_h = proj(np.clip(best_h, 0, 1), best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_n
