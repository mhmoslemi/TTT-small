        if ct < cc or rng.rand() < np.exp(-(ct - cc) / Tc):
            hc, cc = ht, ct
            if ct < cb:
                cb, hb = ct, hc.copy()
        Tc *= 0.996
    return hb, cb

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = proj(h, n)
    hbest, nbest = h, n
    cbest = compute_c5(hbest)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B quick pass (faster per-iter than SLSQP)
    if rem() > 180:
        for T in [0.02, 0.008, 0.003, 0.001]:
            if rem() < 50:
                break
            ho = lbfgs_run(hbest.copy(), nbest, T, 1500)
            c = compute_c5(ho)
            if c < cbest:
                hbest, cbest = ho, c

    # Phase 2: SLSQP with top-k focus
    if rem() > 150:
        corr = corr_full(hbest)
        cmax = np.max(corr)
        thr = cmax * 0.98 + 1e-8
        nk = min(int(np.sum(corr >= thr)), 25)
        for T in [0.006, 0.002, 0.0008]:
            if rem() < 50:
                break
            ho = slsqp_run(hbest.copy(), nbest, T, 900, nk)
            c = compute_c5(ho)
            if c < cbest:
                hbest, cbest = ho, c

    # Phase 3: DE global exploration
    if rem() > 180:
        hd, cd = de_search(hbest, nbest, 28, 0.025, 0.003, 35, seed + 7)
        if cd < cbest:
            hbest, cbest = hd, cd
        if rem() > 80:
            hbest, cbest = local_polish(hbest, nbest, 0.004, 700, 12)

    # Phase 4: Multi-resolution SLSQP
    for nt in [256, 448, 336, 280]:
        if rem() < 90:
            break
        if nt == nbest:
            hc = hbest.copy()
        else:
            hc = upsample(hbest, nt)
        for T in [0.008, 0.002]:
            if rem() < 45:
                break
            ho = slsqp_run(hc, nt, T, 600, 0)
            c = compute_c5(ho)
            if c < cbest:
                hbest, cbest, nbest = ho, c, nt

    # Phase 5: Hinge-objective optimization
    if rem() > 80:
        corr = corr_full(hbest)
        thr = np.percentile(corr, 95)
        def hinge_wrap(u, T2, th2):
            return hinge_obj(u, T2, th2)
        cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (nbest,)}]
        res = opt.minimize(hinge_wrap, hbest.copy(), args=(0.01, thr),
                           method='SLSQP', bounds=[(0.0, 1.0)] * nbest,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-14, 'iprint': 0})
        ho = proj(res.x, nbest)
        c = compute_c5(ho)
        if c < cbest:
            hbest, cbest = ho, c
        hbest, cbest = local_polish(hbest, nbest, 0.003, 600, 10)

    # Phase 6: Multi-start perturbations + polish
    if rem() > 120:
        for i in range(8):
            if rem() < 60:
                break
            rng = np.random.RandomState(seed + i * 104729 + 13)
            sg = 0.008 + 0.006 * (i % 3)
            hp = proj(np.clip(hbest + rng.randn(nbest) * sg, 0, 1), nbest)
            ho = slsqp_run(hp, nbest, 0.003, 500, 10)
            c = compute_c5(ho)
            if c < cbest:
                hbest, cbest = ho, c
        if rem() > 60:
            hbest, cbest = local_polish(hbest, nbest, 0.002, 700, 10)

    # Phase 7: Simulated annealing refinement
    if rem() > 80:
        hsa, csa = block_sa(hbest, nbest, t0, budget_s, 250, seed + 333)
        if csa < cbest:
            hbest, cbest = hsa, csa
        hbest, cbest = local_polish(hbest, nbest, 0.001, 800, 8)

    # Phase 8: Ultra-fine final polish
    if rem() > 50:
        for T in [0.0005, 0.0002, 0.0001]:
            if rem() < 25:
                break
            ho = slsqp_run(hbest.copy(), nbest, T, 2000, 8)
            c = compute_c5(ho)
            if c < cbest:
                hbest, cbest = ho, c

    # Final L-BFGS-B squeeze
    if rem() > 25:
        ho = lbfgs_run(hbest.copy(), nbest, 0.0001, 4000, 8)
        c = compute_c5(ho)
        if c < cbest:
            hbest, cbest = ho, c

    hbest = proj(hbest, nbest)
    cfinal = compute_c5(hbest)
    return (hbest, cfinal, nbest)
