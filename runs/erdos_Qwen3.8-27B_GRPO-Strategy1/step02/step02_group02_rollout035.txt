from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def proj_binary(h, n):
    """Project to binary with exactly n/2 ones"""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_needed = n // 2
    if n_ones == n_needed:
        return h_bin
    if n_ones > n_needed:
        # Need to turn off some 1s
        idx_ones = np.where(h_bin == 1)[0]
        # Turn off the ones with lowest h values
        order = np.argsort(h[idx_ones])
        turn_off = n_ones - n_needed
        h_bin[idx_ones[order[:turn_off]]] = 0
    else:
        # Need to turn on some 0s
        idx_zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx_zeros])
        turn_on = n_needed - n_ones
        h_bin[idx_zeros[order[:turn_on]]] = 1
    return h_bin

def binary_local_search(h, n, max_iter=200, rng=None):
    """Greedy pairwise swap local search on binary vector"""
    h = h.copy()
    dx = 2.0 / n
    best_c5, corr = compute_c5(h)
    
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    
    for it in range(max_iter):
        best_delta = 0
        best_pair = None
        ones_list = list(ones)
        zeros_list = list(zeros)
        
        for i in zeros_list:
            for j in ones_list:
                # Try swapping: h[i] 0->1, h[j] 1->0
                h_test = h.copy()
                h_test[i] = 1
                h_test[j] = 0
                c5_test = float(np.max(np.correlate(h_test, 1.0-h_test, mode='full')) * dx)
                delta = c5_test - best_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_pair = (i, j)
        
        # Also try reverse swaps
        for i in ones_list:
            for j in zeros_list:
                h_test = h.copy()
                h_test[i] = 0
                h_test[j] = 1
                c5_test = float(np.max(np.correlate(h_test, 1.0-h_test, mode='full')) * dx)
                delta = c5_test - best_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        best_c5 += best_delta
        # Update sets

            ones.add(i)
        else:

            zeros.add(i)
        if h[j] == 1:
            zeros.discard(j)
            ones.add(j)
        else:
            ones.discard(j)
            zeros.add(j)
    
    return h, best_c5

    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:

        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)

def proj_binary(h, n, rng=None):

        if rng is not None:

        else:

        turn_on = n_needed - n_ones
        h_bin[idx_zeros[:turn_on]] = 1
    return h_bin

    h = h.copy()
    c5 = compute_c5(h)
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        improved = False
        best_delta = 1e-15
        best_swaps = []
        
        # Try all zero->one and one->zero swaps
        for i in zeros:

        ones.remove(j)
        zeros.append(j)
        ones.append(i)
        improved = True

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_needed = n // 2
    if n_ones == n_needed:
        return h_bin
    if n_ones > n_needed:
        idx_ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx_ones])
        idx_ones = idx_ones[order]
        h_bin[idx_ones[:n_ones - n_needed]] = 0
    else:
        idx_zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx_zeros])
        idx_zeros = idx_zeros[order]
        h_bin[idx_zeros[:n_needed - n_ones]] = 1
    return h_bin

def compute_F(h, n):
    v = 1.0 - h
    num_lags = 2 * n - 1
    F = np.zeros((n, num_lags))
    lags = np.arange(num_lags) - (n - 1)
    p = np.arange(n)
    for k in range(num_lags):
        d = lags[k]
        idx1 = p - d
        m1 = (idx1 >= 0) & (idx1 < n)
        F[m1, k] += v[idx1[m1]]
        idx2 = p + d
        m2 = (idx2 >= 0) & (idx2 < n)
        F[m2, k] -= h[idx2[m2]]
    return F

Let me think about what the optimal solution looks like. We want to minimize max_k ∫h(x)(1-h(x+k))dx. This is like saying: for the worst shift k, we want h and its shifted complement to have minimal overlap.

If h is the indicator of a set S (binary), then h(x)(1-h(x+k)) = 1 if x ∈ S and x+k ∉ S. So the integral is the measure of S \ (S-k), i.e., the part of S that, when shifted by -k, doesn't overlap with S.

Alternatively: ∫h(x)(1-h(x+k))dx = |S| - |S ∩ (S-k)| = 1 - |S ∩ (S-k)| (since |S|=1).

So we want to maximize min_k |S ∩ (S-k)|, which is the minimum autocorrelation of the set S.

This is exactly the problem of finding a subset of [0,2] with maximum minimum autocorrelation! This is related to difference sets and Golay sequences.

For the binary case with n points, we have n/2 ones. The autocorrelation at lag d is the number of pairs (i, i+d) both in S. We want to minimize the maximum over d of (n/2 - autocorr[d]) * dx, which is equivalent to maximizing the minimum autocorr[d].

This is a well-studied problem in combinatorics! The best known constructions are related to:
- Welch/BCH bounds
- Cyclic difference sets
- Golay/Rice sequences

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_needed = n // 2
    if n_ones == n_needed:
        return h_bin
    if n_ones > n_needed:
        idx = np.where(h_bin == 1)[0]
        ord_ = np.argsort(h[idx])
        h_bin[idx[ord_[:n_ones - n_needed]]] = 0
    else:
        idx = np.where(h_bin == 0)[0]
        ord_ = np.argsort(-h[idx])
        h_bin[idx[ord_[:n_needed - n_ones]]] = 1
    return h_bin

def proj_cont(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_F(h, n):
    v = 1.0 - h
    nl = 2 * n - 1
    F = np.zeros((n, nl))
    lags = np.arange(nl) - (n - 1)
    p = np.arange(n)
    for k in range(nl):
        d = lags[k]
        i1 = p - d
        m1 = (i1 >= 0) & (i1 < n)
        F[m1, k] += v[i1[m1]]
        i2 = p + d
        m2 = (i2 >= 0) & (i2 < n)
        F[m2, k] -= h[i2[m2]]
    return F

def binary_greedy(h, n, max_iters=300):
    dx = 2.0 / n
    h = h.copy()
    c5 = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        F = compute_F(h, n)
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        D = corr[np.newaxis, :] + F[zeros, :]
        Fo = F[ones, :]
        scores = np.max(D[:, np.newaxis, :] - Fo[np.newaxis, :, :], axis=2)
        flat_idx = int(np.argmin(scores))
        if scores.ravel()[flat_idx] >= c5 / dx - 1e-12:
            break
        i, j = np.unravel_index(flat_idx, scores.shape)
        zi, oj = zeros[i], ones[j]
        h[zi] = 1.0
        h[oj] = 0.0
        new_c5 = compute_c5(h)
        if new_c5 < c5 - 1e-14:
            c5 = new_c5
        else:
            h[zi] = 0.0
            h[oj] = 1.0
            break
    return h, c5

def binary_sa(h, n, rng, n_iters=500, t_init=0.002, t_final=1e-6):
    dx = 2.0 / n
    h = h.copy()
    c5 = compute_c5(h)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = list(np.where(h == 0)[0])
    cur = 0
    for it in range(n_iters):
        t = t_init * (t_final / t_init) ** (it / max(1, n_iters - 1))
        zi = zeros[rng.randint(len(zeros))]
        oj = list(ones)[rng.randint(len(ones))]
        h[zi] = 1.0
        h[oj] = 0.0
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta < 0 or rng.rand() < np.exp(-delta / t):
            c5 = new_c5
            ones.add(zi)
            ones.discard(oj)
            zeros.remove(zi)
            zeros.append(oj)
        else:
            h[zi] = 0.0
            h[oj] = 1.0
            cur += 1
    return h, c5

def smooth_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s = np.clip(s, -500, 500)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def continuous_refine(h, n, t_val, maxiter=5000):
    bnds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones((1, n))},)
    r = minimize(smooth_obj, h, args=(t_val,), method='L-BFGS-B',
                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    return proj_cont(r.x, n)

def interp_rescale(h, n_from, n_to):
    xs = (np.arange(n_from) + 0.5) * 2.0 / n_from
    xd = (np.arange(n_to) + 0.5) * 2.0 / n_to
    return proj_cont(np.interp(xd, xs, h), n_to)
