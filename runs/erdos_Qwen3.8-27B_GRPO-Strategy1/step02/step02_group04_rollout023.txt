    """Projected subgradient step for the active shifts"""
    ov = compute_overlaps(h)

    # corr[m] for m in 0..2n-2, shift s = m-(n-1)
    # d corr[m] / d h[j] = (1-h[j+s]) if j is from first arg, -h[j-s] if from second

    h_new = h - alpha * grad

            active_idx2 = get_top_idx(h_best, 8)
            h_opt = slsqp(h_best, n_best, obj_active, (0.0005, active_idx2), 2500)
            try_update(h_opt, n_best)

    # Phase 4: Fourier-structured multi-start
    for nm in [2, 3, 5, 8, 12]:
        for sc in [0.005, 0.01, 0.02, 0.035, 0.06]:
            if budget_s - elapsed() < 180:
                break
            hp = fourier_perturb(h_best, n_best, nm, sc, rng)
            h_opt = slsqp(hp, n_best, obj_full, (0.005,), 2000)
            active = get_top_idx(h_opt, 6)
            h_opt = slsqp(h_opt, n_best, obj_active, (0.001, active), 2000)
            try_update(h_opt, n_best)

    # Phase 5: Multi-resolution exploration
    for n_try in [280, 360, 420, 500, 600]:
        if budget_s - elapsed() < 180:
            break
        if n_try != n_best:
            h_init = interp_h(h_best, n_try)
        else:
            h_init = h_best.copy()
        for T_ in [0.01, 0.003, 0.001, 0.0003]:
            if budget_s - elapsed() < 90:
                break
            h_opt = slsqp(h_init, n_try, obj_full, (T_,), 3000)
            if try_update(h_opt, n_try):
                h_init = interp_h(h_best, n_try)
            else:
                h_init = h_opt

    # Phase 6: Bundle-style active set refinement
    for it in range(20):
        if budget_s - elapsed() < 50:
            break
        ov = compute_overlaps(h_best)
        c_max = np.max(ov)
        threshold = c_max - 0.002
        active = np.where(ov >= threshold)[0]
        if len(active) == 0:
            active = get_top_idx(h_best, 5)
        T_ = 0.0004
        h_opt = slsqp(h_best, n_best, obj_active, (T_, active), 4000)
        if not try_update(h_opt, n_best):
            # Try subset: only exact max
            active2 = get_top_idx(h_best, 3)
            h_opt = slsqp(h_best, n_best, obj_active, (0.0002, active2), 3000)
            try_update(h_opt, n_best)

    # Phase 7: Final polish with very small T
    for T in [0.0002, 0.0001, 0.00005, 0.00003, 0.00002]:
        if budget_s - elapsed() < 35:
            break
        active = get_top_idx(h_best, 5)
        h_opt = slsqp(h_best, n_best, obj_active, (T, active), 5000)
        try_update(h_opt, n_best)

    # Phase 8: Micro-perturbation escape
    if budget_s - elapsed() > 60:
        for trial in range(5):
            if budget_s - elapsed() < 30:
                break
            hm = h_best + rng.randn(n_best) * 0.004
            hm = fix_sum(hm, n_best)
            for T in [0.001, 0.0003, 0.0001]:
                if budget_s - elapsed() < 20:
                    break
                h_opt = slsqp(hm, n_best, obj_full, (T,), 3000)
                try_update(h_opt, n_best)
                hm = h_best.copy()

    h_best = fix_sum(np.clip(h_best, 0.0, 1.0), len(h_best))
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
