                deltas[qq, :] = 0.0; deltas[qq, qq:qq+N] = s[qq]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_improve = 0
            T = max(T, 0.02)
    
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h_in, N, deadline, topK=8, lr0=0.01, beta0=0.8):
    h = h_in.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    lr = lr0
    beta = beta0
    m1 = 0.001; m2 = 0.999  # Adam
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:

        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Top-K subgradient
        top = np.argpartition(corr, -topK)[-topK:]
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= topK
        
        # Adam step
        it += 1
        m1_ = m1**it if it < 1000 else 1.0  # skip bias correction for simplicity
        h_new = project_sum(y - lr * grad / 0.9, target)
        m = beta * (h_new - h)
        h = h_new
        
        if it % 50 == 0:
            lr *= 0.97
            beta = min(0.95, beta0 + it * 0.0002)
    
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def build_deltas(s, N):
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s[i]
    return deltas

                contrib[L:] += 1
            else:

    v = np.zeros(N)
    beta1, beta2 = 0.85, 0.999
    eps = 1e-8
    lr = lr0
    it = 0

        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        top = np.argpartition(corr, -topK)[-topK:]
        grad = np.zeros(N)
        for t in top:

                grad += 1.0 - 2.0 * y

            else:

                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= topK
        it += 1
        b1c = 1 - beta1**it
        b2c = 1 - beta2**it
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        mhat = m / b1c

        h = h_new
