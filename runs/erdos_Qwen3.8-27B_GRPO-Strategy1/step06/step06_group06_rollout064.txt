def continuous_lse(h, N, deadline, beta_max=50, K=30):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = 0.003
    center = N - 1
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    
    while time.time() < deadline - 3:
        beta = min(beta_max, 5 + t * 0.01)
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Use top-K lags with softmax weights
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
        c_shifted = c_top - c_top.max()
        w = np.exp(beta * c_shifted)
        w /= w.sum()
        
        grad = np.zeros(N)
        for i, ai in enumerate(top_idx):
            wi = w[i]
            s = int(ai - center)
            if s > 0:
                grad[s:] += wi * (1.0 - h[:N-s])
                grad[:N-s] -= wi * h[s:]
            elif s < 0:
                L2 = -s
                grad[:N-L2] += wi * (1.0 - h[L2:])
                grad[L2:] -= wi * h[:N-L2]
            else:
                grad += wi * (1.0 - 2.0*h)
        
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N/2.0)
        if t % 300 == 0:
            lr *= 0.96
    
    return best_h, best_pk

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h, N):
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h==1)[0]
        sel = np.random.choice(len(idx), n1-target, replace=False)
        h[idx[sel]] = 0
    elif n1 < target:
        idx = np.where(h==0)[0]
        sel = np.random.choice(len(idx), target-n1, replace=False)
        h[idx[sel]] = 1
    return h

def find_best_swap(h, corr, K):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-K:]
    c_top = corr[top_idx]
    G = np.zeros((K, N))
    for i, ai in enumerate(top_idx):
        s = int(ai - center)
        if s > 0:
            G[i, s:N] += (1.0 - h[:N-s])
            G[i, :N-s] -= h[s:]
        elif s < 0:
            L2 = -s
            G[i, :N-L2] += (1.0 - h[L2:])
            G[i, L2:] -= h[:N-L2]
        else:
            G[i, :] = 1.0 - 2.0*h
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return -1, -1, np.inf
    A = G[:, ones]
    B = G[:, zeros]
    scores = np.max(c_top[:,None,None] - A[:,None,:] + B[:,:,None], axis=0)
    flat_idx = np.argmin(scores)
    pi, qi = np.unravel_index(flat_idx, scores.shape)
    return int(ones[pi]), int(zeros[qi]), float(scores[pi, qi])

def directed_search(h, N, deadline, K=15, max_kicks=60):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    kicks = 0
    no_improve = 0
    while time.time() < deadline - 3:
        p, q, score = find_best_swap(h, corr, K)
        if p < 0:
            break
        if score < float(corr.max()) - 1e-10:
            h[p] = 0
            h[q] = 1
            corr = fft_corr(h)
            pk = float(corr.max())
            if pk < best_pk - 1e-12:
                best_pk = pk
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            kicks += 1
            if kicks > max_kicks:
                break
            n_kick = np.random.randint(3, 10)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            ps = np.random.choice(ones, min(n_kick, len(ones)), replace=False)
            qs = np.random.choice(zeros, min(n_kick, len(zeros)), replace=False)
            h[ps] = 0
            h[qs] = 1
            corr = fft_corr(h)
            no_improve = 0
    return best_h, best_pk

def continuous_opt(h, N, deadline, K=25, beta_max=60, lr0=0.004):

def anneal(h, N, deadline, T0=0.05, cool=0.9995, max_no_improve=400):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    center = N - 1
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    T = T0
    no_imp = 0
    s_arr = np.arange(-(N-1), N)
    while time.time() < deadline - 3 and T > 0.001:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        ps = p + s_arr; pm = p - s_arr
        qs = q + s_arr; qm = q - s_arr
        vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
        vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
        d = np.zeros(2*N-1)
        pc=np.clip(ps,0,N-1); pmc=np.clip(pm,0,N-1)
        qsc=np.clip(qs,0,N-1); qmc=np.clip(qm,0,N-1)
        d[vps] -= (1.0 - h[pc[vps]])
        d[vpm] += h[pmc[vpm]]
        d[vqs] += (1.0 - h[qsc[vqs]])
        d[vqm] -= h[qmc[vqm]]
        nc = corr + d
        npk = float(nc.max())
        delta = npk - current_pk
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
            h[p] = 0; h[q] = 1
            corr = nc
            current_pk = npk
            if npk < best_pk:
                best_pk = npk
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        T *= cool
        if no_imp > max_no_improve:
            h = best_h.copy()
            corr = fft_corr(h)
            current_pk = best_pk
            no_imp = 0
            T = max(T * 0.6, 0.008)
    return best_h, best_pk
