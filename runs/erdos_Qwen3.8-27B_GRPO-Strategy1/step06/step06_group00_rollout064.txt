def bin_search(h_in, dl, t0=0.3, max_it=8000):
    N = len(h_in)
    h = (h_in > 0.5).astype(float).copy()
    nh = N // 2
    diff = int(h.sum()) - nh
    if diff > 0:
        h[np.random.choice(np.where(h==1)[0], diff, replace=False)] = 0
    elif diff < 0:
        h[np.random.choice(np.where(h==0)[0], -diff, replace=False)] = 1
    best, bv = h.copy(), c5_of(h)
    cv = bv
    temp = t0
    stuck = 0
    for it in range(max_it):
        if time.time() > dl: break
        c = fft_corr(h)
        v = c.max() * 2.0/N
        if v < bv - 1e-10:
            bv, best, stuck = v, h.copy(), 0
        else:
            stuck += 1
        if stuck > 400:
            temp *= 0.85
            if temp < 1e-5: break
            stuck = 0
        pk = int(np.argmax(c))
        kp = pk - (N-1)
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        # Targeted candidates
        if kp != 0:
            if kp > 0:
                m = (ones + kp < N) & (h[np.minimum(ones+kp, N-1)] == 0)
            else:
                m = (ones + kp >= 0) & (h[np.maximum(ones+kp, 0)] == 0)
            cands = ones[m]
            if len(cands) < 3: cands = ones
        else:
            cands = ones
        # Try up to 6 random swaps
        ns = min(6, len(cands), len(zeros))
        ok = False
        for _ in range(ns):
            a = int(np.random.choice(cands))
            b = int(np.random.choice(zeros))
            h[a], h[b] = 0.0, 1.0
            nv = float(np.max(fft_corr(h))) * 2.0/N
            d = nv - cv
            if d <= 0 or (temp > 1e-5 and np.random.random() < np.exp(-d/(temp+1e-8))):
                cv = nv
                ok = True
                break
            h[a], h[b] = 1.0, 0.0
        if not ok and temp > 0.01:
            a, b = int(np.random.choice(ones)), int(np.random.choice(zeros))
            h[a], h[b] = 0.0, 1.0
            cv = c5_of(h)
        temp *= 0.9996
    return best, bv

    if diff > 0:

    elif diff < 0:
        h[np.random.choice(np.where(h == 0)[0], -diff, False)] = 1
    best, bv = h.copy(), c5(h)
    cv = bv
    tp = t0
    st = 0
    for it in range(mi):
        if time.time() > dl:
            break
        c = fft_corr(h)
        v = c.max() * 2.0 / N
        if v < bv - 1e-10:
            bv, best, st = v, h.copy(), 0
        else:
            st += 1
        if st > 350:
            tp *= 0.82
            if tp < 1e-5:
                break
            st = 0
        pk = int(np.argmax(c))
        kp = pk - (N - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if kp > 0:
            msk = (ones + kp < N) & (h[np.minimum(ones + kp, N - 1)] == 0)
        elif kp < 0:
            msk = (ones + kp >= 0) & (h[np.maximum(ones + kp, 0)] == 0)
        else:
            msk = np.ones(len(ones), dtype=bool)
        cd = ones[msk]
        if len(cd) < 4:
            cd = ones
        ns = min(ntry, len(cd), len(zeros))
        ok = False
        for _ in range(ns):
            a = int(np.random.choice(cd))
            b = int(np.random.choice(zeros))
            h[a], h[b] = 0.0, 1.0
            nv = float(np.max(fft_corr(h))) * 2.0 / N
            d = nv - cv
            if d <= 0 or (tp > 1e-5 and np.random.random() < np.exp(-d / (tp + 1e-8))):
                cv = nv
                ok = True
                break
            h[a], h[b] = 1.0, 0.0
        if not ok and tp > 0.008:
            a = int(np.random.choice(ones))
            b = int(np.random.choice(zeros))
            h[a], h[b] = 0.0, 1.0
            cv = c5(h)
        tp *= 0.9996
    return best, bv

def steep(h, N, ntry=15):
    """One steepest-descent swap targeting the peak."""
    c = fft_corr(h)
    cur = c.max()
    pk = int(np.argmax(c))
    kp = pk - (N - 1)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if kp > 0:
        msk = (ones + kp < N) & (h[np.minimum(ones + kp, N - 1)] == 0)
    elif kp < 0:
        msk = (ones + kp >= 0) & (h[np.maximum(ones + kp, 0)] == 0)
    else:
        msk = np.ones(len(ones), dtype=bool)
    cd = ones[msk]
    if len(cd) < 5:
        cd = ones
    best_nv = cur
    ba, bb = -1, -1
    picks = np.random.choice(cd, min(ntry, len(cd)), replace=False)
    for a in picks:
        b = int(np.random.choice(zeros))
        h[a], h[b] = 0.0, 1.0
        nv = float(np.max(fft_corr(h)))
        if nv < best_nv:
            best_nv = nv
            ba, bb = a, b
        h[a], h[b] = 1.0, 0.0
    if ba >= 0:
        h[ba], h[bb] = 0.0, 1.0
    return h, best_nv * 2.0 / N

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    bh = h0.copy()
    bc = c5(bh)

    # Phase 1: intensive binary search on initial
    d1 = min(dl - 500, t0 + 300)
    h1, b1 = binsrch(bh, d1, t0=0.4, mi=15000, ntry=7)
    if b1 < bc:
        bh, bc = h1.copy(), b1

    # Phase 2: steepest descent sweep
    if time.time() < dl - 400:
        d2 = min(dl - 380, time.time() + 100)
        hs = (bh > 0.5).astype(float).copy()
        Ns = len(hs)
        for _ in range(20000):
            if time.time() > d2:
                break
            hs, vs = steep(hs, Ns, ntry=12)
            if vs < bc:
                bc, bh = vs, hs.copy()

    # Phase 3: Nesterov continuous
    if time.time() < dl - 300:
        d3 = min(dl - 280, time.time() + 160)
        hc = proj(bh + np.random.randn(len(bh)) * 0.05, len(bh) / 2.0)
        h3, b3 = nest(hc, d3, alpha=60, lr0=0.03)
        if b3 < bc:
            bh, bc = h3.copy(), b3

    # Phase 4: binary on Nesterov result
    if time.time() < dl - 200:
        d4 = min(dl - 180, time.time() + 150)
        h4, b4 = binsrch(bh, d4, t0=0.3, mi=12000, ntry=7)
        if b4 < bc:
            bh, bc = h4.copy(), b4

    # Phase 5: try N=200
    if time.time() < dl - 140:
        d5 = min(dl - 120, time.time() + 80)
        N5 = 200
        x_src = np.linspace(0, 2, len(bh), endpoint=False)
        x_dst = np.linspace(0, 2, N5, endpoint=False)
        hi5 = np.interp(x_dst, x_src, bh)
        h5, b5 = binsrch(hi5, d5, t0=0.5, mi=8000, ntry=8)
        if b5 < bc:
            bh, bc = h5.copy(), b5

    # Phase 6: try N=300
    if time.time() < dl - 100:
        d6 = min(dl - 80, time.time() + 60)
        N6 = 300
        x_src = np.linspace(0, 2, len(bh), endpoint=False)
        x_dst = np.linspace(0, 2, N6, endpoint=False)
        hi6 = np.interp(x_dst, x_src, bh)
        h6, b6 = binsrch(hi6, d6, t0=0.5, mi=7000, ntry=7)
        if b6 < bc:
            bh, bc = h6.copy(), b6

    # Phase 7: try N=500
    if time.time() < dl - 60:
        d7 = min(dl - 40, time.time() + 40)
        N7 = 500
        x_src = np.linspace(0, 2, len(bh), endpoint=False)
        x_dst = np.linspace(0, 2, N7, endpoint=False)
        hi7 = np.interp(x_dst, x_src, bh)
        h7, b7 = binsrch(hi7, d7, t0=0.5, mi=5000, ntry=6)
        if b7 < bc:
            bh, bc = h7.copy(), b7

    # Phase 8: final Nesterov polish
    if time.time() < dl - 8:
        Nf = len(bh)
        hf = proj(bh + np.random.randn(Nf) * 0.02, Nf / 2.0)
        h8, b8 = nest(hf, dl - 2, alpha=120, lr0=0.015)
        if b8 < bc:
            bh, bc = h8.copy(), b8

    # Final binary cleanup
    if time.time() < dl - 0.5:
        h9, b9 = binsrch(bh, dl - 0.2, t0=0.08, mi=1500, ntry=10)
        if b9 < bc:
            bh, bc = h9.copy(), b9

    # Ensure valid binary output
    bh = (bh > 0.5).astype(float)
    final_c = c5(bh)
    return bh, final_c, len(bh)
