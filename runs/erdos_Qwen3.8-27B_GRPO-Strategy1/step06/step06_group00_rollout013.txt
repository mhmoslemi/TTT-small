def subgrad_opt(h, N, deadline, n_active=20, lr_init=0.05):
    """Subgradient method for min max_s c_s."""
    h = h.astype(float).copy()
    target_sum = N / 2.0
    dx = 2.0 / N
    L = 2 * N - 1
    center = N - 1
    best_h = h.copy()
    best_val = np.max(fft_corr(h))
    y = h.copy()  # momentum variable
    beta = 0.7
    lr = lr_init
    it = 0
    stagnant = 0
    
    while time.time() < deadline:
        corr = fft_corr(y)
        pk = float(np.max(corr))
        
        if pk < best_val - 1e-10:
            best_val = pk
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Get top-n_active shifts
        top_idx = np.argpartition(corr, -n_active)[-n_active:]
        
        # Compute subgradient averaged over top shifts
        grad = np.zeros(N)
        for ai in top_idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * y)
            elif s > 0:
                # dc_s/dh[j] = (1-h[j-s]) - h[j+s] for valid j
                # j-s valid: s <= j < N
                # j+s valid: 0 <= j < N-s
                neg = np.zeros(N)
                neg[s:] = 1.0 - y[:N-s]  # (1-h[j-s]) part for j >= s
                neg[:N-s] -= y[s:]  # -h[j+s] part for j < N-s
                grad += neg
            else:
                m = -s
                neg = np.zeros(N)
                # j-s = j+m valid: 0 <= j+m < N, i.e., -m <= j < N-m
                # Since m > 0: 0 <= j < N-m
                neg[:N-m] += 1.0 - y[m:]  # (1-h[j+m]) for j < N-m
                # j+s = j-m valid: m <= j < N
                neg[m:] -= y[:N-m]  # -h[j-m] for j >= m
                grad += neg
        grad /= n_active
        
        # Update
        h_new = y - lr * grad
        # Project to [0,1] with sum = target_sum
        h_new = np.clip(h_new, 0, 1)
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-10:
            # Water-filling
            if diff > 0:
                # Need to add more: increase positions not at 1
                mask = h_new < 1.0
                if mask.sum() > 0:
                    h_new[mask] += diff / mask.sum()
                    h_new = np.clip(h_new, 0, 1)
            else:
                mask = h_new > 0.0
                if mask.sum() > 0:
                    h_new[mask] += diff / mask.sum()
                    h_new = np.clip(h_new, 0, 1)
        
        y = h_new
        h = h_new
        it += 1

                y = h.copy()
                stagnant = 0
                lr = max(lr, 0.01)
    
    return best_h, best_val * dx

def project_sum(h, target):
    """Project h to [0,1]^N with sum = target."""
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -max(s - target, target - s), max(s - target, target - s)
    # We want to find theta such that sum(clip(h - theta, 0, 1)) = target
    for _ in range(40):
        mid = (lo + hi) / 2
        s_mid = np.clip(h - mid, 0, 1).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h.astype(float), L)
    G = np.fft.rfft(np.flip(1.0 - h.astype(float)), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def find_best_swap(h, N, corr, sample_sz=100):
    L = 2 * N - 1
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) < 2 or len(zeros) < 2:
        return None, None, np.inf
    so = np.random.choice(ones, min(len(ones), sample_sz), replace=False)
    sz = np.random.choice(zeros, min(len(zeros), sample_sz), replace=False)
    s_vals = np.arange(-(N - 1), N)
    a_s = so[:, None] - s_vals[None, :]
    a_ps = so[:, None] + s_vals[None, :]
    D = np.zeros((len(so), L))
    D -= np.where((a_s >= 0) & (a_s < N), 1.0 - h[np.clip(a_s, 0, N - 1)], 0.0)
    D += np.where((a_ps >= 0) & (a_ps < N), h[np.clip(a_ps, 0, N - 1)], 0.0)
    b_s = sz[:, None] - s_vals[None, :]
