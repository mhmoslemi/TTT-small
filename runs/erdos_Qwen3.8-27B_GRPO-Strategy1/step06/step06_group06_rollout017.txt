def fft_corr(h):
    """Compute full linear correlation of h with 1-h using FFT."""
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    result = np.fft.irfft(H * G, L)[:2*N-1]
    return result

def proj_sum(h, target):

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:

    G = np.fft.rfft(np.flip(1.0 - h), L)

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

def to_binary(h, N):
    target = N // 2
    h_c = np.clip(h, 0, 1)
    order = np.argsort(-h_c, kind='stable')

def swap_delta(h, N, p, q, s_arr):

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def to_binary(h, N):
    target = N // 2
    h_c = np.clip(h, 0, 1)
    order = np.argsort(-h_c, kind='stable')
    out = np.zeros(N)
    out[order[:target]] = 1.0
    return out

def swap_delta(h, N, p, q, s_arr):
    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)
    d = np.zeros(2 * N - 1)
    d[vps] -= (1.0 - h[ps[vps]])
    d[vpm] += h[pm[vpm]]
    d[vqs] += (1.0 - h[qs[vqs]])
    d[vqm] -= h[qm[vqm]]
    return d
