```python
def compute_c5(h):

    return float(np.max(fft_corr(h))) * (2.0/N)

def binarize(h, N):

    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h==1)[0]

        np.random.shuffle(idx)

def swap_delta(h, p, q, top_idx, N):

    k = top_idx
    p1 = np.clip(p + k - c, 0, N-1); v1 = (p+k-c >= 0) & (p+k-c < N)
```

```python
def subgrad_step(h, N, T=5.0, lr=0.05):
    c = fft_corr(h)
    cmax = c.max()
    w = np.exp(T*(c - cmax)); w /= w.sum()
    
    # Compute gradient: grad[j] = sum_k w[k] * dc[k]/dh[j]
    # dc[k]/dh[j] = (1-h[j+k-N+1])_valid - h[j-k+N-1]_valid
    center = N - 1
    grad = np.zeros(N)
    
    # Vectorized: for each k, the contribution to grad is w[k] * ((1-h) shifted - h shifted other way)
    # (1-h)[j+k-center] for valid j+k-center means: shift (1-h) left by (k-center)
    # h[j-k+center] for valid j-k+center means: shift h right by (k-center)
    
    for k in range(2*N-1):
        shift = k - center
        if np.abs(w[k]) < 1e-10:
            continue
        # First part: w[k] * (1-h[j+shift]) for valid j+shift
        idx1 = np.arange(N) + shift
        valid1 = (idx1 >= 0) & (idx1 < N)
        grad[valid1] += w[k] * (1 - h[idx1[valid1]])
        # Second part: -w[k] * h[j-shift] for valid j-shift
        idx2 = np.arange(N) - shift
        valid2 = (idx2 >= 0) & (idx2 < N)
        grad[valid2] -= w[k] * h[idx2[valid2]]
    
    # Constraint gradient
    grad += 0.1 * (h - N/(2*N))  # Push towards sum = N/2
    
    return grad
```

```python
    best = compute_c5(h)
    best_h = h.copy()
    target = N / 2.0
    lr = lr0
    
    for it in range(steps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        # Soft-max weight
        cmax = c.max()
        w = np.exp(T * (c - cmax))
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        
        # Gradient computation (focused on top peaks)
        k_top = np.argsort(w)[-min(20, 2*N-1):]  # Top peaks by weight
        w_top = w[k_top]
        
        grad = np.zeros(N)
        center = N - 1

        for ki in k_top:
            wk = w[ki]

            idx1 = j + s
            m1 = (idx1 >= 0) & (idx1 < N)
            grad[m1] += wk * (1 - h[idx1[m1]])
            idx2 = j - s
            m2 = (idx2 >= 0) & (idx2 < N)
            grad[m2] -= wk * h[idx2[m2]]
        
        # Normalize gradient
        gnorm = np.sqrt(np.sum(grad**2) + 1e-12)
        if gnorm < 1e-12:
            break
        
        # Update
        h_new = h - lr * grad / gnorm
        # Project: clip to [0,1], adjust sum
        h_new = np.clip(h_new, 0, 1)
        diff = h_new.sum() - target
        h_new -= diff / N
        h_new = np.clip(h_new, 0, 1)
        
        h = h_new
        cur = compute_c5(h)
        if cur < best:
            best = cur
            best_h = h.copy()
        
        lr *= 0.9995
    
    return best_h, best
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def binarize(h, N):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def swap_delta(h, p, q, top_idx, N):
    c = N - 1
    k = top_idx
    p1 = np.clip(p + k - c, 0, N - 1)
    v1 = (p + k - c >= 0) & (p + k - c < N)
    p2 = np.clip(k - p + c, 0, N - 1)
    v2 = (k - p + c >= 0) & (k - p + c < N)
    q1 = np.clip(q + k - c, 0, N - 1)
    v3 = (q + k - c >= 0) & (q + k - c < N)
    q2 = np.clip(k - q + c, 0, N - 1)
    v4 = (k - q + c >= 0) & (k - q + c < N)
    return (np.where(v1, -(1 - h[p1]), 0) + np.where(v2, h[p2], 0) +
            np.where(v3, (1 - h[q1]), 0) + np.where(v4, -h[q2], 0))

def guided_local_search(h, N, deadline, top_k=20, T0=2.0, max_iter=200000):
    h = binarize(h, N)
    c = fft_corr(h)
    top_idx = np.argsort(c)[-top_k:]
    best_max = float(c[top_idx].max())
    best_h = h.copy()
    T = T0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        P = float(c[top_idx].max())
        if P < best_max:
            best_max = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Focused: target the worst peak
        worst_pos = top_idx[np.argmax(c[top_idx])]
        center = N - 1
        # Find best p (1->0) and q (0->1) for worst peak
        pk = np.clip(ones + worst_pos - center, 0, N - 1)
        pv = (ones + worst_pos - center >= 0) & (ones + worst_pos - center < N)
        qk = np.clip(worst_pos - ones + center, 0, N - 1)
        qv = (worst_pos - ones + center >= 0) & (worst_pos - ones + center < N)
        dp = np.where(pv, -(1 - h[pk]), 0) + np.where(qv, h[qk], 0)
        pkq = np.clip(zeros + worst_pos - center, 0, N - 1)
        pvz = (zeros + worst_pos - center >= 0) & (zeros + worst_pos - center < N)
        qkz = np.clip(worst_pos - zeros + center, 0, N - 1)
        qvz = (worst_pos - zeros + center >= 0) & (worst_pos - zeros + center < N)
        dq = np.where(pvz, (1 - h[pkz]), 0) + np.where(qvz, -h[qkz], 0)
        # Score each pair by worst-peak reduction (approx: dp[p] + dq[q])
        best_score = -10.0
        best_pq = None
        dp_sorted = np.argsort(dp)
        dq_sorted = np.argsort(dq)
        n_cand = min(40, len(ones), len(zeros))
        for i in range(n_cand):
            for j in range(n_cand):
                p = int(ones[dp_sorted[i]])
                q = int(zeros[dq_sorted[j]])
                sc = dp[dp_sorted[i]] + dq[dq_sorted[j]]
                if sc < best_score:
                    best_score = sc
                    best_pq = (p, q)
        # Try best candidates with full verification
        accepted = False
        for i in range(min(30, n_cand * n_cand)):
            if time.time() > deadline:
                break
            di = i // n_cand
            dj = i % n_cand
            p = int(ones[dp_sorted[di]])
            q = int(zeros[dq_sorted[dj]])
            delta = swap_delta(h, p, q, top_idx, N)
            new_vals = c[top_idx] + delta
            new_max = float(new_vals.max())
            if new_max < P or (T > 0.3 and np.random.random() < np.exp(-(new_max - P) / T)):
                h[p], h[q] = 0.0, 1.0
                c = fft_corr(h)
                top_idx = np.argsort(c)[-top_k:]
                P = new_max
                accepted = True
                break
        T *= 0.9997
        if it % 2000 == 0:
            T = max(T, 0.3)
    return best_h, best_max * (2.0 / N)

    best_h = h.copy()
    target = N / 2.0
    lr = lr0
    j = np.arange(N)
    center = N - 1
    it = 0
    while time.time() < deadline:
        c = fft_corr(h)
        cmax = c.max()
        w = np.exp(T_sm * (c - cmax))
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        k_order = np.argsort(w)[-top_peak_k:]
        grad = np.zeros(N)
        for ki in k_order:
            wk = w[ki]
            if wk < 1e-10:

            s = ki - center
            idx1 = j + s
            m1 = (idx1 >= 0) & (idx1 < N)
            if m1.any():
                grad[m1] += wk * (1 - h[idx1[m1]])
            idx2 = j - s
            m2 = (idx2 >= 0) & (idx2 < N)
            if m2.any():
                grad[m2] -= wk * h[idx2[m2]]
        grad += 0.3 * (h - 0.5)
        gnorm = np.sqrt(np.sum(grad * grad) + 1e-15)
        if gnorm < 1e-13:
            break
        h = h - lr * grad / gnorm
        h = np.clip(h, 0, 1)
        h -= (h.sum() - target) / N
        h = np.clip(h, 0, 1)
        cur = compute_c5(h)
        if cur < best:
            best = cur
            best_h = h.copy()
        lr = max(0.001, lr0 * (0.999 ** it))
        it += 1
        if it % 2000 == 0:
            T_sm *= 0.8
    return best_h, best

                cp_dl = min(deadline - 15, time.time() + 25)
                h_cp = best_h + rng.randn(len(best_h)) * 0.02
                h_cp = np.clip(h_cp, 0, 1)
                h_cp -= (h_cp.sum() - len(best_h) / 2.0) / len(best_h)
                h_cp = np.clip(h_cp, 0, 1)
                h_cp, c5_cp = continuous_refine(h_cp, len(best_h), cp_dl, T_sm=4.0, top_peak_k=12)
                if c5_cp < best_c5:
                    best_c5, best_h = c5_cp, h_cp.copy()

    # Phase 5: Final local search + continuous
    if time.time() < deadline - 30:
        d5 = min(deadline - 20, time.time() + 60)
        h_f, c5_f = guided_local_search(best_h, len(best_h), d5, top_k=20, T0=1.0)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    if time.time() < deadline - 10:
        hcF = best_h + rng.randn(len(best_h)) * 0.02
        hcF = np.clip(hcF, 0, 1)
        hcF -= (hcF.sum() - len(best_h) / 2.0) / len(best_h)
        hcF = np.clip(hcF, 0, 1)
        hcF, c5F = continuous_refine(hcF, len(best_h), deadline - 2, T_sm=6.0, top_peak_k=15)
        if c5F < best_c5:
            best_c5, best_h = c5F, hcF.copy()

    # Final: try binary version too
    h_bin = binarize(best_h, len(best_h))
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5, best_h = c5_bin, h_bin

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
```
