    beta = beta0
    shifts = np.arange(-(N - 1), N)
    pos = np.arange(N)[:, None]
    ip = pos + shifts[None, :]
    im = pos - shifts[None, :]
    vp = ((ip >= 0) & (ip < N)).astype(float)
    vm = ((im >= 0) & (im < N)).astype(float)
    ip_c = np.clip(ip, 0, N - 1)
    im_c = np.clip(im, 0, N - 1)
    while time.time() < deadline:
        c = fft_corr(h)
        cs = beta * c
        cs -= cs.max()
        w = np.exp(cs)
        w /= w.sum()
        grad = (w[None, :] * ((1.0 - h[ip_c]) * vp - h[im_c] * vm)).sum(axis=1)
        beta *= 1.01
        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        mc = m / (1 - 0.9 ** t)
        vc = v / (1 - 0.999 ** t)
        h = project_sum(h - lr * mc / (np.sqrt(vc) + 1e-8), N / 2.0)
        pk = float(np.max(fft_corr(h)))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        if t % 300 == 0:
            lr *= 0.95
    return best_h, best_pk * (2.0 / N)

def upscale(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h)
    return fix_sum_binary((h_interp > 0.5).astype(float), N_new)

def make_periodic(N, period, duty, offset=0):
    h = np.zeros(N)
    x = np.arange(N)
    h[((x + offset) % period) < period * duty] = 1.0
    return fix_sum_binary(h, N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = (h0 > 0.5).astype(float)
    best_c5 = compute_c5(best_h)
    # Phase 1: Thorough binary SA on parent
    d1 = min(deadline - 350, t0 + 180)
    h1, c1 = binary_sa(h0, N0, d1, max_sweeps=30000, t0=0.014)
    if c1 < best_c5:
        best_c5, best_h = c1, h1
    # Phase 2: 2-swap SA from best
    if time.time() < deadline - 250:
        d2 = min(deadline - 200, time.time() + 120)
        h2, c2 = binary_sa_2swap(best_h, N0, d2, max_sweeps=15000, t0=0.016)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
    # Phase 3: Try different resolutions
    for N_try in [N0 * 2, N0, N0 // 2 + N0 // 2 * 0]:
        if time.time() > deadline - 180:
            break
        N_try = max(N_try, 100)
        if N_try % 2 != 0:
            N_try += 1
        if N_try == N0:
            continue
        d3 = min(deadline - 120, time.time() + 70)
        h_up = upscale(best_h, N_try)
        h3, c3 = binary_sa(h_up, N_try, d3, max_sweeps=15000, t0=0.01)
        if c3 < best_c5:
            best_c5, best_h = c3, h3
    # Phase 4: Periodic starts
    if time.time() < deadline - 120:
        for period in [7, 9, 11, 13, 17, 19, 23, 10, 6]:
            if time.time() > deadline - 80:
                break
            for duty in [0.45, 0.5, 0.55]:
                if time.time() > deadline - 70:
                    break
                h_per = make_periodic(N0, period, duty)
                dp = min(deadline - 40, time.time() + 12)
                hp, cp = binary_sa(h_per, N0, dp, max_sweeps=5000, t0=0.02)
                if cp < best_c5:
                    best_c5, best_h = cp, hp
    # Phase 5: Adam continuous
    if time.time() < deadline - 50:
        d5 = min(deadline - 25, time.time() + 60)
        hf, cf = adam_smooth(best_h, len(best_h), d5, beta0=100.0, lr=0.004)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    # Phase 6: Final binary polish
    if time.time() < deadline - 3:
        d6 = deadline - 1
        Nf = len(best_h)
        hf2, cf2 = binary_sa(best_h, Nf, d6, max_sweeps=50000, t0=0.006)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2
    # Phase 7: 2-swap polish
    if time.time() < deadline - 1:
        h3f, c3f = binary_sa_2swap(best_h, len(best_h), deadline - 0.5, max_sweeps=10000, t0=0.005)
        if c3f < best_c5:
            best_c5, best_h = c3f, h3f
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
