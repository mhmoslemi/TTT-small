import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    # D[p, k] = change in corr[k] when h[p] goes from 1 to 0
    # = -(1-h[p+k-N+1]) + h[p-k+N-1] (with bounds)
    shifts = np.arange(2*N-1) - (N-1)  # L values: -(N-1), ..., N-1
    pos = np.arange(N)[:, None]  # (N, 1)
    sv = shifts[None, :]  # (1, 2N-1)
    # corr[k] = sum_i h[i]*(1-h[i+L]) where L = k-(N-1)
    # D[p,k] = -(1-h[p+L]) + h[p-L] (with bounds checks)
    pm = pos + sv  # p + L = p + k - (N-1)
    pp = pos - sv  # p - L = p - k + (N-1)
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return hm * vm - hp * vp  # This is the change when h[p] decreases by 1
    
def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(42 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_greedy(h, N, deadline, max_sweeps=2000, sample_frac=1.0, perturb_every=50):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(777)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        D = compute_D(h, N)
        Dz = D[zeros, :]  # change in corr when zero->one, negated: actually D[p,:] is change when h[p] 1->0
        # When we flip 1 at p to 0, corr changes by -D[p,:] (D is the contribution)
        # When we flip 0 at q to 1, corr changes by D[q,:]
        # Net change: D[q,:] - D[p,:]
        # We want to minimize max(corr - D[p,:] + D[q,:])
        
        if sample_frac < 1.0 and len(ones) > 50:
            n_s = max(30, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]  # corr after removing p
            cand = base[None, :] + D[zeros, :]  # corr after adding q
            npk = cand.max(axis=1)  # (n_zeros,)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    
    return best_h, best_c * (2.0 / N)

def nesterov_full(h, N, deadline, k=100, lr0=0.02, beta0=0.85):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        
        # Subgradient: use top-k lags
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += (1.0 - y[M:])
                    grad[M:] -= y[:N-M]
        grad /= k
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.97, 0.80 + it * 0.00005)
    
    return best_h, best_pk * dx

def simulated_annealing(h, N, deadline, t_start=0.5, t_end=0.01, max_iters=50000):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    cur_c = best_c
    rng = np.random.RandomState(999)
    
    it = 0
    total = max_iters
    while time.time() < deadline and it < total:
        T = t_start * (t_end / t_start) ** (it / total)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        
        c_old = fft_corr(h)
        D = compute_D(h, N)
        new_corr = c_old - D[p, :] + D[q, :]
        new_c = float(new_corr.max())
        
        delta = new_c - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h[p], h[q] = 0.0, 1.0
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        it += 1
    
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
