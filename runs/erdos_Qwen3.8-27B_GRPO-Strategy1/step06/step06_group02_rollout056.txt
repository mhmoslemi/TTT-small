    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_softmax(h, T, N):
    """Compute gradient of soft-max of correlation values."""
    corr = fft_corr(h)
    # Soft-max weights
    mc = corr - corr.max()
    p = np.exp(T * mc)
    p /= p.sum()
    # Gradient: sum_k p_k * d corr[k]/d h[i]
    # d corr[k]/d h[i] = (1-h[i+m_k]) - h[i-m_k] where m_k = k-(N-1)
    grad = np.zeros(N)
    center = N - 1
    # Only use top k for efficiency
    k_vals = np.argsort(p)[::-1]
    # Use all for correctness but could truncate
    for k in range(2*N-1):
        if p[k] < 1e-10:
            continue
        m = k - center
        # (1-h[i+m]) for valid i+m
        if m >= 0:
            # i+m < N => i < N-m
            idx = np.arange(N - m)
            grad[idx] += p[k] * (1.0 - h[idx + m])
            # -h[i-m] for valid i-m: i-m >= 0 => i >= m
            idx2 = np.arange(m, N)
            grad[idx2] -= p[k] * h[idx2 - m]
        elif m < 0:
            mm = -m
            # i+m < N always (since m<0), i >= 0
            idx = np.arange(N)
            grad[idx] += p[k] * (1.0 - h[idx - m])  # i+m = i-mm wait
            # Wait, m is negative. i+m < N means i < N-m which is always true since i<N and m<0.
            # i+m >= 0 means i >= -m = mm
            idx = np.arange(mm, N)
            grad[idx] += p[k] * (1.0 - h[idx + m])

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binary_local_search(h, N, deadline, max_sweeps=500, sample_n=300, 
                        flat_weight=0.1, anneal_start=100):

            break

        P = float(corr.max())

        top2 = sorted_c[:2]
        score = P + flat_weight * (top2[0] - top2[1]) if len(top2) > 1 else P
        
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1

        if len(ones) < 2 or len(zeros) < 2:
            break

            valid_plus = (p + m_range >= 0) & (p + m_range < N)

            idx_minus = np.where(valid_minus)[0]
            delta_p[idx_plus] = (1.0 - h[p + m_range[idx_plus]])
            delta_p[idx_minus] -= h[p - m_range[idx_minus]]

            # We want to minimize max(new_corr)
            # = minimize max(corr - delta_p + delta_q)

            # We want argmin over q of new.max()

            h[p], h[q] = 0.0, 1.0
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        
        if no_improve > 400:
            break
    return h, best_c

def soft_max_opt(h, N, deadline, T_start=2.0, T_end=50.0, k_top=50, lr0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    n_iters = 0
    while time.time() < deadline:
        t_frac = min(1.0, it / 200.0)
        T = T_start + (T_end - T_start) * t_frac * t_frac  # Quadratic ramp
        
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Soft-max gradient
        mc = corr - corr.max()
        p_sm = np.exp(T * mc)
        p_sm /= p_sm.sum()
        
        # Gradient via convolution-like computation
        L = 2 * N - 1
        center = N - 1
        m_arr = np.arange(-center, center + 1)
        
        # Use top-k of p_sm for efficiency
        k_use = min(k_top, L)
        top_idx = np.argsort(p_sm)[::-1][:k_use]
        p_top = p_sm[top_idx]
        p_top /= p_top.sum()  # Renormalize
        m_top = m_arr[top_idx]  # shifts
        
        # grad[i] = sum_k p_k * [(1-h[i+m_k]) - h[i-m_k]]
        # For top k terms only
        grad = np.zeros(N)
        for kk in range(k_use):
            mm = int(m_top[kk])
            pk_val = p_top[kk]
            if mm >= 0:
                # i+mm in [0,N): i in [0, N-mm)
                if N - mm > 0:
                    grad[:N-mm] += pk_val * (1.0 - h[mm:])
                # i-mm in [0,N): i in [mm, N)
                if mm < N:
                    grad[mm:] -= pk_val * h[:N-mm]
            else:
                mm2 = -mm
                # i+mm in [0,N): i in [mm2, N)
                if N - mm2 > 0:
                    grad[mm2:] += pk_val * (1.0 - h[:N-mm2])
                # i-mm in [0,N): i in [0, N-mm2)
                if N-mm2 > 0:
                    grad[:N-mm2] -= pk_val * h[mm2:]
        
        h_new = project_sum(h - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        n_iters += 1
        if it % 50 == 0:
            lr *= 0.993
            beta = min(0.95, 0.85 + it * 0.0002)
    
    return best_h, best_pk * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    n1 = len(ones)
    if n1 > nh:
        to_remove = ones[np.random.choice(n1, n1 - nh, replace=False)]
        h[to_remove] = 0.0
    elif n1 < nh:
        to_add = zeros[np.random.choice(len(zeros), nh - n1, replace=False)]
        h[to_add] = 1.0
    return h

def binary_local_search(h, N, deadline, max_sweeps=1000, sample_n=300,
                        flat_w=0.06, anneal_start=120):
    h = h.astype(float).copy()
    center = N - 1
    L = 2 * N - 1
    m_all = np.arange(-center, center + 1).astype(int)
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Precompute b matrix for all zeros (addition delta)
        q_col = zeros[:, None]
        m_row = m_all[None, :]
        qp = q_col + m_row
        qm = q_col - m_row
        vp = (qp >= 0) & (qp < N)
        vm = (qm >= 0) & (qm < N)
        qpc = np.clip(qp, 0, N - 1)
        qmc = np.clip(qm, 0, N - 1)
        b = np.where(vp, 1.0 - h[qpc], 0.0) - np.where(vm, h[qmc], 0.0)
        # Sample ones
        if len(ones) > sample_n:
            p_list = ones[np.random.permutation(len(ones))[:sample_n]]
        else:
            p_list = ones
        best_score = P + 1.0
        best_pq = None
        # Flatness: penalize 2nd max
        sorted_c = np.sort(corr)[::-1][:3]
        flat_pen = flat_w * (sorted_c[0] - sorted_c[1]) if len(sorted_c) > 1 else 0.0
        target_score = P + flat_pen
        for p in p_list:
            a = np.zeros(L)
            vp2 = (p + m_all >= 0) & (p + m_all < N)
            vm2 = (p - m_all >= 0) & (p - m_all < N)
            a[vp2] = 1.0 - h[p + m_all[vp2]]
            a[vm2] -= h[p - m_all[vm2]]
            base = corr - a
            new_vals = base[None, :] + b
            new_max = new_vals.max(axis=1)
            # Also compute 2nd max for flatness scoring (approximate)
            mi = int(np.argmin(new_max))
            v = float(new_max[mi])
            sc = v + flat_w * 0.3
            if sc < best_score:
                best_score = sc
                best_pq = (int(p), int(zeros[mi]))
        # Annealing kick
        use_kick = (sweep > anneal_start and no_improve > 80) or \
                   (no_improve > 200 and np.random.random() < 0.15)
        if best_pq is None or use_kick:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        if no_improve > 500:
            break
    return h, best_c

def soft_max_opt(h, N, deadline, T_start=3.0, T_end=100.0, k_top=60, lr0=0.025):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    center = N - 1
    L = 2 * N - 1
    m_all = np.arange(-center, center + 1).astype(int)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        t_frac = min(1.0, it / 300.0)
        T = T_start + (T_end - T_start) * t_frac * t_frac
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        mc = corr - corr.max()
        p_sm = np.exp(T * mc)
        p_sm /= p_sm.sum()
        k_use = min(k_top, L)
        top_idx = np.argsort(p_sm)[::-1][:k_use]
        p_top = p_sm[top_idx].copy()
        p_top /= p_top.sum()
        m_top = m_all[top_idx]
        grad = np.zeros(N)
        for kk in range(k_use):
            mm = int(m_top[kk])
            pv = p_top[kk]
            if pv < 1e-12:
                continue
            if mm >= 0:
                if N - mm > 0:
                    grad[:N - mm] += pv * (1.0 - h[mm:])
                if mm < N:
                    grad[mm:] -= pv * h[:N - mm]
            else:
                mm2 = -mm
                if N - mm2 > 0:
                    grad[mm2:] += pv * (1.0 - h[:N - mm2])
                if N - mm2 > 0:
                    grad[:N - mm2] -= pv * h[mm2:]
        h_new = project_sum(h - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.992
            beta = min(0.95, 0.85 + it * 0.0002)
    return best_h, best_pk * dx
