```python
import numpy as np
import time

def compute_corr(h):
    N = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_h(h, target_sum):
    """Project h to [0,1] with sum = target_sum using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target_sum) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_grad(h, corr, tau):
    """Compute gradient of log-sum-exp(corr * tau / N) w.r.t. h."""
    N = len(h)
    # weights
    w = np.exp(tau * corr / N - tau * corr.max() / N)  # for numerical stability
    w /= w.sum()
    # grad[i] = (1/N) * sum_ell w[ell] * ((1-h[i+ell]) - h[i-ell])
    # where ell maps to corr index: corr index = ell + (N-1)
    # Let's compute term1[i] = sum_ell w[ell] * h[i+ell] and term2[i] = sum_ell w[ell] * h[i-ell]
    # Use direct computation (O(N^2) but N=380 is fine)
    grad = np.zeros(N)
    half = N - 1
    for i in range(N):
        s1 = 0.0
        s2 = 0.0
        for d in range(-half, half + 1):
            idx = d + half  # index into corr/w
            j1 = i + d
            j2 = i - d
            if 0 <= j1 < N:
                s1 += w[idx] * h[j1]
            if 0 <= j2 < N:
                s2 += w[idx] * h[j2]
```

Then update A using the incremental update.

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target_sum) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_A(h, N):
    half = N - 1
    r = np.arange(N)
    d = np.arange(-half, half + 1)
    j1 = r[:, None] + d[None, :]
    j2 = r[:, None] - d[None, :]
    m1 = (j1 >= 0) & (j1 < N)
    m2 = (j2 >= 0) & (j2 < N)
    A = np.where(m1, h[np.clip(j1, 0, N-1)], 0.0) + np.where(m2, h[np.clip(j2, 0, N-1)], 0.0)
    return A

def swap_optimize(h, N, deadline, max_sweeps=800, sample_frac=0.4, anneal_p=0.0):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        oi, zi = np.where(h == 1)[0], np.where(h == 0)[0]
        h[:] = 0.0
        if n1 > nh:
            h[oi[:nh]] = 1.0
        else:
            h[oi] = 1.0
            h[zi[:nh-n1]] = 1.0
    
    corr = compute_corr(h)
    A = compute_A(h, N)
    half = N - 1
    r = np.arange(N)
    best_c = float(corr.max())
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(corr.max())
        if P < best_c - 1e-12:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 120:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        ns = max(2, int(len(ones) * sample_frac))
        p_cand = np.random.choice(ones, min(ns, len(ones)), replace=False)
        
        best_delta = 1e10
        best_pq = None
        for p in p_cand:
            B = corr + A[p]
            diffs = B[None, :] - A[zeros, :]
            new_max = diffs.max(axis=1)
            mi = int(np.argmin(new_max))
            delta = float(new_max[mi]) - P
            if delta < best_delta:
                best_delta = delta
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is not None and best_delta < -1e-12:
            p, q = best_pq
            corr += A[p] - A[q]
            # Update A incrementally
            A[r, p - r + half] -= 1
            A[r, r - p + half] -= 1
            A[r, q - r + half] += 1
            A[r, r - q + half] += 1
            h[p], h[q] = 0.0, 1.0
        elif anneal_p > 0 and np.random.random() < anneal_p:

            A[r, r - q + half] += 1
            h[ones[p]], h[zeros[q]] = 0.0, 1.0
            stall = 0
        else:

    return h, best_c

def lse_gradient(h, corr, tau, N):
    x = tau * corr / N
    w = np.exp(x - x.max())
    w /= w.sum()
    half = N - 1
    d = np.arange(-half, half + 1)
    i = np.arange(N)
    j1 = i[:, None] + d[None, :]
    j2 = i[:, None] - d[None, :]
    m1 = (j1 >= 0) & (j1 < N)
    m2 = (j2 >= 0) & (j2 < N)
    term1 = (w[None, :] * m1 * h[np.clip(j1, 0, N-1)]).sum(axis=1)
    term2 = (w[None, :] * m2 * h[np.clip(j2, 0, N-1)]).sum(axis=1)
    return (1.0 - term1 - term2) / N
