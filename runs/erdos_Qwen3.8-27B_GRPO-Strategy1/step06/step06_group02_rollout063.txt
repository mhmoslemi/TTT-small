    sign = 2.0 * h[:, 0] - 1.0

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def flip_delta(h, N):
    lags = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    lag = lags[None, :]
    plus = pos + lag
    minus = pos - lag
    vp = (plus >= 0) & (plus < N)
    vm = (minus >= 0) & (minus < N)
    hp = np.where(vp, h[np.clip(plus, 0, N-1)], 0.0)
    one_minus_hm = np.where(vm, 1.0 - h[np.clip(minus, 0, N-1)], 0.0)
    base = hp - one_minus_hm
    sign = 2.0 * h - 1.0
    return sign[:, None] * base

def binary_optimize(h, N, deadline, max_sweeps=5000, tau=5.0, elite_k=10):
    h = h.copy().astype(float)
    target = N // 2
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    T = 1.0  # annealing temperature
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep % 100 == 0 and no_improve > 200:
            T = max(0.1, T * 0.95)
        
        corr = fft_corr(h)
        P = float(corr.max())
        
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        D = flip_delta(h, N)
        
        # Score function: soft max to flatten peaks
        # For each (p, q) candidate, compute new_corr and score it
        # Vectorize over p: for each p, compute best q
        
        best_score = P + 10.0
        best_pq = None
        
        # Sample ones if too many
        if len(ones) > 200:
            perm = np.random.permutation(len(ones))
            p_list = ones[perm[:150]]
        else:
            p_list = ones
        
        for p in p_list:
            base = corr + D[int(p), :]
            # For all zeros: new_corr = base + D[q]
            cand = base[None, :] + D[zeros, :]
            # Use soft max score: tau * log(sum(exp(cand/tau)))
            # But this is expensive. Use top-k approach:
            top_idx = np.argpartition(cand, -elite_k, axis=1)[:, -elite_k:]
            top_vals = np.take_along_axis(cand, top_idx, axis=1)
            # Score: max + small * mean of top-k
            scores = top_vals[:, 0] + 0.1 * top_vals.mean(axis=1)
            best_q_local = int(np.argmin(scores))
            best_val_local = float(scores[best_q_local])
            
            if best_val_local < best_score:
                best_score = best_val_local
                best_pq = (int(p), int(zeros[best_q_local]))
        
        # Annealing: sometimes accept bad moves
        if best_pq is None or (no_improve > 100 and np.random.random() < T * 0.1):
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
        
        if no_improve > 500:
            # Restart with perturbation
            for _ in range(10):
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                ones = np.flatnonzero(h == 1)
                zeros = np.flatnonzero(h == 0)
            no_improve = 0
            best_c = float(np.max(fft_corr(h)))
    
    return h, best_c * (2.0/N)

def nesterov_smooth(h_init, N, deadline, k=30, lr0=0.01, beta0=0.9):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    tau = 3.0  # softmax temperature
    
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        c5_val = float(corr.max()) * (2.0/N)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Softmax weights for gradient
        tau = max(1.0, tau * (1 - it * 0.0005))
        shifted = corr / tau
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        
        # Gradient of weighted sum of correlations
        grad = np.zeros(N)
        lags = np.arange(2*N-1) - (N-1)
        # For each correlation position k with weight w[k], 
        # grad gets: w[k] * (1-2*h at the relevant positions)
        # This is complex to vectorize perfectly, so use top-k
        top_idx = np.argsort(w)[-k:]
        for kidx in top_idx:
            L = int(lags[kidx])
            wk = float(w[kidx])
            if L == 0:
                grad += wk * (1.0 - 2.0 * y)
            elif L > 0:
                # corr[k] involves h[i+L]*(1-h[i]) for i=0..N-1-L
                # d/dh[j]: if j in [L, N-1]: (1-h[j-L])
                #          if j in [0, N-1-L]: -h[j+L]
                # But y is the current estimate
                n_valid = N - L
                grad[L:N] += wk * (1.0 - y[0:n_valid])
                grad[0:n_valid] -= wk * y[L:N]
            else:
                M = -L
                n_valid = N - M
                grad[0:n_valid] += wk * (1.0 - y[M:N])
                grad[M:N] -= wk * y[0:n_valid]
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 50 == 0:
            lr *= 0.99
            beta = min(0.98, beta0 + it * 0.0002)
    
    return best_h, best_c5

def fix_binary_sum(h, N):
    h = h.copy()
    target = N // 2
    cur = int(h.sum())
    if cur > target:

        h[zeros[order[:target-cur]]] = 1.0
    elif cur == target:
        h[h > 0.5] = 1.0
        h[h <= 0.5] = 0.0
    return h

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def flip_delta(h, N):
    lags = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    lag = lags[None, :]
    plus = pos + lag
    minus = pos - lag
    vp = (plus >= 0) & (plus < N)
    vm = (minus >= 0) & (minus < N)
    hp = np.where(vp, h[np.clip(plus, 0, N - 1)], 0.0)
    omhm = np.where(vm, 1.0 - h[np.clip(minus, 0, N - 1)], 0.0)
    base = hp - omhm
    sign = 2.0 * h - 1.0
    return sign[:, None] * base

def fix_binary(h, N):
    h = h.copy()
    target = N // 2
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    cur = int(h.sum())
    if cur > target:
        ones = np.flatnonzero(h == 1)
        order = np.random.permutation(len(ones))
        h[ones[order[target:]]] = 0.0
    elif cur < target:
        zeros = np.flatnonzero(h == 0)
        order = np.random.permutation(len(zeros))
        h[zeros[order[:target - cur]]] = 1.0
    return h

def binary_opt(h, N, deadline, max_sweeps=20000, elite_k=6, flatten_w=0.05):
    h = fix_binary(h, N)
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    corr = fft_corr(h)
    best_c = float(corr.max())
    no_improve = 0
    T = 0.5

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if no_improve > 300:
            # Large perturbation: flip 5-15 pairs
            n_flips = int(np.random.randint(5, 16))
            o = np.flatnonzero(h == 1)
            z = np.flatnonzero(h == 0)
            for _ in range(n_flips):
                p = int(o[np.random.randint(len(o))])
                q = int(z[np.random.randint(len(z))])
                h[p], h[q] = 0.0, 1.0
                o = np.flatnonzero(h == 1)
                z = np.flatnonzero(h == 0)
            no_improve = 0
            best_c = float(fft_corr(h).max())
            T = min(1.0, T + 0.3)

        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-13:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) < 2 or len(zeros) < 2:
            break

        D = flip_delta(h, N)
        best_score = P + 10.0
        best_pq = None

        # Evaluate all ones (N/2 <= 250 is manageable)
        p_list = ones
        if len(p_list) > 220:
            perm = np.random.permutation(len(p_list))
            p_list = p_list[perm[:180]]

        for ip in range(len(p_list)):
            p = int(p_list[ip])
            base = corr + D[p, :]
            cand = base[None, :] + D[zeros, :]
            # Score: max + flatten_w * mean of top elite_k
            part = np.argpartition(cand, -elite_k, axis=1)[:, -elite_k:]
            vals = np.take_along_axis(cand, part, axis=1)
            scores = vals[:, 0] + flatten_w * vals.mean(axis=1)
            bqi = int(np.argmin(scores))
            if scores[bqi] < best_score:
                best_score = float(scores[bqi])
                best_pq = (p, int(zeros[bqi]))

        # Annealing kick
        accept_worse = no_improve > 80 and np.random.random() < T * 0.15
        if best_pq is None or accept_worse:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            T = max(0.05, T * 0.98)
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0

    return h, best_c * (2.0 / N)

def nesterov_smooth(h0, N, deadline, k=30, lr0=0.012, beta0=0.92):
    h = project_sum(h0.astype(float).copy(), N / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    tau = 4.0
    lags = np.arange(2 * N - 1) - (N - 1)
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        c5v = float(corr.max()) * (2.0 / N)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        tau = max(1.5, 4.0 - it * 0.005)
        sh = corr / tau
        sh -= sh.max()
        w = np.exp(sh)
        w /= w.sum()
        grad = np.zeros(N)
        top = np.argsort(w)[-k:]
        for ki in top:
            L = int(lags[ki])
            wk = float(w[ki])
            if L == 0:
                grad += wk * (1.0 - 2.0 * y)
            elif L > 0:
                nv = N - L
                grad[L:] += wk * (1.0 - y[:nv])
                grad[:nv] -= wk * y[L:]
            else:
                M = -L
                nv = N - M
                grad[:nv] += wk * (1.0 - y[M:])
                grad[M:] -= wk * y[:nv]
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.992
            beta = min(0.98, beta0 + it * 0.00015)
    return best_h, best_c5

def modular_pattern(N, p):
    h = np.zeros(N)
    h[(np.arange(N) * p) % N < N // 2] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
