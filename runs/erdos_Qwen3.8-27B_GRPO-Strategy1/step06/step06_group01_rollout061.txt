def compute_flip_effect(p, h, N):
    """Compute the change in correlation when h[p] is flipped (h[p] += dh)."""
    # For dh = 1 (0→1) or dh = -1 (1→0)
    # Δcorr[j] = dh * ((1-h[p+j-(N-1)]) - h[p-j+(N-1)]) for valid indices
    effect = np.zeros(2*N-1)
    v = 1.0 - h
    # First term: valid when 0 <= p+j-(N-1) <= N-1, i.e., N-1-p <= j <= 2N-1-p
    j_lo = N-1-p
    j_hi = 2*N-1-p
    j_lo = max(0, j_lo)
    j_hi = min(2*N-2, j_hi)
    idx1 = p + np.arange(j_lo, j_hi+1) - (N-1)  # p+j-(N-1)
    effect[j_lo:j_hi+1] += v[idx1]  # will be multiplied by dh later
    
    # Second term: valid when 0 <= p-j+(N-1) <= N-1, i.e., N-1-p <= j <= N-1+p
    j_lo2 = N-1-p
    j_hi2 = N-1+p

    effect[j_lo2:j_hi2+1] -= h[idx2]
    
    return effect
