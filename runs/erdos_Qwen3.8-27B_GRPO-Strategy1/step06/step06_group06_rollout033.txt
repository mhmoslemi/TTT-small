    n = len(h)
    corr = fft_corr(h)  # shape (2n-1,)
    w = np.exp(tau * corr - tau * corr.max())  # numerical stability
    W = w.sum()
    w = w / W  # normalized weights

    shifts = np.arange(2*n-1) - (n-1)  # -(n-1) to (n-1)
    
    grad = np.zeros(n)
    for j in range(n):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    # Use bisection to find the right shift
    def sum_after_shift(shift):
        return float(np.clip(h - shift, 0, 1).sum())
    
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if sum_after_shift(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin == 1)[0]
        np.random.shuffle(idx)
        h_bin[idx[:n1-target]] = 0
    elif n1 < target:
        idx = np.where(h_bin == 0)[0]
        np.random.shuffle(idx)
        h_bin[idx[:target-n1]] = 1
    return h_bin

def softmax_gradient(h, tau):
    n = len(h)
    corr = fft_corr(h)
    # Numerically stable softmax
    max_c = np.max(corr)
    log_w = tau * (corr - max_c)
    w = np.exp(log_w)
    W = w.sum()
    if W < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= W
    
    shifts = np.arange(2*n-1) - (n-1)
    s1 = np.arange(n)[:, None] - shifts[None, :]
    s2 = np.arange(n)[:, None] + shifts[None, :]
    valid1 = (s1 >= 0) & (s1 < n)
    valid2 = (s2 >= 0) & (s2 < n)
    
    oh = 1.0 - h
    g1 = np.sum(w[None, :] * oh[np.clip(s1, 0, n-1)] * valid1, axis=1)
    g2 = np.sum(w[None, :] * h[np.clip(s2, 0, n-1)] * valid2, axis=1)
    
    return (g1 - g2) / n  # normalize by n for scale-invariance

def softmax_descent(h, N, deadline, tau0=5.0, tau1=40.0, lr=0.002, sweeps=20):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    
    n_steps = 0
    while time.time() < deadline:
        # Schedule tau
        frac = min(1.0, n_steps / (sweeps * N))
        tau = tau0 + (tau1 - tau0) * frac
        
        grad = softmax_gradient(h, tau)
        h = proj_sum(h - lr * grad, N / 2.0)
        n_steps += 1
        
        if n_steps % 50 == 0:
            pk = float(np.max(fft_corr(h)))
            if pk < best_pk:
                best_pk = pk
                best_h = h.copy()
                
        # Decrease learning rate slowly
        if n_steps % 200 == 0:
            lr *= 0.95
    
    # Final evaluation
    pk = float(np.max(fft_corr(h)))
    if pk < best_pk:
        best_pk = pk
        best_h = h.copy()
    
    return best_h, best_pk

def binary_local_search(h, N, deadline, n_trials=500):
    h = fix_count(h.astype(float).copy(), N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    s_arr = np.arange(-(N-1), N)
    
    trials = 0
    while time.time() < deadline and trials < n_trials:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        
        ps = p + s_arr
        pm = p - s_arr
        qs = q + s_arr
        qm = q - s_arr
        vps = (ps >= 0) & (ps < N)
        vpm = (pm >= 0) & (pm < N)
        vqs = (qs >= 0) & (qs < N)
        vqm = (qm >= 0) & (qm < N)
        
        d = np.zeros(2*N - 1)
        pc = np.clip(ps, 0, N-1)
        pmc = np.clip(pm, 0, N-1)
        qsc = np.clip(qs, 0, N-1)
        qmc = np.clip(qm, 0, N-1)
        
        d[vps] -= (1.0 - h[pc[vps]])
        d[vpm] += h[pmc[vpm]]
        d[vqs] += (1.0 - h[qsc[vqs]])
        d[vqm] -= h[qmc[vqm]]
        
        nc = corr + d
        npk = float(nc.max())
        
        if npk < best_pk or npk < float(corr.max()):
            h[p] = 0
            h[q] = 1
            corr = nc
            if npk < best_pk:
                best_pk = npk
                best_h = h.copy()
        trials += 1
    
    return best_h, best_pk

def anneal_binary(h, N, deadline, T0=0.05, cool=0.9995, max_no_improve=500):
    h = fix_count(h.astype(float).copy(), N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    T = T0
    no_imp = 0
    iterations = 0
    
    while time.time() < deadline and T > 0.0005:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        
        ps = p + s_arr; pm = p - s_arr
        qs = q + s_arr; qm = q - s_arr
        vps = (ps >= 0) & (ps < N)
        vpm = (pm >= 0) & (pm < N)
        vqs = (qs >= 0) & (qs < N)
        vqm = (qm >= 0) & (qm < N)
        
        d = np.zeros(2*N - 1)
        pc = np.clip(ps, 0, N-1); pmc = np.clip(pm, 0, N-1)
        qsc = np.clip(qs, 0, N-1); qmc = np.clip(qm, 0, N-1)
        
        d[vps] -= (1.0 - h[pc[vps]])

        d[vqs] += (1.0 - h[qsc[vqs]])
        d[vqm] -= h[qmc[vqm]]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin == 1)[0]
        np.random.shuffle(idx)
        h_bin[idx[:n1 - target]] = 0
    elif n1 < target:
        idx = np.where(h_bin == 0)[0]
        np.random.shuffle(idx)
        h_bin[idx[:target - n1]] = 1
    return h_bin

def softmax_gradient(h, tau):
    n = len(h)
    corr = fft_corr(h)
    mc = float(np.max(corr))
    log_w = tau * (corr - mc)
    w = np.exp(log_w)
    W = float(w.sum())
    if W < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= W
    shifts = np.arange(2 * n - 1) - (n - 1)
    s1 = np.arange(n)[:, None] - shifts[None, :]
    s2 = np.arange(n)[:, None] + shifts[None, :]
    valid1 = (s1 >= 0) & (s1 < n)
    valid2 = (s2 >= 0) & (s2 < n)
    oh = 1.0 - h
    c1 = np.clip(s1, 0, n - 1)
    c2 = np.clip(s2, 0, n - 1)
    g1 = np.sum(w[None, :] * oh[c1] * valid1, axis=1)
    g2 = np.sum(w[None, :] * h[c2] * valid2, axis=1)
    return (g1 - g2) / n

def softmax_descent(h, N, deadline, tau0=8.0, tau1=50.0, lr=0.003):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    steps = 0
    while time.time() < deadline:
        frac = min(1.0, steps / max(1, 1000))
        tau = tau0 + (tau1 - tau0) * frac
        grad = softmax_gradient(h, tau)
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        steps += 1
        if steps % 40 == 0:
            pk = float(np.max(fft_corr(h)))
            if pk < best_pk:
                best_pk = pk
                best_h = h.copy()
        if steps % 500 == 0:
            lr *= 0.96
    pk = float(np.max(fft_corr(h)))
    if pk < best_pk:
        best_pk = pk
        best_h = h.copy()
    return best_h, best_pk

def swap_delta(h, N, p, q):
    s_arr = np.arange(-(N - 1), N)
    ps, pm = p + s_arr, p - s_arr
    qs, qm = q + s_arr, q - s_arr
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    d = np.zeros(2 * N - 1)
    d[vps] -= (1.0 - h[np.clip(ps, 0, N - 1)[vps]])
    d[vpm] += h[np.clip(pm, 0, N - 1)[vpm]]
    d[vqs] += (1.0 - h[np.clip(qs, 0, N - 1)[vqs]])
    d[vqm] -= h[np.clip(qm, 0, N - 1)[vqm]]
    return d

def anneal_binary(h, N, deadline, T0=0.05, cool=0.9994, max_no_imp=600):
    h = fix_count(h.astype(float).copy(), N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    T = T0
    no_imp = 0
    while time.time() < deadline and T > 0.0005:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        d = swap_delta(h, N, p, q)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            h[p] = 0
            h[q] = 1
            corr = nc
            cur_pk = npk
            if npk < best_pk:
                best_pk = npk
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        T *= cool
        if no_imp > max_no_imp:
            h = best_h.copy()
            corr = fft_corr(h)
            cur_pk = best_pk
            no_imp = 0
            T = max(T * 2.5, 0.01)
    return best_h, best_pk

def interpolate_h(h_old, N_new):
    N_old = len(h_old)

    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
