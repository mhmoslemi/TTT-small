    grad = np.zeros(N)

        m = p - N + 1

    return grad

import numpy as np
import time

def fft_corr_full(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    T = np.fft.rfft((1.0 - h)[::-1], L)
    return np.fft.irfft(H * T, L)

def project_01(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, w):
    N = len(h)
    grad = np.zeros(N)
    for p in range(2 * N - 1):
        s = w[p]
        if s < 1e-14:
            continue
        m = p - N + 1
        if m >= 0:
            n = N - m
            grad[m:] += s * (1.0 - h[:n])
            grad[:n] -= s * h[m:]
        else:
            mm = -m
            n = N - mm
            grad[:n] += s * (1.0 - h[mm:])
            grad[mm:] -= s * h[:n]
    return grad

def softmax_subgrad(h, N, deadline, beta_init=8.0, lr_init=0.015, momentum=0.85):
    target = N / 2.0
    best_h = h.copy()
    best_obj = float(np.max(fft_corr_full(h)))
    x = h.astype(float).copy()
    v = np.zeros(N)
    lr = lr_init
    beta = beta_init
    it = 0
    while time.time() < deadline:
        corr = fft_corr_full(x)
        bc = beta * corr - beta * np.max(corr)
        exp_bc = np.exp(bc)
        w = exp_bc / exp_bc.sum()
        grad = compute_grad(x, w)
        v = momentum * v - lr * grad
        x_new = project_01(x + v, target)
        corr_new = fft_corr_full(x_new)
        obj_new = float(np.max(corr_new))
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = x_new.copy()
        x = x_new
        it += 1
        if it % 300 == 0:
            lr *= 0.96
            momentum = min(0.96, 0.80 + it * 0.0001)
            beta *= 1.2
    return best_h, best_obj

def compute_effect_matrix(h, N):
    """Compute effect of flipping each position from 1 to 0."""
    L = 2 * N - 1
    E = np.zeros((N, L))
    for p in range(N):
        for m in range(-(N-1), N):
            idx = m + N - 1
            val = 0.0
            if 0 <= p + m < N:
                val -= (1.0 - h[p + m])
            if 0 <= p - m < N:
                val += h[p - m]
            E[p, idx] = val
    return E

    nh = N // 2
