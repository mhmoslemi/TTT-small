def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    hp = np.where(vp, h[pp_c], 0.0)
    hm = np.where(vm, h[pm_c], 0.0)
    return -(1.0 - hp) * vp.astype(float) + hm * vm.astype(float)

def soft_max_grad_c(corr, h, N, T):
    corr_s = corr * (2.0 / N)
    m = corr_s.max()
    exp_T = np.exp(T * (corr_s - m))
    w = exp_T / exp_T.sum()
    A = np.zeros(N)
    B = np.zeros(N)
    for s in range(-(N-1), N):
        wi = w[s + N - 1]
        if wi < 1e-12:
            continue
        j_lo = max(0, -s)
        j_hi = min(N, N - s)
        if j_lo < j_hi:
            A[j_lo:j_hi] += wi * h[j_lo+s:j_hi+s]
        j_lo2 = max(0, s)
        j_hi2 = min(N, s + N)
        if j_lo2 < j_hi2:
            B[j_lo2:j_hi2] += wi * h[j_lo2-s:j_hi2-s]
    return (2.0 / N) * (1.0 - A - B), float(m + np.log(float(exp_T.sum())) / T)

def adam_opt(h, N, deadline, T0=20, T1=100, n_iter_max=2000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h))) * (2.0/N)
    m_adam = np.zeros(N)
    v_adam = np.zeros(N)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    while time.time() < deadline and t < n_iter_max:
        T = T0 + (T1 - T0) * min(1.0, t / (n_iter_max * 0.8))
        corr = fft_corr(h)
        grad, fval = soft_max_grad_c(corr, h, N, T)
        t += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad**2
        m_hat = m_adam / (1 - beta1**t)
        v_hat = v_adam / (1 - beta2**t)
        lr = 0.05 / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h - lr * m_hat, target)
        h = h_new
        c = float(np.max(fft_corr(h))) * (2.0/N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def binary_opt(h, N, deadline, max_sweeps=2000, sample_n=300):
    h = h.astype(float).copy()
    target = N / 2.0
    best_c = float(np.max(fft_corr(h))) * (2.0/N)
    best_h = h.copy()
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max()) * (2.0/N)
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        # For flip p(1→0), q(0→1): new_corr[k] ≈ corr[k] - D[p,k] - D[q,k]
        # (ignoring cross terms for 1st approximation)
        # We want to minimize max_k new_corr[k]
        if len(ones) > sample_n:
            perm = np.random.permutation(len(ones))[:sample_n]
            p_list = ones[perm]
        else:
            p_list = ones
        if len(zeros) > sample_n:
            perm_z = np.random.permutation(len(zeros))[:sample_n]
            q_list = zeros[perm_z]
        else:
            q_list = zeros
        
        best_score = P + 1.0
        best_pq = None
        for p in p_list:
            base = corr - D[p, :]
            cand = base[None, :] - D[q_list, :]
            pk = cand.max(axis=1)
            mi = int(np.argmin(pk))
            if pk[mi] < best_score:
                best_score = float(pk[mi])
                best_pq = (int(p), int(q_list[mi]))
        
        if best_pq is None or (no_improve > 100 and np.random.random() < 0.1):
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
        else:
            p, q = best_pq
        h[p] = 0.0
        h[q] = 1.0
        if no_improve > 500:
            break
    return best_h, best_c

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int((h > 0.5).sum())
    if n1 > nh:
        idx = np.where(h > 0.5)[0]
        rm = idx[np.argsort(np.random.random(len(idx)))][nh:]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h < 0.5)[0]
        add = idx[np.argsort(np.random.random(len(idx)))][:nh - n1]
        h[add] = 1.0
    return h
