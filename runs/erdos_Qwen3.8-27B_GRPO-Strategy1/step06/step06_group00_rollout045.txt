import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_swap_deltas(h, N):
    """Precompute the effect of swapping h[i]=1 to 0 and h[j]=0 to 1 on each correlation value.
    Returns: for each i with h[i]=1, delta_one[i, k] = change in corr[k]
             for each j with h[j]=0, delta_zero[j, k] = change in corr[k]
    """
    corr = fft_corr(h)
    center = N - 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # For each position i, the contribution to corr[k] is:
    # corr[k] includes h[i] * (1-h[i+k-center]) (approximately)
    # Swapping h[i] from 1 to 0 changes corr[k] by -(1-h[i+k-center]) for those k
    # Swapping h[j] from 0 to 1 changes corr[k] by +1*(1-h[j+k-center]) for those k... 
    # Actually it's more subtle. Let me think...
    
    # corr[k] = sum_i h[i] * (1-h[(i+k-center) mod ...]) - but it's "full" not "circular"
    # For full correlation: corr[k] = sum_{valid i} h[i] * (1-h[i+k-(N-1)])
    
    # The effect of changing h[i] from 1 to 0:
    #   delta_corr[k] -= (1-h[i+k-(N-1)]) for valid (i+k-(N-1))
    #   AND ALSO: for all k' where i appears as the (1-h[...]) term:
    #   delta_corr[k'] -= h[i'-...]  ... this is getting complicated
    
    # Simpler: just compute delta by the formula
    # If h[i]: 1->0, the change in corr is:
    #   For the h part: -(1 - h[i+s]) for each shift s where i+s is in [0,N-1]
    #   For the (1-h) part: -h[i-s] for each shift s where i-s is in [0,N-1] (these become 1-h[i-s] increasing)
    # Wait, I need to be more careful.
    
    # corr[k] = sum_i h[i] * (1-h[i + (k-(N-1))])
    # Let s = k - (N-1), so k = s + (N-1), s in [-(N-1), N-1]
    # corr[s+(N-1)] = sum_i h[i] * (1-h[i+s]) where i+s in [0,N-1]
    
    # If h[i] changes from 1 to 0:
    #   delta = -(1-h[i+s]) for s where i+s in [0,N-1], s != 0 (s=0 gives 0 for binary)
    #   AND for s' where i = i'+s', i.e., s' = i-i', h[i'] changes contribution:
    #   Actually no, h[i] only appears in the first factor. (1-h[...]) involves other indices.
    #   
    # Wait, h[i] also appears in (1-h[i-s]) terms for other positions!
    # corr[s+(N-1)] = sum_i' h[i'] * (1-h[i'+s])
    # h[i] appears when i'+s = i, i.e., i' = i-s
    # So: corr[s+(N-1)] includes h[i-s] * (1-h[i]) for valid i-s
    # 
    # So changing h[i] from 1 to 0:
    #   Effect on term h[i]*(1-h[i+s]): goes from (1-h[i+s]) to 0, delta = -(1-h[i+s])
    #   Effect on term (1-h[i]) in h[i-s]*(1-h[i]): goes from h[i-s]*0 to h[i-s]*1, delta = +h[i-s]
    #   Total delta for shift s: -(1-h[i+s]) + h[i-s] (with bounds)
    
    # This is O(N) per position, O(N^2) total for all positions. For N=380, that's 144K, fine.
    
    n_ones = len(ones)
    n_zeros = len(zeros)
    n_corr = 2 * N - 1
    
    delta_one = np.zeros((n_ones, n_corr))
    delta_zero = np.zeros((n_zeros, n_corr))
    
    for ii, i in enumerate(ones):
        for s in range(-(N-1), N):
            k = s + center
            # h[i] contributes to corr[k] via:
            # (a) h[i] * (1-h[i+s]) if 0<=i+s<N
            # (b) h[i-s] * (1-h[i]) if 0<=i-s<N (h[i] is in the (1-h) factor)
            d = 0.0
            if 0 <= i + s < N:
                d -= (1 - h[i + s])
            if 0 <= i - s < N:
                d += h[i - s]
            delta_one[ii, k] = d
    
    for jj, j in enumerate(zeros):
        for s in range(-(N-1), N):
            k = s + center
            d = 0.0
            if 0 <= j + s < N:
                d += (1 - h[j + s])
            if 0 <= j - s < N:
                d -= h[j - s]
            delta_zero[jj, k] = d
    
    return corr, delta_one, delta_zero, ones, zeros

def binary_search_fast(h, N, deadline, max_iters=2000, sample_sz=40):
    """Fast binary search using precomputed swap deltas."""
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    
    corr, delta_one, delta_zero, ones, zeros = compute_swap_deltas(h, N)
    best_c = np.max(corr)
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        P = np.max(corr)
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 30:
            # Recompute deltas (shouldn't happen if we update correctly, but safety)
            corr, delta_one, delta_zero, ones, zeros = compute_swap_deltas(h, N)
            stall = 0
        
        n_o = len(ones)
        n_z = len(zeros)
        if n_o < 2 or n_z < 2:
            break
        
        # Sample
        so = min(n_o, sample_sz)
        sz = min(n_z, sample_sz)
        oi = np.random.choice(n_o, so, replace=False)
        zi = np.random.choice(n_z, sz, replace=False)
        
        # new_corr = corr + delta_one[oi] - delta_zero[zi] (broadcast)
        new = corr[None, None, :] + delta_one[oi][None, :, :] - delta_zero[zi][:, None, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        
        if mv < P - 1e-12:
            i_pos = ones[oi[mi[1]]]
            j_pos = zeros[zi[mi[0]]]
            h[i_pos] = 0
            h[j_pos] = 1
            # Update corr
            corr += delta_one[oi[mi[1]]] - delta_zero[zi[mi[0]]]
            # Update delta arrays (expensive, just recompute periodically)
            if it % 50 == 0:
                corr, delta_one, delta_zero, ones, zeros = compute_swap_deltas(h, N)
        elif stall > 50 and np.random.random() < 0.1:
            # Random perturbation
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p] = 0
            h[q] = 1
            corr, delta_one, delta_zero, ones, zeros = compute_swap_deltas(h, N)
            stall = 0
        else:
            break
    
    return best_h, best_c * (2.0 / N)

def nesterov_gradient(h, N, deadline, lr0=0.015, top_k=80):
    """Nesterov with top-k subgradient."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h))) * dx
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    center = N - 1
    
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(np.max(corr)) * dx
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        top = np.argpartition(corr, -top_k)[-top_k:]
        top = top[corr[top] > np.max(corr) * 0.5]  # only "active" ones
        if len(top) == 0:
            top = np.array([np.argmax(corr)])
        
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[L:] += (1.0 - h[:N-L])
                grad[:N-L] -= h[L:]
            elif -N < L < 0:
                M = -L
                grad[:N-M] += (1.0 - h[M:])
                grad[M:] -= h[:N-M]
        grad /= len(top)
        
        h_new = project_01(h - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        lr *= 0.9997
        beta = min(0.96, beta + 0.00005)
    
    return best_h, best_pk
