def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def ensure_sum_binary(h, N):

    n1 = int(h_bin.sum())

        return h_bin

    else:

        idx = rng.choice(zeros, nh - n1, replace=False)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

    return best_h, best_cv * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h, N, deadline, k=100, lr0=0.018, beta0=0.87):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr = max(0.001, lr * 0.96)
            beta = min(0.95, beta0 + it * 0.0003)
        if it % 500 == 0 and lr < 0.01:
            lr = 0.012
            beta = 0.85
    return best_h, best_pk * dx

def upsample(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def perturb_binary(h, n_flips, rng):
    hp = (h > 0.5).astype(float)
    ones = np.where(hp == 1)[0]
    zeros = np.where(hp == 0)[0]
    for _ in range(n_flips):
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        hp[p], hp[q] = 0.0, 1.0
        ones = np.where(hp == 1)[0]
        zeros = np.where(hp == 0)[0]
    return hp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Greedy at N0
    h_bin = ensure_sum_binary(h0, N0)
    d1 = min(deadline - 550, t0 + 50)
    h1, c51 = greedy_search(h_bin, N0, d1, max_sweeps=300, sample_frac=1.0)
    if c51 < best_c5:
        best_c5, best_h = c51, h1.copy()

    # Phase 2: Annealed search at N0, perturbation 1
    if time.time() < deadline - 450:
        rng2 = np.random.RandomState(seed + 11)
        hp = perturb_binary(best_h, 4, rng2)
        d2 = min(deadline - 400, time.time() + 130)
        h2, c52 = anneal_search(hp, N0, d2, max_sweeps=600, t_init=4.0, t_final=0.003, sample_frac=0.8)
        if c52 < best_c5:
            best_c5, best_h = c52, h2.copy()

    # Phase 3: Annealed search at N0, perturbation 2 (larger)
    if time.time() < deadline - 350:
        rng3 = np.random.RandomState(seed + 23)
        hp3 = perturb_binary(best_h, 8, rng3)
        d3 = min(deadline - 300, time.time() + 150)
        h3, c53 = anneal_search(hp3, N0, d3, max_sweeps=700, t_init=6.0, t_final=0.002, sample_frac=0.7)
        if c53 < best_c5:
            best_c5, best_h = c53, h3.copy()

    # Phase 4: Upsample to 2*N0, annealed search
    N1 = 2 * N0
    if time.time() < deadline - 180:
        h_up = ensure_sum_binary(upsample(best_h, N1), N1)
        d4 = min(deadline - 130, time.time() + 120)
        h4, c54 = anneal_search(h_up, N1, d4, max_sweeps=350, t_init=2.5, t_final=0.005, sample_frac=0.55)
        if c54 < best_c5:
            best_c5, best_h = c54, h4.copy()

    # Phase 5: Continuous Nesterov at N0
    if time.time() < deadline - 80:
        rng5 = np.random.RandomState(seed + 37)
        hc = (best_h > 0.5).astype(float)
        hc = hc + rng5.randn(N0) * 0.1
        hc = project_sum(hc, N0 / 2.0)
        d5 = min(deadline - 50, time.time() + 60)
        h5, c55 = nesterov_opt(hc, N0, d5, k=120, lr0=0.02, beta0=0.84)
        if c55 < best_c5:
            best_c5, best_h = c55, h5.copy()

    # Phase 6: Second Nesterov with different init
    if time.time() < deadline - 40:
        Nn = len(best_h)
        hc2 = upsample(best_h, Nn) if len(best_h) != Nn else best_h.astype(float).copy()
        hc2 = hc2 + np.random.randn(Nn) * 0.06
        hc2 = project_sum(hc2, Nn / 2.0)
        d6 = min(deadline - 25, time.time() + 40)
        h6, c56 = nesterov_opt(hc2, Nn, d6, k=90, lr0=0.012, beta0=0.88)
        if c56 < best_c5:
            best_c5, best_h = c56, h6.copy()

    # Phase 7: Final greedy polish
    Nf = len(best_h)
    if time.time() < deadline - 3:
        hF = ensure_sum_binary(best_h, Nf)
        sf = 1.0 if Nf <= N0 else 0.5
        hF2, c5F = greedy_search(hF, Nf, deadline - 1, max_sweeps=250, sample_frac=sf, stall_limit=50)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
