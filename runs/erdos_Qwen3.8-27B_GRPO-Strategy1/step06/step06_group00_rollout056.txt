def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    # C[i, k] = change in corr[k] when h[i] goes 0->1
    # For lag index k (0 to 2N-2), lag = k - (N-1)
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals  # i - k (for the (1-h) part)
    i_pk = pos + k_vals  # i + k (for the h part)
    valid_a = (i_mk >= 0) & (i_mk < N)
    valid_b = (i_pk >= 0) & (i_pk < N)
    A = np.where(valid_a, 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where(valid_b, h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

    i = np.arange(N)
    return ((i * alpha) % 1.0 < 0.5).astype(float)

    """Fix the sum of a binary vector to exactly target"""
    n = len(h)
    h = (h > 0.5).astype(float)
    s = int(h.sum())
    if s == target:
        return h
    if s > target:

        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[:s-target]] = 0
    else:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[:target-s]] = 1
    return h

        P = float(np.max(corr))

            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if len(ones) < 2 or len(zeros) < 2:
            break
        
        C = compute_C_mat(h, N)
        Co = C[ones]  # (n_ones, 2N-1)

    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()

    m = np.zeros(N)

    beta = 0.9
    it = 0

        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:

        it += 1
        if it % 200 == 0:

def compute_C_mat(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    valid_a = (i_mk >= 0) & (i_mk < N)
    valid_b = (i_pk >= 0) & (i_pk < N)
    A = np.where(valid_a, 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where(valid_b, h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

                kz = np.random.choice(zeros, ks, replace=False)

                h[kz] = 1

        Cz = C[zeros]

        best_mi = -1
        best_mj = -1

            Cz_c = Cz[ci:cj]
            # (n_ones, chunk, L2N)
            new = corr[None, None, :] - Co[:, None, :] + Cz_c[None, :, :]
            newP = new.max(axis=2)
            mi, mci = np.unravel_index(np.argmin(newP), newP.shape)
            if newP[mi, mci] < best_newP:
                best_newP = newP[mi, mci]
                best_mi = mi
                best_mj = ci + mci
        
        i = ones[best_mi]
        j = zeros[best_mj]
        
        if (i, j) in tabu:
            if stall > max_stall // 2:
                # Allow tabu move to escape
                pass
            else:

                # ... for simplicity, just kick
                ones2 = np.where(h == 1)[0]
                zeros2 = np.where(h == 0)[0]
                ks = min(2, len(ones2)//4, len(zeros2)//4)
                ko = np.random.choice(ones2, ks, replace=False)
                kz = np.random.choice(zeros2, ks, replace=False)
                h[ko] = 0
                h[kz] = 1
                continue

            tabu.pop()
        # Also add reverse as tabu
        tabu.add((j, i))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    va = (i_mk >= 0) & (i_mk < N)
    vb = (i_pk >= 0) & (i_pk < N)
    A = np.where(va, 1.0 - h[np.clip(i_mk, 0, N-1)], 0.0)
    B = np.where(vb, h[np.clip(i_pk, 0, N-1)], 0.0)
    return A - B

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, target):
    h = (h > 0.5).astype(float)
    s = int(h.sum())
    if s == target:
        return h
    if s > target:

        h[idx[:s - target]] = 0
    else:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - s]] = 1
    return h

def beatty(N, alpha):
    i = np.arange(N)
    return ((i * alpha) % 1.0 < 0.5).astype(float)
