import numpy as np
import time

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    return float(np.max(full_corr(h))) * (2.0 / N)

def compute_A(h, N):
    L = 2 * N - 1
    shifts = np.arange(L) - (N - 1)
    A = np.zeros((N, L))
    for i in range(N):
        im = i - shifts
        ip = i + shifts
        vm = (im >= 0) & (im < N)
        vp = (ip >= 0) & (ip < N)
        A[i, :] = np.where(vm, h[np.clip(im, 0, N-1)], 0.0) - np.where(vp, 1.0 - h[np.clip(ip, 0, N-1)], 0.0)
    return A

def binary_search(h, N, deadline, max_iters=3000, reheat_every=200, T0=0.3):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = float(full_corr(h).max())
    T_decay = 0.93
    for restart in range(40):
        if time.time() > deadline:
            break
        A = compute_A(h, N)
        c = full_corr(h)
        cur_max = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        stall = 0
        T = T0 * (T_decay ** restart)
        for it in range(max_iters):
            if time.time() > deadline:
                break
            if stall > reheat_every:
                break
            A_zeros = A[zeros, :]
            best_delta = 1e10
            best_pq = None
            for p in ones:
                base = c + A[p, :]
                cands_max = np.max(base[None, :] - A_zeros, axis=1)
                idx = np.argmin(cands_max)
                d = cands_max[idx] - cur_max
                if d < best_delta:
                    best_delta = d
                    best_pq = (p, zeros[idx])
            accept = False
            if best_pq is not None and best_delta < -1e-12:
                accept = True
            elif T > 0.005:

                p, q = best_pq
