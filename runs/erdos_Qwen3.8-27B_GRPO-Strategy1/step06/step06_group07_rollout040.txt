import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(60):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 1e-15
            n = mask.sum()
            if n == 0:
                break
            h[mask] -= diff / n
        else:
            mask = h < 1.0 - 1e-15
            n = mask.sum()
            if n == 0:
                break
            h[mask] += (-diff) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_R(h, N, k):
    g = np.zeros(N)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        if k >= N:
            return g
        g[:N-k] += (1.0 - h[k:])
        g[k:] -= h[:N-k]
    else:
        M = -k
        if M >= N:
            return g
        g[M:] += (1.0 - h[:N-M])
        g[:N-M] -= h[M:]
    return g

def continuous_opt(h, N, deadline, top_k=5, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    lr = lr0
    beta = 0.9
    m = np.zeros(N)
    stall = 0
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        active = np.argpartition(corr, -(top_k))[-(top_k):]
        cw = corr[active] - corr[active].min()
        w = np.exp(cw * 4.0)
        w /= w.sum()
        g = np.zeros(N)
        for ai, wi in zip(active, w):
            k = int(ai - (N - 1))
            g += wi * grad_R(h, N, k)
        g /= top_k
        m = beta * m + g
        h_new = project_box_sum(h - lr * m, target)
        c = float(np.max(fft_corr(h_new)))
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.975
            if lr < 0.0008:
                lr = 0.003
                beta = 0.82
            if stall > 200:
                h += np.random.randn(N) * 0.025
                h = project_box_sum(h, target)
                stall = 0
        if stall > 500:
            break
    return best_h, best_c * dx

def binary_opt(h, N, deadline, max_sweeps=500, full_enum=True, perturb_stall=60):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        rng = np.random.RandomState(7)
        if n1 > nh:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones, n1 - nh, replace=False)] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, nh - n1, replace=False)] = 1
    dx = 2.0 / N
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    shifts = np.arange(2 * N - 1) - (N - 1)
    k_idx = shifts[None, :]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_ones, n_zeros = len(ones), len(zeros)
        # D: change in corr when ones[i] goes 1->0
        p_idx = ones[:, None]
        pk = p_idx + k_idx
        pm = p_idx - k_idx
        vk = (pk >= 0) & (pk < N)
        vm = (pm >= 0) & (pm < N)
        D = -(1.0 - np.where(vk, h[np.clip(pk, 0, N-1)], 0.0)) * vk.astype(float) + np.where(vm, h[np.clip(pm, 0, N-1)], 0.0) * vm.astype(float)
        # E: change in corr when zeros[j] goes 0->1
        q_idx = zeros[:, None]
        qk = q_idx + k_idx
        qm = q_idx - k_idx
        vqk = (qk >= 0) & (qk < N)
        vqm = (qm >= 0) & (qm < N)
        E = (1.0 - np.where(vqk, h[np.clip(qk, 0, N-1)], 0.0)) * vqk.astype(float) - np.where(vqm, h[np.clip(qm, 0, N-1)], 0.0) * vqm.astype(float)
        # Determine sampling
        use_full = full_enum and (n_ones <= 250 and n_zeros <= 250)
        if use_full:
            Ds, Es, ps, qs = D, E, ones, zeros
        else:
            ns = max(30, int(n_ones * 0.6))
            pi = np.random.choice(n_ones, ns, replace=False)
            Ds, ps = D[pi], ones[pi]
            nz = min(n_zeros, 250)
            qi = np.random.choice(n_zeros, nz, replace=False)
            Es, qs = E[qi], zeros[qi]
        best_new_P = P
        best_pq = None
        for i in range(Ds.shape[0]):
            base = corr - Ds[i]
            cand = base[None, :] + Es
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (ps[i], qs[mi])
        if best_pq is None:
            if stall > perturb_stall:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return best_h, best_c * dx

def upsample(h, Ns, Nd):
    xs = np.linspace(0, 2, Ns, endpoint=False)
    xd = np.linspace(0, 2, Nd, endpoint=False)
    return np.interp(xd, xs, h)
