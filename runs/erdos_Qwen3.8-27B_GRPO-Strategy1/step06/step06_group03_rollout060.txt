    """Precompute swing vectors for all positions"""
    # For position p, swing[p, k] = h[p+k] - h[p-k] (with boundary handling)
    # This gives the contribution to correlation change
    center = N - 1

            val = 0
            if 0 <= p+L < N:

            if 0 <= p-L < N:
                val -= h[p-L]

def build_swing(h, N):
    n_corr = 2 * N - 1
    swing = np.zeros((N, n_corr))

        if mk == 0:
            swing[:, k] = 2.0 * h
        elif mk > 0:
            swing[:N-mk, k] = h[mk:]

def adjust_binary(h, N, rng):

        to_one = idx[rng.choice(len(idx), target - n1, replace=False)]

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
