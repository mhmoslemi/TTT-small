import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin==1)[0]
        sel = np.random.choice(len(idx), n1-target, replace=False)
        h_bin[idx[sel]] = 0
    elif n1 < target:
        idx = np.where(h_bin==0)[0]
        sel = np.random.choice(len(idx), target-n1, replace=False)
        h_bin[idx[sel]] = 1
    return h_bin

def swap_delta(h, N, p, q, s_arr):
    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
    vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
    d = np.zeros(2*N-1)
    pc=np.clip(ps,0,N-1); pmc=np.clip(pm,0,N-1)
    qsc=np.clip(qs,0,N-1); qmc=np.clip(qm,0,N-1)
    d[vps] -= (1.0-h[pc[vps]])
    d[vpm] += h[pmc[vpm]]
    d[vqs] += (1.0-h[qsc[vqs]])
    d[vqm] -= h[qmc[vqm]]
    return d

def hill_climb(h, N, max_iters, deadline=None):
    s_arr = np.arange(-(N-1), N)
    corr = fft_corr(h)
    cur_pk = float(corr.max())
    best_pk = cur_pk
    best_h = h.copy()
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    for _ in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        d = swap_delta(h, N, p, q, s_arr)
        nc = corr + d
        npk = float(nc.max())
        if npk < cur_pk - 1e-15:
            h[p]=0; h[q]=1
            corr=nc; cur_pk=npk
            ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
            if npk < best_pk - 1e-15:
                best_pk=npk; best_h=h.copy()
    return best_h, best_pk, h, cur_pk

def iterated_ls(h0, N, deadline, kick_sizes=(3,5,7), max_hc=8000):
    h = h0.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    s_arr = np.arange(-(N-1), N)
    kick_cycle = 0
    stall = 0
    while time.time() < deadline:
        h_c, pk_c = h.copy(), best_pk
        h, pk, _, cur_pk = hill_climb(h, N, max_hc, deadline)
        if pk < best_pk - 1e-15:
            best_pk = pk; best_h = h.copy(); stall = 0
        else:
            stall += 1
        if time.time() > deadline:
            break
        h = h_c.copy()
        ks = kick_sizes[kick_cycle % len(kick_sizes)]
        kick_cycle += 1
        if stall > 4:
            ks = min(ks + 3, 15)
            stall = 0
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        k = min(ks, len(ones), len(zeros))
        fo = np.random.choice(ones, k, replace=False)
        fi = np.random.choice(zeros, k, replace=False)
        h[fo] = 0
        h[fi] = 1
    return best_h, best_pk

def continuous_opt(h_init, N, deadline, lr0=0.003):
    h = h_init.astype(float).copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_o = np.zeros(N); v_o = np.zeros(N)
    t = 0
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    s_arr = np.arange(-(N-1), N)
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-8:
            best_pk = pk; best_h = h.copy()
        beta = 8.0 * (1.0 + t/400)
        cm = corr.max()
        ex = np.exp(beta*(corr-cm))
        w = ex / ex.sum()
        grad = np.zeros(N)
        for k in range(2*N-1):
            wk = w[k]
            if wk < 1e-13:
                continue
            m = k - (N-1)
            if m >= 0:
                grad[:N-m] += wk*(1.0-h[m:])
                grad[m:] -= wk*h[:N-m]
            elif m < 0:
                p = -m
                grad[p:] += wk*(1.0-h[:N-p])
                grad[:N-p] -= wk*h[p:]
        nact = float(w.sum())
        grad /= nact
        m_o = b1*m_o + (1-b1)*grad
        v_o = b2*v_o + (1-b2)*grad*grad
        mh = m_o/(1-b1**t); vh = v_o/(1-b2**t)
        lr = lr0/(1+0.05*t)
        h = proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
    return best_h, best_pk

def binary_anneal(h_bin, N, deadline, T0=0.1, cool=0.9997, max_stall=3000):
    h = h_bin.astype(float).copy()
    s_arr = np.arange(-(N-1), N)
    corr = fft_corr(h)
    best_pk = float(corr.max()); best_h = h.copy(); cur_pk = best_pk
    T = T0; stall = 0
    while time.time() < deadline and T > 0.001:
        p = int(np.random.choice(np.where(h==1)[0]))
        q = int(np.random.choice(np.where(h==0)[0]))
        d = swap_delta(h, N, p, q, s_arr)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        if delta < 0 or np.random.random() < np.exp(-delta/max(T,1e-9)):
            h[p]=0; h[q]=1; corr=nc; cur_pk=npk
            if npk < best_pk:
                best_pk=npk; best_h=h.copy(); stall=0
            else:
                stall+=1
        else:
            stall+=1
        T *= cool
        if stall > max_stall:
            stall = 0
            T = min(T*3, T0)
            if npk > best_pk:
                h = best_h.copy(); corr = fft_corr(h); cur_pk = best_pk
    return best_h, best_pk
