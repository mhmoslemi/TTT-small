import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0

def swap_delta(p, q, h, N):
    """Compute change in full correlation when h[p]:1->0 and h[q]:0->1."""
    L = 2 * N - 1

        cng = 0.0

            cng -= h[q + lag]
        if 0 <= q - lag < N:
            cng += (1.0 - h[q - lag])
        dC[idx] = cng

        d2 = min(deadline - 250, time.time() + 100)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.03
        hc = project_sum(hc, N0 / 2.0)
        hn, c2 = prox_grad(hc, N0, d2, lr0=0.015, k_top=50)
        if c2 < best_c5:
            best_c5, best_h = c2, hn.copy()

    # Phase 3: SA binary again from improved
    if time.time() < deadline - 200:
        d3 = min(deadline - 150, time.time() + 180)
        h3, _ = sa_binary((best_h > 0.5).astype(float), N0, d3, max_iter=200000, T0=0.5, T1=0.005)
        c3 = compute_c5(h3, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Try different resolutions
    for N_try in [360, 420, 480, 560]:
        if time.time() > deadline - 80:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        h_bt = fix_binary(h_bt, N_try // 2)
        sub_dl = min(deadline - 40, time.time() + 60)
        h_rt, _ = sa_binary(h_bt, N_try, sub_dl, max_iter=80000, T0=0.8, T1=0.03)
        c_rt = compute_c5(h_rt, N_try)
        if c_rt < best_c5:
            best_c5, best_h = c_rt, h_rt.copy()

    # Phase 5: Golden ratio restart + SA
    if time.time() < deadline - 100:
        N_g = N0
        h_g = golden_init(N_g)
        h_g = fix_binary(h_g, N_g // 2)
        d5 = min(deadline - 60, time.time() + 80)
        h_g, _ = sa_binary(h_g, N_g, d5, max_iter=100000, T0=1.2, T1=0.02)
        c_g = compute_c5(h_g, N_g)
        if c_g < best_c5:
            best_c5, best_h = c_g, h_g.copy()

    # Phase 6: Final SA polish
    if time.time() < deadline - 10:
        d6 = deadline - 3
        h6, _ = sa_binary((best_h > 0.5).astype(float), len(best_h), d6, max_iter=150000, T0=0.3, T1=0.001)
        c6 = compute_c5(h6, len(best_h))
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    # Final continuous micro-tweak
    if time.time() < deadline - 5:
        hF = best_h.astype(float) + np.random.randn(len(best_h)) * 0.01
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, cF = prox_grad(hF, len(best_h), deadline - 1, lr0=0.005, k_top=60)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
