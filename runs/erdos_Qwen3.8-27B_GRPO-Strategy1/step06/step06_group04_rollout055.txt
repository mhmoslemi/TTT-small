```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    """Project h to [0,1]^N with sum = target using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -max(h.max()-0, 0.01), 1 - min(h.min(), 1.0) + 0.01
    # Actually, we need shift such that sum(clip(h - shift, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_gradient(corr, h, k_top, temp):
    """Compute gradient of log-sum-exp surrogate."""
    N = len(h)
    # Use top-k for efficiency
    top_idx = np.argpartition(corr, -k_top)[-k_top:]
    c_top = corr[top_idx]
    c_max = c_top.max()
    w = np.exp((c_top - c_max) / temp)
    w /= w.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for wi, ki in zip(w, top_idx):
        L = int(ki - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            seg = N - M
            grad[:seg] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:seg]
    return grad

def adam_continuous(h0, N, deadline, k=80, temp0=0.002, lr0=0.03):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    
    m = np.zeros(N)  # Adam first moment
    v = np.zeros(N)  # Adam second moment
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = lr0
    temp = temp0
    stall = 0
    
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        grad = smooth_gradient(corr, h, k, temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h_new, target)
        
        # Schedule
        if t % 150 == 0:
            lr = max(0.003, lr * 0.98)
            temp = max(0.0003, temp * 0.97)
            k = min(120, k + 2)
        
        if stall > 500:
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)

def corr_update(corr, h, p, q, N):
    """Update correlation array after swapping h[p]:1->0 and h[q]:0->1.
    Returns the new corr array (in-place modification)."""
    # corr[k] where k = N-1+L corresponds to lag L
    center = N - 1
    for L in range(-(N-1), N):
        if L == 0:
            continue  # overlap(0) = 0 for binary
        # Change in overlap at lag L
        delta = 0
        # p was 1, now 0:
        # as h[p] factor: -(1-h_old[p+L]) if p+L in [0,N)
        # as (1-h) factor at i=p-L: +h_old[p-L] if p-L in [0,N)
        if 0 <= p + L < N:
            delta -= (1.0 - h[p+L])  # h[p+L] is still old value
        if 0 <= p - L < N:
            delta += h[p-L]
        # q was 0, now 1:
        # as h[q] factor: +(1-h_old[q+L]) if q+L in [0,N)
        # as (1-h) factor at i=q-L: -h_old[q-L] if q-L in [0,N)
        if 0 <= q + L < N:
            delta += (1.0 - h[q+L])
        if 0 <= q - L < N:
            delta -= h[q-L]
        corr[center + L] += delta
    
    # Now perform the swap
    h[p] = 0.0
    h[q] = 1.0
    return corr
```
