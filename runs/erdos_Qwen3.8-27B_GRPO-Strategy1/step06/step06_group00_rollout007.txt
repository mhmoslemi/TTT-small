Let me think about the continuous optimization more carefully.

The objective: minimize max_k f_k(h) where f_k(h) = Σ_i h[i] * (1-h[i+shift_k])

f_k = Σ_i h[i] * (1 - h[i+s])

The indices i range over valid positions where both i and i+s are in [0,N).

∂f_k/∂h[j]:
- j appears in h[i] when i=j: contributes (1-h[j+s])
- j appears in h[i+s] when i+s=j, i.e., i=j-s: contributes -h[j-s]

So ∂f_k/∂h[j] = (1-h[j+s]) - h[j-s] (with h[·] = 0 outside [0,N))

This is O(N) per lag, and there are O(N) lags, so total is O(N²) for the full gradient.

For the "max" gradient, we only need the gradient of the active lags (those achieving the max).

The previous code uses top-k lags in the gradient, which is a good approximation.

Let me think about what else could help:

**Idea: Use the "water-filling" principle**

The optimal solution should have all the worst lags approximately equal (by the KKT conditions for min max). So we should look for solutions where the top few lags are all close to the same value.

**Idea: Post-processing / local adjustment**

After getting a near-optimal continuous solution, do a "local adjustment" where we slightly tweak individual h[i] values to reduce the max, while maintaining the sum constraint.

For each i, if we change h[i] by δ (and adjust h[j] by -δ to maintain sum), the change in f_k is:
δ * [(1-h[i+s_k]) - h[i-s_k] - (1-h[j+s_k]) + h[j-s_k]]
= δ * [(1-h[i+s_k]-h[i-s_k]) - (1-h[j+s_k]+h[j-s_k])]
= δ * [(h[j+s_k]-h[j-s_k]) - (h[i+s_k]-h[i-s_k])]

This is a 2D search (over pairs (i,j)) for the best adjustment.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_corr(h, corr, N, top_lags):
    """Compute gradient of sum over top_lags of corr[lag]."""
    grad = np.zeros(N)
    center = N - 1
    for idx in top_lags:
        s = int(idx - center)
        # df/dh[j] = (1-h[j+s]) - h[j-s]
        if s >= 0:
            # h[j+s] valid for j in [0, N-s)
            grad[:N-s] += (1.0 - h[s:])
            # h[j-s] valid for j in [s, N)
            grad[s:] -= h[:N-s]
        else:
            s2 = -s
            # h[j+s] = h[j-s2] valid for j in [s2, N)
            grad[s2:] += (1.0 - h[:N-s2])
            # h[j-s] = h[j+s2] valid for j in [0, N-s2)
            grad[:N-s2] -= h[s2:]
    return grad

def nesterov_opt(h, N, deadline, k=30, lr0=0.015, beta0=0.9):
    h = h.astype(float).copy()
    m = np.zeros(N)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    lr = lr0
    beta = beta0
    target = N / 2.0
    stagnant = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Get top k lags
        top = np.argpartition(corr, -k)[-k:]
        grad = grad_corr(y, corr, N, top)
        grad /= k
        
        # Adaptive learning rate
        if stagnant > 50:
            lr *= 0.95
            stagnant = 0
        
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if lr < 0.001:
            break
    return best_h, best_pk * (2.0 / N)

def binary_local_search(h, N, deadline, max_iters=3000, sample_sz=80):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1-nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh-n1, replace=False)] = 1
    best_c = float('inf')
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        # Compute the C matrix for sampled elements
        center = N - 1
        best_new = P
        best_oi = best_zi = -1
        # For each pair, evaluate change in max corr
        # Use FFT for a few candidates
        ntry = min(15, ns * nz)
        for _ in range(ntry):
            oi = int(np.random.randint(ns))
            zi = int(np.random.randint(nz))
            o, z = os_[oi], zs[zi]
            h[o] = 0
            h[z] = 1
            new_P = float(np.max(fft_corr(h)))
            h[z] = 0
            h[o] = 1
            if new_P < best_new:
                best_new = new_P
                best_oi = oi
                best_zi = zi
        if best_new < P - 1e-12:
            h[os_[best_oi]] = 0
            h[zs[best_zi]] = 1
        else:
            break
    return best_h, best_c

def continuous_local_search(h, N, deadline, max_iters=5000):
    """Local search in continuous space using pair adjustments."""
    h = h.astype(float).copy()
    target = N / 2.0
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    nh = N // 2
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
        # Find the worst lag
        worst = int(np.argmax(corr))
        center = N - 1
        s = worst - center
        # Compute gradient at worst lag
        # df/dh[j] = (1-h[j+s]) - h[j-s]
        grad = np.zeros(N)
        if s >= 0:
            grad[:N-s] = (1.0 - h[s:])
            grad[s:] -= h[:N-s]
        else:
            s2 = -s
            grad[s2:] = (1.0 - h[:N-s2])
            grad[:N-s2] -= h[s2:]
        # To reduce f_worst, we want to increase h where grad < 0 and decrease where grad > 0
        # But we need to maintain sum constraint
        # Find i with most negative grad (want to increase h[i]) and j with most positive (decrease h[j])
        i = int(np.argmin(grad))
        j = int(np.argmax(grad))
        if grad[i] >= grad[j]:
            break
        # Adjust: h[i] += eps, h[j] -= eps
        # Choose eps to stay in [0,1]
        eps = min(1.0 - h[i], h[j], 0.05)
        h[i] += eps
        h[j] -= eps
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        # Fix sum if needed
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                frac = np.where((h < 1) & (h > 0))[0]
                h[frac[-1]] += min(diff, 1 - h[frac[-1]])
            else:
                frac = np.where((h < 1) & (h > 0))[0]
                h[frac[0]] -= min(-diff, h[frac[0]])
    return best_h, best_c * (2.0 / N)
