```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binary_search(h, N, deadline, max_sweeps, anneal=0.0, kick_every=30):
    h = h.astype(float).copy()
    nh = N // 2
    if int(round(h.sum())) != nh:
        ones_i = np.where(h > 0.5)[0]
        zeros_i = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if len(ones_i) > nh:
            h[ones_i[:nh]] = 1
        else:
            h[ones_i] = 1
            h[zeros_i[:nh - len(ones_i)]] = 1
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for it in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        bd, bpq = 0.0, None
        for p in ones:
            base = c - D[p, :]
            cand_max = np.max(base[None, :] - Dz, axis=1)
            mi = int(np.argmin(cand_max))
            d = float(cand_max[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        elif stall > kick_every and stall % max(1, kick_every // 3) == 0:
            # Kick: 2-3 random swaps to escape local minimum
            for _ in range(np.random.randint(2, 5)):
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            stall = 0
        else:
            if stall > kick_every * 2:
                break
    return h, best_c

def mw_smooth(h, N, deadline, T0=0.3, lr0=0.04):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    lr = lr0
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        # Multiplicative weights
        cs = c - c.max()
        w = np.exp(cs / T)
        w /= w.sum()
        D = compute_D(h, N)
        grad = -(D @ w)
        grad -= grad.mean()
        bc1 = 1 - 0.9 ** t
        bc2 = 1 - 0.999 ** t
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad ** 2
        h = project_sum(h - lr * (m / bc1) / (np.sqrt(v / bc2) + 1e-8), target)
        if t % 100 == 0:
            T *= 0.96
            lr *= 0.97
        if t % 150 == 0:
            hb = (h > 0.5).astype(float)
            n1 = int(hb.sum())
            if n1 != N // 2:
                oi = np.where(hb == 1)[0]
                zi = np.where(hb == 0)[0]
                if n1 > N // 2:
                    idx = oi[np.argsort(h[oi])[:-(n1 - N//2)]]
                    hb[idx] = 0
                else:
                    idx = zi[np.argsort(-h[zi])[:(N//2 - n1)]]
                    hb[idx] = 1
            pkb = float(np.max(fft_corr(hb)))
            if pkb < best_pk:
                best_pk = pkb
                best_h = hb.copy()
            h = 0.5 * h + 0.5 * hb
    return best_h, best_pk * dx
```
