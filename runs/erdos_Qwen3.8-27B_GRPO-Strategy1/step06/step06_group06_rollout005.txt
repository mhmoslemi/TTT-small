import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0-h, L)
    corr = np.fft.irfft(np.conj(G) * H, L)
    return corr[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def binary_proj(h, N):
    target = N // 2
    h_b = (h >= 0.5).astype(float)
    n1 = int(h_b.sum())
    if n1 > target:
        idx = np.where(h_b == 1)[0]
        perm = np.random.permutation(len(idx))[:n1-target]
        h_b[idx[perm]] = 0
    elif n1 < target:
        idx = np.where(h_b == 0)[0]
        perm = np.random.permutation(len(idx))[:target-n1]
        h_b[idx[perm]] = 1
    return h_b

def grad_smooth_max(h, corr, K, T):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-K:]
    tc = corr[top_idx]
    w = np.exp(T * (tc - tc.max()))
    w /= w.sum()
    grad = np.zeros(N)
    for idx, wi in zip(top_idx, w):
        s = int(idx - center)
        if s == 0:
            grad += wi * (1.0 - 2.0*h)
        elif s > 0:
            grad[:N-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:N-s]
        else:
            s2 = -s
            grad[s2:] += wi * (1.0 - h[:N-s2])
            grad[:N-s2] -= wi * h[s2:]
    return grad

def swap_delta(h, N, p, q):
    s_arr = np.arange(-(N-1), N)
    d = np.zeros(2*N-1)
    ps = p + s_arr; pm = p - s_arr
    qs = q + s_arr; qm = q - s_arr
    vps = (ps>=0)&(ps<N); vpm = (pm>=0)&(pm<N)
    vqs = (qs>=0)&(qs<N); vqm = (qm>=0)&(qm<N)
    d[vps] -= (1.0 - h[np.clip(ps,0,N-1)[vps]])
    d[vpm] += h[np.clip(pm,0,N-1)[vpm]]
    d[vqs] += (1.0 - h[np.clip(qs,0,N-1)[vqs]])
    d[vqm] -= h[np.clip(qm,0,N-1)[vqm]]
    return d
