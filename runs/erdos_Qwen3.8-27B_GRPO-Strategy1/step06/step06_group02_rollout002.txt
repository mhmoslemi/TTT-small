    target = N / 2.0
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    m = np.zeros(N)
    lr = lr0
    beta = 0.7
    temp = temp0
    it = 0
    while time.time() < deadline:
        y = h + m
        g, _ = compute_gradient(y, temp, top_k=k)
        h_new = project_sum(y - lr * g, target)

            beta = min(0.9, 0.7 + it * 0.001)

                best_pq = (p, q)
            h[p], h[q] = 1.0, 0.0
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            # Random perturbation
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
    return h, best_c * (2.0 / N)

    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    d1 = min(deadline - 400, t0 + 300)
    h1, c5_1 = nesterov_optimize(best_h, N0, d1, temp0=8.0, temp_min=0.5, lr0=0.04, k=200)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1
    
    # Phase 2: Nesterov with low temperature (polish)
    if time.time() < deadline - 200:
        d2 = min(deadline - 180, time.time() + 150)
        h2, c5_2 = nesterov_optimize(best_h, N0, d2, temp0=2.0, temp_min=0.1, lr0=0.02, k=300)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2
    
    # Phase 3: Restart with perturbation
    if time.time() < deadline - 150:
        d3 = min(deadline - 130, time.time() + 100)
        h3_init = best_h + np.random.randn(N0) * 0.05
        h3_init = project_sum(h3_init, N0 / 2.0)
        h3, c5_3 = nesterov_optimize(h3_init, N0, d3, temp0=4.0, temp_min=0.3, lr0=0.03, k=200)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3
    
    # Phase 4: Another restart
    if time.time() < deadline - 100:
        d4 = min(deadline - 80, time.time() + 70)
        h4_init = best_h + np.random.randn(N0) * 0.03
        h4_init = project_sum(h4_init, N0 / 2.0)
        h4, c5_4 = nesterov_optimize(h4_init, N0, d4, temp0=3.0, temp_min=0.2, lr0=0.025, k=250)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4
    
    # Phase 5: Binary local search from best
    if time.time() < deadline - 50:
