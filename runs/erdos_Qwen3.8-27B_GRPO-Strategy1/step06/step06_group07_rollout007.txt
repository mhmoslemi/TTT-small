def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

        return h
    lo, hi = -1.0, 1.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    hp = np.where(vp, h[pp_c], 0.0)
    hm = np.where(vm, h[pm_c], 0.0)
    return hp * vp + hm * vm - vp

def lse_grad(h, N, k, temp):
    y = h
    corr = fft_corr(y)
    top = np.argpartition(corr, -k)[-k:]
    tv = corr[top]
    ev = np.exp((tv - tv.max()) / temp)
    w = ev / ev.sum()
    center = N - 1
    grad = np.zeros(N)
    for ai, wi in zip(top, w):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += wi * (1.0 - y[:N - L])
            grad[:N - L] -= wi * y[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N - M]
    return grad / k, float(corr.max())
