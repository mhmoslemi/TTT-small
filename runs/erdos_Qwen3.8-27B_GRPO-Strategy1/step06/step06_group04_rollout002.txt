def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(corr, h, N, top_k, temp):
    """Compute subgradient of max_k corr_k weighted by softmax."""
    center = N - 1
    top = np.argpartition(corr, -top_k)[-top_k:]
    w = np.exp((corr[top] - corr[top].max()) / max(temp, 1e-10))
    w /= w.sum()
    grad = np.zeros(N)
    for wi, ai in zip(w, top):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                seg = N - M
                grad[:seg] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:seg]
    return grad

        corr = fft_corr(h)
        pk = float(np.max(corr))

            best_val = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        grad = compute_grad(corr, h, N, k, temp)

        h = h_new

            stall = 0

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(corr, h, N, top_k, temp):
    center = N - 1
    top = np.argpartition(corr, -top_k)[-top_k:]
    w = np.exp((corr[top] - corr[top].max()) / max(temp, 1e-10))
    w /= w.sum()
    grad = np.zeros(N)
    for wi, ai in zip(w, top):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                seg = N - M
                grad[:seg] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:seg]
    return grad

def smooth_opt(h0, N, deadline, k=100, lr0=0.010, temp0=0.001, mom0=0.82):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    mom = np.zeros(N)
    lr = lr0
    temp = temp0
    beta = mom0
    stall = 0
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_val - 1e-14:
            best_val = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        grad = compute_grad(corr, h, N, k, temp)
        mom = beta * mom + (1.0 - beta) * grad
        # Line search: try current lr and a few smaller steps
        best_step = lr
        best_obj = pk
        for frac in [1.0, 0.7, 0.5, 0.3]:
            h_cand = project_sum(h - lr * frac * mom, target)
            c_cand = float(np.max(fft_corr(h_cand)))
            if c_cand < best_obj - 1e-15:
                best_obj = c_cand
                best_step = lr * frac
                h = h_cand
                break
        else:
            h = project_sum(h - best_step * mom, target)
        it += 1
        if it % 150 == 0:
            lr = max(0.0008, lr * 0.996)
            temp = max(0.00005, temp * 0.992)
            beta = min(0.94, beta + 0.0003)
        if stall > 250:
            h = project_sum(best_h + np.random.randn(N) * 0.015, target)
            mom = np.zeros(N)
            lr = max(lr, lr0 * 0.7)
            stall = 0
    return best_h, best_val * (2.0 / N)

def targeted_binary(h0, N, deadline, n_tries=25):
    h = (h0 > 0.5).astype(float).copy()

    n1 = int(h.sum())
    if n1 != nh:

        else:

    c = fft_corr(h)

    best_h = h.copy()
    stall = 0
    center = N - 1

        zeros = np.where(h == 0)[0]

            break
        worst_idx = np.argmax(c)
        worst_lag = int(worst_idx - center)
        best_sweep_int = int(c.max())
        best_sweep_h = None
        for _ in range(n_tries):
            if worst_lag > 0:

                best_sweep_h = h.copy()
                best_sweep_c = c_t.copy()
            h[p] = 1
            h[q] = 0
        if best_sweep_h is not None:

            c = best_sweep_c
            stall = 0

                    i2 = np.random.choice(len(zeros), 2, replace=False)
                    for pp in ones[i1]:
                        h[pp] = 0
                    for qq in zeros[i2]:
                        h[qq] = 1
                    c = fft_corr(h)
                stall = 0
    return best_h, best_int * (2.0 / N)
