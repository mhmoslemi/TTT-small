```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def sa_binary(h_in, N, deadline, max_iter=50000, t0_init=0.5, t0_final=0.01):
    """Simulated annealing on binary space with single and double flips."""
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    no_improve = 0
    it = 0
    
    while time.time() < deadline and it < max_iter:
        it += 1
        T = t0_init * (t0_final / t0_init) ** (it / max_iter)
        
        # Choose move type: single swap (80%), double swap (20%)
        if np.random.random() < 0.8 or len(ones) < 2 or len(zeros) < 2:
            # Single swap: one 1→0 and one 0→1
            p = np.random.choice(ones)
            q = np.random.choice(zeros)
            # Delta: change in corr when h[p] 1→0 and h[q] 0→1
            # corr changes by: -flip effect at p + flip effect at q
            # Using the correlation structure:
            # We need to compute the new max after the swap
            # For efficiency, track the delta vector
            delta = -corr_pattern(h, p, N) + corr_pattern(h, q, N)
            new_corr = corr + delta
            new_c = float(new_corr.max())
            
            if new_c < best_c or np.random.random() < np.exp(-(new_c - best_c) / T):
                if new_c < best_c - 1e-12:
                    best_c = new_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                # Apply swap
                h[p], h[q] = 0.0, 1.0
                corr = new_corr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                no_improve += 1
        else:
            # Double swap: two 1s→0s and two 0s→1s
            pp = np.random.choice(ones, 2, replace=False)
            qq = np.random.choice(zeros, 2, replace=False)
            delta = (-corr_pattern(h, pp[0], N) - corr_pattern(h, pp[1], N) 
                     + corr_pattern(h, qq[0], N) + corr_pattern(h, qq[1], N))
            new_corr = corr + delta
            new_c = float(new_corr.max())
            
            if new_c < best_c or np.random.random() < np.exp(-(new_c - best_c) / T):
                if new_c < best_c - 1e-12:
                    best_c = new_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                h[pp[0]], h[qq[0]] = 0.0, 1.0
                h[pp[1]], h[qq[1]] = 0.0, 1.0
                corr = new_corr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                no_improve += 1
        
        if no_improve > 5000:
            # Restart temperature
            T = t0_init * 0.5
            no_improve = 0
    
    return best_h, best_c
```

```python
def flip_delta_binary(h, p, N):
    """Change in correlate(h, 1-h, 'full') when h[p] flips (binary). Returns array of length 2N-1."""
    delta = 1.0 - 2.0 * h[p]  # +1 if 0→1, -1 if 1→0
    L = 2 * N - 1
    S = L - 1  # = 2N-2, index offset for k=0
    result = np.zeros(L)
    
    # For each lag k (can be negative), compute contribution
    # k ranges from -(N-1) to N-1
    # corr at index j = k + (N-1)
    
    # Case A: h[p] is in first factor h[n+k], so n = p-k
    #   Valid: 0 <= n <= N-1, i.e., 0 <= p-k <= N-1
    #   k <= p and k >= p-(N-1)
    #   Also n+k = p, and 1-h[n] = 1-h[p-k]
    #   Change: delta * (1-h[p-k])
    #   Exception: if p-k = p (k=0), skip (handled separately)
    for k in range(-(N-1), N):
        if k == 0:
            continue
        n = p - k
        if 0 <= n < N:
            j = k + (N-1)
            result[j] += delta * (1.0 - h[n])
    
    # Case B: h[p] is in second factor (1-h[n]), so n = p
    #   Valid: 0 <= p < N (always true)
    #   Also n+k = p+k must be in [0, N-1]
    #   Change: h[p+k] * (-delta)
    #   Exception: if p+k = p (k=0), skip
    for k in range(-(N-1), N):
        if k == 0:
            continue
        ip = p + k
        if 0 <= ip < N:
            j = k + (N-1)
            result[j] -= delta * h[ip]
    
    return result
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def flip_delta(h, p, N):
    delta = 1.0 - 2.0 * h[p]
    L = 2 * N - 1
    S = N - 1
    result = np.zeros(L)
    n_vals = np.arange(N)
    kA = p - n_vals
    validA = (kA >= -(N-1)) & (kA <= N-1) & (kA != 0)
    jA = kA[validA] + S
    result[jA] += delta * (1.0 - h[n_vals[validA]])
    kA2 = n_vals - p
    validB = (kA2 >= -(N-1)) & (kA2 <= N-1) & (kA2 != 0)
    jB = kA2[validB] + S
    result[jB] -= delta * h[n_vals[validB]]
    return result

            qq = np.random.choice(zeros, 2, replace=False)
            d1 = flip_delta(h, pp[0], N)
            h[pp[0]] = 1.0 - h[pp[0]]
            d2 = flip_delta(h, pp[1], N)
            h[pp[1]] = 1.0 - h[pp[1]]
            d3 = flip_delta(h, qq[0], N)
            h[qq[0]] = 1.0 - h[qq[0]]
            d4 = flip_delta(h, qq[1], N)
            h[qq[1]] = 1.0 - h[qq[1]]
            # Undo for delta computation: actually this is wrong. 
            # The delta for flipping two positions doesn't just add up because

            # For simplicity, let me just recompute corr for double swaps.

                    best_h = h.copy()

                zeros = np.where(h == 0)[0]
            else:
                # Undo
                h[p] = 1.0
                h[q] = 0.0
                no_improve += 1

        if no_improve > max_no_improve:
            T = t_start * 0.3
            no_improve = 0
            max_no_improve = max(2000, max_no_improve // 2)
    
    return best_h, best_c
```
