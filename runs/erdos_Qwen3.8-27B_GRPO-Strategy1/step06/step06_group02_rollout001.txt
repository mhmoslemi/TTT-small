def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

    if abs(s - target) < 1e-10:
        return h

            lo = mid
        else:

    """Simulated annealing on binary h."""
    h = h.astype(float).copy()
    dx = 2.0 / N
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    T = T0

    no_improve = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_iter):

            break
        T = max(T_end, T0 * cool ** it)

        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        # Evaluate: new_corr = corr - M[p,:] + M[q,:]
        new_corr = corr - M[p, :] + M[q, :]

            if new_max < best_c - 1e-12:

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            M[p, center-(N-1-p):center] -= h[p+1:]
    return M
