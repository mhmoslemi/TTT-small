```python
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:

        it += 1
        if it % 80 == 0:
            lr *= 0.994

    return best_h, best_pk * (2.0 / N)
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h > 0.5)[0]
        rm = idx[np.random.permutation(len(idx))[:n1 - nh]]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h < 0.5)[0]
        add = idx[np.random.permutation(len(idx))[:nh - n1]]
        h[add] = 1.0
    return h

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def full_local_search(h, N, deadline, penalty2=0.03, penalty3=0.01, max_sweeps=5000, stuck_limit=400):
    h = h.astype(float).copy()
    c = fft_corr(h)
    cmax = float(c.max())
    best_cmax = cmax
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        D = compute_D(h, N)
        cmax = float(c.max())
        
        best_score = cmax + 1.0
        best_pq = None
        
        for p in ones:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            cm = cand.max(axis=1)
            # Add secondary terms for tie-breaking
            if penalty2 > 0 and len(cm) > 1:
                # 2nd max
                max_idx = np.argmax(cm)
                cm2 = cm.copy()
                cm2[max_idx] = -1e9
                second_max = float(cm2.max())
                s_vals = cm + penalty2 * second_max
            else:
                s_vals = cm.copy()
            mi = int(np.argmin(s_vals))
            if s_vals[mi] < best_score:

def ils_search(h, N, deadline, n_restarts=5, penalty2=0.03):

    best_c = compute_c5(best_h)

        if c5r < best_c:

def nesterov_opt(h, N, deadline, k=50, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.994
            beta = min(0.97, 0.88 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h > 0.5)[0]
        rm = idx[np.random.permutation(len(idx))[:n1 - nh]]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h < 0.5)[0]
        add = idx[np.random.permutation(len(idx))[:nh - n1]]
        h[add] = 1.0
    return h

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def full_local_search(h, N, deadline, pen2=0.02, pen3=0.008, max_sweeps=5000, stuck_limit=600):
    h = h.astype(float).copy()
    c = fft_corr(h)
    cmax = float(c.max())
    best_cmax = cmax
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        cmax = float(c.max())
        best_score = cmax + 1.0
        best_pq = None
        for p in ones:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            cm = cand.max(axis=1)
            if pen2 > 0 and len(cm) > 1:
                mi_max = int(np.argmax(cm))
                cm2 = cm.copy()
                cm2[mi_max] = -9999.0
                second = float(cm2.max())
                if pen3 > 0 and len(cm) > 2:
                    cm2[mi_max] = cm[mi_max]
                    cm2[int(np.argmax(cm2))] = -9999.0
                    third = float(cm2.max())
                    s_vals = cm + pen2 * second + pen3 * third
                else:
                    s_vals = cm + pen2 * second
            else:
                s_vals = cm
            mi = int(np.argmin(s_vals))
            if s_vals[mi] < best_score:
                best_score = float(s_vals[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)
        new_cmax = float(c.max())
        if new_cmax < best_cmax - 1e-12:
            best_cmax = new_cmax
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > stuck_limit:
            break
    return h, float(np.max(fft_corr(h))) * (2.0 / N)

def ils_search(h, N, deadline, n_restarts=4, pen2=0.02):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    remain = deadline - time.time()

        zeros_r = np.where(hr < 0.5)[0]

        elif r == 0:

def nesterov_opt(h, N, deadline, k=50, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.994
            beta = min(0.97, 0.88 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def two_swap_search(h, N, deadline, max_sweeps=2000):
    h = h.astype(float).copy()
    c = fft_corr(h)
    cmax = float(c.max())
    best_cmax = cmax
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        best_score = cmax + 1.0
        best_pq = None
        sample_p = min(80, len(ones))
        sample_z = min(80, len(zeros))
        p_sel = ones[np.random.permutation(len(ones))[:sample_p]]
        z_sel = zeros[np.random.permutation(len(zeros))[:sample_z]]
        for i in range(len(p_sel)):
            p1 = int(p_sel[i])
            for j in range(i + 1, len(p_sel)):
                p2 = int(p_sel[j])
                base12 = c - D[p1, :] - D[p2, :]
                cand = base12[None, :] - D[z_sel[None, :], :]
                cm = cand.max(axis=2)
                mi, mj = np.unravel_index(int(np.argmin(cm)), cm.shape)
                if cm[mi, mj] < best_score:
                    best_score = float(cm[mi, mj])
                    best_pq = (p1, p2, int(z_sel[mi]), int(z_sel[mj]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 0.0
        h[best_pq[2]] = 1.0
        h[best_pq[3]] = 1.0
        c = fft_corr(h)
        new_cmax = float(c.max())
        if new_cmax < best_cmax - 1e-12:
            best_cmax = new_cmax
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
    return h, float(np.max(fft_corr(h))) * (2.0 / N)
```
