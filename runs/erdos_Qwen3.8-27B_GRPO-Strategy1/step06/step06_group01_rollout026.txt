import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_binary(h, target):
    """Project to binary with exactly target ones"""
    h = h.copy()
    n1 = int(np.sum(h > 0.5))
    if n1 == target:
        h[h > 0.5] = 1.0
        h[h <= 0.5] = 0.0
        return h
    vals = np.sort(h)[::-1]
    threshold = vals[target] if target < len(vals) else 0
    h = (h >= threshold).astype(float)
    n1 = int(h.sum())
    if n1 > target:
        # Randomly set some to 0
        ones = np.where(h == 1)[0]
        extra = np.random.choice(ones, n1-target, replace=False)
        h[extra] = 0.0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        need = target - n1
        add = np.random.choice(zeros, need, replace=False)
        h[add] = 1.0
    return h

def local_search_binary(h, deadline, n_try=50, max_sweeps=50000):
    """Local search: try n_try random swaps per sweep, accept best improvement"""
    N = len(h)
    target = N // 2
    h = project_binary(h, target)
    corr = fft_corr(h).copy()
    best_c = float(corr.max())
    best_h = h.copy()
    no_improve = 0
    it = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        n1 = min(n_try, len(ones))
        n0 = min(n_try, len(zeros))
        p_cand = np.random.choice(ones, n1, replace=False)
        q_cand = np.random.choice(zeros, n0, replace=False)
        
        best_new_c = best_c + 1.0
        best_p, best_q = -1, -1
        
        # Try all pairs in the candidate set
        for p in p_cand:
            for q in q_cand:
                if p == q:
                    continue
                h[p], h[q] = 0.0, 1.0
                c_new = np.max(fft_corr(h))
                h[p], h[q] = 1.0, 0.0  # revert
                if c_new < best_new_c:
                    best_new_c = c_new
                    best_p, best_q = p, q
        
        if best_p < 0 or best_new_c >= best_c - 1e-15:
            no_improve += 1
            if no_improve > 20:
                # Random kick: swap 3 random pairs
                for _ in range(3):
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p], h[q] = 0.0, 1.0
                corr = fft_corr(h)
                best_c = float(corr.max())
                no_improve = 0
            continue
        
        # Apply best swap
        h[best_p], h[best_q] = 0.0, 1.0
        corr = fft_corr(h)
        c = float(corr.max())
        if c < best_c:
            best_c = c
            best_h = h.copy()
        no_improve = 0
        it += 1
    
    return best_h, best_c * (2.0/N)

import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0/N)

def make_binary(h, target=None):
    N = len(h)
    if target is None:
        target = N // 2
    # Keep top 'target' values as 1
    sorted_idx = np.argsort(h)[::-1]
    h_new = np.zeros(N)
    h_new[sorted_idx[:target]] = 1.0
    return h_new

def local_search(h_init, deadline, batch_size=50, kick_size=3, max_no_improve=30):
    N = len(h_init)
    target = N // 2
    h = make_binary(h_init, target)
    corr = fft_corr(h).copy()
    best_c = float(corr.max())
    best_h = h.copy()
    no_improve = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        n1 = min(batch_size, len(ones))
        n0 = min(batch_size, len(zeros))
        p_cand = np.random.choice(ones, n1, replace=False)
        q_cand = np.random.choice(zeros, n0, replace=False)
        
        best_new_c = best_c + 1.0
        best_p, best_q = -1, -1
        
        # Evaluate all pairs in batch
        for ip in range(n1):
            p = p_cand[ip]
            h[p] = 0.0
            for iq in range(n0):

                if q == p:
                    continue
                h[q] = 1.0
                c = np.max(fft_corr(h))
                if c < best_new_c:
                    best_new_c = c
                    best_p, best_q = p, q
                h[q] = 0.0
            h[p] = 1.0
        
        if best_p is None or best_new_c >= best_c - 1e-15:
            no_improve += 1
            if no_improve >= max_no_improve:
                # Kick
                n_kick = min(kick_size, min(len(ones)//2, len(zeros)//2))
                pk = np.random.choice(ones, n_kick, replace=False)
                qk = np.random.choice(zeros, n_kick, replace=False)
                h[pk] = 0.0
                h[qk] = 1.0
                corr = fft_corr(h)
                best_c = float(corr.max())
                no_improve = 0
            continue

        c = float(corr.max())

            best_c = c
            best_h = h.copy()
        no_improve = 0
    
    return best_h, best_c * (2.0/N)

def nesterov_continuous(h_init, deadline, k_top=30, lr=0.02):
    N = len(h_init)
    target = N / 2.0
    h = h_init.astype(float).copy()
    # Project to sum constraint
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > target:
        excess = s - target
        h = h - excess / N
        h = np.clip(h, 0, 1)
    elif s < target:
        deficit = target - s
        h = h + deficit / N
        h = np.clip(h, 0, 1)
    
    vx = h.copy()
    m = np.zeros(N)
    best_c5 = float('inf')
    best_h = h.copy()
    beta = 0.85
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        c5 = float(corr.max()) * (2.0/N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient: subgradient of max_k f_k(h)
        # f_k(h) = sum_i h[i]*(1-h[i+k])
        # df_k/dh[j] = (1-h[j+k]) - h[j-k] (for valid k)
        
        # Use top-k active correlations
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        top_idx = np.sort(top_idx)
        
        grad = np.zeros(N)
        for t in top_idx:
            L_shift = int(t - (N-1))  # actual shift
            if L_shift == 0:
                grad += (1 - 2*y)
            elif L_shift > 0:
                grad[L_shift:] += (1 - y[:N-L_shift])
                grad[:N-L_shift] -= y[L_shift:]
            else:
                M = -L_shift
                grad[:N-M] += (1 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= len(top_idx)
        
        h_new = y - lr * grad
        # Project to [0,1] and sum constraint
        h_new = np.clip(h_new, 0, 1)
        s = h_new.sum()
        if abs(s - target) > 1e-6:
            # Bisection to find offset
            lo, hi = -1.0, 1.0
            for _ in range(30):
                mid = (lo+hi)/2
                if np.clip(h_new - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo+hi)/2, 0, 1)
        
        m = beta * (h_new - h)
        h = h_new
    
    return best_h, best_c5

        if N_try == N0:

        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h + 0.5) - 0.5
        h_interp = np.clip(h_interp, 0, 1)
        sub_dl = min(deadline - 150, time.time() + 80)
        h_r, c_r = local_search(h_interp, sub_dl, batch_size=50, kick_size=3, max_no_improve=20)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r
            N0 = N_try  # Update base N

    if time.time() < deadline - 100:

        hc = best_h + np.random.randn(len(best_h)) * 0.03
        hc = np.clip(hc, 0, 1)
        hn, c_n = nesterov_continuous(hc, d3)

            best_c5, best_h = c4, h4

    return best_h, final_c5, len(best_h)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, target):
    N = len(h)
    idx = np.argsort(h)[::-1]
    h_new = np.zeros(N)
    h_new[idx[:target]] = 1.0
    return h_new

def compute_deltas(h, corr):
    N = len(h)
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        h[i] = 1.0 - h[i]
        c2 = fft_corr(h)
        deltas[i] = c2 - corr
        h[i] = 1.0 - h[i]
    return deltas

def local_search(h_init, deadline, n_cand=200, chunk=25, recompute_every=6, kick_sz=4, max_stale=30):
    N = len(h_init)
    target = N // 2
    h = make_binary(h_init, target)
    corr = fft_corr(h).copy()
    best_c = float(corr.max())
    best_h = h.copy()
    deltas = compute_deltas(h, corr)
    swaps = 0
    no_imp = 0

    while time.time() < deadline:
        ones = np.where(h == 1)[0]

def nesterov_cont(h_init, deadline, k_top=40, lr=0.015, beta=0.82):

    target = N / 2.0
    h = np.clip(h_init, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) > 1e-6:
        lo, hi = -1.0, 1.0

            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    m = np.zeros(N)

            best_h = h.copy()

        grad = np.zeros(N)
        for t in top:

            if Ls == 0:

                grad[Ls:] += (1.0 - y[:N - Ls])
                grad[:N - Ls] -= y[Ls:]
            else:

                grad[:N - M] += (1.0 - y[M:])

        h_new = np.clip(h_new, 0.0, 1.0)
        s2 = h_new.sum()
        if abs(s2 - target) > 1e-6:

        m = beta * (h_new - h)
        h = h_new
        if lr > 0.002:
            lr *= 0.9995
    return best_h, best_c5
