import time

def compute_c5(h, N):
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def to_binary(h, N, rng):
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:n1-nh]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:nh-n1]]] = 1
    return h

        h_rt, c5_rt = local_search(h_bt, N_try, rng, sub_dl, K=25, stag_lim=60, pert_n=15)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    if time.time() < deadline - 40:
        d5 = min(deadline - 20, time.time() + 60)
        h_a, c5_a = local_search(best_h, len(best_h), rng, d5, K=30, stag_lim=50, pert_n=30)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a.copy()

    if time.time() < deadline - 10:
        hF = continuous_polish(best_h, len(best_h), deadline - 5, rounds=20)
        hFb = to_binary(hF, len(best_h), rng)
        c5F = compute_c5(hFb, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hFb.copy()

    if time.time() < deadline - 2:
        hF2 = to_binary(best_h, len(best_h), rng)
        hF3, c5F3 = local_search(hF2, len(best_h), rng, deadline - 1, K=30, stag_lim=30, pert_n=10)
        if c5F3 < best_c5:
            best_c5, best_h = c5F3, hF3.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
