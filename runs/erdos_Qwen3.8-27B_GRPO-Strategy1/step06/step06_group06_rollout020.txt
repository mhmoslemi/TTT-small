def corr_full(h):
    """Compute full cross-correlation of h with 1-h."""
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    c = np.fft.irfft(H * np.conj(G), L)
    return np.concatenate([c[L-(N-1):], c[:N]])

def proj_sum(h, target):
    """Project h onto {0≤h≤1, sum=target}."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def fix_count(h, N):
    """Fix binary count to N/2."""
    target = N // 2

    elif n1 < target:

        sel = np.random.choice(len(idx), target-n1, replace=False)

def corr_full(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    c = np.fft.irfft(H * np.conj(G), L)
    return np.concatenate([c[L - (N - 1):], c[:N]])

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

        else:
