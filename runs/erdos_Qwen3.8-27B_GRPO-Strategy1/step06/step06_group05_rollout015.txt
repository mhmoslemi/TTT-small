    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    target = N // 2
    h = np.clip(h, 0, 1)
    n1 = int(np.round(h.sum()))
    if n1 > target:
        idx = np.where(h > 0.5)[0]
        if len(idx) >= n1:
            order = np.argsort(h[idx])
            h[idx[order[:n1 - target]]] = 0
    elif n1 < target:
        idx = np.where(h <= 0.5)[0]
        order = np.argsort(-h[idx])
        h[idx[order[:target - n1]]] = 1
    return (h > 0.5).astype(float)

def compute_deltas(h, N):
    """Compute A (delta for 1->0) and B (delta for 0->1) matrices."""
    shifts = np.arange(-(N - 1), N)  # (2N-1,)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    s_row = shifts[None, :]  # (1, 2N-1)
    
    # A: (n_ones, 2N-1) - effect of flipping 1 to 0
    p_col = ones[:, None]
    ps = p_col + s_row
    pm = p_col - s_row
    v_ps = (ps >= 0) & (ps < N)
    v_pm = (pm >= 0) & (pm < N)
    ps_c = np.clip(ps, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    A = np.where(v_ps, -(1.0 - h[ps_c]), 0.0) + np.where(v_pm, h[pm_c], 0.0)
    
    # B: (n_zeros, 2N-1) - effect of flipping 0 to 1
    q_col = zeros[:, None]
    qs = q_col + s_row
    qm = q_col - s_row
    v_qs = (qs >= 0) & (qs < N)
    v_qm = (qm >= 0) & (qm < N)
    qs_c = np.clip(qs, 0, N - 1)
    qm_c = np.clip(qm, 0, N - 1)
    B = np.where(v_qs, (1.0 - h[qs_c]), 0.0) - np.where(v_qm, h[qm_c], 0.0)
    
    return A, B, ones, zeros, shifts

def binary_optimize(h, N, deadline, max_sweeps=500, anneal=0.0, top_k=50):
    h = h.astype(float).copy()
    h = make_binary(h, N)
    best_h = h.copy()
    best_max = float(np.max(fft_corr(h)))
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_max - 1e-12:
            best_max = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 60:
            if anneal > 0:
                # Random move
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                p = int(np.random.randint(len(ones)))
                q = int(np.random.randint(len(zeros)))
                h[ones[p]], h[zeros[q]] = 0.0, 1.0
                stall = 0
                continue
            else:
                break
        
        A, B, ones, zeros, shifts = compute_deltas(h, N)
        
        # Find best swap
        best_d, best_pq = 0.0, None
        n_ones, n_zeros = len(ones), len(zeros)
        batch = min(40, n_ones)
        
        for bi in range(0, n_ones, batch):
            be = min(bi + batch, n_ones)
            temp = corr[None, :] + A[bi:be, :]  # (batch, 2N-1)
            # (batch, n_zeros, 2N-1)
            cmax = (temp[:, :, None] + B[None, :, :]).max(axis=2)  # (batch, n_zeros)

            d = float(cmax[mi, mj]) - P

            h[ones[best_pq[0]]] = 0.0
            h[zeros[best_pq[1]]] = 1.0
    
    return best_h, best_max

    dx = 2.0 / N

        top_k = min(20, len(corr))
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        # For each h[i], gradient = sum over top shifts of d(corr[s])/d(h[i])
        # d(corr[s])/d(h[i]) = (1-h[i+s]) - h[i-s] (with boundary checks)
        # This is the same as -A[i,s] for h[i]=1, or B[i,s] for h[i]=0
        # Actually: d(corr[s])/d(h[i]) = (1-h[i+s]) if i+s valid (from h[i]*(1-h[i+s]))
        #                           - h[i-s] if i-s valid (from h[i-s]*(1-h[i]))
        # = (1-h[i+s]) - h[i-s] (with boundary conditions)
        
        shifts = np.arange(-(N-1), N)
        s_row = shifts[None, :]
        i_col = np.arange(N)[:, None]
        is_ = i_col + s_row[None, :]  # wait, s_row is already (1, 2N-1)
        # Actually I need this for all i, only for top shifts
        top_s = shifts[top_idx - (N-1)]  # top shift indices
        
        i_col2 = np.arange(N)[:, None]
        ts_row = top_s[None, :]
        is2 = i_col2 + ts_row
        im2 = i_col2 - ts_row
        v2_pos = (is2 >= 0) & (is2 < N)
        v2_neg = (im2 >= 0) & (im2 < N)
        is2_c = np.clip(is2, 0, N-1)
        im2_c = np.clip(im2, 0, N-1)
        
        # grad[i] = sum_s (1-h[i+s]) - h[i-s] for top s
        grad = np.where(v2_pos, 1.0 - h[is2_c], 0.0).sum(axis=1) - np.where(v2_neg, h[im2_c], 0.0).sum(axis=1)
        grad *= dx
        
        # Center gradient (to maintain sum constraint approximately)
        grad -= grad.mean()
        
        # Adam update
        bc1 = 1 - b1**t
        bc2 = 1 - b2**t
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad**2
        h = h - lr * (m / bc1) / (np.sqrt(v / bc2) + eps)
        
        # Project to sum constraint and [0,1]
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            # Binary search for offset
            lo, hi = -1.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2
                if np.clip(h + mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h = np.clip(h + (lo + hi)/2, 0, 1)
        
        if t % 200 == 0:
            lr *= 0.95
    
    return best_h, best_val

        ps = p_col + s_row
        pm = p_col - s_row
        A_c = np.where((ps >= 0) & (ps < N), -(1.0 - h[np.clip(ps, 0, N - 1)]), 0.0) + \
              np.where((pm >= 0) & (pm < N), h[np.clip(pm, 0, N - 1)], 0.0)

        q_col = zeros[:, None]
        qs = q_col + s_row
        qm = q_col - s_row
        B_c = np.where((qs >= 0) & (qs < N), (1.0 - h[np.clip(qs, 0, N - 1)]), 0.0) - \
              np.where((qm >= 0) & (qm < N), h[np.clip(qm, 0, N - 1)], 0.0)

        base = corr[crit][None, None, :]
        best_d, best_pq = 0.0, None
        batch = min(40, len(ones))
        for bi in range(0, len(ones), batch):
            be = min(bi + batch, len(ones))
            comb = base + A_c[bi:be, None, :] + B_c[None, :, :]
            nm = comb.max(axis=2)
            mi, mj = np.unravel_index(nm.argmin(), nm.shape)
            d = float(nm[mi, mj]) - P
            if d < best_d:
                best_d = d
                best_pq = (bi + int(mi), int(mj))

        if best_pq is not None and best_d < -1e-12:
            h[ones[best_pq[0]]] = 0.0
            h[zeros[best_pq[1]]] = 1.0

    return best_h, best_max

def continuous_optimize(h, N, deadline, lr0=0.008):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    shifts = np.arange(-(N - 1), N)

    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_val - 1e-10:
            best_val = P
            best_h = h.copy()
        K = min(30, len(corr))
        top = np.argpartition(corr, -K)[-K:]
        top_s = shifts[top - (N - 1)]
        i_col = np.arange(N)[:, None]
        ts = top_s[None, :]
        is_ = i_col + ts
        im_ = i_col - ts
        vp = (is_ >= 0) & (is_ < N)
        vn = (im_ >= 0) & (im_ < N)
        isc = np.clip(is_, 0, N - 1)
        imc = np.clip(im_, 0, N - 1)
        grad = (np.where(vp, 1.0 - h[isc], 0.0).sum(axis=1) -
                np.where(vn, h[imc], 0.0).sum(axis=1)) * dx
        grad -= grad.mean()
        bc1 = 1.0 - b1 ** t
        bc2 = 1.0 - b2 ** t
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        h = h - lr * (m / bc1) / (np.sqrt(v / bc2) + eps)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            lo, hi = -1.0, 1.0
            for _ in range(35):
                mid = (lo + hi) * 0.5
                if np.clip(h + mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h = np.clip(h + (lo + hi) * 0.5, 0, 1)
        if t % 300 == 0:
            lr *= 0.93

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    d1 = min(deadline - 650, t0 + 180)
    hb = make_binary(h0, N0)
    hb, _ = binary_optimize(hb, N0, d1, max_sweeps=2000, anneal=0.0)
    c5 = compute_c5(hb, N0)
    if c5 < best_c5:
        best_c5, best_h = c5, hb.copy()

    if time.time() < deadline - 450:
        d2 = min(deadline - 380, time.time() + 170)
        hb2 = make_binary(best_h, N0)
        hb2, _ = binary_optimize(hb2, N0, d2, max_sweeps=1000, anneal=0.06)
        c5 = compute_c5(hb2, N0)
        if c5 < best_c5:
            best_c5, best_h = c5, hb2.copy()

    if time.time() < deadline - 320:
        d3 = min(deadline - 250, time.time() + 170)
        hc = best_h.astype(float).copy()
        hc += np.random.randn(N0) * 0.015
        hc, _ = continuous_optimize(hc, N0, d3, lr0=0.01)
        c5 = compute_c5(hc, N0)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    if time.time() < deadline - 200:
        d4 = min(deadline - 120, time.time() + 180)
        hb4 = make_binary(best_h, N0)
        hb4, _ = binary_optimize(hb4, N0, d4, max_sweeps=1500, anneal=0.02)
        c5 = compute_c5(hb4, N0)
        if c5 < best_c5:
            best_c5, best_h = c5, hb4.copy()

    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [378, 382, 384, 376, 386]:
        if time.time() > deadline - 100:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        ht = make_binary(h_interp, N_try)
        sub_dl = min(deadline - 50, time.time() + 45)
        ht, _ = binary_optimize(ht, N_try, sub_dl, max_sweeps=500, anneal=0.03)
        c5 = compute_c5(ht, N_try)
        if c5 < best_c5:
            best_c5, best_h = c5, ht.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    if time.time() < deadline - 80:
        d5 = min(deadline - 40, time.time() + 70)
        for attempt in range(5):
            if time.time() > d5:
                break
            hp = make_binary(best_h, len(best_h))
            nf = np.random.randint(4, 10)
            oi = np.where(hp == 1)[0]
            zi = np.where(hp == 0)[0]
            for _ in range(nf):
                pi = int(np.random.randint(len(oi)))
                qi = int(np.random.randint(len(zi)))
                hp[oi[pi]], hp[zi[qi]] = 0.0, 1.0
                oi = np.where(hp == 1)[0]
                zi = np.where(hp == 0)[0]
            sub_dl = min(d5, time.time() + 12)
            hp, _ = binary_optimize(hp, len(best_h), sub_dl, max_sweeps=200, anneal=0.0)
            c5 = compute_c5(hp, len(best_h))
            if c5 < best_c5:
                best_c5, best_h = c5, hp.copy()

    if time.time() < deadline - 3:
        hF = make_binary(best_h, len(best_h))
        hF, _ = binary_optimize(hF, len(best_h), deadline - 1, max_sweeps=300, anneal=0.0)
        c5 = compute_c5(hF, len(best_h))
        if c5 < best_c5:
            best_c5, best_h = c5, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
