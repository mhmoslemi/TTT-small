        valid_plus = (p + deltas >= 0) & (p + deltas < N)
        valid_minus = (p - deltas >= 0) & (p - deltas < N)
        dp[valid_plus] += h[p + deltas[valid_plus]]
        dp[valid_minus] += h[p - deltas[valid_minus]]
        
        # Now for each q in zeros:
        # total_change[k] = dp[k] + dq[k] where dq[k] = 1 - h[q-deltas[k]] - h[q+deltas[k]]
        # new_corr[k] = c[k] + dp[k] + dq[k]
        # We want min over q of max(new_corr)
        
        for q in zeros:
            dq = np.full(2*N-1, 1.0)
            valid_plus_q = (q + deltas >= 0) & (q + deltas < N)
            valid_minus_q = (q - deltas >= 0) & (q - deltas < N)
            dq[valid_plus_q] -= h[q + deltas[valid_plus_q]]
            dq[valid_minus_q] -= h[q - deltas[valid_minus_q]]
            
            new_c = c + dp + dq
            nm = int(new_c.max())
            if nm < best_max:
                best_max = nm
                best_p, best_q = p, q
                best_c = new_c
    
    return best_p, best_q, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        idx = idx[np.random.permutation(len(idx))][:n1 - nh]
        h[idx] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        idx = idx[np.random.permutation(len(idx))][:nh - n1]
        h[idx] = 1
    return h

def binary_opt(h0, N, deadline, n_tries=20, kick_freq=60):
    h = make_binary(h0, N)
    c = fft_corr(h)
    best_int = int(np.max(c))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_sweep_int = int(c.max())
        best_sweep_h = None
        best_sweep_c = None
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_t = fft_corr(h)
            ct_int = int(c_t.max())
            if ct_int < best_sweep_int:
                best_sweep_int = ct_int
                best_sweep_h = h.copy()
                best_sweep_c = c_t.copy()
            h[p] = 1
            h[q] = 0
        if best_sweep_h is not None:
            h = best_sweep_h
            c = best_sweep_c
            stall = 0
            if best_sweep_int < best_int:
                best_int = best_sweep_int
                best_h = h.copy()
        else:
            stall += 1
            if stall > kick_freq:
                n_flips = max(2, N // 15)
                if len(ones) >= n_flips and len(zeros) >= n_flips:
                    i1 = np.random.choice(len(ones), n_flips, replace=False)
                    i2 = np.random.choice(len(zeros), n_flips, replace=False)
                    h[ones[i1]] = 0
                    h[zeros[i2]] = 1
                    c = fft_corr(h)
                stall = 0
    return best_h, best_int * (2.0 / N)

def continuous_adam(h0, N, deadline, k=60, lr0=0.02, temp0=0.003):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    temp = temp0
    stall = 0
    center = N - 1
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        top = np.argpartition(corr, -k)[-k:]
        w = np.exp((corr[top] - corr[top].max()) / max(temp, 1e-10))
        w /= w.sum()
        grad = np.zeros(N)
        for wi, ai in zip(w, top):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                seg = N - L
                grad[L:] += wi * (1.0 - h[:seg])
                grad[:seg] -= wi * h[L:]
            else:
                M = -L
                if M < N:
                    seg = N - M
                    grad[:seg] += wi * (1.0 - h[M:])
                    grad[M:] -= wi * h[:seg]
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        h = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        if t % 400 == 0:
            lr = max(0.002, lr * 0.995)
            temp = max(0.0002, temp * 0.985)
        if stall > 400:
            h = best_h + np.random.randn(N) * 0.02
            h = project_sum(h, target)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(lr, lr0 * 0.7)
            stall = 0
    return best_h, best_pk * (2.0 / N)

    return np.interp(x_new, x_old, h)
