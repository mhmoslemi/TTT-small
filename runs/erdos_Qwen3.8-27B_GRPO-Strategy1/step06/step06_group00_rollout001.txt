                grad[M:] -= h[:N - M]
    return grad / k

def nesterov_opt(h, N, deadline, k=50, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        grad = topk_grad(corr, y, k)
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.993
            beta = min(0.97, 0.88 + it * 0.00012)
    return best_h, best_pk

def compute_C_mat(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def binary_il(h, N, deadline, max_iters=800, sample_sz=80, temp=0.0):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - n1, replace=False)] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    it = 0
    while it < max_iters and time.time() < deadline:
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 200:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        if int(it) % 40 == 0 or True:
            C = compute_C_mat(h, N)
        Co = C[os_]
        Cz = C[zs]
        newP = (corr[None, None, :] + Co[None, :, :] - Cz[:, None, :]).max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-12:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
        elif temp > 0 and np.random.random() < temp:
            h[os_[np.random.randint(ns)]] = 0
            h[zs[np.random.randint(nz)]] = 1
        else:
            h[os_[np.random.randint(ns)]] = 0
            h[zs[np.random.randint(nz)]] = 1
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    dx0 = 2.0 / N0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    def elapsed():
        return time.time() - t0

    # Phase 1: Nesterov continuous optimization (2 restarts)
    for r in range(2):
        if time.time() > deadline - 400:
            break
        d1 = min(deadline - 380 + r * 30, time.time() + 120)
        noise = 0.06 if r == 0 else 0.03
        hc = best_h + np.random.randn(N0) * noise
        hc = project_01(hc, N0 / 2.0)
        hn, pk = nesterov_opt(hc, N0, d1, k=60, lr0=0.012)
        c5n = pk * dx0
        if c5n < best_c5:
            best_c5 = c5n
            best_h = hn.copy()

    # Phase 2: Binary IL search on best
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 150)
        hb = (best_h > 0.5).astype(float)
        hs, pk2 = binary_il(hb, N0, d2, max_iters=700, sample_sz=90)
        c5_2 = pk2 * dx0
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = hs.copy()

    # Phase 3: Multi-resolution with different N values
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    N_tries = []
    for n in range(340, 660, 12):
        if n % 2 == 0:
            N_tries.append(n)
    np.random.shuffle(N_tries)
    N_tries = sorted(N_tries[:12])
    for Nt in N_tries:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        h_int = np.interp(x_new, x_best, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh = Nt // 2
        n1 = int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            h_bt[np.random.choice(idx, n1 - nh, replace=False)] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            h_bt[np.random.choice(idx, nh - n1, replace=False)] = 1
        sub_dl = min(deadline - 100, time.time() + 60)
        h_rt, pk_rt = binary_il(h_bt, Nt, sub_dl, max_iters=300, sample_sz=70)
        c5_rt = pk_rt * (2.0 / Nt)
        if c5_rt < best_c5:
            best_c5 = c5_rt
            best_h = h_rt.copy()
            x_best = np.linspace(0, 2, Nt, endpoint=False)

    # Phase 4: Second Nesterov polish on best
    if time.time() < deadline - 60:
        d4 = min(deadline - 40, time.time() + 80)
        hf = best_h + np.random.randn(len(best_h)) * 0.02
        hf = project_01(hf, len(best_h) / 2.0)
        hF, pkF = nesterov_opt(hf, len(best_h), d4, k=80, lr0=0.008)
        c5F = pkF * (2.0 / len(best_h))
        if c5F < best_c5:
            best_c5 = c5F
            best_h = hF.copy()

    # Phase 5: Final binary polish
    if time.time() < deadline - 5:
        hB = (best_h > 0.5).astype(float)
        hB, pkB = binary_il(hB, len(best_h), deadline - 1, max_iters=150, sample_sz=100)
        c5B = pkB * (2.0 / len(best_h))
        if c5B < best_c5:
            best_c5 = c5B
            best_h = hB

    # Final exact evaluation
    Nf = len(best_h)
    final_c5 = compute_c5(best_h, Nf)
    h_bin = (best_h > 0.5).astype(float)
    final_c5b = compute_c5(h_bin, Nf)
    if final_c5b < final_c5:
        best_h = h_bin
        final_c5 = final_c5b
    return best_h, final_c5, Nf
