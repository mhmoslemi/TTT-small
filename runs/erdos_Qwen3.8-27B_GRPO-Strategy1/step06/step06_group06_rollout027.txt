def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]

def corr_max(h):
    return float(np.max(fft_corr(h)))

def proj_box_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    # Binary search for threshold
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        clipped = np.clip(h - mid, 0, 1)
        if clipped.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)*0.5, 0, 1)

def subgrad_step(h, N, top_k, lr):
    """Take a subgradient step on max_k corr(h,1-h)"""
    corr = fft_corr(h)
    center = N - 1
    idx = np.argpartition(corr, -top_k)[-top_k:]
    # Weight active k's by their corr value
    weights = corr[idx]
    weights = np.exp(weights * 5.0)  # Softmax-like weighting
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for w, ai in zip(weights, idx):
        s = int(ai) - center
        if s == 0:
            grad += w * (1.0 - 2.0*h)
        elif s > 0:
            grad[:N-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:N-s]
        else:
            s2 = -s
            grad[s2:] += w * (1.0 - h[:N-s2])

    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def grad_topk(h, N, top_k, sharp):
    corr = fft_corr(h)
    center = N-1
    idx = np.argpartition(corr, -top_k)[-top_k:]
    w = (corr[idx]-corr[idx].max()+1.0)**sharp
    w /= w.sum()
    grad = np.zeros(N)
    for wi, ai in zip(w, idx):
        s = int(ai)-center
        if s == 0:
            grad += wi*(1.0-2.0*h)
        elif s > 0:
            grad[:N-s] += wi*(1.0-h[s:])
            grad[s:] -= wi*h[:N-s]
        else:
            s2=-s
            grad[s2:] += wi*(1.0-h[:N-s2])
            grad[:N-s2] -= wi*h[s2:]
    return grad, float(corr.max())

def cont_opt(h, N, deadline, top_k=50, lr0=0.004, sharp=8.0):
    h = h.astype(float).copy()
    m=np.zeros(N); v=np.zeros(N)
    b1,b2,eps=0.8,0.999,1e-8
    t=0; best=float(np.max(fft_corr(h))); best_h=h.copy(); lr=lr0
    while time.time()<deadline:
        g,pk=grad_topk(h,N,top_k,sharp)
        if pk<best: best=pk; best_h=h.copy()
        t+=1
        m=b1*m+(1-b1)*g; v=b2*v+(1-b2)*g*g
        h=proj_box_sum(h-lr*m/(1-b1**t)/np.sqrt(v/(1-b2**t)+eps),N/2.0)
        if t%400==0: lr*=0.985
        if t%2500==0: h=proj_box_sum(h+np.random.randn(N)*0.01,N/2.0)
    return best_h, best

def bin_search(h,N,deadline,T0=0.05,mni=2000,dd=0.35):
    h=(h>0.5).astype(float)
    tgt=N//2; n1=int(h.sum())
    if n1>tgt:
        ix=np.where(h==1)[0]; h[ix[np.random.permutation(len(ix))[:n1-tgt]]]=0
    elif n1<tgt:
        ix=np.where(h==0)[0]; h[ix[np.random.permutation(len(ix))[:tgt-n1]]]=1
    s_arr=np.arange(-(N-1),N)
    corr=fft_corr(h); best=float(corr.max()); best_h=h.copy()
    cur=best; no_imp=0; T=T0; cool=0.9996
    ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
    while time.time()<deadline and T>0.0003:
        if np.random.random()<dd:
            ai=int(np.argmax(corr)); s=ai-(N-1)
            if s>0:
                ir=np.arange(0,N-s)
                msk=(h[ir]==1)&(h[ir+s]==0)
                vp=ir[msk]; vq=ir[msk]+s
            elif s<0:
                ir=np.arange(-s,N)
                msk=(h[ir]==1)&(h[ir+s]==0)
                vp=ir[msk]; vq=ir[msk]+s
            else:
                vp=ones; vq=zeros
            if len(vp)>0:
                ii=np.random.randint(len(vp))
                p=int(vp[ii]); q=int(vq[ii])
            else:
                p=int(ones[np.random.randint(len(ones))]); q=int(zeros[np.random.randint(len(zeros))])
        else:
            p=int(ones[np.random.randint(len(ones))])
            q=int(zeros[np.random.randint(len(zeros))])
        ps=p+s_arr; pm=p-s_arr; qs=q+s_arr; qm=q-s_arr
        vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
        vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
        d=np.zeros(2*N-1)
        d[vps]-=(1.0-h[np.clip(ps,0,N-1)[vps]])
        d[vpm]+=h[np.clip(pm,0,N-1)[vpm]]
        d[vqs]+=(1.0-h[np.clip(qs,0,N-1)[vqs]])
        d[vqm]-=h[np.clip(qm,0,N-1)[vqm]]
        nc=corr+d; npk=float(nc.max()); delta=npk-cur
        if delta<0 or np.random.random()<np.exp(-delta/max(T,1e-8)):
            h[p]=0; h[q]=1; corr=nc; cur=npk
            ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
            if npk<best: best=npk; best_h=h.copy(); no_imp=0
            else: no_imp+=1
        else:
            no_imp+=1
        T*=cool
        if no_imp>mni:
            no_imp=0; h=best_h.copy(); corr=fft_corr(h); cur=best
            ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
            T=max(T,T0*0.4)
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0=time.time(); dl=t0+budget_s-5
    np.random.seed(seed)
    h0=np.array(initial_h_values,dtype=float)
    N0=len(h0); dx=2.0/N0
    best_h=h0.copy(); best_c5=float(np.max(fft_corr(h0)))*dx

    # Phase 1: Multiple continuous restarts with varied params
    for i in range(5):
        rem=dl-time.time()
        if rem<160: break
        alloc=min(60+15*i, rem-140)
        ns=0.015+0.013*i
        hc=proj_box_sum(h0+np.random.randn(N0)*ns,N0/2.0)
        hk,pk=cont_opt(hc,N0,time.time()+alloc,top_k=25+15*i,lr0=0.0025+0.001*i,sharp=4.0+6*i)
        c5=pk*dx
        if c5<best_c5: best_c5=c5; best_h=hk.copy()

    # Phase 2: Directed binary search from best
    rem=dl-time.time()
    if rem>100:
        alloc=min(200,rem-70)
        hb,pb=bin_search((best_h>0.5).astype(float),N0,time.time()+alloc,T0=0.06,mni=2500,dd=0.35)
        c5=pb*dx
        if c5<best_c5: best_c5=c5; best_h=hb.copy()

    # Phase 3: Continuous refinement around binary result
    rem=dl-time.time()
    if rem>50:
        alloc=min(70,rem-40)
        hc=proj_box_sum(best_h+np.random.randn(len(best_h))*0.012,len(best_h)/2.0)
        hk,pk=cont_opt(hc,len(best_h),time.time()+alloc,top_k=60,lr0=0.0015,sharp=14.0)
        c5=pk*(2.0/len(best_h))
        if c5<best_c5: best_c5=c5; best_h=hk.copy()

    # Phase 4: Multi-N exploration
    for Nt in [280,320,340,400,440,480,540,640,780,960]:
        rem=dl-time.time()
        if rem<25: break
        xo=np.linspace(0,2,len(best_h),endpoint=False)
        xn=np.linspace(0,2,Nt,endpoint=False)
        hi=np.interp(xn,xo,best_h)
        hbt=(hi>0.5).astype(float)
        tgt=Nt//2; n1=int(hbt.sum())
        if n1>tgt:
            ix=np.where(hbt==1)[0]; hbt[ix[np.random.permutation(len(ix))[:n1-tgt]]]=0
        elif n1<tgt:
            ix=np.where(hbt==0)[0]; hbt[ix[np.random.permutation(len(ix))[:tgt-n1]]]=1
        a1=min(16,rem-18)
        if a1>4:
            hr,pr=bin_search(hbt,Nt,time.time()+a1,T0=0.04,mni=600,dd=0.35)
            c5r=pr*(2.0/Nt)
            if c5r<best_c5: best_c5=c5r; best_h=hr.copy()
        rem=dl-time.time()
        if rem>12 and len(best_h)==Nt:
            hc=proj_box_sum(best_h+np.random.randn(Nt)*0.01,Nt/2.0)
            ht,pt=cont_opt(hc,Nt,time.time()+min(7,rem-4),top_k=45,lr0=0.0025,sharp=10)
            c5t=pt*(2.0/Nt)
            if c5t<best_c5: best_c5=c5t; best_h=ht.copy()

    # Phase 5: Final binary polish
    rem=dl-time.time()
    if rem>10:
        Nf=len(best_h)
        hf,pf=bin_search((best_h>0.5).astype(float),Nf,dl-2,T0=0.015,mni=1200,dd=0.3)
        c5f=pf*(2.0/Nf)
        if c5f<best_c5: best_c5=c5f; best_h=hf.copy()

    fc5=float(np.max(fft_corr(best_h)))*(2.0/len(best_h))
    return best_h,fc5,len(best_h)
