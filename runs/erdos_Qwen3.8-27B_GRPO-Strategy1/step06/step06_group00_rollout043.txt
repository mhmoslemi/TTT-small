import numpy as np
import time

def compute_corr(h):
    N = len(h)
    return np.correlate(h, 1.0 - h, 'full').astype(float)

def compute_c5(h):
    N = len(h)
    return float(compute_corr(h).max()) * (2.0 / N)

def init_patterns(h, N):
    lags = np.arange(-(N-1), N)  # shape (2N-1,)
    o_idx = np.arange(N)[:, None]  # shape (N, 1)
    pos_p = o_idx + lags[None, :]  # shape (N, 2N-1)
    pos_m = o_idx - lags[None, :]  # shape (N, 2N-1)
    P_plus = np.where((pos_p >= 0) & (pos_p < N), h[np.clip(pos_p, 0, N-1)], 0.0)
    P_minus = np.where((pos_m >= 0) & (pos_m < N), h[np.clip(pos_m, 0, N-1)], 0.0)
    return P_plus, P_minus

def binary_greedy_il(h, N, deadline, max_restarts=5):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    
    best_h = h.copy()
    best_c = compute_c5(h)
    corr = compute_corr(h)
    P_plus, P_minus = init_patterns(h, N)
    
    for restart in range(max_restarts):
        if time.time() > deadline:
            break
        # If restart > 0, perturb
        if restart > 0:
            n_perturb = N // 20  # 5%
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(n_perturb):
                o = np.random.choice(ones)
                z = np.random.choice(zeros)
                h[o], h[z] = h[z], h[o]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = compute_corr(h)
            P_plus, P_minus = init_patterns(h, N)
        
        # Greedy loop
        improved = True
        while improved and time.time() < deadline:
            improved = False
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            
            current_max = corr.max()
            
            # Batch scan
            best_val = current_max
            best_o, best_z = -1, -1
            batch_sz = 20
            for i in range(0, len(ones), batch_sz):
                o_batch = ones[i:i+batch_sz]
                # Delta shape: (batch, len(zeros), 2N-1)
                # To save memory, sub-batch zeros too
                for j in range(0, len(zeros), batch_sz):
                    z_batch = zeros[j:j+batch_sz]
                    delta = (P_plus[o_batch][:, None, :] - P_plus[z_batch][None, :, :] +
                             P_minus[o_batch][:, None, :] - P_minus[z_batch][None, :, :])
                    new_corr = corr[None, None, :] + delta
                    new_max = new_corr.max(axis=2)  # shape (batch_o, batch_z)
                    mi = np.unravel_index(np.argmin(new_max), new_max.shape)
                    val = new_max[mi]
                    if val < best_val - 1e-12:
                        best_val = val
                        best_o = o_batch[mi[0]]
                        best_z = z_batch[mi[1]]
            
            if best_o >= 0:
                # Apply swap
                h[best_o], h[best_z] = h[best_z], h[best_o]
                # Update corr
                corr += (P_plus[best_o] - P_plus[best_z] + P_minus[best_o] - P_minus[best_z])
                # Update patterns (approximate - recompute for safety)

                improved = True
                c5_val = corr.max() * (2.0 / N)
                if c5_val < best_c:
                    best_c = c5_val
                    best_h = h.copy()
        
        # Check exact
        c5_exact = compute_c5(h)
        if c5_exact < best_c:
            best_c = c5_exact
            best_h = h.copy()
    
    return best_h, best_c

import time

def compute_corr(h):
    N = len(h)
    return np.correlate(h, 1.0 - h, 'full').astype(np.float64)

def compute_c5(h):
    N = len(h)
    return float(np.correlate(h, 1.0 - h, 'full').max()) * (2.0 / N)

def init_patterns(h, N):
    lags = np.arange(-(N-1), N)
    o_idx = np.arange(N)[:, None]
    pos_p = o_idx + lags[None, :]
    pos_m = o_idx - lags[None, :]
    P_plus = np.where((pos_p >= 0) & (pos_p < N), h[np.clip(pos_p, 0, N-1)], 0.0)
    P_minus = np.where((pos_m >= 0) & (pos_m < N), h[np.clip(pos_m, 0, N-1)], 0.0)
    return P_plus, P_minus

def update_patterns(P_plus, P_minus, N, o, z, val_o, val_z):
    """Update patterns after h[o]: val_o→0, h[z]: val_z→1 (for binary swap)."""
    w = np.arange(N)
    # P_plus: h[valid(w + lag)] changes at positions o and z
    r_po = o - w + N - 1  # where w+lag = o
    r_pz = z - w + N - 1  # where w+lag = z
    P_plus[w, r_po] -= val_o  # was val_o (1), now 0
    P_plus[w, r_pz] += (1 - val_z)  # was val_z (0), now 1
    # P_minus: h[valid(w - lag)] changes
    r_mo = w - o + N - 1  # where w-lag = o
    r_mz = w - z + N - 1  # where w-lag = z
    P_minus[w, r_mo] -= val_o
    P_minus[w, r_mz] += (1 - val_z)

                z = int(np.random.choice(zeros))
                h[o], h[z] = h[z], h[o]
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            corr = compute_corr(h)
            P_plus, P_minus = init_patterns(h, N)
        
        # Greedy loop
        stall = 0
        while time.time() < deadline:

            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            cur_max = float(corr.max())
            best_val = cur_max
            best_o, best_z = -1, -1
            
            # Batch scan
            bs = 25
            for i in range(0, len(ones), bs):
                ob = ones[i:i+bs]
                for j in range(0, len(zeros), bs):
                    zb = zeros[j:j+bs]
                    delta = (P_plus[ob][:, None, :] - P_plus[zb][None, :, :] +
                             P_minus[ob][:, None, :] - P_minus[zb][None, :, :])
                    nm = (corr[None, None, :] + delta).max(axis=2)
                    mi = np.unravel_index(np.argmin(nm), nm.shape)
                    v = float(nm[mi])

                        best_val = v
                        best_o = int(ob[mi[0]])
                        best_z = int(zb[mi[1]])
            
            if best_o < 0:
                stall += 1
                if stall > 3:
                    break
                continue

            # Incremental update (approximate - good enough for next iteration's ranking)
            update_patterns(P_plus, P_minus, N, best_o, best_z, 1.0, 0.0)
            corr += (P_plus[best_o] - P_plus[best_z] + P_minus[best_o] - P_minus[best_z])

    if c5_exact < best_c:
        best_c = c5_exact
        best_h = h.copy()

Actually, let me also add a "steepest descent in correlation space" idea. Instead of just doing single swaps, what if I do a "gradient step" in the space of h?

The gradient of max_k corr[k] with respect to h[i] (for binary h, this is the change in corr when flipping h[i]):
- If h[i] = 1 (flipping to 0): Δcorr[k] = -(1-h[i+k]) + h[i-k] (simplified)
- If h[i] = 0 (flipping to 1): Δcorr[k] = +(1-h[i+k]) - h[i-k]

The "descent direction" for position i is -Δcorr, and we want to flip positions that most reduce the max. This is exactly what the greedy IL does (find the swap that most reduces the max).
