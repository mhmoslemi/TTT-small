def alternating_minmax(h, N, deadline, n_iter_max=5000):
    h = h.astype(float).copy()
    dx = 2.0 / N
    num_shifts = 2*N - 1
    center = N - 1
    
    # Initial mu: uniform
    mu = np.ones(num_shifts) / num_shifts
    
    best_h = h.copy()
    best_c5 = np.max(fft_corr(h))
    
    lr = 0.05
    tau = 0.01  # temperature for mu update
    
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        
        # Compute current corr
        corr = fft_corr(h)  # length 2N-1
        
        # Update mu (softmax on negative corr)
        mu = np.exp(-corr / tau)
        mu /= mu.sum()

        # corr[k] = sum_i h[i] * (1-h[i-k]) for linear correlation
        # d/dh[i] = sum_k mu[k] * [(1-h[i-k]) - h[i+k]] (with bounds)

        # d/dh[i] corr[center+L] = (1-h[i-L])_valid - h[i+L]_valid
        for L in range(-(N-1), N):

                # Actually for continuous: d/dh[i] of h[i]*(1-h[i]) = 1-2h[i]

                #           for i in [N-L, N-1]: -h[i-L]
                g[:N-L] += mu_L * (1 - h[L:])
                g[N-L:] -= mu_L * h[:L]
            else:

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_Dmat(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp
