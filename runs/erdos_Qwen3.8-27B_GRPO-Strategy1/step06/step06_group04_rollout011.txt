def fft_corr(h):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:

            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh - n1]
            h[idx] = 1
    return h

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def soft_gradient(h, c, N, frac=0.20, temp=0.3):
    center = N - 1
    k = max(8, int(frac * (2 * N - 1)))
    top = np.argpartition(c, -k)[-k:]

    w = np.exp((c[top] - cm) / max(temp, 1e-8))

            grad[L:] += wi * (1.0 - h[:seg])

            M = -L

        zeros = np.where(h == 0)[0]

    # Final binary nudge
    if time.time() < deadline - 2:
        hB = make_binary(best_h, len(best_h))
        cB = compute_c5(hB)
        if cB < best_c5:
            best_c5, best_h = cB, hB.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
