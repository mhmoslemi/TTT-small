import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def soft_max_grad_topk(corr, h, N, k=50, T=20.0):
    """Compute gradient of soft-max of top-k correlation values."""
    # Fallback to hard max gradient if T is very large
    if T > 500:
        return hard_max_grad(corr, h, N, k)
    
    # Use soft-max weights over top-k
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    weights = np.exp(T * (c_top - c_top.max()))
    weights /= weights.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for i, ai in enumerate(idx):
        w = weights[i]
        s = int(ai - center)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            # (1-h[j-s]) for valid j-s, minus h[j+s] for valid j+s
            # j-s valid: j >= s, j < N → h[s:N] gets (1-h[0:N-s])
            # j+s valid: j >= 0, j < N-s → h[0:N-s] gets -h[s:N]
            grad[s:] += w * (1.0 - h[:N-s])
            grad[:N-s] -= w * h[s:]
        else:
            M = -s
            # j-s valid: j >= 0, j < N-M → h[0:N-M] gets (1-h[M:N])  
            # j+s valid: j >= M, j < N → h[M:N] gets -h[0:N-M]
            grad[:N-M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:N-M]
    return grad / k  # normalize

def hard_max_grad(corr, h, N, k=10):
    idx = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    for ai in idx:
        s = int(ai - center)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += (1.0 - h[:N-s])
            grad[:N-s] -= h[s:]
        else:
            M = -s
            grad[:N-M] += (1.0 - h[M:])
            grad[M:] -= h[:N-M]
    return grad / len(idx)

def project_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    # Bisection to find offset
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        grad = soft_max_grad_topk(corr, y, N, k=k, T=T)

        m = beta * (h_new - h)

        it += 1

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(corr, h, N, k, T):
    idx = np.argpartition(corr, -k)[-k:]
    c_top = corr[idx]
    w = np.exp(T * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        s = int(idx[ii] - center)
        wi = w[ii]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad / k

def nesterov_opt(h_init, N, deadline, k=40, T=60.0, lr0=0.02):
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = h.copy()
        grad = soft_max_grad(corr, y, N, k, T)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.996
            beta = min(0.98, 0.85 + it * 0.00015)
        if it % 400 == 0:
            cs = np.sort(corr)[::-1]
            gap = cs[0] - cs[min(k, len(cs) - 1)]
            T = max(15.0, min(300.0, T * (0.9 if gap < 0.005 else 1.05)))
    return best_h, best_pk * dx

def ensure_binary(h, N):
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    target = N // 2
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(77)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_local_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, perturb_every=40):

    best_h = h.copy()

            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        s_all = shifts[None, :]
        pm = p_all - s_all
        pp = p_all + s_all
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        pm_c = np.clip(pm, 0, N - 1)
        pp_c = np.clip(pp, 0, N - 1)
        hm = np.where(vm, h[pm_c], 0.0)
        hp = np.where(vp, h[pp_c], 0.0)
