import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

        perm = np.random.permutation(len(idx))
        h[idx[perm[:n1 - nh]]] = 0
    elif n1 < nh:

        perm = np.random.permutation(len(idx))

def local_search_exhaustive(h, N, deadline, top_m=30, anneal_after=50, anneal_prob=0.3):
    h = fix_binary_sum(h.copy(), N)
    dx = 2.0 / N
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    no_improve = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline:
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        Q = len(zeros)
        nhalf = N - 1
        top_idx = np.argsort(c)[-top_m:]
        L_vals = (top_idx - nhalf).astype(int)
        M = len(L_vals)
        c_top = c[top_idx]
        
        # Build A and B matrices: (len(ones), M) and (len(zeros), M)
        p_plus = ones[:, None] + L_vals[None, :]
        p_minus = ones[:, None] - L_vals[None, :]
        q_plus = zeros[:, None] + L_vals[None, :]
        q_minus = zeros[:, None] - L_vals[None, :]
        
        A = (np.where((p_plus >= 0) & (p_plus < N), h[np.clip(p_plus, 0, N-1)], 0.0) +
             np.where((p_minus >= 0) & (p_minus < N), h[np.clip(p_minus, 0, N-1)], 0.0))
        B = (np.where((q_plus >= 0) & (q_plus < N), h[np.clip(q_plus, 0, N-1)], 0.0) +
             np.where((q_minus >= 0) & (q_minus < N), h[np.clip(q_minus, 0, N-1)], 0.0))
        
        # new_c[p,q,j] = c_top[j] + A[p,j] - B[q,j]
        new_c = c_top[None, None, :] + A[:, None, :] - B[None, :, :]
        max_new = new_c.max(axis=2)
        
        best_flat = int(np.argmin(max_new))
        bp = best_flat // Q
        bq = best_flat % Q
        
        if max_new[bp, bq] < P - 1e-12:
            h[ones[bp]] = 0.0
            h[zeros[bq]] = 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c = fft_corr(h)
        else:
            # Stuck: try annealing
            if no_improve > anneal_after and np.random.random() < anneal_prob:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p] = 0.0
                h[q] = 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                c = fft_corr(h)
            else:
                break
    
    return best_h, best_c * dx

def subgrad_vec(y, corr, top_k, N):
    grad = np.zeros(N)
    center = N - 1
    top = np.argpartition(corr, -top_k)[-top_k:]
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += (1.0 - y[:N - L])
            grad[:N - L] -= y[L:]
        else:
            M = -L
            grad[:N - M] += (1.0 - y[M:])
            grad[M:] -= y[:N - M]
    grad /= len(top)
    return grad

def nesterov_opt(h, N, deadline, k=80, lr0=0.015, beta0=0.90):
    h = h.astype(float).copy()

    best_h = h.copy()

        it += 1
        if it % 100 == 0:

            beta = min(0.97, beta0 + it * 0.00012)

                best_c5, best_h = cr, hr.copy()

    # Phase 6: Final exhaustive polish
    if time.time() < deadline - 3:
        hF = fix_binary(best_h, len(best_h))
        hF, cF = exhaustive_ls(hF, len(best_h), deadline - 1, top_m=40)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, final_c5, len(best_h)
