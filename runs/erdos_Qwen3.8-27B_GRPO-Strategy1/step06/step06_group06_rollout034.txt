import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

def corr_val(h):
    return float(fft_corr(h).max())

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h, N):
    target = N // 2
    h = h.copy()
    n1 = int(np.round(h.sum()))
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        n_remove = n1 - target
        perm = np.random.permutation(len(idx))[:n_remove]
        h[idx[perm]] = 0.0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        n_add = target - n1
        perm = np.random.permutation(len(idx))[:n_add]
        h[idx[perm]] = 1.0
    return h

def compute_grad(corr, h, N, top_k):
    center = N - 1
    idx = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros(N)
    w = 1.0 / top_k
    for ai in idx:
        s = int(ai - center)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:N-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:N-s]
        else:
            s2 = -s
            grad[s2:] += w * (1.0 - h[:N-s2])
            grad[:N-s2] -= w * h[s2:]
    return grad

def continuous_phase(h, N, deadline, lr=0.004, top_k=50):
    h = np.asarray(h, dtype=float).copy()
    m1 = np.zeros(N)
    m2 = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    best = corr_val(h)
    best_h = h.copy()
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best:
            best = pk
            best_h = h.copy()
        grad = compute_grad(corr, h, N, top_k)
        t += 1
        m1 = b1 * m1 + (1 - b1) * grad
        m2 = b2 * m2 + (1 - b2) * grad * grad
        mh = m1 / (1 - b1**t)
        vh = m2 / (1 - b2**t)
        h = proj_sum(h - lr * mh / (np.sqrt(vh) + eps), N / 2.0)
        if t % 400 == 0:
            lr *= 0.99
    return best_h, best

def binary_phase(h, N, deadline, T0=0.06, cool=0.9994):
    h = fix_count(h, N)
    corr = fft_corr(h)
    best_pk = corr.max()
    best_h = h.copy()
    curr_pk = best_pk
    T = T0
    no_imp = 0
