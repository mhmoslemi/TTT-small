import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h.astype(float), L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h, N, rng):
    target = N // 2
    h = h.copy().astype(float)
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        c = rng.choice(len(idx), n1 - target, replace=False)
        h[idx[c]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        c = rng.choice(len(idx), target - n1, replace=False)
        h[idx[c]] = 1
    return h

def swap_delta(h, N, p, q):
    s = np.arange(-(N - 1), N)
    d = np.zeros(2 * N - 1)
    ps = p + s; pm = p - s
    m1 = (ps >= 0) & (ps < N); m2 = (pm >= 0) & (pm < N)
    d[m1] -= (1.0 - h[ps[m1]])
    d[m2] += h[pm[m2]]
    qs = q + s; qm = q - s
    m3 = (qs >= 0) & (qs < N); m4 = (qm >= 0) & (qm < N)
    d[m3] += (1.0 - h[qs[m3]])
    d[m4] -= h[qm[m4]]
    return d

def greedy_finish(h, N, deadline, rng):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        cur = float(corr.max())
        found = False
        n_try = min(len(ones) * len(zeros), 5000)
        for _ in range(n_try):
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            d = swap_delta(h, N, p, q)
            nm = float((corr + d).max())
            if nm < cur - 1e-12:
                h[p] = 0; h[q] = 1
                corr = corr + d
                cur = nm
                if cur < best_pk:
                    best_pk = cur; best_h = h.copy()
                found = True
                break
        if not found:
            break
    return best_h, best_pk

def anneal(h, N, deadline, rng, T0=0.05, cool=0.9993, stall_lim=600, kick_size=1):
    h = h.astype(float).copy()
    h = fix_count(h, N, rng)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    stall = 0
    T = T0
    while time.time() < deadline and T > 0.0003:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = int(ones[rng.randint(len(ones))])
        q = int(zeros[rng.randint(len(zeros))])
        d = swap_delta(h, N, p, q)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        if delta < -1e-12 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h[p] = 0; h[q] = 1
            corr = nc; cur_pk = npk
            if npk < best_pk - 1e-12:
                best_pk = npk; best_h = h.copy(); stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T *= cool
        if stall > stall_lim:
            if kick_size == 1:
                h = best_h.copy()
                corr = fft_corr(h); cur_pk = best_pk
            else:
                # Kick: flip kick_size random pairs
                for _ in range(kick_size):
                    oo = np.where(h == 1)[0]
                    zz = np.where(h == 0)[0]
                    if len(oo) < 2 or len(zz) < 2:

                    pp = int(oo[rng.randint(len(oo))])
                    qq = int(zz[rng.randint(len(zz))])
                    h[pp] = 0; h[qq] = 1
                h = fix_count(h, N, rng)
                corr = fft_corr(h); cur_pk = float(corr.max())
            stall = 0
            T = T0 * 0.4
    # Greedy finish after annealing
    d_rem = deadline - time.time()
    if d_rem > 3:
        best_h, best_pk = greedy_finish(best_h, N, deadline, rng)
    return best_h, best_pk
