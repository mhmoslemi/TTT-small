def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 1e-10:

    lo, hi = -1.0, 1.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_max(corr, tau):
    """Log-sum-exp approximation of max."""
    m = corr.max()
    if tau * m > 500:
        return m
    s = np.sum(np.exp(tau * (corr - m)))
    return m + np.log(s) / tau

def lse_weights(corr, tau):
    m = corr.max()
    if tau * m > 500:
        w = np.zeros_like(corr)
        w[corr == m] = 1.0
        w /= w.sum()
        return w
    e = np.exp(tau * (corr - m))
    return e / e.sum()

def compute_D(h, N):
    """Compute the effect of flipping each position on the correlation."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

        corr = fft_corr(y)
        # Anneal tau
        tau = tau0 + (tau_final - tau0) * min(1.0, it / 300.0)

            best_h = h.copy()

            else:

                    grad[:N - M] += wi * (1.0 - y[M:])

        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.98
            beta = min(0.95, 0.85 + it * 0.00005)
    return best_h, best_pk * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N, rng=None):
    nh = N // 2
    h = np.clip(h, 0, 1)
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    if rng is None:
        rng = np.random.RandomState(42 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def lse_weights(corr, tau):
    m = corr.max()
    t = tau * (corr - m)
    t_max = t.max()
    if t_max > 500:
        w = np.zeros_like(corr)
        w[t == t_max] = 1.0
        w /= w.sum()
        return w
    e = np.exp(t)
    return e / e.sum()

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, perturb_every=0, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.astype(float).copy()
    nh = N // 2
    if int(h.sum()) != nh:
        h = ensure_sum(h, N, rng)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 50:
            n_s = max(50, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones.copy()
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                n_flips = rng.randint(2, 5)
                for _ in range(n_flips):
                    ttl = len(ones)
                    tzt = len(zeros)
                    if ttl < 1 or tzt < 1:
                        break

                    q = int(zeros[rng.randint(tzt)])
                    h[p], h[q] = 0.0, 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                c = fft_corr(h)
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def nesterov_lse(h, N, deadline, k=80, lr0=0.012, tau0=4.0, tau_final=60.0):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        tau = tau0 + (tau_final - tau0) * min(1.0, it / 400.0)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        w = lse_weights(corr, tau)
        thresh = w.max() * 0.005
        top_idx = np.where(w > thresh)[0]
        if len(top_idx) > k:
            top_idx = np.argpartition(corr, -k)[-k:]
        elif len(top_idx) < 5:
            top_idx = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top_idx:
            wi = w[ai]
            if wi < 1e-8:
                continue
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N - L])
                grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:N - M]
        gn = np.linalg.norm(grad)
        if gn > 1e-12:
            grad /= gn
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.97
            beta = min(0.95, 0.85 + it * 0.00006)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
