    qq_m = q_pos - d_list
    vp2 = (qq_p >= 0) & (qq_p < N) & nz
    vm2 = (qq_m >= 0) & (qq_m < N) & nz
    dq = np.where(vm2, (1.0 - h[np.clip(qq_m, 0, N - 1)]), 0.0)
    dq += np.where(vp2, -h[np.clip(qq_p, 0, N - 1)], 0.0)
    
    new_top = c[top_idx] + dp + dq
    rest_mask = np.ones(len(c), dtype=bool)
    rest_mask[top_idx] = False
    rest_max = float(c[rest_mask].max())
    new_P = max(float(new_top.max()), rest_max)
    delta = new_P - P
    if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
        h[p_pos] = 0.0
        h[q_pos] = 1.0
        return h, new_P, True
    return h, P, False

def simulated_anneal(h, N, deadline, n_steps, T0, T1):
    h = balance_binary(h.astype(float).copy(), N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    log_ratio = np.log(T1 / T0)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = T0 * np.exp(log_ratio * (step / max(n_steps - 1, 1)))
        c = fft_corr(h)
        P = float(c.max())
        h, P, _ = anneal_step(h, c, P, T, N)
        if P < best_c:
            best_c = P
            best_h = h.copy()
    return best_h, best_c

def continuous_polish(h, N, deadline, k=60, lr0=0.015):
    h = project_sum(h.astype(float).copy(), N / 2.0)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.90
    center = N - 1
    for it in range(5000):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argsort(corr)[-k:]
        d_list = (top - center).astype(float)
        grad = np.zeros(N)
        nz = d_list != 0
        for di in d_list:
            if di == 0:
                grad += (1.0 - 2.0 * y)
            else:
                di_i = int(di)
                if di_i > 0:
                    grad[di_i:] += (1.0 - y[:N - di_i])
                    grad[:N - di_i] -= y[di_i:]
                else:
                    M = -di_i
                    if M < N:
                        grad[:N - M] += (1.0 - y[M:])
                        grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        if it % 100 == 0:
            lr *= 0.993
            beta = min(0.97, 0.90 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    
    # Phase 1: Intensive binary search
    d1 = min(deadline - 450, t0 + 280)
    h1, _ = binary_search(best_h, N0, d1, 2000, 180, 22, 4)
    c5_1 = compute_c5(h1, N0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()
    
    # Phase 2: Simulated annealing
    if time.time() < deadline - 280:
        d2 = min(deadline - 220, time.time() + 130)
        h2, _ = simulated_anneal(best_h, N0, d2, 6000, 0.5, 0.004)
        c5_2 = compute_c5(h2, N0)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
    
    # Phase 3: Continuous polish
    if time.time() < deadline - 200:
        d3 = min(deadline - 160, time.time() + 60)
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        hc, c5c = continuous_polish(hc, N0, d3, k=70, lr0=0.012)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc.copy()
    
    # Phase 4: Multi-resolution
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    res_list = [340, 360, 400, 420, 440, 460, 500, 540, 600, 700]
    for N_try in res_list:
        if time.time() > deadline - 40:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = balance_binary(h_interp, N_try)
        sub_dl = min(deadline - 25, time.time() + 35)
        h_rt, _ = binary_search(h_bt, N_try, sub_dl, 500, 140, 18, 3)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)
    
    # Phase 5: More binary search on best
    if time.time() < deadline - 100:
        d5 = min(deadline - 70, time.time() + 80)
        h5, _ = binary_search(best_h, len(best_h), d5, 1500, 200, 25, 4)
        c5_5 = compute_c5(h5, len(best_h))
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()
    
    # Phase 6: Annealing
    if time.time() < deadline - 50:
        d6 = min(deadline - 30, time.time() + 50)
        h6, _ = simulated_anneal(best_h, len(best_h), d6, 4000, 0.3, 0.003)
        c5_6 = compute_c5(h6, len(best_h))
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()
    
    # Phase 7: Final polish
    if time.time() < deadline - 10:
        d7 = min(deadline - 2, time.time() + 25)
        h7, _ = binary_search(best_h, len(best_h), d7, 300, 200, 25, 3)
        c5_7 = compute_c5(h7, len(best_h))
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()
    
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
