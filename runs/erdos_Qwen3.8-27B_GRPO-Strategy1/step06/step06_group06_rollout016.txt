def corr_grad_topk(h, corr, topk, weights=None):
    """Compute gradient using weighted top-k correlations."""
    N = len(h)
    center = N - 1
    if weights is None:
        idx = np.argpartition(corr, -topk)[-topk:]
        weights = np.ones(topk) / topk
    else:
        idx = np.argsort(corr)[-topk:]
    grad = np.zeros(N)

            grad += w * (1.0 - 2.0*h)

            grad[:N-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:N-s]

            s2 = -s
            grad[s2:] += w * (1.0 - h[:N-s2])

def compute_corr(h):
    N = len(h)

        L *= 2

    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5

            lo = mid
        else:
            hi = mid

def compute_grad(h, corr, topk, softmax_T=0.3):
    N = len(h)
    center = N - 1
    # Softmax weighting
    w_all = np.exp((corr - corr.max()) / max(softmax_T, 1e-6))
    w_all /= w_all.sum()

            d_sub2 = min(sub_budget, time.time() + 25)
            hb_an, pk_an = anneal_binary(hb_n, Nt, d_sub2, T0=0.05, cool=0.9995, max_no_imp=400)
            c5_an = pk_an * (2.0 / Nt)
            if c5_an < best_c5:
                best_c5, best_h = c5_an, hb_an

    # Phase 5: Final deep polish on best (remaining time)
    Nf = len(best_h)
    if time.time() < deadline - 10:
        d5a = min(deadline - 10, time.time() + 100)
        hf = best_h.astype(float) + np.random.randn(Nf) * 0.015
        hf = proj_sum(hf, Nf / 2.0)
        hfc, pkfc = continuous_opt(hf, Nf, d5a, topk=60, lr0=0.002, smT0=0.15)
        c5fc = pkfc * (2.0 / Nf)
        if c5fc < best_c5:
            best_c5, best_h = c5fc, hfc
    if time.time() < deadline - 8:
        hf_bin = binarize(best_h, Nf)
        hfb, pkfb = anneal_binary(hf_bin, Nf, deadline - 3, T0=0.03, cool=0.9997, max_no_imp=1000)
        c5fb = pkfb * (2.0 / Nf)
        if c5fb < best_c5:
            best_c5, best_h = c5fb, hfb

    # Final verification
    final_c5 = float(np.max(compute_corr(best_h))) * (2.0 / len(best_h))
    best_c5 = min(best_c5, final_c5)
    return best_h, best_c5, len(best_h)
