    j = np.arange(N)[:, None]

In simulated annealing:
- Generate a candidate move
- If it improves: accept
- If it worsens by Δ > 0: accept with probability exp(-Δ/T)
- Otherwise: reject (stay at current state)

import numpy as np
import time
import math

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def D_matrix(h, N):
    j = np.arange(N)[:, None]
    k = np.arange(-(N - 1), N)[None, :]
    jp = j + k
    jm = j - k
    b = np.where((jp >= 0) & (jp < N), 1.0 - h[np.clip(jp, 0, N - 1)], 0.0)
    c = np.where((jm >= 0) & (jm < N), h[np.clip(jm, 0, N - 1)], 0.0)
    return b - c

def binary_anneal(h, N, deadline, max_sweeps=3000, sample_n=100, T0=8.0, Tmin=0.5):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    T = T0
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 700:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = D_matrix(h, N)
        if len(ones) > sample_n:
            perm = np.random.permutation(len(ones))[:sample_n]
            p_list = ones[perm]
        else:
            p_list = ones
        best_delta = 0.0
        best_pq = None
        for p in p_list:
            base = corr - D[p, :]
            cand = base[None, :] + D[zeros]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            delta = float(npk[mi]) - P
            if delta < best_delta:
                best_delta = delta
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None:
            if best_delta <= 0 or (T > Tmin and np.random.random() < math.exp(max(-50.0, -best_delta / T))):
                h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
            elif no_improve > 150:
                rp = int(ones[np.random.randint(len(ones))])
                rq = int(zeros[np.random.randint(len(zeros))])
                h[rp], h[rq] = 0.0, 1.0
        else:
            if no_improve > 50:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
        T = max(Tmin, T * 0.9994)
    return best_h, best_c

def lse_optimize(h, N, deadline, T0=2.5, lr=0.05):
    h = h.astype(float).copy()
    target = N / 2.0
    T = T0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
        x = corr / T
        x -= x.max()
        w = np.exp(x)
        w /= w.sum()
        D = D_matrix(h, N)
        grad = D @ w
        h = project_sum(h - lr * grad, target)
        it += 1
        if it % 150 == 0:
            lr *= 0.97
        T = max(0.4, T * 0.9993)
    return best_h, best_pk * (2.0 / N)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rm = idx[np.random.permutation(len(idx))[:n1 - nh]]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        add = idx[np.random.permutation(len(idx))[:nh - n1]]
        h[add] = 1.0
    return h

def perturb_binary(h, N, frac):
    hp = h.copy()
    n_flip = max(2, int(N * frac))
    ones = np.where(hp == 1)[0]
    zeros = np.where(hp == 0)[0]
    n1 = min(n_flip // 2, len(ones))
    if n1 > 0 and len(ones) >= n1:
        pick_ones = ones[np.random.permutation(len(ones))[:n1]]
        hp[pick_ones] = 0.0
    n0 = min(n_flip - n1, len(zeros))
    if n0 > 0 and len(zeros) >= n0:
        pick_zeros = zeros[np.random.permutation(len(zeros))[:n0]]
        hp[pick_zeros] = 1.0
    return fix_sum_binary(hp, N)

    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)

    h_bin = (h0 > 0.5).astype(float)
    h1, _ = binary_anneal(h_bin, N0, d1, max_sweeps=4000, sample_n=80, T0=10.0, Tmin=0.5)
    c5_1 = compute_c5(h1, N0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: LSE continuous optimization (120s)
    if time.time() < deadline - 350:
        d2 = min(deadline - 330, time.time() + 120)
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = lse_optimize(hc, N0, d2, T0=3.0, lr=0.04)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary annealing from best (200s)
    if time.time() < deadline - 200:
        d3 = min(deadline - 180, time.time() + 200)
        h_bin3 = (best_h > 0.5).astype(float)
        h3, _ = binary_anneal(h_bin3, N0, d3, max_sweeps=3000, sample_n=100, T0=6.0, Tmin=0.3)
        c5_3 = compute_c5(h3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Binary from perturbed start (100s)
    if time.time() < deadline - 120:
        d4 = min(deadline - 100, time.time() + 100)
        h_pert = perturb_binary((best_h > 0.5).astype(float), N0, 0.08)
        h4, _ = binary_anneal(h_pert, N0, d4, max_sweeps=1500, sample_n=90, T0=8.0, Tmin=0.5)
        c5_4 = compute_c5(h4, N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Upscale to 2x resolution (180s)
    N_up = N0 * 2
    if time.time() < deadline - 80:
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_up, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        h_bt = fix_sum_binary(h_bt, N_up)
        d5 = min(deadline - 40, time.time() + 170)
        h5, _ = binary_anneal(h_bt, N_up, d5, max_sweeps=1500, sample_n=80, T0=8.0, Tmin=0.5)
        c5_5 = compute_c5(h5, N_up)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final LSE polish (40s)
    Nf = len(best_h)
    if time.time() < deadline - 5:
        hF = best_h + np.random.randn(Nf) * 0.015
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = lse_optimize(hF, Nf, deadline - 2, T0=1.2, lr=0.02)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 7: Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = (best_h > 0.5).astype(float)
        hF2 = fix_sum_binary(hF2, len(hF2))
        hF2, _ = binary_anneal(hF2, len(hF2), deadline - 0.5, max_sweeps=80, sample_n=80, T0=2.0, Tmin=0.3)
        c5F2 = compute_c5(hF2, len(hF2))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
