import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def to_binary(h, N):
    nb = (h.astype(float) > 0.5).astype(float)
    nh = N // 2
    n1 = int(nb.sum())

        idx = np.where(nb == 1)[0]
        sel = np.random.choice(idx, n1 - nh, replace=False)
        nb[sel] = 0

        sel = np.random.choice(idx, nh - n1, replace=False)
        nb[sel] = 1
    return nb

def subgrad_vec(y, corr, top_k, N):
    grad = np.zeros(N)
    center = N - 1
    kk = min(top_k, len(corr))
    top = np.argpartition(corr, -kk)[-kk:]
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += (1.0 - y[:N - L])
            grad[:N - L] -= y[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
    grad /= len(top)
    return grad

                x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 6: Short anneal on best
    if time.time() < deadline - 50:
        d6 = min(deadline - 30, time.time() + 45)
        h6, _ = anneal_binary(best_h, len(best_h), d6, T0=0.3, Tend=0.005)
        c6 = compute_c5(h6, len(best_h))
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    # Phase 7: Final Nesterov polish
    if time.time() < deadline - 12:
        hF = best_h + np.random.randn(len(best_h)) * 0.03
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, cF = nesterov_opt(hF, len(best_h), deadline - 6, k=110, lr0=0.007, beta0=0.95)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    # Phase 8: Final greedy
    if time.time() < deadline - 2:
        hF2 = to_binary(best_h, len(best_h))
        hF2, _ = greedy_binary(hF2, len(best_h), deadline - 0.5, sample_n=200, top_m=25, max_kick=2)
        cF2 = compute_c5(hF2, len(best_h))
        if cF2 < best_c5:
            best_c5, best_h = cF2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
