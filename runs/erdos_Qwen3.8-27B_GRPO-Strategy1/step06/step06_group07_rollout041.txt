    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Gradient of max(corr) w.r.t. h
        # Use top-k lags
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += (1.0 - h[:N - L])
                grad[:N - L] -= h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - h[M:])
                    grad[M:] -= h[:N - M]
        grad /= k_top
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = lr0 / np.sqrt(t + 1)
        h_new = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), target)
        h = h_new
    
    return best_h, best_pk * dx
