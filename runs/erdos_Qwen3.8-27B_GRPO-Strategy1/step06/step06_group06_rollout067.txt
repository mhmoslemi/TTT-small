import numpy as np
import time

def full_corr(h):
    """Compute full correlation of h with (1-h)."""
    return np.correlate(h, 1.0-h, mode='full')

def proj_sum(h, target):
    """Project h onto {0≤h≤1, sum(h)=target}."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Bisection for the threshold
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def lse_objective(corr, lam):
    """Smooth approximation to max(corr)."""
    cmax = corr.max()
    shifted = corr - cmax
    return cmax + np.log(np.sum(np.exp(lam*shifted)))/lam

def lse_gradient(h, corr, lam):
    """Gradient of LSE(corr,h)/lam w.r.t. h."""
    cmax = corr.max()
    w = np.exp(lam*(corr - cmax))
    w /= w.sum()
    N = len(h)
    grad = np.zeros(N)
    # grad[j] = sum_k w[k] * (1-h[j-k])*[0<=j-k<N] - sum_k w[k]*h[j+k]*[0<=j+k<N]
    # First term: cross-correlation of w with (1-h)
    # Second term: cross-correlation of w with h (reversed)
    # For efficiency with moderate N, do it directly:
    for j in range(N):
        # First term: sum over k where 0 <= j-k < N
        # k ranges from max(0, j-N+1) to min(j, 2*N-2)
        k_lo = max(0, j - N + 1)
        k_hi = min(j, 2*N-2)
        ks = np.arange(k_lo, k_hi+1)
        idx = j - ks  # = j-k, ranges 0..N-1
        grad[j] += w[ks] @ (1.0 - h[idx])
        # Second term: sum over k where 0 <= j+k < N
        # k ranges from max(0, -(j)) to min(N-1-j, 2*N-2)
        k_lo2 = max(0, -j)  # j+k >= 0 always since j>=0, k>=0
        k_hi2 = min(N-1-j, 2*N-2)
        ks2 = np.arange(k_lo2, k_hi2+1)
        idx2 = j + ks2  # = j+k, ranges 0..N-1
        grad[j] -= w[ks2] @ h[idx2]
    return grad

```python
import numpy as np
import time

def full_corr(h):
    return np.correlate(h, 1.0-h, mode='full')

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def lse_grad(h, corr, lam):
    cmax = corr.max()
    w = np.exp(lam*(corr - cmax))
    w /= w.sum()
    N = len(h)
    term1 = np.convolve(1.0-h, w, mode='full')[:N]
    term2 = np.correlate(h, w, mode='full')[:N]
    return term1 - term2

def continuous_opt(h0, N, deadline, lam0=20.0, lam_max=500.0, lr0=0.01):
    h = h0.astype(float).copy()
    h = proj_sum(h, N/2.0)
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0; lr = lr0
    dx = 2.0/N
    best_c5 = float(full_corr(h).max()) * dx
    best_h = h.copy()
    
    while time.time() < deadline - 2:
        corr = full_corr(h)
        lam = min(lam_max, lam0 * (1.0 + t*0.01))
        pk = float(corr.max()) * dx
        if pk < best_c5:
            best_c5 = pk; best_h = h.copy()
        
        grad = lse_grad(h, corr, lam)
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m/(1-b1**t)
        vh = v/(1-b2**t)
        lr_now = lr / (1.0 + 0.001*t)
        h = proj_sum(h - lr_now*mh/np.sqrt(vh+eps), N/2.0)
        
        if t % 1000 == 0:
            lr *= 0.95
    
    return best_h, best_c5
```

import time

def full_corr(h):
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_gradient(h, corr, lam):
    N = len(h)
    cmax = float(corr.max())
    w = np.exp(lam * (corr - cmax))
    w /= float(w.sum())
    term1 = np.convolve(1.0 - h, w, mode='full')[:N]
    term2 = np.correlate(h, w, mode='full')[:N]
    return (term1 - term2).astype(float)

def lse_max(corr, lam):
    cmax = float(corr.max())
    shifted = corr - cmax
    return cmax + float(np.log(np.sum(np.exp(lam * shifted)))) / lam

    h = h0.astype(float).copy()

    while time.time() < deadline - 1:

            best_c5 = float(corr.max()) * dx

        grad = lse_gradient(h, corr, lam)

        v = b2 * v + (1 - b2) * grad * grad
