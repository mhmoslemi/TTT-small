```python
def compute_D(h, N):

    vp = (pp >= 0) & (pp < N)

    pp_c = np.clip(pp, 0, N - 1)

    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
```

```python
def continuous_optimize(h, N, deadline, top_k=10, lr0=0.01):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    m = np.zeros(N)  # Adam momentum
    v = np.zeros(N)  # Adam variance
    t = 0
    best_h = h.copy()
    best_pk = float(max(fft_corr(h)))
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Top-k subgradient
        topk = np.argpartition(c, -top_k)[-top_k:]
        # Weight by how close to max
        weights = np.exp((c[topk] - pk) * 10)
        weights /= weights.sum()
        
        grad = np.zeros(N)
        D = compute_D(h, N)
        for j in range(top_k):
            grad += weights[j] * D[:, topk[j]]
        
        # Center gradient (to maintain sum constraint approximately)
        grad -= grad.mean()
        
        # Adam update
        m = b1 * m + (1-b1) * grad
        v = b2 * v + (1-b2) * grad**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)
        
        if t % 100 == 0:
            lr *= 0.95
            # Periodic perturbation to escape local minima
            if t % 500 == 0:
                h += np.random.randn(N) * 0.01
                h = project_sum(h, target)
    
    return best_h, best_pk * dx
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0/len(h))

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s-target) < 1e-10: return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:,None]
    sv = shifts[None,:]
    pm, pp = pos-sv, pos+sv
    vm = (pm>=0)&(pm<N); vp = (pp>=0)&(pp<N)
    hm = np.where(vm, h[np.clip(pm,0,N-1)], 0.0)
    hp = np.where(vp, h[np.clip(pp,0,N-1)], 0.0)
    return -(1.0-hm)*vm.astype(float) + hp*vp.astype(float)

def binarize(h, N):
    idx = np.argsort(-h)
    r = np.zeros(N)
    r[idx[:N//2]] = 1.0
    return r

def adam_continuous(h, N, deadline, top_k=10, lr0=0.02):

    t, best_pk, best_h = 0, float('inf'), h.copy()
    lr = lr0

    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk, best_h = pk, h.copy()
        # Weighted top-k subgradient

            h = project_sum(h, target)

            ci, cj = divmod(fi, len(zeros))
            d = float(mx[ci,cj]) - P
            if d < best_d: best_d, best_pq = d, (int(oc[ci]), int(zeros[cj]))
        if best_pq:
            if best_d < -1e-14:
                h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
            elif anneal > 0 and np.random.random() < np.exp(best_d/anneal):
                h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
                stall = 0
            else: break
    return h, best_c*(2.0/N)
```

Actually, I think the key insight is: for the continuous problem, we want to make ALL correlations as small as possible, not just the max. The max will naturally be the bottleneck. A good approach is:

1. Compute all correlations c[k]
2. Compute a "smoothed max" like log-sum-exp: LSE = (1/β) * log(Σ exp(β*c[k]))
3. Optimize LSE (which is smooth and approximates max as β→∞)
4. Gradually increase β

This gives a smooth objective whose gradient is a weighted average of all subgradients, with weights proportional to exp(β*c[k]).

This is actually what the top-k weighting approximates. Let me use the log-sum-exp approach with adaptive β:

```python
def lse_optimize(h, N, deadline):
    h = h.astype(float).copy()
    target = N/2.0
    m, v = np.zeros(N), np.zeros(N)
    t = 0
    best_h, best_pk = h.copy(), float('inf')
    lr = 0.02
    beta = 10.0  # start moderate, increase over time
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk, best_h = pk, h.copy()
        # Log-sum-exp gradient
        c_shifted = c - c.max()
        w = np.exp(beta * c_shifted)
        w /= w.sum()
        D = compute_D(h, N)
        grad = D @ w  # (N,) - weighted average of subgradients
        grad -= grad.mean()
        m = 0.9*m + 0.1*grad
        v = 0.999*v + 0.001*grad**2
        h -= lr*m/(np.sqrt(v)+1e-8)
        h = project_sum(h, target)
        if t%100 == 0:
            beta *= 1.1  # increase sharpness
            lr *= 0.97
        if t%400 == 0:
            h += np.random.randn(N)*0.01
            h = project_sum(h, target)
    return best_h, best_pk*(2.0/N)
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm, pp = pos - sv, pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    hm = np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0)
    hp = np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binarize(h, N):
    idx = np.argsort(-h)
    r = np.zeros(N)
    r[idx[:N // 2]] = 1.0
    return r

def lse_adam(h, N, deadline, lr0=0.015, beta0=15.0):
    h = h.astype(float).copy()
    target = N / 2.0
    m, v = np.zeros(N), np.zeros(N)
    t = 0
    best_pk = float('inf')
    best_h = h.copy()
    lr = lr0
    beta = beta0
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        c_sh = c - c.max()
        w = np.exp(beta * c_sh)
        w /= w.sum()
        D = compute_D(h, N)
        grad = D @ w
        grad -= grad.mean()
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        h -= lr * m / (np.sqrt(v) + 1e-8)
        h = project_sum(h, target)
        if t % 80 == 0:
            beta = min(beta * 1.2, 400.0)
            lr *= 0.96
        if t % 300 == 0:
            h += np.random.randn(N) * 0.015
            h = project_sum(h, target)
    return best_h, best_pk * (2.0 / N)

def bin_climb(h, N, deadline, anneal=0.0):
    h = binarize(h, N).copy()
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    while time.time() < deadline and stall < 100:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c, stall = P, 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        best_d, best_pq = 0.0, None
        chunk = 40
        for i in range(0, len(ones), chunk):
            oc = ones[i:i + chunk]
            vals = c[None, None, :] - D[oc, :, None] - D[zeros][None, :, :]
            mx = vals.max(axis=2)
            fi = int(np.argmin(mx))
            ci, cj = divmod(fi, len(zeros))
            d = float(mx[ci, cj]) - P
            if d < best_d:
                best_d, best_pq = d, (int(oc[ci]), int(zeros[cj]))
        if best_pq is not None:
            if best_d < -1e-14:
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
            elif anneal > 0 and np.random.random() < np.exp(max(best_d, -50.0) / anneal):
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
                stall = 0
            else:
                break
    return h, best_c * (2.0 / N)

def two_swap_climb(h, N, deadline):
    """Try 2-1 and 1-2 swaps for more flexibility."""
    h = h.astype(float).copy()
    best_c = compute_c5(h)
    stall = 0
    while time.time() < deadline and stall < 80:
        c = fft_corr(h)
        P = float(np.max(c))
        if P < best_c - 1e-14:
            best_c, stall = P, 0
        else:
            stall += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Random 2-swap: remove 2 ones, add 2 zeros
        p1, p2 = np.random.choice(ones, 2, replace=False)
        q1, q2 = np.random.choice(zeros, 2, replace=False)
        h[p1], h[p2], h[q1], h[q2] = 0, 0, 1, 1
        c_new = fft_corr(h)
        P_new = float(np.max(c_new))
        if P_new < best_c:
            best_c = P_new
        else:
            # Reject
            h[p1], h[p2], h[q1], h[q2] = 1, 1, 0, 0
    return h, best_c * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h, best_c5 = h0.copy(), compute_c5(h0)

    # Phase 1: LSE Adam continuous optimization (main phase)
    d1 = deadline - 350
    h1, c5_1 = lse_adam(h0, N0, d1, lr0=0.015, beta0=15.0)
    if c5_1 < best_c5:
        best_h, best_c5 = h1.copy(), c5_1

    # Phase 2: Binary hill-climb from continuous result
    if time.time() < deadline - 200:
        d2 = min(deadline - 180, time.time() + 130)
        h2, c5_2 = bin_climb(best_h, N0, d2, anneal=0.0)
        if c5_2 < best_c5:
            best_h, best_c5 = h2.copy(), c5_2

    # Phase 3: LSE Adam from perturbed binary
    if time.time() < deadline - 120:
        d3 = min(deadline - 100, time.time() + 80)
        h3_init = best_h + np.random.randn(N0) * 0.08
        h3_init = project_sum(h3_init, N0 / 2.0)
        h3, c5_3 = lse_adam(h3_init, N0, d3, lr0=0.012, beta0=25.0)
        if c5_3 < best_c5:
            best_h, best_c5 = h3.copy(), c5_3

    # Phase 4: Bin-climb from phase 3 result
    if time.time() < deadline - 80:
        d4 = min(deadline - 60, time.time() + 50)
        h4, c5_4 = bin_climb(best_h, N0, d4, anneal=0.0)
        if c5_4 < best_c5:
            best_h, best_c5 = h4.copy(), c5_4

    # Phase 5: Multi-resolution at N=384, 400
    for N_try in [384, 400]:
        if time.time() > deadline - 50:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, N_try / 2.0)
        d5 = min(deadline - 25, time.time() + 30)
        h5, c5_5 = lse_adam(h_interp, N_try, d5, lr0=0.01, beta0=20.0)
        if time.time() < deadline - 10:
            h5b, c5_5b = bin_climb(h5, N_try, deadline - 5, anneal=0.0)
            if c5_5b < c5_5:
                h5, c5_5 = h5b, c5_5b
        if c5_5 < best_c5:
            best_h, best_c5 = h5.copy(), c5_5

    # Phase 6: Annealed binary climb (escape local minima)
    if time.time() < deadline - 15:
        Nf = len(best_h)
        h6, c5_6 = bin_climb(best_h, Nf, deadline - 5, anneal=0.15)
        if c5_6 < best_c5:
            best_h, best_c5 = h6.copy(), c5_6

    # Phase 7: Final short LSE polish
    if time.time() < deadline - 3:
        h7, c5_7 = lse_adam(best_h, len(best_h), deadline - 1, lr0=0.005, beta0=30.0)
        if c5_7 < best_c5:
            best_h, best_c5 = h7.copy(), c5_7

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
```
