import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0-h), L)))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(np.float64).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        rem = np.random.choice(np.where(h > 0.5)[0], n1-nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        add = np.random.choice(np.where(h < 0.5)[0], nh-n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def make_deltas(h, N):
    s = 2.0*h - 1.0
    d = np.zeros((N, 2*N-1))
    for i in range(N):
        d[i, i:i+N] = s
    return d

def upd_delta(D, s, p, N):
    D[p, :] = 0.0
    D[p, p:p+N] = s

def bsearch(h_in, N, deadline, ns=40, T0=0.003, max_it=10000):
    h = fix_binary(h_in, N//2)
    corr = fft_corr(h).copy()
    s = 2.0*h - 1.0
    D = make_deltas(h, N)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    no_imp = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        T = T0 * (0.997 ** it)
        np_ = min(ns, len(ones))
        ps = np.random.choice(ones, np_, replace=False)
        dp = D[ps]
        dq = D[zeros]
        bv, bi, bj = 1e10, 0, 0
        ch = 10
        for ci in range(0, np_, ch):
            ce = min(ci+ch, np_)
            v = corr[None,None,:] + dp[ci:ce,None,:] - dq[None,:,:]
            m = v.max(axis=2)
            idx = np.unravel_index(int(m.argmin()), m.shape)
            if float(m[idx]) < bv:
                bv = float(m[idx])
                bi, bj = idx[0]+ci, idx[1]
        p = int(ps[bi]); q = int(zeros[bj])
        if bv > best_c + 1e-12:
            dC = (bv - best_c) * 2.0/N
            if np.random.random() > np.exp(-dC/max(T,1e-12)):
                no_imp += 1
                if no_imp > 50:
                    for _ in range(3):
                        kk = int(np.random.choice(ones)); qq = int(np.random.choice(zeros))
                        h[kk],h[qq] = 0.0,1.0
                        corr += D[kk].copy() - D[qq].copy()
                        s[kk],s[qq] = -s[kk],-s[qq]
                        upd_delta(D,s,kk,N); upd_delta(D,s,qq,N)
                    ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
                    no_imp = 0
                    continue
                continue
        h[p],h[q] = 0.0,1.0
        corr += D[p].copy() - D[q].copy()
        s[p],s[q] = -s[p],-s[q]
        upd_delta(D,s,p,N); upd_delta(D,s,q,N)
        ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
        c = float(corr.max())
        if c < best_c - 1e-12:
            best_c = c; best_h = h.copy(); no_imp = 0
        else:
            no_imp += 1
        if no_imp > 120:
            no_imp = 0
            for _ in range(5):
                kk = int(np.random.choice(ones)); qq = int(np.random.choice(zeros))

                s[kk],s[qq] = -s[kk],-s[qq]

def grad_desc(h_in, N, deadline, k=35, lr=0.01):
    h = h_in.astype(np.float64).copy()
    tgt = N/2.0
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    m = np.zeros(N)
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        top = np.sort(top)
        g = np.zeros(N)
        for t in top:
            L = int(t - (N-1))
            if L == 0:
                g += 1.0 - 2.0*y
            elif L > 0:
                g[L:] += (1.0-y[:N-L])
                g[:N-L] -= y[L:]
            else:
                M = -L
                g[:N-M] += (1.0-y[M:])
                g[M:] -= y[:N-M]
        g /= len(top)
        h2 = project_sum(y - lr*g, tgt)
        m = beta*(h2-h)
        h = h2
        it += 1
        if it % 300 == 0:
            lr *= 0.95
            beta = min(0.95, 0.82 + it*0.0001)
    return best_h, best_pk*2.0/N
