import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:L]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh, rng):
    h = h.astype(float).copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n1 = int(h.sum())
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        h[rng.choice(ones, n1 - nh, replace=False)] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        h[rng.choice(zeros, nh - n1, replace=False)] = 1.0
    return h

def nesterov_cont(h_in, N, deadline, rng, k=30, lr0=0.008):
    h = h_in.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    corr = fft_corr(h)
    best_pk = float(np.max(corr))
    best_h = h.copy()
    m = np.zeros(N)

    beta = 0.8
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top_k = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)

            if d == 0:
                grad += 1.0 - 2.0 * y

                grad[:N - d] += (1.0 - y[d:])
                grad[d:] -= y[:N - d]
            else:

                grad[:N - M] -= y[M:]
        grad /= len(top_k)

            lr *= 0.965
