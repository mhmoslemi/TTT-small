```python
    # Phase 1: Upsample initial point to double resolution for higher clarity
    N0 = 380  # Double the initial grid (not half)
    h_orig = initial_h_values.astype(float)
    h_up = upsample(h_orig, N0)
    # Projection to stay in bounds and ensure sum
    h0 = ensure_sum(h_up, N0/2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0) * (2.0/N0)  # Account for dx during eval

    # Primary loop over increased resolutions
    N_vals = [N0*2, N0*4]  # Progressively denser: 760, 1520 points
    N_grid = N_vals[0]   # Current grid size
    
    # Phase 1: Binary search on original candidate at N0=380
    h1 = binary_search_mod(best_h, N0, deadline, sample_frac=0.7, max_sweeps=300)
    if compute_c5(h1, N0) < best_c5:
        best_c5, best_h = compute_c5(h1, N0), h1.copy()
    
    # Phase 2: Move to 760-point grid from best N0 candidate
    h_re sampled = upsample(best_h, N_vals[0])
    h_re sampled = ensure_sum(h_re sampled, N_vals[0]/2.0)
    budget_remaining = deadline - time.time()
    h2, new_c5 = binary_search_mod(h_re sampled, N_vals[0], deadline - time.time(), 
                                   max_sweeps=min(budget_remaining//2, 400))
    if new_c5 < best_c5:
        best_c5, best_h = new_c5, h2.copy()

    # Phase 3: Continuous optimization at current best
    # Use Nesterov-style with heuristic damping
    budget_remaining = deadline - time.time()
    h3, new_c5 = nesterov_mod(best_h, N_cur=len(best_h), budget_remaining, k=90)
    if new_c5 < best_c5:
        best_c5, best_h = new_c5, h3.copy()
    
    # Final polish if time allows
    if time.time() < deadline:
        h_final = best_h.copy()
        h_final = final_project_optimality(h_final, len(h_final))
        best_c5 = compute_c5(h_final, len(h_final))
    
    # Ensure bounds and sum
    h_final = np.clip(h_final, 0.0, 1.0)
    h_final = ensure_sum(h_final, len(h_final)/2.0)
    final_c5 = compute_c5(h_final, len(h_final))
    return h_final, final_c5, len(h_final)
```
