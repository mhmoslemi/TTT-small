def grad_topk(h, N, k, temp):

    corr = fft_corr(h)
    center = N - 1
    # Softmax weights over all lags, but truncate to top-k for efficiency

    z = corr[idx] - corr[idx].max()
    w = np.exp(z / temp)
    w /= w.sum()
    
    grad = np.zeros(N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_full_grad(corr, h, N, temp):
    center = N - 1
    z = (corr - corr.max()) / max(temp, 1e-10)
    w = np.exp(z)
    w /= w.sum()
    grad = np.zeros(N)
    for L in range(-(N-1), N):
        wi = w[center + L]
        if wi < 1e-8:
            continue
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            seg = N - M
            grad[:seg] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:seg]
    return grad

def adam_opt(h0, N, deadline, temp_start=0.02, temp_end=0.003, lr0=0.015):

    target = N / 2.0
    best_h = h.copy()
    best_corr = fft_corr(h).copy()
    best_pk = float(np.max(best_corr))
    
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = lr0
    stall = 0
    
    total_iter_target = 5000
    current_iter_target = max(500, total_iter_target // 4)
    
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
            best_corr = corr.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive temperature
        progress = min(1.0, t / total_iter_target)
        temp = temp_start * (temp_end / temp_start) ** progress
        
        grad = compute_full_grad(corr, h, N, temp)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        
        # Adaptive LR
        lr_eff = lr / (1 + t * 0.0001)

            v[:] = 0

            lr = lr0
            stall = 0

def binary_refine(h0, N, deadline, n_tries=25):

    stall = 0

        found = False

            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c = fft_corr(h)

            if c_max < int(corr.max()):

            else:
                h[p] = 1
                h[q] = 0
        
        if found:

                best_h = h.copy()
        else:

                    p = ones[np.random.randint(len(ones))]
                    q = zeros[np.random.randint(len(zeros))]
                    h[p] = 0
                    h[q] = 1
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                corr = fft_corr(h)
                stall = 0
    
    return best_h, best_max * (2.0 / N)

    x_new = np.linspace(0, 2.0, N_new, endpoint=False)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_full_grad(corr, h, N, temp):
    center = N - 1
    z = (corr - corr.max()) / max(temp, 1e-10)
    w = np.exp(z)
    w /= w.sum()
    grad = np.zeros(N)
    threshold = 1e-9
    for L in range(-(N - 1), N):
        wi = w[center + L]
        if wi < threshold:
            continue
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            seg = N - M
            grad[:seg] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:seg]
    return grad

def adam_opt(h0, N, deadline, temp_start=0.02, temp_end=0.003, lr0=0.012):

    target = N / 2.0
    best_h = h.copy()

    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = lr0
    stall = 0
    best_stall_reset = t0_global
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-13:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        progress = min(1.0, t / 8000.0)
        temp = temp_start * (temp_end / temp_start) ** progress
        grad = compute_full_grad(corr, h, N, temp)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        lr_eff = lr / (1.0 + 0.0002 * t)

def binary_refine(h0, N, deadline, n_tries=25):
    h = h0.copy()
    h_bin = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h_bin.sum())
    if n1 > nh:
