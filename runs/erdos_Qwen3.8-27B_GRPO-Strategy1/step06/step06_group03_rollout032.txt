def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * np.conj(G)))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def sh(h, idx, N):
    m = (idx >= 0) & (idx < N)
    r = np.zeros(len(idx), dtype=float)
    r[m] = h[np.clip(idx, 0, N - 1)[m]]
    return r

def subgrad_vec(y, corr, top_k, N):
    grad = np.zeros(N)
    center = N - 1
    top = np.argpartition(corr, -top_k)[-top_k:] if top_k < len(corr) else np.arange(len(corr))
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += (1.0 - y[:N - L])
            grad[:N - L] -= y[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
    grad /= max(1, len(top))
    return grad

def nesterov_opt(h, N, deadline, k=50, lr0=0.012, beta0=0.90):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    mom = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + mom
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = subgrad_vec(y, corr, k, N)
        h_new = project_sum(y - lr * grad, target)
        mom = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr *= 0.988
            beta = min(0.96, beta0 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

                for _ in range(3):
                    p = int(ones[np.random.randint(len(ones))])

            if row_max[bq] < best_val:

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive binary search at native resolution
    d1 = min(deadline - 400, t0 + 320)
    h_s, _ = binary_opt(h0, N0, d1, top_m=30, sample_n=250, two_swap_prob=0.2, anneal_prob=0.05)
    c5_s = compute_c5(h_s)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Continuous Nesterov polish
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 120)
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k=40, lr0=0.010, beta0=0.92)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary search again (more aggressive)
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 200)
        h_s3, _ = binary_opt(best_h, N0, d3, top_m=35, sample_n=280, two_swap_prob=0.25, anneal_prob=0.08)
        c5_3 = compute_c5(h_s3)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Try alternative resolution N=400
    N_alt = 400
    if time.time() < deadline - 150:
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N_alt, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        nh_alt = N_alt // 2
        n1 = int(h_bt.sum())
        if n1 > nh_alt:
            idx = np.where(h_bt == 1)[0]
            h_bt[idx[np.random.permutation(len(idx))[nh_alt:]]] = 0
        elif n1 < nh_alt:
            idx = np.where(h_bt == 0)[0]
            h_bt[idx[np.random.permutation(len(idx))[:nh_alt - n1]]] = 1
        d4 = min(deadline - 100, time.time() + 100)
        h_rt, _ = binary_opt(h_bt, N_alt, d4, top_m=30, sample_n=200, two_swap_prob=0.15, anneal_prob=0.05)
        c5_rt = compute_c5(h_rt)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 5: Final Nesterov on best
    if time.time() < deadline - 60:
        Nf = len(best_h)
        d5 = min(deadline - 50, time.time() + 80)
        hf = best_h + np.random.randn(Nf) * 0.02
        hf = project_sum(hf, Nf / 2.0)
        hf, c5f = nesterov_opt(hf, Nf, d5, k=50, lr0=0.008, beta0=0.93)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf

    # Phase 6: Final binary cleanup
    if time.time() < deadline - 5:
        Nf = len(best_h)
        hF = (best_h > 0.5).astype(float)
        hF, _ = binary_opt(hF, Nf, deadline - 1, top_m=40, sample_n=300, two_swap_prob=0.3, anneal_prob=0.1)
        c5F = compute_c5(hF)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
