- Δ_i = (h[p+L_i]+h[p-L_i]) - (h[q+L_i]+h[q-L_i])

```python
def pos_shift_sum(h, positions, shifts, N):
    """For each position p and shift L, compute h[p+L] + h[p-L] (0 if out of bounds)"""
    # positions: (P,), shifts: (M,)
    # result: (P, M)
    result = np.zeros((len(positions), len(shifts)))
    for j, L in enumerate(shifts):
        pp = positions + L
        valid_p = (pp >= 0) & (pp < N)
        result[valid_p, j] += h[pp[valid_p]]
        pm = positions - L
        valid_m = (pm >= 0) & (pm < N)
        result[valid_m, j] += h[pm[valid_m]]
    return result
```

Then for 1-1 swap: Δ = pos_shift_sum(h, [p], shifts, N) - pos_shift_sum(h, [q], shifts, N)
For 2-2 swap: Δ = pos_shift_sum(h, [p1,p2], shifts, N).sum(0) - pos_shift_sum(h, [q1,q2], shifts, N).sum(0)

This is efficient!

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h):
    N = len(h)
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        idx = idx[np.random.permutation(len(idx))][:n1 - nh]
        h[idx] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        idx = idx[np.random.permutation(len(idx))][:nh - n1]
        h[idx] = 1
    return h

def pos_contrib(h, positions, shifts, N):
    P = len(positions)
    M = len(shifts)
    result = np.zeros((P, M))
    for j in range(M):
        L = int(shifts[j])
        if L == 0:
            valid = (positions >= 0) & (positions < N)
            result[valid, j] = 2.0 * h[positions[valid]]
        else:
            pp = positions + L
            vp = (pp >= 0) & (pp < N)
            result[:, j] += np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
            pm = positions - L
            vm = (pm >= 0) & (pm < N)
            result[:, j] += np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0)
    return result

def binary_search(h, N, deadline, top_m=18, sample_n=110, two_swap_prob=0.4, patience=300):
    h = make_binary(h).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    center = N - 1
    no_improve = 0

    while time.time() < deadline:
        P = float(c.max())
        if P < best_c:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) < 3 or len(zeros) < 3:
                    break
                n_rand = min(3, len(ones) // 2, len(zeros) // 2)
                ps = np.random.choice(ones, n_rand, replace=False)
                qs = np.random.choice(zeros, n_rand, replace=False)
                h[ps] = 0.0
                h[qs] = 1.0
                c = fft_corr(h)
                no_improve = 0
                continue

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        top_idx = np.argsort(c)[-top_m:]
        top_shifts = (top_idx - center).astype(int)
        c_top = c[top_idx].copy()

        s_p = min(sample_n, len(ones))
        p_list = np.random.choice(ones, s_p, replace=False) if len(ones) > s_p else ones
        s_q = min(sample_n, len(zeros))
        q_list = np.random.choice(zeros, s_q, replace=False) if len(zeros) > s_q else zeros

        p_contrib = pos_contrib(h, p_list, top_shifts, N)
        q_contrib = pos_contrib(h, q_list, top_shifts, N)

        best_new_P = P
        best_move = None

        for i in range(len(p_list)):
            pi = p_contrib[i]
            for j in range(len(q_list)):
                nv = c_top + pi - q_contrib[j]
                nm = float(nv.max())
                if nm < best_new_P:
                    best_new_P = nm
                    best_move = ((int(p_list[i]), int(q_list[j])), 1)

        if two_swap_prob > 0 and np.random.random() < two_swap_prob and len(p_list) >= 2 and len(q_list) >= 2:
            sp = min(12, len(p_list))
            sq = min(12, len(q_list))
            for i in range(sp):
                for j in range(i + 1, sp):
                    pp_sum = p_contrib[i] + p_contrib[j]
                    for k in range(sq):
                        for l in range(k + 1, sq):
                            nv = c_top + pp_sum - q_contrib[k] - q_contrib[l]
                            nm = float(nv.max())
                            if nm < best_new_P:
                                best_new_P = nm
                                best_move = ((int(p_list[i]), int(p_list[j]), int(q_list[k]), int(q_list[l])), 2)

        if best_move is not None:
            mv, typ = best_move
            if typ == 1:
                h[mv[0]] = 0.0
                h[mv[1]] = 1.0
            else:
                h[mv[0]] = 0.0
                h[mv[1]] = 0.0
                h[mv[2]] = 1.0
                h[mv[3]] = 1.0
            c = fft_corr(h)

                    h[mv[1]] = 0.0
                else:
                    h[mv[0]] = 1.0
                    h[mv[1]] = 1.0
                    h[mv[2]] = 0.0
                    h[mv[3]] = 0.0
                c = fft_corr(h)

    return h, best_c

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h, N, deadline, k=60, lr0=0.012, beta0=0.92):
    h = project_sum(h.astype(float).copy(), N / 2.0)
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.995
            beta = min(0.98, beta0 + it * 0.0001)
    return best_h, best_pk * dx
