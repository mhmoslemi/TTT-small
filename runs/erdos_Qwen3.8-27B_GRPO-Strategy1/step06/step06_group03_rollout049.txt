def compute_c5(h, N):

def nesterov_soft(h, N, deadline, k=80, lr0=0.012, beta0=0.90, tau=0.05):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        # Softmax subgradient
        shifted = corr - corr.max()
        w = np.exp(shifted / tau)
        w /= w.sum()
        # Gradient: sum_k w[k] * d corr[k] / dh[i]
        grad = np.zeros(N)
        active = np.where(w > 1e-6)[0]
        for ai in active:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y) * w[ai]
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L]) * w[ai]
                grad[:N-L] -= y[L:] * w[ai]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:]) * w[ai]
                grad[M:] -= y[:N-M] * w[ai]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.97, beta0 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

        else:
            q_list = zeros.copy()
        d1, d2 = compute_deltas(h, p_list, q_list, top_L, N)
        nv, pi, qi = eval_swaps(top_c, d1, d2, batch)
        if pi >= 0 and nv < P:
            h[int(p_list[pi])] = 0.0
            h[int(q_list[qi])] = 1.0
    return best_h, best_c

def anneal_search(h, N, deadline, rng, top_m=25, sample_n=200, T0=0.04, T1=0.003, batch=50):
    h = binarize_fix(h, N, rng)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    T = T0
    it = 0
    while time.time() < deadline:
        it += 1
        T = max(T1, T0 * (T1 / T0) ** (it / 8000.0))
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        top_idx = np.argsort(c)[-top_m:]
        top_c = c[top_idx].copy()
        top_L = (top_idx - (N - 1)).astype(int)
        if len(ones) > sample_n:
            p_list = ones[rng.permutation(len(ones))[:sample_n]]
        else:
            p_list = ones.copy()
        if len(zeros) > sample_n:
            q_list = zeros[rng.permutation(len(zeros))[:sample_n]]
        else:
            q_list = zeros.copy()
        d1, d2 = compute_deltas(h, p_list, q_list, top_L, N)
        # Collect all new values for annealing
        all_nm = []
        for pb in range(0, len(p_list), batch):
            pe = min(pb + batch, len(p_list))
            nc = top_c[None, None, :] + d1[pb:pe, None, :] + d2[None, :, :]
            all_nm.append(nc.max(axis=2))
        all_nm = np.concatenate(all_nm, axis=0)
        fi = np.argmin(all_nm)
        pi, qi = int(fi // len(q_list)), int(fi % len(q_list))
        if all_nm[pi, qi] < P:
            h[int(p_list[pi])] = 0.0
            h[int(q_list[qi])] = 1.0
        else:
            ri = rng.randint(len(p_list))
            rj = rng.randint(len(q_list))
            delta = all_nm[ri, rj] - P
            if delta <= 0 or rng.random() < np.exp(-delta / T):
                h[int(p_list[ri])] = 0.0
                h[int(q_list[rj])] = 1.0
    return best_h, best_c

def nesterov_soft(h, N, deadline, lr0=0.012, beta0=0.90, tau=0.05):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        shifted = corr - corr.max()
        w = np.exp(shifted / tau)
        w /= w.sum()
        grad = np.zeros(N)
        active = np.where(w > 1e-5)[0]
        for ai in active:
            L = int(ai - center)
            wi = w[ai]
            if L == 0:
                grad += (1.0 - 2.0 * y) * wi
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L]) * wi
                grad[:N-L] -= y[L:] * wi
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:]) * wi
                grad[M:] -= y[:N-M] * wi
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.996
            beta = min(0.97, beta0 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = binarize_fix(h0, N0, rng)
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Greedy local search
    d1 = min(deadline - 350, t0 + 220)
    h1, c1 = greedy_search(h0, N0, d1, rng, top_m=35, sample_n=350, batch=40, max_stall=500)
    c5_1 = compute_c5(h1, N0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Simulated annealing
    if time.time() < deadline - 280:
        d2 = min(deadline - 220, time.time() + 180)
        h2, c2 = anneal_search(best_h, N0, d2, rng, top_m=25, sample_n=200, T0=0.05, T1=0.004)
        c5_2 = compute_c5(h2, N0)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2

    # Phase 3: Nesterov continuous polish
    if time.time() < deadline - 150:
        d3 = min(deadline - 100, time.time() + 100)
        hc = best_h + rng.randn(N0) * 0.05
        hc = project_sum(hc, N0 / 2.0)
        h3, c5_3 = nesterov_soft(hc, N0, d3, lr0=0.010, tau=0.04)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3

    # Phase 4: Greedy from best (larger search)
    if time.time() < deadline - 100:
        d4 = min(deadline - 60, time.time() + 80)
        h4, c4 = greedy_search(best_h, N0, d4, rng, top_m=40, sample_n=400, batch=35, max_stall=300)
        c5_4 = compute_c5(h4, N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4

    # Phase 5: Multi-restart perturbation
    n_restarts = 0
    while time.time() < deadline - 40 and n_restarts < 8:
        n_restarts += 1
        noise_scale = 0.25 + 0.1 * n_restarts
        pert = best_h + rng.randn(N0) * noise_scale
        h5, c5_5 = greedy_search(pert, N0, min(deadline - 35 - n_restarts * 2, time.time() + 35), rng, top_m=30, sample_n=250, batch=50, max_stall=200)
        c5_5 = compute_c5(h5, N0)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5

    # Phase 6: Final greedy cleanup
    if time.time() < deadline - 3:
        hF = binarize_fix(best_h, N0, rng)
        hF, c5F = greedy_search(hF, N0, deadline - 1, rng, top_m=50, sample_n=500, batch=30, max_stall=100)
        c5F = compute_c5(hF, N0)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, N0)
    return best_h, final_c5, N0
