def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def anneal_binary(h, deadline, T0=0.1, cool=0.9995, max_no_imp=2000):
    N = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    no_imp = 0
    
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    T = T0
    
    while time.time() < deadline and T > 0.002:
        p = ones[np.random.randint(len(ones))]
        q = zeros[np.random.randint(len(zeros))]
        d = swap_delta(h, p, q)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - current_pk
        
        if delta < 0 or np.random.random() < np.exp(-delta/max(T, 1e-9)):
            h[p] = 0; h[q] = 1
            corr = nc
            current_pk = npk
            # Update ones/zeros
            idx_p = np.where(ones == p)[0]
            ones = np.delete(ones, idx_p[0])
            zeros = np.delete(zeros, np.where(zeros == q)[0][0])
            ones = np.append(ones, q)
            zeros = np.append(zeros, p)
            
            if npk < best_pk - 1e-12:
                best_pk = npk
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        
        T *= cool
        if no_imp > max_no_imp:
            # Restart
            h = best_h.copy()
            corr = compute_corr(h)
            current_pk = best_pk
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            no_imp = 0
            T = T0 * 0.5
    
    return best_h, best_pk

def continuous_opt(h, deadline, top_k=40, lr0=0.003):
    N = len(h)
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    corr = compute_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    center = N-1
    
    while time.time() < deadline:
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0*h)
            elif s > 0:
                grad[:N-s] += (1.0 - h[s:])
                grad[s:] -= h[:N-s]
            else:
                s2 = -s
                grad[s2:] += (1.0 - h[:N-s2])
                grad[:N-s2] -= h[s2:]
        
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m/(1-b1**t)
        vh = v/(1-b2**t)
        h = proj_sum(h - lr0*mh/np.sqrt(vh+eps), N/2.0)
        corr = compute_corr(h)
        if t % 1000 == 0:
            lr0 *= 0.95
    
    return best_h, best_pk

def local_search(h, deadline):
    N = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    
    while time.time() < deadline:
        improved = False
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        # Try all single swaps
        for p_idx in range(len(ones)):
            if time.time() > deadline:
                break
            p = ones[p_idx]
            for q_idx in range(len(zeros)):
                if time.time() > deadline:
                    break
                q = zeros[q_idx]
                d = swap_delta(h, p, q)
                npk = float((corr + d).max())
                if npk < best_pk - 1e-12:
                    h[p] = 0; h[q] = 1
                    corr = corr + d
                    best_pk = npk
                    best_h = h.copy()
                    improved = True
                    break  # restart q loop
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_pk

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    return np.fft.irfft(np.fft.rfft(h, L) * np.fft.rfft(np.flip(1.0-h), L), L)[:2*N-1]

def swap_delta(h, p, q):
    N = len(h)
    d = np.zeros(2*N-1)
    f1h = np.flip(1.0-h)
    n1 = min(p+N, 2*N-1) - p
    d[p:p+n1] -= f1h[:n1]
    n2 = min(q+N, 2*N-1) - q
    d[q:q+n2] += f1h[:n2]
    off1 = N-1-p
    n3 = min(off1+N, 2*N-1) - off1
    d[off1:off1+n3] += h[:n3]
    off2 = N-1-q
    n4 = min(off2+N, 2*N-1) - off2
    d[off2:off2+n4] -= h[:n4]
    return d

def proj_sum(h, target):

            h[p] = 0; h[q] = 1
            corr = nc
            current_pk = npk
            ip = np.searchsorted(ones, p)
            ones = np.delete(ones, ip)
            iq = np.searchsorted(zeros, q)
            zeros = np.delete(zeros, iq)
            ones = np.append(ones, q)
            zeros = np.append(zeros, p)
            # Keep sorted for searchsorted
            if ip < len(ones): ones = np.sort(ones)
