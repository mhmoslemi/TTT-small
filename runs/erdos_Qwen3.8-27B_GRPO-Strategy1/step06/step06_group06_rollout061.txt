import numpy as np
import time

def corr_full(h):
    N = len(h)
    return np.correlate(h, 1.0-h, mode='full')

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    a = np.zeros(L); a[:N] = h
    b = np.zeros(L); b[:N] = 1.0 - h[::-1]
    return np.real(np.fft.ifft(np.fft.fft(a)*np.fft.fft(b)))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def lse_grad(h, corr, beta):
    N = len(h)
    center = N-1
    bc = beta*corr
    bc -= bc.max()
    ew = np.exp(bc)
    w = ew / ew.sum()
    oh = 1.0 - h
    grad = np.zeros(N)
    for s in range(1, N):
        wp = w[center+s]
        wn = w[center-s]
        if wp > 1e-12:
            grad[s:] += wp * oh[:N-s]
            grad[:N-s] -= wp * h[s:]
        if wn > 1e-12:
            grad[:N-s] += wn * oh[s:]
            grad[s:] -= wn * h[:N-s]
    grad += w[center] * (1.0 - 2.0*h)
    return grad

def lse_optimize(h, N, deadline, beta0=5.0, beta_end=50.0, top_k=None):
    h = h.astype(float).copy()
    h = proj_sum(h, N/2.0)
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    lr = 0.01
    best_pk = np.max(fft_corr(h))
    best_h = h.copy()
    corr = fft_corr(h)
    while time.time() < deadline:
        if t % 100 == 0:
            frac = min(1.0, t / max(1, int((deadline - time.time()) * 10)))
            beta = beta0 + (beta_end - beta0) * frac
        else:
            beta = beta0 + (beta_end - beta0) * min(1.0, t / 3000)
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:

        grad = lse_grad(h, corr, beta)
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m/(1-b1**t); vh = v/(1-b2**t)
        h = proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
        if t % 500 == 0:

    return best_h, best_pk

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    a = np.zeros(L); a[:N] = h
    b = np.zeros(L); b[:N] = 1.0 - h[::-1]
    return np.real(np.fft.ifft(np.fft.fft(a)*np.fft.fft(b)))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s-target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid,0,1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N):
    target = N//2
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin==1)[0]
        sel = np.random.choice(len(idx), n1-target, replace=False)
        h_bin[idx[sel]] = 0
    elif n1 < target:
        idx = np.where(h_bin==0)[0]
        sel = np.random.choice(len(idx), target-n1, replace=False)
        h_bin[idx[sel]] = 1
    return h_bin

def lse_grad(h, corr, beta):
    N = len(h)
    center = N-1
    bc = beta*corr
    bc -= bc.max()
    ew = np.exp(bc)
    w = ew / ew.sum()
    oh = 1.0 - h
    grad = np.zeros(N)
    for s in range(1, N):
        wp = w[center+s]
        wn = w[center-s]
        if wp > 1e-12:
            grad[s:] += wp*oh[:N-s]
            grad[:N-s] -= wp*h[s:]
        if wn > 1e-12:
            grad[:N-s] += wn*oh[s:]
            grad[s:] -= wn*h[:N-s]
    grad += w[center]*(1.0-2.0*h)
    return grad

        t += 1
        m = b1*m+(1-b1)*grad
        v = b2*v+(1-b2)*grad*grad
        mh = m/(1-b1**t); vh = v/(1-b2**t)
        h = proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
        if t % 400 == 0:

    return best_h, best_pk
