                best_d = d
                best_pq = (int(p), int(qs[cands[np.argmin(nm-P)]]))
        if best_pq is not None and best_d < -1e-12:
            h[best_pq[0]] = 0; h[best_pq[1]] = 1
        else:
            if np.random.random() < 0.02 and len(oi)>0 and len(zi)>0:
                p = int(oi[np.random.randint(len(oi))])
                q = int(zi[np.random.randint(len(zi))])
                h[p], h[q] = 0, 1
            else:
                break
    return h, best

def cont_opt(h, N, dl, lr=0.04, T=6.0):
    h = h.astype(float).copy()
    tgt = N/2.0
    dx = 2.0/N
    N1 = N-1
    best_h = h.copy(); best_pk = np.max(fft_corr(h))
    m = np.zeros(N); v = np.zeros(N); t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    kt = min(8, 2*N-1)
    while time.time() < dl:
        t += 1
        c = fft_corr(h)
        pk = c.max()
        if pk < best_pk: best_pk = pk; best_h = h.copy()
        top = np.argpartition(c, -kt)[-kt:]
        w = np.exp(T*(c[top]-c.max())); w /= w.sum()
        grad = np.zeros(N)
        j = np.arange(N)
        for wi, ki in zip(w, top):
            sh = int(ki) - N1
            v1 = (j+sh>=0) & (j+sh<N)
            v2 = (j-sh>=0) & (j-sh<N)
            t1 = np.zeros(N); t1[v1] = 1.0-h[j[v1]+sh]
            t2 = np.zeros(N); t2[v2] = -h[j[v2]-sh]
            grad += wi*(t1+t2)
        grad -= grad.mean()
        bc1 = 1-b1**t; bc2 = 1-b2**t
        m = b1*m+(1-b1)*grad; v = b2*v+(1-b2)*grad**2
        h = h - lr*(m/bc1)/(np.sqrt(v/bc2)+eps)
        h = proj(h, tgt)
        if t % 400 == 0: lr *= 0.94
    return best_h, best_pk*dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = c5_val(best_h)

    # Phase 1: Binary search on initial
    d1 = min(deadline-400, t0+300)
    h1, _ = bsearch(h0, N0, d1, sweeps=4000, nq_max=50)
    c1 = c5_val(h1)
    if c1 < best_c5: best_c5, best_h = c1, h1.copy()

    # Phase 2: Continuous optimization
    if time.time() < deadline-300:
        d2 = min(deadline-250, time.time()+180)
        hc = best_h.astype(float).copy()
        hc += np.random.randn(N0)*0.03
        hc = proj(hc, N0/2.0)
        h2, c2r = cont_opt(hc, N0, d2, lr=0.04, T=6.0)
        if c2r < best_c5: best_c5, best_h = c2r, h2.copy()

    # Phase 3: Binary from continuous
    if time.time() < deadline-200:
        d3 = min(deadline-160, time.time()+180)
        h3, _ = bsearch(best_h, N0, d3, sweeps=3000, nq_max=40)
        c3 = c5_val(h3)
        if c3 < best_c5: best_c5, best_h = c3, h3.copy()

    # Phase 4: Try different N values
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [280, 460, 200]:
        if time.time() > deadline-80: break
        if N_try % 2 != 0: N_try += 1
        if N_try == len(best_h): continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(x_new, x_best, best_h)
        ht = (hi > 0.5).astype(float)
        nh, n1 = N_try//2, int(ht.sum())
        if n1 > nh:
            idx = np.where(ht==1)[0][nh:]
            ht[idx] = 0
        elif n1 < nh:
            idx = np.where(ht==0)[0][:nh-n1]
            ht[idx] = 1
        sub_dl = min(deadline-40, time.time()+60)
        h_r, _ = bsearch(ht, N_try, sub_dl, sweeps=2000, nq_max=40)
        c_r = c5_val(h_r)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 5: More binary search with different nq
    if time.time() < deadline-40:
        d5 = min(deadline-20, time.time()+50)
        h5, _ = bsearch(best_h, len(best_h), d5, sweeps=2000, nq_max=80)
        c5v = c5_val(h5)
        if c5v < best_c5: best_c5, best_h = c5v, h5.copy()

    # Phase 6: Final polish
    if time.time() < deadline-2:
        hF = best_h.astype(float).copy()
        hF += np.random.randn(len(hF))*0.01
        hF = proj(hF, len(hF)/2.0)
        hF, cF = cont_opt(hF, len(hF), deadline-1, lr=0.02, T=10.0)
        if cF < best_c5: best_c5, best_h = cF, hF.copy()

    # Final binary cleanup
    if time.time() < deadline-0.5:
        hB = (best_h > 0.5).astype(float)
        hB, _ = bsearch(hB, len(hB), deadline-0.2, sweeps=500, nq_max=60)
        cB = c5_val(hB)
        if cB < best_c5: best_c5, best_h = cB, hB

    final_c5 = c5_val(best_h)
    return best_h, final_c5, len(best_h)
