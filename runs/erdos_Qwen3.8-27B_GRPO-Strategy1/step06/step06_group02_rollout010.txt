def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

    N = len(h)

    return np.real(np.fft.ifft(H * G))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0.0
    elif n1 < nh:
        idx = np.where(h <= 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1.0
    return h

def topk_weights(k, alpha=2.0):
    w = np.exp(-np.linspace(0, alpha, k))
    return w / w.sum()

def compute_E(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pmc = np.clip(pm, 0, N - 1)
    ppc = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pmc], 0.0)
    hp = np.where(vp, h[ppc], 0.0)
    return (1.0 - hm) * vm.astype(float) - hp * vp.astype(float)

def binary_local(h, N, deadline, max_sweeps=5000, k_score=5, alpha=2.0):
    h = h.astype(float).copy()
    best_h = h.copy()
    c = fft_corr(h)
    best_max = float(c.max())
    no_improve = 0
    temp = 0.05
    tw = topk_weights(k_score, alpha)
    tw_rev = tw[::-1]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_max - 1e-12:
            best_max = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        E = compute_E(h, N)
        best_sc = np.inf
        best_pq = None
        for p in ones:
            base = c - E[p, :]
            cand = base[None, :] + E[zeros, :]
            kk = min(k_score, 2 * N - 1)
            top = np.partition(cand, -kk, axis=1)[:, -kk:]
            top_s = np.sort(top, axis=1)[:, ::-1]
            scores = top_s @ tw_rev
            mi = int(np.argmin(scores))
            if scores[mi] < best_sc:
                best_sc = scores[mi]
                best_pq = (int(p), int(zeros[mi]))
        curr_sc = float(np.sort(c)[-k_score:][::-1] @ tw_rev)
        if best_pq is None:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
        else:
            delta = best_sc - curr_sc
            if delta > 0:
                if np.random.random() < np.exp(-delta / max(temp, 1e-4)):
                    p, q = best_pq
                else:
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
            else:
                p, q = best_pq
        h[p] = 0.0
        h[q] = 1.0
        c = fft_corr(h)
        if sweep > 50:
            temp *= 0.999
        if no_improve > 1000:
            break
    return best_h, best_max

def nesterov(h, N, deadline, k_grad=50, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top_idx = np.argpartition(corr, -k_grad)[-k_grad:]
        top_vals = corr[top_idx]
        w = np.exp(-10.0 * (pk - top_vals))
        wn = w / w.sum()
        grad = np.zeros(N)
        center = N - 1
        for ii in range(k_grad):
            L = int(top_idx[ii] - center)
            wi = wn[ii]
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N - L])
                grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:N - M]
        h_new = project_sum(y - lr * grad / k_grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 50 == 0:
            lr *= 0.992
            beta = min(0.96, 0.88 + it * 0.0002)
    return best_h, best_pk * dx
