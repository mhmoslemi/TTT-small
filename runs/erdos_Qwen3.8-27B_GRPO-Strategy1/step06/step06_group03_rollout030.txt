```python
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)

    nh = N // 2
    rank = np.argsort(h)[::-1]  # indices from highest to lowest

                 cool_rate=0.998, init_temp=2.0):

    elif int(h.sum()) < nh:

        if time.time() > deadline:
            break

        P = float(c[top_idx].max()) if len(top_idx) > 0 else float(c.max())
        
        if P < best_c:
            best_c = P
            best_h = h.copy()
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Sample candidates
        if len(ones) > sample_n:
            p_list = np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        if len(zeros) > sample_n:
            q_list = np.random.choice(zeros, sample_n, replace=False)
        else:
            q_list = zeros
        
        # Evaluate all pairs
        best_gain = 0
        best_pq = None
        for p in p_list:
            for q in q_list:
                # Quick check: compute delta at top lags
                new_max = 0.0
                for ki in top_idx:
                    L = int(ki - center)
                    d = 0.0
                    ip = p - L
                    iq = q - L

                    iq2 = q + L

                    if 0 <= ip2 < N:
                        d += h[ip2]
                    if 0 <= iq < N:
                        d += (1.0 - h[iq])
                    if 0 <= iq2 < N:
                        d -= h[iq2]
                    val = c[ki] + d
                    if val > new_max:
                        new_max = val
                gain = P - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_pq = (int(p), int(q))

            # Accept if improves, or with probability exp(gain/temp) if not
            if best_gain > 0:
```

```python
    iq2 = q_arr + lags

        iq = q_list - L
        valid_iq = (iq >= 0) & (iq < N)
        dq[valid_iq] += (1.0 - h[iq[valid_iq]])
        iq2 = q_list + L
        valid_iq2 = (iq2 >= 0) & (iq2 < N)
        dq[valid_iq2] -= h[iq2[valid_iq2]]

        new_vals = c[top_idx[k]] + delta_k  # (P, Q)
        d[:, :, k] = new_vals
```

```python
def eval_swaps(h, c, p_list, q_list, lags_arr, top_idx, N):
    """
    p_list: (P,) positions of 1s to potentially flip to 0
    q_list: (Q,) positions of 0s to potentially flip to 1
    lags_arr: (K,) signed lags
    top_idx: (K,) indices into c corresponding to lags_arr
    Returns: (P, Q) array of new max correlation values
    """
    P = len(p_list)
    Q = len(q_list)
    K = len(lags_arr)
    result = np.zeros((P, Q))
    center = N - 1
    
    for k in range(K):
        L = int(lags_arr[k])
        if L == 0:
            # Delta is 0 for L=0 (binary)
            result[:] = np.maximum(result, c[top_idx[k]])
            continue
        
        dp = np.zeros(P)
        ip = p_list + L
        v1 = (ip >= 0) & (ip < N)
        dp[v1] -= (1.0 - h[ip[v1]])
        ip2 = p_list - L
        v2 = (ip2 >= 0) & (ip2 < N)
        dp[v2] += h[ip2[v2]]
        
        dq = np.zeros(Q)
        iq = q_list + L
        v3 = (iq >= 0) & (iq < N)
        dq[v3] += (1.0 - h[iq[v3]])
        iq2 = q_list - L
        v4 = (iq2 >= 0) & (iq2 < N)
        dq[v4] -= h[iq2[v4]]
        
        delta = dp[:, np.newaxis] + dq[np.newaxis, :]
        vals = c[top_idx[k]] + delta
        result = np.maximum(result, vals)
    
    return result
```

Positions with high pressure are the ones that contribute most to the top lags. Flipping them would reduce the top lags the most.

Positions with high relief benefit the most from being set to 1.

```python
def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
```
