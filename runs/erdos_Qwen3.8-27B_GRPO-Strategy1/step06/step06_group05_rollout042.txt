```python
    #              -w[j] * h[i-l] if 0<=i-l<N

    dn = i_idx - l_arr  # (N, 2N-1)
    
    mask_up = (up >= 0) & (up < N)  # (N, 2N-1)
    mask_dn = (dn >= 0) & (dn < N)
    
    up_c = np.clip(up, 0, N-1)
    dn_c = np.clip(dn, 0, N-1)
    
    w_2d = w[None, :]  # (1, 2N-1)
    
    term1 = (w_2d * (1 - h[up_c]) * mask_up).sum(axis=1)
    term2 = (w_2d * h[dn_c] * mask_dn).sum(axis=1)
    
    grad = (term1 - term2) * dx
    return grad
```

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_E(h, N):
    """E[i,m] = change in c[m] when setting h[i] from 0 to 1 (for binary h)"""
    lags = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    lv = lags[None, :]
    pp = pos + lv  # i + lag
    pm = pos - lv  # i - lag
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    hp = np.where(vp, h[pp_c], 0.0)
    hm = np.where(vm, h[pm_c], 0.0)
    return hp - hm

def bin_search(h, N, deadline, max_sweeps=2000, anneal=0.0):

    n1 = int(h.sum())
    if n1 != nh:

        zi = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[oi[:nh]] = 1
        else:
            h[oi] = 1
            h[zi[:nh - n1]] = 1
    c = fft_corr(h)
    best_c = float(c.max())
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:

        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 120:

        ones = np.where(h == 1)[0]

        E = compute_E(h, N)

        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 100:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        E = compute_E(h, N)
        # Sample a subset of ones to speed up
        if len(ones) > 80:
            p_sample = rng.choice(ones, 80, replace=False)
        else:
            p_sample = ones
        bd, bpq = 0.0, None
        for p in p_sample:
            delta = E[zeros, :] - E[p, :]
            new_c = c[None, :] + delta
            npk = new_c.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
            c = fft_corr(h)
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
            stall = 0
        else:
            break
    return h, best_c
