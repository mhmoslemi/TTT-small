def compute_corr(h):

For the continuous part, I'll use a simple projected subgradient method:
- Compute the correlation
- Find the top-K lags (where correlation is highest)
- For each top lag, compute the subgradient
- Average the subgradients
- Take a step in the negative subgradient direction
- Project to [0,1] with sum constraint

The subgradient at position i for lag k (where corr[k] = sum_j h[j]*(1-h[j+k-(N-1)])):
∂corr[k]/∂h[i] = (1-h[i+k-(N-1)]) - h[i-k+(N-1)] (with boundary checks)

For the max over a set S of lags, the subgradient is the sum of subgradients over S (or the average, depending on normalization).

def continuous_polish(h, N, deadline, alpha=0.01, K=50):
    h = h.astype(np.float64).copy()
    target = N / 2.0
    best_h = h.copy()
    best_v = corr_max(h)
    
    for it in range(10000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        v = float(np.max(corr))
        if v < best_v:
            best_v = v
            best_h = h.copy()
        
        # Top K lags
        top_k = np.argpartition(corr, -K)[-K:]
        # Subgradient
        grad = np.zeros(N)
        center = N - 1
        for k in top_k:
            d = int(k - center)  # lag offset
            if d == 0:
                grad += (1.0 - 2.0 * h)
            elif d > 0:
                # h[i] appears in corr as: (1-h[i+d]) for i in [0,N-d-1]
                # and as h[i-d] for i in [d, N-1] (as the 1-h factor)
                grad[:N-d] += (1.0 - h[d:])
                grad[d:] -= h[:N-d]
            else:
                dd = -d
                # h[i] appears as: (1-h[i-dd]) for i in [dd, N-1]
                # and as h[i+dd] for i in [0, N-dd-1]
                grad[dd:] += (1.0 - h[:N-dd])
                grad[:N-dd] -= h[dd:]
        grad /= K
        
        # Step
        h_new = h - alpha * grad
        # Project to [0,1] with sum = target
        h_new = np.clip(h_new, 0, 1)
        excess = h_new.sum() - target
        if excess > 0:
            # Need to decrease: subtract from largest values
            # Simple: subtract excess/N from all, clip at 0
            h_new -= excess / N
            h_new = np.clip(h_new, 0, 1)
            # Re-normalize iteratively
            for _ in range(20):
                e = h_new.sum() - target
                if abs(e) < 1e-10:
                    break
                # Adjust: if e > 0, reduce values > 0.5; if e < 0, boost values < 0.5
                if e > 0:
                    mask = h_new > 0
                    h_new[mask] -= e / mask.sum()
                else:
                    mask = h_new < 1
                    h_new[mask] += (-e) / mask.sum()
                h_new = np.clip(h_new, 0, 1)

        corr_cur = np.correlate(h, 1.0 - h, mode='full')
        base = corr_cur.copy()
        best_pair = None
        best_new_v = cur_v
        for oi in range(len(s_ones)):
            p = s_ones[oi]
            for zi in range(len(s_zeros)):
                q = s_zeros[zi]
                if p == q:
                    continue
                h[p] = 0.0
                h[q] = 1.0
                v = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                if v < best_new_v:
                    best_new_v = v
                    best_pair = (p, q)
                h[p] = 1.0
                h[q] = 0.0
        if best_pair is not None:
            p, q = best_pair
            h[p] = 0.0
            h[q] = 1.0
            idx_p = np.where(ones == p)[0][0]
            idx_q = np.where(zeros == q)[0][0]
            ones[idx_p] = q
            zeros[idx_q] = p
            cur_v = best_new_v
            if cur_v < best_v:
                best_v = cur_v
                best_h = h.copy()
            improved = True
            stall = 0
        else:
            stall += 1
            if stall > 3:
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                stall = 0
    return best_h, best_v

def continuous_opt(h, N, deadline, K=60, lr0=0.03):
    h = h.astype(np.float64).copy()
    target = N / 2.0
    best_h = h.copy()
    best_v = corr_max(h)
    lr = lr0
    beta = 0.85
    m = np.zeros(N)
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = np.correlate(y, 1.0 - y, mode='full')
        v = float(np.max(corr))
        if v < best_v:
            best_v = v
            best_h = y.copy()
        top = np.argpartition(corr, -K)[-K:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0 and L < N:
                grad[:N - L] += (1.0 - y[L:])
                grad[L:] -= y[:N - L]
            elif L < 0 and -L < N:
                M = -L
                grad[M:] += (1.0 - y[:N - M])
                grad[:N - M] -= y[M:]
        grad /= K
        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if int(best_v * N / 2.0) >= int(best_v * N / 2.0):
            pass
    return best_h, best_v * (2.0 / N)
