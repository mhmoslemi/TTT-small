```python
def swap_delta_fast(h, p, q, N):
    s_arr = np.arange(-(N-1), N)
    d = np.zeros(2*N-1)
    
    # p goes 1->0
    ps = p + s_arr  # p+s for each s
    pm = p - s_arr  # p-s for each s
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    d[vps] -= (1.0 - h[ps[vps]])
    d[vpm] -= h[pm[vpm]]
    
    # q goes 0->1
    qs = q + s_arr
    qm = q - s_arr
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    d[vqs] += (1.0 - h[qs[vqs]])
    d[vqm] -= h[qm[vqm]]
    
    return d
```

```python
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    while time.time() < deadline:
        t += 1
        corr = full_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Softmax weights
        denom = np.max(corr)
        weights = np.exp(beta * (corr - denom))
        weights /= weights.sum()
        
        # Weighted gradient
        grad = np.zeros(N)
        for idx in np.where(weights > 0.001)[0]:
            s = int(idx - (N-1))
            if s == 0:
                grad += weights[idx] * (1.0 - 2.0*h)
            elif s > 0:
                grad[:N-s] += weights[idx] * (1.0 - h[s:])
                grad[s:] -= weights[idx] * h[:N-s]
            else:
                s2 = -s
                grad[s2:] += weights[idx] * (1.0 - h[:N-s2])
                grad[:N-s2] -= weights[idx] * h[s2:]
        
        # ADAM update
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m/(1-b1**t)
        vh = v/(1-b2**t)
        h = proj_sum(h - lr*mh/np.sqrt(vh+eps), target)
        
        # Adaptive learning rate
        if t % 200 == 0:
            lr *= 0.95
        # Periodically increase beta
        if t % 500 == 0:
            beta *= 1.3
    
    return best_h, best_pk
```

Let me also add a "block move" to the binary annealing: move a contiguous block of 2-3 ones to a different location.

Actually, for the binary case, "move a block" is equivalent to multiple simultaneous swaps. If I move k consecutive 1s from position p to position q, that's k individual swaps. The total delta is the sum of individual deltas (if positions don't overlap in the shift calculation).

Let me simplify and just do:
- Single swap: swap one 1 with one 0
- Double swap: swap two 1s with two 0s (4 choices of pairing)

For the double swap, the delta is approximately the sum of individual deltas if the positions are far apart.

```python
import numpy as np
import time

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binarize_smart(h, N):
    target = N // 2
    h_bin = (h >= 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx])[::-1]
        for i in range(n1 - target):
            h_bin[idx[order[i]]] = 0
    elif n1 < target:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(h[idx])[::-1]
        for i in range(target - n1):
            h_bin[idx[order[i]]] = 1
    if int(h_bin.sum()) != target:
        # Fallback random
        n1 = int(h_bin.sum())
        if n1 > target:
            idx = np.where(h_bin == 1)[0]
            perm = np.random.permutation(len(idx))[:(n1 - target)]
            h_bin[idx[perm]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            perm = np.random.permutation(len(idx))[:(target - n1)]
            h_bin[idx[perm]] = 1
    return h_bin

def swap_delta(h, p, q, N, s_arr):
    d = np.zeros(2 * N - 1)
    ps, pm = p + s_arr, p - s_arr
    qs, qm = q + s_arr, q - s_arr
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    d[vps] -= (1.0 - h[ps[vps]])
    d[vpm] -= h[pm[vpm]]
    d[vqs] += (1.0 - h[qs[vqs]])
    d[vqm] -= h[qm[vqm]]
    return d

def double_swap_delta(h, p1, p2, q1, q2, N, s_arr):
    d1 = swap_delta(h, p1, q1, N, s_arr)
    h[p1], h[q1] = 0.0, 1.0
    d2 = swap_delta(h, p2, q2, N, s_arr)
    h[p1], h[q1] = 1.0, 0.0
    return d1 + d2

def softmax_adam(h, N, deadline, beta0=30.0, lr0=0.008, top_k=40):
    h = h.astype(float).copy()
    target = N / 2.0
    s_arr = np.arange(-(N - 1), N)
    center = N - 1
    best_pk = float(full_corr(h).max())
    best_h = h.copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = lr0
    beta = beta0
    while time.time() < deadline:
        t += 1
        corr = full_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        denom = np.max(corr)
        w = np.exp(beta * (corr - denom))
        ws = w.sum()
        if ws < 1e-30:
            break
        w /= ws
        top_idx = np.argsort(w)[::-1][:top_k]
        grad = np.zeros(N)
        for idx in top_idx:
            wi = w[idx]
            s = int(idx - center)
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N-s] += wi * (1.0 - h[s:])
                grad[s:] -= wi * h[:N-s]
            else:
                s2 = -s
                grad[s2:] += wi * (1.0 - h[:N-s2])
                grad[:N-s2] -= wi * h[s2:]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), target)
        if t % 300 == 0:
            lr *= 0.96
            beta *= 1.25
    return best_h, best_pk

def binary_anneal(h, N, deadline, T0=0.05, cool=0.9995, max_no_improve=600, n_restarts=3):
    h = h.astype(float).copy()
    s_arr = np.arange(-(N - 1), N)
    corr = full_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for restart in range(n_restarts):
        T = T0 * (1.0 if restart == 0 else 0.3)
        no_imp = 0
        while time.time() < deadline and T > 0.0005:
            if np.random.random() < 0.85:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                if p == q:
                    T *= cool
                    continue
                d = swap_delta(h, p, q, N, s_arr)
            else:
                if len(ones) < 3 or len(zeros) < 3:
                    p = int(ones[np.random.randint(len(ones))])
                    q = int(zeros[np.random.randint(len(zeros))])
                    if p == q:
                        T *= cool
                        continue
                    d = swap_delta(h, p, q, N, s_arr)
                else:
                    pii = np.random.choice(len(ones), 2, replace=False)
                    p1, p2 = int(ones[pii[0]]), int(ones[pii[1]])
                    qii = np.random.choice(len(zeros), 2, replace=False)
                    q1, q2 = int(zeros[qii[0]]), int(zeros[qii[1]])
                    if p1 == p2 or q1 == q2 or {p1,p2} == {q1,q2}:
                        T *= cool
                        continue
                    d = double_swap_delta(h, p1, p2, q1, q2, N, s_arr)
            nc = corr + d
            npk = float(nc.max())
            delta = npk - cur_pk
            if delta < -1e-12 or np.random.random() < np.exp(-delta / max(T, 1e-9)):
                if np.random.random() < 0.85 or len(ones) < 3:
                    h[p] = 0.0; h[q] = 1.0
                else:
                    h[p1] = 0.0; h[p2] = 0.0; h[q1] = 1.0; h[q2] = 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                corr = nc
                cur_pk = npk
                if npk < best_pk - 1e-14:
                    best_pk = npk
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1
            T *= cool
            if no_imp > max_no_improve:
                no_imp = 0
                if restart < n_restarts - 1:
                    break
                else:
                    h = best_h.copy()
                    corr = full_corr(h)
                    cur_pk = best_pk
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    T = T0 * 0.2
    return best_h, best_pk

        c5f = pkf * (2.0 / len(best_h))
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hfp.copy()

    # Final validation
    final_corr = full_corr(best_h)
    final_c5 = float(final_corr.max()) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
```
