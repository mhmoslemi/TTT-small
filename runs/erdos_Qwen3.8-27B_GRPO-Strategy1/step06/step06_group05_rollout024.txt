```python
def penalty_grad(c, D, sigma):
    # Smooth max approximation
    c_max = c.max()
    scaled = sigma * (c - c_max)  # prevent overflow
    w = np.exp(scaled)
    w /= w.sum()
    return D @ w  # (N,)
```

```python
def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv

    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -hm * vm + (1.0 - hp) * vp
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -hm * vm + (1.0 - hp) * vp

def smooth_max_grad(c, D, sigma):
    cm = c.max()
    w = np.exp(sigma * (c - cm))
    w /= w.sum()
    return D @ w

def cont_opt(h, N, deadline, sig_lo=15.0, sig_hi=120.0, lr0=0.04):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    t_start = time.time()
    total = max(deadline - t_start, 1.0)
    while time.time() < deadline:
        t += 1
        frac = min(1.0, (time.time() - t_start) / total)
        sigma = sig_lo + (sig_hi - sig_lo) * frac
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        D = compute_D(h, N)
        grad = smooth_max_grad(c, D, sigma) * dx
        grad -= grad.mean()
        bc1 = 1.0 - b1 ** t
        bc2 = 1.0 - b2 ** t
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        step = lr * (m / bc1) / (np.sqrt(v / bc2) + eps)
        h = project_sum(h + step, target)
        if t % 250 == 0:
            lr *= 0.93
    return best_h, best_pk * dx

def bin_search(h, N, deadline, max_sweeps=500, anneal=0.0, sample_n=None):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        oi = np.where(h > 0.5)[0]
        zi = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[oi[:nh]] = 1
        else:
            h[oi] = 1
            h[zi[:nh - n1]] = 1
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 50:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        if sample_n and len(ones) > sample_n:
            p_list = np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        bd, bpq = 0.0, None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    return h, best_c

def to_binary(h, N):
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    nh = N // 2
    if n1 != nh:
        oi, zi = np.where(hb == 1)[0], np.where(hb == 0)[0]
        hb[:] = 0.0
        if n1 > nh:
            idx = np.argsort(-h[oi])[:nh]
            hb[oi[idx]] = 1.0
        else:
            hb[oi] = 1.0
            idx = np.argsort(-h[zi])[:nh - n1]
            hb[zi[idx]] = 1.0
    return hb
```
