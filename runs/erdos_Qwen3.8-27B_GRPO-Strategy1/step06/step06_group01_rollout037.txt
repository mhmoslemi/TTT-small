    for i in range(N):

            if 0 <= i - ell < N:
                val -= h[i-ell]
            G[i, idx] = val
    return G

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        rem = np.random.choice(np.where(h > 0.5)[0], n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        add = np.random.choice(np.where(h < 0.5)[0], nh - n1, replace=False)
        h[add] = 1.0
    return (h > 0.5).astype(float)

def build_A(h, N):
    L = N - 1
    A = np.zeros((N, 2 * N - 1))
    for i in range(N):
        lo1 = max(-(N - 1 - i), -L)
        hi1 = min(i, L)
        if lo1 <= hi1:
            A[i, lo1 + L:hi1 + L + 1] -= (1.0 - h[i - hi1:i - lo1 + 1][::-1])
        lo2 = max(-i, -L)
        hi2 = min(N - 1 - i, L)
        if lo2 <= hi2:
            A[i, lo2 + L:hi2 + L + 1] += h[i + lo2:i + hi2 + 1]
    return A

        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.98
            beta = min(0.95, 0.80 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
