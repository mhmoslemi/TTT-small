    corr = fft_corr(h)

    center = N - 1

    top_lags = top_idx - center

            dq[j, valid2] -= h[np.clip(iq2, 0, N-1)[valid2]]

        min_cand = cand_max.min()
        if min_cand < best_P:
            best_P = min_cand
            best_q = zeros[cand_max.argmin()]
            best_pq = (int(p), int(best_q))
    
    if best_pq is not None:
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    
    return h, best_P

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N):
    """Convert to binary with exactly N/2 ones."""
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    target = N // 2
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[target:]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target-n1]] = 1
    return h

def greedy_sweep(h, N, top_m=30, max_sweeps=100, deadline=None):
    """Repeated greedy single-swap local search."""
    h = h.copy()
    C = (2.0 / N)
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        center = N - 1
        top_idx = np.argsort(corr)[-top_m:]
        lags = (top_idx - center).astype(int)
        top_vals = corr[top_idx]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        best_improve = 0.0
        best_pq = None
        
        for p in ones:
            dp = np.zeros(top_m)
            for j in range(top_m):
                L = lags[j]
                ip = p + L
                if 0 <= ip < N:
                    dp[j] -= (1.0 - h[ip])
                ip2 = p - L
                if 0 <= ip2 < N:
                    dp[j] += h[ip2]
            
            # Vectorized over zeros
            dq = np.zeros((top_m, len(zeros)))
            for j in range(top_m):
                L = lags[j]
                iq = zeros + L
                v1 = (iq >= 0) & (iq < N)
                if v1.any():
                    dq[j, v1] += (1.0 - h[np.clip(iq, 0, N-1)[v1]])
                iq2 = zeros - L
                v2 = (iq2 >= 0) & (iq2 < N)
                if v2.any():
                    dq[j, v2] -= h[np.clip(iq2, 0, N-1)[v2]]
            
            new_vals = top_vals[:, None] + dp[:, None] + dq
            cand_max = new_vals.max(axis=0)
            min_cand = float(cand_max.min())
            improve = P - min_cand
            if improve > best_improve:
                best_improve = improve
                best_pq = (int(p), int(zeros[cand_max.argmin()]))
        
        if best_pq is None or best_improve < 1e-12:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    
    return h, float(np.max(fft_corr(h))) * C

def greedy_sweep(h, N, top_m=30, max_sweeps=200, deadline=None):
    h = h.copy()
    C = 2.0 / N
    for sweep in range(max_sweeps):
        if deadline and time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        P = float(corr.max())
        center = N - 1
        top_idx = np.argsort(corr)[-top_m:]
        lags = (top_idx - center).astype(int)
        top_vals = corr[top_idx].copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_improve = 1e-12
        best_p = -1
        best_q = -1
        for p in ones:
            dp = np.zeros(top_m)
            for j in range(top_m):
                L = int(lags[j])
                # delta_p(L) = -(1-h[p-L]) if 0<=p-L<N + h[p+L] if 0<=p+L<N
                idx1 = p - L
                if 0 <= idx1 < N:
                    dp[j] -= (1.0 - h[idx1])
                idx2 = p + L
                if 0 <= idx2 < N:
                    dp[j] += h[idx2]
            # Vectorized over zeros: delta_q(L) = (1-h[q-L]) if 0<=q-L<N - h[q+L] if 0<=q+L<N
            dq = np.zeros((top_m, len(zeros)))
            for j in range(top_m):
                L = int(lags[j])
                iq1 = zeros - L
                v1 = (iq1 >= 0) & (iq1 < N)
                if v1.any():
                    dq[j, v1] += (1.0 - h[iq1[v1]])
                iq2 = zeros + L
                v2 = (iq2 >= 0) & (iq2 < N)
                if v2.any():
                    dq[j, v2] -= h[iq2[v2]]
            new_vals = top_vals[:, None] + dp[:, None] + dq
            cand_max = new_vals.max(axis=0)
            min_cand = float(cand_max.min())
            imp = P - min_cand
            if imp > best_improve:
                best_improve = imp
                best_p = int(p)
                best_q = int(zeros[cand_max.argmin()])
        if best_p < 0:
            break
        h[best_p] = 0.0
        h[best_q] = 1.0
    return h, float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * C
