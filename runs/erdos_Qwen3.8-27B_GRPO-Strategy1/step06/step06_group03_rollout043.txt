def nesterov_opt(h, N, deadline, k=60, lr0=0.02, beta0=0.92):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m if it > 0 else h.copy()
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = subgrad_topk(y, corr, k, N)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.95
            beta = min(0.97, beta0 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

def make_binary(h, N):
    nh = N // 2
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        perm = np.random.permutation(len(idx))
        for i in perm[:n1 - nh]:
            hb[idx[i]] = 0
    elif n1 < nh:
        idx = np.where(hb == 0)[0]
        perm = np.random.permutation(len(idx))
        for i in perm[:nh - n1]:
            hb[idx[i]] = 1
    return hb

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgrad_topk(y, corr, top_k, N):
    grad = np.zeros(N)
    center = N - 1
    n_act = min(top_k, len(corr))
    top = np.argpartition(corr, -n_act)[-n_act:]
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += (1.0 - y[:N - L])
            grad[:N - L] -= y[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
    grad /= n_act
    return grad

def nesterov_opt(h, N, deadline, k=60, lr0=0.02, beta0=0.92):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m if it > 0 else h.copy()
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = subgrad_topk(y, corr, k, N)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.95
            beta = min(0.97, beta0 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

def make_binary(h, N):
    nh = N // 2
    hb = (h > 0.5).astype(float).copy()
    n1 = int(hb.sum())
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        perm = np.random.permutation(len(idx))
        hb[idx[perm[:n1 - nh]]] = 0.0
    elif n1 < nh:
        idx = np.where(hb == 0)[0]
        perm = np.random.permutation(len(idx))
        hb[idx[perm[:nh - n1]]] = 1.0
    return hb

def binary_opt(h, N, deadline, max_sweeps=500, sample_n=100, top_m=12):
    h = make_binary(h, N)
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) >= 1 and len(zeros) >= 1:
                    h[ones[np.random.randint(len(ones))]] = 0.0
                    h[zeros[np.random.randint(len(zeros))]] = 1.0
                no_improve = 0
                continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        center = N - 1
        top_idx = np.argsort(c)[-top_m:]
        top_lags = (top_idx - center).astype(int)
        top_vals = c[top_idx].copy()
        ns = min(sample_n, len(ones), len(zeros))
        p_list = np.random.choice(ones, ns, replace=False) if len(ones) > ns else ones.copy()
        q_list = np.random.choice(zeros, ns, replace=False) if len(zeros) > ns else zeros.copy()
        A = np.zeros((len(p_list), top_m))
        B = np.zeros((len(q_list), top_m))
        for j in range(top_m):
            L = top_lags[j]
            ip = p_list + L
            vs = (ip >= 0) & (ip < N)
            A[:, j] += -np.where(vs, 1.0 - h[np.clip(ip, 0, N-1)], 0.0)
            ip2 = p_list - L
            vs2 = (ip2 >= 0) & (ip2 < N)
            A[:, j] += np.where(vs2, h[np.clip(ip2, 0, N-1)], 0.0)
            iq = q_list + L
            vq = (iq >= 0) & (iq < N)
            B[:, j] += np.where(vq, 1.0 - h[np.clip(iq, 0, N-1)], 0.0)
            iq2 = q_list - L
            vq2 = (iq2 >= 0) & (iq2 < N)
            B[:, j] -= np.where(vq2, 1.0 - h[np.clip(iq2, 0, N-1)], 0.0)
        new_c = top_vals[None, None, :] + A[:, None, :] + B[None, :, :]
        max_new = new_c.max(axis=2)
        best_val = float(max_new.min())
        if best_val < P - 1e-12:
            bi = np.unravel_index(np.argmin(max_new), max_new.shape)
            h[p_list[bi[0]]] = 0.0
            h[q_list[bi[1]]] = 1.0
        else:
            no_improve += 1
            if no_improve > 40:
                h[ones[np.random.randint(len(ones))]] = 0.0
                h[zeros[np.random.randint(len(zeros))]] = 1.0
                no_improve = 0
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

        hb = make_binary(hi, Nt)
        sd = min(time.time() + 50, deadline - 10)

        c5_rt = compute_c5(hrt)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, hrt
            x_best = xn
    # Phase 5: final Nesterov
    if time.time() < deadline - 20:
        d5 = min(time.time() + 40, deadline - 5)
        h5 = project_sum(best_h + np.random.randn(len(best_h)) * 0.04, len(best_h) / 2.0)

            best_c5, best_h = c5_5, h5
    # Phase 6: final binary

        c5_6 = compute_c5(h6)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgrad_topk(y, corr, top_k, N):
    grad = np.zeros(N)
    center = N - 1
    n_act = min(top_k, len(corr))
    top = np.argpartition(corr, -n_act)[-n_act:]
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += (1.0 - y[:N - L])
            grad[:N - L] -= y[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
    grad /= n_act
    return grad

def nesterov_opt(h, N, deadline, k=60, lr0=0.02, beta0=0.92):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m if it > 0 else h.copy()
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = subgrad_topk(y, corr, k, N)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.95
            beta = min(0.97, beta0 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

def make_binary(h, N):
    nh = N // 2
    hb = (h > 0.5).astype(float).copy()
    n1 = int(hb.sum())
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        perm = np.random.permutation(len(idx))
        hb[idx[perm[:n1 - nh]]] = 0.0
    elif n1 < nh:
        idx = np.where(hb == 0)[0]
        perm = np.random.permutation(len(idx))
        hb[idx[perm[:nh - n1]]] = 1.0
    return hb

def binary_opt(h, N, deadline, max_sweeps=500, sample_n=120, top_m=15):
    h = make_binary(h, N)
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    center = N - 1
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 35:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) >= 1 and len(zeros) >= 1:
                    h[ones[np.random.randint(len(ones))]] = 0.0
                    h[zeros[np.random.randint(len(zeros))]] = 1.0
                no_improve = 0
                continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        top_idx = np.argsort(c)[-top_m:]
        top_lags = (top_idx - center).astype(int)
        top_vals = c[top_idx].copy()
        ns = min(sample_n, len(ones), len(zeros))
        if len(ones) > ns:
            p_list = np.random.choice(ones, ns, replace=False)
        else:
            p_list = ones.copy()
        if len(zeros) > ns:
            q_list = np.random.choice(zeros, ns, replace=False)
        else:
            q_list = zeros.copy()
        A = np.zeros((len(p_list), top_m))
        B = np.zeros((len(q_list), top_m))
        for j in range(top_m):
            L = top_lags[j]
            ip = p_list + L
            vs = (ip >= 0) & (ip < N)
            A[:, j] += -np.where(vs, 1.0 - h[np.clip(ip, 0, N - 1)], 0.0)
            ip2 = p_list - L
            vs2 = (ip2 >= 0) & (ip2 < N)
            A[:, j] += np.where(vs2, h[np.clip(ip2, 0, N - 1)], 0.0)
            iq = q_list + L
            vq = (iq >= 0) & (iq < N)
            B[:, j] += np.where(vq, 1.0 - h[np.clip(iq, 0, N - 1)], 0.0)
            iq2 = q_list - L
            vq2 = (iq2 >= 0) & (iq2 < N)
            B[:, j] -= np.where(vq2, 1.0 - h[np.clip(iq2, 0, N - 1)], 0.0)
        new_c = top_vals[None, None, :] + A[:, None, :] + B[None, :, :]
        max_new = new_c.max(axis=2)
        best_val = float(max_new.min())
        if best_val < P - 1e-12:
            bi = np.unravel_index(np.argmin(max_new), max_new.shape)
            h[p_list[bi[0]]] = 0.0
            h[q_list[bi[1]]] = 1.0
        else:
            no_improve += 1
            if no_improve > 35:
                h[ones[np.random.randint(len(ones))]] = 0.0
                h[zeros[np.random.randint(len(zeros))]] = 1.0
                no_improve = 0
    return h, best_c
