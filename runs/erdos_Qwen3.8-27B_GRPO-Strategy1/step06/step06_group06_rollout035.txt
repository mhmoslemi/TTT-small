def corr_delta_single(h, p, q, N):
    """Compute delta in full correlation for swap: h[p]: 1->0, h[q]: 0->1"""
    d = np.zeros(2*N-1)
    c = N - 1  # center index
    
    # p: 1->0
    # First factor (n=p): removes (1-h[p+m-c]) for valid m
    m_lo_p = max(0, c-p)
    m_hi_p = min(2*N-2, 2*N-2-p)
    idx = np.arange(m_lo_p, m_hi_p+1)
    j = idx + p - c  # = p+m-c, the index in h for second factor
    d[idx] -= (1.0 - h[j])
    
    # Second factor (i=p): adds h[n] for m=n+p-c, n in valid range
    n_lo_p = max(0, c-p)
    n_hi_p = N-1
    if n_lo_p <= n_hi_p:
        ns = np.arange(n_lo_p, n_hi_p+1)
        ms = ns + p - c
        d[ms] += h[ns]
    
    # q: 0->1
    # First factor (n=q): adds (1-h[q+m-c]) for valid m
    m_lo_q = max(0, c-q)
    m_hi_q = min(2*N-2, 2*N-2-q)
    idx = np.arange(m_lo_q, m_hi_q+1)

def fft_corr(h):

def corr_delta(h, p, q, N):
    """Delta for swap h[p]:1->0, h[q]:0->1. Returns (2N-1) array."""
    d = np.zeros(2 * N - 1)
    c = N - 1
    # p removes as first factor, adds as second factor
    a = max(0, c - p)
    b = min(2 * N - 2, 2 * N - 2 - p)
    if a <= b:
        m = np.arange(a, b + 1)
        d[m] -= (1.0 - h[m + p - c])
    a2 = max(0, c - p)
    if a2 <= N - 1:
        n = np.arange(a2, N)
        d[n + p - c] += h[n]
    # q adds as first factor, removes as second factor
    a = max(0, c - q)
    b = min(2 * N - 2, 2 * N - 2 - q)
    if a <= b:
        m = np.arange(a, b + 1)
        d[m] += (1.0 - h[m + q - c])
    a2 = max(0, c - q)
    if a2 <= N - 1:
        n = np.arange(a2, N)
        d[n + q - c] -= h[n]
    return d

    ks = np.arange(2*N-1)
    delta = np.zeros(2*N-1)

    j1 = p + ks - c
    valid1 = (j1 >= 0) & (j1 < N)
    delta[valid1] -= (1.0 - h[j1[valid1]])

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.real(np.fft.irfft(H * G, L))[:2 * N - 1]

def compute_delta(h, p, q, N):
    c = N - 1
    ks = np.arange(2 * N - 1)
    delta = np.zeros(2 * N - 1)
    j1 = p + ks - c
    valid1 = (j1 >= 0) & (j1 < N)
    delta[valid1] -= (1.0 - h[j1[valid1]])
    j2 = q + ks - c
    valid2 = (j2 >= 0) & (j2 < N)
    hj2 = h[j2[valid2]].copy()
    hj2[j2[valid2] == p] = 0.0
    hj2[j2[valid2] == q] = 1.0
    delta[valid2] += (1.0 - hj2)
    n1 = p + c - ks
    valid3 = (n1 >= 0) & (n1 < N) & (n1 != p) & (n1 != q)
    delta[valid3] += h[n1[valid3]]
    n2 = q + c - ks
    valid4 = (n2 >= 0) & (n2 < N) & (n2 != p) & (n2 != q)
    delta[valid4] -= h[n2[valid4]]
    return delta

def make_binary(h, N):

    h = (h >= 0.5).astype(float)
    target = N // 2
    n1 = int(h.sum())

        idx = np.where(h == 1)[0]
        h[idx[np.random.choice(len(idx), n1 - target, replace=False)]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[idx[np.random.choice(len(idx), target - n1, replace=False)]] = 1
    return h

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if np.random.random() < guide_frac:
            threshold = cur_pk * 0.93
            hot = corr > threshold
            n_hot = int(hot.sum())
            if n_hot > 0:
                k_hot = int(np.random.choice(np.where(hot)[0]))
                s = k_hot - c
                if s >= 0:
                    vp = np.intersect1d(ones, np.arange(0, N - s))
                    vq = np.intersect1d(zeros, np.arange(s, N))
                else:
                    vp = np.intersect1d(ones, np.arange(-s, N))
                    vq = np.intersect1d(zeros, np.arange(0, N + s))
                if len(vp) > 0 and len(vq) > 0:
                    p = int(vp[np.random.randint(len(vp))])
                    q = int(vq[np.random.randint(len(vq))])
                else:
                    p = int(ones[np.random.randint(len(ones))])
                    q = int(zeros[np.random.randint(len(zeros))])
            else:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
        d = compute_delta(h, p, q, N)
        nc = corr + d
        npk = float(nc.max())
        delta_val = npk - cur_pk
        if delta_val < -1e-12 or (T > 0.003 and np.random.random() < np.exp(-delta_val / max(T, 1e-8))):
            h[p] = 0
            h[q] = 1
            corr = nc
            cur_pk = npk
            if npk < best_pk - 1e-12:
                best_pk = npk
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        T *= cool
        iters += 1
        if no_imp > 600:
            no_imp = 0
            T = max(T, T0 * 0.35)
            if iters % 3000 < 300:
                for _ in range(3):
                    p2 = int(ones[np.random.randint(len(ones))])
                    q2 = int(zeros[np.random.randint(len(zeros))])
                    d2 = compute_delta(h, p2, q2, N)
                    h[p2] = 0
                    h[q2] = 1
                    corr = corr + d2
    return best_h, best_pk

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def continuous_refine(h, N, deadline, top_k=40, lr0=0.004):
    h = h.astype(float).copy()
    h = proj_sum(h, N / 2.0)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    center = N - 1
    lr = lr0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N - s] += (1.0 - h[s:])
                grad[s:] -= h[:N - s]
            else:
                s2 = -s
                grad[s2:] += (1.0 - h[:N - s2])
                grad[:N - s2] -= h[s2:]
        grad /= top_k
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 400 == 0:
            lr *= 0.97
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = make_binary(h0, N0)
    best_c5 = float(np.max(fft_corr(best_h))) * (2.0 / N0)

    # Phase 1: Long guided anneal (45% budget)
    d1 = t0 + budget_s * 0.45
    h1, pk1 = guided_anneal(best_h, N0, d1, T0=0.06, cool=0.99935, guide_frac=0.6)
    c5_1 = pk1 * (2.0 / N0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Perturbed restart (20% budget)
    d2 = min(deadline - budget_s * 0.3, time.time() + budget_s * 0.2)
    h_pert = best_h.copy()
    for _ in range(8):
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        h_pert[p] = 0
        h_pert[q] = 1
    h2, pk2 = guided_anneal(h_pert, N0, d2, T0=0.045, cool=0.99945, guide_frac=0.7)
    c5_2 = pk2 * (2.0 / N0)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Multi-N exploration (20% budget)
    x_old = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in [320, 400, 440, 500]:
        if time.time() > deadline - budget_s * 0.15:
            break
        if Nt == len(best_h) or Nt % 2 != 0:
            continue
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        h_bin = make_binary(hi, Nt)
        rem = min(budget_s * 0.08, deadline - time.time() - 10)
        if rem < 20:
            continue
        hr, pkr = guided_anneal(h_bin, Nt, time.time() + rem, T0=0.05, cool=0.9994, guide_frac=0.6)
        c5r = pkr * (2.0 / Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()
            x_old = np.linspace(0, 2, Nt, endpoint=False)

    # Phase 4: Continuous refine then binarize (10% budget)
    if time.time() < deadline - 40:
        Nf = len(best_h)
        hc = best_h.astype(float) + np.random.randn(Nf) * 0.03
        hc = proj_sum(hc, Nf / 2.0)
        d_cont = min(deadline - 30, time.time() + budget_s * 0.08)
        hc2, pkc2 = continuous_refine(hc, Nf, d_cont, top_k=50, lr0=0.003)
        h_bin2 = make_binary(hc2, Nf)
        c5_cont = float(np.max(fft_corr(h_bin2))) * (2.0 / Nf)
        if c5_cont < best_c5:
            best_c5, best_h = c5_cont, h_bin2

    # Phase 5: Final intensive polish
    if time.time() < deadline - 3:
        Nf = len(best_h)
        hf = make_binary(best_h, Nf)
        hf, pkf = guided_anneal(hf, Nf, deadline - 1, T0=0.025, cool=0.9996, guide_frac=0.8)
        c5f = pkf * (2.0 / Nf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    final_c5 = float(np.max(fft_corr(best_h))) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
