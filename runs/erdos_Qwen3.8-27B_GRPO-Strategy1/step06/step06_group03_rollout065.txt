import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:

        order = np.random.permutation(len(idx))
        idx = idx[order[:nh - n1]]

    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Subgradient: average over top-k lags
        top_k = min(k, 2*N-1)
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        center = N - 1
        grad = np.zeros(N)
        for ai in top_idx:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M2 = -L
                grad[:N-M2] += (1.0 - y[M2:])
                grad[M2:] -= y[:N-M2]

        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.97, beta0 + it * 0.0002)
    
    return best_h, best_pk * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        order = np.random.permutation(len(idx))
        h[idx[order[:n1 - nh]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        order = np.random.permutation(len(idx))
        h[idx[order[:nh - n1]]] = 1
    return h

        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        center = N - 1
        grad = np.zeros(N)
        for ai in top_idx:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M2 = -L
                grad[:N - M2] += (1.0 - y[M2:])
                grad[M2:] -= y[:N - M2]
        grad /= len(top_idx)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.97, 0.90 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: aggressive fast binary search on parent
    d1 = min(deadline - 350, t0 + 300)
    h_s, _ = fast_binary_search(h0, N0, d1, top_m=30, reheat_interval=200)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: multi-resolution exploration
    N_list = [200, 220, 240, 260, 280, 300, 320, 340, 360, 380, 400, 440, 480]
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in N_list:
        if time.time() > deadline - 40:
            break
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = make_binary(h_interp, N_try)
        sub_dl = min(deadline - 25, time.time() + 90)
        h_rt, _ = fast_binary_search(h_bt, N_try, sub_dl, top_m=28, reheat_interval=180)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 3: Nesterov polish
    if time.time() < deadline - 80:
        hF = best_h + np.random.randn(len(best_h)) * 0.04
        hF = project_sum(hF, len(best_h) / 2.0)
        d3 = min(deadline - 50, time.time() + 70)
        hn, c5n = nesterov_opt(hF, len(best_h), d3, k=40, lr0=0.018)
        if c5n < best_c5:
            best_h, best_c5 = hn.copy(), c5n
            x_best = np.linspace(0, 2, len(best_h), endpoint=False)

    # Phase 4: final binary polish
    if time.time() < deadline - 3:
        h_fin = make_binary(best_h, len(best_h))
        h_fin, _ = fast_binary_search(h_fin, len(best_h), deadline - 1, top_m=30, reheat_interval=100)
        c5_f = compute_c5(h_fin, len(best_h))
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_fin.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
