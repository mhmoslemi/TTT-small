def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_G(h, N):
    pos = np.arange(N)[:, None]
    shifts = np.arange(2 * N - 1)[None, :] - (N - 1)
    ip = pos + shifts
    im = pos - shifts
    vp = ((ip >= 0) & (ip < N)).astype(np.float64)
    vm = ((im >= 0) & (im < N)).astype(np.float64)
    ip_c = np.clip(ip, 0, N - 1)
    im_c = np.clip(im, 0, N - 1)
    return (1.0 - h[ip_c]) * vp - h[im_c] * vm
