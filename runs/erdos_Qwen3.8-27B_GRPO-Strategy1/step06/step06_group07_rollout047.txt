            stall = 0

def anneal_binary(h, N, deadline, rng_seed=12345, T0=1.5, Tcool=0.997, tabu_len=15):

    R = corr_full(h)
    best_c = float(R.max())
    best_h = h.copy()

    no_improve = 0
    max_sweeps = 100000
    for it in range(max_sweeps):
        if time.time() > deadline:
            break
        T = T0 * (Tcool ** it)

            T = 0.005

def nesterov_cont(h, N, deadline, k=60, lr0=0.012):
    target = N / 2.0
    dx = 2.0 / N
    h = h.astype(float).copy()
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) > 1e-9:
        h = h / s * target
        h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_pk = float(np.max(np.correlate(h, 1 - h, mode='full')))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    for it in range(50000):
        if time.time() > deadline:
            break
        y = h + m
        corr = np.correlate(y, 1.0 - y, mode='full')
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = y - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        ss = h_new.sum()
        if ss > 1e-10:
            h_new = h_new * target / ss
            h_new = np.clip(h_new, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        if it % 200 == 0:
            lr *= 0.993
            beta = min(0.96, 0.82 + it * 0.0002)
    return best_h, best_pk * dx
