    center = N - 1
    Ls = shifts - center  # actual shift values

def delta_at_shifts(h, p, q, Ls, N):
    # Vectorized delta computation for multiple shifts
    d = np.zeros(len(Ls))
    m = (p - Ls >= 0) & (p - Ls < N)

    m = (p + Ls >= 0) & (p + Ls < N)

    m = (q - Ls >= 0) & (q - Ls < N)
    if m.any(): d[m] -= h[q - Ls[m]]
    m = (q + Ls >= 0) & (q + Ls < N)
    if m.any(): d[m] -= h[q + Ls[m]]
