Let me also think about a different approach: **gradient-based optimization with smooth max**.

The smooth max approach:
f(h) = (1/β) * log(Σ_k exp(β * corr_k / Σ_k' exp(β * corr_k')))
     = max_c + (1/β) * log(Σ_k exp(β * (corr_k - max_c)))

This is differentiable. The gradient w.r.t. h[j]:
∂f/∂h[j] = Σ_k w_k * ∂corr_k/∂h[j]
where w_k = exp(β*(corr_k - max_c)) / Σ exp(β*(corr_{k'} - max_c))

And ∂corr_k/∂h[j] = (1-h[j+k]) - h[j-k] (with validity checks)

This can be computed efficiently:
- Compute w_k (the softmax weights over shifts)
- The gradient at position j is: Σ_k w_k * [(1-h[j+k]) - h[j-k]]
  = (w * (1-h))[j] ... no, it's a convolution-like operation.

Actually: Σ_k w_k * (1-h[j+k]) is a "correlation" of w with (1-h).
And Σ_k w_k * h[j-k] is a "correlation" of w with h.

So ∂f/∂h[j] = corr(w, 1-h)[j] - corr(w, h)[j] (approximately, with appropriate index shifts)

This can be computed with FFT in O(N log N)!

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum_binary(h, N, seed=0):
    h = (h > 0.5).astype(float).copy()
    target = N // 2
    n1 = int(h.sum())
    rng = np.random.RandomState(seed + N)
    if n1 > target:
        idx = rng.choice(np.where(h == 1)[0], n1 - target, replace=False)
        h[idx] = 0
    elif n1 < target:
        idx = rng.choice(np.where(h == 0)[0], target - n1, replace=False)
        h[idx] = 1
    return h

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def project_sum(h, target):

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

        w = exps / exps.sum()

        wc = w  # wc[k] corresponds to shift (k - (N-1))
        for s in range(-(N - 1), N):
            wi = wc[N - 1 + s]
            if wi < 1e-14:

                    grad[j1] += wi * (1.0 - h[j1 + s])

                    grad[j2] -= wi * h[j2 - s]
            else:

                j1 = np.arange(ss, N)
                grad[j1] += wi * (1.0 - h[j1 - ss])
                j2 = np.arange(N - ss)
                grad[j2] -= wi * h[j2 + ss]
