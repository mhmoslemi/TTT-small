def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):

    if n1 > nh:

        h[add] = 1.0

    return h

def grad_corr(h, corr, top_k):
    """Compute gradient of max of top_k correlations w.r.t. h"""
    N = len(h)
    top = np.argpartition(corr, -top_k)[-top_k:]
    # Weight by how close to max
    vals = corr[top]
    mx = vals.max()
    mn = vals.min()
    if mx - mn > 1e-10:
        w = np.exp(3.0 * (vals - mx) / (mx - mn))  # weights in (exp(-3), 1]
        w /= w.sum()
    else:
        w = np.ones(len(top)) / len(top)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def grad_topk(h, corr, k, alpha=4.0):
    N = len(h)
    top = np.argpartition(corr, -k)[-k:]
    vals = corr[top]
    mx = vals.max()
    spread = mx - vals.min()
    if spread > 1e-10:
        w = np.exp(-alpha * (mx - vals) / spread)
        w /= w.sum()
    else:
        w = np.ones(len(top)) / len(top)
    grad = np.zeros(N)
    for idx in range(len(top)):
        t = int(top[idx])
        L = t - (N - 1)
        wi = w[idx]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def nesterov_opt(h_in, N, deadline, k=50, lr0=0.025, alpha=4.0):
    h = h_in.astype(float).copy()
    target = N / 2.0
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        c = float(corr.max())
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        grad = grad_topk(y, corr, k, alpha)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 40 == 0:
            lr *= 0.96
            beta = min(0.98, 0.85 + it * 0.0004)
    return best_h, best_c * (2.0 / N)

def binary_sa(h_in, N, deadline, max_sweeps=4000, n_samp=20, tabu_sz=40):
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    s = 2.0 * h - 1.0
    best_c = float(corr.max())
    best_h = h.copy()
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s[i]
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    tabu = set()
    no_imp = 0
    T = 0.8
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        n_p = min(n_samp, len(ones))
        n_q = min(n_samp * 2, len(zeros))
        p_list = np.random.choice(ones, n_p, replace=False)
        q_list = np.random.choice(zeros, n_q, replace=False)
        dp = deltas[p_list]
        dq = deltas[q_list]
        chunk = 8
        best_nc = 1e10
        best_pi, best_qi = 0, 0
        for ci in range(0, n_p, chunk):
            dc = min(ci + chunk, n_p)
            vals = corr[None, None, :] + dp[ci:dc, None, :] - dq[None, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_nc:
                best_nc = float(nm[mi])
                best_pi, best_qi = mi[0] + ci, mi[1]
        p = int(p_list[best_pi])
        q = int(q_list[best_qi])
        dc_ = best_nc - best_c
        if dc_ > 0 and ((p, q) in tabu or np.random.random() > np.exp(-dc_ / max(T, 0.01))):
            no_imp += 1
            if no_imp > 50:
                break
            if no_imp > 20:
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
        else:
            no_imp = 0
        h[p], h[q] = 0.0, 1.0
        corr += deltas[p] - deltas[q]
        s[p], s[q] = -s[p], -s[q]
        deltas[p, :].fill(0.0); deltas[p, p:p+N] = s[p]
        deltas[q, :].fill(0.0); deltas[q, q:q+N] = s[q]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        tabu.add((p, q))
        if len(tabu) > tabu_sz:
            tabu.pop()
        c = float(corr.max())
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        T *= 0.996
    return best_h, best_c * (2.0 / N)

def golden_init(N):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(N)
    for i in range(N):
        h[i] = 1.0 if (np.floor((i + 1) * phi) - np.floor(i * phi)) else 0.0
    return fix_binary(h, N // 2)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

        if time.time() > deadline - 60:

        x_old = np.linspace(0, 2, len(best_h), endpoint=False)

    if time.time() < deadline - 3:
