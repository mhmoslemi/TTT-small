import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0 - h), L)))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, corr, top_k):
    N = len(h)
    top = np.argsort(corr)[-top_k:]
    grad = np.zeros(N)
    for idx in top:
        d = int(idx - (N - 1))
        if d == 0:
            grad += (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += (1.0 - h[:N - d])
            grad[:N - d] -= h[d:]
        else:
            e = -d
            grad[:N - e] += (1.0 - h[e:])
            grad[e:] -= h[:N - e]
    return grad / len(top)

def adam_opt(h_init, N, deadline, lr=0.004, top_k=35):
    h = project_sum(h_init, N / 2.0)
    m = np.zeros(N)
    v = np.zeros(N)
    best_h = h.copy()
    corr = fft_corr(h)
    best_c = float(corr.max()) * (2.0 / N)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        g = compute_grad(h, corr, top_k)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        mh = m / (1 - beta1 ** t)
        vh = v / (1 - beta2 ** t)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h_new, N / 2.0)
        corr = fft_corr(h)
        c = float(corr.max()) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
        if t % 2000 == 0:
            lr *= 0.97
    return best_h, best_c

def fix_binary_count(h, target):
    h = h.copy()
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, n1 - target, replace=False)] = 0.0
    elif n1 < target:
        zeros = np.where(h <= 0.5)[0]
        h[np.random.choice(zeros, target - n1, replace=False)] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def sa_binary(h_in, N, deadline, max_iter=30000):
    target = N // 2
    h = fix_binary_count(h_in, target)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = 2.0
    T_min = 0.05
    cool = 0.9998
    for it in range(max_iter):
        if time.time() > deadline:
            break
        mode = np.random.random()
        if mode < 0.3 and len(ones) >= 2 and len(zeros) >= 2:
            p1 = int(np.random.choice(ones))
            rem = np.setdiff1d(ones, [p1])
            p2 = int(np.random.choice(rem))
            q1 = int(np.random.choice(zeros))
            rem2 = np.setdiff1d(zeros, [q1])
            q2 = int(np.random.choice(rem2))
            h[p1], h[q1], h[p2], h[q2] = 0, 1, 0, 1
            nc = fft_corr(h)

            if nc_max < best_c or np.random.random() < np.exp(-(nc_max - best_c) / T):

                    best_h = h.copy()

def smooth_max_opt(h_init, N, deadline):
    h = project_sum(h_init, N / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h, N)
    tau = 50.0
    t = 0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        c = float(corr.max()) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
        # Smooth max gradient
        vals = tau * corr
        vals -= vals.max()
        w = np.exp(vals)
        w /= w.sum()
        grad = np.zeros(N)
        active = np.argsort(w)[-15:]
        a = w[active] / w[active].sum()
        for i, idx in enumerate(active):
            d = int(idx - (N - 1))
            if d == 0:
                grad += a[i] * (1.0 - 2.0 * h)
            elif d > 0:
                grad[d:] += a[i] * (1.0 - h[:N - d])
                grad[:N - d] -= a[i] * h[d:]
            else:
                e = -d
                grad[:N - e] += a[i] * (1.0 - h[e:])
                grad[e:] -= a[i] * h[:N - e]
        grad *= (2.0 / N) / tau
        h = project_sum(h - 0.05 * grad, N / 2.0)
        tau *= 1.01
    return best_h, best_c
