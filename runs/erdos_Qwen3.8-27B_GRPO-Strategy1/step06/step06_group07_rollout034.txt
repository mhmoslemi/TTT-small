    target = N // 2
    hb = (h > 0.5).astype(float)

    if n1 == target:
        return hb
    rng = np.random.RandomState(777)

        idx = rng.choice(np.where(hb == 1)[0], n1 - target, replace=False)
        hb[idx] = 0
    else:
        idx = rng.choice(np.where(hb == 0)[0], target - n1, replace=False)
        hb[idx] = 1
    return hb

def greedy_once(h, N, deadline, max_iter=300, sample_z=15):

    corr = fft_corr(h)
    best = float(corr.max())
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        improved = False
        # Try all p, sample zeros
        zero_sample = zeros if len(zeros) <= sample_z else np.random.choice(zeros, sample_z, replace=False)
        for p in ones:
            h[p] = 0.0
            for q in zero_sample:
                h[q] = 1.0
                new_corr = fft_corr(h)
                m = float(new_corr.max())
                if m < best:
                    best = m
                    best_h = h.copy()
                    corr = new_corr
                    improved = True
                    break
                else:
                    h[q] = 0.0
            h[p] = 1.0
            if improved:
                break
        if not improved:
            break
    best_h = ensure_binary(best_h, N)
    best = float(np.max(fft_corr(best_h)))
    return best_h, best * (2.0 / N)

    zeros_pos = np.where(h == 0)[0].copy()
    
    n_steps = max_iter
    decay = (Tmin / T0) ** (1.0 / max(n_steps, 1))
    T = T0
    improves = 0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T *= decay
        if T < Tmin:
            T = Tmin
        
        # Pick random swap
        pi = rng.randint(len(ones_pos))
        qi = rng.randint(len(zeros_pos))
        p = int(ones_pos[pi])
        q = int(zeros_pos[qi])
        if p == q:
            continue
        
        h[p] = 0.0
        h[q] = 1.0
        new_corr = fft_corr(h)
        new_max = float(new_corr.max())
        delta = new_max - cur_max
        
        if delta < 0 or (T > Tmin * 0.5 and rng.random() < np.exp(-delta / T)):
            # Accept
            corr = new_corr
            cur_max = new_max
            ones_pos[pi] = q
            zeros_pos[qi] = p

                best_h = h.copy()
                improves += 1

            h[p] = 1.0
            h[q] = 0.0

    x_new = np.linspace(0, 2, N_new, endpoint=False)

    """Project to binary with exact sum, using randomized threshold."""
    target = N / 2.0

    return hb

        # Subgradient: average over shifts in top-k
        top_idx = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top_idx:
            L_shift = int(ai - center)
            if L_shift == 0:
                grad += (1.0 - 2.0 * y)
            elif L_shift > 0:
                M = L_shift
                grad[M:] += (1.0 - y[:N-M])
                grad[:N-M] -= y[M:]
            else:
                M = -L_shift
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= k
        # Project
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        # Enforce sum
        s = h_new.sum()
        if abs(s - target) > 1e-10:
            # Binary search for offset

                else:
                    hi = mid

        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, beta0 + it * 0.00005)
    return best_h, best_pk * dx

    fc5 = c5_from_h(best_h)
    if fc5 < best_c5:
        best_c5 = fc5
    return best_h, best_c5, Nf
