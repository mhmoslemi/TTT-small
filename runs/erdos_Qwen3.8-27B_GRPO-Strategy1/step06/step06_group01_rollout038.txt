def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    h1, c1 = sa_search(h0, N0, d1, seed=seed, T0=2.5, cool=0.9995, smart_frac=0.4)
    if c1 < best_c5:
        best_c5, best_h = c1, h1
    
    # Phase 2: Block SA for escaping local optima (18%)
    if time.time() < deadline - 120:
        d2 = min(deadline - 90, time.time() + (budget_s - 10) * 0.18)
        h2, c2 = block_sa(best_h, N0, d2, seed=seed + 7)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
    
    # Phase 3: SA different starting seed (15%)
    if time.time() < deadline - 80:
        d3 = min(deadline - 60, time.time() + (budget_s - 10) * 0.15)
        h3_init = best_h.copy()
        for _ in range(20):
            pp = int(np.random.choice(np.flatnonzero(h3_init)))
            qq = int(np.random.choice(np.flatnonzero(1 - h3_init)))
            h3_init[pp], h3_init[qq] = 0.0, 1.0
        h3, c3 = sa_search(h3_init, N0, d3, seed=seed + 200, T0=1.5, cool=0.9994, smart_frac=0.5)
        if c3 < best_c5:
            best_c5, best_h = c3, h3
    
    # Phase 4: Try N=400 (10%)
    if time.time() < deadline - 50:
        N4 = 400
        xo = np.linspace(0, 2, N0, endpoint=False)
        xn = np.linspace(0, 2, N4, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        h4i = fix_binary((hi > 0.45).astype(float), N4 // 2)
        d4 = min(deadline - 30, time.time() + (budget_s - 10) * 0.10)
        h4, c4 = sa_search(h4i, N4, d4, seed=seed + 500, T0=1.5, cool=0.9996, smart_frac=0.35)
        if c4 < best_c5:
            best_c5, best_h = c4, h4
    
    # Phase 5: Very cold final polish (remaining)
    if time.time() < deadline - 3:
        h5, c5 = sa_search(best_h, len(best_h), deadline - 1,
                           seed=seed + 999, T0=0.2, cool=0.997, smart_frac=0.6)
        if c5 < best_c5:
            best_c5, best_h = c5, h5
    
    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
