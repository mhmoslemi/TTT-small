def compute_delta_remove(h, N):
    """For each position p with h[p]=1, compute the delta in corr if h[p] is set to 0."""
    # delta[p, k] = change in corr at index k when h[p] goes from 1 to 0
    # corr[k+N-1] = sum_i h[i+k]*(1-h[i]) where k = idx - (N-1)
    k_vals = np.arange(-(N-1), N)  # length 2N-1
    L = 2*N - 1
    delta = np.zeros((N, L))
    
    # For position p (h[p]=1), setting to 0:
    # Term i+k=p (i=p-k): was (1-h[p-k]), now 0. Change: -(1-h[p-k]) if valid.
    # Term i=p (i+k=p+k): was 0, now h[p+k]. Change: +h[p+k] if valid.
    # At k=0: both terms are at i=p, and net change is 0.
    
    for p in np.where(h == 1)[0]:
        n1 = p - k_vals
        v1 = (n1 >= 0) & (n1 < N)
        delta[p, v1] -= (1.0 - h[n1[v1]])
        
        n2 = p + k_vals
        v2 = (n2 >= 0) & (n2 < N)
        delta[p, v2] += h[n2[v2]]
        
        delta[p, N-1] = 0.0  # fix k=0
    
    return delta

def compute_delta_add(h, N):
    """For each position q with h[q]=0, compute the delta in corr if h[q] is set to 1."""
    k_vals = np.arange(-(N-1), N)
    L = 2*N - 1
    delta = np.zeros((N, L))
    
    for q in np.where(h == 0)[0]:

        # Term i+k=q (i=q-k): was 0*(1-h[q-k])=0, now 1*(1-h[q-k])=1-h[q-k]. Change: +(1-h[q-k]).
        # Term i=q (i+k=q+k): was h[q+k]*1=h[q+k], now h[q+k]*0=0. Change: -h[q+k].
        # At k=0: net change is 0.

        v4 = (n4 >= 0) & (n4 < N)
        delta[q, v4] -= h[n4[v4]]
        
        delta[q, N-1] = 0.0
    
    return delta

        if len(ones) < 5 or len(zeros) < 5:
            break
        
        n_t = min(n_try, len(ones))
        if len(ones) > n_t:
            p_sample = ones[np.random.choice(len(ones), n_t, replace=False)]
        else:
            p_sample = ones
        
        best_score = best_peak + T
        best_pq = None
        best_new_c = None
        
        for p in p_sample:
            n1 = p - k_vals
            v1 = (n1 >= 0) & (n1 < N)
            dp = np.zeros(L)
            dp[v1] -= (1.0 - h[n1[v1]])
            n2 = p + k_vals
            v2 = (n2 >= 0) & (n2 < N)
            dp[v2] += h[n2[v2]]
            dp[N-1] = 0.0
            
            base = c + dp
            peak_idx = int(np.argmax(base))
            
            k_peak = k_vals[peak_idx]
            n3 = zeros - k_peak
            v3 = (n3 >= 0) & (n3 < N)
            n4 = zeros + k_peak
            v4 = (n4 >= 0) & (n4 < N)
            dq_peak = np.zeros(len(zeros))
            if np.any(v3):
                dq_peak[v3] += (1.0 - h[n3[v3]])
            if np.any(v4):
                dq_peak[v4] -= h[n4[v4]]
            if k_peak == 0:
                dq_peak[:] = 0.0
            
            top_idx = np.argsort(dq_peak)[:min(5, len(zeros))]
            
            for qi in top_idx:
                q = int(zeros[qi])
                n3f = q - k_vals
                v3f = (n3f >= 0) & (n3f < N)
                n4f = q + k_vals
                v4f = (n4f >= 0) & (n4f < N)
                dq = np.zeros(L)
                dq[v3f] += (1.0 - h[n3f[v3f]])
                dq[v4f] -= h[n4f[v4f]]
                dq[N-1] = 0.0
                
                new_c = base + dq
                score = float(new_c.max())
                
                if score < best_score:
                    best_score = score
                    best_pq = (int(p), q)
                    best_new_c = new_c
        
        if best_pq is not None and best_score < best_peak + T:
            if best_score < best_peak - 1e-10:
                accept = True
            else:
                delta = best_peak - best_score
                accept = np.log(np.random.exponential(1.0)) > -delta / (T + 1e-10)
            
            if accept:
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
                c = best_new_c.copy()
                if float(c.max()) < best_peak:
                    best_peak = float(c.max())
                    no_improve = 0
                else:
                    no_improve += 1
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
            no_improve += 1
        
        if no_improve > 40:
            T *= 0.96
        if no_improve > 150:
            # Perturbation
            n_pert = np.random.randint(2, 5)
            for _ in range(n_pert):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) < 2 or len(zeros) < 2:
                    break
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
            T = max(T, 0.005)
            no_improve = 0
        
        if no_improve > 500:
            break
    
    return h, best_peak * (2.0 / N)

def nesterov_continuous(h_in, N, deadline, k_top=40, lr0=0.01):
    h = h_in.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    
    def project(x):
        x = np.clip(x, 0.0, 1.0)
        s = float(x.sum())
        if abs(s - target) < 1e-10:
            return x
        lo, hi = -1.0, 1.0
        for _ in range(40):
            mid = (lo + hi) * 0.5
            if float(np.clip(x - mid, 0.0, 1.0).sum()) > target:
                lo = mid
            else:
                hi = mid
        return np.clip(x - (lo+hi)*0.5, 0.0, 1.0)
    
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L_shift = int(ai - center)
            if L_shift == 0:
                grad += (1.0 - 2.0 * y)
            elif L_shift > 0:
                grad[L_shift:] += (1.0 - y[:N-L_shift])
                grad[:N-L_shift] -= y[L_shift:]
            else:
                M = -L_shift
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        
        grad /= k_top
        h_new = project(y - lr * grad)
        m = beta * (h_new - h)
        h = h_new
        lr *= 0.9995
        beta = min(0.97, beta + 0.0001)
    
    return best_h, best_c * dx

def fibonacci_pattern(N):
    """Create a binary pattern based on golden ratio (Fibonacci word)."""
    phi = (1 + np.sqrt(5)) / 2
    pos = np.floor(np.arange(N) * phi) % N
    h = np.zeros(N)
    h[pos[:N//2]] = 1.0
    # Ensure exactly N/2 ones
    h = fix_binary_sum(h, N)
    return h
