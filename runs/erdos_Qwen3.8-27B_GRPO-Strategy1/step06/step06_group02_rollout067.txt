Let me also think about a key algorithmic improvement: instead of computing D as a full N×(2N-1) matrix, I can compute it on the fly for a given p using the correlation structure.

D[p, j] = (1-h[p-j]) - h[p+j]

This is just two lookups into h! So for a given p, D[p, :] can be computed in O(N) by just indexing into h.

We want to minimize max(c_new) over all lags.

- For each candidate q (one of the 0s), compute Dq = D[q,:]
- Score = max(c - Dp + Dq)
- Pick the (p,q) pair with minimum score

```python
import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    Lp = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, Lp)
    G = np.fft.rfft(np.flip(1.0 - h), Lp)
    return np.real(np.fft.irfft(H * G, Lp))[:L]

def compute_D_row(h, p):
    """D[p, m] = (1-h[p-j]) - h[p+j], j = m-(N-1)"""
    N = len(h)
    js = np.arange(2*N-1) - (N-1)

    pp = p + js

    vp = (pp >= 0) & (pp < N)
    D[vm] = 1.0 - h[pm[vm]]
    D[vp] -= h[pp[vp]]

    best_h = h.copy()
```

```python
    beta = 0.9
    lr = lr0

    it = 0
    while time.time() < deadline:
        c = fft_corr(h)
        top = np.argpartition(c, -K)[-K:]
        # Compute subgradient
        center = N - 1
        grad = np.zeros(N)
        for mi in top:
            j = int(mi - center)
            # D[k, mi] = (1-h[k-j]) - h[k+j]
            k_arr = np.arange(N)
            km = k_arr - j
            kp = k_arr + j
            vm = (km >= 0) & (km < N)
            vp = (kp >= 0) & (kp < N)
            d = np.zeros(N)
            d[vm] = 1.0 - h[km[vm]]
            d[vp] -= h[kp[vp]]
            grad += d
        grad /= K
        # Nesterov step
        y = h + m_vec
        grad_y = np.zeros(N)
        for mi in top:
            j = int(mi - center)
            k_arr = np.arange(N)

            d[vp] -= y[kp[vp]]
```

A lower bound: base[m*] where m* = argmax(base) is the minimum possible max (achieved if D[q, m*] ≤ 0 for all m where base is near its max). But D[q, m*] = (1-h[q-j*]) - h[q+j*] which can be positive or negative.

Greedy approach: m* = argmax(base), then pick q to maximize D[q, m*] (to reduce the peak most). This is O(|zeros|) per p.

Total: O(|ones| * |zeros|) ≈ 36K per sweep. Very fast!

```python
    k = np.arange(N)[:, None]
    j = (np.arange(L) - (N - 1))[None, :]
    pm = k - j
    pp = k + j
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    D = np.where(vm, 1.0 - h[np.clip(pm, 0, N-1)], 0.0)
    D -= np.where(vp, h[np.clip(pp, 0, N-1)], 0.0)
    return D

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h > 0.5)[0]
        sel = idx[np.random.permutation(len(idx))][:n1 - nh]
        h[sel] = 0.0
    elif n1 < nh:
        idx = np.where(h <= 0.5)[0]
        sel = idx[np.random.permutation(len(idx))][:nh - n1]
        h[sel] = 1.0
    return h
```

Hmm, the Nesterov momentum is tricky. Let me redo:

Standard Nesterov:
y = x + m  (lookahead)
grad = f'(y)  (gradient at lookahead)
x_new = x - lr * grad  (gradient step from current)
m = beta * (x_new - x)  (update momentum)
x = x_new

Wait, that's not standard either. Let me use the standard:
m = beta * m - lr * grad(y)
x_new = x + m

Or the more common form:
y = x + beta * m  (lookahead: x + beta * velocity)
grad = f'(y)
m_new = beta * m + grad(y)  (new velocity)
x_new = x - lr * m_new  (update)

Actually, let me just use momentum SGD:
v = beta * v - lr * grad
x = x + v

This is simpler and works well.

```python
def subgrad_opt(h_in, N, deadline, K=30, lr0=0.04):
    h = h_in.astype(float)
    tgt = N / 2.0
    dx = 2.0 / N
    best_h, best_val = h.copy(), float(np.max(fft_corr(h)) * dx)
    v = np.zeros(N)
    lr, beta, it = lr0, 0.9, 0
    ka = np.arange(N)
    while time.time() < deadline:
        y = h + beta * v  # lookahead
        c = fft_corr(y)
        top = np.argpartition(c, -K)[-K:]
        g = np.zeros(N)
        for mi in top:
            j = int(mi - (N - 1))
            km, kp = ka - j, ka + j
            vm = (km >= 0) & (km < N)
            vp = (kp >= 0) & (kp < N)
            d = np.zeros(N)
            d[vm] = 1.0 - y[km[vm]]
            d[vp] -= y[kp[vp]]
            g += d
        v = beta * v - (lr / K) * g
        h = project_sum(h + v, tgt)
        val = float(np.max(fft_corr(h)) * dx)
        if val < best_val:
            best_val, best_h = val, h.copy()
        it += 1
        if it % 200 == 0:
            lr = max(0.002, lr * 0.99)
            beta = min(0.97, 0.88 + it * 0.0001)
    return best_h, best_val
```
