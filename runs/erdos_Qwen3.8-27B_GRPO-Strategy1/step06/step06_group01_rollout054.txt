def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def init_deltas(s, N):
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    return deltas

def sa_binary(h_in, N, deadline, T0=0.015, Tend=0.0003, n_sample=25, chunk=8):
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    s = 2.0 * h - 1.0
    deltas = init_deltas(s, N)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    tabu = []
    no_improve = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if len(ones) < 2 or len(zeros) < 2:
            break
        T = max(T0 * (Tend / T0) ** (it / 20000.0), 1e-7)
        n_p = min(n_sample, len(ones))
        p_list = np.random.choice(ones, n_p, replace=False)
        q_list = zeros
        dp = deltas[p_list]
        dq = deltas[q_list]
        best_new_val = 1e10
        best_pi, best_qi = 0, 0
        for ci in range(0, n_p, chunk):
            dc = min(ci + chunk, n_p)
            vals = corr[None, None, :] + dp[ci:dc, None, :] - dq[None, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new_val:
                best_new_val = float(nm[mi])
                best_pi, best_qi = mi[0] + ci, mi[1]
        p = int(p_list[best_pi])
        q = int(q_list[best_qi])
        is_tabu = any(p == tp and q == tq for tp, tq in tabu)
        delta_c = best_new_val - best_c
        if (delta_c < -1e-12) or (not is_tabu and T > 1e-7 and np.random.random() < np.exp(-max(delta_c, 0.0) / T)):
            h[p], h[q] = 0.0, 1.0
            corr = corr + deltas[p] - deltas[q]
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0; deltas[p, p:p + N] = s
            deltas[q, :] = 0.0; deltas[q, q:q + N] = s
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            tabu.append((p, q))
            if len(tabu) > 40:
                tabu.pop(0)
            c_now = float(corr.max())
            if c_now < best_c - 1e-12:
                best_c = c_now
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 50:
            n_sw = min(4, len(ones), len(zeros))
            rp = np.random.choice(ones, n_sw, replace=False)
            rq = np.random.choice(zeros, n_sw, replace=False)
            for ii in range(n_sw):
                pp, qq = int(rp[ii]), int(rq[ii])
                h[pp], h[qq] = 0.0, 1.0
                corr += deltas[pp] - deltas[qq]
                s[pp], s[qq] = -s[pp], -s[qq]
                deltas[pp, :] = 0.0; deltas[pp, pp:pp+N] = s
                deltas[qq, :] = 0.0; deltas[qq, qq:qq+N] = s
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            tabu.clear()
            no_improve = 0
    return best_h, best_c * (2.0 / N)

def continuous_opt(h_in, N, deadline, k_top=40, lr0=0.008):
    h = h_in.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_c5 = compute_c5(h, N)
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        it += 1
        y = h
        corr = fft_corr(y)
        peak = float(corr.max())
        c5_now = peak * (2.0 / N)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        top = np.sort(top)
        grad = np.zeros(N)
        for t in top:
            L = int(t - (N - 1))
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N - L] += (1.0 - y[L:])
                grad[L:] -= y[:N - L]
            else:
                M = -L
                grad[M:] += (1.0 - y[:N - M])
                grad[:N - M] -= y[M:]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = project_sum(y + m, target)
        if it % 80 == 0:
            lr *= 0.96
            beta = min(0.95, 0.80 + it * 0.0004)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: SA binary search (250s)
    d1 = min(deadline - 500, t0 + 250)
    h_bin = (h0 > 0.5).astype(float)
    h1, c1 = sa_binary(h_bin, N0, d1, T0=0.018, Tend=0.0004, n_sample=28)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Continuous relaxation (200s)
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 200)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.04
        h2, c2 = continuous_opt(hc, N0, d2, k_top=50, lr0=0.010)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Binary search from relaxed (200s)
    if time.time() < deadline - 200:
        d3 = min(deadline - 150, time.time() + 200)
        h3, c3 = sa_binary((best_h > 0.5).astype(float), N0, d3, T0=0.012, Tend=0.0003, n_sample=30)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Multi-start SA (250s)
    for trial in range(3):
        if time.time() > deadline - 150:
            break
        pert = best_h.astype(float).copy()
        n_flip = N0 // 10
        flip_pos = np.random.choice(N0, n_flip, replace=False)
        for fp in flip_pos:
            pert[fp] = 1.0 - pert[fp]
        pert = project_sum(pert, N0 / 2.0)
        d4 = min(deadline - 150 - trial * 50, time.time() + 80)
        h4, c4 = sa_binary((pert > 0.5).astype(float), N0, d4, T0=0.02, Tend=0.0005, n_sample=32)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Phase 5: Try N=400
    if time.time() < deadline - 100:
        N_alt = 400
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_alt, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bt = fix_binary(h_interp, N_alt // 2)
        d5 = min(deadline - 80, time.time() + 80)
        h5, c5 = sa_binary(h_bt, N_alt, d5, T0=0.015, Tend=0.0003, n_sample=25)
        if c5 < best_c5:
            best_c5, best_h = c5, h5
        else:
            h5c, c5c = continuous_opt(h_bt.astype(float), N_alt, min(deadline - 60, time.time() + 40), k_top=40, lr0=0.008)
            if c5c < best_c5:
                best_c5, best_h = c5c, h5c

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        d6 = deadline - 3
        h6, c6 = sa_binary((best_h > 0.5).astype(float), len(best_h), d6, T0=0.008, Tend=0.0001, n_sample=35)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    # Final continuous micro-adjust
    if time.time() < deadline - 4:
        hF = best_h.astype(float)
        hF = project_sum(hF, len(hF) / 2.0)
        hF, cF = continuous_opt(hF, len(hF), deadline - 2, k_top=60, lr0=0.005)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
