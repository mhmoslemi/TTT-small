    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def greedy_pass(h, N, deadline, max_sweeps=500, sample_frac=1.0):
    """Pure greedy: always take the best improving flip."""
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 50:
            n_s = max(50, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            npk = (base[None, :] - Dz).max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None or best_new_P >= P - 1e-14:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
    return best_h, best_c * (2.0 / N)

def sa_pass(h, N, deadline, max_sweeps=500, sample_frac=1.0, T0=0.008, T_min=0.0005):
    """Simulated annealing on binary flips."""
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    T = T0
    stall = 0
    cool = max(0.998, (T_min / T0) ** (1.0 / max(max_sweeps, 1)))
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 50:
            n_s = max(50, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            npk = (base[None, :] - Dz).max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        if delta < -1e-14:
            accept = True
        elif delta < T:
            accept = np.random.random() < np.exp(-delta / max(T, 1e-10))
        else:
            accept = False
        if accept:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            P = float(c.max())
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T = max(T_min, T * cool)
    return best_h, best_c * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h, N, deadline, k=50, lr0=0.018):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.0001)
    return best_h, best_pk * dx

        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 50:
            n_s = max(50, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            npk = (base[None, :] - Dz).max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        delta = best_new_P - P
        if delta < -1e-14:
            accept = True
        elif delta < 0.5 * T:
            accept = np.random.random() < np.exp(-delta / max(T, 1e-10))
        else:
            accept = False
        if accept:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            P = float(c.max())
            if P < best_c - 1e-14:
                best_c = P
                best_h = h.copy()
        T = max(T_min, T * cool)
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=100, lr0=0.02, beta0=0.85):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        kk = min(k, 2 * N - 1)
        top = np.argpartition(corr, -kk)[-kk:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= kk
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.985
            beta = min(0.96, beta0 + it * 0.0002)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Long Nesterov at N0 (continuous, allows fractional values)
    d1 = min(deadline - 600, t0 + 200)
    h1, c5_1 = nesterov_opt(best_h, N0, d1, k=120, lr0=0.02, beta0=0.82)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Round to 0/1 and greedy binary search
    h_bin = ensure_sum(best_h, N0)
    d2 = min(deadline - 450, time.time() + 80)
    h2, c5_2 = greedy_binary(h_bin, N0, d2, max_sweeps=800, sample_frac=1.0)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Nesterov from 0/1 solution with perturbation
    if time.time() < deadline - 350:
        hc = best_h + np.random.randn(N0) * 0.03
        hc = project_sum(hc, N0 / 2.0)
        d3 = min(deadline - 300, time.time() + 100)
        h3, c5_3 = nesterov_opt(hc, N0, d3, k=150, lr0=0.015, beta0=0.80)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: ILS with binary search (multiple restarts)
    for i in range(3):
        if time.time() > deadline - 250:
            break
        h_pert = ensure_sum(best_h, N0)
        n_flip = 2 + 2 * i
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(min(n_flip, len(ones), len(zeros))):
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h_pert[p], h_pert[q] = 0.0, 1.0
        h_pert = ensure_sum(h_pert, N0)
        d_pert = min(deadline - 200, time.time() + 60)
        hp, cp = greedy_binary(h_pert, N0, d_pert, max_sweeps=400, sample_frac=1.0)
        if cp < best_c5:
            best_c5, best_h = cp, hp.copy()

    # Phase 5: SA binary search for escape
    if time.time() < deadline - 150:
        h_sa = ensure_sum(best_h, N0)
        d5 = min(deadline - 100, time.time() + 80)
        h5, c5_5 = sa_binary(h_sa, N0, d5, max_sweeps=5000, sample_frac=1.0, T0=0.005, T_min=0.0002)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Nesterov at higher resolution N=760
    N1 = 2 * N0
    if time.time() < deadline - 120:
        h_up = upsample(best_h, len(best_h), N1)
        h_up = project_sum(h_up, N1 / 2.0)
        d6 = min(deadline - 60, time.time() + 80)
        h6, c5_6 = nesterov_opt(h_up, N1, d6, k=150, lr0=0.015, beta0=0.82)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final Nesterov polish at N0
    if time.time() < deadline - 20:
        h_fin = upsample(best_h, len(best_h), N0) if len(best_h) != N0 else best_h.copy()
        h_fin = project_sum(h_fin, N0 / 2.0)
        d7 = max(deadline - 1, time.time() + 40)
        h7, c5_7 = nesterov_opt(h_fin, N0, d7, k=200, lr0=0.008, beta0=0.75)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
