import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1-nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh-n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def interp_resample(h_old, N_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    cur_c = best_c
    s = 2.0*h - 1.0
    deltas = np.zeros((N, 2*N-1))
    for i in range(N):
        deltas[i, i:i+N] = s
    T = T0
    it = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline:
        it += 1
        T = max(Tmin, T0 * (cool ** it))
        
        if np.random.random() < pair_frac:
            if len(ones) < 2 or len(zeros) < 2:
                break
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))

                if new_c < best_c:

                    best_h = h.copy()
                h[p], h[q] = 0.0, 1.0
                corr += delta
                s[p], s[q] = -s[p], -s[q]
                deltas[p, :] = 0.0
                deltas[p, p:p+N] = s
                deltas[q, :] = 0.0
                deltas[q, q:q+N] = s
                cur_c = new_c
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        else:

            start = int(np.random.choice(ones))

                    best_h = h.copy()

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def interp_resample(h_old, N_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def sa_optimize(h_in, N, deadline, T0=0.003, Tmin=0.00004, cool=0.9997, pair_frac=0.78, block_frac=0.20):

    it = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    rbuf = np.random.random
    while time.time() < deadline:
        it += 1
        T = max(Tmin, T0 * (cool ** it))
        r = rbuf()
        if r < pair_frac:
            if len(ones) < 2 or len(zeros) < 2:
                break
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            delta = deltas[p] - deltas[q]
            new_c = float((corr + delta).max())

                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                h[p], h[q] = 0.0, 1.0
                corr += delta
                s[p], s[q] = -s[p], -s[q]
                deltas[p, :] = 0.0
                deltas[p, p:p + N] = s
                deltas[q, :] = 0.0

                    deltas[pos, pos:pos + N] = s
                cur_c = new_c
                ones = np.where(h == 1)[0]

def continuous_opt(h_in, N, deadline, k_top=50):
    h = h_in.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    m = np.zeros(N)
    best_h = h.copy()
    best_val = compute_c5(h, N)
    lr = 0.007
    beta = 0.89
    it = 0
    while time.time() < deadline:
        it += 1
        y = h + m
        corr = fft_corr(y)
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        for t in top:
            shift = int(t - (N - 1))
            if shift == 0:
                grad += (1.0 - 2.0 * y)
            elif shift > 0:
                grad[shift:] += (1.0 - y[:N - shift])
                grad[:N - shift] -= y[shift:]
            else:
                M = -shift
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:M]
        grad /= k_top
        m = beta * m + (1.0 - beta) * grad
        h = project_sum(y - lr * m, target)
        cval = compute_c5(h, N)
        if cval < best_val:
            best_val = cval
            best_h = h.copy()
        if it % 400 == 0:
            lr *= 0.94
            beta = min(0.96, 0.88 + it * 0.00003)
    return best_h, best_val
