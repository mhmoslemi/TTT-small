def exact_swap_search(h, N, deadline, n_sweeps=2000, sample_size=100):
    """Binary swap optimization with exact evaluation."""
    h = h.copy()
    target = N // 2
    # Ensure exactly N/2 ones
    n_ones = int(h.sum())
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if n_ones > target:
        remove = np.random.choice(ones, n_ones - target, replace=False)
        h[remove] = 0
        ones = ones[~np.isin(ones, remove)]
    elif n_ones < target:
        add = np.random.choice(zeros, target - n_ones, replace=False)
        h[add] = 1
        zeros = zeros[~np.isin(zeros, add)]
    
    best_c = np.max(fft_corr(h))
    best_h = h.copy()
    temp = 0.01
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Sample candidates
        p_list = np.random.choice(ones, min(len(ones), sample_size), replace=False)
        q_list = np.random.choice(zeros, min(len(zeros), sample_size), replace=False)
        
        best_delta = 0
        best_pq = None
        curr_c = np.max(fft_corr(h))
        
        for p in p_list:
            for q in q_list:
                h[p], h[q] = 0, 1
                new_c = np.max(fft_corr(h))
                h[p], h[q] = 1, 0  # revert
                delta = new_c - curr_c
                if delta < best_delta:
                    best_delta = delta
                    best_pq = (p, q)
        
        if best_pq is not None:
            h[best_pq[0]], h[best_pq[1]] = 0, 1
            if curr_c + best_delta < best_c:
                best_c = curr_c + best_delta
                best_h = h.copy()
        else:
            # Annealing: accept random swap with some probability
            if np.random.random() < 0.3:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0, 1
            temp *= 0.999
    
    return best_h, best_c

def swap_corr_delta_vec(h, p, q, N):
    center = N - 1
    L = np.arange(-(N-1), N)
    delta = np.zeros(2*N-1)
    
    # p as n: valid when 0 <= p-L < N, i.e., p-N+1 <= L <= p
    # delta contribution: -(1-h[p-L])
    mask1 = (L >= p-N+1) & (L <= p)
    if np.any(mask1):
        delta[mask1] -= (1.0 - h[p - L[mask1]])
    
    # p as n+L: valid when 0 <= p+L < N, i.e., -p <= L <= N-1-p
    mask2 = (L >= -p) & (L <= N-1-p)
    if np.any(mask2):
        delta[mask2] += h[p + L[mask2]]
    
    # q as n: valid when 0 <= q-L < N
    mask3 = (L >= q-N+1) & (L <= q)
    if np.any(mask3):
        delta[mask3] += (1.0 - h[q - L[mask3]])
    
    # q as n+L: valid when 0 <= q+L < N
    mask4 = (L >= -q) & (L <= N-1-q)
    if np.any(mask4):
        delta[mask4] -= h[q + L[mask4]]
    
    return delta

def swap_delta_vec(h, p, q, N):
    """Vectorized swap delta computation."""
    L = np.arange(-(N-1), N)
    delta = np.zeros(2*N-1)
    m1 = (L >= p-N+1) & (L <= p)
