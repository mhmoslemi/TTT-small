            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-14:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    return h, best_c

def try_N(h_best, N_best, N_try, deadline, sub_time=60):
    x_best = np.linspace(0, 2, len(h_best), endpoint=False)
    x_new = np.linspace(0, 2, N_try, endpoint=False)
    h_interp = np.interp(x_new, x_best, h_best)
    h_bt = (h_interp > 0.5).astype(float)
    nh = N_try // 2
    n1 = int(h_bt.sum())
    if n1 > nh:
        idx = np.where(h_bt == 1)[0]
        idx = idx[np.argsort(h_interp[idx])[: n1 - nh]]
        h_bt[idx] = 0
    elif n1 < nh:
        idx = np.where(h_bt == 0)[0]
        idx = idx[np.argsort(-h_interp[idx])[: nh - n1]]
        h_bt[idx] = 1
    dl = min(deadline, time.time() + sub_time)
    h_r, _ = bin_search(h_bt, N_try, dl, max_sweeps=300)
    c5_r = compute_c5(h_r)
    return h_r, c5_r

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Continuous optimization from parent (250s)
    d1 = min(deadline - 700, t0 + 250)
    hc = best_h.copy() + np.random.randn(N0) * 0.01
    hc = project_sum(hc, N0 / 2.0)
    h1, c1 = continuous_opt(hc, N0, d1, k_top=5, lr_init=0.015)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()
    # Also try from parent directly
    h1b, c1b = continuous_opt(best_h.copy(), N0, min(deadline - 650, time.time() + 100), k_top=3, lr_init=0.01)
    if c1b < best_c5:
        best_c5, best_h = c1b, h1b.copy()

    # Phase 2: Binarize + binary search (250s)
    if time.time() < deadline - 400:
        d2 = min(deadline - 350, time.time() + 250)
        hb = (best_h > 0.5).astype(float)
        n1 = int(hb.sum())
        if n1 != N0 // 2:
            oi = np.where(hb == 1)[0]
            zi = np.where(hb == 0)[0]
            hb[:] = 0
            if n1 > N0 // 2:
                hb[oi[np.argsort(-best_h[oi])[:N0//2]]] = 1
            else:
                hb[oi] = 1
                hb[zi[np.argsort(-best_h[zi])[:N0//2-n1]]] = 1
        h2, _ = bin_search(hb, N0, d2, max_sweeps=800)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Continuous from best binary (150s)
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 150)
        h3c = best_h.copy() + np.random.randn(N0) * 0.008
        h3c = project_sum(h3c, N0 / 2.0)
        h3, c3 = continuous_opt(h3c, N0, d3, k_top=4, lr_init=0.01)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Binary from continuous result (150s)
    if time.time() < deadline - 200:
        d4 = min(deadline - 150, time.time() + 150)
        hb4 = (best_h > 0.5).astype(float)
        n1 = int(hb4.sum())
        if n1 != N0 // 2:
            oi = np.where(hb4 == 1)[0]
            zi = np.where(hb4 == 0)[0]
            hb4[:] = 0
            if n1 > N0 // 2:
                hb4[oi[np.argsort(-best_h[oi])[:N0//2]]] = 1
            else:
                hb4[oi] = 1
                hb4[zi[np.argsort(-best_h[zi])[:N0//2-n1]]] = 1
        h4, _ = bin_search(hb4, N0, d4, max_sweeps=500)
        c4 = compute_c5(h4)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Multi-resolution (200s)
    if time.time() < deadline - 100:
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        for N_try in [400, 420, 390, 440, 370]:
            if time.time() > deadline - 40:
                break
            if N_try == len(best_h) or N_try % 2 != 0:
                continue
            sub_dl = min(deadline - 30, time.time() + 35)
            h_rt, c5_rt = try_N(best_h, len(best_h), N_try, sub_dl, sub_time=30)
            if c5_rt < best_c5:
                best_c5, best_h = c5_rt, h_rt.copy()
                x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 6: Annealed binary search on best (100s)
    if time.time() < deadline - 60:
        d6 = min(deadline - 50, time.time() + 100)
        hb6 = (best_h > 0.5).astype(float)
        n1 = int(hb6.sum())
        if n1 != len(best_h) // 2:
            oi = np.where(hb6 == 1)[0]
            zi = np.where(hb6 == 0)[0]
            hb6[:] = 0
            if n1 > len(best_h) // 2:
                hb6[oi[np.argsort(-best_h[oi])[:len(best_h)//2]]] = 1
            else:
                hb6[oi] = 1
                hb6[zi[np.argsort(-best_h[zi])[:len(best_h)//2-n1]]] = 1
        h6, _ = bin_search(hb6, len(best_h), d6, max_sweeps=300, anneal=0.05)
        c6 = compute_c5(h6)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    # Phase 7: Multi-start perturbations (80s)
    if time.time() < deadline - 30:
        d7 = min(deadline - 20, time.time() + 80)
        hb7 = (best_h > 0.5).astype(float)
        Nc = len(hb7)
        for attempt in range(6):
            if time.time() > d7:
                break
            hp = hb7.copy()
            n_flip = np.random.randint(2, 6)
            oi, zi = np.where(hp == 1)[0], np.where(hp == 0)[0]
            for _ in range(n_flip):
                p = int(np.random.randint(len(oi)))
                q = int(np.random.randint(len(zi)))
                hp[oi[p]], hp[zi[q]] = 0, 1
                oi = np.where(hp == 1)[0]
                zi = np.where(hp == 0)[0]
            sub_dl = min(d7, time.time() + 12)
            h_pt, _ = bin_search(hp, Nc, sub_dl, max_sweeps=120)
            c5_pt = compute_c5(h_pt)
            if c5_pt < best_c5:
                best_c5, best_h = c5_pt, h_pt.copy()

    # Phase 8: Final continuous polish (50s)
    if time.time() < deadline - 5:
        hF = best_h.copy() + np.random.randn(len(best_h)) * 0.005
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = continuous_opt(hF, len(best_h), deadline - 2, k_top=3, lr_init=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Final binary snap
    hF2 = (best_h > 0.5).astype(float)
    n1f = int(hF2.sum())
    Nf = len(hF2)
    if n1f != Nf // 2:
        oi = np.where(hF2 == 1)[0]
        zi = np.where(hF2 == 0)[0]
        hF2[:] = 0
        if n1f > Nf // 2:
            hF2[oi[np.argsort(-best_h[oi])[:Nf//2]]] = 1
        else:
            hF2[oi] = 1
            hF2[zi[np.argsort(-best_h[zi])[:Nf//2-n1f]]] = 1
    hF2, _ = bin_search(hF2, Nf, deadline - 0.5, max_sweeps=40)
    c5F2 = compute_c5(hF2)
    if c5F2 < best_c5:
        best_c5, best_h = c5F2, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
