def compute_corr(h):

    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

    dx = 2.0 / N
    return float(np.max(compute_corr(h))) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_step(h, corr, N, k, lr, beta, m):
    """One Nesterov step"""
    y = h + m
    # Recompute corr at y
    corr_y = compute_corr(y)
    # Subgradient: average gradient over top-k lags
    center = N - 1
    if k >= 2 * N - 1:
        top = np.arange(2 * N - 1)
    else:
        top = np.argpartition(corr_y, -k)[-k:]
    
    grad = np.zeros(N)
    for ci in top:
        L_lag = int(ci - center)
        if L_lag == 0:
            grad += (1.0 - 2.0 * y)
        elif L_lag > 0:
            # corr contribution from positions 0..N-1-L_lag: h[i]*(1-h[i+L_lag])
            # d/dh[i] = (1-h[i+L]) for i < N-L, d/dh[i+L] = -h[i] for i < N-L
            n_valid = N - L_lag
            grad[:n_valid] += (1.0 - y[L_lag:N])
            grad[L_lag:N] -= y[:n_valid]
        else:
            M = -L_lag
            n_valid = N - M
            grad[:n_valid] += (1.0 - y[M:N])
            grad[M:N] -= y[:n_valid]
    grad /= len(top)
    
    h_new = project_sum(y - lr * grad, N / 2.0)
    m_new = beta * (h_new - h)
    return h_new, m_new, float(np.max(corr_y))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = c5_val(best_h)

    # Phase 1: Nesterov continuous (80s)
    if time.time() < deadline - 550:
        d1 = min(deadline - 550, t0 + 80)
        h_c = best_h + rng.randn(N0) * 0.04
        h_c = project_sum(h_c, N0 / 2.0)
        m = np.zeros(N0)
        lr, beta = 0.014, 0.88
        for it in range(3000):
            if time.time() > d1:
                break
            h_c, m, pk = nesterov_iter(h_c, N0, top_k=25 + (it % 5) * 10, lr=lr, beta=beta, m=m)
            c5t = pk * (2.0 / N0)
            if c5t < best_c5:
                best_c5 = c5t
                best_h = h_c.copy()
            if it % 120 == 0:
                lr *= 0.97
        # Multiple threshold snaps
        for thresh in [0.35, 0.45, 0.5, 0.55, 0.65]:
            h_t = fix_binary_sum(best_h * 1.0, N0, rng)
            if thresh != 0.5:
                h_t = (best_h > thresh).astype(float)
                h_t = fix_binary_sum(h_t, N0, rng)
            c5t = c5_val(h_t)
            if c5t < best_c5:
                best_c5 = c5t
                best_h = h_t.copy()

    # Phase 2: Greedy binary local search (180s)
    if time.time() < deadline - 350:
        d2 = min(deadline - 350, time.time() + 180)
        h_s, c_s = binary_greedy(best_h, N0, rng, d2, n_sweeps=2000, top_m=12, sample_sz=140)
        c5s = c_s * (2.0 / N0)
        if c5s < best_c5:
            best_c5 = c5s
            best_h = h_s.copy()

    # Phase 3: Simulated annealing (150s)
    if time.time() < deadline - 250:
        d3 = min(deadline - 250, time.time() + 150)
        h_a, c_a = binary_anneal(best_h, N0, rng, d3, T0=0.12, T1=0.002, sample_sz=120)
        c5a = c_a * (2.0 / N0)
        if c5a < best_c5:
            best_c5 = c5a
            best_h = h_a.copy()

    # Phase 4: 2-swap search (100s)
    if time.time() < deadline - 150:
        d4 = min(deadline - 150, time.time() + 100)
        h_2, c_2 = two_swap_search(best_h, N0, rng, d4, n_iters=3000, sample_sz=50)
        c5_2 = c_2 * (2.0 / N0)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h_2.copy()
        # Follow up with greedy
        if time.time() < deadline - 60:
            d4b = min(deadline - 60, time.time() + 50)
            h_2b, c_2b = binary_greedy(best_h, N0, rng, d4b, n_sweeps=1000, top_m=10, sample_sz=120)
            c5_2b = c_2b * (2.0 / N0)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h_2b.copy()

    # Phase 5: Multi-resolution (180s)
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    n_list = [300, 320, 360, 400, 440, 480, 520]
    for N_try in n_list:
        if time.time() > deadline - 80:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_i = np.interp(x_new, x_best, best_h)
        h_b = fix_binary_sum(h_i, N_try, rng)
        sub_d = min(deadline - 60, time.time() + 40)
        h_r, c_r = binary_greedy(h_b, N_try, rng, sub_d, n_sweeps=800, top_m=12, sample_sz=130)
        c5_r = c_r * (2.0 / N_try)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 6: Final Nesterov + binary (80s)
    if time.time() < deadline - 40:
        d6 = min(deadline - 40, time.time() + 60)
        Nf = len(best_h)
        h_f = best_h + rng.randn(Nf) * 0.02
        h_f = project_sum(h_f, Nf / 2.0)
        m = np.zeros(Nf)
        lr, beta = 0.008, 0.90
        for it in range(2000):
            if time.time() > d6:
                break
            h_f, m, pk = nesterov_iter(h_f, Nf, top_k=20 + it % 30, lr=lr, beta=beta, m=m)
            c5t = pk * (2.0 / Nf)
            if c5t < best_c5:
                best_c5 = c5t
                best_h = h_f.copy()
            if it % 80 == 0:
                lr *= 0.96
        h_bin = fix_binary_sum(best_h, Nf, rng)
        c5b = c5_val(h_bin)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h_bin.copy()

    # Phase 7: Final greedy polish
    if time.time() < deadline - 2:
        d7 = deadline - 1
        Nf = len(best_h)
        h_p, c_p = binary_greedy(best_h, Nf, rng, d7, n_sweeps=2000, top_m=8, sample_sz=100)
        c5p = c_p * (2.0 / Nf)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = h_p.copy()

    final_c5 = c5_val(best_h)
    return best_h, final_c5, len(best_h)
