import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, n_ones):
    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    if n1 != n_ones:
        if n1 > n_ones:
            idx = np.where(h_b == 1)[0]
            idx = idx[np.random.permutation(len(idx))]
            h_b[idx[:n1 - n_ones]] = 0
        else:
            idx = np.where(h_b == 0)[0]
            idx = idx[np.random.permutation(len(idx))]
            h_b[idx[:n_ones - n1]] = 1
    return h_b

def grad_c5(corr, h, N, topk=80, temp=0.005):
    """Compute subgradient of max(corr) using softmax over top-k."""
    # corr has length 2N-1
    k = min(topk, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    w = np.exp((corr[idx] - corr[idx].max()) / max(temp, 1e-12))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for wi, ci in zip(w, idx):
        L = int(ci - center)  # lag
        if L == 0:
            # c_0 = sum h[i]*(1-h[i]) = 0 for binary, but for general:
            grad += wi * (1.0 - 2.0 * h) * (2.0 / N)
        elif L > 0:
            seg = N - L
            # c_L = sum_{m=0}^{N-1-L} h[m]*(1-h[m+L])*dx
            # d/dh[i]: for m=i: (1-h[i+L])*dx; for m+L=i i.e. m=i-L: -h[i-L]*dx
            grad[:seg] += wi * (1.0 - h[L:]) * (2.0/N)  # first term: (1-h[i+L])
            grad[L:] -= wi * h[:seg] * (2.0/N)  # second term: -h[i-L]
        else:
            M = -L
            if M < N:
                seg = N - M
                # c_{-M} = sum_{m=M}^{N-1} h[m]*(1-h[m-M])*dx
                # = sum_{m=0}^{N-1-M} h[m+M]*(1-h[m])*dx
                # d/dh[i]: for m+M=i i.e. m=i-M: (1-h[i-M])*dx; for m=i: -h[i+M]*dx
                grad[M:] += wi * (1.0 - h[:seg]) * (2.0/N)
                grad[:seg] -= wi * h[M:] * (2.0/N)
    return grad

def subgradient_opt(h0, N, deadline, topk=80, temp0=0.003, lr0=0.05):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    corr = fft_corr(h)
    best_val = float(np.max(corr))
    lr = lr0
    temp = temp0
    beta = 0.9
    h_prev = h.copy()
    stall = 0
    while time.time() < deadline:
        # Nesterov step
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_val - 1e-14:
            best_val = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        g = grad_c5(corr, y, N, topk, temp)
        h = project_sum(y - lr * g, target)
        if stall % 300 == 300:
            lr = max(0.005, lr * 0.98)
            temp = max(0.001, temp * 0.99)
        if stall > 500:
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)
            h_prev = h.copy()
            lr = lr0 * 0.8
            stall = 0
    return best_h, best_val * (2.0 / N)

def binary_opt2(h0, N, deadline, n_candidates=40, restart_after=30):
    """Binary local search with relocating moves."""
    nh = N // 2
    h = make_binary(h0, nh)
    c = fft_corr(h)
    best_int = int(np.max(c))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_move_int = best_int
        best_move = None
        # Try n_candidates random (p,q) pairs
        nc = min(n_candidates, len(ones) * len(zeros))
        ps = ones[np.random.randint(len(ones), nc)]
        qs = zeros[np.random.randint(len(zeros), nc)]
        for p, q in zip(ps, qs):
            if p == q:
                continue
            h[p] = 0
            h[q] = 1
            ci = int(fft_corr(h).max())
            if ci < best_move_int:
                best_move_int = ci
                best_move = (p, q)
            h[p] = 1
            h[q] = 0
        if best_move is not None:
            p, q = best_move
            h[p] = 0
            h[q] = 1
            c = fft_corr(h)
            if best_move_int < best_int:
                best_int = best_move_int
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= restart_after:
                # Random restart: flip a few
                nf = max(3, N // 50)
                fi = ones[np.random.choice(len(ones), min(nf, len(ones)), replace=False)]
                fj = zeros[np.random.choice(len(zeros), min(nf, len(zeros)), replace=False)]
                h[fi] = 0
                h[fj] = 1
                c = fft_corr(h)
                stall = 0
    return best_h, best_int * (2.0 / N)

def interp_to(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)
