import numpy as np

# Helper functions (top-level, no closures)

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.fft.irfft(H*G, L)[:2*N-1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s-target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0.0, 1.0)

def to_binary(h, N):
    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    tgt = N // 2
    if n1 != tgt:
        oi = np.where(h_b==1)[0]
        zi = np.where(h_b==0)[0]
        h_b[:] = 0
        if n1 > tgt:
            h_b[oi] = 1
            h_b[oi[np.argsort(h[oi])[:n1-tgt]]] = 0
        else:
            h_b[oi] = 1
            h_b[zi[np.argsort(-h[zi])[:tgt-n1]]] = 1
    return h_b
