    """Compute gradient of smoothed max of correlation."""
    corr = fft_corr(h)
    # Softmax weights
    top_idx = np.argpartition(corr, -k)[-k:]
    top_corr = corr[top_idx]
    w = np.exp((top_corr - top_corr.max()) / temp)
    w /= w.sum()
    
    center = N - 1
    grad = np.zeros(N)
    # Use the D matrix concept but with weights
    # grad = sum_k w_k * d(corr_k)/dh
    # d(corr_L)/dh[p] = (1-h[p+L])*I(0<=p+L<N) - h[p-L]*I(0<=p-L<N)
    shifts = np.arange(2*N-1) - center
    # For efficiency, only use top-k shifts
    top_shifts = shifts[top_idx]
    
    for i, L in enumerate(top_shifts):
        wi = w[i]
        L = int(L)
        if L >= 0:
            # h[p] as first element: p+L < N, i.e., p < N-L
            # d(corr_L)/dh[p] += (1-h[p+L]) for p in [0, N-L-1]
            # h[p] as second element: p-L >= 0, i.e., p >= L
            # d(corr_L)/dh[p] -= h[p-L] for p in [L, N-1]
            if N - L > 0:
                grad[:N-L] += wi * (1.0 - h[L:])
            if L < N:
                grad[L:] -= wi * h[:N-L]
        else:
            M = -L
            if M < N:
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
    
    return grad, corr, float(corr.max())

    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    hm = np.where(vm, h[np.clip(pm, 0, N-1)], 0.0)
    hp = np.where(vp, h[np.clip(pp, 0, N-1)], 0.0)
    return -(1.0-hm)*vm.astype(float) + hp*vp.astype(float)

def topk_sum(arr, k=8):
    """Sum of top k values along last axis."""
    # arr shape: (n_candidates, L)
    top = np.partition(arr, -k, axis=1)[:, -k:]
    return top.sum(axis=1)

def binary_search_v3(h, N, deadline, max_sweeps=5000, sample_n=180,
                     topk=8, kick_every=200, kick_n=2):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_p = float(np.max(fft_corr(h)))
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_p - 1e-13:
            best_p = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        D = compute_D(h, N)
        
        if len(ones) > sample_n:
            perm = np.random.permutation(len(ones))[:sample_n]
            p_list = ones[perm]
        else:
            p_list = ones
        
        best_score = P * topk + 1.0
        best_pq = None
        
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]  # shape: (n_zeros, L)
            # Score: combination of max and topk-sum
            npk = cand.max(axis=1)
            ntk = topk_sum(cand, topk)
            score = npk + 0.3 * ntk  # weight the sum of top-k
            mi = int(np.argmin(score))
            if float(score[mi]) < best_score:
                best_score = float(score[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        elif no_improve >= kick_every:
            for _ in range(kick_n):
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
            no_improve = 0
        
        if no_improve > 1000:
            break
    
    return h, best_p

    corr = fft_corr(h)
    # Softmax weights on top-k
    top_idx = np.argpartition(corr, -k)[-k:]
    top_corr = corr[top_idx]

        wi = float(w[i])
        # First element contribution: (1-h[p+L]) for p where 0<=p+L<N
