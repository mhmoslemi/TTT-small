    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    """Project h to [0,1] with sum = target using binary search on threshold."""
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def lse_objective_and_grad(h, gamma):
    """Compute LSE(γ * corr) and its gradient."""
    N = len(h)
    corr = fft_corr(h)
    mc = np.max(corr)
    exps = np.exp(gamma * (corr - mc))
    w = exps / exps.sum()
    lse_val = mc + np.log(exps.sum()) / gamma
    # Gradient: sum_m w[m] * ((1-h)[i+k_m] - h[i-k_m])
    ks = np.arange(2 * N - 1) - (N - 1)
    i_idx = np.arange(N)[:, None]
    pos = i_idx + ks[None, :]
    neg = i_idx - ks[None, :]
    pos_valid = (pos >= 0) & (pos < N)
    neg_valid = (neg >= 0) & (neg < N)
    pos_c = np.clip(pos, 0, N - 1)
    neg_c = np.clip(neg, 0, N - 1)

    term2 = np.sum(w[None, :] * h[neg_c] * neg_valid, axis=1)

        P = float(np.max(corr))
        if P * (2.0 / N) < best_c5:
            best_c5 = P * (2.0 / N)
            best_h = h.copy()
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for m in idx:
            k = int(m) - (N - 1)
            ii = np.arange(N)
            pp = ii + k
            nn = ii - k
            vp = (pp >= 0) & (pp < N)
            vn = (nn >= 0) & (nn < N)
            grad[vp] += (1.0 - h[pp[vp]]) / top_k
            grad[vn] -= h[nn[vn]] / top_k
        h = project_sum(h - lr * grad, target)
        it += 1
        if it % 200 == 0:
            lr *= 0.97
    c5 = compute_c5(h, N)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def binary_il(h, N, deadline, max_iters=500, sample_sz=50):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    best_c = compute_c5(h, N)
    best_h = h.copy()
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P * (2.0 / N) < best_c - 1e-10:
            best_c = P * (2.0 / N)
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        best_mv = P
        best_pair = None
        for a in os_:
            for b in zs:
                h[a], h[b] = 0, 1
                nv = float(np.max(fft_corr(h)))
                h[a], h[b] = 1, 0
                if nv < best_mv:
                    best_mv = nv
                    best_pair = (a, b)
        if best_pair is not None:
            h[best_pair[0]] = 0
            h[best_pair[1]] = 1
        else:
            if stuck > 30:
                break
            a = int(np.random.choice(ones))
            b = int(np.random.choice(zeros))
            h[a], h[b] = 0, 1
            nv = float(np.max(fft_corr(h)))
            h[a], h[b] = 1, 0
            if nv < P + 0.3:
                h[a], h[b] = 0, 1
    c5 = compute_c5(h, N)
    if c5 < best_c:
        best_c, best_h = c5, h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Nesterov LSE - high gamma
    if time.time() < deadline - 350:
        d1 = min(deadline - 340, t0 + 120)
        hc = best_h + np.random.randn(N0) * 0.08
        hc = project_sum(hc, N0 / 2.0)
        h1, c5_1 = nesterov_lse(hc, N0, d1, gamma=150, lr=0.015, beta=0.92)
        if c5_1 < best_c5:
            best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Subgradient (faster, multi-lag)
    if time.time() < deadline - 280:
        d2 = min(deadline - 270, t0 + 200)
        h2, c5_2 = subgrad_search(best_h, N0, d2, top_k=20, lr=0.04)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Nesterov LSE - low gamma (balance all lags)
    if time.time() < deadline - 200:
        d3 = min(deadline - 190, time.time() + 100)
        hc3 = best_h + np.random.randn(N0) * 0.05
        hc3 = project_sum(hc3, N0 / 2.0)
        h3, c5_3 = nesterov_lse(hc3, N0, d3, gamma=30, lr=0.008, beta=0.94)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Binary local search
    if time.time() < deadline - 120:
        hb = (best_h > 0.5).astype(float)
        d4 = min(deadline - 110, time.time() + 80)
        h4, c5_4 = binary_il(hb, N0, d4, max_iters=400, sample_sz=45)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Continue LSE from best (moderate gamma)
    if time.time() < deadline - 80:
        d5 = min(deadline - 70, time.time() + 60)
        hc5 = best_h.copy()
        h5, c5_5 = nesterov_lse(hc5, N0, d5, gamma=60, lr=0.006, beta=0.93)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Multi-resolution
    for N_try in [220, 300, 420, 500]:
        if time.time() > deadline - 40:
            break
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_best, best_h)
        h_int = project_sum(h_int, N_try / 2.0)
        sub_dl = min(deadline - 15, time.time() + 45)
        h_rt, c5_rt = nesterov_lse(h_int, N_try, sub_dl, gamma=80, lr=0.01, beta=0.91)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Final: try both continuous and binary
    if time.time() < deadline - 3:
        hb_fin = (best_h > 0.5).astype(float)
        dh = len(best_h) // 2
        n1 = int(hb_fin.sum())
        if n1 > dh:
            idx = np.where(hb_fin == 1)[0]
            hb_fin[np.random.choice(idx, n1 - dh, replace=False)] = 0
        elif n1 < dh:
            idx = np.where(hb_fin == 0)[0]
            hb_fin[np.random.choice(idx, dh - n1, replace=False)] = 1
        c5_bin = compute_c5(hb_fin, len(best_h))
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, hb_fin

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
