    """Find the best (one→0, zero→1) swap. Returns (i, j, new_max_value)"""
    best_val = np.inf
    best_i, best_j = -1, -1
    for i in ones:
        # new_corr for swapping i (1→0) and each zero j (0→1)
        # new_corr = corr + C[j] - C[i] (j goes 0→1 adds C[j], i goes 1→0 removes C[i])
        new_corr = corr[None, :] + C[zeros, :] - C[i:i+1, :]  # shape (n_zeros, 2N-1)
        new_max = new_corr.max(axis=1)  # shape (n_zeros,)
        j_idx = np.argmin(new_max)
        if new_max[j_idx] < best_val:
            best_val = new_max[j_idx]
            best_i, best_j = i, zeros[j_idx]
    return best_i, best_j, best_val

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_C_mat(h, N):
    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def full_pair_local_search(h, deadline, max_iters=500, wl=0.1, use_top3=False):
    """Full-pair binary local search. h is binary with N/2 ones."""
    N = len(h)
    h = h.astype(float).copy()
    best_c = np.inf
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c:
            best_c = P
            best_h = h.copy()
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        C = compute_C_mat(h, N)
        
        # Full pair search
        best_val = np.inf
        best_i, best_j = -1, -1
        for i in ones:
            new_corr = corr[None, :] + C[zeros, :] - C[i:i+1, :]
            if use_top3:
                top3 = np.partition(new_corr, -3, axis=1)[:, -3:]
                obj = top3[:, -1] + wl * top3[:, -2] + wl*0.5 * top3[:, -3]
            else:
                obj = new_corr.max(axis=1)
            j_idx = int(np.argmin(obj))
            if obj[j_idx] < best_val:
                best_val = obj[j_idx]
                best_i, best_j = i, zeros[j_idx]
        
        if best_i < 0 or best_val >= P - 1e-12:
            break
        
        h[best_i] = 0
        h[best_j] = 1
    
    return best_h, best_c * (2.0 / N)

def ils_perturb(h, block_len=8):
    """Perturb by flipping a random block (maintaining sum constraint)."""
    N = len(h)
    h = h.copy()
    # Find a random starting position and flip a block
    # To maintain sum: flip k ones to 0 and k zeros to 1
    half_bl = block_len // 2
    start = np.random.randint(0, N - block_len)
    # In the block, flip some 1s to 0s and some 0s to 1s
    block = h[start:start+block_len]
    ones_in_block = np.where(block == 1)[0]
    zeros_in_block = np.where(block == 0)[0]
    n_flip = min(half_bl, len(ones_in_block), len(zeros_in_block))
    if n_flip > 0:
        flip_o = np.random.choice(ones_in_block, n_flip, replace=False)
        flip_z = np.random.choice(zeros_in_block, n_flip, replace=False)
        for idx in flip_o:
            h[start + idx] = 0
        for idx in flip_z:
            h[start + idx] = 1
    return h
