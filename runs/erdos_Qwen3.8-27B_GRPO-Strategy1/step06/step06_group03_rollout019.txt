import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G, L))[:L]

def compute_c5(h, N):
    dx = 2.0 / N
    return float(np.max(fft_corr(h))) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_weights(corr, beta):
    # Softmax of correlation values for log-sum-exp gradient
    shifted = corr - corr.max()
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    return weights

def compute_grad_lse(h, weights):
    """Gradient of log-sum-exp objective."""
    N = len(h)
    grad = np.zeros(N)
    center = N - 1
    for m in range(2 * N - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        L = m - center  # lag
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            grad[:N - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:N - L]
        elif L < 0 and -L < N:
            M = -L
            grad[M:] += w * (1.0 - h[:N - M])
            grad[:N - M] -= w * h[M:]
    return grad

def lse_optimize(h, N, deadline, beta0=5.0, lr0=0.01, max_iter=2000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)  # momentum
    beta = beta0
    lr = lr0
    mu = 0.9
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_val:
            best_val = pk
            best_h = h.copy()
        weights = softmax_weights(corr, beta)
        grad = compute_grad_lse(h, weights)
        h_new = project_sum(h - lr * grad, target)
        m = mu * m + (1 - mu) * (h_new - h)
        h = project_sum(h + m, target)
        # Decay learning rate and increase beta
        if it % 200 == 0 and it > 0:
            lr *= 0.95
            beta *= 1.05
    return best_h, best_val * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G, L))[:L]

def compute_c5(h, N=None):
    if N is None:
        N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        order = np.argsort(np.random.random(len(idx)))
        h[idx[order[nh:]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        order = np.argsort(np.random.random(len(idx)))
        h[idx[order[:nh - n1]]] = 1
    return h

def lse_gradient(h, beta):
    N = len(h)
    c = fft_corr(h)
    shifted = c - c.max()
    exps = np.exp(beta * shifted)
    w = exps / exps.sum()
    grad = np.zeros(N)
    center = N - 1
    for m in range(2 * N - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        L = m - center
        if L == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wm * (1.0 - h[:N - L])
            grad[:N - L] -= wm * h[L:]
        elif -N < L < 0:
            M = -L
            grad[:N - M] += wm * (1.0 - h[M:])
            grad[M:] -= wm * h[:N - M]
    return grad

def lse_opt(h, N, deadline, beta0=8.0, lr0=0.012, mu=0.9):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    mom = np.zeros(N)
    beta = beta0
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + mom
        c = fft_corr(y)
        pk = float(c.max())
        if pk < best_val:
            best_val = pk
            best_h = h.copy()
        grad = lse_gradient(y, beta)
        h_new = project_sum(h - lr * grad, target)
        mom = mu * mom + (1.0 - mu) * (h_new - h)
        h = project_sum(h + mom, target)
        it += 1
        if it % 150 == 0:
            lr *= 0.96
            beta = min(50.0, beta * 1.04)
    return best_h, best_val * (2.0 / N)

def vec_swap_eval(c, h, p, q_arr, top_idx, N, non_top_max):
    """Vectorized: for fixed p, evaluate all q in q_arr at top positions."""
    cm = (top_idx - (N - 1)).astype(int)
    nm = len(top_idx)
    nq = len(q_arr)
    pp = p + cm
    pm = p - cm
    dp = np.zeros(nm)
    vpp = (pp >= 0) & (pp < N)
    vpm = (pm >= 0) & (pm < N)
    dp[vpp] += h[np.clip(pp[vpp], 0, N - 1)]
    dp[vpm] += h[np.clip(pm[vpm], 0, N - 1)]
    dp -= 1.0
    qm = q_arr[:, None] - cm[None, :]
    qp = q_arr[:, None] + cm[None, :]
    vqm = (qm >= 0) & (qm < N)
    vqp = (qp >= 0) & (qp < N)
    hqm = np.where(vqm, h[np.clip(qm, 0, N - 1)], 0.0)
    hqp = np.where(vqp, h[np.clip(qp, 0, N - 1)], 0.0)
    total = c[top_idx][None, :] + dp[None, :] + 1.0 - hqm - hqp
    return np.maximum(total.max(axis=1), non_top_max)

def local_search(h, N, deadline, top_m=25, samp=60, max_sw=2000, anneal=0.08):
    h = make_binary(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for sweep in range(max_sw):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        b_ones = np.array([i for i in ones if (i > 0 and h[i - 1] == 0) or (i < N - 1 and h[i + 1] == 0)])
        if len(b_ones) < 4:
            b_ones = ones
        b_zeros = np.array([i for i in zeros if (i > 0 and h[i - 1] == 1) or (i < N - 1 and h[i + 1] == 1)])
        if len(b_zeros) < 4:
            b_zeros = zeros
        np_ = min(samp, len(b_ones))
        nq = min(samp, len(b_zeros))
        p_list = np.random.choice(b_ones, np_, replace=False)
        q_list = np.random.choice(b_zeros, nq, replace=False)
        top_idx = np.argsort(c)[-top_m:]
        nt_mask = np.ones(len(c), dtype=bool)
        nt_mask[top_idx] = False
        ntm = c[nt_mask].max() if nt_mask.any() else 0.0
        best_new = P
        best_mv = None
        for p in p_list:
            candidates = vec_swap_eval(c, h, int(p), q_list, top_idx, N, ntm)
            bi = int(np.argmin(candidates))
            if candidates[bi] < best_new:
                best_new = candidates[bi]
                best_mv = (int(p), int(q_list[bi]))
        if best_mv is not None:
            h[best_mv[0]] = 0.0
            h[best_mv[1]] = 1.0
        elif np.random.random() < anneal:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p], h[q] = 0.0, 1.0
    return makes_binary(h, N), best_c
