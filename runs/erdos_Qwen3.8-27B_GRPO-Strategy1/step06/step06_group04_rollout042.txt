    return delta

def corr_delta(h, p, q, m, N):
    """Delta in correlation at lag m for swap h[p]:1→0, h[q]:0→1"""
    center = N - 1
    delta = 0
    seen = set()
    
    # Check pair (i=p, j=p+m-center)
    j = p + m - center
    if 0 <= j < N and (p, j) not in seen:
        seen.add((p, j))
        # New: h'[p]*(1-h')[j] = 0 * (1 - h'[j])
        # Old: h[p]*(1-h)[j] = 1 * (1-h[j])
        # h'[j] = h[j] unless j==q (then 1) or j==p (then 0, but j=p means m=center)
        h_j_new = 1.0 if j == q else (0.0 if j == p else h[j])
        h_j_old = h[j]
        delta += 0.0 * (1.0 - h_j_new) - 1.0 * (1.0 - h_j_old)
    
    # Check pair (i=p-m+center, j=p)
    i = p - m + center
    if 0 <= i < N and (i, p) not in seen:
        seen.add((i, p))
        # New: h'[i]*(1-h')[p] = h'[i] * 1
        # Old: h[i]*(1-h)[p] = h[i] * 0 = 0
        h_i_new = 1.0 if i == q else (0.0 if i == p else h[i])
        delta += h_i_new * 1.0 - h[i] * 0.0
    
    # Check pair (i=q, j=q+m-center)
    j = q + m - center
    if 0 <= j < N and (q, j) not in seen:
        seen.add((q, j))
        # New: h'[q]*(1-h')[j] = 1 * (1-h'[j])
        # Old: h[q]*(1-h)[j] = 0 * ... = 0
        h_j_new = 1.0 if j == q else (0.0 if j == p else h[j])
        delta += 1.0 * (1.0 - h_j_new) - 0.0
    
    # Check pair (i=q-m+center, j=q)
    i = q - m + center
    if 0 <= i < N and (i, q) not in seen:
        seen.add((i, q))
        # New: h'[i]*(1-h')[q] = h'[i] * 0 = 0
        # Old: h[i]*(1-h)[q] = h[i] * 1 = h[i]
        h_i_new = 1.0 if i == q else (0.0 if i == p else h[i])
        delta += 0.0 - h[i] * 1.0
    
    return delta
