def fft_corr(h):
    N = len(h)

    H = np.fft.fft(h, L)

    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def build_A(h):
    N = len(h)
    center = N - 1
    L = 2 * N - 1
    i = np.arange(N)[:, None]
    lags = np.arange(L) - center  # (L,)
    # A[i, L_idx] = (1-h[i+lag])*[valid] - h[i-lag]*[valid]
    j = i + lags[None, :]  # (N, L) positions for (1-h[.])
    k = i - lags[None, :]  # (N, L) positions for -h[.]
    valid_j = (j >= 0) & (j < N)
    valid_k = (k >= 0) & (k < N)
    # Safe indexing
    j_clipped = np.clip(j, 0, N-1)
    k_clipped = np.clip(k, 0, N-1)
    A = np.where(valid_j, 1.0 - h[j_clipped], 0.0) - np.where(valid_k, h[k_clipped], 0.0)
    return A

def update_A_single(A, pos, diff, N):
    center = N - 1
    L = 2 * N - 1
    i_range = np.arange(N)
    idx1 = pos - i_range + center
    v1 = (idx1 >= 0) & (idx1 < L)
    A[i_range[v1], idx1[v1]] -= diff
    idx2 = i_range - pos + center
    v2 = (idx2 >= 0) & (idx2 < L)
    A[i_range[v2], idx2[v2]] -= diff

def binary_best_swap(h0, N, deadline):
    h = h0.astype(float).copy()
    if h.sum() != N // 2:
        h = project_sum(h, N / 2.0)
        h = (h > 0.5).astype(float)
        # Fix count
        nh = N // 2
        n1 = int(h.sum())
        if n1 > nh:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:n1-nh]] = 0
        elif n1 < nh:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:nh-n1]] = 1
    A = build_A(h)
    c = fft_corr(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    zero_arr = np.array(zeros)
    cur_max = float(c.max())
    best_h = h.copy()
    best_val = cur_max
    stall = 0
    
    while time.time() < deadline:
        best_new_max = cur_max
        best_p = best_q = -1
        best_delta = None
        
        for p in ones:
            base = -A[p]
            sub_A = A[zero_arr, :]
            new_c = c[None, :] + base[None, :] + sub_A
            new_maxes = new_c.max(axis=1)
            idx = np.argmin(new_maxes)
            if new_maxes[idx] < best_new_max - 1e-14:
                best_new_max = new_maxes[idx]
                best_p = p
                best_q = int(zero_arr[idx])
                best_delta = base + A[best_q]
        
        if best_p == -1:
            stall += 1
            if stall > 30:
                break
            # Random perturbation
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
        else:
            stall = 0
        
        # Apply swap
        c = c + (-A[best_p] + A[best_q]) if best_p != -1 else c
        h[best_p] = 0
        h[best_q] = 1
        update_A_single(A, best_p, -1.0, N)
        update_A_single(A, best_q, 1.0, N)
        
        ones.remove(best_p)
        zeros.remove(best_q)
        ones.append(best_q)
        zeros.append(best_p)
        zeros = list(zeros)
        zero_arr = np.array(zeros)
        
        cur_max = float(c.max())
        if cur_max < best_val - 1e-14:
            best_val = cur_max
            best_h = h.copy()
        
        # Periodic FFT verification
        if len(ones) % 500 == 0:
            c_check = fft_corr(h)
            if abs(c_check.max() - c.max()) > 1e-8:
                c = c_check
                A = build_A(h)
    
    return best_h, best_val * (2.0 / N)

def continuous_opt(h0, N, deadline, beta0=50.0, lr0=0.01):
    h = h0.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    dx = 2.0 / N
    A = build_A(h)
    c = fft_corr(h)
    best_h = h.copy()
    best_c5 = float(c.max()) * dx
    h_mom = h.copy()
    lr = lr0
    beta = beta0
    stall = 0
    it = 0
    
    while time.time() < deadline:
        # Compute smoothed gradient
        bc = beta * c * dx
        bc -= bc.max()
        w = np.exp(bc)
        w /= w.sum()
        grad = dx * (A.T @ w)
        
        # Nesterov extrapolation
        y = h + 0.8 * (h - h_mom)
        h_mom = h
        
        # Update A for y (approximate, or just use h's A)
        # For simplicity, use h's A (small error when step is small)
        
        # Projected gradient step
        h_new = y - lr * grad
        h_new = project_sum(h_new, target)
        
        # Compute new c and objective
        c = fft_corr(h_new)
        new_c5 = float(c.max()) * dx
        
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        h = h_new
        
        # Update A (full rebuild for simplicity, or incremental)
        # Actually A depends on h, and h changed significantly, so rebuild
        # But this is O(N^2) which is fine
        if it % 10 == 0:
            A = build_A(h)
        
        it += 1
        if it % 500 == 0:
            lr = max(0.001, lr * 0.99)
            beta = min(500.0, beta * 1.02)
            if stall > 200:
                # Restart
                h = best_h + np.random.randn(N) * 0.03
                h = project_sum(h, target)
                h_mom = h.copy()
                A = build_A(h)
                c = fft_corr(h)
                lr = lr0 * 0.7
                stall = 0
    
    return best_h, best_c5

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new
