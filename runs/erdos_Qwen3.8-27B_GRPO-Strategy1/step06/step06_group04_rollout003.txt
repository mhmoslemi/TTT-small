import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_gradient(h, N, top_k, temp):
    """Compute gradient of log-sum-exp of correlation."""
    corr = fft_corr(h)
    # top_k weights
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c_top = corr[idx]
    w = np.exp((c_top - c_top.max()) / temp)
    w /= w.sum()

    for wi, ai in zip(w, idx):
        s = int(ai) - center  # shift
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            # corr contribution: sum_{n=s}^{N-1} h[n]*(1-h[n-s])

            grad[s:] -= wi * h[:seg1]           # -h[i-s] for i=s..N-1
        else:

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

        else:

def lse_opt(h0, N, deadline, T0=0.08, Tend=0.002, top_k=80, lr0=0.03, beta0=0.9):

    lr = lr0
    beta = beta0

        prev = h
        
        corr = fft_corr(y)

            best_h = y.copy()
            stall = 0

def lse_grad(h, corr, top_k, T):
    N = len(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c_top = corr[idx]
    w = np.exp((c_top - c_top.max()) / T)
    w /= w.sum()
    
    two_h = 2.0 * h - 1.0  # = -(1-2h)
    grad = np.zeros(N)
    
    for wi, m in zip(w, idx):
        m = int(m)
        lo = max(0, m - N + 1)
        hi = min(N - 1, m)
        # grad[lo:hi+1] += wi * (1 - 2*h[m-hi : m-lo : -1])
        # = wi * (1 - 2*h[lo to m reversed... )
        # h[m-i] for i=lo..hi gives h[m-lo] down to h[m-hi]
        seg = h[m-hi : m-lo+1 : -1] if m-hi > m-lo else h[m-lo+1 : m-hi-1 : -1]
        # Actually: i goes from lo to hi, m-i goes from m-lo to m-hi
        # So we need h[m-lo], h[m-lo-1], ..., h[m-hi]
        # That's h[m-hi : m-lo+1][::-1]
        seg = h[m-hi : m-lo+1][::-1]
        grad[lo:hi+1] += wi * (1.0 - 2.0 * seg)
    
    return grad
