import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    full = np.fft.irfft(H * G)
    return full[:2*N-1]

def proj_binary(h, N):
    """Project to binary with exactly N/2 ones"""
    h = h.copy()
    n1 = int(np.sum(h > 0.5))
    target = N // 2
    if n1 == target:
        return (h > 0.5).astype(float)
    scores = h.copy()
    if n1 > target:
        # Need to flip some 1s to 0s: pick the lowest-scoring 1s
        ones_idx = np.where(h > 0.5)[0]
        to_flip = len(ones_idx) - target
        scores[ones_idx] = scores[ones_idx]  # keep as is
        flip_candidates = np.argsort(scores[ones_idx])[:to_flip]
        h[ones_idx[flip_candidates]] = 0
    else:
        zeros_idx = np.where(h <= 0.5)[0]
        to_flip = target - len(zeros_idx)  # wait, target - n1
        to_flip = target - n1
        flip_candidates = np.argsort(-scores[zeros_idx])[:to_flip]
        h[zeros_idx[flip_candidates]] = 1
    return h

def compute_corr_delta_single(h, N, p, is_one=True):
    """Compute delta in correlation for single position change"""
    s = np.arange(-(N-1), N)
    # p going from one state to other
    ps = p + s  # i = p, shift = s: h[p]*(1-h[p+s])
    pm = p - s  # i+s = p, i = p-s: h[p-s]*(1-h[p])
    
    d = np.zeros(2*N-1)
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    
    pc = np.clip(ps, 0, N-1)
    pmc = np.clip(pm, 0, N-1)
    
    if is_one:
        # p: 1 -> 0
        d[vps] -= (1.0 - h[pc[vps]])
        d[vpm] += h[pmc[vpm]]
    else:
        # p: 0 -> 1
        d[vps] += (1.0 - h[pc[vps]])
        d[vpm] -= h[pmc[vpm]]
    return d

    current_pk = best_pk

    T = T0

    while time.time() < deadline and T > 0.002:
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        
        ps = p + s_arr; pm = p - s_arr
        qs = q + s_arr; qm = q - s_arr
        vps = (ps >= 0) & (ps < N); vpm = (pm >= 0) & (pm < N)
        vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)
        
        d = np.zeros(2*N-1)

def _fix_count(h, N):
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        perm = idx[np.random.permutation(len(idx))][:n1-target]
        h[perm] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        perm = idx[np.random.permutation(len(idx))][:target-n1]
        h[perm] = 1
    return h

def continuous_refine(h, N, deadline, top_k=20, lr=0.002):
    """ADAM continuous optimization focusing on top-k correlations"""
    h = h.astype(float).copy()
    h = _proj_sum(h, N / 2.0)
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    center = N - 1
    
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-10:
            best_pk = pk; best_h = h.copy()
        
        # Top-k indices
        idx = np.argpartition(corr, -top_k)[-top_k:]
        
        # Gradient from top-k
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0 and s < N:
                grad[:N-s] += (1.0 - h[s:])
                grad[s:] -= h[:N-s]
            elif s < 0 and -s < N:
                s2 = -s
                grad[s2:] += (1.0 - h[:N-s2])
                grad[:N-s2] -= h[s2:]
        grad /= top_k
        
        # ADAM
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = _proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        
        if t % 200 == 0:
            lr *= 0.98
    
    return best_h, best_pk

def _proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -3.0, 3.0

        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)
