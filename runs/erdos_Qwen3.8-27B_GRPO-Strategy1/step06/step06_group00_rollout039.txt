def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, N, rng):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:n1 - nh]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:nh - n1]]] = 1
    return h

def project_01_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5(h, N, k_top=20):
    corr = fft_corr(h)
    top = np.argpartition(corr, -k_top)[-k_top:]
    grad = np.zeros(N)
    center = N - 1
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]
    return grad / k_top, corr

def nesterov_opt(h, N, deadline, k=30, lr0=0.03, max_iters=200000):

            break
        grad, _ = grad_c5(y, N, k)
        h_new = project_01_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        y = h + m
        pk = float(np.max(fft_corr(h)))

            best_h = h.copy()

def fourier_kick(h, N, rng, sigma=0.3):
    H = np.fft.rfft(h)
    n = len(H)
    noise = rng.randn(n) + 1j * rng.randn(n)
    freqs = np.arange(n) / n
    w = sigma * (0.1 + 0.9 * freqs)
    noise[0] = 0
    Hn = H + noise * w

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, N, rng):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:n1 - nh]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:nh - n1]]] = 1
    return h

def project_01_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5(h, N, k_top=20):
    corr = fft_corr(h)
    top = np.argpartition(corr, -k_top)[-k_top:]
    grad = np.zeros(N)
    center = N - 1
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]
    return grad / k_top, corr

def nesterov_opt(h, N, deadline, k=30, lr0=0.04, max_iters=200000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    y = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad, _ = grad_c5(y, N, k)
        h_new = project_01_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        y = h + m
        pk = float(np.max(fft_corr(h)))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        if it % 200 == 199:
            lr *= 0.991
            beta = min(0.97, 0.85 + it * 0.0003)
    return best_h, best_pk * dx

    if rng is None:
        rng = np.random
    h = fix_binary(h, N, rng)
    corr = fft_corr(h)
    cur_c = float(np.max(corr))
    best_c = cur_c
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_iters):

            break
        T = T0 * (Tf / T0) ** (it / max(max_iters, 1))

                break
            i1 = int(ones[rng.randint(len(ones))])
            i2 = int(ones[rng.randint(len(ones))])

            j2 = int(zeros[rng.randint(len(zeros))])
            if j1 == j2:

            new_c = float(np.max(fft_corr(h)))
            d = new_c - cur_c
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):

                ones = np.where(h == 1)[0]
