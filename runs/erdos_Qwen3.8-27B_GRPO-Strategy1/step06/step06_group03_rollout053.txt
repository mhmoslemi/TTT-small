def eval_swaps(c, h, p_list, q_list, top_idx, N, center):
    """Vectorized evaluation of swap effects on top correlations."""
    Lags = top_idx.astype(int) - center  # shape (top_k,)
    # For each p, q, compute delta at each lag
    # delta[p,q,k] = h[p+Lk] - h[q+Lk] + h[p-Lk] - h[q-Lk]
    
    # p_list shape (n_o,), q_list shape (n_z,), Lags shape (top_k,)
    # We want to compute for all (p, q) pairs
    # This is a (n_o, n_z, top_k) array
    
    pL = p_list[:, None, None] + Lags[None, None, :]  # (n_o, 1, top_k)
    qL = q_list[None, :, None] + Lags[None, None, :]  # (1, n_z, top_k)
    p_mL = p_list[:, None, None] - Lags[None, None, :]
    q_mL = q_list[None, :, None] - Lags[None, None, :]
    
    # Boundary checks
    hpL = np.where((pL >= 0) & (pL < N), h[pL.clip(0, N-1)], 0.0)
    hqL = np.where((qL >= 0) & (qL < N), h[qL.clip(0, N-1)], 0.0)
    hp_mL = np.where((p_mL >= 0) & (p_mL < N), h[p_mL.clip(0, N-1)], 0.0)
    hq_mL = np.where((q_mL >= 0) & (q_mL < N), h[q_mL.clip(0, N-1)], 0.0)
    
    deltas = hpL - hqL + hp_mL - hq_mL  # (n_o, n_z, top_k)
    new_corrs = c[top_idx][None, None, :] + deltas  # (n_o, n_z, top_k)
    new_max = new_corrs.max(axis=2)  # (n_o, n_z)
    return new_max

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def eval_swaps_vec(c, h, p_list, q_list, top_idx, N, center):
    Lags = (top_idx - center).astype(np.int64)
    pL = p_list[:, None, None].astype(np.int64) + Lags[None, None, :]
    qL = q_list[None, :, None].astype(np.int64) + Lags[None, None, :]
    p_mL = p_list[:, None, None].astype(np.int64) - Lags[None, None, :]
    q_mL = q_list[None, :, None].astype(np.int64) - Lags[None, None, :]
    hpL = np.where((pL >= 0) & (pL < N), h[np.clip(pL, 0, N - 1)], 0.0)
    hqL = np.where((qL >= 0) & (qL < N), h[np.clip(qL, 0, N - 1)], 0.0)
    hp_mL = np.where((p_mL >= 0) & (p_mL < N), h[np.clip(p_mL, 0, N - 1)], 0.0)
    hq_mL = np.where((q_mL >= 0) & (q_mL < N), h[np.clip(q_mL, 0, N - 1)], 0.0)
    deltas = hpL - hqL + hp_mL - hq_mL
    new_corrs = c[top_idx][None, None, :] + deltas
    return new_corrs.max(axis=2)

def fix_binary(h, N):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - nh, replace=False)] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, nh - n1, replace=False)] = 1.0
    return h

def intensive_search(h_init, N, deadline, top_k=30, n_sample=120):
    h = fix_binary(h_init, N)
    c = fft_corr(h)
    best_c = float(c.max())
    center = N - 1
    no_improve = 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    tabu_pq = {}

    while time.time() < deadline:
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        ns_o = min(len(ones), n_sample)
        o_s = ones[np.random.choice(len(ones), ns_o, replace=False)] if ns_o < len(ones) else ones
        ns_z = min(len(zeros), n_sample)
        z_s = zeros[np.random.choice(len(zeros), ns_z, replace=False)] if ns_z < len(zeros) else zeros
        if len(o_s) == 0 or len(z_s) == 0:
            break
        new_max = eval_swaps_vec(c, h, o_s, z_s, top_idx, N, center)
        # Apply tabu
        if tabu_pq:
            for (p, q) in list(tabu_pq.keys()):
                ip = np.where(o_s == p)[0]
                iq = np.where(z_s == q)[0]
                if len(ip) > 0 and len(iq) > 0:
                    new_max[ip[0], iq[0]] = np.inf
            tabu_pq = {k: v for k, v in tabu_pq.items() if no_improve - v < 80}
        flat_idx = int(new_max.argmin())
        i, j = np.unravel_index(flat_idx, new_max.shape)
        new_P = float(new_max[i, j])
        p = int(o_s[i])
        q = int(z_s[j])
        if new_P < P - 1e-10:
            h[p] = 0.0
            h[q] = 1.0
            tabu_pq[(p, q)] = 0
            tabu_pq[(q, p)] = 0
            c = fft_corr(h)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            no_improve += 1
            if no_improve > 50:
                n_pert = min(1 + (no_improve - 50) // 35, 7)
                for _ in range(n_pert):
                    pp = int(ones[np.random.randint(len(ones))])
                    qq = int(zeros[np.random.randint(len(zeros))])
                    h[pp] = 0.0
                    h[qq] = 1.0
                c = fft_corr(h)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                no_improve = 0
                tabu_pq = {}
    return h, best_c

def full_polish(h, N, max_rounds=30):
    h = h.copy()
    c = fft_corr(h)
    center = N - 1
    for _ in range(max_rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        top_idx = np.argpartition(c, -30)[-30:]
        new_max = eval_swaps_vec(c, h, ones, zeros, top_idx, N, center)
        flat_idx = int(new_max.argmin())
        i, j = np.unravel_index(flat_idx, new_max.shape)
        P = float(c.max())
        if new_max[i, j] >= P - 1e-10:
            break
        h[int(ones[i])] = 0.0
        h[int(zeros[j])] = 1.0
        c = fft_corr(h)
    return h
