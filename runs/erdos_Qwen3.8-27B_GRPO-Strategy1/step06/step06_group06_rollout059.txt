        # Gradient from top-k hottest shifts
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                # corr[center+s] = sum_{i=0}^{N-1-s} h[i+s]*(1-h[i])
                # d/dh[i]: for i in [0, N-1-s): -h[i+s] (from (1-h[i]) term) 
                #          for i in [s, N-1]: (1-h[i-s]) (from h[i+s] where i+s=n, n=i, so h[i]=[i-s+s], (1-h[i-s]))
                # Wait: h[n+s]*(1-h[n]), d/dh[i]:
                #   h[i] appears as h[n+s] when n+s=i → n=i-s: contribution (1-h[i-s]) if 0<=i-s<=N-1-s
                #   h[i] appears in (1-h[n]) when n=i: contribution -h[i+s] if 0<=i<=N-1-s and 0<=i+s<N

                    grad[s:] += (1.0 - h[:N-s])  # (1-h[i-s]) for i in [s, N-1]

                #   h[i] appears as h[n-s2] when n-s2=i → n=i+s2: contribution (1-h[i+s2]) if valid
                #   h[i] appears in (1-h[n]) when n=i: contribution -h[i-s2] if valid
                if s2 < N:

                    grad[:N-s2] += (1.0 - h[s2:])  # (1-h[i+s2]) for i in [0, N-1-s2]

        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0-h, L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def make_binary(h, N):
    target = N // 2
    h_bin = (h >= 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin==1)[0]
        np.random.shuffle(idx)
        h_bin[idx[:n1-target]] = 0
    elif n1 < target:
        idx = np.where(h_bin==0)[0]
        np.random.shuffle(idx)
        h_bin[idx[:target-n1]] = 1
    return h_bin

def compute_delta(h, N, p, q, s_arr):
    ps, pm = p + s_arr, p - s_arr
    qs, qm = q + s_arr, q - s_arr
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    d = np.zeros(2*N-1)
    d[vps] -= (1.0 - h[np.clip(ps,0,N-1)[vps]])
    d[vpm] += h[np.clip(pm,0,N-1)[vpm]]
    d[vqs] += (1.0 - h[np.clip(qs,0,N-1)[vqs]])
    d[vqm] -= h[np.clip(qm,0,N-1)[vqm]]
    return d

def comp_score(corr):
    return float(corr.max()) + 0.005 * float(np.sort(corr)[-3:].sum())

def anneal_ils(h_init, N, deadline, T0=0.06, min_T=0.002, cool=0.9995,
               max_stall=2000, n_ils=5, pert_size=4):
    h = h_init.copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    total_iters = 0

    for ils_round in range(n_ils):
        T = T0 * (0.55 ** ils_round)
        no_imp = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        while time.time() < deadline and T > min_T:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            d = compute_delta(h, N, p, q, s_arr)
            nc = corr + d
            npk = float(nc.max())
            delta = npk - current_pk
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                h[p] = 0; h[q] = 1
                corr = nc; current_pk = npk
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if npk < best_pk:
                    best_pk = npk; best_h = h.copy(); no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1
            T *= cool
            total_iters += 1
            if no_imp > max_stall:

                corr = fft_corr(h)

        if time.time() < deadline - 30 and ils_round < n_ils - 1:

                h[p] = 0; h[q] = 1

def lse_continuous(h, N, deadline, beta=15.0, lr0=0.003, top_k=40):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    center = N - 1
    lr = lr0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < s < N:
                grad[:N-s] += -h[s:]
                grad[s:] += (1.0 - h[:N-s])
            elif -N < s < 0:
                s2 = -s
                grad[s2:] += -h[:N-s2]
                grad[:N-s2] += (1.0 - h[s2:])
        grad /= top_k
        t += 1
        m = b1*m + (1.0-b1)*grad
        v = b2*v + (1.0-b2)*grad*grad
        mh = m / (1.0 - b1**t)
        vh = v / (1.0 - b2**t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 300 == 0:
            lr *= 0.96
    return best_h, best_pk

def golden_pattern(N):
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    x = np.arange(N) * phi
    h = (np.modf(x)[0] > 0.5).astype(float)
    return h

def block_pattern(N, block=7):
    h = np.zeros(N)
    reps = N // block + 1
    for r in range(reps):
        s = r * block
        e = min(s + block, N)
        if r % 2 == 0:
            h[s:e] = 1.0
    return h
