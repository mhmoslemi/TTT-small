    G = np.fft.fft(np.flip(1.0 - h), L)

    return float(np.max(fft_corr(h))) * (2.0 / N)

    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(h, c, k=40, tau=8.0):
    N = len(h)
    center = N - 1
    top = np.argpartition(c, -k)[-k:]
    w = np.exp(tau * (c[top] - c[top].max()))
    w /= w.sum()

        if d == 0:

        elif d > 0:

            grad[d:] -= w[i] * h[:N-d]
        else:
            M = -d
            grad[M:] += w[i] * (1.0 - h[:N-M])
            grad[:N-M] -= w[i] * h[M:]

    cy = fft_corr(y)
    g = smooth_max_grad(y, cy, k, tau)
    h_new = project_sum(y - lr * g, target)
    m_new = beta * (h_new - h)
    return h_new, m_new, float(cy.max())
