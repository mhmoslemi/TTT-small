def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    dx = 2.0 / N
    return float(np.max(fft_corr(h))) * dx

    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    corr = fft_corr(h)
    cmax = corr.max()
    vals = beta * (corr - cmax)
    w = np.exp(vals)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return cmax * (2.0/N), np.zeros(N)
    w /= w_sum
    obj = (cmax + np.log(w_sum) / beta) * (2.0/N)
    
    grad = np.zeros(N)
    # Only compute for significant weights
    thresh = w.max() * 1e-6
    active = np.where(w > thresh)[0]
    
    for k in active:
        ss = k - (N - 1)
        i_lo = max(0, ss)
        i_hi = min(N - 1, N - 1 + ss)
        # First term: +w[k]*(1-h[j-ss]) for j in [i_lo, i_hi]
        idx = np.arange(i_lo, i_hi + 1)
        idx_s = idx - ss
        grad[idx] += w[k] * (1.0 - h[idx_s])
        # Second term: -w[k]*h[j+ss] for j+ss in [i_lo, i_hi]
        j_lo = max(0, i_lo - ss)
        j_hi = min(N - 1, i_hi - ss)
        if j_lo <= j_hi:
            idx2 = np.arange(j_lo, j_hi + 1)
            idx2_s = idx2 + ss
            grad[idx2] -= w[k] * h[idx2_s]
    
    grad *= (2.0 / N)
    return obj, grad

The ideal scenario is all corr[k] = C for all k, giving C₅ = C * dx. The minimum possible C is related to the structure of h.

For a binary sequence with N/2 ones, the average of corr[k] over all k is related to the Fourier transform of the sequence. The "flat spectral" property (all Fourier components equal) would give uniform correlations.

Actually, let me try a different approach: **spectral shaping**. 

If I design h in the frequency domain to have a flat power spectrum (white noise), the autocorrelation should be flat (except at lag 0). This is the basis of "m-sequences" and "Zadoff-Chu sequences".

But we need h(x)(1-h(x+k)), not h(x)h(x+k). However, for binary h:
h(x)(1-h(x+k)) = h(x) - h(x)h(x+k)

So we want to minimize max_k (∫h dx - ∫h(x)h(x+k)dx) over the valid ranges.

∫h dx over a range of length L is approximately L/2 (for balanced h). So we want to maximize min_k ∫h(x)h(x+k)dx.

For a binary sequence with N/2 ones out of N:
∫h(x)h(x+k)dx ≈ (count of positions where both h[i]=1 and h[i+k]=1) * dx

We want this to be as uniform as possible across k.

The "Ward's identity" or "balanced sequences" from coding theory give us a starting point. But let me not go down that path and instead focus on the optimization.

    v1 = (p - ss_arr >= 0) & (p - ss_arr < N)
    v2 = (p + ss_arr >= 0) & (p + ss_arr < N)
    D_p = -np.where(v1, 1.0 - h[np.clip(p - ss_arr, 0, N-1)], 0.0) + np.where(v2, h[np.clip(p + ss_arr, 0, N-1)], 0.0)
    new_c = c + D_p
    
    # For all z in z_sample:
    # D_z (for 0->1 flip) = (1-h[z-ss])*v1 - h[z+ss]*v2
    # Vectorized over z:
    z_ss = z_sample[:, None] - ss_arr[None, :]  # (n_z, 2N-1)
    z_ss2 = z_sample[:, None] + ss_arr[None, :]
    v1_z = (z_ss >= 0) & (z_ss < N)
    v2_z = (z_ss2 >= 0) & (z_ss2 < N)
    # Clip for indexing
    z_ss_c = np.clip(z_ss, 0, N-1)
    z_ss2_c = np.clip(z_ss2, 0, N-1)
    D_z = np.where(v1_z, 1.0 - h[z_ss_c], 0.0) - np.where(v2_z, h[z_ss2_c], 0.0)
    
    new_c_all = new_c[None, :] + D_z  # (n_z, 2N-1)
    scores = new_c_all.max(axis=1)  # (n_z,)
    mi = int(scores.argmin())
    if scores[mi] < best_score:
        best_score = scores[mi]
        best_pq = (int(p), int(z_sample[mi]))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, N, beta, n_active=30):
    corr = fft_corr(h)
    cmax = corr.max()
    ex = np.exp(beta * (corr - cmax))
    w = ex / ex.sum()
    obj = (cmax + np.log(float(ex.sum())) / beta) * (2.0 / N)
    grad = np.zeros(N)
    top_idx = np.argsort(-w)[:n_active]
    ss_arr = np.arange(2 * N - 1) - (N - 1)
    for k in top_idx:
        ss = int(ss_arr[k])
        i_lo = max(0, ss)
        i_hi = min(N - 1, N - 1 + ss)
        if i_lo > i_hi:
            continue
        j1 = np.arange(i_lo, i_hi + 1)
        j1s = j1 - ss
        grad[j1] += w[k] * (1.0 - h[j1s])
        j2_lo = max(0, i_lo - ss)
        j2_hi = min(N - 1, i_hi - ss)
        if j2_lo <= j2_hi:
            j2 = np.arange(j2_lo, j2_hi + 1)
            j2s = j2 + ss
            grad[j2] -= w[k] * h[j2s]
    grad *= (2.0 / N)
    return obj, grad

def nesterov_opt(h_init, N, deadline, beta=40, lr=0.004):
    h = h_init.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    best_c5 = compute_c5(h, N)
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        y = h + m
        _, grad = smooth_grad(y, N, beta)
        h_new = project_sum(y - lr * grad, target)
        m = 0.82 * (h_new - h)
        h = h_new
        it += 1
        if it % 20 == 0:
            c5 = compute_c5(h, N)
            if c5 < best_c5 - 1e-11:
                best_c5 = c5
                best_h = h.copy()
        if it % 300 == 0:
            lr *= 0.93
    return best_h, best_c5

def binary_search(h_init, N, deadline, max_iter=3000, n_ps=60, n_zs=40):
    h = (h_init > 0.5).astype(float).copy()
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[idx[np.random.permutation(len(idx))][target:]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        h[idx[np.random.permutation(len(idx))][:target - n1]] = 1
    best_c = float(fft_corr(h).max())
    no_imp = 0
    ss_arr = np.arange(2 * N - 1) - (N - 1)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            no_imp = 0
        else:
            no_imp += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        np_s = min(n_ps, len(ones))
        p_sample = ones[np.random.permutation(len(ones))[:np_s]]
        nz_s = min(n_zs, len(zeros))
        z_sample = zeros[np.random.permutation(len(zeros))[:nz_s]]
        best_score = P + 1.0
        best_pq = None
        for p in p_sample:
            v1 = (p - ss_arr >= 0) & (p - ss_arr < N)
            v2 = (p + ss_arr >= 0) & (p + ss_arr < N)

                   + np.where(v2, h[np.clip(p + ss_arr, 0, N-1)], 0.0))
            new_c = c + D_p
            zz = z_sample[:, None]
            zs1 = zz - ss_arr[None, :]

        if best_pq is not None:
