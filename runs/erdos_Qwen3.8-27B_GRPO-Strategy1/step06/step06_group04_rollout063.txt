import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_gradient(h, T, topk=80):
    """Compute gradient of log-sum-exp approximation to max correlation"""
    N = len(h)
    c = fft_corr(h)
    # Find top-k correlations
    idx = np.argpartition(c, -topk)[-topk:]
    c_top = c[idx]
    w = np.exp((c_top - c_top.max()) / T)
    w /= w.sum()
    
    # Compute gradient: d/dh[j] sum_lag w_lag * c[lag]
    # c[lag] = sum_i h[i] * (1-h[i+lag])
    # dc[lag]/dh[j] = 1 - h[j+lag] - h[j-lag] (with h=0 outside)
    
    grad = np.zeros(N)
    for wi, ci in zip(w, idx):
        lag = int(ci) - (N - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            seg = N - lag
            grad[lag:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[lag:]
        else:
            m = -lag
            if m < N:
                seg = N - m
                grad[:seg] += wi * (1.0 - h[m:])
                grad[m:] -= wi * h[:seg]
    return grad, c

def optimize_continuous(h0, N, deadline, T0=5.0, T_min=0.5, lr0=0.01, topk=80):
    h = h0.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_val = compute_c5(h, N)
    
    T = T0
    lr = lr0
    stall = 0
    momentum = np.zeros(N)
    beta = 0.8
    
    it = 0
    while time.time() < deadline:
        grad, c = soft_gradient(h, T, topk)
        max_c = float(np.max(c)) * (2.0 / N)
        
        if max_c < best_val - 1e-12:
            best_val = max_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Momentum update
        momentum = beta * momentum - lr * grad
        h_new = h + momentum
        h = project_sum(h_new, target)
        it += 1
        
        # Adaptive scheduling
        if it % 500 == 0:
            lr = max(0.001, lr * 0.97)
            T = max(T_min, T * 0.95)
            beta = min(0.95, 0.8 + it * 0.0001)
        
        if stall > 300:
            # Restart with perturbation
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)
            momentum = np.zeros(N)
            lr = max(lr, lr0 * 0.5)
            T = max(T, T0 * 0.3)
            stall = 0
    
    return best_h, best_val
