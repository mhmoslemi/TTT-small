import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

        deltas[i, i:i + N] = s

    no_improve = 0

            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        n_p = min(n_sample, len(ones))
        p_list = np.random.choice(ones, n_p, replace=False)

        dp = deltas[p_list]

        chunk = 8
        best_new_c = 1e10
        best_pi, best_qi = 0, 0
        for ci in range(0, n_p, chunk):
            dc = min(ci + chunk, n_p)
            vals = corr[None, None, :] + dp[ci:dc, None, :] - dq[None, :, :]
            nm = vals.max(axis=2)
            mi = np.unravel_index(int(nm.argmin()), nm.shape)
            if nm[mi] < best_new_c:

        if best_new_c >= best_c - 1e-12:

            if no_improve > 100:
                break
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
        else:

def sa_binary(h_in, N, deadline, T0=0.3, cooling=0.9995, max_iters=50000):
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    cur_c = float(corr.max())
    best_c = cur_c
    best_h = h.copy()
    s = 2.0 * h - 1.0
    
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    no_improve = 0
    n_iters = 0
    
    while time.time() < deadline:
        n_iters += 1
        if n_iters % 1000 == 0:
            T *= 0.95  # Additional cooling checkpoint
        
        # Choose move type: 80% single, 20% double
        if np.random.random() < 0.8:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            # Apply
            h[p], h[q] = 0.0, 1.0
            old_corr = corr.copy()
            corr += deltas[p].copy() - deltas[q].copy()
            new_c = float(corr.max())
            
            # Accept or reject
            if new_c < cur_c:
                # Accept
                cur_c = new_c
                s[p], s[q] = -s[p], -s[q]
                deltas[p, :] = 0.0; deltas[p, p:p + N] = s
                deltas[q, :] = 0.0; deltas[q, q:q + N] = s
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # Metropolis
                if np.random.random() < np.exp(-(new_c - cur_c) / max(T, 1e-6)):
                    cur_c = new_c
                    s[p], s[q] = -s[p], -s[q]
                    deltas[p, :] = 0.0; deltas[p, p:p + N] = s
                    deltas[q, :] = 0.0; deltas[q, q:q + N] = s
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    no_improve += 1
                else:
                    # Reject
                    h[p], h[q] = 1.0, 0.0
                    corr = old_corr
        
        T *= cooling
        
        if no_improve > 5000 and T > 0.01:
            T = max(T, 0.1)  # Reheat
            no_improve = 0
    
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h.astype(float), L)
    G = np.fft.fft(np.flip(1.0 - h.astype(float)), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

    no_improve = 0

            break
        if len(ones) < 1 or len(zeros) < 1:

        if best_new_c >= best_c - 1e-12:

def sa_binary(h_in, N, deadline, T0=0.5, max_iters=100000):
    """Simulated annealing for binary optimization"""
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h)
    cur_c = float(corr.max())
    best_c = cur_c
    best_h = h.copy()
    s = 2.0 * h - 1.0
    
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    T = T0
    no_improve = 0
    it = 0
    
    while time.time() < deadline:
        it += 1
        # Pick random move
        pi = np.random.randint(len(ones))
        qi = np.random.randint(len(zeros))
        p = ones[pi]
        q = zeros[qi]
        
        # Try the swap
        old_p_delta = deltas[p].copy()
        old_q_delta = deltas[q].copy()
        corr += old_p_delta - old_q_delta
        new_c = float(corr.max())
        
        if new_c <= cur_c or np.random.random() < np.exp(-(new_c - cur_c) / max(T, 1e-8)):
            # Accept
            cur_c = new_c
            h[p], h[q] = 0.0, 1.0
            s[p], s[q] = -s[p], -s[q]
            deltas[p, :] = 0.0; deltas[p, p:p + N] = s.copy()
            deltas[q, :] = 0.0; deltas[q, q:q + N] = s.copy()
            ones[pi] = q
            zeros[qi] = p
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Reject
            corr -= old_p_delta - old_q_delta
        
        # Cool
        T *= 0.9999
        if no_improve > 2000:
            T = max(T, 0.1)
            no_improve = 0
    
    return best_h, best_c

def nesterov_polish(h_in, N, deadline, k=50, lr0=0.01, max_iter=5000):
    h = h_in.astype(float).copy()
    target = N / 2.0
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    
    while time.time() < deadline and it < max_iter:
        y = h + m
        corr_y = fft_corr(y)
        pk = float(corr_y.max())
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        
        # Subgradient: average gradient of top-k correlations
        top = np.argpartition(corr_y, -k)[-k:]
        grad = np.zeros(N)
        for t in top:
            L_flag = int(t - (N - 1))
            if L_flag == 0:
                grad += 1.0 - 2.0 * y
            elif L_flag > 0:
                grad[L_flag:] += (1.0 - y[:N - L_flag])
                grad[:N - L_flag] -= y[L_flag:]
            else:
                M = -L_flag
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        
        if it % 200 == 0:
            lr *= 0.97
            beta = min(0.97, 0.80 + it * 0.0002)
    
    return best_h, best_c * (2.0 / N)
