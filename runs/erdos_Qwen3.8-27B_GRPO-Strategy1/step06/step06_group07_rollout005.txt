import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    """Compute D matrix where D[p, k] gives change in corr[k] when h[p] goes from 1 to 0."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):

def softmax_nesterov(h, N, deadline, beta0=50.0, beta1=500.0, k_smooth=100, lr0=0.01):
    """Nesterov with softmax-smoothed max objective."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    s = fft_corr(h)
    best_pk = float(s.max())
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        s = fft_corr(y)
        pk = float(s.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        # Softmax weights
        b = beta0 + (beta1 - beta0) * min(1.0, it / 200.0)
        s_max = s.max()
        weights = np.exp(b * (s - s_max))
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.zeros_like(s)
            weights[int(np.argmax(s))] = 1.0
            wsum = 1.0
        else:
            weights /= wsum
        # Gradient: weighted sum of derivatives
        grad = np.zeros(N)
        # For each shift, add weight * derivative
        # Use efficient computation: for shift l, derivative at position i is:
        #   if l > 0: [i<N-l](1-y[i+l]) - [i>=l,i<N]y[i-l]  (for positive shift)
        #   if l < 0: [i>=|l|,i<N](1-y[i+|l|]) - [i<N-|l|]y[i-|l|]  (for negative shift)  
        #   if l = 0: 1 - 2*y[i]
        # But computing this for all shifts is O(N^2). Instead use the top weighted shifts.
        # Take top k_smooth shifts by weight
        top_idx = np.argsort(-weights)[:k_smooth]
        top_w = weights[top_idx]
        for ai in top_idx:
            L = int(ai - center)
            w = top_w[top_idx == ai][0] if len(top_idx) == 1 else float(weights[ai])
            if L == 0:
                grad += weights[ai] * (1.0 - 2.0 * y)
            elif L > 0:
                grad[:(N - L)] += weights[ai] * (1.0 - y[L:N])
                grad[L:N] -= weights[ai] * y[:N - L]
            else:
                M = -L
                if M < N:
                    grad[M:N] += weights[ai] * (1.0 - y[:N - M])
                    grad[:N - M] -= weights[ai] * y[M:N]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.98
            beta = min(0.95, 0.85 + it * 0.0002)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, perturb_every=0, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                npert = 1 + (stall // (perturb_every * 2))
                for _ in range(min(npert, 4)):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                c = fft_corr(h)
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def simulated_annealing(h, N, deadline, T0=1.0, alpha=0.9998, max_iter=100000, rng=None):
    if rng is None:
        rng = np.random.RandomState(7)
    h = ensure_sum(h, N).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    T = T0
    it = 0
    while time.time() < deadline and it < max_iter:
        if it % 500 == 0:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
        p = int(ones[rng.randint(len(ones))])
        q = int(zeros[rng.randint(len(zeros))])
        h[p] = 0.0
        h[q] = 1.0
        c_new = fft_corr(h)
        P_new = float(c_new.max())
        delta = P_new - best_c
        if delta < 0:
            c = c_new
            best_c = P_new
            best_h = h.copy()
        else:
            accept = delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T))
            if not accept:
                h[p] = 1.0
                h[q] = 0.0
                c = c_new
                # restore
                h[p] = 1.0
                h[q] = 0.0
        T *= alpha
        it += 1
    return best_h, best_c * (2.0 / N)

def softmax_nesterov(h, N, deadline, beta0=30.0, beta1=300.0, k_smooth=80, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    s = fft_corr(h)
    best_pk = float(s.max())
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        s = fft_corr(y)
        pk = float(s.max())
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
        b = beta0 + (beta1 - beta0) * min(1.0, it / 300.0)
        s_shift = s - s.max()
        weights = np.exp(b * s_shift)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.zeros_like(s)
            weights[int(np.argmax(s))] = 1.0
            wsum = 1.0
        else:
            weights /= wsum
        top_idx = np.argsort(-weights)[:k_smooth]
        grad = np.zeros(N)
        for ai in top_idx:
            L = int(ai - center)
            w = float(weights[ai])
            if w < 1e-15:
                continue
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N - L] += w * (1.0 - y[L:])
                grad[L:] -= w * y[:N - L]
            else:
                M = -L
                if M < N:
                    grad[M:] += w * (1.0 - y[:N - M])
                    grad[:N - M] -= w * y[M:]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.985
            beta = min(0.96, 0.85 + it * 0.00015)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
