def swap_delta_exact(c, h, p, q, top_idx, N):
    """Exact delta in corr at top shifts after swapping h[p]:1->0, h[q]:0->1."""
    center = N - 1
    L_arr = top_idx - center  # shape (top_m,)
    
    # For each L in L_arr, compute the delta
    # delta[L] = (h[p+L]-1 if valid) + (h[p-L] if valid) + (1-h[q+L] if valid) + (-h[q-L] if valid)
    
    ip = p + L_arr  # p+L
    valid_ip = (ip >= 0) & (ip < N)
    ip_c = np.clip(ip, 0, N-1)
    term1 = np.where(valid_ip, h[ip_c] - 1.0, 0.0)
    
    ip2 = p - L_arr  # p-L (this is the i such that i+L=p)
    valid_ip2 = (ip2 >= 0) & (ip2 < N)
    ip2_c = np.clip(ip2, 0, N-1)
    term2 = np.where(valid_ip2, h[ip2_c], 0.0)
    
    iq = q + L_arr  # q+L
    valid_iq = (iq >= 0) & (iq < N)
    iq_c = np.clip(iq, 0, N-1)
    term3 = np.where(valid_iq, 1.0 - h[iq_c], 0.0)
    
    iq2 = q - L_arr  # q-L
    valid_iq2 = (iq2 >= 0) & (iq2 < N)
    iq2_c = np.clip(iq2, 0, N-1)
    term4 = np.where(valid_iq2, -h[iq2_c], 0.0)
    
    delta = term1 + term2 + term3 + term4
    new_top = c[top_idx] + delta
    return float(new_top.max())

        if len(ones) > sample_n:

        if len(zeros) > sample_n:
            q_list = rng.choice(zeros, sample_n, replace=False)
        else:
            q_list = zeros
        best_new_P = P + 1e-15
        best_pq = None
        for p in p_list:
            for q in q_list:

                if nv < best_new_P:
                    best_new_P = nv
                    best_pq = (int(p), int(q))
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        elif anneal > 0 and rng.random() < anneal:
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        c = fft_corr(h)
        sweep += 1
    return h, best_c

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N, rng):

    m = np.zeros(N)

    beta = beta0
    it = 0

        y = h + m
        corr = fft_corr(y)
