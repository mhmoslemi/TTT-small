def make_effects(h, N, top_ci):
    """Compute p_eff and q_eff for positions 0..N-1 at correlation indices top_ci."""
    n = len(top_ci)
    p_eff = np.zeros((N, n))
    q_eff = np.zeros((N, n))
    for j, ci in enumerate(top_ci):
        k = int(ci) - (N - 1)
        if k >= 0:
            # Valid i as j+k: i in [k, N-1]
            for i in range(k, N):
                p_eff[i, j] -= (1.0 - h[i-k])
                q_eff[i, j] += (1.0 - h[i-k])
            # Valid i as j: i in [0, N-1-k]
            for i in range(N - k):
                if i + k < N:
                    p_eff[i, j] -= h[i+k]
                    q_eff[i, j] -= h[i+k]
        else:
            m = -k
            # Valid i as j: i in [0, N-1-m]
            for i in range(N - m):
                p_eff[i, j] -= (1.0 - h[i+m])
                q_eff[i, j] += (1.0 - h[i+m])
            # Valid i as j+m: i in [m, N-1]
            for i in range(m, N):
                p_eff[i, j] -= h[i-m]
                q_eff[i, j] -= h[i-m]
    return p_eff, q_eff

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def c5_of(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_binary(h, rng):
    N = len(h)
    h_b = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(h_b.sum())
    if n1 > target:
        ones = np.where(h_b == 1)[0]
        sel = rng.choice(len(ones), size=n1 - target, replace=False)
        h_b[ones[sel]] = 0.0
    elif n1 < target:
        zeros = np.where(h_b == 0)[0]
        sel = rng.choice(len(zeros), size=target - n1, replace=False)
        h_b[zeros[sel]] = 1.0
    return h_b

def make_effects(h, N, top_ci):

            p_eff[idx1, j] -= (1.0 - h[idx1 - k])
            q_eff[idx1, j] += (1.0 - h[idx1 - k])
            idx2 = np.arange(0, N - k)
            p_eff[idx2, j] -= h[idx2 + k]

            idx2 = np.arange(m, N)
