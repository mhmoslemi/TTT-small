import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_binary(h, N, rng):
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    nh = N // 2
    if n1 > nh:
        ones = np.where(h == 1)[0]
        to_flip = rng.choice(ones, n1 - nh, replace=False)
        h[to_flip] = 0
    elif n1 < nh:
        zeros = np.where(h == 0)[0]
        to_flip = rng.choice(zeros, nh - n1, replace=False)
        h[to_flip] = 1
    return h

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def local_search(h, N, deadline, max_iters=2000, sample_frac=1.0, rng=None, perturb_every=0):
    """Greedy local search with optional perturbation."""
    h = h.astype(float).copy()
    n1 = int(h.sum())
    nh = N // 2
    if n1 != nh:
        h = ensure_binary(h, N, rng)
    
    c = fft_corr(h)
    D = compute_D(h, N)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        P = float(c.max())
        Dz = D[zeros, :]
        
        if sample_frac < 1.0 and len(ones) > 30:
            n_s = max(30, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:

                h[p], h[q] = 0.0, 1.0

                stall = 0
            else:

        else:

            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            D = compute_D(h, N)
            stall += 1
        
        if P < best_c - 1e-13:

            best_h = h.copy()
            stall = 0

def sa_search(h, N, deadline, rng, T0=0.5, T_min=0.01, max_iters=5000):
    """Simulated annealing with single swaps."""
    h = h.astype(float).copy()
    n1 = int(h.sum())
    nh = N // 2
    if n1 != nh:
        h = ensure_binary(h, N, rng)
    
    c = fft_corr(h)
    D = compute_D(h, N)
    best_c = float(c.max())
    best_h = h.copy()
    cur_c = best_c
    T = T0
    
    n_iters = max_iters
    cool_rate = (T_min / T0) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T *= cool_rate
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        # Random swap
        p = int(ones[rng.randint(len(ones))])
        q = int(zeros[rng.randint(len(zeros))])
        
        new_corr = c - D[p, :] - D[q, :]
        new_P = float(new_corr.max())
        
        if new_P <= cur_c + 1e-13:
            # Accept (improvement or same)
            h[p], h[q] = 0.0, 1.0
            c = new_corr
            D = compute_D(h, N)
            cur_c = new_P
            if new_P < best_c - 1e-13:
                best_c = new_P
                best_h = h.copy()
        else:
            # Accept with probability
            delta = new_P - cur_c
            if delta * N / 2.0 < T:  # scale by dx
                if rng.random() < np.exp(-delta * (N / 2.0) / T):
                    h[p], h[q] = 0.0, 1.0
                    c = new_corr
                    D = compute_D(h, N)
                    cur_c = new_P
    
    return best_h, best_c * (2.0 / N)

For large β, this approximates max. The gradient is:
∂C_smooth/∂h[j] = Σ_k (exp(β*corr[k]) / Σ_m exp(β*corr[m])) * ∂corr[k]/∂h[j]

This is a probability-weighted sum of the per-lag gradients.

def adam_continuous(h, N, deadline, beta=50.0, lr=0.01, max_iters=2000, rng=None):
    h = h.astype(float).copy()
    h = project_sum(h, N / 2.0)
    dx = 2.0 / N
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    
    m_s = np.zeros(N)  # Adam first moment
    v_s = np.zeros(N)  # Adam second moment
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        
        # Softmax weights
        shifted = corr - corr.max()
        weights = np.exp(beta * shifted)
        weights /= weights.sum()
        
        # Gradient: weighted sum of per-lag gradients
        # corr[k] = Σ_j h[j] * (1-h[j+k])  (with bounds)
        # ∂corr[k]/∂h[j] = (1-h[j+k]) + h[j-k]*(1-h[j]) ... wait
        # Actually: ∂/∂h[j] [h[j]*(1-h[j+k])] = (1-h[j+k]) + h[j]*0 = (1-h[j+k]) (if in range)
        # ∂/∂h[j] [h[j-k]*(1-h[j])] = -(h[j-k]) (if in range)
        # So ∂corr[k]/∂h[j] = (1-h[j+k])*I(j+k valid) - h[j-k]*I(j-k valid)
        
        # This is exactly D[j, k] but with opposite sign!
        # D[j, k] = -(1-h[j-k])*vm + h[j+k]*vp
        # ∂corr[k]/∂h[j] = (1-h[j+k])*vp - h[j-k]*vm
        # = -D[j, k]! (from our earlier analysis)
        
        # So gradient = -D^T @ weights
        # where D is the Nx(2N-1) matrix and weights is (2N-1)
        
        # But computing full D is expensive. Let's compute it.
        D = compute_D(h, N)  # shape (N, 2N-1)
        grad = -D.T @ weights  # shape (N,)
        
        # Adam update
        m_s = b1 * m_s + (1 - b1) * grad
        v_s = b2 * v_s + (1 - b2) * grad * grad
        m_hat = m_s / (1 - b1**t)
        v_hat = v_s / (1 - b2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h_new, N / 2.0)
        
        # Check constraint
        if not np.all(h_new >= 0) and not np.all(h_new <= 1):
            h_new = np.clip(h_new, 0, 1)
            h_new = project_sum(h_new, N / 2.0)
        
        h = h_new
    
    return best_h, best_pk * dx

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_binary(h, N, rng):
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    nh = N // 2
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx, nh - n1, replace=False)] = 1
    return h

def compute_D(h, N):
    pos = np.arange(N)[:, None]
    sh = (np.arange(2*N-1) - (N-1))[None, :]
    pm, pp = pos - sh, pos + sh
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pc = np.clip(pm, 0, N-1)
    qc = np.clip(pp, 0, N-1)
    return -(1.0-h[pc])*vm + h[qc]*vp

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def upsample(h, Nn):
    return np.interp(np.linspace(0,2,Nn,endpoint=False), np.linspace(0,2,len(h),endpoint=False), h)

    st = 0

        if time.time() > deadline:
            break
        ones, zeros = np.where(h==1)[0], np.where(h==0)[0]
        if len(ones)<1 or len(zeros)<1:
            break
        P = c.max()
        Dz = D[zeros,:]
        n_s = len(ones) if (sf>=1.0 or len(ones)<=40) else max(40, int(len(ones)*sf))
        pl = rng.choice(ones, n_s, replace=False) if n_s<len(ones) else ones
        bnP, bpq = P, None
        for p in pl:
            cd = (c - D[p,:])[None,:] - Dz
            mk = cd.max(1)
            mi = mk.argmin()
            if mk[mi] < bnP:
                bnP, bpq = float(mk[mi]), (int(p), int(zeros[mi]))
        if bpq is None:
            if perturb>0 and st>perturb:
                p,q = int(ones[rng.randint(len(ones))]), int(zeros[rng.randint(len(zeros))])
                h[p],h[q] = 0,1
                c,D = fft_corr(h), compute_D(h,N)
                st=0
            else:
                break
        else:
            h[bpq[0]],h[bpq[1]] = 0,1
            c,D = fft_corr(h), compute_D(h,N)
            st+=1
        if P < bp - 1e-13:
            bp, bh = P, h.copy()
            st = 0
    return bh, bp*(2.0/N)

def adam_cont(h, N, deadline, rng, beta=80.0, lr=0.008, max_it=3000):
    h = project_sum(h, N/2.0).astype(float).copy()
    dx = 2.0/N
    bp, bh = fft_corr(h).max(), h.copy()
    ms, vs, t = np.zeros(N), np.zeros(N), 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        t += 1
        c = fft_corr(h)
        pk = c.max()
        if pk < bp - 1e-12:
            bp, bh = pk, h.copy()
        sw = np.exp(beta*(c - c.max()))
        sw /= sw.sum()
        D = compute_D(h, N)
        g = -(D.T @ sw)
        b1,b2,eps = 0.9, 0.999, 1e-8
        ms = b1*ms + (1-b1)*g
        vs = b2*vs + (1-b2)*g*g
        mh, vh = ms/(1-b1**t), vs/(1-b2**t)
        h = project_sum(h - lr*mh/(np.sqrt(vh)+eps), N/2.0)
    return bh, bp*dx

new_corr = c - D[p1,:] - D[p2,:] - D[q1,:] - D[q2,:]

1. Do the greedy 1-for-1 to a local optimum
2. Try random 2-for-2 swaps: pick random p1,p2 from ones, q1,q2 from zeros, compute new_corr, accept if better
3. Repeat for many iterations

        if two_swap_every > 0 and it > 0 and it % two_swap_every == 0 and len(ones) >= 2 and len(zeros) >= 2:
            best_np, best_2idx = P, None
            for _ in range(30):
                p1, p2 = rng.choice(ones, 2, replace=False)
                q1, q2 = rng.choice(zeros, 2, replace=False)
                nc = c - D[p1] - D[p2] - D[q1] - D[q2]
                mv = float(nc.max())
                if mv < best_np - 1e-13:
                    best_np, best_2idx = mv, (int(p1), int(p2), int(q1), int(q2))
            if best_2idx is not None:
                h[best_2idx[0]] = 0.0
                h[best_2idx[1]] = 0.0
                h[best_2idx[2]] = 1.0
                h[best_2idx[3]] = 1.0
                c, D = fft_corr(h), compute_D(h, N)
                improved = True
        if not improved:
            Dz = D[zeros, :]
            n_s = len(ones) if (sf >= 1.0 or len(ones) <= 40) else max(40, int(len(ones) * sf))
            pl = rng.choice(ones, n_s, replace=False) if n_s < len(ones) else ones
            best_nP, best_pq = P, None
            for p in pl:
                cd = (c - D[p, :])[None, :] - Dz
                mk = cd.max(1)
                mi = int(np.argmin(mk))
                if mk[mi] < best_nP:
                    best_nP, best_pq = float(mk[mi]), (int(p), int(zeros[mi]))
            if best_pq is None:
                if perturb > 0 and st > perturb:
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                    c, D = fft_corr(h), compute_D(h, N)
                    st = 0
                else:
                    break
            else:
                h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
                c, D = fft_corr(h), compute_D(h, N)
                st += 1
        if float(c.max()) < bp - 1e-13:
            bp, bh = float(c.max()), h.copy()
            st = 0
    return bh, bp * (2.0 / N)

def adam_cont(h, N, deadline, rng, beta=80.0, lr=0.007, max_it=3000):
    h = project_sum(h.astype(float), N / 2.0).copy()
    dx = 2.0 / N
    bp, bh = float(fft_corr(h).max()), h.copy()
    ms, vs, t = np.zeros(N), np.zeros(N), 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    for it in range(max_it):
        if time.time() > deadline:
            break
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < bp - 1e-12:
            bp, bh = pk, h.copy()
        sw = np.exp(beta * (c - c.max()))
        sw /= sw.sum()
        D = compute_D(h, N)
        g = -(D.T @ sw)
        ms = b1 * ms + (1 - b1) * g
        vs = b2 * vs + (1 - b2) * g * g
        mh, vh = ms / (1 - b1 ** t), vs / (1 - b2 ** t)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), N / 2.0)
    return bh, bp * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h, best_c5 = h0.copy(), compute_c5(h0, N0)

    # Phase 1: Thorough local search at N0 with 2-for-2
    d1 = min(dl - 750, t0 + 80)
    h1, c1 = greedy_ls(h0, N0, d1, rng, sf=1.0, max_it=6000, two_swap_every=40)
    if c1 < best_c5:
        best_h, best_c5 = h1.copy(), c1

    # Phase 2: Iterated perturbation at N0
    for k in range(6):
        if time.time() > dl - 650:
            break
        hk = best_h.copy()
        nf = 3 + 2 * k
        ones = np.where(hk == 1)[0]
        zeros = np.where(hk == 0)[0]
        for _ in range(nf):
            hk[ones[rng.randint(len(ones))]] = 0
            hk[zeros[rng.randint(len(zeros))]] = 1
        hk = ensure_binary(hk, N0, rng)
        d2 = min(dl - 620, time.time() + 70)
        h2, c2 = greedy_ls(hk, N0, d2, rng, sf=0.85, max_it=2500, perturb=20, two_swap_every=60)
        if c2 < best_c5:
            best_h, best_c5 = h2.copy(), c2

    # Phase 3: Upscale to 2*N0, local search
    N1 = 2 * N0
    h_up = ensure_binary(upsample(best_h, N1), N1, rng)
    d3 = min(dl - 450, time.time() + 140)
    h3, c3 = greedy_ls(h_up, N1, d3, rng, sf=0.7, max_it=2000, two_swap_every=80)
    if c3 < best_c5:
        best_h, best_c5 = h3.copy(), c3

    # Phase 4: Adam continuous at N1
    if time.time() < dl - 300:
        hc = upsample(best_h, N1) if len(best_h) != N1 else best_h.copy()
        hc = hc + rng.randn(N1) * 0.03
        hc = project_sum(hc, N1 / 2.0)
        d4 = min(dl - 250, time.time() + 110)
        h4, c4 = adam_cont(hc, N1, d4, rng, beta=90, lr=0.005)
        if c4 < best_c5:
            best_h, best_c5 = h4.copy(), c4
        # Round and local search
        hr = ensure_binary(h4, N1, rng)
        d4b = min(dl - 230, time.time() + 60)
        h4b, c4b = greedy_ls(hr, N1, d4b, rng, sf=0.7, max_it=1000)
        if c4b < best_c5:
            best_h, best_c5 = h4b.copy(), c4b

    # Phase 5: Local search at current best resolution
    Nc = len(best_h)
    if time.time() < dl - 180:
        h5b = ensure_binary(best_h, Nc, rng)
        d5 = min(dl - 140, time.time() + 90)
        sf5 = 1.0 if Nc <= 400 else 0.6
        h5, c5 = greedy_ls(h5b, Nc, d5, rng, sf=sf5, max_it=3000, perturb=25, two_swap_every=50)
        if c5 < best_c5:
            best_h, best_c5 = h5.copy(), c5

    # Phase 6: Upscale to 4*N0
    N2 = 4 * N0
    if N2 <= 1600 and time.time() < dl - 140:
        h6 = ensure_binary(upsample(best_h, N2), N2, rng)
        d6 = min(dl - 90, time.time() + 90)
        h7, c7 = greedy_ls(h6, N2, d6, rng, sf=0.5, max_it=800, two_swap_every=100)
        if c7 < best_c5:
            best_h, best_c5 = h7.copy(), c7

    # Phase 7: Adam at higher resolution
    if time.time() < dl - 50:
        Na = min(N2 if N2 <= 1600 else N1, max(N0, len(best_h)))
        ha = upsample(best_h, Na) if len(best_h) != Na else best_h.copy()
        ha = ha + rng.randn(Na) * 0.02
        ha = project_sum(ha, Na / 2.0)
        d7 = min(dl - 30, time.time() + 60)
        h8, c8 = adam_cont(ha, Na, d7, rng, beta=110, lr=0.004)
        if c8 < best_c5:
            best_h, best_c5 = h8.copy(), c8
        hr8 = ensure_binary(h8, Na, rng)
        d7b = min(dl - 10, time.time() + 30)
        h8b, c8b = greedy_ls(hr8, Na, d7b, rng, sf=0.5, max_it=500)
        if c8b < best_c5:
            best_h, best_c5 = h8b.copy(), c8b

    # Phase 8: Final polish
    if time.time() < dl - 2:
        Nf = len(best_h)
        hF = ensure_binary(best_h, Nf, rng)
        sfF = 1.0 if Nf <= 500 else 0.5
        hF2, cF = greedy_ls(hF, Nf, dl - 0.5, rng, sf=sfF, max_it=800, two_swap_every=60)
        if cF < best_c5:
            best_h, best_c5 = hF2.copy(), cF

    final = compute_c5(best_h, len(best_h))
    return best_h, final, len(best_h)
