    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)

    nh = N // 2
    h = (h > 0.5).astype(float)

    if n1 > nh:
        idx = np.where(h == 1)[0]
        removal = rng.choice(idx, n1 - nh, replace=False)
        h[removal] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        addition = rng.choice(idx, nh - n1, replace=False)
        h[addition] = 1
    return h

def swap_delta_at_lags(h, p, q, lags):
    """Compute delta at specific lag values (positive integers)."""
    d = np.zeros(len(lags))
    for i, M in enumerate(lags):

            continue
        if p + M < len(h):
            d[i] -= (1.0 - h[p + M])
        if q + M < len(h):
            d[i] += (1.0 - h[q + M])
        if p >= M:
            d[i] += h[p - M]
        if q >= M:
            d[i] -= h[q - M]
