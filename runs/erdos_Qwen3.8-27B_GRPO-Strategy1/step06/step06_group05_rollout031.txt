```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
```

```python
def lse_gradient(h, N, T):
    c = fft_corr(h)
    # log-sum-exp for numerical stability
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    p = exp_c / exp_c.sum()
    D = compute_D(h, N)
    grad = D.T @ p  # shape (N,)
    return grad, c, p

def lse_optimize(h, N, deadline, T0=50.0, T1=200.0, n_steps=5000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.05
    for t in range(1, n_steps + 1):
        if time.time() > deadline:
            break
        T = T0 + (T1 - T0) * min(1.0, t / (n_steps * 0.5))
        grad, c, p = lse_gradient(h, N, T)
        c5 = c.max() * (2.0 / N)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
        bc1 = 1 - beta1 ** t
        bc2 = 1 - beta2 ** t
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        h_new = h - lr * (m / bc1) / (np.sqrt(v / bc2) + eps)
        h = project_sum(h_new, target)
        if t % 100 == 0:
            lr *= 0.97
    return best_h, best_c5
```

def lse_grad(h, N, T):
    c = fft_corr(h)
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    p = exp_c / exp_c.sum()
    # grad_h[i] = Σ_m p[m] * [(1-h[i+s_m])*1{valid} - h[i-s_m])*1{valid}]
    # where s_m = m - (N-1)
    shifts = np.arange(2*N-1) - (N-1)
    idx = np.arange(N)
    grad = np.zeros(N)
    for s in shifts:
        w = p[N-1+s]  # weight for this shift
        # Term 1: (1-h[i+s]) for valid i+s
        ps = idx + s
        v1 = (ps >= 0) & (ps < N)
        grad[v1] += w * (1.0 - h[ps[v1]])
        # Term 2: -h[i-s] for valid i-s
        ms = idx - s
        v2 = (ms >= 0) & (ms < N)
        grad[v2] -= w * h[ms[v2]]
    return grad, c, p

def lse_grad_fast(h, N, T):
    c = fft_corr(h)
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    p = exp_c / exp_c.sum()
    # Prefix sums
    P = np.concatenate([[0.0], np.cumsum(p)])
    # P[u+1] = sum_{v=0}^{u} p[v]
    # P[2N-1] = sum of all p = 1
    S1 = np.zeros(N)  # S1[j] = sum_{u=j}^{2N-2} p[u] = P[2N-1] - P[j]
    S1 = P[2*N-1] - P[np.arange(N)]
    # S2[j] = sum_{u=N-1-j}^{2N-2-j} p[u] = P[2N-1-j] - P[N-1-j]
    S2 = P[2*N-1 - np.arange(N)] - P[N-1 - np.arange(N)]
    # Handle edge: P[N-1-j] when j=N-1 gives P[0]=0, fine
    grad = (1.0 - h) * S1 - h * S2
    return grad, c, p
