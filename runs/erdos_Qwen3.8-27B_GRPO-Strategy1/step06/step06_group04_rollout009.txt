```python
import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def c5_value(h):
    N = len(h)
    dx = 2.0 / N
    c = compute_corr(h)
    return float(np.max(c)) * dx

def make_binary(h, N):
    """Convert to binary with exactly N/2 ones."""
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    nh = N // 2
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1-nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh-n1]] = 1
    return h

def lag_delta(h, i, j, top_lags, N):
    """Compute change in correlation at given lags when swapping h[i]:1->0, h[j]:0->1.
    top_lags are indices into the full correlation array (0 to 2N-2).
    Returns array of changes."""
    changes = np.zeros(len(top_lags))
    for idx, k in enumerate(top_lags):
        # k is the index in the full correlation (0 to 2N-2)
        # Full corr index k corresponds to lag = k - (N-1)
        lag = k - (N - 1)
        if lag >= 0:
            # c[k] = sum_{m=0}^{N-1-lag} h[m]*(1-h[m+lag])
            # Flipping i from 1->0: change = -(1-h[i+lag]) if 0<=i<=N-1-lag
            # Flipping j from 0->1: change = +(1-h[j+lag]) if 0<=j<=N-1-lag
            if 0 <= i <= N-1-lag:
                changes[idx] -= (1 - h[i + lag])
            if 0 <= j <= N-1-lag:
                changes[idx] += (1 - h[j + lag])
        else:
            # lag < 0, let M = -lag
            M = -lag
            # c[k] = sum_{m=M}^{N-1} h[m]*(1-h[m-M])
            if M <= i <= N-1:
                changes[idx] -= (1 - h[i - M])
            if M <= j <= N-1:
                changes[idx] += (1 - h[j - M])
    return changes
```
