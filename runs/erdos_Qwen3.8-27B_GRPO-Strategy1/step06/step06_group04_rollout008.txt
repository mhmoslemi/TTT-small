        best_c5, best_h = c5b3, hb3.copy()

    # Phase 4: Short continuous refinement (100s)
    d4 = max(d3, t0) + 100
    d4 = min(deadline - 80, d4)
    hc4 = best_h + np.random.randn(N0) * 0.03
    hc4 = project_sum(hc4, N0 / 2.0)
    hn4, c5n4 = continuous_nesterov(hc4, N0, d4, k_top=80, lr_init=0.012, temp_init=0.0015)
    if c5n4 < best_c5:
        best_c5, best_h = c5n4, hn4.copy()

    # Phase 5: Final binary polish with all remaining time
    if time.time() < deadline - 5:
        hb5, cb5 = binary_multiswap(best_h, N0, deadline - 3, n_trials=30, kick_thresh=35)
        c5b5 = compute_c5(hb5, N0)
        if c5b5 < best_c5:
            best_c5, best_h = c5b5, hb5.copy()

    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    s = float(best_h.sum())
    if abs(s - N0 / 2.0) > 1e-9:
        best_h = project_sum(best_h, N0 / 2.0)
    final_c5 = compute_c5(best_h, N0)
    return best_h, final_c5, N0
