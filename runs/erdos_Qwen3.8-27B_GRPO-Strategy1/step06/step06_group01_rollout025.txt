import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    one_minus_h = 1.0 - h
    G = np.fft.rfft(one_minus_h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def build_deltas(s):
    N = len(s)
    L = 2 * N - 1
    deltas = np.zeros((N, L))
    for i in range(N):
        deltas[i, i:i+N] = s
    return deltas

def steepest_descent(h, N, deadline, max_iters=5000):
    """Full-neighborhood steepest descent for binary h."""
    h = h.astype(float).copy()
    # Ensure exactly N//2 ones
    n_ones = int(h.sum())
    if n_ones != N // 2:
        order = np.argsort(-h)
        h = np.zeros(N)
        h[order[:N//2]] = 1.0
    else:
        h[h >= 0.5] = 1.0
        h[h < 0.5] = 0.0
    
    s = 2.0 * h - 1.0
    L = 2 * N - 1
    deltas = build_deltas(s)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        # Evaluate all swaps
        best_val = np.inf
        best_p, best_q = -1, -1
        
        for p in ones:
            base = corr + deltas[p]
            # For all q: max(base - deltas[q])
            vals = base[None, :] - deltas[zeros]  # (n_zeros, L)
            mx = vals.max(axis=1)  # (n_zeros,)
            qi = int(mx.argmin())
            if mx[qi] < best_val:
                best_val = float(mx[qi])
                best_p, best_q = p, zeros[qi]
        
        if best_val >= best_c - 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        # Apply swap
        h[best_p], h[best_q] = 0.0, 1.0
        delta_vec = deltas[best_p] - deltas[best_q]
        corr += delta_vec
        s[best_p], s[best_q] = -s[best_p], -s[best_q]
        deltas[best_p] = 0; deltas[best_p, best_p:best_p+N] = s
        deltas[best_q] = 0; deltas[best_q, best_q:best_q+N] = s
        
        c = float(corr.max())
        if c < best_c:
            best_c = c
            best_h = h.copy()
        no_improve = 0
    
    return best_h, best_c * (2.0 / N)

def simulated_annealing(h, N, deadline, T0=2.0, cooling=0.95, iters_per_temp=50):
    """SA with full neighborhood."""
    h = h.astype(float).copy()
    s = 2.0 * h - 1.0
    L = 2 * N - 1
    deltas = build_deltas(s)
    corr = fft_corr(h)
    
    best_c = float(corr.max())
    best_h = h.copy()
    T = T0
    
    while time.time() < deadline and T > 0.01:
        for _ in range(iters_per_temp):
            if time.time() > deadline:
                break
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            
            delta_vec = deltas[p] - deltas[q]
            new_corr = corr + delta_vec
            new_c = float(new_corr.max())
            old_c = float(corr.max())
            
            if new_c < old_c or np.random.random() < np.exp((old_c - new_c) / T):
                # Accept
                h[p], h[q] = 0.0, 1.0
                corr = new_corr
                s[p], s[q] = -s[p], -s[q]
                deltas[p] = 0; deltas[p, p:p+N] = s
                deltas[q] = 0; deltas[q, q:q+N] = s
                
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
        
        T *= cooling
    
    return best_h, best_c * (2.0 / N)

def double_swap_search(h, N, deadline, max_iters=2000):
    """Search for best double swap (two 1s exchanged with two 0s)."""
    h = h.astype(float).copy()
    s = 2.0 * h - 1.0
    L = 2 * N - 1
    deltas = build_deltas(s)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try random double swaps
        improved = False
        for _ in range(20):
            p1 = int(np.random.choice(ones))
            p2 = int(np.random.choice(ones))
            if p1 == p2: continue
            q1 = int(np.random.choice(zeros))
            q2 = int(np.random.choice(zeros))
            if q1 == q2: continue
            
            dvec = (deltas[p1] - deltas[q1]) + (deltas[p2] - deltas[q2])
            new_corr = corr + dvec
            new_c = float(new_corr.max())
            
            if new_c < best_c - 1e-10:
                h[p1], h[q1] = 0.0, 1.0
                h[p2], h[q2] = 0.0, 1.0
                corr = new_corr
                for p, q in [(p1, q1), (p2, q2)]:
                    s[p], s[q] = -s[p], -s[q]
                    deltas[p] = 0; deltas[p, p:p+N] = s
                    deltas[q] = 0; deltas[q, q:q+N] = s
                best_c = new_c
                best_h = h.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0
    
    return best_h, best_c * (2.0 / N)

def make_random_binary(N, seed):
    rng = np.random.RandomState(seed)

def make_golden_binary(N, seed=0):
    """Place ones at golden-ratio positions."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    N_half = N // 2
    positions = []
    for i in range(N_half):
        pos = int((i * phi) % 1.0 * N) % N
        if pos not in positions:
            positions.append(pos)
    # Fill remaining
    while len(positions) < N_half:
        pos = len(positions)
        if pos < N and pos not in positions:
            positions.append(pos)
    h = np.zeros(N)
    h[positions[:N_half]] = 1.0
    return h

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Project onto simplex with box constraints
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_continuous(h_in, N, deadline, k=30):
    """Nesterov gradient descent on continuous h."""
    h = h_in.astype(float).copy()
    h = project_sum(h, N / 2.0)
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = 0.01
    beta = 0.8
    it = 0
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Gradient: focus on top-k peaks
        top = np.argpartition(corr, -k)[-k:]
        top = np.sort(top)
        grad = np.zeros(N)
        for t in top:
            d = int(t - (N - 1))
            if d == 0:
                grad += 1.0 - 2.0 * y
            elif d > 0:
                grad[d:] += (1.0 - y[:N-d])
                grad[:N-d] -= y[d:]
            else:
                M = -d
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= len(top)
        
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 50 == 0:
            lr *= 0.97
            beta = min(0.95, 0.75 + it * 0.0005)
    
    return best_h, best_pk * (2.0 / N)

    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_cont(h_in, N, deadline, k=40):
    h = project_sum(h_in.astype(float), N / 2.0)
    hx = h.copy()
    corr = fft_corr(hx)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = 0.01
    beta = 0.8
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        top = np.sort(top)
        grad = np.zeros(N)
        for t in top:
            d = int(t - (N - 1))
            if d == 0:
                grad += 1.0 - 2.0 * y
            elif d > 0:
                grad[d:] += (1.0 - y[:N - d])
                grad[:N - d] -= y[d:]
            else:
                M = -d
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.97
            beta = min(0.95, 0.75 + it * 0.0004)
    return best_h, best_pk * (2.0 / N)
