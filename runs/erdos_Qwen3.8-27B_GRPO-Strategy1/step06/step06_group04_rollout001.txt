def continuous_opt_v2(h0, N, deadline, k0=30, k_end=5, T0=0.005, Tend=0.0005):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    h_prev = h.copy()
    v = np.zeros(N)  # momentum
    lr = 0.02
    beta = 0.9
    T = T0
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        cmax = corr.max()
        if cmax < best_pk - 1e-14:
            best_pk = cmax
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive top-k
        k = max(k_end, int(k0 * (Tend / max(T, 1e-10)) ** 0.3))
        
        # Softmax weighting over top-k
        top_k = np.argpartition(corr, -k)[-k:]
        z = corr[top_k] / max(T, 1e-10)
        z -= z.max()
        w = np.exp(z)
        w /= w.sum()
        
        # Gradient computation
        grad = np.zeros(N)
        center = N - 1
        for wi, ai in zip(w, top_k):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                seg = N - L
                if seg > 0:
                    grad[L:L+seg] += wi * (1.0 - y[:seg])
                    grad[:seg] -= wi * y[L:L+seg]
            else:
                M = -L
                if M < N:
                    seg = N - M
                    if seg > 0:
                        grad[:seg] += wi * (1.0 - y[M:M+seg])
                        grad[M:M+seg] -= wi * y[:seg]

        h_new = project_sum(y + v, target)
        h = h_new

            h_prev = h.copy()
            v = np.zeros(N)

    best_h = h.copy()
    stall = 0
    it = 0
    while time.time() < deadline:

        T = T0 * (Tend / T0) ** (it / 5000)

        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        p_idx = np.random.choice(len(ones), n_swap, replace=False)
        q_idx = np.random.choice(len(zeros), n_swap, replace=False)
        ps = ones[p_idx]
        qs = zeros[q_idx]
        
        # Apply move
        for p in ps: h[p] = 0
        for q in qs: h[q] = 1

                stall = 0

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(y, corr, top_k, w):
    N = len(y)
    grad = np.zeros(N)
    center = N - 1
    for wi, ai in zip(w, top_k):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif L > 0:
            seg = N - L
            if seg > 0:
                grad[L:L+seg] += wi * (1.0 - y[:seg])
                grad[:seg] -= wi * y[L:L+seg]
        else:
            M = -L
            if M < N:
                seg = N - M
                if seg > 0:
                    grad[:seg] += wi * (1.0 - y[M:M+seg])
                    grad[M:M+seg] -= wi * y[:seg]
    return grad

def continuous_opt(h0, N, deadline, k_start=40, k_end=4, T_start=0.006, T_end=0.0004, lr_init=0.02):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    h_prev = h.copy()
    v = np.zeros(N)
    lr = lr_init
    beta = 0.90
    T = T_start
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        progress = min(1.0, it / 3000.0)
        k = max(k_end, int(k_start * (1.0 - progress) + k_end * progress))
        top_k = np.argpartition(corr, -k)[-k:]
        z = corr[top_k] / max(T, 1e-12)
        z -= z.max()
        w = np.exp(z)
        w /= w.sum()
        grad = grad_topk(y, corr, top_k, w)
        v = 0.85 * v - lr * grad
        h_new = project_sum(y + v, target)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr = max(0.002, lr_init * (1.0 - progress) ** 1.5 + 0.002)
            T = max(T_end, T_start * (T_end / T_start) ** progress)
            beta = min(0.97, 0.88 + 0.09 * progress)
        if stall > 300:
            h = best_h + np.random.randn(N) * 0.015 * (1.0 + progress)
            h = project_sum(h, target)
            h_prev = h.copy()
            v = np.zeros(N)
            lr = max(lr, lr_init * 0.5)
            stall = 0
    return best_h, best_pk * (2.0 / N)

def binary_sa(h0, N, deadline, T0=0.06, Tend=0.002, multi_frac=0.25):

    n1 = int(h.sum())
    if n1 != nh:

            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh - n1]
            h[idx] = 1
    c = fft_corr(h)
    best_int = int(c.max())
    best_h = h.copy()
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        T = max(Tend, T0 * np.exp(-it / 3000.0))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        n_swap = 2 if (np.random.random() < multi_frac and len(ones) >= 2 and len(zeros) >= 2) else 1
        ps = ones[np.random.choice(len(ones), n_swap, replace=False)]
        qs = zeros[np.random.choice(len(zeros), n_swap, replace=False)]
        cur_int = int(c.max())
        for p in ps: h[p] = 0
        for q in qs: h[q] = 1
        c_new = fft_corr(h)
        new_int = int(c_new.max())
        delta = new_int - cur_int
        if delta <= 0 or (delta > 0 and np.random.random() < np.exp(-delta * 0.8 / max(T, 1e-10))):
            c = c_new
            if new_int < best_int:
                best_int = new_int
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            for p in ps: h[p] = 1
            for q in qs: h[q] = 0
            stall += 1

            h = (best_h + np.random.randn(N) * 0.15 > 0.5).astype(float)

            if n1 > nh:

                h[idx] = 0
            elif n1 < nh:
                idx = np.random.choice(np.where(h==0)[0], nh-n1, replace=False)
                h[idx] = 1
            c = fft_corr(h)
            stall = 0
    return best_h, best_int * (2.0 / N)

def interp_to_N(h, N_new):

    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)
