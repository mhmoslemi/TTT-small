import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

        return h
    lo, hi = -1.0, 1.0

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    """D[p, s] = change in corr[s] when h[p] goes 1→0"""
    s_idx = np.arange(2 * N - 1)
    shifts = s_idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]

    pp = pos + sv

    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    s_idx = np.arange(2 * N - 1)
    shifts = s_idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binary_search(h, N, deadline, max_sweeps=600, sample_n=200,
                  anneal_temp=0.5, cool=0.9995, patience=300):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    no_improve = 0
    temp = anneal_temp
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        if len(ones) > sample_n:
            perm = np.random.permutation(len(ones))[:sample_n]
            p_list = ones[perm]
        else:
            p_list = ones
        qs = min(len(zeros), sample_n * 2)
        if len(zeros) > qs:
            perm_z = np.random.permutation(len(zeros))[:qs]
            q_list = zeros[perm_z]
        else:
            q_list = zeros
        best_score = P + 1e-9
        best_pq = None
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - D[q_list, :].T
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_score:
                best_score = float(npk[mi])
                best_pq = (int(p), int(q_list[mi]))
        accept = True
        if best_pq is not None and best_score > P and temp > 1e-8:
            if np.random.random() > np.exp(min(50.0, -(best_score - P) / temp)):
                accept = False
        if best_pq is not None and accept:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p] = 0.0
            h[q] = 1.0
        temp *= cool
        if no_improve > patience and temp < 1e-6:
            break
    return best_h, best_c

def two_swap_search(h, N, deadline, max_sweeps=200, sample_n=80):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break
        D = compute_D(h, N)
        ns = min(len(ones), sample_n)
        nz = min(len(zeros), sample_n)
        p_a = ones[np.random.permutation(len(ones))[:ns]]
        p_b = ones[np.random.permutation(len(ones))[:ns]]
        q_a = zeros[np.random.permutation(len(zeros))[:nz]]
        q_b = zeros[np.random.permutation(len(zeros))[:nz]]
        best_score = P + 1e-9
        best_move = None
        for i in range(min(ns, 30)):
            for j in range(min(nz, 15)):
                if p_a[i] == p_b[j]:
                    continue
                base = c + D[p_a[i]] + D[p_b[j]]
                # Sample a few q pairs
                nq = min(nz, 10)
                cand = base[None, :] - D[q_a[:nq], :].T - D[q_b[:nq], :].T
                npk = cand.max(axis=1)
                mi = int(np.argmin(npk))
                if npk[mi] < best_score:
                    best_score = float(npk[mi])
                    best_move = (int(p_a[i]), int(p_b[j]), int(q_a[mi]), int(q_b[mi]))
        if best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 0.0
            h[best_move[2]] = 1.0
            h[best_move[3]] = 1.0
        else:
            break
        if no_improve > 50:
            break
    return best_h, best_c

def nesterov_opt(h, N, deadline, k=50, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.993
            beta = min(0.97, 0.86 + it * 0.0004)
    return best_h, best_pk * (2.0 / N)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rm = idx[np.argsort(np.random.random(len(idx)))][nh:]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        add = idx[np.argsort(np.random.random(len(idx)))][:nh - n1]
        h[add] = 1.0
    return h
