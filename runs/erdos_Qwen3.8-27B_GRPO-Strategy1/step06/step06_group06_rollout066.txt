import numpy as np

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h.astype(float), L)
    G = np.fft.fft(np.flip(1.0-h.astype(float)), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N, rng=None):
    target = N // 2
    h_bin = h_bin.copy()
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin==1)[0]
        np.random.shuffle(idx)
        h_bin[idx[:n1-target]] = 0
    elif n1 < target:
        idx = np.where(h_bin==0)[0]
        np.random.shuffle(idx)
        h_bin[idx[:target-n1]] = 1

def delta_p_remove(h, N, p):
    """Correlation change when h[p] goes 1->0"""
    s = np.arange(-(N-1), N)
    d = np.zeros(2*N-1)
    ps = p + s
    v1 = (ps >= 0) & (ps < N)
    d[v1] -= (1.0 - h[ps[v1]])
    pm = p - s
    v2 = (pm >= 0) & (pm < N)
    d[v2] += h[pm[v2]]
    return d

def delta_q_add(h, N, q):
    """Correlation change when h[q] goes 0->1"""
    s = np.arange(-(N-1), N)
    d = np.zeros(2*N-1)
    qs = q + s
    v1 = (qs >= 0) & (qs < N)
    d[v1] += (1.0 - h[qs[v1]])
    qm = q - s
    v2 = (qm >= 0) & (qm < N)
    d[v2] -= h[qm[v2]]
    return d

    return delta_p_remove(h, N, p) + delta_q_add(h, N, q)

def continuous_subgrad(h, N, deadline, beta0=50, top_k=30):
    """Subgradient method with log-sum-exp approximation"""
    h = h.astype(float).copy()
    h = proj_sum(h, N/2.0)
    beta = beta0
    lr = 1.0 / np.sqrt(1.0)
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    center = N - 1
    
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        
        # Subgradient of log-sum-exp
        idx = np.argpartition(corr, -top_k)[-top_k:]
        g = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                g += (1.0 - 2.0*h)
            elif s > 0:
                g[:N-s] += (1.0 - h[s:])
                g[s:] -= h[:N-s]
            else:
                s2 = -s
                g[s2:] += (1.0 - h[:N-s2])
                g[:N-s2] -= h[s2:]
        g /= top_k
        
        t += 1
        lr = 1.0 / np.sqrt(t)
        m = b1*m + (1-b1)*g
        v = b2*v + (1-b2)*g*g
        mh = m/(1-b1**t)
        vh = v/(1-b2**t)
        h = proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
        
        if t % 2000 == 0:
            beta = max(beta*0.95, 5)
    
    return best_h, best_pk

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h.astype(float), L)
    G = np.fft.fft(np.flip(1.0-h.astype(float)), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    h_bin = h_bin.copy()
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin==1)[0]
        idx = idx[np.random.permutation(len(idx))]
        h_bin[idx[:n1-target]] = 0
    elif n1 < target:
        idx = np.where(h_bin==0)[0]
        idx = idx[np.random.permutation(len(idx))]
        h_bin[idx[:target-n1]] = 1
    return h_bin

def sdelta(h, N, p, q):
    s = np.arange(-(N-1), N)
    d = np.zeros(2*N-1)
    ps = p + s
    v1 = (ps >= 0) & (ps < N)
    d[v1] -= (1.0 - h[ps[v1]])
    pm = p - s
    v2 = (pm >= 0) & (pm < N)
    d[v2] += h[pm[v2]]
    qs = q + s
    v3 = (qs >= 0) & (qs < N)
    d[v3] += (1.0 - h[qs[v3]])
    qm = q - s
    v4 = (qm >= 0) & (qm < N)
    d[v4] -= h[qm[v4]]
    return d

def anneal(h, N, deadline, T0=0.05, cool=0.9997, restart_gap=2500, mfrac=0.12):
    h = h.astype(float).copy()
    h = fix_count(h, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    T = T0
    no_imp = 0
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    ctr = N - 1
    while time.time() < deadline:
        if np.random.random() < mfrac and len(ones)>=2 and len(zeros)>=2:
            p1 = int(ones[np.random.randint(len(ones))])
            p2 = int(ones[np.random.randint(len(ones))])
            q1 = int(zeros[np.random.randint(len(zeros))])
            q2 = int(zeros[np.random.randint(len(zeros))])
            if p1==p2 or q1==q2 or p1 in(q1,q2) or p2 in(q1,q2):
                T *= cool
                no_imp += 1
                continue
            hs = h.copy()
            h[p1]=0; h[p2]=0; h[q1]=1; h[q2]=1
            nc = fft_corr(h)
            npk = float(nc.max())
            if npk-cur_pk < 0 or np.random.random() < np.exp(-(npk-cur_pk)/max(T,1e-10)):
                corr=nc; cur_pk=npk
                ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
                if npk < best_pk: best_pk=npk; best_h=h.copy(); no_imp=0
                else: no_imp+=1
            else:
                h=hs; no_imp+=1
        else:
            # Peak-biased selection: 30% chance to pick p from top-peak contributors
            if np.random.random() < 0.30:
                topk = int(np.argmax(corr))
                sk = topk - ctr
                if sk > 0:
                    cands = np.where((h[sk:]==1) & (h[:N-sk]==0))[0]
                    if len(cands)>0: p = int(cands[np.random.randint(len(cands))])
                    else: p = int(ones[np.random.randint(len(ones))])
                else:
                    cands = np.where((h[:abs(sk)]==0) & (h[abs(sk):]==1))[0]
                    if len(cands)>0: p = int(cands[np.random.randint(len(cands))])
                    else: p = int(ones[np.random.randint(len(ones))])
            else:
                p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            d = sdelta(h, N, p, q)
            nc = corr + d
            npk = float(nc.max())
            if npk-cur_pk < 0 or np.random.random() < np.exp(-(npk-cur_pk)/max(T,1e-10)):
                h[p]=0; h[q]=1; corr=nc; cur_pk=npk
                ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
                if npk < best_pk: best_pk=npk; best_h=h.copy(); no_imp=0
                else: no_imp+=1
            else:
                no_imp+=1
        T *= cool
        if no_imp > restart_gap:
            h=best_h.copy(); corr=fft_corr(h); cur_pk=best_pk
            ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
            no_imp=0; T=max(T,0.025)
    return best_h, best_pk

def greedy_peak(h, N, max_passes=60, ntry=120):
    h = h.astype(float).copy()
    h = fix_count(h, N)
    corr = fft_corr(h)
    cur_pk = float(corr.max())
    ctr = N - 1
    for _ in range(max_passes):
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        best_d = 0; best_p=-1; best_q=-1; best_nc=None
        for _t in range(ntry):
            # Bias toward peak-reducing swaps
            if np.random.random() < 0.4:
                topidx = np.argpartition(corr, -5)[-5:]
                ti = int(topidx[np.random.randint(len(topidx))])
                sk = ti - ctr
                if sk > 0 and len(h)-sk > 0:
                    cands_p = np.where((h[sk:]==1) & (h[:N-sk]==0))[0]
                elif sk < 0:
                    cands_p = np.where((h[:abs(sk)]==1) & (h[abs(sk):]==0))[0]
                    sk = -sk
                else:
                    cands_p = ones
                if len(cands_p) > 0:
                    p = int(cands_p[np.random.randint(len(cands_p))])
                else:
                    p = int(ones[np.random.randint(len(ones))])
            else:
                p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            d = sdelta(h, N, p, q)
            nc = corr + d
            npk = float(nc.max())
            dd = npk - cur_pk
            if dd < best_d:
                best_d=dd; best_p=p; best_q=q; best_nc=nc
        if best_p >= 0:
            h[best_p]=0; h[best_q]=1; corr=best_nc; cur_pk=float(corr.max())
        else:
            break
    return h, cur_pk

def cont_opt(h, N, deadline, topk=40, lr0=0.004):
    h = h.astype(float).copy()
    h = proj_sum(h, N/2.0)
    m = np.zeros(N); v = np.zeros(N)
    b1,b2,eps = 0.85,0.999,1e-8
    t=0; best_pk=float(np.max(fft_corr(h))); best_h=h.copy()
    ctr = N-1
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk: best_pk=pk; best_h=h.copy()
        idx = np.argpartition(corr, -topk)[-topk:]
        g = np.zeros(N)
        for ai in idx:
            s = int(ai - ctr)
            if s == 0:
                g += (1.0 - 2.0*h)
            elif s > 0:
                g[:N-s] += (1.0-h[s:]); g[s:] -= h[:N-s]
            else:
                s2=-s; g[s2:] += (1.0-h[:N-s2]); g[:N-s2] -= h[s2:]
        g /= topk
        t+=1
        lr = lr0/np.sqrt(t+1)
        m=b1*m+(1-b1)*g; v=b2*v+(1-b2)*g*g
        mh=m/(1-b1**t); vh=v/(1-b2**t)
        h = proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
    return best_h, best_pk
