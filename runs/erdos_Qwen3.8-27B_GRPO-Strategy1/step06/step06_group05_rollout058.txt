import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def compute_grad(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    g1 = np.where(vp, 1.0 - h[pp_c], 0.0)
    g2 = np.where(vm, h[pm_c], 0.0)
    return g1 - g2

def to_binary(h, N):
    nh = N // 2
    idx = np.argsort(-h)
    b = np.zeros(N)
    b[idx[:nh]] = 1.0
    return b

def smooth_opt(h, N, deadline, k=5, lr0=0.04, n_steps=8000):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    target = N / 2.0
    dx = 2.0 / N
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        c = fft_corr(h) * dx
        c5 = float(c.max())
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        top_idx = np.argpartition(c, -k)[-k:]
        G = compute_grad(h, N)
        grad = G[:, top_idx].sum(axis=1) * dx
        grad -= grad.mean()
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        if t % 800 == 0:
            lr *= 0.92
    return best_h, best_c5

    T = T0
    stall = 0
    for sweep in range(max_sweeps):

            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
        stall += 1
        if T == 0 and stall > 120:
            break
        if T > 0 and stall > 500:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        bd, bpq = 0.0, None
        do_two = (two_swap_every > 0 and sweep % two_swap_every == 37 and len(ones) >= 2 and len(zeros) >= 2)
        if do_two:
            for i_p1 in range(0, len(ones), max(1, len(ones)//80)):
                p1 = int(ones[i_p1])
                for i_p2 in range(i_p1+1, len(ones), max(1, len(ones)//80)):
                    p2 = int(ones[i_p2])
                    base = c - D[p1, :] - D[p2, :]
                    cand = base[None, :] - D[zeros, :]
                    npk = cand.max(axis=1)
                    mi = int(np.argmin(npk))
                    d = float(npk[mi]) - P
                    if d < bd:
                        bd = d
                        bpq = (p1, p2, int(zeros[mi]), -1)
        else:
            for p in ones:
                base = c - D[p, :]
                cand = base[None, :] - D[zeros, :]
                npk = cand.max(axis=1)
                mi = int(np.argmin(npk))
                d = float(npk[mi]) - P
                if d < bd:
                    bd, bpq = d, (int(p), -1, int(zeros[mi]), -1)
        if bpq is not None:
            accept = False
            if bd < -1e-12:
                accept = True
            elif T > 0 and np.random.random() < np.exp(min(0, bd / max(T * 2.0 / N, 1e-10))):
                accept = True
            if accept:
                if len(bpq) >= 4 and bpq[3] == -1 and bpq[1] >= 0:
                    h[bpq[0]] = 0.0
                    h[bpq[1]] = 0.0
                    h[bpq[2]] = 1.0
                    q2 = int(zeros[np.argmin(D[zeros, :].sum(axis=1) - D[zeros[bpq[2]], :])])
                    h[q2] = 1.0
                else:
                    h[bpq[0]] = 0.0
                    h[bpq[2]] = 1.0
                stall = 0
            if T > 0:
                T *= cool
                if T < 1e-12:
                    T = 0
    return best_h, best_c
