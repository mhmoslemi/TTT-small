def subgradient(h, N, tol=1e-6):
    """Compute subgradient of max_k corr[k] at point h."""
    corr = fft_corr(h)
    pk = np.max(corr)
    # Active set: indices where corr is within tol of max
    active = np.where(corr >= pk - tol)[0]
    n_active = len(active)
    
    grad = np.zeros(N)
    center = N - 1
    for ai in active:
        L = int(ai - center)
        g = linear_grad(L, h, N)
        grad += g
    return grad / n_active, pk

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def linear_grad(L, h, N):
    """Gradient of corr at lag L with respect to h."""
    g = np.zeros(N)
    if L == 0:
        return 1.0 - 2.0 * h
    elif L > 0:
        # j in [0, L-1]: -h[j+L]
        if L < N:
            g[:L] = -h[L:]
        # j in [L, N-1-L]: (1-h[j-L]) - h[j+L]
        if N - L > L:

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

# FFT correlation
def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

# Gradient of linear corr at lag L
def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0*h
    elif L > 0:
        g[L:] += 1.0 - h[:N-L]
        g[:N-L] -= h[L:]
    else:
        M = -L
        g[:N-M] += 1.0 - h[M:]
        g[M:] -= h[:N-M]
    return g

# Projection
def proj01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo+hi)/2
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

# Smoothed objective for L-BFGS
def smooth_c5(h, T=30.0):
    c = fft_corr(h)
    mx = c.max()
    return mx + np.log(np.sum(np.exp(T*(c-mx))))/T

def smooth_c5_grad(h, T=30.0):

    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T*(c-mx))
    w = e / e.sum()
    g = np.zeros(N)
    ctr = N-1
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * lin_grad(i-ctr, h, N)
    return g

def lbfgs_phase(h, N, deadline, T=30.0, max_iter=200):

    target = float(N)//2
    best_h = h.copy()
    best_val = smooth_c5(h, T)
    
    # Use penalty method for the sum constraint

    for outer in range(5):
        if time.time() > deadline:
            break
        def obj(x):
            c = fft_corr(x)
            mx = c.max()
            val = mx + np.log(np.sum(np.exp(T*(c-mx))))/T
            s = x.sum() - target
            return val + lam*0.5*s*s

            c = fft_corr(x)
            mx = c.max()
            e = np.exp(T*(c-mx))

            g = np.zeros(N)

            g += lam*(x.sum()-target)*np.ones(N)
            return g
        
        res = scipy_minimize(obj, h, jac=grad, method='L-BFGS-B',

            v = smooth_c5(h, T)
            if v < best_val:

import numpy as np
import time
from scipy.optimize import minimize as sp_min

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def lin_grad(L, h, N):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:N]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def proj01(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, T, target, lam):
    N = len(h)
    c = fft_corr(h)
    mx = c.max()
    e = np.exp(T * (c - mx))
    w = e / e.sum()
    val = mx + np.log(e.sum()) / T
    g = np.zeros(N)
    ctr = N - 1
    for i in range(len(c)):
        if w[i] > 1e-9:
            g += w[i] * lin_grad(i - ctr, h, N)
    s = h.sum() - target
    val += 0.5 * lam * s * s
    g += lam * s * np.ones(N)
    return val, g

def lbfgs_smooth(h, N, deadline, T=30.0, max_outer=4):
    h = h.astype(float).copy()
    target = float(N) // 2
    best_h = h.copy()
    best_v = smooth_obj_and_grad(h, T, target, 0.0)[0]
    lam = 5.0
    for _ in range(max_outer):
        if time.time() > deadline:
            break
        res = sp_min(smooth_obj_and_grad, h, args=(T, target, lam),
                     jac=True, method='L-BFGS-B', bounds=[(0, 1)] * N,
                     options={'maxiter': 150, 'maxfun': 400, 'ftol': 1e-14})
        h = proj01(res.x, target)
        v = smooth_obj_and_grad(h, T, target, 0.0)[0]
        if v < best_v:
            best_v = v
            best_h = h.copy()
        lam *= 2.5
    return best_h, best_v * (2.0 / N)

def subgrad_ne(h, N, deadline, alpha0=0.04, beta=0.9, atol=1e-5):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = np.max(fft_corr(h))
    m = np.zeros(N)
    alpha = alpha0
    it = 0
    while time.time() < deadline and it < 8000:
        y = np.clip(h + m, 0, 1)
        corr = fft_corr(y)
        pk = np.max(corr)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        active = np.where(corr >= pk - atol)[0]
        grad = np.zeros(N)
        ctr = N - 1
        for ai in active:
            grad += lin_grad(int(ai - ctr), y, N)
        grad /= len(active)
        h_new = proj01(y - alpha * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            alpha *= 0.96
            beta = min(0.98, 0.88 + 0.0002 * it)
    return best_h, best_pk * (2.0 / N)

def bin_search(h, N, deadline, max_iters=400, ntri=20, anneal=0.0):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:

    best_c = np.max(fft_corr(h))
    best_h = h.copy()
    stall = 0
    temp = anneal
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        improved = False
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            improved = True
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd, bi, bj = 0.0, -1, -1
        for _ in range(ntri):
            i = np.random.randint(len(ones))
            j = np.random.randint(len(zeros))
            oi, zi = int(ones[i]), int(zeros[j])
            h[oi], h[zi] = 0.0, 1.0
            nP = float(np.max(fft_corr(h)))
            d = P - nP
            if d > bd + temp:

            h[oi], h[zi] = 1.0, 0.0
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
            improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
        temp = max(0.0, anneal * (1.0 - it / float(max_iters)))
    return best_h, best_c
