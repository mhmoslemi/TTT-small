def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

    return float(np.max(corr)) * (2.0 / N)

        max_c = corr[top_idx].max()
        w = np.exp((corr[top_idx] - max_c) / temp)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.ones(k) / k
        
        grad = np.zeros(N)
        for wi, ai in zip(w, top_idx):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                seg = N - L
                grad[L:] += wi * (1.0 - h[:seg])
                grad[:seg] -= wi * h[L:]
            elif L < 0 and L > -N:
                M = -L
                seg = N - M
                grad[:seg] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:seg]
        
        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h_new, target)
        
        # Schedule
        if t % 100 == 0:
            temp = max(0.005, temp * 0.985)
            lr = max(0.002, lr * 0.997)
        
        # Restart
        if stall > stall_limit:
            h = best_h + np.random.randn(N) * 0.015
            h = project_sum(h, target)
            m = np.zeros(N)
            v = np.zeros(N)
            t = 0
            stall = 0
            temp = max(0.05, temp * 3)
            lr = max(lr, lr0 * 0.5)
    
    return best_h, best_val * (2.0 / N)

                zeros = np.where(h == 0)[0]

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))[:2 * N - 1]

def compute_c5(h, N):
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    lo, hi = (0.0, diff) if diff > 0 else (diff, 0.0)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_of_corr(h, N, top_idx, weights):
    """Compute weighted subgradient at given lags."""
    grad = np.zeros(N)
    center = N - 1
    for wi, ai in zip(weights, top_idx):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        elif -N < L < 0:
            M = -L
            seg = N - M
            grad[:seg] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:seg]
    return grad
