def project_to_binary_sum(h, N):
    """Project continuous h to binary with sum = N/2"""
    target = N / 2.0
    s = float(h.sum())
    if s == target:
        # Already binary? Check
        if np.allclose(h, np.round(h)):
            return np.round(h).astype(float)
    # Threshold with level set
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        nb = np.sum(h > mid)
        if nb > target:
            lo = mid
        else:
            hi = mid
    result = np.where(h > (lo+hi)*0.5, 1.0, 0.0)
    # Fix exact count
    cnt = int(result.sum())
    if cnt != target:
        # Adjust
        idx_below = np.where(result == 0)[0]
        idx_above = np.where(result == 1)[0]
        if cnt < target:
            # Need more 1s
            n_add = int(target) - cnt
            # Add from highest values among 0s
            scores = h[idx_below]
            order = np.argsort(-scores)[:n_add]
            result[idx_below[order]] = 1.0
        else:
            n_rem = cnt - int(target)
            scores = h[idx_above]
            order = np.argsort(scores)[:n_rem]
            result[idx_above[order]] = 0.0
    return result

def compute_D(h, N):
    """Compute the change matrix D[p, k] = change in c[k] when h[p] flips 1->0"""
    # If h[p] flips from 1 to 0, the change in c[k] for each k:
    # c[k] = sum_i h[i]*(1-h)[valid indices at lag k]
    # Removing h[p]=1: c[k] decreases by (1-h)[p shifted by k] if valid
    # This is complex to vectorize perfectly, let me use a simpler approach
    # Actually for binary h, flipping h[p] from 1 to 0:
    # c_new[k] = c[k] - (1-h[valid companion at shift k for position p])
    # The companion of p at shift k is the index j such that the pair (p, j) contributes to c[k]
    # In the correlation, position p in h pairs with position p+k in (1-h) (for lag k)
    # So the contribution of h[p]=1 to c[k] is (1-h[p+k]) if p+k is valid (0 <= p+k < N)
    # After flip: contribution becomes 0 (since h[p]=0)
    # Change: -（1-h[p+k]) if p+k valid, 0 otherwise
    # Also, for the companion in h: if some position q has q+k = p, i.e., q = p-k,
    # then h[q]*(1-h[p]) was contributing. After flip, (1-h[p]) changes from 0 to 1.
    # So change for that pair: +h[q] if q valid
    # Net change in c[k]: +h[p-k] (if valid) - (1-h[p+k]) (if valid)
    # For binary: h[p-k] is 0 or 1, (1-h[p+k]) is 0 or 1
    
    idx = np.arange(2*N - 1)
    # lag k = idx - (N-1), ranges from -(N-1) to N-1
    lags = idx - (N - 1)
    # For position p, contribution to c at lag k involves:
    # - h[p] * (1-h)[p+k] (p in h, p+k in 1-h)
    # - h[p-k] * (1-h)[p] (p-k in h, p in 1-h)
    # When we flip h[p] from 1 to 0:
    # Term 1 changes from (1-h[p+k]) to 0: delta = -(1-h[p+k]) if 0<=p+k<N
    # Term 2 changes from h[p-k]*(1-h[p])=h[p-k]*0 to h[p-k]*1: delta = +h[p-k] if 0<=p-k<N
    
    # For all p at once, compute D[p, k_idx]:
    p = np.arange(N)[:, None]  # N x 1
    k = lags[None, :]  # 1 x (2N-1)
    # p+k valid: 0 <= p+k < N
    pk = p + k
    pm = p - k
    valid_pk = (pk >= 0) & (pk < N)
    valid_pm = (pm >= 0) & (pm < N)
    # (1-h[p+k]) where valid
    pk_clipped = np.clip(pk, 0, N-1)
    pm_clipped = np.clip(pm, 0, N-1)
    one_minus_h_pk = np.where(valid_pk, 1.0 - h[pk_clipped], 0.0)
    h_pm = np.where(valid_pm, h[pm_clipped], 0.0)
    # D = -(1-h[p+k]) + h[p-k] (net change in c[k] when h[p] goes 1->0)
    D = h_pm - one_minus_h_pk
    return D

def binary_local_search(h, N, deadline, max_sweeps=1000, sample_n=200, top_k=5):
    """Improved binary local search with multi-peak awareness"""
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    sweep = 0
    while sweep < max_sweeps:
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        # Use top-k peaks for scoring
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        # Score: weighted combination of peaks

def subgradient_step(h, N, deadline, n_iter=2000, lr0=0.01):
    """Sub-gradient descent on the continuous relaxation, focusing on top peaks"""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    k_top = 10
    it = 0
    while time.time() < deadline:
        # Compute correlation
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        # Sub-gradient: sum of gradients of top-k peaks
        top_i = np.argpartition(c, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top_i:
            L = int(ai - center)
            if L == 0:
                # c[center] = sum h[i]*(1-h[i]), grad w.r.t. h[i] = (1-h[i]) - h[i] = 1-2h[i]
                grad += (1.0 - 2.0 * h)
            elif L > 0:
                # c[center+L] = sum_{i=L}^{N-1} h[i]*(1-h[i-L])
                # grad[i] for i in [L, N-1]: (1-h[i-L])
                # grad[i] for i in [0, N-1-L]: -h[i+L] (from h[i+L]*(1-h[i]) term)
                # Wait: c = sum_{i=L}^{N-1} h[i]*(1-h[i-L])
                # d/dh[i] for i in [L, N-1]: (1-h[i-L])
                # d/dh[j] for j in [0, N-1-L]: from term h[j+L]*(1-h[j]), = -h[j+L]
                grad[L:] += (1.0 - h[:N-L])
                grad[:N-L] -= h[L:]
            else:
                M = -L  # M > 0
                # c[center-M] = sum_{i=0}^{N-1-M} h[i+M]*(1-h[i])
                # d/dh[i] for i in [M, N-1]: (1-h[i-M])
                # d/dh[j] for j in [0, N-1-M]: -h[j+M]
                grad[M:] += (1.0 - h[:N-M])
                grad[:N-M] -= h[M:]
        grad /= k_top
        # Nesterov update with projection
        y = h + beta * m
        h_new = y - lr * grad
        # Project to [0,1] and fix sum
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum via bisection on threshold
        s = h_new.sum()
        if abs(s - target) > 1e-8:
            adj = (target - s) / N
            h_new = np.clip(h_new + adj, 0.0, 1.0)
        m = h_new - h
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
        if it % 500 == 0:
            beta = min(0.97, 0.85 + it * 0.0002)
    return best_h, best_pk * (2.0/N)

def spectral_smooth(h, N, n_steps=5, alpha=0.3):
    """Apply spectral low-pass smoothing to h"""
    h = h.astype(float).copy()
    H = np.fft.rfft(h)
    freqs = np.fft.rfftfreq(N)
    # Low-pass: attenuate high frequencies
    cutoff = 3.0 / N  # Keep first few frequencies
    mask = np.exp(-alpha * (freqs / cutoff)**2)
    H *= mask
    h_smooth = np.fft.irfft(H, N)
    return h_smooth
