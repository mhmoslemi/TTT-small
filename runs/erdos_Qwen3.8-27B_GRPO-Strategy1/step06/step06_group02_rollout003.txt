        else:

        else:
            weights = np.ones(len(top_idx)) / len(top_idx)

        center = N - 1

            L = int(ai - center)
            if L > 0:
                grad[L:] += w * (1.0 - y[:N-L])
                grad[:N-L] -= w * y[L:]
            elif L < 0:
                M = -L
                grad[:N-M] += w * (1.0 - y[M:])
                grad[M:] -= w * y[:N-M]
            else:
                grad += w * (1.0 - 2.0 * y)
        
        # Add variance regularization
        if var_weight > 0:
            # Gradient of var(c) w.r.t. h is more complex, so skip
            pass
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        
        # Decay learning rate
        if it % 50 == 0:
            lr *= 0.997
            beta = min(0.97, 0.85 + it * 0.0005)
    
    return best_h, best_c5

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h0, N, deadline, k_act=60, lr0=0.04):

    target = N / 2.0
    dx = 2.0 / N

    m = np.zeros(N)

    beta = 0.88
    it = 0

        y = h + m
        c = fft_corr(y)
        pk = float(c.max())

            best_val = pk
            best_h = y.copy()

        k = min(k_act, 2 * N - 1)

        c_top = c[top]
        c_med = np.median(c_top)
        w = np.maximum(c_top - c_med, 1e-6)
        w = w / w.sum()

        for i in range(k):

            L = int(top[i] - center)

        h = h_new
        it += 1
        if it % 40 == 0:

            beta = min(0.97, 0.85 + it * 0.0004)

    no_imp = 0
    center = N - 1
    k_range = np.arange(-(N-1), N)

        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:

            no_imp = 0
        else:
            no_imp += 1

        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:

        if len(ones) > sample_n:

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h0, N, deadline, k_act=70, lr0=0.04):
    h = h0.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.87
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        if pk < best_val - 1e-15:
            best_val = pk
            best_h = y.copy()
        k = min(k_act, 2 * N - 1)
        top = np.argpartition(c, -k)[-k:]
        c_top = c[top]
        c_med = np.median(c_top)
        w = np.maximum(c_top - c_med, 1e-8)
        w = w / w.sum()
        grad = np.zeros(N)
        for i in range(k):
            wi = w[i]
            L = int(top[i] - center)
            if L > 0:
                grad[L:] += wi * (1.0 - y[:N-L])
                grad[:N-L] -= wi * y[L:]
            elif L < 0:
                M = -L
                grad[:N-M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N-M]
            else:
                grad += wi * (1.0 - 2.0 * y)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 30 == 0:
            lr *= 0.9985
            beta = min(0.97, 0.84 + it * 0.0005)
    return best_h, best_val * dx

def binary_local(h, N, deadline, max_sweeps=500, sample_n=80):
    h = np.array(h, dtype=float)
    best_c = float(np.max(fft_corr(h)))
    no_imp = 0
    center = N - 1
    k_range = np.arange(-(N-1), N)
    ncorr = 2 * N - 1
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            no_imp = 0
        else:
            no_imp += 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if len(ones) > sample_n:
            p_list = ones[np.random.choice(len(ones), sample_n, replace=False)]
        else:
            p_list = ones
        if len(zeros) > sample_n:
            q_list = zeros[np.random.choice(len(zeros), sample_n, replace=False)]
        else:
            q_list = zeros
        nq = len(q_list)
        pos_m = q_list[:, None].astype(int) - k_range[None, :]
        pos_p = q_list[:, None].astype(int) + k_range[None, :]
        Dq = np.zeros((nq, ncorr))
        vm = (pos_m >= 0) & (pos_m < N)
        vp = (pos_p >= 0) & (pos_p < N)
        Dq[vm] = 1.0 - h[np.clip(pos_m, 0, N-1)[vm]]
        Dq[vp] -= h[np.clip(pos_p, 0, N-1)[vp]]
        best_score = P + 1.0
        best_pq = None
        for p in p_list:
            pm = p - k_range
            pp = p + k_range
            dp = np.zeros(ncorr)
            vm_ = (pm >= 0) & (pm < N)
            vp_ = (pp >= 0) & (pp < N)
            dp[vm_] = -(1.0 - h[np.clip(pm, 0, N-1)[vm_]])
            dp[vp_] += h[np.clip(pp, 0, N-1)[vp_]]
            base = c + dp
            scores = (base[None, :] + Dq).max(axis=1)
            mi = int(np.argmin(scores))
            if scores[mi] < best_score:
                best_score = float(scores[mi])
                best_pq = (int(p), int(q_list[mi]))
        temp = 0.015 * np.exp(-sw / 100.0)
        do_kick = (best_pq is None) or (no_imp > 70 and np.random.random() < temp)
        if do_kick:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
        else:
            p, q = best_pq
        h[p], h[q] = 0.0, 1.0
        if no_imp > 400:
            break
    return h, best_c * (2.0 / N)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h > 0.5)[0]
        idx = idx[np.argsort(np.random.random(len(idx)))]
        h[idx[nh:]] = 0.0
        h[idx[:nh]] = 1.0
    elif n1 < nh:
        idx = np.where(h < 0.5)[0]
        idx = idx[np.argsort(np.random.random(len(idx)))]
        h[idx[:nh - n1]] = 1.0
    else:
        h[h > 0.5] = 1.0
        h[h <= 0.5] = 0.0
    return h
