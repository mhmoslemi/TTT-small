import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, nh, rng):
    h = (h > 0.5).astype(float).copy()
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx, n1 - nh, replace=False)] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx, nh - n1, replace=False)] = 1
    return h

def hill_climb(h, N, deadline, rng, sample_sz=100, chunk=10, max_no_improve=40):
    """Hill climbing with single swaps. Returns (h, c5)."""
    h = h.copy()
    corr = fft_corr(h)
    s = 2.0 * h - 1.0
    c5 = float(corr.max()) * (2.0 / N)
    
    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i+N] = s[i]
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_imp = 0
    
    while time.time() < deadline and no_imp < max_no_improve:
        if len(ones) < 1 or len(zeros) < 1:
            break
        n_s = min(sample_sz, len(ones))
        p_sel = rng.choice(ones, n_s, replace=False)
        cur_mx = float(corr.max())
        best_d = 0.0
        best_p, best_q = -1, -1
        
        for st in range(0, n_s, chunk):
            ed = min(st + chunk, n_s)
            dp = deltas[p_sel[st:ed]]
            dq = deltas[zeros]
            mvals = (corr[None, None, :] + dp[:, None, :] - dq[None, :, :]).max(axis=2)
            for ii in range(ed - st):
                mi = int(mvals[ii].argmin())
                d = cur_mx - mvals[ii, mi]
                if d > best_d:
                    best_d = d
                    best_p = int(p_sel[st + ii])
                    best_q = int(zeros[mi])
        
        if best_p < 0:
            no_imp += 1
            continue
        
        h[best_p], h[best_q] = 0.0, 1.0
        corr += deltas[best_p] - deltas[best_q]
        s[best_p], s[best_q] = -s[best_p], -s[best_q]
        deltas[best_p, :] = 0; deltas[best_p, best_p:best_p+N] = s[best_p]
        deltas[best_q, :] = 0; deltas[best_q, best_q:best_q+N] = s[best_q]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        no_imp = 0
    
    c5 = float(corr.max()) * (2.0 / N)
    return h, c5

def perturb(h, n_swaps, rng):
    """Apply n_swaps random 1↔0 swaps."""
    h = h.copy()
    N = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_swaps):
        p = int(rng.choice(ones))
        q = int(rng.choice(zeros))
        h[p], h[q] = 0.0, 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h

        if c_r < best_c5:

        x_old = np.linspace(0, 2, len(best_h), endpoint=False)

    # Phase 4: Golden ratio / structured init
    alpha = (np.sqrt(5.0) - 1.0) / 2.0
    for N_g in [best_h.shape[0], best_h.shape[0] + 4]:
        if time.time() > deadline - 40:

        x = np.linspace(0, 2, N_g, endpoint=False)
        h_g = ((x * 2.0 * alpha) % 1.0 < 0.5).astype(float)
        h_g = make_binary(h_g, N_g // 2, rng)
        c_g = compute_c5(h_g, N_g)
        if c_g < best_c5 + 0.02:
            sub_dl = min(deadline - 5, time.time() + 40)
