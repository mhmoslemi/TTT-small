        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            temp *= temp_rate
        if no_improve > 600:
            break
        if no_improve < 30:
            bs = 1
        elif no_improve < 100:
            bs = int(np.random.choice([1, 2, 3]))
        elif no_improve < 250:
            bs = int(np.random.choice([2, 3, 5, 7]))
        else:
            bs = int(np.random.choice([3, 5, 8, 12, 16]))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if bs == 1:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            continue
        # Find blocks
        one_st = []
        for i in range(len(ones) - bs + 1):
            if int(ones[i + bs - 1]) - int(ones[i]) == bs - 1:
                one_st.append(int(ones[i]))
        zero_st = []
        for i in range(len(zeros) - bs + 1):
            if int(zeros[i + bs - 1]) - int(zeros[i]) == bs - 1:
                zero_st.append(int(zeros[i]))
        if len(one_st) < 1 or len(zero_st) < 1:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            continue
        n_try = min(6, len(one_st) * len(zero_st))
        best_new_P = P + 1.0
        best_move = None
        for _ in range(n_try):
            op = int(one_st[np.random.randint(len(one_st))])
            zp = int(zero_st[np.random.randint(len(zero_st))])
            if op + bs > zp and zp + bs > op:
                continue
            h[op:op + bs] = 0.0
            h[zp:zp + bs] = 1.0
            new_P = float(np.max(fft_corr(h)))
            h[op:op + bs] = 1.0
            h[zp:zp + bs] = 0.0
            if new_P < best_new_P:
                best_new_P = new_P
                best_move = (op, zp)
        if best_move is not None:
            delta = best_new_P - P
            if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-4)):
                h[best_move[0]:best_move[0] + bs] = 0.0
                h[best_move[1]:best_move[1] + bs] = 1.0
            else:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
    h = best_h.copy()
    return h, float(np.max(fft_corr(h)))

def nesterov_soft(h, N, deadline, k_top=30, power=5.0, lr0=0.02, max_iter=8000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline and it < max_iter:
        y = h + m
        yc = np.clip(y, 0, 1)
        corr = fft_corr(yc)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = yc.copy()
        top_idx = np.argsort(corr)[::-1][:k_top]
        grad = np.zeros(N)
        center = N - 1
        for ai in top_idx:
            L = int(ai - center)
            w = power * (corr[ai] ** (power - 1))
            if L == 0:
                grad += w * (1.0 - 2.0 * yc)
            elif L > 0:
                grad[L:] += w * (1.0 - yc[:N - L])
                grad[:N - L] -= w * yc[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += w * (1.0 - yc[M:])
                    grad[M:] -= w * yc[:N - M]
        h_new = project_sum(yc - lr * grad / k_top, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr *= 0.992
            beta = min(0.95, 0.82 + it * 0.0002)
    return best_h, float(np.max(fft_corr(best_h))) * (2.0 / N)

def gen_thue_morse(N):
    tm = np.array([bin(i).count('1') % 2 for i in range(N)], dtype=float)
    return fix_sum_binary(tm, N)

def gen_random_balanced(N, seed=0):
    rng = np.random.RandomState(seed)
    x = rng.random(N)
    x = (x * 2).clip(0, 1)
    return fix_sum_binary((x > 0.5).astype(float), N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    candidates = []
    candidates.append(fix_sum_binary((h0 > 0.5).astype(float), N0))
    candidates.append(gen_thue_morse(N0))
    candidates.append(gen_random_balanced(N0, 7))
    candidates.append(gen_random_balanced(N0, 99))
    # Phase 1: Block-swap on each candidate with time budget
    t_rem = deadline - 350
    per_cand = max(60, (t_rem - t0) // max(len(candidates), 1))
    best_h = candidates[0]
    best_c5 = compute_c5(best_h, N0)
    for ci, cand in enumerate(candidates):
        tc = compute_c5(cand, N0)
        if tc < best_c5:
            best_c5, best_h = tc, cand.copy()
        if time.time() > deadline - 300:
            break
        d_i = min(deadline - 300, time.time() + per_cand)
        hi, _ = block_swap_search(cand, N0, d_i, max_iters=2500, temp0=0.3, temp_rate=0.9995)
        tc = compute_c5(hi, N0)
        if tc < best_c5:
            best_c5, best_h = tc, hi.copy()
    # Phase 2: Nesterov soft
    if time.time() < deadline - 250:
        hc = best_h.astype(float) + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 200, time.time() + 120)
        hn, c5n = nesterov_soft(hc, N0, d2, k_top=25, power=6.0, lr0=0.018)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
    # Phase 3: Re-binarize + block search
    if time.time() < deadline - 150:
        hb = fix_sum_binary((best_h > 0.5).astype(float), N0)
        d3 = min(deadline - 120, time.time() + 160)
        h3, _ = block_swap_search(hb, N0, d3, max_iters=3000, temp0=0.2, temp_rate=0.9997)
        c3 = compute_c5(h3, N0)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()
    # Phase 4: Upscale to 2x
    N_up = N0 * 2
    if time.time() < deadline - 80:
        xb = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, N_up, endpoint=False)
        hi2 = np.interp(xn, xb, best_h)
        h_up = fix_sum_binary((hi2 > 0.5).astype(float), N_up)
        d4 = min(deadline - 40, time.time() + 130)
        h4, _ = block_swap_search(h_up, N_up, d4, max_iters=1200, temp0=0.3, temp_rate=0.9995)
        c4 = compute_c5(h4, N_up)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()
    # Phase 5: Nesterov at best res
    Nf = len(best_h)
    if time.time() < deadline - 15:
        hF = best_h.astype(float) + np.random.randn(Nf) * 0.025
        hF = project_sum(hF, Nf / 2.0)
        d5 = min(deadline - 5, time.time() + 80)
        hF, c5F = nesterov_soft(hF, Nf, d5, k_top=40, power=8.0, lr0=0.012)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
    # Phase 6: Final binary polish
    if time.time() < deadline - 2:
        hF2 = fix_sum_binary((best_h > 0.5).astype(float), len(best_h))
        d6 = deadline - 1
        hF2, _ = block_swap_search(hF2, len(hF2), d6, max_iters=300, temp0=0.1)
        c6 = compute_c5(hF2, len(hF2))
        if c6 < best_c5:
            best_c5, best_h = c6, hF2
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
