import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N, rng):
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        idx = idx[rng.permutation(len(idx))][nh:]
        h[idx] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        idx = idx[rng.permutation(len(idx))][:nh - n1]
        h[idx] = 1
    return h

def swap_delta(h, p, q, top_s, N):
    center = N - 1
    ip = p + top_s
    valid_ip = (ip >= 0) & (ip < N)
    dr = np.where(valid_ip, 2.0 * h[np.clip(ip, 0, N - 1)] - 1.0, 0.0)
    iq = q + top_s
    valid_iq = (iq >= 0) & (iq < N)
    da = np.where(valid_iq, 1.0 - 2.0 * h[np.clip(iq, 0, N - 1)], 0.0)
    return dr + da

def binary_search_2swap(h, N, deadline, rng, max_sweeps=300, sample_n=100, top_m=20, anneal_temp=0.0):
    h = h.astype(float).copy()
    h = make_binary(h, N, rng)
    c = fft_corr(h)
    best_c = float(c.max())
    tabu = set()
    temp = anneal_temp
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_idx = np.argsort(c)[-top_m:]
        top_s = top_idx - (N - 1)
        best_new_P = P
        best_move = None
        n_p = min(sample_n, len(ones))
        n_q = min(sample_n, len(zeros))
        p_list = rng.choice(ones, n_p, replace=False) if n_p < len(ones) else ones
        q_list = rng.choice(zeros, n_q, replace=False) if n_q < len(zeros) else zeros
        for p in p_list:
            d_p = swap_delta(h, p, -1, top_s, N)
            for q in q_list:
                d_q = swap_delta(h, -1, q, top_s, N)
                new_vals = c[top_idx] + d_p + d_q
                nm = float(new_vals.max())
                key = (int(p), int(q))
                if nm < best_new_P and key not in tabu:
                    best_new_P = nm
                    best_move = key
        if best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
            tabu.add(best_move)
            if len(tabu) > 50:
                tabu = set(list(tabu)[-20:])
        elif anneal_temp > 0 and rng.random() < 0.1:
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        if sweep % 50 == 0:
            anneal_temp *= 0.85
    return h, best_c

def soft_max_opt(h, N, deadline, T_init=0.05, T_end=0.005, lr0=0.02, beta0=0.85, k=50):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h))) * dx
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    T = T_init
    it = 0
    n_it = 0
    while time.time() < deadline and n_it < 5000:
        n_it += 1
        y = h + m
        corr = fft_corr(y)
        P_raw = float(corr.max())
        P_soft = T * np.log(np.sum(np.exp(corr / T)))
        if P_raw * dx < best_pk:
            best_pk = P_raw * dx
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 60 == 0:
            lr *= 0.99
            beta = min(0.95, beta0 + it * 0.0002)
            T = max(T_end, T * 0.97)
    return best_h, best_pk

def structured_init(N, rng, pattern=0):
    x = np.linspace(0, 2, N, endpoint=False)
    if pattern == 0:
        h = (np.sin(2 * np.pi * 3 * x) > 0).astype(float)
    elif pattern == 1:
        h = (np.sin(2 * np.pi * 5 * x) * np.sin(2 * np.pi * 2 * x) > 0).astype(float)
    elif pattern == 2:
        h = (np.sin(2 * np.pi * 7 * x + np.pi / 4) > 0).astype(float)
    else:
        h = rng.rand(N)
    h = make_binary(h, N, rng)
    return h
