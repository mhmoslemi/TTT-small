        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1 = len(ones)
        nz = len(zeros)
        if n1 < 2 or nz < 2:
            break
        
        # Move selection
        if rng.random() < block_p and N > 20:

            q_block = None
            for _ in range(8):

                cb = np.arange(ps, min(ps + bs, N))

                    p_block = cb
                    break

                    q_block = cb
                    break

            q_block = np.array([int(zeros[rng.randint(nz)])])

def continuous_refine(h, N, deadline, rng, top_k=40):
    """Adam optimization in continuous space, then round."""
    h = h.astype(float).copy()
    target = N / 2.0
    # Project to [0,1] with correct sum
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) > 1e-10:
        # Binary search for offset
        lo, hi = -1.0, 1.0
        for _ in range(50):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    lr = 0.003
    center = N - 1
    best_pk = float(fft_corr(h).max())
    best_h = h.copy()
    
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Gradient from top-k shifts
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(N)
        for ai in idx:
            s = int(ai - center)
            if s == 0:
                grad[:] += (1.0 - 2.0 * h)
            elif s > 0:
                grad[:N-s] += (1.0 - h[s:])
                grad[s:] -= h[:N-s]
            else:
                s2 = -s
                grad[s2:] += (1.0 - h[:N-s2])
                grad[:N-s2] -= h[s2:]
        grad /= top_k
        
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = h - lr * mh / np.sqrt(vh + eps)
        # Project
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target) > 1e-10:
            lo, hi = -1.0, 1.0
            for _ in range(40):
                mid = (lo + hi) * 0.5
                if np.clip(h - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h = np.clip(h - (lo+hi)*0.5, 0, 1)
        
        if t % 1000 == 0:
            lr *= 0.95
    
    return best_h, best_pk
