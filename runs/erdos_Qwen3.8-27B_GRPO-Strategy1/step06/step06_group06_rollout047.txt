import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h_bin, N, rng):
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin == 1)[0]
        remove = rng.choice(idx, n1 - target, replace=False)
        h_bin[remove] = 0
    elif n1 < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(idx, target - n1, replace=False)
        h_bin[add] = 1
    return h_bin

def softmax_grad(corr, beta):
    """Compute gradient of log-sum-exp(β*corr)/β with respect to corr."""
    m = float(corr.max())
    exp_bc = np.exp(beta * (corr - m))
    weights = exp_bc / float(exp_bc.sum())
    return weights / beta

def compute_grad(h, N, weights):
    """Compute gradient of weighted sum of correlations w.r.t. h."""
    center = N - 1
    grad = np.zeros(N)
    for k in range(-(N-1), N):
        if k == 0:
            # corr term: sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
            # d/dh[i] = 1 - 2*h[i]
            grad += weights[center] * (1.0 - 2.0 * h)
        elif k > 0:
            # corr[center+k] = sum_{i=0}^{N-1-k} h[i]*(1-h[i+k])
            # d/dh[i] for i < N-k: (1-h[i+k]) - h[i] (from h[i] term and -h[i+k] term)
            # Wait, d/dh[i] of h[i]*(1-h[i+k]) = (1-h[i+k])
            # d/dh[i+k] of h[i]*(1-h[i+k]) = -h[i]
            # So for j < N-k: grad[j] += (1-h[j+k])
            # And for j >= k: grad[j] += -h[j-k]
            w = weights[center + k]
            if w < 1e-10:
                continue
            n_valid = N - k
            grad[:n_valid] += w * (1.0 - h[k:])
            grad[k:] += -w * h[:n_valid]
        else:
            # k < 0, let j = -k > 0
            # corr[center-k] = corr[center+(-j)] ... 
            # Actually corr[center + k] where k < 0
            # = sum_{i=-k}^{N-1} h[i]*(1-h[i+k])
            # = sum_{i=j}^{N-1} h[i]*(1-h[i-j]) where j=-k
            # d/dh[i] for i >= j: (1-h[i-j])
            # d/dh[i-j] for i >= j: -h[i], i.e., d/dh[m] for m < N-j: -h[m+j]
            j = -k
            w = weights[center + k]
            if w < 1e-10:
                continue
            n_valid = N - j
            grad[j:] += w * (1.0 - h[:n_valid])
            grad[:n_valid] += -w * h[j:]
    return grad

def continuous_opt_lse(h, N, deadline, beta_start=20, beta_end=200, lr0=0.003):
    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    best = float(np.max(fft_corr(h)))
    best_h = h.copy()
    total_steps = max(1, int((deadline - time.time()) * 3))  # estimate
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best:
            best = pk
            best_h = h.copy()
        # Progressively increase beta
        frac = t / max(total_steps, 1)
        beta = beta_start + (beta_end - beta_start) * min(frac, 1.0)
        weights = softmax_grad(corr, beta)
        grad = compute_grad(h, N, weights)
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        lr = lr0 / (1 + t * 0.001)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 200 == 0 and t > 100:
            lr *= 0.9
    return best_h, best

def binary_anneal(h, N, deadline, T0=0.05, cool=0.9993, max_no_imp=600, multi_restart=2, rng=None):
    if rng is None:
        rng = np.random
    h = h.astype(float).copy()
    h = fix_count((h > 0.5).astype(float), N, rng)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    for restart in range(multi_restart):
        T = T0 if restart == 0 else T0 * 0.3
        no_imp = 0
        while time.time() < deadline and T > 0.0005:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            # Compute delta for swapping p (1→0) and q (0→1)
            ps = p + s_arr
            pm = p - s_arr
            qs = q + s_arr
            qm = q - s_arr
            vps = (ps >= 0) & (ps < N)
            vpm = (pm >= 0) & (pm < N)
            vqs = (qs >= 0) & (qs < N)
            vqm = (qm >= 0) & (qm < N)
            d = np.zeros(2 * N - 1)
            # p goes 1→0: removes contribution of h[p]
            # In corr[center+s] = sum h[i]*(1-h[i+s]):
            # if p = i: removes (1-h[p+s])
            # if p = i+s: removes h[p-s] from (1-h[i]) term → adds h[p-s] (removing a -)
            d[vps] -= (1.0 - h[np.clip(ps, 0, N-1)[vps]])
            d[vpm] += h[np.clip(pm, 0, N-1)[vpm]]
            # q goes 0→1: adds contribution
            d[vqs] += (1.0 - h[np.clip(qs, 0, N-1)[vqs]])
            d[vqm] -= h[np.clip(qm, 0, N-1)[vqm]]
            nc = corr + d
            npk = float(nc.max())
            delta = npk - current_pk
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                h[p] = 0
                h[q] = 1
                corr = nc
                current_pk = npk
                if npk < best_pk:
                    best_pk = npk
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1
            T *= cool
            if no_imp > max_no_imp:
                no_imp = 0
                if restart < multi_restart - 1:
                    break
                else:
                    h = best_h.copy()
                    corr = fft_corr(h)
                    current_pk = best_pk
    return best_h, best_pk

    T = T0

    while time.time() < deadline and T > 0.001:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < n_swap or len(zeros) < n_swap:
            break
        ps = rng.choice(ones, n_swap, replace=False)
        qs = rng.choice(zeros, n_swap, replace=False)
        d = np.zeros(2 * N - 1)
        for p in ps:

        nc = corr + d

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h_bin, N, rng):
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[rng.choice(idx, n1 - target, replace=False)] = 0
    elif n1 < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[rng.choice(idx, target - n1, replace=False)] = 1
    return h_bin

def lse_weights(corr, beta):
    m = float(corr.max())
    eb = np.exp(beta * (corr - m))
    return eb / float(eb.sum())

def compute_grad(h, N, weights):
    center = N - 1
    grad = np.zeros(N)
    for ks in range(1, N):
        wk = weights[center + ks]
        wkn = weights[center - ks]
        if abs(wk) > 1e-12:
            nv = N - ks
            grad[:nv] += wk * (1.0 - h[ks:])
            grad[ks:] += -wk * h[:nv]
        if abs(wkn) > 1e-12:
            nv = N - ks
            grad[ks:] += wkn * (1.0 - h[:nv])
            grad[:nv] += -wkn * h[ks:]
    w0 = weights[center]
    if abs(w0) > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    return grad

def continuous_lse(h, N, deadline, beta0=20, beta1=500, lr0=0.005):
    h = h.astype(float).copy()
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best = float(np.max(fft_corr(h)))
    best_h = h.copy()
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best:
            best = pk
            best_h = h.copy()
        t += 1
        frac = min(t / 500.0, 1.0)
        beta = beta0 + (beta1 - beta0) * frac
        w = lse_weights(corr, beta)
        grad = compute_grad(h, N, w)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        lr = lr0 * (1.0 + t * 0.002) ** (-0.5)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
    return best_h, best

def swap_delta(corr, h, N, p, s_arr):
    ps = p + s_arr
    pm = p - s_arr
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    d = np.zeros(2 * N - 1)
    d[vps] -= (1.0 - h[np.clip(ps, 0, N - 1)[vps]])
    d[vpm] += h[np.clip(pm, 0, N - 1)[vpm]]
    return d

def swap_delta_add(corr, h, N, q, s_arr):
    qs = q + s_arr
    qm = q - s_arr
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    d = np.zeros(2 * N - 1)
    d[vqs] += (1.0 - h[np.clip(qs, 0, N - 1)[vqs]])
    d[vqm] -= h[np.clip(qm, 0, N - 1)[vqm]]
    return d

def b_anneal(h, N, deadline, T0=0.06, cool=0.9994, max_no_imp=800, n_swaps=1, multi_restart=2, rng=None):

        rng = np.random

    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    s_arr = np.arange(-(N - 1), N)
    for restart in range(multi_restart):
        T = T0 if restart == 0 else T0 * 0.3
        no_imp = 0
        while time.time() < deadline and T > 0.0004:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < n_swaps or len(zeros) < n_swaps:
                break
            ps = rng.choice(ones, n_swaps, replace=False)
            qs = rng.choice(zeros, n_swaps, replace=False)
            d = np.zeros(2 * N - 1)
            for p in ps:
                d += swap_delta(corr, h, N, p, s_arr)

                d += swap_delta_add(corr, h, N, q, s_arr)
            nc = corr + d

                    h[p] = 0

                    h[q] = 1
                corr = nc
                cur_pk = npk

                    no_imp = 0
                else:

                if restart < multi_restart - 1:
                    break
                h = best_h.copy()

    return best_h, best_pk
