import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * 2.0 / N

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def make_binary(h, N):
    hb = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(hb.sum())
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        idx = idx[np.random.permutation(len(idx))[:n1 - nh]]
        hb[idx] = 0
    elif n1 < nh:
        idx = np.where(hb == 0)[0]
        idx = idx[np.random.permutation(len(idx))[:nh - n1]]
        hb[idx] = 1
    return hb

def lse_grad(h, N, T):
    corr = fft_corr(h)
    m = corr.max()
    ef = np.exp(T * (corr - m))
    p = ef / ef.sum()
    active = p > (0.005 / len(p))
    if active.sum() < 4:
        ai = np.argpartition(p, -4)[-4:]
        active = np.zeros_like(p, dtype=bool)
        active[ai] = True
    grad = np.zeros(N)
    center = N - 1
    for ki in np.where(active)[0]:
        L = int(ki - center)
        pi = p[ki]
        if L >= 0:
            seg = N - L
            if seg > 0:
                grad[:seg] += pi * (1.0 - h[L:])
                grad[L:] -= pi * h[:seg]
        else:
            M = -L
            seg = N - M
            if seg > 0:
                grad[M:] += pi * (1.0 - h[:seg])
                grad[:seg] -= pi * h[M:]
    return grad, m

def adam_opt(h0, N, deadline, T0=4.0, T1=100.0, lr=0.008):
    h = project_sum(h0.copy(), N / 2.0)
    target = N / 2.0
    best_h = h.copy()
    best_pk = np.max(fft_corr(h))
    mom = np.zeros(N)
    var = np.zeros(N)
    t = 0
    b1, b2, eps = 0.82, 0.998, 1e-9
    while time.time() < deadline:
        T = T0 + (T1 - T0) * min(1.0, (t % 8000) / 6000.0)
        grad, pk = lse_grad(h, N, T)
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
        mom = b1 * mom + (1 - b1) * grad
        var = b2 * var + (1 - b2) * grad * grad
        mh = mom / (1.0 - b1 ** (t + 1))
        vh = var / (1.0 - b2 ** (t + 1))
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        t += 1
        if t % 6000 == 0 and t > 3000:
            h = project_sum(h + np.random.randn(N) * 0.012, target)
            mom = var = np.zeros(N)
    return best_h, best_pk * 2.0 / N

def binary_opt(h0, N, deadline, n_tries=20, kick=70, sa_temp=0.0):
    h = (h0 > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            idx = idx[np.random.permutation(len(idx))[:n1 - nh]]
            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))[:nh - n1]]
            h[idx] = 1
    corr = fft_corr(h)
    best_int = int(corr.max())
    best_h = h.copy()
    stall = 0
    cur_temp = sa_temp
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_sweep = int(corr.max())
        bestsh = None
        bestsc = None
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            ct = fft_corr(h)
            ctmax = int(ct.max())
            accept = ctmax < best_sweep
            if not accept and cur_temp > 0:
                d = ctmax - best_sweep
                if np.random.rand() < np.exp(-d / cur_temp):
                    accept = True
            if accept and ctmax < best_sweep:
                best_sweep = ctmax
                bestsh = h.copy()
                bestsc = ct.copy()
            h[p] = 1
            h[q] = 0
        if bestsh is not None:
            h = bestsh
            corr = bestsc
            stall = 0
            if best_sweep < best_int:
                best_int = best_sweep
                best_h = h.copy()
        else:
            stall += 1
            cur_temp *= 0.95
            if stall > kick:
                if len(ones) >= 3 and len(zeros) >= 3:
                    kk = min(3, len(ones), len(zeros))
                    i1 = np.random.choice(len(ones), kk, replace=False)
                    i2 = np.random.choice(len(zeros), kk, replace=False)
                    h[ones[i1]] = 0
                    h[zeros[i2]] = 1
                    corr = fft_corr(h)
                stall = 0
                cur_temp = max(sa_temp * 0.5, 0.01)
    return best_h, best_int * 2.0 / N

def interp_to(h, N_new):
    N_old = len(h)
    xo = np.linspace(0, 2, N_old, endpoint=False)
    xn = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Adam LSE continuous (280s)
    d1 = min(deadline - 820, t0 + 280)
    hc = project_sum(h0 + np.random.randn(N0) * 0.04, N0 / 2.0)
    hn, c5n = adam_opt(hc, N0, d1, T0=3.0, T1=80.0, lr=0.007)

    return best_h, final_c5, len(best_h)
