import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_sum(h, N):

        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, perturb_every=0):
    h = h.astype(float).copy()
    if int(round(h.sum())) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    D = compute_D(h, N)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c + D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                D = compute_D(h, N)
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            D = compute_D(h, N)
    return best_h, best_c * (2.0 / N)

def softmax_nesterov(h, N, deadline, tau0=3.0, tau1=60.0, lr=0.01, beta=0.9):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = compute_c5(h, N)
    m = np.zeros(N)
    tau = tau0
    it = 0
    D = compute_D(h, N)
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        c_shift = c - c.max()
        e = np.exp(tau * c_shift)
        w = e / e.sum()
        grad = -(D.T @ w)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        D = compute_D(h, N)
        it += 1
        if it % 20 == 0:
            c_now = compute_c5(h, N)
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
        if it % 50 == 0:
            tau = min(tau1, tau * 1.06)
            lr *= 0.997
            beta = min(0.97, 0.88 + it * 0.00005)
    c_final = compute_c5(h, N)
    if c_final < best_c:
        best_c, best_h = c_final, h.copy()
    return best_h, best_c

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def downsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary search at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 500, t0 + 40)
    h1, c5_1 = binary_search(h_bin, N0, d1, max_sweeps=600, sample_frac=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Softmax Nesterov at N0
    if time.time() < deadline - 400:
        h_frac = project_sum(best_h, N0 / 2.0)
        d2 = min(deadline - 350, time.time() + 70)
        h2, c5_2 = softmax_nesterov(h_frac, N0, d2, tau0=4.0, tau1=50.0, lr=0.009, beta=0.9)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Re-binarize and binary search
    if time.time() < deadline - 300:
        h_bin3 = ensure_sum(best_h, N0)
        d3 = min(deadline - 270, time.time() + 50)
        h3, c5_3 = binary_search(h_bin3, N0, d3, max_sweeps=400, sample_frac=1.0, perturb_every=35)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Upsample to 760, binary search
    N1 = 2 * N0
    if time.time() < deadline - 200:
        h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        d4 = min(deadline - 160, time.time() + 70)
        h4, c5_4 = binary_search(h_up, N1, d4, max_sweeps=150, sample_frac=0.5)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Softmax at 760
    if time.time() < deadline - 120:
        h_up2 = project_sum(upsample(best_h, len(best_h), N1), N1 / 2.0)
        d5 = min(deadline - 90, time.time() + 50)
        h5, c5_5 = softmax_nesterov(h_up2, N1, d5, tau0=3.0, tau1=40.0, lr=0.006, beta=0.9)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Downsample to N0, binary search
    if time.time() < deadline - 60:
        N_cur = len(best_h)
        h_ds = downsample(best_h, N_cur, N0) if N_cur != N0 else best_h.copy()
        h_bin6 = ensure_sum(h_ds, N0)
        d6 = min(deadline - 35, time.time() + 40)
        h6, c5_6 = binary_search(h_bin6, N0, d6, max_sweeps=350, sample_frac=1.0, perturb_every=30)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final softmax polish at N0
    if time.time() < deadline - 15:
        h_fin = project_sum(best_h, N0 / 2.0)
        d7 = deadline - 3
        h7, c5_7 = softmax_nesterov(h_fin, N0, d7, tau0=8.0, tau1=80.0, lr=0.004, beta=0.93)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Final binary polish
    if time.time() < deadline - 1:
        h_fin2 = ensure_sum(best_h, len(best_h))
        N_fin = len(h_fin2)
        sf = 1.0 if N_fin <= 400 else 0.6
        h8, c5_8 = binary_search(h_fin2, N_fin, deadline - 0.5, max_sweeps=200, sample_frac=sf)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
