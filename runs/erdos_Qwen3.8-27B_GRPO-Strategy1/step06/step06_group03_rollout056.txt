def subgrad_topk(corr, h, top_k, N):
    """Subgradient at top-k lags (averaged)"""

    if top_k >= 2*N - 1:
        top = np.arange(2*N - 1)
    else:
        top = np.argpartition(corr, -top_k)[-top_k:]
    top = np.sort(top)
    for k in top:
        lag = int(k - center)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:-lag] -= h[lag:]
        else:
            M = -lag
            if M < N:
                grad[:-M] += (1.0 - h[M:])
                grad[M:] -= h[:-M]
    grad /= len(top)
    return grad

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def eval_swap_max(h, c, top_idx, p, q, N):
    center = N - 1
    result = -1e18
    for k in top_idx:
        L = int(k - center)
        delta = 0.0
        seen = set()
        for m in (p, p - L, q, q - L):
            if 0 <= m < N and 0 <= m + L < N and m not in seen:
                seen.add(m)
                old_v = h[m+L] * (1.0 - h[m])
                hmn = 0.0 if m == p else (1.0 if m == q else h[m])
                hml = 0.0 if m+L == p else (1.0 if m+L == q else h[m+L])
                delta += hml * (1.0 - hmn) - old_v
        v = c[k] + delta
        if v > result:
            result = v
    return result

def local_search(h, N, deadline, max_sweeps=600, sample_n=200, top_m=25):

    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        idx = idx[np.random.permutation(len(idx))][:n1 - nh]
        h[idx] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        idx = idx[np.random.permutation(len(idx))][:nh - n1]
        h[idx] = 1
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_idx = np.argsort(c)[-top_m:]
        if len(ones) > sample_n:
            p_list = np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones.copy()
        if len(zeros) > sample_n:
            q_list = np.random.choice(zeros, sample_n, replace=False)
        else:
            q_list = zeros.copy()
        best_new_P = P
        best_move = None
        for p in p_list:
            pi = int(p)
            for q in q_list:
                if pi == int(q):
                    continue
                mv = eval_swap_max(h, c, top_idx, pi, int(q), N)
                if mv < best_new_P:
                    best_new_P = mv
                    best_move = (pi, int(q))
        if best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
        else:
            no_improve += 5
    best_c = min(best_c, float(np.max(fft_corr(h))))
    return h, best_c

def nesterov_polish(h, N, deadline, k=60, lr0=0.006):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.92
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        center = N - 1
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N-L])
                grad[:N-L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += (1.0 - y[M:])
                    grad[M:] -= y[:N-M]
        grad /= len(top)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 40 == 0:
            lr *= 0.993
            beta = min(0.97, beta + 0.001)
    return best_h, best_pk * dx
