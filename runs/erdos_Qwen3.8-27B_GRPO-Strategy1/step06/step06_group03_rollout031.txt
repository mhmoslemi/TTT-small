def influence_top(h, top_idx, N):
    """Compute influence vectors for top lags. Returns shape (N, len(top_idx))."""
    center = N - 1
    K = len(top_idx)
    phi = np.zeros((N, K))
    for ki, ai in enumerate(top_idx):
        L = int(ai - center)
        if L >= 0:
            # phi[j, L] = (1-h[j-L]) if L <= j <= N-1, plus -h[j+L] if 0 <= j <= N-1-L
            valid1 = np.arange(N) >= L
            phi[valid1, ki] += (1.0 - h[np.arange(N)[valid1] - L])
            valid2 = np.arange(N) <= N - 1 - L
            phi[valid2, ki] += -h[np.arange(N)[valid2] + L]
        else:
            M = -L
            # R(-M) = sum h[m](1-h[m+M])
            # phi[j, -M] = (1-h[j+M]) if 0 <= j <= N-1-M, plus -h[j-M] if M <= j <= N-1
            valid1 = np.arange(N) <= N - 1 - M
            phi[valid1, ki] += (1.0 - h[np.arange(N)[valid1] + M])
            valid2 = np.arange(N) >= M
            phi[valid2, ki] += -h[np.arange(N)[valid2] - M]
    return phi

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h, N, deadline, k_start=100, k_end=15, lr0=0.02, beta0=0.9):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    max_it = 5000
    while time.time() < deadline and it < max_it:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        # Adaptive K
        progress = it / max_it
        k_cur = max(k_end, int(k_start * (1 - progress) + k_end * progress))
        temp = max(0.05, 0.5 * (1 - progress))
        
        top = np.argpartition(corr, -k_cur)[-k_cur:]
        c_top = corr[top]
        c_max = c_top.max()
        w = np.exp((c_top - c_max) / (temp * max(abs(c_max), 1e-10)))
        w /= w.sum()
        
        center = N - 1
        grad = np.zeros(N)
        for idx in range(len(top)):
            L = int(top[idx] - center)
            wi = w[idx]
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wi * (1.0 - h[:N - L])
                grad[:N - L] -= wi * h[L:]
            else:
                M = -L
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.992
            beta = min(0.97, beta0 + it * 0.00008)
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def influence_top(h, top_idx, N):
    center = N - 1
    K = len(top_idx)
    phi = np.zeros((N, K))
    ar = np.arange(N)
    for ki in range(K):
        L = int(top_idx[ki] - center)
        if L == 0:
            phi[:, ki] = 1.0 - 2.0 * h
        elif L > 0:
            v1 = ar >= L
            phi[v1, ki] += 1.0 - h[ar[v1] - L]
            v2 = ar <= N - 1 - L
            phi[v2, ki] += -h[ar[v2] + L]
        else:
            M = -L
            v1 = ar <= N - 1 - M
            phi[v1, ki] += 1.0 - h[ar[v1] + M]
            v2 = ar >= M
            phi[v2, ki] += -h[ar[v2] - M]
    return phi
