            best_h = h.copy()
        t += 1

    hb = np.zeros(N)
    # Sort by h value, take top N/2
    idx = np.argsort(-h)
    hb[idx[:N//2]] = 1.0

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)*0.5, 0.0, 1.0)

def compute_D(h):
    N = len(h)
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp>=0) & (pp<N)
    vm = (pm>=0) & (pm<N)
    return (np.where(vp, h[np.clip(pp,0,N-1)], 0.0) -
            np.where(vm, h[np.clip(pm,0,N-1)], 0.0))

def validate_binary(h, N):
    """Ensure h is binary with exactly N/2 ones."""

        return h
    h[:] = 0
    if n1 > N//2:

        h[:N//2] = 1

def iterated_local_search(h, N, deadline, n_iter=200, k_min=3, k_max=15):

def subgradient_guide(h0, N, deadline):
    """Continuous subgradient to find a good direction for binarization."""
    h = project_sum(h0.astype(float), N/2.0)
    best_h = h.copy()
    best_val = compute_c5(h)
    lr = 0.5
    for t in range(10000):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        k_star = int(np.argmax(c))
        shift = k_star - (N-1)
        pos = np.arange(N)
        jp = pos + shift
        jm = pos - shift
        vp = (jp>=0) & (jp<N)
        vm = (jm>=0) & (jm<N)
        hp = np.where(vp, h[np.clip(jp,0,N-1)], 0.0)
        hm = np.where(vm, h[np.clip(jm,0,N-1)], 0.0)
        # To reduce f_{k*}, we want to increase h where (1-hp-hm) is large
        # i.e., where both neighbors are 0. This creates 1-1 pairs at shift.
        # gradient of f_{k*} at h: 1 - hp - hm

        h = project_sum(h + lr*(hp + hm - 1.0), N/2.0)
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        lr *= 0.9995
    return best_h, best_val
