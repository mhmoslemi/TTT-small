        q_list = rng.choice(zeros, n_s, replace=False).astype(int)
        ip = p_list[:, None] + km[None, :]
        vi = (ip >= 0) & (ip < N)
        dr = np.where(vi, 2.0 * h[np.clip(ip, 0, N - 1)] - 1.0, 0.0)
        iq = q_list[:, None] + km[None, :]
        vq = (iq >= 0) & (iq < N)
        da = np.where(vq, 1.0 - 2.0 * h[np.clip(iq, 0, N - 1)], 0.0)
        nv = c[top_idx][None, None, :] + dr[:, None, :] + da[None, :, :]
        nm = nv.max(axis=2)
        if tabu_pairs:
            pm = np.zeros((N, N), dtype=bool)
            for (a, b) in tabu_pairs:
                pm[a, b] = True
                pm[b, a] = True
            mask = pm[np.ix_(p_list, q_list)]
            if mask.all():
                tabu_pairs.clear()
                continue
            nm = np.where(mask, np.inf, nm)
            if not np.isfinite(nm).any():
                tabu_pairs.clear()
                continue
        mi = np.unravel_index(np.argmin(nm), nm.shape)
        bp = int(p_list[mi[0]])
        bq = int(q_list[mi[1]])
        bv = float(nm[mi[0], mi[1]])
        d = bv - P
        accept = d < -1e-10
        if not accept and anneal_temp > 0:
            temp = anneal_temp * (0.97 ** (it // 400))
            if d < 3.0 and rng.random() < np.exp(-max(d, 0.0) / max(temp, 1e-8)):
                accept = True
        if accept:
            h[bp] = 0.0
            h[bq] = 1.0
            tabu_pairs.add((bp, bq))
            tabu_pairs.add((bq, bp))
        else:
            tabu_pairs.clear()
        if it % tabu_len == 0 and len(tabu_pairs) > 80:
            tabu_pairs.clear()
        it += 1
    fc = float(np.max(fft_corr(h)))
    if fc < best_c:
        best_c = fc
        best_h = h.copy()
    return best_h, best_c

def nesterov_opt(h, N, deadline, k=60, lr0=0.015):
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) > 1e-10:
        lo, hi = -1.0, 1.0
        for _ in range(40):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    m = np.zeros(N)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    beta = 0.92
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top_k = min(k, len(corr))
        top = np.argpartition(corr, -top_k)[-top_k:]
        center = N - 1
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= len(top)
        h_new = np.clip(y - lr0 * grad, 0.0, 1.0)
        s2 = h_new.sum()
        if abs(s2 - target) > 1e-8:
            lo, hi = -1.0, 1.0
            for _ in range(30):
                mid = (lo + hi) * 0.5
                if np.clip(h_new - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo + hi) * 0.5, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr0 *= 0.995
            beta = min(0.97, 0.92 + it * 0.0001)
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    # Phase 1: Direct local search on parent
    d1 = min(deadline - 450, t0 + 150)
    h1, c1 = local_search(h0, N0, d1, rng, top_m=50, base_sample=130)
    c5_1 = compute_c5(h1, N0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1
    # Phase 2: Annealed search from perturbation
    if time.time() < deadline - 380:
        h_pert = best_h.copy()
        fl = min(8, N0 // 20)
        o1 = np.where(h_pert == 1)[0]
        z1 = np.where(h_pert == 0)[0]
        h_pert[rng.choice(o1, fl, replace=False)] = 0
        h_pert[rng.choice(z1, fl, replace=False)] = 1
        h_pert = make_binary(h_pert, N0 // 2, rng)
        d2 = min(deadline - 330, time.time() + 250)
        h2, c2 = local_search(h_pert, N0, d2, rng, top_m=50, base_sample=140, anneal_temp=0.25)
        c5_2 = compute_c5(h2, N0)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2
    # Phase 3: Nesterov + threshold
    if time.time() < deadline - 280:
        d3 = min(deadline - 240, time.time() + 80)
        hc = best_h + rng.randn(N0) * 0.04
        hn, c5n = nesterov_opt(hc, N0, d3)
        h_bin = make_binary(hn, N0 // 2, rng)
        c5b = compute_c5(h_bin, N0)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin
    # Phase 4: Intensive local search on best
    if time.time() < deadline - 200:
        d4 = min(deadline - 150, time.time() + 200)
        h4, c4 = local_search(best_h, N0, d4, rng, top_m=60, base_sample=150, tabu_len=25)
        c5_4 = compute_c5(h4, N0)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4
    # Phase 5: Multi-resolution
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [400, 420, 460, 500, 600]:
        if time.time() > deadline - 60:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_i = np.interp(x_new, x_best, best_h)
        h_bt = make_binary(h_i, N_try // 2, rng)
        sub_dl = min(deadline - 30, time.time() + 45)
        h_rt, _ = local_search(h_bt, N_try, sub_dl, rng, top_m=50, base_sample=120)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt
            x_best = np.linspace(0, 2, N_try, endpoint=False)
    # Phase 6: Final tight search
    if time.time() < deadline - 10:
        hF, cF = local_search(best_h, len(best_h), deadline - 2, rng, top_m=60, base_sample=160)
        c5F = compute_c5(hF, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
