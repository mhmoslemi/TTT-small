import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))[:2*N-1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def compute_grad_smoothed(corr, h, T, N, n_top=150):
    """Gradient of log-sum-exp(T*corr)/T with respect to h."""
    shifted = T * (corr - corr.max())
    p = np.exp(shifted)
    p /= p.sum()
    top_idx = np.argpartition(p, -n_top)[-n_top:]
    grad = np.zeros(N)
    center = N - 1
    for idx in top_idx:
        wi = p[idx]
        if wi < 1e-10:
            continue
        L = int(round(float(idx) - center))
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        elif -N < L < 0:
            M = -L
            seg = N - M
            grad[:seg] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:seg]
    return grad

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N-1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]

def c5_of(h):
    return float(np.max(fft_corr(h))) * (2.0/len(h))

def proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum())-target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        m = 0.5*(lo+hi)
        if float(np.clip(h-m,0,1).sum()) > target: lo = m
        else: hi = m
    return np.clip(h-0.5*(lo+hi), 0, 1)

def grad_sm(corr, h, T, N, ktop=200):
    s = T*(corr-corr.max())
    p = np.exp(s); p /= p.sum()
    ti = np.argpartition(p, -ktop)[-ktop:]
    g = np.zeros(N)
    c = N-1
    for i in ti:
        w = p[i]
        if w < 1e-9: continue
        L = int(round(float(i)-c))
        if L == 0:
            g += w*(1.0-2.0*h)
        elif 0 < L < N:
            seg = N-L
            g[L:] += w*(1.0-h[:seg])
            g[:seg] -= w*h[L:]
        elif -N < L < 0:
            M = -L; seg = N-M
            g[:seg] += w*(1.0-h[M:])
            g[M:] -= w*h[:seg]
    return g
