import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(np.float64) + hp * vp.astype(np.float64)

def smooth_grad(h, N, t0, beta, deadline, lr0=0.01, max_iter=5000):
    """Smooth gradient descent with softmax objective + Adam."""
    h = h.astype(np.float64).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr0
    beta_cur = beta
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_c:
            best_c = pk
            best_h = h.copy()
        # Softmax weights
        c_shifted = c - c.max()
        w = np.exp(beta_cur * c_shifted)
        w /= w.sum()
        # Gradient
        D = compute_D(h, N)
        grad = (D @ w) * dx
        # Center gradient (since sum constraint)
        grad -= grad.mean()
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        # Schedule
        if t % 500 == 0:
            lr *= 0.9
            beta_cur *= 1.15
    return best_h, best_c * dx

def bin_search(h, N, deadline, max_sweeps=2000, anneal=0.0):

    n1 = int(round(h.sum()))

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(np.float64) + hp * vp.astype(np.float64)

    v = np.zeros(N)
    t = 0

    lr = lr0

    while time.time() < deadline:
        t += 1
        c = fft_corr(h)

        if pk < best_c - 1e-12:
            best_c = pk

        frac = min(t / total_iters, 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        cs = c - c.max()
        w = np.exp(beta * cs)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= wsum
        D = compute_D(h, N)

        grad -= grad.mean()
        m = b1 * m + (1 - b1) * grad

        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if t % 800 == 0:
            lr *= 0.88
    return best_h, best_c * dx

            best_c5, best_h = c57, h_7.copy()

    # Phase 8: Final smooth + binary polish
    if time.time() < deadline - 15:
        d8 = min(deadline - 5, time.time() + 12)
        hF = project_sum((best_h > 0.5).astype(float) + np.random.randn(len(best_h)) * 0.015, len(best_h) / 2.0)
        hF, c5F = softmax_grad(hF, len(best_h), 100, 200, d8, lr0=0.003)
        hbF = to_binary(hF, hF, len(hF))
        c5Fb = compute_c5(hbF, len(hbF))
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
        elif c5Fb < best_c5:
            best_c5, best_h = c5Fb, hbF.copy()

    if time.time() < deadline - 2:
        hbF = to_binary(best_h, best_h, len(best_h))
        hbF, _ = bin_search(hbF, len(best_h), deadline - 0.5, 80, 0.0)
        c5F2 = compute_c5(hbF, len(hbF))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hbF.copy()

    # Final evaluation
    final_h = (best_h > 0.5).astype(float)
    nchk = int(final_h.sum())
    nexp = len(final_h) // 2
    if nchk != nexp:
        final_h = to_binary(best_h, best_h, len(best_h))
    final_c5 = compute_c5(final_h, len(final_h))
    return final_h, final_c5, len(final_h)
