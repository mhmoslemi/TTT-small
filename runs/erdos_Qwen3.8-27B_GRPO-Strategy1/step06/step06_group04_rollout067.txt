def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Adjust: add diff/N to non-saturated entries
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    # Final binary search refinement
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        sh = np.clip(h - mid, 0.0, 1.0).sum()
        if sh > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_c5(corr, h, top_k=None):
    """Gradient of max(corr) w.r.t. h, using top_k smoothing"""
    N = len(h)
    if top_k is None or top_k >= len(corr):
        top_k = len(corr)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c = corr[idx]
    T = 0.01 * max(c.max() - c.min(), 1e-6)
    w = np.exp((c - c.max()) / T)
    w /= w.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for wi, ai in zip(w, idx):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            if seg > 0:
                grad[L:] += wi * (1.0 - h[:seg])
                grad[:seg] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                seg = N - M
                if seg > 0:
                    grad[:seg] += wi * (1.0 - h[M:])
                    grad[M:] -= wi * h[:seg]
    return grad

    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    return project_sum(h_new, target), m, v

    return float(np.max(fft_corr(h))) * (2.0 / len(h))

        elif L > 0:

            if 0 < M < N:

                g[:s] += wi * (1.0 - h[M:])

        v = b2 * v + (1.0 - b2) * g * g

    c = fft_corr(h)
    bi = int(c.max())
    bh = h.copy()
    stall = 0
    while time.time() < dl:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_i = int(c.max())
        best_h = None
        best_c = None
        for _ in range(ntr):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            ct = fft_corr(h)
            ci = int(ct.max())
            if ci < best_i:
                best_i = ci
                best_h = h.copy()
                best_c = ct.copy()
            h[p], h[q] = 1.0, 0.0
        if best_h is not None:
            h, c = best_h, best_c
            stall = 0
            if best_i < bi:
                bi = best_i
                bh = h.copy()
        else:
            stall += 1
            if stall > 50:

                st = np.random.randint(0, max(1, N - bl))
                h[st:st + bl] = 1.0 - h[st:st + bl]
                h = to_binary(h, N)
                c = fft_corr(h)
                stall = 0
    return bh, bi * (2.0 / N)

def local_refine(h, N, dl):
    h = h.copy()
    while time.time() < dl - 1:
        improved = False
        c = fft_corr(h)
        best = int(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if time.time() > dl - 1:
                break
            for j in zeros:
                if h[i] != 1.0 or h[j] != 0.0:
                    continue
                h[i], h[j] = 0.0, 1.0
                ct = fft_corr(h)
                cv = int(ct.max())
                if cv < best:
                    best = cv
                    c = ct
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, int(np.max(fft_corr(h))) * (2.0 / N)
