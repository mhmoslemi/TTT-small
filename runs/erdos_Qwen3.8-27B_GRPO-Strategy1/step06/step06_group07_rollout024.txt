```python
def compute_grad(h, w):
    """Compute gradient of w @ corr(h) w.r.t. h, where corr(h) = correlate(h, 1-h, 'full')"""
    N = len(h)
    grad = np.zeros(N)
    for t in range(2*N - 1):
        wt = w[t]
        if abs(wt) < 1e-14:
            continue
        L = t - (N - 1)
        if L >= 0:
            if N - L > 0:
                grad[:N-L] += wt * (1.0 - h[L:N])
            if L < N:
                grad[L:] -= wt * h[:N-L]
        else:
            M = -L
            if M < N:
                grad[M:] += wt * (1.0 - h[:N-M])
            if N - M > 0:
                grad[:N-M] -= wt * h[M:N]
    return grad
```

This is O(N * 2N) = O(N²) per gradient eval. For N=380, that's ~288K ops, very fast.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    L = 2 * N - 1
    D = np.zeros((N, L))
    one = np.ones(N)
    for p in range(N):
        for t in range(L):
            lag = t - (N - 1)
            if lag >= 0:
                if 0 <= p <= N - 1 - lag:
                    D[p, t] = -(1.0 - h[p + lag])
                if lag <= p <= N - 1:
                    D[p, t] -= h[p - lag]
            else:
                M = -lag
                if M <= p <= N - 1:
                    D[p, t] = -(1.0 - h[p - M])
                if 0 <= p <= N - 1 - M:
                    D[p, t] -= h[p + M]
    return D

def compute_grad(h, w):
    N = len(h)
    grad = np.zeros(N)
    for t in range(2 * N - 1):
        wt = w[t]
        if abs(wt) < 1e-14:
            continue
        L = t - (N - 1)
        if L >= 0:
            if N - L > 0:
                grad[:N - L] += wt * (1.0 - h[L:N])
            if L < N:
                grad[L:] -= wt * h[:N - L]
        else:
            M = -L
            if M < N:
                grad[M:] += wt * (1.0 - h[:N - M])
            if N - M > 0:
                grad[:N - M] -= wt * h[M:N]
    return grad

def ensure_sum(h, N):
    target = N // 2
    h = h.astype(float).copy()
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    n1 = int(h_bin.sum())
    if n1 == target:
        return h_bin
    rng = np.random.RandomState(9999 + N)
    if n1 > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, perturb_every=0):
    h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 50:
            n_s = max(30, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def smooth_opt(h_init, N, deadline, beta_start=8.0, beta_end=150.0):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_max = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta = beta_start
    lr_base = 0.4
    momentum = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        cur_max = float(np.max(c))
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
        scaled = beta * c
        scaled -= scaled.max()
        ew = np.exp(scaled)
        w = ew / ew.sum()
        grad = compute_grad(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        lr = lr_base / (gnorm + 1e-8)
        h_new = project_sum(y - lr * grad, target)
        m = momentum * (h_new - h)
        h = h_new
        it += 1
        if it % 50 == 0:
            beta = min(beta * 1.04, beta_end)
            momentum = min(0.95, 0.8 + it * 0.0005)
            if it % 100 == 0:
                hr = ensure_sum(h, N)
                hr_c = float(np.max(fft_corr(hr)))
                if hr_c < best_max:
                    best_max = hr_c
                    best_h = hr.copy()
                    h = hr + 0.02 * np.random.randn(N)
                    h = project_sum(h, target)
                    m = np.zeros(N)
    return best_h, best_max * (2.0 / N)

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
