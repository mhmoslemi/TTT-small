def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_sum(h, target):

def fix_count(h_bin, N):

    vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
    vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
    d = np.zeros(2*N-1)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    h_bin = h_bin.copy()
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin==1)[0]
        perm = np.random.permutation(len(idx))[:n1-target]
        h_bin[idx[perm]] = 0
    elif n1 < target:
        idx = np.where(h_bin==0)[0]
        perm = np.random.permutation(len(idx))[:target-n1]
        h_bin[idx[perm]] = 1
    return h_bin

def corr_delta(h, N, p, q):
    s_arr = np.arange(-(N-1), N)
    ps, pm = p+s_arr, p-s_arr
    qs, qm = q+s_arr, q-s_arr
    vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
    vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
    d = np.zeros(2*N-1)
    pc=np.clip(ps,0,N-1)[vps]; pmc=np.clip(pm,0,N-1)[vpm]
    qsc=np.clip(qs,0,N-1)[vqs]; qmc=np.clip(qm,0,N-1)[vqm]
    d[vps] -= (1.0-h[pc])
    d[vpm] += h[pmc]
    d[vqs] += (1.0-h[qsc])
    d[vqm] -= h[qmc]
    return d

def anneal_adv(h, N, deadline, T0=0.09, cool=0.9994, max_ni=900, n_restart=4, p2s=0.35):
    h = h.astype(float).copy()
    h = fix_count(h, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    for r in range(n_restart):
        T = T0 * (0.55 ** r)
        ni = 0
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        while time.time() < deadline and T > 0.0008:
            if ni > max_ni:
                h = best_h.copy(); corr = fft_corr(h); cur_pk = best_pk
                ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
                T = T0 * 0.25; ni = 0
                continue
            use2 = np.random.random() < p2s and len(ones)>=2 and len(zeros)>=2
            if use2:
                i1=np.random.randint(len(ones)); i2=np.random.randint(len(ones))
                j1=np.random.randint(len(zeros)); j2=np.random.randint(len(zeros))
                if i1==i2 or j1==j2: continue
                p1=int(ones[i1]); p2=int(ones[i2]); q1=int(zeros[j1]); q2=int(zeros[j2])
                if p1==q1 or p1==q2 or p2==q1 or p2==q2: continue
                d = corr_delta(h,N,p1,q1) + corr_delta(h,N,p2,q2)
                # fix interaction at q1 and p2
                nc = corr + d
                npk = float(nc.max())
                delta = npk - cur_pk
                if delta < -1e-12 or np.random.random() < np.exp(-delta/max(T,1e-9)):
                    h[p1]=0; h[q1]=1; h[p2]=0; h[q2]=1
                    corr=nc; cur_pk=npk
                    ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
                    if npk < best_pk: best_pk=npk; best_h=h.copy(); ni=0
                    else: ni+=1
                else: ni+=1
            else:
                p=int(ones[np.random.randint(len(ones))])
                q=int(zeros[np.random.randint(len(zeros))])
                d = corr_delta(h,N,p,q)
                nc = corr + d
                npk = float(nc.max())
                delta = npk - cur_pk
                if delta < -1e-12 or np.random.random() < np.exp(-delta/max(T,1e-9)):
                    h[p]=0; h[q]=1; corr=nc; cur_pk=npk

                    if npk < best_pk: best_pk=npk; best_h=h.copy(); ni=0
                    else: ni+=1
                else: ni+=1
            T *= cool
    return best_h, best_pk

def greedy_local(h, N, deadline, n_cand=12):
    h = h.astype(float).copy()
    h = fix_count(h, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    stall = 0
    while time.time() < deadline and stall < 500:
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)==0 or len(zeros)==0: break
        best_np = float('inf'); bp=-1; bq=-1; bd=None
        for _ in range(n_cand):
            p=int(ones[np.random.randint(len(ones))])
            q=int(zeros[np.random.randint(len(zeros))])
            d = corr_delta(h,N,p,q)
            npk = float((corr+d).max())
            if npk < best_np:
                best_np=npk; bp=p; bq=q; bd=d
        nc = corr + bd
        if best_np < best_pk - 1e-12:
            h[bp]=0; h[bq]=1; corr=nc
            if best_np < best_pk:
                best_pk=best_np; best_h=h.copy(); stall=0
            else: stall+=1
        else:
            stall+=1
    return best_h, best_pk

def continuous_step(h, N, deadline, top_k=40, lr0=0.006):
    h = h.astype(float).copy()
    m=np.zeros(N); v=np.zeros(N)
    b1a,b2a,eps=0.82,0.999,1e-8
    t_s=0; lr=lr0
    best_pk=float('inf'); best_h=h.copy()
    center=N-1
    while time.time() < deadline:
        corr=fft_corr(h); pk=float(corr.max())
        if pk<best_pk: best_pk=pk; best_h=h.copy()
        idx=np.argpartition(corr,-top_k)[-top_k:]
        grad=np.zeros(N)
        for ai in idx:
            s=int(ai-center)
            if s==0: grad+=(1.0-2.0*h)
            elif s>0:
                grad[:N-s]+=(1.0-h[s:]); grad[s:]-=h[:N-s]
            else:
                s2=-s; grad[s2:]+=(1.0-h[:N-s2]); grad[:N-s2]-=h[s2:]
        grad/=len(idx)
        t_s+=1
        m=b1a*m+(1-b1a)*grad; v=b2a*v+(1-b2a)*grad*grad
        mh=m/(1-b1a**t_s); vh=v/(1-b2a**t_s)
        h=proj_sum(h-lr*mh/np.sqrt(vh+eps), N/2.0)
        if t_s%300==0: lr*=0.985
    return best_h, best_pk
