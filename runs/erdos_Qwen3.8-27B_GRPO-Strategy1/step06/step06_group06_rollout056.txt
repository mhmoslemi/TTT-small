    # Combined:
    # At shift k: delta = -(1-h[p-k]) + h[p+k] + (1-h[q-k]) - h[q+k]
    # where all h indices must be in [0, N-1]
    
    for k in range(-(N-1), N):
        # p as first factor: n=p-k, need 0<=p-k<N
        dp = 0.0
        if 0 <= p - k < N:
            dp -= (1.0 - h[p - k])
        # p in (1-h[n]): n=p, need 0<=p+k<N
        if 0 <= p + k < N:
            dp += h[p + k]
        # q as first factor: n=q-k, need 0<=q-k<N
        dq = 0.0
        if 0 <= q - k < N:
            dq += (1.0 - h[q - k])
        # q in (1-h[n]): n=q, need 0<=q+k<N
        if 0 <= q + k < N:
            dq -= h[q + k]
        d[k + N - 1] = dp + dq
    return d

    d[vqm] -= h[qmc[vqm]]
    return d

def bin_anneal(h, N, deadline, T0=0.06, cool=0.9994, max_stall=600):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    cur_pk = best_pk
    T = T0
    stall = 0
    while time.time() < deadline and T > 0.0003:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = int(ones[np.random.randint(len(ones))])
        q = int(zeros[np.random.randint(len(zeros))])
        if p == q:
            T *= cool
            continue
        d = swap_delta(h, N, p, q)
        nc = corr + d
        npk = float(nc.max())
        delta = npk - cur_pk
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            h[p] = 0
            h[q] = 1
            corr = nc
            cur_pk = npk
            if npk < best_pk:
                best_pk = npk
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T *= cool
        if stall > max_stall:
            break
    return best_h, best_pk

def targeted_fix(h, N, rng, n_iter=4000):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    center = N - 1
    for _ in range(n_iter):
        worst_idx = int(np.argmax(corr))
        worst_k = worst_idx - center
        if worst_k >= 0:
            bad_i = np.where(h[:N - worst_k] == 1)[0]
            bad_i = bad_i[h[bad_i + worst_k] == 0]
            good_j = np.where(h[:N - worst_k] == 0)[0]
            good_j = good_j[h[good_j + worst_k] == 1]
        else:
            s2 = -worst_k
            bad_i = np.where(h[s2:] == 1)[0] + s2
            bad_i = bad_i[h[bad_i + worst_k] == 0]
            good_j = np.where(h[:N - s2] == 0)[0]
            good_j = good_j[h[good_j + s2] == 1]
        if len(bad_i) == 0 or len(good_j) == 0:
            continue
        i = int(bad_i[rng.randint(len(bad_i))])
        j = int(good_j[rng.randint(len(good_j))])
        if h[i] != 1 or h[j] != 0:
            continue
        d = swap_delta(h, N, i, j)
        nc = corr + d
        npk = float(nc.max())
        if npk < best_pk - 1e-12:
            h[i] = 0
            h[j] = 1
            corr = nc
            best_pk = npk
            best_h = h.copy()
    return best_h, best_pk

def interp_to_N(h, Nt):
    N0 = len(h)
    x0 = np.arange(N0) * (2.0 / N0)
    x1 = np.arange(Nt) * (2.0 / Nt)
    return np.interp(x1, x0, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = float(fft_corr(h0).max()) * (2.0 / N0)

    # Phase 1: Continuous LSE optimization
    d1 = min(deadline - 400, t0 + 220)
    hc = h0 + rng.randn(N0) * 0.025
    hc = proj_sum(hc, N0 / 2.0)
    hC, pkC = cont_opt(hc, N0, d1, topk=55, lr0=0.004)
    c5C = pkC * (2.0 / N0)
    if c5C < best_c5:
        best_c5, best_h = c5C, hC

    # Phase 2: Binary anneal from best
    d2 = min(deadline - 250, time.time() + 220)
    hb = fix_count((best_h > 0.5).astype(float), N0, rng)
    hB, pkB = bin_anneal(hb, N0, d2, T0=0.05, cool=0.9994, max_stall=700)
    c5B = pkB * (2.0 / N0)
    if c5B < best_c5:
        best_c5, best_h = c5B, hB

    # Phase 3: Targeted repair
    if time.time() < deadline - 180:
        d3 = min(deadline - 170, time.time() + 90)
        ni = max(3000, int((d3 - time.time()) * 40))
        hT, pkT = targeted_fix(best_h, len(best_h), rng, n_iter=ni)
        c5T = pkT * (2.0 / len(best_h))
        if c5T < best_c5:
            best_c5, best_h = c5T, hT

    # Phase 4: Second continuous pass
    if time.time() < deadline - 140:
        d4 = min(deadline - 130, time.time() + 120)
        Nf = len(best_h)
        hc2 = best_h.astype(float) + rng.randn(Nf) * 0.015
        hc2 = proj_sum(hc2, Nf / 2.0)
        hC2, pkC2 = cont_opt(hc2, Nf, d4, topk=60, lr0=0.002)
        c5C2 = pkC2 * (2.0 / Nf)
        if c5C2 < best_c5:
            best_c5, best_h = c5C2, hC2

    # Phase 5: Multi-N exploration
    for Nt in [360, 390, 420, 500]:
        if time.time() > deadline - 50:
            break
        if Nt % 2 != 0:
            Nt += 1
        hi = interp_to_N(best_h, Nt)
        hbi = fix_count((hi > 0.5).astype(float), Nt, rng)
        sub = min(deadline - 10, time.time() + 40)
        hrt, pkr = bin_anneal(hbi, Nt, sub, T0=0.04, cool=0.9997, max_stall=350)
        c5r = pkr * (2.0 / Nt)
        if c5r < best_c5:
            best_c5, best_h = c5r, hrt

    # Phase 6: Final binary polish
    if time.time() < deadline - 5:
        Nf = len(best_h)
        hf = fix_count((best_h > 0.5).astype(float), Nf, rng)
        hfp, pkf = bin_anneal(hf, Nf, deadline - 2, T0=0.015, cool=0.9998, max_stall=300)
        c5f = pkf * (2.0 / Nf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hfp

    final_c5 = float(fft_corr(best_h).max()) * (2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
