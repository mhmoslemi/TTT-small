    H = np.fft.rfft(h, L)

    beta = beta_init
    lr = lr_init
    t_k = 1.0

    stall = 0

        c_max = np.max(c_scaled)

        w = np.exp(c_scaled - logsumexp)  # softmax weights, sum to 1

        # FISTA update

        t_k = t_new

        h_new = project_sum(h_new, target)
        h = h_new

            h_old = h.copy()
            t_k = 1.0
            lr = lr_init
            beta = beta_init
            stall = 0

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    return np.fft.irfft(H * G, L)[:L]  # might get tiny imaginary parts

def compute_c5(h):
    N = len(h)
    corr = fft_corr(h)
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_grad(h, w, N):
    """Compute gradient of sum_k w[k]*corr[k] w.r.t. h.
    w has length 2N-1, indexed by lag: w[m+N-1] = weight for lag m."""
    grad = np.zeros(N)
    for m in range(-(N-1), N):
        wm = w[m + N - 1]
        if wm < 1e-14:
            continue
        if m >= 0:
            n = N - m
            grad[:n] += wm * (1.0 - h[m:])
            grad[m:] -= wm * h[:n]
        else:
            M = -m
            n = N - M
            grad[M:] += wm * (1.0 - h[:n])
            grad[:n] -= wm * h[M:]
    return grad

def fista_optimize(h0, N, deadline, beta0=10.0, beta_max=100.0):
    h = project_sum(h0.astype(float).copy(), N / 2.0)
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = beta0
    t_k = 1.0
    h_prev = h.copy()
    stall = 0
    
    while time.time() < deadline:
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = h + ((t_k - 1.0) / t_new) * (h - h_prev)
        h_prev = h
        t_k = t_new
        
        # Compute correlation and gradient at y
        corr = fft_corr(np.clip(y, 0, 1))
        c_scaled = beta * corr
        c_max = np.max(c_scaled)
        w = np.exp(c_scaled - c_max)
        w /= np.sum(w)
        
        smooth_obj = (c_max + np.log(np.sum(np.exp(c_scaled - c_max)))) / beta
        
        if smooth_obj < best_c5 * 1.001:
            actual_c5 = compute_c5(y)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = np.clip(y, 0, 1).copy()
            stall = 0
        else:
            stall += 1
        
        grad = compute_grad(np
