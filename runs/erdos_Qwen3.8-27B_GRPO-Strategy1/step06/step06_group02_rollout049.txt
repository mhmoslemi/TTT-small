def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

def nesterov_opt(h, N, deadline, k=50, lr0=0.02):
    h = h.astype(float).copy()
    target = N/2.0
    dx = 2.0/N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        
        # Subgradient: weighted average over top-k active shifts
        top = np.argpartition(corr, -k)[-k:]
        weights = np.zeros(2*N-1)
        active_corr = corr[top]
        w = np.exp((active_corr - active_corr.max()) / 0.01)
        w /= w.sum()
        
        grad = np.zeros(N)
        center = N - 1
        for ai in range(k):
            s = int(top[ai])
            shift = s - center
            L = np.float64(w[ai])
            if shift == 0:
                grad += L * (1.0 - 2.0*y)
            elif shift > 0:
                grad[shift:] += L * (1.0 - y[:N-shift])
                grad[:N-shift] -= L * y[shift:]
            else:
                M = -shift
                grad[:N-M] += L * (1.0 - y[M:])
                grad[M:] -= L * y[:N-M]
        
        grad /= k
        h_new = project_sum(y - lr*grad, target)
        m = beta*(h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it*0.0002)
    
    return best_h, best_pk*dx
