    corr = fft_corr(h)
    # Shift to avoid underflow
    cm = corr - corr.max()

    # Use only top shifts for efficiency
    top_idx = np.argsort(weights)[-top_k:]

        w = weights[ai]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))[:2*N-1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin==1)[0]
        h_bin[idx[np.random.choice(len(idx), n1-target, replace=False)]] = 0
    elif n1 < target:
        idx = np.where(h_bin==0)[0]
        h_bin[idx[np.random.choice(len(idx), target-n1, replace=False)]] = 1
    return h_bin

def corr_grad(h, N, top_k, beta):
    corr = fft_corr(h)
    pk = float(corr.max())
    w = np.exp(beta*(corr - pk))
    ws = float(w.sum())
    if ws < 1e-30:
        w = np.zeros_like(corr); w[np.argmax(corr)]=1.0; ws=1.0
    w /= ws
    center = N - 1
    grad = np.zeros(N)
    idx = np.argpartition(w, -min(top_k, len(w)))[-min(top_k,len(w)):]
    for ai in idx:
        s = int(ai - center)
        wi = w[ai]
        if s == 0:
            grad += wi*(1.0-2.0*h)
        elif s > 0:
            grad[:N-s] += wi*(1.0-h[s:])
            grad[s:] += wi*(-h[:N-s])
        else:
            s2=-s
            grad[s2:] += wi*(1.0-h[:N-s2])
            grad[:N-s2] += wi*(-h[s2:])
    return grad, pk

def cont_opt(h, N, deadline, top_k=45, beta0=25.0, beta_max=250.0, lr0=0.005):
    h = h.astype(float).copy()
    h = proj_sum(h, N/2.0)
    m=np.zeros(N); v=np.zeros(N)
    b1,b2,eps=0.8,0.999,1e-8
    t=0; best_pk=float(np.max(fft_corr(h))); best_h=h.copy()
    beta=beta0
    while time.time()<deadline:
        grad,pk=corr_grad(h,N,top_k,beta)
        if pk<best_pk: best_pk=pk; best_h=h.copy()
        t+=1
        m=b1*m+(1-b1)*grad; v=b2*v+(1-b2)*grad*grad
        mh=m/(1-b1**t); vh=v/(1-b2**t)
        lr=lr0*np.exp(-0.5*t/500.0)
        h=proj_sum(h-lr*mh/np.sqrt(vh+eps),N/2.0)
        if t%200==0: beta=min(beta*1.06,beta_max)
        if t%1000==0: beta=max(beta*0.9,10.0)
    return best_h,best_pk

def bin_anneal(h,N,deadline,T0=0.07,cool=0.9993,max_stall=800):
    h=h.astype(float).copy(); h=fix_count(h,N)

        p=int(ones[np.random.randint(len(ones))]); q=int(zeros[np.random.randint(len(zeros))])
        ps=p+s_arr; pm=p-s_arr; qs=q+s_arr; qm=q-s_arr
        vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
        vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
        d=np.zeros(2*N-1)
        d[vps]-=(1.0-h[np.clip(ps,0,N-1)[vps]])
        d[vpm]+=h[np.clip(pm,0,N-1)[vpm]]
        d[vqs]+=(1.0-h[np.clip(qs,0,N-1)[vqs]])

            stall=0; h=best_h.copy(); corr=fft_corr(h); cur_pk=best_pk

    return best_h,best_pk

def bin_local(h,N,deadline):
    """Targeted local search on the peak correlation."""
    h=h.astype(float).copy(); h=fix_count(h,N)
    corr=fft_corr(h); best_pk=float(corr.max()); best_h=h.copy()
    s_arr=np.arange(-(N-1),N)
    stalling=0
    while time.time()<deadline and stalling<3:
        ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
        peak_i=np.argmax(corr); peak_s=int(peak_i-(N-1))
        # Find positions contributing to peak: h[i]=1, h[i+peak_s]=0
        if peak_s>0:
            cand_p=np.where((h==1)&(h[peak_s:]==0)[np.arange(N-peak_s)])[0]
        else:
            s2=-peak_s
            cand_p=np.where((h[s2:]==1)&(h==0)[np.arange(N-s2)])[0]
        best_d=0; best_pq=(-1,-1)
        n_cand=min(len(cand_p),80)
        if n_cand>0:
            picks=np.random.choice(cand_p,n_cand,replace=False)
            for p in picks:
                q_cands=zeros
                n_q=min(len(q_cands),30)
                q_picks=np.random.choice(q_cands,n_q,replace=False)
                for q in q_picks:
                    ps=p+s_arr;pm=p-s_arr;qs=q+s_arr;qm=q-s_arr
                    vps=(ps>=0)&(ps<N);vpm=(pm>=0)&(pm<N)
                    vqs=(qs>=0)&(qs<N);vqm=(qm>=0)&(qm<N)
                    d=np.zeros(2*N-1)
                    d[vps]-=(1.0-h[np.clip(ps,0,N-1)[vps]])
                    d[vpm]+=h[np.clip(pm,0,N-1)[vpm]]
                    d[vqs]+=(1.0-h[np.clip(qs,0,N-1)[vqs]])
                    d[vqm]-=h[np.clip(qm,0,N-1)[vqm]]
                    npk=float((corr+d).max())
                    delta=npk-best_pk
                    if delta<best_d: best_d=delta;best_pq=(p,q)
        if best_pq[0]>=0 and best_d<0:

            corr=fft_corr(h);best_pk=float(corr.max());best_h=h.copy()
            stalling=0
        else:
            stalling+=1
    return best_h,best_pk
