```python
def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

        else:

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:

    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    """Compute the change matrix D[p, shift] for flipping h[p] from 1 to 0."""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp
```

```python
def soft_gradient(h, N, T=0.01, top_k=100):
    """Compute gradient using soft-max weighting over shifts."""
    corr = fft_corr(h)
    # Soft weights
    c_shifted = (corr - corr.max()) / T
    w = np.exp(np.minimum(c_shifted, 0))  # cap to avoid underflow
    w /= w.sum()
    # Only use top shifts for efficiency
    top_idx = np.argsort(w)[-top_k:]
    top_w = w[top_idx]
    top_w /= top_w.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for idx, wi in zip(top_idx, top_w):
        L = int(idx - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad
```

This is better than hard top-k because it gives smooth gradients and responds to near-maximal shifts.

```python
                    grad[M:] -= wi * y[:N - M]

            beta = min(0.97, beta0 + it * 0.00005)
```

```python
    rng = np.random.RandomState(54321 + N)
    if n1 > nh:
        idx = rng.choice(np.where(h_bin == 1)[0], n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        idx = rng.choice(np.where(h_bin == 0)[0], nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def binary_search(h, N, deadline, max_sweeps=2000, sample_frac=1.0, perturb_every=25):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_binary(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(31337)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 30:
            n_s = max(30, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P - 1e-15:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if stall >= perturb_every:
                n_flip = 1 if stall < perturb_every * 3 else 2
                for _ in range(n_flip):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    h[p], h[q] = 0.0, 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                c = fft_corr(h)
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
    return best_h, best_c * (2.0 / N)

def soft_nesterov(h, N, deadline, T=0.007, k=80, lr0=0.014, beta0=0.82):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_P = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    patience = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        P = float(corr.max())
        if P < best_P - 1e-14:
            best_P = P
            best_h = y.copy()
            patience = 0
        else:
            patience += 1
        c_shift = (corr - corr.max()) / T
        w = np.exp(np.clip(c_shift, -60, 0))
        ws = w.sum()
        if ws < 1e-10:
            w = np.zeros_like(corr)
            w[int(np.argmax(corr))] = 1.0
            ws = 1.0
        else:
            w /= ws
        if k < len(w):
            top_idx = np.argpartition(w, -k)[-k:]
            w_t = w[top_idx]
            w_t /= w_t.sum()
            w_u = np.zeros_like(w)
            w_u[top_idx] = w_t
        else:
            w_u = w
        active = np.where(w_u > 1e-6)[0]
        grad = np.zeros(N)
        center = N - 1
        for ai, wi in zip(active, w_u[active]):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += wi * (1.0 - y[:N - L])
                grad[:N - L] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:N - M]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.994
            beta = min(0.97, beta0 + it * 0.00004)
        if patience > 400:
            break
    return best_h, best_P * dx

def upsample(h_old, N_new):
    N_old = len(h_old)
    if N_old == N_new:
        return h_old.astype(float).copy()
    x_old = np.arange(N_old, dtype=float)
    x_new = np.arange(N_new, dtype=float) * (N_old / N_new)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick binary search at N0
    h_bin = ensure_binary(h0, N0)
    d1 = min(deadline - 500, t0 + 45)
    h1, c5_1 = binary_search(h_bin, N0, d1, max_sweeps=5000, sample_frac=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Soft Nesterov at N0 (two tries)
    if time.time() < deadline - 400:
        hc = best_h + np.random.randn(N0) * 0.06
        hc = project_sum(hc, N0 / 2.0)
        d2 = min(deadline - 350, time.time() + 150)
        h2, c5_2 = soft_nesterov(hc, N0, d2, T=0.007, k=90, lr0=0.016)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
    if time.time() < deadline - 300:
        hc2 = ensure_binary(best_h, N0).astype(float) + np.random.randn(N0) * 0.1
        hc2 = project_sum(hc2, N0 / 2.0)
        d2b = min(deadline - 260, time.time() + 80)
        h2b, c5_2b = soft_nesterov(hc2, N0, d2b, T=0.005, k=60, lr0=0.012)
        if c5_2b < best_c5:
            best_c5, best_h = c5_2b, h2b.copy()

    # Phase 3: Binary search from continuous at N0
    if time.time() < deadline - 250:
        h_b3 = ensure_binary(best_h, N0)
        d3 = min(deadline - 200, time.time() + 70)
        h3, c5_3 = binary_search(h_b3, N0, d3, max_sweeps=8000, sample_frac=1.0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Binary search at N1=2*N0 (main effort)
    N1 = 2 * N0
    if time.time() < deadline - 200:
        h_up = ensure_binary(upsample(best_h, N1), N1)
        d4 = min(deadline - 160, time.time() + 180)
        h4, c5_4 = binary_search(h_up, N1, d4, max_sweeps=800, sample_frac=0.45, perturb_every=20)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Soft Nesterov at N1
    if time.time() < deadline - 130:
        hc5 = upsample(ensure_binary(best_h, len(ensure_binary(best_h, N0))), N1)
        hc5 = project_sum(hc5 + np.random.randn(N1) * 0.04, N1 / 2.0)
        d5 = min(deadline - 100, time.time() + 120)
        h5, c5_5 = soft_nesterov(hc5, N1, d5, T=0.005, k=100, lr0=0.01)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Binary search at N1 from continuous
    if time.time() < deadline - 80:
        h_b6 = ensure_binary(best_h, N1)
        d6 = min(deadline - 50, time.time() + 80)
        h6, c5_6 = binary_search(h_b6, N1, d6, max_sweeps=600, sample_frac=0.4, perturb_every=20)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Try N2=3*N0 if time allows
    N2 = 3 * N0
    if N2 <= 1600 and time.time() < deadline - 100:
        h_up2 = ensure_binary(upsample(best_h, N2), N2)
        d7 = min(deadline - 60, time.time() + 90)
        h7, c5_7 = binary_search(h_up2, N2, d7, max_sweeps=300, sample_frac=0.3, perturb_every=30)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Final polish - binary at best N
    N_b = len(best_h)
    if time.time() < deadline - 10:
        hF = ensure_binary(best_h, N_b)
        sf_fin = 1.0 if N_b <= N0 else (0.5 if N_b <= N1 else 0.3)
        hF2, c5F = binary_search(hF, N_b, deadline - 2, max_sweeps=5000, sample_frac=sf_fin, perturb_every=15)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    # Phase 9: Tiny soft polish
    if time.time() < deadline - 2:
        Nb9 = len(best_h)
        hc9 = project_sum(best_h.astype(float) + np.random.randn(Nb9) * 0.015, Nb9 / 2.0)
        h9, c5_9 = soft_nesterov(hc9, Nb9, deadline - 1, T=0.003, k=40, lr0=0.006, beta0=0.75)
        if c5_9 < best_c5:
            best_c5, best_h = c5_9, h9.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
```
