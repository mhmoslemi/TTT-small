    # c_full[m] corresponds to shift k = m - (N-1)
    # c_full[m] = sum_i h[i]*(1-h[i+k]) where k = m-(N-1)
    #
    # Actually let me use the convention where shifts are indexed -N+1 to N-1
    # and stored in an array of length 2N-1.
    
    shifts = np.arange(2*N-1) - (N-1)  # -N+1, ..., N-1
    # For each p, the delta_remove at shift k is:
    # -(1-h[p+k]) if 0<=p+k<N, else 0, plus h[p-k] if 0<=p-k<N, else 0
    
    # Vectorized computation:
    # delta_remove[p, s] where s indexes shift k=s-(N-1)
    # = -(1 - h[p + k]) * (0<=p+k<N) + h[p-k] * (0<=p-k<N)
    
    pos = np.arange(N)[:, None]  # N x 1
    sv = shifts[None, :]  # 1 x (2N-1)
    pk = pos + sv  # p + k
    pm = pos - sv  # p - k
    
    valid_k = (pk >= 0) & (pk < N)
    valid_m = (pm >= 0) & (pm < N)
    
    pk_c = np.clip(pk, 0, N-1)
    pm_c = np.clip(pm, 0, N-1)
    
    # delta_remove[p] = -(1-h[p+k])*valid_k + h[p-k]*valid_m
    delta_remove = -(1.0 - h[pk_c]) * valid_k.astype(float) + h[pm_c] * valid_m.astype(float)
    
    # delta_add[p] = (1-h[p+k])*valid_k - h[p-k]*valid_m
    delta_add = (1.0 - h[pk_c]) * valid_k.astype(float) - h[pm_c] * valid_m.astype(float)
    
    return delta_remove, delta_add

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / N)

def project_sum(h, target):

        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h_new = h + lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h_new, target)
        lr *= lr_decay
        if t % 400 == 0:
            hb = make_binary(h, N)
            h = 0.5 * h + 0.5 * hb
    return best_h, best_val

def local_search(h, N, deadline, max_sweeps=5000, sample_frac=0.5, anneal=0.0):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pk_mat = pos + sv
    pm_mat = pos - sv
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > (80 if anneal == 0 else 40):
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        vk = ((pk_mat >= 0) & (pk_mat < N)).astype(float)
        vm = ((pm_mat >= 0) & (pm_mat < N)).astype(float)
        pkc = np.clip(pk_mat, 0, N-1)
        pmc = np.clip(pm_mat, 0, N-1)
        dr = -(1.0 - h[pkc]) * vk + h[pmc] * vm
        da = (1.0 - h[pkc]) * vk - h[pmc] * vm
        ns = max(1, int(len(ones) * sample_frac))
        samp = np.random.choice(ones, ns, replace=False) if ns < len(ones) else ones
        bd = 0.0
        bpq = None
        for p in samp:
            base = c - dr[p]
            nm = (base[None, :] + da[zeros]).max(axis=1)
            mi = int(np.argmin(nm))
            d = float(nm[mi]) - P
            if d < bd:
                bd = d
                bpq = (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    best_N = N0

    # Phase 1: Continuous gradient optimization on initial
    d1 = min(deadline - 500, t0 + 250)
    h_cont = h0.copy()
    h_cont, pk1 = grad_opt(h_cont, N0, d1, lr0=0.05, T0=12.0)
    c5_1 = pk1 * (2.0 / N0)
    if c5_1 < best_c5:
        best_c5, best_h, best_N = c5_1, h_cont, N0

    # Phase 2: Binary local search from continuous solution
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 150)
        h_bin = make_binary(best_h, best_N)
        h_bin, cb = local_search(h_bin, best_N, d2, max_sweeps=3000, sample_frac=0.4)
        c5_2 = cb * (2.0 / best_N)
        if c5_2 < best_c5:
            best_c5, best_h, best_N = c5_2, h_bin, N0

    # Phase 3: Second continuous pass (finer)
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 120)
        h_c2 = best_h.astype(float).copy()
        h_c2 += np.random.randn(N0) * 0.01
        h_c2 = project_sum(h_c2, N0 / 2.0)
        h_c2, pk3 = grad_opt(h_c2, N0, d3, lr0=0.03, T0=8.0, T_decay=0.9995)
        c5_3 = pk3 * (2.0 / N0)
        if c5_3 < best_c5:
            best_c5, best_h, best_N = c5_3, h_c2, N0

    # Phase 4: Intensive binary local search (full sampling)
    if time.time() < deadline - 200:
        d4 = min(deadline - 150, time.time() + 120)
        h_b2 = make_binary(best_h, best_N)
        h_b2, c4 = local_search(h_b2, best_N, d4, max_sweeps=2000, sample_frac=1.0)
        c5_4 = c4 * (2.0 / best_N)
        if c5_4 < best_c5:
            best_c5, best_h, best_N = c5_4, h_b2, best_N

    # Phase 5: Multi-resolution
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [420, 400, 380, 440, 360, 460]:
        if time.time() > deadline - 50:
            break
        if N_try % 2 != 0:
            N_try += 1
        if N_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        # Continuous optimization at new resolution
        sub_dl1 = min(deadline - 40, time.time() + 50)
        h_new, pn = grad_opt(h_interp, N_try, sub_dl1, lr0=0.04, T0=10.0)
        c5n = pn * (2.0 / N_try)
        if time.time() > deadline - 15:
            continue
        # Binary refinement
        sub_dl2 = min(deadline - 10, time.time() + 50)
        h_bn = make_binary(h_new, N_try)
        h_bn, cbn = local_search(h_bn, N_try, sub_dl2, max_sweeps=1000, sample_frac=0.5)
        c5_bn = cbn * (2.0 / N_try)
        if c5_bn < c5n:
            c5n = c5_bn
            h_new = h_bn
        if c5n < best_c5:
            best_c5, best_h, best_N = c5n, h_new, N_try
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 6: Annealed binary search
    if time.time() < deadline - 40:
        d6 = min(deadline - 20, time.time() + 50)
        h_a = make_binary(best_h, best_N)
        h_a, ca = local_search(h_a, best_N, d6, max_sweeps=500, sample_frac=0.5, anneal=0.12)
        c5_6 = ca * (2.0 / best_N)
        if c5_6 < best_c5:
            best_c5, best_h, best_N = c5_6, h_a, best_N

    # Phase 7: Final continuous polish
    if time.time() < deadline - 10:
        d7 = deadline - 5
        h_f = best_h.astype(float).copy()
        h_f = project_sum(h_f, best_N / 2.0)
        h_f, pk7 = grad_opt(h_f, best_N, d7, lr0=0.02, T0=6.0, T_min=3.0)
        c5_7 = pk7 * (2.0 / best_N)
        if c5_7 < best_c5:
            best_c5, best_h, best_N = c5_7, h_f, best_N

    # Final: try binary version of best and compare
    if time.time() < deadline - 2:
        h_fb = make_binary(best_h, best_N)
        c5_fb = compute_c5(h_fb, best_N)
        c5_fc = compute_c5(best_h, best_N)
        if c5_fb < c5_fc:
            best_h = h_fb
            best_c5 = c5_fb
        else:
            best_c5 = c5_fc

    return best_h, best_c5, len(best_h)
