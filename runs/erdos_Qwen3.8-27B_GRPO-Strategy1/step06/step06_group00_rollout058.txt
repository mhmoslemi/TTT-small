def gs_init(N, n_iters=200, seed=42):

    rng = np.random.RandomState(seed)
    h = np.zeros(N)
    idx = rng.choice(N, N // 2, replace=False)
    h[idx] = 1.0

        amp = np.abs(H)
        # Target: flat amplitude (except DC)
        target_amp = np.ones(N)
        target_amp[0] = amp[0]  # Keep DC (sum constraint)
        # Scale: mean amplitude should match
        target_amp[1:] = np.mean(amp[1:])
        phase = H / (amp + 1e-15)
        H_new = target_amp * phase
        h_new = np.real(np.fft.ifft(H_new))

def compute_D(h, N):
    D = np.zeros((N, 2*N-1))
    # D[i, N-1+d] = (1-h[i+d]) * [0<=i+d<N] - h[i-d] * [0<=i-d<N]

    #   (1-h[i+d]) term: valid for i in [0, N-1-d] (d>=0) or i in [-d, N-1] (d<0)
    #   h[i-d] term: valid for i in [d, N-1] (d>=0) or i in [0, N-1-|d|] (d<0)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_D(h, N):
    D = np.zeros((N, 2*N-1))
    for d in range(-(N-1), N):
        k = N - 1 + d
        if d >= 0:
            D[:N-d, k] = (1.0 - h[d:])
            D[d:, k] -= h[:N-d]
        else:
            s = -d
            D[s:, k] = (1.0 - h[:N-s])
            D[:N-s, k] -= h[s:]
    return D

def binary_greedy(h, N, deadline, max_iters=20000):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1-nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh-n1, replace=False)] = 1
    
    best_c = np.inf
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        if stall > 100:
            # Perturb: random swaps
            n_pert = min(3, len(ones)//20)
            for _ in range(n_pert):
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                h[i], h[j] = 0, 1
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            stall = 0
            continue
        
        # Full greedy: try all pairs
        D = compute_D(h, N)

        # new_corr = corr + D[q,:] - D[p,:]
        # Want to minimize max(new_corr)
        
        # Vectorized: for all (p, q) pairs
        # n_ones * n_zeros can be up to 190*190 = 36100
        # Each entry: 759 values
        # Total: 36100 * 759 = 27.4M - too much memory?
        # 27.4M * 8 bytes = 219 MB - should be OK
        
        c = corr[None, None, :] + D[None, zeros, :] - D[ones, None, :]
        new_max = c.max(axis=2)
        idx = np.unravel_index(np.argmin(new_max), new_max.shape)
        p = ones[idx[0]]

        new_P = float(new_max[idx])

        elif stall > 30:

def fft_corr(h):
    N = len(h)
