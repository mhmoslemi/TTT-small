def full_subgradient(h, c, N, active_lags, weights):
    """Compute subgradient of max(c[active_lags]) w.r.t. h."""
    grad = np.zeros(N)
    center = N - 1
    for w, lag_idx in zip(weights, active_lags):
        L = int(lag_idx - center)
        if L > 0:
            # corr_pos[L] = sum_{i=0}^{N-1-L} h[i]*(1-h[i+L])
            seg1 = N - L
            grad[:seg1] += w * (1.0 - h[L:])  # h[i] * (1-h[i+L]): derivative w.r.t h[i]

def subgrad(h, c, N, n_act=80):
    center = N-1
    top = np.argpartition(c, -n_act)[-n_act:]
    w = np.exp((c[top]-c.max())/0.8)
    w /= w.sum()
    g = np.zeros(N)

            g[L:] -= wi*h[:s]
        elif L < 0:
            M = -L
            s = N-M
            g[M:] += wi*(1.0-h[:s])
            g[:s] -= wi*h[M:]
    return g
