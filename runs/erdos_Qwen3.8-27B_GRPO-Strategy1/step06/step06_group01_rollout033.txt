import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    return h

def make_binary(h, nh, rng):
    h = h.astype(float).copy()
    n1 = int(np.sum(h > 0.5))
    if n1 > nh:
        idx = np.where(h > 0.5)[0]
        rem = rng.choice(idx, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        idx = np.where(h <= 0.5)[0]
        add = rng.choice(idx, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def recompute_delta(h, pos, N):
    N2 = 2 * N - 1
    d = np.zeros(N2)
    lo1, hi1 = pos, pos + N - 1
    idx1 = np.arange(lo1, hi1 + 1)
    si1 = pos + N - 1 - idx1

    if h[pos] == 1:
        d[lo1:hi1 + 1] = -(1.0 - h[si1])
        d[lo2:hi2 + 1] += h[si2]
    else:
        d[lo1:hi1 + 1] = (1.0 - h[si1])
        d[lo2:hi2 + 1] -= h[si2]
    return d

    N2 = 2 * N - 1
    corr = fft_corr(h).copy()
    best_c = float(corr.max())
    best_h = h.copy()
    deltas = np.zeros((N, N2))
    for i in range(N):
        deltas[i] = recompute_delta(h, i, N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if len(ones) < 1 or len(zeros) < 1:
            break
        top_idx = np.argpartition(corr, -peak_k)[-peak_k:]
        weight = np.zeros(N)
        for pk in top_idx:
            pk = int(pk)
            lo_i = max(0, N - 1 - pk)
            hi_i = min(N - 1, 2 * N - 2 - pk)
            if hi_i >= lo_i:
                cand = np.arange(lo_i, hi_i + 1)
                si = cand + pk - N + 1
                mask = (h[cand] == 1) & (h[si] == 0)
                if mask.any():

def nesterov_cont(h_in, N, deadline, k_start=35, k_min=5, lr0=0.018):
    h = h_in.astype(float).copy()
    target = N / 2.0
    hx = h.copy()
    best_c = compute_c5(hx)
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.82
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        k = max(k_min, k_start - it // 30)
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for t in top:
            t = int(t)
            L = t - (N - 1)
            if L == 0:
                grad += 1.0 - 2.0 * y
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 40 == 0:
            lr *= 0.975
            beta = min(0.96, 0.82 + 0.0002 * it)
        if it % 15 == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    c = compute_c5(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c
