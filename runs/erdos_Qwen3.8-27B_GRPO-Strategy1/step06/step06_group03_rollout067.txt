def swap_delta(corr, h, p, q, top_idx, N):
    """Compute new values of corr at top_idx after swapping h[p]:1→0, h[q]:0→1.
    Returns the new maximum over top_idx."""
    center = N - 1
    new_max = -np.inf
    for ai in top_idx:
        L = ai - center
        dv = 0.0
        if L > 0:
            if p < N - L: dv -= (1.0 - h[p+L])
            if p >= L: dv += h[p-L]
            if q < N - L: dv += (1.0 - h[q+L])
            if q >= L: dv -= h[q-L]
        elif L < 0:
            M = -L
            if p >= M: dv -= (1.0 - h[p-M])
            if p+M < N: dv += h[p+M]
            if q >= M: dv += (1.0 - h[q-M])
            if q+M < N: dv -= h[q+M]
        new_val = corr[ai] + dv
        if new_val > new_max:
            new_max = new_val
    return new_max

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def swap_delta_max(corr, h, p, q, top_idx, N):
    center = N - 1
    new_max = 0.0
    for ai in top_idx:
        L = int(ai) - center
        dv = 0.0
        if L > 0:
            if p < N - L: dv -= (1.0 - h[p + L])
            if p >= L: dv += h[p - L]
            if q < N - L: dv += (1.0 - h[q + L])
            if q >= L: dv -= h[q - L]
        elif L < 0:
            M = -L
            if p >= M: dv -= (1.0 - h[p - M])
            if p + M < N: dv += h[p + M]
            if q >= M: dv += (1.0 - h[q - M])
            if q + M < N: dv -= h[q + M]
        nv = corr[ai] + dv
        if nv > new_max:

    return new_max

def subgrad_topk(y, corr, k, N):
    grad = np.zeros(N)
    center = N - 1
    top = np.argpartition(corr, -k)[-k:] if k < len(corr) else np.arange(len(corr))

        elif L < 0:
            M = -L
            if p >= M: dv -= (1.0 - h[p - M])
            if p + M < N: dv += h[p + M]
            if q >= M: dv += (1.0 - h[q - M])
            if q + M < N: dv -= h[q + M]
        nv = corr[ai] + dv
        if nv > new_max:
            new_max = nv
    return new_max

def make_binary(h, N):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[nh:]] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1.0
    return h

def binary_opt(h, N, deadline, k=20, T0=4.0, sample_n=140):
    h = make_binary(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    T = T0
    stall = 0
    while time.time() < deadline:
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_idx = np.argsort(c)[-k:]
        ns = min(len(ones), sample_n)
        nz = min(len(zeros), sample_n)
        p_list = np.random.choice(ones, ns, replace=False) if len(ones) > ns else ones
        q_list = np.random.choice(zeros, nz, replace=False) if len(zeros) > nz else zeros
        pairs = [(int(p), int(q)) for p in p_list for q in q_list]
        np.random.shuffle(pairs)
        n_try = min(len(pairs), 3000)
        best_nm = P
        best_pq = None
        accepted = False
        for p, q in pairs[:n_try]:
            nm = swap_delta_max(c, h, p, q, top_idx, N)
            if nm < best_nm:
                best_nm = nm
                best_pq = (p, q)
            elif nm < P and T > 0.01 and np.random.random() < np.exp(-(nm - P) / T):
                best_pq = (p, q)
                break
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            accepted = True
        if not accepted:
            T *= 0.998
        if stall > 60 and T > 0.01:
            T *= 0.95
    return best_h, best_c

def subgrad_topk(y, corr, k, N):
    grad = np.zeros(N)
    center = N - 1
    top = np.argpartition(corr, -k)[-k:] if k < len(corr) else np.arange(len(corr))
    for ai in top:
        L = int(ai) - center
        if L == 0:
            grad += 1.0 - 2.0 * y
        elif L > 0:
            grad[:N - L] += 1.0 - y[L:]
            grad[L:] -= y[:N - L]
        else:
            M = -L
            grad[:N - M] -= y[M:]
            grad[M:] += 1.0 - y[:N - M]
    grad /= len(top)
    return grad

def adam_opt(h, N, deadline, k=50, lr=0.04):
    h = h.astype(float).copy()
    target = N / 2.0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        g = subgrad_topk(h, corr, k, N)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    d1 = min(deadline - 350, t0 + 300)
    h1, _ = binary_opt(best_h, N0, d1, k=25, T0=5.0, sample_n=155)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    if time.time() < deadline - 280:
        d2 = min(deadline - 220, time.time() + 100)
        hc = make_binary(best_h, N0).astype(float) + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        ha, ca = adam_opt(hc, N0, d2, k=60, lr=0.03)
        if ca < best_c5:
            best_c5, best_h = ca, ha

    if time.time() < deadline - 200:
        d3 = min(deadline - 150, time.time() + 140)
        h3 = make_binary(best_h, N0)
        h3b, _ = binary_opt(h3, N0, d3, k=30, T0=3.0, sample_n=170)
        c3 = compute_c5(h3b)
        if c3 < best_c5:
            best_c5, best_h = c3, h3b

    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in [400, 420, 440, 460, 480, 500, 550, 600, 700, 800]:
        if time.time() > deadline - 25:
            break
        if Nt % 2 != 0:
            Nt += 1
        if Nt == len(best_h):
            continue
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(xn, x_best, best_h)
        hb = make_binary(hi, Nt)
        dl = min(deadline - 15, time.time() + 40)
        hr, _ = binary_opt(hb, Nt, dl, k=20, T0=2.0, sample_n=110)
        c5r = compute_c5(hr)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr
            x_best = np.linspace(0, 2, Nt, endpoint=False)

    if time.time() < deadline - 12:
        hF = best_h + np.random.randn(len(best_h)) * 0.02
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, cF = adam_opt(hF, len(best_h), deadline - 5, k=70, lr=0.015)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    if time.time() > t0 + 5:
        hF2 = make_binary(best_h, len(best_h))
        hF2, _ = binary_opt(hF2, len(best_h), deadline - 0.5, k=20, T0=0.5, sample_n=80)
        cF2 = compute_c5(hF2)
        if cF2 < best_c5:
            best_c5, best_h = cF2, hF2

    return best_h, best_c5, len(best_h)
