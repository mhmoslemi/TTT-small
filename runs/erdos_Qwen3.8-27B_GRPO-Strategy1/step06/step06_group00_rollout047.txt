def smooth_max_grad(h, N, temp):
    corr = fft_corr(h)
    dx = 2.0/N
    scaled = corr / temp
    max_sc = np.max(scaled)
    weights = np.exp(scaled - max_sc)
    weights /= np.sum(weights)

    # For each k, gradient of R(k) w.r.t. h[i] is as before

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(z, target):
    z = np.clip(z, 0.0, 1.0)
    s = z.sum()
    if abs(s - target) < 1e-12:
        return z.copy()
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(z - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)

def compute_deltas(positions, h, N):
    n_pos = len(positions)
    if n_pos == 0:
        return np.zeros((0, 2*N-1))
    L_arr = np.arange(2*N-1) - (N-1)
    pL = positions[:, None] + L_arr[None, :]
    pLn = positions[:, None] - L_arr[None, :]
    D = np.zeros((n_pos, 2*N-1))
    m1 = (pL >= 0) & (pL < N)
    m2 = (pLn >= 0) & (pLn < N)
    D[m1] += (1.0 - h[pL[m1]])
    D[m2] += (-h[pLn[m2]])
    return D

def subgrad_topk(h, corr, N, k_top=50):
    grad = np.zeros(N)
    top = np.argpartition(corr, -k_top)[-k_top:]
    center = N - 1
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += (1.0 - h[:N-L])
            grad[:N-L] += (-h[L:])
        elif -N < L < 0:
            M = -L
            grad[:N-M] += (1.0 - h[M:])
            grad[M:] += (-h[:N-M])
    grad /= k_top
    return grad

def smooth_grad(h, corr, N, temp):
    scaled = corr / temp
    mx = np.max(scaled)
    w = np.exp(scaled - mx)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for ki in range(2*N-1):
        if w[ki] < 1e-12:
            continue
        L = ki - center
        wi = w[ki]
        if L == 0:
            grad += wi * (1.0 - 2.0*h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] += wi * (-h[L:])
        elif -N < L < 0:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] += wi * (-h[:N-M])
    return grad * (2.0 / N)

def smooth_max_opt(h, N, deadline, temp0=0.05, temp_min=0.005, lr0=0.04):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(np.max(corr)) * (2.0/N)
    beta = 0.5
    vel = np.zeros(N)
    it = 0
    while time.time() < deadline:
        y = h + vel
        corr = fft_corr(y)
        pk = float(np.max(corr)) * (2.0/N)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        progress = min(1.0, it / 500.0)
        temp = temp0 * (temp_min/temp0) ** progress
        grad = smooth_grad(y, corr, N, temp)
        lr = lr0 / (1.0 + 0.005 * it)
        z = y - lr * grad * 10.0
        h_new = project_01(z, target)
        beta_new = (1.0 + np.sqrt(1.0 + 4.0*beta*beta)) / 2.0
        vel = ((h - h_new) * beta) / beta_new
        beta = beta_new
        h = h_new
        it += 1
    return best_h, best_pk

def subgrad_descent(h, N, deadline, k_top=60, lr0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(np.max(corr)) * (2.0/N)
    vel = np.zeros(N)
    beta = 0.5
    it = 0
    while time.time() < deadline:
        y = h + vel
        corr = fft_corr(y)
        pk = float(np.max(corr)) * (2.0/N)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = subgrad_topk(y, corr, N, k_top)
        lr = lr0 / (1.0 + 0.003 * it)
        z = y - lr * grad * 10.0
        h_new = project_01(z, target)
        beta_new = (1.0 + np.sqrt(1.0 + 4.0*beta*beta)) / 2.0
        vel = ((h - h_new) * beta) / beta_new
        beta = beta_new
        h = h_new
        it += 1
    return best_h, best_pk

def binary_ls(h, N, deadline, max_iters=3000, sample_sz=80, temp=0.0):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1-nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh-n1, replace=False)] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c:
            best_c = P
            best_h = h.copy()
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        if ns < 2 or nz < 2:
            break
        os_ = np.random.choice(ones, ns, replace=False) if ns < len(ones) else ones
        zs = np.random.choice(zeros, nz, replace=False) if nz < len(zeros) else zeros
        De = compute_deltas(os_, h, N)
        Df = compute_deltas(zs, h, N)
        new = corr[None, None, :] - De[:, None, :] + Df[None, :, :]
        newP = new.max(axis=2)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-12:
            h[os_[mi[1]]] = 0
            h[zs_[mi[0]]] = 1
        elif temp > 0 and np.random.random() < temp:
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p] = 0
            h[q] = 1
        else:
            for _ in range(3):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0
                h[q] = 1
            if it % 80 == 0:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return best_h, best_c

def make_binary(h, N):
    hb = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(hb.sum())

        else:
            idx = np.where(hb == 0)[0]
            hb[np.random.choice(idx, nh-n1, replace=False)] = 1
    return hb
