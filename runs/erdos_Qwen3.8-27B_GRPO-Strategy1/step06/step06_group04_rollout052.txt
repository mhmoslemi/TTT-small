import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_softmax(h, corr, top_k, temp):
    N = len(h)
    top = np.argpartition(corr, -top_k)[-top_k:]
    m = corr[top].max()
    w = np.exp((corr[top] - m) / max(temp, 1e-10))
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for wi, ai in zip(w, top):
        L = int(ai - center)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            seg = N - M
            if M < N:
                grad[:seg] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:seg]
    return grad

def adam_opt(h0, N, deadline, top_k=80, lr0=0.012, temp0=0.0015):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr0
    temp = temp0
    stall = 0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 400:
            h = best_h + np.random.randn(N) * 0.03
            h = project_sum(h, target)
            m *= 0.3
            v *= 0.3
            lr = max(lr0, lr * 1.2)
            stall = 0
        g = grad_softmax(h, corr, top_k, temp)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        mh = m / (1.0 - b1**t)
        vh = v / (1.0 - b2**t)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        if t % 400 == 0:
            lr *= 0.985
            temp = max(0.0003, temp * 0.99)
    return best_h, best_pk * (2.0 / N)

def binary_search(h0, N, deadline, n_inner=25, use_2swap=True):

def interpol(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, N_old, endpoint=False)
    x_new = np.linspace(0, 2.0, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)
