```python
def compute_gradient(h, w, N):
    """w is the softmax weight vector of length 2N-1."""
    grad = np.zeros(N)
    for m in range(2*N-1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (N - 1)
        if k >= 0:
            # (1-h[j+k]) valid for j = 0..N-1-k
            # -h[j-k] valid for j = k..N-1
            if N-1-k >= 0:
                grad[:N-k] += wm * (1.0 - h[k:])
            if N-1-k >= 0:
                grad[k:] -= wm * h[:N-k]
        else:
            M = -k
            # (1-h[j+M]) valid for j = 0..N-1-M
            # -h[j-M] valid for j = M..N-1
            if N-1-M >= 0:
                grad[:N-M] += wm * (1.0 - h[M:])
            if N-1-M >= 0:
                grad[M:] -= wm * h[:N-M]
    return grad
```

This is O(N²) per call but with very small constants in numpy. For N=400, 799 iterations of vector ops on arrays of size ~400: should take < 1ms.

Let me also think about a different approach: **subgradient method** for the max-of-linears (when h is binary) or max-of-quadratics (when h is continuous).

For continuous h, c_k(h) = Σ_i h[i](1-h[i+k]) is quadratic in h. The objective max_k c_k is the max of quadratic functions, which is convex (max of convex functions is convex). Wait, is c_k convex in h?

c_k = Σ_i h[i] - h[i]h[i+k]

The Hessian of -h[i]h[i+k] with respect to (h[i], h[i+k]) is [[0, -1], [-1, 0]], which is indefinite. So c_k is NOT convex in h!

So the problem is non-convex for continuous h. The soft-max approach will find local minima.

For binary h, c_k is linear in {h[i]} (since h[i]² = h[i]), so c_k is linear, and max of linears is convex. But the feasible set {h ∈ {0,1}^N : sum = N/2} is not convex.

```python
import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
```

```python
def swap_delta(h, p, N, flip_val):
```

c_k = Σ_i h[i]·(1-h[i+k])

If h[p] changes by δ (from 1 to 0, δ=-1, or from 0 to 1, δ=+1):

The terms affected:
- Where i=p: h[p]·(1-h[p+k]) changes by δ·(1-h[p+k])
- Where i+k=p, i.e., i=p-k: h[p-k]·(1-h[p]) changes by h[p-k]·(-δ) = -δ·h[p-k]

So Δc[m] = δ·(1-h[p+k])·[0≤p+k<N] - δ·h[p-k]·[0≤p-k<N]

For δ=-1 (1→0): Δc = -(1-h[p+k])·[valid] + h[p-k]·[valid] ✓
For δ=+1 (0→1): Δc = +(1-h[p+k])·[valid] - h[p-k]·[valid] ✓

Great, so:
Δc[m] = δ·[(1-h[p+k])·[0≤p+k<N] - h[p-k]·[0≤p-k<N]]

```python
def swap_delta(h, p, N, delta):

    k_vals = np.arange(2*N-1) - (N-1)

    dc[valid1] += delta * (1.0 - h[p + k_vals[valid1]])
    dc[valid2] -= delta * h[p - k_vals[valid2]]
```

For a continuous solution, we can get any C₅ value. The soft-max gradient approach should be able to fine-tune the continuous solution to reduce C₅ below 0.380893.

1. Starting from initial_h_values
2. Doing soft-max gradient descent with careful β scheduling
3. Doing binary local search for discrete improvements
4. Trying a few different N values
5. Being time-aware

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_grad(h, w, N):
    L = 2 * N - 1
    grad = np.zeros(N)
    for m in range(L):
        wm = w[m]
        if wm < 1e-15:
            continue
        k = m - (N - 1)
        if k >= 0:
            nv = N - k
            if nv > 0:
                grad[:nv] += wm * (1.0 - h[k:])
                grad[k:] -= wm * h[:nv]
        else:
            M = -k
            nv = N - M
            if nv > 0:
                grad[:nv] += wm * (1.0 - h[M:])
                grad[M:] -= wm * h[:nv]
    return grad

def smax_opt(h0, N, dl, lr=0.003, b0=20, b1=800):
    tgt = N / 2.0
    h = h0.copy()
    bh = h.copy()
    bp = float(np.max(fft_corr(h)))
    it = 0
    while time.time() < dl:
        beta = b0 + (b1 - b0) * min(1.0, it / 2000.0)
        corr = fft_corr(h)
        mc = float(corr.max())
        ec = np.exp(beta * (corr - mc))
        sw = ec.sum()
        w = ec / sw if sw > 1e-300 else np.eye(1, 2*N-1)[int(np.argmax(corr))]
        g = compute_grad(h, w, N)
        hn = project_sum(h - lr * g, tgt)
        pk = float(np.max(fft_corr(hn)))
        if pk < bp:
            bp = pk
            bh = hn.copy()
        h = hn
        it += 1
        if it % 500 == 0:
            lr *= 0.93
    return bh, bp

def bin_search(h0, N, dl):
    h = (h0 > 0.5).astype(float)
    tgt = N // 2
    n1 = int(h.sum())
    if n1 > tgt:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[tgt:]] = 0.0
    elif n1 < tgt:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:tgt-n1]] = 1.0
    bp = float(np.max(fft_corr(h)))
    ni = 0
    ka = np.arange(2*N-1) - (N-1)
    while time.time() < dl:
        c = fft_corr(h)
        pk = float(c.max())
        if pk < bp - 1e-12:
            bp = pk; ni = 0
        else:
            ni += 1
        if ni > 1000: break
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)==0 or len(zeros)==0: break
        bs = pk; bpq = None
        ns = min(len(ones), 35)
        pl = ones if ns==len(ones) else ones[np.random.choice(len(ones), ns, replace=False)]
        for p in pl:
            v1 = (p+ka>=0)&(p+ka<N)
            v2 = (p-ka>=0)&(p-ka<N)
            dcp = np.zeros(2*N-1)
            dcp[v1] = -(1.0-h[p+ka[v1]])
            dcp[v2] += h[p-ka[v2]]
            base = c + dcp
            for q in zeros:
                v1q = (q+ka>=0)&(q+ka<N)
                v2q = (q-ka>=0)&(q-ka<N)
                t = base.copy()
                t[v1q] += (1.0-h[q+ka[v1q]])
                t[v2q] -= h[q-ka[v2q]]
                sc = float(t.max())
                if sc < bs:
                    bs = sc; bpq = (int(p), int(q))
        if bpq:
            h[bpq[0]]=0; h[bpq[1]]=1
        else:
            h[ones[np.random.randint(len(ones))]], h[zeros[np.random.randint(len(zeros))]] = 0.0, 1.0
    return h, float(np.max(fft_corr(h)))
