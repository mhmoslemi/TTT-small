import time

def fft_corr(h):

    L = 2 * N - 1

    G = np.fft.fft(np.flip(1.0 - h), L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_delta_remove(h, p, shifts, N):
    """Effect on correlation of removing 1 at position p."""
    pk = p + shifts
    pm = p - shifts
    vk = (pk >= 0) & (pk < N)
    vm = (pm >= 0) & (pm < N)
    hk = np.where(vk, h[np.clip(pk, 0, N-1)], 0.0)
    hm = np.where(vm, h[np.clip(pm, 0, N-1)], 0.0)
    return -(1.0 - hk) * vk.astype(float) + hm * vm.astype(float)

def get_delta_add(h, positions, shifts, N):
    """Effect on correlation of adding 1 at positions (batch)."""
    qk = positions[:, None] + shifts[None, :]
    qm = positions[:, None] - shifts[None, :]
    vk = (qk >= 0) & (qk < N)
    vm = (qm >= 0) & (qm < N)
    hk = np.where(vk, h[np.clip(qk, 0, N-1)], 0.0)
    hm = np.where(vm, h[np.clip(qm, 0, N-1)], 0.0)
    return (1.0 - hk) * vk.astype(float) - hm * vm.astype(float)

def binary_sa(h, N, deadline, T0=0.5, T_min=0.01, max_sweeps=5000):
    """Binary simulated annealing for the overlap problem."""
    h = h.astype(float).copy()
    N_half = N // 2
    shifts = np.arange(2*N-1) - (N-1)
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    T = T0
    T_factor = (T_min / T0) ** (1.0 / max_sweeps)
    
    for it in range(max_sweeps):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        # Try a random move: pick random p from ones, q from zeros
        p = int(ones[np.random.randint(len(ones))])
        # Sample candidate q's
        n_cand = min(20, len(zeros))
        q_cands = zeros[np.random.choice(len(zeros), n_cand, replace=False)]
        
        c = fft_corr(h)
        P = float(c.max())
        
        delta_p = get_delta_remove(h, p, shifts, N)
        delta_q = get_delta_add(h, q_cands, shifts, N)
        
        base = c + delta_p
        new_peaks = np.max(base[None, :] + delta_q, axis=1)
        
        best_q_i = int(np.argmin(new_peaks))
        q = int(q_cands[best_q_i])
        new_P = float(new_peaks[best_q_i])
        
        # Accept/reject
        delta = new_P - P
        if delta <= 0 or (T > 1e-6 and np.random.random() < np.exp(-delta / T)):
            h[p] = 0.0
            h[q] = 1.0
            if new_P < best_c:
                best_c = new_P
                best_h = h.copy()
        
        T *= T_factor
    
    return best_h, best_c

    """D[p, k] = delta correlation at shift k when 1 is at position p (vs p=background)."""
    shifts = np.arange(2*N-1) - (N-1)
    # D[p, k] for removing 1 at p: effect on corr[k]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    D = np.empty((N, 2*N-1))
    for p in range(N):
        pk = p + shifts
        pm = p - shifts
        vk = (pk >= 0) & (pk < N)
        vm = (pm >= 0) & (pm < N)
        hk = np.where(vk, h[np.clip(pk, 0, N-1)], 0.0)
        hm = np.where(vm, h[np.clip(pm, 0, N-1)], 0.0)
        D[p] = -(1.0 - hk) * vk.astype(float) + hm * vm.astype(float)
    return D

def greedy_binary(h, N, deadline, max_sweeps=2000, sample_p=150):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    T = 0.03
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        T = 0.03 * (0.995 ** no_improve)
        
        ones = np.where(h == 1)[
            :]
        zeros = np.where(h == 0)[
            :]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        D = compute_D(h, N)
        
        if len(ones) > sample_p:
            p_list = ones[np.random.permutation(len(ones))[:sample_p]]
        else:
            p_list = ones
        
        best_score = P + 1.0
        best_pq = None
        
        for p in p_list:
            base = c - D[p]  # corr after removing 1 at p
            cand = base[None, :] - D[zeros]  # corr after also adding 1 at each zero
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))

                best_pq = (int(p), int(zeros[mi]))

                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
        else:
            h[int(ones[np.random.randint(len(ones))])] = 0.0

def adam_continuous(h, N, deadline, T0=80, T_decay=0.997, lr0=0.08):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)) * dx)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2 = 0.8, 0.999
    eps = 1e-8
    lr = lr0
    T = T0
    shifts = np.arange(2*N-1) - (N-1)
    it = 0
    
    while time.time() < deadline and T > 1.0:
        corr = fft_corr(h)
        pk = float(corr.max() * dx)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Soft-max weights
        shifted_corr = corr - corr.max()
        exps = np.exp(T * shifted_corr)
        w = exps / exps.sum()
        
        # Gradient: g[j] = sum_s w[s] * [(1-h[j+s])*[valid] - h[j-s]*[valid]]
        # = sum_s w[s] * (1-2h[j+s shifted stuff])... 
        # Use the D matrix trick: g[j] = sum_s w' [s] where w'[s] = w for the shift
        # Actually: g[j] = sum over shifts s of w_shift[s] * d_corr[s]/d_h[j]
        # d_corr[s]/d_h[j] = -(1-h[j+s])*[0<=j+s<N] + h[j-s]*[0<=j-s<N]  (for removing)
        # Wait no, d/dh[j] of corr[s] = (1-h[j+s])*[j+s<N] - h[j-s]*[j-s>=0]
        # So g[j] = sum_s w[s] * ((1-h[j+s])*[valid] - h[j-s]*[valid])
        
        # Compute using D-like approach
        # For each shift s with weight w_s:
        #   contribution to g[j] from this shift: w_s * ((1-h[j+s])*[0<=j+s<N] - h[j-s]*[0<=j-s<N])
        
        # Vectorized:
        g = np.zeros(N)
        # Only process shifts with significant weight
        thresh = 1.0 / (2*N)
        active = np.where(w > thresh)[0]
        
        for si in active:
            s = int(si - (N-1))  # actual shift value
            ws = float(w[si])
            if s >= 0:
                # (1-h[j+s]) for j < N-s
                g[:N-s] += ws * (1.0 - h[s:])
                # -h[j-s] for j >= s
                g[s:] -= ws * h[:N-s]
            else:
                m_val = -s
                # (1-h[j+m_val]) for j >= m_val... wait
                # (1-h[j+s]) where s<0: j+s = j-m_val, valid when j >= m_val
                g[m_val:] += ws * (1.0 - h[:N-m_val])
                # -h[j-s] = -h[j+m_val], valid when j+m_val < N, i.e., j < N-m_val
                g[:N-m_val] -= ws * h[m_val:]
        
        m = b1 * m + (1-b1) * g
        v = b2 * v + (1-b2) * g**2
        mh = m / (1 - b1**(it+1))
        vh = v / (1 - b2**(it+1))
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        T *= T_decay
        it += 1
    
    return best_h, best_pk

def fix_sum_binary(h, N):
    nh = N // 2
    n1 = int(np.round(h.sum()))
    h = h.astype(float)
    if n1 > nh:

        add = idx[np.random.permutation(len(idx))][:nh-n1]

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    D = np.empty((N, 2 * N - 1))
    for p in range(N):
        pm = p - shifts
        pp = p + shifts
        vm = (pm >= 0) & (pm < N)
        vp = (pp >= 0) & (pp < N)
        hm = np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0)
        hp = np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
        D[p] = -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)
    return D

def fix_sum(h, N):
