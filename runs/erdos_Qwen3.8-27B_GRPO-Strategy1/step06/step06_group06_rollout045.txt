def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:

    G = np.fft.rfft(np.flip(1.0-h.astype(float)), L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]

def delta_corr(h, N, p, q):
    """Compute change in correlation vector when h[p]:1->0, h[q]:0->1"""
    s_arr = np.arange(-(N-1), N)
    d = np.zeros(2*N-1)
    # For shift s: Δcorr[s] = -(1-h[p-s]) + h[p+s] + (1-h[q-s]) - h[q+s]
    # where out-of-bounds = 0
    ps = p - s_arr  # p-s
    pm = p + s_arr  # p+s
    qs = q - s_arr  # q-s
    qm = q + s_arr  # q+s
    v1 = (ps >= 0) & (ps < N)
    v2 = (pm >= 0) & (pm < N)
    v3 = (qs >= 0) & (qs < N)

    d[v2] += h[pm[v2]]
    d[v3] += (1.0 - h[qs[v3]])
    d[v4] -= h[qm[v4]]
    return d

import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]

def delta_vec(h, N, p, q):
    """O(N) computation of correlation change for swap p:1->0, q:0->1"""
    s = np.arange(-(N-1), N)
    d = np.zeros(2*N-1)
    # Δcorr at shift s: -(1-h[p-s]) + h[p+s] + (1-h[q-s]) - h[q+s]
    a = p - s  # p-s
    b = p + s  # p+s  
    c = q - s  # q-s
    e = q + s  # q+s
    m1 = (a>=0)&(a<N); d[m1] -= (1.0-h[a[m1]])
    m2 = (b>=0)&(b<N); d[m2] += h[b[m2]]
    m3 = (c>=0)&(c<N); d[m3] += (1.0-h[c[m3]])
    m4 = (e>=0)&(e<N); d[m4] -= h[e[m4]]
    return d

def to_binary(h, N):
    """Ensure exactly N//2 ones"""
    h = h.astype(float).copy()
    target = N // 2
    h[(h < 0.5)] = 0
    h[(h >= 0.5)] = 1
    n1 = int(h.sum())
    if n1 > target:
        idx = np.random.permutation(np.where(h==1)[0])[:n1-target]
        h[idx] = 0
    elif n1 < target:
        idx = np.random.permutation(np.where(h==0)[0])[:target-n1]
        h[idx] = 1
    return h

def guided_moves(h, N, corr, ones, zeros, top_k=8):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))[:2*N-1]

def delta_vec(h, N, p, q):
    s = np.arange(-(N-1), N)
    d = np.zeros(2*N-1)
    a = p - s; b = p + s; c = q - s; e = q + s
    m1 = (a>=0)&(a<N); d[m1] -= (1.0 - h[a[m1]])
    m2 = (b>=0)&(b<N); d[m2] += h[b[m2]]
    m3 = (c>=0)&(c<N); d[m3] += (1.0 - h[c[m3]])
    m4 = (e>=0)&(e<N); d[m4] -= h[e[m4]]
    return d

def to_binary(h, N):
    h = h.astype(float).copy()
    target = N // 2
    h[h < 0.5] = 0
    h[h >= 0.5] = 1
    n1 = int(h.sum())
    if n1 > target:
        idx = np.random.permutation(np.where(h==1)[0])[:n1-target]
        h[idx] = 0
    elif n1 < target:
        idx = np.random.permutation(np.where(h==0)[0])[:target-n1]
        h[idx] = 1
    return h

def pick_guided_moves(h, N, corr, ones, zeros, top_k=5):
    cands = []
    top_idx = np.argpartition(-corr, top_k)[:top_k]
    for si in top_idx:
        s = int(si - (N-1))
        ps = ones - s; pm = ones + s
        qs = zeros - s; qm = zeros + s
        p_sc = np.zeros(len(ones))
        q_sc = np.zeros(len(zeros))
        m1=(ps>=0)&(ps<N); p_sc[m1]+=h[ps[m1]]
        m2=(pm>=0)&(pm<N); p_sc[m2]+=h[pm[m2]]
        m3=(qs>=0)&(qs<N); q_sc[m3]+=h[qs[m3]]
        m4=(qm>=0)&(qm<N); q_sc[m4]+=h[qm[m4]]
        for pi in np.argsort(p_sc)[:2]:
            for qi in np.argsort(-q_sc)[:2]:
                p = int(ones[pi]); q = int(zeros[qi])
                d = delta_vec(h, N, p, q)
                cands.append((p, q, float((corr+d).max())))
    return cands

def anneal_h(h_init, N, deadline, T0=0.035, cool=0.9996, top_k=5, restarts=4):
    h = h_init.astype(float).copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    for r in range(restarts):
        T = T0 * (0.55 ** r)
        no_imp = 0
        while time.time() < deadline and T > 0.0002:
            cands = pick_guided_moves(h, N, corr, ones, zeros, top_k)
            if not cands:
                break
            maxes = np.array([c[2] for c in cands])
            deltas = maxes - current_pk
            w = np.exp(-np.maximum(deltas, 0) / max(T, 1e-7))
            if w.sum() < 1e-12:
                w = np.ones(len(cands))
            w /= w.sum()
            idx = np.random.choice(len(cands), p=w)
            p, q, _ = cands[idx]
            d = delta_vec(h, N, p, q)
            nc = corr + d
            npk = float(nc.max())
            delta = npk - current_pk
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
                h[p]=0; h[q]=1
                corr = nc; current_pk = npk
                ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
                if npk < best_pk:
                    best_pk = npk; best_h = h.copy(); no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1
            T *= cool
            if no_imp > 400:
                h = best_h.copy(); corr = fft_corr(h)
                current_pk = best_pk
                ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
                no_imp = 0
                T = max(T*1.8, T0*0.25)
    return best_h, best_pk

def perturb(h, N, frac=0.05):
    h = h.copy()
    n_flip = max(2, int(N * frac))
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    pi = np.random.choice(ones, min(n_flip, len(ones)), replace=False)
    qi = np.random.choice(zeros, min(n_flip, len(zeros)), replace=False)
    h[pi[:n_flip]] = 0
    h[qi[:n_flip]] = 1
    return to_binary(h, N)
