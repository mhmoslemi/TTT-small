            L = int(k - (N - 1))
            if L == 0:
                g += (1 - 2 * y)
            elif L > 0:
                g[L:] += (1 - y[:N - L])
                g[:N - L] -= y[L:]
            else:
                M = -L
                g[:N - M] += (1 - y[M:])
                g[M:] -= y[:N - M]
        g /= len(active)
        h_new = proj(y - lr * g, tgt)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            lr *= 0.96
            beta = min(0.95, 0.8 + it * 0.00005)
    return bh, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h_init = np.array(initial_h_values, dtype=float)
    N0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = c5_of(best_h, N0)

    # Phase 1: SA single-flip on initial
    d1 = min(deadline - 400, t0 + 250)
    h1, c1 = sa_binary(h_init, N0, d1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: SA block moves
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 200)
        h2, c2 = sa_block(best_h, N0, d2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Nesterov continuous
    if time.time() < deadline - 200:
        d3 = min(deadline - 150, time.time() + 150)
        hn, c3 = nesterov(best_h, N0, d3)
        if c3 < best_c5:
            best_c5, best_h = c3, hn.copy()

    # Phase 4: SA single-flip again (2 flips)
    if time.time() < deadline - 120:
        d4 = min(deadline - 80, time.time() + 150)
        h4, c4 = sa_binary(to_bin(best_h, N0 // 2), N0, d4, n_flips=2)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Try other resolutions
    for Nt in [420, 460, 500]:
        if time.time() > deadline - 60:
            break
        if Nt % 2 != 0:
            Nt += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hb = to_bin(hi, Nt // 2)
        d5 = min(deadline - 20, time.time() + 60)
        hr, cr = sa_binary(hb, Nt, d5)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 6: Final Nesterov polish
    if time.time() < deadline - 5:
        hF = proj(best_h.astype(float) + 0.02 * np.random.randn(len(best_h)), len(best_h) / 2.0)
        hF, cF = nesterov(hF, len(best_h), deadline - 1)
        if cF < best_c5:
            best_c5, best_h = cF, hF.copy()

    # Final SA polish on best
    if time.time() < deadline - 2:
        hP = to_bin(best_h, len(best_h) // 2)
        hP, cP = sa_binary(hP, len(best_h), deadline - 1)
        if cP < best_c5:
            best_c5, best_h = cP, hP.copy()

    final_c5 = c5_of(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
