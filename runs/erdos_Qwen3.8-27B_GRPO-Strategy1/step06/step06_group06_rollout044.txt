```python
def smooth_max_grad(corr, h, beta):
    """Compute smooth max and its gradient w.r.t. h"""
    N = len(h)
    # Numerically stable log-sum-exp
    max_c = corr.max()
    exp_bc = np.exp(beta * (corr - max_c))
    Z = exp_bc.sum()
    w = exp_bc / Z  # normalized weights, sum to 1
    
    # Smooth max value
    smooth_val = max_c + np.log(Z) / beta
    
    # Gradient: grad[p] = sum_c w_c * [(1-h[p+c-(N-1)]) * I - h[p-c+(N-1)] * I]
    center = N - 1
    c_idx = np.arange(2*N-1) - center  # c_idx[m] = m - (N-1), ranges from -(N-1) to N-1
    
    grad = np.zeros(N)
    for p in range(N):
        # First term: (1-h[p+c]) for valid c where 0<=p+c<N
        valid1 = (p >= 0) & (p + c_idx >= 0) & (p + c_idx < N)
        terms1 = np.zeros(2*N-1)
        terms1[valid1] = (1.0 - h[p + c_idx[valid1]])
        # Second term: -h[p-c] for valid c where 0<=p-c<N
        valid2 = (p >= 0) & (p - c_idx >= 0) & (p - c_idx < N)
        terms2 = np.zeros(2*N-1)
        terms2[valid2] = h[p - c_idx[valid2]]
        
        grad[p] = np.dot(w, terms1 - terms2)
    
    return smooth_val, grad
```

```python
def fft_corr(h):

    while L < 2*N-1:

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin==1)[0]
        perm = np.random.permutation(len(idx))
        h_bin = h_bin.copy()
        h_bin[idx[perm[:n1-target]]] = 0
    elif n1 < target:
        idx = np.where(h_bin==0)[0]
        perm = np.random.permutation(len(idx))
        h_bin = h_bin.copy()
        h_bin[idx[perm[:target-n1]]] = 1
    return h_bin

                    d[vqs] += sgn*(1.0-h[qsc[vqs]])
                    d[vqm] -= sgn*h[qmc[vqm]]
                nc = corr + d
                npk = float(nc.max())
            elif len(ones) >= 1 and len(zeros) >= 1:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                ps = p+s_arr; pm = p-s_arr
                qs = q+s_arr; qm = q-s_arr
                vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
                vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
                pc=np.clip(ps,0,N-1); pmc=np.clip(pm,0,N-1)
                qsc=np.clip(qs,0,N-1); qmc=np.clip(qm,0,N-1)
                d = np.zeros(2*N-1)
                d[vps] -= (1.0-h[pc[vps]])
                d[vpm] += h[pmc[vpm]]
                d[vqs] += (1.0-h[qsc[vqs]])
                d[vqm] -= h[qmc[vqm]]
                nc = corr + d
                npk = float(nc.max())
            else:
                break
            delta = npk - cur_pk
            if delta < 0 or np.random.random() < np.exp(-delta/max(T,1e-9)):
                if 'p1' in dir() and 'q1' in dir() and move_type < 0.7 and locals().get('p1',-1)>=0:
                    h[p1]=0; h[p2]=0; h[q1]=1; h[q2]=1
                elif 'p' in dir() and locals().get('p',-1)>=0:
                    h[p]=0; h[q]=1
                corr=nc; cur_pk=npk
                ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
                if npk < best_pk:
                    best_pk=npk; best_h=h.copy(); no_imp=0
                else:
                    no_imp+=1
            else:
                no_imp+=1
            T*=cool
            if no_imp > max_no_imp:
                no_imp=0
                if r < n_restarts-1:
                    break
                else:
                    h=best_h.copy(); corr=fft_corr(h); cur_pk=best_pk
                    ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
    return best_h, best_pk

def smooth_max_grad(corr, h, beta):
    N = len(h)
    max_c = float(corr.max())
    exp_bc = np.exp(beta*(corr - max_c))
    Z = float(exp_bc.sum())
    w = exp_bc / Z
    smooth_val = max_c + np.log(Z)/beta
    center = N - 1
    c_idx = np.arange(2*N-1) - center
    grad = np.zeros(N)
    for p in range(N):
        v1 = (p + c_idx >= 0) & (p + c_idx < N)
        v2 = (p - c_idx >= 0) & (p - c_idx < N)
        t1 = np.zeros(2*N-1); t1[v1] = 1.0 - h[p + c_idx[v1]]
        t2 = np.zeros(2*N-1); t2[v2] = h[p - c_idx[v2]]
        grad[p] = np.dot(w, t1 - t2)
    return smooth_val, grad

def continuous_smooth(h, N, deadline, beta_start=3.0, beta_end=80.0, lr0=0.003, top_refine=True):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_val = float(fft_corr(h).max())
    best_h = h.copy()
    n_steps = max(1, int((deadline - time.time()) / 0.05))
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max(1, n_steps*0.7))
        beta = beta_start + (beta_end - beta_start) * frac * frac
        corr = fft_corr(h)
        cur_max = float(corr.max())
        if cur_max < best_val:
            best_val = cur_max; best_h = h.copy()
        _, grad = smooth_max_grad(corr, h, beta)
        t += 1
        lr = lr0 * (1.0 - 0.5*frac)
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m/(1-b1**t); vh = v/(1-b2**t)
        h = proj_sum(h - lr*mh/np.sqrt(vh + eps), N/2.0)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    corr0 = fft_corr(h0)
    best_c5 = float(corr0.max()) * (2.0/N0)
    best_h = h0.copy()

    # Phase 1: Binary annealing from initial (200s)
    d1 = min(deadline - 400, t0 + 200)
    hb = (h0 > 0.5).astype(float)
    hb = fix_count(hb, N0)
    hB, pkB = binary_anneal(hb, N0, d1, T0=0.06, cool=0.9993, max_no_imp=500, n_restarts=2)
    c5B = pkB * (2.0/N0)
    if c5B < best_c5:
        best_c5 = c5B; best_h = hB.copy()

    # Phase 2: Smooth-max continuous optimization (250s)
    d2 = min(deadline - 250, time.time() + 250)
    hc = best_h.astype(float) + np.random.randn(len(best_h))*0.03
    hc = proj_sum(hc, len(best_h)/2.0)
    hC, pkC = continuous_smooth(hc, len(best_h), d2, beta_start=5.0, beta_end=100.0, lr0=0.004)
    c5C = pkC * (2.0/len(best_h))
    if c5C < best_c5:
        best_c5 = c5C; best_h = hC.copy()

    # Phase 3: Round and re-anneal (150s)
    if time.time() < deadline - 200:
        d3 = min(deadline - 150, time.time() + 150)
        hb2 = (best_h > 0.5).astype(float)
        hb2 = fix_count(hb2, len(best_h))
        hB2, pkB2 = binary_anneal(hb2, len(best_h), d3, T0=0.04, cool=0.9995, max_no_imp=800, n_restarts=2)
        c5B2 = pkB2 * (2.0/len(best_h))
        if c5B2 < best_c5:
            best_c5 = c5B2; best_h = hB2.copy()

    # Phase 4: Multi-N exploration (150s)
    for Nt in [320, 400, 460, 540, 680]:
        if time.time() > deadline - 80:
            break
        if Nt % 2 != 0:
            Nt += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hbt = fix_count((hi > 0.5).astype(float), Nt)
        sub_deadline = min(deadline - 20, time.time() + 40)
        hrt, pkr = binary_anneal(hbt, Nt, sub_deadline, T0=0.04, cool=0.9995, max_no_imp=400, n_restarts=1)
        c5r = pkr * (2.0/Nt)
        if c5r < best_c5:
            best_c5 = c5r; best_h = hrt.copy()

    # Phase 5: Final continuous polish (remaining)
    if time.time() < deadline - 5:
        d5 = deadline - 2
        hc2 = best_h.astype(float)
        hC2, pkC2 = continuous_smooth(hc2, len(best_h), d5, beta_start=20.0, beta_end=150.0, lr0=0.002)
        c5C2 = pkC2 * (2.0/len(best_h))
        if c5C2 < best_c5:
            best_c5 = c5C2; best_h = hC2.copy()

    # Final: round to binary if it doesn't hurt
    hb_final = (best_h > 0.5).astype(float)
    hb_final = fix_count(hb_final, len(best_h))
    c5_bin = float(fft_corr(hb_final).max()) * (2.0/len(best_h))
    if c5_bin < best_c5:
        best_c5 = c5_bin; best_h = hb_final

    final_c5 = float(np.max(fft_corr(best_h))) * (2.0/len(best_h))
    return best_h, final_c5, len(best_h)
```
