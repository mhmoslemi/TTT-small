        ck = corr_y[top_k]

        lse = ck_max + np.log(np.sum(np.exp(ck - ck_max)))  # smooth max

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h[::-1], L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def binary_anneal(h_in, N, deadline, max_iters=80000, T0=0.04, Tend=0.0005):

    s = 2.0 * h - 1.0
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    cur_c = best_c

    deltas = np.zeros((N, 2 * N - 1))
    for i in range(N):
        deltas[i, i:i + N] = s

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    it = 0
    no_improve = 0
    stall_restart = 0

    while time.time() < deadline and it < max_iters:
        frac = min(1.0, it / max_iters)
        T = max(Tend, T0 * np.exp(-3.0 * frac))

        # Targeted: 40% near peak, 60% random
        if np.random.random() < 0.4 and len(ones) > 10:

            w = np.exp(-d / (N * 0.15))
            p = int(np.random.choice(ones, p=w / w.sum()))
        else:
            p = int(np.random.choice(ones))
        q = int(np.random.choice(zeros))

def lse_nesterov(h_in, N, deadline, k_top=60):
    h = np.clip(h_in.astype(float).copy(), 0.0, 1.0)
    h = project_sum(h, N / 2.0)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = 0.012
    beta = 0.82
    it = 0

    while time.time() < deadline:
        y = h + m
        corr_y = fft_corr(y)
        top_k = np.argpartition(corr_y, -k_top)[-k_top:]
        ck = corr_y[top_k]
        ck_max = ck.max()
        exp_ck = np.exp(ck - ck_max)
        w = exp_ck / exp_ck.sum()

        grad = np.zeros(N)
        for idx, l in enumerate(top_k):
            wi = w[idx]
            if wi < 0.005:
                continue
            Li = int(l - (N - 1))
            if Li == 0:
                g = 1.0 - 2.0 * y
            elif Li > 0:
                g = np.zeros(N)
                g[Li:] += (1.0 - y[:N - Li])
                g[:N - Li] -= y[Li:]
            else:
                M = -Li
                g = np.zeros(N)
                g[:N - M] += (1.0 - y[M:])
                g[M:] -= y[:N - M]
            grad += wi * g

        h_new = project_sum(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1

        if it % 100 == 0:
            c_h = float(fft_corr(h).max())
            if c_h < best_c:
                best_c = c_h
                best_h = h.copy()
            lr *= 0.97
            beta = min(0.95, 0.80 + it * 0.0002)

    c_final = float(fft_corr(h).max())
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c * (2.0 / N)
