    pos = np.arange(N)[:, None]
    k_vals = np.arange(2 * N - 1)[None, :] - (N - 1)
    i_mk = pos - k_vals
    i_pk = pos + k_vals
    A = np.where((i_mk >= 0) & (i_mk < N), 1.0 - h[np.clip(i_mk, 0, N - 1)], 0.0)
    B = np.where((i_pk >= 0) & (i_pk < N), h[np.clip(i_pk, 0, N - 1)], 0.0)
    return A - B

def binary_il(h, N, deadline, max_iters=10000, sample_sz=80, kick_every=200, kick_sz=3):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    best_c = np.inf
    best_h = h.copy()
    it = 0
    stuck = 0
    while time.time() < deadline and it < max_iters:
        corr = fft_corr(h)
        P = float(np.max(corr))
        if P < best_c - 1e-14:
            best_c = P
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        C = compute_C_mat(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        ns = min(len(ones), sample_sz)
        nz = min(len(zeros), sample_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        Co = C[os_]  # (ns, 2N-1)
        Cz = C[zs]  # (nz, 2N-1)
        # new[i,j,k] = corr[k] + Co[i,k] - Cz[j,k]
        # newP[i,j] = max_k new[i,j,k]
        newP = (Co[None, :, :] - Cz[:, None, :]).max(axis=2) + P  # wrong
        # Actually: max_k (corr[k] + Co[i,k] - Cz[j,k])
        # = max_k (corr[k] + Co[i,k] - Cz[j,k])
        # We need the full 3D for correctness
        # But we can do: for each i, compute fi = corr + Co[i], then newP[i,j] = max_k(fi[k] - Cz[j,k])
        fi = corr[None, :] + Co  # (ns, 2N-1)
        newP = (fi[:, None, :] - Cz[None, :, :]).max(axis=2)  # (ns, nz)
        mi = np.unravel_index(np.argmin(newP), newP.shape)
        mv = float(newP[mi])
        if mv < P - 1e-12:
            h[os_[mi[1]]] = 0
            h[zs[mi[0]]] = 1
        else:
            if stuck >= kick_every:
                for _ in range(kick_sz):
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p] = 0
                    h[q] = 1
                stuck = 0
            else:
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p] = 0
                h[q] = 1
        it += 1
    return best_h, best_c

def nesterov(h, N, deadline, k=40, lr0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m_vec = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline and it < 15000:
        y = h + m_vec
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_01(y - lr * grad, target)
        m_vec = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            lr *= 0.985
            beta = min(0.97, 0.87 + it * 0.00008)
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    
    # Phase 2: Nesterov continuous
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 120)
        hc = best_h + np.random.randn(N0) * 0.04
        hc = project_01(hc, N0 / 2.0)

        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
    
    # Phase 3: Deeper binary search with larger samples
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 200)
        h3 = (best_h > 0.5).astype(float)
        h_s3, _ = binary_il(h3, N0, d3, max_iters=8000, sample_sz=120, kick_every=250, kick_sz=3)
        c5_3 = compute_c5(h_s3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()
    
    # Phase 4: Multi-resolution
    for N_try in [260, 320, 400, 440, 500]:
        if time.time() > deadline - 80:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_best, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh_t = N_try // 2
        n1_t = int(h_bt.sum())
        if n1_t > nh_t:
            idx = np.where(h_bt == 1)[0]
            h_bt[np.random.choice(idx, n1_t - nh_t, replace=False)] = 0
        elif n1_t < nh_t:
            idx = np.where(h_bt == 0)[0]
            h_bt[np.random.choice(idx, nh_t - n1_t, replace=False)] = 1
        sub_dl = min(deadline - 30, time.time() + 70)
        h_rt, _ = binary_il(h_bt, N_try, sub_dl, max_iters=5000, sample_sz=80, kick_every=150, kick_sz=3)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
    
    # Phase 5: Final Nesterov + binary
    if time.time() < deadline - 5:
        hF = best_h + np.random.randn(len(best_h)) * 0.02
        hF = project_01(hF, len(best_h) / 2.0)
        hF, c5F = nesterov(hF, len(best_h), deadline - 3, k=70, lr0=0.01)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    
    if time.time() < deadline - 0.5:
        hB = (best_h > 0.5).astype(float)
        hB, _ = binary_il(hB, len(best_h), deadline - 0.2, max_iters=300, sample_sz=100)
        c5B = compute_c5(hB, len(best_h))
        if c5B < best_c5:
            best_c5, best_h = c5B, hB
        final_c5 = compute_c5(best_h, len(best_h))
    else:
        final_c5 = compute_c5(best_h, len(best_h))
    
    # Prefer binary
    hB = (best_h > 0.5).astype(float)
    c5B = compute_c5(hB, len(best_h))
    if c5B < final_c5:
        best_h = hB
        final_c5 = c5B
    
    return best_h, final_c5, len(best_h)

Actually, let me think about this differently. The computation:
newP[i,j] = max_k (fi[i,k] - Cz[j,k])

For fixed i, this is: for each j, compute max_k (fi[i,:]-Cz[j,:]). This is an outer max operation.

We can vectorize: newP[:,j] = max_k (fi[:,k] - Cz[0,j,k]) ... no that doesn't work.

Actually, the most memory-efficient way:
```python
newP = np.empty((ns, nz))
for i in range(ns):
    newP[i,:] = (fi[i][None,:] - Cz).max(axis=1)
```
This uses only (nz, 2N-1) = 80*759 ≈ 60K elements at a time. But it's a Python loop over 80-120 iterations, each doing a numpy op on (nz, 759). Should be fast.

Actually even better: we can do it as a single numpy op:
```python
diff = fi[:, None, :] - Cz[None, :, :]  # (ns, nz, 2N-1)
newP = diff.max(axis=2)
```

This allocates the full 3D array. For ns=nz=80, 2N-1=759: 80*80*759 = 4.86M floats = 39MB. Modern CPUs can handle this. The .max(axis=2) is also fast.

**Alternative: SLSQP or other NLP solver on the continuous relaxation**

The continuous problem (h ∈ [0,1]) is non-convex, but we could try using scipy.optimize.minimize with SLSQP. The issue is that the objective is max of many functions, which is a max-min problem. We could reformulate as:
minimize t
subject to: corr_k(h) * dx ≤ t for all k, sum(h)*dx = 1, 0 ≤ h ≤ 1

But this has 2N-1 constraints (759 for N=380), which is a lot for SLSQP.

**Alternative: Use the active set method**

At each iteration, identify the top-K active constraints and only enforce those. This is what the Nesterov approach does implicitly.
