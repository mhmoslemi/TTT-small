def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)

    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

        if P < best_c - 1e-12:

            no_improve = 0
        else:
            no_improve += 1

def nesterov_pressure(h, N, deadline, k=40, lr0=0.02):

    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    center = N - 1
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = y.copy()
        
        # Top-k peaks with Gaussian weighting
        top_idx = np.argpartition(corr, -k)[-k:]
        peak_vals = corr[top_idx]
        peak_max = peak_vals.max()
        # Weight: exponential decay from max
        weights = np.exp(-(peak_max - peak_vals) / (0.1 * peak_max + 1e-10))
        weights /= weights.sum()
        
        grad = np.zeros(N)
        for ai in top_idx:
            L = int(ai - center)
            w = weights[np.argmax(top_idx == ai)]
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += w * (1.0 - y[:N - L])
                grad[:N - L] -= w * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += w * (1.0 - y[M:])
                    grad[M:] -= w * y[:N - M]
        
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - y)
        h = h_new
        it += 1
        
        if it % 50 == 0:
            lr *= 0.99
            beta = min(0.95, 0.82 + it * 0.0002)
    
    return best_h, best_pk * (2.0 / N)

    h_s, _ = binary_refine(h_bin, N0, d1, alpha=0.12, sa_temp=0.04)
    c5_s = compute_c5(h_s)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    best_h, best_c5 = try_rotations(best_h, N0, n_rot=4)
    # Phase 2: Nesterov
    if time.time() < deadline - 320:
        d2 = min(deadline - 300, time.time() + 120)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.03
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k=60, lr0=0.015)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
    # Phase 3: Binary from Nesterov
    if time.time() < deadline - 170:
        d3 = min(deadline - 150, time.time() + 160)
        h3 = (best_h > 0.5).astype(float)
        h3 = fix_sum_binary(h3, N0)
        h_s3, _ = binary_refine(h3, N0, d3, alpha=0.10, sa_temp=0.02)
        c5_3 = compute_c5(h_s3)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()
    # Phase 4: Perturb + reoptimize (iterated local search)
    if time.time() < deadline - 130:
        d4 = min(deadline - 110, time.time() + 100)
        h_p, c5_p = perturb_reopt(best_h, N0, d4, n_kicks=3, kick_size=6, n_sweeps=50)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p
    # Phase 5: 2x resolution
    N_up = N0 * 2
    if time.time() < deadline - 80:
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_up, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        h_bt = fix_sum_binary(h_bt, N_up)
        d5 = min(deadline - 40, time.time() + 80)
        h_rt, _ = binary_refine(h_bt, N_up, d5, alpha=0.12, sa_temp=0.03, max_stuck=250)
        c5_rt = compute_c5(h_rt)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
    # Phase 6: Final Nesterov polish
    Nf = len(best_h)
    if time.time() < deadline - 12:
        hF = best_h.astype(float) + np.random.randn(Nf) * 0.012
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = nesterov_opt(hF, Nf, deadline - 6, k=70, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
    # Phase 7: Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = (best_h > 0.5).astype(float)
        hF2 = fix_sum_binary(hF2, len(hF2))
        hF2, _ = binary_refine(hF2, len(hF2), deadline - 0.5, alpha=0.1, sa_temp=0.0, max_stuck=100)
        c5F2 = compute_c5(hF2)
        if c5F2 < best_c5:
            best_c5 = c5F2
            best_h = hF2
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
