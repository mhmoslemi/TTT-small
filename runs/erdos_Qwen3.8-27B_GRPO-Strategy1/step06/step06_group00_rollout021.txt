def swap_delta(h, N, a, b):
    """Compute the change in correlation when swapping h[a]:1->0 and h[b]:0->1"""
    # For each shift k (represented as index 0..2N-2, where k=0 is at N-1)
    # Shift k (signed) = idx - (N-1)
    delta = np.zeros(2*N-1)
    center = N - 1
    # Flip a: 1->0
    # For shift s >= 0: contributes at positions where i=a, need a <= N-1-s, i.e., s <= N-1-a
    #   change to corr[s] = -(1-h[a+s]) = h[a+s]-1
    for s in range(N - a):
        if s <= N-1:
            delta[center + s] += h[a+s] - 1
    # For shift s < 0 (m=-s): contributes where i=a, need a >= m, i.e., m <= a
    #   change to corr[-m] = -(1-h[a-m]) = h[a-m]-1
    for m in range(1, a + 1):
        if m < N:
            delta[center - m] += h[a-m] - 1
    # Flip b: 0->1
    for s in range(N - b):
        if s <= N-1:
            delta[center + s] += 1 - h[b+s]
    for m in range(1, b + 1):
        if m < N:
            delta[center - m] += 1 - h[b-m]
    return delta

def swap_delta_fast(h, N, a, b):
    """Compute delta in correlation for swapping h[a]:1->0, h[b]:0->1. Returns array of length 2N-1."""
    delta = np.zeros(2*N-1)
    c = N - 1  # center index
    
    # Term 1: h[a-k]-1 for k in [a-N+1, a], i.e., idx in [a, a+N-1]
    k_lo = max(a - N + 1, -(N-1))
    k_hi = min(a, N-1)
    if k_lo <= k_hi:
        idx_lo = c + k_lo
        idx_hi = c + k_hi
        # h[a-k] for k=k_lo..k_hi → h[a-k_lo]..h[a-k_hi] → indices a-k_hi..a-k_lo
        # a-k ranges from a-k_lo to a-k_hi (decreasing as k increases)
        # h indices: a-k_lo, a-k_lo-1, ..., a-k_hi
        h_slice = h[a-k_hi:a-k_lo+1][::-1]  # reversed
        delta[idx_lo:idx_hi+1] += h_slice - 1
    
    # Term 2: h[a+k] for k in [-a, N-1-a], i.e., idx in [c-a, c+N-1-a]
    k_lo = max(-a, -(N-1))
    k_hi = min(N-1-a, N-1)
    if k_lo <= k_hi:
        idx_lo = c + k_lo
        idx_hi = c + k_hi
        # h[a+k] for k=k_lo..k_hi → h[a+k_lo]..h[a+k_hi]
        h_slice = h[a+k_lo:a+k_hi+1]
        delta[idx_lo:idx_hi+1] += h_slice
    
    # Term 3: 1-h[b-k] for k in [b-N+1, b], i.e., idx in [b, b+N-1]

        h_new = project_01(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.993
            beta = min(0.97, 0.88 + it * 0.0002)
    return best_h, best_pk * (2.0/N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    h_bin = (h0 > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 != N0 // 2:
        if n1 > N0 // 2:
            idx = np.where(h_bin == 1)[0]
            h_bin[np.random.choice(idx, n1 - N0//2, replace=False)] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[np.random.choice(idx, N0//2 - n1, replace=False)] = 1
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()

    # Phase 1: Fast local search with single swaps
    d1 = min(deadline - 400, t0 + 300)
    h1, c1 = fast_local_search(h_bin, N0, d1, max_iters=30000, ns=25, ne=40, bp=0.10, t0=0.005)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Second local search pass with more block moves
    if time.time() < deadline - 300:
        d2 = min(deadline - 250, time.time() + 250)
        h2, c2 = fast_local_search(best_h, N0, d2, max_iters=25000, ns=30, ne=50, bp=0.25, t0=0.003)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Try different resolutions
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [390, 400, 420, 440]:
        if time.time() > deadline - 80:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_best, best_h)
        h_bt = (h_int > 0.5).astype(float)
        nh = N_try // 2
        n1r = int(h_bt.sum())
        if n1r > nh:
            idx = np.where(h_bt == 1)[0]
            h_bt[np.random.choice(idx, n1r - nh, replace=False)] = 0
        elif n1r < nh:
            idx = np.where(h_bt == 0)[0]
            h_bt[np.random.choice(idx, nh - n1r, replace=False)] = 1
        sub_dl = min(deadline - 30, time.time() + 60)
        h_rt, c_rt = fast_local_search(h_bt, N_try, sub_dl, max_iters=8000, ns=20, ne=30, bp=0.15, t0=0.004)
        c5_rt = compute_c5(h_rt)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 4: Nesterov continuous polish
    if time.time() < deadline - 40:
        d4 = min(deadline - 10, time.time() + 80)
        Nb = len(best_h)
        hc = best_h + np.random.randn(Nb) * 0.04
        hc = project_01(hc, Nb / 2.0)
        hn, c5n = nesterov_polish(hc, Nb, d4, k=50, lr0=0.012)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn

    # Phase 5: Final binary local search
    if time.time() < deadline - 2:
        hB = (best_h > 0.5).astype(float)
        Nb = len(hB)
        hB, cB = fast_local_search(hB, Nb, deadline - 0.5, max_iters=5000, ns=30, ne=50, bp=0.15, t0=0.002)
        c5B = compute_c5(hB)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB

    # Final validation
    Nb = len(best_h)
    final_h = (best_h > 0.5).astype(float)
    nf = int(final_h.sum())
    if nf != Nb // 2:
        if nf > Nb // 2:
            idx = np.where(final_h == 1)[0]
            final_h[np.random.choice(idx, nf - Nb//2, replace=False)] = 0
        else:
            idx = np.where(final_h == 0)[0]
            final_h[np.random.choice(idx, Nb//2 - nf, replace=False)] = 1
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5
    # Ensure valid
    final_h = np.clip(final_h, 0.0, 1.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
