    L = 2 * N - 1
    H = np.fft.fft(h, L)

    return np.real(np.fft.ifft(H * G))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 1e-12:

    lo, hi = -1.0, 1.0

def subgrad_continuous(h0, N, deadline, k=40, lr0=0.02, temp0=0.003):
    """Nesterov-accelerated subgradient with softmax weighting on top lags"""
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    y = h.copy()
    m = np.zeros(N)  # Adam first moment
    v = np.zeros(N)  # Adam second moment
    t_adam = 0
    lr = lr0
    beta = 0.9
    temp = temp0
    stall = 0
    
    while time.time() < deadline:
        # Nesterov step
        h_new = y + beta * (y - h)
        h = h_new
        
        corr = compute_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Softmax weights on top-k lags
        top_idx = np.argpartition(corr, -k)[-k:]
        top_vals = corr[top_idx]
        w = np.exp((top_vals - top_vals.max()) / max(temp, 1e-10))
        w /= w.sum()
        
        # Gradient at top lags
        grad = np.zeros(N)
        center = N - 1
        for wi, ai in zip(w, top_idx):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                seg = N - L
                grad[L:] += wi * (1.0 - h[:seg])
                grad[:seg] -= wi * h[L:]
            else:
                M = -L
                if M < N:
                    seg = N - M
                    grad[:seg] += wi * (1.0 - h[M:])
                    grad[M:] -= wi * h[:seg]
        
        y = h  # update y for Nesterov
        t_adam += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.99 * v + 0.01 * grad * grad
        step = lr * m / (np.sqrt(v) + 1e-8)
        h = project_sum(h - step, target)

            lr = max(0.002, lr * 0.995)

            m = np.zeros(N)
            v = np.zeros(N)

            stall = 0
    
    return best_h, best_pk * (2.0 / N)

import numpy as np

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def to_binary(h, N):
    """Snap to binary with exactly N//2 ones"""
    h = (h > 0.5).astype(float)
    half = N // 2
    n1 = int(h.sum())
    if n1 > half:
        # Remove excess ones (from smallest h values originally, approximated by random)
        idx = np.where(h == 1)[0]
        idx = idx[np.random.permutation(len(idx))][:n1 - half]
        h[idx] = 0
    elif n1 < half:
        idx = np.where(h == 0)[0]
        idx = idx[np.random.permutation(len(idx))][:half - n1]
        h[idx] = 1
    return h
