```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def clip_project(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}."""
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h
    # Binary search for the threshold
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient(h, N, hot_indices):
    """Compute subgradient of max_k f_k at given h."""
    grad = np.zeros(N)
    center = N - 1
    for idx in hot_indices:
        L = int(idx - center)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N - L] += (1.0 - h[L:])
            grad[L:] -= h[:N - L]
        else:
            M = -L
            if M < N:
                grad[M:] += (1.0 - h[:N - M])
                grad[:N - M] -= h[M:]
    return grad

def continuous_opt(h, N, deadline, hot_sz=30, lr_init=0.01, momentum=0.85):
    """Nesterov-like accelerated projected gradient for C5 minimization."""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    m = np.zeros(N)
    lr = lr_init
    it = 0
    stall = 0
    
    while time.time() < deadline:
        # Nesterov lookahead
        y = h + m
        
        corr = fft_corr(y)
        cur_c = float(np.max(corr))
        if cur_c * (2.0/N) < best_c - 1e-10:
            best_c = cur_c * (2.0/N)
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Hot set: top hot_sz shifts
        hot = np.argpartition(corr, -hot_sz)[-hot_sz:]
        grad = subgradient(y, N, hot)
        grad /= hot_sz
        
        # Adaptive LR
        if stall > 50:
            lr *= 0.95
            stall = 0
        
        # Step
        h_new = clip_project(y - lr * grad, target)
        m = momentum * (h_new - h)
        h = h_new
        it += 1
        
        # Periodic perturbation
        if it % 200 == 0:
            h += np.random.randn(N) * lr * 0.3
            h = clip_project(h, target)
        
        # Decay LR
        if it % 500 == 0:
            lr *= 0.97
            momentum = min(0.95, momentum + 0.005)
    
    return best_h, best_c
```

Actually, I want to think about coordinate descent as an alternative. For each pair (i,j), we can compute the effect of swapping h[i] and h[j] (or adjusting them) on the objective. But with N=380, that's O(N^2) pairs, and evaluating each requires O(N) work for the correlation. Total: O(N^3) per pass, which is 380^3 ≈ 55M operations. Too slow for many passes.

Better: use the FFT to compute the full correlation once (O(N log N)), then for each pair, compute the delta in the objective in O(1) using precomputed arrays.

Actually, for continuous h, changing h[i] by δ changes the correlation at each lag k by:
- For lag k: the terms involving h[i] are h[i]*(1-h[i+k]) and (if i-k>0) h[i-k]*(1-h[i])
- Delta = δ*(1-h[i+k]) - δ*h[i-k] (approximately, for small δ)

This is essentially the subgradient at position i. So the subgradient approach IS the right way to do coordinate descent here.

Let me focus on making the subgradient method as effective as possible:

Key idea: **Proximal gradient with Nesterov acceleration**

The objective is F(h) = max_k f_k(h) where each f_k is smooth (bilinear).
F is convex if each f_k is convex, but they're not (they're non-convex bilinear). However, locally around a point, the max is achieved by a subset of the f_k's, and on that subset, F is locally smooth.

So the approach is:
1. At each iteration, identify the active set S = {k: f_k ≈ max}
2. Compute g = (1/|S|) sum_{k in S} ∇f_k
3. Take a step: h ← P(h - α*g) where P is the projection
4. Use Nesterov momentum

The challenge is choosing α. Too large → oscillation. Too small → slow.

Let me use an adaptive scheme:
- Start with α = 0.01
- If F decreased: α ← α * 1.05
- If F increased: α ← α * 0.8
- Cap α at [0.001, 0.1]

Also, I'll use a "damped" Nesterov: m ← β*m + (1-β)*(h_new - h_old) with β starting at 0.85 and increasing to 0.95.

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def clip_project(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_subgrad(h, N, hot_idx):
    grad = np.zeros(N)
    center = N - 1
    for idx in hot_idx:
        L = int(idx - center)
        if L == 0:
            grad += 1.0 - 2.0 * h
        elif 0 < L < N:
            grad[:N-L] += 1.0 - h[L:]
            grad[L:] -= h[:N-L]
        elif -N < L < 0:
            M = -L
            grad[M:] += 1.0 - h[:N-M]
            grad[:N-M] -= h[M:]
    return grad

def nesterov_opt(h_init, N, deadline, hot_sz=40, lr0=0.015, mom0=0.85):
    h = h_init.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    m = np.zeros(N)
    lr = lr0
    momentum = mom0
    it = 0
    last_imp = 0
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        c5_val = pk * (2.0 / N)
        if c5_val < best_c - 1e-12:
            best_c = c5_val
            best_h = h.copy()
            last_imp = it
        
        # Weighted hot set
        top_k = np.argpartition(corr, -hot_sz)[-hot_sz:]
        # Weight by how close to max
        weights = np.maximum(corr[top_k] - (pk - 0.5), 0)
        if weights.sum() < 1e-10:
            weights = np.ones(hot_sz)
        
        grad = np.zeros(N)
        center = N - 1
        for j, idx in enumerate(top_k):
            L = int(idx - center)
            w = weights[j]
            if L == 0:
                grad += w * (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[:N-L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:N-L]
            elif -N < L < 0:
                M = -L
                grad[M:] += w * (1.0 - h[:N-M])
                grad[:N-M] -= w * h[M:]
        grad /= weights.sum()
        
        h_new = clip_project(y - lr * grad, target)
        m = momentum * (h_new - h)
        h = h_new
        it += 1
        
        # Adaptive LR
        if it - last_imp > 100 and it > 200:
            lr *= 0.9
            momentum = min(0.95, momentum + 0.01)
            # Perturb
            h += np.random.randn(N) * lr * 0.5
            h = clip_project(h, target)
        
        if it % 300 == 0:
            lr *= 0.97
            momentum = min(0.95, 0.85 + it * 0.00005)
    
    return best_h, best_c
```

```python
def weighted_subgrad(h, corr, N, hot_sz, width):
```
