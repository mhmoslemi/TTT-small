corr(k) = sum_j h[j]*(1-h[j+k])

The contribution of h[i] to corr(k):
- As first factor (j=i): h[i]*(1-h[i+k])
- As second factor (j+k=i, j=i-k): h[i-k]*(1-h[i])

If h[i] goes 1→0:
- First term: (1-h[i+k]) → 0, change = -(1-h[i+k])
- Second term: h[i-k]*(1-1)=0 → h[i-k]*(1-0)=h[i-k], change = +h[i-k]
- Net change in corr(k): h[i-k] - (1-h[i+k]) = h[i-k] + h[i+k] - 1

For this to decrease corr(k) (change < 0): h[i-k] + h[i+k] < 1, i.e., both are 0. ✓

If h[i] goes 0→1:
- First term: 0 → (1-h[i+k]), change = +(1-h[i+k])
- Second term: h[i-k]*(1-0)=h[i-k] → h[i-k]*(1-1)=0, change = -h[i-k]
- Net change: (1-h[i+k]) - h[i-k] = 1 - h[i+k] - h[i-k]

For this to decrease corr(k): 1 - h[i+k] - h[i-k] < 0, i.e., h[i+k] + h[i-k] > 1, i.e., at least one is 1. For both to be 1: change = 1-1-1 = -1. ✓

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_opt(h0, N, deadline, k_top=80, lr0=0.02, temp0=0.003, beta0=0.9):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    h_prev = h.copy()
    lr = lr0
    beta = beta0
    temp = temp0
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        top = np.argpartition(corr, -k_top)[-k_top:]
        w = np.exp((corr[top] - corr[top].max()) / max(temp, 1e-10))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for wi, ai in zip(w, top):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif 0 < L < N:
                seg = N - L
                grad[L:] += wi * (1.0 - y[:seg])
                grad[:seg] -= wi * y[L:]
            elif -N < L < 0:
                M = -L
                seg = N - M
                grad[:seg] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:seg]
        h = project_sum(y - lr * grad, target)
        it += 1
        if it % 150 == 0:
            lr = max(0.002, lr * 0.994)
            temp = max(0.0003, temp * 0.992)
            beta = min(0.98, beta0 + it * 0.0001)
        if stall > 250:
            h = best_h + np.random.randn(N) * 0.02
            h = project_sum(h, target)
            h_prev = h.copy()
            lr = max(lr, lr0 * 0.7)
            stall = 0
    return best_h, best_pk * (2.0 / N)

def binary_informed(h0, N, deadline, n_trials=15):
    h = (h0 > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            idx = idx[np.random.permutation(len(idx))][:n1 - nh]
            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh - n1]
            h[idx] = 1
    c = fft_corr(h)
    best_int = int(np.max(c))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        top_k = np.argpartition(c, -8)[-8:]
        top_k_sorted = top_k[np.argsort(c[top_k])[::-1]]
        best_sw_int = int(c.max())
        best_sw_h = None
        best_sw_c = None
        for _ in range(n_trials):
            p = None
            q = None
            for k_pos in top_k_sorted[:4]:
                k_off = int(np.round(k_pos - (N - 1)))
                if k_off == 0:
                    continue
                mp = (h == 1) & (np.roll(h, -k_off) == 0) & (np.roll(h, k_off) == 0)
                cp = np.where(mp)[0]
                if len(cp) > 0:
                    mq = (h == 0) & (np.roll(h, -k_off) == 1) & (np.roll(h, k_off) == 1)
                    cq = np.where(mq)[0]
                    if len(cq) > 0:
                        p = cp[np.random.randint(len(cp))]
                        q = cq[np.random.randint(len(cq))]
                        break
            if p is None:
                # secondary strategy: look for "good-enough" moves
                for k_pos in top_k_sorted[:6]:
                    k_off = int(np.round(k_pos - (N - 1)))
                    if k_off == 0:
                        continue
                    mp = (h == 1) & ((np.roll(h, -k_off) + np.roll(h, k_off)) == 0)
                    cp = np.where(mp)[0]
                    mq = (h == 0) & ((np.roll(h, -k_off) + np.roll(h, k_off)) == 2)
                    cq = np.where(mq)[0]
                    if len(cp) > 0 and len(cq) > 0:
                        p = cp[np.random.randint(len(cp))]
                        q = cq[np.random.randint(len(cq))]
                        break
            if p is None:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_t = fft_corr(h)
            ct = int(c_t.max())
            if ct < best_sw_int:
                best_sw_int = ct
                best_sw_h = h.copy()
                best_sw_c = c_t.copy()
            h[p] = 1
            h[q] = 0
        if best_sw_h is not None:
            h = best_sw_h
            c = best_sw_c
            stall = 0
            if best_sw_int < best_int:
                best_int = best_sw_int
                best_h = h.copy()
        else:
            stall += 1
            if stall > 35:
                nf = np.random.randint(2, max(3, N // 50))
                nf = min(nf, len(ones), len(zeros))
                i1 = np.random.choice(len(ones), nf, replace=False)
                i2 = np.random.choice(len(zeros), nf, replace=False)
                h[ones[i1]] = 0
                h[zeros[i2]] = 1
                c = fft_corr(h)
                stall = 0
                if stall > 100:
                    # larger perturbation
                    nf2 = np.random.randint(4, max(5, N // 25))
                    nf2 = min(nf2, len(ones), len(zeros))
                    i1 = np.random.choice(len(ones), nf2, replace=False)
                    i2 = np.random.choice(len(zeros), nf2, replace=False)
                    h[ones[i1]] = 0
                    h[zeros[i2]] = 1
                    c = fft_corr(h)
    return best_h, best_int * (2.0 / N)
