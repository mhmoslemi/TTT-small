def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv

    vp = (pp >= 0) & (pp < N)

f_k(h) = sum_i h[i+k] * (1-h[i])

This is NOT positive semidefinite in general. So f_k is not convex!

Hmm, this means the problem min_h max_k f_k(h) is not a convex optimization problem. That explains why simple gradient methods struggle.

Wait, but f_k(h) = sum_i h[i+k] - h[i+k]h[i]. The term -h[i+k]h[i] is concave (for h ≥ 0). So f_k is neither convex nor concave in general.

def simulated_annealing(h, N, deadline, T0=0.5, T1=0.01, cooling=0.999):
    h = h.astype(float).copy()
    c = fft_corr(h)
    current_max = float(c.max())
    best_c = current_max
    best_h = h.copy()
    T = T0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    rng = np.random.RandomState(42)
    
    while time.time() < deadline and T > T1:
        D = compute_D(h, N)
        # Choose random p from ones, q from zeros
        p = rng.choice(ones)
        q = rng.choice(zeros)
        
        # Compute delta
        delta = D[p, :] - D[q, :]  # change in corr
        new_c = c + delta
        new_max = float(new_c.max())
        
        # Accept
        if new_max < current_max or rng.random() < np.exp(-(new_max - current_max) / T):
            # Accept
            c = new_c
            h[p], h[q] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            current_max = new_max
            if current_max < best_c:
                best_c = current_max
                best_h = h.copy()
        
        T *= cooling
    
    return best_h, best_c * (2.0 / N)
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum_binary(h, N):
    nh = N // 2
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def greedy_binary(h, N, deadline, max_sweeps=200, sample_frac=1.0):
    """Greedy binary local search."""
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum_binary(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(20, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            cand = (c - D[p, :])[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P - 1e-13:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = fft_corr(h)  # recompute
    return best_h, best_c * (2.0 / N), h

def iterated_local_search(h, N, deadline, n_restarts=10, kick_size=5):
    """Iterated local search with random kicks."""
    best_h = h.copy()
    best_c = compute_c5(h, N)
    current_h = h.copy()
    
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        # Local search
        deadline_ls = min(deadline, time.time() + 30)
        h_ls, c_ls, h_end = greedy_binary(current_h, N, deadline_ls, max_sweeps=100)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls.copy()
        
        # Kick: flip kick_size random pairs
        current_h = h_end.copy()
        if r < n_restarts - 1:  # don't kick the last one
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            n_kick = kick_size + r
            for _ in range(min(n_kick, len(ones), len(zeros))):
                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                current_h[p], current_h[q] = 0.0, 1.0
                ones = np.where(current_h == 1)[0]
                zeros = np.where(current_h == 0)[0]
    
    return best_h, best_c
