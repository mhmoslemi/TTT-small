    state['m'] = m
    state['v'] = v

        PM_c = np.clip(PM, 0, N - 1)
        PP_c = np.clip(PP, 0, N - 1)
        # D[p, m] = -(1-h[PM]) * VM - h[PP] * VP
        D = -((1.0 - h[PM_c]) * VM.astype(float) + h[PP_c] * VP.astype(float))
        
        # For each q in zeros, the change when h[q]: 0->1 is -D (with h updated)
        # Approximation: use original h, so Dq = -D evaluated at q
        # But more precisely: Dq[q, m] = (1-h[q-lags[m]])*[valid] + h[q+lags[m]]*[valid]
        # = -(D with 1->0 formula) ... actually it's the negative of the 1->0 change
        # because 0->1 adds what 1->0 removes (approximately, ignoring cross terms)
        
        # For the combined swap (p: 1->0, q: 0->1):
        # corr_new ≈ c + D[p,:] - Dq[q,:] where Dq is the 0->1 change at q
        # 0->1 change at q: (1-h[q-lags])*[valid] + h[q+lags]*[valid]
        # = -D[q, :] (using the 1->0 formula but at position q where h[q]=0)
        # Wait, D is defined for 1->0. For 0->1, it's the negative.
        # But D was computed with the assumption h[p]=1. For q where h[q]=0, the formula is different.
        # Let me just compute it directly.
        
        # Actually, the cleanest approach:
        # Change in corr[m] when h[p] changes from a to b (a->b, where b-a = ±1):
        # = (b-a) * [(1-h[p-lag]) * [0<=p-lag<N] + h[p+lag] * [0<=p+lag<N]]
        # Wait no. Let me re-derive.
        # corr[m] = sum_i h[i] * (1-h[i-lag_m]) where lag_m = m-(N-1)
        # = sum_i h[i] - h[i]*h[i-lag_m]
        # d(corr[m])/d(h[p]) = 1 - h[p-lag_m] (from h[p] term, i=p)
        #                       - h[p]  (from -h[p]*h[p-lag_m]... no)
        # Let me be more careful:
        # corr[m] = sum_i h[i]*(1-h[i-lag_m])
        # Expand: = sum_i h[i] - sum_i h[i]*h[i-lag_m]
        # d/dh[p]: = 1 - [h[p-lag_m] + h[p+lag_m]*... wait
        # From first term sum_i h[i]: d/dh[p] = 1
        # From second term -sum_i h[i]*h[i-lag_m]:
        #   = -h[p-lag_m] (from i=p term: -h[p]*h[p-lag_m], derivative w.r.t h[p] is -h[p-lag_m])
        #   - h[p+lag_m] (from i=p+lag_m term: -h[p+lag_m]*h[p], derivative w.r.t h[p] is -h[p+lag_m])
        # Wait, that's only if p+lag_m is in range and i-lag_m = p, i.e., i = p+lag_m
        # So: d(corr[m])/d(h[p]) = 1 - h[p-lag_m]*[0<=p-lag_m<N] - h[p+lag_m]*[0<=p+lag_m<N]
        
        # For 1->0 (change = -1): Δcorr[m] = -(1 - h[p-lag_m]*[valid] - h[p+lag_m]*[valid])
        # For 0->1 (change = +1): Δcorr[m] = +(1 - h[p-lag_m]*[valid] - h[p+lag_m]*[valid])
        
        # So D_1to0[p, m] = -(1 - h[p-lag_m]*[valid] - h[p+lag_m]*[valid])
        # And D_0to1[q, m] = +(1 - h[q-lag_m]*[valid] - h[q+lag_m]*[valid])
        
        # For the combined swap p:1->0, q:0->1:
        # Δcorr[m] ≈ D_1to0[p,m] + D_0to1[q,m]
        # = -1 + h[p-lag]*[v1] + h[p+lag]*[v2] + 1 - h[q-lag]*[v2'] - h[q+lag]*[v3']
        # = h[p-lag]*[v1] + h[p+lag]*[v2] - h[q-lag]*[v2'] - h[q+lag]*[v3']
        # (the -1 and +1 cancel!)
        
        # This is nice. But for the objective, we want max(corr_new) = max(c + Δcorr)
        # = max(c[m] + h[p-lag]*[v1] + h[p+lag]*[v2] - h[q-lag]*[v2'] - h[q+lag]*[v3'])
        
        # For a fixed p, this is: c[m] + D1[p,m] - D0[q,m] where D1 and D0 are as defined
        # We want to find q that minimizes max_m (c[m] + D1[p,m] - D0[q,m])
        
        # Precompute for sampled p's: base[p, m] = c[m] + D1[p,m] (shape S x N2)
        # Then for each q: score = max_m (base[p,m] - D0[q,m])
        # D0[q,m] = 1 - h[q-lag_m]*[valid] - h[q+lag_m]*[valid]
        
        # But this is still O(S * Q * N2) which is too slow.
        
        # Alternative: for each p, just find the q that minimizes the max.
        # Use the approach from the original code but with the correct formula.
        
        best_score = P + 1.0
        best_pq = None
        for p in p_list:
            # D1 for this p: change for 1->0
            pm = p - lags  # (N2,)
            pp = p + lags
            vm = (pm >= 0) & (pm < N)
            vp = (pp >= 0) & (pp < N)
            D1 = np.zeros(N2)
            D1[vm] = h[pm[vm]]
            D1[vp] += h[pp[vp]]
            # D1[m] = h[p-lag_m]*[v] + h[p+lag_m]*[v2]
            # Δcorr[m] for p:1->0 = -1 + D1[m]
            base = c - 1.0 + D1  # = c + (-1 + D1) = c + Δ for p:1->0
            
            # Now find best q (0->1):
            # Δcorr_total[m] = (-1 + D1_p[m]) + (1 + D0_q[m]) = D1_p[m] + D0_q[m]
            # where D0_q[m] = -h[q-lag]*[v] - h[q+lag]*[v2]... wait
            # For 0->1: Δ = +1 - h[q-lag]*[v] - h[q+lag]*[v2]
            # Total: (-1 + D1_p) + (1 - h[q-l]v - h[q+l]v2) = D1_p[m] - h[q-lag_m]*[v] - h[q+lag_m]*[v2]
            
            # So cand[m] = base[m] - h[q-lag_m]*[v] - h[q+lag_m]*[v2]
            # = base[m] - (h[q-lag_m]*[v] + h[q+lag_m]*[v2])
            # Let me define: S[q, m] = h[q-lag_m]*[v] + h[q+lag_m]*[v2]
            # Then cand[m] = base[m] - S[q, m]
            # We want min over q of max over m of (base[m] - S[q, m])
            
            # Precompute S for all zeros: shape (Q, N2)
            Q_arr = zeros[:, None]
            QM = Q_arr - lags[None, :]
            QP = Q_arr + lags[None, :]
            QVM = (QM >= 0) & (QM < N)
            QVP = (QP >= 0) & (QP < N)
            QM_c = np.clip(QM, 0, N-1)
            QP_c = np.clip(QP, 0, N-1)
            S = h[QM_c] * QVM.astype(float) + h[QP_c] * QVP.astype(float)  # (Q, N2)
            
            # cand for all q: base[None,:] - S  (Q, N2)
            # max over m for each q:
            npk = (base[None, :] - S).max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_score:
                best_score = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        if no_improve > 400:
            break
        sweep += 1
    return h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lag_grad(correlation, h, N, k, beta=8.0):
    c = correlation.copy()
    top = np.argpartition(c, -(k))[-k:]
    w = np.exp(beta * (c[top] - c[top].max()))
    w /= w.sum()
    center = N - 1
    grad = np.zeros(N)
    for ii, ai in enumerate(top):
        L = int(ai - center)
        wi = w[ii]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad

def adam_continuous(h, N, deadline, k=40, lr0=0.04, beta=8.0):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m_ = np.zeros(N)
    v_ = np.zeros(N)
    t_step = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_c - 1e-14:
            best_c = pk
            best_h = h.copy()
        grad = lag_grad(corr, h, N, k, beta)
        t_step += 1
        beta_t = (1.0 - 0.95 ** t_step) ** 0.5
        m_ = 0.9 * m_ + 0.1 * grad
        v_ = 0.999 * v_ + 0.001 * grad ** 2
        h = project_sum(h - lr0 * m_ / (beta_t * (np.sqrt(v_) + 1e-8)), target)
        if t_step % 200 == 0:
            lr0 *= 0.995
    return best_h, best_c * (2.0 / N)

def binary_swap(h, N, deadline, max_sweeps=3000, sample_n=200, top_k_penalty=0.15):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    sweep = 0
    N2 = 2 * N - 1
    lags = np.arange(N - 1, -N, -1)
    while sweep < max_sweeps and time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if len(ones) > sample_n:
            perm = np.random.permutation(len(ones))[:sample_n]
            p_list = ones[perm]
        else:
            p_list = ones
        # Precompute S for all zeros: S[q, m] = h[q-lag_m]*[valid] + h[q+lag_m]*[valid]
        Q_arr = zeros[:, None]
        QM = Q_arr - lags[None, :]
        QP = Q_arr + lags[None, :]
        QVM = (QM >= 0) & (QM < N)
        QVP = (QP >= 0) & (QP < N)
        QMc = np.clip(QM, 0, N - 1)
        QPc = np.clip(QP, 0, N - 1)
        S = h[QMc] * QVM.astype(np.float64) + h[QPc] * QVP.astype(np.float64)
        # Top-k penalty vector: penalize correlations near the max
        c_pen = c.copy()
        thr = c.max() - top_k_penalty
        c_pen[c_pen > thr] = c.max() + (c_pen[c_pen > thr] - c.max()) * 2.0
        best_score = P + 1.0
        best_pq = None
        for p in p_list:
            pm = p - lags
            pp = p + lags
            vm = (pm >= 0) & (pm < N)
            vp = (pp >= 0) & (pp < N)
            D1 = np.zeros(N2)
            pmc = np.clip(pm, 0, N - 1)
            ppc = np.clip(pp, 0, N - 1)
            D1[vm] = h[pmc[vm]]
            D1[vp] += h[ppc[vp]]
            base = c_pen - 1.0 + D1
            npk = (base[None, :] - S).max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_score:
                best_score = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if no_improve > 60:
                break
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        if no_improve > 500:
            break
        sweep += 1
    return h, best_c

def perturbed_binary(h, N, n_flips=15):
    h = h.astype(float).copy()
    one_idx = np.where(h == 1)[0]
    zero_idx = np.where(h == 0)[0]
    to_flip_o = np.random.choice(one_idx, size=min(n_flips, len(one_idx)), replace=False)
    to_flip_z = np.random.choice(zero_idx, size=min(n_flips, len(zero_idx)), replace=False)
    h[to_flip_o] = 0.0
    h[to_flip_z] = 1.0
    return h

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rm = idx[np.random.choice(len(idx), size=n1 - nh, replace=False)]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        add = idx[np.random.choice(len(idx), size=nh - n1, replace=False)]
        h[add] = 1.0
    return h

    if time.time() < deadline - 2:
        hF2 = (best_h > 0.5).astype(float)
        hF2 = fix_sum_binary(hF2, len(hF2))
        hF2, _ = binary_swap(hF2, len(hF2), deadline - 1, max_sweeps=200, sample_n=150, top_k_penalty=0.10)
        c5F2 = compute_c5(hF2, len(hF2))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
