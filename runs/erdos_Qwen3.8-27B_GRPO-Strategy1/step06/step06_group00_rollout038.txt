        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        ns = min(len(ones), sample_size)
        nz = min(len(zeros), sample_size)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        
        # For each critical lag L, precompute delta for each candidate
        # delta_corr(L) = h[oi+L]+h[oi-L]-h[zi+L]-h[zi-L]
        center = N - 1
        Ls = top_lags - center  # actual lags
        
        best_move = None
        best_val = c
        
        for oi in os_:
            for zi in zs:
                # Check if this swap reduces the max
                new_max = 0.0
                ok = True
                for L in Ls:
                    L = int(L)
                    # h[oi+L] + h[oi-L] - h[zi+L] - h[zi-L] with bounds
                    d = 0.0
                    if 0 <= oi + L < N: d += h[oi + L]
                    if 0 <= oi - L < N: d += h[oi - L]
                    if 0 <= zi + L < N: d -= h[zi + L]
                    if 0 <= zi - L < N: d -= h[zi - L]
                    new_corri = corr[center + int(L)] + d
                    if new_corri > new_max:
                        new_max = new_corri
                # Also need to check it doesn't increase other lags significantly
                # For simplicity, just check the critical ones
                if new_max * (2.0/N) < best_val - 1e-9:
                    best_val = new_max * (2.0/N)
                    best_move = (oi, zi)
        
        if best_move is not None:
            oi, zi = best_move
            h[oi] = 0
            h[zi] = 1
        else:
            # Random perturbation (simulated annealing style)
            if no_improve > 50:
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                h[oi] = 0
                h[zi] = 1
        it += 1
    return best_h, best_c

import numpy as np
import time

def compute_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def grad_corr_top(h, corr, k=40):
    N = len(h)
    center = N - 1
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(N)
    for m in top_idx:
        M = abs(int(m) - center)
        if M == 0:
            grad += (1.0 - 2.0 * h)
        elif M < N:
            hi_idx = np.arange(M, N)
            lo_idx = np.arange(0, N - M)
            grad[hi_idx] += (1.0 - h[hi_idx - M])
            grad[lo_idx] -= h[lo_idx + M]
    return grad / k

def cont_opt(h, deadline, k=40, lr0=0.025, mom=0.85):
    N = len(h)
    tgt = N / 2.0
    h = h.astype(float).copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 80:
            lr0 *= 0.85
            stale = 0
        g = grad_corr_top(h, corr, k)
        h_new = project(h + mom * v - lr0 * g, tgt)
        v = 0.8 * v + 0.3 * (h_new - h)
        h = h_new
    return best_h, best_c

def fix_binary(h):
    N = len(h)
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1
    return h

def binary_il(h, deadline, max_iters=2000, n_crit=20, s_sz=35):
    N = len(h)
    h = fix_binary(h.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    it = 0
    no_imp = 0
    center = N - 1
    while time.time() < deadline and it < max_iters:
        corr = compute_corr(h)
        c = float(np.max(corr)) * (2.0 / N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 300:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        top_lags = (np.argsort(corr)[-n_crit:] - center).astype(int)
        ns = min(len(ones), s_sz)
        nz = min(len(zeros), s_sz)
        os_ = np.random.choice(ones, ns, replace=False)
        zs = np.random.choice(zeros, nz, replace=False)
        best_move = None
        best_new = c
        cur_peak = float(np.max(corr))
        for oi in os_:
            d_oi = np.zeros(2 * N - 1)
            for L in top_lags:
                idx = center + L
                d = 0.0
                if 0 <= oi + L < N:
                    d += h[oi + L]
                if 0 <= oi - L < N:
                    d += h[oi - L]
                d_oi[idx] = d
            for zi in zs:
                new_peak = 0.0
                for L in top_lags:
                    idx = center + L
                    d = 0.0
                    if 0 <= zi + L < N:
                        d -= h[zi + L]
                    if 0 <= zi - L < N:
                        d -= h[zi - L]
                    val = corr[idx] + d_oi[idx] + d
                    if val > new_peak:
                        new_peak = val
                if new_peak < cur_peak - 1e-8:
                    nv = new_peak * (2.0 / N)
                    if nv < best_new:
                        best_new = nv
                        best_move = (oi, zi)
        if best_move is not None:
            h[best_move[0]] = 0
            h[best_move[1]] = 1
        elif no_imp > 100 and np.random.random() < 0.4:
            oi = int(np.random.choice(ones))
            zi = int(np.random.choice(zeros))
            h[oi] = 0
            h[zi] = 1
        it += 1
    return best_h, best_c
