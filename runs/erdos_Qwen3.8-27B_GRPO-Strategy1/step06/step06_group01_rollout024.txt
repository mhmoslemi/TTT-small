```python
    return np.fft.irfft(H * G, L)

def project_binary(h, n_ones):

    h = h.copy()
    current_ones = int((h >= 0.5).sum())
    if current_ones > n_ones:

        idx = np.where(h >= 0.5)[0]
        order = np.argsort(h[idx])[:current_ones - n_ones]
        h[idx[order]] = 0.0

        order = np.argsort(-h[idx])[:n_ones - current_ones]
        h[idx[order]] = 1.0
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0

    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
```

```python
    T_min = T * 0.01
    no_improve = 0
    recompute_interval = 30
    
    for it in range(max_iters):
        if time.time() > deadline:

        if len(ones) < 2 or len(zeros) < 2:

                fd[idx] = fft_corr(h) - corr
                h[pos] = 1 - h[pos]
            fd_p = fd[:n_p]
            fd_q = fd[n_p:]

        vals = corr[None, None, :] + fd_p[:, None, :] + fd_q[None, :, :]

            zeros = np.where(h == 0)[0]

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

                qq = int(np.random.choice(zeros))
                h[pp], h[qq] = 0.0, 1.0

            ones = np.where(h == 1)[0]
```

Wait, there's an issue: when I recompute fd, I need to be at the current corr state. And after a swap, the fd for other positions changes (because the bilinear interaction). So the fd values become stale.

The recomputation every 30 iterations handles this. Between recomputations, the fd values are approximate (they don't account for the bilinear interaction with the most recent swaps). This is an approximation that works well in practice for local search.

Actually, let me think about this more carefully. The correlation is:
corr[k] = sum_j h[j+k] * (1-h[j])

This is bilinear in h. If I flip h[p] (from 1 to 0), the change is:
- For terms where j+k = p: h[p]*(1-h[j]) → 0, change = -(1-h[j]) = -h_old[j]... wait h[p] was 1, now 0.
  Change in corr[k] for j+k=p: was 1*(1-h[j]), now 0*(1-h[j]). Change: -(1-h[j]).
- For terms where j = p: h[j+k]*(1-h[p]) → h[j+k]*(1-0)=h[j+k]. Was h[j+k]*0=0. Change: h[j+k].
  But wait, (1-h[p]) was 0 (since h[p]=1), now is 1. So the term goes from 0 to h[j+k].

So the flip_delta for position p:
flip_delta[p][k] = -sum over valid j where j+k=p of (1-h[j]) + sum over valid j+k where j=p of h[j+k]
= -sum_{j: j+k=p, valid} (1-h[j]) + sum_{j: j=p, j+k valid} h[j+k]

The first term: j=p-k, so only valid if 0≤p-k<N and 0≤j<N (clearly) and j+k valid in the correlation.
The second term: j=p, so j+k=p+k, valid if 0≤p+k<2N-1 (for the full correlation) but also p+k < N (since h is only defined for [0,N)).

This is getting complicated. The FFT approach to recompute flip_delta for a subset of positions is simpler and fast enough.

For n_p + n_q = 60 positions, computing 60 FFTs of size 759 takes about 60 * 759 * 9 ≈ 400K flops. Very fast.

Let me refine my approach. I'll precompute flip_delta for the candidate positions at each recomputation step.

Now, for the continuous optimization, let me use a subgradient method:

The objective is f(h) = max_k corr[k] where corr = convolve(h, flip(1-h)).

The subgradient with respect to h[i]:
∂corr[k]/∂h[i] = (1-h[i-k]) for k such that i = j+k (i.e., j = i-k in range)
                  + (-1) * h[i-k'] for k' such that i-k' = i... 

Hmm, let me think differently. For the bilinear form:
corr[k] = sum_j h[j+k] * (1-h[j])

∂corr[k]/∂h[i] = (1-h[i-k]) [if i-k >= 0 and i-k < N, i.e., the "first" factor]
                + h[i+k] * (-1) [if i+k < N, i.e., the "second" factor through (1-h[j])]

Wait: ∂/∂h[i] of h[j+k]*(1-h[j]):
- If i = j+k (i.e., j = i-k): contributes (1-h[j]) = (1-h[i-k])
- If i = j: contributes h[j+k]*(-1) = -h[i+k]

So: ∂corr[k]/∂h[i] = (1-h[i-k]) * [0≤i-k<N] - h[i+k] * [i+k<N]

And the subgradient of f(h) = max_k corr[k] is the average of ∂corr[k]/∂h over k in argmax.

This gives a direction to move. We then project onto the constraint set (0≤h≤1, sum=1/dx = N/2).

```python
def subgradient_opt(h0, N, deadline, max_iters=2000):
    h = h0.astype(float).copy()
    target = N / 2.0
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    
    lr = 0.01
    momentum = np.zeros(N)
    beta = 0.5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Find top shifts
        top_k = 20
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        
        # Compute subgradient
        grad = np.zeros(N)
        for k in top_idx:
            k_int = int(k) - (N - 1)  # actual shift
            if k_int == 0:
                grad += (1.0 - 2.0 * h)
            elif k_int > 0:
                # ∂corr[k]/∂h[i] for i in [k, N-1]: (1-h[i-k])
                grad[k:] += (1.0 - h[:N-k])
                # ∂corr[k]/∂h[i] for i in [0, N-k-1]: -h[i+k]
                grad[:N-k] -= h[k:]
            else:
                m = -k_int
                # ∂corr[k]/∂h[i] for i in [0, N-m-1]: -h[i+m]... 
                # Actually for shift -m: corr[k] = sum_j h[j-m]*(1-h[j])
                # ∂/∂h[i]: (1-h[i+m]) for j=i+m (if i+m < N), -h[i-m] for j=i (if i-m >= 0)
                grad[:N-m] += (1.0 - h[m:])
                grad[m:] -= h[:N-m]
        
        grad /= len(top_idx)
        
        # Momentum update
        momentum = beta * momentum + lr * grad
        h_new = h - momentum
        
        # Project onto {0<=h<=1, sum(h)=N/2}
        h_new = np.clip(h_new, 0.0, 1.0)
        # Adjust sum
        excess = h_new.sum() - target
        if abs(excess) > 1e-10:
            # Simple: subtract excess/N (works if not too many boundary violations)
            h_new -= excess / N
            h_new = np.clip(h_new, 0.0, 1.0)
            # If still off, use iterative projection
            for _ in range(20):
                h_new = np.clip(h_new, 0.0, 1.0)
                excess = h_new.sum() - target
                if abs(excess) < 1e-8:
                    break
                h_new -= excess / N
        
        h = h_new
        corr = fft_corr(h)
        c = float(corr.max())
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Decay
        if it % 200 == 0:
            lr *= 0.95
            beta = min(0.9, 0.5 + it * 0.001)
    
    return best_h, best_c * (2.0 / N)
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_ones(h, n_ones):
    h = h.copy()
    cur = int(h.sum())
    if cur > n_ones:
        idx = np.where(h > 0.5)[0]
        rem = np.random.choice(idx, cur - n_ones, replace=False)
        h[rem] = 0.0
    elif cur < n_ones:
        idx = np.where(h < 0.5)[0]
        add = np.random.choice(idx, n_ones - cur, replace=False)
        h[add] = 1.0
    return (h > 0.5).astype(float)
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:L]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_ones(h, n_ones):
    h = h.astype(float).copy()
    cur = int(np.round(h.sum()))
    if cur > n_ones:
        idx = np.where(h > 0.5)[0]
        rem = np.random.choice(idx, cur - n_ones, replace=False)
        h[rem] = 0.0
    elif cur < n_ones:
        idx = np.where(h < 0.5)[0]
        add = np.random.choice(idx, n_ones - cur, replace=False)
        h[add] = 1.0
    return (h > 0.5).astype(float)

def binary_sa(h0, N, deadline, n_samp=30, max_it=50000, pert_size=5):
    h = fix_ones(h0, N // 2)
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    sc = np.sort(corr)[::-1]
    T = max(float(sc[0] - sc[min(5, len(sc)-1)]), 0.3)
    T_min = 0.015
    no_imp = 0
    L = 2 * N - 1
    fd = np.zeros((0, L))

    for it in range(max_it):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        n_p = min(n_samp, len(ones))
        n_q = min(n_samp, len(zeros))
        ps = np.random.choice(ones, n_p, replace=False)
        qs = np.random.choice(zeros, n_q, replace=False)
        all_c = np.concatenate([ps, qs])
        if len(fd) != len(all_c):
            fd = np.zeros((len(all_c), L))
        for ii in range(len(all_c)):
            pp = int(all_c[ii])
            h[pp] = 1.0 - h[pp]
            fd[ii] = fft_corr(h) - corr
            h[pp] = 1.0 - h[pp]

        vals = corr[None, None, :] + fd[:n_p][:, None, :] + fd[n_p:][None, :, :]
        nm = vals.max(axis=2)
        mi, mj = np.unravel_index(int(nm.argmin()), nm.shape)
        cand_c = float(nm[mi, mj])

        if cand_c <= best_c + 1e-12 or np.random.random() < np.exp(-(cand_c - best_c) / max(T, 1e-9)):
            h[int(ps[mi]), int(qs[mj])] = 0.0, 1.0
            corr += fd[mi] + fd[n_p + mj]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            ac = float(corr.max())
            if ac < best_c - 1e-12:
                best_c = ac
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        T = max(T * 0.99985, T_min)
        if no_imp > 180:
            ns = np.random.randint(3, pert_size + 3)
            for _ in range(ns):
                pp = int(np.random.choice(ones))
                qq = int(np.random.choice(zeros))
                h[pp], h[qq] = h[qq], h[pp]
            corr = fft_corr(h)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_imp = 0
            T = max(T, 0.4)

    return best_h, best_c * (2.0 / N)
```
