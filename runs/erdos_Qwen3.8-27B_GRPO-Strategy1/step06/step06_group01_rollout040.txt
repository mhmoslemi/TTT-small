```python
def sa_optimize(h_init, N, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = make_binary(h_init, N)
    corr = full_corr_fft(h)
    best_c = float(corr.max())
    best_h = h.copy()
    
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    
    T = 0.02  # initial temperature
    T_min = 1e-6
    cool_rate = 0.999
    
    for it in range(50000):
        if time.time() > deadline:
            break
        
        # Choose move: single swap or double swap
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        if rng.random() < 0.3:
            # Double swap
            p1, p2 = rng.choice(sorted(ones), 2)
            q1, q2 = rng.choice(sorted(zeros), 2)
            h[p1] = h[p2] = 0
            h[q1] = h[q2] = 1
            old_corr = corr
            corr = full_corr_fft(h)
            delta = float(corr.max()) - float(old_corr.max())
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                ones.discard(p1); ones.discard(p2)
                ones.add(q1); ones.add(q2)
                zeros.discard(q1); zeros.discard(q2)
                zeros.add(p1); zeros.add(p2)
            else:
                h[p1] = h[p2] = 1
                h[q1] = h[q2] = 0
                corr = old_corr
        else:
            # Single swap
            p = rng.choice(sorted(ones))
            q = rng.choice(sorted(zeros))
            h[p] = 0; h[q] = 1
            old_max = float(corr.max())
            old_corr = corr
            corr = full_corr_fft(h)
            new_max = float(corr.max())
            delta = new_max - old_max
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                ones.discard(p); ones.add(q)
                zeros.discard(q); zeros.add(p)
            else:
                h[p] = 1; h[q] = 0
                corr = old_corr
        
        c = float(corr.max())
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        
        if it % 500 == 0:
            T *= cool_rate
    
    return best_h, best_c * (2.0/N)
```

```python
import numpy as np
import time

def full_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))

def c5_of(h):
    return float(full_corr(h).max()) * 2.0 / len(h)

def snap_bin(h, rng=None):
    N = len(h)
    tgt = N // 2
    h = np.clip(h, 0, 1)
    # Round to binary
    thresh = 0.5
    h = (h >= thresh).astype(float)
    s = int(h.sum())
    if s != tgt:
        if s > tgt:
            idx = np.where(h == 1)[0]
            if rng:
                idx = idx[rng.permutation(len(idx))]
            h[idx[:s-tgt]] = 0
        else:
            idx = np.where(h == 0)[0]
            if rng:
                idx = idx[rng.permutation(len(idx))]
            h[idx[:tgt-s]] = 1
    return h

    h = h.copy()

    best_m = float(corr.max())
```

import time

def full_corr(h):
    N = len(h)
    L = 2*N-1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0-h), L)
    return np.real(np.fft.irfft(H*G, L))

def c5_fun(h):
    return float(full_corr(h).max()) * 2.0 / len(h)

def snap_bin(h):
    N = len(h)
    tgt = N//2
    h = (h >= 0.5).astype(float)
    s = int(h.sum())
    if s != tgt:
        if s > tgt:
            idx = np.where(h==1)[0]
            np.random.shuffle(idx)
            h[idx[:s-tgt]] = 0
        else:
            idx = np.where(h==0)[0]
            np.random.shuffle(idx)
            h[idx[:tgt-s]] = 1

    corr = full_corr(h)
    best_m = float(corr.max())
    best_h = h.copy()

        if time.time() > deadline:
            break

        p0 = np.where(h==0)[0]
        if len(p1)<2 or len(p0)<2:

            b = rng.choice(p0, 2, replace=False)
            h[a] = 0; h[b] = 1
        else:
            a = np.array([rng.choice(p1)])
            b = np.array([rng.choice(p0)])
            h[a[0]] = 0; h[b[0]] = 1
        nc = full_corr(h)
        nv = float(nc.max())
        ov = float(corr.max())
        if nv <= ov or (T > 0.01 and rng.random() < np.exp(-(nv-ov)/T)):
            corr = nc
            no_improve = 0 if nv < best_m else no_improve + 1

            h[a] = 1; h[b] = 0
        
        v = float(corr.max())
        if v < best_m - 1e-12:
            best_m = v
            best_h = h.copy()
            no_improve = 0
        
        if it % 10 == 0:
            T *= 0.997
        if no_improve > 500:
            # Kick: random perturb
            for _ in range(5):
                x = rng.choice(p1); y = rng.choice(p0)
                h[x]=0; h[y]=1
            corr = full_corr(h)
            no_improve = 0
            T = max(T, 0.2)
    return best_h, best_m*2.0/N

def subgrad(h, N, deadline, lr0=0.05, ktop=20):
    h = h.copy().astype(float).clip(0,1)
    tgt = N/2.0
    corr = full_corr(h)
    best_pk = float(corr.max())
    bh = h.copy()
    lr = lr0
    for it in range(100000):
        if time.time() > deadline:
            break
        # Find top ktop lags
        top = np.argsort(corr)[-ktop:]
        grad = np.zeros(N)
        for k in top:
            d = int(k) - N + 1
            if d >= 0:
                grad[d:] += 1.0 - h[:N-d]
                grad[:N-d] -= h[d:]
            else:
                dd = -d
                grad[:N-dd] += 1.0 - h[dd:]
                grad[dd:] -= h[:N-dd]
        grad /= ktop
        h_new = h - lr * grad
        # Project to [0,1] with sum constraint
        h_new = np.clip(h_new, 0, 1)
        s = h_new.sum()
        if abs(s - tgt) > 1e-8:
            lo, hi = -2.0, 2.0
            for _ in range(50):
                mid = (lo+hi)*0.5
                if np.clip(h_new-mid, 0, 1).sum() > tgt:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new-(lo+hi)*0.5, 0, 1)
        h = h_new
        corr = full_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            bh = h.copy()
        if it % 200 == 0:
            lr *= 0.99
    return bh, best_pk*2.0/N

def interp_h(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)
