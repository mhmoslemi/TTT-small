                if pk > best_pk + 1e-10:
                    h[best_pq[0]], h[best_pq[1]] = 1.0, 0.0
                    c = fft_corr(h)
        else:
            no_improve += 1
        if no_improve > 80 and np.random.random() < 0.25:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
            if float(c.max()) > best_pk + 0.5:
                h[p], h[q] = 1.0, 0.0
                c = fft_corr(h)
        if no_improve > 350:
            break
    return h, best_pk * (2.0 / N)

def nesterov_opt(h, N, deadline, top_k=100, lr0=0.012):
    h = h.astype(np.float64).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        indices = np.argpartition(c, -top_k)[-top_k:]
        grad = np.zeros(N)
        center = N - 1
        for j in indices:
            L = int(j - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif 0 < L < N:
                n = N - L
                grad[:n] += (1.0 - y[L:])
                grad[L:] -= y[:n]
            elif -N < L < 0:
                M = -L
                n = N - M
                grad[M:] += (1.0 - y[:n])
                grad[:n] -= y[M:]
        grad /= top_k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.993
            beta = min(0.96, 0.88 + 0.08 * min(1.0, it / 1500))
    return best_h, best_pk * (2.0 / N)

def fix_binary(h, N):
    h = h.copy()
    target = N // 2
    n1 = int(np.sum(h >= 0.5))
    h[h < 0.5] = 0.0
    h[h >= 0.5] = 1.0
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1.0)[0]
        rm = np.random.choice(ones, n1 - target, replace=False)
        h[rm] = 0.0
    elif n1 < target:
        zeros = np.where(h == 0.0)[0]
        add = np.random.choice(zeros, target - n1, replace=False)
        h[add] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=np.float64)
    N = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive binary search (all ones checked)
    d1 = min(deadline - 350, t0 + 280)
    hb = fix_binary(h0, N)
    if compute_c5(hb) < best_c5:
        best_c5, best_h = compute_c5(hb), hb.copy()
    hb, c5b = binary_search(hb, N, d1, max_sweeps=3000, alpha_multipeak=0.05)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 2: Nesterov continuous optimization
    if time.time() < deadline - 220:
        d2 = min(deadline - 200, time.time() + 130)
        hc = best_h + np.random.randn(N) * 0.03
        hc = project_sum(hc, N / 2.0)
        hn, c5n = nesterov_opt(hc, N, d2, top_k=100, lr0=0.012)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary search from improved continuous solution
    if time.time() < deadline - 120:
        d3 = min(deadline - 100, time.time() + 140)
        hb3 = fix_binary(best_h, N)
        hb3, c53 = binary_search(hb3, N, d3, max_sweeps=2000, alpha_multipeak=0.03)
        if c53 < best_c5:
            best_c5, best_h = c53, hb3.copy()

    # Phase 4: Try slightly higher resolution
    if time.time() < deadline - 90:
        N2 = min(N + 20, 400)
        x_old = np.linspace(0, 2.0, N, endpoint=False)
        x_new = np.linspace(0, 2.0, N2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h2 = fix_binary(h_interp, N2)
        d4 = min(deadline - 50, time.time() + 80)
        h2, c52 = binary_search(h2, N2, d4, max_sweeps=800, alpha_multipeak=0.05)
        if c52 < best_c5:
            best_c5, best_h = c52, h2.copy()

    # Phase 5: Final Nesterov polish
    Nf = len(best_h)
    if time.time() < deadline - 8:
        hF = best_h + np.random.randn(Nf) * 0.015
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = nesterov_opt(hF, Nf, deadline - 5, top_k=120, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    # Phase 6: Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = fix_binary(best_h, len(best_h))
        hF2, c5F2 = binary_search(hF2, len(best_h), deadline - 0.5, max_sweeps=50, alpha_multipeak=0.0)
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
