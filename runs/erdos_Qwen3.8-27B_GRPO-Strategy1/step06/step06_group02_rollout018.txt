Now for the Nesterov optimization with proper subgradient:

The objective is f(h) = max_k c(k) where c(k) = sum_i h[i](1-h[i+k]).

The subgradient of f at h is:
- If k* is unique argmax: grad = ∂c(k*)/∂h
- If multiple k* tie: grad = average of ∂c(k*)/∂h over tied k*

∂c(k)/∂h[j] = (1-h[j+k])*[0<=j+k<N] - h[j-k]*[0<=j-k<N]

But note we also have the constraint sum(h) = N/2, so the effective gradient is the projection of the above onto the constraint hyperplane. Actually, for projected gradient methods, we just apply the gradient and then project.

For Nesterov:
y = h + m (momentum)
grad = subgradient of max at y
h_new = project_sum(y - lr*grad, N/2)
m = beta*(h_new - h)
h = h_new

def fft_corr(h):
    N = len(h)
    L = 2*N-1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0-h), L)
    return np.real(np.fft.ifft(H*G))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s-target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo+hi)*0.5
        if float(np.sum(np.clip(h-mid, 0, 1))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def compute_D(h, N):
    L = 2*N-1
    D = np.zeros((N, L))
    ks = np.arange(-N+1, N)
    for p in range(N):
        pk = p + ks
        pm = p - ks
        D[p] = np.where((pk>=0)&(pk<N), 1.0-h[np.clip(pk,0,N-1)], 0.0) - \
               np.where((pm>=0)&(pm<N), h[np.clip(pm,0,N-1)], 0.0)

        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        
        # Subgradient: average gradient over top-k peaks
        top_idx = np.argpartition(corr, -k)[-k:]

        weights = weights - weights.min() + 1e-10
        weights /= weights.sum()

        for ai in top_idx:

Actually, I just realized: for binary h with N/2 ones, the mean of the correlation is:
mean(k) of c(k) = (1/L) * sum_k c(k)

And sum_k c(k) = sum_k sum_i h[i](1-h[i+k]) = sum_i h[i] * (number of valid k for i) - sum_k sum_i h[i]h[i+k]

The first term is fixed (depends only on N and the positions of 1s). The second term is the total autocorrelation energy. So the mean of c(k) is:
= (fixed - total_autocorr) / L

Minimizing the mean is equivalent to maximizing the total autocorrelation, which is NOT what we want! We want to MINIMIZE the max, not the mean.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if float(np.sum(np.clip(h - mid, 0, 1))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def compute_D(h, N):
    L = 2 * N - 1
    D = np.zeros((N, L))
    ks = np.arange(-(N - 1), N)
    for p in range(N):
        pk = p + ks
        pm = p - ks
        D[p] = np.where((pk >= 0) & (pk < N), 1.0 - h[np.clip(pk, 0, N - 1)], 0.0) - \
               np.where((pm >= 0) & (pm < N), h[np.clip(pm, 0, N - 1)], 0.0)
    return D

def upd_D_swap(D, N, p, q):
    c0 = N - 1
    ks = np.arange(-(N - 1), N)
    rmp, rpp = p - ks, p + ks
    rmq, rqq = q - ks, q + ks
    m1, m2 = (rmp >= 0) & (rmp < N), (rpp >= 0) & (rpp < N)
    m3, m4 = (rmq >= 0) & (rmq < N), (rqq >= 0) & (rqq < N)
    D[rmp[m1], ks[m1] + c0] += 1.0
    D[rpp[m2], ks[m2] + c0] += 1.0
    D[rmq[m3], ks[m3] + c0] -= 1.0
    D[rqq[m4], ks[m4] + c0] -= 1.0

def binary_opt(h, N, deadline, max_sweeps=800, sample_n=250):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_pk = float(c.max())
    best_h = h.copy()
    D = compute_D(h, N)
    no_imp = 0
    temp = 0.006

    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        pk = float(c.max())
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        ns = min(sample_n, len(ones))
        p_list = ones[np.random.choice(len(ones), ns, replace=False)] if ns < len(ones) else ones
        Dz = D[zeros]
        best_sc = pk + 1.0
        best_pq = None
        best_c = None

        for p in p_list:
            base = c - D[p]
            cand = base[None, :] + Dz
            mx = cand.max(axis=1)
            mi = int(np.argmin(mx))
            sc = float(mx[mi])
            if sc < best_sc:
                best_sc = sc
                best_pq = (int(p), int(zeros[mi]))
                best_c = cand[mi].copy()

        acc = False
        if best_pq is not None:
            d = best_sc - pk
            if d <= 1e-14 or np.random.random() < np.exp(-d / max(temp, 1e-10)):
                acc = True

        if acc and best_pq is not None:
            p, q = best_pq
            h[p], h[q] = 0.0, 1.0
            c = best_c
            upd_D_swap(D, N, p, q)
        elif no_imp > 50 and np.random.random() < 0.12:
            for _ in range(4):
                pp = int(ones[np.random.randint(len(ones))])
                qq = int(zeros[np.random.randint(len(zeros))])
                if pp != qq:
                    h[pp], h[qq] = 1.0 - h[pp], 1.0 - h[qq]
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            c = fft_corr(h)
            D = compute_D(h, N)
            no_imp = 20

        temp = max(0.0008, temp * 0.9965)
        if no_imp > 450:
            break

    return h, best_pk, best_h

def nesterov_opt(h, N, deadline, k=50, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        top_idx = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top_idx:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N - L] += (1.0 - y[L:])
                grad[L:] -= y[:N - L]
            else:
                M = -L
                if M < N:
                    grad[M:] += (1.0 - y[:N - M])
                    grad[:N - M] -= y[M:]
        grad /= float(k)
        grad -= grad.mean()
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr = max(0.001, lr0 * 0.992 ** (it // 80))
            beta = min(0.96, 0.86 + 0.001 * it)
    best_h = project_sum(best_h, target)
    return best_h, best_pk * dx

def fix_binary(h, N):
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    nh = N // 2
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rm = idx[np.random.choice(len(idx), n1 - nh, replace=False)]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        add = idx[np.random.choice(len(idx), nh - n1, replace=False)]
        h[add] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
