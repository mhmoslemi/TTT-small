        if c2 < best_c5:
            best_h, best_c5 = h2r, c2

    # Phase 3: fine-tune with low temp
    if time.time() < deadline - 250:
        d3 = min(deadline - 200, time.time() + 200)
        h3, c3 = binary_opt(best_h, N0, d3, max_sweeps=4000, sample_n=300,
                            T0=0.0008, T_min=0.00005, cool=0.9995, multi_prob=0.15)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: upscale to 2x
    N4 = N0 * 2
    if time.time() < deadline - 120:
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N4, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h4 = (h_interp > 0.5).astype(float)
        h4 = fix_sum(h4, N4)
        d4 = min(deadline - 40, time.time() + 140)
        h4r, c4 = binary_opt(h4, N4, d4, max_sweeps=1500, sample_n=200,
                             T0=0.002, cool=0.9993, multi_prob=0.08)
        if c4 < best_c5:
            best_h, best_c5 = h4r, c4
        else:
            # Try also 3x
            N5 = N0 * 3
            if time.time() < deadline - 80:
                x5 = np.linspace(0, 2, N5, endpoint=False)
                h5i = np.interp(x5, x_old, best_h)
                h5 = (h5i > 0.5).astype(float)
                h5 = fix_sum(h5, N5)
                d5 = min(deadline - 20, time.time() + 100)
                h5r, c5 = binary_opt(h5, N5, d5, max_sweeps=1000, sample_n=180,
                                     T0=0.002, cool=0.9994, multi_prob=0.08)
                if c5 < best_c5:
                    best_h, best_c5 = h5r, c5

    # Phase 5: final polish
    Nf = len(best_h)
    if time.time() < deadline - 2:
        d6 = deadline - 1
        h6, c6 = binary_opt(best_h, Nf, d6, max_sweeps=2000, sample_n=250,
                            T0=0.0005, T_min=0.00003, cool=0.9996, multi_prob=0.10)
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
