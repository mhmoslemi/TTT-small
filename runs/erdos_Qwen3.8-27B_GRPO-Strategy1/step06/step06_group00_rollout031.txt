```python
    # For h[p] going 1->0:
    # Term 1: -(1-h[p+s]) for valid p+s
    lo1 = max(0, -p)  # s >= -p
    hi1 = min(N-1, N-1-p)  # s <= N-1-p
    if lo1 <= hi1:
        s_range = np.arange(lo1, hi1+1)
        delta[s_range + center] += -(1.0 - h[p + s_range])
    
    # Term 2: +h[p-s] for valid p-s
    lo2 = max(0, p-N+1)  # s <= p (p-s >= 0 → s <= p)
    hi2 = min(N-1, p)  # s >= 0 for p-s < N → s > p-N → s >= p-N+1
    # Actually: 0 <= p-s < N → p-N < s <= p → s in [p-N+1, p]
    lo2 = max(0, p - N + 1)
    hi2 = min(N-1, p)
    if lo2 <= hi2:
        s_range = np.arange(lo2, hi2+1)
        delta[s_range + center] += h[p - s_range]
    
    # For h[q] going 0->1:
    # Term 3: +(1-h[q+s]) for valid q+s
    lo3 = max(0, -q)
    hi3 = min(N-1, N-1-q)
    if lo3 <= hi3:
        s_range = np.arange(lo3, hi3+1)
        delta[s_range + center] += (1.0 - h[q + s_range])
    
    # Term 4: -h[q-s] for valid q-s
    lo4 = max(0, q - N + 1)
    hi4 = min(N-1, q)
    if lo4 <= hi4:
        s_range = np.arange(lo4, hi4+1)
        delta[s_range + center] -= h[q - s_range]
    
    return delta
```

```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_01(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def grad_c5(h, N, temp):
    corr = fft_corr(h)
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= w.sum()
    # Use top indices
    sorted_w = np.sort(w)[::-1]
    cum = np.cumsum(sorted_w)
    n_act = int(np.searchsorted(cum, 0.999)) + 1
    n_act = min(n_act, 2*N-1)
    top_idx = np.argsort(w)[::-1][:n_act]
    
    grad = np.zeros(N)
    center = N - 1
    for mi in top_idx:
        wi = w[mi]
        s = mi - center
        if s >= 0:
            grad[:N-s] += wi * (1.0 - h[s:])
            if s > 0:
                grad[s:] -= wi * h[:N-s]
        else:
            M = -s
            grad[M:] += wi * (1.0 - h[:N-M])
            grad[:N-M] -= wi * h[M:]
    return grad * (2.0 / N)

def adam_run(h_init, N, deadline):
    h = h_init.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    m_a = np.zeros(N)
    v_a = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    it = 0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    temp0, temp1 = 4.0, 0.05
    total_temps = 2500
    
    while time.time() < deadline:
        frac = min(1.0, it / float(total_temps))
        temp = temp0 * (temp1 / temp0) ** frac
        
        c = float(np.max(fft_corr(h))) * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        g = grad_c5(h, N, temp)
        g -= g.mean()
        
        it += 1
        m_a = b1 * m_a + (1 - b1) * g
        v_a = b2 * v_a + (1 - b2) * g * g
        m_hat = m_a / (1 - b1 ** it)
        v_hat = v_a / (1 - b2 ** it)
        
        lr = 0.025 / (1 + 0.005 * it)
        h = project_01(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
    
    return best_h, best_c

def swap_delta(h, N, p, q):
    center = N - 1
    L = 2 * N - 1
    delta = np.zeros(L)
    # h[p] 1->0:
    lo1 = max(0, -p)
    hi1 = min(N - 1, N - 1 - p)
    if lo1 <= hi1:
        sr = np.arange(lo1, hi1 + 1)
        delta[sr + center] -= (1.0 - h[p + sr])
    lo2 = max(0, p - N + 1)
    hi2 = min(N - 1, p)
    if lo2 <= hi2:
        sr = np.arange(lo2, hi2 + 1)
        delta[sr + center] += h[p - sr]
    # h[q] 0->1:
    lo3 = max(0, -q)
    hi3 = min(N - 1, N - 1 - q)
    if lo3 <= hi3:
        sr = np.arange(lo3, hi3 + 1)
        delta[sr + center] += (1.0 - h[q + sr])
    lo4 = max(0, q - N + 1)
    hi4 = min(N - 1, q)
    if lo4 <= hi4:
        sr = np.arange(lo4, hi4 + 1)
        delta[sr + center] -= h[q - sr]
    return delta

def binary_search(h_init, N, deadline, max_iters=100000):
    h = (h_init > 0.5).astype(float)
    target = N // 2
    n1 = int(h.sum())
    if n1 != target:
        if n1 > target:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - target, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, target - n1, replace=False)] = 1
    
    dx = 2.0 / N
    corr = fft_corr(h)
    cur_max = float(np.max(corr))
    best_c = cur_max * dx
    best_h = h.copy()
    temp = 0.03
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 1000 == 0:
            temp = max(0.001, temp * 0.85)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        
        improved = False
        n_tries = min(80, n_ones * n_zeros)
        for _ in range(n_tries):
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            d = swap_delta(h, N, p, q)
            new_corr = corr + d
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                h[p] = 0
                h[q] = 1
                corr = new_corr
                cur_max = new_max
                improved = True
                if cur_max * dx < best_c:
                    best_c = cur_max * dx
                    best_h = h.copy()
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 50:
                if np.random.random() < 0.3:
                    # Annealing: accept random swap
                    p = int(np.random.choice(ones))
                    q = int(np.random.choice(zeros))
                    h[p] = 0
                    h[q] = 1
                    corr = fft_corr(h)
                    cur_max = float(np.max(corr))
                    no_improve = 0
                elif no_improve > 200:
                    # Larger perturbation
                    npert = min(5, n_ones, n_zeros)
                    ps = np.random.choice(ones, npert, replace=False)
                    qs = np.random.choice(zeros, npert, replace=False)
                    h[ps] = 0
                    h[qs] = 1
                    corr = fft_corr(h)
                    cur_max = float(np.max(corr))
                    no_improve = 0
    
    return best_h, best_c
```
