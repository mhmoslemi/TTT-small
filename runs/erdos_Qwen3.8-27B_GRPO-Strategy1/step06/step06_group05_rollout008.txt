def fft_corr(h):

    L = 2 * N - 1

    h = np.clip(h, 0.0, 1.0)

    lo, hi = -1.0, 1.0

    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pk = pos + sv
    pm = pos - sv

    pmc = np.clip(pm, 0, N - 1)

    tm = np.where(vm, h[pmc], 0.0)

    # Smooth max approximation
    cmax = c.max()
    if T < 1e-10:
        return cmax
    exp_c = np.exp((c - cmax) * T)
    return cmax + np.log(exp_c.sum()) / T

    # Gradient of smooth max w.r.t. c
    cmax = c.max()
    exp_c = np.exp((c - cmax) * T)
    return exp_c / exp_c.sum()

def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)

    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]

    pm = pos - sv

    vm = (pm >= 0) & (pm < N)
    pkc = np.clip(pk, 0, N - 1)
    pmc = np.clip(pm, 0, N - 1)
    tp = np.where(vp, 1.0 - h[pkc], 0.0)
    tm = np.where(vm, h[pmc], 0.0)
    return tp - tm

            best_c5, best_h = c5F, hF.copy()

    # Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = (best_h > 0.5).astype(float)
        hF2, _ = tabu_binary(hF2, len(hF2), deadline - 0.5, tabu_size=30, max_sweeps=60, sample_p=50)
        c5F2 = compute_c5(hF2, len(hF2))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
