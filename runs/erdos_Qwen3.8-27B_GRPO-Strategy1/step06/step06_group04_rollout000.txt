def fft_corr(h):

    L = 2 * N - 1

    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    corr = fft_corr(h)

            h = project_sum(h, target)
            m[:] = 0
            v[:] = 0
            t = 0
            lr = 0.02
            temp = 0.005
            stall = 0
    return best_h, best_pk * (2.0 / N)

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad(h, N, k_top, temp):
    corr = fft_corr(h)
    top = np.argpartition(corr, -k_top)[-k_top:]
    w = np.exp((corr[top] - corr[top].max()) / temp)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for wi, ai in zip(w, top):
        L = int(ai) - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            seg = N - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] -= wi * h[L:]
        else:
            M = -L
            seg = N - M
            grad[:seg] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:seg]
    return grad, corr
