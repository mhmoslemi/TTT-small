def anneal_fast(h, N, deadline, T0=0.05, cool=0.9995):
    # h is binary {0,1} with sum = N/2
    dx = 2.0/N
    corr = correlate_ffth(h, np.ones(N) - h)  # or use the FFT approach
    current_max = float(corr.max())
    best_max = current_max
    best_h = h.copy()
    
    T = T0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline and T > 0.001:
        p = ones[np.random.randint(len(ones))]
        q = zeros[np.random.randint(len(zeros))]
        
        # Compute delta
        delta = compute_delta(h, N, p, q)
        new_corr = corr + delta
        new_max = float(new_corr.max())
        
        if new_max < current_max or np.random.random() < np.exp(-(new_max - current_max) / T):
            # Accept
            h[p] = 0
            h[q] = 1
            corr = new_corr
            current_max = new_max
            # Update ones/zeros
            # (need to be careful here)
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        T *= cool
    
    return best_h, best_max * dx

The compute_delta function:
```python
def compute_delta(h, N, p, q):
    # p: 1→0, q: 0→1
    delta = np.zeros(2*N-1)
    s = np.arange(-(N-1), N)  # lags
    
    # Effect of h[p]: 1→0
    # Term n=p: -(1-h[p+s]) for valid p+s
    # Term n=p-s: +h[p-s] for valid p-s
    ps = p + s
    pm = p - s
    vps = (ps >= 0) & (ps < N)
    vpm = (pm >= 0) & (pm < N)
    
    delta[vps] -= (1.0 - h[ps[vps]])
    pc = np.clip(pm, 0, N-1)
    delta[vpm] += h[pc[vpm]]
    
    # Effect of h[q]: 0→1
    qs = q + s
    qm = q - s
    vqs = (qs >= 0) & (qs < N)
    vqm = (qm >= 0) & (qm < N)
    
    qsc = np.clip(qs, 0, N-1)
    qmc = np.clip(qm, 0, N-1)
    delta[vqs] += (1.0 - h[qsc[vqs]])
    delta[vqm] -= h[qmc[vqm]]
    
    return delta
```

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2 * N - 1]

def fix_count(h, N):
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        pick = np.random.choice(len(idx), n1 - target, replace=False)
        h[idx[pick]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        pick = np.random.choice(len(idx), target - n1, replace=False)
        h[idx[pick]] = 1
    return h

def compute_delta(h, N, p, q, s):
    pm_s = p - s
    pp_s = p + s
    qm_s = q - s
    qp_s = q + s
    d = np.zeros(2 * N - 1)
    m1 = (pm_s >= 0) & (pm_s < N)
    m2 = (pp_s >= 0) & (pp_s < N)
    m3 = (qm_s >= 0) & (qm_s < N)
    m4 = (qp_s >= 0) & (qp_s < N)
    d[m1] -= (1.0 - h[pm_s[m1]])
    d[m2] += h[pp_s[m2]]
    d[m3] += (1.0 - h[qm_s[m3]])
    d[m4] -= h[qp_s[m4]]
    return d

def anneal_fast(h, N, deadline, T0=0.06, cool=0.9994, max_streak=1200):
    h = h.astype(float).copy()
    s = np.arange(-(N - 1), N)
    corr = fft_corr(h)
    cur_max = float(corr.max())
    best_max = cur_max
    best_h = h.copy()
    T = T0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    streak = 0
    iters = 0
    while time.time() < deadline and T > 0.0015:
        pi = np.random.randint(len(ones))
        qi = np.random.randint(len(zeros))
        p = int(ones[pi])
        q = int(zeros[qi])
        if p == q:
            continue
        d = compute_delta(h, N, p, q, s)
        nc = corr + d
        nk = float(nc.max())
        delta = nk - cur_max
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            h[p] = 0.0
            h[q] = 1.0
            corr = nc
            cur_max = nk
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if nk < best_max:
                best_max = nk
                best_h = h.copy()
                streak = 0
            else:
                streak += 1
        else:
            streak += 1
        iters += 1
        if iters % 2000 == 0:
            T *= cool
            if streak > max_streak:
                streak = 0
                T *= 1.5
        if streak > max_streak * 3:
            h = best_h.copy()
            corr = fft_corr(h)
            cur_max = best_max
            T = max(T, 0.03)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            streak = 0
    return best_h, best_max

def thue_morse(N):
    t = np.zeros(N)
    for i in range(N):
        t[i] = bin(i).count('1') % 2
    return t

def quad_residue_seq(N):
    p = max(3, N if N % 2 == 1 else N)
    while p < N + 20:
        if is_prime(p):
            break
        p += 2
    qr = np.zeros(p, dtype=np.float64)
    for x in range(1, p):
        qr[(x * x) % p] = 1.0
    if p > N:
        qr = qr[:N]
    else:
        pad = np.zeros(N - p)
        qr = np.concatenate([qr, pad])
    return qr

def is_prime(n):
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

def staggered_pattern(N):
    h = np.zeros(N)
    block = 4
    for i in range(N):
        phase = (i // block) % block
        pos = i % block
        if phase == 0 and pos < 2:
            h[i] = 1
        elif phase == 1 and pos >= 2:
            h[i] = 1
        elif phase == 2 and pos < 1:
            h[i] = 1
        elif phase == 3 and pos >= 3:
            h[i] = 1
    return h
