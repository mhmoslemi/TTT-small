def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L <<= 1
    hm = 1.0 - h
    rev = hm[::-1]
    H = np.fft.fft(h, L)
    R = np.fft.fft(rev, L)
    return np.real(np.fft.ifft(H * R, L))[:2 * N - 1]

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    """Project h to [0,1] with sum = target."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

    ones = np.where(h == 1)[0]

def nesterov(h, N, deadline, k_peaks=50, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    y = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    best_h = h.copy()
    best_peak = float(np.max(fft_corr(h)))
    center = N - 1
    
    it = 0
    while time.time() < deadline and it < 20000:
        y = h + beta * m
        corr = fft_corr(y)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = y.copy()
        
        # Compute gradient weighted by top-k peaks
        top_idx = np.argpartition(corr, -k_peaks)[-k_peaks:]
        # Reduce weight for lower peaks in the top set
        w = np.zeros(len(corr))
        w[top_idx] = 1.0
        
        grad = np.zeros(N)
        for idx in top_idx:
            r = int(idx - center)
            if r == 0:
                grad += (1.0 - 2.0 * y)
            elif r > 0:
                grad[:N - r] += (1.0 - y[r:])
                grad[r:] -= y[:N - r]
            else:
                a = -r
                grad[a:] += (1.0 - y[:N - a])
                grad[:N - a] -= y[a:]
        
        grad /= k_peaks
        
        # Projected step
        h_new = y - lr * grad
        h_new = project_sum(h_new, target)
        
        m = h_new - h
        h = h_new
        it += 1
        
        if it % 100 == 0:
            lr *= 0.992
            beta = min(0.96, 0.85 + it * 0.0001)
    
    return best_h, best_peak * (2.0 / N)

Now let me also think about the Nesterov step. The gradient computation for the max of multiple correlations is:

If we define g(y) = max_k corr_k(y), the subgradient at y is the average of gradients of all k that achieve the max (or a weighted combination for a smoothed version).

For a "soft-max" version: g_ε(y) = ε * log(sum_k exp(corr_k/ε))
∂g_ε/∂y[i] = (1/sum_k exp) * sum_k exp(corr_k/ε) * ∂corr_k/∂y[i]

This is a weighted average of the individual correlation gradients, weighted by their "importance" (exp(corr_k/ε)).

For the discrete binary case, the Nesterov approach with continuous interpolation might not help much. Let me focus on making the binary SA as effective as possible.

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def delta_corr(h, p, q, N):
    """O(N) computation of correlation change for swap p:1->0, q:0->1"""
    ks = np.arange(-(N - 1), N)
    d = np.zeros(2 * N - 1)
    pk = p + ks
    qk = q + ks
    pm = p - ks
    qm = q - ks
    m = (pk >= 0) & (pk < N)
    d[m] -= 1.0 - h[pk[m]]
    m = (qk >= 0) & (qk < N)
    d[m] += 1.0 - h[qk[m]]
    m = (pm >= 0) & (pm < N)
    d[m] -= h[pm[m]]
    m = (qm >= 0) & (qm < N)
    d[m] -= h[qm[m]]
    return d

def fix_sum(h, N):
    h = (h >= 0.5).astype(float)
    t = N // 2
    n1 = int(h.sum())
    if n1 > t:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[t:]] = 0.0
    elif n1 < t:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:t - n1]] = 1.0
    return h

def project_continuous(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):

            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_opt(h, N, deadline, k_peek=50, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    y = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.88
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    center = N - 1
    
    it = 0
    while time.time() < deadline and it < 30000:
        y = h + beta * m
        corr = compute_corr(y)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = y.copy()
        
        # Soft-max weighted gradient
        # Use top-k for efficiency
        top = np.argpartition(corr, -k_peek)[-k_peek:]
        # Soft weights based on excess over median of top
        w = np.zeros(len(corr))
        ct = corr[top]
        w_max = ct.max()
        w[top] = np.exp(5.0 * (ct - w_max))  # softmax-like with temperature 0.2
        
        grad = np.zeros(N)
        for idx in top:
            wi = w[idx]
            r = int(idx - center)
            if r == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif r > 0:
                grad[:N-r] += wi * (1.0 - y[r:])
                grad[r:] -= wi * y[:N-r]
            else:
                a = -r
                grad[a:] += wi * (1.0 - y[:N-a])
                grad[:N-a] -= wi * y[a:]
        
        grad /= w.sum()
        
        h_new = project_continuous(y - lr * grad, target)
        m = h_new - h
        h = h_new
        it += 1
        
        if it % 100 == 0:
            lr *= 0.993
            beta = min(0.97, 0.86 + it * 0.00015)
    
    return best_h, best_peak * (2.0 / N)
