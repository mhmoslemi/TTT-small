import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    """Convert to binary with exactly N//2 ones."""
    h_b = (h > 0.5).astype(float)
    n_one = int(h_b.sum())
    target = N // 2
    if n_one > target:
        ones = np.where(h_b == 1)[0]
        np.random.shuffle(ones)
        h_b[ones[:n_one-target]] = 0
    elif n_one < target:
        zeros = np.where(h_b == 0)[0]
        np.random.shuffle(zeros)
        h_b[zeros[:target-n_one]] = 1
    return h_b

def swap_delta_vec(h, p, q):
    N = len(h)
    L = np.arange(-(N-1), N)
    delta = np.zeros(2*N-1)
    
    mask1 = (0 <= p + L) & (p + L < N)
    delta[mask1] -= (1.0 - h[p + L[mask1]])
    mask2 = (0 <= p - L) & (p - L < N)
    delta[mask2] += h[p - L[mask2]]
    mask3 = (0 <= q + L) & (q + L < N)
    delta[mask3] += (1.0 - h[q + L[mask3]])
    mask4 = (0 <= q - L) & (q - L < N)
    delta[mask4] -= h[q - L[mask4]]
    delta[N-1] = 0.0
    return delta

def binary_sa(h0, N, deadline, max_iter=None):
    """Simulated annealing for binary optimization with fast swap evaluation."""
    h = h0.copy()
    corr = fft_corr(h).copy()
    cur_max = int(corr.max())
    best_h = h.copy()
    best_int = cur_max
    
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    
    if max_iter is None:
        max_iter = 10_000_000
    
    n_iter = 0
    T = 1.0  # temperature
    cool = 0.9999
    
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        p = ones[np.random.randint(len(ones))]
        q = zeros[np.random.randint(len(zeros))]
        
        delta = swap_delta_vec(h, p, q)
        new_corr = corr + delta
        new_max = int(new_corr.max())
        
        # Accept if improvement or with probability exp(-delta/T)
        if new_max < cur_max:
            # Accept
            h[p] = 0
            h[q] = 1
            corr = new_corr
            cur_max = new_max
            # Update ones/zeros tracking
            idx_p = np.where(ones == p)[0]
            ones = np.delete(ones, idx_p[0])
            zeros = np.append(zeros, p)
            idx_q = np.where(zeros == q)[0]
            zeros = np.delete(zeros, idx_q[0])
            ones = np.append(ones, q)
            
            if cur_max < best_int:
                best_int = cur_max
                best_h = h.copy()
        else:
            # Reject (for SA, could accept with prob, but for ILB just reject)
            pass
        
        n_iter += 1
        if n_iter % 10000 == 0:
            T *= cool
            # Kick: random 5-swap
            if n_iter % 50000 == 0:
                kick = min(5, len(ones) // 2, len(zeros) // 2)
                for _ in range(kick):
                    pp = ones[np.random.randint(len(ones))]
                    qq = zeros[np.random.randint(len(zeros))]
                    h[pp], h[qq] = h[qq], h[pp]
                    corr = fft_corr(h)
                    cur_max = int(corr.max())
                    idx_p = np.where(ones == pp)[0]
                    ones = np.delete(ones, idx_p[0])
                    zeros = np.append(zeros, pp)
                    idx_q = np.where(zeros == qq)[0]
                    zeros = np.delete(zeros, idx_q[0])
                    ones = np.append(ones, qq)
                
    return best_h, best_int * (2.0 / N)

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    tgt = N // 2
    if n1 > tgt:
        idx = np.where(h_b == 1)[0]
        np.random.shuffle(idx)
        h_b[idx[:n1 - tgt]] = 0
    elif n1 < tgt:
        idx = np.where(h_b == 0)[0]
        np.random.shuffle(idx)
        h_b[idx[:tgt - n1]] = 1
    return h_b

def swap_delta_vec(h, p, q, L_arr):
    N = len(h)
    delta = np.zeros(2 * N - 1)
    m1 = (0 <= p + L_arr) & (p + L_arr < N)
    delta[m1] -= (1.0 - h[p + L_arr[m1]])
    m2 = (0 <= p - L_arr) & (p - L_arr < N)
    delta[m2] += h[p - L_arr[m2]]
    m3 = (0 <= q + L_arr) & (q + L_arr < N)
    delta[m3] += (1.0 - h[q + L_arr[m3]])
    m4 = (0 <= q - L_arr) & (q - L_arr < N)
    delta[m4] -= h[q - L_arr[m4]]
    delta[N - 1] = 0.0
    return delta

def binary_hill(h0, N, deadline, n_cand=8, kick_size=5, kick_freq=3000):

    corr = fft_corr(h).copy()
    cur_max = int(corr.max())
    best_h = h.copy()
    best_int = cur_max
    L_arr = np.arange(-(N - 1), N)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    stall = 0
    it = 0
    while time.time() < deadline:
        if stall > kick_freq:
            # Kick
            k = min(kick_size, len(ones) // 2, len(zeros) // 2)
            for _ in range(k):
                pp = ones[np.random.randint(len(ones))]
                qq = zeros[np.random.randint(len(zeros))]
                h[pp], h[qq] = 0.0, 1.0
                # update lists
                ones.remove(pp)
                zeros.remove(qq)
                ones.append(qq)
                zeros.append(pp)
            corr = fft_corr(h).copy()
            cur_max = int(corr.max())
            stall = 0
            it = 0
            continue
        
        best_new = None
        best_new_max = cur_max
        for _ in range(n_cand):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            delta = swap_delta_vec(h, p, q, L_arr)
            new_corr = corr + delta
            nm = int(new_corr.max())
            if nm < best_new_max:
                best_new_max = nm
                best_new = (p, q, new_corr)

            h[p], h[q] = 0.0, 1.0
            corr = new_corr
            cur_max = best_new_max

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def continuous_opt(h0, N, deadline, k=40, lr0=0.02, temp0=0.003):
    h = h0.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_max = float(np.max(fft_corr(h)))
    h_prev = h.copy()

    beta = 0.9
    temp = temp0
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_max - 1e-14:
            best_max = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        top = np.argpartition(corr, -k)[-k:]

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N):
    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    tgt = N // 2
    if n1 > tgt:
        idx = np.where(h_b == 1)[0]
        np.random.shuffle(idx)
        h_b[idx[:n1 - tgt]] = 0.0
    elif n1 < tgt:

        np.random.shuffle(idx)
        h_b[idx[:tgt - n1]] = 1.0
