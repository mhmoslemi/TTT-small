def fft_corr(h):

    return np.fft.irfft(H * G, L)

def compute_c5(h, N):

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_sum(h, N):
    h = h.copy()
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        rm = np.random.choice(idx, n1 - target, replace=False)
        h[rm] = 0.0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        add = np.random.choice(idx, target - n1, replace=False)
        h[add] = 1.0
    return h

def precomp_indices(N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    return pm_c, pp_c, vm, vp

def compute_D(h, pre):
    pm_c, pp_c, vm, vp = pre
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binary_opt(h, N, deadline, max_sweeps=5000, sample_n=200,
               flat_w=0.06, shake_interval=150, shake_size=4):
    h = h.astype(float).copy()
    corr = fft_corr(h)
    D = compute_D(h, precomp_indices(N))
    best_c = float(corr.max())
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(corr.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        # Shake to escape local minima
        if sweep > 50 and no_improve > shake_interval and np.random.random() < 0.3:
            for _ in range(shake_size):
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = fft_corr(h)
            D = compute_D(h, precomp_indices(N))
            no_improve = 0
            continue

        best_score = P + 1.0
        best_pq = None

        if len(ones) > sample_n:
            perm = np.random.permutation(len(ones))[:sample_n]
            p_list = ones[perm]
        else:
            p_list = ones

        zD = D[zeros, :]
        for p in p_list:
            base = corr - D[p, :]
            npk = (base[None, :] - zD).max(axis=1)
            # Flat penalty: add small weight to 2nd peak
            mi = int(np.argmin(npk))
            sc = npk[mi]
            if sc < best_score:
                best_score = sc
                best_pq = (int(p), int(zeros[mi]))

        # Occasionally try random q for diversity
        if len(zeros) > 10:
            for _ in range(5):
                ri = int(np.random.randint(len(p_list)))
                p = int(p_list[ri])
                rq = int(np.random.randint(len(zeros)))
                nc = corr - D[p, :] - zD[rq, :]
                sc = float(nc.max())
                if sc < best_score:
                    best_score = sc
                    best_pq = (p, int(zeros[rq]))

        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0

        corr = fft_corr(h)
        D = compute_D(h, precomp_indices(N))

        if no_improve > 600:
            break

    return h, best_c

def nesterov_opt(h, N, deadline, k=40, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = np.clip(y - lr * grad, 0.0, 1.0)
        # Project to correct sum
        diff = target - h_new.sum()
        h_new += diff / N
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum exactly
        h_new = np.clip(h_new, 0.0, 1.0)
        s = h_new.sum()
        if abs(s - target) > 1e-6:
            lo, hi = -0.5, 0.5
            for _ in range(40):
                mid = (lo + hi) * 0.5
                if np.clip(h_new - mid, 0, 1).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo + hi) * 0.5, 0.0, 1.0)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.993
            beta = min(0.96, 0.85 + it * 0.0004)
    return best_h, best_pk * (2.0 / N)
