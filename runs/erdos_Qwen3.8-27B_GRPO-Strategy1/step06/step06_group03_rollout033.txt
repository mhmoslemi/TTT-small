def local_search_v2(h, N, deadline, max_sweeps=200):
    h = h.copy()
    nh = N // 2
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = N - 1
    best_max = float(corr.max())
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        M = float(corr.max())
        if M < best_max:
            best_max = M
            best_h = h.copy()
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Critical lags: those with corr >= M - 1
        crit_mask = corr >= M - 1
        crit_lags = np.where(crit_mask)[0]
        
        # Top lags for scoring
        top_k = min(10, len(corr))
        top_lags = np.argsort(corr)[-top_k:]
        
        best_new_max = M
        best_pq = None
        
        # Try all pairs (vectorized in batches)
        n_o = len(ones)
        n_z = len(zeros)
        
        for i in range(n_o):
            p = ones[i]
            for j in range(n_z):
                q = zeros[j]
                # Compute delta at critical lags
                ok = True
                new_maxes = []
                for L in crit_lags:
                    d = L - center
                    # delta at lag d
                    delta = 0
                    # Effect of removing 1 at p
                    if 0 <= p + d < N:
                        delta -= (1 - h[p+d])
                    if 0 <= p - d < N:
                        delta -= h[p-d]
                    # Effect of adding 1 at q
                    if 0 <= q + d < N:
                        delta += (1 - h[q+d])
                    if 0 <= q - d < N:
                        delta -= h[q-d]
                    # Interference correction
                    if p + d == q or q - d == p:
                        # d = q - p: overcounted by 1
                        delta += 1
                    if p - d == q or q + d == p:
                        # d = p - q: overcounted by 1 (but in opposite direction)
                        delta -= 1  # Actually need to check sign
                    if corr[L] + delta >= M:
                        ok = False
                        break
                if ok:
                    # Compute actual new max
                    actual_new_max = 0
                    for L in top_lags:
                        d = L - center
                        delta = 0
                        if 0 <= p + d < N:
                            delta -= (1 - h[p+d])
                        if 0 <= p - d < N:
                            delta -= h[p-d]
                        if 0 <= q + d < N:
                            delta += (1 - h[q+d])
                        if 0 <= q - d < N:
                            delta -= h[q-d]
                        if p + d == q or q - d == p:
                            delta += 1
                        if p - d == q or q + d == p:
                            delta -= 1
                        actual_new_max = max(actual_new_max, corr[L] + delta)
                    # Also check non-top lags could increase
                    # (but for binary, max increase is 1, so if corr[L]+1 < M for all non-top, we're fine)
                    if M - corr.max() >= 1:  # always true since M is the max
                        pass
                    # Check if any non-critical lag could exceed
                    non_crit_max = corr[~crit_mask].max() if (~crit_mask).any() else 0
                    if non_crit_max + 2 >= M:  # could be a problem
                        # Need to check these too, but for now skip
                        pass
                    if actual_new_max < best_new_max:
                        best_new_max = actual_new_max
                        best_pq = (int(p), int(q))
        
        if best_pq is None:
            # Stuck, try random perturbation
            if np.random.random() < 0.5:
                p = int(ones[np.random.randint(n_o)])
                q = int(zeros[np.random.randint(n_z)])
                h[p], h[q] = 0.0, 1.0
                corr = np.correlate(h, 1.0 - h, mode="full")
            else:
                # Try multi-swap: swap two pairs
                if n_o >= 2 and n_z >= 2:
                    p1 = int(ones[np.random.randint(n_o)])
                    p2 = int(ones[np.random.randint(n_o)])
                    q1 = int(zeros[np.random.randint(n_z)])
                    q2 = int(zeros[np.random.randint(n_z)])
                    if p1 != p2 and q1 != q2:
                        h[p1], h[q1] = 0.0, 1.0
                        h[p2], h[q2] = 0.0, 1.0
                        corr = np.correlate(h, 1.0 - h, mode="full")
                else:
                    break
        else:
            p, q = best_pq
            h[p], h[q] = 0.0, 1.0
            corr = np.correlate(h, 1.0 - h, mode="full")
    
    return best_h, best_max
