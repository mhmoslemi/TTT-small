def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)

    return np.real(np.fft.ifft(H * G))

def compute_c5(h):

    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -0.5, 0.5

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -0.5, 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, corr, temp):
    N = len(h)
    L = 2 * N - 1
    center = N - 1
    shifted = (corr - corr.max()) / max(temp, 1e-10)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(L)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(N)
    Bh = 1.0 - h
    for k in range(L):
        wk = w[k]
        if wk < 1e-10:
            continue
        shift = k - center
        if shift == 0:
            grad += wk * (Bh - h)
        elif 0 < shift < N:
            seg = N - shift
            grad[:seg] += wk * Bh[shift:]
            grad[shift:] -= wk * h[:seg]
        elif -N < shift < 0:
            s = -shift
            seg = N - s
            grad[s:] += wk * Bh[:seg]
            grad[:seg] -= wk * h[s:]
    return grad

def continuous_opt(h0, deadline, lr0=0.025, temp0=0.008, kick_noise=0.03, kick_stall=300):
    N = len(h0)
    target = N / 2.0
    h = project_sum(h0.copy(), target)
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    v = np.zeros(N)
    beta = 0.88
    lr = lr0
    temp = temp0
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + beta * v
        h_new = project_sum(y, target)
        v = h_new - h
        h = h_new
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-14:
            best_pk = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        grad = compute_grad(h, corr, temp)
        h = project_sum(h - lr * grad, target)
        it += 1
        if it % 80 == 0:
            lr = max(0.001, lr * 0.996)
            temp = max(0.0005, temp * 0.994)
            beta = min(0.97, 0.85 + it * 0.0003)
        if stall > kick_stall:
            h = best_h + np.random.randn(N) * kick_noise
            h = project_sum(h, target)
            v = np.zeros(N)
            lr = max(lr, lr0 * 0.6)
            temp = max(temp, temp0 * 0.4)
            stall = 0
    return best_h, best_pk * (2.0 / N)

def binary_opt(h0, deadline, n_tries=15, kick_freq=60):
    N = len(h0)
    h = (h0 > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            idx = idx[np.random.permutation(len(idx))][:n1 - nh]
            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh - n1]
            h[idx] = 1
    c = fft_corr(h)
    best_int = int(np.max(c))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_sweep_int = int(c.max())
        best_sweep_h = None
        best_sweep_c = None
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_t = fft_corr(h)
            ct_int = int(c_t.max())
            if ct_int < best_sweep_int:
                best_sweep_int = ct_int
                best_sweep_h = h.copy()
                best_sweep_c = c_t.copy()
            h[p] = 1
            h[q] = 0
        if best_sweep_h is not None:
            h = best_sweep_h
            c = best_sweep_c
            stall = 0
            if best_sweep_int < best_int:
                best_int = best_sweep_int
                best_h = h.copy()
        else:
            stall += 1
            if stall > kick_freq:
                n_kick = min(3, len(ones), len(zeros))
                i1 = np.random.choice(len(ones), n_kick, replace=False)
                i2 = np.random.choice(len(zeros), n_kick, replace=False)
                for pp in ones[i1]:
                    h[pp] = 0
                for qq in zeros[i2]:
                    h[qq] = 1
                c = fft_corr(h)
                stall = 0
    return best_h, best_int * (2.0 / N)

    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)
