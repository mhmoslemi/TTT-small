    N = len(h)
    return float(np.max(correlate_full(h))) * (2.0 / N)

def proj_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def smooth_grad(h, beta=80):
    N = len(h)
    corr = correlate_full(h)
    center = N - 1
    cs = corr - corr.max()
    e = np.exp(beta * cs)
    wsum = e.sum()
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w = e / wsum
    grad = np.zeros(N)
    active = np.where(w > 1e-8)[0]
    for k in active:
        wk = w[k]
        l = k - center
        if l == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            j = abs(l)
            g = np.zeros(N)
            g[j:] += (1.0 - h[:N-j])
            g[:N-j] -= h[j:]
            grad += wk * g
    return grad, float(corr.max())

def grad_opt(h, N, deadline, beta=80, lr0=0.015):
    target = N / 2.0
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = c5_of(best_h)
    lr = lr0
    mu = 0.85
    v = np.zeros(N)
    it = 0
    while time.time() < deadline and it < 80000:
        g, _ = smooth_grad(h, beta)
        v = mu * v + (1.0 - mu) * g
        h = proj_box_sum(h - lr * v, target)
        it += 1
        if it % 300 == 0:
            c = c5_of(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr *= 0.99995
            mu = min(0.96, 0.85 + it * 0.00001)
    c = c5_of(h)
    if c < best_c:
        return h, c
    return best_h, best_c

def anneal_binary(h, N, deadline, seed, n_move=1):
    rng = np.random.RandomState(seed)
    h = (h > 0.5).astype(float).copy()
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        rm = idx[rng.choice(len(idx), n1 - target, replace=False)]
        h[rm] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        add = idx[rng.choice(len(idx), target - n1, replace=False)]
        h[add] = 1
    corr = correlate_full(h)
    peak = float(corr.max())
    best_h = h.copy()
    best_peak = peak
    T = 0.004
    cool = 0.9997
    it = 0
    while time.time() < deadline and it < 800000:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if n_move >= 2 and rng.random() < 0.3:
            p1 = ones[rng.randint(len(ones))]
            p2 = ones[rng.randint(len(ones))]
            q1 = zeros[rng.randint(len(zeros))]
            q2 = zeros[rng.randint(len(zeros))]
            if p1 == p2 or q1 == q2:
                continue
            h[[p1,p2]], h[[q1,q2]] = h[[q1,q2]], h[[p1,p2]]
            corr_new = correlate_full(h)
            pk_new = float(corr_new.max())
            d = pk_new - peak
            if d < 0 or rng.random() < np.exp(-d/T):
                peak = pk_new
                corr = corr_new
                if peak < best_peak:
                    best_peak = peak
                    best_h = h.copy()
            else:
                h[[p1,p2]], h[[q1,q2]] = h[[q1,q2]], h[[p1,p2]]
        else:
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            corr_new = correlate_full(h)
            pk_new = float(corr_new.max())
            d = pk_new - peak
            if d < 0 or rng.random() < np.exp(-d/T):
                peak = pk_new
                corr = corr_new
                if peak < best_peak:
                    best_peak = peak
                    best_h = h.copy()
            else:
                h[p], h[q] = 1.0, 0.0
        T = max(0.00005, T * cool)
    return best_h, best_peak * (2.0/N)

def duplicate_upscale(h, factor=2):
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of(best_h)

    # Phase 1: Annealing at N0 with pair swaps
    d1 = min(deadline - 500, t0 + 200)
    h1, c1 = anneal_binary(h0, N0, d1, seed=seed, n_move=2)
    if c1 < best_c5:
        best_h, best_c5 = h1.copy(), c1

    # Phase 2: Gradient polish at N0
    if time.time() < deadline - 400:
        d2 = min(deadline - 380, time.time() + 120)
        h2 = best_h.astype(float) + np.random.randn(N0) * 0.025
        h2 = proj_box_sum(h2, N0 / 2.0)
        h2, c2 = grad_opt(h2, N0, d2, beta=90, lr0=0.012)
        if c2 < best_c5:
            best_h, best_c5 = h2.copy(), c2

    # Phase 3: Annealing again from gradient result
    if time.time() < deadline - 280:
        d3 = min(deadline - 260, time.time() + 100)
        h3, c3 = anneal_binary(best_h, N0, d3, seed=seed+7, n_move=2)
        if c3 < best_c5:
            best_h, best_c5 = h3.copy(), c3

    # Phase 4: Upscale to 2*N0 and optimize
    N_up = N0 * 2
    if time.time() < deadline - 120:
        h_up = duplicate_upscale(best_h, 2)
        target_up = N_up // 2
        n1 = int(h_up.sum())
        if n1 > target_up:
            idx = np.where(h_up == 1)[0]
            rm = idx[np.argsort(np.random.random(len(idx)))][target_up:]
            h_up[rm] = 0
        elif n1 < target_up:
            idx = np.where(h_up == 0)[0]
            add = idx[np.argsort(np.random.random(len(idx)))][:target_up - n1]
            h_up[add] = 1
        d4 = min(deadline - 100, time.time() + 80)
        h4, c4 = anneal_binary(h_up, N_up, d4, seed=seed+13, n_move=2)
        c4v = c5_of(h4)
        if c4v < best_c5:
            best_h, best_c5 = h4.copy(), c4v

    # Phase 5: Gradient on best (whichever N)
    Nf = len(best_h)
    if time.time() < deadline - 60:
        d5 = min(deadline - 40, time.time() + 50)
        hf = best_h.astype(float) + np.random.randn(Nf) * 0.015
        hf = proj_box_sum(hf, Nf / 2.0)
        hf, c5f = grad_opt(hf, Nf, d5, beta=100, lr0=0.01)
        if c5f < best_c5:
            best_h, best_c5 = hf, c5f

    # Phase 6: Final quick anneal
    if time.time() < deadline - 1:
        Nf2 = len(best_h)
        d6 = deadline - 0.5
        h6, c6 = anneal_binary(best_h, Nf2, d6, seed=seed+21, n_move=1)
        if c6 < best_c5:
            best_h, best_c5 = h6.copy(), c6
    # Final gradient micro-polish
    if time.time() < deadline - 0.5:
        Nf3 = len(best_h)
        hf3 = best_h.astype(float) + np.random.randn(Nf3) * 0.01
        hf3 = proj_box_sum(hf3, Nf3 / 2.0)
        hf3, c5f3 = grad_opt(hf3, Nf3, deadline - 0.2, beta=120, lr0=0.005)
        if c5f3 < best_c5:
            best_h, best_c5 = hf3, c5f3

    final_c5 = c5_of(best_h)
    return best_h, final_c5, len(best_h)
