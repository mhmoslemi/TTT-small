def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def nesterov_smooth(h_in, N, deadline, T0=0.03, k=40):

    target = N / 2.0
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = 0.03
    beta = 0.8
    T = T0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(c_top / T - c_top.max() / T)
        w /= w.sum()
        grad = np.zeros(N)
        for ki in range(len(idx)):
            kv = int(idx[ki] - (N - 1))
            if kv == 0:
                grad += w[ki] * (1.0 - 2.0 * y)
            elif kv > 0:
                grad[kv:] += w[ki] * (1.0 - y[:N-kv])
                grad[:N-kv] -= w[ki] * y[kv:]
            else:
                m2 = -kv
                grad[:N-m2] += w[ki] * (1.0 - y[m2:])
                grad[m2:] -= w[ki] * y[:m2]
        h_new = project_sum(y - lr * grad, target)

        it += 1

            lr *= 0.995
            beta = min(0.95, 0.78 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_binary(h, nh):
    h = h.astype(float).copy()
    n1 = int(round(h.sum()))
    if n1 > nh:
        ones = np.where(h > 0.5)[0]
        rem = np.random.choice(ones, n1 - nh, replace=False)
        h[rem] = 0.0
    elif n1 < nh:
        zeros = np.where(h < 0.5)[0]
        add = np.random.choice(zeros, nh - n1, replace=False)
        h[add] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def nesterov_smooth(h_in, N, deadline, T0=0.04, k=50):
    h = h_in.astype(float).copy()
    target = N / 2.0
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    m = np.zeros(N)
    lr = 0.03
    beta = 0.80
    T = T0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[idx]
        w = np.exp(c_top / T - c_top.max() / T)
        w /= w.sum()
        grad = np.zeros(N)
        for ki in range(len(idx)):
            kv = int(idx[ki] - (N - 1))
            if kv == 0:
                grad += w[ki] * (1.0 - 2.0 * y)
            elif kv > 0:
                grad[kv:] += w[ki] * (1.0 - y[:N - kv])
                grad[:N - kv] -= w[ki] * y[kv:]
            else:
                m2 = -kv
                grad[:N - m2] += w[ki] * (1.0 - y[m2:])
                grad[m2:] -= w[ki] * y[:m2]
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 30 == 0:
            T = max(0.004, T * 0.96)
            lr *= 0.995
            beta = min(0.95, 0.78 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

            best_c5, best_h = c_rt, h_rt.copy()

    # Phase 5: Final polish (remaining time)
    if time.time() < deadline - 10:
        d5 = deadline - 3
        h5_bin = (best_h > 0.5).astype(float)
        h5, c5_raw = sa_search(h5_bin, len(best_h), d5, T0=0.2, T1=0.001, max_stall=5000)
        c5v = c5_raw * (2.0 / len(best_h))
        if c5v < best_c5:
            best_c5, best_h = c5v, h5.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
