def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h + mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rm = idx[np.argsort(np.random.random(len(idx)))][:n1 - nh]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        n_add = nh - n1
        perm = np.random.permutation(len(idx))[:n_add]
        h[idx[perm]] = 1.0
    return h

def flip_1to0(p, h, N):
    """Return vector of length 2N-1: change in corr when h[p] goes 1->0."""
    nc = 2 * N - 1
    n_arr = np.arange(nc)
    x1 = p + N - 1 - n_arr
    x2 = p - N + 1 + n_arr
    d = np.zeros(nc)
    m1 = (x1 >= 0) & (x1 < N)
    m2 = (x2 >= 0) & (x2 < N)
    d[m1] = (h[x1[m1]] - 1.0)
    d[m2] += h[x2[m2]]
    return d

def flip_0to1(q, h, N):
    """Return vector of length 2N-1: change in corr when h[q] goes 0->1."""
    nc = 2 * N - 1
    n_arr = np.arange(nc)

    m2 = (x2 >= 0) & (x2 < N)

    return e

    best_h = h.copy()
    best_P = P
    T = T0
    no_improve = 0
    sweep = 0
    while sweep < max_sweeps and time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        nc = 2 * N - 1
        # Precompute flip effects
        if len(ones) <= sample_n:
            p_list = ones
            fp = np.array([flip_1to0(int(p), h, N) for p in p_list])
        else:
            perm = np.random.permutation(len(ones))[:sample_n]
            p_list = ones[perm]
            fp = np.array([flip_1to0(int(p), h, N) for p in p_list])

        for i in range(len(p_list)):

            mi = int(np.argmin(scores))
            if scores[mi] < best_score:

                best_pq = (int(p_list[i]), int(zeros[mi]))

            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:

            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        
        c = fft_corr(h)
        P = float(c.max())
        if P < best_P - 1e-12:
            best_P = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        T *= cool

    return best_h, best_P

def nesterov_refine(h, N, deadline, lr0=0.02, k=50):

    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)

    beta = 0.9
    it = 0

        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = y.copy()
            stagnant = 0
        else:
            stagnant += 1
        top = np.argpartition(corr, -k)[-k:]

            L = int(ai - center)
            if L == 0:

        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.99
            beta = min(0.98, 0.88 + it * 0.0002)
        if stagnant > 600:
            break
    return best_h, best_pk * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h + mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def fix_sum_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        rm = idx[np.argsort(np.random.random(len(idx)))][:n1 - nh]
        h[rm] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        n_add = nh - n1
        perm = np.random.permutation(len(idx))[:n_add]
        h[idx[perm]] = 1.0
    return h

def binary_sa(h, N, deadline, sample_n=250, T0=0.5, T_min=0.005, cool=0.997, max_sweeps=10000):
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    best_h = h.copy()
    best_P = P
    T = T0
    no_improve = 0
    sweep = 0
    while sweep < max_sweeps and time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        nc = 2 * N - 1
        if len(ones) <= sample_n:
            p_list = ones
            fp = np.array([_flip1(p, h, N) for p in p_list])
        else:
            perm = np.random.permutation(len(ones))[:sample_n]
            p_list = ones[perm]
            fp = np.array([_flip1(int(p), h, N) for p in p_list])
        fq = np.array([_flip0(int(q), h, N) for q in zeros])
        best_score = P + 2.0
        best_pq = None
        for i in range(len(p_list)):
            base = c + fp[i]
            scores = (base[None, :] + fq).max(axis=1)
            mi = int(np.argmin(scores))
            if scores[mi] < best_score:
                best_score = float(scores[mi])
                best_pq = (int(p_list[i]), int(zeros[mi]))
        if best_pq is not None and (best_score <= P + 1e-10 or np.random.random() < np.exp(-(best_score - P) / (T + 1e-6))):
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        c = fft_corr(h)
        P = float(c.max())
        if P < best_P - 1e-12:
            best_P = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        T = max(T * cool, T_min)
        sweep += 1
        if no_improve > 1200 and T <= T_min * 1.1:
            break
    return best_h, best_P

def _flip1(p, h, N):
    nc = 2 * N - 1
    n_arr = np.arange(nc)
    x1 = p + N - 1 - n_arr
    x2 = p - N + 1 + n_arr
    d = np.zeros(nc)
    m1 = (x1 >= 0) & (x1 < N)
    m2 = (x2 >= 0) & (x2 < N)
    d[m1] = h[x1[m1]] - 1.0
    d[m2] += h[x2[m2]]
    return d

def _flip0(q, h, N):
    nc = 2 * N - 1
    n_arr = np.arange(nc)
    x1 = q + N - 1 - n_arr
    x2 = q - N + 1 + n_arr
    e = np.zeros(nc)
    m1 = (x1 >= 0) & (x1 < N)
    m2 = (x2 >= 0) & (x2 < N)
    e[m1] = 1.0 - h[x1[m1]]
    e[m2] += -h[x2[m2]]
    return e

def nesterov_refine(h, N, deadline, lr0=0.02, k=50):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    stagnant = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = y.copy()
            stagnant = 0
        else:
            stagnant += 1
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.99
            beta = min(0.98, 0.88 + it * 0.0002)
        if stagnant > 800:
            break
    return best_h, best_pk * (2.0 / N)

    if time.time() < deadline - 80:
        d3 = min(deadline - 60, time.time() + 180)
        h_b3 = (best_h > 0.5).astype(float)
        h3, P3 = binary_sa(h_b3, N0, d3, sample_n=300, T0=0.2, T_min=0.002, cool=0.998)
        c5_3 = P3 * (2.0 / N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
    # Phase 4: Upscale to 2x
    N_up = N0 * 2
    if time.time() < deadline - 45:
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_up, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        h_bt = fix_sum_binary(h_bt, N_up)
        d4 = min(deadline - 18, time.time() + 55)
        h_up, P_up = binary_sa(h_bt, N_up, d4, sample_n=120, T0=0.3, T_min=0.01, cool=0.997)
        c5_up = P_up * (2.0 / N_up)
        if c5_up < best_c5:
            best_c5, best_h = c5_up, h_up.copy()
    # Phase 5: Final Nesterov polish
    Nf = len(best_h)
    if time.time() < deadline - 5:

        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = nesterov_refine(hF, Nf, deadline - 2, lr0=0.008, k=80)

        c5F2 = P2 * (2.0 / len(hF2))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2.copy()
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
