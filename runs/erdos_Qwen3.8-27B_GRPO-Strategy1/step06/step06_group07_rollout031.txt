import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def ensure_sum_bin(h, N, seed=42):
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 == nh:
        return h
    rng = np.random.RandomState(seed)
    if n1 > nh:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h[idx] = 0
    else:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h[idx] = 1
    return h

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def thue_morse(N):
    """Generate Thue-Morse binary sequence of length N."""
    t = np.zeros(N)
    for i in range(1, N):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def greed_bin_search(h, N, deadline, max_sweeps=1000, perturb_every=30):
    """Binary search with greedy best swap."""
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum_bin(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(999 + N)
    L = 2 * N - 1
    center = N - 1
    s_vals = np.arange(L) - center  # lags -(N-1)..(N-1)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Compute A for all ones: A_p[lag] = h[p+lag] - (1-h[p-lag])
        # And B for all zeros: B_q[lag] = (1-h[q-lag]) - h[q+lag] = -A_q[lag]
        # For swap (p,q): new_c = c + A_p - A_q
        # Want: min_{p,q} max_lag (c + A_p - A_q)
        
        # Compute A matrix for all positions (N x L)
        pos = np.arange(N)[:, None]
        s_row = s_vals[None, :]
        idx_p = pos + s_row
        v_p = (idx_p >= 0) & (idx_p < N)
        idx_m = pos - s_row
        v_m = (idx_m >= 0) & (idx_m < N)
        
        A = np.zeros((N, L))
        A[v_p] += h[np.clip(idx_p, 0, N-1)[v_p]]
        A[v_m] -= (1.0 - h[np.clip(idx_m, 0, N-1)[v_m]])
        
        # For each p in ones: base_p = c + A[p]
        # For each q in zeros: new_max = max(base_p - A[q])
        
        base = c[None, :] + A  # shape (N, L)
        
        # Sample ones if too many
        if len(ones) > 60:
            p_list = rng.choice(ones, 60, replace=False)
        else:
            p_list = ones
        
        A_q = A[zeros, :]  # shape (len(zeros), L)
        
        best_new_P = P
        best_pq = None
        for p in p_list:
            bp = base[p, :]  # shape (L,)
            vals = bp[None, :] - A_q  # shape (len(zeros), L)
            mxs = vals.max(axis=1)
            mi = int(np.argmin(mxs))
            if mxs[mi] < best_new_P:
                best_new_P = float(mxs[mi])
                best_pq = (int(p), int(zeros[mi]))
        
        if best_pq is None:
            if stall > perturb_every:
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    
    return best_h, best_c * (2.0 / N)

def block_swap_search(h, N, deadline, max_sweeps=200, block=2):
    """Try block swaps of size 'block'."""
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    rng = np.random.RandomState(555 + N)
    L = 2 * N - 1
    center = N - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
        
        # Try random block swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < block or len(zeros) < block:
            break
        
        # Pick a random block of 'block' consecutive ones
        # and a random block of 'block' consecutive zeros
        # Try the swap, check if it improves
        
        p_start = int(rng.randint(0, N - block + 1))
        # Check if h[p_start:p_start+block] is all 1s
        while not np.all(h[p_start:p_start+block] == 1):
            p_start = int(rng.randint(0, N - block + 1))
        
        q_start = int(rng.randint(0, N - block + 1))
        while not np.all(h[q_start:q_start+block] == 0):
            q_start = int(rng.randint(0, N - block + 1))
        
        h_new = h.copy()
        h_new[p_start:p_start+block] = 0
        h_new[q_start:q_start+block] = 1
        c_new = fft_corr(h_new)
        if c_new.max() < P:
            h = h_new
        # Keep best_h update in main loop
    
    return best_h, best_c * (2.0 / N)

def nesterov_refine(h, N, deadline, k=50, lr0=0.02):
    """Nesterov accelerated projected gradient for continuous relaxation."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    L = 2 * N - 1
    center = N - 1
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
        
        # Gradient: for top-k correlations
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        y_safe = np.clip(y, 0, 1)
        # For each active lag ai, add contribution to grad
        for ai in top:
            lag = int(ai - center)
            if lag == 0:
                grad += (1.0 - 2.0 * y_safe)
            elif lag > 0:
                # c[lag] = sum_i h[i+lag]*(1-h[i])
                # d/dh[j]: from h[j] as h[i+lag] term: (1-h[j-lag])
                #          from h[j] as h[i] term: -h[j+lag]
                grad[lag:] += (1.0 - y_safe[:N-lag])
                grad[:N-lag] += (-y_safe[lag:])  # wait, this adds negative
            else:
                M = -lag
                grad[:N-M] += (1.0 - y_safe[M:])  # from h[i+lag] where i+lag=j, j>=M
                grad[M:] += (-y_safe[:N-M])  # from h[i] term
        
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.98
            beta = min(0.97, 0.85 + it * 0.0001)
        if it % 500 == 0:
            # Periodic binarization to escape
            h_bin = ensure_sum_bin(h, N, seed=it)
            if fft_corr(h_bin).max() < best_pk:
                best_pk = float(fft_corr(h_bin).max())
                best_h = h_bin.copy()
    
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
