            break

import time

def corr_full(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)

    N = len(h)
    return float(np.max(corr_full(h))) * (2.0 / N)

def contrib_vec(h, p):
    """Vector of contributions of position p to corr (assuming h[p]=1)."""
    N = len(h)
    c = np.zeros(2 * N - 1)
    s = max(0, p)
    e = min(2 * N - 2, p + N - 1)
    if s > e:
        return c
    js = N - 1 + p - np.arange(s, e + 1)
    c[s:e+1] = 1.0 - h[js]
    return c

def fast_local_search(h, N, deadline, max_iters=20000, S=40):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_P = np.inf
    stall = 0

        if time.time() > deadline:
            break
        corr = corr_full(h)
        P = float(np.max(corr))

            best_P = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 500:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        S_a = min(S, len(ones))
        S_b = min(S, len(zeros))
        a_cand = np.random.choice(ones, S_a, replace=False)
        b_cand = np.random.choice(zeros, S_b, replace=False)
        
        # Compute contributions
        CA = np.array([contrib_vec(h, a) for a in a_cand])  # (S_a, 2N-1)
        CB = np.array([contrib_vec(h, b) for b in b_cand])  # (S_b, 2N-1)

        # new_P[i,j] = max(corr + CB[j] - CA[i])

def fix_sum_binary(h, target):
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, n1 - target, replace=False)] = 0

def greedy_search(h, N, deadline, max_iters=20000, S=40, temperature=0.0):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_P = np.inf
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = corr_full(h)
        P = float(np.max(corr))
        if P < best_P - 1e-14:
            best_P = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200 and temperature == 0:
            break
        if no_improve > 500:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        Sa = min(S, len(ones))
        Sb = min(S, len(zeros))
        a_c = np.random.choice(ones, Sa, replace=False)
        b_c = np.random.choice(zeros, Sb, replace=False)
        
        # Compute contribution vectors
        # For position p, contribution to corr at index idx:
        #   valid for idx in [max(0,p), min(2N-2, p+N-1)]
        #   value = 1 - h[N-1+p-idx]

        for i, a in enumerate(a_c):

            e = min(L - 1, a + N - 1)
            js = N - 1 + a - np.arange(s, e + 1)
            CA[i, s:e+1] = 1.0 - h[js]

        for j, b in enumerate(b_c):
            s = max(0, b)
            e = min(L - 1, b + N - 1)
            js = N - 1 + b - np.arange(s, e + 1)
