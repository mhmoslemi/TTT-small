import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binary_search(h, N, deadline, max_sweeps=1000, sample_n=200):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        n_to_try = min(len(ones), sample_n)
        perm = np.random.choice(len(ones), n_to_try, replace=False)
        p_list = ones[perm]
        best_score = P + 1.0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            # Composite score: max + 0.3*2nd + 0.1*3rd
            sorted_c = np.sort(cand, axis=1)[:, ::-1]
            score = sorted_c[:, 0] + 0.3 * sorted_c[:, 1] + 0.1 * sorted_c[:, 2]
            mi = int(np.argmin(score))
            if score[mi] < best_score:
                best_score = score[mi]
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
        else:
            p, q = best_pq
        h[p] = 0.0
        h[q] = 1.0
    return h, best_c
