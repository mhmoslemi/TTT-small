import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2*N-1:
        L *= 2
    H = np.fft.fft(h, L)

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    h_bin = h_bin.copy()
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin == 1)[0]
        perm = np.random.permutation(len(idx))[:n1-target]
        h_bin[idx[perm]] = 0
    elif n1 < target:
        idx = np.where(h_bin == 0)[0]
        perm = np.random.permutation(len(idx))[:target-n1]
        h_bin[idx[perm]] = 1
    return h_bin

def lse_grad(h, corr, N, temp):
    """Gradient of log-sum-exp(corr) w.r.t. h."""
    center = N - 1
    w = np.exp((corr - corr.max()) / temp)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(N)
    w /= wsum
    grad = np.zeros(N)
    # Only process shifts with significant weight
    thresh = 1e-4 / N
    active = np.where(w > thresh)[0]
    for ai in active:
        s = int(ai - center)
        wi = w[ai]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:N-s] += wi * (1.0 - h[s:])
            grad[s:] += wi * (-h[:N-s])
        else:
            s2 = -s
            grad[s2:] += wi * (1.0 - h[:N-s2])
            grad[:N-s2] += wi * (-h[s2:])
    return grad

def continuous_opt(h, N, deadline, top_k=50):
    """Smooth gradient descent with log-sum-exp."""
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    temp = 0.05
    lr = 0.01
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        grad = lse_grad(h, corr, N, temp)
        t += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 200 == 0:
            temp *= 0.95
            lr *= 0.98
    return best_h, best_pk

    ones = list(np.where(h == 1)[0])

    while time.time() < deadline:

        q = zeros[np.random.randint(len(zeros))]
        if p == q:

        ps = p + s_arr; pm = p - s_arr
        qs = q + s_arr; qm = q - s_arr

        vqs = (qs >= 0) & (qs < N); vqm = (qm >= 0) & (qm < N)

        qsc = np.clip(qs, 0, N-1); qmc = np.clip(qm, 0, N-1)
        
        d[vps] -= (1.0 - h[pc[vps]])

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N - 1:
        L *= 2
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2 * N - 1]

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    h_bin = h_bin.copy()
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin == 1)[0]
        perm = np.random.permutation(len(idx))[:n1 - target]
        h_bin[idx[perm]] = 0
    elif n1 < target:
        idx = np.where(h_bin == 0)[0]
        perm = np.random.permutation(len(idx))[:target - n1]
        h_bin[idx[perm]] = 1
    return h_bin

def lse_grad(h, corr, N, temp):
    center = N - 1
    shifted = corr - corr.max()
    w = np.exp(shifted / temp)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(N)
    w /= wsum
    thresh = 1e-5 / N
    active = np.where(w > thresh)[0]
    grad = np.zeros(N)
    for ai in active:
        s = int(ai - center)
        wi = w[ai]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:N - s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:N - s]
        else:
            s2 = -s
            grad[s2:] += wi * (1.0 - h[:N - s2])
            grad[:N - s2] -= wi * h[s2:]
    return grad

def continuous_opt(h, N, deadline):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0

    best_h = h.copy()
    temp = 0.04
    lr = 0.008
    stall = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-9:
            best_pk = pk; best_h = h.copy(); stall = 0
        else:
            stall += 1
        if stall > 300:
            h = best_h + np.random.randn(N) * 0.02
            h = proj_sum(h, N / 2.0)
            m = np.zeros(N); v = np.zeros(N); t = 0
            temp = max(temp * 1.5, 0.03)
            stall = 0
        grad = lse_grad(h, corr, N, temp)
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = proj_sum(h - lr * mh / np.sqrt(vh + eps), N / 2.0)
        if t % 300 == 0:
            temp *= 0.92
            lr *= 0.97
    return best_h, best_pk
