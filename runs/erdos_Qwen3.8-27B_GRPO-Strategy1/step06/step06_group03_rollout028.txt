import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_signatures(h):
    N = len(h)
    L = 2 * N - 1
    lags = np.arange(L) - (N - 1)
    S = np.zeros((N, L))
    for p in range(N):
        idx1 = p + lags
        m1 = (idx1 >= 0) & (idx1 < N)
        idx2 = p - lags
        m2 = (idx2 >= 0) & (idx2 < N)
        S[p, m1] = h[idx1[m1]]
        S[p, m2] += h[idx2[m2]]
    return S

def full_local_search(h, deadline, max_perturbs=80, perturb_size=3):
    N = len(h)
    h = h.copy()
    c = fft_corr(h)
    c_max = float(c.max())
    best_c5 = c_max * (2.0 / N)
    best_h = h.copy()
    S = compute_signatures(h)
    npert = 0
    while time.time() < deadline and npert < max_perturbs:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        S_zeros = S[zeros]
        found_swap = False
        inner = 0
        while time.time() < deadline and inner < 8000:
            inner += 1
            best_nm = c_max
            best_pq = None
            for p in ones:
                nm = (c + S[p] - S_zeros).max(axis=1)
                ix = nm.argmin()
                if nm[ix] < best_nm:
                    best_nm = nm[ix]
                    best_pq = (p, zeros[ix])
            if best_pq is not None and best_nm < c_max - 1e-10:
                p, q = best_pq
                h[p] = 0.0
                h[q] = 1.0
                S = compute_signatures(h)
                S_zeros = S[zeros]
                c = fft_corr(h)
                c_max = float(c.max())
                c5v = c_max * (2.0 / N)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                S_zeros = S[zeros]
                found_swap = True
            else:
                break
        if not found_swap:
            npert += 1
            ps = perturb_size if npert < 20 else max(2, perturb_size + npert // 10)
            for _ in range(ps):
                o = np.where(h == 1)[0]
                z = np.where(h == 0)[0]

                    pi = o[np.random.randint(len(o))]
                    qi = z[np.random.randint(len(z))]
                    h[pi], h[qi] = 0.0, 1.0
            S = compute_signatures(h)
            c = fft_corr(h)
            c_max = float(c.max())
    return best_h, best_c5

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_vec(y, corr, top_k, N):
    grad = np.zeros(N)
    center = N - 1
    top = np.argpartition(corr, -top_k)[-top_k:]
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += (1.0 - y[:N - L])
            grad[:N - L] -= y[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
    grad /= len(top)
    return grad

def nesterov_opt(h, N, deadline, k=70, lr0=0.012):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.90
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = subgrad_vec(y, corr, k, N)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.99
            beta = min(0.97, 0.90 + it * 0.0001)
    return best_h, best_pk * dx

def make_binary(h, N):
    hb = (h > 0.5).astype(float)
    t = N // 2
    n1 = int(hb.sum())
    if n1 > t:
        idx = np.where(hb == 1)[0]
        idx = idx[np.argsort(np.random.random(len(idx)))][t:]
        hb[idx] = 0
    elif n1 < t:
        idx = np.where(hb == 0)[0]
        idx = idx[np.argsort(np.random.random(len(idx)))][:t - n1]
        hb[idx] = 1
    return hb
