    best_c5 = compute_c5(h, N)
    best_h = h.copy()

    beta = 0.9

        c5 = float(corr.max()) * 2/N
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find top shifts
        top_k = np.argpartition(corr, -n_top)[-n_top:]
        # Average gradient of top shifts
        grad = np.zeros(N)
        for k in top_k:
            s = int(k) - (N-1)
            if s >= 0:
                j_range = np.arange(N - s)
                grad[j_range] += (1 - h[j_range + s])
                if s > 0:
                    grad[s:] -= h[:-s]
            else:
                m = -s
                grad[m:] += (1 - h[:-m])
                grad[:N-m] -= h[m:]
        grad /= n_top
        
        # Gradient step with Nesterov momentum
        h_new = h + vel - lr_current * grad
        h_new = project_sum(h_new, N/2)
        vel = beta * (h_new - h)
        h = h_new
        
        lr_current *= 0.999
        if lr_current < 0.001:
            lr_current = 0.001
            beta = min(0.99, beta + 0.001)
    
    return best_h, best_c5

import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def fix_binary(h, nh):
    h = np.array(h, dtype=float)
    cur = int(round(h.sum()))
    if cur > nh:
        ones = np.where(h > 0.5)[0]
        h[np.random.choice(ones, cur - nh, replace=False)] = 0.0
    elif cur < nh:
        zeros = np.where(h < 0.5)[0]
        h[np.random.choice(zeros, nh - cur, replace=False)] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

                p = int(np.random.choice(ones))
                q = int(np.random.choice(zeros))
                h[p], h[q] = 0.0, 1.0
                corr += deltas[p] - deltas[q]
                sgn[p], sgn[q] = -sgn[p], -sgn[q]
                deltas[p, :] = 0.0; deltas[p, p:p + N] = sgn[p]
                deltas[q, :] = 0.0; deltas[q, q:q + N] = sgn[q]
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]

            T = max(T * 2.0, 0.1)

def cont_opt(h_init, N, deadline, lr0=0.015, n_top=10):
    h = project_sum(np.array(h_init, dtype=float).copy(), N / 2.0)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    vel = np.zeros(N)
    lr, beta, it = lr0, 0.85, 0

        c5v = float(corr.max()) * 2.0 / N

            best_h = h.copy()
        top = np.argpartition(corr, -n_top)[-n_top:]

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(fft_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def fix_binary(h, nh):
    h = np.array(h, dtype=float)
    cur = int(round(h.sum()))
    if cur > nh:
        idx = np.random.choice(np.where(h > 0.5)[0], cur - nh, replace=False)
        h[idx] = 0.0
    elif cur < nh:
        idx = np.random.choice(np.where(h < 0.5)[0], nh - cur, replace=False)
        h[idx] = 1.0
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def binary_sa(h_in, N, deadline, T0=0.4, cool=0.9998, stall_max=5000):
    h = fix_binary(h_in.copy(), N // 2)
    corr = fft_corr(h).copy()
    sg = 2.0 * h - 1.0
    D = np.zeros((N, 2 * N - 1))
    for i in range(N):
        D[i, i:i + N] = sg[i]
    bc = float(corr.max())
    bh = h.copy()
    cc = bc
    T = T0
    st = 0
    while time.time() < deadline:
        if st > stall_max:
            ow = np.where(h == 1)[0]
            zw = np.where(h == 0)[0]
            nk = min(4, len(ow), len(zw))
            for _ in range(nk):
                p = int(np.random.choice(ow))
                q = int(np.random.choice(zw))
                h[p], h[q] = 0.0, 1.0
                corr += D[p] - D[q]
                sg[p], sg[q] = -sg[p], -sg[q]
                D[p, :] = 0.0; D[p, p:p + N] = sg[p]
                D[q, :] = 0.0; D[q, q:q + N] = sg[q]
                ow = np.where(h == 1)[0]
                zw = np.where(h == 0)[0]
            cc = float(corr.max())
            T = max(T * 2.0, 0.1)

        ow = np.where(h == 1)[0]

        q = int(zw[np.random.randint(len(zw))])
        dc = D[p] - D[q]
        nc = corr + dc
        mc = float(nc.max())
        dl = mc - cc
        if dl < 0 or np.random.random() < np.exp(-dl / max(T, 1e-14)):
            h[p], h[q] = 0.0, 1.0
            corr = nc
            sg[p], sg[q] = -sg[p], -sg[q]
            D[p, :] = 0.0; D[p, p:p + N] = sg[p]
            D[q, :] = 0.0; D[q, q:q + N] = sg[q]

def cont_opt(hi, N, dl, lr0=0.015, kt=10):

    bc5 = compute_c5(h)
