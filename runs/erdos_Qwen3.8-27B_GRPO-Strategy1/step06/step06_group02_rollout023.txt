import numpy as np

def fft_corr(h):
    N = len(h)
    L = 2 * N
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

def build_eff(h, N):
    """Build efficiency vectors for all 1-positions and 0-positions."""
    M = 2 * N - 1
    j = np.arange(M)
    k = j - (N - 1)
    pos = np.arange(N)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    # eff0[p, j] = -[(1-h[p-k_j])*valid(p-k_j)] + [h[p+k_j]*valid(p+k_j)]
    # Compute for all positions at once:
    # p-k_j ranges over... for each p, p-k ranges from p-(N-2) to p-(-(N-2)) = p+(N-2)
    # This is a Toeplitz-like structure.

    # D[p, j] = (1-h[p-k_j])*[0≤p-k_j<N] - h[p+k_j]*[0≤p+k_j<N]
    #         = d1[p,j] - d2[p,j]

    #       d2[p,j] = h[max(0,min(N-1,p+k_j))] * [0≤p+k_j<N]
    
    pm = pos[:, None] - k[None, :]  # N x M
    pp = pos[:, None] + k[None, :]  # N x M
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    
    one_minus_h = 1.0 - h
    D = one_minus_h[pm_c] * vm.astype(float) - h[pp_c] * vp.astype(float)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))[:2*N-1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    target = float(target)
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def build_D(h, N):
    M = 2 * N - 1
    k = np.arange(M) - (N - 1)
    pos = np.arange(N)[:, None]
    pm = pos - k[None, :]
    pp = pos + k[None, :]
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    D = (1.0 - h[pm_c]) * vm.astype(float) - h[pp_c] * vp.astype(float)
    return D

def sa_binary(h0, N, deadline, T0=0.10, Tend=0.001, samples=200, max_steps=4000):
    h = (h0 > 0.5).astype(float)
    target = N // 2
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        rm = np.random.choice(len(idx), n1 - target, replace=False)
        h[idx[rm]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        add = np.random.choice(len(idx), target - n1, replace=False)
        h[idx[add]] = 1
    T = T0
    cool = (Tend / T0) ** (1.0 / max_steps)
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    no_imp = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        T = max(Tend, T * cool)
        D = build_D(h, N)
        corr = fft_corr(h)
        pk = float(corr.max())
        if pk < best_pk - 1e-12:
            best_pk = pk
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 800:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_s = min(samples, len(ones), len(zeros))
        pi = np.random.randint(len(ones), n_s)
        qi = np.random.randint(len(zeros), n_s)
        eff0 = -D[ones[pi], :]
        eff1 = D[zeros[qi], :]
        new_maxes = np.max(corr[None, :] + eff0 + eff1, axis=1)
        bi = int(np.argmin(new_maxes))
        best_score = float(new_maxes[bi])
        p = ones[pi[bi]]
        q = zeros[qi[bi]]
        delta = best_score - pk
        if delta >= 0 or np.random.random() < np.exp(-max(delta, -50) / max(T, 1e-6)):
            h[p] = 0.0
            h[q] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0.0
            h[q] = 1.0

    v = np.zeros(N)
    t = 0
    center = N - 1
    beta_sm = 40.0
    b1, b2 = 0.9, 0.999
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(corr.max()) * dx
        if pk < best_pk - 1e-9:

            best_h = h.copy()

        corr_top = corr[top_idx].copy()
        corr_top -= corr_top.max()
        w = np.exp(corr_top * beta_sm)
        ws = w.sum()
