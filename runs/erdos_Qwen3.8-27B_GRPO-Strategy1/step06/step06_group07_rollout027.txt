    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    return np.real(np.fft.ifft(H * G))

This is related to "perfect sequences" in the theory of orthogonal frequency division multiplexing. For a binary sequence of length N with exactly N/2 ones, the autocorrelation at lag 0 is N/2. At other lags, we want it to be as close to N/8 (the average) as possible.

This gives us a clear optimization target: maximize min_{k≠0} autocorr_h[k], which is equivalent to minimizing max_{k≠0} (N/2 - autocorr_h[k]) = max_{k≠0} corr[k].

(Note: at k=0, corr[0] = Σ h[i](1-h[i]) = 0 for binary h. So the max is over k≠0.)

This reframing might help with the optimization. The autocorrelation of a binary sequence is well-studied. Sequences with good autocorrelation properties include:
- m-sequences (but these have (N-1)/2 or (N+1)/2 ones, not N/2)
- Barker sequences (short, specific)
- CAZAC sequences (complex)

For our binary case with exactly N/2 ones, this is more like a combinatorial design problem.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_binary(h, N, seed=0):
    nh = N // 2
    h = np.clip(h, 0, 1).astype(float)
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(seed)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def adam_opt(h, N, deadline, top_frac=0.15, lr=0.015, beta1=0.8, beta2=0.999):
    h = h.astype(np.float64).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    eps = 1e-8
    lr_cur = lr
    center = N - 1
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        val = float(np.max(corr))
        if val < best_val - 1e-14:
            best_val = val
            best_h = h.copy()
        n_active = max(20, int(len(corr) * top_frac))
        active = np.argpartition(corr, -n_active)[-n_active:]
        grad = np.zeros(N)
        for ai in active:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += (1.0 - h[:N-L])
                grad[:N-L] -= h[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - h[M:])
                grad[M:] -= h[:N-M]
        grad /= float(len(active))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        h_new = h - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h_new, target)
        if t % 2000 == 0:
            lr_cur *= 0.9
    return best_h, best_val * dx
