            hi = mid
        else:
            lo = mid

    h_sorted_idx = np.argsort(-h)  # highest first
    h_bin = np.zeros(n, dtype=np.int8)
    h_bin[h_sorted_idx[:n//2]] = 1
    return h_bin

    return np.correlate(h_bin, (1-h_bin).astype(np.int32), mode='full').astype(np.int32)

    h = h_bin.copy()
    corr = corr_int(h)
    cur_max = int(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()

        else:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    vec = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    if abs(vec.sum() - target) < 1e-12:
        return vec
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(-np.asarray(h, dtype=float))
    b = np.zeros(n, dtype=np.int32)
    b[idx[:n // 2]] = 1
    return b

def random_binary(n, rng):
    b = np.zeros(n, dtype=np.int32)
    idx = rng.choice(n, size=n // 2, replace=False)
    b[idx] = 1
    return b

def corr_of(h):
    return np.correlate(h, (1 - h).astype(np.int32), mode='full').astype(np.int32)

def binary_search(h_init, n, t_end, rng):
    h = h_init.copy()
    corr = corr_of(h)
    cur_max = int(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    stuck_count = 0
    while time.time() < t_end:
        improved = False
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:

            for j in zeros:
                if i == j:
                    continue
                h[i], h[j] = 0, 1
                mx = int(np.max(corr_of(h)))
                if mx < cur_max:
                    cur_max = mx
                    corr = corr_of(h)
                    ones.remove(i); ones.append(j)
                    zeros.remove(j); zeros.append(i)
                    if cur_max < best_max:

                        best_h = h.copy()
                    improved = True

        if improved:
            stuck_count = 0
        else:

            kick_sz = min(stuck_count // 5, 5) + 1
