import time

    return np.correlate(h, 1.0-h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0/n)

def smooth_c5(h, T):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0/n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T*(corr-m))))/T) * dx

    n = len(h)
    dx = 2.0/n
    corr = compute_corr(h)
    m = np.max(corr)
    log_lam = T*(corr-m) - np.log(np.sum(np.exp(T*(corr-m))))
    lam = np.exp(log_lam)
    grad = np.zeros(n)

        w = lam[k_idx] * dx
        if w < 1e-16: continue
        k = k_idx - (n-1)
        if k >= 0:
            for i in range(k, n):
                grad[i] += w * (1.0 - h[i-k])
            for i in range(0, n-k):
                grad[i] += w * (-h[i+k])
        else:

    dx = 2.0 / n

    exps = np.exp(T * (corr - m))
    S = exps.sum()

    obj = (m + np.log(S) / T) * dx
    grad = np.zeros(n)

            grad[mv:] += wi * (-h[:n - mv])

def project(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    orig = h.copy()
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(orig + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    np.clip(orig + (lo + hi) * 0.5, 0.0, 1.0, out=h)
    return h

        return smooth_obj_grad(x, T)[0]

        return smooth_obj_grad(x, T)[1]

                    return np.max(corr)
                mx = np.max(corr)
                for k in range(n):
                    s = 0.0

                        s += d * (1.0 - h[i - k])
                    if k < n - i:
                        s += d * (-h[i + k])
                    v = corr[n - 1 + k] + s

                    if i - m >= 0:

                return mx
