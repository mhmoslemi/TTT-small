                corr_new[n-1-m] += t * d_neg[m]

    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})

            break
        scale = 0.15 + 0.1 * r
        pert = best_h + scale * rng.standard_normal(n)
        pert = project(pert, n)
        for T in [100, 500, 2000]:
            if time.time() > t_phase:
                break
            try:
                hc = slsqp_opt(pert, T, n, maxiter=400)
            except Exception:
                hc = pert
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
                pert = hc.copy()
        hc, cv = proj_grad_descent(pert, n, 2000, t_phase, max_steps=1000)
        if cv < best_c5:
            best_h, best_c5 = hc, cv
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish
    t1 = t0 + budget_s * 0.12
    for T in [50, 300, 1500, 8000]:
        if time.time() > t1:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=800, ftol=1e-15)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 2: Projected gradient descent
    t2 = t0 + budget_s * 0.25
    best_h, best_c5 = proj_grad_descent(best_h, n, 300, t2, max_steps=2000)
    best_h, best_c5 = proj_grad_descent(best_h, n, 2000, t2, max_steps=2000)

    # Phase 3: Priority coordinate descent
    t3 = t0 + budget_s * 0.45
    best_h, best_c5 = priority_coord_descent(best_h, n, t3, T=500, max_sweeps=60)

    # Phase 4: Block swap exploration
    t4 = t0 + budget_s * 0.58
    best_h, best_c5 = block_swap_descent(best_h, n, t4, rng, n_iters=3000)

    # Phase 5: More coord descent after swaps
    t5 = t0 + budget_s * 0.68
    best_h, best_c5 = priority_coord_descent(best_h, n, t5, T=2000, max_sweeps=40)

    # Phase 6: Multi-restart
    t6 = t0 + budget_s * 0.85
    best_h, best_c5 = multi_restart(best_h, n, t6, rng, n_restarts=3)

    # Phase 7: Multi-resolution
    t7 = t0 + budget_s * 0.93
    for n_try in [160, 200, 280, 320]:
        if time.time() > t7:
            break
        h_interp = interpolate_h(best_h, n_try)
        try:
            hc = slsqp_opt(h_interp, 800, n_try, maxiter=600)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5, n = hc, cv, n_try
                h_interp = interpolate_h(best_h, n_try)
        except Exception:
            pass
        if n == n_try:
            best_h, best_c5 = proj_grad_descent(best_h, n, 1000, t7, max_steps=1500)
            best_h, best_c5 = priority_coord_descent(best_h, n, t7, T=1000, max_sweeps=20)

    # Phase 8: Final polish
    t8 = t0 + budget_s * 0.98
    best_h, best_c5 = priority_coord_descent(best_h, n, t8, T=5000, max_sweeps=30)
    try:
        hc = slsqp_opt(best_h, 20000, n, maxiter=1500, ftol=1e-15)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv
    except Exception:
        pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
