import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    vec = np.array(h, dtype=float)
    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -np.max(vec) - 1.0
    hi = 1.0 - np.min(vec) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for i in range(n):
        # Part 1: sum_s w[s+(n-1)] * (1-h[i+s]) for valid 0<=i+s<n
        s_lo = max(-i, -(n-1))
        s_hi = min(n-1-i, n-1)
        if s_lo <= s_hi:
            s_range = np.arange(s_lo, s_hi+1)
            grad[i] += np.dot(w[s_range+(n-1)], one_minus_h[i+s_range])
        # Part 2: sum_s w[s+(n-1)] * h[i-s] for valid 0<=i-s<n
        s_lo2 = max(i-(n-1), -(n-1))
        s_hi2 = min(i, n-1)
        if s_lo2 <= s_hi2:
            s_range2 = np.arange(s_lo2, s_hi2+1)
            grad[i] -= np.dot(w[s_range2+(n-1)], h[i-s_range2])
    grad *= dx
    return grad

def cd_sweep(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(50000):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            # Optimize h[i] in [0,1]
            def f1d(hi):
                h_try = best_h.copy()
                h_try[i] = hi
                corr = np.correlate(h_try, 1.0 - h_try, mode='full')
                return np.max(corr) * (2.0/n)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            if abs(res.x - best_h[i]) > 1e-10:
                best_h[i] = res.x
                c5_try = compute_c5(best_h)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def fw_step(h, T, n, gamma):
    dx = 2.0/n
    grad = smooth_grad(h, T)
    # Find extreme point: n/2 entries with smallest gradient
    idx = np.argsort(grad)
    v = np.zeros(n)
    v[idx[:n//2]] = 1.0
    h_new = h + gamma * (v - h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project(h_new, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals() and initial_h_values is not None:

        best_h = np.zeros(160)

    n = len(best_h)

        try:

            hc = project(res.x, n)
            c5v = compute_c5(hc)
            if c5v < best_c5:

        except Exception:

    # Phase 2: Intensive CD (40% budget)

    # Phase 3: FW + CD alternation (25% budget)

**Speed optimization for CD:**

The key insight: when we change h[i] from old to new, the correlation changes in a predictable way. Specifically:
- Δcorr_k = (new - old) * [(1-h[i+s_k])*valid - h[i-s_k]*valid] for each shift s_k

So instead of recomputing the full correlation, we can update it incrementally:
- Store the current corr array
- When h[i] changes by δ: corr += δ * g_i where g_i is the "influence vector" for coordinate i

This makes each CD step O(n) instead of O(n^2)!

            grad[i] += np.dot(w[sr + (n - 1)], omh[i + sr])
