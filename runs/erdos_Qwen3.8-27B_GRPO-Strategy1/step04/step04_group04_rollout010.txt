```python
def grad_smooth(corr, h, T):
    n = len(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        pk = p[k]
        lag = k - (n - 1)
        # (1-h[i-lag]) for 0<=i-lag<n
        i_min = max(0, lag)  # i-lag >= 0
        i_max = min(n, n + lag)  # i-lag < n
        if i_min < i_max:
            # i ranges from i_min to i_max-1
            # 1-h[i-lag]: index i-lag ranges from i_min-lag to i_max-1-lag
            grad[i_min:i_max] += pk * (1.0 - h[i_min-lag:i_max-lag])
        # -h[i+lag] for 0<=i+lag<n
        j_min = max(0, -lag)  # i+lag >= 0
        j_max = min(n, n - lag)  # i+lag < n
        if j_min < j_max:
            grad[j_min:j_max] -= pk * h[j_min+lag:j_max+lag]
    return grad
```

This is O(n * num_active) but with numpy vectorized slices for each active k. For n=180 and all 359 k's active, that's 359 iterations of small numpy operations - should be fast enough.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += room * (diff / tot)
        else:
            if s < 1e-15:
                break
            h *= (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)

   active = np.where(p > 1e-13)[0]
