```python
            # So coeff[c0] = 1-2*old is correct for linear part
            # Quadratic correction: -(v-old)² added at d=0
            
            c0_base = corr[c0]
            
            def obj(v, coeff=coeff, c0_base=c0_base, old=old):
                dv = v - old
                cn = corr + dv * coeff
                cn[c0] = c0_base + dv * (1.0 - v - old)
                return np.max(cn)
            
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            best_v = res.x
            best_f = res.fun
            # Check boundaries
            f0 = obj(0.0)
            f1 = obj(1.0)
            if f0 < best_f:
                best_v, best_f = 0.0, f0
            if f1 < best_f:
                best_v, best_f = 1.0, f1
            
            if best_f * dx < best_c5 - tol:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr += (best_v - old) * coeff
                corr[c0] = c0_base + (best_v - old) * (1.0 - best_v - old)
                improved = True
                max_improve = max(max_improve, best_c5 - best_f * dx)
        if not improved:
            break
    return best_h, best_c5
```

Wait, there's an issue with the incremental corr update. When I do `corr += (best_v - old) * coeff`, I'm using the OLD best_h in coeff. - Before: corr_old = correlate(best_h, 1-best_h, full)
- After changing h[i] from old to v:

Let me also add a **swap-based local search** for 0-1 solutions:

For 0-1 valued h, a swap means: pick i with h[i]=1 and j with h[j]=0, set h[i]=0, h[j]=1. The change in corr:
- At position i: h goes 1→0, so contributions involving h[i] as first factor change
- At position j: h goes 0→1, so contributions involving h[j] as first factor appear, and (1-h[j]) as second factor disappears

Actually for 0-1 valued h, let's think in terms of the set S = {i : h[i]=1}.
corr[c0+d] = |{i ∈ S : i+d ∈ S^c, i+d ∈ [0,n)}| + |{i ∉ S : i-d ∈ S, i-d ∈ [0,n)}| 

Wait no. corr[c0+d] = sum_i h[i]*(1-h[i+d]) where i+d must be in [0,n).
For 0-1 h: = |{i ∈ S : i+d ∈ [0,n), i+d ∉ S}|

When we swap i (in S) and j (in S^c):
- S' = S - {i} + {j}
- For corr[c0+d]:
  - i was contributing 1 if i+d ∈ [0,n) and i+d ∉ S. Now 0.
  - j is contributing 1 if j+d ∈ [0,n) and j+d ∉ S'. 
    - j+d ∉ S' means j+d ∉ S or j+d = i (since i was removed)
    - Actually j+d ∉ S' = (j+d ∉ S) or (j+d = i)
  - Also, for the term where i appears in the (1-h[.]) factor:
    - corr[c0+d] includes h[m]*(1-h[m+d]) for all m.
    - If m+d = i, i.e., m = i-d: h[i-d]*(1-h[i]) was 1 if i-d ∈ S and i ∉ S... but i ∈ S, so this was 0.
    - Now h[i-d]*(1-h[i]): i is no longer in S, so (1-h[i])=1. This contributes h[i-d] = 1 if i-d ∈ S'.
    - Similarly for j: if m+d = j, i.e., m = j-d: h[j-d]*(1-h[j]) was 0 (since j ∉ S, (1-h[j])=1, but h[j-d] needs to be 1).
    - Wait, j ∉ S means h[j]=0, so (1-h[j])=1. The term h[j-d]*(1-h[j]) = h[j-d] if j-d ∈ [0,n).
    - This was already contributing (since j was 0 before, (1-h[j])=1).
    - After swap, j ∈ S, so (1-h[j])=0. This term disappears.

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=1000):
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    c0 = n - 1
    ks = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx_fwd = i + ks
            idx_bwd = i - ks
            vf = (idx_fwd >= 0) & (idx_fwd < n)
            vb = (idx_bwd >= 0) & (idx_bwd < n)
            coeff = np.zeros(2*n-1)
            coeff[vf] += 1.0 - best_h[idx_fwd[vf]]
            coeff[vb] += -best_h[idx_bwd[vb]]
            c0b = corr[c0]
            
            def obj(v):
                dv = v - old
                cn = corr + dv * coeff
                cn[c0] = c0b + dv * (1.0 - v - old)
                return np.max(cn)
            
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            bv, bf = res.x, res.fun
            f0, f1 = obj(0.0), obj(1.0)
            if f0 < bf: bv, bf = 0.0, f0
            if f1 < bf: bv, bf = 1.0, f1
            
            if bf * dx < best_c5 - 1e-14:
                dv = bv - old
                best_h[i] = bv
                corr += dv * coeff
                corr[c0] = c0b + dv * (1.0 - bv - old)
                best_c5 = bf * dx
                improved = True
        if not improved:
            break
    return best_h, best_c5

def swap_search(h, n, t_end, max_sweeps=200):
    """0-1 swap local search."""
    h = np.where(h >= 0.5, 1.0, 0.0).copy()
    # Ensure exactly n/2 ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    while len(ones) > n//2:
        j = ones.pop(np.random.randint(len(ones)))
        h[j] = 0.0
        zeros = np.append(zeros, j)
    while len(ones) < n//2:
        j = zeros.pop(np.random.randint(len(zeros)))
        h[j] = 1.0
        ones = np.append(ones, j)
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_swap = None
        best_val = best_c5
        # Try all swaps (limited for speed)
        n_ones, n_zeros = len(ones), len(zeros)
        # Sample if too many
        max_tries = 2000
        if n_ones * n_zeros > max_tries:
            idx = np.random.randint(0, n_ones * n_zeros, max_tries)
            i_list = idx // n_zeros
            j_list = idx % n_zeros
        else:
            i_list = np.arange(n_ones)
            j_list = np.repeat(np.arange(n_zeros), n_ones)
            # Actually this gives n_ones*n_zeros entries, let me fix
            I, J = np.meshgrid(np.arange(n_ones), np.arange(n_zeros), indexing='ij')
            i_list, j_list = I.flatten(), J.flatten()
        
        for ii, jj in zip(i_list, j_list):
            i, j = ones[ii], zeros[jj]
            # Swap
            h[i], h[j] = 0.0, 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_val - 1e-14:
                best_val = new_max
                best_swap = (i, j)
                corr = new_corr
            else:
                h[i], h[j] = 1.0, 0.0
        
        if best_swap is not None:
            i, j = best_swap
            ones = np.delete(ones, np.where(ones == i)[0][0])
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            best_c5 = best_val
        else:
            break
    return h, best_c5

def basin_hop(h, c5, n, t_end, max_iters=100):
    best_h, best_c5 = h.copy(), c5
    work_h, work_c5 = h.copy(), c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.1, 1.0 - 0.9 * it / max_iters)
        step = 0.05 + 0.2 * cool
        h_p = work_h + step * np.random.randn(n)
        h_p = project(h_p, n)
        h_c, c5_c = coord_descent(h_p, n, min(time.time() + 5, t_end), max_sweeps=50)
        temp = 0.001 * cool + 0.0001
        if c5_c < work_c5 or np.random.rand() < np.exp(-(c5_c - work_c5) / temp):
            work_h, work_c5 = h_c, c5_c
            if c5_c < best_c5:
                best_c5, best_h = c5_c, h_c.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) > 0:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n = 160
        best_h = np.zeros(n)
        best_h[::2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)[0]
    
    # Phase 1: CD
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.15, max_sweeps=2000)
    
    # Phase 2: 0-1 projection + swap search
    h01 = np.where(best_h >= 0.5, 1.0, 0.0)
    h01, c5_01 = swap_search(h01, n, t0 + budget_s * 0.25, max_sweeps=100)
    if c5_01 < best_c5:
        best_c5, best_h = c5_01, h01
    
    # Phase 3: More CD on the 0-1 solution (relaxing)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.35, max_sweeps=1000)
    
    # Phase 4: Basin hopping
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.60, max_iters=80)
    
    # Phase 5: Final CD
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.72, max_sweeps=2000)
    
    # Phase 6: Try 0-1 again
    h01 = np.where(best_h >= 0.5, 1.0, 0.0)
    h01, c5_01 = swap_search(h01, n, t0 + budget_s * 0.78, max_sweeps=100)
    if c5_01 < best_c5:
        best_c5, best_h = c5_01, h01
    
    # Phase 7: Multi-resolution
    for n_try in [n*2, n//2 + 1, n + 20]:
        if elapsed() > budget_s * 0.88:
            break
        if n_try < 40 or n_try > 600:
            continue
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, min(time.time() + budget_s*0.02, t0 + budget_s*0.90), max_sweeps=500)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try
    
    # Phase 8: Final polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=1000)
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)[0]
    
    if elapsed() > budget_s * 0.95:
        return best_h, best_c5, len(best_h)
    
    # Phase 9: One more basin hop if time
    best_h, best_c5 = basin_hop(best_h, best_c5, len(best_h), t0 + budget_s * 0.98, max_iters=30)
    best_h, best_c5 = coord_descent(best_h, len(best_h), t0 + budget_s * 0.995, max_sweeps=500)
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)[0]
    
    return best_h, best_c5, len(best_h)
```

Wait, there's a problem with my swap_search function. The `np.correlate` call for each swap is O(n²), and with 6400 swaps that's O(n² * n_ones * n_zeros) which is too slow.

Let me optimize: for the swap, I can compute the change in correlation incrementally.

When h[i] changes from 1 to 0 and h[j] changes from 0 to 1:
- For corr at lag offset d:
  - Position i in first factor: was 1*(1-h[i+d]), now 0. Change: -(1-h[i+d])
  - Position j in first factor: was 0*(1-h[j+d]), now 1. Change: +(1-h[j+d])
  - Position i in second factor (1-h[.]): was h[m]*(1-1)=0 for m where m+d=i. Now h[m]*(1-0)=h[m]. Change: +h[m] where m=i-d
  - Position j in second factor: was h[m]*(1-0)=h[m] for m where m+d=j. Now h[m]*(1-1)=0. Change: -h[m] where m=j-d

So the total change at lag d:
Δ(d) = -(1-h[i+d]) + (1-h[j+d]) + h[i-d] - h[j-d]
(where we only include terms where the indices are in [0,n))

For d=0: Δ(0) = -(1-h[i]) + (1-h[j]) + h[i] - h[j] = -1+h[i]+1-h[j]+h[i]-h[j] = 2h[i]-2h[j] = 2-2=0 (since h[i]=1, h[j]=0)
Wait that's 0? Let me recheck.
At d=0: corr[c0] = sum_m h[m]*(1-h[m])
Change: position i: was 1*0=0, now 0*1=0. No change.
Position j: was 0*1=0, now 1*0=0. No change.
So at d=0, no change! That makes sense for 0-1 valued h.

For d≠0:

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=1000):
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    c0 = n - 1
    d_vals = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            fwd = i + d_vals
            bwd = i - d_vals
            vf = (fwd >= 0) & (fwd < n)
            vb = (bwd >= 0) & (bwd < n)
            coeff = np.zeros(2 * n - 1)
            coeff[vf] += 1.0 - best_h[fwd[vf]]
            coeff[vb] += -best_h[bwd[vb]]
            c0b = corr[c0]
            def obj(v):
                dv = v - old
                cn = corr + dv * coeff
                cn[c0] = c0b + dv * (1.0 - v - old)
                return np.max(cn)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            bv, bf = res.x, res.fun
            f0, f1 = obj(0.0), obj(1.0)
            if f0 < bf:
                bv, bf = 0.0, f0
            if f1 < bf:
                bv, bf = 1.0, f1
            if bf * dx < best_c5 - 1e-14:
                dv = bv - old
                best_h[i] = bv
                corr += dv * coeff
                corr[c0] = c0b + dv * (1.0 - bv - old)
                best_c5 = bf * dx
                improved = True
        if not improved:
            break
    return best_h, best_c5

def swap_search(h, n, t_end, max_sweeps=100):
    h = np.where(h >= 0.5, 1.0, 0.0).copy()
    s = int(h.sum())
    target = n // 2
    if s > target:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[:s - target]] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[:target - s]] = 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = min(1000, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            nc = np.correlate(h, 1.0 - h, mode='full')
            nv = np.max(nc) * dx
            if nv < best_c5 - 1e-14:
                best_c5 = nv
                corr = nc
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, best_c5

def make_initial(n, variant):
    h = np.zeros(n)
    if variant == 0:
        h[::2] = 1.0
    elif variant == 1:
        h[:n // 2] = 1.0
    elif variant == 2:
        pat = np.array([1, 1, 0, 1, 0, 0])
        h[:] = np.tile(pat, n // len(pat))[:n]
    elif variant == 3:
        h[:] = 0.5
        h[np.random.permutation(n)[:n // 2]] = 1.0
    else:
        h[:] = 0.5
        for k in range(1, n):
            h[k] = 0.5 + 0.5 * np.sin(2 * np.pi * k * 7.0 / n)
        h = np.where(h > 0.5, 1.0, 0.0)
    return project(h, n)

def basin_hop(h, c5, n, t_end, max_iters=80):
    best_h, best_c5 = h.copy(), c5
    work_h, work_c5 = h.copy(), c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.05, 1.0 - 0.85 * it / max_iters)
        step = 0.03 + 0.20 * cool
        h_p = work_h + step * np.random.randn(n)
        h_p = project(h_p, n)
        h_c, c5_c = coord_descent(h_p, n, min(time.time() + 8.0, t_end), max_sweeps=40)
        h_c, c5_c2 = swap_search(h_c, n, min(time.time() + 3.0, t_end), max_sweeps=20)
        if c5_c2 < c5_c:
            h_c, c5_c = h_c, c5_c2
        temp = 0.0015 * cool + 0.0001
        if c5_c < work_c5 or np.random.rand() < np.exp(-(c5_c - work_c5) / temp):
            work_h, work_c5 = h_c.copy(), c5_c
            if c5_c < best_c5:
                best_c5, best_h = c5_c, h_c.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) > 0:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_initial(160, 0)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)[0]

    # Phase 1: Aggressive CD
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.12, max_sweeps=3000)

    # Phase 2: 0-1 swap refinement
    h01, c5_01 = swap_search(best_h, n, t0 + budget_s * 0.20, max_sweeps=60)
    if c5_01 < best_c5:
        best_c5, best_h = c5_01, h01

    # Phase 3: CD on refined solution
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.32, max_sweeps=2000)

    # Phase 4: Multi-start basin hopping
    for var in range(5):
        if elapsed() > budget_s * 0.55:
            break
        h_init = make_initial(n, var) if var > 0 else best_h.copy()
        h_init, c5_init = coord_descent(h_init, n, min(time.time() + 3.0, t0 + budget_s * 0.55), max_sweeps=80)
        h_res, c5_res = basin_hop(h_init, c5_init, n,
                                  min(time.time() + budget_s * 0.04, t0 + budget_s * 0.55), max_iters=25)
        if c5_res < best_c5:
            best_c5, best_h = c5_res, h_res

    # Phase 5: Polish with CD + swap
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.62, max_sweeps=2000)
    h01, c5_01 = swap_search(best_h, n, t0 + budget_s * 0.68, max_sweeps=80)
    if c5_01 < best_c5:
        best_c5, best_h = c5_01, h01
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.72, max_sweeps=1000)

    # Phase 6: Multi-resolution search
    for n_try in [n, 120, 200, 240, 320]:
        if elapsed() > budget_s * 0.82:
            break
        if n_try == n:
            h_tr = best_h.copy()
        else:
            h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, min(time.time() + 15.0, t0 + budget_s * 0.85), max_sweeps=500)
        h01t, c5_t = swap_search(h_tr, n_try, min(time.time() + 5.0, t0 + budget_s * 0.86), max_sweeps=30)
        if c5_t < c5_tr:
            h_tr, c5_tr = h01t, c5_t
        if c5_tr < best_c5:
            best_c5, best_h = c5_tr, h_tr

    # Phase 7: Final basin hop on best
    best_h, best_c5 = basin_hop(best_h, best_c5, len(best_h),
                                min(time.time() + budget_s * 0.08, t0 + budget_s * 0.93), max_iters=40)

    # Phase 8: Final polish
    fn = len(best_h)
    best_h, best_c5 = coord_descent(best_h, fn, t0 + budget_s * 0.97, max_sweeps=3000)
    h01, c5_01 = swap_search(best_h, fn, t0 + budget_s * 0.99, max_sweeps=30)
    if c5_01 < best_c5:
        best_c5, best_h = c5_01, h01
        best_h, best_c5 = coord_descent(best_h, fn, t0 + budget_s * 0.995, max_sweeps=500)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)[0]
    return best_h, best_c5, len(best_h)
```
