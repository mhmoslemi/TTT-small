```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T, corr=None):
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))

    target = n / 2.0

    if abs(diff) < 1e-13:

    if diff > 0:

        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:

        h *= (1.0 + diff / s)
```

```python
def coord_opt_fast(h, n, i, corr):
    """Optimize h[i] given all others, using precomputed corr."""
    v = h[i]
    dx = 2.0 / n
    
    # Compute coefficients A[k] for linear part
    # corr_full[k] is at index k + (n-1) in the np.correlate output
    # For lag k, delta = (t-v) * A[k] (linear) + (if k=0: quadratic)
    A = np.zeros(2*n-1)
    n1 = n - 1
    for k in range(-(n-1), n):
        ak = 0.0
        j1 = i + k  # "first" position partner
        j2 = i - k  # "second" position partner  
        if k == 0:
            pass  # handled specially
        else:
            if 0 <= j1 < n:
                ak += (1.0 - h[j1])
            if 0 <= j2 < n and j2 != i:
                ak -= h[j2]
        A[k + n1] = ak
    
    # Now minimize max_k (corr[k+n1] + (t-v)*A[k+n1] + quad_term)
    # where quad_term = -(t-v)*(t+v) if k=0, else 0
    
    k0 = n1  # index of lag 0 in the corr array
    def f1d(tt):
        dv = tt - v
        new_corr = corr + dv * A
        new_corr[k0] = corr[k0] + dv * (1.0 - tt - v)
        return float(np.max(new_corr)) * dx
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i, corr):
    v = h[i]
    n1 = n - 1
    k0 = n1
    A = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        if k == 0:
            continue
        ak = 0.0
        j1 = i + k
        j2 = i - k
        if 0 <= j1 < n:
            ak += (1.0 - h[j1])
        if 0 <= j2 < n:
            ak -= h[j2]
        A[k + n1] = ak
    c0 = corr[k0]
    def f1d(t):
        dv = t - v
        if dv == 0:
            return float(np.max(corr)) 
        nc = corr + dv * A
        nc[k0] = c0 + dv * (1.0 - t - v)
        return float(np.max(nc))
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr)) * 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            t_new = coord_opt(best_h, n, i, corr)
            if abs(t_new - best_h[i]) > 1e-10:
                old = best_h[i]
                best_h[i] = t_new
                # Update corr incrementally
                # (For simplicity, recompute; for n=180 it's fast)
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                c5_new = float(np.max(corr)) * 2.0 / n
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
                else:
                    best_h[i] = old
                    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        if not improved:
            break
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i, corr):
    v = h[i]
    n1 = n - 1
    k0 = n1
    A = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        if k == 0:
            continue
        j1 = i + k
        j2 = i - k
        if 0 <= j1 < n:
            A[k + n1] += (1.0 - h[j1])
        if 0 <= j2 < n:
            A[k + n1] -= h[j2]
    c0 = corr[k0]
    def f1d(t):
        dv = t - v
        nc = corr.copy()
        nc += dv * A
        nc[k0] = c0 + dv * (1.0 - t - v)
        return float(np.max(nc))
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=600):
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr)) * 2.0 / n
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            t_new = coord_opt(best_h, n, i, corr)
            if abs(t_new - best_h[i]) > 1e-12:
                best_h[i] = t_new
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                c5_new = float(np.max(corr)) * 2.0 / n
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
                else:
                    best_h[i] = t_new  # keep (neutral)
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return best_h, best_c5

def swap_local(h, n, t_end, n_iter=20000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.default_rng(999)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or abs(best_h[i] - best_h[j]) < 1e-10:
            continue
        old_i, old_j = best_h[i], best_h[j]
        best_h[i], best_h[j] = old_j, old_i
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new
        else:
            best_h[i], best_h[j] = old_i, old_j
    return best_h, best_c5

def sa_phase(h, n, t_end, n_iters=800):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = best_h.copy()
    work_c5 = best_c5
    rng = np.random.default_rng(1234)
    temp = 0.004
    for it in range(n_iters):
        if time.time() > t_end:
            break
        frac = it / float(n_iters)
        temp = 0.004 * (1.0 - 0.92 * frac) + 0.0001
        move_type = rng.random()
        if move_type < 0.5:
            n_pert = rng.integers(3, 15)
            start = rng.integers(0, n - n_pert)
            idx = np.arange(start, start + n_pert)
            work_h[idx] += rng.standard_normal(n_pert) * (0.05 + 0.1 * (1.0 - frac))
        elif move_type < 0.8:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                work_h[i], work_h[j] = work_h[j], work_h[i]
        else:
            i = rng.integers(0, n)
            work_h[i] = 1.0 - work_h[i]
        work_h = proj(work_h, n)
        c5_new = compute_c5(work_h)
        delta = c5_new - work_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            work_h = work_h.copy()
            work_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = work_h.copy()
        else:
            work_h = work_h.copy()
        if it % 100 == 99:
            work_h, work_c5 = coord_descent(work_h, n, t_end, max_sweeps=5)
            if work_c5 < best_c5:
                best_c5 = work_c5
                best_h = work_h.copy()
    return best_h, best_c5

def focused_slsqp(h, n, maxiter=1500):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val * 0.9999)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    def obj_f(x):
        c = np.correlate(x, 1.0 - x, mode='full')[active]
        return np.max(c) * dx
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj_f, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj(np.clip(res.x, 0, 1), n)
```
