For k ≥ 0: h[i] appears as h[j] when j=i (if i ≤ n-1-k) with coeff (1-h[i+k]), and as h[j+k] when j+k=i, i.e., j=i-k (if i-k ≥ 0 and i-k ≤ n-1-k) with coeff -h[i-k]*... wait, (1-h[j+k]) so the derivative of (1-h[i]) wrt h[i] is -1, times h[j] = h[i-k].

So for k ≥ 0: ∂corr/∂h[i] = (1-h[i+k])*1_{i≤n-1-k} - h[i-k]*1_{i-k≥0}
For k < 0 (m=-k): corr = sum_{j=0}^{n-1-m} h[j+m]*(1-h[j])
h[i] appears as h[j+m] when j=i-m (if 0≤i-m≤n-1-m) with coeff (1-h[i-m]), and as (1-h[j]) when j=i (if 0≤i≤n-1-m) with coeff -h[i+m].

So for k < 0, m=-k: ∂corr/∂h[i] = (1-h[i-m])*1_{m≤i≤n-1} - h[i+m]*1_{0≤i≤n-1-m}

Combined: ∂corr[k_idx]/∂h[i] = (1-h[i+k])*1_{i+k<n, i+k≥0, i≥0, i<n, i≤n-1-|k| if k≥0 else i≥|k|} - h[i-k]*1_{i-k≥0, i-k<n}

def project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def sm_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def sm_obj_topk(h, T, n, k=3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = np.sort(corr)[::-1]
    top = sc[:k].sum()
    rest = sc[k:]
    if len(rest) > 0:
        m2 = np.max(rest)
        top += 0.3 * (m2 + np.log(np.sum(np.exp(T * (rest - m2)))) / T)
    return top * (2.0 / n)

def slsqp_solve(h0, T, n, mi=600, ft=1e-15, topk=1):
    ob = sm_obj_topk if topk > 1 else sm_obj
    def f(x):
        return ob(x, T, n) if topk > 1 else sm_obj(x, T, n)
    def g(x):
        return x.sum() - n / 2.0
    r = minimize(f, h0, method='SLSQP', bounds=[(0, 1)] * n,
                 constraints=[{'type': 'eq', 'fun': g}],
                 options={'maxiter': mi, 'ftol': ft})
    return project(r.x, n)

def golden_1d(h, n, i, lo, hi, t_end):
    """Golden section on h[i] in [lo,hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i] = c
    fc = compute_c5(h)
    h[i] = d
    fd = compute_c5(h)
    n_iter = 0
    while (b - a) > 1e-10 and n_iter < 40:
        if time.time() > t_end:
            break
        n_iter += 1
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h[i] = c
            fc = compute_c5(h)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h[i] = d
            fd = compute_c5(h)
    best = (a + b) * 0.5
    h[i] = best
    return compute_c5(h), best

def coord_descent(h0, n, t_end, max_sweeps=5000):
    h = h0.copy()
    c5 = compute_c5(h)
    no_imp = 0
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                break
            ov = h[i]
            # Quick grid scan
            cands = [0.0, 1.0, float(np.clip(ov + 0.5, 0, 1)), float(np.clip(ov - 0.5, 0, 1)),
                     float(np.clip(ov + 0.25, 0, 1)), float(np.clip(ov - 0.25, 0, 1)),
                     float(np.clip(ov + 0.1, 0, 1)), float(np.clip(ov - 0.1, 0, 1))]
            bv, bh = c5, ov
            for cv in cands:
                if abs(cv - ov) < 1e-13:
                    continue
                h[i] = cv
                v = compute_c5(h)
                if v < bv - 1e-14:
                    bv, bh = v, cv
            if bh == ov:
                h[i] = ov
                continue
            # Golden section refinement around best candidate
            lo = max(0.0, bh - 0.15)
            hi = min(1.0, bh + 0.15)
            h[i] = ov
            cv2, bv2 = golden_1d(h, n, i, lo, hi, t_end)
            # Also check the boundaries of [0,1]
            h[i] = 0.0
            v0 = compute_c5(h)
            h[i] = 1.0
            v1 = compute_c5(h)
            h[i] = ov
            best_v = min(cv2 if h[i] != ov else c5, v0, v1, c5)
            if cv2 < c5 - 1e-14:
                h[i] = bv2
                c5 = cv2
                imp = True
            elif v0 < c5 - 1e-14:
                h[i] = 0.0
                c5 = v0
                imp = True
            elif v1 < c5 - 1e-14:
                h[i] = 1.0
                c5 = v1
                imp = True
        no_imp = 0 if imp else no_imp + 1
        if no_imp >= 4:
            break
    return h, c5

def interp_h(h, n_old, n_new):
    xo = np.linspace(0, 2, n_old)
    xn = np.linspace(0, 2, n_new)
    return np.interp(xn, xo, h)

def binary_sturmian(n, a):
    return np.array([1.0 if ((i + 1) * a) % 1.0 < a else 0.0 for i in range(n)])

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        phi = (np.sqrt(5.0) - 1.0) / 2.0
        h_init = project(binary_sturmian(160, phi), 160)
    n_base = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: SLSQP with increasing T (15% budget)
    t_end1 = t0 + budget_s * 0.15
    for T in [50, 300, 2000, 10000, 50000]:
        if time.time() > t_end1:
            break
        try:
            hc = slsqp_solve(best_h.copy(), T, n_base, mi=1000, ft=1e-16, topk=3)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 2: Coordinate descent (25%)
    t_end2 = t0 + budget_s * 0.40
    best_h, best_c5 = coord_descent(best_h, n_base, t_end2, max_sweeps=5000)

    # Phase 3: SLSQP hot phase with top-k (12%)
    t_end3 = t0 + budget_s * 0.52
    for T in [5000, 30000, 100000]:
        if time.time() > t_end3:
            break
        try:
            hc = slsqp_solve(best_h.copy(), T, n_base, mi=1500, ft=1e-16, topk=1)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 4: Fine coordinate descent (20%)
    t_end4 = t0 + budget_s * 0.72
    best_h, best_c5 = coord_descent(best_h, n_base, t_end4, max_sweeps=5000)

    # Phase 5: Try n=200 (18%)
    t_end5 = t0 + budget_s * 0.90
    n_alt = 200
    h_alt = project(interp_h(best_h, n_base, n_alt), n_alt)
    try:
        h_alt = slsqp_solve(h_alt, 5000, n_alt, mi=800, ft=1e-15, topk=3)
    except Exception:
        pass
    h_alt, c5_alt = coord_descent(h_alt, n_alt, t_end5, max_sweeps=3000)
    # Compare: interpolate back
    h_back = project(interp_h(h_alt, n_alt, n_base), n_base)
    c5_back = compute_c5(h_back)
    if c5_back < best_c5:
        best_c5, best_h = c5_back, h_back
    # Also check n=240
    if time.time() < t0 + budget_s * 0.97 and time.time() < t_end5:
        n_alt2 = 240
        h_alt2 = project(interp_h(best_h, n_base, n_alt2), n_alt2)
        h_alt2, c5_alt2 = coord_descent(h_alt2, n_alt2, t0 + budget_s * 0.97, max_sweeps=1500)
        h_back2 = project(interp_h(h_alt2, n_alt2, n_base), n_base)
        c5_back2 = compute_c5(h_back2)
        if c5_back2 < best_c5:
            best_c5, best_h = c5_back2, h_back2

    # Phase 6: Final polish (3%)
    t_end6 = t0 + budget_s * 0.99
    try:
        hc = slsqp_solve(best_h.copy(), 100000, n_base, mi=1000, ft=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
    except Exception:
        pass
    if time.time() < t_end6:
        best_h, best_c5 = coord_descent(best_h, n_base, t_end6, max_sweeps=500)

    best_h = project(best_h, n_base)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_base
