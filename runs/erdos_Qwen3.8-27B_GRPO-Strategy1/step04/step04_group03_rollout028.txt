import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        h = np.clip(h + diff / len(h), 0.0, 1.0)
    return h

def min_cpl(a, b, lo, hi):
    if hi - lo < 1e-13:
        return 0.5 * (lo + hi)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        mid = 0.5 * (lo + hi)
        vals = a + mid * b
        t = vals.max()
        active = vals > t - 1e-9
        ba = b[active]
        if ba.min() <= 0.0 and ba.max() >= 0.0:
            return mid
        elif ba.max() < 0.0:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def cd_fast(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    k_all = np.arange(-(n - 1), n)
    for sweep in range(200000):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            h_old = best_h[i]
            lo = max(-h_old, 0.0)
            hi = min(1.0 - h_old, 1.0)
            if hi <= lo:
                continue
            idx_a = i - k_all
            idx_b = i + k_all
            mask_a = (idx_a >= 0) & (idx_a < n)
            mask_b = (idx_b >= 0) & (idx_b < n)
            coef = np.zeros(2 * n - 1)
            coef[mask_a] = 1.0 - best_h[idx_a[mask_a]]
            coef[mask_b] -= best_h[idx_b[mask_b]]
            s_opt = min_cpl(corr, coef, lo, hi)
            if abs(s_opt) > 1e-12:
                best_h[i] += s_opt
                corr += s_opt * coef
                nm = corr.max()
                c5_new = nm * dx
                if c5v := c5_new:
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        improved = True
        if sweep % 200 == 199:
            corr = compute_corr(best_h)
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        sm = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return sm * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)
