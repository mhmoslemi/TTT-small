    dx = 2.0 / n

                grad[:n-nl] += pk*(-h[nl:])

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj(h, T, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        c = corr[idx]
    else:
        c = corr
    m = np.max(c)
    sh = np.clip(T * (c - m), -500, 500)
    s = np.sum(np.exp(sh))
    return (m + np.log(s) / T) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-13:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-13:
            return h
        h += r * (d / t)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, miter=400, topk=None):
    def obj(x):
        return smooth_obj(x, T, topk)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': miter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

            coef = np.zeros(2 * n - 1)

                c = 0.0
                j1 = i + lag
                j2 = i - lag
