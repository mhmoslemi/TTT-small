import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    # softmax smoothing
    shifted = T * (corr - m)
    # clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    val = m + np.log(np.sum(np.exp(shifted)) + 1e-30) / T
    return val

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_c5(x, T)
    def con_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt(h, i):
    n = len(h)
    dx = 2.0 / n
    h0 = h.copy()
    h0[i] = 0.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    m_vals = np.arange(-(n-1), n)
    j1 = i - m_vals
    j2 = i + m_vals
    coefs = np.zeros(2*n-1)
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[j1[v1]])
    coefs[v2] += (-h[j2[v2]])
    
    def f1d(hi):
        return float(np.max(corr0 + hi * coefs) * dx)
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-15})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500, verbose=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stagnation = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, i)
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
                stagnation = 0
            elif c5_try < best_c5 + 1e-14:
                # Acceptable, update anyway for exploration
                pass
        if verbose:
            print(f"Sweep {sweep}: c5={best_c5:.8f}")
        if not improved:
            stagnation += 1
            if stagnation > 3:
                break
    return best_h, best_c5

def targeted_perturbation(h, n, magnitude=0.1):
    """Perturb coordinates that most affect the max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argmax(corr)
    m_val = top_idx - (n - 1)  # the lag
    
    # Find which coordinates most affect this correlation
    # corr[top_idx] = sum_i h[i+m_val] * (1-h[i])
    # d(corr)/d(h[j]) = (1-h[j-m_val]) - h[j+m_val] (when valid)
    sensitivity = np.zeros(n)
    for j in range(n):
        s = 0
        jm = j - m_val
        jp = j + m_val
        if 0 <= jm < n:
            s += (1.0 - h[jm])
        if 0 <= jp < n:
            s += (-h[jp])
        sensitivity[j] = s
    
    # Perturb the most sensitive coordinates
    k = max(5, n // 10)
    top_coords = np.argsort(np.abs(sensitivity))[-k:]
    pert = np.zeros(n)
    signs = np.random.choice([-1, 1], size=k)
    # If sensitivity > 0, decreasing h[j] decreases corr -> perturb negative
    # If sensitivity < 0, increasing h[j] decreases corr -> perturb positive
    for idx, j in enumerate(top_coords):
        direction = -np.sign(sensitivity[j]) if sensitivity[j] != 0 else 1
        pert[j] = magnitude * direction * np.random.uniform(0.5, 1.0)
    
    return project_h(h + pert)
