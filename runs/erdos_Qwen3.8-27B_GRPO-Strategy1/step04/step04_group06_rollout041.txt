       """Find optimal h[p] given current correlation c and h (with h[p] to be optimized).
       Returns new value of h[p]."""
       center = n - 1
       old = h[p]
       dx = 2.0 / n
       
       # Compute A[m] and Q[m] for all m
       # For lag j = m - center:
       # If j == 0: contribution is h[p]*(1-h[p]) = t - t², so A=1, Q=-1

       A = np.zeros(2*n - 1)
