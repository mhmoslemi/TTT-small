import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    m = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            coeff = np.zeros(2 * n - 1)
            lo1, hi1 = max(0, m - i), min(2 * n - 2, m - i + n - 1)
            if lo1 <= hi1:
                coeff[lo1:hi1 + 1] += 1.0 - best_h[i + np.arange(lo1, hi1 + 1) - m]
            lo2, hi2 = max(0, i + m - n + 1), min(2 * n - 2, i + m)
            if lo2 <= hi2:
                coeff[lo2:hi2 + 1] -= best_h[i - np.arange(lo2, hi2 + 1) + m]
            f0 = float(np.max(corr - old * coeff))
            f1 = float(np.max(corr + (1.0 - old) * coeff))
            best_f = min(f0, f1)
            best_v = 0.0 if f0 <= f1 else 1.0
            res = minimize_scalar(lambda v: float(np.max(corr + (v - old) * coeff)),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 35})
            if res.fun < best_f:
                best_v, best_f = res.x, res.fun
            if best_f * dx < best_c5 - 1e-15:
                corr += (best_v - old) * coeff
                best_h[i] = best_v
                best_c5 = best_f * dx
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_smooth(h0, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mm = np.max(c)
        return (mm + np.log(np.sum(np.exp(T * (c - mm)))) / T) * dx
    res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

def basin_hop(h_best, c5_best, n, t_end, max_iters=400):
    h_work = h_best.copy()
    c5_work = c5_best
    evals = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.02, 1.0 - 0.95 * it / max_iters)
        step = 0.02 + 0.30 * cool
        r = np.random.rand()
        h_p = h_work.copy()
        if r < 0.35:
            h_p += step * np.random.randn(n)
        elif r < 0.55:
            a = np.random.randint(0, max(1, n - 12))
            b = min(a + np.random.randint(4, max(5, n // 4)), n)
            h_p[a:b] = 1.0 - h_p[a:b]
        elif r < 0.72:
            ln = np.random.randint(3, max(4, 15))
            a = np.random.randint(0, max(1, n - 2 * ln - 1))
            c = np.random.randint(a + ln + 1, max(a + ln + 2, n - ln))
            h_p[a:a+ln], h_p[c:c+ln] = h_p[c:c+ln].copy(), h_p[a:a+ln].copy()
        elif r < 0.88:
            nf = np.random.randint(2, max(3, n // 10))
            idx = np.random.choice(n, nf, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        else:
            h_p = 1.0 - h_work
        h_p = project(h_p, n)
        t_inner = min(3.0, max(0.5, (t_end - time.time()) * 0.02))
        h_c, c5_c = coord_descent(h_p, n, time.time() + t_inner, max_sweeps=40)
        evals += 1
        temp = 0.004 * cool + 0.0002
        if c5_c < c5_work or np.random.rand() < np.exp(-(c5_c - c5_work) / temp):
            h_work, c5_work = h_c, c5_c
            if c5_c < c5_best:
                c5_best, h_best = c5_c, h_c.copy()
    if c5_work < c5_best:
        c5_best, h_best = c5_work, h_work
    return h_best, c5_best
