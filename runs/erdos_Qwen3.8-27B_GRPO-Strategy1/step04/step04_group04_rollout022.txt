    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def smooth_c5_grad(h, T):

    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    val = (m + np.log(exps.sum()) / T) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            r = n - lag
            grad[:r] += pk * (1.0 - h[lag:n])
            grad[lag:] += pk * (-h[:r])
        else:
            nl = -lag
            r = n - nl
            grad[nl:] += pk * (1.0 - h[:r])
            grad[:r] += pk * (-h[nl:])
    return val, grad * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    d = target - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (d / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def lbfgs_opt(h, T, n, maxiter=400):
    res = minimize(smooth_c5_grad, h, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 40})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    hz = h.copy()
    hv = hz[i]
    hz[i] = 0.0
    cb = corr_full(hz)
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += 1.0 - h[j1]
        if 0 <= j2 < n:
            c -= h[j2]
        coef[k] = c
    k0 = n - 1
    def f1(t):
        cv = cb + t * coef
        cv[k0] += t * (1.0 - t)
        return float(cv.max()) * dx
    return float(minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12}).x)

def cd_sweep(h, n, t_end, max_sweeps=99999):
    best = h.copy()
    bc5 = compute_c5(best)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, bc5
            new_v = coord_opt(best, n, i)
            old_v = best[i]
            if abs(new_v - old_v) < 1e-15:
                continue
            best[i] = new_v
            best = proj(best, n)
            c = compute_c5(best)
            if c < bc5 - 1e-15:
                bc5 = c
                improved = True
            else:
                best[i] = old_v
                best = proj(best, n)
        if not improved and sw > 2:
            break
    return best, bc5

def squeeze(h, n):
    h = np.where(h > 0.97, 1.0, h)
    h = np.where(h < 0.03, 0.0, h)
    return proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h, best_c5 = cd_sweep(best_h, n, t0 + budget_s * 0.50)
    
    # Phase 4: High-T soft-max
    for T in [50000, 200000]:
        if time.time() - t0 > budget_s * 0.60:
            break
        hc = lbfgs_opt(best_h, T, n, 500)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5, best_h = c, hc
    
    # Phase 5: Basin hopping
    wh, wc = best_h.copy(), best_c5
    for it in range(50):
        if time.time() - t0 > budget_s * 0.78:
            break
        cool = max(0.1, 1.0 - it / 45.0)
        r = np.random.rand()
        if r < 0.4:
            hp = wh + (0.05 + 0.15 * cool) * np.random.randn(n)
        elif r < 0.7:
            hp = wh.copy()
            b = np.random.randint(3, 15)
            s = np.random.randint(0, n - b)
            hp[s:s+b] = 1.0 - hp[s:s+b]
        else:
            hp = wh.copy()
            p, q = np.random.choice(n, 2, replace=False)
            hp[p], hp[q] = 1.0 - hp[p], 1.0 - hp[q]
        hp = proj(hp, n)
        try:
            hc = lbfgs_opt(hp, 3000, n, 100)
            c = compute_c5(hc)
            temp = 0.005 * cool
            if c < wc or np.random.rand() < np.exp(-(c - wc) / max(temp, 1e-10)):
                wh, wc = hc, c
                if c < best_c5:
                    best_c5, best_h = c, hc
        except Exception:
            pass
    
    # Phase 6: Final coord descent
    best_h, best_c5 = cd_sweep(best_h, n, t0 + budget_s * 0.92)
    
    # Phase 7: Ultimate polish

        hc = lbfgs_opt(best_h, T, n, 600)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5, best_h = c, hc
    best_h, best_c5 = cd_sweep(best_h, n, t0 + budget_s * 0.995)
    
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    sp = exps.sum()
    p = exps / sp
    val = (m + np.log(sp) / T) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            r = n - lag
            if r > 0:
                grad[:r] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:r])
        else:
            nl = -lag
            r = n - nl
            if r > 0:
                grad[nl:] += pk * (1.0 - h[:r])
                grad[:r] += pk * (-h[nl:])
    return val, grad * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    d = target - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        rm = 1.0 - h
        tot = rm.sum()
        if tot < 1e-14:
            return h
        h += rm * (d / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def lbfgs_opt(h, T, n, maxiter=400):
    res = minimize(smooth_c5_grad, h, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 40})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    hz = h.copy()
    hv = hz[i]
    hz[i] = 0.0
    cb = corr_full(hz)
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += 1.0 - h[j1]
        if 0 <= j2 < n:
            c -= h[j2]
        coef[k] = c
    k0 = n - 1
    cb0 = cb[k0]
    def f1(t):
        cv = cb + t * coef
        cv[k0] = cb0 + t * (1.0 - t)
        return float(cv.max()) * dx
    return float(minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12}).x)

def cd_sweep(h, n, t_end):
    best = h.copy()
    bc5 = compute_c5(best)
    for sw in range(999999):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, bc5
            nv = coord_opt(best, n, i)
            ov = best[i]
            if abs(nv - ov) < 1e-14:
                continue
            best[i] = nv
            best = proj(best, n)
            c = compute_c5(best)
            if c < bc5 - 1e-15:
                bc5 = c
                improved = True
            else:
                best[i] = ov
                best = proj(best, n)
        if not improved and sw > 3:
            break
    return best, bc5

def squeeze(h, n):
    h = np.where(h > 0.97, 1.0, h)
    h = np.where(h < 0.03, 0.0, h)
    return proj(h, n)

def active_softmax(h, n, T, frac=0.3):
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    nact = max(5, int(np.ceil((2 * n - 1) * frac)))
    idx = np.argsort(corr)[-nact:]
    sub = corr[idx]
    sm = np.max(sub)
    sh = np.clip(T * (sub - sm), -500, 500)
    ev = np.exp(sh)
    sp = ev.sum()
    psub = ev / sp
    val = (sm + np.log(sp) / T) * dx
    grad = np.zeros(n)
    for j, k in enumerate(idx):
        pk = psub[j]
        if pk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            r = n - lag
            if r > 0:
                grad[:r] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:r])
        else:
            nl = -lag
            r = n - nl
            if r > 0:
                grad[nl:] += pk * (1.0 - h[:r])
                grad[:r] += pk * (-h[nl:])
    return val, grad * dx

def lbfgs_active(h, T, n, frac, maxiter=400):
    res = minimize(active_softmax, h, args=(n, T, frac), method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 40})
    return proj(np.clip(res.x, 0, 1), n)

    t0 = time.time()

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():

    for T in [8, 40, 150, 600, 2500, 10000]:
        if time.time() - t0 > budget_s * 0.08:

        try:
            hc = lbfgs_opt(best_h, T, n, 300)
            c = compute_c5(hc)
            if c < best_c5:
                best_c5, best_h = c, hc
        except Exception:
            pass

    # Phase 2: Heavy coordinate descent
    best_h, best_c5 = cd_sweep(best_h, n, t0 + budget_s * 0.30)

    # Phase 3: Squeeze + more CD
    h_sq = squeeze(best_h, n)
    c_sq = compute_c5(h_sq)
    if c_sq < best_c5:
        best_c5, best_h = c_sq, h_sq
    best_h, best_c5 = cd_sweep(best_h, n, t0 + budget_s * 0.42)

    # Phase 4: Active-set soft-max (focus on top lags)
    for T, fr in [(10000, 0.2), (50000, 0.15), (200000, 0.1)]:
        if time.time() - t0 > budget_s * 0.55:
            break
        try:
            hc = lbfgs_active(best_h, T, n, fr, 500)
            c = compute_c5(hc)
            if c < best_c5:
                best_c5, best_h = c, hc
        except Exception:
            pass
    best_h, best_c5 = cd_sweep(best_h, n, t0 + budget_s * 0.58)

    # Phase 5: Basin hopping with structured perturbations
    wh, wc = best_h.copy(), best_c5
    n_it = 45
    for it in range(n_it):
        if time.time() - t0 > budget_s * 0.76:

            ii = np.random.choice(n, 2, replace=False)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
