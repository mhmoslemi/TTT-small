            best_x = old_val
            best_f = np.max(base + old_val * coef)
            for x in cands:
                f = np.max(base + x * coef)
                if f < best_f - 1e-14:
                    best_f = f
                    best_x = x
            if abs(best_x - old_val) > 1e-10:
                delta = best_x - old_val
                best_h[i] = best_x
                corr += delta * coef
                c5 = float(np.max(corr) * dx)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                else:
                    best_h[i] = old_val
                    corr -= delta * coef
        best_h = project_h(best_h, n)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    ratio = n_new / n_old
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    new_h = np.interp(x_new, x_old, h)
    return project_h(new_h, n_new)

def make_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def make_structured_init(n, rng):
    pattern = np.zeros(n)
    block = n // 4
    for i in range(n // block + 1):
        start = i * block
        if start < n:
            if i % 2 == 0:
                pattern[start:min(start + block, n)] = 1.0
    h = pattern + 0.1 * rng.randn(n)
    return project_h(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 160

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)

    n_main = 160
    if best_h is None:
        best_h = make_binary_init(n_main, rng)

    # Phase 1: Gradient descent from initial (15% budget)
    t1 = t0 + budget_s * 0.15
    h1, c5_1 = fg_momentum(best_h, best_n, t1, lr_init=0.05, T0=30.0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Try a coarse resolution (10% budget)
    if 'initial_h_values' not in globals() or initial_h_values is None:
        pass
    n_coarse = 80
    t2 = t0 + budget_s * 0.25
    h_coarse = make_binary_init(n_coarse, rng)
    h_coarse2 = make_structured_init(n_coarse, rng)
    for hstart in [h_coarse, h_coarse2]:
        if time.time() > t2:
            break
        hc, c5c = fg_momentum(hstart, n_coarse, min(time.time() + 5, t2), lr_init=0.06, T0=20.0)
        hc_up = upsample(hc, n_main)
        hc_up, c5_up = coord_descent_fast(hc_up, n_main, min(time.time() + 10, t2), max_sweeps=50)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = hc_up
            best_n = n_main

    # Phase 3: Basin hopping with gradient descent (25% budget)
    t3 = t0 + budget_s * 0.50
    work_h = best_h.copy()
    work_c5 = best_c5
    n_work = best_n
    for it in range(60):
        if time.time() > t3:
            break
        cool = 1.0 - 0.8 * (it / 60.0)
        step = 0.03 + 0.12 * cool
        if it % 3 == 0:
            pert = step * rng.randn(n_work)
        elif it % 3 == 1:
            freq = rng.randint(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * freq * np.arange(n_work) / n_work + phase)
        else:
            mask = rng.random(n_work) < 0.3
            pert = np.zeros(n_work)
            pert[mask] = step * rng.randn(mask.sum())
        h_pert = project_h(work_h + pert, n_work)
        t_end_it = min(time.time() + 3.0, t3)
        hc, c5c = fg_momentum(h_pert, n_work, t_end_it, lr_init=0.03, T0=200.0 + 3000 * cool)
        temp = 0.003 * cool + 0.0003
        if c5c < work_c5 or rng.rand() < np.exp(-(c5c - work_c5) / temp):
            work_h = hc
            work_c5 = c5c
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc.copy()
                best_n = n_work
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()
        best_n = n_work

    # Phase 4: Coordinate descent main (30% budget)
    t4 = t0 + budget_s * 0.80
    best_h, best_c5 = coord_descent_fast(best_h, best_n, t4, max_sweeps=2000)

    # Phase 5: More basin hopping (10% budget)
    t5 = t0 + budget_s * 0.90
    work_h = best_h.copy()
    work_c5 = best_c5
    n_work = best_n
    for it in range(30):
        if time.time() > t5:
            break
        step = 0.02 + 0.05 * (1.0 - it / 30.0)
        h_pert = project_h(work_h + step * rng.randn(n_work), n_work)
        t_end_it = min(time.time() + 2.0, t5)
        hc, c5c = fg_momentum(h_pert, n_work, t_end_it, lr_init=0.02, T0=500.0)
        if c5c < work_c5:
            work_h = hc
            work_c5 = c5c
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc.copy()
                best_n = n_work

    # Phase 6: Final coordinate descent (10% budget)
    t6 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent_fast(best_h, best_n, t6, max_sweeps=1500)

    best_h = project_h(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
