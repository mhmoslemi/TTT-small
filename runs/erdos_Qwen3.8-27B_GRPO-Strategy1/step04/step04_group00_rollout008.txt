from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_corr(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smcv(h, T):
    """Return (value, gradient) of smooth max correlation."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    se = e.sum()
    w = e / se
    val = (m + np.log(se) / T) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        ws = max(0, lag)
        we = min(n, n + lag)
        if ws < we:
            grad[ws:we] += w[k] * omh[ws - lag:we - lag]
        ws2 = max(0, -lag)
        we2 = min(n, n - lag)
        if ws2 < we2:
            grad[ws2:we2] -= w[k] * h[ws2 + lag:we2 + lag]
    return val, dx * grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h0, T, n, maxiter=2000, ftol=1e-15):
    def obj(x):
        v, _ = smcv(x, T)
        return v
    def jac(x):
        _, g = smcv(x, T)
        return g
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def coord_step(h, n, i):
    """Optimize h[i] with others fixed."""
    h_base = h.copy()
    old = h_base[i]
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    dx = 2.0 / n
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coef = np.zeros(2 * n - 1)
    coef[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n-1)])
    coef[valid2] += (-h[np.clip(j2[valid2], 0, n-1)])
    def f1d(v):
        return np.max(corr_base + v * coef) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100):

                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

numpy.correlate(a, v): "Cross-correlation of two 1 dimensional sequences."
The formula is: c[k] = Σ_n a[n+k] * conj(v[n])

where k ranges such that the product is defined.

So np.correlate(h, 1-h, mode='full'):
Let a = h (length n), v = 1-h (length n)

For mode='full', k ranges from -(n-1) to (n-1), giving 2n-1 values.
The output is indexed 0 to 2n-2, corresponding to k = -(n-1), -(n-2), ..., 0, ..., n-2, n-1.

For this to be valid: 0 ≤ n < n and 0 ≤ n + lag < n

  Contribution: -h[j+lag] if 0 ≤ j+lag < n

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smcv(h, T):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    se = e.sum()
    w = e / se
    val = (m + np.log(se) / T) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        ws = max(0, lag)
        we = min(n, n + lag)
        if ws < we:
            grad[ws:we] += w[k] * omh[ws - lag:we - lag]
        ws2 = max(0, -lag)
        we2 = min(n, n - lag)
        if ws2 < we2:
            grad[ws2:we2] -= w[k] * h[ws2 + lag:we2 + lag]
    return val, dx * grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def slsqp_smooth(h0, T, n, maxiter=2000, ftol=1e-15):
    def obj(x):
        v, _ = smcv(x, T)
        return v
    def jac(x):
        _, g = smcv(x, T)
        return g
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: sum_constraint(x, n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def coord_step(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    dx = 2.0 / n
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coef = np.zeros(2 * n - 1)
    coef[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n - 1)])
    coef[valid2] += (-h[np.clip(j2[valid2], 0, n - 1)])
    def f1d(v):
        return np.max(corr_base + v * coef) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=150):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            v_new = coord_step(best_h, n, i)
            if abs(v_new - best_h[i]) > 1e-12:
                h_try = best_h.copy()
                h_try[i] = v_new
                h_try = project(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hopping(h, n, t_end, n_iter, scale_base, T_base, maxiter):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(n_iter, 1))
        scale = scale_base * (0.3 + cool)
        h_pert = project(work_h + scale * np.random.randn(n), n)
        T_bh = T_base * (0.2 + cool)
        hc = slsqp_smooth(h_pert, T_bh, n, maxiter=maxiter, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.002 * cool + 0.0002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def make_structured(n, pattern_idx):
    x = np.arange(n) / n
    if pattern_idx == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * x)
    elif pattern_idx == 1:
        h = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.sin(2 * np.pi * x)
    elif pattern_idx == 2:
        h = 0.5 + 0.25 * np.sin(6 * np.pi * x + 0.5)
    elif pattern_idx == 3:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x) * np.cos(2 * np.pi * x)
    else:
        h = 0.5 + 0.2 * np.sin(2 * np.pi * (1 + pattern_idx) * x)
    return project(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP annealing with analytical gradient (n=160)
    for T in [200, 1000, 4000, 15000, 60000]:
        if time.time() - t0 > budget_s * 0.18:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=2000, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Basin hopping (n=160)
    work_h, work_c5 = basin_hopping(best_h, n, t0 + budget_s * 0.42,
                                     70, 0.12, 3000, 700)
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 3: Coordinate descent (n=160)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.62, max_sweeps=120)

    # Phase 4: Try n=80 with structured starts
    n80 = 80
    best80_h = None
    best80_c5 = 1.0
    for pi in range(5):
        if time.time() - t0 > budget_s * 0.82:
            break
        h80 = make_structured(n80, pi)
        for T in [500, 3000]:
            if time.time() - t0 > budget_s * 0.83:
                break
            hc80 = slsqp_smooth(h80, T, n80, maxiter=1500, ftol=1e-15)
            c5_80 = compute_c5(hc80)
            if c5_80 < best80_c5:
                best80_c5 = c5_80
                best80_h = hc80
        if time.time() - t0 > budget_s * 0.84:
            break
        h80, c80 = coord_descent(best80_h, n80, t0 + budget_s * 0.85, max_sweeps=60)
        if c80 < best80_c5:
            best80_c5 = c80

    if best80_h is not None and best80_c5 < best_c5:

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, max_sweeps=60)

    # Phase 7: Ultra-high-T polish

                best_c5 = c5v

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
