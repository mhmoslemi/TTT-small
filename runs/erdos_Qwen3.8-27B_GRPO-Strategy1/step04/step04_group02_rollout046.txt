import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    dx = 2.0 / len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += h * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_fast(h, n, t_end, max_sweeps=200, rng=None):

        rng = np.random.default_rng(42)
    best_h = h.copy()

        for idx in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            p = order[idx]
            hp = best_h[p]
            c0_base = corr[n-1] - hp * (1.0 - hp)

            # Precompute derivatives for all lags
            a_fwd = np.zeros(n - 1)
            a_bwd = np.zeros(n - 1)
            for k in range(1, n):
                af = 0.0
                ab = 0.0
                pk = p - k
                pkp = p + k
                if 0 <= pk < n:
                    af += (1.0 - best_h[pk])
                    ab -= best_h[pk]
                if 0 <= pkp < n:
                    af -= best_h[pkp]
                    ab += (1.0 - best_h[pkp])
                a_fwd[k-1] = af
                a_bwd[k-1] = ab

            vals_fwd_base = corr[n:2*n-1].copy()
            vals_bwd_base = corr[0:n-1].copy()

            def f1d(t):
                c0 = c0_base + t * (1.0 - t)
                vf = np.max(vals_fwd_base + (t - hp) * a_fwd)
                vb = np.max(vals_bwd_base + (t - hp) * a_bwd)
                return max(c0, vf, vb) * dx

            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12, 'maxiter': 100})
            t_new = res.x
            if abs(t_new - hp) > 1e-12:
                dt = t_new - hp
                corr[n:2*n-1] += dt * a_fwd
                corr[0:n-1] += dt * a_bwd
                corr[n-1] = c0_base + t_new * (1.0 - t_new)
                best_h[p] = t_new

def slsqp_opt(h, T, n, maxiter=1500, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)
