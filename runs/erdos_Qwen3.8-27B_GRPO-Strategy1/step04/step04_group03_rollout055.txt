import time

def compute_c5(h):

    m = np.max(vals)
    if T < 1e-6:
        return m
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

    h = np.clip(h, 0.0, 1.0).copy()

    if abs(diff) < 1e-15:

        if sg < -1e-15:
            lo = mid
        elif sg > 1e-14:

        else:

        val = cc[im]
        tol = val * 1e-6 + 1e-12
        active = np.where(cc >= val - tol)[0]
        sg = np.zeros(n)
        lags = np.arange(-(n - 1), n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            for i in range(n):
                ja = i + lag; jb = i - lag
                c = 0.0
                if 0 <= ja < n: c += (1.0 - bh[ja])
                if 0 <= jb < n: c -= bh[jb]
                sg[i] += c
        sg *= dx
        step = lr * np.exp(-it / 200.0)
        bh_new = bh - step * sg
        bh_new = project(bh_new, n / 2.0)
        c5n = compute_c5(bh_new)
        if c5n < bc5:
            bh = bh_new; bc5 = c5n
            cc = get_corr(bh)
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def el(): return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(224); best_h[:112] = 1.0
    n = len(best_h)
    best_h = project(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial CD
    best_h, best_c5 = cd(best_h, t0 + budget_s * 0.06, 500)

    # Phase 2: SLSQP T-schedule with CD between
    for T in [30, 100, 500, 2000, 10000]:
        if el() > budget_s * 0.22: break
        hc = slsqp(best_h, T, n, 1000, 1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v
        best_h, best_c5 = cd(best_h, t0 + budget_s * 0.22, 80)

    # Phase 3: Extended CD
    t3 = t0 + budget_s * 0.50
    best_h, best_c5 = cd(best_h, t3, 5000)

    # Phase 4: Fourier diversification
    K = 8
    a0, cs = fourier_coeffs(best_h, n, K)
    work_h, work_c5 = best_h.copy(), best_c5
    n_restarts = 20
    for it in range(n_restarts):
        if el() > budget_s * 0.76: break
        cool = max(1.0 - 0.97 * (it / n_restarts), 0.02)
        pert = 0.04 * cool + 0.005
        cs_p = [(a + pert * np.random.randn(), b + pert * np.random.randn()) for a, b in cs]
        hp = fourier_eval(a0, cs_p, n)
        hp = project(hp, n / 2)
        # Also add small random noise
        hp = project(hp + 0.01 * cool * np.random.randn(n), n / 2)
        try:
            hc = slsqp(hp, max(500, int(5000 * cool)), n, 400, 1e-12)
            hc, c5v = cd(hc, time.time() + 3.0, 80)
            acc = c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(0.0005 * cool, 1e-6))
            if acc:
                work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = hc.copy(), c5v
        except Exception:
            pass

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Subgradient nudge
    if el() < budget_s * 0.85:
        best_h, best_c5 = subgrad_nudge(best_h, t0 + budget_s * 0.85, 300, 0.003)
        best_h, best_c5 = cd(best_h, t0 + budget_s * 0.87, 200)

    # Phase 6: High-T SLSQP final
    for T in [50000, 200000, 800000]:
        if el() > budget_s * 0.94: break
        hc = slsqp(best_h, T, n, 2000, 1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Phase 7: Final CD
    best_h, best_c5 = cd(best_h, t0 + budget_s * 0.995, 3000)

    best_h = project(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
