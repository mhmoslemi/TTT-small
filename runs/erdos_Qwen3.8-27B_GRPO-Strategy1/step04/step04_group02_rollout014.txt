def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        h = h * (target / s)
        h = np.clip(h, 0, 1)
        s2 = h.sum()
        diff = target - s2
        if diff > 1e-13:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += room * (diff / tot)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * (target - s) / tot
        h = np.clip(h, 0, 1)

def grad_corr(h, n, k):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        k2 = -k
        g[k2:] += (1.0 - h[:n - k2])
        g[:n - k2] -= h[k2:]
    return g

    m = len(active_keys)

    C_var = cp.Variable()
    obj = cp.Minimize(C_var)
    cons = []

        cons.append(C_var >= corr[active_keys[ii]] + G[ii] @ delta)

        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 2: LP subgradient (main phase)
    h_work = best_h.copy()
    c5_work = best_c5
    t_lp_end = t0 + budget_s * 0.50
    iters = 0
    while elapsed() < t_lp_end:
        corr = get_correlations(h_work)
        center = n - 1
        max_v = np.max(corr)
        # Active set: top correlations within threshold
        thresh = max_v - 3.0
        active = np.where(corr >= thresh)[0]
        if len(active) < 8:
            active = np.argsort(corr)[-20:]
        active = active[:50]
        # Try LP with decreasing step
        accepted = False
        for sc in [0.04, 0.02, 0.01]:
            h_new = lp_step(h_work, n, active, step_cap=sc)
            c5_new = compute_c5(h_new)
            if c5_new < c5_work - 1e-13:
                h_work, c5_work = h_new, c5_new
                accepted = True
                break
        if not accepted:
            # Fallback: subgradient direction
            h_new = subgradient_dir_step(h_work, n, active, alpha=0.01)
            c5_new = compute_c5(h_new)
            if c5_new < c5_work - 1e-13:
                h_work, c5_work = h_new, c5_new
        if c5_work < best_c5:
            best_h, best_c5 = h_work.copy(), c5_work
        iters += 1
        # Periodic SLSQP polish
        if iters % 300 == 0:
            hc = slsqp_opt(h_work, 2000, n, maxiter=200, ftol=1e-13)
            cv = compute_c5(hc)
            if cv < c5_work:
                h_work, c5_work = hc, cv
            if cv < best_c5:
                best_h, best_c5 = h_work.copy(), c5_work

    # Phase 3: Coordinate descent
    if elapsed() < budget_s * 0.70:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.70,
                                        max_sweeps=30, rng=rng)

    # Phase 4: Multi-resolution exploration
    for n_try in [200, 280, 320, 160, 240]:
        if elapsed() > budget_s * 0.87:
            break
        h_interp = interpolate_h(best_h, n_try)
        h_m = h_interp.copy()
        c5_m = compute_c5(h_m)
        n_m = n_try
        t_m_end = min(t0 + budget_s * 0.91, elapsed() + 30)
        for _ in range(300):
            if elapsed() > t_m_end:
                break
            corr = get_correlations(h_m)
            max_v = np.max(corr)
            thresh = max_v - 3.0
            active = np.where(corr >= thresh)[0]
            if len(active) < 8:
                active = np.argsort(corr)[-20:]
            active = active[:50]
            accepted = False
            for sc in [0.04, 0.02]:
                h_new = lp_step(h_m, n_m, active, step_cap=sc)
                cv = compute_c5(h_new)
                if cv < c5_m - 1e-13:
                    h_m, c5_m = h_new, cv
                    accepted = True
                    break
            if not accepted:
                break
        if c5_m < best_c5:
            best_h, best_c5, n = h_m.copy(), c5_m, n_try
        else:
            n = best_h.size

    # Phase 5: Final coordinate descent + SLSQP
    if elapsed() < budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95,
                                        max_sweeps=25, rng=rng)
    for T in [10000, 50000, 200000]:
        if elapsed() > budget_s * 0.985:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=1500, ftol=1e-15)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
