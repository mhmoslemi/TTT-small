import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def bin_proj(h, n):
    t = n // 2
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:t]] = 1.0
    return hb

def ls_binary(h, n, t_end, max_sw=50000, four_opt=True):
    dx = 2.0 / n
    h = h.copy()
    corr = corr_of(h)
    c5 = corr.max() * dx
    lags = np.arange(-(n-1), n)
    for _ in range(max_sw):
        if time.time() > t_end:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        rem = np.zeros((len(ones), 2*n-1))
        for ia, a in enumerate(ones):
            ip = a + lags; mp = (ip >= 0) & (ip < n)
            im = a - lags; mm = (im >= 0) & (im < n)
            rem[ia, mp] += h[ip[mp]]
            rem[ia, mm] -= (1.0 - h[im[mm]])
        add = np.zeros((len(zeros), 2*n-1))
        for ib, b in enumerate(zeros):
            ip = b + lags; mp = (ip >= 0) & (ip < n)
            im = b - lags; mm = (im >= 0) & (im < n)
            add[ib, mm] += (1.0 - h[im[mm]])
            add[ib, mp] -= h[ip[mp]]
        best_c = c5; ba = -1; bb = -1
        for ia in range(len(ones)):
            base = corr + rem[ia]
            nms = (base[None, :] + add).max(axis=1)
            bi = nms.argmin()
            if nms[bi] * dx < best_c - 1e-13:
                best_c = nms[bi] * dx; ba = ia; bb = bi
        if ba >= 0:
            h[ones[ba]] = 0.0; h[zeros[bb]] = 1.0
            corr = corr + rem[ba] + add[bb]
            corr[n-1] = 0.0
            c5 = best_c
        elif four_opt:
            ns = min(14, len(ones)); nz = min(14, len(zeros))
            oi = np.random.choice(len(ones), ns, replace=False) if ns > 2 else np.arange(ns)
            zi = np.random.choice(len(zeros), nz, replace=False) if nz > 2 else np.arange(nz)
            found = False
            for i1 in range(ns):
                for i2 in range(i1 + 1, ns):
                    rp = rem[oi[i1]] + rem[oi[i2]]
                    for j1 in range(nz):
                        b12 = corr + rp + add[zi[j1]]
                        nms2 = (b12[None, :] + add[zi]).max(axis=1)
                        nms2[int(zi[j1])] = np.inf
                        jc = np.where(nms2 < np.inf)[0]
                        if len(jc) == 0:
                            continue
                        j2 = jc[np.argmin(nms2[jc])]
                        if nms2[j2] * dx < c5 - 1e-13:
                            h[ones[oi[i1]]] = 0; h[ones[oi[i2]]] = 0
                            h[zeros[zi[j1]]] = 1; h[zeros[zi[j2]]] = 1
                            corr = b12 + add[zi[j2]]
                            corr[n-1] = 0.0
                            c5 = nms2[j2] * dx
                            found = True; break
                    if found:
                        break
                if found:
                    break
            if not found:
                break
        else:
            break
    return h, c5

def coord_cont(h, n, t_end, max_pass=500):
    dx = 2.0 / n
    h = h.copy(); corr = corr_of(h); c5 = corr.max() * dx
    lags = np.arange(-(n-1), n)
    ci = n - 1
    for _ in range(max_pass):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            old = h[i]
            if old <= 1e-8 or old >= 1.0 - 1e-8:
                continue
            ip = i + lags; mp = (ip >= 0) & (ip < n)
            im = i - lags; mm = (im >= 0) & (im < n)
            coeff = np.zeros(2*n-1)
            coeff[mm] += (1.0 - h[im[mm]])
            coeff[mp] -= h[ip[mp]]
            def obj(v):
                d = v - old
                cn = corr + d * coeff
                cn[ci] -= d * d
                return cn.max()
            r = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-10})
            bv = r.x
            if abs(bv - old) > 1e-9 and r.fun * dx < c5 - 1e-14:
                d = bv - old
                h[i] = bv
                corr = corr + d * coeff
                corr[ci] -= d * d
                c5 = r.fun * dx
                imp = True
        if not imp:
            break
    return h, c5

def coord_cont_fast(h, n, t_end, max_pass=2000):
    """Faster: only perturb active coordinates (near 0 or 1 boundary)"""
    dx = 2.0 / n
    h = h.copy(); corr = corr_of(h); c5 = corr.max() * dx
    lags = np.arange(-(n-1), n)
    ci = n - 1
    for _ in range(max_pass):
        if time.time() > t_end:
            break
        imp = False
        # Focus on coordinates that are near 0 or 1
        for i in range(n):
            old = h[i]
            if old < 0.02 or old > 0.98:
                continue
            ip = i + lags; mp = (ip >= 0) & (ip < n)
            im = i - lags; mm = (im >= 0) & (im < n)
            coeff = np.zeros(2*n-1)
            coeff[mm] += (1.0 - h[im[mm]])
            coeff[mp] -= h[ip[mp]]
            def obj(v):
                d = v - old
                cn = corr + d * coeff
                cn[ci] -= d * d
                return cn.max()
            r = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-11})
            bv = r.x
            if abs(bv - old) > 1e-9 and r.fun * dx < c5 - 1e-14:
                d = bv - old
                h[i] = bv
                corr = corr + d * coeff
                corr[ci] -= d * d
                c5 = r.fun * dx
                imp = True
        if not imp:
            break
    return h, c5

def interp_h(h, nn):
    no = len(h)
    xo = np.linspace(0, 2, no, endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return bin_proj(np.clip(np.interp(xn, xo, h), 0, 1), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def el():
        return time.time() - t0

    best_h, best_c5, best_n = None, np.inf, 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(160) * 0.5
        h0[0::2] = 1.0
    n0 = len(h0)

    # Phase 1: Binary local search (most critical)
    hb = bin_proj(h0, n0)
    cb, c5b = ls_binary(hb, n0, t0 + budget_s * 0.25, max_sw=50000)
    if c5b < best_c5:
        best_h, best_c5, best_n = cb, c5b, n0

    # Phase 2: Continuous refinement on best binary
    hc, c5c = coord_cont(cb, n0, t0 + budget_s * 0.35, max_pass=400)
    if c5c < best_c5:
        best_h, best_c5, best_n = hc, c5c, n0
    else:
        hc, c5c = coord_cont(best_h, best_n, t0 + budget_s * 0.35, max_pass=400)
        if c5c < best_c5:
            best_h, best_c5, best_n = hc, c5c, best_n

    # Phase 3: Multi-start with kicks (escape local optima)
    n_kicks = 30
    for k in range(n_kicks):
        if el() > budget_s * 0.60:
            break
        # Random kick: flip 4-8 random pairs
        n_flip = 4 + (k % 5)
        hp = best_h.copy()
        ones_p = np.where(hp > 0.5)[0]
        zeros_p = np.where(hp < 0.5)[0]
        if len(ones_p) < n_flip or len(zeros_p) < n_flip:
            continue
        fa = np.random.choice(ones_p, n_flip, replace=False)
        fb = np.random.choice(zeros_p, n_flip, replace=False)
        hp[fa] = 0.0; hp[fb] = 1.0
        hp = bin_proj(hp, best_n)
        h_try, c5_try = ls_binary(hp, best_n, t0 + budget_s * 0.60, max_sw=8000)
        if c5_try < best_c5:
            best_c5, best_h, best_n = c5_try, h_try, len(h_try)
            # Quick continuous refinement
            hc2, c5c2 = coord_cont(h_try, best_n, t0 + budget_s * 0.60, max_pass=100)
            if c5c2 < best_c5:
                best_c5, best_h = c5c2, hc2

    # Phase 4: Try different resolutions
    for nt in [120, 200, 240, 320, 400]:
        if el() > budget_s * 0.80:
            break
        ht = interp_h(best_h, nt)
        ht, c5t = ls_binary(ht, nt, t0 + budget_s * 0.82, max_sw=10000)
        ht, c5t = coord_cont(ht, nt, t0 + budget_s * 0.84, max_pass=200)
        if c5t < best_c5:
            best_c5, best_h, best_n = c5t, ht, nt

    # Phase 5: More binary local search at best resolution
    hb2 = bin_proj(best_h, best_n)
    hb2, c5b2 = ls_binary(hb2, best_n, t0 + budget_s * 0.88, max_sw=30000)
    if c5b2 < best_c5:

    # Phase 6: Final continuous refinement
    best_h, best_c5 = coord_cont(best_h, best_n, t0 + budget_s * 0.95, max_pass=800)
    # One more pass
    best_h, best_c5 = coord_cont_fast(best_h, best_n, t0 + budget_s * 0.98, max_pass=500)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
