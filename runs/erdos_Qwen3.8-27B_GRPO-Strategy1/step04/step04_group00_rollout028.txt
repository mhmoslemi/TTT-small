import numpy as np
from scipy.optimize import minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def grad_smooth_c5(h, T):

    #   Additional terms: v*(1-h[i+s]) [if valid] + h[i-s]*(1-v) - h[i-s]*1 = v*(1-h[i+s]) - v*h[i-s]

    #   With h[i]=v instead of h[i]=0:
    #   = sum_{j≠i} h[j]*(1-h[j+s]) + v*(1-h[i+s]) + sum_{j≠i} h[j]*(-1)*[j+s=i]
    #   The last term: when j+s=i, j=i-s, contribution is -h[i-s] (was 0 before since h[i]=0, now (1-h[i])=(1-v))
    #   Wait no. Let me restart.
    #   corr[m] = sum_j h[j] * (1-h[j+s])
    #   With h[i]=v: the j=i term gives v*(1-h[i+s]) [if i+s valid]

    #   With h[i]=0: j=i term gives 0, j=i-s term gives h[i-s]*(1-0)=h[i-s]
    #   Difference: v*(1-h[i+s])*[valid] + h[i-s]*(1-v)*[valid] - h[i-s]*[valid]

    #   = v * [(1-h[i+s])*[valid] - h[i-s]*[valid]]
    
    coeff = np.zeros(2*n-1)
    for m_idx in range(2*n-1):
        s = m_idx - (n-1)
        c = 0.0
        if 0 <= i+s < n:
            c += (1.0 - h[i+s])
        if 0 <= i-s < n:
            c -= h[i-s]
        coeff[m_idx] = c

    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

1. Compute corr once
2. For each i (in random order):
   a. Compute coeff_i (O(n))
   b. Find optimal h[i] (1D maximize)
   c. Update corr += (new_h[i] - old_h[i]) * coeff_i
3. End of sweep: recompute corr from scratch (to avoid accumulated error)

This makes each coordinate update O(n) instead of O(n) for corr + O(n) for minimize_scalar = still O(n) but with smaller constant.

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def grad_smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wv = w[m_idx]
        if wv < 1e-12:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                grad[:L] += wv * (1.0 - h[s:])
                grad[s:] -= wv * h[:L]
        else:
            ss = -s
            L = n - ss
            if L > 0:
                grad[ss:] += wv * (1.0 - h[:L])
                grad[:L] -= wv * h[ss:]
    return grad * dx

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h0 = best_h.copy()
            h0[i] = 0.0
            corr0 = np.correlate(h0, 1.0 - h0, mode='full')
            coeff = np.zeros(2 * n - 1)
            for m_idx in range(2 * n - 1):
                s = m_idx - (n - 1)
                c = 0.0
                if 0 <= i + s < n:
                    c += (1.0 - best_h[i + s])
                if 0 <= i - s < n:
                    c -= best_h[i - s]
                coeff[m_idx] = c
            def f1d(v):
                return np.max(corr0 + v * coeff) * (2.0 / n)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

        elif mod == 2:

def resample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old + 1)
    h_mid = (np.concatenate([[h_old[0]], (h_old[:-1] + h_old[1:]) / 2, [h_old[-1]]]))
    x_mid = np.linspace(0, 2, len(h_mid))
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_mid, h_mid, left=h_old[0], right=h_old[-1])
    return project_h(h_new)
