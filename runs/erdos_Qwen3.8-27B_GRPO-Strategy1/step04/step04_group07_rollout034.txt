    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    bh = h.copy()

    bh = np.zeros(n)
    bh[sorted_idx[:n//2]] = 1.0

    c5 = float(np.max(corr)) * dx
    best_h = bh.copy()
    best_c5 = c5
    m = 2 * n - 1
    ks = np.arange(-(n - 1), n)
    it = 0
    while time.time() < t_end:
        it += 1
        temp = t0_temp * (t1_temp / t0_temp) ** (it / max(1, int((t_end - time.time()) / 0.5)))
        temp = max(temp, t1_temp)
        # Pick random pair to swap
        i, j = np.random.choice(n, 2, replace=False)
        if bh[i] == bh[j]:
            continue
        di = bh[j] - bh[i]  # change at i
        # Compute delta corr
        ip, im = i + ks, i - ks
        jp, jm = j + ks, j - ks
        mpi = (ip >= 0) & (ip < n)
        mmi = (im >= 0) & (im < n)
        mpj = (jp >= 0) & (jp < n)
        mmj = (jm >= 0) & (jm < n)
        ci = np.zeros(m)
        ci[mpi] = 1.0 - bh[ip[mpi]]
        ci[mmi] -= bh[im[mmi]]
        cj = np.zeros(m)
        cj[mpj] = 1.0 - bh[jp[mpj]]
        cj[mmj] -= bh[jm[mmj]]
        delta = di * ci - di * cj  # since delta_j = -delta_i
        new_corr = corr + delta
        new_c5 = float(np.max(new_corr)) * dx
        if new_c5 < c5 or np.random.rand() < np.exp(-(new_c5 - c5) / temp):
            corr = new_corr
            bh[i], bh[j] = bh[j], bh[i]
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = bh.copy()
    return best_h, best_c5

def slsqp_refine(h, n, T, maxiter=500):
    dx = 2.0 / n
    m = 2 * n - 1
    def soft_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(corr)
        return float(np.log(np.sum(np.exp(T * (corr - mx)))) / T + mx) * dx
    res = minimize(soft_obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

    # Phase 8: Final CD
    best_h, best_c5 = fast_cd(best_h, len(best_h), time.time() + 4.0, max_sweeps=300)
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
