def slsqp_opt(h, T, n, maxiter=800):
    def obj(x):
        return smooth_obj(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def structured_basin_hop(h, n, t_end, rng, n_iters=200):

    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5

        if time.time() > t_end:

        cool = 1.0 - 0.9 * (it / max(n_iters, 1))
        step = 0.06 * (0.3 + 0.7 * cool)

        if pert_type == 0:
            # Global sinusoidal perturbation
            freq = rng.integers(2, n // 4)
            phase = rng.uniform(0, 2 * np.pi)
            amp = step * rng.uniform(0.5, 1.5)
            pert = work + amp * np.sin(2 * np.pi * freq * np.linspace(0, 1, n) + phase)
        elif pert_type == 1:

            pert[start:start+block_len] = 1.0 - pert[start:start+block_len]
            pert += step * 0.3 * rng.standard_normal(n)
        elif pert_type == 2:

            pert = work + step * rng.standard_normal(n)
        else:

            width = rng.integers(n // 20, n // 5)
            kernel = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
            pert = work + step * 2 * kernel * (rng.choice([-1, 1]))

            work, work_c5 = hc, cv
