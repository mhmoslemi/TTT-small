            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final CD
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=500)

    best_h = proj_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
