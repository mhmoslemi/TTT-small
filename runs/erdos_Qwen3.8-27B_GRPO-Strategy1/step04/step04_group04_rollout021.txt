import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(corr, T):
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    e = np.exp(shifted)
    s = np.sum(e)
    return m + np.log(s) / T, e / s

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, _ = smooth_max(corr, T)

    else:
        if s < 1e-13:

def active_set_slsqp(h, n, top_k=3, maxiter=2000):
    corr = corr_full(h)
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    dx = 2.0 / n
    def obj(x):
        c = corr_full(x)[sorted_idx]
        return float(np.max(c)) * dx
    def obj_grad(x):
        c = corr_full(x)[sorted_idx]
        k_max = sorted_idx[np.argmax(c)]
        lag = k_max - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                g[:n-lag] = (1.0 - x[lag:n])
                g[lag:] += (-x[:n-lag])
        else:
            nl = -lag
            if n - nl > 0:
                g[nl:] = (1.0 - x[:n-nl])
                g[:n-nl] += (-x[nl:])
        return g * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

        if time.time() > t0 + budget_s * 0.88:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n_try)
        if time.time() < t0 + budget_s * 0.88:
            hc, c5v = nesterov_opt(h_interp, n_try, time.time() + 15, T0=100, Tmax=20000)
            if c5v < best_c5:
                hc, c5v = coord_descent(hc, n_try, t0 + budget_s * 0.88, max_sweeps=30)
                best_h, best_c5, n = hc, c5v, n_try

    # Phase 7: Final active-set + coord polish
    if time.time() < t0 + budget_s * 0.95:
        hc = active_set_slsqp(best_h, n, top_k=3, maxiter=3000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=30)

    # Final subgradient polish
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = subgrad_exact(best_h, n, t0 + budget_s * 0.995, lr0=0.0003)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
