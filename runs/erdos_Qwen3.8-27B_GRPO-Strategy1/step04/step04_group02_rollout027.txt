def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Proportional adjustment
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = h + room * (target - s) / tot
    else:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)

        if s2 < target:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (target - s2) / tot
    return h

import numpy as np
from scipy.optimize import minimize_scalar
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = h + room * (target - s) / tot
    else:
        h = h * (target / max(s, 1e-15))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if s2 < target:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (target - s2) / tot
    return h

def grad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    omh = 1.0 - h
    if k == 0:
        g[:] = omh[:] - h[:]
    elif k > 0:
        g[k:] += omh[:n - k]
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += omh[-k:]
        g[-k:] -= h[:n + k]
    return g

def active_set_lp(h, n, top_k=15):
    corr = compute_corr(h)
    max_c = corr.max()
    idx = np.argsort(corr)[::-1][:top_k]
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
    ]
    for m in idx:
        k = m - (n - 1)
        g = grad_for_lag(h, k) * dx
        c_val = corr[m]
        constraints.append(c_val + g @ (h_var - h) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return h, max_c
    if h_var.value is None:
        return h, max_c
    h_new = np.clip(h_var.value, 0.0, 1.0)
    h_new = project(h_new, n)
    return h_new, compute_c5(h_new)

def coord_opt(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        vals = [0.0]
        if 0 <= i - k < n:
            vals[0] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            vals[0] += (-h[i + k])
        d[n - 1 + k] = vals[0]
        vals2 = [0.0]
        if 0 <= i + k < n:
            vals2[0] += (1.0 - h[i + k])
        if 0 <= i - k < n:
            vals2[0] += (-h[i - k])
        d[n - 1 - k] = vals2[0]
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=40, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-13:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

    # Phase 5: Multi-resolution
    for n_try in [200, 160, 280]:
        if elapsed() > budget_s * 0.82:
            break
        h_interp = interpolate_h(best_h, n_try)
        for rk in [10, 20]:
            for _ in range(10):
                if elapsed() > budget_s * 0.84:
                    break
                h_new, c5_new = active_set_lp(h_interp, n_try, top_k=rk)
                if c5_new < compute_c5(h_interp) - 1e-14:
                    h_interp = h_new
                else:
                    break
        h_interp, c5_interp = coord_descent(h_interp, n_try, t0 + budget_s * 0.85, max_sweeps=12, rng=rng)
        if c5_interp < best_c5 - 1e-14:
            best_h, best_c5, n = h_interp, c5_interp, n_try

    # Phase 6: Final tight optimization
    t_final = t0 + budget_s * 0.93
    best_h, best_c5 = coord_descent(best_h, n, t_final, max_sweeps=25, rng=rng)
    for rk in [25, 40, 60]:
        if elapsed() > budget_s * 0.96:
            break
        for _ in range(8):
            if elapsed() > budget_s * 0.97:
                break
            h_new, c5_new = active_set_lp(best_h, n, top_k=rk)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new
            else:
                break

    # Final coordinate polish
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=10, rng=rng)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
