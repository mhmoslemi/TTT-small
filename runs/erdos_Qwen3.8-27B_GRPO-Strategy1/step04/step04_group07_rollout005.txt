def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp_opt(h0, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0.copy()

def interp_h(h, n_new):

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def basin_hop(work_h, work_c5, n, t_end, max_iters=100):
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.1, 1.0 - 0.9 * it / max_iters)
        # Mix of continuous and binary perturbation
        if np.random.rand() < 0.4:
            # Binary perturbation: flip k random bits
            k_flip = max(1, int(n * 0.01 * cool))
            h_p = work_h.copy()
            b = to_binary(h_p, n)
            flip_idx = np.random.choice(n, k_flip, replace=False)
            for idx in flip_idx:
                b[idx] = 1.0 - b[idx]
            # Fix sum
            n_ones = int(b.sum())
            if n_ones != n // 2:
                # Adjust
                excess = n_ones - n // 2
                if excess > 0:
                    ones_mask = b == 1
                    zero_mask = b == 0
                    if zero_mask.sum() >= excess:
                        ones_to_flip = np.where(ones_mask)[0]
                        zeros_available = np.where(zero_mask)[0]
                        to_flip = np.random.choice(ones_to_flip, excess, replace=False)
                        b[to_flip] = 0
                elif excess < 0:
                    zeros_to_flip = np.where(b == 0)[0]
                    to_flip = np.random.choice(zeros_to_flip, -excess, replace=False)
                    b[to_flip] = 1
            h_p = b + 0.1 * np.random.randn(n) * cool
        else:
            step = 0.05 + 0.15 * cool
            h_p = work_h + step * np.random.randn(n)
        h_p = project(h_p, n)
        
        T_use = int(100 + 2000 * cool)
        hc = slsqp_opt(h_p, T_use, n, maxiter=300)
        c5v = compute_c5(hc)
        
        temp = 0.003 * cool + 0.0003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()
    return best_h, best_c5
