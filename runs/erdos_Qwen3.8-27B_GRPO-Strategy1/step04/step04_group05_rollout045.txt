                grad[M:] -= w * h[:N-M]
    return grad, peak

def adam_loop(h0, N, target, deadline, lr0=0.04, n_top=100, decay=0.995, b1=0.85, b2=0.999):
    h = h0.copy()
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_at_active(h, N, n_top)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= decay
    return best_h, best_peak * dx

def binary_swap_search(h0, N, deadline, max_sweeps=200):
    h = (h0 > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n1]] = 1
    c = compute_corr(h)
    cur = float(c.max())
    center = N - 1
    L_vals = np.arange(2*N-1) - center
    pi = np.arange(N)[:, None]
    pL = pi - L_vals[None, :]
    pP = pi + L_vals[None, :]
    vL = (pL >= 0) & (pL < N)
    vP = (pP >= 0) & (pP < N)
    hL = np.where(vL, h[np.clip(pL, 0, N-1)], 0.0)
    hP = np.where(vP, h[np.clip(pP, 0, N-1)], 0.0)
    D = (1.0 - hL) * vL.astype(float) - hP * vP.astype(float)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_new = cur
        best_pq = None
        for p in ones:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            peaks = cand.max(axis=1)
            mi = np.argmin(peaks)
            if peaks[mi] < best_new:
                best_new = float(peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = compute_corr(h)
        cur = float(c.max())
        # Update D for changed positions
        for pos in [best_pq[0], best_pq[1]]:
            pL2 = pos - L_vals
            pP2 = pos + L_vals
            vL2 = (pL2 >= 0) & (pL2 < N)
            vP2 = (pP2 >= 0) & (pP2 < N)
            hL2 = np.where(vL2, h[np.clip(pL2, 0, N-1)], 0.0)
            hP2 = np.where(vP2, h[np.clip(pP2, 0, N-1)], 0.0)
            D[pos, :] = (1.0 - hL2) * vL2.astype(float) - hP2 * vP2.astype(float)
    return h, compute_c5(h, N)

def anneal_binary(h0, N, n_iters, rng, T0=2.0, T1=0.01):
    h = (h0 > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n1]] = 1
    best = h.copy()
    c = compute_corr(h)
    best_pk = float(c.max())
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        T = T0 * (T1/T0)**(it/n_iters)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c = compute_corr(h)
        new_pk = float(c.max())
        d = new_pk - cur_pk
        if d < 0 or (T > 0.005 and rng.random() < np.exp(-d/T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_pk * (2.0/N)
