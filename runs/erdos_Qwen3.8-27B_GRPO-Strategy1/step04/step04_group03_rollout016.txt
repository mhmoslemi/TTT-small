            hc = slsqp(h_p, 500, mi=300, ft=1e-12)

            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 5: Multi-n exploration (18% budget)
    t_multi_end = t0 + budget_s * 0.85
    for n_try in [128, 160, 192, 256, 288, 320, 384]:
        if elapsed() > t_multi_end:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project(h_i, n_try / 2.0)
        for T in [200, 2000]:
            if elapsed() > t_multi_end:
                break
            hc = slsqp(h_i, T, mi=500, ft=1e-13)
            hc, c5v = cd(hc, time.time() + 5.0, ns=80)
            if c5v < best_c5:
                best_c5, best_h, n = c5v, hc, len(hc)

    # Phase 6: Final intensive CD + SLSQP (13% budget)
    t_final = t0 + budget_s * 0.97
    best_h, best_c5 = cd(best_h, t_final, ns=800)
    if elapsed() < t_final:
        for T in [5000, 30000, 100000]:
            if elapsed() > t_final:
                break
            hc = slsqp(best_h, T, mi=1500, ft=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v

    # Final micro-CD
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = cd(best_h, t0 + budget_s * 0.995, ns=100)

    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
