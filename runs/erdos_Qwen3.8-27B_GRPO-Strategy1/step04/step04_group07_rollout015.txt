import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, n, t_end, max_sweeps=5000):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr)
    ks = np.arange(-(n - 1), n)
    c0 = n - 1
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    grc = 1.0 - 1.0 / gr
    improved_total = True
    sweep = 0
    while improved_total and sweep < max_sweeps:
        if time.time() > t_end:
            break
        improved_total = False
        for i in range(n):
            old = best_h[i]
            idx1 = i + ks
            idx2 = i - ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_coeff = coeff[c0]
            base = corr.copy()
            def max_at(v):
                d = v - old
                m = np.max(base + d * coeff)
                c0v = base[c0] + d * c0_coeff - d * d
                return max(m, c0v)
            lo, hi = 0.0, 1.0
            c1 = lo + grc * (hi - lo)
            c2 = hi - grc * (hi - lo)
            f1, f2 = max_at(c1), max_at(c2)
            for _ in range(28):
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = lo + grc * (hi - lo)
                    f1 = max_at(c1)
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = hi - grc * (hi - lo)
                    f2 = max_at(c2)
            best_v = (lo + hi) * 0.5
            best_val = max_at(best_v)
            for vc in (0.0, 1.0):
                fv = max_at(vc)
                if fv < best_val:
                    best_val, best_v = fv, vc
            if best_val < best_c5 - 1e-13:
                d = best_v - old
                corr += d * coeff
                corr[c0] -= d * d
                best_h[i] = best_v
                best_c5 = best_val
                improved_total = True
        sweep += 1
        if sweep % 3 == 0:

            best_c5 = np.max(corr)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr)
    return best_h, best_c5 * dx

def slsqp_opt(h0, T, n, maxiter=500, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def basin_hop(work_h, work_c5, n, t_end, max_iters=300):
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.03, 1.0 - 0.9 * it / max_iters)
        step = 0.03 + 0.22 * cool
        h_p = work_h + step * np.random.randn(n)
        h_p = project(h_p, n)
        T_use = int(100 + 3000 * cool)
        try:
            hc = slsqp_opt(h_p, T_use, n, maxiter=250)
            c5v = compute_c5(hc)
        except Exception:
            continue
        temp = 0.004 * cool + 0.0004
        gap = c5v - work_c5
        if c5v < work_c5 or (gap < 0.01 and np.random.rand() < np.exp(-gap / max(temp, 1e-9))):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def interp_h(h, n_new):

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)
