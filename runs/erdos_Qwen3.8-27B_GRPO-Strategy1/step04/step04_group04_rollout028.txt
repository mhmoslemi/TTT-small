import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def active_set_opt(h, n, top_k=10, T=200, maxiter=1000):
    """Optimize focusing on the top_k largest correlation values."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    indices = np.argsort(corr)[-top_k:]
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')[indices]
        # Use max as objective (piecewise linear)
        return float(np.max(c)) * dx
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def weighted_smooth_opt(h, n, T=500, maxiter=1000):
    """Use weighted smooth max."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    weights = np.exp(shifted)
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mm = np.max(c)
        sh = np.clip(T * (c - mm), -500, 500)
        return (mm + np.log(np.sum(np.exp(sh))) / T) * dx
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def pair_swap_descent(h, n, t_end, max_rounds=50):
    """Try pairwise swaps: h[i]+=e, h[j]-e for pairs of near-boundary points."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        # Try all pairs is O(n^2) which is too much for n=180
        # Instead, try random pairs
        n_pairs = min(500, n * 5)
        for _ in range(n_pairs):
            if time.time() > t_end:
                break
            i, j = np.random.choice(n, 2, replace=False)
            # small swap

                h_try = best_h.copy()
                h_try[i] += e

                    best_h = h_try
                    improved = True
                    break
        if not improved:

    return best_h, best_c5

= (t*(1-h[i+off]) + h[i-off]*(1-t)) - (v*(1-h[i+off]) + h[i-off]*(1-v))  [for off≠0, with boundary checks]
= (t-v)*(1-h[i+off]) + h[i-off]*(-t+v)
= (t-v)*[(1-h[i+off]) - h[i-off]]
= (t-v) * a[k]

And for k=n-1 (off=0):
Δc[n-1] = t*(1-t) - v*(1-v) = (t-v)*(1-v-t)

So I can update c_full in O(n) per coordinate! That makes the whole thing O(n²) per sweep.

            #   if 0<=i+off<n: v*(1-h[i+off]) [or v*(1-v) if off=0]
            #   if 0<=i-off<n and off!=0: h[i-off]*(1-v)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def precise_coord_descent(h, n, t_end, max_sweeps=3000, shuffle=True):
    best_h = h.copy()
    dx = 2.0 / n
    c_full = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    best_c5 = float(c_full.max()) * dx
    idx = np.arange(2 * n - 1)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n) if shuffle else np.arange(n)
        for i in order:
            v = best_h[i]
            a = np.zeros(2 * n - 1)
            contrib = np.zeros(2 * n - 1)
            for k in idx:
                off = k - (n - 1)
                if off == 0:
                    a[k] = 1.0
                    contrib[k] = v * (1.0 - v)
                else:
                    p1 = i + off
                    p2 = i - off
                    if 0 <= p1 < n:
                        a[k] += (1.0 - best_h[p1])
                        contrib[k] += v * (1.0 - best_h[p1])
                    if 0 <= p2 < n:
                        a[k] += (-best_h[p2])
                        contrib[k] += best_h[p2] * (1.0 - v)
            c_base = c_full - contrib
            cn1_base = c_base[n - 1]
            def f_1d(t):
                cv = c_base + a * t
                cv[n - 1] = cn1_base + t - t * t
                return float(cv.max())
            res = minimize_scalar(f_1d, bounds=(0.0, 1.0), method='bounded',
                               options={'xatol': 1e-12})
            t_opt = res.x
            f_val = f_1d(t_opt)
            if f_val * dx < best_c5 - 1e-14:
                delta = t_opt - v
                if abs(delta) > 1e-13:
                    for k in idx:
                        if k == n - 1:
                            c_full[k] = cn1_base + t_opt - t_opt * t_opt
                        else:
                            c_full[k] = c_base[k] + a[k] * t_opt
                    best_h[i] = t_opt
                    best_c5 = f_val * dx
                    improved = True
        if not improved:
            break
        if sweep % 20 == 0:
            best_h = proj(best_h, n)
            c_full = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
            best_c5 = float(c_full.max()) * dx
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def weighted_smooth_slsqp(h, n, T=1000, maxiter=800):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = np.clip(T * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj(np.clip(res.x, 0, 1), n)

def active_set_slsqp(h, n, top_k=8, maxiter=1200):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    indices = np.argsort(corr)[-top_k:]
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')[indices]
        return float(c.max()) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj(np.clip(res.x, 0, 1), n)

def pair_swap_local(h, n, t_end, n_iters=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() > t_end:
            break
        i, j = np.random.choice(n, size=2, replace=False)
        if abs(best_h[i] - best_h[j]) < 0.02:
            continue
        for e in [0.02, 0.05, 0.1, 0.2]:
            h_try = best_h.copy()
            h_try[i] += e
            h_try[j] -= e
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                break
    return best_h, best_c5

def basin_hopping(h, n, t_end, n_iters=40):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    best_h = work_h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.88 * (it / float(n_iters))
        step = 0.01 + 0.06 * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = proj(h_pert, n)
        try:
            hc = weighted_smooth_slsqp(h_pert, n, T=800, maxiter=300)
            c5v = compute_c5(hc)
            temp = 0.002 * cool + 0.0001
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass
    return best_h, best_c5, work_c5
