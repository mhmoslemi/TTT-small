    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()

    if abs(diff) < 1e-12:

    if diff > 0:

        if tot < 1e-12:

def coord_update(h, corr, i, n):
    """Compute best value for h[i] given current corr, return (best_v, new_corr or None, new_c5)"""
    old = h[i]
    dx = 2.0 / n
    lag_offset = n - 1
    # Compute coefficient vector
    ks = np.arange(2 * n - 1)
    lags = ks - lag_offset  # -(n-1) to n-1

    idx1 = i - lags  # needs to be in [0, n-1]
    v1 = (idx1 >= 0) & (idx1 < n)
    
    # Coefficient from h[i] appearing in (1-h[n-lag]) term: -h[i + lag]
    idx2 = i + lags  # needs to be in [0, n-1]
    v2 = (idx2 >= 0) & (idx2 < n)
    
    coeff = np.zeros(2 * n - 1)
    coeff[v1] += 1.0 - h[idx1[v1]]
    coeff[v2] -= h[idx2[v2]]
    
    # Objective: max_k (corr[k] + (v - old) * coeff[k])
    # = max_k (corr[k] - old*coeff[k] + v*coeff[k])
    base = corr - old * coeff
    
    def obj(v):
        return np.max(base + v * coeff)
    
    # Evaluate at boundaries and use minimize_scalar
    f0 = obj(0.0)
    f1 = obj(1.0)
    best_v = 0.0 if f0 <= f1 else 1.0
    best_f = min(f0, f1)
    
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    if res.fun < best_f:

    return best_v, best_f, coeff

                corr = corr + (best_v - old) * coeff
                best_c5 = best_f * dx
                improved = True
        if not improved:
            break
    return best_h, best_c5

    if s > target:
        ones = np.where(h == 1)[0]
        to_remove = rng.choice(ones, s - target, replace=False)
        h[to_remove] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        to_add = rng.choice(zeros, target - s, replace=False)
        h[to_add] = 1
    return h.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = random_binary(160, seed)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Fast coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.3, max_sweeps=3000, temperature=0.0)
    
    # Phase 2: SLSQP with smooth max (multiple T values)
    for T in [50, 200, 1000, 5000]:
        if elapsed() > budget_s * 0.40:
            break
        try:
            hc = slsqp_smooth(best_h, T, n, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass
    
    # Phase 3: More coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.50, max_sweeps=3000, temperature=0.0)
    
    # Phase 4: Diversified search - try random perturbations
    work_h, work_c5 = best_h.copy(), best_c5
    n_perturb = 50
    t_perturb_end = t0 + budget_s * 0.70
    for it in range(n_perturb):
        if time.time() > t_perturb_end:
            break
        # Random perturbation
        step_size = 0.1 * (1.0 + 0.5 * np.sin(it * 0.3))
        h_p = best_h + step_size * np.random.randn(n)
        h_p = project(h_p, n)
        
        # Quick coordinate descent
        T_use = max(1.0, 5.0 - it * 0.05)
        h_c, c5_c = coord_descent(h_p, n, t0 + budget_s * 0.70, max_sweeps=200, temperature=T_use)
        
        if c5_c < work_c5:
            work_h, work_c5 = h_c, c5_c
            if c5_c < best_c5:
                best_h, best_c5 = work_h, work_c5
    
    # Phase 5: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.85, max_sweeps=5000, temperature=0.0)
    
    # Phase 6: Try alternate resolutions
    for n_try in [128, 200, 256, 320]:
        if elapsed() > budget_s * 0.93:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.94, max_sweeps=1000, temperature=0.0)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try
    
    # Phase 7: Final polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=2000, temperature=0.0)

            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass
    
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
