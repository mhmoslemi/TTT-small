def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, n, t_end, max_sweeps=5000, tol=1e-13):
    dx = 2.0 / n
    m = 2 * n - 1
    best_h = h.astype(float).copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    best_c5 = float(np.max(corr)) * dx
    j_arr = np.arange(m)
    prev_recompute = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i + j_arr - (n - 1)
            idx2 = i - j_arr + (n - 1)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(m)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            a = corr - old * coeff
            b = coeff
            f0 = float(np.max(a))
            f1 = float(np.max(a + b))
            lo, hi = 0.0, 1.0
            for _ in range(25):
                third = (hi - lo) / 3.0
                m1 = lo + third
                m2 = hi - third
                if float(np.max(a + m1 * b)) < float(np.max(a + m2 * b)):
                    hi = m2
                else:
                    lo = m1
            v_best = 0.5 * (lo + hi)
