    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()

def coord_opt_fixed(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += (1.0 - h[i - k])
        if 0 <= i + k < n:
            dk += (-h[i + k])
        d[n - 1 + k] = dk
        d[n - 1 - k] = dk  # symmetric
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fixed(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=800, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def slsqp_direct(h, n, maxiter=1500, ftol=1e-15):

def interpolate_h(h_old, n_new):

def fw_gradient(h, T):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= wsum
    omh = 1.0 - h
    g = np.zeros(n)
    for ell in range(-(n-1), n):
        wv = w[n-1+ell]
        if abs(wv) < 1e-14:
            continue
        # Term 1: (1-h[i+ell]) * I(0<=i+ell<n)
        if ell >= 0:
            # i+ell < n => i < n-ell; i+ell >= 0 always for i>=0
            g[:n-ell] += wv * omh[ell:]
        else:
            # i+ell >= 0 => i >= -ell; i+ell < n always for i<n
            e = -ell
            g[e:] += wv * omh[:n-e]

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        h = np.clip(h - h * (diff / max(s, 1e-14)), 0, 1)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = np.clip(h + room * (diff2 / tot), 0, 1)
    return h

def smooth_c5(h, T):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def fw_gradient(h, T):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= wsum
    omh = 1.0 - h
    g = np.zeros(n)
    for ell in range(-(n - 1), n):
        wv = w[n - 1 + ell]
        if abs(wv) < 1e-14:
            continue
        if ell >= 0:
            g[:n - ell] += wv * omh[ell:]
            g[ell:] += wv * (-h[:n - ell])
        else:
            e = -ell
            g[e:] += wv * omh[:n - e]
            g[:n - e] += wv * (-h[e:])
    return g

def fw_lmo(g, n):
    order = np.argsort(g)
    v = np.zeros(n)
    v[order[:n // 2]] = 1.0
    return v

def fw_solve(h, n, t_end, T_start=30, T_end=5000, n_iters=400):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_fw = h.copy()
    for it in range(n_iters):
        if time.time() > t_end:
            break
        T = T_start * (T_end / T_start) ** (it / max(n_iters - 1, 1))
        g = fw_gradient(h_fw, T)
        v = fw_lmo(g, n)
        best_gamma = 0.0
        best_obj = smooth_c5(h_fw, T)
        for gamma in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            h_try = (1 - gamma) * h_fw + gamma * v
            h_try = project(h_try, n)
            obj_try = smooth_c5(h_try, T)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_gamma = gamma
        if best_gamma > 0:
            h_fw = (1 - best_gamma) * h_fw + best_gamma * v
            h_fw = project(h_fw, n)
        else:
            step = 0.02 / (1 + it * 0.01)
            h_try = project(h_fw - step * g, n)
        c5_curr = compute_c5(h_fw)
        if c5_curr < best_c5 - 1e-14:
            best_h, best_c5 = h_fw.copy(), c5_curr
    return best_h, best_c5

def projected_grad(h, n, t_end, T=500, n_steps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_pg = h.copy()
    lr = 0.05
    for it in range(n_steps):
        if time.time() > t_end:
            break
        g = fw_gradient(h_pg, T)
        h_try = project(h_pg - lr * g, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h_pg = h_try
            best_h, best_c5 = h_pg.copy(), c5_try
        else:
            lr *= 0.7
            if lr < 1e-6:
                lr = 0.05
    return best_h, best_c5

def coord_opt_fixed(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += (1.0 - h[i - k])
        if 0 <= i + k < n:
            dk += (-h[i + k])
        d[n - 1 + k] = dk
        d[n - 1 - k] = dk
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fixed(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=800, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def random_start(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    freqs = rng.integers(1, 10, size=8)
    phases = rng.uniform(0, 2 * np.pi, size=8)
    amps = rng.uniform(0.05, 0.4, size=8)
    h = 0.5
    for j in range(8):
        h = h + amps[j] * np.sin(np.pi * freqs[j] * x + phases[j])
    return project(h, n)

def combined_opt(h, n, t_end, rng):
    best_h, best_c5 = h.copy(), compute_c5(h)
    for T in [50, 200, 800]:
        if time.time() > t_end:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=500, ftol=1e-13)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass
    tc1 = t0 + (t_end - t0) * 0.5
    if time.time() < t_end:
        fh, fc = fw_solve(best_h, n, tc1, T_start=50, T_end=3000, n_iters=200)
        if fc < best_c5:
            best_h, best_c5 = fh, fc
    if time.time() < t_end:
        best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=30, rng=rng)
    return best_h, best_c5

t0 = 0.0
