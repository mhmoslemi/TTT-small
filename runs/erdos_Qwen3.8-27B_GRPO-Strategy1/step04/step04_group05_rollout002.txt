        c = compute_corr(h)
        peak = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        dv1 = compute_dv1(h, N)
        dv1_ones = dv1[ones]
        dv1_zeros = dv1[zeros]
        best = peak
        best_pq = None
        for ip in range(len(ones)):
            base = c + dv1_ones[ip]
            npk = (base[None, :] - dv1_zeros).max(axis=1)
            mi = np.argmin(npk)
            if npk[mi] < best:
                best = float(npk[mi])
                best_pq = (int(ones[ip]), int(zeros[mi]))
        moved = False
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            moved = True
            stall = 0
        elif allow_2swap and rng is not None and sweep % 4 == 3 and len(ones) >= 2 and len(zeros) >= 2:
            for _ in range(30):
                p1, p2 = rng.choice(ones, 2, replace=False)
                q1, q2 = rng.choice(zeros, 2, replace=False)
                delta = dv1[p1] + dv1[p2] - dv1[q1] - dv1[q2]
                npk = float(np.max(c + delta))
                if npk < peak - 1e-12:
                    h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
                    moved = True
                    stall = 0
                    break
        if not moved:
            stall += 1
            if stall > 15:
                break
        if float(np.max(compute_corr(h))) < best_peak - 1e-12:
            best_peak = float(np.max(compute_corr(h)))
    return h, compute_c5(h, N)

def anneal_binary(h, N, n_iters, rng, deadline, T0=2.0, T1=0.01):
    h = make_binary(h, N)
    nw = 2 * N - 1
    best = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        p = int(ones[rng.randint(len(ones))])
        q = int(zeros[rng.randint(len(zeros))])
        h[p], h[q] = h[q], h[p]
        npk = float(np.max(compute_corr(h)))
        delta = npk - cur_pk
        if delta < 0 or (T > 0.008 and rng.random() < np.exp(-delta / T)):
            cur_pk = npk
            if npk < best_pk:
                best_pk = npk
                best = h.copy()
        else:
            h[p], h[q] = h[q], h[p]
    return best, best_pk * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def adam_continuous(h, N, target, deadline, k=40, lr0=0.03, decay=0.993):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    nw = 2 * N - 1
    center = N - 1
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                grad[L:] += wk * (1.0 - h[:N - L])
                grad[:N - L] -= wk * h[L:]
            elif L < 0 and -L < N:
                M = -L
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 150 == 0:
            lr *= decay
    return best_h, best_peak * dx

def make_init(N, pattern, rng):
    if pattern == 'alt':
        h = np.zeros(N)
        h[::2] = 1.0
    elif pattern == 'blocks':
        h = np.zeros(N)
        h[:N // 2] = 1.0
    elif pattern == 'half_alt':
        h = np.zeros(N)
        h[::4] = 1.0
        h[1::4] = 1.0
    else:
        h = (rng.random(N) > 0.5).astype(float)
    return make_binary(h, N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = make_binary(h0, N0)
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Steepest descent on initial binary
    d1 = deadline - 400
    h_s, c5_s = binary_descent(best_h, N0, max_sweeps=600, deadline=d1, allow_2swap=True, rng=rng)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Iterated anneal + descent
    d2 = deadline - 120
    for restart in range(30):
        if time.time() > d2:
            break
        remaining = d2 - time.time()
        ann_time = max(5.0, min(20.0, remaining * 0.3))
        n_ann = int(ann_time * 80)
        h_a, c5_a = anneal_binary(best_h, N0, n_ann, rng, deadline=time.time() + ann_time, T0=2.5, T1=0.005)
        h_s, c5_s = binary_descent(h_a, N0, max_sweeps=400, deadline=d2 - 5, allow_2swap=True, rng=rng)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        if time.time() > d2:
            break

    # Phase 3: Additional structured starts
    if time.time() < deadline - 80:
        for pat in ['alt', 'half_alt', 'blocks']:
            if time.time() > deadline - 80:
                break
            h_init = make_init(N0, pat, rng)
            h_s, c5_s = binary_descent(h_init, N0, max_sweeps=400, deadline=deadline - 75, allow_2swap=True, rng=rng)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s.copy()

    # Phase 4: Try nearby N values
    for N_try in [N0 + 2, N0 - 2, N0 + 4, N0 + 6]:
        if time.time() > deadline - 30 or N_try < 100:
            break
        if N_try % 2 != 0:
            N_try -= 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_b = make_binary(h_interp, N_try)
        h_s, c5_s = binary_descent(h_b, N_try, max_sweeps=300, deadline=deadline - 20, allow_2swap=True, rng=rng)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        if time.time() < deadline - 10:
            h_a, c5_a = anneal_binary(h_s, N_try, 800, rng, deadline=deadline - 5, T0=1.5, T1=0.005)
            h_s2, c5_s2 = binary_descent(h_a, N_try, max_sweeps=200, deadline=deadline - 3, allow_2swap=True, rng=rng)
            if c5_s2 < best_c5:
                best_c5, best_h = c5_s2, h_s2.copy()

    # Phase 5: Continuous Adam polish
    if time.time() < deadline - 5:
        Nf = len(best_h)
        h_pol = best_h.astype(float) + rng.randn(Nf) * 0.05
        h_pol = project_sum(h_pol, Nf / 2.0)
        h_ad, c5_ad = adam_continuous(h_pol, Nf, Nf / 2.0, deadline - 1, k=50, lr0=0.012, decay=0.995)
        h_bin = make_binary(h_ad, Nf)
        h_s, c5_s = binary_descent(h_bin, Nf, max_sweeps=200, deadline=deadline, allow_2swap=True, rng=rng)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
