```python
def project_onto_polytope(h, n):
    """Project h onto {0<=h_i<=1, sum(h)=n/2}"""
    h = h.copy()
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    # Then adjust sum
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to increase some elements
        room = 1.0 - h
        if room.sum() < diff:
            h = np.ones(n) * (target / n)
            return h
        # Find threshold tau such that sum(min(1, h+tau)) = target
        # h_i + tau where h_i < 1, clipped to 1
        # This is a water-filling problem
        candidates = np.sort(1.0 - h)  # amounts of room
        cumsum = np.cumsum(candidates)
        idx = np.searchsorted(cumsum, diff)
        if idx >= n:
            tau = diff / n
        else:
            remaining = diff - (cumsum[idx-1] if idx > 0 else 0)
            tau = remaining / (n - idx)
        h = np.clip(h + tau, 0, 1)
    else:
        # Need to decrease
        if h.sum() < -diff:  # can't decrease enough
            h = np.zeros(n)
            return h
        candidates = np.sort(h)
        cumsum = np.cumsum(candidates)
        idx = np.searchsorted(cumsum, -diff)
        if idx >= n:
            tau = -diff / n
        else:
            remaining = -diff - (cumsum[idx-1] if idx > 0 else 0)
            tau = remaining / (n - idx)
        h = np.clip(h - tau, 0, 1)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def coord_sweep(h, n, t_end):
    """One sweep of coordinate descent with incremental corr updates."""
    center = n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corr.max() * dx
    improved = False
    order = np.random.permutation(n)
    for i in order:
        if time.time() > t_end:
            break
        # Compute sensitivity a[j]
        a = np.zeros(2 * n - 1)
        a[center] = 1.0
        for k in range(1, n):
            jp = center + k
            jm = center - k
            if k <= i:
                a[jp] += (1.0 - h[i - k])
                a[jm] -= h[i - k]
            if i <= n - 1 - k:
                a[jp] -= h[i + k]
                a[jm] += (1.0 - h[i + k])
        hi_old = h[i]
        # Find optimal new value
        cb = corr - hi_old * a
        cb[center] += hi_old * hi_old
        
        def f1d(x):
            v = cb + x * a
            v[center] -= x * x
            return v.max()
        
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                            options={'xatol': 1e-12})
        x_new = res.x
        if abs(x_new - hi_old) > 1e-12:
            diff = x_new - hi_old
            corr += diff * a
            corr[center] -= diff * (x_new + hi_old)
            h[i] = x_new
            new_c5 = corr.max() * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
            improved = True
    h = project(h, n)
    return h, best_c5, improved, corr

def slsqp_smooth(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        ec = np.exp(T * (c * dx - m))
        w = ec / ec.sum()
        center = n - 1
        g = np.zeros(n)
        for j in range(2 * n - 1):
            if w[j] < 1e-15:
                continue
            k = j - center
            if k == 0:
                g += w[j] * dx * (1.0 - 2.0 * x)
            elif k > 0:
                g[k:] += w[j] * dx * (1.0 - x[:n-k])
                g[:n-k] -= w[j] * dx * x[k:]
            else:
                m2 = -k
                g[:n-m2] += w[j] * dx * (1.0 - x[m2:])
                g[m2:] -= w[j] * dx * x[:n-m2]
        return g
    def con_eq(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_eq}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)
```

```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def coord_sweep(h, n, t_end):
    center = n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corr.max() * dx
    improved = False
    order = np.random.permutation(n)
    is_c = np.zeros(2 * n - 1)
    is_c[center] = 1.0
    for i in order:
        if time.time() > t_end:
            break
        a = np.zeros(2 * n - 1)
        a[center] = 1.0
        for k in range(1, n):
            jp = center + k
            jm = center - k
            if k <= i:
                a[jp] += (1.0 - h[i - k])
                a[jm] -= h[i - k]
            if i <= n - 1 - k:
                a[jp] -= h[i + k]
                a[jm] += (1.0 - h[i + k])
        old = h[i]
        res = minimize_scalar(
            lambda x: (corr + (x - old) * a - (x - old) * (x - old) * is_c).max(),
            bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
        x_new = res.x
        if abs(x_new - old) > 1e-13:
            d = x_new - old
            corr += d * a
            corr[center] -= d * d
            h[i] = x_new
            improved = True
    h = project(h, n)
    return h, best_c5, improved

def slsqp_smooth(h, T, n, maxiter=400):
    dx = 2.0 / n
    center = n - 1
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        ec = np.exp(T * (c * dx - m))
        w = ec / ec.sum()
        g = np.zeros(n)
        for j in range(2 * n - 1):
            if w[j] < 1e-14:
                continue
            k = j - center
            if k == 0:
                g += w[j] * dx * (1.0 - 2.0 * x)
            elif k > 0:
                g[k:] += w[j] * dx * (1.0 - x[:n - k])
                g[:n - k] -= w[j] * dx * x[k:]
            else:
                m2 = -k
                g[:n - m2] += w[j] * dx * (1.0 - x[m2:])
                g[m2:] -= w[j] * dx * x[:n - m2]
        return g
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)
```
