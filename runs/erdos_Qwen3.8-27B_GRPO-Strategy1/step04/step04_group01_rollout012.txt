        frac = min((time.time() - t_start) / total_time, 1.0)
        T = T0 * (1.0 - frac) ** 2.0 + 1e-8
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if len(set([i1, i2, j1, j2])) < 4:
            continue
        cur_h[i1] = 0.0; dh[i1] = 1.0
        cur_h[i2] = 0.0; dh[i2] = 1.0
        cur_h[j1] = 1.0; dh[j1] = 0.0
        cur_h[j2] = 1.0; dh[j2] = 0.0
        new_c5 = float(np.max(np.correlate(cur_h, dh, mode='full'))) * dx
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[i1] = 1.0; dh[i1] = 0.0
            cur_h[i2] = 1.0; dh[i2] = 0.0
            cur_h[j1] = 0.0; dh[j1] = 1.0
            cur_h[j2] = 0.0; dh[j2] = 1.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    init_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    best_h = None
    best_c5 = 1.0

    def try_solution(h, n, t_limit, rng_local):
        nonlocal best_h, best_c5
        # Hill climb to local opt
        hc_t = min(t_limit, time.time() + (t_limit - time.time()) * 0.25)
        h, c5 = hill_climb(h, n, hc_t, max_sweeps=150)
        # SA single swaps
        sa_t = min(t_limit, time.time() + (t_limit - time.time()) * 0.35)
        if time.time() < sa_t:
            h, c5 = simulated_annealing(h, n, sa_t, rng_local, T0=0.005)
            h, c5 = hill_climb(h, n, min(t_limit, time.time() + (t_limit - time.time()) * 0.15), max_sweeps=80)
        # Multi-swap SA
        ms_t = min(t_limit, time.time() + (t_limit - time.time()) * 0.25)
        if time.time() < ms_t:
            h, c5 = multi_swap_sa(h, n, ms_t, rng_local, T0=0.012)
            h, c5 = hill_climb(h, n, t_limit, max_sweeps=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return h, c5

    # Build starts for n=160
    n_p = 160
    starts_p = []
    if init_h is not None and len(init_h) == n_p:
        starts_p.append(make_binary(init_h, n_p, rng))
    starts_p.append(structured_binary(n_p, rng))
    starts_p.append(random_binary(n_p, rng))
    starts_p.append(random_binary(n_p, rng))

    # Phase 1: Primary n=160 starts (70% budget)
    t_p_end = t0 + budget_s * 0.70
    t_alloc = (t_p_end - t0) / len(starts_p)
    for idx, s in enumerate(starts_p):
        t_lim = min(t_p_end, t0 + (idx + 1) * t_alloc, time.time() + t_alloc * 1.2)
        if time.time() >= t_lim:
            t_lim = time.time() + t_alloc * 0.5
        try_solution(s, n_p, t_lim, rng)

    # Phase 2: Other n values (25% budget)
    t_o_end = t0 + budget_s * 0.93
    t_alloc_o = (t_o_end - max(t_p_end, time.time())) / 2
    for n_alt in [128, 200]:
        if time.time() > t_o_end:
            break
        s = structured_binary(n_alt, rng)
        try_solution(s, n_alt, t_o_end, rng)
        t_o_end = max(t_o_end - t_alloc_o, time.time() + t_alloc_o * 0.5)

    # Phase 3: Final intensive polish (7% budget)
    if best_h is not None:
        n = len(best_h)
        t_final = t0 + budget_s * 0.99
        best_h, best_c5 = hill_climb(best_h, n, t_final, max_sweeps=100)

    if best_h is None:
        n = n_p
        best_h = random_binary(n, rng)
        best_c5 = compute_c5(best_h)

    # Verify
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
