def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n=None):
    if n is None:
        n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_fast(h, i, n):

    dx = 2.0 / n
    # Compute S_k for all lags (offsets)
    # Lag k ranges from -(n-1) to n-1, stored at index k + n - 1
    # S_k = sum_{valid i} h[i] * (1 - h[i+k])
    # When we change h[i] to h[i]+d:
    #   S_k changes by d*(1-h[i+k]) [from term i] + h[i-k]*(-d) [from term i-k]
    #   = d*(1 - h[i+k] - h[i-k])
    
    # Precompute a_k for all lags
    lags = np.arange(-(n-1), n)  # 2n-1 lags

    h_i_minus = np.zeros(2*n-1)

One more optimization: instead of minimize_scalar (which does ~50-100 function evaluations), I can solve the 1D problem analytically.

The 1D problem is: min_d max_m (c_m + d * a_m) over d ∈ [lo, hi]

where c_m = corr[m], a_m = coeff[m].

This is the classical "minimize the envelope of lines" problem. The optimal d* is:
- If all a_m ≥ 0: the max is non-decreasing, so d* = lo
- If all a_m ≤ 0: the max is non-increasing, so d* = hi
- Otherwise: d* is where two lines cross (the ones that are最大 at d*)

More precisely, define g(d) = max_m (c_m + d*a_m). This is convex in d. The minimum is where the subgradient contains 0, i.e., there exist m1, m2 with a_m1 < 0 < a_m2 such that c_m1 + d*a_m1 = c_m2 + d*a_m2 (the "upper envelope" changes at this point), and g is minimized there. Or at a boundary.

To find this exactly:
1. For each m, the line c_m + d*a_m has slope a_m and intercept c_m.
2. The upper envelope is a convex piecewise-linear function.
3. Find the point where the envelope's slope changes from negative to positive.

This can be done in O(n log n) using a convex hull trick, but for n=319, it's faster to just use bisection:
- Check if g is decreasing at lo (subgradient < 0 means there's a line with negative slope on the envelope)
- Check if g is increasing at hi
- If g(lo) is the max, return lo
- If g(hi) is the max, return hi
- Otherwise, bisect to find where the active set changes from "all negative slope" to "all positive slope"

Actually, bisection on the subgradient:
- At point d, the subgradient of g is the set {a_m : c_m + d*a_m = g(d)}
- g is minimized at d* iff 0 ∈ subgradient, i.e., there exist m+ and m- with a_m+ > 0 and a_m- < 0 (or a_m = 0) on the envelope at d*

Simpler approach: just ternary search on [lo, hi]. Since g is convex, ternary search works. Each evaluation is O(n). ~40 iterations of ternary search for precision 1e-12 on an interval of length at most 1. 40 * 319 = 12760 ops per coordinate. For 160 coordinates: 2M per sweep. Very fast!

2. Find d_right where g is still increasing

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, corr, n, order):
    """One sweep of coordinate descent. Updates corr in-place."""
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)
    improved = False
    for i in order:
        # Compute coefficient
        coeff = np.zeros(2*n-1)
        vp = (i + lags >= 0) & (i + lags < n)
        vm = (i - lags >= 0) & (i - lags < n)
        coeff[vp] += (1.0 - h[i + lags[vp]])
        coeff[vm] -= h[i - lags[vm]]
        
        lo, hi = -h[i], 1.0 - h[i]
        if hi - lo < 1e-14:
            continue
        
        # Minimize max(corr + d*coeff)
        res = minimize_scalar(lambda d: np.max(corr + d * coeff),
                            bounds=(lo, hi), method='bounded',
                            options={'xatol': 1e-13})
        d_opt = res.x
        if abs(d_opt) < 1e-13:
            continue
        new_val = np.clip(h[i] + d_opt, 0.0, 1.0)
        d_actual = new_val - h[i]
        if abs(d_actual) < 1e-13:
            continue
        
        old_c5 = np.max(corr) * dx
        corr_new = corr + d_actual * coeff
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < old_c5 - 1e-15:
            h[i] = new_val
            corr[:] = corr_new
            improved = True
    return improved

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def min_1d_convex(corr, coeff, lo, hi, tol=1e-12):
    if hi - lo < tol:
        return (lo + hi) * 0.5
    gr = (sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(corr + c * coeff)
    fd = np.max(corr + d * coeff)
    for _ in range(200):
        if b - a < tol:
            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = np.max(corr + c * coeff)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = np.max(corr + d * coeff)
    return (a + b) * 0.5

def make_coeff(h, i, n, lags):
    coeff = np.zeros(2 * n - 1)
    vp = (i + lags >= 0) & (i + lags < n)
    vm = (i - lags >= 0) & (i - lags < n)
    coeff[vp] += (1.0 - h[i + lags[vp]])
    coeff[vm] -= h[i - lags[vm]]
    return coeff

            new_val = 1.0
        d_actual = new_val - h[i]
        if abs(d_actual) < 1e-13:

        new_corr = corr + d_actual * coeff
        if np.max(new_corr) < np.max(corr) - 1e-15:
            h[i] = new_val
            corr[:] = new_corr
            improved = True
    return improved
