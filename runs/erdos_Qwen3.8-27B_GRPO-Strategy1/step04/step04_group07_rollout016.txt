            # Let's compute the coefficient vector
            coeff = np.zeros(2 * n - 1)
            # h[i] as first factor: corr[i - (i+m) + n-1]... 
            # Actually, lag m corresponds to index m + n - 1 in the full correlate output
            # corr_idx[m + n - 1] = sum_{p: 0<=p<n, 0<=p+m<n} h[p] * (1-h[p+m])
            
            # h[i] as h[p]: p=i, need 0<=i+m<n, so m in [-i, n-1-i]
            # Contribution to corr[m+n-1]: (1-h[i+m])
            for m in range(-i, n - i):
                coeff[m + n - 1] += (1.0 - best_h[i + m])
            
            # h[i] as h[p+m]: p+m=i, p=i-m, need 0<=i-m<n
            # Contribution to corr[m+n-1]: -h[i-m]
            for m in range(i - (n-1), i + 1):
                if 0 <= i - m < n:
                    coeff[m + n - 1] -= best_h[i - m]
            
            # Special case: lag 0, h[i] appears as both h[p] and h[p] (since p+m=p when m=0)
            # corr at lag 0: sum_p h[p](1-h[p])
            # h[i] contributes: h[i](1-h[i]) = h[i] - h[i]^2
            # dv = v - old
            # change = dv*(1-h[i]) + h[i]*(-dv) + ... wait
            # Actually (v)(1-v) - old(1-old) = v - v^2 - old + old^2 = (v-old) - (v^2 - old^2)
            # = (v-old)(1 - v - old)
            # So the coeff at lag 0 needs correction: linear approx gives (1-2*old)*dv
            # But exact: (v-old)(1-v-old)
            # Let me handle this separately
            c0_base = corr[c0_idx]
            
            def obj(v):
                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                return np.max(cn)
            
            v0 = obj(0.0)
            v1r = obj(1.0)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-8})
            best_v = res.x
            best_f = res.fun
            if v0 < best_f:
                best_v, best_f = 0.0, v0
            if v1r < best_f:
                best_v, best_f = 1.0, v1r
            
            if best_f * dx < best_c5 - 1e-13:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = corr + (best_v - old) * coeff
                corr[c0_idx] = c0_base + (best_v - old) * (1.0 - best_v - old)
                improved = True
        
        if not improved:
            # Try with tighter tolerance
            break
    
    # Final verification
    corr_final = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr_final) * dx
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=500):

    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    c0_idx = n - 1
    h_rev = best_h[::-1]  # for the second term

        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = best_h[i]
            
            coeff = np.zeros(2 * n - 1)
            # First term: m from -i to n-1-i, h[i+m] = h[0:n]
            coeff[n - i : n - 1 + (n - i)] += (1.0 - best_h)
            # Wait: ms1 + n - 1 where ms1 = range(-i, n-i)
            # Index range: (-i + n - 1) to (n-1-i + n - 1) = (n-1-i) to (2n-2-i)
            coeff[n - 1 - i : 2 * n - 1 - i] += (1.0 - best_h)
            
            # Second term: m from i-(n-1) to i, h[i-m] = h reversed
            # Index range: (i-(n-1) + n - 1) to (i + n - 1) = i to i+n-1
            coeff[i : i + n] -= h_rev
            
            c0_base = corr[c0_idx]

                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                return np.max(cn)
            
            v0 = obj(0.0)
            v1r = obj(1.0)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-8})
            best_v = res.x
            best_f = res.fun
            if v0 < best_f:
                best_v, best_f = 0.0, v0
            if v1r < best_f:
                best_v, best_f = 1.0, v1r
            
            if best_f * dx < best_c5 - 1e-13:
                dv = best_v - old
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr[n - 1 - i : 2 * n - 1 - i] += dv * (1.0 - best_h)
                corr[i : i + n] -= dv * h_rev
                corr[c0_idx] = c0_base + dv * (1.0 - best_v - old)
                # Update h_rev
                h_rev[n - 1 - i] = best_v
                improved = True
        
        if not improved:
            break
    
    corr_final = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr_final) * dx
    return best_h, best_c5

Wait, I have a bug. When I update corr incrementally, I'm using the OLD values of best_h and h_rev in the coefficient computation. But after updating best_h[i], the coefficients for FUTURE coordinate updates will be wrong because they depend on the current h values.

The issue is that `coeff` depends on the current h (specifically 1-best_h and h_rev). After I update best_h[i], the coeff for the next coordinate will be different.

So I need to recompute coeff each time. But the vectorized version is O(n) per coordinate, so the total is O(n²) per sweep. For n=160, that's 25600 per sweep, which is fine.

But the incremental update of corr is O(n) per coordinate. The total is also O(n²) per sweep. This is fine.

However, the issue is that after updating best_h[i] and h_rev, the coeff for the next coordinate is automatically correct because we recompute it from the updated best_h and h_rev. Good.

But wait, there's a subtle issue with the incremental corr update. When I update h[i] from old to best_v, the change in corr should be:
- For the h[i] as first factor: (best_v - old) * (1 - h[i+m]) for each m
- For h[i] as second factor: (best_v - old) * (-h[i-m]) for each m
- Special correction at lag 0

But I'm computing coeff BEFORE the update, using the old h values. The coeff already accounts for the linear change. The only non-linear part is the lag-0 correction (v-old)(1-v-old) vs the linear approximation dv*(1-2*old).

So the incremental update should be:
corr += dv * coeff  (this is the linear part, which is exact for all lags except lag 0)
corr[c0_idx] = c0_base + (v - old) * (1 - v - old)  (exact correction for lag 0)

But wait, at lag 0, the coeff already includes the linear contribution. Let me check:
- At lag 0 (m=0), first factor: 1-h[i+0] = 1-h[i] = 1-old
- At lag 0 (m=0), second factor: -h[i-0] = -h[i] = -old
- Total coeff at lag 0: (1-old) + (-old) = 1-2*old

The linear approximation of the change would be: dv * (1-2*old)
The exact change: (v)(1-v) - old(1-old) = v - v² - old + old² = (v-old) - (v²-old²) = (v-old)(1-v-old)

So the correction is: exact - linear = (v-old)(1-v-old) - (v-old)(1-2*old) = (v-old)[(1-v-old) - (1-2old)] = (v-old)(old-v) = -(v-old)²

So corr[c0_idx] += dv * (1-2*old) + (-(dv)²) = dv*(1-2*old) - dv²
Or equivalently: corr[c0_idx] = c0_base + dv*(1-2*old) - dv² = c0_base + dv*(1-v-old)

Yes, that's what I had. Good.

                g[:n+m] -= h[-m:]
        g /= len(active)
        
        h_new = np.clip(h - step * g, 0.0, 1.0)
        h_new = project(h_new, n)
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=2000):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    c0_idx = n - 1
    h_rev = best_h[::-1]
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = best_h[i]
            coeff = np.zeros(2 * n - 1)
            coeff[n - 1 - i: 2 * n - 1 - i] += (1.0 - best_h)
            coeff[i: i + n] -= h_rev
            c0_base = corr[c0_idx]
            def obj(v):
                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                return np.max(cn)
            v0 = obj(0.0)
            v1r = obj(1.0)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            best_v = res.x
            best_f = res.fun
            if v0 < best_f:
                best_v, best_f = 0.0, v0
            if v1r < best_f:
                best_v, best_f = 1.0, v1r
            if best_f * dx < best_c5 - 1e-14:
                dv = best_v - old
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr[n - 1 - i: 2 * n - 1 - i] += dv * (1.0 - best_h)
                corr[i: i + n] -= dv * (best_h)[::-1]
                corr[c0_idx] = c0_base + dv * (1.0 - best_v - old)
                h_rev[n - 1 - i] = best_v
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if (sweep + 1) % 50 == 0:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_c5 = np.max(corr) * dx
            h_rev = best_h[::-1]
    corr_f = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr_f) * dx
    return best_h, best_c5

def subgradient_opt(h, n, t_end, max_iters=50000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    step = 0.08
    patience = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = np.max(corr)
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            step = min(step * 1.03, 0.4)
        else:
            patience += 1
            if patience > 500:
                step *= 0.85
                patience = 0
                if step < 0.005:
                    break
        thresh = c5 - 0.03 * c5
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        g = np.zeros(n)
        for idx in active:
            m = idx - (n - 1)
            if m >= 0:
                g[:n - m] += (1.0 - h[m:])
                g[m:] -= h[:n - m]
            else:
                g[-m:] += (1.0 - h[:n + m])
                g[:n + m] -= h[-m:]
        g /= len(active)
        h = project(h - step * g, n)
    return best_h, best_c5

def perturbation_search(h, n, t_end, max_rounds=200):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        frac = 0.15 + 0.15 * np.random.rand()
        n_pert = max(1, int(frac * n))
        idx_pert = np.random.choice(n, n_pert, replace=False)
        h_p = best_h.copy()
        noise_scale = 0.3 + 0.4 * (1.0 - r / max_rounds)
        h_p[idx_pert] += noise_scale * np.random.randn(n_pert)
        h_p = project(h_p, n)
        h_opt, c5_opt = coord_descent(h_p, n, min(t_end, time.time() + 8.0), max_sweeps=100)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)

    best_c5 = compute_c5(best_h)

        return time.time() - t0

    # Phase 3: Perturbation search (fast, no SLSQP)
    best_h, best_c5 = perturbation_search(best_h, n, t0 + budget_s * 0.55, max_rounds=150)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.62, max_sweeps=600)

    # Phase 4: Multi-resolution
    for n_try in [100, 120, 80, 200, 240]:

        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.84, max_sweeps=400)
        h_tr, c5_tr = subgradient_opt(h_tr, n_try, t0 + budget_s * 0.86, max_iters=10000)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.87, max_sweeps=300)

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=1000)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
