```python
def optimal_hi(corr_base, coefs, n):
    # f(hi) = max_m (corr_base[m] + coefs[m] * hi) for hi in [0,1]
    # Find argmin
    vals0 = corr_base  # at hi=0
    vals1 = corr_base + coefs  # at hi=1
    max0 = np.max(vals0)
    max1 = np.max(vals1)
    if max0 <= max1:
        # Non-decreasing at 0? Check subgradient at 0
        active0 = np.where(vals0 == max0)[0]
        if np.min(coefs[active0]) >= 0:
            return 0.0  # minimum at hi=0
    if max1 <= max0:
        active1 = np.where(vals1 == max1)[0]
        if np.max(coefs[active1]) <= 0:
            return 1.0  # minimum at hi=1
    # Interior: find where the upper envelope has a minimum
    # The upper envelope is max of lines. The minimum is at a kink.
    # Kink between lines m1, m2 at hi = (a_m2 - a_m1)/(b_m1 - b_m2)
    # where b_m1 != b_m2 and the kink hi is in [0,1]
    
    # Efficient approach: the minimum of the max of lines is at the point
    # where the "two highest lines" cross (or at the boundary)
    # Sort by value at hi=0: order = argsort(corr_base, descending)
    # The top few lines determine the envelope near their crossing points
    
    # Simpler: ternary search (since convex)
    a, b = 0.0, 1.0
    for _ in range(50):
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        f1 = np.max(corr_base + m1 * coefs)
        f2 = np.max(corr_base + m2 * coefs)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2
```
