import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-14 and s2 > 1e-14:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def grad_c_at_lag(h, p):
    """Gradient of correlation at lag index p (0-indexed in full correlate output)."""
    n = len(h)
    shift = n - 1 - p
    g = np.zeros(n)
    for j in range(n):
        if 0 <= j + shift < n:
            g[j] += (1.0 - h[j + shift])
        if 0 <= j - shift < n:
            g[j] -= h[j - shift]
    return g

def sg_step(h, p, alpha):
    """Stochastic gradient step for lag p."""
    g = grad_c_at_lag(h, p)
    h_new = h - alpha * g
    return h_new

def stochastic_subgrad(h, n, t_end, rng, n_iters=50000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work = h.copy()
    work_c5 = best_c5
    alpha = 0.1
    for it in range(n_iters):
        if time.time() > t_end:
            break
        if it % 100 == 0:
            alpha = max(0.01, 0.1 * (1 - it / n_iters) ** 0.5)
        p = rng.integers(0, 2 * n - 1)
        g = grad_c_at_lag(work, p)
        work = work - alpha * g
        work = project(work, n)
        if it % 2000 == 0:
            cv = compute_c5(work)
            if cv < best_c5:
                best_c5 = cv
                best_h = work.copy()
    cv = compute_c5(work)
    if cv < best_c5:
        best_c5 = cv
        best_h = work.copy()
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(np.clip(res.x, 0, 1), n)
    except:
        return h

def coord_opt(h, n, i):
    """Optimize coordinate i, fixing all others."""
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if 0 <= i - k < n:
            d[n - 1 - k] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            d[n - 1 + k] += (-h[i + k])
        if 0 <= i + k < n:
            d[n - 1 - k] += (1.0 - h[i + k])
        if 0 <= i - k < n:
            d[n - 1 + k] += (-h[i - k])
    c0 = corr_base[n - 1]
    # f(t) = max(corr_base + t*d, with c0 replaced by c0 + t*(1-t)) * dx
    from scipy.optimize import minimize_scalar
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, rng, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-12:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-14 and s2 > 1e-14:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def grad_c_vec(h, p):
    n = len(h)
    shift = n - 1 - p
    g = np.zeros(n)
    if shift >= 0:
        g[:n - shift] += (1.0 - h[shift:])
        g[shift:] -= h[:n - shift]
    else:
        s = -shift
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g

def nesterov_sg(h, n, t_end, rng, n_iters=30000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    x = h.copy()
    alpha = 0.05
    for it in range(n_iters):
        if time.time() > t_end:
            break
        decay = alpha * (1.0 / (1.0 + 0.001 * it)) ** 0.5
        momentum = 0.9
        y = x + momentum * v
        p = rng.integers(0, 2 * n - 1)
        g = grad_c_vec(y, p)
        v = momentum * v - decay * g
        x = project(y + v, n)
        if it % 2000 == 0:
            cv = compute_c5(x)
            if cv < best_c5:
                best_c5 = cv
                best_h = x.copy()
    cv = compute_c5(x)
    if cv < best_c5:
        best_c5 = cv
        best_h = x.copy()
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter=2000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if 0 <= i - k < n:
            d[n - 1 - k] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            d[n - 1 + k] += (-h[i + k])
        if 0 <= i + k < n:
            d[n - 1 - k] += (1.0 - h[i + k])
        if 0 <= i - k < n:
            d[n - 1 + k] += (-h[i - k])
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, rng, max_sweeps=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-12:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def pair_descent(h, n, t_end, rng, n_pairs=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        for eps in [0.01, 0.03, 0.05]:
            for di in [-1, 1]:
                for dj in [-1, 1]:
                    hi = best_h[i] + di * eps
                    hj = best_h[j] + dj * eps
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        test = best_h.copy()
                        test[i], test[j] = hi, hj
                        test = project(test, n)
                        cv = compute_c5(test)
                        if cv < best_c5 - 1e-15:
                            best_h, best_c5 = test, cv
                            break
                else:
                    continue
                break
            else:
                continue
            break
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish with increasing T
    for T in [50, 200, 1000, 5000]:
        if elapsed() > budget_s * 0.10:
            break
        hc = slsqp_polish(best_h, T, n, maxiter=2000, ftol=1e-16)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 2: Coordinate descent (strong local)
    t_cd1 = t0 + budget_s * 0.30
    best_h, best_c5 = coord_descent(best_h, n, t_cd1, rng, max_sweeps=80)

    # Phase 3: SLSQP with very high T after coord descent
    for T in [10000, 50000, 200000]:
        if elapsed() > budget_s * 0.40:
            break
        hc = slsqp_polish(best_h, T, n, maxiter=3000, ftol=1e-16)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 4: Nesterov stochastic subgradient
    t_sg = t0 + budget_s * 0.55

    if c5_sg < best_c5:
        best_h, best_c5 = h_sg, c5_sg
    # Polish after SG
    hc = slsqp_polish(best_h, 50000, n, maxiter=3000, ftol=1e-16)
    cv = compute_c5(hc)
    if cv < best_c5:
        best_h, best_c5 = hc, cv

    # Phase 5: Pairwise descent for fine adjustments
    t_pair = t0 + budget_s * 0.68
    best_h, best_c5 = pair_descent(best_h, n, t_pair, rng, n_pairs=8000)

    # Phase 6: Basin hopping with small perturbations
    if elapsed() < budget_s * 0.90:
        work = best_h.copy()
        work_c5 = best_c5
        n_hop = 400
        for it in range(n_hop):
            if time.time() > t0 + budget_s * 0.90:

                work, work_c5 = hc, cv
            if work_c5 < best_c5:
                best_h, best_c5 = work.copy(), work_c5
        # Final coord descent on basin hop result
        if elapsed() < budget_s * 0.94:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, rng, max_sweeps=40)
