def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    lam = exps / exps.sum()
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        l = lam[k_idx]
        if l < 1e-15:

            grad[m_arr:] += l * (1.0 - h[:n-m_arr])
            grad[:n-m_arr] -= l * h[m_arr:]

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    lam = exps / exps.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        l = lam[k_idx]
        if l < 1e-15:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            if k == 0:
                grad[:] += l * (1.0 - h)
            else:
                grad[:n-k] += l * (1.0 - h[k:])
                grad[k:] -= l * h[:n-k]
        else:
            ma = -k
            grad[ma:] += l * (1.0 - h[:n-ma])
            grad[:n-ma] -= l * h[ma:]
    grad *= dx
    return grad

def project_to_constraint(h, n):
    target = n / 2.0
    vc = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-13:
        return vc
    def sum_at(sh):
        return np.clip(vc + sh, 0.0, 1.0).sum()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if sum_at(mid) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vc + 0.5 * (lo + hi), 0.0, 1.0)

def slsqp_step(h, T, n, maxiter=800, ftol=1e-15):
    def obj(x):
        return smooth_c5_val(x, T)
    def objg(x):
        return smooth_c5_grad(x, T)
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(objg, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_constraint(res.x, n)

def compute_dk(h, n, i):
    dk = np.zeros(2 * n - 1)
    off = n - 1
    for k_idx in range(2 * n - 1):
        k = k_idx - off
        if k >= 0:
            if i <= n - 1 - k:
                dk[k_idx] += 1.0 - h[i + k]
            if k > 0 and i >= k:
                dk[k_idx] -= h[i - k]
        else:
            ma = -k
            if i >= ma:
                dk[k_idx] += 1.0 - h[i - ma]
            if ma > 0 and i <= n - 1 - ma:
                dk[k_idx] -= h[i + ma]
    return dk

def fast_coord_descent(h, n, t_end, max_sweeps=20000):
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    no_improve = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            dk = compute_dk(best_h, n, i)
            hv = best_h[i]
            A = corr - dk * hv
            B = dk
            a, b = 0.0, 1.0
            cp = b - gr * (b - a)
            dp = a + gr * (b - a)
            fc = np.max(A + B * cp)
            fd = np.max(A + B * dp)
            for _ in range(32):
                if fc < fd:
                    b = dp; dp = cp; fd = fc
                    cp = b - gr * (b - a)
                    fc = np.max(A + B * cp)
                else:
                    a = cp; cp = dp; fc = fd
                    dp = a + gr * (b - a)
                    fd = np.max(A + B * dp)
            a_opt = 0.5 * (a + b)
            f0 = np.max(A)
            f1 = np.max(A + B)
            fm = np.max(A + B * a_opt)
            if f0 <= fm and f0 <= f1:
                a_opt = 0.0
            elif f1 <= fm and f1 <= f0:
                a_opt = 1.0
            if abs(a_opt - hv) < 1e-11:
                continue
            delta = a_opt - hv
            best_h[i] = a_opt
            corr += dk * delta
            improved = True
        if improved:
            c5_new = np.max(corr) * dx
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve >= 4:
            break
    return best_h, best_c5
