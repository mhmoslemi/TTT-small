import numpy as np
from scipy.optimize import minimize, minimize_scalar, Powell
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_minimize(h, T, maxiter=400):
    n = len(h)
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_topk(h, K):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for m in top_idx:
        s = m - (n - 1)
        j1 = np.arange(n) + s
        j2 = np.arange(n) - s
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        grad += (1.0 - h[np.clip(j1, 0, n-1)]) * v1.astype(float)
        grad -= h[np.clip(j2, 0, n-1)] * v2.astype(float)
    return (grad / K) * (2.0 / n)

def coord_descent(h, t_end, max_sweeps=200, tol=1e-12):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            s_arr = np.arange(len(corr_base)) - (n - 1)
            j1 = i + s_arr
            j2 = i - s_arr
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(len(corr_base))
            coefs[v1] += (1.0 - h_base[np.clip(j1[v1], 0, n-1)])
            coefs[v2] += (-h_base[np.clip(j2[v2], 0, n-1)])
            quad = (s_arr == 0).astype(float)
            def f1d(v):
                return (np.max(corr_base + v * coefs - v * v * quad) - 0) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < tol:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved and sweep > 2:
            break
    return best_h, best_c5

def free_var_opt(h, t_end, max_iter=500):
    n = len(h)
    tol = 0.005
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        free = np.where((best_h > tol) & (best_h < 1.0 - tol))[0]
        if len(free) < 2:
            break
        # Also consider near-boundary vars
        near = np.where((best_h > 0.0) & (best_h < tol) |
                        (best_h > 1.0 - tol) & (best_h < 1.0))[0]
        idx = np.unique(np.concatenate([free, near]))
        if len(idx) < 2:
            break
        h_sub = best_h[idx].copy()
        def sub_obj(v):
            trial = best_h.copy()
            trial[idx] = np.clip(v, 0, 1)
            trial = project_h(trial)
            return compute_c5(trial)
        res = Powell(sub_obj, x0=h_sub, bounds=[(0.0, 1.0)] * len(idx),
                     maxiter=80, tol=1e-10)
        h_try = best_h.copy()
        h_try[idx] = np.clip(res.x, 0, 1)
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_h = h_try
            best_c5 = c5_try
        else:
            break
    return best_h, best_c5
