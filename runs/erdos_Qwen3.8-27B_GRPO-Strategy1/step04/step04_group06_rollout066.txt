```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.minimum(h + diff / n, 1.0)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_fast(h, n, i):
    """Optimize h[i] given all other coordinates fixed. O(n) time."""
    # Compute corr with h[i] = 0
    h0 = h.copy()
    h0[i] = 0.0
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    
    # Compute coefficients a[m] for m = 0, ..., 2n-1
    # a[m] is such that corr_new[m] = corr[m] + x*a[m] (for the linear part)
    # Special: for m = n-1, corr_new[n-1] = corr[n-1] + x - x^2
    a = np.zeros(2*n - 1)
    center = n - 1
    a[center] = 1.0  # linear coefficient for shift 0
    
    for k in range(1, n):
        # Shift k > 0: m = center + k
        # a[center+k] = (1-h[i-k])*1_{k<=i} - h[i+k]*1_{i<=n-1-k}
        m_pos = center + k
        if k <= i <= n - 1:
            a[m_pos] += 1.0 - h[i-k]
        if 0 <= i <= n - 1 - k:
            a[m_pos] -= h[i+k]
        
        # Shift -k: m = center - k
        m_neg = center - k
        # a[center-k] = (1-h[i+k])*1_{i<=n-1-k} - h[i-k]*1_{k<=i<=n-1}
        if 0 <= i <= n - 1 - k:
            a[m_neg] += 1.0 - h[i+k]
        if k <= i <= n - 1:
            a[m_neg] -= h[i-k]
    
    # Now minimize f(x) = max(corr + x*a) * dx with x² correction
    # f(x) = max(corr[m] + x*a[m] for m≠center, corr[center] + x - x²) * dx
    dx = 2.0 / n
    
    def f1d(x):
        vals = corr + x * a
        vals[center] = corr[center] + x - x * x
        return np.max(vals) * dx
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x
```

```python
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    
    for i in range(n):
        old_val = h[i]
        if old_val == 0.0:
            # Need to compute a with h[i]=0 (already is)
            pass
        # Subtract old contribution from corr
        # corr currently includes h[i]
        # The contribution of h[i] to corr[m]:
        # For m = center: h[i] - h[i]²
        # For m = center + k (k>0): (1-h[i-k])*h[i]*1_{k<=i} - h[i+k]*h[i]*1_{i<=n-1-k}
        #   = h[i] * [(1-h[i-k])*1_{k<=i} - h[i+k]*1_{i<=n-1-k}]
        # For m = center - k (k>0): (1-h[i+k])*h[i]*1_{i<=n-1-k} - h[i-k]*h[i]*1_{k<=i}
        #   = h[i] * [(1-h[i+k])*1_{i<=n-1-k} - h[i-k]*1_{k<=i}]
```

```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.minimum(h + diff / n, 1.0)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    """Optimize h[i] given all others fixed. Returns optimal value."""
    h0 = h.copy()
    h0[i] = 0.0
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    a[center] = 1.0
    for k in range(1, n):
        if k <= i:
            a[center + k] += 1.0 - h[i - k]
            a[center - k] -= h[i - k]
        if i <= n - 1 - k:
            a[center + k] -= h[i + k]
            a[center - k] += 1.0 - h[i + k]
    base = corr
    bc = base[center]
    def f1d(x):
        m = np.max(base + x * a)
        v = bc + x - x * x
        if v > m:
            m = v
        return m
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-9})
    return res.x

def cd_sweeps(h, n, t_end, max_sweeps=10000, patience=8):
    """Multiple coordinate descent sweeps with early stopping."""
    best_c5 = compute_c5(h)
    n_stale = 0
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        order = np.random.permutation(n)
        for i in order:
            nv = coord_opt(h, n, i)
            h[i] = nv
        h = project(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            n_stale = 0
        else:
            n_stale += 1
            if n_stale >= patience:
                break
    return h, best_c5

def binary_local_search(h, n, t_end, n_trials=20000):
    """Local search on the binary rounding of h."""
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n//2 ones
    while h_bin.sum() > n // 2:
        idx = np.where(h_bin == 1.0)[0]
        h_bin[np.random.choice(idx)] = 0.0
    while h_bin.sum() < n // 2:
        idx = np.where(h_bin == 0.0)[0]
        h_bin[np.random.choice(idx)] = 1.0
    
    best_c5 = compute_c5(h_bin)
    best_bin = h_bin.copy()
    temp = 0.05
    
    for it in range(n_trials):
        if time.time() > t_end:
            break
        # Swap: pick random 1 and random 0, swap them
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < best_c5 or np.random.random() < np.exp(-(c5_new - best_c5) / temp):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_bin = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
        temp = max(temp * 0.9995, 0.001)
    
    return best_bin, best_c5

def slsqp_smooth(h, n, T, maxiter=800):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        sm = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return sm * dx
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        weights = np.exp(T * (corr - m))
        weights /= weights.sum()
        center = n - 1
        g = np.zeros(n)
        for idx in range(2 * n - 1):
            w = weights[idx]
            if w < 1e-10:
                continue
            k = idx - center
            if k == 0:
                g += w * (1.0 - 2.0 * x)
            elif k > 0:
                g[:n - k] += w * (1.0 - x[k:])
                g[k:] -= w * x[:n - k]
            else:
                m2 = -k
                g[m2:] += w * (1.0 - x[:n - m2])
                g[:n - m2] -= w * x[m2:]
        return g * dx
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    
    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        n0 = 160

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Intensive coordinate descent (30% of budget)
    p1_end = t0 + budget_s * 0.30
    best_h, best_c5 = cd_sweeps(best_h, n, p1_end, max_sweeps=5000, patience=10)
    
    # Phase 2: SLSQP polish (15% of budget)
    for T in [100, 500, 2000]:
        if time.time() > t0 + budget_s * 0.42:
            break
        try:
            hc = slsqp_smooth(best_h, n, T, maxiter=600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    # Quick CD after SLSQP
    if time.time() < t0 + budget_s * 0.48:
        best_h, best_c5 = cd_sweeps(best_h, n, t0 + budget_s * 0.48, max_sweeps=1000, patience=5)
    
    # Phase 3: Binary local search (15% of budget)
    if time.time() < t0 + budget_s * 0.60:
        h_bin, c5_bin = binary_local_search(best_h, n, t0 + budget_s * 0.58, n_trials=15000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            # Interpolate: blend binary with continuous
            best_h = 0.5 * h_bin + 0.5 * best_h
            best_h = project(best_h, n)
            best_h, best_c5 = cd_sweeps(best_h, n, t0 + budget_s * 0.63, max_sweeps=500, patience=5)
    
    # Phase 4: Multi-start exploration (15% of budget)
    for restart in range(5):
        if time.time() > t0 + budget_s * 0.75:
            break
        h_try = best_h + 0.05 * np.random.randn(n)
        h_try = project(h_try, n)
        h_try, c5_try = cd_sweeps(h_try, n, t0 + budget_s * (0.75 + 0.02 * (restart - 4)), max_sweeps=300, patience=3)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    # Phase 5: Final intensive CD + SLSQP (25% of budget)
    best_h, best_c5 = cd_sweeps(best_h, n, t0 + budget_s * 0.88, max_sweeps=5000, patience=15)
    for T in [5000, 20000]:
        if time.time() > t0 + budget_s * 0.95:
            break
        try:

            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    best_h, best_c5 = cd_sweeps(best_h, n, t_end - 5, max_sweeps=2000, patience=10)
    
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
