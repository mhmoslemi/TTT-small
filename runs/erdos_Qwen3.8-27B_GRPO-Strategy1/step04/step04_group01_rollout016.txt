import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def smooth_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-12:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n-k] += wm * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= wm * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - h[:n-kk])
            grad[:n-kk] -= wm * h[kk:]
    return grad * dx

def project(h, n):
    target = n / 2.0
    vec = np.clip(h, 0.0, 1.0)
    s = vec.sum()
    if abs(s - target) < 1e-10:
        return vec
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def grad_step(h, T, n, lr):
    g = smooth_grad(h, T)
    f0 = smooth_obj(h, T)
    alpha = lr
    best_h, best_f = h, f0
    for _ in range(30):
        h_new = project(h - alpha * g, n)
        f_new = smooth_obj(h_new, T)
        if f_new < best_f:
            best_h, best_f = h_new, f_new
        if f_new <= f0 - 1e-4 * alpha * np.dot(h - h_new, g):
            return best_h
        alpha *= 0.5
    return best_h

def coord_desc(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_imp = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            def f1d(x):
                h_try = best_h.copy()
                h_try[i] = x
                h_try = project(h_try, n)
                return compute_c5(h_try)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            if res.x != best_h[i]:
                h_try = best_h.copy()
                h_try[i] = res.x
                h_try = project(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-14:
                    best_c5 = c_try
                    best_h = h_try
                    improved = True
        no_imp = 0 if improved else no_imp + 1
        if no_imp >= 5:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init//2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Gradient descent with T annealing (30% budget)
    T_schedule = [20, 50, 100, 300, 1000]
    t_phase1_end = t0 + budget_s * 0.30
    work_h = best_h.copy()
    work_T = 20
    it = 0
    while elapsed() < t_phase1_end:
        work_h = grad_step(work_h, work_T, n, lr=0.05)
        it += 1
        if it % 50 == 0:
            c5v = compute_c5(work_h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = work_h.copy()
            work_T = min(work_T * 1.5, 5000)
    
    # Phase 2: Coordinate descent (30% budget)
    t_phase2_end = t0 + budget_s * 0.60
    best_h, best_c5 = coord_desc(best_h, n, t_phase2_end, max_sweeps=200)
    
    # Phase 3: More gradient with high T (20% budget)
    t_phase3_end = t0 + budget_s * 0.80
    work_h = best_h.copy()
    for T_hi in [500, 2000, 10000]:
        for _ in range(50):
            if elapsed() > t_phase3_end:
                break
            work_h = grad_step(work_h, T_hi, n, lr=0.02)
        c5v = compute_c5(work_h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = work_h.copy()
    
    # Phase 4: Final coordinate descent (15% budget)
    t_phase4_end = t0 + budget_s * 0.97
    best_h, best_c5 = coord_desc(best_h, n, t_phase4_end, max_sweeps=500)
    
    # Phase 5: Final polish
    if elapsed() < budget_s * 0.99:
        work_h = grad_step(best_h, 5000, n, lr=0.01)
        for _ in range(10):
            work_h = grad_step(work_h, 5000, n, lr=0.005)
        c5v = compute_c5(work_h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = work_h
    
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

If we set h[i]=1, h[i+1]=0: contribution at lag 1 is 1*1 = 1 (worse!).
If we set h[i]=0, h[i+1]=1: contribution at lag 1 is 0*0 = 0 (better!).

So 0-1 is not always worse than fractional. But the coordinate descent analysis showed that for the exact objective, the optimum is at the boundary. The issue is that when we're constrained to sum = n/2, we can't independently set each variable to its boundary.

I think the best approach is:
1. Use 0-1 structure for the "hard" problem
2. Allow fractional values for fine-tuning (the coordinate descent will find the optimal fractional values given the 0-1 "skeleton")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wm = w[mi]
        if wm < 1e-12:
            continue
        k = mi - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n-k] += wm * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= wm * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - h[:n-kk])
            grad[:n-kk] -= wm * h[kk:]
    return grad * dx

def project(h, n):
    target = n / 2.0
    vec = np.clip(h, 0.0, 1.0)
    if abs(vec.sum() - target) < 1e-10:
        return vec
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def grad_opt(h, T, n, n_steps, lr0):
    h_cur = h.copy()

        lr = lr0 * (1.0 - s / n_steps) + 0.001
        g = smooth_grad(h_cur, T)
        f0 = smooth_obj(h_cur, T)
        alpha = lr

            if f_new < f0:
                h_cur = h_new
                break
            alpha *= 0.5
        else:
            h_cur = project(h_cur - alpha * g, n)
    return h_cur
