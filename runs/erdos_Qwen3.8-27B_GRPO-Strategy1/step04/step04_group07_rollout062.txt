    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_simplex_box(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s < 1e-14:
            return h
        h = np.clip(h * (1.0 + diff / s), 0, 1)
    # Second pass to fix any residual
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = np.clip(h + room * (diff2 / tot), 0, 1)
        else:
            s2 = h.sum()
            if s2 > 1e-14:
                h = np.clip(h * (1.0 + diff2 / s2), 0, 1)
    return h

        if tot < 1e-14:
            return h
        h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s < 1e-14:
            return h
        h = np.clip(h * (1.0 + diff / s), 0, 1)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = np.clip(h + room * (diff2 / tot), 0, 1)
        else:
            s2 = h.sum()
            if s2 > 1e-14:
                h = np.clip(h * (1.0 + diff2 / s2), 0, 1)
    return h

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new)

def build_coef_matrix(h0, n):
    N = 2 * n - 1
    coef = np.zeros((N, n))
    idx = np.arange(n)
    tauris = np.arange(-(n - 1), n)
    for ki, tau in enumerate(tauris):
        v1 = (idx + tau >= 0) & (idx + tau < n)
        v2 = (idx - tau >= 0) & (idx - tau < n)
        if np.any(v1):
            coef[ki, v1] += (1.0 - h0[idx[v1] + tau])
        if np.any(v2):
            coef[ki, v2] -= h0[idx[v2] - tau]
    return coef

def lp_linearize(h0, n, max_iter=100, max_step=None, t_end=None):
    if max_step is None:
        max_step = 0.3
    h_work = h0.copy()
    c5_work = compute_c5(h_work)
    best_h = h_work.copy()
    best_c5 = c5_work
    N = 2 * n - 1
    num_vars = n + 1
    c_obj = np.zeros(num_vars)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    for it in range(max_iter):
        if t_end is not None and time.time() > t_end:
            break
        corr0 = get_corrs(h_work)
        coef = build_coef_matrix(h_work, n)
        A_ub = np.hstack([coef, -np.ones((N, 1))])
        b_ub = -corr0 + coef.dot(h_work)
        bounds = [(max(0.0, h_work[i] - max_step), min(1.0, h_work[i] + max_step))
                  for i in range(n)]
        bounds.append((0.0, None))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        new_h = project_h(res.x[:n], n)
        new_c5 = compute_c5(new_h)
        if new_c5 < c5_work - 1e-14:
            h_work = new_h
            c5_work = new_c5
            if c5_work < best_c5:
                best_c5, best_h = c5_work, new_h.copy()
        else:
            mid_h = project_h(0.5 * h_work + 0.5 * res.x[:n], n)
            mid_c5 = compute_c5(mid_h)
            if mid_c5 < c5_work - 1e-14:
                h_work = mid_h
                c5_work = mid_c5
                if c5_work < best_c5:
                    best_c5, best_h = c5_work, mid_h.copy()
            else:
                max_step *= 0.65
                if max_step < 0.005:
                    break
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    corr = get_corrs(best_h)
    best_c5 = np.max(corr) * (2.0 / n)
    c0_idx = n - 1
    tauris = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            v1 = (i + tauris >= 0) & (i + tauris < n)
            v2 = (i - tauris >= 0) & (i - tauris < n)
            coeff = np.zeros(2 * n - 1)
            if np.any(v1):
                coeff[v1] += (1.0 - best_h[i + tauris[v1]])
            if np.any(v2):
                coeff[v2] -= best_h[i - tauris[v2]]
            lo, hi = 0.0, 1.0
            cands = [lo, hi]
            for kk in range(2 * n - 2):
                if abs(coeff[kk] - coeff[kk + 1]) < 1e-15:
                    continue
                t_cross = (corr[kk] - corr[kk + 1]) / (coeff[kk + 1] - coeff[kk])
                if lo < t_cross < hi:
                    cands.append(t_cross)
            cands = np.array(cands)
            best_f = np.inf
            best_v = old
            for vc in cands:
                vc = np.clip(vc, 0.0, 1.0)
                cn = corr + (vc - old) * coeff
                cn[c0_idx] = corr[c0_idx] + (vc - old) * (1.0 - vc - old)
                fv = cn.max()
                if fv < best_f:
                    best_f, best_v = fv, vc
            if best_f * (2.0 / n) < best_c5 - 1e-13:
                best_h[i] = best_v
                best_c5 = best_f * (2.0 / n)
                corr = get_corrs(best_h)
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_refine(h, n, T=500, maxiter=400, t_end=None):
    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / len(x))
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if t_end is not None and time.time() > t_end:
            return h, compute_c5(h)
        return project_h(np.clip(res.x, 0, 1), n), compute_c5(project_h(np.clip(res.x, 0, 1), n))
    except Exception:
        return h, compute_c5(h)

def basin_hop(work_h, work_c5, n, t_end, max_iters=100):
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.1, 1.0 - 0.9 * it / max_iters)
        step = 0.04 + 0.22 * cool
        h_p = work_h + step * np.random.randn(n)
        h_p = project_h(h_p, n)
        iter_end = min(t_end, time.time() + 4.0)
        h_c, c5_c = lp_linearize(h_p, n, max_iter=40, max_step=0.25, t_end=iter_end)
        if time.time() < t_end:
            h_c, c5_c = coord_descent(h_c, n, min(t_end, time.time() + 3.0), max_sweeps=30)
        temp = 0.003 * cool + 0.0002
        if c5_c < work_c5 or np.random.rand() < np.exp(-(c5_c - work_c5) / temp):
            work_h, work_c5 = h_c, c5_c
            if c5_c < best_c5:
                best_c5, best_h = c5_c, h_c.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-resolution LP + coord descent
    for n_try in [80, 120, 160, 200, 240]:
        if time.time() > t0 + budget_s * 0.18:
            break
        h_tr = best_h if n_try == n else interp_h(best_h, n_try)
        h_tr, c5_tr = lp_linearize(h_tr, n_try, max_iter=150, max_step=0.3,
                                   t_end=t0 + budget_s * 0.22)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.24, max_sweeps=60)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 2: Extended LP refinement at best n
    best_h, best_c5 = lp_linearize(best_h, n, max_iter=250, max_step=0.15,
                                   t_end=t0 + budget_s * 0.38)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.48, max_sweeps=120)

    # Phase 3: SLSQP polish
    for T in [200, 800, 3000]:
        if time.time() > t0 + budget_s * 0.58:
            break
        h_s, c5_s = slsqp_refine(best_h, n, T=T, maxiter=500,
                                 t_end=t0 + budget_s * 0.60)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s

    # Phase 4: Basin hopping
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.78,
                                max_iters=120)

    # Phase 5: Final LP + coord descent
    best_h, best_c5 = lp_linearize(best_h, n, max_iter=200, max_step=0.08,
                                   t_end=t0 + budget_s * 0.88)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, max_sweeps=150)

    # Phase 6: Try neighboring resolutions
    for n_try in [n - 20, n + 20, n + 40]:
        n_try = max(40, min(400, n_try))
        if n_try == n or time.time() > t0 + budget_s * 0.97:
            continue
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = lp_linearize(h_tr, n_try, max_iter=80, max_step=0.15,
                                   t_end=t0 + budget_s * 0.975)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.978, max_sweeps=40)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 7: Final coord descent polish
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = coord_descent(best_h, n,
                                        t0 + budget_s * 0.999, max_sweeps=100)

    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
