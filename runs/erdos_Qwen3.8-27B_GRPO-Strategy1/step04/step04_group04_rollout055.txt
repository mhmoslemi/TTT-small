import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h))) * 2.0 / n

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14:
            return h
        h -= h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def compute_A(h, n):
    m = 2 * n - 1
    A = np.zeros((n, m))
    idx = np.arange(n)
    for k in range(m):
        s = k - (n - 1)
        if s >= 0:
            valid1 = (idx >= s) & (idx < n)
            valid2 = (idx < n - s)
            A[np.where(valid1)[0], k] += (1.0 - h[np.where(valid1)[0] - s])
            A[np.where(valid2)[0], k] += (-h[np.where(valid2)[0] + s])
        else:
            t = -s
            valid1 = (idx < n - t)
            valid2 = (idx >= t) & (idx < n)
            A[np.where(valid1)[0], k] += (1.0 - h[np.where(valid1)[0] + t])
            A[np.where(valid2)[0], k] += (-h[np.where(valid2)[0] - t])
    return A

def best_pair_move(h, n, corr, A, subset_size=4000):
    m = 2 * n - 1
    best_improvement = 0.0
    best_ij = None
    best_t = 0.0
    nc = len(corr)
    total_pairs = n * (n - 1) // 2
    npairs = min(subset_size, total_pairs)
    ii, jj = np.random.randint(0, n, size=npairs), np.random.randint(0, n, size=npairs)
    mask = ii < jj
    ii, jj = ii[mask], jj[mask]
    if len(ii) == 0:
        return 0.0, -1, -1, 0.0
    D = A[ii] - A[jj]
    t_lo = np.maximum(h[ii] - 1.0, h[jj] - 1.0 + 1.0)
    t_lo = np.maximum(-h[ii], h[jj] - 1.0)
    t_hi = np.minimum(1.0 - h[ii], h[jj])
    valid = t_hi > t_lo + 1e-14
    if not valid.any():
        return 0.0, -1, -1, 0.0
    ii, jj, t_lo, t_hi, D = ii[valid], jj[valid], t_lo[valid], t_hi[valid], D[valid]
    dm = np.abs(D).max(axis=1)
    mid = (t_lo + t_hi) * 0.5
    g0 = np.max(corr[np.newaxis, :] + t_lo[:, np.newaxis:1] * D, axis=1)
    g1 = np.max(corr[np.newaxis, :] + t_hi[:, np.newaxis:1] * D, axis=1)
    gm = np.max(corr[np.newaxis, :] + mid[:, np.newaxis:1] * D, axis=1)
    base = np.max(corr)
    improve = base - np.minimum(np.minimum(g0, g1), gm)
    good = improve > 1e-12
    if not good.any():
        return 0.0, -1, -1, 0.0
    gi = np.argmax(improve * good - (1 - good) * 1e30)
    i, j = int(ii[gi]), int(jj[gi])
    d = D[gi]
    c = corr
    def g(t):
        return float(np.max(c + t * d))
    res = minimize_scalar(g, bounds=(t_lo[gi], t_hi[gi]), method='bounded',
                          options={'xatol': 1e-12})
    t = res.x
    new_max = res.fun
    return base - new_max, i, j, float(t)

def subgradient_step(h, n, corr, A, lr, T=200):
    m = 2 * n - 1
    mmax = np.max(corr)
    shifted = np.clip(T * (corr - mmax), -300, 300)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    grad = (A.T @ w) * (2.0 / n)
    step = lr
    for _ in range(30):
        trial = proj(h - step * grad, n)
        c5_new = compute_c5(trial)
        c5_old = compute_c5(h)
        if c5_new < c5_old:
            return trial, c5_new, step
        step *= 0.5
    return h, compute_c5(h), 0.0

def subgrad_descent(h, n, t_end, lr_init=0.02):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    T = 100
    stall = 0
    while time.time() < t_end:
        corr = compute_corr(best_h)
        A = compute_A(best_h, n)
        trial_h, c5_new, used_lr = subgradient_step(best_h, n, corr, A, lr, T)
        if c5_new < best_c5 - 1e-15:
            best_h = trial_h
            best_c5 = c5_new
            T = min(T * 1.1, 5000)
            lr = min(lr * 1.05, 0.1)
            stall = 0
        else:
            lr *= 0.6
            stall += 1
            if stall > 40:
                break
    return best_h, best_c5

def pairwise_descent(h, n, t_end, n_subsets=50000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = compute_corr(best_h)
    A = compute_A(best_h, n)
    iters_since_recompute = 0
    iterations = 0
    while time.time() < t_end:
        if iters_since_recompute >= 50 or iterations == 0:
            corr = compute_corr(best_h)
            A = compute_A(best_h, n)
            iters_since_recompute = 0
        imp, i, j, t = best_pair_move(best_h, n, corr, A, n_subsets)
        if imp > 1e-14 and i >= 0:
            best_h[i] += t
            best_h[j] -= t
            best_h = np.clip(best_h, 0, 1)
            best_h = proj(best_h, n)
            new_c5 = compute_c5(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
            else:
                corr = compute_corr(best_h)
                A = compute_A(best_h, n)
            iters_since_recompute += 1
        else:
            corr = compute_corr(best_h)
            A = compute_A(best_h, n)
            iters_since_recompute = 0
        iterations += 1
        if iterations % 200 == 0:
            time.sleep(0)
    return best_h, best_c5

def make_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_smooth(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    n_peaks = rng.randint(3, 8)
    h = np.zeros(n)
    for _ in range(n_peaks):
        c = rng.uniform(0, 2)
        w = rng.uniform(0.1, 0.6)
        h += np.exp(-0.5 * ((x - c) / w) ** 2)
    h /= h.max() + 1e-10
    return proj(h, n)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        c = 0
        while v:
            c ^= 1
            v >>= 1
        h[i] = c
    # need to fix: use bit_count
    h = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    return proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s * 0.97

    starts = []
    if 'initial_h_values' in globals() and initial_h_values is not None:

        starts.append(h_bin)
        h_sm = make_smooth(n_try, seed + 100 + n_try)

    starts.append(thue_morse(180))

    best_h = starts[0].copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    for si, h0 in enumerate(starts):
        if time.time() > t_end:
            break
        n = len(h0)
        h0 = proj(h0.copy(), n)
        c5_cur = compute_c5(h0)
        if c5_cur > best_c5 + 0.001:

        time_left = t_end - time.time()
        if time_left < 30:
            break
        alloc = min(time_left, budget_s * 0.25)
        t_phase = t0 + (time.time() - t0) + alloc

        # Subgradient descent
        if time_left > 20:

        # Pairwise descent

        # Quick SLSQP polish
        if time.time() < t_end and c5_cur <= best_c5 + 5e-4:
            try:
                m = len(compute_corr(h0))
                active = np.where(compute_corr(h0) >= np.max(compute_corr(h0)) * 0.999)[0]
                dx = 2.0 / n
                def obj(x):
                    c = np.correlate(x, 1.0 - x, mode='full')
                    return float(np.max(c[active])) * dx
                res = minimize(obj, h0, method='SLSQP',
                               bounds=[(0,1)]*n,
                               constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2}],
                               options={'maxiter': 500, 'ftol': 1e-15})
                h_trial = proj(res.x, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_cur:
                    h0, c5_cur = h_trial, c5_trial
            except Exception:

        if c5_cur < best_c5:

    best_h = proj(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
