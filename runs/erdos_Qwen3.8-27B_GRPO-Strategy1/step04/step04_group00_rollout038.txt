```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, T):
    """Returns (smooth_c5_value, gradient) for the log-sum-exp smoothed C5."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    m = corr.max()
    exp_t = np.exp(T * (corr - m))
    exp_sum = exp_t.sum()
    w = exp_t / exp_sum  # softmax weights, length 2n-1
    val = (m + np.log(exp_sum) / T) * dx
    
    # Gradient: grad[j] = dx * sum_r w[r+n-1] * [(1-h[j-r])*1[0≤j-r<n] - h[j+r]*1[0≤j+r<n]]
    grad = np.zeros(n)
    for r in range(-(n-1), n):
        wr = w[r + n - 1]
        if wr < 1e-16:
            continue
        wr_dx = wr * dx
        # Term 1: (1-h[j-r]) for j-r in [0, n-1]
        # j-r in [0,n-1] → j in [r, r+n-1], intersect [0, n-1]
        js = max(0, r)
        je = min(n-1, r + n - 1)
        if js <= je:
            grad[js:je+1] += wr_dx * (1.0 - h[js - r : je - r + 1])
        # Term 2: -h[j+r] for j+r in [0, n-1]
        # j+r in [0,n-1] → j in [-r, -r+n-1], intersect [0, n-1]
        js2 = max(0, -r)
        je2 = min(n-1, -r + n - 1)
        if js2 <= je2:
            grad[js2:je2+1] -= wr_dx * h[js2 + r : je2 + r + 1]
    
    return val, grad

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        h[:] = 0.5
        return h
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=2000):
    def obj(x):
        v, _ = smooth_obj_and_grad(x, T)
        return v
    def grad(x):
        _, g = smooth_obj_and_grad(x, T)
        return g
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        h[:] = 0.5
        return h
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_t = np.exp(T * (corr - m))
    exp_sum = exp_t.sum()
    w = exp_t / exp_sum
    val = (m + np.log(exp_sum) / T) * dx
    grad = np.zeros(n)
    for r in range(-(n - 1), n):
        wr = w[r + n - 1]
        if wr < 1e-15:
            continue
        c = wr * dx
        js = max(0, r)
        je = min(n - 1, r + n - 1)
        if js <= je:
            grad[js:je + 1] += c * (1.0 - h[js - r:je - r + 1])
        js2 = max(0, -r)
        je2 = min(n - 1, -r + n - 1)
        if js2 <= je2:
            grad[js2:je2 + 1] -= c * h[js2 + r:je2 + r + 1]
    return val, grad

class SmoothObj:
    def __init__(self, T):
        self.T = T
    def __call__(self, x):
        v, _ = smooth_obj_grad(x, self.T)
        return v
    def grad(self, x):
        _, g = smooth_obj_grad(x, self.T)
        return g

class PenObj:
    def __init__(self, T, n, pen):

        self.n = n
        self.pen = pen
        self.ones_vec = np.ones(n)
    def __call__(self, x):

class SLSQPObj:
    def __init__(self, T):
        self.T = T
    def __call__(self, x):
        v, _ = smooth_obj_grad(x, self.T)
        return v
    def grad(self, x):
        _, g = smooth_obj_grad(x, self.T)
        return g
    def constraint(self, x):
        return x.sum() - x.shape[0] / 2.0
    def constraint_jac(self, x):
        return np.ones(x.shape[0])

def lbgfs_step(h, T, n, maxiter=2000, pen=1e6):

def slsqp_step(h, T, n, maxiter=1000):

    try:

def subgrad_step(h, T, n, alpha, n_topk=10):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = np.argsort(corr)[-n_topk:]
    grad = np.zeros(n)
    for idx in topk:
        r = idx - (n - 1)
        js = max(0, r)
        je = min(n - 1, r + n - 1)
        if js <= je:
            grad[js:je + 1] += (1.0 - h[js - r:je - r + 1])
        js2 = max(0, -r)
        je2 = min(n - 1, -r + n - 1)
        if js2 <= je2:
            grad[js2:je2 + 1] -= h[js2 + r:je2 + r + 1]
    grad *= dx / len(topk)
    h_new = project_h(h - alpha * grad)
    return h_new

def coord_opt_1d(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n - 1)])
    dx = 2.0 / n
    def f1d(hi):
        return np.max(corr_base + hi * coefs) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_1d(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

            pert[mask] = (0.08 + 0.15 * cool) * np.random.randn(mask.sum())
        elif mod == 3:
            block = max(2, n // 16)
            start_i = np.random.randint(0, n - block)
            pert = np.zeros(n)
            pert[start_i:start_i + block] = (0.1 + 0.2 * cool) * np.random.randn(block)
        else:
            pert = (0.03 + 0.08 * cool) * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T_bh = 2000 + 50000 * cool
        hc = lbgfs_step(h_pert, T_bh, n, maxiter=500, pen=1e7)
        c5v = compute_c5(hc)
        temp = 0.003 * cool + 0.0002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Coordinate descent
    t_phase5_end = t0 + budget_s * 0.82
    best_h, best_c5 = coord_descent(best_h, n, t_phase5_end, max_sweeps=300)

    # Phase 6: SLSQP polish with high T
    t_phase6_end = t0 + budget_s * 0.92
    for T in [100000, 500000, 2000000]:
        if time.time() > t_phase6_end:
            break
        hc = slsqp_step(best_h, T, n, maxiter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Final coordinate descent
    t_phase7_end = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t_phase7_end, max_sweeps=100)

    # Final SLSQP
    if time.time() < t0 + budget_s * 0.99:
        hc = slsqp_step(best_h, 1000000, n, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
