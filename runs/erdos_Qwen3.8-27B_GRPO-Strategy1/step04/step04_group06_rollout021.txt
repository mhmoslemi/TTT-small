import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h * (target / max(s, 1e-14)), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_optimize(h, i):
    n = len(h)
    old = h[i]
    h[i] = 0.0
    corr0 = get_corr(h)
    h[i] = old
    center = n - 1
    d = np.zeros(2 * n - 1)
    for l in range(-(n - 1), n):
        eff = 0.0
        if l != 0:
            if 0 <= i + l < n:
                eff -= h[i + l]
            if 0 <= i - l < n:
                eff += 1.0 - h[i - l]
        d[center + l] = eff
    lo, hi = 0.0, 1.0
    for _ in range(42):
        mid = (lo + hi) * 0.5
        vals = corr0 + mid * d
        vals[center] = corr0[center] + mid - mid * mid
        mv = vals.max()
        active = vals >= mv - 1e-10
        am = active.copy()
        am[center] = False
        if am.any():
            sl = d[am]
            lo_sg, hi_sg = float(sl.min()), float(sl.max())
        else:
            lo_sg, hi_sg = 0.0, 0.0
        if active[center]:
            sg = 1.0 - 2.0 * mid
            lo_sg = min(lo_sg, sg)
            hi_sg = max(hi_sg, sg)
        if lo_sg > 1e-13:
            hi = mid
        elif hi_sg < -1e-13:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_sweep(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(800):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            nv = coord_optimize(best_h, i)
            if abs(nv - best_h[i]) > 1e-12:
                best_h[i] = nv
                improved = True
        if not improved:
            break
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def _sm_fun(x, T):
    corr = get_corr(x)
    n = len(x)
    dx = 2.0 / n
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def _sm_grad(x, T):
    corr = get_corr(x)
    n = len(x)
    dx = 2.0 / n
    m = corr.max()
    ex = T * (corr - m)
    ex -= ex.max()
    w = np.exp(ex)
    w /= w.sum()
    grad = np.zeros(n)
    center = n - 1
    nz = np.where(w > 1e-14)[0]
    for idx in nz:
        l = idx - center
        wi = w[idx]
        if l >= 0:
            if l < n:
                grad[l:] += wi * (1.0 - x[:n - l])
            if l > 0:
                grad[:n - l] -= wi * x[l:]
        else:
            ll = -l
            if ll < n:
                grad[:n - ll] += wi * (1.0 - x[ll:])
            grad[ll:] -= wi * x[:n - ll]
    return grad * dx

def _eq_fun(x):
    return x.sum() - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x)) / len(x)

def slsqp_step(h, T, n, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, args=(T,), jac=_sm_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 160
        x0 = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x0) + 0.1 * np.sin(4 * np.pi * x0), 0, 1)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate convergence
    t1 = t0 + budget_s * 0.12
    best_h, best_c5 = coord_sweep(best_h, n, t1, rng)

    # Phase 2: SLSQP smoothing
    for T in [30, 100, 500, 2000]:
        if time.time() > t0 + budget_s * 0.25:
            break
        hc = slsqp_step(best_h, T, n, 600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        if time.time() < t0 + budget_s * 0.25:
            best_h, best_c5 = coord_sweep(best_h, n, min(time.time() + 4, t0 + budget_s * 0.25), rng)

    # Phase 3: Multi-start with diverse perturbations
    t3_end = t0 + budget_s * 0.50
    for trial in range(6):
        if time.time() > t3_end:
            break
        x_arr = np.linspace(0, 2, n, endpoint=False)
        if trial == 0:
            m = rng.randint(2, 8)
            phi = rng.uniform(0, 2 * np.pi)
            pert = 0.15 * np.sin(2 * np.pi * m * x_arr + phi)
        elif trial == 1:
            pert = 0.10 * np.random.randn(n)
        elif trial == 2:
            pert = np.zeros(n)
            c = rng.randint(n)
            w = rng.randint(8, 30)
            for di in range(-w, w + 1):
                idx = (c + di) % n
                pert[idx] = 0.2 * np.exp(-0.5 * (di / (w * 0.35)) ** 2) * rng.randn()
        elif trial == 3:

            i1, i2 = rng.choice(n, size=2, replace=False)
            pert[i1] = 0.3
            pert[i2] = -0.3
        elif trial == 4:

            pert = 0.12 * (np.sin(2 * np.pi * m1 * x_arr) + np.cos(2 * np.pi * m2 * x_arr + 0.7))
        else:
            pert = 0.25 * rng.randn(n)
        h_try = project(best_h + pert, n)
        h_try, c5_try = coord_sweep(h_try, n, min(time.time() + 12, t3_end), rng)
        if c5_try < best_c5:

            best_h = h_try

    # Phase 4: Simulated annealing
    T_anneal = 0.003
    t4_end = t0 + budget_s * 0.68
    for it in range(3000):
        if time.time() > t4_end:
            break
        r = rng.random()
        h_try = best_h.copy()
        if r < 0.65:
            i = rng.randint(n)
            h_try[i] = np.clip(h_try[i] + rng.normal(0, 0.12), 0, 1)
        elif r < 0.80:
            i, j = rng.choice(n, size=2, replace=False)
            h_try[i], h_try[j] = h_try[j], h_try[i]
        elif r < 0.90:
            i = rng.randint(n)
            j = (i + rng.randint(2, 10)) % n
            h_try[i] = np.clip(h_try[i] + rng.normal(0, 0.15), 0, 1)
            h_try[j] = np.clip(h_try[j] + rng.normal(0, 0.15), 0, 1)
        else:
            c = rng.randint(n)
            w = rng.randint(4, 15)
            amp = rng.uniform(0.05, 0.15)
            for di in range(-w, w + 1):
                idx = (c + di) % n
                h_try[idx] = np.clip(h_try[idx] + amp * np.exp(-0.5*(di/(w*0.4))**2) * rng.randn(), 0, 1)
        h_try = project(h_try, n)
        c5_try = compute_c5(h_try)
        dc = c5_try - best_c5
        if dc < 0 or rng.random() < np.exp(-dc / max(T_anneal, 1e-5)):
            best_h = h_try
            best_c5 = c5_try
            T_anneal = max(T_anneal * 0.9985, 5e-5)
        else:
            T_anneal = min(T_anneal * 1.002, 0.01)
        if it % 200 == 100:
            best_h, best_c5 = coord_sweep(best_h, n, min(time.time() + 3, t4_end), rng)

    # Phase 5: Extended coordinate descent
    t5_end = t0 + budget_s * 0.82
    best_h, best_c5 = coord_sweep(best_h, n, t5_end, rng)

    # Phase 6: Resolution experiment

        if time.time() > t0 + budget_s * 0.90:

        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n_try)
