def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))

def grad_smooth(h, T):

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            r = n - lag
            if r > 0:
                grad[:r] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:r])
        else:
            nl = -lag
            r = n - nl
            if r > 0:
                grad[nl:] += pk * (1.0 - h[:r])
                grad[:r] += pk * (-h[nl:])
    return grad * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def focused_slsqp(h, n, maxiter=3000):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val * 0.9998)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    def obj_f(x):
        c = np.correlate(x, 1.0 - x, mode='full')[active]
        return np.max(c) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_f, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj(np.clip(res.x, 0, 1), n)

def grad_descent(h, n, t_end, lr_init=0.002):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    T = 300
    stalled = 0
    for it in range(8000):
        if time.time() > t_end:
            break
        g = grad_smooth(best_h, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        step = lr / (gnorm + 1e-10)
        improved = False
        for _ in range(30):
            trial = proj(best_h - step * g, n)
            c5_t = compute_c5(trial)
            if c5_t < best_c5 - 1e-16:
                best_h = trial
                best_c5 = c5_t
                T = min(T * 1.04, 50000)
                stalled = 0
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.6
            stalled += 1
            if lr < 1e-13 or stalled > 200:
                break
    return best_h, best_c5

def fast_coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old_val = best_h[i]
            # Try a few candidate values
            candidates = [0.0, 1.0, old_val * 0.5, old_val * 1.5]
            best_cand = old_val
            best_c5_cand = best_c5
            for cv in candidates:
                if cv < -0.01 or cv > 1.01:
                    continue
                best_h[i] = np.clip(cv, 0.0, 1.0)
                best_h = proj(best_h, n)
                c5_try = compute_c5(best_h)
                if c5_try < best_c5_cand:
                    best_c5_cand = c5_try
                    best_cand = best_h[i]
            best_h[i] = best_cand
            best_h = proj(best_h, n)
            if best_c5_cand < best_c5 - 1e-15:
                best_c5 = best_c5_cand
                improved = True
        # Pair swaps (preserve sum automatically)
        n_swaps = max(5, n // 10)
        for _ in range(n_swaps):
            if time.time() > t_end:
                return best_h, best_c5
            i, j = np.random.choice(n, 2, replace=False)
            if abs(best_h[i] - best_h[j]) < 1e-12:
                continue
            best_h[i], best_h[j] = best_h[j], best_h[i]
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                improved = True
            else:
                best_h[i], best_h[j] = best_h[j], best_h[i]
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter=100):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    best_h = work_h.copy()
    best_c5 = work_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.88 * (it / float(n_iter))
        strat = it % 4
        if strat == 0:
            step = 0.01 + 0.06 * cool
            h_pert = work_h + step * np.random.randn(n)
        elif strat == 1:
            # Structured perturbation: shift a block
            w = np.random.randint(10, 40)
            start = np.random.randint(0, n - w)
            h_pert = work_h.copy()
            h_pert[start:start+w] += 0.03 * cool * (2.0 * np.random.rand(w) - 1.0)
        elif strat == 2:
            # Mirror-like perturbation
            h_pert = work_h.copy()
            h_pert[n//2:] += 0.02 * cool * np.random.randn(n//2)
        else:
            # Small random perturbation
            h_pert = work_h + 0.005 * np.random.randn(n)
        h_pert = proj(h_pert, n)
        T_opt = 800
        try:
            hc = slsqp_opt(h_pert, T_opt, n, maxiter=300, ftol=1e-12)
            c5v = compute_c5(hc)
            temp = 0.0015 * cool + 0.00005
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass
    return best_h, best_c5
