import numpy as np
from scipy.optimize import minimize_scalar
import time

_gs = {}

def _obj(v):
    old = _gs['old']
    corr = _gs['corr']
    L = _gs['L']
    c0 = _gs['c0']
    d = v - old
    cn = corr + d * L
    cn[c0] -= d * d
    return np.max(cn)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return np.clip(h, 0, 1)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_cd(h0, n, t_end, max_sweeps=5000):
    dx = 2.0 / n
    h = h0.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    c0 = n - 1
    
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = h[i]
            L = np.zeros(2 * n - 1)
            ks = n - 1 - i
            ke = n - 1 + i
            L[ks:ke+1] += 1.0 - h[i + np.arange(ks, ke+1) - (n-1)]
            ks2 = max(0, i)
            ke2 = min(2*n-2, i + n - 1)
            L[ks2:ke2+1] -= h[i - np.arange(ks2, ke2+1) + (n-1)]
            
            _gs['old'] = old
            _gs['corr'] = corr
            _gs['L'] = L
            _gs['c0'] = c0
            
            res = minimize_scalar(_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 40})
            bv, bf = res.x, res.fun
            f0 = _obj(0.0)
            f1 = _obj(1.0)
            if f0 < bf:
                bv, bf = 0.0, f0
            if f1 < bf:
                bv, bf = 1.0, f1
            
            if bf * dx < best_c5 - 1e-13:
                d = bv - old
                h[i] = bv
                corr += d * L
                corr[c0] -= d * d
                best_c5 = bf * dx
                improved = True
        if not improved:
            break
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    return h, best_c5

def hop(h, n, t_end, iters=200, seed=0):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    work_h = h.copy()
    work_c5 = best_c5
    
    for it in range(iters):
        if time.time() > t_end:
            break
        cool = max(0.03, 1.0 - 0.85 * it / max(iters, 1))
        step = 0.01 + 0.12 * cool
        h_p = work_h + step * rng.randn(n)
        h_p = project(h_p, n)
        hc, c5v = fast_cd(h_p, n, min(t_end, time.time() + 10), max_sweeps=300)
        temp = 0.0005 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    
    if work_c5 < best_c5:
        best_h, best_c5 = work_h, work_c5
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def make_periodic(n, period):
    h = np.zeros(n)
    half = period // 2
    for i in range(n):
        if (i % period) < half:
            h[i] = 1.0
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_periodic(160, 4)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Thorough coordinate descent (22%)
    end1 = t0 + budget_s * 0.22
    best_h, best_c5 = fast_cd(best_h, n, end1, max_sweeps=20000)
    
    # Phase 2: Basin hopping (33%)
    end2 = t0 + budget_s * 0.55
    best_h, best_c5 = hop(best_h, n, end2, iters=150, seed=seed)
    
    # Phase 3: Coordinate descent after hop (10%)
    end3 = t0 + budget_s * 0.65
    best_h, best_c5 = fast_cd(best_h, n, end3, max_sweeps=10000)
    
    # Phase 4: Multi-resolution exploration (23%)
    for n_try in [200, 240, 320, 120]:
        if time.time() > t0 + budget_s * 0.88:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = fast_cd(h_tr, n_try, t0 + budget_s * 0.89, max_sweeps=3000)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try
    
    # Phase 5: Final CD + short hop (9%)
    end5 = t0 + budget_s * 0.96
    best_h, best_c5 = fast_cd(best_h, n, end5, max_sweeps=20000)
    if time.time() < t0 + budget_s * 0.985:
        best_h, best_c5 = hop(best_h, n, t0 + budget_s * 0.985, iters=30, seed=seed+99)
        best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.995, max_sweeps=5000)
    
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
