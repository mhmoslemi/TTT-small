```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def cd_breakeven(h, n, t_end, n_sweeps=1000):
    """Fast coordinate descent using breakpoint analysis."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            h_old = best_h[i]
            # Compute coefficient: corr changes by s * coef when h[i] changes by s
            # corr[k] = sum_j h[j] * (1-h[j+(k-(n-1))])
            # When h[i] changes by s:
            #   In h[i]*(1-h[i+k]) term: contributes s*(1-h[i+k])
            #   In h[i-k]*(1-h[i]) term: contributes -s*h[i-k]
            # So coef[k] = (1-h[i+k]) - h[i-k] for appropriate k
            
            coef = np.zeros(2 * n - 1)
            for k in range(-(n-1), n):
                c = 0.0
                j1 = i - k  # h[i-k] appears as (1-h) factor
                j2 = i + k  # h[i+k] appears as h factor
                if 0 <= j1 < n:
                    c -= best_h[j1]  # h[i-k] is the (1-h) part: -(1 - (1-h[j1])) = -h[j1]
                if 0 <= j2 < n:
                    c += (1.0 - best_h[j2])  # h[i+k] is the h part
                coef[k + n - 1] = c
            
            # Minimize max(corr + s*coef) over s in [-h_old, 1-h_old]
            # This is convex piecewise linear in s
            # Use scipy's minimize_scalar for robustness
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            
            def f1d(s):
                return np.max(corr + s * coef)

                                  options={'xatol': 1e-10})

                best_h[i] += s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved:

    return best_h, best_c5
```

```python
    # result[j] = sum_i h[i] * p_centered[j + n - 1 - i]... 
    
    # This is getting confusing. Let me just use direct computation with numpy broadcasting.
    
    # For each i, s1[i] = sum over valid (i+lag) of p[m] * h[i+lag]
    # where lag = m - (n-1)
    
    # Build index matrix
    lags = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # For each i in range(n), valid lags are those with 0 <= i+lag < n
    # i.e., -i <= lag < n-i
    
    # Vectorized: create n x (2n-1) matrix of valid h values
    # indices = np.arange(n)[:,None] + lags[None,:]  # shape (n, 2n-1)
    # valid = (indices >= 0) & (indices < n)
    # h_vals = np.where(valid, h[clipped_indices], 0)
    # s1 = p @ h_vals  # shape (n,)
    
    # Let's do it properly:
    idx1 = np.arange(n)[:, None] + lags[None, :]  # (n, 2n-1)
    idx2 = np.arange(n)[:, None] - lags[None, :]  # (n, 2n-1)
    
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    h1 = np.where(valid1, h[np.clip(idx1, 0, n-1)], 0.0)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    
    s1 = p @ h1  # (n,)
    s2 = p @ h2  # (n,)
    
    grad = dx * (1.0 - s1 - s2)
    return grad
```

Actually, I realize there might be an even better approach. Since the objective is max of bilinear forms, and we're constrained to a polytope (hbox with sum constraint), we can think of this as a **min-max** problem:

min_h max_k F_k(h)

where F_k(h) = dx * Σ_i h[i] * (1-h[i+k])

For a fixed k, F_k is bilinear. The max makes it non-convex.

But here's a trick: We can use **mirror descent** or **follow-the-regularized-leader** (FTRL) in the h-space, treating the k-index as an adversarial player. The regret bound gives us a guarantee.

In each iteration:
1. The "adversary" picks k* = argmax_k F_k(h_t)
2. We update: h_{t+1} = argmin_{h ∈ K} <∂F_{k*}/∂h · (h - h_t)> + (1/η) D(h, h_t)

where K = {h : 0≤h≤1, sum(h)=n/2} and D is a divergence (e.g., KL or L2).

With L2: h_{t+1} = P_K(h_t - η * ∂F_{k*}/∂h_t)

This is just projected gradient descent in the direction of the active constraint! And it's very simple.

The step size η should decrease over time. With η_t = c/sqrt(t), we get O(1/sqrt(T)) regret, hence an O(1/sqrt(T)) gap to the optimum.

For our purposes, this might not be the fastest, but it's very simple and robust. Let me combine it with CD for local refinement.

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def make_indices(n):
    lags = np.arange(2 * n - 1) - (n - 1)
    ii = np.arange(n)[:, None]
    idx1_c = np.clip(ii + lags[None, :], 0, n - 1)
    idx2_c = np.clip(ii - lags[None, :], 0, n - 1)
    v1 = (ii + lags[None, :]) >= 0
    v1 &= (ii + lags[None, :]) < n
    v2 = (ii - lags[None, :]) >= 0
    v2 &= (ii - lags[None, :]) < n
    return idx1_c, v1, idx2_c, v2

def smooth_grad(h, T, idx):
    n = len(h)
    dx = 2.0 / n
    idx1_c, v1, idx2_c, v2 = idx
    corr = compute_corr(h)
    sh = T * corr * dx
    sh -= sh.max()
    p = np.exp(sh)
    p /= p.sum()
    h1 = np.where(v1, h[idx1_c], 0.0)
    h2 = np.where(v2, h[idx2_c], 0.0)
    s1 = (p[None, :] * h1).sum(axis=1)
    s2 = (p[None, :] * h2).sum(axis=1)
    return dx * (1.0 - s1 - s2)

def smooth_gd_momentum(h, n, T, n_iter, lr, momentum=0.85, t_end=None):
    h = h.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idx = make_indices(n)
    for it in range(n_iter):
        if t_end is not None and time.time() > t_end:
            break
        grad = smooth_grad(h, T, idx)
        v = momentum * v - lr * grad
        h = project_to_sum(h + v, n / 2)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, n, t_end, n_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            coef = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                c = 0.0
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    c -= best_h[j1]
                if 0 <= j2 < n:
                    c += (1.0 - best_h[j2])
                coef[k + n - 1] = c
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            res = minimize_scalar(lambda s: np.max(corr + s * coef),
                                  bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-11})
            s_new = res.x
            if abs(s_new) > 1e-13:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def fourier_perturb(h, n, rng, strength):
    h_p = h.copy()
    for mode in range(1, 8):
        amp = strength / mode
        if rng.rand() < 0.5:
            h_p += amp * np.cos(2 * np.pi * mode * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        else:
            h_p += amp * np.sin(2 * np.pi * mode * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    return project_to_sum(h_p, n / 2)

def block_swap_perturb(h, n, rng):
    h_p = h.copy()
    blen = rng.randint(8, 40)
    a = rng.randint(0, n - blen)
    b = rng.randint(0, n - blen)
    if abs(a - b) < blen:
        return h_p
    tmp = h_p[a:a+blen].copy()
    h_p[a:a+blen] = h_p[b:b+blen]
    h_p[b:b+blen] = tmp
    return project_to_sum(h_p, n / 2)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(224)
        best_h[:112] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick CD (10%)
    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = coord_descent(best_h, n, t1, n_sweeps=200)

    # Phase 2: Smooth GD with momentum, adaptive T (20%)
    t2 = t0 + budget_s * 0.30
    T_list = [50, 200, 800, 3000, 10000]
    for T in T_list:
        if elapsed() > t2:
            break
        n_it = min(800, int(budget_s * 0.03 / (2 * n / 1e6 + 0.001)))
        lr = 0.5 / max(T * 0.1, 1.0)
        hc, c5v = smooth_gd_momentum(best_h, n, T, n_it, lr, momentum=0.8, t_end=t2)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v
        hc2, c5v2 = coord_descent(best_h, n, min(elapsed() + 3, t2), n_sweeps=30)
        if c5v2 < best_c5:
            best_h, best_c5 = hc2, c5v2

    # Phase 3: Basin hopping with diverse perturbations (35%)
    t3 = t0 + budget_s * 0.65
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = int(budget_s * 0.35)
    it_count = 0
    while elapsed() < t3:
        it_count += 1
        cool = max(1.0 - 0.95 * (it_count / float(n_iter + 50)), 0.03)
        r = rng.rand()
        if r < 0.25:
            step = 0.02 + 0.10 * cool
            h_p = work_h + step * rng.randn(n)
        elif r < 0.45:
            h_p = fourier_perturb(work_h, n, rng, 0.05 + 0.15 * cool)
        elif r < 0.60:
            h_p = block_swap_perturb(work_h, n, rng)
        elif r < 0.75:
            step = 0.01 + 0.05 * cool
            mask = rng.rand(n) < 0.25
            h_p = work_h.copy()
            h_p[mask] += step * rng.randn(mask.sum())
        else:
            shift = rng.randint(-5, 6)
            h_p = np.roll(work_h, shift)
            h_p = 0.8 * h_p + 0.2 * work_h
        h_p = project_to_sum(h_p, n / 2)

        # Local optimization: quick smooth GD + CD
        T_loc = int(200 + 3000 * cool)
        lr_loc = 0.4 / max(T_loc * 0.1, 1.0)
        hc, c5v = smooth_gd_momentum(h_p, n, T_loc, 100, lr_loc, momentum=0.7,
                                      t_end=min(t3, elapsed() + 2.0))
        hc, c5v = coord_descent(hc, n, min(t3, elapsed() + 2.0), n_sweeps=8)

        temp = 0.003 * cool + 0.00005
        if c5v < work_c5 or (c5v - work_c5) < temp and rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 4: Multi-resolution (10%)
    t4 = t0 + budget_s * 0.75
    for n_try in [256, 320, 200]:
        if elapsed() > t4 or n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2)
        hc, c5v = coord_descent(h_i, n_try, min(elapsed() + 3, t4), n_sweeps=40)
        hc, c5v2 = smooth_gd_momentum(hc, n_try, 2000, 200, 0.02, t_end=min(t4, elapsed() + 3))
        if c5v2 < c5v:
            c5v = c5v2
        if c5v < best_c5:
            best_c5, best_h, n = c5v, hc, n_try

    # Phase 5: Final intensive CD (20%)
    t5 = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, n, t5, n_sweeps=1000)

    # Final smooth GD pass
    if elapsed() < t5 + budget_s * 0.03:
        hc, c5v = smooth_gd_momentum(best_h, n, 5000, 300, 0.01, momentum=0.9,
                                      t_end=t0 + budget_s * 0.97)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, n_sweeps=100)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
