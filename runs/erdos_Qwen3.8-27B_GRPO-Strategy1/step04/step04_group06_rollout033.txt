            result = np.clip(result, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    dx = 2.0 / len(h)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    weights = np.exp(T * (corr - m))
    weights /= weights.sum()
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(corr)):

            m2 = -k
            grad[:n-m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n-m2]

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def waterfill_project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        free = (result > 0.01) & (result < 0.99)
        if free.sum() > 0:
            result[free] += diff / free.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    ev = np.exp(T * (corr - m))
    sw = ev.sum()
    weights = ev / sw
    val = (m + np.log(sw) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for idx in active:
        w = weights[idx]
        k = idx - center
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < k < n:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        elif -n < k < 0:
            m2 = -k
            grad[:n-m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n-m2]
    return val, grad * dx

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_obj_grad(x, T)[0]
    def jac(x):
        return smooth_obj_grad(x, T)[1]
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = ({'type': 'eq', 'fun': con_fun},)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return waterfill_project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, max_sweeps=200, t_end=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    center = n - 1
    dx = 2.0 / n

        if t_end is not None and time.time() > t_end:
            break
        improved = False
        order = list(range(n))
        np.random.shuffle(order)

def compute_c5(h):
    n = len(h)

def waterfill_project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))

            hi = mid
        else:

    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

        free = (result > 0.01) & (result < 0.99)

def smooth_obj_val(h, T):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
