        if c5_now < best_c5 - 1e-14:

            improved = True
        elif not improved:

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_h(h):
    n = len(h)

        tot = room.sum()

            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)

                       options={'maxiter': maxiter, 'ftol': 1e-15})

                changed = True

            if c5_now < best_c5 - 1e-14:

        elif sweep > 3:

def subgrad_direction(h, n, topk=8):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-topk:]
    grad = np.zeros(n)
    for k in top_idx:
        c = np.zeros(n)
        kr = np.arange(2 * n - 1)

            idx_a = i + k - (n - 1)
