def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def to_binary(h, n):
    n1 = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n1]] = 1.0
    return b

def project_cont(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h0, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_cont(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=300):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            def obj_v(v):
                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                return np.max(cn)
            v0 = obj_v(0.0)
            v1r = obj_v(1.0)
            res = minimize_scalar(obj_v, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            best_v = res.x
            best_f = res.fun
            if v0 < best_f:
                best_v, best_f = 0.0, v0
            if v1r < best_f:
                best_v, best_f = 1.0, v1r
            if best_f * dx < best_c5 - 1e-14:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def to_binary(h, n):
    n1 = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n1]] = 1.0
    return b

def project_cont(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def _sum_con(x, n):
    return x.sum() - n / 2.0

def slsqp_opt(h0, T, n, maxiter=500, ftol=1e-14):
    res = minimize(smooth_c5, h0, args=(T,), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_cont(np.clip(res.x, 0, 1), n)

def _obj_coord(v, corr, old, coeff, c0_idx):
    cn = corr + (v - old) * coeff
    cn[c0_idx] = corr[c0_idx] + (v - old) * (1.0 - v - old)
    return np.max(cn)

def coord_descent(h, n, t_end, max_sweeps=300):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx2 = i + ks
            v2 = (idx2 >= 0) & (idx2 < n)
            idx1 = i - ks
            v1 = (idx1 >= 0) & (idx1 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v2] += 1.0 - best_h[idx2[v2]]
            coeff[v1] -= best_h[idx1[v1]]
            res = minimize_scalar(_obj_coord, bounds=(0.0, 1.0), method='bounded',
                                  args=(corr, old, coeff, c0_idx), options={'xatol': 1e-10})
            best_v, best_f = res.x, res.fun
            val0 = _obj_coord(0.0, corr, old, coeff, c0_idx)
            val1 = _obj_coord(1.0, corr, old, coeff, c0_idx)
            if val0 < best_f:
                best_v, best_f = 0.0, val0
            if val1 < best_f:
                best_v, best_f = 1.0, val1
            if best_f * dx < best_c5 - 1e-14:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_sa(h_bin, n, t_end, max_rounds=30000, T_init=0.5, T_min=0.003):

    best_c5 = compute_c5(best_h)
    work_h = h_bin.copy()
    work_c5 = best_c5

    cool_rate = 0.9997
