            return h
        h = h + room * (diff / tot)

            return h
        h = h * (1.0 + diff / s)

            idx = rng.choice(n, nf, replace=False)

        else:
            h_p = 1.0 - work_h + (0.05 + 0.1 * cool) * rng.randn(n)
        h_p = project(h_p, n)
        hc, c5v = coord_descent(h_p, n, time.time() + 4.0, max_sweeps=40, recompute_every=4)
        temp = 0.0015 * cool + 0.0001
        if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def _slsqp_smooth(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

_slsqp_state = {'T': 100, 'n': 160}

def _slsqp_obj(x):
    return _slsqp_smooth(x, _slsqp_state['T'], _slsqp_state['n'])

def _slsqp_con(x):
    return x.sum() - _slsqp_state['n'] / 2.0

def slsqp_opt(h0, T, n, maxiter=400):
    _slsqp_state['T'] = T
    _slsqp_state['n'] = n
    res = minimize(_slsqp_obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _slsqp_con, 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Heavy coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.20, max_sweeps=3000)

    # Phase 2: Binary rounding exploration
    if elapsed() < budget_s * 0.28:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project(h_bin, n)
        h_bin, c5_bin = coord_descent(h_bin, n, t0 + budget_s * 0.30, max_sweeps=600)
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, h_bin

    # Phase 3: SLSQP with increasing T
    if elapsed() < budget_s * 0.40:
        for T in [50, 200, 800]:
            if time.time() > t0 + budget_s * 0.40:
                break
            try:
                hc = slsqp_opt(best_h, T, n, maxiter=500)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
            except Exception:
                pass

    # Phase 4: Coordinate descent after SLSQP
    if elapsed() < budget_s * 0.47:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.47, max_sweeps=1500)

    # Phase 5: Basin hopping with diverse perturbations
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.70, max_iters=500)

    # Phase 6: Post-hop refinement
    if elapsed() < budget_s * 0.80:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.80, max_sweeps=1500)

    # Phase 7: Second basin hopping round (cooler)
    if elapsed() < budget_s * 0.86:
        best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.86, max_iters=200)
        if elapsed() < budget_s * 0.88:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.88, max_sweeps=800)

    # Phase 8: Try alternate resolutions
    for n_try in [144, 180, 200, 240]:
        if elapsed() > budget_s * 0.93:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, time.time() + 12, max_sweeps=400)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 9: Final polish
    if elapsed() < budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=1000)
    if elapsed() < budget_s * 0.98:
        for T in [3000, 10000]:
            if time.time() > t0 + budget_s * 0.98:
                break
            try:
                hc = slsqp_opt(best_h, T, n, maxiter=2000)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
            except Exception:
                pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
