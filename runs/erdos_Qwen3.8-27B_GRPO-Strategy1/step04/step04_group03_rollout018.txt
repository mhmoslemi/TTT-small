import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent_exact(h, n, t_end, n_sweeps=200):
    """Coordinate descent with exact 1D convex subproblem via ternary search."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            coef = np.zeros(2 * n - 1)
            for k_idx in range(-(n - 1), n):
                c = 0.0
                j1 = i - k_idx
                j2 = i + k_idx
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[k_idx + n - 1] = c
            
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            
            # Ternary search on convex function f(s) = max(corr + s*coef)
            a, b = lo, hi
            for _ in range(50):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(corr + m1 * coef)
                f2 = np.max(corr + m2 * coef)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            
            s_new = (a + b) / 2.0
            if abs(s_new) > 1e-12:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def coord_descent(h_init, t_end, n_sweeps=500):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = np.max(compute_corr(best_h)) * (2.0/n)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            coef = np.zeros(2 * n - 1)
            for k in range(-(n-1), n):
                j1 = i + k
                j2 = i - k
                c = 0.0
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[k + n - 1] = c
            
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            
            # Ternary search on convex f(s) = max(corr + s*coef)
            a, b = lo, hi
            for _ in range(40):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(corr + m1 * coef)
                f2 = np.max(corr + m2 * coef)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            s_new = (a + b) / 2.0
            if abs(s_new) > 1e-10:
                best_h[i] = np.clip(h_old + s_new, 0, 1)
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

        else:

            x = np.linspace(0, 2, n, endpoint=False)
            freqs = np.arange(1, 8)
            h_p = work_h.copy()
            for k in freqs:

                h_p += amp * (np.cos(2*np.pi*k*x/2) + np.sin(2*np.pi*k*x/2))
