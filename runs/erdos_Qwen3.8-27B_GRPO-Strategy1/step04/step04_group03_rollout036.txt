import numpy as np

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_to_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    d = target - s
    if abs(d) < 1e-15:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-15:
            return h
        return np.clip(h + r * (d / t), 0.0, 1.0)
    else:
        if s < 1e-15:
            return h
        return np.clip(h * (1.0 + d / s), 0.0, 1.0)

def corr_full(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def solve_1d_pwl(a, b, lo, hi):
    if hi - lo < 1e-14:
        return 0.5 * (lo + hi)
    xl, xh = lo, hi
    for _ in range(45):
        xm = 0.5 * (xl + xh)
        v = a + b * xm
        mv = v.max()
        act = b[v >= mv - 1e-11]
        if len(act) == 0:
            act = b[np.argmax(v):np.argmax(v)+1]
        if act.min() >= -1e-12:
            xh = xm
        elif act.max() <= 1e-12:
            xl = xm
        else:
            break
    return 0.5 * (xl + xh)

def cd_optimize(h, t_end, n_sweeps=2000):

    bc5 = compute_c5(best)
    dx = 2.0 / n

    m = 2 * n - 1

    for _ in range(n_sweeps):
        if time.time() > t_end:
            break

def slsqp_polish(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0-x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T*(corr-m))))/T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(np.clip(res.x, 0, 1), n/2.0)
    except:
        return h

min_{x,y} max_k (a_k + b_k*x + c_k*y) subject to 0≤x≤1, 0≤y≤1

This is a 2D LP (minimize max of 447 planes in a 2D box). Can be solved with scipy.linprog or manually.

For 2D: the minimum of max of m linear functions in a 2D box occurs at:
- A corner of the box (4 points)
- Or the intersection of two plane boundaries on the box edges
- Or the intersection of three plane boundaries in the interior

For m planes, there are O(m²) intersection points. With m=447, that's about 100k points. Checking each is O(m) (evaluate max). Total: 100k * 447 = 45M per pair step. For n/2 pairs per sweep: 112 * 45M = 5B. Too slow.

Let me skip pair CD. The single-coordinate CD should be sufficient.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_to_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    d = target - s
    if abs(d) < 1e-15:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-15:
            return h
        return np.clip(h + r * (d / t), 0.0, 1.0)
    else:
        if s < 1e-15:
            return h
        return np.clip(h * (1.0 + d / s), 0.0, 1.0)

def corr_full(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def solve_1d_pwl(a, b, lo, hi):
    if hi - lo < 1e-14:
        return 0.5 * (lo + hi)
    xl, xh = lo, hi
    for _ in range(48):
        xm = 0.5 * (xl + xh)
        v = a + b * xm
        mv = v.max()
        act = b[v >= mv - 1e-11]
        if len(act) == 0:
            idx = int(np.argmax(v))
            act = b[idx:idx+1]
        if act.min() >= -1e-12:
            xh = xm
        elif act.max() <= 1e-12:
            xl = xm
        else:
            break
    return 0.5 * (xl + xh)

def cd_optimize(h, t_end, n_sweeps=2000):
    n = len(h)
    best = h.copy().astype(float)
    bc5 = compute_c5(best)
    dx = 2.0 / n
    corr = corr_full(best)
    m = 2 * n - 1
    coeff = np.zeros(m)
    for _ in range(n_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            ho = best[i]
            for k in range(m):
                l = k - n + 1
                c = 0.0
                jp = i + l
                jm = i - l
                if 0 <= jp < n:
                    c += 1.0 - best[jp]
                if 0 <= jm < n:
                    c -= best[jm]
                coeff[k] = c
            a = corr - ho * coeff
            hn = solve_1d_pwl(a, coeff, 0.0, 1.0)
            if abs(hn - ho) > 1e-13:
                d = hn - ho
                best[i] = hn
                corr += d * coeff
                c5n = np.max(corr) * dx
                if c5n < bc5:
                    bc5 = c5n
                    imp = True
        if not imp:
            break
    return best, bc5

def fourier_perturb(h, amp, nmode, rng):
    n = len(h)
    hf = np.fft.rfft(h.astype(float))
    hf[0] = 0.0
    nm = min(nmode, len(hf))
    if nm > 1:
        sz = nm - 1
        hf[1:nm] += amp * (rng.standard_normal(sz) + 1j * rng.standard_normal(sz))
    return np.fft.irfft(hf, n=n)

def smooth_max_obj(x, n, T):
    dx = 2.0 / n
    xc = np.clip(x, 0.0, 1.0)
    co = np.correlate(xc, 1.0 - xc, mode='full')
    m = co.max()
    return (m + np.log(np.sum(np.exp(T * (co - m)))) / T) * dx

def slsqp_n(n):
    return n / 2.0

def slsqp_polish(h, T, n, maxiter=400, ftol=1e-14):
    dx = 2.0 / n
    target = n / 2.0
    try:
        def obj(x):
            return smooth_max_obj(x, n, T)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
        res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(np.clip(res.x, 0, 1), target)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(128)
        best_h[:64] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial intensive CD (8% budget)
    t1 = t0 + budget_s * 0.08
    best_h, best_c5 = cd_optimize(best_h, t1, n_sweeps=500)

    # SLSQP polish at start
    if time.time() < t0 + budget_s * 0.12:
        hc = slsqp_polish(best_h, T=500, n=n, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Phase 2: Basin hopping with diverse perturbations (48% budget)
    t2 = t0 + budget_s * 0.60
    work = best_h.copy()
    wc5 = best_c5
    n_hop = 0
    while time.time() < t2:
        n_hop += 1
        cool = max(1.0 - 0.65 * n_hop / 400.0, 0.03)
        amp = 0.004 + 0.055 * cool
        r = rng.rand()
        if r < 0.35:
            h_p = fourier_perturb(work, amp, 20, rng)
        elif r < 0.55:
            h_p = work + amp * rng.randn(n)
        elif r < 0.75:
            p = rng.randint(0, max(1, n - 12))
            h_p = work.copy()
            h_p[p:p+12] += amp * 2.5 * rng.randn(12)
        elif r < 0.9:
            sh = rng.randint(-6, 7)
            h_p = 0.6 * np.roll(work, sh) + 0.4 * work
        else:
            # High-frequency burst
            h_p = work.copy()
            idx = rng.choice(n, size=max(2, n//8), replace=False)
            h_p[idx] += amp * 3.0 * rng.randn(len(idx))
        h_p = project_to_sum(h_p, n / 2.0)
        h_q, c5_q = cd_optimize(h_p, time.time() + 2.5, n_sweeps=100)
        temp = 0.0015 * cool + 0.00002
        if c5_q < wc5 or rng.rand() < np.exp(-(c5_q - wc5) / max(temp, 1e-12)):
            work, wc5 = h_q, c5_q
            if c5_q < best_c5:
                best_h, best_c5 = h_q.copy(), c5_q

    if wc5 < best_c5:
        best_h, best_c5 = work.copy(), wc5

    # Phase 3: Multi-resolution refinement (15% budget)
    t3 = t0 + budget_s * 0.75
    for n_try in [128, 160, 192, 256, 320]:
        if n_try == n or time.time() > t3:
            continue
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, n_try, endpoint=False)
        hi_ = np.interp(xn, xo, best_h)
        hi_ = project_to_sum(hi_, n_try / 2.0)
        hi_, c5v = cd_optimize(hi_, time.time() + 6.0, n_sweeps=200)
        if c5v < best_c5:
            best_h, best_c5, n = hi_, c5v, n_try

    # Phase 4: SLSQP with high smoothness parameter
    if time.time() < t0 + budget_s * 0.85:
        for T in [2000, 10000]:
            if time.time() > t0 + budget_s * 0.85:
                break
            hc = slsqp_polish(best_h, T=T, n=n, maxiter=800, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v

    # Phase 5: Final intensive CD (15% budget)
    t5 = t0 + budget_s * 0.99
    best_h, best_c5 = cd_optimize(best_h, t5, n_sweeps=10000)

    # Tiny final jitter attempts
    if time.time() < t0 + budget_s * 0.995:
        for _ in range(5):
            if time.time() > t0 + budget_s * 0.995:
                break
            h_j = best_h + 0.002 * rng.randn(n)
            h_j = project_to_sum(h_j, n / 2.0)
            h_j, c5v = cd_optimize(h_j, time.time() + 3.0, n_sweeps=100)
            if c5v < best_c5:
                best_h, best_c5 = h_j, c5v

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
