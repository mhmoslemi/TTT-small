                best[oi] = 0.0
                best[zi] = 1.0
                c5 = compute_c5(best)
                best[oi] = 1.0
                best[zi] = 0.0
                delta = best_c5 - c5
                if delta > best_delta:
                    best_delta = delta
                    best_oi, best_zi = oi, zi
        if best_oi >= 0:
            best[best_oi] = 0.0
            best[best_zi] = 1.0
            best_c5 = compute_c5(best)
        else:
            break
    return best, best_c5

def smooth_grad(h, T):
    dx = 2.0 / len(h)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    lam = np.exp(T * (corr - m))
    lam /= lam.sum()
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        l = lam[ki]
        if l < 1e-14:
            continue
        k = ki - (n - 1)
        if k >= 0:
            if k < n:
                g[:n - k] += l * (1.0 - h[k:])
            if k > 0:
                g[k:] -= l * h[:n - k]
        else:
            kk = -k
            g[kk:] += l * (1.0 - h[:n - kk])
            g[:n - kk] -= l * h[kk:]
    return g * dx

def grad_phase(h, n, t_end, n_steps=3000):
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    v = np.zeros(n)
    for it in range(n_steps):
        if time.time() > t_end:
            break
        T = 200.0 * (1.0 + 0.01 * it)
        g = smooth_grad(cur, T)
        alpha = 0.04 / (1.0 + 0.003 * it)
        v = 0.9 * v - alpha * g
        cur = project(cur + v, n)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best = cur.copy()
            best_c5 = c5
    return best, best_c5

def make_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[:n // 2] = 1.0
    h += 0.02 * rng.randn(n)
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    iv = globals().get('initial_h_values', None)
    if iv is not None and len(iv) > 0:
        best_h = np.array(iv, dtype=float)
    else:
        best_h = make_init(160, seed)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: CD polish initial (12%)
    t1 = t0 + budget_s * 0.12
    best_h, best_c5 = cd_polish(best_h, n, t1, 400)

    # Phase 2: Gradient phase for global movement (10%)
    t2 = t0 + budget_s * 0.22
    hg, c5g = grad_phase(best_h, n, t2, 2000)
    if c5g < best_c5:
        best_h, best_c5 = hg, c5g

    # Phase 3: Binary local search (18%)
    t3 = t0 + budget_s * 0.40
    hb, cb = bin_search(best_h, n, t3)
    if cb < best_c5:
        best_h, best_c5 = hb, cb

    # Phase 4: Heavy CD (30%)
    t4 = t0 + budget_s * 0.70
    best_h, best_c5 = cd_polish(best_h, n, t4, 8000)

    # Phase 5: Diversification + CD (20%)
    t5 = t0 + budget_s * 0.90
    for r in range(20):
        if time.time() > t5:
            break
        h_p = best_h + 0.04 * np.random.randn(n)
        h_p = project(h_p, n)
        h_c, c5_c = cd_polish(h_p, n, min(t5, time.time() + budget_s * 0.025), 100)
        if c5_c < best_c5:
            best_h, best_c5 = h_c, c5_c

    # Phase 6: Binary restart from current (5%)
    t6 = t0 + budget_s * 0.95
    hb2, cb2 = bin_search(best_h, n, t6)
    if cb2 < best_c5:
        best_h, best_c5 = hb2, cb2

    # Phase 7: Final CD polish (5%)
    t7 = t0 + budget_s * 0.995
    best_h, best_c5 = cd_polish(best_h, n, t7, 2000)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
