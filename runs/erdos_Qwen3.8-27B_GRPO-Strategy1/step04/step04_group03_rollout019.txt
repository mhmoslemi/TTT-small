    best_h, best_c5 = coord_descent(best_h, n, t1, n_sweeps=600)

    # Phase 2: Multi-resolution exploration (23% budget)
    t2 = t0 + budget_s * 0.35
    for n_try in [64, 96, 128, 160, 192]:
        if n_try == n or elapsed() > t2 - 5:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2)
        trials = 4
        for tr in range(trials):
            if elapsed() > t2 - 3:
                break
            if tr == 0:
                h_try = h_i.copy()
            elif tr == 1:
                h_try = h_i + 0.06 * np.random.randn(n_try)
            elif tr == 2:
                freq = np.random.uniform(1.0, 4.0)
                xx = np.linspace(0, 2, n_try, endpoint=False)
                h_try = h_i + 0.05 * np.sin(2 * np.pi * freq * xx + np.random.uniform(0, 6.28))
            else:
                h_try = np.sort(h_i)[::-1]  # antitone structure
            h_try = project_to_sum(h_try, n_try / 2)
            t_sub = min(elapsed() + 15, t2 - 2)
            h_o, c5o = coord_descent(h_try, n_try, t_sub, n_sweeps=80)
            if c5o < best_c5 * (n / float(n_try)):
                x_f = np.linspace(0, 2, n_try, endpoint=False)
                x_full = np.linspace(0, 2, n, endpoint=False)
                h_back = np.interp(x_full, x_f, h_o)
                h_back = project_to_sum(h_back, n / 2)
                h_back, c5b = coord_descent(h_back, n, min(elapsed() + 10, t2 - 2), n_sweeps=60)
                if c5b < best_c5:
                    best_h, best_c5 = h_back, c5b

    # Phase 3: Basin hopping with diverse moves (35% budget)
    t3 = t0 + budget_s * 0.70
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 80
    for it in range(n_iter):
        if elapsed() > t3 - 3:
            break
        cool = max(1.0 - 0.95 * (it / float(n_iter)), 0.02)
        r = np.random.rand()
        if r < 0.2:
            freq = np.random.uniform(0.5, 6.0)
            ph = np.random.uniform(0, 2 * np.pi)
            amp = 0.02 + 0.1 * cool
            xx = np.linspace(0, 2, n, endpoint=False)
            h_p = work_h + amp * np.sin(2 * np.pi * freq * xx + ph)
        elif r < 0.4:
            blen = np.random.randint(max(4, n // 10), n // 4)
            bs = np.random.randint(0, n - blen)
            sh = np.random.randint(-blen // 2, blen // 2 + 1)
            h_p = work_h.copy()
            blk = h_p[bs:bs + blen].copy()
            h_p[bs:bs + blen] = 0.5
            np2 = max(0, min(n - blen, bs + sh))
            h_p[np2:np2 + blen] = 0.7 * blk + 0.3 * h_p[np2:np2 + blen]
        elif r < 0.6:
            step = 0.02 + 0.12 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.8:
            nflip = np.random.randint(n // 15, n // 4)
            idx = np.random.choice(n, nflip, replace=False)
            h_p = work_h.copy()
            h_p[idx] = np.clip(h_p[idx] + np.random.choice([-1, 1], nflip).astype(float)
                               * np.random.uniform(0.1, 0.4, nflip), 0, 1)
        else:
            # Mix with a random binary structure
            h_bin = (np.random.rand(n) < 0.5).astype(float)
            h_bin = project_to_sum(h_bin, n / 2)
            h_p = (1 - 0.3) * work_h + 0.3 * h_bin

        h_p = project_to_sum(h_p, n / 2)
        per_trial = max(2.0, 0.8 * (t3 - t0) / n_iter)
        t_sub = min(elapsed() + per_trial, t3 - 2)
        h_o, c5o = coord_descent(h_p, n, t_sub, n_sweeps=30)
        temp = 0.0025 * cool + 5e-5
        accept = c5o < work_c5 or (np.random.rand() < np.exp(-(c5o - work_c5) / max(temp, 1e-10)))
        if accept:
            work_h, work_c5 = h_o, c5o
            if c5o < best_c5:
                best_h, best_c5 = h_o.copy(), c5o

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 4: SLSQP polish at various T (15% budget)
    t4 = t0 + budget_s * 0.85
    for T in [300, 1500, 8000, 40000]:
        if elapsed() > t4 - 2:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=1000, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass
        # Quick CD after each SLSQP
        if elapsed() < t4 - 2:
            best_h, best_c5 = coord_descent(best_h, n, min(elapsed() + 8, t4 - 2), n_sweeps=80)

    # Phase 5: Final intensive coordinate descent (15% budget)
    t5 = t0 + budget_s * 0.995
    best_h, best_c5 = coord_descent(best_h, n, t5, n_sweeps=800)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
