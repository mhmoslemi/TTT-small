def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(np.float64), 1.0 - h, mode='full').astype(np.float64)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12: return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14: return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14: return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def slsqp_polish(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)
    except:
        return h

def make_coef(h, i, n):
    coef = np.zeros(2 * n - 1)
    coef[i:i+n] += (1.0 - h)[::-1]
    coef[n-1-i:2*n-1-i] -= h
    return coef

def coord_descent(h0, n, t_end, max_sweeps=5000):
    h = h0.copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    stagn = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end: break
        improved = False
        for i in range(n):
            h_old = h[i]
            lo, hi = -h_old, 1.0 - h_old
            if hi - lo < 1e-13: continue
            coef = np.zeros(2 * n - 1)
            coef[i:i+n] += (1.0 - h)[::-1]
            coef[n-1-i:2*n-1-i] -= h
            def f1d(s):
                return np.max(corr + s * coef)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 25})
            ds = res.x
            if abs(ds) > 1e-11:
                h[i] += ds
                corr += ds * coef
                if np.max(corr) * dx < best_c5:
                    best_c5 = np.max(corr) * dx
                    improved = True
        if not improved:
            stagn += 1
            if stagn >= 3: break
        else:
            stagn = 0
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    h = project_to_sum(h, n / 2.0)
    best_c5 = compute_c5(h)
    return h, best_c5
