```python
def find_min_of_max_lines(intercepts, slopes, n_lines):
    # Each line: f_k(t) = intercepts[k] + t * slopes[k], t in [0,1]
    # Find min of max_k f_k(t)
    
    # Sort by slope
    order = np.argsort(slopes)
    s = slopes[order]
    ic = intercepts[order]
    
    # Build upper hull
    # A line j is above line i at t if ic[j] + s[j]*t > ic[i] + s[i]*t
    # Intersection of lines i and j: t* = (ic[i] - ic[j]) / (s[j] - s[i])
    
    hull = [0]  # indices into sorted array
    for j in range(1, n_lines):
        # Check if j makes the last hull element obsolete
        while len(hull) >= 1:
            i = hull[-1]
            if s[j] == s[i]:
                # Same slope, keep the one with higher intercept
                if ic[j] > ic[i]:
                    hull.pop()
                else:
                    break
            else:
                # Intersection of i and j
                t_ij = (ic[i] - ic[j]) / (s[j] - s[i])
                if len(hull) >= 2:
                    # Intersection of hull[-2] and i
                    i2 = hull[-2]
                    t_i2i = (ic[i2] - ic[i]) / (s[i] - s[i2])
                    if t_ij <= t_i2i:
                        hull.pop()
                        continue
                # Check if i is below j everywhere in [0,1]
                if ic[i] <= ic[j] and s[i] <= s[j]:
                    hull.pop()
                    continue
                if ic[i] >= ic[j] and s[i] >= s[j]:
                    # i dominates j, skip j
                    break
                # Check intersection with previous hull element
                if len(hull) >= 2:
                    i2 = hull[-2]
                    t_i2i = (ic[i2] - ic[i]) / (s[i] - s[i2])
                    t_i2j = (ic[i2] - ic[j]) / (s[j] - s[i2])
                    if t_i2j <= t_i2i:
                        hull.pop()
                        continue
                break
        hull.append(j)
    
    # Now find the minimum of the upper envelope on [0,1]
    # The envelope is convex (since it's a max of lines)
    # Minimum is where the slope changes from negative to positive
    # or at an endpoint
    
    # Evaluate at vertices of the hull in [0,1]
    best_t = 0.0
    best_val = np.max(intercepts)  # f(0)
    
    for idx in range(len(hull) - 1):
        i1 = hull[idx]
        i2 = hull[idx + 1]
        if s[i2] == s[i1]:
            continue
        t_cross = (ic[i1] - ic[i2]) / (s[i2] - s[i1])
        if 0.0 <= t_cross <= 1.0:
            val = ic[i1] + t_cross * s[i1]
            if val < best_val:
                best_val = val
                best_t = t_cross
    
    # Also check t=1
    f1 = np.max(ic + slopes)  # wait, this is max of (ic[k] + 1*slopes[k]) = max of f_k(1)
    if f1 < best_val:
        best_val = f1
        best_t = 1.0
    
    return best_t
```

def project(h, n):
    target = n / 2.0
    vec = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    diff = target - vec.sum()
    if abs(diff) < 1e-14:
        return vec
    # Bisection on additive shift
    lo = min(-vec.min(), 0.0) - 0.01 if diff < 0 else 0.0
    hi = max(1.0 - vec.max(), 0.0) + 0.01 if diff > 0 else 0.0
    if diff > 0:

        else:

def fast_cd(h, n, t_end, max_sweeps=50000):
    best_h = h.copy()
    m_total = 2 * n - 1
    gr_val = 0.6180339887
    best_raw = np.max(np.correlate(best_h, 1.0 - best_h, mode='full'))
    no_improve = 0
    
    # Precompute p array
    p_arr = np.arange(-(n-1), n)  # shape (2n-1)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        current_max = np.max(corr)
        changed = False
        
        for i in range(n):
            slopes = np.zeros(m_total)
            
            # p >= 0: slope += (1-h[i-p]) if p <= i
            pos = p_arr >= 0
            m1 = pos & (p_arr <= i)
            if np.any(m1):
                slopes[m1] += 1.0 - best_h[i - p_arr[m1]]
            # p >= 0: slope -= h[i+p] if i+p <= n-1
            m2 = pos & (p_arr <= n - 1 - i)
            if np.any(m2):
                slopes[m2] -= best_h[i + p_arr[m2]]
            # p < 0: q = -p. slope += (1-h[i+q]) if i+q <= n-1
            neg = p_arr < 0
            q_arr = -p_arr
            m3 = neg & (q_arr <= n - 1 - i)
            if np.any(m3):
                slopes[m3] += 1.0 - best_h[i + q_arr[m3]]
            # p < 0: slope -= h[i-q] if q <= i
            m4 = neg & (q_arr <= i)
            if np.any(m4):
                slopes[m4] -= best_h[i - q_arr[m4]]
            
            intercepts = corr - slopes * best_h[i]
            
            # Golden section
            lo_t, hi_t = 0.0, 1.0
            c_t = hi_t - gr_val * (hi_t - lo_t)
