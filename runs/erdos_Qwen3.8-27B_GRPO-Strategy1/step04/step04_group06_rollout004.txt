import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-14:
        return np.full(n, 0.5)
    if s >= target:
        h = h * (target / s)
        return np.clip(h, 0.0, 1.0)
    room = 1.0 - h
    tot = room.sum()
    if tot < 1e-14:
        return h
    h = h + room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def fast_coord_opt(h, n, i):
    center = n - 1
    save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = save
    delta = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        vp = (1.0 - h[ip]) if ip < n else 0.0
        vm = h[im] if im >= 0 else 0.0
        delta[center + k] = vp - vm
        delta[center - k] = vm - vp
    c0 = corr_base[center]
    xs = np.linspace(0.0, 1.0, 256)
    ss = corr_base[None, :] + xs[:, None] * delta[None, :]
    ss[:, center] = c0 + xs - xs * xs
    fs = ss.max(axis=1)
    ib = np.argmin(fs)
    lo = xs[max(0, ib - 5)]
    hi = xs[min(255, ib + 5)]
    xs2 = np.linspace(lo, hi, 64)
    ss2 = corr_base[None, :] + xs2[:, None] * delta[None, :]
    ss2[:, center] = c0 + xs2 - xs2 * xs2
    fs2 = ss2.max(axis=1)
    ib2 = np.argmin(fs2)
    lo2 = xs2[max(0, ib2 - 2)]
    hi2 = xs2[min(63, ib2 + 2)]
    return 0.5 * (lo2 + hi2)

    rng = np.random.RandomState(seed)
    h_work = h.copy()
    for i in rng.permutation(n):
        h_work[i] = fast_coord_opt(h_work, n, i)
    return h_work

def coord_descent(h, n, t_end, max_sweeps=300, seed=42, min_gain=1e-14):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        h_new = coord_sweep(best_h, n, rng.randint(2**31))
        h_new = project(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - min_gain:
            best_h = h_new
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 4:
            break
    return best_h, best_c5

def gen_random_h(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0, 1, n)
    return project(h, n)

def gen_structured_h(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    kind = rng.randint(4)
    if kind == 0:
        k1, k2 = rng.randint(1, 5), rng.randint(5, 12)
        p1, p2 = rng.uniform(0, 2*np.pi), rng.uniform(0, 2*np.pi)
        a1, a2 = rng.uniform(0.1, 0.4), rng.uniform(0.05, 0.2)

    elif kind == 1:

        kernel = np.array([0.25, 0.5, 0.25])
        h = np.convolve(h, kernel, 'same')

            pos += int(s * 1.5)
            if pos >= n:

    else:

def smooth_max_opt(h, T, n, t_end):
    center = n - 1
    def obj_and_grad(x):
        cx = np.clip(x, 0, 1)
        corr = np.correlate(cx, 1.0 - cx, mode='full')
        m = corr.max()
        ec = np.exp(T * (corr - m))
        sm = m + np.log(ec.sum()) / T
        grad = -T * ec / ec.sum()
        dc = np.zeros(2 * n - 1)
        for k_idx in range(len(grad)):
            k = k_idx - center
            if k == 0:
                dc[k_idx] = 1.0 - 2.0 * cx
            elif k > 0:
                dc[k_idx, :n-k] += (1.0 - cx[k:]) if False else 0
        # Use full gradient via finite diff on smooth max
        eps = 1e-6
        grad_full = np.zeros(n)
        for i in range(n):
            xp = cx.copy()
            xp[i] += eps
            cp = np.correlate(xp, 1.0 - xp, mode='full')
            mp = cp.max()
            fp = (mp + np.log(np.exp(T*(cp-mp)).sum())/T)
            xm = cx.copy()
            xm[i] = max(xm[i] - eps, 0)
            cm = np.correlate(xm, 1.0 - xm, mode='full')
            mm = cm.max()
            fm = (mm + np.log(np.exp(T*(cm-mm)).sum())/T)
            grad_full[i] = (fp - fm) / (2 * eps)
        return sm, grad_full
    # Just use SLSQP with numeric gradient
    bnds = [(0.0, 1.0)] * n
    res = minimize(lambda x: obj_and_grad(x)[0], h, method='SLSQP',
                   bounds=bnds, options={'maxiter': 200, 'ftol': 1e-12})
    return project(np.clip(res.x, 0, 1), n)
