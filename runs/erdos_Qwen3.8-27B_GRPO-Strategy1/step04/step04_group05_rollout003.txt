```python
def smooth_max_objective(h, N, tau=0.01):
    c = compute_corr(h)
    # Softmax max
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    val = c_max + tau * np.log(exp_c.sum())
    return val * (2.0 / N)

def smooth_max_grad(h, N, tau=0.01):
    c = compute_corr(h)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    weights = exp_c / exp_c.sum()
    # Gradient of c(k) w.r.t. h
    grad = np.zeros(N)
    center = N - 1
    for k_idx in range(len(c)):
        L = k_idx - center
        if abs(weights[k_idx]) < 1e-10:
            continue
        w = weights[k_idx]
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:N-L])
            grad[:N-L] -= w * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += w * (1.0 - h[M:])
                grad[M:] -= w * h[:N-M]
    return grad * (2.0 / N)
```

def topk_avg_grad(h, N, K=50):

            if M < N:

    return val, grad

        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c_new = compute_corr(h)
            new_pk = float(c_new.max())
            delta = new_pk - cur
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
                cur, c = new_pk, c_new
                if new_pk < best_pk:
                    best_pk, best = new_pk, h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i], h[j] = 1.0, 0.0
    return best, best_pk * (2.0 / N)

def lbfgs_polish(h0, N, K, deadline, max_iter=3000):
    target = N / 2.0
    h = project_sum(h0, target)

    def obj(x):
        c = compute_corr(x)
        topk = np.sort(c)[-K:]
        pen = 500.0 * (x.sum() - target) ** 2
        return topk.mean() * (2.0 / N) + pen

    def grad(x):
        c = compute_corr(x)
        top_idx = np.argpartition(c, -K)[-K:]
        w = (1.0 / K) * (2.0 / N)
        g = np.zeros(N)
        center = N - 1
        for ki in top_idx:
            L = ki - center
            if L == 0:
                g += w * (1.0 - 2.0 * x)
            elif L > 0:
                g[L:] += w * (1.0 - x[:N - L])
                g[:N - L] -= w * x[L:]
            else:
                M = -L
                if M < N:
                    g[:N - M] += w * (1.0 - x[M:])
                    g[M:] -= w * x[:N - M]
        g += 1000.0 * (x.sum() - target)
        return g

    bounds = [(0.0, 1.0)] * N
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-12,
                            'factr': 1e10})
    h_out = project_sum(res.x, target)
    return h_out, compute_c5(h_out, N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    nh = N0 // 2

    h = (h0 > 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1

    best_h = h.copy()
    best_c5 = compute_c5(h, N0)

    # Phase 1: Steepest descent binary
    d1 = min(deadline - 250, t0 + 80)
    h = steepest_descent(h, N0, max_sweeps=400, deadline=d1)
    c5 = compute_c5(h, N0)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    # Phase 2: Iterated continuous + binary cycles
    d2 = deadline - 120
    h_cur = best_h.copy()
    n_cycles = 10
    for cyc in range(n_cycles):
        if time.time() > d2:
            break
        rem = d2 - time.time()
        # Nesterov continuous
        h_n, c5_n = nesterov_cont(h_cur + rng.randn(N0) * 0.04, N0,
                                  int(rem * 0.35), 0.15, 0.008, 0.03,
                                  d1=0)
        if time.time() < d2:
            h_n, c5_n = nesterov_cont(h_n, N0, int(min(6000, rem * 0.25)),
                                      0.08, 0.005, 0.02, d=0) if False else (h_n, c5_n)
        # Round and binary descent
        h_bin = (h_n > 0.5).astype(float)
        no = int(h_bin.sum())
        if no > nh:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[nh:]] = 0
        elif no < nh:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:nh - no]] = 1
        h_sd = steepest_descent(h_bin, N0, max_sweeps=150,
                                deadline=max(d2 - 5, time.time() + 10))
        c5_sd = compute_c5(h_sd, N0)
        if c5_sd < best_c5:
            best_c5, best_h = c5_sd, h_sd.copy()
        c5_cont = compute_c5(h_n, N0)
        if c5_cont < best_c5:
            best_c5, best_h = c5_cont, h_n.copy()
        h_cur = best_h.copy()

    # Phase 3: Mixed-anneal with 2-swaps + steepest descent
    d3 = deadline - 60
    for r in range(15):
        if time.time() > d3:
            break
        h_a, c5_a = anneal_mixed(best_h, N0, rng, int(8000), 3.0, 0.02,
                                  min(d3 - 2, time.time() + 25), use_2swap=True)
        h_sd = steepest_descent(h_a, N0, max_sweeps=200,
                                deadline=max(d3 - 2, time.time() + 10))
        c5_s = compute_c5(h_sd, N0)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_sd.copy()

    # Phase 4: L-BFGS-B polish
    if time.time() < deadline - 20:
        for K in [30, 50, 80]:
            if time.time() > deadline - 10:
                break
            h_p = best_h + rng.randn(len(best_h)) * 0.03
            h_p = project_sum(h_p, len(best_h) / 2.0)
            h_lb, c5_lb = lbfgs_polish(h_p, len(best_h), K,
                                       deadline=deadline - 8, max_iter=2000)
            if c5_lb < best_c5:
                best_c5, best_h = c5_lb, h_lb.copy()

    # Phase 5: Different N
    for dN in [-6, 6, -12, 12]:
        if time.time() > deadline - 20:
            break
        Nt = N0 + dN
        if Nt < 100 or Nt % 2 != 0:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = (hi > 0.5).astype(float)
        nth = Nt // 2
        no = int(hi.sum())
        if no > nth:
            idx = np.where(hi == 1)[0]
            hi[idx[nth:]] = 0
        elif no < nth:
            idx = np.where(hi == 0)[0]
            hi[idx[:nth - no]] = 1
        h_sd = steepest_descent(hi, Nt, max_sweeps=150, deadline=deadline - 10)
        c5_t = compute_c5(h_sd, Nt)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_sd.copy()
    # Final polish
    if time.time() < deadline - 3:
        hF = best_h + rng.randn(len(best_h)) * 0.015
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = lbfgs_polish(hF, len(best_h), 60, deadline=deadline - 1, max_iter=1500)
        if c5F < best_c5:
            best_h, best_c5 = hF, c5F

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
