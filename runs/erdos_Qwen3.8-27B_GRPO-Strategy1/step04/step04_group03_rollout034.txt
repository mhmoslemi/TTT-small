import time

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    """Project h to [0,1]^n with sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)

            return h

        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, n_sweeps=500, niters=40):
    best_h = h.copy().astype(float)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n)
    k_arr = np.arange(2 * n - 1)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        order = np.random.permutation(n) if sweep % 50 == 0 else range(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            u = i - k_arr + n - 1
            v = i + k_arr - n + 1
            c = np.zeros(2 * n - 1)
            mu = (u >= 0) & (u < n)
            mv = (v >= 0) & (v < n)
            c[mu] += 1.0 - best_h[u[mu]]
            c[mv] -= best_h[v[mv]]
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if hi - lo < 1e-15:
                continue
            a, b = lo, hi
            for _ in range(niters):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(corr + m1 * c) <= np.max(corr + m2 * c):
                    b = m2
                else:
                    a = m1
            delta = (a + b) * 0.5
            if abs(delta) < 1e-16:
                continue
            corr = corr + delta * c
            best_h[i] += delta
            improved += 1
        best_c5 = np.max(corr) * (2.0 / n)
        if improved == 0:
            break
    return best_h, best_c5

def basin_hopping(h, n, c5_cur, t_end, n_iters=50):
    work_h = h.copy()
    work_c5 = c5_cur
    best_h = h.copy()
    best_c5 = c5_cur
    half = n // 2
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(1.0 - it / float(n_iters), 0.03)
        r = np.random.rand()
        if r < 0.30:
            step = 0.05 + 0.30 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.50:
            h_p = work_h.copy()
            h_p[:half] += 0.15 * cool * np.random.randn(half)
            h_p[half:] = 1.0 - h_p[:half]
        elif r < 0.70:
            shift = np.random.randint(-10, 11)
            h_p = 0.5 * np.roll(work_h, shift) + 0.5 * work_h
        elif r < 0.85:
            mask = np.random.rand(n) < 0.2
            step = 0.05 + 0.15 * cool
            h_p = work_h.copy()
            h_p[mask] += step * np.random.randn(mask.sum())
        else:
            h_p = (work_h > 0.5).astype(float)
            h_p = project_to_sum(h_p, n / 2.0)
            h_p += 0.05 * cool * np.random.randn(n)
        h_p = project_to_sum(h_p, n / 2.0)
        t_local = time.time() + max(1.0, 25.0 * cool)
        hc, c5v = coord_descent(h_p, n, t_local, n_sweeps=50, niters=35)
        temp = 0.002 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
            work_h, work_c5 = hc, c5v
        if c5v < best_c5:
            best_c5, best_h = c5v, hc.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(224)
        best_h[:112] = 1.0
    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate descent (10%)
    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = coord_descent(best_h, n, t1, n_sweeps=800, niters=40)

    # Phase 2: Basin hopping (35%)
    t2 = t0 + budget_s * 0.45
    best_h, best_c5 = basin_hopping(best_h, n, best_c5, t2, n_iters=70)

    # Phase 3: Multi-resolution search (20%)
    for n_try in [128, 160, 200, 224, 256, 320, 400]:
        if n_try == n or time.time() > t0 + budget_s * 0.65:
            continue
        x_old = np.arange(len(best_h)) * (2.0 / len(best_h))
        x_new = np.arange(n_try) * (2.0 / n_try)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2.0)
        hc, c5v = coord_descent(h_i, n_try, time.time() + 25, n_sweeps=200, niters=40)
        if c5v < best_c5:
            best_c5, best_h, n = c5v, hc, n_try

    # Phase 4: Extended basin hopping (20%)
    t4 = t0 + budget_s * 0.85
    best_h, best_c5 = basin_hopping(best_h, n, best_c5, t4, n_iters=100)

    # Phase 5: Final coordinate descent polish (15%)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, n_sweeps=2000, niters=40)

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
