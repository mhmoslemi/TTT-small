def subgradient_step(h, n, step_size):
    """Move h in the direction that reduces the max correlation."""
    corr = get_corr(h)
    max_val = corr.max()
    # Active set: lags within 1e-6 of max
    active = corr > max_val - 1e-4
    
    # Compute gradient of max via subgradient (average of active gradients)
    grad = np.zeros(n)
    for idx in np.where(active)[0]:
        k = idx - (n - 1)  # lag
        if k == 0:
            # corr_0[i] = h[i]*(1-h[i]), grad = 1-2h[i]
            grad += (1 - 2*h)
        elif k > 0:
            # corr_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # grad[i] = (1-h[i+k]) for i=0..n-1-k
            # grad[i] = -h[i-k] for i=k..n-1
            for i in range(n - k):
                grad[i] += (1 - h[i+k])
            for i in range(k, n):
                grad[i] += (-h[i-k])
        else:
            k_pos = -k
            for i in range(k_pos, n):
                grad[i] += (1 - h[i-k_pos])
            for i in range(0, n - k_pos):
                grad[i] += (-h[i+k_pos])
    
    grad /= active.sum()
    
    # Projected gradient descent
    h_new = h - step_size * grad
    h_new = project(h_new, n)
    return h_new

    """Gradient of correlation at lag m (can be negative) w.r.t. h."""
    g = np.zeros(n)
    if m == 0:
        return 1 - 2*h
    am = abs(m)
    if m > 0:
        # grad[j] = (1-h[j-m]) for j>=m, minus h[j+m] for j<=n-1-m
        g[am:] += (1 - h[:n-am])
        g[:n-am] += (-h[am:])

        # with m negative, -m = am
        g[:n-am] += (1 - h[am:])
        g[am:] += (-h[:n-am])
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-12))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                room = 1.0 - h
                h = h + room * (diff2 / max(room.sum(), 1e-14))
            else:
                h = h * (target / max(s2, 1e-14))
                h = np.clip(h, 0.0, 1.0)
    return h

    h_save = h[i]
    h[i] = 0.0
    corr_base = get_corr(h).copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if 0 <= i - k < n:
            d[n - 1 + k] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            d[n - 1 + k] += (-h[i + k])
        if 0 <= i + k < n:
            d[n - 1 - k] += (1.0 - h[i + k])
        if 0 <= i - k < n:
            d[n - 1 - k] += (-h[i - k])
    c0 = corr_base[n - 1]

        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-15})
    return res.x

        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0

                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    use_priority = False
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if use_priority:
            sens = compute_sensitivity(best_h, n)
            order = np.argsort(-sens)
        else:
            order = rng.permutation(n)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fixed(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-13:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
            if sweep % 8 == 0:
                use_priority = not use_priority
        else:
            stall += 1
            if stall >= 5:
                break
    return best_h, best_c5

def partial_slsqp(h, n, rng):
    """Optimize only free variables (mid-range) with L-BFGS-B."""
    free_mask = (h > 0.1) & (h < 0.9)
    free_idx = np.where(free_mask)[0]
    if len(free_idx) < 5 or len(free_idx) > 80:
        return h
    if len(free_idx) > 40:
        free_idx = rng.choice(free_idx, 40, replace=False)
    before = compute_c5(h)
    def obj(x):
        ht = h.copy()
        ht[free_idx] = x
        return compute_c5(ht)
    x0 = h[free_idx].copy()
    bnds = [(0.0, 1.0)] * len(free_idx)
    try:
        res = minimize(obj, x0, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': 800, 'ftol': 1e-15, 'gtol': 1e-14})
        if res.fun < before - 1e-12:
            h[free_idx] = np.clip(res.x, 0, 1)
            h = project(h, n)
    except Exception:
        pass
    return h

def perturbation_polish(h, n, t_end, rng, n_perturb=120, base_step=0.025):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for p in range(n_perturb):
        if time.time() > t_end:
            break
        s = base_step * (0.4 + 1.2 * rng.random())
        if p % 4 == 3:
            s *= 2.0
        pert = best_h + s * rng.standard_normal(n)
        pert = project(pert, n)
        hc = pert.copy()
        n_inner = 6 if s < 0.03 else 4
        for _ in range(n_inner):
            if time.time() > t_end:
                break
            for i in rng.permutation(n):
                if time.time() > t_end:
                    break
                hi = coord_opt_fixed(hc, n, i)
                if abs(hi - hc[i]) > 1e-12:
                    hc[i] = hi
                    hc = project(hc, n)
        cv = compute_c5(hc)
        if cv < best_c5 - 1e-12:
            best_h, best_c5 = hc, cv
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP initial polish
    for T in [30, 200, 1500]:
        if elapsed() > budget_s * 0.04:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=600, ftol=1e-14)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 2: Main coordinate descent (priority-based, many sweeps)
    t2 = t0 + budget_s * 0.38
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=400, rng=rng)

    # Phase 3: Partial SLSQP on free variables
    if elapsed() < budget_s * 0.45:
        for _ in range(5):
            if elapsed() > budget_s * 0.45:
                break
            best_h = partial_slsqp(best_h, n, rng)
            best_c5 = compute_c5(best_h)
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.45, max_sweeps=30, rng=rng)

    # Phase 4: SLSQP with high T
    for T in [8000, 30000, 100000]:
        if elapsed() > budget_s * 0.55:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=1200, ftol=1e-15)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.56, max_sweeps=20, rng=rng)

    # Phase 5: Perturbation + quick polish (key exploration phase)
    t5 = t0 + budget_s * 0.78
    best_h, best_c5 = perturbation_polish(best_h, n, t5, rng, n_perturb=150, base_step=0.025)

    # Phase 6: More CD after perturbations
    t6 = t0 + budget_s * 0.85
    best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=200, rng=rng)

    # Phase 7: Multi-resolution
    for n_try in [160, 200, 280, 300, 320, 360]:
        if elapsed() > budget_s * 0.92:
            break
        h_interp = interpolate_h(best_h, n_try)
        hc, cv = coord_descent(h_interp, n_try, t0 + budget_s * 0.93, max_sweeps=40, rng=rng)
        if cv < best_c5 - 1e-6:
            best_h, best_c5, n = hc, cv, n_try

    # Phase 8: Final polish
    if elapsed() < budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=60, rng=rng)
    best_h = partial_slsqp(best_h, n, rng)
    best_c5 = compute_c5(best_h)
    if elapsed() < budget_s * 0.99:
        for T in [50000, 200000]:
            if elapsed() > budget_s * 0.995:
                break
            hc = slsqp_opt(best_h, T, n, maxiter=1500, ftol=1e-15)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
