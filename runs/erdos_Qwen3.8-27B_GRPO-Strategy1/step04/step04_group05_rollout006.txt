def corr(h, g):

def c5_val(h, N):

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -3.0, 3.0

        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid

def compute_D(h, N):
    idx = np.arange(2*N-1)
    d_lags = idx - (N-1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N-1)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N-1)], 0.0)
    return (1.0-h_pmd)*valid_pmd.astype(float) - h_pd*valid_pd.astype(float)

def grad_smooth(h, N, beta, top_k=80):
    c = corr(h, 1.0-h)
    cs = c - c.max()
    w = np.exp(beta*cs)
    sw = w.sum()
    if sw < 1e-30:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= sw
    center = N-1
    grad = np.zeros(N)
    idx = np.argsort(c)[-top_k:]
    w_top = w[idx]
    sw2 = w_top.sum()
    if sw2 < 1e-20:
        return grad
    w_top /= sw2
    for ai, wi in zip(idx, w_top):
        L = int(ai) - center
        if L == 0:
            grad += wi*(1.0-2.0*h)
        elif L > 0:
            nv = N-L
            grad[L:] += wi*(1.0-h[:nv])
            grad[:nv] -= wi*h[L:]
        else:
            M = -L
            if M < N:
                nv = N-M
                grad[:nv] += wi*(1.0-h[M:])
                grad[M:] -= wi*h[:nv]
    return grad

def adam_smooth(h, N, target, deadline, beta=30.0, lr=0.025, top_k=70):
    best_h = h.copy()
    best_peak = float(np.max(corr(h, 1.0-h)))
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    it = 0
    cur_lr = lr
    while time.time() < deadline:
        g = grad_smooth(h, N, beta, top_k)
        m = b1*m + (1-b1)*g
        v = b2*v + (1-b2)*g*g
        it += 1
        mh = m/(1-b1**it)
        vh = v/(1-b2**it)
        h = h - cur_lr*mh/(np.sqrt(vh)+eps)
        h = project(h, target)
        peak = float(np.max(corr(h, 1.0-h)))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        if it % 150 == 0:
            cur_lr *= 0.985
    return best_h, best_peak*(2.0/N)

    if n1 > nh:
        ones = np.where(h==1)[0]
        order = np.argsort(h[ones])[::-1]
        h[ones[order[nh-1:]]] = 0
    elif n1 < nh:
        zeros = np.where(h==0)[0]
        h[zeros[:nh-n1]] = 1
    return h

def perturb_binary(h, N, n_flips, rng):
    h = h.copy()
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    n = min(n_flips, len(ones), len(zeros))
    ps = rng.choice(ones, size=n, replace=False)
    qs = rng.choice(zeros, size=n, replace=False)
    h[ps] = 0.0
    h[qs] = 1.0
    return h

def make_binary(n, pattern, rng):
    h = np.zeros(n)
    if pattern == 0:
        h[::2] = 1.0
    elif pattern == 1:
        h[:n//2] = 1.0
    else:
        h = (rng.random(n) > 0.5).astype(float)
    nh = n//2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h==1)[0]
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h==0)[0]
        h[idx[:nh-n1]] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = c5_val(best_h, N0)
    N = N0
    target = N / 2.0

    # Phase 1: Binary local search on initial
    h_bin = (h0 > 0.5).astype(float)
    h_s = local_search(h_bin, N, max_sweeps=150, deadline=deadline-500)
    c5_s = c5_val(h_s, N)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Smooth Adam with multiple betas
    for beta in [50.0, 30.0, 15.0, 8.0]:
        if time.time() > deadline - 250:
            break
        h_cur = best_h.astype(float) + rng.randn(N)*0.06
        h_cur = project(h_cur, target)
        dl = min(time.time()+70, deadline-180)
        h_ad, c5_ad = adam_smooth(h_cur, N, target, dl, beta=beta, lr=0.02, top_k=60)
        if c5_ad < best_c5:
            best_c5, best_h = c5_ad, h_ad.copy()
        h_b = smart_round(h_ad, N)
        h_s = local_search(h_b, N, max_sweeps=120, deadline=deadline-175)
        c5_s = c5_val(h_s, N)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

    # Phase 3: Iterated perturb + local search
    d3 = deadline - 150
    h_cur = best_h.copy()
    for it in range(25):
        if time.time() > d3:
            break
        nflips = rng.randint(2, 8)
        h_p = perturb_binary(h_cur, N, nflips, rng)
        h_s = local_search(h_p, N, max_sweeps=200, deadline=d3-10)
        c5_s = c5_val(h_s, N)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
            h_cur = h_s.copy()

    # Phase 4: Simulated annealing + local search
    d4 = deadline - 80
    for restart in range(8):
        if time.time() > d4:
            break
        rem = d4 - time.time()
        n_ann = int(max(30, min(60, rem*0.25))*100)
        h_a, c5_a = anneal_binary(h_cur, N, n_ann, rng, T0=2.0, T1=0.001, deadline=d4-15)
        h_s = local_search(h_a, N, max_sweeps=200, deadline=d4-10)
        c5_s = c5_val(h_s, N)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
            h_cur = h_s.copy()

    # Phase 5: Different N values
    for N_try in [N0-10, N0+10, N0+20, N0-20]:
        if time.time() > deadline - 40 or N_try < 100:
            break
        if N_try % 2 != 0:
            N_try -= 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = (hi > 0.5).astype(float)
        nh = N_try//2
        n1 = int(hi.sum())
        if n1 > nh:
            idx = np.where(hi==1)[0]
            hi[idx[nh:]] = 0
        elif n1 < nh:
            idx = np.where(hi==0)[0]
            hi[idx[:nh-n1]] = 1
        h_s = local_search(hi, N_try, max_sweeps=150, deadline=deadline-25)
        c5_s = c5_val(h_s, N_try)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        if time.time() < deadline - 15:
            h_pol = h_s.astype(float) + rng.randn(N_try)*0.04
            h_pol = project(h_pol, N_try/2.0)
            h_ad, c5_ad = adam_smooth(h_pol, N_try, N_try/2.0, deadline-10, beta=40, lr=0.015, top_k=50)
            h_rb = smart_round(h_ad, N_try)
            c5_rb = c5_val(h_rb, N_try)
            if c5_rb < best_c5:
                best_c5, best_h = c5_rb, h_rb.copy()

    # Phase 6: Final continuous polish
    if time.time() < deadline - 8:
        hF = best_h.astype(float) + rng.randn(len(best_h))*0.03
        hF = project(hF, len(best_h)/2.0)
        hF, c5F = adam_smooth(hF, len(best_h), len(best_h)/2.0, deadline-3, beta=60, lr=0.008, top_k=50)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = c5_val(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
