    n = len(h)

def project_binary(h, n):
    h = h.astype(float).copy()
    target = n // 2
    n_ones = int(np.sum(h >= 0.5))
    if n_ones > target:
        ones = np.where(h >= 0.5)[0]
        order = np.argsort(h[ones])[:n_ones - target]
        h[ones[order]] = 0.0
    elif n_ones < target:
        zeros = np.where(h < 0.5)[0]
        order = np.argsort(-h[zeros])[:target - n_ones]
        h[zeros[order]] = 1.0
    return h

def project_cont(h, n):
    target = n / 2.0
    vec = np.clip(h, 0.0, 1.0)
    if abs(vec.sum() - target) < 1e-12:

    lo, hi = -2.0, 2.0

            hi = mid
        else:
            lo = mid

        order = np.argsort(-h[zeros])[:target - n_ones]
        h[zeros[order]] = 1.0
    return h

def project_cont(h, n):
    target = n / 2.0
    vec = np.clip(h, 0.0, 1.0)
    if abs(vec.sum() - target) < 1e-12:
        return vec
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def sa_binary(h, n, t_end, T0, T1, seed):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    dur = max(t_end - t_start, 0.01)
    half = n // 2
    ones_list = list(np.where(cur == 1)[0])
    zeros_list = list(np.where(cur == 0)[0])
    stale = 0.0
    last_improve = time.time()

    while time.time() < t_end:
        frac = min(1.0, (time.time() - t_start) / dur)
        T = T0 * (T1 / T0) ** frac
        # Reheat if stuck
        if time.time() - last_improve > 5.0:
            T = max(T, 0.1)

        mt = rng.rand()
        if mt < 0.55:
            i = ones_list[rng.randint(half)]
            j = zeros_list[rng.randint(half)]
            cur[i], cur[j] = 0.0, 1.0
            nc = compute_c5(cur)
            if nc < cur_c5:
                cur_c5 = nc
                ones_list.remove(i); ones_list.append(j)
                zeros_list.remove(j); zeros_list.append(i)
                if nc < best_c5:
                    best_c5 = nc; best = cur.copy(); last_improve = time.time()
            elif rng.rand() < np.exp(-(nc - cur_c5) / max(T, 1e-9)):
                cur_c5 = nc
                ones_list.remove(i); ones_list.append(j)
                zeros_list.remove(j); zeros_list.append(i)
            else:
                cur[i], cur[j] = 1.0, 0.0
        elif mt < 0.8:
            a = rng.randint(0, n - 2)
            b = rng.randint(a + 1, n)
            cur[a:b+1] = cur[b:a-1:-1].copy()
            nc = compute_c5(cur)
            if nc < cur_c5 or rng.rand() < np.exp(-(nc - cur_c5) / max(T, 1e-9)):
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc; best = cur.copy(); last_improve = time.time()
                ones_list = list(np.where(cur == 1)[0])
                zeros_list = list(np.where(cur == 0)[0])
            else:
                cur[a:b+1] = cur[b:a-1:-1].copy()
        else:
            k = rng.randint(4, min(10, n))
            idx = rng.choice(n, k, replace=False)
            saved = cur[idx].copy()
            rng.shuffle(saved)
            cur[idx] = saved
            nc = compute_c5(cur)
            if nc < cur_c5 or rng.rand() < np.exp(-(nc - cur_c5) / max(T, 1e-9)):
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc; best = cur.copy(); last_improve = time.time()
                ones_list = list(np.where(cur == 1)[0])
                zeros_list = list(np.where(cur == 0)[0])
            else:
                cur[idx] = saved
    return best, best_c5

def greedy_local(h, n, t_end):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    while time.time() < t_end:
        improved = False
        for i in ones:
            if time.time() > t_end:
                break
            for j in zeros:
                cur[i], cur[j] = 0.0, 1.0
                nc = compute_c5(cur)
                if nc < cur_c5 - 1e-15:
                    cur_c5 = nc
                    improved = True
                    ones = np.where(cur == 1)[0]
                    zeros = np.where(cur == 0)[0]
                else:
                    cur[i], cur[j] = 1.0, 0.0
        if not improved:
            break
    return cur, cur_c5

def cd_polish(h, n, t_end, seed=0):
    rng = np.random.RandomState(seed)
    cur = h.astype(float).copy()
    cur_c5 = compute_c5(cur)
    deltas = np.array([0.08, -0.08, 0.04, -0.04, 0.02, -0.02, 0.15, -0.15])
    for _ in range(50000):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            cands = []
            for d in deltas:
                v = cur[i] + d
                if 0.0 <= v <= 1.0:
                    cands.append(v)
            if not cands:
                continue
            best_v = cur[i]
            best_nc = cur_c5
            for v in cands:
                trial = cur.copy()
                trial[i] = v
                trial = project_cont(trial, n)
                nc = compute_c5(trial)
                if nc < best_nc:
                    best_nc = nc
                    best_v = v
            if best_nc < cur_c5 - 1e-14:
                cur[i] = best_v
                cur = project_cont(cur, n)
                cur_c5 = best_nc
                improved = True
        if not improved:
            break
    return cur, cur_c5

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    h0 = None
    try:
        if initial_h_values is not None:
            h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    best_h = None
    best_c5 = float('inf')

    n_candidates = [160, 200, 240]
    if h0 is not None:
        n_candidates = [len(h0)] + [x for x in n_candidates if x != len(h0)]

    for ci, n in enumerate(n_candidates):
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        frac = 0.45 if ci == 0 else 0.3 if ci == 1 else 0.25
        nb = rem * frac
        t_p1 = time.time() + nb * 0.35
        t_p2 = time.time() + nb * 0.60
        t_p3 = time.time() + nb * 0.80
        t_p4 = time.time() + nb * 0.98

        starts = []
        if h0 is not None and len(h0) == n:
            starts.append(project_binary(h0, n))
        starts.append(random_binary(n, seed + n))
        starts.append(random_binary(n, seed + n + 7331))

        # Phase 1: SA exploration with multiple starts
        per_start = (t_p1 - time.time()) / max(1, len(starts))
        for si, sh in enumerate(starts):
            if time.time() >= t_p1:
                break
            t_end_sa = time.time() + per_start * (len(starts) - si)
            t_end_sa = min(t_end_sa, t_p1)
            h_b, c5_b = sa_binary(sh, n, t_end_sa, 0.35, 0.0003, seed + n * 10 + si)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

        h_ls = best_h.copy() if (best_h is not None and len(best_h) == n) else starts[0].copy()

        # Phase 2: Greedy local search
        if time.time() < t_p2:
            h_b, c5_b = greedy_local(h_ls, n, t_p2)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

        # Phase 3: SA refinement + greedy
        if best_h is not None and len(best_h) == n and time.time() < t_p3:
            t_sa3 = time.time() + (t_p3 - time.time()) * 0.45
            h_b, c5_b = sa_binary(best_h, n, t_sa3, 0.06, 0.0001, seed + 500)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
            if time.time() < t_p3:
                h_b, c5_b = greedy_local(best_h, n, t_p3)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()

        # Phase 4: Continuous polish
        if best_h is not None and len(best_h) == n and time.time() < t_p4:
            h_cont = best_h.copy()
            h_b, c5_b = cd_polish(h_cont, n, t_p4, seed + 2000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
            # Also try binary version after polish
            h_bin = project_binary(h_cont, n)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin

    if best_h is None:
        best_h = random_binary(160, seed)
        best_c5 = compute_c5(best_h)

    nf = len(best_h)
    best_h = project_binary(best_h, nf)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, nf
