from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        # Scale down proportionally
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def golden_1d(f, lo, hi, iters=18):
    """Minimize f on [lo, hi] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0

    center = n - 1

    h[i] = 0.0

    #   x * (1 - h[i+k]) [if 0<=i+k<n] 
    #   - x * h[i-k] [if 0<=i-k<n]
    #   At k=0: also -x² (the h[i](1-h[i]) term)

            new_val = coord_opt_fast(best_h, n, i)

But this has a problem: `coord_opt_fast` modifies `best_h[i]` in place (sets it to 0 then to the new value). And the projection at the end of the sweep might change things. Let me be more careful.

Actually, let me just accept that each `coord_opt_fast` call recomputes the correlation (O(n²) via FFT), and evaluate the 1D objective ~20 times (O(n) each). Total per coordinate: O(n² + 20n) ≈ O(n²). Per sweep: O(n³). For n=160: 160 × (160² + 20×160) ≈ 160 × 28800 ≈ 4.6M operations. At 1ns/op, that's ~5ms per sweep. 1000 sweeps: ~5s. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def golden_1d(f, lo, hi, iters=16):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(iters):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0

def coord_opt_fast(h, n, i):
    dx = 2.0 / n
    center = n - 1
    orig = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    a = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        coeff = 0.0
        ik = i + k
        im = i - k
        if 0 <= ik < n:
            coeff += 1.0 - h[ik]
        if 0 <= im < n:
            coeff -= h[im]
        a[center + k] = coeff
    def f1d(x):
        v = corr_base + x * a
        v[center] -= x * x
        return v.max() * dx
    best_x = golden_1d(f1d, 0.0, 1.0, iters=16)
    h[i] = best_x
    return best_x

def coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            new_val = coord_opt_fast(best_h, n, i)
            if abs(new_val - old) > 1e-10:
                improved = True
        best_h = project(best_h, n)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
        else:
            best_h = project(best_h, n)
        if not improved:
            break
    return best_h, best_c5

def smooth_obj_and_grad(params, T):
    h = params
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    exps = np.exp(T * (corr - m))
    s = exps.sum()
    w = exps / s
    obj = m + np.log(s) / T
    grad = np.zeros(n)
    for k in range(-n + 1, n):
        wk = w[n - 1 + k] * dx
        if wk < 1e-15:
            continue
        if k >= 0:
            if n - k > 0:
                grad[:n - k] += wk * (1.0 - h[k:])
                grad[k:] -= wk * h[:n - k]
        else:
            m2 = -k
            if n - m2 > 0:
                grad[m2:] += wk * (1.0 - h[:n - m2])
                grad[:n - m2] -= wk * h[m2:]
    return obj, grad

def lbfgs_opt(h, T, n, maxiter=300):
    x0 = h.copy()
    res = minimize(
        lambda x: smooth_obj_and_grad(x, T)[0],
        x0,
        jac=lambda x: smooth_obj_and_grad(x, T)[1],
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12}
    )
    return project(np.clip(res.x, 0, 1), n)

def pair_move(h, n, i, j):
    dx = 2.0 / n
    hi, hj = h[i], h[j]
    h[i] = 0.0
    h[j] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    ai = np.zeros(2 * n - 1)
    aj = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        ci = 0.0
        cj = 0.0
        for idx in [i, j]:
            coeff = 0.0
            ix = idx + k
            im = idx - k
            if 0 <= ix < n:
                coeff += 1.0 - h[ix]
            if 0 <= im < n:
                coeff -= h[im]
            if idx == i:
                ci = coeff
            else:
                cj = coeff
        ai[center + k] = ci
        aj[center + k] = cj
    def f2d(x, y):
        v = corr0 + x * ai + y * aj
        v[center] -= x * x + y * y
        return v.max() * dx
    # Alternating 1D optimization (4 iterations)
    xi, yj = hi, hj
    for _ in range(4):
        xi = golden_1d(lambda x: f2d(x, yj), 0.0, 1.0, iters=12)
        yj = golden_1d(lambda y: f2d(xi, y), 0.0, 1.0, iters=12)
    h[i] = xi
    h[j] = yj
    return h

def multi_start_explore(best_h, best_c5, n, t_end, n_starts=15):
    for s in range(n_starts):
        if time.time() > t_end:
            break
        noise_scale = 0.04 + 0.02 * s
        h_try = best_h + noise_scale * np.random.randn(n)
        h_try = project(h_try, n)
        t_sub = min(time.time() + 15.0, t_end - 10.0)
        h_try, c5_try = coord_descent(h_try, n, t_sub, max_sweeps=100)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
