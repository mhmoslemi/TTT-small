    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_pk = float(corr.max())
    best = h.copy()
    best_pk = cur_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        
        if use_2swap and rng.random() < 0.2 and len(ones) >= 2 and len(zeros) >= 2:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                continue
            n1, n2 = i1, i2
            z1, z2 = j1, j2
            h[n1], h[z1], h[n2], h[z2] = 0, 1, 0, 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            new_pk = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            delta = new_pk - cur_pk
            if delta < 0 or (T > 0.01 and rng.random() < np.exp(-delta * N / (4 * T))):
                cur_pk = new_pk
                if new_pk < best_pk:
                    best_pk = new_pk
                    best = h.copy()
            else:
                h[n1], h[z1], h[n2], h[z2] = 1, 0, 1, 0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0, 1
            new_pk = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            delta = new_pk - cur_pk
            if delta < 0 or (T > 0.01 and rng.random() < np.exp(-delta * N / (4 * T))):

                if new_pk < best_pk:

                    best = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i], h[j] = 1, 0
    
    return best, best_pk * (2.0 / N)
