import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    vc = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    diff = target - vc.sum()
    if abs(diff) < 1e-13:
        return vc
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(vc + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vc + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_vals(h, T):
    """Returns (f_value, lam) for smooth max."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ex = np.exp(T * (corr - m))
    s = ex.sum()
    lam = ex / s
    f = (m + np.log(s) / T) * (2.0 / n)
    return f, lam

def smooth_grad_full(h, T):
    """Returns (f, grad) for smooth_c5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ex = np.exp(T * (corr - m))
    s = ex.sum()
    lam = ex / s
    f = (m + np.log(s) / T) * dx
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        l = lam[mi]
        if l < 1e-15:
            continue
        k = mi - (n - 1)
        if k == 0:
            grad += l * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += l * (1.0 - h[k:])
            grad[k:] -= l * h[:n-k]
        else:
            kk = -k
            grad[kk:] += l * (1.0 - h[:n-kk])
            grad[:n-kk] -= l * h[kk:]
    return f, dx * grad

    """Momentum gradient descent on smooth max."""
    x = h.copy()
    v = np.zeros(n)
    best_h = x.copy()
    best_c5 = compute_c5(best_h)
    gamma = 0.9

                best_h = x.copy()
        if it > 100:

    return best_h, best_c5
