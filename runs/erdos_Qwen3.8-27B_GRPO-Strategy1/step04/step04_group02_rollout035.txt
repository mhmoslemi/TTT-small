from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            s2 = h.sum()
            if s2 < 1e-14:
                break
            h *= (target / s2)
            np.clip(h, 0, 1, out=h)
    return h

def fast_coord_descend(h, n, t_end, max_sweeps=300, rng=None, stall_limit=6):
    if rng is None:
        rng = np.random.default_rng(42)
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            v = h[i]
            if v < 1e-12:
                v = 0.0
            elif v > 1 - 1e-12:
                v = 1.0
            h[i] = v
            # Compute delta vector
            d = np.zeros(2 * n - 1)
            for k in range(1, n):
                # corr[n-1+k]: Δ = (t-v)*d_k
                dk = 0.0
                if k <= i and i + k < n:
                    dk = (1.0 - h[i - k]) - h[i + k]
                elif k <= i:
                    dk = 1.0 - h[i - k]
                elif i + k < n:
                    dk = -h[i + k]
                d[n - 1 + k] = dk
                # corr[n-1-k]: Δ = (t-v)*dk2
                dk2 = 0.0
                if k <= i:
                    dk2 += 1.0 - h[i - k]
                if i + k < n:
                    dk2 -= h[i + k]
                d[n - 1 - k] = dk2
            
            c0 = corr[n - 1]
            # 1D optimization: minimize max(corr + (t-v)*d) with corr[n-1] = c0 - v(1-v) + t(1-t)
            def g(t):
                vals = corr + (t - v) * d
                vals[n - 1] = c0 - v * (1.0 - v) + t * (1.0 - t)
                return np.max(vals)
            res = minimize_scalar(g, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - v) < 1e-13:
                continue
            # Update corr
            delta = t_new - v
            corr += delta * d
            corr[n - 1] = c0 - v * (1.0 - v) + t_new * (1.0 - t_new)
            h[i] = t_new
            c5 = np.max(corr) * dx
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
        # End of sweep: project
        h = project(h, n)
        corr = compute_corr(h)
        c5 = np.max(corr) * dx
        if c5 <= best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h = best_h.copy()
            corr = compute_corr(h)
        if not improved:
            stall += 1
            if stall >= stall_limit:
                break
        else:
            stall = 0
    return best_h, best_c5

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': coneq}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)

    T_start = 0.1

        if h[i] == 1:

        if c5 < best_c5 or rng.random() < np.exp(-(c5 - best_c5) / T):

        else:
            h[i], h[j] = h[j], h[i]  # revert
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return project(best_h, n)

def init_structured(n):
    """Create a structured initial solution."""
    h = np.zeros(n)
    # Use a "spread out" pattern
    phi = (1 + 5**0.5) / 2
    positions = set()
    for i in range(n // 2):
        positions.add(int(i * phi) % n)
    for p in positions:
        h[p] = 1.0
    # If we don't have exactly n/2 ones, adjust
    if h.sum() < n / 2:
        zeros = np.where(h == 0)[0]
        for i in range(int(n / 2 - h.sum())):
            h[zeros[i]] = 1.0
    elif h.sum() > n / 2:
        ones = np.where(h == 1)[0]
        for i in range(int(h.sum() - n / 2)):
            h[ones[i]] = 0.0
    return project(h, n)

    return project(h_new, n_new)

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def _project(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + room * (diff / tot)
        else:
            if s < 1e-15:
                break
            h = h * (target / s)
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _g_eval(t, corr, d, v, c0, n):
    vals = corr + (t - v) * d
    vals[n - 1] = c0 - v * (1.0 - v) + t * (1.0 - t)
    return np.max(vals)

def _fast_coord(h, n, t_end, max_sweeps, rng, stall_lim=8):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    stall = 0
    ks = np.arange(1, n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            v = h[i]
            if v < 5e-13:
                v = 0.0
            elif v > 1.0 - 5e-13:
                v = 1.0
            # Compute d vector (symmetric)
            ib = i - ks
            vf_b = ib >= 0
            ifd = i + ks
            vf_f = ifd < n
            d_half = np.zeros(n - 1)
            if vf_b.any():
                d_half[vf_b] -= 0.0  # will add below
                d_half[vf_b] += (1.0 - h[ib[vf_b]])
            if vf_f.any():
                d_half[vf_f] -= h[ifd[vf_f]]
            d = np.zeros(2 * n - 1)
            d[n:] = d_half
            d[:n - 1] = d_half[::-1]
            c0 = corr[n - 1]
            res = minimize_scalar(_g_eval, bounds=(0.0, 1.0),
                                  args=(corr, d, v, c0, n),
                                  method='bounded', options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - v) < 1e-13:
                continue
            delta = t_new - v
            corr += delta * d
            corr[n - 1] = c0 - v * (1.0 - v) + t_new * (1.0 - t_new)
            h[i] = t_new
            c5 = np.max(corr) * dx
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
        h = _project(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        c5 = np.max(corr) * dx
        if c5 <= best_c5 + 1e-15:
            best_c5 = min(best_c5, c5)
            best_h = h.copy()
        else:
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        stall = 0 if improved else stall + 1
        if stall >= stall_lim:
            break
    return best_h, best_c5

def _smooth_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def _smooth_con(x):
    return x.sum() - len(x) / 2.0

def _slsqp(h, T, n, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    res = minimize(_smooth_obj, h, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': _smooth_con}],
                   args=(T,), options={'maxiter': maxiter, 'ftol': 1e-15})
    return _project(np.clip(res.x, 0, 1), n)

def _interp(h, n_new):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return _project(np.interp(xn, xo, h), n_new)

def _bin_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

                if cv
