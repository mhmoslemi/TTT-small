def compute_grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    p = np.exp(T * (corr - m))
    p /= p.sum()
    
    grad_h = np.zeros(n)
    for m_idx in range(2 * n - 1):
        pm = p[m_idx]
        if pm < 1e-15:
            continue
        k = abs(m_idx - (n - 1))
        ka = k  # absolute shift
        # d/dh[j] = [ka<=j<=n-1]*(1-h[j-ka]) - [0<=j<=n-1-ka]*h[j+ka]
        if ka == 0:

            grad_h += pm * (1.0 - 2.0 * h)

            grad_h[ka:] += pm * (1.0 - h[:n-ka])
            # Second term: j from 0 to n-1-ka
            grad_h[:n-ka] -= pm * h[ka:n]
    
    return grad_h * dx

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    p = np.exp(T * (corr - m))
    p /= p.sum()
    grad_h = np.zeros(n)
    for m_idx in range(2 * n - 1):
        pm = p[m_idx]
        if pm < 1e-15:
            continue
        ka = abs(m_idx - (n - 1))
        if ka == 0:
            grad_h += pm * (1.0 - 2.0 * h)
        else:
            grad_h[ka:] += pm * (1.0 - h[:n-ka])
            grad_h[:n-ka] -= pm * h[ka:n]
    return grad_h * dx

def project(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    def sum_at(s):
        return np.clip(h + s, 0.0, 1.0).sum()
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if sum_at(mid) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def momentum_descent(h, n, t_end, T_init=1000, T_end=50000, n_steps=None):
    """Momentum gradient descent on smooth C5."""
    h_cur = h.copy()
    v = np.zeros(n)
    beta = 0.9
    c5_cur = compute_c5(h_cur)
    
    if t_end is None:
        n_steps = int(n_steps)
    else:
        n_steps = None
    
    n_total = 5000 if n_steps is not None else 10**6
    for step in range(n_total):
        if t_end is not None and time.time() > t_end:
            break
        
        frac = step / max(n_total - 1, 1)
        T = T_init * (T_end / T_init) ** frac
        alpha = 0.5 * (1 - frac) + 0.01
        
        grad = smooth_c5_grad(h_cur, T)
        v = beta * v - alpha * grad
        h_new = project(h_cur + v, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
            v = 0.5 * v  # dampen on success
        else:
            h_cur = project(h_cur - 0.5 * v, n)
            c5_t = compute_c5(h_cur)
            if c5_t < c5_cur:
                c5_cur = c5_t
            v = -0.5 * v
    
    return h_cur, c5_cur

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Try optimizing h[i] in [0,1] while adjusting for sum constraint
            # Simple: try h[i]+delta for small deltas, or use scipy minimize_scalar
            # For speed, try a few candidate values
            old = best_h[i]
            best_local = old
            best_c5_local = best_c5
            # Try: current, 0, 1, and a few interpolation points
            candidates = [old]
            if old > 0.01:
                candidates += [max(0, old - 0.1), max(0, old - 0.01)]
            if old < 0.99:
                candidates += [min(1, old + 0.1), min(1, old + 0.01)]
            candidates += [0.0, 1.0, 0.5]
            for c in candidates:
                if abs(c - old) < 1e-10:
                    continue
                h_try = best_h.copy()
                h_try[i] = c
                h_try = project(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5_local - 1e-15:
                    best_c5_local = c5_try
                    best_local = c
            if best_local != old:
                h_try = best_h.copy()
                h_try[i] = best_local
                h_try = project(h_try, n)
                best_h = h_try
                best_c5 = best_c5_local
                improved = True
                no_improve = 0
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
    return best_h, best_c5

def slsqp_step(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5_val(x, T)
    def obj_grad(x):
        return smooth_c5_grad(x, T)
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < smooth_c5_val(h, T):
            return project(res.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:

        return time.time() - t0
    def remaining():

    # Phase 1: SLSQP polish (5%)

            best_h = hc

    best_h = project(h_mom, n)

    best_h, best_c5 = coord_descent(best_h, n, t_end=t_phase3, max_sweeps=2000)
    
    # Phase 4: Diversification (25%)
    t_phase4 = t0 + budget_s * 0.85
    work_h = best_h.copy()

            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:

                best_h = hc
    if work_c5 < best_c5:

        best_h = work_h

Let me try a different approach: instead of changing one coordinate at a time, swap values between two coordinates. This preserves the sum exactly!

**Swap-based local search**:
- Pick a random pair (i, j)
- Try h_new[i] = h[j], h_new[j] = h[i] (swap)
- Or try h_new[i] = h[i] + δ, h_new[j] = h[j] - δ (transfer)
- This preserves sum(h) and (mostly) bounds

This avoids projection entirely! And it's a different type of move than coordinate descent, so it might escape different local minima.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    p = np.exp(T * (corr - m))
    p /= p.sum()
    grad_h = np.zeros(n)
    for m_idx in range(2 * n - 1):
        pm = p[m_idx]
        if pm < 1e-15:
            continue
        ka = abs(m_idx - (n - 1))
        if ka == 0:
            grad_h += pm * (1.0 - 2.0 * h)
        else:
            grad_h[ka:] += pm * (1.0 - h[:n - ka])
            grad_h[:n - ka] -= pm * h[ka:n]
    return grad_h * dx

def project(h, n):
    target = n * 0.5
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -h.max() - 0.1
    hi = 1.0 - h.min() + 0.1
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if (np.clip(h + mid, 0, 1)).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def momentum_descent(h, n, t_end):
    h_cur = h.copy()
    v = np.zeros(n)
    beta = 0.85
    c5_cur = compute_c5(h_cur)
    step = 0
    while time.time() < t_end:
        step += 1
        frac = min(step / 8000.0, 1.0)
        T = 500.0 * (100.0 ** frac)
        alpha = 0.3 * (1.0 - frac) + 0.02
        grad = smooth_c5_grad(h_cur, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        v = beta * v - alpha * grad / gnorm
        h_try = project(h_cur + v, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur:
            h_cur = h_try
            c5_cur = c5_try
            v *= 0.8
        else:
            h_back = project(h_cur - 0.3 * v, n)
            c5_back = compute_c5(h_back)
            if c5_back < c5_cur:
                c5_cur = c5_back
                h_cur = h_back
            v = -0.3 * v
        if step % 200 == 0:
            h_cur = project(h_cur, n)
    return h_cur, c5_cur

def coord_descent_fast(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_imp = 0
    deltas = np.array([-0.2, -0.1, -0.05, -0.02, 0.02, 0.05, 0.1, 0.2])
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            best_d = 0.0
            best_c5_l = best_c5
            for d in deltas:
                nv = old + d
                if nv < -0.01 or nv > 1.01:
                    continue
                best_h[i] = nv
                best_h = project(best_h, n)
                c5_l = compute_c5(best_h)
                if c5_l < best_c5_l - 1e-14:
                    best_c5_l = c5_l
                    best_d = d
                else:
                    best_h[i] = old
            if best_d != 0.0:
                best_h = project(best_h, n)
                best_c5 = best_c5_l
                improved = True
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 4:
                break
    return best_h, best_c5

def swap_search(h, n, t_end, t0_sa=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    if t0_sa is None:
        t0_sa = time.time()
    it = 0
    while time.time() < t_end:
        it += 1
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        oi, oj = best_h[i], best_h[j]
        for d in (0.03, 0.07, 0.12, 0.2, -0.03, -0.07, -0.12, -0.2):
            ni, nj = oi + d, oj - d
            if ni < -0.001 or ni > 1.001 or nj < -0.001 or nj > 1.001:
                continue
            best_h[i], best_h[j] = ni, nj
            c5_t = compute_c5(best_h)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                break
            best_h[i], best_h[j] = oi, oj
    return best_h, best_c5

def slsqp_step(h, T, n, maxiter=400):
    def obj(x):
        return smooth_c5_val(x, T)
    def jobj(x):
        return smooth_c5_grad(x, T)
    def con(x):
        return x.sum() - n * 0.5
    try:
        res = minimize(obj, h, jac=jobj, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if hasattr(res, 'fun') and res.fun < smooth_c5_val(h, T):
            return project(res.x, n)
    except Exception:
        pass
    return h

    if 'initial_h_values' in globals() and initial_h_values is not None:

    work_h, work_c5 = best_h.copy(), best_c5
