    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        # Scale down positive entries, preserving zeros
        pos = h > 1e-12
        if pos.sum() == 0:
            return h
        h = h * (target / s)
    else:
        # Scale up, preserving ones
        neg = h < 1.0 - 1e-12
        if neg.sum() == 0:
            return h
        room = (1.0 - h)[neg].sum()
        if room < 1e-13:
            return h
        h = h + (target - s) * (1.0 - h) / (n - h.sum())
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    # Log-sum-exp for numerical stability
    shifted = T * (corr - m)
    max_shifted = shifted.max()
    val = m + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        h *= target / s
    else:
        room = (1.0 - h).sum()
        if room < 1e-12:
            return h
        h += (target - s) * (1.0 - h) / room
    return np.clip(h, 0.0, 1.0)

def smooth_max_h(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = T * (corr - m)

        if time.time() > t_end:
            break
        imp = False
        for _ in range(samples_per_sweep):
            if time.time() > t_end:
                return best, best_c5
            i, j = np.random.choice(n, 2, replace=False)
            hi, hj = best[i], best[j]
            t_lo = max(-hi, hj - 1.0)
            t_hi = min(1.0 - hi, hj)
            if t_hi - t_lo < 1e-12:
                continue
            best_t = 0.0
            best_v = best_c5
            fracs = [0.0, 0.25, 0.5, 0.75, 1.0]
            for f in fracs:
                t_try = t_lo + f * (t_hi - t_lo)
                h_try = best.copy()
                h_try[i] = hi + t_try
                h_try[j] = hj - t_try
                c5_try = compute_c5(h_try)
                if c5_try < best_v:
                    best_v = c5_try
                    best_t = t_try
            # Refine around best_t
            if best_t != 0.0:
                lo_r = max(t_lo, best_t - 0.05 * (t_hi - t_lo))
                hi_r = min(t_hi, best_t + 0.05 * (t_hi - t_lo))
                for _ in range(8):
                    t_mid = (lo_r + hi_r) / 2
                    h_a = best.copy(); h_a[i] = hi + lo_r + 0.01*(hi_r-lo_r); h_a[j] = hj - lo_r - 0.01*(hi_r-lo_r)
                    h_b = best.copy(); h_b[i] = hi + t_mid; h_b[j] = hj - t_mid
                    if compute_c5(h_b) < compute_c5(h_a):
                        lo_r = t_mid
                    else:
                        hi_r = t_mid
                h_try = best.copy()
                h_try[i] = hi + (lo_r + hi_r) / 2
                h_try[j] = hj - (lo_r + hi_r) / 2
                c5_try = compute_c5(h_try)
                if c5_try < best_v:
                    best_v = c5_try
                    best_t = (lo_r + hi_r) / 2
            if best_v < best_c5 - 1e-14:
                best[i] = hi + best_t
                best[j] = hj - best_t
                best_c5 = best_v
                imp = True
        if not imp:
            break
    return best, best_c5

def critical_repairs(h, n, t_end, max_rounds=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        top_lags = np.argsort(corr)[-4:]
        improved = False
        for lag in top_lags:
            d = lag - (n - 1)
            if d > 0:
                valid = np.arange(n - d)
                contrib = best[valid] * (1.0 - best[valid + d])
            elif d < 0:
                valid = np.arange(-d)
                contrib = best[valid - d] * (1.0 - best[valid])
            else:
                continue
            top_pos = np.argsort(contrib)[-6:]
            for p in top_pos:
                if d > 0:
                    j1, j2 = valid[p], valid[p] + d
                else:
                    j1, j2 = valid[p] - d, valid[p]
                for flip_j in [j1, j2]:
                    h_try = best.copy()
                    h_try[flip_j] = 1.0 - best[flip_j]
                    h_try = project_h(h_try)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best = h_try
                        improved = True
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP annealing
    for T in [50, 300, 1500, 8000, 40000, 150000]:
        if elapsed() > budget_s * 0.08:
            break
        hc = slsqp(best_h, T, 800)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-14:
            best_c5, best_h = c5v, hc

    # Phase 2: Pair coordinate descent
    best_h, best_c5 = pair_cd(best_h, n, t0 + budget_s * 0.18, 300, 250)

    # Phase 3: Critical lag repairs
    best_h, best_c5 = critical_repairs(best_h, n, t0 + budget_s * 0.25, 40)

    # Phase 4: Basin hopping (diverse)
    work, work_c5 = best_h.copy(), best_c5
    t4_end = t0 + budget_s * 0.55
    it = 0
    while elapsed() < t4_end:
        cool = max(0.0, 1.0 - it / 100.0)
        mod = it % 5
        if mod == 0:
            p = (0.02 + 0.08 * cool) * np.random.randn(n)
            hp = project_h(work + p)
        elif mod == 1:
            hp = work.copy()
            s = np.random.randint(0, n - 3)
            l = np.random.randint(3, 14)
            hp[s:s+l] = 1.0 - hp[s:s+l]
            hp = project_h(hp)
        elif mod == 2:
            hp = work.copy()
            a, b = np.random.choice(n, 2, replace=False)
            hp[a], hp[b] = hp[b], hp[a]
        elif mod == 3:
            f = np.random.randint(1, 6)
            ph = np.random.uniform(0, 2 * np.pi)
            p = (0.02 + 0.06 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
            hp = project_h(work + p)
        else:
            # Complement restart (exploits C5(h)=C5(1-h) symmetry)
            if it % 20 == 15:
                hp = project_h(1.0 - work + 0.01 * np.random.randn(n))
            else:
                hp = project_h(work + (0.03 + 0.05*cool) * np.random.randn(n))

        hc = slsqp(hp, 500 + 4000 * cool, 300)
        c5v = compute_c5(hc)
        temp = 5e-4 * cool + 1e-4
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        it += 1

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work
        n = len(best_h)

    # Phase 5: Pair CD + critical repairs
    best_h, best_c5 = pair_cd(best_h, n, t0 + budget_s * 0.68, 250, 200)
    best_h, best_c5 = critical_repairs(best_h, n, t0 + budget_s * 0.73, 30)

    # Phase 6: Fine basin hopping
    work, work_c5 = best_h.copy(), best_c5
    t6_end = t0 + budget_s * 0.85
    it6 = 0
    while elapsed() < t6_end:
        cool = max(0.0, 1.0 - it6 / 60.0)
        step = 0.01 + 0.03 * cool
        hp = project_h(work + step * np.random.randn(n))
        hc = slsqp(hp, 2000 + 5000 * cool, 250)
        c5v = compute_c5(hc)
        temp = 3e-4 * cool + 5e-5
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        it6 += 1

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work
        n = len(best_h)

    # Phase 7: Final pair CD
    best_h, best_c5 = pair_cd(best_h, n, t0 + budget_s * 0.93, 200, 150)

    # Phase 8: Ultra-high-T SLSQP polish
    for T in [200000, 800000]:
        if elapsed() > budget_s * 0.97:
            break
        hc = slsqp(best_h, T, 1200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Final micro CD
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = pair_cd(best_h, n, t0 + budget_s * 0.98, 50, 100)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
