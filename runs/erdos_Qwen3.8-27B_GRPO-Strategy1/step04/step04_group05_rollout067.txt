import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_smooth_val_grad(h, N, alpha):
    corr = compute_corr(h)
    cmax = corr.max()
    exp_c = np.exp(alpha * (corr - cmax))
    S = exp_c.sum()
    w = exp_c / S
    val = cmax + np.log(S) / alpha
    center = N - 1
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-7:
            continue
        L = k - center
        if L >= 0:
            nv = N - L
            if nv > 0:
                grad[:nv] += wk * (1.0 - h[L:L+nv])
                grad[L:] -= wk * h[:nv]
        else:
            M = -L
            nv = N - M
            if nv > 0:
                grad[M:] += wk * (1.0 - h[:nv])
                grad[:nv] -= wk * h[M:M+nv]
    return val, grad

def adam_continuous(h, N, target, deadline, alpha0=30.0, alpha1=300.0, lr=0.008):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    it = 0
    total_iters = max(1, int((deadline - time.time()) / 0.05))
    while time.time() < deadline:
        t_frac = min(1.0, it / total_iters)
        alpha = alpha0 + (alpha1 - alpha0) * t_frac ** 0.5
        _, grad = lse_smooth_val_grad(h, N, alpha)
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = project_sum(h_new, target)
        h = h_new
        if it % 100 == 0:
            pk = float(np.max(compute_corr(h)))
            if pk < best_peak:
                best_peak = pk
                best_h = h.copy()
    pk = float(np.max(compute_corr(h)))
    if pk < best_peak:
        best_peak = pk
        best_h = h.copy()
    return best_h, best_peak * (2.0 / N)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    d_lags = idx - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N - 1)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N - 1)], 0.0)
    A_part = np.where(valid_pd, 1.0 - h_pd, 0.0)
    B_part = np.where(valid_pmd, -h_pmd, 0.0)
    return A_part + B_part

def steepest_binary(h, N, max_sweeps=500, deadline=None):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    for _ in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        corr = compute_corr(h)
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        D = compute_D(h, N)
        best_peak = P
        best_pq = None
        for p in ones:
            base = corr - D[p, :]
            candidates = base[None, :] + D[zeros, :]
            new_peaks = candidates.max(axis=1)
            mi = int(np.argmin(new_peaks))
            if new_peaks[mi] < best_peak:
                best_peak = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def anneal_binary(h, N, n_iters, rng, deadline):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    best = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    cur_pk = best_pk
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    T0, T1 = 5.0, 0.01
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / n_iters)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_pk = float(np.max(compute_corr(h)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.02 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_pk * (2.0 / N)

    elif n_ones < nh:

    return h
