            best_c5 = c5v
            best_h = hc

    # Phase 2: Projected gradient (fast, many iters)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(300):
        if elapsed() > budget_s * 0.25:
            break
        frac = it / 300.0
        T = 200 + 3000 * (1 - frac)
        lr = 0.8 / (T * 0.02)
        h_new, _ = proj_grad_step(work_h, T, lr, n)
        c5v = compute_c5(h_new)
        if c5v < work_c5:
            work_h = h_new
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 3: Basin hopping round 1 (diverse)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(100):
        if elapsed() > budget_s * 0.42:
            break
        cool = 1.0 - 0.85 * (it / 100.0)
        mod = it % 5
        if mod == 0:
            pert = (0.04 + 0.14 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(2, 12)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.10 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.25
            pert = np.zeros(n)
            pert[mask] = (0.08 + 0.18 * cool) * np.random.randn(mask.sum())
        elif mod == 3:
            block = n // 4
            b = np.random.randint(0, 4)
            pert = np.zeros(n)
            pert[b*block:(b+1)*block] = (0.05 + 0.10*cool) * np.random.randn(block)
        else:
            pert = (0.02 + 0.06 * cool) * np.random.randn(n)
        h_p = project_h(work_h + pert)
        T_bh = 300 + 5000 * cool
        hc = slsqp_opt(h_p, T_bh, n, maxiter=800, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 4: Coordinate descent (thorough)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.62, max_sweeps=300)

    # Phase 5: Basin hopping round 2 (fine)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(50):
        if elapsed() > budget_s * 0.75:
            break
        cool = 1.0 - 0.7 * (it / 50.0)
        step = 0.015 + 0.05 * cool
        h_p = project_h(work_h + step * np.random.randn(n))
        T_bh = 500 + 6000 * cool
        hc = slsqp_opt(h_p, T_bh, n, maxiter=600, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 6: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=200)

    # Phase 7: Ultra-high T SLSQP polish
    for T in [500000, 2000000]:
        if elapsed() > budget_s * 0.95:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=3000, ftol=1e-18)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc.copy()

    # Phase 8: Final coordinate descent polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=80)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
