def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):

    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    def sum_at(s):
        return np.clip(vec + s, 0.0, 1.0).sum()

    hi = 1.0 - np.min(vec) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if sum_at(mid) > target:
            hi = mid
        else:
            lo = mid

        if np.any(pos):

        t_lo_b = 0.0

        else:
