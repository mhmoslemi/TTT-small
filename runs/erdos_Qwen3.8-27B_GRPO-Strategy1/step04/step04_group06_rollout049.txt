def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    return np.max(compute_corr(h)) * 2.0 / n

def project(h, n):

    target = n / 2.0

    if diff > 0:

        tot = room.sum()
        if tot < 1e-12:

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * 2.0 / n

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return np.clip(h * (target / s), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    center = n - 1
    h0 = h.copy()
    h0[i] = 0.0
    cb = compute_corr(h0)
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        a[center + k] = (1.0 - h[ip] if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
        a[center - k] = (1.0 - h[im] if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
    ct = cb[center]
    def f1d(x):
        vals = cb + x * a
        vals[center] = ct + x - x * x
        return np.max(vals)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def smooth_obj_grad(h, T):

        else:
            p = -m
            g[:n-p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n-p]
