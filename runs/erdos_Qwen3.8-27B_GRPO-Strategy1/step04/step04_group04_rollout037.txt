```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx
```

```python
def compute_grad(h, p, n):
    # Build index matrix: shape (n, n)
    # M1[j, i] = j + n - 1 - i
    # M2[j, i] = n - 1 - j + i
    j = np.arange(n)[:, None]
    i = np.arange(n)[None, :]
    M1 = j + n - 1 - i  # shape (n, n)
    M2 = n - 1 - j + i  # shape (n, n)
    # grad[j] = sum_i p[M1[j,i]]*(1-h[i]) - sum_i p[M2[j,i]]*h[i]
    P1 = p[M1]  # shape (n, n), P1[j,i] = p[j+n-1-i]
    P2 = p[M2]  # shape (n, n), P2[j,i] = p[n-1-j+i]
    g1 = P1 @ (1.0 - h)  # shape (n,)
    g2 = P2 @ h  # shape (n,)
    return g1 - g2
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.sum(np.exp(np.clip(T * (corr - m), -500, 500)))
    return (m + np.log(s) / T) * dx

def proj(h, n):
    h = np.clip(h.astype(float), 0, 1).copy()
    target = n / 2.0
    for _ in range(40):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            fr = h < 1 - 1e-13
            if fr.sum() == 0:
                break
            h[fr] += d / fr.sum()
        else:
            fr = h > 1e-13
            if fr.sum() == 0:
                break
            h[fr] += d / fr.sum()
        h = np.clip(h, 0, 1)
    return h

def compute_grad(h, p, n):
    j = np.arange(n)[:, None]
    i = np.arange(n)[None, :]
    M1 = j + n - 1 - i
    M2 = n - 1 - j + i
    g1 = p[M1] @ (1.0 - h)
    g2 = p[M2] @ h
    return g1 - g2

def slsqp_smooth(h, T, n, maxiter=400, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.sum(np.exp(np.clip(T * (c - m), -500, 500)))
        return (m + np.log(s) / T) * dx
    def con_f(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_f}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def grad_proj_descent(h, n, T, t_end, lr_init=0.005, momentum=0.85):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    lr = lr_init
    y = best_h.copy()
    v = np.zeros(n)
    stagnant = 0
    for it in range(50000):
        if time.time() > t_end:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        m = corr.max()
        p = np.exp(np.clip(T * (corr - m), -500, 500))
        p /= p.sum()
        g = compute_grad(y, p, n) * dx
        v = momentum * v + g
        y_new = y - lr * v
        y = proj(y_new, n)
        c5 = compute_c5(y)
        if c5 < best_c5 - 1e-16:
            best_c5 = c5
            best_h = y.copy()
            stagnant = 0
            T = min(T * 1.01, 50000)
        else:
            stagnant += 1
            if stagnant > 200:
                lr *= 0.7
                if lr < 1e-12:
                    break
                stagnant = 0
    return best_h, best_c5

def fast_coord_desc(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    ncorr = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        for j in range(n):
            a = best_h[j]
            idx1 = np.arange(j + n - 1, j - 1, -1)
            idx2 = np.arange(n - 1 - j, 2 * n - 1 - j)
            coeff1 = 1.0 - best_h  # (1-h[i]) for first part
            coeff2 = -best_h       # -h[i] for second part
            # coeff for this coordinate:
            # new_corr[k] = corr[k] + (t-a)*c[k] - (t-a)^2 * [k==n-1]
            c = np.zeros(ncorr)
            c[idx1] += coeff1
            c[idx2] += coeff2
            k_diag = n - 1
            def f1d(t):
                d = t - a
                nc = corr + d * c
                nc[k_diag] -= d * d
                return np.max(nc)
            best_t = a
            best_val = np.max(corr)
            # Try a few candidate points + minimize_scalar
            cands = [0.0, 1.0, a, 0.1, 0.3, 0.5, 0.7, 0.9]
            for tc in cands:
                if tc < 0 or tc > 1:
                    continue
                d = tc - a
                nc = corr + d * c
                nc[k_diag] -= d * d
                vm = np.max(nc)
                if vm < best_val - 1e-15:
                    best_val = vm
                    best_t = tc
            try:
                res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-11})
                if res.fun < best_val - 1e-14:
                    best_t = res.x
                    best_val = res.fun
            except Exception:
                pass
            if abs(best_t - a) > 1e-12:
                best_h[j] = best_t
                best_h = proj(best_h, n)
                c5_new = compute_c5(best_h)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_active(h, n, maxiter=1500):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    active = np.where(corr >= mx * 0.9999)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return np.max(c[active]) * dx
    def con_f(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_f}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP warm-start
    for T in [50, 200, 1000, 5000]:
        if elapsed() > budget_s * 0.06:
            break
        try:
            hc = slsqp_smooth(best_h, T, n, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Fast coordinate descent
    if elapsed() < budget_s * 0.20:
        best_h, best_c5 = fast_coord_desc(best_h, n, t0 + budget_s * 0.20, max_sweeps=200)

    # Phase 3: Gradient projection descent
    if elapsed() < budget_s * 0.35:
        best_h, best_c5 = grad_proj_descent(best_h, n, 500, t0 + budget_s * 0.35, lr_init=0.004)

    # Phase 4: More coordinate descent
    if elapsed() < budget_s * 0.45:
        best_h, best_c5 = fast_coord_desc(best_h, n, t0 + budget_s * 0.45, max_sweeps=200)

    # Phase 5: Basin hopping with structured perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 60
    for it in range(n_bh):
        if elapsed() > budget_s * 0.65:
            break
        cool = 1.0 - 0.85 * (it / float(n_bh))
        # Structured perturbation
        r = np.random.rand()
        if r < 0.3:
            # Swap-based perturbation
            h_p = work_h.copy()
            for _ in range(int(5 + 20 * cool)):
                i1, i2 = np.random.randint(0, n, 2)
                h_p[i1], h_p[i2] = h_p[i2], h_p[i1]
        elif r < 0.5:
            # Sinusoidal perturbation
            freq = np.random.randint(2, 10)
            phase = np.random.uniform(0, 2 * np.pi)
            amp = 0.05 + 0.1 * cool
            x = np.arange(n) * 2.0 / n
            h_p = work_h + amp * np.sin(2 * np.pi * freq * x + phase)
        elif r < 0.7:
            # Gaussian noise
            amp = 0.03 + 0.15 * cool

        else:
            # Block swap
            w = np.random.randint(3, 15)
            s1 = np.random.randint(0, n - w)
            s2 = np.random.randint(0, n - w)
            h_p = work_h.copy()
            h_p[s1:s1+w], h_p[s2:s2+w] = work_h[s2:s2+w], work_h[s1:s1+w]
        h_p = proj(h_p, n)
        T_bh = 500 + 5000 * (1 - cool)
        try:
            hc = slsqp_smooth(h_p, T_bh, n, maxiter=250)
            c5v = compute_c5(hc)
            temp = 0.002 * cool + 0.0001
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

            if elapsed() > budget_s * 0.90:
                break
            try:
                hc = slsqp_smooth(h_interp, T, n_try, maxiter=500)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    n = n_try
            except Exception:
                pass

    # Phase 8: Final polish

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
