    return np.correlate(h, 1.0 - h, mode='full')

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        h *= target / s
        h = np.clip(h, 0, 1)
    else:

        tot = room.sum()
        if tot > 1e-12:
            h += room * ((target - s) / tot)
    h = np.clip(h, 0, 1)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0, 1)
    return h

    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

Let me also think about whether there's a fundamentally better approach. 

One idea: the problem is equivalent to maximizing min_k R(k) where R(k) is the autocorrelation. This is related to "optimal autocorrelation" problems in signal processing. The optimal signal for flat autocorrelation is related to Zadoff-Chu sequences and other low-PSF sequences.

But our constraint is different: we have a step function with values in [0,1] and fixed mean 1/2. The "signal" is h - 1/2 (zero mean, values in [-1/2, 1/2]).

The autocorrelation of (h - 1/2) should be as flat as possible (all zero ideally, but that's impossible with our constraints).

Actually, let's define g = h - 1/2. Then:
R_h(k) = ∑ h[i]*h[i+k] = ∑(g[i]+1/2)(g[i+k]+1/2)
= ∑g[i]g[i+k] + (1/2)∑g[i] + (1/2)∑g[i+k] + (1/4)*count
= R_g(k) + 0 + 0 + (1/4)*count (since ∑g = 0)

where count is the number of valid pairs. For periodic correlation, count = n for all k.
So R_h(k) = R_g(k) + n/4.

And C₅ = max_k (n/2 - R_h(k)) * 2/n = max_k (n/2 - R_g(k) - n/4) * 2/n = max_k (n/4 - R_g(k)) * 2/n = 1/2 - 2*min_k R_g(k)/n

So minimizing C₅ is equivalent to maximizing min_k R_g(k), where g = h - 1/2, ∑g = 0, -1/2 ≤ g ≤ 1/2.

This is: maximize the minimum autocorrelation of a zero-mean, bounded-amplitude signal.

For a white noise signal, R_g(k) ≈ 0 for k≠0 and ∑g² > 0 for k=0. The minimum would be at the non-zero lags, close to 0.

But we want to MAXIMIZE the minimum, so we want all R_g(k) to be as large as possible. Since ∑g = 0, R_g(0) = ∑g² ≥ n*(mean of g²) = n*var(g). The other R_g(k) depend on the structure.

For a sinusoid g(i) = A*sin(2π*i/n), R_g(k) = A²*n/2 * cos(2π*k/n), which is negative for some k. Not ideal.

For g to have all positive autocorrelation, it needs to be very structured. import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-12:
            h = h + room * ((target - s) / tot)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def get_influence(h, n, i):
    a = np.zeros(2 * n - 1)
    c = n - 1
    for k in range(-(n - 1), n):
        hp = h[i + k] if 0 <= i + k < n else 0.0
        hm = h[i - k] if 0 <= i - k < n else 0.0
        a[c + k] = (1.0 - hp) - hm
    return a

class F1D:
    def __init__(self, corr, ci, a, center):
        self.corr = corr
        self.ci = ci
        self.a = a
        self.center = center
    def __call__(self, v):
        nc = self.corr + (v - self.ci) * self.a
        nc[self.center] = self.corr[self.center] + (v - self.ci) * (1.0 - v - self.ci)
        return np.max(nc)

def smooth_max_val(corr, T):
    m = corr.max()
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def obj_smooth(x, T, n):
    c = compute_corr(x)
    return smooth_max_val(c, T) * (2.0 / n)

def grad_smooth(x, T, n):
    c = compute_corr(x)
    m = c.max()
    w = T * np.exp(T * (c - m))
    w /= w.sum()
    g = np.zeros(len(x))
    center = len(x) - 1
    for ki in range(len(c)):
        if w[ki] < 1e-12:
            continue
        k = ki - center
        if k == 0:
            g += w[ki] * (1.0 - 2.0 * x)
        elif k > 0:
            g[:len(x)-k] += w[ki] * (1.0 - x[k:])
            g[k:] -= w[ki] * x[:len(x)-k]
        else:
            mk = -k
            g[mk:] += w[ki] * (1.0 - x[:len(x)-mk])
            g[:len(x)-mk] -= w[ki] * x[mk:]
    return g * (2.0 / n)

def con_eq(x):
    return x.sum() - len(x) / 2.0

def con_eq_jac(x):
    return np.ones(len(x)) / len(x)

class ObjSmooth:
    def __init__(self, T, n):
        self.T = T
        self.n = n
    def f(self, x):
        return obj_smooth(x, self.T, self.n)
    def g(self, x):
        return grad_smooth(x, self.T, self.n)

class ConEq:
    def f(self, x):
        return con_eq(x)
    def g(self, x):
        return con_eq_jac(x)

def cd_sweep(h, corr, n, center, t_end, max_iter=50000, random_order=False):
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    improved = 0
    it = 0
    while it < max_iter and time.time() < t_end:
        if it % n == 0:
            order = np.random.permutation(n) if random_order else np.arange(n)
        i = order[it % n]
        ci = h[i]
        a = get_influence(h, n, i)
        f1 = F1D(corr, ci, a, center)
        res = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
        v = res.x
        dv = v - ci
        if abs(dv) < 1e-11:
            it += 1
            continue
        h[i] = v
        corr += dv * a
        corr[center] -= dv * dv
        s = h.sum()
        if abs(s - n * 0.5) > 1e-9:
            h = project(h, n)
            corr = compute_corr(h)
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5 - 1e-13:
            best_c5 = new_c5
            improved += 1
        it += 1
        if it % n == 0 and improved == 0:
            break
        if it % n == 0:
            improved = 0
    return h, corr, best_c5

def slsqp_run(h, T, n, maxiter=500):
    os = ObjSmooth(T, n)
    ce = ConEq()
    cons = [{'type': 'eq', 'fun': ce.f, 'jac': ce.g}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(os.f, h, jac=os.g, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def fourier_perturb(h, n, sigma, t_end):
    x = np.linspace(0, 2, n, endpoint=False)
    nmodes = np.random.randint(2, min(9, n // 8 + 1))
    freqs = np.random.randint(1, n // 4, size=nmodes)
    phases = np.random.uniform(0, 2 * np.pi, size=nmodes)
    amps = np.random.uniform(0.005, sigma, size=nmodes)
    pert = np.zeros(n)
    for j in range(nmodes):
        pert += amps[j] * np.cos(2.0 * np.pi * freqs[j] * x + phases[j])
    return project(h + pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        x_init = np.linspace(0, 2, n_init, endpoint=False)
        best_h = np.clip(0.5 + 0.3 * np.cos(2 * np.pi * x_init) + 0.15 * np.cos(4 * np.pi * x_init), 0, 1)

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Main coordinate descent (30% time)
    cd1_end = t0 + budget_s * 0.30
    corr = compute_corr(best_h)
    center = n - 1
    best_h, corr, best_c5 = cd_sweep(best_h, corr, n, center, cd1_end, max_iter=200000, random_order=True)

    # Phase 2: SLSQP with smooth max (20% time)
    slsqp_end = t0 + budget_s * 0.50
    for T in [50, 200, 1000, 5000, 20000]:
        if time.time() > slsqp_end:
            break
        try:
            hc = slsqp_run(best_h, T, n, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                corr = compute_corr(best_h)
        except Exception:
            pass

    # Phase 3: More CD (15% time)
    cd2_end = t0 + budget_s * 0.65
    corr = compute_corr(best_h)
    best_h, corr, best_c5 = cd_sweep(best_h, corr, n, center, cd2_end, max_iter=200000, random_order=True)

    # Phase 4: Fourier perturbation exploration (20% time)
    pert_end = t0 + budget_s * 0.85
    for it in range(100):
        if time.time() > pert_end:
            break
        h_try = fourier_perturb(best_h, n, 0.06, pert_end)
        corr_try = compute_corr(h_try)
        h_try, corr_try, c5_try = cd_sweep(h_try, corr_try, n, center, min(time.time() + budget_s * 0.02, pert_end), max_iter=3000, random_order=False)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            corr = corr_try.copy()

    # Phase 5: Multi-resolution (10% time)
    res_end = t_end - 15
    for n_try in [192, 256, 200]:
        if time.time() > res_end:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project(h_interp, n_try)
        corr_d = compute_corr(h_interp)
        center_d = n_try - 1
        c5_d = np.max(corr_d) * (2.0 / n_try)
        sub_end = min(time.time() + budget_s * 0.03, res_end)
        h_interp, corr_d, c5_d = cd_sweep(h_interp, corr_d, n_try, center_d, sub_end, max_iter=10000, random_order=False)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_interp
            n = n_try
            center = center_d
            corr = corr_d

    # Phase 6: Final SLSQP + CD polish
    final_end = t_end - 5
    if time.time() < final_end - 30:
        try:
            hc = slsqp_run(best_h, 50000, n, maxiter=2000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final CD polish
    corr = compute_corr(best_h)
    center = n - 1
    best_h, corr, best_c5 = cd_sweep(best_h, corr, n, center, t_end - 2, max_iter=50000, random_order=False)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
