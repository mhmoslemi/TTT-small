    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

    for rnd in range(max_rounds):
        if t_limit is not None and time.time() > t_limit:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        n_trials = min(4000, len(ones) * len(zeros))

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def to_binary(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def local_search(h, n, max_rounds, n_trials, t_limit=None):
    h = h.copy()
    cur_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if t_limit is not None and time.time() > t_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_i, best_j, best_c5 = -1, -1, cur_c5
        for _ in range(n_trials):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_i, best_j = i, j
            h[i], h[j] = h[j], h[i]
        if best_i >= 0:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            cur_c5 = best_c5
        else:
            break
    return h, cur_c5

def perturb_multi(h, n, K):
    h_p = h.copy()
    ones = np.where(h_p == 1)[0]
    zeros = np.where(h_p == 0)[0]
    k_eff = min(int(K), len(ones), len(zeros))
    if k_eff < 1:
        return h_p
    si = np.random.choice(ones, k_eff, replace=False)
    sj = np.random.choice(zeros, k_eff, replace=False)
    h_p[si] = 0.0
    h_p[sj] = 1.0
    return h_p

def ilds(h, n, t_start, t_end, seed_offset=0):
    np.random.seed(10000 + seed_offset)
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    stale = 0
    it = 0
    total = max(1.0, t_end - t_start)
    while time.time() < t_end:
        frac = min(1.0, (time.time() - t_start) / total)
        # Adaptive perturbation size
        if frac < 0.3:
            K = np.random.choice([2, 3, 4, 5])
        elif frac < 0.7:
            K = np.random.choice([3, 5, 8, 10])
        else:
            K = np.random.choice([2, 3, 4, 5])
        h_p = perturb_multi(cur_h, n, K)
        # Local search from perturbed state
        h_p, c5_p = local_search(h_p, n, 30, 3000, time.time() + 2.5)
        # SA acceptance
        T = 0.03 * (0.001 / 0.03) ** frac
        if c5_p < cur_c5 - 1e-14:
            cur_h = h_p
            cur_c5 = c5_p
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                stale = 0
            else:
                stale += 1
        elif T > 1e-15 and np.random.rand() < np.exp(-(c5_p - cur_c5) / T):
            cur_h = h_p
            cur_c5 = c5_p
            stale += 1
        else:
            stale += 1
        # Restart if stuck too long
        if stale > 80:
            if np.random.rand() < 0.3:
                cur_h = best_h.copy()
                cur_c5 = best_c5
            else:
                cur_h = to_binary(np.random.rand(n), n)
                cur_h, cur_c5 = local_search(cur_h, n, 30, 3000, time.time() + 2.0)
            stale = 0
        it += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s * 0.985

    # Initialize from parent or default
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        n = 160
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0

    h = to_binary(h_init, n)

    # Phase 1: Initial local search
    h, c5 = local_search(h, n, 80, 4000, t0 + budget_s * 0.04)
    best_h = h.copy()
    best_c5 = c5

    # Phase 2: ILS main search
    t_ils_end = t0 + budget_s * 0.82
    h, c5 = ilds(h, n, t0 + budget_s * 0.04, t_ils_end, seed_offset=0)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Second ILS from best with different seed
    t2_end = min(t0 + budget_s * 0.90, time.time() + budget_s * 0.08)
    if time.time() < t2_end:
        h2, c5_2 = ilds(best_h, n, time.time(), t2_end, seed_offset=999)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 4: Final intensive local search
    t_ls_end = min(t_end, time.time() + budget_s * 0.08)
    best_h, best_c5 = local_search(best_h, n, 300, 5000, t_ls_end)

    # Verify and return
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, n
