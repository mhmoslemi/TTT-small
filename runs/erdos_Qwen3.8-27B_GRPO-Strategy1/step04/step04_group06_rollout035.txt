            if rs < 1e-14:

    """Compute how each correlation entry changes per unit change in h[i]"""
    n = len(h)
    a = np.zeros(2 * n - 1)
    c = n - 1
    x = h[i]
    a[c] = 1.0 - 2.0 * x  # linear part at lag 0 (quadratic handled separately)
    for k in range(1, n):
        ip = i + k
        im = i - k
        if ip < n:

        if im >= 0:

        if im >= 0:
            a[c - k] += 1.0 - h[im]
        if ip < n:
            a[c - k] -= h[ip]
    return a

    """One coordinate descent sweep"""
    if order is None:

    corr = np.correlate(h, 1.0 - h, mode='full')

from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):

    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            rs = r.sum()
            if rs < 1e-14:
                break
            h += r * (d / rs)
        else:
            h *= target / s
        h = np.clip(h, 0.0, 1.0)
    return h

def change_vec(h, i):
    n = len(h)
    a = np.zeros(2 * n - 1)
    c = n - 1
    x = h[i]
    a[c] = 1.0 - 2.0 * x
    for k in range(1, n):
        ip, im = i + k, i - k
        if ip < n:
            a[c + k] += 1.0 - h[ip]
        if im >= 0:
            a[c + k] -= h[im]
        if im >= 0:
            a[c - k] += 1.0 - h[im]
        if ip < n:
            a[c - k] -= h[ip]
    return a

def coord_sweep(h, n, order=None, max_eval=35):
    if order is None:
        order = np.random.permutation(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = n - 1
    for i in order:
        x = h[i]
        a = change_vec(h, i)
        base_x = x * (1.0 - x)
        def f(v):

            vals[c] = corr[c] + v * (1.0 - v) - base_x
            return vals.max()
        res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                              options={'maxiter': max_eval, 'xatol': 1e-10})
        h[i] = res.x
    return h

def grad_update(h, n, top_k=6, lr=0.03):

        else:
            mm = -k
            g[mm:] += 1.0 - h[:n-mm]
            g[:n-mm] -= h[mm:]
    g /= top_k
    return h - lr * g, m

        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)
