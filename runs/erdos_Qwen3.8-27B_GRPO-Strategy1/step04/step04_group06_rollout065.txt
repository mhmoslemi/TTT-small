    n = len(h)
    dx = 2.0 / n

    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + room * (diff / tot), 0.0, 1.0)
        else:
            h = np.clip(h * (target / s), 0.0, 1.0)
    return h

def _fv(x, cb, a, c):
    v = cb + x * a
    v[c] = cb[c] + x - x * x
    return np.max(v)

    h0 = h.copy()
    h0[i] = 0.0
    cb = np.correlate(h0, 1.0 - h0, mode='full')
    c = n - 1

        if stall % 30 == 0 and stall > 0 and time.time() < t_end:
            best, best_c5 = coord_descent(best, n, min(time.time() + 2, t_end), 30)
    return best, best_c5

def perturb_phase(h, n, t_end, temp0=0.035):
    best = h.copy()
    best_c5 = compute_c5(best)
    temp = temp0
    xg = np.arange(n, dtype=float)
    no_improve = 0
    while time.time() < t_end:
        mode = np.random.rand()
        if mode < 0.5:
            freq = np.random.choice([1, 2, 3, 4, 5, 6])
            phase = np.random.uniform(0, 2 * np.pi)
            amp = temp * np.random.uniform(0.4, 1.2)
            pert = best + amp * np.sin(2 * np.pi * freq * xg / n + phase)
            pert += temp * 0.3 * np.random.randn(n)
        elif mode < 0.8:
            bi = np.random.randint(n)
            bw = np.random.randint(2, max(3, n // 8))
            bump = np.exp(-0.5 * ((xg - bi) / bw) ** 2)
            sign = np.random.choice([-1.0, 1.0])
            pert = best + sign * temp * np.random.uniform(0.5, 1.5) * bump
        else:
            i1 = np.random.randint(n)
            i2 = np.random.randint(n)
            pert = best.copy()
            delta = temp * np.random.uniform(0.5, 1.5) * np.random.choice([-1.0, 1.0])
            pert[i1] += delta
            pert[i2] -= delta
        pert = project(pert, n)
        hc = pert.copy()
        for s in range(6):
            if time.time() > t_end:
                break
            hc, _ = _one_sweep(hc, n)
        hc = project(hc, n)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-13:
            best_c5 = c5v
            best = hc
            no_improve = 0
            temp = min(temp * 1.08, 0.2)
        else:
            no_improve += 1
            temp = max(temp * 0.96, 0.006)
        if no_improve > 60:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.25 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x), 0, 1)

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate descent (22%)
    p1 = t0 + budget_s * 0.22
    best_h, best_c5 = coord_descent(best_h, n, p1, max_sweeps=500)

    # Phase 2: Gradient-based exploration (15%)
    if time.time() < t0 + budget_s * 0.37:
        p2 = t0 + budget_s * 0.37
        best_h, best_c5 = gradient_phase(best_h, n, p2, T0=30, T1=3000, alpha0=0.04)

    # Phase 3: Structured perturbations + CD (33%)
    p3 = t0 + budget_s * 0.70
    best_h, best_c5 = perturb_phase(best_h, n, p3, temp0=0.035)

    # Phase 4: Extended coordinate descent (18%)
    p4 = t0 + budget_s * 0.88
    best_h, best_c5 = coord_descent(best_h, n, p4, max_sweeps=2000)

    # Phase 5: Final gradient polish (6%)
    if time.time() < t_end - 5:
        best_h, best_c5 = gradient_phase(best_h, n, t_end - 5, T0=500, T1=20000, alpha0=0.02)

    # Phase 6: Final CD polish
    best_h, best_c5 = coord_descent(best_h, n, t_end - 2, max_sweeps=500)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
