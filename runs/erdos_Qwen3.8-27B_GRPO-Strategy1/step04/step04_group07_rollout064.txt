```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
        h = np.clip(h, 0, 1)
        # Fix any overflow
        s2 = h.sum()
        if abs(s2 - target) > 1e-10:
            h = h / s2 * target
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def swap_delta(corr, h, i, j, n):
    """Compute delta to corr after swapping i(1)->0, j(0)->1."""
    d = np.arange(-(n-1), n)
    # Indices
    ip = i + d
    im = i - d
    jp = j + d
    jm = j - d
    
    hip = np.zeros(2*n-1)
    him = np.zeros(2*n-1)
    hjp = np.zeros(2*n-1)
    hjm = np.zeros(2*n-1)
    
    m1 = (ip >= 0) & (ip < n)
    m2 = (im >= 0) & (im < n)
    m3 = (jp >= 0) & (jp < n)
    m4 = (jm >= 0) & (jm < n)
    
    hip[m1] = h[ip[m1]]
    him[m2] = h[im[m2]]
    hjp[m3] = h[jp[m3]]
    hjm[m4] = h[jm[m4]]
    
    delta = (hip + him) - (hjp + hjm)
    delta[n-1] = 0.0  # d=0 has no change for binary swap
    return delta

def binary_local_search(h, n, t_end, max_sweeps=5000):
    """Greedy binary swap local search."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    ones = np.where(h == 1)[0]  # positions with h=1
    zeros = np.where(h == 0)[0]  # positions with h=0
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        # Try all swaps (or a random subset for speed)
        best_delta_c5 = 0
        best_i, best_j = -1, -1
        best_new_c5 = best_c5
        
        # For speed, only check a random subset if too many
        n_trial = min(len(ones) * len(zeros), 2000)
        if n_trial > 2000:
            # Random sample
            indices = np.random.randint(0, n_trial, 500)
            pairs = [(ones[indices[k] // len(zeros)], zeros[indices[k] % len(zeros)]) for k in range(500)]
        else:
            pairs = [(i, j) for i in ones for j in zeros]
        
        for i, j in pairs:
            if h[i] != 1 or h[j] != 0:
                continue
            delta = swap_delta(corr, h, i, j, n)
            new_max = np.max(corr + delta)
            if new_max * dx < best_new_c5 - 1e-15:
                best_new_c5 = new_max * dx
                best_i, best_j = i, j
                best_delta = delta
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr += best_delta
            best_c5 = best_new_c5
            improved = True
    
    return h, best_c5
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s > 1e-14:
            h = np.clip(h * (target / s), 0, 1)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def greedy_binary_ls(h, n, t_end, max_sweeps=50000):
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    d = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_val = best_c5
        best_pair = None
        best_delta = None
        
        for i in ones:
            ip = i + d
            im = i - d
            ci = np.zeros(2 * n - 1)
            m1 = (ip >= 0) & (ip < n)
            m2 = (im >= 0) & (im < n)
            ci[m1] += h[ip[m1]]
            ci[m2] += h[im[m2]]
            
            jp = zeros[:, None] + d[None, :]
            jm = zeros[:, None] - d[None, :]
            jp_c = np.clip(jp, 0, n - 1)
            jm_c = np.clip(jm, 0, n - 1)
            cj = (np.where((jp >= 0) & (jp < n), h[jp_c], 0.0) +
                  np.where((jm >= 0) & (jm < n), h[jm_c], 0.0))
            cj[:, n - 1] = 0.0
            ci[n - 1] = 0.0
            
            dm = ci[None, :] - cj
            nmax = np.max(corr[None, :] + dm, axis=1)
            k = np.argmin(nmax)
            if nmax[k] * dx < best_val - 1e-15:
                best_val = nmax[k] * dx
                best_pair = (i, zeros[k])
                best_delta = dm[k]
        
        if best_pair is not None:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            corr += best_delta
            best_c5 = best_val
        else:
            break
    
    return h, best_c5

def simulated_annealing(h, n, t_end, max_iters=200000, T_start=2.0, T_end=0.01):
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = np.max(corr) * dx
    best_c5 = cur_c5
    best_h = h.copy()
    d = np.arange(-(n - 1), n)
    n_iters = max_iters
    
    for it in range(n_iters):
        if time.time() > t_end:
            break
        T = T_start * (T_end / T_start) ** (it / n_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Compute delta
        ip = i + d
        im = i - d
        jp = j + d
        jm = j - d
        ci = np.zeros(2 * n - 1)
        m1 = (ip >= 0) & (ip < n)
        m2 = (im >= 0) & (im < n)
        ci[m1] += h[ip[m1]]
        ci[m2] += h[im[m2]]
        cj = np.zeros(2 * n - 1)
        m3 = (jp >= 0) & (jp < n)
        m4 = (jm >= 0) & (jm < n)
        cj[m3] += h[jp[m3]]
        cj[m4] += h[jm[m4]]
        ci[n - 1] = 0.0
        cj[n - 1] = 0.0
        delta = ci - cj
        
        new_max = np.max(corr + delta)
        new_c5 = new_max * dx
        accept = new_c5 < cur_c5 or (T > 1e-10 and np.random.rand() < np.exp(-(new_c5 - cur_c5) / T))
        
        if accept:
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5

def slsqp_refine(h, n, t_end, maxiter=300):
    """Short SLSQP for continuous refinement."""
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        # Smooth max
        T = 200.0
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    
    def con_fun(x):
        return x.sum() - target
    
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def interp_h(h, n_new):
    n_old = len(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s > 1e-14:
            h = np.clip(h * (target / s), 0, 1)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def greedy_binary_ls(h, n, t_end, max_sweeps=50000):
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    d = np.arange(-(n - 1), n)

        if time.time() > t_end:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_val = best_c5
        best_i, best_j = -1, -1
        best_delta = None
        for i in ones:
            ip = i + d
            im = i - d
            ci = np.zeros(2 * n - 1)
            m1 = (ip >= 0) & (ip < n)
            m2 = (im >= 0) & (im < n)
            ci[m1] += h[ip[m1]]
            ci[m2] += h[im[m2]]
            ci[n - 1] = 0.0
            jp = zeros[:, None] + d[None, :]
            jm = zeros[:, None] - d[None, :]
            jp_c = np.clip(jp, 0, n - 1)
            jm_c = np.clip(jm, 0, n - 1)
            cj = (np.where((jp >= 0) & (jp < n), h[jp_c], 0.0) +
                  np.where((jm >= 0) & (jm < n), h[jm_c], 0.0))

            if nmax[k] * dx < best_val - 1e-15:

            h[best_i] = 0.0
            h[best_j] = 1.0
            corr += best_delta
            best_c5 = best_val
        else:
            break
    return h, best_c5
```
