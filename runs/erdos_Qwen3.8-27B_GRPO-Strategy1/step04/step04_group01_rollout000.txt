import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m) + 1e-30))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-13):
    def obj(x):
        return smooth_c5(x, T)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def fix_sum(h, n):
    target = n / 2.0
    hc = np.clip(h, 0.0, 1.0)
    s = hc.sum()
    if abs(s - target) < 1e-14:
        return hc
    if s > target:
        order = np.argsort(hc)
        need = s - target
        for i in order:
            d = min(hc[i], need)
            hc[i] -= d
            need -= d
            if need <= 1e-14:
                break
    else:
        order = np.argsort(-hc)
        need = target - s
        for i in order:
            d = min(1.0 - hc[i], need)
            hc[i] += d
            need -= d
            if need <= 1e-14:
                break
    return hc

def coord_opt(h, n, i):
    def f1d(hi):
        ht = h.copy()
        ht[i] = hi
        corr = np.correlate(ht, 1.0 - ht, mode='full')
        return np.max(corr) * (2.0 / n)
    r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                        options={'xatol': 1e-13, 'maxiter': 50})
    return r.x

def coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_imp = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = fix_sum(ht, n)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        no_imp = 0 if improved else no_imp + 1
        if no_imp >= 4:
            break
    return best_h, best_c5

def make_start(n, seed):
    idx = np.random.RandomState(seed)
    p = idx.random(n)
    p = np.clip((p - 0.5) * 0.6 + 0.5, 0.0, 1.0)
    return fix_sum(p, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    ts = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_start(160, 42)
    n = len(best_h)
    best_h = fix_sum(best_h, n)
    best_c5 = compute_c5(best_h)

    n_starts = 7
    trial_ns = [120, 140, 160, 180, 200, 220, 240]
    used_n_set = set()
    if n in trial_ns:
        used_n_set.add(n)
    for j, ns in enumerate(trial_ns):
        if ns in used_n_set or ts() > budget_s * 0.06:
            continue
        used_n_set.add(ns)
        hh = make_start(ns, 100 + j * 17)
        ls = 0.6
        for T in [30, 100, 300, 800]:
            if ts() > budget_s * 0.06:

            try:
                hh = slsqp_opt(hh, T, ns, int(320 * ls + 80), 1e-13)

                ls *= 0.85
            except Exception:
                pass
        hh, c5h = coord_descent(hh, ns, min(t0 + budget_s * 0.06, ts() + 12), 200)
        nm = n if abs(c5h - best_c5) < 0.002 else ns
        if nm != ns:
            xq = np.linspace(0.0, 2.0, ns + 1)[:-1] + 1.0 / ns
            xx = np.linspace(0.0, 2.0, nm + 1)[:-1] + 1.0 / nm
            hh = fix_sum(np.interp(xx, xq, hh), nm)
        hh, c5h = coord_descent(hh, nm, min(t0 + budget_s * 0.06, ts() + 8), 80)

        if ts() > budget_s * 0.12:
            break
        try:
            hc = slsqp_opt(best_h.copy(), T, n, 600, 1e-14)
            hc = fix_sum(hc, n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

            dxv = 2.0 / n

                if l < 1e-13:
                    continue
                k = ki - (n - 1)

                    if k < n:

                    if k > 0:
                        gr[k:] += l * (-best_h[:n - k])
                else:
                    kk = -k
                    gr[kk:] += l * (1.0 - best_h[:n - kk])
                    gr[:n - kk] += l * (-best_h[kk:])
            gr *= dxv
            hp = fix_sum(best_h - (0.008 * cool + 0.001) * gr, n)
        hp = fix_sum(hp, n)
        try:
            hc = slsqp_opt(hp, 400 + 2000 * cool, n, 200, 1e-13)
            hc = fix_sum(hc, n)
        except Exception:
            hc = hp
        hc, c5v = coord_descent(hc, n, min(t0 + budget_s * 0.65, ts() + budget_s * 0.012), 60)
        tm = 0.0025 * cool + 0.0003
        if c5v < best_c5 or np.random.rand() < np.exp(-(c5v - best_c5) / max(tm, 1e-9)):
            if c5v < best_c5:
                best_c5, best_h = c5v, hc

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.85, 3000)

    for T in [2000, 8000, 25000, 60000]:
        if ts() > budget_s * 0.95:
            break
        try:
            hc = slsqp_opt(best_h.copy(), T, n, 1200, 1e-15)
            hc = fix_sum(hc, n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    if ts() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, 500)

    best_h = fix_sum(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
