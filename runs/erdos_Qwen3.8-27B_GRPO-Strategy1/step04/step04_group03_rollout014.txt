import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    hf = h.astype(float)
    return np.correlate(hf, 1.0 - hf, mode='full').astype(float)


def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)


def sum_eq(x):
    return x.sum() - x.size / 2.0


def smooth_max_obj(x, T):
    vals = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / len(x)
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(T * (vals - m)))) / T) * dx


def slsqp_polish(h, T, maxiter=400, ftol=1e-14):
    n = len(h)
    res = minimize(smooth_max_obj, h, args=(T,),
                   method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': sum_eq}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)


def min1d_convex_pl(corv, coef, lo, hi):
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = corv + mid * coef
        mx = np.max(vals)
        active = vals >= mx - 1e-15
        if not np.any(active):
            active = (vals == mx)
        sg_min = np.min(coef[active])
        sg_max = np.max(coef[active])
        if sg_min > 1e-13:
            hi = mid
        elif sg_max < -1e-13:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5


def coord_descent(h_in, t_end, n_sweeps=5000):
    h = h_in.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    n_ro = 2 * n - 1
    ks = np.arange(-(n - 1), n)
    idx = ks + (n - 1)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return h, best_c5
            h_old = h[i]
            j1 = i - ks
            j2 = i + ks
            valid1 = (j1 >= 0) & (j1 < n)
            valid2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(n_ro)
            if np.any(valid1):
                coef[idx[valid1]] += (1.0 - h[j1[valid1]])
            if np.any(valid2):
                coef[idx[valid2]] -= h[j2[valid2]]
            lov = -h_old
            hiv = 1.0 - h_old
            if hiv - lov < 1e-14:
                continue
            s_new = min1d_convex_pl(corr, coef, lov, hiv)
            if abs(s_new) > 1e-14:
                new_max = np.max(corr + s_new * coef)
                if new_max < best_c5 / dx - 1e-15:
                    h[i] += s_new
                    corr += s_new * coef
                    best_c5 = new_max * dx
                    improved = True
        if not improved:
            break
    return h, best_c5


def perturb_gauss(h, target, step, rng):
    return project_to_sum(h + step * rng.randn(len(h)), target)


def perturb_swap(h, target, n_pairs, rng):
    hc = h.copy()
    n = len(h)
    for _ in range(n_pairs):
        i, j = rng.choice(n, 2, replace=False)
        delta = (hc[j] - hc[i]) * rng.uniform(0.3, 1.0)
        hc[i] += delta
        hc[j] -= delta
    return project_to_sum(hc, target)


def perturb_block(h, target, rng):
    n = len(h)
    blen = rng.randint(max(4, n // 20), max(8, n // 5))
    start = rng.randint(0, n - blen)
    hc = h.copy()
    block = hc[start:start + blen].copy()
    block = np.sort(block)[::-1] if rng.rand() < 0.5 else np.sort(block)
    hc[start:start + blen] = block
    noise = 0.03 * rng.randn(blen)
    hc[start:start + blen] += noise
    return project_to_sum(hc, target)


def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=None)
    return project_to_sum(h_new, new_n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(224)
        best_h[:112] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP to refine (5% budget)
    t_p1 = t0 + budget_s * 0.05
    for T in [20, 100, 500, 2000]:
        if time.time() > t_p1:
            break
        try:
            hc = slsqp_polish(best_h, T, maxiter=300, ftol=1e-13)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 2: Main coordinate descent (50% budget)
    t_p2 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, t_p2, n_sweeps=20000)

    # Phase 3: Basin hopping (25% budget)
    t_p3 = t0 + budget_s * 0.80
    work_h = best_h.copy()
    work_c5 = best_c5
    n_it = 0
    while time.time() < t_p3 and n_it < 200:
        n_it += 1
        cool = max(1.0 - 0.95 * n_it / 200.0, 0.02)
        r = rng.rand()
        if r < 0.35:
            h_p = perturb_gauss(work_h, len(work_h) / 2, 0.02 + 0.08 * cool, rng)
        elif r < 0.65:
            h_p = perturb_swap(work_h, len(work_h) / 2, rng.randint(2, 6), rng)
        else:
            h_p = perturb_block(work_h, len(work_h) / 2, rng)
        # Quick CD polish (limit to ~2s)
        hc, c5v = coord_descent(h_p, time.time() + 2.0, n_sweeps=500)
        temp = 0.003 * cool + 0.00005
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 4: Try other n values (10% budget)
    t_p4 = t0 + budget_s * 0.90
    for n_try in [192, 256, 288]:
        if time.time() > t_p4:
            break
        h_s = upsample(best_h, n_try)
        hc, c5v = coord_descent(h_s, min(time.time() + 15, t_p4), n_sweeps=2000)
        if c5v < best_c5:
            best_c5, best_h, n = c5v, hc, n_try

    # Phase 5: Final deep CD polish (10% budget)
    t_p5 = t0 + budget_s * 0.995
    best_h, best_c5 = coord_descent(best_h, t_p5, n_sweeps=50000)

    # Final SLSQP with very high T
    if elapsed() < budget_s * 0.98:
        try:
            hc = slsqp_polish(best_h, 50000, maxiter=800, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
