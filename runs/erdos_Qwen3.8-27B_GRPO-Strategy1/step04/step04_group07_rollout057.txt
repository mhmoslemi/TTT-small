    hn = np.interp(xn, xo, h)
    return project(hn, n_new)

def block_perturb(h, n, scale, n_blocks=3):
    hp = h.copy()
    for _ in range(n_blocks):
        blen = np.random.randint(2, max(3, n // 6))
        start = np.random.randint(0, n - blen)
        hp[start:start + blen] += np.random.randn(blen) * scale
    return project(hp, n)

def lag_pair_opt(h, n, t_end, max_iters=200):
    """Focus on reducing the top 2 lags simultaneously via coordinate descent on relevant variables."""
    bh = h.copy()
    dx = 2.0 / n
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    cs = corr * dx
    for it in range(max_iters):
        if time.time() > t_end:
            break
        top2 = np.argsort(-corr)[:2]
        k1, k2 = int(top2[0]), int(top2[1])
        target = min(corr[k1], corr[k2])
        # Try to reduce corr[k1] by adjusting variables that most affect it
        # Variable i affects corr[k1] with coefficient (1-h[i+l])-h[i-l] where l = k1-(n-1)
        l1 = k1 - (n - 1)
        l2 = k2 - (n - 1)
        ks = np.arange(-(n - 1), n)
        # Find variable with largest negative impact on k1
        best_i, best_gain = -1, 0.0
        for i in range(n):
            old = bh[i]
            if old < 0.01 or old > 0.99:
                continue
            v = 0.0 if old > 0.5 else 1.0
            ip = i + ks
            im = i - ks
            coeff = np.zeros(2 * n - 1)
            m1 = (ip >= 0) & (ip < n)
            coeff[m1] = 1.0 - bh[ip[m1]]
            m2 = (im >= 0) & (im < n)
            coeff[m2] -= bh[im[m2]]
            dv = v - old
            cn = corr + dv * coeff
            cn[n-1] -= dv * dv
            gain = corr[k1] - float(cn[k1])
            if gain > best_gain and float(np.max(cn)) < coral * 1.001:
                best_gain = gain
                best_i = i
                best_v = v
                best_cn = cn
        if best_i < 0:
            break
        bh[best_i] = best_v
        corr = best_cn
    return bh, float(np.max(corr)) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def remain():
        return max(0, deadline - time.time())

    # Phase 1: Thorough CD on initial
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.20, max_sweeps=2000)

    # Phase 2: Try 1-h symmetry
    h_flip = project(1.0 - best_h, n)
    h_flip, c5_fl = coord_descent(h_flip, n, t0 + budget_s * 0.25, max_sweeps=500)
    if c5_fl < best_c5:
        best_c5, best_h = c5_fl, h_flip

    # Phase 3: Multi-resolution search
    for n_try in [100, 120, 140, 180, 200, 240, 280, 320]:
        if time.time() > t0 + budget_s * 0.42:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.427, max_sweeps=300)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try
        h_fl = project(1.0 - h_tr, n_try)
        h_fl, c5_fl = coord_descent(h_fl, n_try, t0 + budget_s * 0.427, max_sweeps=200)
        if c5_fl < best_c5:
            best_c5, best_h, n = c5_fl, h_fl, n_try

    # Phase 4: Population-based global search
    pop = [best_h.copy()]
    for _ in range(7):
        h_p = block_perturb(best_h, n, scale=np.random.uniform(0.05, 0.25), n_blocks=np.random.randint(2, 6))
        h_p, _ = coord_descent(h_p, n, t0 + budget_s * 0.46, max_sweeps=80)
        pop.append(h_p)
        h_p2 = project(1.0 - h_p, n)
        h_p2, _ = coord_descent(h_p2, n, t0 + budget_s * 0.46, max_sweeps=60)
        pop.append(h_p2)
    pop_c5 = np.array([compute_c5(p) for p in pop])
    bi = int(np.argmin(pop_c5))
    if pop_c5[bi] < best_c5:
        best_c5, best_h = float(pop_c5[bi]), pop[bi].copy()

    # Evolve best few
    for gen in range(60):
        if time.time() > t0 + budget_s * 0.62:
            break
        n_cands = 4
        for _ in range(n_cands):
            r = np.random.rand()
            if r < 0.4:
                hp = block_perturb(best_h, n, np.random.uniform(0.03, 0.15), np.random.randint(1, 4))
            elif r < 0.7:
                hp = best_h.copy()
                blen = np.random.randint(3, n // 3 + 1)
                st = np.random.randint(0, n - blen)
                hp[st:st+blen] = 1.0 - hp[st:st+blen]
                hp = project(hp, n)
            else:
                hp = block_perturb(pop[np.random.randint(len(pop))], n, 0.1, 2)
            hp, c5_hp = coord_descent(hp, n, t0 + budget_s * 0.625, max_sweeps=100)
            if c5_hp < best_c5:
                best_c5, best_h = c5_hp, hp
            pop.append(hp)
        pop_c5 = np.array([compute_c5(p) for p in pop])
        pop = [pop[i].copy() for i in np.argsort(pop_c5)[:10]]

    # Phase 5: Final intensive CD
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.80, max_sweeps=3000)

    # Phase 6: Try 0-1 rounding
    if time.time() < t0 + budget_s * 0.85:
        h01 = (best_h > 0.5).astype(float)
        h01 = project(h01, n)
        h01, c5_01 = coord_descent(h01, n, t0 + budget_s * 0.86, max_sweeps=500)
        if c5_01 < best_c5:
            best_c5, best_h = c5_01, h01

    # Phase 7: Final polish with very detailed CD
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, max_sweeps=3000)

    # Phase 8: One more resolution check
    if time.time() < t0 + budget_s * 0.97:
        for n_try in [n - 20, n + 20]:
            if n_try < 40:
                continue
            h_tr = interp_h(best_h, n_try)
            h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.975, max_sweeps=300)
            if c5_tr < best_c5:
                best_c5, best_h, n = c5_tr, h_tr, n_try

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
