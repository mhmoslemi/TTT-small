def coord_opt_fast(h, n, i):
    """Optimize h[i] to minimize max correlation, given all other h fixed."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n - 1)])
    dx = 2.0 / n
    
    # The objective is max(corr_base + hi * coefs) * dx
    # This is a piecewise linear function of hi in [0,1]
    # The maximum is at hi=0, hi=1, or at intersection points
    # Since coefs can be positive or negative, the max of corr_base + hi*coefs
    # is a convex piecewise linear function of hi
    # Minimum of convex function is at the steepest crossing or endpoint
    
    # Find where corr_base + hi*coefs changes order
    # For the min of max, we want to find hi that minimizes the maximum
    # This is a 1D minimax problem
    
    # Simple approach: ternary search or golden section on [0,1]
    def f(hi):
        return np.max(corr_base + hi * coefs)
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(T * (corr - m)) / np.sum(np.exp(T * (corr - m)))

    idx = np.argsort(weights)[::-1][:top_k]

    s_vals = idx - (n - 1)
    
    grad = np.zeros(n)
    for wi, s in zip(w, s_vals):
        if wi < 1e-15:
            continue
        # Contribution: wi * [(1-h[j+s]) - h[j-s]] for valid indices
        if s >= 0:
            jmax = n - s
            grad[:jmax] += wi * (1.0 - h[s:s+jmax])  # (1-h[j+s]) for j=0..jmax-1
        else:
            jmin = -s
            grad[jmin:] += wi * (1.0 - h[:n-jmin])  # (1-h[j+s]) for j=jmin..n-1
        if s <= 0:
            jmax2 = n + s if s < 0 else n
            grad[:jmax2] -= wi * h[-s:-s+jmax2] if s < 0 else h[0:jmax2]
        else:
            jmin2 = s
            grad[jmin2:] -= wi * h[0:n-jmin2]
    
    return grad * dx

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, T, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    idx = np.argsort(p)[::-1][:top_k]
    grad = np.zeros(n)
    for k in idx:
        pk = p[k]
        if pk < 1e-16:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n - s]
        else:
            neg_s = -s
            grad[neg_s:] += pk * (1.0 - h[:n + s])
            grad[:n + s] -= pk * h[neg_s:]
    return grad * dx

def nesterov_opt(h, t_end, n_max=15000):
    n = len(h)
    dx = 2.0 / n
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T_init, T_end = 200000.0, 200.0
    beta = 0.92
    for it in range(n_max):
        if time.time() > t_end:
            break
        frac = it / max(n_max, 1)
        T = T_init * (T_end / T_init) ** frac
        lr = 0.8 / (T ** 0.3)
        g = smooth_grad(h, T, top_k=40)
        v = beta * v - lr * g
        h_new = project_h(h + v)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new.copy()
        h = h_new
        if it > 100 and (it % 500 == 0):
            pass
    return best_h, best_c5

def slsqq_polish(h, n, t_end, T_list=None, maxiter=300):
    if T_list is None:
        T_list = [10000, 50000, 200000]
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for T in T_list:
        if time.time() > t_end:
            break
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
        def con_eq(x):
            return x.sum() - n / 2.0
        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': con_eq}],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            hc = project_h(np.clip(res.x, 0, 1))
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    return best_h, best_c5

def coord_single(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    corr_base = np.correlate(h0, 1.0 - h0, mode='full')
    kr = np.arange(-(n - 1), n)
    j1 = i + kr
    j2 = i - kr
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    def f1d(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=600):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def perturb_h(h, n, scale):
    x = np.arange(n)
    f1 = np.random.randint(1, 15)
    f2 = np.random.randint(1, 15)
    p1 = np.random.uniform(0, 2 * np.pi)
    p2 = np.random.uniform(0, 2 * np.pi)
    pert = scale * (np.sin(2 * np.pi * f1 * x / n + p1) +
                    0.5 * np.sin(2 * np.pi * f2 * x / n + p2) +
                    0.3 * np.random.randn(n))
    return project_h(h + pert)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    # Phase 1: Nesterov momentum gradient
    t1 = t0 + budget_s * 0.30
    best_h, best_c5 = nesterov_opt(best_h, t1, n_max=12000)
    # Phase 2: SLSQP polishing
    t2 = t0 + budget_s * 0.50
    best_h, best_c5 = slsqq_polish(best_h, n, t2, T_list=[5000, 50000, 300000], maxiter=250)
    # Phase 3: Basin hopping with quick polish
    t3 = t0 + budget_s * 0.72
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(80):
        if time.time() > t3:
            break
        cool = 1.0 - 0.85 * (it / 80.0)
        scale = 0.02 + 0.08 * cool
        h_p = perturb_h(work_h, n, scale)
        hc, c5v = slsqq_polish(h_p, n, min(time.time() + 4.0, t3),
                                T_list=[20000, 100000], maxiter=120)
        temp = 0.0008 * cool + 0.0001
        accept = c5v < work_c5 or (c5v - work_c5) / max(temp, 1e-9) < -np.log(np.random.rand() + 1e-15)
        if accept:
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()
    # Phase 4: Heavy coordinate descent
    t4 = t0 + budget_s * 0.93
    best_h, best_c5 = coord_descent(best_h, n, t4, max_sweeps=800)
    # Phase 5: Final polish
    if time.time() < t0 + budget_s * 0.96:
        best_h, best_c5 = slsqq_polish(best_h, n, t0 + budget_s * 0.96,
                                        T_list=[500000], maxiter=200)
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=80)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
