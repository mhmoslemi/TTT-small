import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sh = T * (corr - m)
    ms = sh.max()
    return (m + (ms + np.log(np.sum(np.exp(sh - ms)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sh = T * (corr - m)
    ms = sh.max()
    w = np.exp(sh - ms)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wi * (1.0 - h[:n-k])
            grad[:n-k] -= wi * h[k:]
        else:
            mk = -k
            grad[:n-mk] += wi * (1.0 - h[mk:])
            grad[mk:] -= wi * h[:n-mk]
    return grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    for _ in range(20):
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            mass = h
            tot = mass.sum()
            if tot < 1e-14:
                break
            h = h - mass * (abs(diff) / tot)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    return h

def coord_opt_fast(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    center = n - 1
    delta = np.zeros(2 * n - 1)
    ks = np.arange(1, n)
    vp = ks <= i
    vp2 = i + ks < n
    if np.any(vp):
        kp = ks[vp]
        delta[center + kp] += (1.0 - h0[i - kp])
        delta[center - kp] -= h0[i - kp]
    if np.any(vp2):
        kp2 = ks[vp2]
        delta[center + kp2] -= h0[i + kp2]
        delta[center - kp2] += (1.0 - h0[i + kp2])
    dx = 2.0 / n
    cb = corr
    ct = cb[center]
    def f1d(x):
        v = cb + x * delta
        v[center] = ct + x - x * x
        return np.max(v) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-9})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            xi = coord_opt_fast(best_h, n, i)
            if abs(xi - best_h[i]) > 1e-10:
                best_h[i] = xi
                best_h = project(best_h, n)
                c5v = compute_c5(best_h)
                if c5v < best_c5:
                    best_c5 = c5v
                    improved = True
    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=800):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def grad_momentum(h, n, T, t_end, lr=0.002, max_iter=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    beta = 0.9
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = smooth_grad(best_h, T)
        v = beta * v + (1 - beta) * g
        best_h = project(best_h - lr * v, n)
        c5v = compute_c5(best_h)
        if c5v < best_c5:
            best_c5 = c5v
        elif it > 100:
            lr *= 0.95
    return best_h, best_c5

def push_binary(h, n, t_end, max_rounds=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        h_try = best_h.copy()
        mask = h_try < 0.3
        h_try[mask] = 0.0
        mask2 = h_try > 0.7
        h_try[mask2] = 1.0
        h_try = project(h_try, n)
        c5v = compute_c5(h_try)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_try
        else:
            h_mix = 0.5 * best_h + 0.5 * np.where(best_h < 0.3, 0.0,
                   np.where(best_h > 0.7, 1.0, best_h))
            h_mix = project(h_mix, n)
            c5m = compute_c5(h_mix)
            if c5m < best_c5:
                best_c5 = c5m
                best_h = h_mix
            else:
                break
    return best_h, best_c5
