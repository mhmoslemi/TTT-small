import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    for _ in range(10):
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                h = h + diff / n
                break
            h = h + room * (diff / tot)
        else:
            h = h - h * (diff / max(s, 1e-14))
            h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    return h

def subgradient_descent(h, n, t_end, rng, max_iter=5000):
    """Subgradient method with momentum for C5 minimization."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_cur = h.copy()
    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum coefficient
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = compute_corr(h_cur)
        c5_vals = corr * dx
        c5_max = c5_vals.max()
        
        if c5_max < best_c5:
            best_c5 = c5_max
            best_h = h_cur.copy()
        
        # Find active shifts (within 5% of max)
        threshold = c5_max * 0.95
        active = c5_vals > threshold
        if not active.any():
            active[np.argmax(c5_vals)] = True
        
        # Compute subgradient (weighted by excess)
        excess = np.maximum(c5_vals - threshold, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            w = active.astype(float) / active.sum()
        else:
            w = excess / total_excess
        
        grad = np.zeros(n)
        # Vectorized gradient computation
        for k in range(1, n):
            wp = w[n-1+k] if w[n-1+k] > 0 else 0
            wn = w[n-1-k] if w[n-1-k] > 0 else 0
            if wp + wn < 1e-15:
                continue
            if wp > 0:
                grad[0:n-k] += wp * (1 - h_cur[k:n])
                grad[k:n] += -wp * h_cur[0:n-k]
            if wn > 0:
                grad[0:n-k] += -wn * h_cur[k:n]
                grad[k:n] += wn * (1 - h_cur[0:n-k])
        # k=0
        w0 = w[n-1]
        if w0 > 0:
            grad += w0 * (1 - 2 * h_cur)
        
        # Step size (diminishing)
        step = 2.0 / (10 + it * 0.5)
        
        # Momentum update
        v = beta * v - step * grad
        h_new = project(h_cur + v, n)
        
        if not np.all(np.isfinite(h_new)):
            h_new = project(h_cur, n)
        
        # Acceptance: always accept in subgradient method (or use averaging)
        h_cur = h_new
    
    return best_h, best_c5

    v = np.zeros(n)
    beta = 0.85
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = compute_corr(xc)
        c5v = corr * dx
        c5m = c5v.max()
        if c5m < best_c5:
            best_c5 = c5m
            best_h = xc.copy()
        thr = c5m * 0.97
        excess = np.maximum(c5v - thr, 0)
        te = excess.sum()
        if te < 1e-14:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c5v)] = 1.0
        else:
            w = excess / te
        grad = np.zeros(n)
        for k in range(1, n):
            wp = w[n - 1 + k]
            wn = w[n - 1 - k]
            if wp + wn < 1e-14:
                continue
            if wp > 0:
                grad[0:n-k] += wp * (1.0 - xc[k:n])
                grad[k:n] += -wp * xc[0:n-k]
            if wn > 0:
                grad[0:n-k] += -wn * xc[k:n]
                grad[k:n] += wn * (1.0 - xc[0:n-k])
        w0 = w[n - 1]
        if w0 > 0:
            grad += w0 * (1.0 - 2.0 * xc)
        step = 1.5 / (5.0 + 0.3 * it)
        v = beta * v - step * grad
        xn = project(xc + v, n)
        if not np.all(np.isfinite(xn)):
            xn = project(xc, n)
        xc = xn
    return best_h, best_c5

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=400):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': coneq}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h_old), n_new)

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return project(h, n)

rng_global = np.random.default_rng(0)

def run(seed=42, budget_s=1100, **kwargs):
    global rng_global
    rng_global = np.random.default_rng(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def time_left(frac):
        return t0 + budget_s * frac

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = random_binary(240, rng)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: coordinate descent (20%)
    best_h, best_c5 = coord_sweeps(best_h, n, time_left(0.20), max_sweeps=150, rng=rng)

    # Phase 2: subgradient Nesterov (20%)
    if elapsed() < budget_s * 0.40:
        h2, c2 = subgradient_nesterov(best_h, n, time_left(0.40), max_iter=2500)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
        # Polish after subgradient
        best_h, best_c5 = coord_sweeps(best_h, n, min(time_left(0.48), time_left(0.40) + 30), max_sweeps=50, rng=rng)

    # Phase 3: binary swap search (15%)
    if elapsed() < budget_s * 0.55:
        b_h = np.where(best_h > 0.5, 1.0, 0.0)
        b_h = project(b_h, n)
        b_h, b_c = binary_swap_search(b_h, n, time_left(0.55), rng, n_iters=4000)
        if b_c < best_c5:
            best_h, best_c5 = b_h, b_c
            best_h, best_c5 = coord_sweeps(best_h, n, time_left(0.60), max_sweeps=30, rng=rng)

    # Phase 4: multi-resolution (25%)
    for n_try in [200, 220, 280, 300, 320]:
        if elapsed() > budget_s * 0.82:
            break
        h_interp = interp_h(best_h, n_try)
        h_interp, c_interp = coord_sweeps(h_interp, n_try, min(time_left(0.85), time_left(0.82) + 40), max_sweeps=40, rng=rng)
        # SLSQP polish
        try:
            for T in [100, 800]:
                if elapsed() > budget_s * 0.84:
                    break
                hc = slsqp_opt(h_interp, T, n_try, maxiter=300)
                cv = compute_c5(hc)
                if cv < c_interp:
                    h_interp, c_interp = hc, cv
        except Exception:
            pass
        h_interp, c_interp = coord_sweeps(h_interp, n_try, min(time_left(0.87), time_left(0.82) + 60), max_sweeps=30, rng=rng)
        if c_interp < best_c5:
            best_h, best_c5, n = h_interp, c_interp, n_try

    # Phase 5: targeted multi-start (10%)
    if elapsed() < budget_s * 0.90:
        for trial in range(3):
            if elapsed() > budget_s * 0.89:
                break
            h_rand = random_binary(n, rng)
            # Blend with best
            h_mix = 0.7 * best_h + 0.3 * h_rand
            h_mix = project(h_mix, n)
            h_mix, c_mix = coord_sweeps(h_mix, n, min(time_left(0.90), time_left(0.89) + 30), max_sweeps=25, rng=rng)
            if c_mix < best_c5:
                best_h, best_c5 = h_mix, c_mix

    # Phase 6: final polish (10%)
    if elapsed() < budget_s * 0.95:
        best_h, best_c5 = coord_sweeps(best_h, n, time_left(0.95), max_sweeps=60, rng=rng)
    # Final SLSQP
    for T in [2000, 10000]:
        if elapsed() > budget_s * 0.98:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=1500)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
