def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_val(h, n, i):
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1]) if j1 != i else (1.0 - 0.0)
        else:
            pass
        if 0 <= j2 < n and j2 != i:
            c += (-h[j2])
        if j1 == i:
            c = (1.0)  # 1-hz[i] = 1-0 = 1 for lag=0 contribution
            if 0 <= j2 < n and j2 != i:
                c += (-h[j2])
        coef[k] = c
    k0 = n - 1
    def f1d(t):
        cv = corr_base + t * coef
        cv[k0] = corr_base[k0] + t * (1.0 - t)
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-14})
    return res.x

def slsqp_opt(h, T, n, maxiter=600, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def active_slsqp(h, n, n_active=5, T=5000, maxiter=3000):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    thresh = m - (m * 0.01 + 1e-10)
    active = np.where(corr >= thresh)[0][:n_active]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    def obj_f(x):
        c = np.correlate(x, 1.0 - x, mode='full')[active]
        cs = np.clip(T * (c - c.max()), -500, 500)
        return (c.max() + np.log(np.sum(np.exp(cs))) / T) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj_f, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj(np.clip(res.x, 0, 1), n)

def coord_sweeps(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(12345)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_val(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def perturb_reopt(h, n, best_c5, t_end, n_trials=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(99)
    work_h = h.copy()
    work_c5 = best_c5

        if time.time() > t_end:

        frac = it / float(n_trials)
        step = 0.01 + 0.06 * (1.0 - frac)

        elif strategy == 1:
            blk = rng.randint(1, n // 3)

        else:
            mask = (rng.rand(n) < 0.15)

            c5v = compute_c5(hc)
        except Exception:
            continue
        if c5v < work_c5:
            work_c5 = c5v
            work_h = hc
    return work_h, work_c5
