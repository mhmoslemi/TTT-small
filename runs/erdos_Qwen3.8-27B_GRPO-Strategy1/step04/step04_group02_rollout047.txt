    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + logsumexp(T * (corr - m)) / T) * dx

def project(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14: return h
        h = h + room * (diff / tot)
    else:
        h = h * (target / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        d2 = target - s2
        if d2 > 1e-13:
            room = 1.0 - h
            if room.sum() > 1e-14:
                h = h + room * (d2 / room.sum())
    return h

def coord_opt(h, n, i):
    hs = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = hs
    dx = 2.0 / n
    ks = np.arange(1, n)
    jm = i - ks
    jp = i + ks
    mm = (jm >= 0) & (jm < n)
    mp = (jp >= 0) & (jp < n)
    d = np.zeros(2*n - 1)
    dr = np.zeros(n-1); dl = np.zeros(n-1)
    dr[mm] += 1.0 - h[jm[mm]]
    dr[mp] += -h[jp[mp]]
    dl[mm] += -h[jm[mm]]
    dl[mp] += 1.0 - h[jp[mp]]
    d[n-1+ks] = dr
    d[n-1-ks] = dl
    c0b = cb[n-1] - hs*(1.0-hs)
    def f1d(t):
        vals = cb + t*d
        vals[n-1] = c0b + t*(1.0-t)
        return np.max(vals)*dx
    res = minimize_scalar(f1d, bounds=(0.0,1.0), method='bounded', options={'xatol':1e-12})
    return res.x

def slsqp_opt(h, T, n, mi=800, ft=1e-14):
    def obj(x): return smooth_c5(x, T)
    b = [(0.0,1.0)]*n
    r = minimize(obj, h, method='SLSQP', bounds=b,
                 constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0}],
                 options={'maxiter':mi,'ftol':ft})
    return project(np.clip(r.x,0,1),n)

def cd(h, n, t_end, sweeps=100, rng=None):

    bh = h.copy(); bc = compute_c5(bh)

            bh[i] = nv
            bh = project(bh, n)
            c = compute_c5(bh)
            if c < bc - 1e-14: bc = c; imp = True
        if imp: stall = 0
        else:
            stall += 1
            if stall >= 5: break
    return bh, bc

    xn = np.linspace(0, 2, nn, endpoint=False)
