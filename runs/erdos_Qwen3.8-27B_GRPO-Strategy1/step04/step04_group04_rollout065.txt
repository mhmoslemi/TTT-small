def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += room * (diff / tot)
        else:
            if s < 1e-13:
                break
            h *= (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    dx = 2.0 / len(h)
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - n1
        if lag == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif lag > 0:
            L = n - lag
            if L > 0:
                grad[:L] += pk * (1.0 - h[lag:])
            if lag < n:
                grad[lag:] += -pk * h[:L]
        else:
            nl = -lag
            L = n - nl
            if L > 0:
                grad[nl:] += pk * (1.0 - h[:L])
                grad[:L] += -pk * h[nl:]
    return grad * dx

def grad_descent_nesterov(h, n, t_end, T_start=50, T_end=20000, max_iter=10000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    beta = 0.9
    T = T_start
    lr = 0.005
    for it in range(max_iter):
        if time.time() > t_end:
            break
        T = T_start * (T_end / T_start) ** (it / max(max_iter - 1, 1))
        g = softmax_grad(best_h, T)
        # Nesterov: grad at h + beta*v
        h_look = proj(best_h + beta * v, n)
        g_look = softmax_grad(h_look, T)
        v = beta * v - lr * g_look
        h_new = proj(best_h + v, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            # Occasionally increase lr slightly
            if it % 200 == 0 and it > 0:
                lr *= 1.02
        else:
            lr *= 0.9
            v *= 0.5
    return best_h, best_c5

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h += room * (diff / tot)
        else:
            if s < 1e-12:
                break
            h *= (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - n1
        if lag == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif lag > 0:
            L = n - lag
            if L > 0:
                grad[:L] += pk * (1.0 - h[lag:])
            grad[lag:] += -pk * h[:L]
        else:
            nl = -lag
            L = n - nl
            if L > 0:
                grad[nl:] += pk * (1.0 - h[:L])
                grad[:L] += -pk * h[nl:]
    return grad * dx

def grad_desc(h, n, t_end, T_init=100, max_iter=20000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T = T_init
    lr = 0.02
    no_improve = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = softmax_grad(best_h, T)
        step = lr
        accepted = False
        for _ in range(20):
            h_try = proj(best_h - step * g, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                best_h = h_try
                accepted = True
                no_improve = 0
                break
            step *= 0.4
        if accepted:
            if no_improve % 50 == 0:
                T *= 1.05
                lr = min(lr * 1.03, 0.05)
        else:
            no_improve += 1
            lr *= 0.5
            if no_improve > 200:
                break
    return best_h, best_c5

def coord_sweep(h, n, t_end, max_sweeps=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            hz = best_h.copy()
            hz[i] = 0.0
            corr_base = np.correlate(hz, 1.0 - hz, mode='full')
            dx = 2.0 / n
            coef = np.zeros(2 * n - 1)
            n1 = n - 1
            for k in range(2 * n - 1):
                lag = k - n1
                if lag == 0:

                c = 0.0
                jp = i + lag
                jm = i - lag
                if 0 <= jp < n:

                best_h = h_try

def compute_coef(hz, i, n):
    n1 = n - 1
    coef = np.zeros(2*n - 1)
    for lag in range(1, n):
        # Positive lag
        k_pos = n1 + lag
        L = n - lag
        # coef[k_pos] involves hz[i+lag] and hz[i-lag]
        # For the CORRELATION: corr[k_pos] = sum_{j=0}^{L-1} hz[j]*(1-hz[j+lag])
        # d/dhz[i]: (1-hz[i+lag]) if i < L, -hz[i-lag] if i >= lag
        if i < L:
            coef[k_pos] += (1.0 - hz[i + lag])
        if i >= lag:
            coef[k_pos] -= hz[i - lag]
        # Negative lag
        k_neg = n1 - lag
        # corr[k_neg] = sum_{j=lag}^{n-1} hz[j]*(1-hz[j-lag])
        # d/dhz[i]: (1-hz[i-lag]) if i >= lag, -hz[i+lag] if i < n-lag
        if i >= lag:
            coef[k_neg] += (1.0 - hz[i - lag])
        if i < n - lag:
            coef[k_neg] -= hz[i + lag]
    return coef
