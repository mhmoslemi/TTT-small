def compute_corr(h):
    return np.correlate(h.astype(float), (1.0 - h).astype(float), mode='full')

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

**Idea: Exploit the structure of the correlation.**

The correlation C[k] for shift l = k-(n-1) is:
C[n-1+l] = Σ_j h[j]·(1-h[j+l]) for valid j, j+l

= Σ_{j=max(0,-l)}^{n-1-min(1,l)} h[j]·(1-h[j+l])
= Σ_{j} h[j]·(1-h[j+l]) (with implicit bounds)

For large |l|, the number of terms is small. For l=0: C[n-1] = Σ h[j](1-h[j]) = Σ h[j] - Σ h[j]². With sum(h) = n/2, this is n/2 - Σ h[j]².

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), (1.0 - h).astype(float), mode='full')

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def find_min_pl(a, b, lo, hi):
    vals_lo = a + lo * b
    m_lo = np.max(vals_lo)
    act = vals_lo >= m_lo - 1e-12
    if np.min(b[act]) >= -1e-14:
        return lo, m_lo
    vals_hi = a + hi * b
    m_hi = np.max(vals_hi)
    act = vals_hi >= m_hi - 1e-12
    if np.max(b[act]) <= 1e-14:
        return hi, m_hi
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        vals = a + mid * b
        m = np.max(vals)
        act = vals >= m - 1e-12
        min_s = np.min(b[act])
        max_s = np.max(b[act])
        if max_s < -1e-14:
            lo = mid
        elif min_s > 1e-14:
            hi = mid
        else:
            break
    s_opt = 0.5 * (lo + hi)
    return s_opt, np.max(a + s_opt * b)

def fast_cd(h, t_end, n_sweeps=5000):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    l = np.arange(-(n - 1), n)
    idx1_all = np.arange(n)[:, None] + l[None, :]
    idx2_all = np.arange(n)[:, None] - l[None, :]
    v1_all = (idx1_all >= 0) & (idx1_all < n)
    v2_all = (idx2_all >= 0) & (idx2_all < n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            coef = np.zeros(2 * n - 1)
            coef[v1_all[i]] += (1.0 - best_h[idx1_all[i][v1_all[i]]])
            coef[v2_all[i]] -= best_h[idx2_all[i][v2_all[i]]]
            h_old = best_h[i]
            lo, hi = -h_old, 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            s_new, _ = find_min_pl(corr, coef, lo, hi)
            if abs(s_new) > 1e-13:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5 - 1e-16:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = compute_corr(x)
        m = np.max(corr)
        sm = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return sm * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup (10% budget)
    t1 = t0 + budget_s * 0.10

        hc = slsqp_polish(best_h, T, n, maxiter=600, ftol=1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Phase 2: Heavy fast coordinate descent (45% budget)
    t2 = t0 + budget_s * 0.55

    # Phase 3: SLSQP + CD refinement (15% budget)
    t3 = t0 + budget_s * 0.70

        if time.time() > t3:

        hc = slsqp_polish(best_h, T, n, maxiter=1000, ftol=1e-15)

    # Phase 4: Basin hopping (10% budget)

    n_bh = 30
