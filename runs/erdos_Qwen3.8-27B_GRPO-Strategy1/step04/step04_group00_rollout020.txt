def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_step(h, n, i):

    old = h[i]
    if old <= 1e-10 or old >= 1.0 - 1e-10:

        pass
    k_range = np.arange(-(n-1), n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def bisect_1d(corr, coeff, nm1, lo, hi):
    """Bisection on subgradient to find delta minimizing max_k(corr[k]+d*coeff[k]-[k==nm1]*d^2)."""
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = corr + mid * coeff
        vals[nm1] -= mid * mid
        m = vals.max()
        active = vals >= m - 1e-12
        slope = coeff[active].sum()
        if active[nm1]:
            slope -= 2.0 * mid
        if slope > 0:
            hi = mid
        elif slope < 0:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_sweep(h, n, corr, order, t_end):
    """One sweep of coordinate descent with incremental corr update."""
    nm1 = n - 1
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    improved = False
    for i in order:
        if time.time() > t_end:
            break
        old = h[i]
        k_range = np.arange(-(n - 1), n)
        i_mk = i - k_range
        i_pk = i + k_range
        v1 = (i_mk >= 0) & (i_mk < n)
        v2 = (i_pk >= 0) & (i_pk < n)
        coeff = (1.0 - h[np.clip(i_mk, 0, n - 1)]) * v1.astype(np.float64) - \

        lo = -old
        hi = 1.0 - old

        new_val = old + delta

def make_perturbation(rng, n, cool, mod):
    if mod == 0:
        step = 0.04 + 0.12 * cool
        return step * rng.randn(n)
    elif mod == 1:
        freq = rng.randint(1, 15)
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.04 + 0.10 * cool
        return amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mod == 2:
        mask = rng.random(n) < 0.15
        pert = np.zeros(n)
        pert[mask] = (0.08 + 0.15 * cool) * rng.randn(int(mask.sum()) + 1)
        return pert
    elif mod == 3:
        # Flip-based perturbation
        n_flip = max(1, int(n * 0.05 * (0.5 + cool)))
        idx = rng.choice(n, size=n_flip, replace=False)
        pert = np.zeros(n)
        pert[idx] = rng.choice([-1, 1], size=n_flip) * (0.15 + 0.2 * cool)
        return pert
    elif mod == 4:
        # Low-frequency distortion
        x = np.arange(n) / n
        pert = np.zeros(n)
        f1 = rng.uniform(0.5, 3.0)
        f2 = rng.uniform(3.0, 8.0)
        pert += (0.05 + 0.08 * cool) * np.sin(2 * np.pi * f1 * x + rng.uniform(0, 6.28))
        pert += (0.03 + 0.05 * cool) * np.cos(2 * np.pi * f2 * x + rng.uniform(0, 6.28))
        return pert
    else:
        step = 0.02 + 0.06 * cool
        return step * rng.randn(n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=np.float64)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Initial coordinate descent refinement (10% budget)
    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=300, rng=rng)

    # Phase 2: Basin hopping (45% budget)
    t2 = t0 + budget_s * 0.55
    work_h = best_h.copy()
    work_c5 = best_c5
    it = 0
    while elapsed() < t2 - t0:
        cool = max(0.0, 1.0 - it / 300.0)
        mod = it % 6
        pert = make_perturbation(rng, n, cool, mod)
        h_pert = project_h(work_h + pert)
        cd_sweeps = 60 if cool > 0.3 else 40
        h_opt, c5_opt = coord_descent(h_pert, n, min(t2, t0 + elapsed() + 8), max_sweeps=cd_sweeps, rng=rng)
        temp = 0.0008 * cool + 0.00005
        if c5_opt < work_c5 or rng.rand() < np.exp(-(c5_opt - work_c5) / temp):
            work_h, work_c5 = h_opt, c5_opt
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
        it += 1

    if work_c5 < best_c5:
        best_h, best_c5 = work_h, work_c5

    # Phase 3: Multi-resolution exploration (20% budget)
    t3 = t0 + budget_s * 0.75
    x_old = np.linspace(0, 2, n, endpoint=False)
    for n_try in [80, 120, 200, 240, 320]:
        if elapsed() > t3 - t0 - 3:
            break
        if n_try == n:
            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_h(h_interp)
        h_try, c5_try = coord_descent(h_interp, n_try, min(t3, t0 + elapsed() + 4), max_sweeps=150, rng=rng)
        if c5_try < best_c5:
            best_h, best_c5, n = h_try, c5_try, n_try
            x_old = np.linspace(0, 2, n, endpoint=False)

    # Phase 4: Second basin hopping round with smaller steps (15% budget)
    t4 = t0 + budget_s * 0.90
    work_h = best_h.copy()
    work_c5 = best_c5
    it2 = 0
    while elapsed() < t4 - t0:
        cool = max(0.0, 1.0 - it2 / 100.0)
        step = 0.01 + 0.05 * cool
        pert = step * rng.randn(n)
        if it2 % 3 == 1:
            freq = rng.randint(1, 20)
            pert += 0.02 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 6.28))
        h_pert = project_h(work_h + pert)
        h_opt, c5_opt = coord_descent(h_pert, n, min(t4, t0 + elapsed() + 5), max_sweeps=80, rng=rng)
        temp = 0.0003 * cool + 0.00002
        if c5_opt < work_c5 or rng.rand() < np.exp(-(c5_opt - work_c5) / temp):
            work_h, work_c5 = h_opt, c5_opt
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
        it2 += 1

    if work_c5 < best_c5:
        best_h, best_c5 = work_h, work_c5

    # Phase 5: Final coordinate descent polish (10% budget)
    t5 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=500, rng=rng)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
