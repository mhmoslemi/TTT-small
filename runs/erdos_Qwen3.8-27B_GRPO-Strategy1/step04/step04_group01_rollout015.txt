        ord_ = np.random.permutation(n)
        for i in ord_:
            if time.time() > t_end:
                return bh, bc5
            ov = bh[i]
            def f1(v):
                bh[i] = v
                return np.max(np.correlate(bh, 1.0 - bh, mode='full'))
            r = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-13})
            bh[i] = r.x
            if abs(r.x - ov) > 1e-12:
                bh = project(bh)
                c5n = compute_c5(bh)
                if c5n < bc5 - 1e-13:
                    bc5 = c5n
                    imp = True
        stall = 0 if imp else stall + 1
        if stall >= 5:
            break
    return bh, bc5

def subgrad_direction(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        if w[ki] < 1e-16:
            continue
        k = ki - (n - 1)
        if k == 0:
            grad += w[ki] * (1.0 - 2.0 * h)
        elif k > 0:
            g1 = np.zeros(n)
            g1[k:] = 1.0 - h[:n - k]
            g2 = np.zeros(n)
            g2[:n - k] = -h[k:]
            grad += w[ki] * (g1 + g2)
        else:
            kk = -k
            g1 = np.zeros(n)
            g1[:n - kk] = 1.0 - h[kk:]
            g2 = np.zeros(n)
            g2[kk:] = -h[:n - kk]
            grad += w[ki] * (g1 + g2)
    return grad * dx

def grad_step(h, T, alpha):
    n = len(h)
    g = subgrad_direction(h, T)
    return project(h - alpha * g)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float)
    else:
        bh = np.zeros(160)
        bh[:80] = 1.0

    n = len(bh)
    bh = project(bh)
    bc5 = compute_c5(bh)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP warm-up with smooth obj (8% budget)
    for T in [30, 100, 500, 2000, 10000]:
        if elapsed() > budget_s * 0.08:
            break
        try:
            hc = slsqp_step(bh, T, n, 300, 1e-14)
            cv = compute_c5(hc)
            if cv < bc5:
                bc5, bh = cv, hc
        except Exception:
            pass

    # Phase 2: Coordinate descent (32% budget)
    bh, bc5 = cd_exact(bh, n, t0 + budget_s * 0.40, 4000)

    # Phase 3: Exploration with perturbations (28% budget)
    t3end = t0 + budget_s * 0.68
    for it in range(120):
        if elapsed() > budget_s * 0.68:
            break
        fr = 1.0 - it / 120.0
        strat = it % 4
        if strat == 0:
            sc = 0.02 + 0.10 * fr
            hp = bh + sc * np.random.randn(n)
        elif strat == 1:
            hp = bh.copy()
            nf = np.random.randint(2, 6)
            idx = np.random.choice(n, nf, replace=False)
            hp[idx] = 1.0 - hp[idx]
        elif strat == 2:
            hp = bh.copy()
            ns = np.random.randint(1, 4)
            for _ in range(ns):
                i, j = np.random.choice(n, 2, replace=False)
                t = np.random.rand()
                hp[i] = (1 - t) * hp[i] + t * hp[j]
                hp[j] = (1 - t) * hp[j] + t * hp[i]
        else:
            hp = grad_step(bh, 500 + 5000 * fr, 0.02 * fr + 0.003)

        hp = project(hp)
        cv = compute_c5(hp)

        if cv < bc5 + 0.004:
            Ts = 200 + 4000 * fr
            try:
                hp = slsqp_step(hp, Ts, n, 200, 1e-13)
            except Exception:
                pass
            time_budget = min(2.5, max(0.5, t3end - time.time()))
            hp, cv = cd_exact(hp, n, time.time() + time_budget, 80)

        if cv < bc5:
            bc5, bh = cv, hp

    # Phase 4: Intensive CD (20% budget)
    bh, bc5 = cd_exact(bh, n, t0 + budget_s * 0.88, 5000)

    # Phase 5: Final SLSQP polish (10% budget)
    for T in [8000, 40000, 150000]:
        if elapsed() > budget_s * 0.96:
            break
        try:
            hc = slsqp_step(bh, T, n, 2000, 1e-15)
            cv = compute_c5(hc)
            if cv < bc5:
                bc5, bh = cv, hc
        except Exception:
            pass

    if elapsed() < budget_s * 0.98:
        bh, bc5 = cd_exact(bh, n, t0 + budget_s * 0.99, 100)

    bh = project(bh)
    bc5 = compute_c5(bh)
    return bh, bc5, n
