def project_sum(h, target):

        return h
    lo, hi = -2.0, 2.0

def make_binit(h, N):
    """Convert to binary with exactly N/2 ones."""
    h = (h > 0.5).astype(float)
    n_ones = int(h.sum())
    nh = N // 2
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    return h

    h = h.copy()
    nh = N // 2
    corr = compute_corr(h)
    cur_peak = float(corr.max())
    best = h.copy()
    best_peak = cur_peak
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iters):

            break

For continuous h, the objective is:
C₅ = max_L [sum_i h[i+L]*(1-h[i])] * dx (with appropriate bounds)

This is a quadratic function of h (each term is h[i+L] - h[i+L]*h[i]). It's not convex in general because of the h[i+L]*h[i] terms (which are bilinear).

However, for fixed "which pair values each h[i] takes", it becomes linear. The problem is non-convex.

Approach for continuous:
1. Gradient descent / Adam with the top-k gradient
2. Project onto {0 ≤ h ≤ 1, sum(h) = N/2}
3. The gradient with respect to h[p] for the term at lag L:
   - d/dh[p] of sum_i h[i+L]*(1-h[i]):
     - If p = i+L (i.e., i = p-L): contributes (1-h[p-L]) if valid
     - If p = i: contributes -h[p+L] if valid
   - So gradient at h[p] for lag L: (1-h[p-L])*valid(p-L) - h[p+L]*valid(p+L)

For the max, we use the subgradient: if multiple lags achieve the max, the subgradient is the average of their gradients.

def continuous_optimize(h, N, deadline, top_k=20, lr_start=0.05, lr_end=0.001):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    Ncorr = 2 * N - 1
    center = N - 1
    
    # Adam parameters
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(corr.max())
        c5 = peak * (2.0 / N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient using top-k lags
        top = np.argpartition(corr, -top_k)[-top_k:]
        # Weight by softmax for smoothing
        top_corr = corr[top]
        w = np.exp((top_corr - top_corr.max()) * 5)  # sharpness
        w /= w.sum()
        
        grad = np.zeros(N)
        for idx, ai in enumerate(top):
            L = ai - center
            if L == 0:
                grad += w[idx] * (1.0 - 2.0 * h)
            elif L > 0:
                # c[L] = sum_{i=0}^{N-L-1} h[i+L]*(1-h[i])
                # d/dh[p]: if p >= L: (1-h[p-L]) ; if p < N-L: -h[p+L]
                valid_low = np.arange(L, N)  # p where p-L >= 0
                valid_high = np.arange(0, N-L)  # p where p+L < N
                grad[valid_low] += w[idx] * (1.0 - h[valid_low - L])
                grad[valid_high] -= w[idx] * h[valid_high + L]
            else:
                M = -L  # M > 0
                # c[-M] = sum_{i=M}^{N-1} h[i-M]*(1-h[i]) [by numpy convention]
                # Actually for negative lag in correlate: c[k] = sum_i v[i+k]*a[i]
                # With a=h, v=1-h: c[k] = sum_i (1-h[i+k])*h[i] for k<0
                # Let k = -M: c[-M] = sum_i (1-h[i-M])*h[i] for i>=M
                # = sum_i h[i] - h[i]*h[i-M]
                # d/dh[p]: 
                #   p appears in h[i] when i=p: contributes (1-h[p-M]) if p>=M
                #   p appears in h[i-M] when i-M=p, i=p+M: contributes -h[p+M] if p+M<N
                valid_low = np.arange(M, N)
                valid_high = np.arange(0, N-M)
                grad[valid_low] += w[idx] * (1.0 - h[valid_low - M])
                grad[valid_high] -= w[idx] * h[valid_high + M]
        
        # Adam update
        lr = lr_start * (lr_end / lr_start) ** (it / max(1, int(300)))
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    
    return best_h, best_c5

import numpy as np

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binit(h, N):
    h = (h > 0.5).astype(float)
    n_ones = int(h.sum())
    nh = N // 2
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    return h

def anneal_binary(h, N, n_iters, rng, T0, T1, deadline):

    corr = compute_corr(h)
    cur_peak = float(corr.max())
    best = h.copy()
    best_peak = cur_peak

    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h[i], h[j] = h[j], h[i]
        corr = compute_corr(h)
        new_peak = float(corr.max())
        delta = new_peak - cur_peak
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_peak * (2.0 / N)

def steepest_descent(h, N, max_sweeps, deadline, rng):
    h = h.copy()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        peak = float(corr.max())
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        if not ones or not zeros:
            break
        best_pk = peak
        best_moving = None
        rng.shuffle(zeros)
        for p in ones:
            h[p] = 0.0
            for q in zeros:
                h[q] = 1.0
                npk = float(compute_corr(h).max())
                h[q] = 0.0
                if npk < best_pk:
                    best_pk = npk
                    best_moving = (p, q)
            h[p] = 1.0
        if best_moving is None:
            break
        h[best_moving[0]] = 0.0
        h[best_moving[1]] = 1.0
    return h
