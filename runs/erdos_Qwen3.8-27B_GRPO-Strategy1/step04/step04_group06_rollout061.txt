            best_x = x0

    #   at center-k (i-k>=0): h[i-k] * (1-old)
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save
    
    # a[m] = coefficient of x in corr_new[m] (linear part)
    a = np.zeros(2*n-1)
    ks = np.arange(1, n)
    ip = i + ks
    im = i - ks
    vp = ip < n
    vm = im >= 0
    if vp.any():
        a[center + ks[vp]] = 1.0 - h[ip[vp]]
    if vm.any():
        a[center - ks[vm]] = -h[im[vm]]

    # (since a[center] = 0)
    def f1d(x):
        v = corr_base + x * a
        v[center] += x - x*x
        return v.max()
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 50})
    return float(res.x)

    """Fast coordinate descent with incremental correlation."""
    center = n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    c5 = float(corr.max() * dx)
    best_h = h.copy()
    best_c5 = c5
    ks = np.arange(1, n)
    
    stagnant = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            old = h[i]
            # Compute a vector
            a = np.zeros(2*n-1)
            ip = i + ks
            im = i - ks
            vp = ip < n
            vm = im >= 0
            if vp.any():
                a[center + ks[vp]] = 1.0 - h[ip[vp]]
            if vm.any():
                a[center - ks[vm]] = -h[im[vm]]
            
            # 1D optimization
            cb_center = corr[center] - old * (1.0 - old)
            def f1d(x):
                v = corr + (x - old) * a
                v[center] = cb_center + x * (1.0 - x)
                return float(v.max())
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11, 'maxiter': 60})
            x_new = float(res.x)
            delta = x_new - old
            if abs(delta) > 1e-13:
                corr += delta * a
                corr[center] = cb_center + x_new * (1.0 - x_new)
                h[i] = x_new
                improved = True
        
        # Project and recompute
        h = project_wf(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        c5 = float(corr.max() * dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 50:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_wf(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 0) & (h < 1)
        nf = int(free.sum())
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_1d(x, corr_base, a, center):
    v = corr_base + x * a
    v[center] += x - x * x
