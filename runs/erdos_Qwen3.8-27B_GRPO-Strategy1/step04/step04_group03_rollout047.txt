    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def lp_step(h, n, t_end):
    """LP-based subgradient step using cvxpy"""
    if time.time() > t_end:
        return h
    dx = 2.0 / n
    corr = compute_corr(h)
    # Build gradient matrix: grad[k, j] = d(corr[k])/d(h[j])
    # grad[k, j] = (1-h[j+k])*[0≤j+k<n] - h[j-k]*[0≤j-k<n]
    # Shape: (2n-1, n)
    ks = np.arange(-(n-1), n)[:, None]  # (2n-1, 1)
    js = np.arange(n)[None, :]  # (1, n)
    j1 = js + ks  # (2n-1, n)
    j2 = js - ks
    valid1 = (j1 >= 0) & (j1 < n)

    h_mat = np.tile(h[None, :], (2*n-1, 1))
    grad = np.zeros((2*n-1, n))
