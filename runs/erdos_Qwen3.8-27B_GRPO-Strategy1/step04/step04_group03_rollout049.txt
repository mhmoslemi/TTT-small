                else:

        if time.time() > t4:
            break
        cool = max(1.0 - 0.9 * (it / float(n_restart)), 0.05)
        h_p = fourier_perturb(best_h, n, cool, rng)
        h_new, c5_new = cd_fast(h_p, n, time.time() + 1.5, n_sweeps=8)
        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new
            work_h = best_h.copy()

    # Phase 5: Mixed restarts (10%)
    t5 = t0 + budget_s * 0.90
    for it in range(20):
        if time.time() > t5:
            break
        h_p = best_h + 0.03 * rng.standard_normal(n)
        h_p = project_to_sum(h_p, n / 2)
        h_new, c5_new = cd_fast(h_p, n, time.time() + 1.0, n_sweeps=6)
        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new

    # Phase 6: Final long CD (7%)
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = cd_fast(best_h, n, t0 + budget_s * 0.97, n_sweeps=300)

    # Final SLSQP with very high T
    if time.time() < t0 + budget_s * 0.99:
        for T in [30000, 100000]:
            if time.time() > t0 + budget_s * 0.99:
                break
            try:
                hc = slsqp_smooth(best_h, T, n, 600, 1e-15)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5 = hc, cv
            except Exception:
                pass

    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
