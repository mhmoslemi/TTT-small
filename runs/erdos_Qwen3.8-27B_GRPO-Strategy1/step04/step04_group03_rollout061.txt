```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def coord_descent(h_init, n, t_end, n_sweeps=500, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h_init.copy().astype(float)
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    lags = np.arange(2*n - 1) - (n - 1)
    zero_idx = n - 1  # index of k=0 in full correlate
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n) if sweep % 3 == 0 else range(n)
        for i in order:
            if time.time() > t_end - 0.3:
                return best_h, best_c5
            # Compute a, b for linear part
            v1 = i - lags
            v2 = i + lags
            ok1 = (v1 >= 0) & (v1 < n)
            ok2 = (v2 >= 0) & (v2 < n)
            a = np.zeros(2*n - 1)
            a[ok1] = 1.0 - best_h[v1[ok1]]
            a[ok2] -= best_h[v2[ok2]]
            b = corr - a * best_h[i]
            # Save c0_rest for quadratic correction
            hi = best_h[i]
            c0_rest = corr[zero_idx] - hi * (1.0 - hi)
            
            def f1d(x):
                vals = a * x + b
                vals[zero_idx] = c0_rest + x * (1.0 - x)
                return vals.max()
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12, 'maxiter': 40})
            x_new = np.clip(res.x, 0.0, 1.0)
            if abs(x_new - hi) > 1e-13:
                best_h[i] = x_new
                corr = a * x_new + b
                corr[zero_idx] = c0_rest + x_new * (1.0 - x_new)
                c5_new = corr.max() * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5
```

At the current point, compute:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_to_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)
