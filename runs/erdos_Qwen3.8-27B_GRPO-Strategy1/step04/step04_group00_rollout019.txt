def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_fast(h, i):
    """Optimize single coordinate h[i] given all others fixed."""
    n = len(h)
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    
    # When h[i] changes by delta, corr changes by delta * coefs
    # corr[m] = sum_j h[j] * (1-h[j+m-n+1])
    # Changing h[i] from 0 to t:

def slsqp_smooth(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_coefs(h, i):
    """Compute coefficients for changing h[i]."""
    n = len(h)
    h0 = h.copy()
    h0[i] = 0.0

    j2 = i - k_range

    v2 = (j2 >= 0) & (j2 < n)

    coefs[v2] += -h0[j2[v2]]
    return coefs

    return bh, bc5

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(r.x, 0, 1))
    except Exception:
        return h

def basin_hop(h, c5, t_end, n_iter=200, rng_seed=0):
    rng = np.random.RandomState(rng_seed)
    n = len(h)
    wh, wc = h.copy(), c5
    bh, bc = h.copy(), c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - 0.92 * it / max(n_iter, 1))
        r = rng.randint(6)
        if r == 0:
            p = (0.02 + 0.10 * cool) * rng.randn(n)
        elif r == 1:
            f = rng.randint(2, 25)
            ph = rng.uniform(0, 2 * np.pi)
            p = (0.02 + 0.07 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif r == 2:
            m = rng.random(n) < 0.35
            p = np.zeros(n)
            p[m] = (0.04 + 0.14 * cool) * rng.randn(m.sum())
        elif r == 3:
            bsz = rng.randint(4, 18)
            st = rng.randint(0, max(1, n - bsz))
            p = np.zeros(n)
            p[st:st + bsz] = (0.03 + 0.10 * cool) * rng.randn(bsz)
        elif r == 4:
            # Two-swap perturbation
            p = np.zeros(n)
            for _ in range(2):
                i1 = rng.randint(n)
                i2 = rng.randint(n)
                if i1 != i2:
                    p[i1] += 0.5 * (wh[i2] - wh[i1]) * cool
                    p[i2] += 0.5 * (wh[i1] - wh[i2]) * cool
        else:
            p = (0.01 + 0.03 * cool) * rng.randn(n)
        hp = project_h(wh + p)
        t_loc = min(time.time() + 4.0, t_end)
        hl, cl = coord_desc(hp, t_loc, max_sweeps=60)
        temp = 0.0004 * cool + 0.00001
        if cl < wc or rng.rand() < np.exp(-(cl - wc) / max(temp, 1e-10)):
            wh, wc = hl, cl
            if cl < bc:
                bh, bc = hl, cl
    return bh, bc

def multi_round_cd(h, t_end, rounds=5):
    bh, bc5 = h, compute_c5(h)
    per_time = (t_end - time.time()) / rounds
    for r in range(rounds):
        te = time.time() + per_time
        if time.time() > t_end:
            break
        bh, bc5 = coord_desc(bh, te, max_sweeps=200)
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float)
    else:
        bh = np.zeros(160)
        bh[:80] = 1.0
    n = len(bh)
    bh = project_h(bh)
    bc5 = compute_c5(bh)

    # Phase 1: Quick coordinate descent polish
    bh, bc5 = coord_desc(bh, t0 + budget_s * 0.06, max_sweeps=80)

    # Phase 2: SLSQP with increasing smoothness
    for T in [80, 400, 2000]:
        if time.time() > t0 + budget_s * 0.18:
            break
        hc = slsqp_smooth(bh, T, maxiter=350)
        cv = compute_c5(hc)
        if cv < bc5:
            bh, bc5 = hc, cv
        bh, bc5 = coord_desc(bh, min(time.time() + 5, t0 + budget_s * 0.20), max_sweeps=30)

    # Phase 3: Main basin hopping (exploration)
    t_end3 = t0 + budget_s * 0.55
    bh, bc5 = basin_hop(bh, bc5, t_end3, n_iter=250, rng_seed=seed)

    # Phase 4: Long coordinate descent (exploitation)
    bh, bc5 = multi_round_cd(bh, t0 + budget_s * 0.78, rounds=4)

    # Phase 5: Fine basin hopping
    t_end5 = t0 + budget_s * 0.87
    bh, bc5 = basin_hop(bh, bc5, t_end5, n_iter=80, rng_seed=seed + 999)

    # Phase 6: Final coordinate descent
    bh, bc5 = multi_round_cd(bh, t0 + budget_s * 0.95, rounds=3)

    # Phase 7: SLSQP high-T polish
    if time.time() < t0 + budget_s * 0.95:
        hc = slsqp_smooth(bh, 30000, maxiter=500)
        cv = compute_c5(hc)
        if cv < bc5:
            bh, bc5 = hc, cv

    # Phase 8: Final coordinate descent
    if time.time() < t0 + budget_s * 0.985:
        bh, bc5 = coord_desc(bh, t0 + budget_s * 0.985, max_sweeps=50)

    bh = project_h(bh)
    bc5 = compute_c5(bh)
    return bh, bc5, n
