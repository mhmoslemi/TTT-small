    corr = np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    if abs(s - target) < 1e-12:

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s > target:
        h = h * (target / s)
        h = np.clip(h, 0, 1)
        s2 = h.sum()
        if s2 < target - 1e-10:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += room * ((target - s2) / tot)
                h = np.clip(h, 0, 1)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * ((target - s) / tot)
            h = np.clip(h, 0, 1)
        else:
            h = h * (target / max(s, 1e-14))
            h = np.clip(h, 0, 1)
    return h

def compute_grad_matrix(h, active_lags, n):
    K = len(active_lags)
    A = np.zeros((K, n))
    j = np.arange(n)
    for k in range(K):
        lag = active_lags[k] - (n - 1)
        j1 = j - lag
        j2 = j + lag
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        j1c = np.clip(j1, 0, n - 1)
        j2c = np.clip(j2, 0, n - 1)
        A[k] = (1.0 - h[j1c]) * v1.astype(np.float64) - h[j2c] * v2.astype(np.float64)
    return A

def solve_lp(h_base, corr_base, active_lags, n):
    K = len(active_lags)
    if K == 0:
        return h_base.copy()
    A = compute_grad_matrix(h_base, active_lags, n)
    d = A @ h_base
    rhs = d - corr_base[active_lags]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return h_base.copy()

def lp_optimize(h, n, t_end, max_iter=500):
    h = h.copy()
    corr = compute_all_corr(h)
    c5 = float(np.max(corr) * 2.0 / n)
    n_corr = 2 * n - 1
    for it in range(max_iter):
        if time.time() > t_end:
            break
        K_target = min(5 + it // 3, 40)
        top_all = np.argsort(corr)[-K_target:]
        h_new = solve_lp(h, corr, top_all, n)
        h_new = project_h(h_new)
        corr_new = compute_all_corr(h_new)
        c5_new = float(np.max(corr_new) * 2.0 / n)
        if c5_new < c5 - 1e-13:
            h = h_new
            corr = corr_new
            c5 = c5_new
        else:
            K_sm = min(3 + it // 5, 15)
            top_sm = np.argsort(corr)[-K_sm:]

            h_try = project_h(h_try)
            corr_try = compute_all_corr(h_try)

            if c5_try < c5 - 1e-13:
                h = h_try
                corr = corr_try
                c5 = c5_try
            else:
                break
    return h, c5

def coord_sweep(h, n, t_end, max_sweeps=300):
