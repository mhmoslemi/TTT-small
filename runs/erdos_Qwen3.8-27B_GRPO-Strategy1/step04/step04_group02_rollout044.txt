import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)) + 1e-300)) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-14 and s2 > 1e-14:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def coord_opt(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += (1.0 - h[i - k])
        if 0 <= i + k < n:
            dk += (-h[i + k])
        d[n - 1 + k] = dk
        dk2 = 0.0
        if 0 <= i + k < n:
            dk2 += (1.0 - h[i + k])
        if 0 <= i - k < n:
            dk2 += (-h[i - k])
        d[n - 1 - k] = dk2
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-13:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                break
    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=600):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def block_basin_hop(h, n, t_end, rng, n_iters=150):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.7 * (it / max(n_iters, 1))
        # Block perturbation: flip a random block
        blen = rng.integers(3, max(4, n // 15))
        start = rng.integers(0, n - blen)
        pert = work.copy()
        step = 0.08 * (0.4 + 0.6 * cool)
        pert[start:start + blen] += step * rng.standard_normal(blen)
        # Also gaussian noise on a few random coords
        npts = rng.integers(2, 8)
        idxs = rng.integers(0, n, npts)
        pert[idxs] += step * 0.5 * rng.standard_normal(npts)
        pert = project(pert, n)
        # Local polish via SLSQP
        try:
            hc = slsqp_smooth(pert, 400, n, maxiter=120)
        except Exception:
            hc = pert
        cv = compute_c5(hc)
        temp = 0.004 * cool + 0.0003
        if cv < work_c5 or rng.random() < np.exp(-(cv - work_c5) / max(temp, 1e-12)):
            work, work_c5 = hc, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def multi_start_opt(h, n, t_end, rng, T=800):
    """Try SLSQP from multiple perturbations of h, keep best."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_tries = 8
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        pert = best_h + 0.04 * rng.standard_normal(n)
        pert = project(pert, n)
        try:
            hc = slsqp_smooth(pert, T, n, maxiter=400)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass
    # Final CD on best
    best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=30, rng=rng)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent
    t1 = t0 + budget_s * 0.18
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=150, rng=rng)

    # Phase 2: SLSQP polish
    if elapsed() < budget_s * 0.25:
        for T in [200, 1000, 4000]:
            if elapsed() > t0 + budget_s * 0.25:
                break
            try:
                hc = slsqp_smooth(best_h, T, n, maxiter=800)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5 = hc, cv
            except Exception:
                pass

    # Phase 3: Coord descent + block basin hopping (alternating)
    t3 = t0 + budget_s * 0.55
    round_cd = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, round_cd, max_sweeps=80, rng=rng)
    if elapsed() < t3:
        best_h, best_c5 = block_basin_hop(best_h, n, t3, rng, n_iters=100)

    # Phase 4: Multi-start from current best
    if elapsed() < budget_s * 0.65:
        best_h, best_c5 = multi_start_opt(best_h, n, t0 + budget_s * 0.65, rng, T=1500)

    # Phase 5: More coordination
    t5 = t0 + budget_s * 0.72
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=100, rng=rng)

    # Phase 6: Multi-resolution
    sizes = [160, 180, 200, 220, 240]
    if n not in sizes:
        sizes.append(n)
    for n_try in sizes:
        if elapsed() > budget_s * 0.88:
            break
        h_int = interp_h(best_h, n_try)
        if n_try != n:
            try:
                hc = slsqp_smooth(h_int, 800, n_try, maxiter=500)
                cv = compute_c5(hc)
            except Exception:
                hc, cv = h_int, compute_c5(h_int)
        else:
            hc, cv = h_int, best_c5
        if cv < best_c5 - 1e-7:
            best_h, best_c5, n = hc, cv, n_try
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=40, rng=rng)

    # Phase 7: Final basin hop + coord
    if elapsed() < budget_s * 0.92:
        best_h, best_c5 = block_basin_hop(best_h, n, t0 + budget_s * 0.92, rng, n_iters=60)

    # Phase 8: Final polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=60, rng=rng)
    if elapsed() < budget_s * 0.99:
        try:
            hc = slsqp_smooth(best_h, 10000, n, maxiter=1500)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
