def cd_sweep(h, corr, n, order):
    improved = True
    sweeps_done = 0
    while improved:

        for i in order:

            if abs(t_new - h[i]) > 1e-10:
                corr = update_corr(corr, h, n, i, t_new)
                h[i] = t_new
                improved = True
        sweeps_done += 1

        h, corr = h2, corr2
    return h_best, h, corr, best_c5

def perturb(h, n, rng, mode=0):
    hp = h.copy()
    if mode == 0:
        hp += 0.03 * rng.randn(n)
    elif mode == 1:
        k = rng.randint(10, 30)
        start = rng.randint(0, n - k)
        seg = hp[start:start+k].copy()
        hp[start:start+k] = hp[(start+k):((start+2*k) % n)]
        hp[(start+k):((start+2*k) % n)] = seg
    elif mode == 2:
        m = max(5, n // 10)
        idx = rng.choice(n, m, replace=False)
        hp[idx] = rng.uniform(0, 1, m)
    elif mode == 3:
        hp = 1.0 - hp
    elif mode == 4:
        shift = rng.randint(1, n // 2)
        hp = np.roll(hp, shift)
    return project(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x + 0.5), 0, 1)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    best_corr = get_corr(best_h)

    phase1_end = t0 + budget_s * 0.10
    h, corr, _ = cd_sweep(best_h, best_corr, n, rng.permutation(n))
    best_h = project(h, n)
    best_corr = get_corr(best_h)
    best_c5 = float(np.max(best_corr) * 2.0 / n)

    phase2_end = t0 + budget_s * 0.65
    current_sigma = 0.04
    stall_count = 0
    restart_idx = 0
    h_work = best_h.copy()
    corr_work = best_corr.copy()
    while time.time() < phase2_end:
        order = rng.permutation(n)
        h2, corr2, _ = cd_sweep(h_work, corr_work, n, order)
        h2 = project(h2, n)
        corr2 = get_corr(h2)
        c5v = float(np.max(corr2) * 2.0 / n)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()
            best_corr = corr2.copy()
            stall_count = 0
            current_sigma = max(0.01, current_sigma * 0.9)
        else:
            h_work, corr_work = h2, corr2
            stall_count += 1

        if stall_count >= 8 or restart_idx >= 30:
            mode = restart_idx % 5
            h_work = perturb(best_h, n, rng, mode)
            corr_work = get_corr(h_work)
            h_work, corr_work, _ = cd_sweep(h_work, corr_work, n, rng.permutation(n))
            h_work = project(h_work, n)
            corr_work = get_corr(h_work)
            c5p = float(np.max(corr_work) * 2.0 / n)
            if c5p < best_c5:
                best_c5 = c5p
                best_h = h_work.copy()
                best_corr = corr_work.copy()
            stall_count = 0
            restart_idx += 1
            current_sigma = max(0.01, current_sigma * 0.8)

    phase4_end = t0 + budget_s * 0.82
    for n_try in [128, 200]:
        if time.time() > phase4_end:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n_try)
        corr_interp = get_corr(h_interp)
        order = rng.permutation(n_try)
        h_r, corr_r, _ = cd_sweep(h_interp, corr_interp, n_try, order)
        h_r = project(h_r, n_try)
        corr_r = get_corr(h_r)
        c5r = float(np.max(corr_r) * 2.0 / n_try)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h_r
            best_corr = corr_r
            n = n_try

    for T in [30, 100, 400]:
        if time.time() > t0 + budget_s * 0.92:
            break
        try:
            def smooth_obj(x):
                c = np.correlate(x, 1.0 - x, mode='full')
                m = c.max()
                return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * (2.0 / len(x))
            def con_fun(x):
                return x.sum() - len(x) / 2.0
            res = minimize_scalar(lambda ai: smooth_obj(np.clip(best_h + ai * rng.randn(n), 0, 1)),
                                  bounds=(-0.05, 0.05), method='bounded')
        except Exception:
            pass

    final_end = max(t_end - 5, time.time() + 10)
    h_work = best_h.copy()
    corr_work = best_corr.copy()
    n_best = len(h_work)
    h_f, corr_f, _ = cd_timed(h_work, corr_work, n_best, final_end, seed=seed + 1)
    h_f = project(h_f, n_best)
    corr_f = get_corr(h_f)
    c5_f = float(np.max(corr_f) * 2.0 / n_best)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f
    else:
        best_h = h_work

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
