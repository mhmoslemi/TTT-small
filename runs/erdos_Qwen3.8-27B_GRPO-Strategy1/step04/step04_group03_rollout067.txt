def equalize_step(h, n_top=5, step_size=0.01):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-n_top:]
    # For each top correlation, compute gradient
    grads = []
    for m in top_idx:
        d = m - (n-1)
        g = np.zeros(n)
        for i in range(n):
            a, b = i-d, i+d
            if 0 <= a < n: g[i] += (1.0-h[a])
            if 0 <= b < n: g[i] -= h[b]
        grads.append(g)
    grads = np.array(grads)  # shape (n_top, n)
    # Find direction that reduces all top correlations
    # Minimize max_k (corr_k + alpha * g_k · d)
    # This is: find d to minimize max_k g_k · d
    # Simple approach: d = -mean of normalized gradients
    normed = grads / (np.linalg.norm(grads, axis=1, keepdims=True) + 1e-10)
    d = -normed.mean(axis=0)
    d /= (np.linalg.norm(d) + 1e-10)
    h_new = h + step_size * d
    return project_to_sum(h_new, n/2)
