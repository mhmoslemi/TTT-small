    h = project_sum(h, target)

    best_c5 = compute_c5(h, N)

    Nc = 2 * N - 1
    center = N - 1

    lr = lr0
    
    for it in range(n_iters):
        if time.time() > deadline - 2:
            break

        c5 = compute_c5(h, N)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        
        # Top-k indices
        k = min(topk, Nc)
        top_idx = np.argpartition(corr, -k)[-k:]
        vals = corr[top_idx]
        mval = vals.max()
        w = np.exp(beta * (vals - mval))
        w /= w.sum()
        
        grad = np.zeros(N)
        for idx_k in range(k):
            L = top_idx[idx_k] - center
            wv = w[idx_k]
            if wv < 1e-12:
                continue
            if L == 0:
                grad += wv * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wv * (1.0 - h[:N-L])
                grad[:N-L] -= wv * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += wv * (1.0 - h[M:])
                    grad[M:] -= wv * h[:N-M]
        
        m = 0.9 * m + (1.0 - 0.9) * grad
        h = h - lr * m
        h = project_sum(h, target)
        
        lr *= 0.9995
    
    # Check final
    final_c5 = compute_c5(h, N)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return best_h, best_c5

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    d_lags = idx - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N - 1)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N - 1)], 0.0)
    return -(1.0 - h_pd) * valid_pd.astype(float) + h_pmd * valid_pmd.astype(float)

            break
        corr = compute_corr(h)
        c5 = compute_c5(h, N)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        k = min(topk, Nc)
        top_idx = np.argpartition(corr, -k)[-k:]
        vals = corr[top_idx]
        mval = vals.max()
        w = np.exp(beta * (vals - mval))
        wsum = w.sum()
        if wsum < 1e-30:
            break
        w /= wsum
        grad = np.zeros(N)
        for ik in range(k):
            L = top_idx[ik] - center
            wv = w[ik]
            if wv < 1e-12:
                continue
            if L == 0:
                grad += wv * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wv * (1.0 - h[:N - L])
                grad[:N - L] -= wv * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wv * (1.0 - h[M:])
                    grad[M:] -= wv * h[:N - M]
        m = momentum * m + (1.0 - momentum) * grad
        h = h - lr * m
        h = project_sum(h, target)
        lr *= 0.9997
    final_c5 = compute_c5(h, N)
    if final_c5 < best_c5:
        best_c5, best_h = final_c5, h.copy()
    return best_h, best_c5

def hybrid_refine(h_bin, N, deadline, rng):
    h = fix_binary_count((h_bin > 0.5).astype(float), N)
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    phase_budget = max(5.0, (deadline - time.time()) * 0.15)
    for phase in range(8):
        if time.time() > deadline - 5:
            break
        if phase % 2 == 0:
            t_end = min(deadline - 5, time.time() + phase_budget)
            h = steepest_descent_binary(h, N, max_sweeps=300, deadline=t_end)
        else:
            t_end = min(deadline - 5, time.time() + phase_budget)
            h_cont = h.copy() + rng.randn(N) * 0.04
            h_cont = project_sum(h_cont, N / 2.0)
            h_c, c5_c = continuous_opt(h_cont, N, t_end, beta=25.0, topk=20, lr0=0.02, n_iters=5000, momentum=0.93)
            h = (h_c > 0.5).astype(float)
            h = fix_binary_count(h, N)
        c5_cur = compute_c5(h, N)
        if c5_cur < best_c5 - 1e-10:
            best_c5, best_h = c5_cur, h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N)

    # Phase 1: Binary steepest descent (primary)
    h_bin = fix_binary_count((h0 > 0.5).astype(float), N)
    d1 = min(t0 + 350, deadline - 350)
    h_s = steepest_descent_binary(h_bin, N, max_sweeps=3000, deadline=d1)
    c5_s = compute_c5(h_s, N)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Aggressive continuous optimization (key for sub-integer improvement)
    d2 = deadline - 80
    cfgs = [
        (15.0, 30, 0.03, 0.91, 10000),
        (30.0, 15, 0.018, 0.93, 12000),
        (50.0, 10, 0.012, 0.94, 15000),
    ]
    for idx, (bt, tk, lr, mom, ni) in enumerate(cfgs):
        if time.time() > d2:
            break
        h_init = best_h.copy() + rng.randn(N) * (0.06 / (idx + 1))
        h_init = project_sum(h_init, N / 2.0)
        t_end = min(d2, t0 + 500 + idx * 120)
        h_c, c5_c = continuous_opt(h_init, N, t_end, beta=bt, topk=tk, lr0=lr, n_iters=ni, momentum=mom)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c.copy()

    # Phase 3: Hybrid binary-continuous refinement
    d3 = deadline - 40
    h_mix, c5_mix = hybrid_refine((best_h > 0.5).astype(float), N, d3, rng)
    if c5_mix < best_c5:
        best_c5, best_h = c5_mix, h_mix.copy()

    # Phase 4: Final continuous polish on best
    if time.time() < deadline - 10:
        h_fin = best_h.copy() + rng.randn(N) * 0.02
        h_fin = project_sum(h_fin, N / 2.0)
        h_f, c5_f = continuous_opt(h_fin, N, deadline - 3, beta=60.0, topk=8, lr0=0.008, n_iters=20000, momentum=0.95)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Phase 5: Try nearby N values
    for N_try in [N - 20, N + 20, N - 40, N + 40]:
        if time.time() > deadline - 12 or N_try < 200:
            break
        if N_try % 2 != 0:
            N_try -= 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = (np.interp(x_new, x_old, best_h) > 0.5).astype(float)
        h_interp = fix_binary_count(h_interp, N_try)
        h_st = steepest_descent_binary(h_interp, N_try, max_sweeps=200, deadline=deadline - 8)
        c5_t = compute_c5(h_st, N_try)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_st.copy()
            if time.time() < deadline - 5:
                h_c2, c5_c2 = continuous_opt(h_st + rng.randn(N_try) * 0.03, N_try, deadline - 2,
                                             beta=25, topk=20, lr0=0.02, n_iters=5000, momentum=0.92)
                if c5_c2 < best_c5:
                    best_c5, best_h = c5_c2, h_c2.copy()

    final_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h, len(final_h))
    return final_h, final_c5, len(final_h)
