        else:

    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, N):
    h = (h > 0.5).astype(float)
    nh = N // 2
    no = int(h.sum())
    if no > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif no < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - no]] = 1
    return h

def smooth_gradient(h, N, alpha):
    corr = compute_corr(h)
    mc = float(corr.max())
    ew = np.exp(alpha * (corr - mc))
    ew[ew < 1e-15] = 0.0
    w = ew / ew.sum()
    center = N - 1
    grad = w[center] * (1.0 - 2.0 * h)
    for L in range(1, N):
        wp = w[center + L]
        wn = w[center - L]
        if wp > 1e-12:
            grad[:N - L] += wp * (1.0 - h[L:])
            grad[L:] -= wp * h[:N - L]
        if wn > 1e-12:
            grad[L:] += wn * (1.0 - h[:N - L])
            grad[:N - L] -= wn * h[L:]
    return grad, corr

def adam_cont(h, N, target, deadline, a0=10.0, a1=800.0, lr0=0.006):
    best_h = h.copy()
    best_pk = float(compute_corr(h).max())
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 400.0)
        alpha = a0 * (a1 / a0) ** frac
        grad, corr = smooth_gradient(h, N, alpha)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        lr = lr0 * (0.94 ** (it // 150))
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_pk * (2.0 / N)

def bin_local_search(h, N, deadline, n_tries=30):
    h = binarize(h, N)
    corr = compute_corr(h)
    P = float(corr.max())
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    rounds = 0
    while time.time() < deadline:
        rounds += 1
        max_idx = int(np.argmax(corr))
        L = max_idx - (N - 1)
        cands = []
        if L > 0:
            for i in range(N - L):
                if h[i] == 1 and h[i + L] == 0:
                    cands.append((i, i + L))
        elif L < 0:
            M = -L
            for i in range(N - M):
                if h[i + M] == 1 and h[i] == 0:
                    cands.append((i + M, i))
        if not cands:
            cands = [(ones[np.random.randint(len(ones))], zeros[np.random.randint(len(zeros))])]
        if len(cands) > n_tries:
            cands = [cands[i] for i in np.random.choice(len(cands), n_tries, replace=False)]
        best_swap = None
        best_newP = P
        for p, q in cands:
            h[p], h[q] = 0.0, 1.0
            newP = float(compute_corr(h).max())
            h[p], h[q] = 1.0, 0.0
            if newP < best_newP:
                best_newP = newP
                best_swap = (p, q)
        if best_swap is not None:
            h[best_swap[0]], h[best_swap[1]] = 0.0, 1.0
            corr = compute_corr(h)
            P = best_newP
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            if rounds % 50 == 0:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                newP = float(compute_corr(h).max())
                T = max(0.005, 0.3 * (1.0 - rounds / 50000.0))
                if newP < P or np.random.random() < np.exp(-(newP - P) / T):
                    corr = compute_corr(h)
                    P = newP
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                else:
                    h[p], h[q] = 1.0, 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Continuous Adam from initial
    d1 = min(t0 + 90, deadline - 400)
    hc, c5a = adam_cont(h0.copy(), N0, N0 / 2.0, d1, a0=5.0, a1=400.0, lr0=0.008)
    if c5a < best_c5:
        best_c5, best_h = c5a, hc.copy()

    # Phase 2: Binary local search
    d2 = min(t0 + 250, deadline - 250)
    hb = bin_local_search(binarize(best_h, N0), N0, d2, n_tries=25)
    c5b = compute_c5(hb, N0)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 3: Continuous refinement from binary
    d3 = min(t0 + 380, deadline - 180)
    hc3 = best_h.astype(float) + rng.randn(N0) * 0.06
    hc3 = project_sum(hc3, N0 / 2.0)
    hc3, c5c = adam_cont(hc3, N0, N0 / 2.0, d3, a0=30.0, a1=2000.0, lr0=0.004)
    if c5c < best_c5:
        best_c5, best_h = c5c, hc3.copy()

    # Phase 4: Binary local search again
    d4 = min(t0 + 550, deadline - 120)
    hb4 = bin_local_search(binarize(best_h, N0), N0, d4, n_tries=30)
    c5d = compute_c5(hb4, N0)
    if c5d < best_c5:
        best_c5, best_h = c5d, hb4.copy()

    # Phase 5: Try larger N
    for Nt in [N0 + 40, N0 + 80, N0 + 20]:
        if time.time() > deadline - 80 or Nt < 100:
            break
        if Nt % 2 != 0:
            Nt -= 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = project_sum(hi, Nt / 2.0)
        d_ad = min(time.time() + 60, deadline - 60)
        ha, c5_ad = adam_cont(hi, Nt, Nt / 2.0, d_ad, a0=10.0, a1=500.0, lr0=0.005)
        d_bs = min(time.time() + 60, deadline - 30)
        hbs = bin_local_search(binarize(ha, Nt), Nt, d_bs, n_tries=20)
        c5_bs = compute_c5(hbs, Nt)
        if c5_bs < best_c5:
            best_c5, best_h = c5_bs, hbs.copy()
        elif c5_ad < best_c5:
            best_c5, best_h = c5_ad, ha.copy()

    # Phase 6: Try smaller N
    for Nt in [N0 - 20, N0 - 40]:
        if time.time() > deadline - 60 or Nt < 100:
            break
        if Nt % 2 != 0:
            Nt -= 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = project_sum(hi, Nt / 2.0)
        d_ad = min(time.time() + 40, deadline - 40)
        ha, c5_ad = adam_cont(hi, Nt, Nt / 2.0, d_ad, a0=10.0, a1=500.0, lr0=0.005)
        d_bs = min(time.time() + 30, deadline - 20)
        hbs = bin_local_search(binarize(ha, Nt), Nt, d_bs, n_tries=20)
        c5_bs = compute_c5(hbs, Nt)
        if c5_bs < best_c5:
            best_c5, best_h = c5_bs, hbs.copy()
        elif c5_ad < best_c5:
            best_c5, best_h = c5_ad, ha.copy()

    # Phase 7: Final binary polish
    if time.time() < deadline - 3:
        Nb = len(best_h)
        hf = bin_local_search(binarize(best_h, Nb), Nb, deadline - 1, n_tries=40)
        c5f = compute_c5(hf, Nb)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
