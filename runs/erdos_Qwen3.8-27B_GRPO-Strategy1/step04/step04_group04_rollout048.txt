            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass
    if el() < budget_s * 0.985:
        best_h, best_c5 = swap_anneal(best_h, n, t0 + budget_s * 0.985, n_iter=30000, seed_rng=42)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
