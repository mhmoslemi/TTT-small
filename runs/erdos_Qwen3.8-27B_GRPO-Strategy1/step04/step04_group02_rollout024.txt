def project(h, n):

    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)

        s = max(h.sum(), 1e-14)

        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:

    return h

def coord_descent_fast(h, n, t_end, max_sweeps=50, rng=None, tolerance=1e-13):

                return best, best_c5
            new_val = coord_opt_fast(best, n, i, None)

                    best[i] = new_val  # keep the new value, projection fixed things
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best, best_c5

def make_periodic_pattern(n, period, ones_per_period, rng):
    """Create a periodic binary pattern."""
    h = np.zeros(n)
    for start in range(0, n, period):
        end = min(start + period, n)
        positions = rng.choice(period, size=min(ones_per_period, period), replace=False)
        h[start:start + min(ones_per_period, period)] = 0
        h[start + positions] = 1.0
    return h

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        s = max(h.sum(), 1e-14)
        h = np.clip(h * (target / s), 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-14 and s2 > 1e-14:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    dx = 2.0 / n
    h0 = h.copy()
    h1 = h.copy()
    h0[i] = 0.0
    h1[i] = 1.0
    c0 = full_corr(h0)
    c1 = full_corr(h1)
    delta = c1 - c0
    def f1d(t):
        v = c0 + t * delta
        v[n - 1] = c0[n - 1] + t * (1.0 - t)
        return np.max(v) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-14})
    return res.x

def slsqp_smooth(h, T, n, maxiter=500, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        corr = full_corr(x)
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=100, rng=None, tol=1e-13):
    if rng is None:
        rng = np.random.default_rng(42)
    best = h.copy()
    best_c5 = compute_c5(best)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            nv = coord_opt(best, n, i)
            if abs(nv - best[i]) > tol:
                best[i] = nv
                best = project(best, n)
                c5 = compute_c5(best)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best, best_c5

def binary_swap(h, n, t_end, rng, n_iters=10000):
    best = (h > 0.5).astype(float)
    best = project(best, n)
    best_c5 = compute_c5(best)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        i = rng.integers(n)
        if best[i] > 0.5:
            j = rng.integers(n)
            while j == i or best[j] > 0.5:
                j = rng.integers(n)
            test = best.copy()
            test[i], test[j] = 0.0, 1.0
        else:
            j = rng.integers(n)
            while j == i or best[j] < 0.5:
                j = rng.integers(n)
            test = best.copy()
            test[i], test[j] = 1.0, 0.0
        cv = compute_c5(test)
        if cv < best_c5 - 1e-15:
            best, best_c5 = test, cv
    # Polish the binary result
    mh = np.where(best > 0.5, 1.0, 0.0).astype(float)
    return mh, best_c5

def basin_hop(h, n, t_end, rng, n_iters=150):
    work = (h > 0.5).astype(float)
    work = project(work, n)
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.1, 1.0 - 0.85 * it / max(n_iters, 1))
        n_swaps = max(1, int(2 + 3 * cool))
        pert = work.copy()
        ones = np.where(pert > 0.5)[0]
        zeros = np.where(pert < 0.5)[0]
        for _ in range(n_swaps):
            if len(ones) < 2 or len(zeros) < 2:
                break
            oi = ones[rng.integers(len(ones))]
            zj = zeros[rng.integers(len(zeros))]
            pert[oi], pert[zj] = 0.0, 1.0
            ones = np.where(pert > 0.5)[0]
            zeros = np.where(pert < 0.5)[0]
        # Polish
        hc, hc5 = coord_descent(pert, n, min(time.time() + 3, t_end),
                                max_sweeps=8, rng=rng)
        temp = 0.004 * cool + 0.0002
        if hc5 < work_c5 or rng.random() < np.exp(-(hc5 - work_c5) / temp):
            work, work_c5 = hc, hc5
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def thue_morse_pat(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)
