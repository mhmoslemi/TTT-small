import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff < 0:
        mask = h > 1e-15
        s_pos = h[mask].sum()
        if s_pos > 0:
            h[mask] = np.clip(h[mask] * (1.0 + diff / s_pos), 0, 1)
    else:
        mask = h < 1.0 - 1e-15
        room = 1.0 - h[mask]
        tot = room.sum()
        if tot > 0:
            h[mask] = np.clip(h[mask] + room * (diff / tot), 0, 1)
    return h

def fast_coord_down(h, t_end, n_sweeps=3000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    c5 = best_c5
    k_arr = np.arange(2 * n - 1)

    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return h, best_c5
            j1 = i + k_arr - n + 1
            v1 = (j1 >= 0) & (j1 < n)
            j2 = i - k_arr + n - 1
            v2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] = 1.0 - h[j1[v1]]
            coefs[v2] -= h[j2[v2]]
            old = h[i]
            lo, hi = -old, 1.0 - old
            if hi - lo < 1e-15:
                continue
            a, b = lo, hi
            for _ in range(28):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(corr + m1 * coefs) < np.max(corr + m2 * coefs):
                    b = m2
                else:
                    a = m1
            d_opt = 0.5 * (a + b)
            if abs(d_opt) > 1e-13:
                h[i] = old + d_opt
                corr += d_opt * coefs
                c5 = float(np.max(corr) * dx)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return h, best_c5

def targeted_perturb(h, n, amp, rng):
    corr = np.correlate(h, 1.0 - h, mode='full')
    worst = np.argsort(corr)[-3:]
    k_arr = np.arange(2 * n - 1)
    pert = np.zeros(n)
    for kw in worst:
        shift = kw - (n - 1)
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                pert[i] += amp * (1.0 - h[j])
                if 0 <= i - shift < n:
                    pert[i] -= amp * 0.5
    return pert
