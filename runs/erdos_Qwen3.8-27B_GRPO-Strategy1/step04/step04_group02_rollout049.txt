import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            r = room.sum()
            if r < 1e-15:
                break
            h = h + room * (diff / r)
        else:
            if s < 1e-15:
                break
            h = np.clip(h * (target / s), 0.0, 1.0)
    return h

def smooth_obj_grad(x, T):
    n = len(x)
    corr = get_corr(x)
    m = np.max(corr)
    vals = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(vals)
    s = exps.sum()
    w_full = exps / s
    obj = m + np.log(s) / T
    w = w_full[n - 1:]
    g = 1.0 - 2.0 * x
    grad = w[0] * g.copy()
    for k in range(1, n):
        wk = w[k]
        grad[k:] += wk * g[:n - k]
        grad[:n - k] += wk * g[k:]
    return obj, grad

def slsqp_opt(h, n, T, maxiter=800):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj_grad(x, T)[0]
    def jac(x):
        return smooth_obj_grad(x, T)[1]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def subgrad_descent(h, n, t_end, T_base=500, alpha_init=0.005):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = alpha_init
    stale = 0
    for it in range(5000):
        if time.time() > t_end:
            break
        T = T_base * (1.0 + 0.1 * math.sin(0.01 * it))
        _, grad = smooth_obj_grad(best_h, T)
        step = alpha / (1.0 + 0.05 * it)
        h_new = project(best_h - step * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
            stale = 0
        else:
            stale += 1
            alpha *= 0.97
            if stale > 200:
                break
    return best_h, best_c5

def coord_opt_one(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').astype(float)
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    c0 = corr_base[n - 1]
    for k in range(1, n):
        val = 0.0
        if i - k >= 0:
            val += (1.0 - h[i - k])
        if i + k < n:
            val += (-h[i + k])
        d[n - 1 + k] = val
        d[n - 1 - k] = -val
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, rng, max_sweeps=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_one(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-11:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:

    return best_h, best_c5

def structured_perturb(h, n, rng, scale=0.05):
    op = rng.integers(0, 4)
    p = h.copy()
    if op == 0:
        idx = rng.integers(0, n)
        width = rng.integers(3, max(4, n // 4))
        p[idx:idx + width] += scale * rng.standard_normal(width)
    elif op == 1:
        i = rng.integers(0, n)
        w = rng.integers(2, max(3, n // 5))
        p[i:i + w] = p[i:i + w][::-1]
    elif op == 2:
        p += scale * rng.standard_normal(n)
    else:
        f = rng.uniform(0.5, 4.0)
        phi = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n, endpoint=False)
        p += scale * 0.5 * np.sin(2 * np.pi * f * x + phi)
    return project(p, n)

def basin_hop(h, n, t_end, rng, n_iters=100, base_scale=0.06):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(n_iters, 1))
        sc = base_scale * (0.3 + 0.7 * cool)
        pert = structured_perturb(work, n, rng, sc)
        try:
            T = 300 + 2000 * cool
            hc = slsqp_opt(pert, n, T, maxiter=300)
        except Exception:
            hc = pert
        cv = compute_c5(hc)
        temp = 0.003 * cool + 0.0002
        if cv < work_c5 or rng.random() < math.exp(-(cv - work_c5) / max(temp, 1e-12)):
            work, work_c5 = hc, cv
            if work_c5 < best_c5:
                best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

            pass

    # Phase 2: Subgradient descent (10%)
    if elapsed() < budget_s * 0.22:
        best_h, best_c5 = subgrad_descent(best_h, n, t0 + budget_s * 0.22,
                                          T_base=800, alpha_init=0.004)

    # Phase 3: Coordinate descent (18%)
    if elapsed() < budget_s * 0.40:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.40,
                                        rng, max_sweeps=80)

    # Phase 4: Basin hopping (22%)
    if elapsed() < budget_s * 0.62:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.62,
                                    rng, n_iters=80, base_scale=0.05)

    # Phase 5: Coordinate descent after hopping (10%)
    if elapsed() < budget_s * 0.72:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.72,
                                        rng, max_sweeps=50)

    # Phase 6: Multi-resolution exploration (15%)
    for n_try in [200, 220, 260, 280, 320]:
        if elapsed() > budget_s * 0.87:
            break
        h_interp = interpolate_h(best_h, n_try)
        for T in [500, 3000]:
            if elapsed() > budget_s * 0.88:
                break
            try:
                hc = slsqp_opt(h_interp, n_try, T, maxiter=800)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5, n = hc, cv, n_try
                    h_interp = interpolate_h(best_h, n_try)
            except Exception:
                pass
        if n == n_try:
            if elapsed() < budget_s * 0.90:
                best_h, best_c5 = subgrad_descent(best_h, n, t0 + budget_s * 0.90,
                                                  T_base=1000, alpha_init=0.003)

    # Phase 7: Final polish (12%)
    if elapsed() < budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95,
                                        rng, max_sweeps=40)
    for T in [20000, 80000]:
        if elapsed() > budget_s * 0.99:
            break
        try:
            hc = slsqp_opt(best_h, n, T, maxiter=1500)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
