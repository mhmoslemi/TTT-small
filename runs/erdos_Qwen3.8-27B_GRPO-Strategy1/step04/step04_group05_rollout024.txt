import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lse(h, T, top_k=80):
    """Gradient of log-sum-exp objective (without the 1/T factor)."""
    N = len(h)
    corr = compute_corr(h)
    Tc = T * corr
    Tc -= Tc.max()
    weights = np.exp(Tc)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(N), float(corr.max())
    weights /= wsum
    
    grad = np.zeros(N)
    idx = np.argpartition(weights, -top_k)[-top_k:]
    center = N - 1
    for k in idx:
        w = weights[k]
        if w < 1e-12:

        shift = k - center

            if m > 0:

    return grad, float(corr.max())

def adam_lse(h, N, target, deadline, T_start=2.0, T_end=30.0, 
             n_steps=None, lr0=0.02, decay=0.995, k=80, seed=42):
    """Adam optimizer on log-sum-exp objective with increasing T."""
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    
    while time.time() < deadline:
        T = T_start + (T_end - T_start) * min(1.0, it / max(1, n_steps if n_steps else 1000))
        grad, peak = grad_lse(h, T, top_k=k)
        
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        
        if it % 500 == 0:

def binary_local_search(h, N, deadline, max_sweeps=2000):
    """Binary local search: try all single swaps, keep the best."""
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        # Remove the "worst" ones
        for i in range(n_ones - nh):
            h[idx[-1-i]] = 0
            idx = np.where(h == 1)[0]
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        for i in range(nh - n_ones):
            h[idx[i]] = 1
            idx = np.where(h == 0)[0]
    
    best_peak = float(np.max(compute_corr(h)))
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        D = compute_D(h, N)
        c = compute_corr(h)
        P = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_new_P = P
        best_pq = None
        for p in ones:
            base = c - D[p, :]
            candidates = base[None, :] + D[zeros, :]
            new_peaks = candidates.max(axis=1)
            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_new_P:
                best_new_P = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        best_peak = best_new_P
    
    return h, best_peak * (2.0 / N)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    d_lags = idx - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N - 1)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N - 1)], 0.0)
    A_part = -(1.0 - h_pd) * valid_pd.astype(float)
    B_part = h_pmd * valid_pmd.astype(float)
    return A_part + B_part

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lse(h, T, top_k=80):
    N = len(h)
    corr = compute_corr(h)
    Tc = T * corr
    Tc -= Tc.max()
    weights = np.exp(Tc)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(N), float(corr.max())
    weights /= wsum
    grad = np.zeros(N)
    idx = np.argpartition(weights, -top_k)[-top_k:]
    center = N - 1
    for k in idx:
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - center
        if shift >= 0:
            nv = N - shift
            if nv <= 0:
                continue
            grad[:nv] += w * (1.0 - h[shift:shift+nv])
            if shift > 0:
                grad[shift:shift+nv] -= w * h[:nv]
        else:
            m = -shift
            nv = N - m
            if nv <= 0:
                continue
            grad[m:m+nv] += w * (1.0 - h[:nv])
            if m > 0:
                grad[:nv] -= w * h[m:m+nv]
    return grad, float(corr.max())

    while time.time() < deadline:
        T = T_start + (T_end - T_start) * min(1.0, it / 2000.0)

        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = project_sum(h_new, target)

            v = np.zeros(N)

def compute_D(h, N):
    idx = np.arange(2*N - 1)
    d_lags = idx - (N-1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N-1)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N-1)], 0.0)
    return -(1.0 - h_pd) * valid_pd.astype(float) + h_pmd * valid_pmd.astype(float)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lse(h, T, top_k):
    N = len(h)
    corr = compute_corr(h)
    Tc = T * corr
    Tc -= Tc.max()
    weights = np.exp(Tc)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(N), float(corr.max())
    weights /= wsum
    grad = np.zeros(N)
    idx = np.argpartition(weights, -top_k)[-top_k:]
    center = N - 1
    for k in idx:
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - center
        if shift >= 0:
            nv = N - shift
            if nv <= 0:
                continue
            grad[:nv] += w * (1.0 - h[shift:shift + nv])
            if shift > 0:
                grad[shift:shift + nv] -= w * h[:nv]
        else:
            m = -shift
            nv = N - m
            if nv <= 0:
                continue
            grad[m:m + nv] += w * (1.0 - h[:nv])
            if m > 0:
                grad[:nv] -= w * h[m:m + nv]
    return grad, float(corr.max())

def adam_optimize(h, N, target, deadline, T_start, T_end, lr0, decay, top_k, seed, ramp_iters=500000):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.82, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    stall = 0
    while time.time() < deadline:
        frac = min(1.0, it / ramp_iters)
        T = T_start + (T_end - T_start) * frac
        grad, peak = grad_lse(h, T, top_k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = project_sum(h_new, target)
        h = h_new
        if it % 80000 == 0:
            lr *= decay
        if stall > 40000:
            h = best_h.copy() + rng.randn(N) * (0.05 * (1.0 + 0.01 * (stall / 40000)))
            h = project_sum(h, target)
            m[:] = 0.0
            v[:] = 0.0
            stall = 0
    return best_h, best_peak * dx

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    d_lags = idx - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    vpd = (pd >= 0) & (pd < N)
    vpmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(vpd, h[np.clip(pd, 0, N - 1)], 0.0)
    h_pmd = np.where(vpmd, h[np.clip(p_md, 0, N - 1)], 0.0)
    return -(1.0 - h_pd) * vpd.astype(float) + h_pmd * vpmd.astype(float)

def binary_search(h, N, deadline, max_sweeps):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[-(n_ones - nh):]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:(nh - n_ones)]] = 1
    best_peak = float(np.max(compute_corr(h)))
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        D = compute_D(h, N)
        c = compute_corr(h)
        P = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_new_P = P
        best_pq = None
        for p in ones:
            base = c - D[p, :]
            cands = base[None, :] + D[zeros, :]
            nps = cands.max(axis=1)
            mi = np.argmin(nps)
            if nps[mi] < best_new_P:
                best_new_P = float(nps[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        best_peak = best_new_P
    return h, best_peak * (2.0 / N)

def stoch_cont(h, N, target, deadline, rng, n_trials, step0, cool):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    for t in range(n_trials):
        if time.time() > deadline:
            break
        step = step0 * np.exp(-float(t) / cool)
        npert = rng.randint(2, min(20, N // 10))
        idx = rng.choice(N, size=npert, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.randn(npert) * step
        h_new = project_sum(h_new, target)
        pk = float(np.max(compute_corr(h_new)))
        if pk < best_peak:
            best_peak = pk
            best_h = h_new.copy()
            h = h_new
        elif rng.random() < np.exp(-(pk - best_peak) * N / (3.0 * max(1, cool - t + 1))):
            h = h_new
    return best_h, best_peak * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary local search (quick)
    d1 = min(t0 + 70, deadline - 500)
    h_b, c5_b = binary_search(h0, N0, d1, max_sweeps=800)
    if c5_b < best_c5:
        best_c5, best_h = c5_b, h_b.copy()

    # Phase 2: Main continuous LSE (bulk)
    h_cont = best_h.astype(float) + rng.randn(N0) * 0.06
    h_cont = project_sum(h_cont, target)
    d2 = deadline - 250
    h_opt, c5_opt = adam_optimize(h_cont, N0, target, d2,
                                   T_start=1.0, T_end=100.0, lr0=0.012,
                                   decay=0.99, top_k=120, seed=seed)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 3: Second LSE with higher T (exploitation)
    if time.time() < deadline - 150:
        h2 = best_h.copy() + rng.randn(N0) * 0.03
        h2 = project_sum(h2, target)
        d3 = deadline - 130
        h_opt2, c5_2 = adam_optimize(h2, N0, target, d3,
                                      T_start=15.0, T_end=300.0, lr0=0.006,

        if c5_2 < best_c5:

    if time.time() < deadline - 80:
