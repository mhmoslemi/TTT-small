from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    n = len(h)
    dx = 2.0 / n
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / e.sum()  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    # For s in range(-(n-1), n): index in corr is n-1+s
    # Term 1: a[j+s] for valid j
    for s in range(n):  # s >= 0
        wv = w[n - 1 + s]
        if wv < 1e-16:
            continue
        if s == 0:
            grad += wv * a
        else:
            grad[:n-s] += wv * a[s:]
        # Term 2: -h[j-s] for valid j
        if s == 0:
            grad -= wv * h
        else:
            grad[s:] -= wv * h[:n-s]
    for s in range(1, n):  # s < 0, let k = s > 0, actual shift = -s
        wv = w[n - 1 - s]
        if wv < 1e-16:

        # Term 1: a[j-s] for j in [s, n-1]
        grad[s:] += wv * a[:n-s]
        # Term 2: -h[j+s] for j in [0, n-1-s]
        grad[:n-s] -= wv * h[s:]
    
    return grad * dx

def project(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-10:
        return vc
    def sum_at(shift):
        return np.clip(h + shift, 0.0, 1.0).sum()
    lo, hi = -np.max(h) - 1.0, 1.0 - np.min(h) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if sum_at(mid) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def projected_grad_descent(h, n, T, max_iters, t_end, lr_schedule='decay'):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, T):
    corr = full_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))

        wv = w[n - 1 + s]
        if wv > 1e-15:
            if s == 0:
                grad += wv * (a - h)
            else:
                grad[:n-s] += wv * a[s:]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, T):
    corr = full_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for s in range(n):
        wv = w[n - 1 + s]
        if wv > 1e-15:
            if s == 0:
                grad += wv * (a - h)
            else:
                grad[:n - s] += wv * a[s:]
                grad[s:] -= wv * h[:n - s]
    for s in range(1, n):
        wv = w[n - 1 - s]
        if wv > 1e-15:
            grad[s:] += wv * a[:n - s]
            grad[:n - s] -= wv * h[s:]
    return grad * dx

def project(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-10:
        return vc
    lo, hi = -np.max(h) - 1.0, 1.0 - np.min(h) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=300):
    def f(x):
        return smooth_obj(x, T)
    def g(x):
        return smooth_grad(x, T)
    def c_eq(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': c_eq}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(f, h, jac=g, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(res.x, n)

def pl_min_delta(c, effect, lo, hi):
    if hi - lo < 1e-13:
        return 0.5 * (lo + hi)
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        c_mid = c + mid * effect
        k = np.argmax(c_mid)
        slope = effect[k]
        if abs(slope) < 1e-14:
            break
        if slope > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-12:
            break
    return 0.5 * (lo + hi)

def fast_cd(h, n, t_end, max_sweeps=200000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    c = full_corr(best_h)
    iks = np.arange(-(n - 1), n)
    dx = 2.0 / n
    no_improve = 0
    for sweep in range(max_sweeps):

            effect[m1] += 1.0 - best_h[j1[m1]]
            j2 = i - iks
            m2 = (j2 >= 0) & (j2 < n)
            effect[m2] -= best_h[j2[m2]]
            delta = pl_min_delta(c, effect, -v, 1.0 - v)
            if abs(delta) < 1e-12:
                continue
            best_h[i] += delta
            c += delta * effect
            nc5 = np.max(c) * dx
            if nc5 < best_c5:
                best_c5 = nc5
                improved = True
        if sweep % 10 == 0:

        no_improve = 0 if improved else no_improve + 1
        if no_improve >= 8:
            break
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5
