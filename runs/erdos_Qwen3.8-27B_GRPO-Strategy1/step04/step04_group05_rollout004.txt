```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    # D[p, L] = change in corr[L] when h[p] goes 1->0 (removal pattern)
    center = N - 1
    L = np.arange(2 * N - 1) - center  # shifts from -(N-1) to (N-1)
    # For position p, the corr[L] contribution from h[p] as the "1" is (1-h[p+L])*valid
    # and as the "0" (i.e., in the (1-h) factor) is h[p-L]*valid
    p_idx = np.arange(N)[:, None]
    L_vals = L[None, :]
    pos1 = p_idx + L_vals  # p + L: where the (1-h) factor is
    pos2 = p_idx - L_vals  # p - L: where the h factor is (for the (1-h) part)
    
    valid1 = (pos1 >= 0) & (pos1 < N)
    valid2 = (pos2 >= 0) & (pos2 < N)
    
    h_at_1 = np.where(valid1, h[np.clip(pos1, 0, N-1)], 0.0)
    h_at_2 = np.where(valid2, h[np.clip(pos2, 0, N-1)], 0.0)
    
    # D[p, L] = -(1 - h[p+L])*valid1 + h[p-L]*valid2
    return -(1.0 - h_at_1) * valid1.astype(float) + h_at_2 * valid2.astype(float)

def steepest_descent(h, N, max_sweeps, deadline):
    h = h.astype(float).copy()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        P = float(c.max())
        D = compute_D(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_peak = P
        best_pq = None
        for p in ones:
            base = c - D[p]
            candidates = base + D[zeros]  # (len(zeros), 2N-1)
            new_peaks = candidates.max(axis=1)
            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_peak - 1e-12:
                best_peak = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def anneal_duration(h, N, n_iters, rng, T0, T1, deadline):
    h = h.astype(float).copy()
    h_b = (h > 0.5).astype(float)
    nh = N // 2
    n_ones = int(h_b.sum())
    if n_ones > nh:
        idx = np.where(h_b == 1.0)[0]
        h_b[idx[nh:]] = 0.0
    elif n_ones < nh:
        idx = np.where(h_b == 0.0)[0]
        h_b[idx[:nh - n_ones]] = 1.0
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur_pk = best_pk
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / n_iters)
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = 0.0, 1.0
        new_pk = float(np.max(compute_corr(h_b)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.01 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0, 0.0
    return best, best_pk * (2.0 / N)

def continuous_polish(h, N, deadline, lr=0.01, n_steps=5000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    dx = 2.0 / N
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak * dx < best_c5:
            best_c5 = peak * dx
            best_h = h.copy()
        # Subgradient: weighted average of gradients at active lags
        max_val = corr.max()
        active = np.where(corr > max_val - 1.0)[0]  # near-active lags
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        w = (corr[active] - max_val + 1.0)
        w = np.maximum(w, 0.01)
        w /= w.sum()
        Nm1 = N - 1
        grad = np.zeros(N)
        for idx in active:
            L = idx - Nm1  # shift
            if L == 0:
                grad += w[np.argmax(active == idx)] * (1.0 - 2.0 * h)
            elif L > 0:
                wi = w[int(np.where(active == idx)[0][0])]
                grad[L:] += wi * (1.0 - h[:N-L])
                grad[:N-L] -= wi * h[L:]
            else:
                M = -L
                wi = w[int(np.where(active == idx)[0][0])]
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
        # Projected gradient step
        h = h - lr * grad
        h = project_sum(h, target)
        # Decay learning rate
        lr *= 0.999
    return best_h, best_c5

def make_thue_morse(N):
    # Thue-Morse: t(n) = popcount(n) % 2
    h = np.zeros(N)
    for i in range(N):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0.0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1.0
    return h
