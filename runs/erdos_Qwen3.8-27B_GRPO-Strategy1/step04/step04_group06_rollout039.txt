def fast_coord_opt(h, n, j, corr):
    """Optimize coordinate j given current correlation."""
    dx = 2.0 / n
    v = h[j]
    center = n - 1
    
    # Precompute coefficients a_k for k != 0
    # delta_corr[k] = (t-v) * a_k
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        # k > 0: a_k = (1-h[j+k])*[j+k<n] + h[j-k]*[j-k>=0]
        if j + k < n:

        if j - k >= 0:
            a[center + k] += h[j-k]
        # k < 0: a_{-k} = (1-h[j-k])*[j>=k] + h[j+k]*[j+k<n]

            a[center - k] += (1.0 - h[j-k])
        if j + k < n:
            a[center - k] += h[j+k]

    base = corr.copy()
    base[center] -= v - v*v  # remove current contribution so we can add new
    
    def f(t):
        vals = base + t * a
        vals[center] += t - t*t
        return np.max(vals) * dx
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})

def steepest_coord_descent(h, n, dx, t_end, max_rounds=100000):
    corr = get_corr(h)
    best_c5 = np.max(corr) * dx
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        # Find the best coordinate to update
        best_j = -1
        best_new_c5 = best_c5
        best_new_val = 0
        for j in range(n):

                break
            new_val = fast_coord_opt(h, n, j, corr, dx)

            v = h[j]
            delta = new_val - v
            center = n - 1
            new_corr = corr.copy()
            for k in range(1, n):
                ak = 0.0
                if j + k < n:
                    ak += 1.0 - h[j+k]
                if j - k >= 0:
                    ak -= h[j-k]
                new_corr[center+k] += delta * ak
                ak2 = 0.0
                if j - k >= 0:

            corr = get_corr(h)

When h[j] changes from v to v', the change in corr is:
corr_new[k] = corr_old[k] + (v'-v) * a_k for all k

So I just add (v'-v) * a to corr. That's O(n) instead of O(n²)!

            v = h[j]
            if abs(new_val - v) < 1e-10:
                continue
            # Update corr incrementally
            delta = new_val - v
            h[j] = new_val
            center = n - 1
            for k in range(1, n):
                ak = 0.0
                if j + k < n:
                    ak += 1.0 - h[j+k]  # Note: h[j] is already updated!
                if j - k >= 0:
                    ak -= h[j-k]

                ak2 = 0.0
                if j - k >= 0:
                    ak2 += 1.0 - h[j-k]
                if j + k < n:
                    ak2 -= h[j+k]
                corr[center-k] += delta * ak2
            corr[center] += new_val*(1-new_val) - v*(1-v)
            c5_new = np.max(corr) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
            improved = True
        if not improved:
            break
        # Project to maintain sum constraint
        h = project(h, n)
        corr = get_corr(h)  # Recompute after projection
        best_c5 = np.max(corr) * dx
    return h, best_c5

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s > target + 1e-12:
        return h * (target / s)
    elif s < target - 1e-12:
        denom = n - s
        if denom < 1e-12:
            return h
        return h + (1.0 - h) * ((target - s) / denom)
    return h

def fast_coord_opt(h, n, j, corr, dx):
    v = h[j]
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ak = 0.0
        if j + k < n:
            ak += 1.0 - h[j + k]
        if j - k >= 0:
            ak -= h[j - k]
        a[center + k] = ak
        ak2 = 0.0
        if j - k >= 0:
            ak2 += 1.0 - h[j - k]
        if j + k < n:
            ak2 -= h[j + k]
        a[center - k] = ak2
    c0 = corr[center] - v * (1.0 - v)
    def f(t):
        delta = t - v
        vals = corr + delta * a
        vals[center] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, dx, t_end, max_sweeps=50000):
    corr = get_corr(h)
    best_c5 = np.max(corr) * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            if time.time() > t_end:
                return project(h, n), best_c5
            new_val = fast_coord_opt(h, n, j, corr, dx)
            v = h[j]
            if abs(new_val - v) < 1e-10:
                continue
            delta = new_val - v
            center = n - 1
            for k in range(1, n):
                ak = 0.0
                if j + k < n:
                    ak += 1.0 - h[j+k]
                if j - k >= 0:
                    ak -= h[j-k]
                corr[center+k] += delta * ak
                ak2 = 0.0
                if j - k >= 0:
                    ak2 += 1.0 - h[j-k]
                if j + k < n:
                    ak2 -= h[j+k]
                corr[center-k] += delta * ak2
            corr[center] += new_val*(1-new_val) - v*(1-v)
            h[j] = new_val
            c5_new = np.max(corr) * dx
            best_c5 = min(best_c5, c5_new)
            improved = True
        if not improved:
            break
        h = project(h, n)
        corr = get_corr(h)
    best_c5 = np.max(corr) * dx
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    n = len(best_h)
    best_h = project(best_h, n)
