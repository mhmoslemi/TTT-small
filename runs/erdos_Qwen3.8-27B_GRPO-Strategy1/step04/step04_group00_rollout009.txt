from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

    if diff > 0:

        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_step(h, n, i):
    """Optimize h[i] assuming all other coordinates fixed."""
    h_i = h[i]
    offset = np.arange(-(n-1), n)  # lag offsets
    # c_k = (1-h[i+offset])*I[0<=i+offset<n] - h[i-offset]*I[0<=i-offset<n]
    j1 = i + offset  # i + offset_k
    j2 = i - offset  # i - offset_k
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    h_j1 = np.where(v1, h[np.clip(j1, 0, n-1)], 0.0)
    h_j2 = np.where(v2, h[np.clip(j2, 0, n-1)], 0.0)
    c = (1.0 - h_j1) * v1.astype(float) - h_j2 * v2.astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')

    # Minimize max_k (A[k] + x*c[k]) * dx over x in [0,1]
    # This is: minimize max_k (A[k] + x*c[k])
    def obj(x):
        return np.max(A + x * c)
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200, random_order=True):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
