def smooth_max(val, T):
    m = val.max()
    if T < 1:
        T = 1.0
    s = T * (val - m)
    s = np.clip(s, -500, 500)
    return m + np.logaddexp.reduce(s) / T

def project(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:

def slsqp_smooth(h, T, n, maxiter=500):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, T) * (2.0 / n)
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        w = np.exp(np.clip(T * (corr - m), -500, 500))
        w /= w.sum()
        center = n - 1
        grad = np.zeros(n)
        for c in range(2 * n - 1):
            wi = w[c]
            if wi < 1e-12:
                continue
            k = c - center
            if k == 0:
                grad += wi * (1.0 - 2.0 * x)
            elif k > 0:
                grad[:n-k] += wi * (1.0 - x[k:])
                grad[k:] -= wi * x[:n-k]
            else:
                m2 = -k
                grad[m2:] += wi * (1.0 - x[:n-m2])
                grad[:n-m2] -= wi * x[m2:]
        return grad * (2.0 / n)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_grad(x):
        return np.ones(n)

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

grad[i] = sum_{k=1}^{min(i,n-1)} w_pos[k]*(1-h[i-k]) - sum_{k=1}^{min(n-1-i,n-1)} w_pos[k]*h[i+k]
         - sum_{k=1}^{min(i,n-1)} w_neg[k]*h[i-k] + sum_{k=1}^{min(n-1-i,n-1)} w_neg[k]*(1-h[i+k])
         + w_zero*(1-2*h[i])

This looks like convolutions! 
- sum_{k=1}^{i} w_pos[k]*(1-h[i-k]) = convolution of w_pos with (1-h), evaluated at i
- sum_{k=1}^{n-1-i} w_pos[k]*h[i+k] = cross-correlation of w_pos with h, evaluated at i

I can compute all these using FFT-based convolution! This would make the gradient O(n log n) instead of O(n²).

def compute_grad_smooth(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(np.clip(T*(corr - m), -500, 500))
    w /= w.sum()
    center = n - 1
    dx = 2.0 / n
    
    # Use FFT for efficient computation
    # grad[i] = sum_k w[k] * d(corr_k)/d(h[i])
    # d(corr_{center+k})/d(h[i]) = (1-h[i-k])*[k<=i] - h[i+k]*[i+k<n] for k>0
    # (cross-correlation structure)
    
    # Let's use numpy convolve/fft
    N = 2 * n  # FFT size
    wf = np.fft.rfft(w, N)
    hf = np.fft.rfft(h, N)
    # Actually, let me think more carefully...
    
    # Alternative: just do the O(n²) computation but vectorized
    # grad = sum over k of w[k] * (influence of lag k at each position)
    # For lag k>0: influence[i] = (1-h[i-k])*[i>=k] - h[i+k]*[i+k<n]
    # This is: (1-h) shifted right by k (masked) minus h shifted left by k (masked)
    
    grad = np.zeros(n)
    # k=0
    grad += w[center] * (1.0 - 2.0 * h)
    # k>0
    for k in range(1, n):
        wk = w[center + k]
        if wk > 1e-14:
            grad[:n-k] += wk * (1.0 - h[k:])  # (1-h[i+k]) for i < n-k
            grad[k:] -= wk * h[:n-k]           # -h[i-k] for i >= k
        wk = w[center - k]
        if wk > 1e-14:
            grad[k:] += wk * (1.0 - h[:n-k])   # (1-h[i-k]) for i >= k... 
            grad[:n-k] -= wk * h[k:]           # -h[i+k] for i < n-k
    return grad * dx

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(val, T):
    m = val.max()
    if T < 1:
        T = 1.0
    s = np.clip(T * (val - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / T

def project(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h - h * (-diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            old = best_h[i]
            infl = np.zeros(2 * n - 1)
            for k in range(1, n):
                vp = 0.0
                if i >= k:
                    vp += 1.0 - best_h[i - k]
                if i + k < n:
                    vp -= best_h[i + k]
                infl[center + k] = vp
                vm = vp * 0.0
                if i + k < n:
                    vm += 1.0 - best_h[i + k]
                if i >= k:
                    vm -= best_h[i - k]
                infl[center - k] = vm
            infl[center] = 1.0 - 2.0 * old
            c0 = corr[center] - old * (1.0 - old)
            res = minimize_scalar(
                lambda x: np.max(corr + (x - old) * infl - (x - old) ** 2 * np.eye(1)[0,0] * 0 + (corr[center] - c0 - (x - old) * infl[center]) * 0 + np.where(np.arange(2*n-1)==center, c0 + x*(1-x) - corr[center] - (x-old)*infl[center], 0)),
                bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10}
            )
            new_x = res.x
            delta = new_x - old
            if abs(delta) > 1e-12:
                corr += delta * infl
                corr[center] = c0 + new_x * (1.0 - new_x)
                best_h[i] = new_x
                nc = np.max(corr) * dx
                if nc < best_c5 - 1e-14:
                    best_c5 = nc
                    improved = True
        if not improved:
            break
    corr_f = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr_f) * dx
    return best_h, best_c5

def cd_f1d(corr, infl, c0, center, old, x):
    d = x - old
    v = corr + d * infl
    v[center] = c0 + x * (1.0 - x)
    return np.max(v)

def fast_coord_descent2(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    infl_buf = np.zeros(2 * n - 1)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            old = best_h[i]
            infl_buf.fill(0.0)
            for k in range(1, n):
                vp = 0.0
                if i >= k:
                    vp += 1.0 - best_h[i - k]
                if i + k < n:
                    vp -= best_h[i + k]
                infl_buf[center + k] = vp
                vm = 0.0
                if i + k < n:
                    vm += 1.0 - best_h[i + k]
                if i >= k:
                    vm -= best_h[i - k]
                infl_buf[center - k] = vm
            infl_buf[center] = 1.0 - 2.0 * old
            c0 = corr[center] - old * (1.0 - old)
            res = minimize_scalar(
                lambda x: cd_f1d(corr, infl_buf, c0, center, old, x),
                bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10}
            )
            new_x = res.x
            delta = new_x - old
            if abs(delta) > 1e-12:
                corr += delta * infl_buf
                corr[center] = c0 + new_x * (1.0 - new_x)
                best_h[i] = new_x
                nc = np.max(corr) * dx
                if nc < best_c5 - 1e-14:
                    best_c5 = nc
                    improved = True
        if not improved:
            break
    corr_f = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr_f) * dx
    return best_h, best_c5

def grad_smooth(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(np.clip(T * (corr - m), -500, 500))
    w /= w.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    grad += w[center] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = w[center + k]
        if wp > 1e-14:
            grad[:n-k] += wp * (1.0 - h[k:])
            grad[k:] -= wp * h[:n-k]
        wm = w[center - k]
        if wm > 1e-14:
            grad[k:] += wm * (1.0 - h[:n-k])
            grad[:n-k] -= wm * h[k:]
    return grad * dx

def obj_smooth(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, T) * (2.0 / n)

def obj_grad_smooth(x, n, T):
    return grad_smooth(x, n, T)

def con_fun(x, n):
    return x.sum() - n / 2.0

def con_grad(x, n):
    return np.ones(len(x))

def slsqp_run(h, T, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: con_fun(x, n), 'jac': lambda x: con_grad(x, n)}]
    res = minimize(obj_smooth, h, args=(n, T), jac=obj_grad_smooth, method='SLSQP',
                   bounds=bnds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def subgrad_explore(h, n, t_end, steps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    center = n - 1
    vel = np.zeros(n)
    mu = 0.85
    lr = 0.4
    K = 15
    for step in range(steps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        idx = np.argpartition(corr, -K)[-K:]
        grad = np.zeros(n)
        for c in idx:
            k = c - center
            if k == 0:
                grad += (1.0 - 2.0 * best_h)
            elif k > 0:
                grad[:n-k] += (1.0 - best_h[k:])
                grad[k:] -= best_h[:n-k]
            else:
                m2 = -k
                grad[m2:] += (1.0 - best_h[:n-m2])
                grad[:n-m2] -= best_h[m2:]
        grad /= K
        vel = mu * vel - lr * grad
        h_new = project(best_h + vel, n)
        c5n = compute_c5(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new
            lr = min(lr * 1.03, 2.0)
        else:
            lr *= 0.88
            vel *= 0.5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x0 = np.linspace(0, 2, n0, endpoint=False)

    n = len(best_h)

    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient exploration
    p1_end = t0 + budget_s * 0.10

    # Phase 2: SLSQP with increasing T + CD polish
    p2_end = t0 + budget_s * 0.35
    for T in [50, 150, 400, 1000, 3000]:
        if time.time() > p2_end:
            break
        try:
            hc = slsqp_run(best_h, T, n, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
        te = min(time.time() + 8, p2_end)
        best_h, best_c5 = fast_coord_descent2(best_h, n, te, max_sweeps=80)

    # Phase 3: Long coordinate descent
    p3_end = t0 + budget_s * 0.70
    best_h, best_c5 = fast_coord_descent2(best_h, n, p3_end, max_sweeps=2000)

    # Phase 4: Multi-resolution
    best_h_res = best_h.copy()
    best_c5_res = best_c5

        x_old = np.linspace(0, 2, n, endpoint=False)

        h_int = np.interp(x_new, x_old, best_h_res, period=2.0)

                    best_h_res = hc
                    best_n_res = n_try
            except Exception:

        te = min(time.time() + 5, t0 + budget_s * 0.82)
        h2, c5v2 = fast_coord_descent2(best_h_res, best_n_res, te, max_sweeps=50)
        if c5v2 < best_c5_res:
            best_c5_res = c5v2
            best_h_res = h2
            best_n_res = best_n_res
    best_h = best_h_res
    best_c5 = best_c5_res
    n = best_n_res

    # Phase 5: Final polish

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
