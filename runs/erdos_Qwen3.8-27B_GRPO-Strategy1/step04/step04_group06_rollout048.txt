```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def softmax_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def softmax_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(T * (corr - m))
    weights /= weights.sum()
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        if weights[idx] < 1e-15:
            continue
        k = idx - center
        if k == 0:
            grad += weights[idx] * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += weights[idx] * (1.0 - h[k:])
            grad[k:] -= weights[idx] * h[:n - k]
        else:
            mk = -k
            grad[mk:] += weights[idx] * (1.0 - h[:n - mk])
            grad[:n - mk] -= weights[idx] * h[mk:]
    return grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h * (target / (s + 1e-14)), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        a[center + k] += (1.0 - h[ip]) if ip < n else 0.0
        a[center + k] -= h[im] if im >= 0 else 0.0
        a[center - k] += (1.0 - h[im]) if im >= 0 else 0.0
        a[center - k] -= h[ip] if ip < n else 0.0
    dx = 2.0 / n
    ct = corr_base[center]
    # f(x) = max(corr_base + x*a, with center = ct + x - x^2) * dx
    # The center term is quadratic: ct + x - x^2
    # Other terms are linear: corr_base[j] + x * a[j]
    # Find min of max of these piecewise linear/quadratic functions on [0,1]
    def f1d(x):
        vals = corr_base + x * a
        vals[center] = ct + x - x * x
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - center
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
        else:
            dk = -k
            grad[dk:] += wi * (1.0 - h[:n - dk])
            grad[:n - dk] -= wi * h[dk:]
    return grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h * max(target / (s + 1e-15), 0), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        a[center + k] += (1.0 - h[ip]) if ip < n else 0.0
        a[center + k] -= h[im] if im >= 0 else 0.0
        a[center - k] += (1.0 - h[im]) if im >= 0 else 0.0
        a[center - k] -= h[ip] if ip < n else 0.0
    dx = 2.0 / n
    ct = corr_base[center]
    def f1d(x):
        vals = corr_base + x * a
        vals[center] = ct + x - x * x
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def lbfgs_smooth(h, T, n, t_end, P=50.0):
    def obj(x):
        s = x.sum() - n / 2.0
        return softmax_obj(x, T) + P * s * s
    def grad(x):
        s = x.sum() - n / 2.0
        return softmax_grad(x, T) + 2.0 * P * s
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': 5000})
    return project(res.x, n)

def pair_opt(h, n, i, j, t_end):
    ci = coord_opt(h, n, i)
    cj = coord_opt(h, n, j)
    h2 = h.copy()
    h2[i] = ci
    h2[j] = h[j]
    h2 = project(h2, n)
    c1 = compute_c5(h2)
    h2[j] = cj
    h2 = project(h2, n)
    c2 = compute_c5(h2)
    if c1 <= c2:
        return h2, c1
    return h2, c2
```
