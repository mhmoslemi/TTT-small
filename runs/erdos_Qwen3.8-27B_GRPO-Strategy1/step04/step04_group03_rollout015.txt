```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_desc(h, t_end, n_sweeps=1000, anneal=True, T0=0.01):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy().astype(float)
    best_h = project_to_sum(best_h, target)
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    A = 1.0 - best_h
    B = best_h
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        T = T0 * (0.95 ** sweep) if anneal else 0.0
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            if time.time() > t_end:
                return best_h, best_c5
            # Compute coefficient vector
            j_min = max(0, i - n + 1)
            js = np.arange(j_min, n)
            coef = np.zeros(2 * n - 1)
            idx1 = n - 1 + i - js
            coef[idx1] += A[js]
            idx2 = n - 1 + js - i
            coef[idx2] += -B[js]
            
            h_old = best_h[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            
            def f1d(d):
                return np.max(corr + d * coef)
            
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10})
            d_opt = res.x
            c5_opt = f1d(d_opt)
            c5_old = np.max(corr)
            
            # Acceptance
            if T > 0 and c5_opt > c5_old:
                if np.random.rand() > np.exp(-(c5_opt - c5_old) / T):
                    continue
            elif c5_opt >= c5_old - 1e-15:
                # No improvement (or very tiny), skip
                if T == 0 and c5_opt > c5_old:
                    continue
            
            if abs(d_opt) > 1e-13:
                best_h[i] = h_old + d_opt
                best_h[i] = np.clip(best_h[i], 0, 1)
                delta_applied = best_h[i] - h_old  # may differ from d_opt due to clip
                corr += delta_applied * coef
                A[i] = 1.0 - best_h[i]
                B[i] = best_h[i]
                c5_new = np.max(corr)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        
        # Project to sum constraint every sweep
        best_h = project_to_sum(best_h, target)
        corr = compute_corr(best_h)
        A = 1.0 - best_h
        B = best_h
        c5_check = np.max(corr) * dx
        if c5_check < best_c5:
            best_c5 = c5_check
        
        if not improved and sweep > 3:
            break
    
    return best_h, compute_c5(best_h)
```

One more optimization: instead of using minimize_scalar (which does ~20 function evaluations), I can analytically find the minimizer of max(corr + d*coef) in 1D.

The function g(d) = max_k (corr[k] + d * coef[k]) is a convex piecewise-linear function of d. Its minimum occurs at a point where the subgradient contains 0, or at a kink point (where two or more k's are simultaneously maximal).

To find the minimum:
- g(d) is convex in d
- The minimum is where g'(d⁻) ≤ 0 ≤ g'(d⁺) (subgradient contains 0)
- For a piecewise linear convex function, this means:
  - Find all kink points (where the argmax changes)
  - The minimum is at the kink where the slope changes sign

The kink points are where corr[k1] + d*coef[k1] = corr[k2] + d*coef[k2] for some k1, k2.
d = (corr[k2] - corr[k1]) / (coef[k1] - coef[k2])

There are O((2n-1)²) such points, which is too many to enumerate for n=224 (that's ~250K).

Alternative: use binary search. g(d) is convex, so we can binary search for where the derivative crosses 0. The derivative at a point d (between kinks) is coef[argmax_k(corr + d*coef)]. We want to find d such that this "derivative" changes from negative to positive.

Actually, I just realized: I can speed up the coordinate descent significantly. Instead of calling minimize_scalar (which has Python overhead), I can implement the 1D minimization myself using binary search on the derivative.

For g(d) = max_k (a_k + d*b_k) where a_k = corr[k], b_k = coef[k]:
- g is convex piecewise linear in d
- The minimum d* satisfies: there exist k1, k2 such that a_k1 + d*b_k1 = a_k2 + d*b_k2, b_k1 < 0 < b_k2 (or similar)
- Equivalently: at d*, the set of "active" k's (those achieving max) has min(b_k) ≤ 0 ≤ max(b_k)

Algorithm:
1. Start with d = 0 (or bound)
2. Compute active set: k* = argmax(a_k + d*b_k)
3. If b_k* < 0, decrease d; if b_k* > 0, increase d
4. Binary search until the active set has both positive and negative b's

This is O(log(range/eps)) iterations, each O(n). Much faster than minimize_scalar.

Actually, even simpler: since g is convex, I can use the fact that:
- g'(d) = b[argmax(a + d*b)] (at non-kink points)
- At the minimum, 0 ∈ [b_min_active, b_max_active] where active = {k: a_k + d*b_k = g(d)}

Let me implement a simple bisection:

```python
def min_1d_max_linears(a, b, lo, hi):
    """Minimize max_k(a[k] + d*b[k]) over d in [lo, hi]."""
    for _ in range(60):  # bisection iterations
        mid = (lo + hi) / 2
        vals = a + mid * b
        k = np.argmax(vals)
        if b[k] < 0:
            hi = mid  # slope negative, go left
        elif b[k] > 0:
            lo = mid  # slope positive, go right
        else:
            break
    return (lo + hi) / 2
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def min_1d_max_linears(a, b, lo, hi):
    """Minimize max_k(a[k] + d*b[k]) for d in [lo,hi] via bisection."""
    for _ in range(50):
        mid = (lo + hi) * 0.5
        idx = np.argmax(a + mid * b)
        bv = b[idx]
        if bv < -1e-14:
            hi = mid
        elif bv > 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_desc(h, t_end, n_sweeps=500, T0=0.0, cool=0.97):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = project_to_sum(h.copy().astype(float), target)
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    A = 1.0 - best_h
    B = best_h
    stagnation = 0
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        T = T0 * (cool ** sweep)
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            if time.time() > t_end:
                break
            j_min = max(0, i - n + 1)
            js = np.arange(j_min, n)
            coef = np.zeros(2 * n - 1)
            coef[n - 1 + i - js] += A[js]
            coef[n - 1 + js - i] += -B[js]
            h_old = best_h[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            d_opt = min_1d_max_linears(corr, coef, lo, hi)
            d_opt = np.clip(d_opt, lo, hi)
            c5_new = np.max(corr + d_opt * coef)
            c5_cur = np.max(corr)
            if c5_new > c5_cur + 1e-15:
                if T > 1e-12:
                    if np.random.rand() > np.exp(-(c5_new - c5_cur) / T):
                        continue
                else:
                    continue
            if abs(d_opt) > 1e-13:
                best_h[i] = np.clip(h_old + d_opt, 0.0, 1.0)
                da = best_h[i] - h_old
                corr += da * coef
                A[i] = 1.0 - best_h[i]
                B[i] = best_h[i]
                if np.max(corr) * dx < best_c5:
                    best_c5 = np.max(corr) * dx
                    improved = True
        best_h = project_to_sum(best_h, target)
        corr = compute_corr(best_h)
        A = 1.0 - best_h
        B = best_h
        c5v = np.max(corr) * dx
        if c5v < best_c5:
            best_c5 = c5v
        if not improved:
            stagnation += 1
            if stagnation > 4:
                break
        else:
            stagnation = 0
    return best_h, compute_c5(best_h)

def structured_init(n, seed_val):
    np.random.seed(seed_val)
    h = np.zeros(n)
    r = np.random.rand()
    if r < 0.3:
        h[:n // 2] = 1.0
    elif r < 0.6:
        h[0:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif r < 0.8:
        h[np.arange(0, n, 2)] = 1.0
    else:
        k = n // 4
        h[np.random.choice(n, k, replace=False)] = 1.0
    return project_to_sum(h, n / 2.0)

def perturb(h, mode, rng):
    n = len(h)
    r = rng.random()
    if mode == 'gauss':
        sig = 0.03 + 0.08 * rng.random()
        hp = h + sig * rng.standard_normal(n)
    elif mode == 'subset':
        mask = rng.random(n) < 0.3
        hp = h.copy()
        hp[mask] += 0.05 * rng.standard_normal(mask.sum())
    elif mode == 'shift':
        s = int(rng.uniform(-0.04, 0.04) * n)
        hp = 0.7 * np.roll(h, s) + 0.3 * h
    else:
        bp = max(4, n // 20)
        b1 = rng.randint(0, n - 2 * bp)
        b2 = (b1 + bp + rng.randint(0, n - 2 * bp)) % max(1, n - bp)
        hp = h.copy()
        hp[b1:b1 + bp], hp[b2:b2 + bp] = hp[b2:b2 + bp].copy(), hp[b1:b1 + bp].copy()
    return project_to_sum(hp, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 1)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = structured_init(128, seed)
    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Quick polish
    best_h, best_c5 = coord_desc(best_h, t0 + 15, n_sweeps=40)

    # Phase 1: Annealed coordinate descent (30% budget)
    t_p1 = t0 + budget_s * 0.30
    best_h, best_c5 = coord_desc(best_h, t_p1, n_sweeps=800, T0=0.008, cool=0.985)

    # Phase 2: Multi-restart basin hopping (35% budget)
    t_p2 = t0 + budget_s * 0.65
    work_h = best_h.copy()
    work_c5 = best_c5
    n_restarts = 40
    for it in range(n_restarts):
        if elapsed() > budget_s * 0.65:
            break
        cool = max(1.0 - 0.9 * (it / float(n_restarts)), 0.1)
        mode = ['gauss', 'subset', 'shift', 'block'][int(rng.random() * 4)]
        hp = perturb(work_h, mode, rng)
        # Quick polish
        hc, c5v = coord_desc(hp, min(time.time() + 3.0, t_p2), n_sweeps=60,
                             T0=0.004 * cool)
        # Anneal if promising
        if c5v < work_c5 + 0.003:
            hc, c5v2 = coord_desc(hc, min(time.time() + 4.0, t_p2), n_sweeps=120,
                                  T0=0.003 * cool)
            c5v = min(c5v, c5v2)
            hc = project_to_sum(hc, n / 2.0)
            c5v = compute_c5(hc)
        if c5v < work_c5:
            work_h, work_c5 = hc.copy(), c5v
            if c5v < best_c5:
                best_h, best_c5 = hc.copy(), c5v

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 3: Try different resolutions (10% budget)
    t_p3 = t0 + budget_s * 0.78
    for n_try in [160, 192, 256]:
        if time.time() > t_p3:
            break
        x_old = np.linspace(0, 2, best_h.size, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2.0)
        hc, c5v = coord_desc(h_i, min(time.time() + 10, t_p3), n_sweeps=200, T0=0.005)
        if c5v < best_c5:
            best_h, best_c5, n = hc, c5v, n_try

    # Phase 4: Final intensive coordinate descent (15% budget)
    t_p4 = t0 + budget_s * 0.95
    best_h = project_to_sum(best_h, best_h.size / 2.0)
    best_h, best_c5 = coord_desc(best_h, t_p4, n_sweeps=1000, T0=0.002, cool=0.99)

    # Phase 5: Ultra-fine final pass (5% budget)
    t_p5 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_desc(best_h, t_p5, n_sweeps=300, T0=0.0)

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_to_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    # If sum is slightly off, do one more adjustment
    if abs(best_h.sum() - n_final / 2.0) > 1e-10:
        best_h = project_to_sum(best_h, n_final / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
```
