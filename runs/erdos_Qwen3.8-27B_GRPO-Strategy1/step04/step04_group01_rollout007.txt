def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(x, T):

    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

        return vc
    lo = -np.max(vec) - 1.0
    hi = 1.0 - np.min(vec) + 1.0

        if np.clip(vec + mid, 0.0, 1.0).sum() > target:

        else:
