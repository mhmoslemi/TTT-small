def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def init_fd(h, n):
    L = 2 * n - 1
    fd = np.zeros((n, L))
    for p in range(n):
        # First term: (1-h[p-m]) for m in [p-(n-1), p], i.e., k in [p, p+n-1]
        lo1 = max(0, p)
        hi1 = min(L-1, p + n - 1)
        if lo1 <= hi1:

            idx = p - m_vals
            fd[p, lo1:hi1+1] += (1.0 - h[idx])
        # Second term: -h[p+m] for m in [-p, n-1-p], i.e., k in [n-1-p, 2n-2-p]
        lo2 = max(0, n-1-p)
        hi2 = min(L-1, 2*n-2-p)
        if lo2 <= hi2:
            m_vals = np.arange(lo2, hi2+1) - (n-1)
            idx = p + m_vals
            fd[p, lo2:hi2+1] -= h[idx]
    return fd

def update_fd(fd, n, p, q):

    L = 2 * n - 1

def binary_local_search(h_bin, n, t_end, max_sweeps=5000):

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h0, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = get_corr(best_h)
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            def obj(v):
                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                return np.max(cn)
            v0 = obj(0.0)
            v1r = obj(1.0)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            best_v = res.x
            best_f = res.fun
            if v0 < best_f:
                best_v, best_f = 0.0, v0
            if v1r < best_f:
                best_v, best_f = 1.0, v1r
            if best_f * dx < best_c5 - 1e-14:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = get_corr(best_h)
                improved = True
        if not improved:
            break
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def simulated_annealing_binary(h0, n, t_end, max_iters=5000, T_start=0.5, T_end=0.01):
    h = h0.copy().astype(float)
    c = get_corr(h)
    L = 2 * n - 1
    dx = 2.0 / n
    fd = init_fd(h, n)
    target = np.max(c)
    best_c5 = target * dx
    best_h = h.copy()
    work_c5 = best_c5
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = ones[np.random.randint(len(ones))]
        q = zeros[np.random.randint(len(zeros))]
        new_c = c - fd[p] + fd[q]
        new_target = np.max(new_c)
        new_c5 = new_target * dx
        delta = new_c5 - work_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            h[p] = 0
            h[q] = 1
            c = new_c
            update_fd(fd, n, p, q)
            work_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(160) * 0.5
        h0[::2] = 1.0
    n = len(h0)
    
    # Round to binary
    h_bin = (h0 >= 0.5).astype(float)
    s = h_bin.sum()
    target_sum = n / 2
    if s != target_sum:
        # Adjust: need to flip (target_sum - s) positions
        if s > target_sum:
            # Need to flip some 1s to 0s
            excess = int(s - target_sum)
            ones = np.where(h_bin == 1)[0]
            np.random.shuffle(ones)
            h_bin[ones[:excess]] = 0
        else:
            deficit = int(target_sum - s)
            zeros = np.where(h_bin == 0)[0]
            np.random.shuffle(zeros)
            h_bin[zeros[:deficit]] = 1
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: Binary local search (35% time)
    h_bl, c5_bl = binary_local_search(h_bin, n, t0 + budget_s * 0.35, max_sweeps=10000)
    if c5_bl < best_c5:
        best_c5 = c5_bl
        best_h = h_bl.copy()
