        else:

    peak = float(np.max(corr))
    center = N - 1

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_for_lag(h, L, N):
    M = abs(int(L))
    grad = np.zeros(N)
    if M >= N or M == 0:
        if M == 0:
            return 1.0 - 2.0 * h
        return grad
    if L > 0:
        grad[L:] += 1.0
        grad[L:] -= h[:N - L]
        grad[:N - L] -= h[L:]
    else:
        grad[:N - M] += 1.0
        grad[:N - M] -= h[M:]
        grad[M:] -= h[:N - M]
    return grad

def smooth_max_grad(h, N, top_k, eps):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    center = N - 1
    top_idx = np.argsort(corr)[-top_k:]
    vals = corr[top_idx] - peak
    w = np.exp(vals / eps)
    wsum = w.sum()
    if wsum < 1e-15:
        L = top_idx[np.argmax(corr[top_idx])] - center
        return grad_for_lag(h, L, N), peak
    w = w / wsum
    grad = np.zeros(N)
    for i, ai in enumerate(top_idx):
        if w[i] < 1e-10:
            continue
        L = ai - center
        grad += w[i] * grad_for_lag(h, L, N)
    return grad, peak

def continuous_descent(h, N, target, deadline, lr0, decay, top_k, eps_s, eps_e, min_steps=10):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    elapsed0 = time.time()
    total_est = max(1.0, deadline - time.time())
    it = 0
    while time.time() < deadline:
        frac = min(1.0, (time.time() - elapsed0) / total_est)
        eps = eps_s * (eps_e / eps_s) ** frac
        grad, peak = smooth_max_grad(h, N, top_k, eps)
        h = h - lr * grad
        h = project_sum(h, target)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        it += 1
        if it % 80 == 0:
            lr *= decay
        if it < min_steps:
            continue
    return best_h, best_peak * dx

def binarize(h, N):
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    nh = N // 2
    if n_ones > nh:
        idx = np.where(h_b == 1)[0]
        h_b[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:nh - n_ones]] = 1
    return h_b

def local_search(h_b, N, n_iter, rng, deadline=None):
    h = h_b.astype(float).copy()
    best = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iter):
        if deadline and time.time() > deadline:
            break
        T = max(0.01, 0.8 * 0.9995 ** it)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        npk = float(np.max(compute_corr(h)))
        if npk < best_pk:
            best_pk = npk
            best = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        elif rng.random() < np.exp(-(npk - best_pk) / T):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_pk * (2.0 / N)

def coord_descent polish(h, N, target, deadline):
    h = h.astype(float).copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    for _ in range(20):
        if time.time() > deadline:
            break
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            for val in [0.0, 1.0]:
                if abs(h[i] - val) < 1e-6:
                    continue
                old = h[i]
                h[i] = val
                h = project_sum(h, target)
                pk = float(np.max(compute_corr(h)))
                if pk < best_peak - 1e-12:
                    best_peak = pk
                    improved = True
                else:
                    h[i] = old
                    h = project_sum(h, target)
        if not improved:
            break
    return h, best_peak * dx
