def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_max(corr, T):
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return soft_max(corr, T) * dx

def get_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    weights = np.exp(T * (corr - m)) / np.sum(np.exp(T * (corr - m)))
    grad = np.zeros(n)
    center = n - 1
    for km in range(2 * n - 1):
        k = km - center
        w = weights[km]
        if w < 1e-10:
            continue
        if k == 0:
            grad += w * (1 - 2 * h)
        elif k > 0:
            grad[:n-k] += w * (1 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            mk = -k
            grad[mk:] += w * (1 - h[:n-mk])
            grad[:n-mk] -= w * h[mk:]
    return grad * dx

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Adjust to meet sum constraint
    # Add to zeros/ones proportionally
    if abs(s - target) > 0.5 * n:
        return np.clip(np.full(n, 0.5), 0, 1)
    diff = target - s
    if diff > 0:
        # Need to increase sum: add to elements with room to grow
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        # Need to decrease sum: subtract proportionally
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

        return get_grad(x, T)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_sum(np.clip(res.x, 0, 1), n)
    except:
        return h

            f_new = smooth_obj(h_new, 100)
        h = h_new
        grad = get_grad(h, 100)
        alpha *= 0.95
    return h

    return np.max(corr) * dx

    m = corr.max()
    if T < 1:
        return m
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return soft_max_val(corr, T) * dx

def smooth_grad(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_vals = np.exp(T * (corr - m))
    w = exp_vals / exp_vals.sum()

        wk = w[km]
