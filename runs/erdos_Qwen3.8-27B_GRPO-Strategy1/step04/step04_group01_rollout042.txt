import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h, n):
    t = n // 2
    idx = np.argpartition(-h, t)[:t]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def rand_binary(n):
    h = np.zeros(n)
    h[np.random.permutation(n)[:n // 2]] = 1.0
    return h

def hadamard_init(n):
    if n < 2 or (n & (n - 1)) != 0:
        return rand_binary(n)
    H = np.array([[1.0]])
    while H.shape[0] < n:
        H = np.concatenate([np.concatenate([H, H], 1),
                            np.concatenate([H, -H], 1)], 0)
    return ((H[-1] + 1.0) * 0.5)

def project_eq(h, n):
    tgt = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0, 1).sum() > tgt:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0, 1)

def sa_core(cur, n, t_end, T0):
    cur = np.asarray(cur, dtype=float).copy()
    cc = compute_c5(cur)
    best = cur.copy()
    bc = cc
    ts = time.time()
    dur = max(t_end - ts, 0.1)
    while time.time() < t_end - 0.003:
        f = min((time.time() - ts) / dur, 0.9999)
        T = max(T0 * 0.5 ** (f * 16), 1e-14)
        on = np.flatnonzero(cur > 0.5)
        zo = np.flatnonzero(cur < 0.5)
        if on.size < 1 or zo.size < 1:
            break
        dbl = (on.size >= 2 and zo.size >= 2 and np.random.rand() < 0.3)
        if dbl:
            p = np.random.choice(on.size, 2, replace=False)
            q = np.random.choice(zo.size, 2, replace=False)
            a, b = on[p[0]], on[p[1]]
            c, d = zo[q[0]], zo[q[1]]
            cur[a] = cur[b] = 0.0
            cur[c] = cur[d] = 1.0
            nc = compute_c5(cur)
            if nc < cc or np.random.rand() < np.exp(-(nc - cc) / T):
                cc = nc
                if nc < bc:
                    bc, best = nc, cur.copy()
            else:
                cur[a] = cur[b] = 1.0
                cur[c] = cur[d] = 0.0
        else:
            i = on[np.random.randint(on.size)]
            j = zo[np.random.randint(zo.size)]
            cur[i] = 0.0
            cur[j] = 1.0
            nc = compute_c5(cur)
            if nc < cc or np.random.rand() < np.exp(-(nc - cc) / T):
                cc = nc
                if nc < bc:
                    bc, best = nc, cur.copy()
            else:
                cur[i] = 1.0
                cur[j] = 0.0
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:

        te = min(time.time() + max(t3 - time.time(), 0) * 0.12, t3 - 0.2)

        hr, cr = sa_core(hp, nb, te, 0.008)

    hc = best_h.copy()

        if ct < best_c5:

        else:
            hc = best_h.copy()

    best_h = project_eq(best_h, nb)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, nb
