                best_c5, best_h = c5v, hc
        except Exception:
            pass
        try:
            hc = slsqp_opt(best_h, 100000, n, maxiter=1500, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Final coord descent + evaluate
    if time.time() - t0 < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=2000)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
