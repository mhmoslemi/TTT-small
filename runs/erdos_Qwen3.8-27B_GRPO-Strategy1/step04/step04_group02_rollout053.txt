    mc = np.max(corr)

        if has_quad:
            vals[quad_m] = corr_base[quad_m] + t - t * t
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=50, seed_offset=0):
    rng = np.random.default_rng(12345 + seed_offset)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def basin_hopping(h, n, t_end, rng, n_iters=80):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.9 * (it / max(n_iters, 1))
        step = 0.04 * (0.2 + 0.8 * cool)
        pert = project(work + step * rng.standard_normal(n), n)
        T_s = int(200 + 4000 * cool)
        hc = slsqp_refine(pert, T_s, n, maxiter=200)
        hc = proj_grad(hc, 5000, n, time.time() + 5)
        cv = compute_c5(hc)
        temp = 0.004 * cool + 0.0003
        if cv < work_c5 or rng.random() < np.exp(-(cv - work_c5) / temp):
            work, work_c5 = hc, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP homotopy with analytical gradient
    for T in [10, 40, 150, 500, 2000, 8000, 25000]:
        if elapsed() > budget_s * 0.22:
            break
        hc = slsqp_refine(best_h, T, n, maxiter=400)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 2: Projected gradient with large T
    for T in [500, 3000, 15000, 60000]:
        if elapsed() > budget_s * 0.40:
            break
        hc = proj_grad(best_h, T, n, t0 + budget_s * 0.40, max_iter=1500)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 3: Coordinate descent (exact C5)
    if elapsed() < budget_s * 0.62:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.62,
                                        max_sweeps=50, seed_offset=0)

    # Phase 4: Basin hopping
    if elapsed() < budget_s * 0.82:
        best_h, best_c5 = basin_hopping(best_h, n, t0 + budget_s * 0.82,
                                        rng, n_iters=60)

    # Phase 5: Second round of coordinate descent + SLSQP
    if elapsed() < budget_s * 0.90:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90,
                                        max_sweeps=30, seed_offset=7)
    for T in [50000, 150000]:
        if elapsed() > budget_s * 0.94:
            break
        hc = slsqp_refine(best_h, T, n, maxiter=500)
        hc = proj_grad(hc, 100000, n, t0 + budget_s * 0.95, max_iter=500)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 6: Multi-resolution exploration
    for n_try in [200, 280, 320, 360]:
        if elapsed() > budget_s * 0.97:
            break
        h_interp = interpolate_h(best_h, n_try)
        hc = slsqp_refine(h_interp, 3000, n_try, maxiter=300)
        hc = coord_descent(hc, n_try, t0 + budget_s * 0.975, max_sweeps=10, seed_offset=n_try)[0]
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5, n = hc, cv, n_try

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
