```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-13:
        return h
    if d > 0:
        rm = 1.0 - h
        tot = rm.sum()
        if tot < 1e-13:
            return h
        h += rm * (d / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def _slsqp_obj(x):
    T, n = _SLSQP_P['T'], _SLSQP_P['n']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = np.clip(T * (c - m), -500, 500)
    e = np.exp(sh)
    return (m + np.log(e.sum()) / T) * dx

def _slsqp_con(x):
    return x.sum() - _SLSQP_P['n'] / 2.0

                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                     options={'maxiter': mxit, 'ftol': ftol})
        return proj(np.clip(r.x, 0, 1), n)
    except Exception:
        return h_init.copy()

    cv = cb + t * coeff
    cv[k0] = cb[k0] + t * (1.0 - t)
    return float(np.max(cv))
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_SP = {}

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt, s = n / 2.0, h.sum()
    d = tgt - s
    if abs(d) < 1e-13:
        return h
    if d > 0:
        rm, tot = 1.0 - h, 0.0
        tot = rm.sum()
        if tot < 1e-13:
            return h
        h += rm * (d / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def _sobj(x):
    T, n = _SP['T'], _SP['n']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = np.clip(T * (c - m), -500, 500)
    e = np.exp(sh)
    return (m + np.log(e.sum()) / T) * 2.0 / n

def _scon(x):
    return x.sum() - _SP['n'] / 2.0

def slsqp(hi, T, n, mx=500, ft=1e-14):
    _SP['T'], _SP['n'] = T, n
    try:
        r = minimize(_sobj, hi, method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': mx, 'ftol': ft})
        return proj(np.clip(r.x, 0, 1), n)
    except Exception:
        return hi.copy()

def _f1d(t, cb, cf, k0):
    cv = cb + t * cf
    cv[k0] = cb[k0] + t * (1.0 - t)
    return float(np.max(cv))

def cdesc(hi, n, te, ms, sd):
    rng = np.random.RandomState(sd)
    best, bc = hi.copy(), compute_c5(hi)
    nc, k0 = 2 * n - 1, n - 1
    lags = np.arange(nc) - (n - 1)
    for sw in range(ms):
        if time.time() > te:
            break
        hw = best.copy()
        for i in rng.permutation(n):
            if time.time() > te:
                break
            hz = hw.copy()
            hz[i] = 0.0
            cb = np.correlate(hz, 1.0 - hz, mode='full')
            j1, j2 = i + lags, i - lags
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            cf = np.zeros(nc)
            cf[v1] += 1.0 - hw[j1[v1]]
            cf[v2] += -hw[j2[v2]]
            r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                args=(cb, cf, k0), options={'xatol': 1e-11})
            hw[i] = r.x
        hw = proj(hw, n)
        c = compute_c5(hw)
        if c < bc:
            bc, best = c, hw
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n = len(bh)

    bc5 = compute_c5(bh)

        if c2 < bc5:

            if c2 < bc5:
```
