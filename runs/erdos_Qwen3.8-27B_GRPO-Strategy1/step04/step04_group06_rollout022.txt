```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = T * (corr - m)
    e -= e.max()  # numerical stability (redundant but safe)
    w = np.exp(e)
    w /= w.sum()
    grad = np.zeros(n)
    center = n - 1
    for la in np.where(w > 1e-8)[0]:
        wl = w[la]
        k = la - center
        ak = abs(k)
        if ak == 0:
            grad += wl * (1.0 - 2.0 * h)
        else:
            idx1 = np.arange(n - ak)
            grad[idx1] += wl * (1.0 - h[idx1 + ak])
            idx2 = np.arange(ak, n)
            grad[idx2] -= wl * h[idx2 - ak]
    return grad * dx

def project(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0, 1, out=h)
    return h

def slsqp(h0, T, n, maxiter=800):
    x0 = h0.copy()
    res = minimize(
        smooth_obj, x0, args=(T,),
        jac=smooth_grad,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                     'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return project(np.clip(res.x, 0, 1), n)

def fast_coord_descent(h, n, t_end, max_sweeps=200):
    """Coordinate descent with precomputed correlation updates."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    center = n - 1
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Compute effect of changing h[i]
            h_base = best_h.copy()
            old_val = h_base[i]
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            
            # Effect vector: corr_try = corr_base + x * b, with quadratic correction at center
            b = np.zeros(2 * n - 1)
            b[center] = 1.0
            for k in range(1, n):
                ip = i + k
                im = i - k
                if ip < n:
                    b[center + k] += (1.0 - best_h[ip])
                    b[center - k] -= best_h[ip]
                if im >= 0:
                    b[center + k] -= best_h[im]
                    b[center - k] += (1.0 - best_h[im])
            
            def f1d(x):
                vals = corr_base + x * b
                vals[center] = corr_base[center] + x - x * x
                return np.max(vals) * dx
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            new_val = res.x
            if abs(new_val - old_val) > 1e-10:
                test_h = best_h.copy()
                test_h[i] = new_val
                test_h = project(test_h, n)
                c5_try = compute_c5(test_h)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = test_h
                    improved = True
        if not improved:
            break
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = T * (corr - m)
    w = np.exp(e - e.max())
    w /= w.sum()
    grad = np.zeros(n)
    center = n - 1
    active = np.where(w > 1e-10)[0]
    for la in active:
        wl = w[la]
        k = la - center
        ak = abs(k)
        if ak == 0:
            grad += wl * (1.0 - 2.0 * h)
        else:
            idx1 = np.arange(n - ak)
            grad[idx1] += wl * (1.0 - h[idx1 + ak])
            idx2 = np.arange(ak, n)
            grad[idx2] -= wl * h[idx2 - ak]
    return grad * dx

def sum_con_fun(x):
    return x.sum() - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

def project(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0, 1, out=h)
    return h

def slsqp_opt(h0, T, n, maxiter=800):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(T,), jac=smooth_grad,
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

def coordinate_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    center = n - 1
    dx = 2.0 / n
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_base = best_h.copy()
            old_val = h_base[i]
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            b = np.zeros(2 * n - 1)
            b[center] = 1.0
            for k in range(1, n):
                ip = i + k
                im = i - k
                if ip < n:
                    b[center + k] += (1.0 - best_h[ip])
                    b[center - k] -= best_h[ip]
                if im >= 0:
                    b[center + k] -= best_h[im]
                    b[center - k] += (1.0 - best_h[im])
            cb = corr_base
            bc = corr_base[center]
            def f1d(x):
                vals = cb + x * b
                vals[center] = bc + x - x * x
                return np.max(vals) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            new_val = res.x
            if abs(new_val - old_val) > 1e-9:
                test_h = best_h.copy()
                test_h[i] = new_val
                test_h = project(test_h, n)
                c5_try = compute_c5(test_h)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = test_h
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return best_h, best_c5

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_budget_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick coordinate descent to polish initial
    best_h, best_c5 = coordinate_descent(best_h, n, t0 + budget_s * 0.05, max_sweeps=80)

    # Phase 2: SLSQP continuation with increasing T
    T_schedule = [20, 50, 100, 200, 500, 1000, 3000]
    t_p2_end = t0 + budget_s * 0.25
    for T in T_schedule:
        if time.time() > min(t_p2_end, t0 + budget_s * 0.30):
            break
        try:
            h_new = slsqp_opt(best_h, T, n, maxiter=1200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            else:
                h_pert = best_h + 0.02 * np.random.randn(n)
                h_pert = project(h_pert, n)
                h_new2 = slsqp_opt(h_pert, T, n, maxiter=600)
                c5_new2 = compute_c5(h_new2)
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2
        except Exception:
            pass

    # Phase 3: Coordinate descent refinement
    t_p3_end = t0 + budget_s * 0.40
    best_h, best_c5 = coordinate_descent(best_h, n, t_p3_end, max_sweeps=300)

    # Phase 4: High-T SLSQP
    T_high = [5000, 15000, 50000]
    t_p4_end = t0 + budget_s * 0.55
    for T in T_high:
        if time.time() > t_p4_end:
            break
        try:
            h_new = slsqp_opt(best_h, T, n, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Phase 5: Multi-resolution exploration
    t_p5_end = t0 + budget_s * 0.80
    for n_try in [200, 240, 320]:
        if time.time() > t_p5_end:
            break
        h_up = upsample(best_h, n, n_try)
        for T in [50, 500, 3000]:
            if time.time() > t_p5_end:
                break
            try:
                h_new = slsqp_opt(h_up, T, n_try, maxiter=600)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    n = n_try
            except Exception:
                pass
        h_cd, c5_cd = coordinate_descent(best_h, n, min(time.time() + 4, t_p5_end), max_sweeps=60)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 6: Random restarts from perturbed best
    t_p6_end = t0 + budget_s * 0.90
    for attempt in range(3):
        if time.time() > t_p6_end:
            break
        scale = 0.04 * (attempt + 1)
        h_pert = best_h + scale * np.random.randn(n)
        h_pert = project(h_pert, n)
        try:
            h_new = slsqp_opt(h_pert, 3000, n, maxiter=800)
            h_new, c5_new = coordinate_descent(h_new, n, min(time.time() + 3, t_p6_end), max_sweeps=50)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Phase 7: Final SLSQP very high T + coordinate descent
    t_p7_end = t0 + budget_s * 0.96
    try:
        h_new = slsqp_opt(best_h, 200000, n, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    except Exception:
        pass

    best_h, best_c5 = coordinate_descent(best_h, n, t_budget_end - 2, max_sweeps=300)
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
