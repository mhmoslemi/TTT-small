    """Compute cross-correlation h * (1-h) using FFT."""
    n = len(h)
    size = 2 * n
    H = fft(h, size)
    G = fft(1.0 - h, size)
    corr = np.real(ifft(H * G, size)[:2*n-1])
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

    n = len(h)
    size = 2 * n
    H = fft(h, size)
    G = fft(1.0 - h, size)
    return np.real(ifft(H * G, size)[:2*n-1])

def compute_c5(h):

    return np.max(compute_corr(h)) * (2.0 / n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            step = d / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0:

    vals[n - 1] = t * (1.0 - t)
    return np.max(vals)

def minimize_f1d(corr, a, h_save, n):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    l, r = 0.0, 1.0
    c = r - gr * (r - l)
    d = l + gr * (r - l)
    fc = f1d_eval(corr, a, h_save, n, c)
    fd = f1d_eval(corr, a, h_save, n, d)
    for _ in range(50):
        if fc < fd:
            r, d, fd = d, c, fc
            c = r - gr * (r - l)
            fc = f1d_eval(corr, a, h_save, n, c)
        else:
            l, c, fc = c, d, fd
            d = l + gr * (r - l)
            fd = f1d_eval(corr, a, h_save, n, d)
        if r - l < 1e-14:
            break
    t_star = (l + r) / 2.0
    f0 = f1d_eval(corr, a, h_save, n, 0.0)
    f1 = f1d_eval(corr, a, h_save, n, 1.0)
    fmid = f1d_eval(corr, a, h_save, n, t_star)
    if f0 <= f1 and f0 <= fmid:
        return 0.0
    if f1 <= f0 and f1 <= fmid:
        return 1.0
    return t_star

def coord_descent(h, n, max_sweeps, rng, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = compute_corr(best_h)
    stall = 0
    for _sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_save = best_h[i]
            a = compute_slopes(best_h, n, i)
            t_star = minimize_f1d(corr, a, h_save, n)
            if abs(t_star - h_save) > 1e-10:
                best_h[i] = t_star
                best_h = project(best_h, n)
                corr = compute_corr(best_h)
                c5 = np.max(corr) * (2.0 / n)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def make_sine_initial(n, freq, amp):
    x = np.arange(n, dtype=float)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n)
    return project(h, n)

def make_square_initial(n, freq):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.45 * np.sign(np.sin(2.0 * np.pi * freq * x / n))
    h[np.abs(h - 0.5) < 1e-10] = 0.5
    return project(h, n)

def make_block_initial(n, frac):
    h = np.zeros(n)
    mon = int(round(frac * n))
    h[:mon] = 1.0
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_sine_initial(240, 3, 0.45)

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from initial (25% budget)
    t1 = t0 + budget_s * 0.25
    best_h, best_c5 = coord_descent(best_h, n, 30, rng, t1)

    # Phase 2: Random restarts with various perturbations (35% budget)
    t2 = t0 + budget_s * 0.60
    n_restarts = 30
    for r in range(n_restarts):
        if time.time() > t2:
            break
        strat = r % 5
        if strat == 0:
            sigma = 0.03 + 0.04 * (r / max(n_restarts, 1))
            pert = best_h + sigma * rng.standard_normal(n)
        elif strat == 1:
            pert = best_h.copy()
            bs = max(n // 5, 5)
            st = int(rng.integers(0, n - bs))
            pert[st:st + bs] = rng.random(bs)
        elif strat == 2:
            pert = best_h.copy()
            sub = rng.choice(n, size=max(n // 4, 4), replace=False)
            pert[sub] = 1.0 - best_h[sub]
        elif strat == 3:
            freq = int(rng.integers(1, 9))
            amp = float(rng.uniform(0.3, 0.5))
            pert = make_sine_initial(n, freq, amp)
        else:
            freq = int(rng.integers(1, 7))
            pert = make_square_initial(n, freq)
        pert = project(pert, n)
        hc, cv = coord_descent(pert, n, 6, rng, t2)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 3: Resolution cycling (20% budget)
    t3 = t0 + budget_s * 0.82
    for n_try in [120, 160, 200, 280, 300]:
        if time.time() > t3:
            break
        h_interp = interpolate_h(best_h, n_try)
        hc, cv = coord_descent(h_interp, n_try, 12, rng, t3)
        if cv < best_c5:
            best_h, best_c5, n = hc, cv, n_try

    # Phase 4: More restarts at best resolution (10% budget)
    t4 = t0 + budget_s * 0.92
    for r in range(10):
        if time.time() > t4:
            break
        sigma = 0.02 + 0.02 * (r / 10.0)
        pert = best_h + sigma * rng.standard_normal(n)
        pert = project(pert, n)
        hc, cv = coord_descent(pert, n, 8, rng, t4)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 5: Final long coordinate descent (8% budget)
    t5 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent(best_h, n, 60, rng, t5)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
