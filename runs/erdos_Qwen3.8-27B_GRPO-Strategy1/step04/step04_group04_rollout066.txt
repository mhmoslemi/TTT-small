    """Gradient of smooth C5 (log-sum-exp approximation of max)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # corr[k] = sum_{j=0}^{n-1-lag} h[j](1-h[j+lag])
            # d/dh[i] for i in [0, n-1-lag]: (1-h[i+lag])
            # d/dh[i] for i in [lag, n-1]: -h[i-lag]
            if n - lag > 0:
                grad[:n-lag] += pk * (1.0 - h[lag:])
                grad[lag:] += pk * (-h[:n-lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n-nl])
                grad[:n-nl] += pk * (-h[nl:])
    return grad * dx
