def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)

    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr *= 0.993
            beta = min(0.97, 0.88 + it * 0.0002)
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary local search on initial
    d1 = min(deadline - 300, t0 + 250)
    h_bin = (h0 > 0.5).astype(float)
    h_s, _ = binary_search(h_bin, N0, d1, max_sweeps=600, sample_n=120)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Nesterov continuous
    if time.time() < deadline - 150:
        d2 = min(deadline - 120, time.time() + 120)
        hc = best_h.astype(float) + np.random.randn(len(best_h)) * 0.06
        hc = project_sum(hc, len(best_h) / 2.0)
        hn, c5n = nesterov_opt(hc, len(best_h), d2, k=60, lr0=0.02)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: More binary search from improved
    if time.time() < deadline - 100:
        d3 = min(deadline - 80, time.time() + 150)
        h_bin3 = (best_h > 0.5).astype(float)
        h_s3, _ = binary_search(h_bin3, len(best_h), d3, max_sweeps=500, sample_n=150)
        c5_3 = compute_c5(h_s3, len(best_h))
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Annealed binary search (escape local minima)
    if time.time() < deadline - 80:
        d4 = min(deadline - 60, time.time() + 80)
        h_bin4 = (best_h > 0.5).astype(float)
        h_s4, _ = binary_search(h_bin4, len(best_h), d4, max_sweeps=300, sample_n=80, anneal_T=0.1)
        c5_4 = compute_c5(h_s4, len(best_h))
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h_s4.copy()

    # Phase 5: Multi-resolution
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [450, 550, 700, 900]:
        if time.time() > deadline - 30:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        nh = N_try // 2
        n1 = int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            idx = idx[np.argsort(np.random.random(len(idx)))][nh:]
            h_bt[idx] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            idx = idx[np.argsort(np.random.random(len(idx)))][:nh - n1]
            h_bt[idx] = 1
        sub_dl = min(deadline - 10, time.time() + 50)
        h_rt, _ = binary_search(h_bt, N_try, sub_dl, max_sweeps=250, sample_n=100)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 6: Final Nesterov polish
    if time.time() < deadline - 3:
        hF = best_h.astype(float) + np.random.randn(len(best_h)) * 0.03
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_opt(hF, len(best_h), deadline - 1, k=80, lr0=0.012)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = (best_h > 0.5).astype(float)
        hF2, _ = binary_search(hF2, len(best_h), deadline - 0.5, max_sweeps=50, sample_n=100)
        c5F2 = compute_c5(hF2, len(best_h))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
