    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-14:
        return h
    # Use bisection on a water-filling approach

    if diff > 0:

        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h + h * (diff / s)
    return np.clip(h, 0.0, 1.0)

def proj_grad_nesterov(h0, n, target_sum, t_end, T_schedule):
    """Projected gradient with Nesterov momentum"""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    best_h = h0.copy()
    best_f = compute_c5(h0)
    corr = compute_corr(h0)
    dx = 2.0 / n
    
    for step in range(100000):
        if time.time() > t_end:
            break
        
        T = T_schedule[min(step, len(T_schedule)-1)]
        # Compute gradient
        grad = compute_grad_fft(h, corr, T, dx)
        
        # Barzilai-Borwein or fixed step
        alpha = 1.0 / (T * dx * 2)  # Lipschitz constant bound
        
        # Nesterov step
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (t - 1) / t_new
        y = h + beta * (h - h_prev) if step > 0 else h.copy()
        
        h_new = project_feasible(y - alpha * grad, target_sum)
        
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
        corr = compute_corr(h)
        t = t_new
    
    return best_h, best_f

Let me also think about the coordinate descent more carefully. The issue with single-coordinate descent is the sum constraint. The standard trick is:

**Two-coordinate descent**: Change h[i] by +δ and h[j] by -δ. This maintains the sum.

For a pair (i,j), the effect on correlation[k] is:
δ*((1-h[i-k])*Iv(i-k valid) - h[i+k]*Iv(i+k valid) - (1-h[j-k])*Iv(j-k valid) + h[j+k]*Iv(j+k valid))

Let me define d[k] = (1-h[i-k])*Iv(0≤i-k<n) - h[i+k]*Iv(0≤i+k<n) - (1-h[j-k])*Iv(0≤j-k<n) + h[j+k]*Iv(0≤j+k<n)

Then new_corr[k] = corr[k] + δ*d[k].

We want to choose δ (and possibly the pair i,j) to minimize max_k (corr[k] + δ*d[k]).

For a fixed pair (i,j), the optimal δ is:
min_{δ} max_k (corr[k] + δ*d[k])

This is a 1D convex optimization (max of linear functions of δ is convex). I can use ternary search or golden section search on δ ∈ [-min(h[i],h[j]), min(1-h[i], 1-h[j])].
