def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_exact(best_h, n, i)
            if abs(hi_new - best_h[i]) > 1e-10:
                h_try = best_h.copy()
                h_try[i] = hi_new
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=1500, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def targeted_perturb(h, n, scale):
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-5:]
    grad = np.zeros(n)
    k = np.arange(-(n - 1), n)
    for mk in topk_idx:
        shift = mk - (n - 1)
        j1 = np.arange(n) + shift
        j2 = np.arange(n) - shift
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        g = np.zeros(n)
        g[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
        j2c = np.clip(j2[v2], 0, n - 1)
        g[v2] += (-h[j2c])
        if shift == 0:
            g = np.ones(n) - 2.0 * h
        grad += g
    pert = -scale * grad / (np.std(grad) + 1e-15)
    return h + pert

def swap_perturb(h, n, nswaps):
    h_try = h.copy()
    ones = np.where(h_try > 0.5)[0]
    zeros = np.where(h_try < 0.5)[0]
    for _ in range(nswaps):
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = one[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_try[i], h_try[j] = h_try[j], h_try[i]
    return h_try

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP annealing (12% budget)
    for T in [50, 300, 2000, 15000, 100000]:
        if elapsed() > budget_s * 0.12:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=1200, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Intensive coordinate descent (35% budget)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.47, max_sweeps=400)

    # Phase 3: Basin hopping with targeted perturbations (25% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iters_bh = 60
    for it in range(n_iters_bh):
        if elapsed() > budget_s * 0.72:
            break
        cool = 1.0 - 0.85 * (it / n_iters_bh)
        mod = it % 5
        if mod == 0:
            h_pert = targeted_perturb(work_h, n, 0.03 + 0.08 * cool)
        elif mod == 1:
            h_pert = work_h + (0.02 + 0.06 * cool) * np.random.randn(n)
        elif mod == 2:
            freq = np.random.randint(1, 7)
            phase = np.random.uniform(0, 2 * np.pi)
            h_pert = work_h + (0.03 + 0.07 * cool) * np.sin(
                2 * np.pi * freq * np.arange(n) / n + phase)
        elif mod == 3:
            h_pert = swap_perturb(work_h, n, 2 + int(4 * cool))
        else:
            mask = np.random.random(n) < 0.4
            h_pert = work_h.copy()
            h_pert[mask] += (0.05 + 0.1 * cool) * np.random.randn(mask.sum())
        h_pert = project_h(h_pert)
        T_bh = 1000 + 8000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=500, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Coordinate descent round 2 (15% budget)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.87, max_sweeps=300)

    # Phase 5: Short basin hopping (8% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(25):
        if elapsed() > budget_s * 0.95:
            break
        cool = 1.0 - 0.7 * (it / 25.0)
        h_pert = targeted_perturb(work_h, n, 0.02 + 0.04 * cool)
        h_pert = project_h(h_pert)
        hc = slsqp_opt(h_pert, 5000, n, maxiter=400, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < work_c5:
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: Final coordinate descent + SLSQP polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=100)
    if elapsed() < budget_s * 0.995:
        hc = slsqp_opt(best_h, 500000, n, maxiter=2000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        if elapsed() < budget_s * 0.998:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.999, max_sweeps=30)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
