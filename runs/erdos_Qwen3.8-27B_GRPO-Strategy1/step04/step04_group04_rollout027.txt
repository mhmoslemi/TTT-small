import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):

def smooth_obj(h, T):

    shifted = np.clip(T * (corr - m), -500, 500)

    return (m + np.log(s) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    nk1 = n - 1
    for kk in range(2 * n - 1):

        if pk < 1e-14:

        lag = kk - nk1

            grad[nl:] += pk * (1.0 - h[:n - nl])
            grad[:n - nl] += pk * (-h[nl:])
    return grad * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def pair_coord_opt(h, n, i, j):
    """Optimize coordinates i and j jointly (2D problem)."""
    hz = h.copy()
    hz[i] = 0.0
    hz[j] = 0.0

    nk1 = n - 1

    coef_j = np.zeros(2 * n - 1)

        cj = 0.0
        if 0 <= i + lag < n:
            ci += (1.0 - h[i + lag])
        if 0 <= i - lag < n:
            ci += (-h[i - lag])
        if 0 <= j + lag < n:
            cj += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            cj += (-h[j - lag])
        coef_i[kk] = ci
        coef_j[kk] = cj
    # Cross terms: when both i and j are at the same lag offset
    # For lag 0: h[i]*(1-h[i]) + h[j]*(1-h[j]) - but these are separate
    # Actually the cross term is: for lag L where i+L=j or j+L=i:

    #   Self at lag 0 for j: t2*(1-t2)
    cross_lag1 = (j - i) + nk1  # lag j-i

    res = minimize(f2d, x0=np.array([best_h[i], best_h[j]]), method='Nelder-Mead',

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    nk1 = n - 1
    for kk in range(2 * n - 1):
        pk = p[kk]
        if pk < 1e-14:
            continue
        lag = kk - nk1
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n - lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += pk * (-h[nl:])
    return grad * dx

def topk_grad(h, k=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    thresh = m - (m - np.min(corr)) * 0.02
    active = np.where(corr >= thresh)[0]
    if len(active) > k:
        csorted = corr[active]
        top = np.sort(csorted)[-k:]
        active = active[csorted >= top[0]]
    w = np.zeros(2 * n - 1)
    w[active] = 1.0
    w /= w.sum()
    grad = np.zeros(n)
    nk1 = n - 1
    for kk in active:
        pk = w[kk]
        lag = kk - nk1
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n - lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += pk * (-h[nl:])
    return grad * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_single(h, n, i):
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    dx = 2.0 / n
    nk1 = n - 1
    coef = np.zeros(2 * n - 1)
    for kk in range(2 * n - 1):
        lag = kk - nk1
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[kk] = c
    def f1d(t):
        cv = corr_base + t * coef
        cv[nk1] += t * (1.0 - t)
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x
