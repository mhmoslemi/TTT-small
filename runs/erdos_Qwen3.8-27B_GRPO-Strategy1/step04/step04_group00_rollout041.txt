    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)

        lg = lag_arr[idx]

            continue
        if lg >= 0:
            if lg < n:

            grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    return val, dx * grad

def lbfgs_opt(h0, T, maxiter=5000, ftol=1e-16):
    def obj(x):
        return softmax_obj_grad(x, T)[0]
    def grad(x):
        return softmax_obj_grad(x, T)[1]
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * len(h0),
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return h0

def coord_descent(h, n, t_end, n_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            # Compute base corr with h[i]=0
            h0 = best_h.copy()
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            # Compute coefficient vector
            m_full = 2*n-1
            lag = np.arange(m_full) - n + 1
            coef = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag[idx]
                if lg >= 0:
                    if i >= lg and i - lg < n:
                        coef[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        coef[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        coef[idx] += (1.0 - best_h[i + p])
                    if i >= p and i - p < n:
                        coef[idx] += -best_h[i - p]
            # Minimize max(cb + x*coef) for x in [0,1]
            res = minimize_scalar(lambda x: np.max(cb + x * coef),
                                  bounds=(0, 1), method='bounded',
                                  options={'xatol': 1e-10})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, T, lam):
    """Soft-max objective with sum penalty and analytical gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n/2.0)**2
    grad = np.zeros(n)
    lag_arr = np.arange(2*n-1) - n + 1
    for idx in range(2*n-1):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
            grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n/2.0)
    return val, grad

        return obj_grad(x, T, lam)[0]

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
            grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x):
        return obj_grad(x, T, lam)[0]
    def grad(x):
        return obj_grad(x, T, lam)[1]
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return x0

def coord_descent(h, n, t_end, max_sweeps):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h0 = best_h.copy()
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            def f1d(x):
                return np.max(cb + x * lin + x * x * quad)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B warm-up (15% budget)
    p1_end = t0 + budget_s * 0.15
    for T in [30, 100, 400, 1500, 5000]:
        if time.time() > p1_end:
            break
        hc = lbfgs_opt(best_h, T, 300.0, 3000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Multi-restart (25% budget)
    p2_end = t0 + budget_s * 0.40
    candidates = []
    h_bip = np.zeros(n); h_bip[:n // 2] = 1.0
    candidates.append(project_h(h_bip))
    h_alt = np.zeros(n); h_alt[::2] = 1.0
    candidates.append(project_h(h_alt))
    for fr in [2, 3, 5, 7]:
        h_p = 0.5 + 0.45 * np.sin(2 * np.pi * fr * np.arange(n) / n)
        candidates.append(project_h(h_p))
    for _ in range(3):
        candidates.append(project_h(np.random.beta(3, 3, n)))
    for sc in [0.03, 0.08, 0.15, 0.3]:
        candidates.append(project_h(best_h + sc * np.random.randn(n)))
    for h_init in candidates:
        if time.time() > p2_end:
            break
        hc = lbfgs_opt(h_init, 800, 300.0, 2000, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Coordinate descent (20% budget)
    p3_end = t0 + budget_s * 0.60
    best_h, best_c5 = coord_descent(best_h, n, p3_end, 500)

    # Phase 4: High-T L-BFGS-B refinement (25% budget)
    p4_end = t0 + budget_s * 0.85
    for T in [5000, 15000, 50000, 100000]:
        if time.time() > p4_end:
            break
        hc = lbfgs_opt(best_h, T, 300.0, 5000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Basin hops with L-BFGS-B (10% budget)
    p5_end = t0 + budget_s * 0.95
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(30):
        if time.time() > p5_end:
            break
        cool = 1.0 - 0.9 * (it / 30.0)
        sc = 0.01 + 0.06 * cool
        h_p = project_h(work_h + sc * np.random.randn(n))
        T_bh = 3000 * cool + 200
        hc = lbfgs_opt(h_p, T_bh, 300.0, 1500, 1e-16)
        c5v = compute_c5(hc)
        temp = 0.0005 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 6: Final coordinate descent polish
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, 300)

    # Final L-BFGS-B polish
    if time.time() < t0 + budget_s * 0.99:
        hc = lbfgs_opt(best_h, 200000, 500.0, 3000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
