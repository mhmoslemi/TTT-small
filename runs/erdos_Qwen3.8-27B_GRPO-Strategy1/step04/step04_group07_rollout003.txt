            if tot < 1e-14:

    coeff = np.zeros(2 * n - 1)
    k_start = max(0, i - (n - 1))
    k_end = min(2 * n - 1, i + 1)

        if 0 <= j < n:

        if 0 <= idx < n:
            coeff[k] -= h[idx]
    return coeff

        if pert_type == 0:
            h_p = work_h + step * rng.randn(n)
        elif pert_type == 1:
            block_len = rng.randint(5, n // 3 + 1)
            start = rng.randint(0, n - block_len)
            h_p = work_h.copy()
            h_p[start:start+block_len] += step * rng.randn(block_len)
        else:
            n_swap = rng.randint(3, max(4, n // 10))
            idxs = rng.choice(n, size=n_swap, replace=False)
            h_p = work_h.copy()
            h_p[idxs] += step * rng.randn(n_swap)
        h_p = project(h_p, n)
        h_c, c5_c = coord_descent(h_p, n, time.time() + 8.0, max_sweeps=80)
        temp = 0.003 * cool + 0.0001
        if c5_c < work_c5 or (np.random.rand() < np.exp(-(c5_c - work_c5) / temp)):
            work_h, work_c5 = h_c, c5_c
            if c5_c < best_c5:
                best_h, best_c5 = h_c.copy(), c5_c
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def slsqp_opt(h0, T, n, maxiter=400):
    dx = 2.0 / n
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(smooth_obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng_local = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.ones(n_init) * 0.5
        for i in range(0, n_init, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Heavy coordinate descent
    t1 = t0 + budget_s * 0.20
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=2000)

    # Phase 2: SLSQP polish at increasing T
    for T in [50, 200, 800, 3000]:
        if elapsed() > budget_s * 0.30:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 3: Coordinate descent after SLSQP
    t3 = t0 + budget_s * 0.40
    best_h, best_c5 = coord_descent(best_h, n, t3, max_sweeps=1500)

    # Phase 4: Basin hopping (main exploration)
    t4 = t0 + budget_s * 0.70
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t4, max_iters=400, rng=rng_local)

    # Phase 5: Post-hop coordinate descent
    t5 = t0 + budget_s * 0.78
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=1500)

    # Phase 6: Second basin hop round (finer)
    t6 = t0 + budget_s * 0.88
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t6, max_iters=300, rng=rng_local)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.92, max_sweeps=800)

    # Phase 7: Try alternate resolutions
    for n_try in [120, 180, 200, 240, 320]:
        if elapsed() > budget_s * 0.95:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.96, max_sweeps=400)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 8: Final polish
    best_h, best_c5 = coord_descent(best_h, len(best_h), t0 + budget_s * 0.98, max_sweeps=500)
    if elapsed() < budget_s * 0.99:
        hc = slsqp_opt(best_h, 10000, len(best_h), maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
        best_h, best_c5 = coord_descent(best_h, len(best_h), t0 + budget_s * 1.0, max_sweeps=200)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
