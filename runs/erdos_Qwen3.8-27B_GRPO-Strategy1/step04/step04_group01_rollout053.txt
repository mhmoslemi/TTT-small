        elif strategy == 1:
            for _ in range(rng.integers(6, 15)):
                z = np.where(h_pert == 0)[0]
                o = np.where(h_pert == 1)[0]
                if len(z) < 1 or len(o) < 1:
                    break
                h_pert[z[rng.integers(len(z))]] = 1.0
                h_pert[o[rng.integers(len(o))]] = 0.0
        elif strategy == 2:
            idx = rng.permutation(n)
            h_pert = np.zeros(n)
            h_pert[idx[:n // 2]] = 1.0
        else:
            # Perturb with block moves
            bsz = rng.integers(4, 12)
            for _ in range(3):
                start = int(rng.integers(n - bsz))
                h_pert[start:start + bsz] = 1.0 - h_pert[start:start + bsz]
                h_pert = project_clip(h_pert, n)
                z = np.where(h_pert == 0)[0]
                o = np.where(h_pert == 1)[0]
                n_ones = int(np.sum(h_pert > 0.5))
                if n_ones > n // 2:
                    for _ in range(n_ones - n // 2):
                        h_pert[o[rng.integers(len(o))]] = 0.0
                        o = np.where(h_pert == 1)[0]
                elif n_ones < n // 2:
                    for _ in range(n // 2 - n_ones):
                        h_pert[z[rng.integers(len(z))]] = 1.0
                        z = np.where(h_pert == 0)[0]

        t_sa = min(t3, time.time() + budget_s * 0.007)
        h_new, c5_new = sa_loop(h_pert, n, rng, t_sa, T0=0.05, T1=0.0002)
        t_ls = min(t3, time.time() + budget_s * 0.003)
        h_new, c5_new = greedy_ls(h_new, n, t_ls, max_passes=1500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 4: Final greedy LS (15%)
    t4 = t0 + budget_s * 0.92
    best_h, best_c5 = greedy_ls(best_h, n, t4, max_passes=8000)

    # Phase 5: Random LS for extra polishing (5%)
    t5 = t0 + budget_s * 0.96
    best_h, best_c5 = random_ls(best_h, n, rng, t5, n_attempts=30000)

    # Phase 6: Continuous refinement (4%)
    if time.time() < t0 + budget_s * 0.99:
        best_h, c5c = continuous_refine(best_h, n, t0 + budget_s * 0.99)
        if c5c < best_c5:
            best_c5 = c5c

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
