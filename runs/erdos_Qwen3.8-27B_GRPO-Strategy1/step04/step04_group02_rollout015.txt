Actually, for the specific k=1: c_1 = #{j: h[j+1]=1, h[j]=0} = number of 0→1 transitions = number of 1→0 transitions (for a binary string on a line, they differ by at most 1, depending on boundary values).

So c_1 ≈ number of runs of 1s in the string.

For the max to be minimized, we want all c_k to be as equal as possible. The minimum possible max is bounded below by average of all c_k.

Average of c_k over k: (1/(2n-1)) * sum_k c_k = (1/(2n-1)) * sum_k sum_j h[j+k](1-h[j])
= (1/(2n-1)) * sum_j (1-h[j]) * sum_k h[j+k]
= (1/(2n-1)) * sum_j (1-h[j]) * (sum of h from j to j+n-1-k_max, approximately)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(3):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            s_pos = h.sum()
            if s_pos < 1e-14:
                break
            h -= h * (diff / s_pos)
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def grad_smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    val = (m + np.log(np.sum(exps)) / T) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            if lag > 0:
                i1 = np.arange(lag, n)
                i2 = np.arange(0, n - lag)
                grad[i1] += wk * (1.0 - h[i1 - lag])
                grad[i2] -= wk * h[i2 + lag]
            else:
                m2 = -lag
                i1 = np.arange(m2, n)
                i2 = np.arange(0, n - m2)
                grad[i1] += wk * (1.0 - h[i1 - m2])
                grad[i2] -= wk * h[i2 + m2]
    return val, grad * dx

def lbfgs_smooth(h, T, n, maxiter=500, t_end=None):
    def obj(x):
        v, _ = grad_smooth_c5(x, T)
        return v
    def grad(x):
        _, g = grad_smooth_c5(x, T)
        return g
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return project(res.x, n)

def coord_opt(h, n, i):
    """Optimize coordinate i given all others fixed."""
    dx = 2.0 / n
    h_save = h[i]
    # Compute base correlations (with h[i]=0)
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    # d[k] = change in corr[k] per unit change in h[i]
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if 0 <= i - k < n:
            d[n-1+k] += (1.0 - h[i-k])
        if 0 <= i + k < n:
            d[n-1+k] += (-h[i+k])
        if 0 <= i + k < n:
            d[n-1-k] += (1.0 - h[i+k])
        if 0 <= i - k < n:
            d[n-1-k] += (-h[i-k])
    # At k=0: change is 1-2*h[i] per unit... but we set h[i]=base, so:
    # c0_new = c0_base + h[i]*(1-h[i]) ... wait
    # Actually c0 = sum_j h[j](1-h[j]). With h[i]=0: c0_base doesn't include i.
    # When h[i]=t: c0 = c0_base + t*(1-t)
    c0_base = corr_base[n-1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n-1] = c0_base + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def active_coord_descent(h, n, t_end, max_sweeps=50, top_k_frac=0.1, rng=None):
    if rng is None:
        rng = np.random.default_rng(123)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_active = max(5, int(top_k_frac * n))
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        # Find active coordinates: those appearing in top correlation lags
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_lags = np.argsort(corr)[-n_active:]
        active_coords = set()
        for lag_idx in top_lags:

            # Coordinates affected by this lag
            if lag >= 0:
                active_coords.update(range(lag, n))
                active_coords.update(range(0, n - lag))
            else:
                active_coords.update(range(-lag, n))
                active_coords.update(range(0, n + lag))
        if not active_coords:
            active_coords = set(range(n))
        order = list(active_coords)
        rng.shuffle(order)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-10:

                best_h = project(best_h, n)
                c5 = compute_c5(best_h)

                    best_c5 = c5
                    improved = True
                else:
                    best_h[i] = hi  # keep the change even if C5 didn't improve (might help later)
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

    x_new = np.linspace(0, 2, n_new, endpoint=False)
