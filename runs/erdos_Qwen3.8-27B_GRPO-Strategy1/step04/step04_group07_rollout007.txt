def project_feasible(h, n):

    target = n / 2.0

    diff = target - s

        tot = room.sum()

            return h
        h = h + room * (diff / tot)

        h = h * (1.0 + diff / s)

            v2 = (j2 >= 0) & (j2 < n)

            f_hi = obj(hi)

                                  options={'xatol': 1e-10, 'maxiter': 30})

    w = np.exp((corr - m) / T)
    w /= w.sum()
    idx = np.arange(2 * n - 1)
    tau = idx - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        j1 = i - tau
        j2 = i + tau
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        grad[i] = np.sum(w[v1] * (1.0 - h[j1[v1]])) - np.sum(w[v2] * h[j2[v2]])
    h_new = h - alpha * grad
    h_new = project_feasible(h_new, n)
