            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def interp_to(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_sum(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        for i in range(0, n_init, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project_sum(best_h, n / 2.0)
    best_c5 = compute_c5_val(best_h)

    # Phase 1: thorough coordinate descent
    t1 = t0 + budget_s * 0.22
    best_h, best_c5 = coord_descent(best_h, t1, max_sweeps=3000)

    # Phase 2: SLSQP with T schedule
    for T in [50, 200, 800, 3000, 10000]:
        if time.time() > t0 + budget_s * 0.35:
            break
        try:
            hc = slsqp_step(best_h, T, n, maxiter=900)
            c5v = compute_c5_val(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 3: more coordinate descent
    t3 = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, t3, max_sweeps=3000)

    # Phase 4: basin hopping (targeted)
    t4 = t0 + budget_s * 0.70
    best_h, best_c5 = basin_hop_targeted(best_h, best_c5, n, t4, max_iters=400)

    # Phase 5: post-hop coordinate descent
    t5 = t0 + budget_s * 0.82
    best_h, best_c5 = coord_descent(best_h, t5, max_sweeps=3000)

    # Phase 6: try other resolutions
    for n_try in [120, 200, 240]:
        if time.time() > t0 + budget_s * 0.92:
            break
        h_tr = interp_to(best_h, n_try)
        t_tr = min(time.time() + budget_s * 0.012, t0 + budget_s * 0.93)
        h_tr, c5_tr = coord_descent(h_tr, t_tr, max_sweeps=500)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 7: final polish
    t7 = t0 + budget_s * 0.96
    best_h, best_c5 = coord_descent(best_h, t7, max_sweeps=2000)
    if time.time() < t0 + budget_s * 0.98:
        try:
            hc = slsqp_step(best_h, 20000, n, maxiter=2000)
            c5v = compute_c5_val(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
                best_h, best_c5 = coord_descent(best_h, t0 + budget_s * 0.995, max_sweeps=500)
        except Exception:
            pass

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5_val(best_h)
    return best_h, best_c5, len(best_h)
