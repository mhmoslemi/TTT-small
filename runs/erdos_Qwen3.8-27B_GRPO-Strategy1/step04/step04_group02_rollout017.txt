from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = h + room * (diff / tot)
    else:
        h = h - h * (abs(diff) / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-14 and s2 > 1e-14:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def compute_grad_smooth(h, T):
    """Compute gradient of smooth_c5 w.r.t. h, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    w = exp_vals / exp_vals.sum()
    
    grad = w[n-1] * (1.0 - 2.0 * h)
    for k in range(1, min(n, 120)):  # limit to relevant lags
        wk = w[n-1+k] + (w[n-1-k] if n-1-k >= 0 else 0)
        if wk < 1e-12:
            continue
        gj = np.zeros(n)
        gj[k:] = 1.0 - h[:n-k]
        gj[:n-k] -= h[k:]
        grad += wk * gj
    return grad * dx, corr * dx

def pgd_optimize(h, n, T, t_end, lr=0.005, momentum=0.9, max_iter=10000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    vel = np.zeros(n)
    lr_cur = lr
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        if it % 500 == 0 and it > 0:
            lr_cur *= 0.7
            T_cur = min(T * (1.0 + 0.5 * it / max_iter), T * 5)
        else:
            T_cur = T
        
        grad, _ = compute_grad_smooth(h, T_cur)
        vel = momentum * vel - lr_cur * grad
        h_new = h + vel
        h_new = project(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    
    return best_h, best_c5

def coord_opt(h, n, i):
    """Optimize coordinate i via 1D minimization."""
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += (1.0 - h[i - k])
        if 0 <= i + k < n:
            dk += (-h[i + k])
        d[n - 1 + k] = dk
        dk2 = 0.0
        if 0 <= i + k < n:
            dk2 += (1.0 - h[i + k])
        if 0 <= i - k < n:
            dk2 += (-h[i - k])
        d[n - 1 - k] = dk2
    
    c0 = corr_base[n - 1]
    
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = h + room * (diff / tot)
    else:
        h = h - h * (abs(diff) / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-14 and s2 > 1e-14:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    expv = np.exp(T * (corr - m))
    w = expv / expv.sum()
    grad = w[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k]
        if n - 1 - k >= 0:
            wk += w[n - 1 - k]
        if wk < 1e-13:
            continue
        gj = np.zeros(n)
        gj[k:] = 1.0 - h[:n - k]
        gj[:n - k] -= h[k:]
        grad += wk * gj
    return grad * dx

def pgd(h, n, T, t_end, lr=0.004, mu=0.93, max_it=20000):

    bh = h.copy()
    bc5 = compute_c5(bh)
    v = np.zeros(n)
    lr_c = lr
    for it in range(max_it):
        if time.time() > t_end:
            break
        if it % 800 == 0 and it > 0:
            lr_c *= 0.75
        g = grad_smooth(h, T)
        v = mu * v - lr_c * g
        h = project(h + v, n)
        c5 = compute_c5(h)
        if c5 < bc5:
            bh, bc5 = h.copy(), c5
    return bh, bc5

def coord_opt(h, n, i):
    dx = 2.0 / n
    hs = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = hs
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += 1.0 - h[i - k]
        if 0 <= i + k < n:
            dk -= h[i + k]
        d[n - 1 + k] = dk
        dk2 = 0.0
        if 0 <= i + k < n:
            dk2 += 1.0 - h[i + k]
        if 0 <= i - k < n:
            dk2 -= h[i - k]
        d[n - 1 - k] = dk2
    c0 = cb[n - 1]
    def f1(t):
        v = cb + t * d
        v[n - 1] = c0 + t * (1.0 - t)
        return np.max(v) * dx
    res = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def slsqp(h, T, n, maxiter=400, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

    return project(np.interp(xn, xo, ho), nn)
