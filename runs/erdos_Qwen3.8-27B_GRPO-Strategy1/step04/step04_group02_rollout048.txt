def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (target / max(s, 1e-12))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if target - s2 > 1e-12:
            room = 1.0 - h
            tot = room.sum()
