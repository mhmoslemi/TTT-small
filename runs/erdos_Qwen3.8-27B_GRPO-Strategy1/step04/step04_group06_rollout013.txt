import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    weights = exps / exps.sum()  # softmax weights
    # grad[j] = dx * sum_m weights[m] * dcorr[m]/dh[j]
    grad = np.zeros(n)
    for m in range(2*n-1):
        if weights[m] < 1e-15:
            continue
        k = m - (n-1)
        wm = weights[m]
        # dcorr[m]/dh[j] = (1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]
        # For first term: j-k in [0,n-1], j in [k, k+n-1] (intersect [0,n-1])
        #   contributes (1-h[j-k]) to grad[j]
        # For second term: j+k in [0,n-1], j in [-k, n-1-k] (intersect [0,n-1])
        #   contributes -h[j+k] to grad[j]
        j_min1 = max(0, k)
        j_max1 = min(n-1, k+n-1)
        if j_max1 >= j_min1:
            grad[j_min1:j_max1+1] += wm * (1.0 - h[j_min1-k:j_max1-k+1])
        j_min2 = max(0, -k)
        j_max2 = min(n-1, n-1-k)
        if j_max2 >= j_min2:
            grad[j_min2:j_max2+1] -= wm * h[j_min2+k:j_max2+k+1]
    return grad * dx

def project(h, n):
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        target = n / 2.0
        if abs(s - target) < 1e-13:
            return h
        h = h + (target - s) / n
    return np.clip(h, 0.0, 1.0)

def coord_opt_one(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    dx = 2.0 / n
    
    # delta[m] = change in corr[m] when h[i] goes from 0 to v (linear in v)
    # plus quadratic term at m=center
    u = np.zeros(2*n-1)
    w = np.zeros(2*n-1)
    for m in range(2*n-1):
        j1 = i + n - 1 - m  # for correlate(e_i, b): b[j1]
        if 0 <= j1 < n:
            u[m] = 1.0 - h[j1]
        j2 = i + m - n + 1  # for correlate(h, e_i): h[j2]
        if 0 <= j2 < n:
            w[m] = h[j2]
    
    h[i] = old
    # f(v) = max((corr_base + v*(u-w) - v²*x) * dx) where x has 1 at center
    # = dx * max(max_{m≠center}(corr_base[m] + v*(u[m]-w[m])), corr_base[center] + v*(u[center]-w[center]) - v²)
    
    delta_l = u - w  # linear coefficients (excluding center quadratic)
    delta_q = (u[center] - w[center])  # linear part at center
    
    def f(v):
        vals = corr_base + v * delta_l
        vals[center] += v * delta_q - v * v
        return np.max(vals) * dx
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=1000, order='random'):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if order == 'random':
            indices = np.random.permutation(n)
        else:
            indices = range(n)
        for i in indices:
            if time.time() > t_end:
                return best_h, best_c5
            v_new = coord_opt_one(best_h, n, i)
            if abs(v_new - best_h[i]) > 1e-10:
                best_h[i] = v_new
                best_h = project(best_h, n)
                c5_new = compute_c5(best_h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                else:
                    # Revert if projection made things worse
                    best_h[i] = v_new
                    best_h = project(best_h, n)
        if not improved:
            break
    return best_h, best_c5

def grad_descent(h, n, T, lr_init, t_end, max_iter=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > t_end:
            break
        f_val = smooth_c5(best_h, T)
        g = smooth_c5_grad(best_h, T)
        h_new = project(best_h - lr * g, n)
        f_new = smooth_c5(h_new, T)
        if f_new < f_val:
            best_h = h_new
            lr = min(lr * 1.1, 0.5)
            c5_real = compute_c5(best_h)
            if c5_real < best_c5:
                best_c5 = c5_real
        else:
            lr *= 0.7
        if lr < 1e-8:
            break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    t_end = t0 + budget_s
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.3*np.sin(2*np.pi*x) + 0.1*np.cos(4*np.pi*x), 0, 1)
    n = len(best_h)

    # Phase 2: Gradient descent on smooth max (15% time)

                best_h = hc

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    s = exps.sum()
    if s < 1e-300:
        return np.zeros(n)
    weights = exps / s
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = weights[m_idx]
        if wm < 1e-14:
            continue
        k = m_idx - (n - 1)
        j1_lo = max(0, k)
        j1_hi = min(n - 1, k + n - 1)
        if j1_hi >= j1_lo:
            grad[j1_lo:j1_hi + 1] += wm * (1.0 - h[j1_lo - k:j1_hi - k + 1])
        j2_lo = max(0, -k)
        j2_hi = min(n - 1, n - 1 - k)
        if j2_hi >= j2_lo:
            grad[j2_lo:j2_hi + 1] -= wm * h[j2_lo + k:j2_hi + k + 1]
    return grad * dx

def project(h, n):
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) < 1e-14:
            return h
        h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def coord_opt_one(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    dx = 2.0 / n
    delta = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        j1 = i + n - 1 - m
        if 0 <= j1 < n:
            delta[m] += 1.0 - h[j1]
        j2 = i + m - n + 1
        if 0 <= j2 < n:
            delta[m] -= h[j2]
    h[i] = old
    delta_c = delta[center]
    def f_obj(v):
        vals = corr_base + v * delta
        vals[center] += v * delta_c - v * v
        return np.max(vals) * dx
    res = minimize_scalar(f_obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=5000, stall_limit=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            v_new = coord_opt_one(best_h, n, i)
            if abs(v_new - best_h[i]) > 1e-10:
                best_h[i] = v_new
                best_h = project(best_h, n)
                c5_new = compute_c5(best_h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= stall_limit:
                break
    return best_h, best_c5

def grad_descent_smooth(h, n, T, lr_init, t_end, max_iter=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > t_end:
            break
        f_val = smooth_c5(best_h, T)
        g = smooth_c5_grad(best_h, T)
        h_new = project(best_h - lr * g, n)
        f_new = smooth_c5(h_new, T)
        if f_new < f_val - 1e-15:
            best_h = h_new
            c5_real = compute_c5(best_h)
            if c5_real < best_c5:
                best_c5 = c5_real
            lr = min(lr * 1.05, 0.4)
        else:
            lr *= 0.6
        if lr < 1e-10:
            break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=400):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def make_bump_init(n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    h[x < 0.7] = 1.0
    h[(x > 1.0) & (x < 1.8)] = 1.0
    return project(h, n)

def make_alt_init(n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.sign(np.sin(2 * np.pi * x)) + 0.1 * np.sin(6 * np.pi * x)
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        best_h = make_bump_init(n0)
    n = len(best_h)

    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent (12%)
    p1_end = t0 + budget_s * 0.12

    # Phase 2: Gradient descent on smooth max (10%)
    p2_end = t0 + budget_s * 0.22
    for T in [100, 500, 2000]:
        if time.time() > p2_end:
            break
        rem = min(p2_end - time.time(), 15)
        if rem < 3:
            break
        h_g, c5_g = grad_descent_smooth(best_h, n, T, 0.008, t0 + 3 + rem * 0.8)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g
        h_g, c5_g = coord_descent(h_g, n, min(time.time() + 5, p2_end), max_sweeps=500, stall_limit=40)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g

    # Phase 3: Coordinate descent (13%)
    p3_end = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, p3_end, max_sweeps=5000, stall_limit=80)

    # Phase 4: SLSQP with multiple T (20%)
    p4_end = t0 + budget_s * 0.55
    for T in [200, 1000, 5000, 20000]:
        if time.time() > p4_end:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=300)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
        if time.time() < p4_end - 3:

    # Phase 5: Annealing / diverse restarts (20%)
    p5_end = t0 + budget_s * 0.75

        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        if it % 5 == 2:
            T_g = 800
            g = smooth_c5_grad(best_h, T_g)
            h_up = project(best_h + 0.015 * g, n)
            h_try2, c5_try2 = coord_descent(h_up, n, min(time.time() + 6, p5_end), max_sweeps=200, stall_limit=30)
            if c5_try2 < best_c5:

                best_h = h_try2
        if it % 10 == 5:
            # Try structured perturbation
            h_struct = best_h.copy()
            # Shift phase of oscillation

            h_struct = project(h_struct, n)
