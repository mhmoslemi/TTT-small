import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    x = np.clip(T * (corr - m), -300, 300)
    return float((m + np.log(np.sum(np.exp(x))) / T) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n * 0.5
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff * room / tot, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff * h / tot, 0.0, 1.0)
    return h

def slsqp_opt(h, T, n, maxiter=500):
    def obj_fn(x):
        return smooth_obj(x, T)
    def con_fn(x):
        return x.sum() - n * 0.5
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_fn, h, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': con_fn}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    dx = 2.0 / n
    hi = h[i]
    h[i] = 0.0
    cb = get_corr(h)
    h[i] = hi
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dp = 0.0
        if k <= i:
            dp += (1.0 - h[i - k])
        if i <= n - 1 - k:
            dp -= h[i + k]
        d[n - 1 + k] = dp
        dm = 0.0
        if i <= n - 1 - k:
            dm += (1.0 - h[i + k])
        if k <= i:
            dm -= h[i - k]
        d[n - 1 - k] = dm
    c0 = cb[n - 1]
    def f1d(t):
        v = cb + t * d
        v[n - 1] = c0 + t - t * t
        return np.max(v) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return float(res.x)

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best = h.copy()
    best_c5 = compute_c5(best)
    stall = 0
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            t = coord_opt(best, n, i)
            if abs(t - best[i]) > 1e-11:
                best[i] = t
                best = project(best, n)
                c5 = compute_c5(best)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
            else:
                c5 = compute_c5(best)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
        if not improved:
            stall += 1
            if stall >= 4:
                break
        else:
            stall = 0
    return best, best_c5

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    x = np.clip(T * (corr - m), -300, 300)
    w = np.exp(x)
    w /= w.sum()
    grad = np.zeros(n)
    grad += w[n-1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k]
        wm = w[n - 1 - k]
        if wk > 1e-16:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        if wm > 1e-16:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n-k]
    return grad * dx

def proj_grad(h, n, t_end, T_init=50, T_end=5000, n_steps=2000):
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    conc = 1.0 / (1.0 + n)
    for it in range(n_steps):
        if time.time() > t_end:
            break
        frac = it / max(n_steps - 1, 1)
        T = T_init * (T_end / T_init) ** frac
        g = grad_smooth(h, T)
        # Backtracking
        obj = smooth_obj(h, T)
        alpha = 2.0 / T
        for _ in range(15):
            h_new = project(h - alpha * g, n)
            if smooth_obj(h_new, T) < obj:
                break
            alpha *= 0.5
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best, best_c5 = h.copy(), c5
    return best, best_c5

def basin_hop(h, n, t_end, rng, n_iters=300, base_step=0.06):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.95 * (it / max(n_iters, 1))
        step = base_step * (0.3 + 0.7 * cool)
        pert = project(work + step * rng.standard_normal(n), n)
        # Cheap local opt: a few CD sweeps
        hc, hc_c5 = coord_descent(pert, n, time.time() + 3.0, max_sweeps=8, rng=rng)
        temp = 0.005 * cool + 0.0003
        if hc_c5 < work_c5 or rng.random() < np.exp(-(hc_c5 - work_c5) / temp):
            work, work_c5 = hc, hc_c5
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h_old), n_new)

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    x = np.clip(T * (corr - m), -300, 300)
    return float((m + np.log(np.sum(np.exp(x))) / T) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n * 0.5
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff * room / tot, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff * h / tot, 0.0, 1.0)
    return h

def f1d_eval(t, cb, d, c0, n, dx):
    v = cb + t * d
    v[n - 1] = c0 + t - t * t
    return np.max(v) * dx

def coord_opt(h, n, i):
    dx = 2.0 / n
    hi = h[i]
    h[i] = 0.0
    cb = get_corr(h)
    h[i] = hi
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dp = 0.0
        if k <= i:
            dp += (1.0 - h[i - k])
        if i <= n - 1 - k:
            dp -= h[i + k]
        d[n - 1 + k] = dp
        dm = 0.0
        if i <= n - 1 - k:
            dm += (1.0 - h[i + k])
        if k <= i:
            dm -= h[i - k]
        d[n - 1 - k] = dm
    c0 = cb[n - 1]
    res = minimize_scalar(f1d_eval, bounds=(0.0, 1.0), method='bounded',
                          args=(cb, d, c0, n, dx), options={'xatol': 1e-13})
    return float(res.x)

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best = h.copy()
    best_c5 = compute_c5(best)
    stall = 0
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            t = coord_opt(best, n, i)
            if abs(t - best[i]) > 1e-11:
                best[i] = t
                best = project(best, n)
                c5 = compute_c5(best)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
        if not improved:
            stall += 1
            if stall >= 4:
                break
        else:
            stall = 0
    return best, best_c5

def slsqp_obj(x, T):
    return smooth_obj(x, T)

def slsqp_con(x, n):
    return x.sum() - n * 0.5

def slsqp_opt(h, T, n, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}]
    res = minimize(slsqp_obj, h, method='SLSQP', args=(T,),
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    x = np.clip(T * (corr - m), -300, 300)
    w = np.exp(x)
    w /= w.sum()
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k]
        wm = w[n - 1 - k]
        if wk > 1e-16:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        if wm > 1e-16:
            grad[:n - k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n - k]
    return grad * dx

def proj_grad(h, n, t_end, T_init=30, T_end=8000, n_steps=3000):
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_steps):
        if time.time() > t_end:
            break
        frac = it / max(n_steps - 1, 1)
        T = T_init * (T_end / T_init) ** frac
        g = grad_smooth(h, T)
        obj = smooth_obj(h, T)
        alpha = 2.0 / T
        for _ in range(15):
            h_new = project(h - alpha * g, n)
            if smooth_obj(h_new, T) < obj:
                break
            alpha *= 0.5
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best, best_c5 = h.copy(), c5
    return best, best_c5

def basin_hop(h, n, t_end, rng, n_iters=400, base_step=0.08):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.95 * (it / max(n_iters, 1))
        step = base_step * (0.3 + 0.7 * cool)
        pert = project(work + step * rng.standard_normal(n), n)
        hc, hc_c5 = coord_descent(pert, n, time.time() + 2.5, max_sweeps=8, rng=rng)
        temp = 0.005 * cool + 0.0003
        if hc_c5 < work_c5 or rng.random() < np.exp(-(hc_c5 - work_c5) / temp):
            work, work_c5 = hc, hc_c5
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h_old), n_new)
