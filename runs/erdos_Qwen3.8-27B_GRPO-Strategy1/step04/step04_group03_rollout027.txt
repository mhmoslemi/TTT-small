def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
            if not m.any():
                break
            h[m] += d / m.sum()
        else:
            m = h > 1e-15
            if not m.any():
                break
            h[m] *= 1.0 + d / h[m].sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_con_target = 0.0
def _sum_con(x):
    return x.sum() - _con_target

_obj_T = 1.0

    x = np.clip(x, 0, 1)
    c = np.correlate(x, 1.0 - x, mode='full')
