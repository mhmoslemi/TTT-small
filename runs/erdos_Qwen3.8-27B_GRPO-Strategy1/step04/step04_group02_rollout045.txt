def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def smooth_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m))))) * 2.0 / n / T * T

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h += r * (d / t)
        else:
            if s < 1e-14:
                break
            h *= target / s
        h = np.clip(h, 0.0, 1.0)

def slsqp_refine(h, T, n, maxiter=3000):

                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project(np.clip(res.x, 0, 1), n)

        return h
