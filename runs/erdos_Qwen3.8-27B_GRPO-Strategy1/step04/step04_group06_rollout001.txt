        if tot < 1e-14:
            return np.full(n, 0.5)
        h = h + room * ((target - s) / tot)
    clipped = np.clip(h, 0.0, 1.0)
    s2 = clipped.sum()
    if abs(s2 - target) > 1e-10:
        clipped = clipped * (target / s2)
        clipped = np.clip(clipped, 0.0, 1.0)
    return clipped

def coord_opt_step(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    cb = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        a[center + k] = (1.0 - h[ip] if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
        a[center - k] = (1.0 - h[im] if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
    scale = 2.0 / n
    ct = cb[center]
    best_x = 0.5
    best_v = np.inf
    for x in [0.0, 0.25, 0.5, 0.75, 1.0]:
        v = np.max(cb + x * a) * scale
        if v < best_v:
            best_v = v
            best_x = x
    res = minimize_scalar(lambda x: np.max(np.where(np.abs(x)<1e-15, cb+a*x, cb+a*x))[0:0] + 0 or 0, bounds=None)
    f1d = lambda x: np.max(cb + x * a)[0] * scale if isinstance(np.max(cb + x * a)[0], (int, float)) else (np.max(cb + x * a) * scale)
    # Simpler: compute max of the whole vector
    def f1d2(x):
        vals = cb.copy()
        vals += x * a
        vals[center] = ct + x - x * x
        return np.max(vals) * scale
    r = minimize_scalar(f1d2, bounds=(min(0.0, best_x - 0.3), max(1.0, best_x + 0.3)), method='bounded', options={'xatol': 1e-12})
    return np.clip(r.x, 0.0, 1.0)

def sensitivity_order(h, n, top_frac=0.3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    k_star = int(np.argmax(corr) - (n - 1))
    sens = np.zeros(n)
    for i in range(n):
        total = 0.0
        for k in [-1, 0, 1]:
            kk = k_star + k
            if kk == 0:
                total += abs(1.0 - 2.0 * h[i])
            elif kk > 0:
                if kk <= n - 1:
                    total += abs((1.0 - h[i+kk]) if (i+kk) < n else 0.0)
                    total += abs(h[i-kk]) if i-kk >= 0 else 0.0
            else:
                ma = -kk
                if ma <= n - 1:
                    total += abs((1.0 - h[i-ma]) if (i-ma) >= 0 else 0.0)
                    total += abs(h[i+ma]) if i+ma < n else 0.0
        sens[i] = total
    n_top = max(1, int(n * top_frac))
    top_idx = np.argsort(-sens)[:n_top]
    rest_idx = np.argsort(-sens)[n_top:]
    order = np.concatenate([top_idx, rest_idx])
    return order

def coord_descent(h, n, t_end, max_sweeps=200, use_sens_order=True):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = sensitivity_order(best_h, n) if use_sens_order and sweep % 5 == 0 else np.random.permutation(n)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            x_new = coord_opt_step(best_h, n, i)
            if abs(x_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = x_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_run(h, T, n, maxiter=500):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_val, h, jac=lambda x: smooth_max_grad(x, T),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 160
        x = np.linspace(0, 2, n_init, endpoint=False)
        best_h = np.clip(0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.sin(2 * np.pi * x), 0, 1)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick coordinate descent (8%)
    p1_end = t0 + budget_s * 0.08
    best_h, best_c5 = coord_descent(best_h, n, p1_end, max_sweeps=80, use_sens_order=True)

    # Phase 2: SLSQP with analytic gradient (22%)
    p2_end = t0 + budget_s * 0.30
    for T in [30, 100, 400, 1500, 5000]:
        if time.time() > p2_end:
            break
        hc = slsqp_run(best_h, T, n, maxiter=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Random perturbations + coordinate descent (25%)
    p3_end = t0 + budget_s * 0.55
    n_candidates = 6
    per_cand = (p3_end - time.time()) / max(1, n_candidates)
    for ci in range(n_candidates):
        if time.time() > p3_end:
            break
        sigma = 0.04 + 0.03 * ci
        h_pert = best_h + sigma * np.random.randn(n)
        h_pert = project(h_pert, n)
        # Quick SLSQP
        hc = slsqp_run(h_pert, 200, n, maxiter=200)
        # Coordinate descent
        hc, c5v = coord_descent(hc, n, min(time.time() + per_cand * 0.6, p3_end), max_sweeps=60)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Intensive coordinate descent (25%)
    p4_end = t0 + budget_s * 0.80
    best_h, best_c5 = coord_descent(best_h, n, p4_end, max_sweeps=1000, use_sens_order=True)

    # Phase 5: Final SLSQP polish (10%)
    p5_end = t0 + budget_s * 0.90
    for T in [20000, 80000]:
        if time.time() > p5_end:
            break
        hc = slsqp_run(best_h, T, n, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Final coordinate descent polish (10%)
    best_h, best_c5 = coord_descent(best_h, n, t_end - 1.0, max_sweeps=500, use_sens_order=True)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
