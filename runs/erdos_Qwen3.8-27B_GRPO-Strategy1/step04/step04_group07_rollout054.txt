    if diff > 0:

        tot = room.sum()
        if tot < 1e-13:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:

            return h

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    hs = h.copy()
    corr = np.correlate(hs, 1.0 - hs, mode='full')
    best_c5 = float(np.max(corr)) * dx
    ks = np.arange(-(n-1), n)
    c0 = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = hs[i]
            idx_a = i - ks
            idx_b = i + ks
            va = (idx_a >= 0) & (idx_a < n)
            vb = (idx_b >= 0) & (idx_b < n)
            coeff = np.zeros(2*n-1)
            coeff[va] += (1.0 - hs[idx_a[va]])
            coeff[vb] -= hs[idx_b[vb]]
            c0_base = corr[c0]
            
            # Find optimal v
            f0 = float(np.max(corr - old * coeff))
            f0 = float(np.max(np.where(np.arange(2*n-1)==c0, c0_base + (0-old)*(1-0-old), corr - old*coeff)))
            f1 = float(np.max(np.where(np.arange(2*n-1)==c0, c0_base + (1-old)*(1-1-old), corr + (1-old)*coeff)))

            bv, bf = res.x, res.fun

            if f1 < bf: bv, bf = 1.0, f1

                best_c5 = bf * dx

        if not improved:
