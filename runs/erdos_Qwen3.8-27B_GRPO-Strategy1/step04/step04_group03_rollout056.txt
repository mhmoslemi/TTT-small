        a, b = lo, hi
        for _ in range(28):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            if np.max(corr + m1 * coef) < np.max(corr + m2 * coef):
                b = m2
            else:
                a = m1
        s_new = (a + b) / 2.0
        if abs(s_new) > 1e-11:
            h[i] += s_new
            corr += s_new * coef
    z_new = np.max(corr)
    if z_new < z_old - 1e-14:
        improved = z_old - z_new
    return improved

def fourier_perturb(h, n, n_coeff, scale, rng):
    f = np.fft.rfft(h)
    idx = rng.choice(range(1, len(f)), size=min(n_coeff, len(f) - 1), replace=False)
    f[idx] += scale * rng.randn(len(idx)) * np.exp(1j * rng.uniform(0, 2 * np.pi, len(idx)))
    return np.fft.irfft(f, n=n)

def structured_perturb(h, n, rng, cool):
    step = 0.05 * cool + 0.02
    mask = rng.rand(n) < 0.25
    h_p = h.copy()
    h_p[mask] += step * rng.randn(mask.sum())
    h_p = project_to_sum(h_p, n / 2)
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s * 0.985

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=np.float64)
    else:
        best_h = np.zeros(224)
        best_h[:112] = 1.0
        best_h = project_to_sum(best_h, 112.0)

    n = len(best_h)
    ks = np.arange(-(n - 1), n)
    dx = 2.0 / n
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)
    work_h = best_h.copy()

    # Phase 1: SLSQP (5%)
    t_p1 = t0 + budget_s * 0.05
    for T in [30, 200, 2000]:
        if time.time() > t_p1:
            break
        try:
            hc = slsqp_polish(best_h, T, n, 400, 1e-13)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
                work_h = best_h.copy()
        except Exception:
            pass

    # Phase 2: Heavy CD (30%)
    t_p2 = t0 + budget_s * 0.35
    corr = compute_corr(best_h)
    while time.time() < t_p2:
        imp = cd_sweep(best_h, corr, n, ks)
        if imp < 1e-15:
            break
    best_c5 = np.max(corr) * dx
    work_h = best_h.copy()

    # Phase 3: Exploration with perturbations + CD (30%)
    t_p3 = t0 + budget_s * 0.65
    n_explore = 40
    for it in range(n_explore):
        if time.time() > t_p3:
            break
        cool = max(1.0 - it / float(n_explore), 0.05)
        r = rng.rand()
        if r < 0.45:
            h_p = fourier_perturb(work_h, n, int(4 + 8 * cool), 0.08 * cool + 0.02, rng)
        elif r < 0.75:
            h_p = structured_perturb(work_h, n, rng, cool)
        else:
            shift = int(rng.uniform(-0.03, 0.03) * n)
            h_roll = np.roll(work_h, shift)
            h_p = 0.6 * h_roll + 0.4 * work_h
        h_p = project_to_sum(h_p, n / 2)
        corr_p = compute_corr(h_p)
        for _ in range(5):
            cd_sweep(h_p, corr_p, n, ks)
        c5v = np.max(corr_p) * dx
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_p.copy()
            work_h = h_p.copy()
        elif c5v < best_c5 + 0.002 * cool:
            work_h = h_p.copy()

    # Phase 4: Targeted active-lag CD (20%)
    t_p4 = t0 + budget_s * 0.85
    corr = compute_corr(best_h)
    for outer in range(30):
        if time.time() > t_p4:
            break
        z = np.max(corr)
        top_M = np.argsort(corr)[-8:]
        h_work = best_h.copy()
        corr_work = corr.copy()
        for i in range(n):
            if time.time() > t_p4:
                break
            h_old = h_work[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            coef = make_coef(h_work, i, n, ks)
            a, b = lo, hi
            for _ in range(22):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(corr_work[top_M] + m1 * coef[top_M])
                f2 = np.max(corr_work[top_M] + m2 * coef[top_M])
                if f1 < f2:
                    b = m2
                else:
                    a = m1
                if np.max(corr_work + m1 * coef) > z + 1e-6:
                    break
            s_new = (a + b) / 2.0
            if abs(s_new) > 1e-11:
                h_work[i] += s_new
                corr_work += s_new * coef
        c5v = np.max(corr_work) * dx
        if c5v < best_c5 - 1e-13:
            best_h = h_work
            best_c5 = c5v
            work_h = h_work.copy()
            corr = corr_work.copy()
        else:
            break

    # Phase 5: Final heavy CD + SLSQP (15%)
    t_p5_start = max(time.time(), t0 + budget_s * 0.85)
    corr = compute_corr(best_h)
    t_cd_final = t_end - 30
    while time.time() < t_cd_final:
        imp = cd_sweep(best_h, corr, n, ks)
        if imp < 1e-15:
            break
    best_c5 = np.max(corr) * dx

    if time.time() < t_end - 5:
        for T in [5000, 50000]:
            if time.time() > t_end - 3:
                break
            try:
                hc = slsqp_polish(best_h, T, n, 500, 1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_h, best_c5 = hc, c5v
            except Exception:
                pass

    # Tiny final CD
    corr = compute_corr(best_h)
    while time.time() < t_end:
        cd_sweep(best_h, corr, n, ks)
    best_c5 = np.max(corr) * dx

    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
