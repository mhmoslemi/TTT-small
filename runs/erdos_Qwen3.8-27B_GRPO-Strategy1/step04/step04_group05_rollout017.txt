def lse_gradient(h, N, beta, p):
    """Compute gradient of LSE approximation of max correlation."""
    grad = np.zeros(N)
    center = N - 1
    grad += p[center] * (1.0 - 2.0 * h)
    for L in range(1, N):
        qL = p[center + L]
        rL = p[center - L]
        if qL > 1e-10:
            grad[L:] += qL * (1.0 - h[:N-L])
            grad[:N-L] -= qL * h[L:]
        if rL > 1e-10:
            grad[:N-L] += rL * (1.0 - h[L:])
            grad[L:] -= rL * h[:N-L]
    return grad

    """Continuous LSE optimization with Adam."""
    h = h.astype(float).copy()
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline and it < n_max:
        beta = beta0 + (beta1 - beta0) * min(1.0, it / (n_max * 0.7))
        corr = compute_corr(h)
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        # Softmax weights
        shifted = beta * (corr - float(np.max(corr)))
        exps = np.exp(shifted)
        p = exps / exps.sum()

        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        step = lr * mh / (np.sqrt(vh) + eps)
        h = h - step
        h = project_sum(h, target)
    return best_h, best_peak * (2.0 / N)

def project_sum(h, target):

    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:

        else:
            hi = mid

def lse_full_grad(h, N, p):
    """Full gradient for LSE-smoothed max correlation."""
    grad = np.zeros(N)
    c = N - 1
    grad += p[c] * (1.0 - 2.0 * h)
    for L in range(1, N):
        qL = p[c + L]
        rL = p[c - L]
        if qL > 1e-8:
            grad[L:] += qL * (1.0 - h[:N - L])
            grad[:N - L] -= qL * h[L:]
        if rL > 1e-8:
            grad[:N - L] += rL * (1.0 - h[L:])
            grad[L:] -= rL * h[:N - L]
    return grad

    m, v = np.zeros(N), np.zeros(N)
    it = 0
    while time.time() < deadline and it < n_max:
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (n_max * 0.6))
        corr = compute_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk:

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_full_grad(h, N, p):
    grad = np.zeros(N)
    c = N - 1
    grad += p[c] * (1.0 - 2.0 * h)
    for L in range(1, N):
        qL = p[c + L]
        rL = p[c - L]
        if qL > 1e-8:
            grad[L:] += qL * (1.0 - h[:N - L])
            grad[:N - L] -= qL * h[L:]
        if rL > 1e-8:
            grad[:N - L] += rL * (1.0 - h[L:])
            grad[L:] -= rL * h[:N - L]
    return grad

def lse_optimize(h, N, target, deadline, beta0=3.0, beta1=40.0, lr=0.007, n_max=5000):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.8, 0.999, 1e-8
    m, v = np.zeros(N), np.zeros(N)
    it = 0
    while time.time() < deadline and it < n_max:
        beta = beta0 + (beta1 - beta0) * min(1.0, it / (n_max * 0.55))
        corr = compute_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        sm = beta * (corr - float(corr.max()))
        p = np.exp(sm)
        p /= p.sum()
        g = lse_full_grad(h, N, p)
        it += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        h -= lr * (m / (1.0 - b1 ** it)) / (np.sqrt(v / (1.0 - b2 ** it)) + eps)
        h = project_sum(h, target)
    return best_h, best_pk * (2.0 / N)

def b_anneal(h, N, deadline, n_tr=4000, T0=1.5, T1=0.005):
    h_b = (h > 0.5).astype(float)
    nh = N // 2
    no = int(h_b.sum())
    if no > nh:
        h_b[np.where(h_b == 1)[0][nh:]] = 0
    elif no < nh:
        h_b[np.where(h_b == 0)[0][:nh - no]] = 1
    best, best_pk = h_b.copy(), float(np.max(compute_corr(h_b)))
    cur_pk = best_pk
    ones, zeros = np.where(h_b == 1)[0], np.where(h_b == 0)[0]
    for it in range(n_tr):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_tr))
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_b[i], h_b[j] = 0.0, 1.0
        npk = float(np.max(compute_corr(h_b)))
        if npk < cur_pk or (T > 0.004 and np.random.random() < np.exp(-(npk - cur_pk) / T)):
            cur_pk = npk
            if npk < best_pk:
                best_pk, best = npk, h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0, 0.0
        ones, zeros = np.where(h_b == 1)[0], np.where(h_b == 0)[0]
    return best, best_pk * (2.0 / N)

            else:

                cands_p = ones
            p = cands_p[np.random.randint(len(cands_p))]
            q = zeros[np.random.randint(len(zeros))]
            h_b[p], h_b[q] = 0.0, 1.0
            npk = float(np.max(compute_corr(h_b)))
            if npk < P:

            if not improved:

    if float(np.max(compute_corr(h_b))) < best_pk:
        best, best_pk = h_b, float(np.max(compute_corr(h_b)))
    return best, best_pk * (2.0 / N)

def make_tm(N):
    h = np.zeros(N)
    for i in range(N):
        h[i] = bin(i).count('1') & 1
    nh = N // 2
    no = int(h.sum())
    if no > nh:
        h[np.where(h == 1)[0][nh:]] = 0
    elif no < nh:
        h[np.where(h == 0)[0][:nh - no]] = 1
    return h

def make_blocks(N):
    h = np.zeros(N)
    period = max(4, N // 19)
    for i in range(N):
        pos = i % period
        h[i] = 1.0 if pos < period // 2 else 0.0
    nh = N // 2
    no = int(h.sum())
    if no > nh:
        h[np.where(h == 1)[0][nh:]] = 0
    elif no < nh:
        h[np.where(h == 0)[0][:nh - no]] = 1
    return h
