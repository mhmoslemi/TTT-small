def coord_step(h, n, i):
    """Optimize coordinate i, return new value."""
    old = h[i]
    c = corr_vec(h)
    # Compute delta vector for changing h[i] to t
    # For lag k (index m = n-1+k in full corr):
    # delta = (1-h[i+k]) if 0<=i+k<n, plus -h[i-k] if 0<=i-k<n
    # (excluding k=0 which is quadratic)
    d = np.zeros(2*n-1)
    for k in range(1, n):
        idx = n-1+k
        if 0 <= i+k < n:
            d[idx] += (1.0 - h[i+k])
        if 0 <= i-k < n:
            d[idx] += (-h[i-k])
        idx2 = n-1-k
        if 0 <= i+k < n:
            d[idx2] += (1.0 - h[i+k])
        if 0 <= i-k < n:
            d[idx2] += (-h[i-k])
    c0 = c[n-1]
    def f1d(t):
        vals = c + (t - old) * d
        vals[n-1] = c0 + t*(1.0-t) - old*(1.0-old)
        return np.max(vals) * (2.0/n)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_step(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-10:
                best_h[i] = hi
                best_h = project(best_h, n)
                best_c5 = compute_c5(best_h)
                improved = True
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return best_h, best_c5

def basin_hop(h, n, t_end, rng, n_iters=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    work = h.copy()
    work_c5 = best_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - it/max(n_iters,1)
        step = 0.03 * (0.2 + 0.8*cool)
        # Diverse perturbation
        r = rng random()
        if r < 0.3:
            # Gaussian
            pert = work + step * rng.standard_normal(n)
        elif r < 0.6:
            # Binary flip
            pert = work.copy()
            idx = rng.integers(0, n, size=int(n*0.05))
            pert[idx] = np.clip(pert[idx] + rng.choice([-1,1], len(idx))*0.3, 0, 1)
        else:
            # Block shift
            pert = work.copy()
            bstart = rng.integers(0, n-20)
            bsize = rng.integers(5, 25)
            shift = rng.integers(-10, 10)
            pert[bstart:bstart+bsize] = np.roll(pert[bstart:bstart+bsize], shift)
        pert = project(pert, n)
        try:
            T = 200 + 3000*cool
            hc = slsqp(pert, T, n, maxiter=150)
        except:
            hc = pert
        cv = compute_c5(hc)
        temp = 0.002*cool + 0.0002
        if cv < work_c5 or rng.random() < np.exp(-max(0,(cv-work_c5))/temp):
            work, work_c5 = hc, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * (2.0 / n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (1.0 - (-diff) / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-13:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-13:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-13 and s2 > 1e-13:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def make_sym(hh, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = hh
    h[m:] = 1.0 - hh[::-1]
    return h

def slsqp_sym(hh, T, n, maxiter=400):
    m = n // 2
    def obj(x):
        hf = make_sym(np.clip(x, 0, 1), n)
        return smooth_obj(hf, T)
    try:
        res = minimize(obj, np.clip(hh, 0, 1), method='SLSQP',
                       bounds=[(0, 1)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return np.clip(res.x, 0, 1)
    except Exception:
        return hh.copy()

def slsqp_full(h0, T, n, maxiter=400):
    def obj(x):
        return smooth_obj(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': coneq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0.copy()

def coord_step(h, n, i):
    old = h[i]
    c = np.correlate(h, 1.0 - h, mode='full')
    d = np.zeros(2 * n - 1)
    k = np.arange(1, n)
    ik = i + k
    im = i - k
    mk = (ik >= 0) & (ik < n)
    mm = (im >= 0) & (im < n)
    vk = np.zeros(n - 1)
    vm = np.zeros(n - 1)
    vk[mk] = 1.0 - h[ik[mk]]
    vm[mm] = -h[im[mm]]
    d[n:] = vk + vm
    d[:n - 1] = vk + vm
    c0 = c[n - 1]
    def f1d(t):
        vals = c + (t - old) * d
        vals[n - 1] = c0 + t * (1.0 - t) - old * (1.0 - old)
        return float(np.max(vals) * (2.0 / n))
    try:
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        return float(res.x)
    except Exception:
        return old

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_step(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-10:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5:
                    best_c5 = c5
                improved = True
        stall = 0 if improved else stall + 1
        if stall >= 4:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, rng, n_iters=120):
    best = h.copy()
    best_c5 = compute_c5(best)
    work = h.copy()
    work_c5 = best_c5
    m = n // 2
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iters, 1)
        step = 0.04 * (0.15 + 0.85 * cool)
        r = rng.random()
        if r < 0.35:
            pert = work + step * rng.standard_normal(n)
        elif r < 0.60:
            pert = work.copy()
            idx = rng.integers(0, n, size=max(3, n // 20))
            pert[idx] = np.clip(pert[idx] + rng.choice([-1, 1], len(idx)) * 0.35, 0, 1)
        elif r < 0.80:
            hh = work[:m].copy()
            idx = rng.integers(0, m, size=max(2, m // 15))
            hh[idx] = np.clip(hh[idx] + step * 2 * rng.standard_normal(len(idx)), 0, 1)
            pert = make_sym(hh, n)
        else:
            pert = work.copy()
            bs = int(rng.integers(0, n - 30))
            bl = int(rng.integers(8, 30))
            sh = int(rng.integers(-10, 10))
            seg = pert[bs:bs + bl].copy()
            pert[bs:bs + bl] = np.roll(seg, sh)
            pert = project(pert, n)
        pert = project(pert, n)
        if time.time() > t_end:
            break
        T_use = 300 + 5000 * cool
        cc = rng.random()
        if cc < 0.4:
            hh = pert[:m]
            hh_new = slsqp_sym(hh, T_use, n, maxiter=120)
            cand = project(make_sym(hh_new, n), n)
        else:
            cand = slsqp_full(pert, T_use, n, maxiter=120)
        cv = compute_c5(cand)
        temp = 0.003 * cool + 0.0003
        if cv < work_c5 or rng.random() < np.exp(-max(0.0, cv - work_c5) / temp):
            work, work_c5 = cand, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interpolate(h, n_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end_total = t0 + budget_s * 0.98

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = project(np.array(initial_h_values, dtype=float), len(initial_h_values))
    else:
        n0 = 240
        best_h = np.zeros(n0)
        best_h[:n0 // 2] = 1.0
        best_h = project(best_h, n0)
    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with symmetry (fast, fewer vars)
    for T in [10, 50, 200, 800, 3000, 10000, 40000]:
        if time.time() > t0 + budget_s * 0.12:
            break
        m = n // 2
        hh = best_h[:m].copy()
        hh_new = slsqp_sym(hh, T, n, maxiter=500)
        cand = project(make_sym(hh_new, n), n)
        cv = compute_c5(cand)
        if cv < best_c5:
            best_h, best_c5 = cand, cv
        # Also try full SLSQP with symmetry-breaking
        cand2 = slsqp_full(best_h, T, n, maxiter=200)
        cv2 = compute_c5(cand2)
        if cv2 < best_c5:
            best_h, best_c5 = cand2, cv2

    # Phase 2: Coordinate descent
    t_end = t0 + budget_s * 0.32
    best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=150, rng=rng)

    # Phase 3: Basin hopping
    t_end = t0 + budget_s * 0.58
    best_h, best_c5 = basin_hop(best_h, n, t_end, rng, n_iters=100)

    # Phase 4: Post-basin coordinate descent
    t_end = t0 + budget_s * 0.65
    best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=60, rng=rng)

    # Phase 5: Multi-resolution
    t_end = t0 + budget_s * 0.85
    for n_try in [200, 280, 320, 160, 240]:
        if time.time() > t_end:
            break
        if n_try == n:
            continue
        h_interp = interpolate(best_h, n_try)
        for T in [500, 5000, 30000]:
            if time.time() > t_end:
                break
            m2 = n_try // 2
            hh = h_interp[:m2].copy()
            hh_new = slsqp_sym(hh, T, n_try, maxiter=300)
            cand = project(make_sym(hh_new, n_try), n_try)
            cv = compute_c5(cand)
            if cv < best_c5:
                best_h, best_c5, n = cand, cv, n_try
                h_interp = interpolate(best_h, n_try)
        if n == n_try and time.time() < t_end:
            best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=30, rng=rng)

    # Phase 6: Final polish
    t_end = min(t_end_total, t0 + budget_s * 0.96)
    best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=60, rng=rng)
    for T in [30000, 100000, 300000]:
        if time.time() > t_end_total:
            break
        m = n // 2
        hh = best_h[:m].copy()
        hh_new = slsqp_sym(hh, T, n, maxiter=600)
        cand = project(make_sym(hh_new, n), n)
        cv = compute_c5(cand)
        if cv < best_c5:
            best_h, best_c5 = cand, cv
        best_h, best_c5 = coord_descent(best_h, n, t_end_total, max_sweeps=20, rng=rng)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
