def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if s < 1e-14:
        return np.full_like(h, target / len(h))
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def golden_min_max(a, b, lo, hi, niter=50):

    if hi - lo < 1e-14:

def min_upper_envelope(a, b, lo, hi, n):
    # Minimize max_k(a[k] + b[k]*x) for x in [lo, hi]
    # a, b: arrays of length n (number of lags)
    # Returns the x that minimizes the max
    
    # Create (slope, intercept) pairs
    # Sort by slope
    order = np.argsort(b)
    b_s = b[order]
    a_s = a[order]
    
    # Build upper convex hull (for max of lines)
    # A line (b_i, a_i) is on the upper hull if it's above all other lines at some x
    # For max, we want the envelope from below... 
    
    # Actually, for MAX of lines, the upper envelope:
    # Line i is on the upper envelope where it's the maximum
    # The vertices of the upper envelope are intersection points of consecutive lines
    
    # For the MAX function, the relevant convex hull is the upper hull
    # Process lines sorted by slope, keep those that form the upper envelope
    
    hull = []  # list of indices into sorted arrays
    for i in range(n):
        while len(hull) >= 2:
            # Check if the last line in hull is dominated
            j = hull[-1]
            k = hull[-2]
            # Intersection of k and j: (a_j - a_k) / (b_k - b_j)
            # Intersection of k and i: (a_i - a_k) / (b_k - b_i)
            # j is dominated if intersection(k,j) >= intersection(k,i)
            if b_k == b_j:  # parallel
                if a_j <= a_k:
                    hull.pop()  # j is dominated by k
                    continue
                else:
                    break
            x_kj = (a[j] - a[k]) / (b[k] - b[j])
            if b[i] == b[k]:
                if a[i] <= a[k]:
                    continue  # i is dominated
                else:
                    break
            x_ki = (a[i] - a[k]) / (b[k] - b[i])
            if x_kj >= x_ki:
                hull.pop()
            else:
                break
        hull.append(i)
    
    # Find the minimum of the upper envelope in [lo, hi]
    # The upper envelope is max of lines, a convex piecewise-linear function
    # Its minimum is at the "valley" - where the slope changes from negative to positive
    # Or at lo or hi
    
    def f(x):
        return np.max(a + b * x)
    
    best_x = lo
    best_f = f(lo)
    
    # Check vertices of upper envelope in [lo, hi]
    for idx in range(len(hull) - 1):
        i = hull[idx]
        j = hull[idx + 1]
        if b[i] == b[j]:
            continue
        x_cross = (a[j] - a[i]) / (b[i] - b[j])
        if lo <= x_cross <= hi:
            fx = f(x_cross)
            if fx < best_f:
                best_f = fx
                best_x = x_cross
    
    # Check hi
    fh = f(hi)
    if fh < best_f:
        best_x = hi
    
    return best_x

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if s < 1e-14:
        return np.full_like(h, target / len(h))
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if s < 1e-14:
        return np.full(len(h), target / len(h))
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def golden_min(a, b, lo, hi, niter=35):
    if hi - lo < 1e-14:
        return lo
    gr = 0.6180339887498949
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = float(np.max(a + b * x1))
    f2 = float(np.max(a + b * x2))
    for _ in range(niter):
        if f1 <= f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = float(np.max(a + b * x1))
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = float(np.max(a + b * x2))
    mid = 0.5 * (lo + hi)
    fm = float(np.max(a + b * mid))
    if fm <= f1:
        return mid if fm <= f2 else x1
    return x1 if f1 <= f2 else x2

def get_coef_vec(h, i, n):
    kvals = np.arange(-(n - 1), n)
    j1 = i - kvals
    j2 = i + kvals
    m1 = (j1 >= 0) & (j1 < n)
    m2 = (j2 >= 0) & (j2 < n)
    j1i = j1[m1]
    j2i = j2[m2]
    coef = np.zeros(2 * n - 1)
    coef[m1] = 1.0 - h[j1i]
    coef[m2] -= h[j2i]
    return coef

def coord_descent(h, n, t_end, n_sweeps, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for ii in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            i = int(order[ii])
            h_old = best_h[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-12:
                continue
            coef = get_coef_vec(best_h, i, n)
            xopt = golden_min(corr, coef, lo, hi)
            if abs(xopt) > 1e-10:
                nv = h_old + xopt
                if nv < 0.0:
                    nv = 0.0
                elif nv > 1.0:
                    nv = 1.0
                s = nv - h_old
                best_h[i] = nv
                corr += s * coef
                c5n = float(np.max(corr) * dx)
                if c5n < best_c5:
                    best_c5 = c5n
                    improved = True
        if sweep % 16 == 15:
            best_h = project_to_sum(best_h, n / 2.0)
            corr = compute_corr(best_h)
            best_c5 = compute_c5(best_h)
        if not improved:
            break
    return best_h, best_c5

def _sm_obj(x, T, n):
    dx = 2.0 / n
    return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx

def _sm_con(x, n):
    return x.sum() - n / 2.0

def slsqp_step(h, n, T, maxiter=400, ftol=1e-13):
    res = minimize(_sm_obj, h, method='SLSQP', args=(T, n),
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _sm_con, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def pair_refine(h, n, tend, rng):
    bh = h.copy()
    bc = compute_c5(bh)
    dx = 2.0 / n
    cr = compute_corr(bh)
    improved_any = True
    while improved_any and time.time() < tend:
        improved_any = False
        # Try all pairs (or a random subset for speed)
        n_tries = min(5000, n*(n-1)//2)
        for _ in range(n_tries):
            if time.time() > tend:
                break
            i, j = rng.choice(n, 2, replace=False)
            di, dj = bh[i], bh[j]
            if abs(di-dj) < 1e-12:
                continue
            # Swap effect on corr:
            # delta = (dj-di)*coef_i - (di-dj)*coef_j = (dj-di)*(coef_i - coef_j)
            co_i = get_coef(bh, i, n)
            co_j = get_coef(bh, j, n)
            # After swap: corr_new = corr + (dj-di)*co_i + (di-dj)*co_j
            # = corr + (dj-di)*(co_i - co_j)
            delta = (dj-di)*(co_i - co_j)
            cr_new = cr + delta
            c5n = float(np.max(cr_new) * dx)
            if c5n < bc - 1e-14:
                bh[i], bh[j] = dj, di
                cr = cr_new
                bc = c5n
                improved_any = True
