        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

        return 0.6 * h_sh + 0.4 * h
    elif r < 0.75:
        k = int(n * 0.15)
        idx = rng.choice(n, size=k, replace=False)
        h2 = h.copy()
        h2[idx] = 1.0
        return 0.5 * h2 + 0.5 * h
    else:
        step = 0.02 + 0
.05 * cool
        h2 = h.copy()
        mask = rng.random(n) < 0.25
        h2[mask] += step * rng.standard_normal(mask.sum())
        return h2

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=np.float64)
    else:
        best_h = np.full(224, 0.5)
    
    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: intensive coordinate descent (25% budget)
    t1 = t0 + budget_s * 0.25
    best_h, best_c5 = fast_coord_desc(best_h, t1, max_sweeps=3000)
    
    # Phase 2: basin hopping with fast local opt (40% budget)
    t2 = t0 + budget_s * 0.65
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 0
    while elapsed() < t2:
        n_iter += 1
        cool = max(1.0 - 0.95 * n_iter / 200.0, 0.02)
        h_p = perturb(work_h, len(work_h), rng, cool)
        h_p = project_to_sum(h_p, len(h_p) / 2.0)
        t_sub = min(time.time() + 3.0 * cool + 0.5, t2 - 0.1)
        h_c, c5_c = fast_coord_desc(h_p, t_sub, max_sweeps=80)
        temp = 0.002 * cool + 1e-5
        if c5_c < work_c5 or rng.random() < np.exp(-(c5_c - work_c5) / max(temp, 1e-10)):
            work_h, work_c5 = h_c, c5_c
            if c5_c < best_c5:
                best_h, best_c5 = h_c.copy(), c5_c
    
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 3: multi-resolution (15% budget)
    t3 = t0 + budget_s * 0.80
    best_n = len(best_h)
    for n_try in [192, 256, 288, 320, 160]:
        if elapsed() > t3:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2.0)
        t_sub = min(time.time() + budget_s * 0.04, t3 - 0.1)
        h_c, c5_c = fast_coord_desc(h_i, t_sub, max_sweeps=200)
        if c5_c < best_c5:
            best_h, best_c5, best_n = h_c, c5_c, n_try
    
    # Phase 4: subgradient refinement + final polish (20% budget)
    t4 = t0 + budget_s * 0.92
    corr = compute_corr(best_h)
    n_cur = len(best_h)
    for sg_iter in range(200):
        if time.time() > t4:
            break
        lr = 0.003 * max(1.0 - 0.5 * sg_iter / 200.0, 0.1) / n_cur
        h_sg = subgrad_step(best_h, corr, lr, topk=10)
        h_c, c5_c = fast_coord_desc(h_sg, min(time.time() + 1.0, t4), max_sweeps=50)
        if c5_c < best_c5:
            best_h, best_c5 = h_c, c5_c
            corr = compute_corr(best_h)
    
    # Final polish
    t5 = t0 + budget_s * 0.99
    best_h, best_c5 = fast_coord_desc(best_h, t5, max_sweeps=500)
    
    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
