            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 3: Coordinated diversified search (25% budget)
    t_phase3 = t0 + budget_s * 0.60
    n_tries = 12
    for trial in range(n_tries):
        if elapsed() > t_phase3:
            break
        cool = 1.0 - trial / float(n_tries)
        # Diversify
        if trial % 3 == 0:
            noise = (0.05 + 0.15 * cool) * np.random.randn(n)
            h_pert = best_h + noise
        elif trial % 3 == 1:
            # Random reordering of mass
            h_pert = best_h.copy()
            mask = np.random.random(n) < 0.3
            h_pert[mask] = np.random.uniform(0, 1, mask.sum())
        else:
            # Scale variation
            h_pert = best_h * (0.7 + 0.3 * np.random.random(n))
        h_pert = project_box_sum(h_pert, n / 2.0)
        # Optimize from perturbed start
        t_sub = min(t_phase3, time.time() + budget_s * 0.02)
        hc, c5_sub = pg_optimize(h_pert, n, t_sub,
                                 T_start=10.0, T_end=10000.0, lr_init=0.2, max_iter=20000)
        # Polish with CD
        t_sub2 = min(t_phase3, time.time() + budget_s * 0.012)
        hc, c5_sub = coord_descent_fast(hc, n, t_sub2, max_sweeps=200)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = hc

    # Phase 4: Intensive coordinate descent (25% budget)
    t_phase4 = t0 + budget_s * 0.85
    best_h, best_c5 = coord_descent_fast(best_h, n, t_phase4, max_sweeps=5000)

    # Phase 5: Final SLSQP polish (10% budget)
    t_phase5 = t0 + budget_s * 0.95
    for T in [10000, 50000, 200000]:
        if elapsed() > t_phase5:
            break
        try:
            hc = slsqp_soft(best_h.copy(), T, n, maxiter=300, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    # Final CD
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.99, max_sweeps=500)

    best_h = project_box_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
