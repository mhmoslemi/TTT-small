```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_hgrad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ev = np.exp(T * (corr - m))
    sw = ev.sum()
    w = ev / sw
    val = dx * (m + np.log(sw)) / T
    
    idx = np.arange(n)[:, None]
    lags = np.arange(-(n-1), n)[None, :]
    pos_a = idx + lags
    pos_b = idx - lags
    valid_a = (pos_a >= 0) & (pos_a < n)
    valid_b = (pos_b >= 0) & (pos_b < n)
    h_a = np.where(valid_a, h[np.clip(pos_a, 0, n-1)], 0.0)
    h_b = np.where(valid_b, h[np.clip(pos_b, 0, n-1)], 0.0)
    grad = dx * (1.0 - h_a @ w - h_b @ w)
    return val, grad

def slsqp_opt(h, T, n, maxiter=2000, ftol=1e-16):
    def obj(x):
        return smooth_c5_hgrad(x, T, n)[0]
    def jac(x):
        return smooth_c5_hgrad(x, T, n)[1]
    def con(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def proj_grad(h, T, n, n_steps=200, lr0=0.01):
    """Projected gradient with momentum."""
    best_h = h.copy()
    c5 = compute_c5(best_h)
    momentum = np.zeros(n)
    for step in range(n_steps):
        val, grad = smooth_c5_hgrad(best_h, T, n)
        momentum = 0.9 * momentum - lr0 * grad
        new_h = best_h + momentum
        new_h = project_h(new_h, n)
        c5_new = compute_c5(new_h)
        if c5_new < c5:
            c5 = c5_new
            best_h = new_h
    return best_h, c5

def coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n-1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n-1)])
    dx = 2.0 / n
    def f1d(hi):
        return np.max(corr_base + hi * coefs) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_hgrad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ev = np.exp(T * (corr - m))
    sw = ev.sum()
    w = ev / sw
    val = dx * (m + np.log(sw)) / T
    idx_i = np.arange(n)[:, None]
    idx_j = np.arange(2*n-1)[None, :]
    pos1 = idx_i - idx_j + (n-1)
    pos2 = idx_i + idx_j - (n-1)
    valid1 = (pos1 >= 0) & (pos1 < n)
    valid2 = (pos2 >= 0) & (pos2 < n)
    vf1 = valid1.astype(np.float64)
    vf2 = valid2.astype(np.float64)
    A = vf1 @ w
    B = (vf1 * h[np.clip(pos1, 0, n-1)]) @ w
    C = (vf2 * h[np.clip(pos2, 0, n-1)]) @ w
    grad = dx * (A - B - C)
    return val, grad

def slsqp_opt(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_c5_hgrad(x, T, n)[0]
    def jac(x):

    def con(x):
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_hgrad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ev = np.exp(T * (corr - m))
    sw = ev.sum()
    w = ev / sw
    val = dx * (m + np.log(sw)) / T
    idx_i = np.arange(n)[:, None]
    idx_j = np.arange(2 * n - 1)[None, :]
    pos1 = idx_i - idx_j + (n - 1)
    pos2 = idx_i + idx_j - (n - 1)
    valid1 = (pos1 >= 0) & (pos1 < n)
    valid2 = (pos2 >= 0) & (pos2 < n)
    vf1 = valid1.astype(np.float64)
    vf2 = valid2.astype(np.float64)
    A = vf1 @ w
    B = (vf1 * h[np.clip(pos1, 0, n - 1)]) @ w
    C = (vf2 * h[np.clip(pos2, 0, n - 1)]) @ w
    grad = dx * (A - B - C)
    return val, grad
```
