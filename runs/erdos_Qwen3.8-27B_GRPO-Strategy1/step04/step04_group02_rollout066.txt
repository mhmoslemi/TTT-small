    mx = np.max(corr)

def coord_opt(h, n, i, T=500):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if 0 <= i - k < n:
            d[n - 1 + k] += (1.0 - h[i - k])
            d[n - 1 - k] += (-h[i - k])
        if 0 <= i + k < n:
            d[n - 1 + k] += (-h[i + k])
            d[n - 1 - k] += (1.0 - h[i + k])

    # When h[i] changes from 0 to t: corr[n-1] changes by t(1-t) - 0 = t(1-t)

    # corr_base[n-1] = sum_{j≠i} h[j](1-h[j]) (since h[i]=0 contributes 0)
    # When h[i]=t: corr[n-1] = corr_base[n-1] + t(1-t)

        vals[n - 1] += t * (1.0 - t)
        m = np.max(vals)

- For each coordinate i, fix all others and solve the 1D problem optimally (minimize the max over lags)

The 1D subproblem: given h with h[i] fixed to t, all other values fixed, minimize max_k corr_k(t).

corr_k(t) is linear in t (except at k=0 where it has a t(1-t) term).
So the subproblem is: min_t max_k (a_k + b_k * t) [with special k=0 term]

This can be solved exactly! The max of linear functions is convex, so we can find the minimizer.

For the linear case (k≠0): min_t max_{k≠0} (a_k + b_k*t)
This is a 1D convex piecewise-linear minimization. The optimum is where the "active" lines cross.

For k=0: the term is a_0 + t(1-t) = a_0 + t - t², which is concave in t.
So the overall max is max(concave, lines) which is not convex in general.

But we can still solve it numerically with bisection or golden section in O(log(precision)) steps, evaluating the max in O(n) each time. Total: O(n log(1/ε)) per coordinate. Very fast.

    def c_eq(x):
        return x.sum() - n / 2.0
    def c_eq_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_eq_j}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt_exact(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').astype(float)
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if i - k >= 0:
            d[n - 1 + k] += (1.0 - h[i - k])
        if i + k < n:
            d[n - 1 + k] += (-h[i + k])
        if i + k < n:
            d[n - 1 - k] += (1.0 - h[i + k])
        if i - k >= 0:
            d[n - 1 - k] += (-h[i - k])
    best_t = 0.0
    best_val = np.inf
    ts = np.linspace(0.0, 1.0, 300)
    for t in ts:
        vals = corr_base + t * d
        vals[n - 1] += t * (1.0 - t)
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_t = t
    lo = max(0.0, best_t - 1.0 / 300.0)
    hi = min(1.0, best_t + 1.0 / 300.0)
    if lo >= hi:
        return best_t
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] += t * (1.0 - t)
        return np.max(vals)
    res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                          options={'xatol': 1e-14})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=50, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = list(range(n))
        if sweep > 0:
            rng.shuffle(order)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            t_opt = coord_opt_exact(best_h, n, i)
            if abs(t_opt - best_h[i]) > 1e-10:
                best_h[i] = t_opt
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with analytical gradients, increasing T
    for T in [10, 50, 200, 1000, 5000, 20000]:
        if elapsed() > budget_s * 0.18:
            break
        hc = slsqp_run(best_h, T, n, maxiter=400, ftol=1e-16)
        cv = compute_c5(hc)
        if cv < best_c5 - 1e-15:
            best_h, best_c5 = hc, cv

    # Phase 2: Coordinate descent with exact 1D optimization
    t_cd = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, n, t_cd, max_sweeps=80, rng=rng)

    # Phase 3: SLSQP re-polish after coordinate descent
    for T in [5000, 50000, 200000]:
        if elapsed() > budget_s * 0.65:
            break
        hc = slsqp_run(best_h, T, n, maxiter=300, ftol=1e-16)
        cv = compute_c5(hc)
        if cv < best_c5 - 1e-15:
            best_h, best_c5 = hc, cv

    # Phase 4: Multi-resolution search
    for n_try in [160, 200, 320]:
        if elapsed() > budget_s * 0.80:
            break
        h_int = interpolate_h(best_h, n_try)
        for T in [500, 5000, 50000]:
            if elapsed() > budget_s * 0.82:
                break
            hc = slsqp_run(h_int, T, n_try, maxiter=200, ftol=1e-15)
            cv = compute_c5(hc)
            if cv < best_c5 - 1e-15:
                best_h, best_c5, n = hc, cv, n_try
                h_int = hc
        if elapsed() < budget_s * 0.84:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.84, max_sweeps=25, rng=rng)

    # Phase 5: Light perturb-and-reoptimize
    if elapsed() < budget_s * 0.93:
        work = best_h.copy()
        work_c5 = compute_c5(work)
        for it in range(40):
            if elapsed() > budget_s * 0.93:
                break
            cool = 1.0 - 0.9 * (it / 40.0)
            step = 0.025 * (0.3 + 0.7 * cool)
            pert = work + step * rng.standard_normal(n)
            pert = project(pert, n)
            hc = slsqp_run(pert, 3000, n, maxiter=100, ftol=1e-14)
            cv = compute_c5(hc)
            temp = 0.002 * cool + 0.0003
            if cv < work_c5 or rng.random() < np.exp(-(cv - work_c5) / temp):
                work, work_c5 = hc, cv
            if work_c5 < best_c5 - 1e-15:
                best_h, best_c5 = work.copy(), work_c5

    # Phase 6: Final coordinate descent
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=30, rng=rng)

    # Final SLSQP polish
    if elapsed() < budget_s * 0.995:
        hc = slsqp_run(best_h, 100000, n, maxiter=500, ftol=1e-16)
        cv = compute_c5(hc)
        if cv < best_c5 - 1e-15:
            best_h, best_c5 = hc, cv

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
