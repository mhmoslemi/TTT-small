This is a zero-sum game if we consider: Player 1 chooses h, Player 2 chooses k. But h is continuous and k is discrete, and the "payoff" c_k(h) is not linear in h (it's quadratic: h[i+k] - h[i+k]*h[i]).

So it's not a standard LP. The quadratic terms make it a min-max QP, which is generally hard.

However, if we fix the "active set" of k values, we get a convex problem:
min_h max_{k ∈ S} c_k(h) = min_h { t : c_k(h) ≤ t ∀ k ∈ S }

And each c_k(h) is the sum of bilinear terms. The maximum of bilinear functions is not convex.

So the problem is genuinely non-convex. The coordinate descent approach makes sense.

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    dx = 2.0 / n
    m = 2 * n - 1
    corr = compute_corr(best_h)
    best_c5 = float(np.max(corr)) * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            h_old = best_h[i]
            coef = np.zeros(m)
            for k in range(-(n-1), n):
                c = 0.0
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    c += 1.0 - best_h[j1]
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[k + n - 1] = c
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            def f1d(d):
                return np.max(corr + d * coef)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-11})
            s_new = res.x
            if abs(s_new) > 1e-13:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = float(np.max(corr)) * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_swap_search(h, n, t_end, n_iters=20000):
    """Local search on binary structure: swap 0s and 1s."""
    h_bin = (h > 0.5).astype(float)
    h_bin = project_to_sum(h_bin, n / 2.0)
    best_h = h_bin.copy()
    corr = compute_corr(best_h)
    best_val = float(np.max(corr))
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() > t_end:
            break
        i = np.random.randint(n)
        if best_h[i] < 0.5:
            # find a 1 to swap with
            ones = np.where(best_h > 0.5)[0]
            if len(ones) == 0:
                break
            j = ones[np.random.randint(len(ones))]
        else:
            zeros = np.where(best_h < 0.5)[0]
            if len(zeros) == 0:
                break
            j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        # Try the swap
        best_h[i], best_h[j] = best_h[j], best_h[i]
        corr_new = compute_corr(best_h)
        val_new = float(np.max(corr_new))
        if val_new < best_val:
            best_val = val_new
            corr = corr_new
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]
    return best_h, best_val * dx

def anti_corr_perturb(h, corr, n, strength):
    """Perturb h in a direction that reduces the highest lags."""
    m = 2 * n - 1
    top_lags = np.argsort(corr)[-min(5, m):]
    grad = np.zeros(n)
    for lag in top_lags:
        k = lag - (n - 1)
        for i in range(n):
            j1 = i - k
            j2 = i + k
            if 0 <= j1 < n:
                grad[i] += 1.0 - h[j1]
            if 0 <= j2 < n:
                grad[i] -= h[j2]
    grad -= np.mean(grad)
    return h + strength * grad / (np.max(np.abs(grad)) + 1e-15)

def slsqp_smooth(h, T, n, maxiter=500, ftol=1e-13):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        vals = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(vals)
        return mx + np.log(np.sum(np.exp(T * (vals - mx)))) / T * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def structured_init(n, pattern='random'):
    """Create structured initial guesses."""
    h = np.zeros(n)
    if pattern == 'square':
        h[:n//2] = 1.0
    elif pattern == 'alternating':
        h[::2] = 1.0
        if n % 2 == 1 and h.sum() < n // 2:
            h[-1] = 1.0
    elif pattern == 'sine':
        h = 0.5 + 0.49 * np.sin(2 * np.pi * np.arange(n) * 2.0 / n)
    elif pattern == 'sine3':
        h = 0.5 + 0.49 * np.cos(2 * np.pi * np.arange(n) * 3.0 / n)
    elif pattern == 'random_bin':
        idx = np.random.permutation(n)
        h[idx[:n//2]] = 1.0
    h = project_to_sum(h, n / 2.0)
    return h

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_i = np.interp(x_new, x_old, h_old)
    return project_to_sum(h_i, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    def remaining():
        return max(t_end - time.time(), 0)

    # Get initial h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=np.float64)
    else:
        best_h = structured_init(224, 'square')
    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick coordinate descent from initial (15% budget)
    t1 = t0 + budget_s * 0.15
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=200)

    # Phase 2: SLSQP polish with increasing T (10% budget)
    t2 = t0 + budget_s * 0.25
    for T in [50, 200, 1000, 5000]:
        if time.time() > t2:
            break
        try:
            hc = slsqp_smooth(best_h, T, n, maxiter=400, ftol=1e-13)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 3: Basin hopping with anti-correlation perturbations (25% budget)
    t3 = t0 + budget_s * 0.50
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 40
    for it in range(n_bh):
        if time.time() > t3:
            break
        cool = 1.0 - 0.9 * (it / float(n_bh))
        r = np.random.rand()
        if r < 0.35:
            # Anti-correlation perturbation
            corr = compute_corr(work_h)
            step = 0.03 + 0.10 * cool
            h_p = anti_corr_perturb(work_h, corr, n, step)
        elif r < 0.55:
            # Random Gaussian
            step = 0.02 + 0.08 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.75:
            # Phase shift
            shift = np.random.uniform(-0.08, 0.08) * n
            x = np.linspace(0, 2, n, endpoint=False)
            x_shifted = (x - 2.0 * shift / n) % 2.0
            h_p = np.interp(x, x_shifted, work_h)
        else:
            # Binary perturbation: randomly flip a few values toward 0/1
            mask = np.random.rand(n) < 0.15
            h_p = work_h.copy()
            h_p[mask] = np.where(h_p[mask] > 0.5, 1.0, 0.0)
            h_p = 0.6 * h_p + 0.4 * work_h
        h_p = project_to_sum(h_p, n / 2.0)
        try:
            T_bh = 200 + 3000 * cool
            hc = slsqp_smooth(h_p, T_bh, n, maxiter=200, ftol=1e-12)
            hc, c5v = coord_descent(hc, n, time.time() + 2.0, max_sweeps=10)
            temp = 0.003 * cool + 0.0001
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h, work_c5 = hc, c5v
                if c5v < best_c5:
                    best_h, best_c5 = hc.copy(), c5v
        except Exception:
            pass
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 4: Binary swap search on top of best (10% budget)
    t4 = t0 + budget_s * 0.60
    h_bin, c5_bin = binary_swap_search(best_h, n, t4, n_iters=30000)
    if c5_bin < best_c5:
        # Interpolate binary back to continuous and optimize
        h_cont = project_to_sum(h_bin, n / 2.0)
        h_cont, c5_cont = coord_descent(h_cont, n, time.time() + 3.0, max_sweeps=30)
        if c5_cont < best_c5:
            best_h, best_c5 = h_cont, c5_cont

    # Phase 5: Multi-resolution (15% budget)
    t5 = t0 + budget_s * 0.75
    for n_try in [96, 128, 160, 192, 256, 288]:
        if n_try == n or time.time() > t5:
            continue
        h_i = interp_h(best_h, n_try)
        h_i, c5_v = coord_descent(h_i, n_try, time.time() + 5.0, max_sweeps=50)
        if c5_v < best_c5:
            best_h, best_c5, n = h_i, c5_v, n_try
            # Interpolate back to original n and refine
            if n_try != 224:
                h_back = interp_h(best_h, 224)
                h_back, c5_b = coord_descent(h_back, 224, time.time() + 3.0, max_sweeps=30)
                if c5_b < best_c5:
                    best_h, best_c5, n = h_back, c5_b, 224

    # Phase 6: Try structured restarts (10% budget)
    t6 = t0 + budget_s * 0.85
    for pat in ['sine', 'sine3', 'alternating', 'random_bin', 'random_bin']:
        if time.time() > t6:
            break
        h_s = structured_init(n, pat)
        h_s, c5_s = coord_descent(h_s, n, time.time() + 4.0, max_sweeps=40)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 7: Final heavy coordinate descent (12% budget)
    t7 = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t7, max_sweeps=500)

    # Phase 8: Final SLSQP very high T
    if time.time() < t0 + budget_s * 0.99:
        for T in [20000, 100000]:
            if time.time() > t0 + budget_s * 0.995:
                break
            try:
                hc = slsqp_smooth(best_h, T, n, maxiter=800, ftol=1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_h, best_c5 = hc, c5v
            except Exception:
                pass
        # Tiny final coord
        if time.time() < t0 + budget_s * 0.999:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.999, max_sweeps=20)

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
