def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_grad(h, T):
    """Compute smooth approximation of C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    # gradient of sum_k p_k * corr_k w.r.t. h
    # corr_k = sum_i h_i * (1 - h_{i+k}) for valid indices
    # d corr_k / d h_j = sum over valid i where i=j: (1-h_{j+k}) + sum over valid i where i+k=j: (-h_i)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n - lag])
        else:

                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += pk * (-h[nl:])
    val = (m + np.log(np.sum(exps)) / T) * dx
    return val, grad * dx

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):

        return x.sum() - n / 2.0

                   options={'maxiter': maxiter, 'ftol': ftol})

        cv[k0] = corr_base[k0] + t * (1.0 - t)
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',

def proj(h, n):
    for _ in range(20):
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h = h + room * (diff / tot)
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h2 = h.copy()
            h2[mask] += diff * (h[mask] / h[mask].sum())
            h = h2

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_st = {'n': 0, 'idx': None}

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10):
        target = n / 2.0
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + room * (diff / tot)
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h2 = h.copy()
            h2[mask] += diff * (h[mask] / h[mask].sum())
            h = h2
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, T):
    n = _st['n']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(sh))
    return (m + np.log(s) / T) * dx

def _con_eq(x):
    return x.sum() - _st['n'] / 2.0

def _active_obj(x):
    n = _st['n']
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(c[_st['idx']])) * dx

def slsqp_smooth(h, T, n, maxiter=600, ftol=1e-15):
    _st['n'] = n
    bnds = [(0.0, 1.0)] * n
    res = minimize(_smooth_obj, h, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': _con_eq}],
                   args=(T,), options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def slsqp_active(h, n, M, maxiter=2000, ftol=1e-16):
    _st['n'] = n
    corr = np.correlate(h, 1.0 - h, mode='full')
    _st['idx'] = np.argsort(corr)[-M:]
    bnds = [(0.0, 1.0)] * n
    res = minimize(_active_obj, h, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': _con_eq}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def _grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    ex = np.exp(sh)
    p = ex / ex.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                g[:n - lag] += pk * (1.0 - h[lag:n])
                g[lag:] += pk * (-h[:n - lag])
        else:
            nl = -lag
            g[nl:] += pk * (1.0 - h[:n - nl])
            g[:n - nl] += pk * (-h[nl:])
    return g * dx

def nesterov(h, n, t_end, lr0=0.003):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    x = best_h.copy()
    v = np.zeros(n)
    mu = 0.85
    lr = lr0
    T = 200
    stall = 0
    for it in range(80000):
        if time.time() > t_end:
            break
        if it % 200 == 0:
            T = min(T * 1.02, 100000)
        g = _grad_smooth(x, T)
        x_new = proj(x - lr * (g + mu * v), n)
        v = x_new - x
        x = x_new
        c5 = compute_c5(x)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = x.copy()
            stall = 0
        else:
            stall += 1
            if stall == 300:
                lr *= 0.7
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
    return best_h, best_c5

def _f1d(t, cb, coef, k0, dx):
    cv = cb + t * coef
    cv[k0] += t * (1.0 - t)
    return float(np.max(cv)) * dx

def coord_opt(h, n, i):
    hz = h.copy()
    hz[i] = 0.0
    cb = np.correlate(hz, 1.0 - hz, mode='full')
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k] = c
    r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                        args=(cb, coef, n - 1, dx), options={'xatol': 1e-12})
    return r.x

def coord_descend(h, n, t_end, pairs=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while time.time() < t_end:
        improved = False
        order = np.random.permutation(n)
        if pairs:
            for idx in range(0, n - 1, 2):
                if time.time() > t_end:
                    return best_h, best_c5
                i, j = order[idx], order[idx + 1]
                ht = best_h.copy()
                ht[i] = coord_opt(best_h, n, i)
                ht = proj(ht, n)
                ht[j] = coord_opt(ht, n, j)
                ht = proj(ht, n)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t
                    best_h = ht
                    improved = True
        else:
            for i in order:
                if time.time() > t_end:
                    return best_h, best_c5
                ht = best_h.copy()
                ht[i] = coord_opt(best_h, n, i)
                ht = proj(ht, n)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t
                    best_h = ht
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def fourier_perturb(h, n, amp, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    x = np.arange(n)
    for _ in range(30):
        if time.time() > t_end:
            break
        freqs = np.random.randint(1, max(2, n // 4), size=3)
        phs = np.random.uniform(0, 2 * np.pi, 3)
        amps = np.random.uniform(0, amp, 3)
        d = np.zeros(n)
        for f, p, a in zip(freqs, phs, amps):
            d += a * np.sin(2 * np.pi * f * x / n + p)
        ht = proj(h + d, n)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_c5 = c5t
            best_h = ht
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
