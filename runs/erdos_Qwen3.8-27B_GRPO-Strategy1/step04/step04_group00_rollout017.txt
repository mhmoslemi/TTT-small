def smooth_c5_and_grad(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    sm = exps.sum()
    val = (m + np.log(sm) / T) * dx
    weights = exps / sm
    # Gradient
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    j1 = i_idx + m_idx - n + 1

    v2 = (j2 >= 0) & (j2 < n)
    d = (1.0 - h[np.clip(j1, 0, n-1)]) * v1 - h[np.clip(j2, 0, n-1)] * v2
    grad = (d @ weights) * dx

            pert = step * 0.5 * np.random.randn(n)
        h_p = _proj(work_h + pert)
        T_cur = T_base + T_range * cool
        hc = _slsqp(h_p, T_cur, n, iters=400)
        c5v = compute_c5(hc)
        temp = 0.002 * cool + 0.0003
        if c5v < work_c5 or (c5v - work_c5) < temp and np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = _proj(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP annealing
    for T in [10, 50, 200, 1000, 5000, 20000, 80000]:
        if time.time() > t_end * 0.12:
            break
        hc = _slsqp(best_h, T, n, iters=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: Subgradient + SLSQP
    for it in range(60):
        if time.time() > t_end * 0.28:
            break
        cool = 1.0 - it / 60.0
        alpha = 0.004 + 0.006 * cool
        nk = 5 + int(10 * cool)
        hs = _subgrad(best_h, alpha, nk)
        T_cur = 200 + 5000 * cool
        hc = _slsqp(hs, T_cur, n, iters=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 3: Basin hopping (broad)
    best_h, best_c5 = _basin_hop(best_h, best_c5, n, 80,
                                  t_end * 0.5, 0.12, 0.03, 500, 8000)

    # Phase 4: Coordinate descent
    best_h, best_c5 = _coord_desc(best_h, n, t_end * 0.72, sweeps=400)

    # Phase 5: Basin hopping (fine)
    best_h, best_c5 = _basin_hop(best_h, best_c5, n, 50,
                                  t_end * 0.85, 0.05, 0.01, 2000, 15000)

    # Phase 6: Final coordinate descent
    best_h, best_c5 = _coord_desc(best_h, n, t_end * 0.94, sweeps=500)

    # Phase 7: Final SLSQP polish
    for T in [50000, 200000]:
        if time.time() > t_end * 0.97:
            break
        hc = _slsqp(best_h, T, n, iters=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Final coordinate descent
    if time.time() < t_end * 0.99:
        best_h, best_c5 = _coord_desc(best_h, n, t_end * 0.99, sweeps=200)

    best_h = _proj(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
