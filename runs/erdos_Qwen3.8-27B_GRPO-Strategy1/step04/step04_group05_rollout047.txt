import numpy as np
import time
from scipy.optimize import minimize

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def obj_smooth(h, N, tau):
    corr = compute_corr(h)
    s = corr / tau
    s -= s.max()
    e = np.exp(s)
    return tau * (np.log(e.sum()) + s.max())

def grad_smooth(h, N, tau):
    corr = compute_corr(h)
    s = corr / tau
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    center = N - 1
    grad = np.zeros(N)
    for m in range(2 * N - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        L = m - center
        if L == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif L > 0:
            nv = N - L
            grad[:nv] -= wm * h[L:]
            grad[L:] += wm * (1.0 - h[:nv])
        else:
            M = -L
            nv = N - M
            grad[:nv] += wm * (1.0 - h[M:])
            grad[M:] -= wm * h[:nv]
    return grad

def adam_smooth(h, N, target, deadline, tau0=5.0, tau1=0.5, n_phases=3):
    best_h = h.copy()
    best_val = compute_c5(best_h, N)
    dx = 2.0 / N
    phase_len = (deadline - time.time()) / n_phases
    for phase in range(n_phases):
        tau = tau0 * (tau1 / tau0) ** (phase / max(1, n_phases - 1))
        phase_end = time.time() + phase_len
        lr = 0.03 / (1 + phase * 0.3)
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = np.zeros(N)
        v = np.zeros(N)
        it = 0
        while time.time() < phase_end:
            g = grad_smooth(h, N, tau)
            m = b1 * m + (1 - b1) * g
            v = b2 * v + (1 - b2) * g * g
            it += 1
            mh = m / (1 - b1 ** it)
            vh = v / (1 - b2 ** it)
            h = h - lr * mh / (np.sqrt(vh) + eps)
            h = project_sum(h, target)
            c5 = compute_c5(h, N)
            if c5 < best_val:
                best_val = c5
                best_h = h.copy()
            if it % 100 == 0:
                lr *= 0.98
    return best_h, best_val

def steepest_binary(h, N, max_sweeps=200, deadline=None):

    center = N - 1
    # Precompute D matrix
    idx = np.arange(2 * N - 1)

    L = idx[None, :] - center
    q1 = p_idx - L
    q2 = p_idx + L
    valid_q1 = (q1 >= 0) & (q1 < N)
    valid_q2 = (q2 >= 0) & (q2 < N)
    
    def compute_D(h_cur):

        D[valid_q1] += 1.0 - h_cur[np.clip(q1[valid_q1], 0, N - 1)]

        return D
    
    D = compute_D(h)

        c = compute_corr(h)

def lbb_optimize(h, N, target, deadline):
    tau = 1.5
    best_h = h.copy()
    best_val = compute_c5(best_h, N)
    
    def f(x):
        x = project_sum(x, target)
        return obj_smooth(x, N, tau)
    
    def g(x):
        x = project_sum(x, target)
        return grad_smooth(x, N, tau)
    
    bounds = [(0, 1)] * N
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(N)}]
    
    x0 = project_sum(h, target)
    t_end = time.time() + (deadline - time.time()) * 0.8
    while time.time() < t_end:
        res = minimize(f, x0, jac=g, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 50, 'gtol': 1e-4})
        x0 = project_sum(res.x, target)
        c5 = compute_c5(x0, N)
        if c5 < best_val:
            best_val = c5
            best_h = x0.copy()
        tau *= 0.7
        if tau < 0.3:
            break
    return best_h, best_val
