                    best_c5 = c5_try
                    improved = True
                else:
                    best_h[i] = v
                    best_h = project(best_h, n)
        if not improved:
            break
    return best_h, best_c5

def smooth_gradient(h, n, T=80):
    corr = full_corr(h)
    dx = 2.0 / n
    center = n - 1
    m = corr.max()
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(len(w)):
        wij = w[j]
        if wij < 1e-8:
            continue
        k = j - center
        if k == 0:
            grad += wij * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] -= wij * h[k:]
            grad[k:] += wij * (1.0 - h[:n - k])
        else:
            mk = -k
            grad[mk:] += wij * (1.0 - h[:n - mk])
            grad[:n - mk] -= wij * h[mk:]
    return grad * dx

def grad_descent(h, n, t_end, max_iters=300):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    grad_prev = None
    x_prev = None
    step = 0.04
    for it in range(max_iters):
        if time.time() > t_end:
            break
        grad = smooth_gradient(h_cur, n, T=80)
        if grad_prev is not None and x_prev is not None:
            s = h_cur - x_prev
            y = grad - grad_prev
            sy = s @ y
            yy = y @ y
            ss = s @ s
            if yy > 1e-14 and abs(sy) > 1e-14:
                step = min(max(ss / abs(sy), 1e-4), 0.3)
        h_new = project(h_cur - step * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            x_prev = h_cur.copy()
            h_cur = h_new
            c5_cur = c5_new
            grad_prev = grad
        else:
            step *= 0.7
        x_prev = h_cur.copy()
        grad_prev = grad
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = 0.5 + 0.15 * np.cos(np.pi * x) + 0.04 * np.cos(2 * np.pi * x)

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive coordinate descent (30%)
    p1_end = t0 + budget_s * 0.30
    best_h, best_c5 = _coord_loop(best_h, n, p1_end)

    # Phase 2: Gradient + CD alternating (20%)
    p2_end = t0 + budget_s * 0.50
    while time.time() < p2_end:
        h_g, c5_g = grad_descent(best_h, n, time.time() + 20, max_iters=150)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g
        seg_end = min(time.time() + 25, p2_end)
        h_cd, c5_cd = _coord_loop(best_h, n, seg_end)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 3: Multi-resolution exploration (20%)
    p3_end = t0 + budget_s * 0.70
    n_candidates = [200, 128, 256, 192, 160]
    for n_try in n_candidates:
        if time.time() > p3_end:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h, period=2.0)
        h_new = project(h_new, n_try)
        seg_end = min(time.time() + budget_s * 0.035, p3_end)
        h_new, c5_new = _coord_loop(h_new, n_try, seg_end)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            n = n_try

    # Phase 4: Perturbation escapes + CD (15%)
    p4_end = t0 + budget_s * 0.85
    for pert_scale in [0.03, 0.05, 0.02, 0.04, 0.06]:
        if time.time() > p4_end:
            break
        h_p = best_h + pert_scale * np.random.randn(n)
        h_p = project(h_p, n)
        seg_end = min(time.time() + 20, p4_end)
        h_p, c5_p = _coord_loop(h_p, n, seg_end)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 5: Final long coordinate descent (13%)
    final_end = t_end - 5
    best_h, best_c5 = _coord_loop(best_h, n, final_end)

    # Final: verify and clean
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Last second: try a few tiny perturbations
    if time.time() < t_end - 2:
        for _ in range(3):
            h_t = best_h + 0.005 * np.random.randn(n)
            h_t = project(h_t, n)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t

    return best_h, best_c5, n
