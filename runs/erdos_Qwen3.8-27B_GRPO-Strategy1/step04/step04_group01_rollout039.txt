```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def to_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    target = n // 2
    h = np.array(h, dtype=float)
    # Round to binary
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    # Fix sum
    n_ones = int(h_bin.sum())
    if n_ones > target:
        # Need to flip (n_ones - target) ones to zeros
        ones_idx = np.where(h_bin == 1.0)[0]
        np.random.shuffle(ones_idx)
        for idx in ones_idx[:n_ones - target]:
            h_bin[idx] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0.0)[0]
        np.random.shuffle(zeros_idx)
        for idx in zeros_idx[:target - n_ones]:
            h_bin[idx] = 1.0
    return h_bin

def local_search(h, n):
    """Try all 1-0 swaps, take best improving one. Repeat until no improvement."""
    target = n // 2
    best_c5 = compute_c5(h)
    improved = True
    while improved:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    best_pair = (i, j)
                    best_delta = c5
                h[i], h[j] = 1.0, 0.0  # restore
        if best_pair is not None:
            h[best_pair[0]], h[best_pair[1]] = 0.0, 1.0
            improved = True
    return h, best_c5
```

c_s = sum_{n: valid} h[n+s] * (1-h[n])

For a swap of positions i (h: 1→0) and j (h: 0→1):
- Term where n+s = i (n = i-s): changes from h[i]*(1-h[i-s]) to 0*(1-h[i-s]) = 0

  Actually: the term is h[n+s]*(1-h[n]) with n+s=i, n=i-s.
  Before: h[i]*(1-h[i-s]) = 1*(1-h[i-s])
  After: 0*(1-h[i-s]) = 0
  Change: -(1-h[i-s]) (if i-s is valid, i.e., 0≤i-s<n)

- Term where n = i: changes from h[i+s]*(1-h[i]) to h[i+s]*(1-0) = h[i+s]

```python
def compute_corr_change(corr, h, i, j, n):
    """Compute new correlation after swapping h[i]:1->0, h[j]:0->1.
    Returns the change in max correlation value."""
    # corr is the current correlation array
    # We need to compute the change for each affected lag
    # Lag s corresponds to index s + (n-1) in the corr array
    offset = n - 1
    new_corr = corr.copy()
    
    for s in range(-(n-1), n):
        # Compute delta_c_s
        delta = 0.0
        # Term 1: -(1-h[i-s]) if 0 <= i-s < n
        is_idx = i - s
        if 0 <= is_idx < n:
            delta -= (1.0 - h[is_idx])
        # Term 2: +h[i+s] if 0 <= i+s < n
        is2_idx = i + s
        if 0 <= is2_idx < n:
            delta += h[is2_idx]

            new_corr[s + offset] += delta
    
    return int(np.max(new_corr))
```

```python
def eval_swap(corr, h, i, j, n):
    """Return max_corr after hypothetical swap of h[i] (1→0) and h[j] (0→1)."""
    offset = n - 1
    # Determine affected lag range
    # Position i affects lags s where |s-i| < n or |s+i-n+1| < n... 
    # Actually, it affects all s in [min(i-s for valid s), max(i+s for valid s)]
    # Which is s from -(n-1) to n-1 (all of them in most cases)
    # But let's just check the relevant range for i and j
    
    # The affected indices in the corr array:
    # For position i: lags s where 0≤i-s<n (s ∈ [i-n+1, i]) or 0≤i+s<n (s ∈ [-i, n-1-i])
    # For position j: lags s where 0≤j-s<n (s ∈ [j-n+1, j]) or 0≤j+s<n (s ∈ [-j, n-1-j])
    
    # Union of affected lags:
    min_s = max(-n+1, min(i-n+1, -i, j-n+1, -j))
    max_s = min(n-1, max(i, n-1-i, j, n-1-j))
    
    # Compute new values at affected positions
    new_vals = corr.copy()  # Start with old values
    for s in range(min_s, max_s + 1):
        delta = 0.0
        is_idx = i - s
        if 0 <= is_idx < n:
            delta -= (1.0 - h[is_idx])
        is2_idx = i + s
        if 0 <= is2_idx < n:
            delta += h[is2_idx]
        js_idx = j - s
        if 0 <= js_idx < n:
            delta += (1.0 - h[js_idx])
        js2_idx = j + s
        if 0 <= js2_idx < n:
            delta -= h[js2_idx]
        if delta != 0:
            new_vals[s + offset] += delta
    
    return int(np.max(new_vals))
```
