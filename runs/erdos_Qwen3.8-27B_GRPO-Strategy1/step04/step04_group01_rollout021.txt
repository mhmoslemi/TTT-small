    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, target):

    vc = np.clip(vec, 0.0, 1.0)

    if abs(diff) < 1e-12:
        return vc

        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vc + (lo + hi) * 0.5, 0.0, 1.0)

def opt_single(h, i, n, target):
    def f(t):
        h[i] = t
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})

def opt_pair(h, i, j, n):
    d_lo = max(-1.0 + h[i], -h[j], -1.0)  # h[i]+d >= 0, h[j]-d >= 0
    d_hi = min(1.0 - h[i], h[j], 1.0)     # h[i]+d <= 1, h[j]-d <= 1
    if d_lo >= d_hi:
        return 0.0
    def f(d):
        h[i] += d
        h[j] -= d
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = np.max(corr)
        h[i] -= d
        h[j] += d
        return val
    res = minimize_scalar(f, bounds=(d_lo, d_hi), method='bounded', options={'xatol': 1e-14})
    return res.x

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, target):
    vec = np.array(h, dtype=float)
    vc = np.clip(vec, 0.0, 1.0)
    diff = target - vc.sum()
    if abs(diff) < 1e-12:
        return vc
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(vc + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vc + (lo + hi) * 0.5, 0.0, 1.0)

def fast_opt_coord(h, i, target):
    """Fast single-coordinate optimization using precomputed correlations."""
    n = len(h)
    a = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Precompute direction for linear part (k != 0)
    # delta[k] = (1-h[i+k]) - h[i-k] (with 0 for out of bounds)
    delta = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        if k == 0:
            continue
        t1 = (1.0 - h[i + k]) if 0 <= i + k < n else 0.0
        t2 = h[i - k] if 0 <= i - k < n else 0.0
        delta[idx] = t1 - t2
    
    def f(t):
        # Linear part update
        new_max = np.max(corr + (t - a) * delta)
        # k=0 correction (index n-1)
        k0_val = corr[n-1] + t*(1-t) - a*(1-a)
        if k0_val > new_max:
            new_max = k0_val
        return new_max
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return res.x

def fast_opt_pair(h, i, j):
    """Fast pair swap optimization: h[i]+=d, h[j]-=d."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ai, aj = h[i], h[j]
    
    # For h[i] += d: delta_i[k] = (1-h[i+k]) - h[i-k]
    # For h[j] -= d: delta_j[k] = -(1-h[j+k]) + h[j-k]
    # Total: d * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
    delta = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        t1 = (1.0 - h[i + k]) if 0 <= i + k < n else 0.0
        t2 = h[i - k] if 0 <= i - k < n else 0.0
        t3 = (1.0 - h[j + k]) if 0 <= j + k < n else 0.0
        t4 = h[j - k] if 0 <= j - k < n else 0.0
        delta[idx] = (t1 - t2) - (t3 - t4)
    
    # Bound on d: 0 <= ai+d <= 1, 0 <= aj-d <= 1
    d_lo = max(-ai, -aj)  # wait: ai+d>=0 → d>=-ai; aj-d>=0 → d<=aj
    d_hi = min(1.0 - ai, aj)
    # Also d >= -ai, d <= aj, d >= -(1-aj)=aj-1... no
    # ai + d in [0,1] → d in [-ai, 1-ai]
    # aj - d in [0,1] → d in [aj-1, aj]
    d_lo = max(-ai, aj - 1.0)
    d_hi = min(1.0 - ai, aj)
    if d_lo >= d_hi:
        return 0.0
    
    def f(d):
        # Linear part
        new_max = np.max(corr + d * delta)

        # k=0 term: h[i]*(1-h[i]) + h[j]*(1-h[j]) changes

        if k0_val > new_max:

        # For k = i-j: h[i] and h[j+k] = h[j+i-j] = h[i] appear
        # Actually this gets complicated. Let me just handle k=0 and
        # for other k, the linear approx is exact (since h[i] and h[j] 
        # don't both appear in the same product term unless i+k=j or j+k=i)
        
        # Check if i+k=j (k=j-i): then h[i]*(1-h[j]) term exists
        # Changing h[i]→ai+d, h[j]→aj-d: (ai+d)(1-aj+d) vs ai(1-aj)
        # Delta: d*(1-aj+d) + ai*d = d*(1-aj) + d² + ai*d
        # This is quadratic in d! 
        # Similarly for k=i-j: h[j]*(1-h[i]) term
        # Let me handle these:
        
        if i != j:
            k_p = j - i  # h[i] * (1-h[j]) at shift k_p
            k_m = i - j  # h[j] * (1-h[i]) at shift k_m
            idx_p = k_p + (n-1) if -n+1 <= k_p <= n-1 else -1
            idx_m = k_m + (n-1) if -n+1 <= k_m <= n-1 else -1
            
            if idx_p >= 0:
                # Original: ai * (1-aj); New: (ai+d)*(1-aj+d)
                # Delta: (ai+d)(1-aj+d) - ai(1-aj) = d(1-aj) + ai*d + d² 
                # = d(1-aj+ai) + d²
                quadratic_extra = d * (1.0 - aj + ai) + d * d
                # But wait, I already added d*delta[idx_p] which assumed linear
                # delta[idx_p] includes the contribution from h[i] and h[j] at this shift
                # Let me just recompute this specific term:
                # corr at shift k_p includes: h[i]*(1-h[j]) + other terms
                # The "other terms" don't change with the swap
                # So: new_corr[idx_p] = corr[idx_p] - ai*(1-aj) + (ai+d)*(1-aj+d)
                # = corr[idx_p] + d*(1-aj) + d*ai + d²
                # My linear approx gave: corr[idx_p] + d * delta[idx_p]
                # delta[idx_p] = (1-h[i+k_p]) - h[i-k_p] - (1-h[j+k_p]) + h[j-k_p]
                # = (1-h[j]) - h[i-(j-i)] - (1-h[j+(j-i)]) + h[j-(j-i)]
                # = (1-h[j]) - h[2i-j] - (1-h[2j-i]) + h[i]
                # Hmm this doesn't seem right. Let me just abandon the fast approach
                # for pairs and use the direct method.
                pass
    
    # For safety, let me just use direct evaluation for pairs
    def f_direct(d):
        h[i] = ai + d
        h[j] = aj - d
        c = np.max(np.correlate(h, 1.0 - h, mode='full'))
        h[i] = ai

    res = minimize_scalar(f_direct, bounds=(d_lo, d_hi), method='bounded', options={'xatol': 1e-14})
