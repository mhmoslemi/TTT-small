import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    target = n / 2.0
    vec = np.array(h, dtype=float)
    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-13:
        return vc
    lo = -np.max(vec) - 1.0
    hi = 1.0 - np.min(vec) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=600, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def cons(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': cons}],
                   options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
    return project(res.x, n)

def coord_opt(h, n, i):
    def f1d(hi):
        h_try = h.copy()
        h_try[i] = hi
        corr = np.correlate(h_try, 1.0 - h_try, mode='full')
        return np.max(corr) * (2.0 / n)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):

            break
        improved = False

            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= 4:
            break
    return best_h, best_c5

def subgradient_step(h, n, T, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    lam = np.exp(T * (corr - m))
    lam /= lam.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        l = lam[k_idx]
        if l < 1e-14:
            continue
        k = k_idx - (n - 1)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if k >= 0:
            if k < n:
                t1[:n - k] = 1.0 - h[k:]
            if k > 0:
                t2[k:] = -h[:n - k]
        else:
            kk = -k
            t1[kk:] = 1.0 - h[:n - kk]
            t2[:n - kk] = -h[kk:]
        grad += l * (t1 + t2)
    grad *= dx
    return project(h - alpha * grad, n)

def interpolate_h(h, n_old, n_new):

    x_new = np.linspace(0.0, 2.0, n_new + 1)[:-1]
    return np.interp(x_new, x_old, h)
