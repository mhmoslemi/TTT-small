```python
def pair_opt(h, n, i, j):
    # Optimize h[i] and h[j] jointly, maintaining sum constraint
    # h[j] = h[j]_old + (h[i]_old - h[i]) to maintain sum
    hi_old = h[i]
    hj_old = h[j]
    
    def f(hi):
        hj = hj_old + hi_old - hi
        if hj < 0 or hj > 1:
            return np.inf
        h_try = h.copy()
        h_try[i] = hi
        h_try[j] = hj
        corr = np.correlate(h_try, 1.0 - h_try, mode='full')
        return np.max(corr) * (2.0/n)
    
    hi_min = max(0, hi_old - hj_old)  # so that hj = hj_old + hi_old - hi >= 0
    hi_max = min(1, hi_old + 1 - hj_old)  # so that hj <= 1
    hi_max = min(hi_max, 1.0)
    hi_min = max(hi_min, 0.0)
    
    if hi_min >= hi_max:
        return hi_old
    
    res = minimize_scalar(f, bounds=(hi_min, hi_max), method='bounded',
                          options={'xatol': 1e-12})
    return res.x
```

```python
import time

    m = np.max(corr)
```

```python
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def compute_g(h, n, i):
    N = 2 * n - 1
    g = np.zeros(N)
    lag = np.arange(N) - (n - 1)
    # g[k] = (1-h[i-lag[k]]) [valid] - h[i+lag[k]] [valid]
    idx1 = i - lag  # index in h for the first term
    idx2 = i + lag  # index in h for the second term
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    g[valid1] += 1.0 - h[idx1[valid1]]
    g[valid2] -= h[idx2[valid2]]
    return g

def ternary_max_linears(corr, d, lo, hi, iters=50):
    """Minimize max(corr + delta*d) over delta in [lo, hi]. Returns optimal delta."""
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr + m1 * d) < np.max(corr + m2 * d):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_sweep(h, n, max_sweeps=1000, t_end=None):
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        improved = False
        for i in range(n):
            g = compute_g(h, n, i)
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-15:

            delta = ternary_max_linears(corr, g, lo, hi)
            new_hi = h[i] + delta
            if abs(new_hi - h[i]) < 1e-13:

            h[i] = new_hi
            # Update corr incrementally

            # Check if C5 improved
            c5_val = np.max(corr) * (2.0 / n)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                improved = True
        if not improved:
            break
    return h, best_c5
```

```python
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
```

```python
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def compute_g(h, n, i):
    lag = np.arange(2 * n - 1) - (n - 1)
    idx1 = i - lag
    idx2 = i + lag
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    g = np.zeros(2 * n - 1)
    g[v1] += 1.0 - h[idx1[v1]]
    g[v2] -= h[idx2[v2]]
    return g

def ternary_opt(corr, d, lo, hi, iters=50):
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr + m1 * d) < np.max(corr + m2 * d):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def single_coord_sweep(h, n, t_end, max_sweeps=5000):
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        any_change = False
        for i in range(n):
            if h[i] <= 0 or h[i] >= 1:
                # Check if we can move into interior
                lo = max(0, h[i]) - h[i]  # = 0 if h[i]=0, h[i] if h[i] in (0,1)
                hi = 1.0 - h[i]
                lo = -h[i]
                if hi - lo < 1e-14:
                    continue
            g = compute_g(h, n, i)
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            delta = ternary_opt(corr, g, lo, hi)
            new_val = h[i] + delta
            if abs(new_val - h[i]) > 1e-12:
                h[i] = new_val
                corr += delta * g
                any_change = True
                c5_new = np.max(corr) * (2.0 / n)
                if c5_new < best_c5:
                    best_c5 = c5_new
    return h, best_c5

def pair_coord_sweep(h, n, rng, n_pairs, t_end):
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n)
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if abs(h[i] - h[j]) < 1e-12:
            continue
        gi = compute_g(h, n, i)
        gj = compute_g(h, n, j)
        d = gi - gj
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi - lo < 1e-12:
            continue
        delta = ternary_opt(corr, d, lo, hi)
        if abs(delta) < 1e-12:
            continue
        h[i] += delta
        h[j] -= delta
        corr += delta * d
        c5_new = np.max(corr) * (2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new
    return h, best_c5

    def obj(x):
        return smooth_c5(x, T)
    def con_eq(x):
        return x.sum() - n / 2.0

    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(res.x, n)
```

```python
    n = len(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def make_lag(n):
    return np.arange(2 * n - 1) - (n - 1)

def compute_g(h, n, i, lag):
    idx1 = i - lag
    idx2 = i + lag
    N = 2 * n - 1
    g = np.zeros(N)
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    g[v1] += 1.0 - h[idx1[v1]]
    g[v2] -= h[idx2[v2]]
    return g

def ternary_opt(corr, d, lo, hi, iters=50):
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr + m1 * d) < np.max(corr + m2 * d):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def single_coord_sweep(h, n, lag, t_end, max_sweeps=5000):
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n)
    order = list(range(n))
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        any_change = False
        for i in order:
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            g = compute_g(h, n, i, lag)
            delta = ternary_opt(corr, g, lo, hi)
            new_val = h[i] + delta
            if abs(new_val - h[i]) > 1e-12:
                h[i] = new_val
                corr += delta * g
                any_change = True
        c5_cur = np.max(corr) * (2.0 / n)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
        elif not any_change:
            break
    return h, best_c5

def pair_coord_sweep(h, n, lag, rng, n_pairs, t_end):
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n)
    count = 0
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi - lo < 1e-12:
            continue
        gi = compute_g(h, n, i, lag)
        gj = compute_g(h, n, j, lag)
        d = gi - gj
        delta = ternary_opt(corr, d, lo, hi)
        if abs(delta) < 1e-12:
            continue
        h[i] += delta
        h[j] -= delta
        corr += delta * d
        count += 1
        if count % 200 == 0:
            c5_cur = np.max(corr) * (2.0 / n)
            if c5_cur < best_c5:
                best_c5 = c5_cur
    best_c5 = min(best_c5, np.max(corr) * (2.0 / n))
    return h, best_c5

def slsqp_polish(h, n, T, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def con_eq(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_eq}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(res.x, n)
    except Exception:
        return project(h.copy(), n)

def perturb_h(h, rng, n_perturb, scale):
    h_new = h.copy()
    idx = rng.choice(len(h), size=n_perturb, replace=False)
    h_new[idx] += scale * rng.randn(n_perturb)
    return project(h_new, len(h))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    lag = make_lag(n)

    def elapsed():
        return time.time() - t0

    # Phase 1: Intensive single-coordinate (25%)
    t_end1 = t0 + budget_s * 0.25
    best_h, best_c5 = single_coord_sweep(best_h, n, lag, t_end1, max_sweeps=20000)

    # Phase 2: Pair-coordinate descent (25%)
    t_end2 = t0 + budget_s * 0.50
    while time.time() < t_end2:
        batch_t = min(time.time() + budget_s * 0.04, t_end2)
        best_h, best_c5 = pair_coord_sweep(best_h, n, lag, rng, 800, batch_t)
        best_h, best_c5 = single_coord_sweep(best_h, n, lag, batch_t, max_sweeps=500)

    # Phase 3: SLSQP + single coord (15%)
    t_end3 = t0 + budget_s * 0.65
    best_h, best_c5 = single_coord_sweep(best_h, n, lag, min(time.time()+budget_s*0.08, t_end3), max_sweeps=5000)
    for T in [200, 1000, 5000, 20000]:
        if elapsed() > budget_s * 0.64:
            break
        hc = slsqp_polish(best_h.copy(), n, T, maxiter=400)
        hc, c5v = single_coord_sweep(hc, n, lag, min(time.time()+budget_s*0.02, t_end3), max_sweeps=1000)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Diversification - perturb + re-optimize (20%)
    t_end4 = t0 + budget_s * 0.85
    work_h = best_h.copy()
    work_c5 = best_c5
    n_trials = 8
    per_trial = (t_end4 - time.time()) / max(n_trials, 1)
    for trial in range(n_trials):
        if time.time() > t_end4:
            break
        scale = 0.05 + 0.15 * rng.rand()
        n_p = int(5 + 15 * rng.rand())
        h_try = perturb_h(work_h, rng, min(n_p, n), scale)
        t_sub = min(time.time() + per_trial * 0.6, t_end4)
        h_try, c5_try = single_coord_sweep(h_try, n, lag, t_sub, max_sweeps=3000)
        t_sub2 = min(time.time() + per_trial * 0.35, t_end4)
        h_try, c5_try = pair_coord_sweep(h_try, n, lag, rng, 400, t_sub2)
        h_try, c5_try = single_coord_sweep(h_try, n, lag, t_sub2, max_sweeps=2000)
        if c5_try < work_c5:
            work_c5 = c5_try
            work_h = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Final intensive pair + single (10%)
    t_end5 = t0 + budget_s * 0.95
    while time.time() < t_end5:
        batch_t = min(time.time() + budget_s * 0.03, t_end5)
        best_h, best_c5 = pair_coord_sweep(best_h, n, lag, rng, 600, batch_t)
        best_h, best_c5 = single_coord_sweep(best_h, n, lag, batch_t, max_sweeps=2000)

    # Phase 6: Final SLSQP polish (5%)
    for T in [50000, 200000]:
        if elapsed() > budget_s * 0.97:
            break
        hc = slsqp_polish(best_h.copy(), n, T, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final coordinate sweep
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = single_coord_sweep(best_h, n, lag, t0 + budget_s * 0.995, max_sweeps=5000)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
