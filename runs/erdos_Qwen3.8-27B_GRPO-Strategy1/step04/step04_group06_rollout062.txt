```python
    target = n / 2.0

                jm2 = i - k
                if 0 <= jm2 < n and jm2 != i:

        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]

                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return np.clip(res.x, 0, 1)
```

The Jacobian dc[k]/dx[i]:
- x[i] appears in the sum as x[i] * (1-x[i+k]) → derivative (1-x[i+k]) if i+k valid
- x[i] appears in the sum as x[i-k] * (1-x[i]) → derivative -x[i-k] if i-k valid

So dc[k]/dx[i] = (1-x[i+k] if i+k<n else 0) - (x[i-k] if i-k>=0 else 0)

For the weighted sum F = sum_k w[k] * c[k]:
dF/dx[i] = sum_k w[k] * [(1-x[i+k] if i+k<n else 0) - (x[i-k] if i-k>=0 else 0)]

I can compute this for all i at once:
dF/dx[i] = sum_{k: i+k<n} w[i+k - (n-1)] * (1-x[i+k]) - sum_{k: i-k>=0} w[i-k + (n-1)-k] * x[i-k]

Let me reindex. Let lag = i+k (for first sum), so k = lag-i, lag ranges from i to n-1:

And second sum: let lag2 = i-k, k = i-lag2, lag2 ranges from 0 to i:
sum_{lag2=0}^{i} w[lag2 + (n-1)-(i-lag2)] * x[lag2]  ... this is getting messy.

```python
def slsqp_grad(x, w_active, n):
    """w_active: dict of {lag_index: weight} for active lags only."""
    center = n - 1
    g = np.zeros(n)
    for idx, wi in w_active.items():
        k = idx - center
        # For each i: contribution = wi * [(1-x[i+k] if 0<=i+k<n) - (x[i-k] if 0<=i-k<n)]
        if k >= 0:
            # first term: (1-x[i+k]) for i=0..n-1-k
            g[:n-k] += wi * (1.0 - x[k:])
            # second term: -x[i-k] for i=k..n-1
            g[k:] -= wi * x[:n-k]
        else:
            m = -k
            # first term: (1-x[i+k]) = (1-x[i-m]) for i=m..n-1
            g[m:] += wi * (1.0 - x[:n-m])
            # second term: -x[i-k] = -x[i+m] for i=0..n-1-m
            g[:n-m] -= wi * x[m:]
    return g
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            s = h.sum()
            return np.clip(h / max(s, 1e-15) * target, 0, 1) if s > 0 else h
        h = h + room * (diff / tot)
    else:
        s = h.sum()
        if s < 1e-12:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip, im = i + k, i - k
        a[center + k] = (1.0 - h[ip] if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
        a[center - k] = (1.0 - h[im] if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
    ct = corr_base[center]
    def f1d(x):
        v = corr_base + x * a
        v[center] = ct + x - x * x
        return v.max()
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def cd_sweep(h, n):
    changed = False
    order = np.random.permutation(n)
    for i in order:
        hi = coord_opt(h, n, i)
        if abs(hi - h[i]) > 1e-11:
            h[i] = hi
            h = project_sum(h, n)
            changed = True
    return h, changed

def slsqp_run(h0, T, n, maxiter=300):
    dx = 2.0 / n
    lenc = 2 * n - 1
    center = n - 1
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return (m + np.log(np.exp(T * (c - m)).sum()) / T) * dx
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        w = T * np.exp(T * (c - m))
        wsum = w.sum()
        if wsum < 1e-30:
            return np.zeros(n)
        w = w / wsum * dx
        g = np.zeros(n)
        for idx in range(lenc):
            wi = w[idx]
            if wi < 1e-14:
                continue
            k = idx - center
            if k >= 0:
                g[:n - k] += wi * (1.0 - x[k:])
                g[k:] -= wi * x[:n - k]
            else:
                m2 = -k
                g[m2:] += wi * (1.0 - x[:n - m2])
                g[:n - m2] -= wi * x[m2:]
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(np.clip(res.x, 0, 1), n)

def cd_loop(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_change = 0
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        h_new, changed = cd_sweep(best_h, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            no_change = 0
        else:
            h_new, changed = cd_sweep(best_h, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            no_change += 1
        if not changed:
            no_change += 1
        if no_change >= 3:
            break
    return best_h, best_c5

def perturb_h(h, n, rng, mode='gauss', scale=0.1):
    h_new = h.copy()
    if mode == 'gauss':
        h_new += scale * rng.standard_normal(n)
    elif mode == 'freq':
        f = rng.integers(2, 6)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) * 2.0 / n
        h_new += scale * np.cos(2 * np.pi * f * x + phase)
    elif mode == 'binary':
        idx = rng.choice(n, size=int(0.2 * n), replace=False)
        h_new[idx] = 1.0 - h_new[idx]
    elif mode == 'block':
        bsz = max(1, n // 8)
        i0 = rng.integers(0, n - bsz)
        i1 = rng.integers(0, n - bsz)
        tmp = h_new[i0:i0+bsz].copy()
        h_new[i0:i0+bsz] = h_new[i1:i1+bsz]
        h_new[i1:i1+bsz] = tmp
    return project_sum(h_new, n)

    best_h, best_c5 = cd_loop(best_h, n, t1, max_sweeps=300)

            break

            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                best_h, best_c5 = cd_loop(best_h, n, min(time.time() + 2, t2), max_sweeps=30)
        except Exception:
            pass

    # Phase 3: Perturbation + coordinate descent (30% time)
    t3 = t0 + 0.70 * budget_s
    modes = ['gauss', 'freq', 'binary', 'block']

    t5 = t_end - 2

        if time.time() > t5:

        try:

            c5_try = compute_c5(h_try)
            if c5_try < best_c5:

                best_h = h_try
        except Exception:
            pass
```
