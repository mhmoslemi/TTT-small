def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h = h + room * (diff / tr)
        else:
            if s < 1e-14:
                break
            h = h * (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_coef(h, n, i):
    """D_m coefficients: corr[m] = corr_base[m] + h[i]*D_m"""
    nsm1 = n - 1
    D = np.zeros(2 * n - 1)
    # Term 1: (1-h[i-l]) for m in [i, i+n-1] ∩ [0, 2n-2]
    mlo, mhi = max(0, i), min(2*n-2, i+n-1)

        D[ms] += (1.0 - h[ms - nsm1 + i])  # h[i - l] where l = m - nsm1, so i-l = i - (m-nsm1) = i + nsm1 - m

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(8):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h = h + room * (diff / tr)
        else:
            if s < 1e-14:
                break
            h = h * (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_coef(h, n, i):
    nsm1 = n - 1
    D = np.zeros(2 * n - 1)
    mlo, mhi = max(0, i), min(2 * n - 2, i + n - 1)
    if mlo <= mhi:
        ms = np.arange(mlo, mhi + 1)
        idx = i + nsm1 - ms
        D[ms] += (1.0 - h[idx])
    mlo2, mhi2 = max(0, nsm1 - i), min(2 * n - 2, 2 * nsm1 - i)
    if mlo2 <= mhi2:
        ms2 = np.arange(mlo2, mhi2 + 1)
        idx2 = i + ms2 - nsm1
        D[ms2] -= h[idx2]
    return D

def coord_sweep(h, n, t_end, max_sweeps=500, rankcent=True):
    best_h = h.copy()
    best_c5 = compute_c5_val(best_h)
    center = n - 1
    staller = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if rankcent:
            corr_all = np.correlate(best_h, 1.0 - best_h, mode='full')
            order = np.argsort(corr_all)  # process least-active lags first (heuristic)
            # Actually just use random order
            order = np.random.permutation(n)
        else:
            order = np.random.permutation(n)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h0 = best_h.copy()
            h0[i] = 0.0
            corr_base = np.correlate(h0, 1.0 - h0, mode='full')
            D = get_coef(best_h, n, i)
            D[center] = 0.0  # handle quadratic separately

            def f_obj(hi):
                v = corr_base + hi * D
                v[center] = corr_base[center] + hi - hi * hi
                return np.max(v) * (2.0 / n)

            res = minimize_scalar(f_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            new_hi = res.x
            if abs(new_hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = new_hi
            # Fix sum constraint
            diff = n / 2.0 - best_h.sum()
            if abs(diff) > 1e-10:
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = best_h[mask]
                if diff > 0:
                    room = 1.0 - others
                    tr = room.sum()
                    if tr > 1e-14:
                        best_h[mask] = others + room * (diff / tr)
                else:
                    os = others.sum()
                    if os > 1e-14:
                        best_h[mask] = others * (1.0 + diff / os)
                best_h = np.clip(best_h, 0, 1)
            c5_try = compute_c5_val(best_h)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                improved = True
        if not improved:
            staller += 1
            if staller >= 4:
                break
        else:
            staller = 0
    return best_h, best_c5

def softmax_grad(corr, h, T):
    n = len(h)
    dx = 2.0 / n
    nsm1 = n - 1
    m = np.max(corr)
    ew = np.exp(T * (corr - m))
    sw = ew.sum()
    if sw < 1e-30:
        ew[:] = 1.0 / len(corr)
        sw = len(corr)
    ew /= sw
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = ew[idx]
        if w < 1e-12:
            continue
        l = idx - nsm1
        lo, hi = max(0, l), min(n - 1, n - 1 + l)
        if lo <= hi:
            grad[lo:hi + 1] += w * (1.0 - h[lo - l:hi + 1 - l])
        lo2, hi2 = max(0, -l), min(n - 1, n - 1 - l)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w * h[lo2 + l:hi2 + 1 + l]
    return grad * dx

def grad_descent(h, n, t_end, n_iter=500, T_start=200, T_end=5000):
    best_h = h.copy()
    best_c5 = compute_c5_val(best_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / n_iter
        T = T_start + (T_end - T_start) * cool
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        g = softmax_grad(corr, best_h, T)
        alpha = 0.02 * (0.5 + 0.5 * cool)
        h_new = project_h(best_h - alpha * g, n)
        c5_new = compute_c5_val(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

        elif mode == 2:

            start = np.random.randint(0, n - block)
            pert = np.zeros(n)
            pert[start:start + block] = (0.05 + 0.15 * cool) * np.random.randn(block)
        elif mode == 3:
            mask = np.random.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = (0.08 + 0.2 * cool) * np.random.randn(mask.sum())
        else:
            freq = np.random.randint(1, 15)
            phase = np.random.uniform(0, 2 * np.pi)

            pert = amp * (np.sin(2 * np.pi * freq * np.arange(n) / n + phase) +
                          0.5 * np.sin(2 * np.pi * (2 * freq) * np.arange(n) / n + phase))
        h_pert = project_h(work_h + pert, n)
        # Quick coord polish on perturbed
        h_pert, c5_pert = coord_sweep(h_pert, n, min(time.time() + 30, t_end), max_sweeps=30)
        temp = 0.003 * cool + 0.0003
        if c5_pert < work_c5 or np.random.rand() < np.exp(-(c5_pert - work_c5) / temp):
            work_h = h_pert
            work_c5 = c5_pert
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h
    return best_h, best_c5

def make_block_h(n, nblocks):
    total = n // 2
    per = total // nblocks
    h = np.zeros(n)
    filled = 0
    for b in range(nblocks):
        start = b * (n // nblocks)
        length = per
        if filled + length > total:
            length = total - filled
        h[start:start + length] = 1.0
        filled += length
    remaining = total - filled
    if remaining > 0:
        h[n - remaining:] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
