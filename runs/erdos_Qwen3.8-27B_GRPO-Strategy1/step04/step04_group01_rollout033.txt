def nesterov_phase(h, n, t_end, T_start=300, T_end=30000):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    x = h.copy()
    v = np.zeros(n)
    mu = 0.85
    alpha = 0.05
    it = 0
    while time.time() < t_end:
        frac = min(1.0, it / 400.0)
        T = T_start * (T_end / T_start) ** frac
        # Nesterov step
        x_new = x + mu * v
        x_clip = np.clip(x_new, 0.0, 1.0)
        # Gradient
        g = smooth_grad(x_clip, T)
        # Step
        x_try = x_clip - alpha * g
        x_try = project_feasible(x_try, n)
        v = x_try - x
        x = x_try
        it += 1
        c5v = compute_c5(x)
        if c5v < best_val - 1e-12:
            best_val = c5v
            best_h = x.copy()
        # Adapt alpha
        if it % 100 == 0:
            alpha *= 0.95
    return best_h, best_val

def swap_diversify(h, n, t_end, n_trials=50):

        val = compute_c5(h_try)

            best_h = h_try
