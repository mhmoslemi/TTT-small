import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(corr, T):
    m = corr.max()
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_T = np.exp(T * (corr - m))
    p = exp_T / exp_T.sum()
    grad = np.zeros(n)
    center = n - 1
    for idx in range(len(corr)):
        if p[idx] < 1e-15:
            continue
        k = idx - center
        if k == 0:
            grad[:] += p[idx] * (1.0 - 2.0 * h)
        elif k > 0 and k < n:
            grad[:n-k] += p[idx] * (1.0 - h[k:])
            grad[k:] -= p[idx] * h[:n-k]
        elif k < 0 and k > -n:
            mk = -k
            grad[mk:] += p[idx] * (1.0 - h[:n-mk])
            grad[:n-mk] -= p[idx] * h[mk:]
    return grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(20):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        h += (target - s) / n
    return np.clip(h, 0.0, 1.0)

def coord_opt_fast(h, n, i, corr):
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip, im = i + k, i - k
        val_p = (1.0 - h[ip]) if ip < n else 0.0
        val_m = h[im] if im >= 0 else 0.0
        a[center + k] = val_p - val_m
        val_p2 = (1.0 - h[im]) if im >= 0 else 0.0
        val_m2 = h[ip] if ip < n else 0.0
        a[center - k] = val_p2 - val_m2
    dx = 2.0 / n
    cb = corr.copy()
    ct = cb[center]
    h_old = h[i]
    ct += h_old * (1.0 - h_old)
    cb[center] = ct

    def f1d(x):
        vals = cb + (x - h_old) * a
        vals[center] = ct + x * (1.0 - x)
        return np.max(vals) * dx

    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def fast_coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    imp_count = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            x_new = coord_opt_fast(best_h, n, i, corr)
            if abs(x_new - best_h[i]) < 1e-10:
                continue
            delta = x_new - best_h[i]
            center = n - 1
            for k in range(1, n):
                ip, im = i + k, i - k
                a_p = ((1.0 - h[ip]) if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
                a_m = ((1.0 - h[im]) if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
                corr[center + k] += delta * a_p
                corr[center - k] += delta * a_m
            corr[center] += x_new * (1.0 - x_new) - best_h[i] * (1.0 - best_h[i])
            best_h[i] = x_new
            improved = True
            imp_count += 1
        if improved:
            c5_val = np.max(corr) * (2.0 / n)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
            else:
                best_c5 = max(best_c5, c5_val)
        else:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            break
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def slsqp_smooth(h_init, T, n, maxiter=300):
    def obj(x):
        return smooth_c5(x, T)
    def obj_grad(x):
        return smooth_c5_grad(x, T)
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        out = project(np.clip(res.x, 0, 1), n)
        return out
    except Exception:
        return h_init

def grad_descent_smooth(h, T, n, lr, t_end, maxiter=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(maxiter):
        if time.time() > t_end:
            break
        g = smooth_c5_grad(best_h, T)
        step = lr / (1.0 + it * 0.01)
        h_new = project(best_h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_s = 160
        x = np.linspace(0, 2, n_s, endpoint=False)
        best_h = np.clip(0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x), 0, 1)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Long coordinate descent
    cd_end = t0 + budget_s * 0.30
    best_h, best_c5 = fast_coord_descent(best_h, n, cd_end, max_sweeps=5000)

    # Phase 2: SLSQP with smooth max, increasing T
    for T in [50, 200, 1000, 5000]:
        if time.time() > t0 + budget_s * 0.45:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Coordinate descent again after SLSQP
    cd2_end = t0 + budget_s * 0.55
    best_h, best_c5 = fast_coord_descent(best_h, n, cd2_end, max_sweeps=5000)

    # Phase 4: Gradient descent with smooth max + coordinate polish
    for T in [1000, 5000, 20000]:
        if time.time() > t0 + budget_s * 0.70:
            break
        h_g, c5_g = grad_descent_smooth(best_h, T, n, lr=0.02, t_end=t0 + budget_s * 0.70, maxiter=2000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g
            best_h, best_c5 = fast_coord_descent(best_h, n, min(time.time() + 5, t0 + budget_s * 0.70), max_sweeps=200)

        h_pert = best_h + noise_scale * np.random.randn(n)
        h_pert = project(h_pert, n)
        h_pert, c5_pert = fast_coord_descent(h_pert, n, min(time.time() + 8, t0 + budget_s * 0.85), max_sweeps=1000)
        if c5_pert < best_c5:
            best_c5 = c5_pert

        if time.time() > t0 + budget_s * 0.93:

        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Final coordinate descent
    best_h, best_c5 = fast_coord_descent(best_h, n, t_end - 2, max_sweeps=2000)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
