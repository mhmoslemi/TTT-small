```python
                    js = np.arange(n-1, n-1-(k1_hi-k1_lo), -1)
                    corr[k1_lo:k1_hi+1] += d * (1.0 - h[js])
                if k2_hi >= k2_lo:
                    js2 = np.arange(k2_hi - k2_lo + 1)
                    corr[k2_lo:k2_hi+1] -= d * h[js2]
                h[p] = best_v
        
        # Periodic projection
        if sweep % 10 == 9:
            h = project(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full').copy()
        
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

Hmm wait, there's an issue. When I compute `nm = base_max` and then check affected entries, I'm not correctly handling the case where the global max was at an affected entry and it decreases. In that case, the new global max could be at an unaffected entry, which is ≤ base_max but could be less than base_max.

Example: corr = [10, 5, 5, 5, 5], and we change h such that corr[0] changes from 10 to 3. The new max is 5, not max(10, 3) = 10.

So `nm = base_max` is wrong. I need `nm = max(max(corr[unaffected]), max(corr[affected] + d*a))`.

The issue is computing max(corr[unaffected]) efficiently. The affected entries are O(n), and the total is 2n-1, so unaffected is also O(n). I could just compute it:

```python
mask = np.ones(2*n-1, dtype=bool)
mask[k1_lo:k1_hi+1] = False
mask[k2_lo:k2_hi+1] = False
unaffected_max = np.max(corr[mask]) if mask.any() else 0.0
```

But this is O(n) and we do it n times per sweep, making it O(n²). That's fine (25600 for n=160).

Actually, I can precompute the top-2 corr values and their indices. If the top-1 is at an unaffected entry, then unaffected_max = corr[top1]. Otherwise, unaffected_max = corr[top2] (if top2 is also affected, then top3, etc.). But this gets complicated.

Simpler: just compute the max of all entries after the update. Since it's O(n) either way:

```python
# Apply update to a copy
vals = corr.copy()  # 319 elements
# effect 1
vals[k1_lo:k1_hi+1] += d * (1.0 - h[js1])
# effect 2
vals[k2_lo:k2_hi+1] -= d * h[js2]
nm = np.max(vals)
```

This is ~6 numpy operations on arrays of size ≤160, so very fast. Let me just do this.

For 4 candidates × n coordinates × ~6 numpy ops = 4*160*6 = 3840 numpy ops per sweep. Each op on ~100 elements. In numpy, ~1μs each. Total per sweep: ~4ms. In 300s: 75K sweeps. More than enough.

OK let me now write the final code. I'll keep it focused and efficient.

One more thought: the previous code's weakness was that it used `minimize_scalar` (scipy) for each coordinate, which is slow (designs ~20-30 function evaluations, each O(n²)). My fast CD avoids this entirely.

Let me also add a smooth-gradient-based phase for the "valley" between local optima. The log-sum-exp smoothed objective is differentiable, so I can use scipy minimize with a gradient.

- Phase 3 (30%): More CD + periodic SLSQP restarts
- Phase 4 (15%): CD with perturbation (restart from best + noise, polish)
- Phase 5 (5%): Final CD polish

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    vec = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    s = vec.sum()
    if abs(s - target) < 1e-12:
        return vec
    lo = -np.max(vec) - 1.0
    hi = 1.0 - np.min(vec) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    if T < 1:
        return m * dx
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def fast_cd(h, n, t_end, max_sweeps=200000):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    two_n_1 = 2 * n - 1
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for p in range(n):
            old = h[p]
            k1_lo, k1_hi = p, min(p + n - 1, two_n_1 - 1)
            k2_lo, k2_hi = max(0, n - 1 - p), min(two_n_1 - 1, two_n_1 - 1 - p)
            js1 = np.arange(n - 1, n - 1 - (k1_hi - k1_lo), -1) if k1_hi >= k1_lo else np.array([], dtype=int)
            js2 = np.arange(0, k2_hi - k2_lo + 1) if k2_hi >= k2_lo else np.array([], dtype=int)
            c1 = (1.0 - h[js1]) if len(js1) else np.array([])
            c2 = h[js2] if len(js2) else np.array([])
            base_max = np.max(corr)
            best_v, best_m = old, base_max
            for v in (0.0, 1.0, old * 0.5, 1.0 - (1.0 - old) * 0.5):
                if v < 0 or v > 1:
                    continue
                d = v - old
                if abs(d) < 1e-13:
                    continue
                nm = base_max
                if k1_hi >= k1_lo:
                    affected = corr[k1_lo:k1_hi + 1] + d * c1
                    am = np.max(affected)
                    if am < nm:
                        # check if nm was at affected entry
                        nm = max(np.max(corr), am) if am > nm else nm
                    nm = max(nm, am)
                    if am < nm:
                        nm = am if am > np.max(np.concatenate([corr[:k1_lo], corr[k1_hi+1:k2_lo], corr[k2_hi+1:]])) else np.max(np.concatenate([corr[:k1_lo], corr[k1_hi+1:k2_lo], corr[k2_hi+1:]]))
                if k2_hi >= k2_lo:
                    affected2 = corr[k2_lo:k2_hi + 1] - d * c2
                    nm = max(nm, np.max(affected2))
                # Proper check: recompute max
                vals = corr.copy()
                if k1_hi >= k1_lo:
                    vals[k1_lo:k1_hi + 1] += d * c1
                if k2_hi >= k2_lo:
                    vals[k2_lo:k2_hi + 1] -= d * c2
                nm = np.max(vals)
                if nm < best_m - 1e-13:
                    best_m = nm
                    best_v = v
            d = best_v - old
            if abs(d) > 1e-13:
                if k1_hi >= k1_lo:
                    corr[k1_lo:k1_hi + 1] += d * c1
                if k2_hi >= k2_lo:
                    corr[k2_lo:k2_hi + 1] -= d * c2
                h[p] = best_v
                improved = True
        if sweep % 8 == 7:
            h = project(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full').copy()
        c5_cur = np.max(corr) * dx
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Intensive fast CD (35% budget)
    t1 = t0 + budget_s * 0.35
    best_h, best_c5 = fast_cd(best_h, n, t1, max_sweeps=500000)

    # Phase 2: SLSQP smoothed (10% budget)
    t2 = t0 + budget_s * 0.45
    for T in [50, 200, 800, 3000]:
        if elapsed() > budget_s * 0.44:
            break
        hc = slsqp_smooth(best_h.copy(), T, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = hc

    # Phase 3: CD + diversification (30% budget)
    t3 = t0 + budget_s * 0.75
    work_h = best_h.copy()
    work_c5 = best_c5
    n_div = 12
    for it in range(n_div):
        if elapsed() > budget_s * 0.74:
            break
        cool = 1.0 - it / float(n_div)
        pert = 0.03 + 0.08 * cool
        h_div = work_h + pert * np.random.randn(n)
        h_div = project(h_div, n)
        # Quick SLSQP
        T_s = int(500 * cool) + 200
        h_div = slsqp_smooth(h_div, T_s, n, maxiter=300)
        # CD polish
        t_polish = min(t3, time.time() + budget_s * 0.012)
        h_div, c5_div = fast_cd(h_div, n, t_polish, max_sweeps=50000)
        temp = 0.002 * cool + 0.0002
        if c5_div < work_c5 or np.random.rand() < np.exp(-(c5_div - work_c5) / max(temp, 1e-9)):
            work_h = h_div
            work_c5 = c5_div
            if c5_div < best_c5 - 1e-14:
                best_c5 = c5_div
                best_h = h_div

    # Phase 4: Final intensive CD (15% budget)
    t4 = t0 + budget_s * 0.90
    best_h, best_c5 = fast_cd(best_h, n, t4, max_sweeps=500000)

    # Phase 5: Final SLSQP + CD (8% budget)
    for T in [3000, 10000, 30000]:
        if elapsed() > budget_s * 0.96:
            break
        hc = slsqp_smooth(best_h.copy(), T, n, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = hc

    # Final CD polish
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.99, max_sweeps=100000)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
