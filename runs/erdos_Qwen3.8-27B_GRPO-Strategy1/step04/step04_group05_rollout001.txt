def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_gradient(h, N, alpha):
    corr = compute_corr(h)
    shifted = corr - corr.max()
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    center = N - 1
    grad = np.zeros(N)
    threshold = w.max() * 0.005
    active = np.where(w > threshold)[0]
    for ai in active:
        wi = w[ai]
        if wi < 1e-6:
            continue
        L = ai - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad, float(corr.max())

def steepest_descent_binary(h, N, max_sweeps=300, deadline=None, subset=50):
    h = (h > 0.5).astype(float)
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 5 or len(zeros) < 5:
            break
        c = compute_corr(h)
        P = float(c.max())
        idx = np.arange(2 * N - 1)
        d_lags = idx - (N - 1)
        p_idx = np.arange(N)[:, None]
        d_vals = d_lags[None, :]
        pd = p_idx + d_vals
        p_md = p_idx - d_vals
        valid_pd = (pd >= 0) & (pd < N)
        valid_pmd = (p_md >= 0) & (p_md < N)
        h_pd = np.where(valid_pd, h[np.clip(pd, 0, N-1)], 0.0)
        h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N-1)], 0.0)
        D = -(1.0 - h_pd) * valid_pd.astype(float) + h_pmd * valid_pmd.astype(float)
        best_peak = P
        best_pq = None
        step = max(1, len(ones) // subset)
        for p in ones[::step]:
            base = c - D[p, :]
            candidates = base[None, :] + D[zeros, :]

                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h
