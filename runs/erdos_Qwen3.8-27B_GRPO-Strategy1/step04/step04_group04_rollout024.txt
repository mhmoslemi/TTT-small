                return bh, best_c5
            old = bh[i]
            new_v = 1.0 - old
            a_idx = i + lags
            b_idx = i - lags
            va = (a_idx >= 0) & (a_idx < n)
            vb = (b_idx >= 0) & (b_idx < n)
            coef = np.zeros(2 * n - 1)
            if va.any():
                coef[va] = (1.0 - bh[a_idx[va]])
            if vb.any():
                coef[vb] += (-bh[b_idx[vb]])
            delta = new_v - old
            corr_new = corr + delta * coef
            corr_new[k0] = corr[k0] + new_v * (1.0 - new_v) - old * (1.0 - old)
            c_try = float(np.max(corr_new)) * dx
            if c_try < best_c5 - 1e-14:
                bh[i] = new_v
                corr = corr_new
                best_c5 = c_try
                improved = True
        if not improved:
            break
    return bh, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup (up to 60s)
    for T in [50, 300, 2000]:
        if elapsed() > 60:
            break
        try:
            hc = slsqp_step(best_h, n, T, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Coordinate descent (up to 250s)
    best_h, best_c5 = coord_descent(best_h, n, t0 + 250, max_sweeps=8000)

    # Phase 3: Perturbation + re-descent cycles (up to 550s)
    n_cyc = 30
    for cyc in range(n_cyc):
        if elapsed() > 550:
            break
        cool = 1.0 - 0.85 * (cyc / float(n_cyc))
        step = 0.01 + 0.08 * cool
        h_pert = best_h + step * np.random.randn(n)
        h_pert = proj(h_pert, n)
        h_try, c5_try = coord_descent(h_pert, n, min(t0 + 550, t0 + elapsed() + 10), max_sweeps=500)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 4: Binary local search (up to 650s)
    bh, c5_b = binary_local_search(best_h, n, t0 + 650, max_rounds=40)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = bh
        best_h, best_c5 = coord_descent(best_h, n, t0 + 680, max_sweeps=2000)

    # Phase 5: More perturbation cycles (up to 800s)
    for cyc in range(20):
        if elapsed() > 800:
            break
        cool = 1.0 - 0.7 * (cyc / 20.0)
        step = 0.005 + 0.04 * cool
        h_pert = best_h + step * np.random.randn(n)
        h_pert = proj(h_pert, n)
        h_try, c5_try = coord_descent(h_pert, n, min(t0 + 800, t0 + elapsed() + 8), max_sweeps=400)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 6: Multi-resolution (up to 900s)
    for n_try in [120, 144, 200, 240]:
        if elapsed() > 900:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = proj(h_interp, n_try)
        try:
            hc = slsqp_step(h_interp, n_try, 1000, maxiter=400)
        except Exception:
            hc = h_interp
        hc, c5v = coord_descent(hc, n_try, min(t0 + 900, t0 + elapsed() + 15), max_sweeps=300)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            n = n_try

    # Phase 7: Final coordinate descent polish (up to 1050s)
    best_h, best_c5 = coord_descent(best_h, n, t0 + 1050, max_sweeps=10000)

    # Phase 8: Focused SLSQP (up to 1090s)
    if elapsed() < 1090:
        try:
            hc = focused_slsqp(best_h, n, maxiter=1000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    # Final coord descent
    if elapsed() < 1095:
        best_h, best_c5 = coord_descent(best_h, n, t0 + 1098, max_sweeps=3000)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
