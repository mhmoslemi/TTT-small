import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    vec = np.array(h, dtype=float)
    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -vec.max() - 1.0
    hi = 1.0 - vec.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def make_dc(h, i, old, delta, n):
    """Compute the change vector for corr when h[i] changes by delta from old."""
    dc = np.zeros(2 * n - 1)
    dc[n - 1] = delta * (1.0 - 2.0 * old) - delta * delta
    m1s = np.arange(n)
    mask = m1s != i
    js_h = i - m1s[mask] + (n - 1)
    valid_h = (js_h >= 0) & (js_h < 2 * n - 1)
    dc[js_h[valid_h]] += delta * (1.0 - h[m1s[mask][valid_h]])
    m2s = np.arange(n)
    mask2 = m2s != i
    js_c = m2s[mask2] - i + (n - 1)
    valid_c = (js_c >= 0) & (js_c < 2 * n - 1)
    dc[js_c[valid_c]] += -delta * h[m2s[mask2][valid_c]]
    return dc

def eval_1d(corr, h, i, x, n):
    """Evaluate C5 if h[i] were x, without modifying h or corr."""
    old = h[i]
    delta = x - old
    if abs(delta) < 1e-14:
        return np.max(corr) * (2.0 / n)
    dc = make_dc(h, i, old, delta, n)
    return np.max(corr + dc) * (2.0 / n)

def commit(h, corr, i, x, n):
    """Commit change h[i] -> x, updating corr in place."""
    old = h[i]
    delta = x - old
    if abs(delta) < 1e-15:
        return
    corr += make_dc(h, i, old, delta, n)
    h[i] = x

def fast_cd(h, n, t_end, max_sweeps, sa_temp):
    """Fast coordinate descent with optional simulated annealing."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    no_improve = 0
    max_no_improve = 5 if sa_temp > 0 else 2
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            # Quick check: try x=0, x=1, and use minimize_scalar
            def f1d(x):
                return eval_1d(corr, best_h, i, x, n)
            try:
                res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-9})
                x_new = res.x
                c5_new = f1d(x_new)
            except Exception:
                x_new = old
                c5_new = best_c5
            
            accept = False
            if c5_new < best_c5 - 1e-13:
                accept = True
            elif sa_temp > 1e-12 and c5_new < best_c5 + 5 * sa_temp:
                if np.random.rand() < np.exp(-(c5_new - best_c5) / sa_temp):
                    accept = True
            
            if accept:
                commit(best_h, corr, i, x_new, n)
                best_c5 = compute_c5_fast(corr) if False else eval_1d(corr, best_h, i, best_h[i], n)
                # Actually just recompute from corr
                best_c5 = np.max(corr) * (2.0 / n)
                improved = True
        
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= max_no_improve and sa_temp <= 1e-12:
            break
        if sa_temp > 1e-12:
            sa_temp = sa_temp * 0.995
    
    return best_h, best_c5

    """Golden section search for minimum of f on [a,b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)

        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0
