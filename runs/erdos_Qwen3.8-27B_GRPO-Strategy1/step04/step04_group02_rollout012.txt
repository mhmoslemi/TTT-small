import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(5):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            scale = max(s, 1e-14)
            h = h - h * (abs(diff) / scale)
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def corr_grad(h, s, n):
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] += (-h[:n-s])
    else:
        sp = -s
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] += (-h[sp:])
    return g

def smooth_fval(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ea = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(ea))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ea = np.clip(T * (corr - m), -500, 500)
    w = np.exp(ea)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-9:
            continue
        s = idx - (n - 1)
        grad += w[idx] * corr_grad(h, s, n)
    return grad * dx

def slsqp_opt(h, T, n, maxiter=1500, ftol=1e-15):
    def obj(x):
        return smooth_fval(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def nesterov_smooth(h, T, n, t_end, lr0=0.005, max_iter=8000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m1 = h.copy()
    lr = lr0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        beta = it / (it + 2.5)
        y = h + beta * (h - m1) if it > 0 else h.copy()
        m1 = h.copy()
        y = np.clip(y, 0.0, 1.0)
        g = smooth_grad(y, T)
        for _ in range(6):
            h_new = project(y - lr * g, n)
            f_new = compute_c5(h_new)
            if f_new <= best_c5 + 1e-14:
                break
            lr *= 0.5

            best_c5 = f_new
            best_h = h_new.copy()

        else:
            lr = max(lr * 0.8, 1e-9)
        h = h_new
        if lr < 1e-10:
            break
    return best_h, best_c5
