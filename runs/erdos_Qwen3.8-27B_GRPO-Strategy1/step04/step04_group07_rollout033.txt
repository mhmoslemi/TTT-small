```python
def compute_c5(h):

    dx = 2.0 / n

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:

            return h
        h = h * (1.0 + diff / s)

        thresh = np.max(corr) - 0.05 * np.std(corr)
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _eval_1d(v, corr, coeff, c0_base, c0_idx, old):
    cn = corr + (v - old) * coeff
    cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
    return np.max(cn)

def _smooth_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def _con_sum(x, n):
    return x.sum() - n / 2.0

def coord_descent(h, n, t_end, max_sweeps=3000):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    tol = 1e-14
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if sweep % 3 == 0:
            order = range(n)
        elif sweep % 3 == 1:
            order = range(n - 1, -1, -1)
        else:
            step = 2
            order = list(range(0, n, step)) + list(range(1, n, step))
        for i in order:
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            res = minimize_scalar(_eval_1d, args=(corr, coeff, c0_base, c0_idx, old),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            bv, bf = res.x, res.fun
            f0 = _eval_1d(0.0, corr, coeff, c0_base, c0_idx, old)
            f1 = _eval_1d(1.0, corr, coeff, c0_base, c0_idx, old)
            if f0 < bf:
                bv, bf = 0.0, f0
            if f1 < bf:
                bv, bf = 1.0, f1
            if bf * dx < best_c5 - tol:
                best_h[i] = bv
                corr = corr + (bv - old) * coeff
                corr[c0_idx] = c0_base + (bv - old) * (1.0 - bv - old)
                best_c5 = bf * dx
                improved = True
        if not improved:
            break
    return best_h, best_c5

def perturb_cd(h, n, t_end, n_pert=40, amp=0.012):
    best_h, best_c5 = h.copy(), compute_c5(h)
    for _ in range(n_pert):
        if time.time() > t_end:
            break
        h_p = h.copy()
        mode = np.random.randint(3)
        if mode == 0:
            idx = np.random.randint(0, n)
            h_p[idx] = np.clip(h_p[idx] + np.random.uniform(-amp, amp), 0, 1)
        elif mode == 1:
            idx = np.random.randint(0, n - 3)
            h_p[idx:idx+4] = np.clip(h_p[idx:idx+4] + np.random.uniform(-amp, amp, 4), 0, 1)
        else:
            n_move = np.random.randint(3, 10)
            idxs = np.random.choice(n, n_move, replace=False)
            h_p[idxs] = np.clip(h_p[idxs] + np.random.uniform(-amp*0.5, amp*0.5, n_move), 0, 1)
        h_p = project_h(h_p, n)
        h_c, c5_c = coord_descent(h_p, n, t_end, max_sweeps=200)
        if c5_c < best_c5:
            best_h, best_c5 = h_c, c5_c
            h = h_c
    return best_h, best_c5

def sa_coord(h, n, t_end, max_iter=200, t_start=0.005, t_end_sa=0.0005):
    best_h, best_c5 = h.copy(), compute_c5(h)
    work_h, work_c5 = h.copy(), best_c5
    for it in range(max_iter):
        if time.time() > t_end:
            break
        temp = t_start * (t_end_sa / t_start) ** (it / max_iter)
        idx = np.random.randint(0, n)
        h_p = work_h.copy()
        h_p[idx] = np.clip(h_p[idx] + np.random.uniform(-0.08, 0.08), 0, 1)
        h_p = project_h(h_p, n)
        c5_p = compute_c5(h_p)
        if c5_p < work_c5 or np.random.rand() < np.exp(-(c5_p - work_c5) / temp):
            work_h, work_c5 = h_p, c5_p
            # Quick CD refinement
            h_c, c5_c = coord_descent(work_h, n, t_end, max_sweeps=30)
            work_h, work_c5 = h_c, c5_c
            if c5_c < best_c5:
                best_h, best_c5 = h_c, c5_c
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new)

def slsqp_opt(h0, T, n, maxiter=400):
    res = minimize(_smooth_obj, h0, args=(T, n), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _con_sum, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_h(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        best_h[::2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Thorough CD
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.18, max_sweeps=2000)

    # Phase 2: Diverse perturbations
    best_h, best_c5 = perturb_cd(best_h, n, t0 + budget_s * 0.32, n_pert=30, amp=0.015)

    # Phase 3: SA + CD
    best_h, best_c5 = sa_coord(best_h, n, t0 + budget_s * 0.42, max_iter=150)

    # Phase 4: More CD
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.50, max_sweeps=2000)

    # Phase 5: Resolution sweep
    for n_try in [120, 200, 240, 320]:
        if elapsed() > budget_s * 0.68:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.69, max_sweeps=800)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try
            best_h, best_c5 = perturb_cd(best_h, n, t0 + budget_s * 0.70, n_pert=5, amp=0.01)
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.72, max_sweeps=1000)

    # Phase 6: SLSQP with T-annealing
    for T in [100, 500, 2000, 8000]:
        if elapsed() > budget_s * 0.82:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 7: Final CD + perturbations
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.91, max_sweeps=3000)
    best_h, best_c5 = perturb_cd(best_h, n, t0 + budget_s * 0.96, n_pert=15, amp=0.008)

    # Final polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=1000)

    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
