import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    center = N - 1
    lags = np.arange(2 * N - 1) - center
    p_idx = np.arange(N)[:, None]
    L_vals = lags[None, :]
    valid_plus = (p_idx + L_vals >= 0) & (p_idx + L_vals < N)
    valid_minus = (p_idx - L_vals >= 0) & (p_idx - L_vals < N)
    h_plus = np.where(valid_plus, h[np.clip(p_idx + L_vals, 0, N-1)], 0.0)
    h_minus = np.where(valid_minus, h[np.clip(p_idx - L_vals, 0, N-1)], 0.0)
    return -(1.0 - h_plus) * valid_plus.astype(np.float64) + h_minus * valid_minus.astype(np.float64)

def steepest_descent(h, N, max_sweeps, deadline):

    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        D = compute_D(h, N)
        c = np.correlate(h, 1.0 - h, mode='full')
        P = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_peak = P
        best_pq = None
        # For each one position, find best zero to swap with

            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_peak:

                best_pq = (p, int(zeros[mi]))
        if best_pq is None:

        h[best_pq[1]] = 1.0

            if new_pk < best_pk:

                best = h_b.copy()

        else:

    zeros = np.where(h_b == 0)[0]

    for it in range(n_iters):
        if time.time() > deadline:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        # O(N) evaluation

        delta = new_c_max - cur_pk
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-9)):
            # Accept

            # Update D: recompute (or update incrementally)

            D = compute_D(h_b, N)  # This is O(N²), but we do it less often

            zeros = np.where(h_b == 0)[0]

                best_pk = cur_pk
                best = h_b.copy()
        # else reject
    return best, best_pk * (2.0 / N)

A better approach: compute D once, then for each candidate swap, evaluate in O(N). Only recompute D after an accepted swap. Since in annealing, the acceptance rate drops as T decreases, we'll have many O(N) evaluations per O(N²) D recompute.

Actually, even better: we don't need to recompute D after each accepted swap if we update it incrementally. When h[p] changes from 1 to 0:
- D[q, li] changes because h[p±L] appears in D[q, li]
- Specifically, D[q, li] = -(1-h[q+L])*[valid] + h[q-L]*[valid]
- If h[p] changes, D[q, li] changes only if q+L = p or q-L = p, i.e., L = p-q or L = q-p

So for each accepted swap (p: 1→0, q: 0→1), we update:
- D for all rows where the lag L = p-q or q-p is relevant
- This affects D[p, *] (the row for p, which is now 0), D[q, *] (row for q, now 1)
- And D[r, li] for any r where r+L=p or r-L=p (i.e., r = p±L for the affected lag)
