import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_wf(h, n):
    """Water-filling projection onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def cd_fast(h, n, t_end, max_sweeps=500):
    bh = h.copy()
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    c = n - 1
    dx = 2.0 / n
    bc5 = float(np.max(corr) * dx)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return bh, bc5
            k_lo = max(0, -i)
            k_hi = min(n, n - i)
            if k_hi <= k_lo:
                continue
            ks = np.arange(k_lo, k_hi)
            a = np.zeros(2 * n - 1)
            a[c + ks] = 1.0 - 2.0 * bh[i + ks]
            ov = bh[i]
            def f1d(d):
                v = corr + d * a
                v[c] -= d * d
                return np.max(v)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                               options={'xatol': 1e-12})
            nv = float(res.x)
            if abs(nv - ov) > 1e-11:
                d = nv - ov
                corr += d * a
                corr[c] -= d * d
                bh[i] = nv
                nc5 = float(np.max(corr) * dx)
                if nc5 < bc5:
                    bc5 = nc5
                    imp = True
        if not imp:
            break
    return bh, bc5

def lbfgs_smooth(h, n, T, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = c.max()
        v = T * (c - mx)
        v[v > 400] = 400
        return (mx + np.log(np.sum(np.exp(v))) / T) * dx
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = c.max()
        v = T * (c - mx)
        v[v > 400] = 400
        w = np.exp(v) / np.sum(np.exp(v))
        g = np.zeros(n)
        ctr = n - 1
        for m in range(2 * n - 1):
            if w[m] < 1e-14:
                continue
            k = m - ctr
            wm = w[m]
            if k == 0:
                g[:] += wm * (1.0 - 2.0 * x)
            elif k > 0:
                g[:n-k] += wm * (1.0 - x[k:n])
                g[k:] -= wm * x[:n-k]
            else:
                mk = -k
                g[mk:] += wm * (1.0 - x[:n-mk])
                g[:n-mk] -= wm * x[mk:]
        return g * dx
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n, options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj_wf(res.x, n)

def perturb_cd(best_h, n, t_check, max_sweeps=15):
    """Gradient perturbation + quick CD."""
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    mx = corr.max()
    ctr = n - 1
    act = np.where(corr >= mx - 0.03 * n)[0]
    if len(act) == 0:
        act = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for idx in act:
        k = idx - ctr
        wm = 1.0 / len(act)
        if k == 0:
            grad += wm * (1.0 - 2.0 * best_h)
        elif k > 0:
            grad[:n-k] += wm * (1.0 - best_h[k:])
            grad[k:] -= wm * best_h[:n-k]
        else:
            mk = -k
            grad[mk:] += wm * (1.0 - best_h[:n-mk])
            grad[:n-mk] -= wm * best_h[mk:]
    best_hc, best_c5c = best_h.copy(), compute_c5(best_h)
    for ss in [0.02, 0.06, 0.12, 0.20]:
        if time.time() > t_check:
            break
        hp = proj_wf(best_h - ss * grad + 0.02 * np.random.randn(n), n)
        ht, cv = cd_fast(hp, n, t_check, max_sweeps)
        if cv < best_c5c:
            best_hc, best_c5c = ht, cv
    return best_hc, best_c5c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.3*np.sin(np.pi*x) + 0.1*np.cos(2*np.pi*x), 0, 1)
    n = len(best_h)
    best_h = proj_wf(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: CD convergence
    best_h, best_c5 = cd_fast(best_h, n, t0 + budget_s * 0.10, 400)

    # Phase 2: L-BFGS smooth at multiple resolutions and T
    for ntr in [n, 120, 200, 240]:
        if time.time() > t0 + budget_s * 0.35:
            break
        if ntr == n:
            ht = best_h.copy()
        else:
            xo = np.linspace(0, 2, n, endpoint=False)
            xn = np.linspace(0, 2, ntr, endpoint=False)
            ht = np.interp(xn, xo, best_h, period=2.0)
            ht = proj_wf(ht, ntr)
        for T in [50, 200, 1000, 5000]:
            if time.time() > t0 + budget_s * 0.35:
                break
            try:
                hc = lbfgs_smooth(ht, ntr, T, 250)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    n = ntr
            except Exception:
                pass
        best_h, best_c5 = cd_fast(proj_wf(best_h, n), n,
                                  min(time.time()+4, t0+budget_s*0.35), 50)

    # Phase 3: Perturbation + CD cycles (escape local optima)
    for cyc in range(60):
        if time.time() > t0 + budget_s * 0.72:
            break
        np.random.seed(seed + cyc * 7)
        ht, cv = perturb_cd(best_h, n, t0 + budget_s * 0.72, 12)
        if cv < best_c5:
            best_c5 = cv
            best_h = ht
            best_h, best_c5 = cd_fast(best_h, n,
                                      min(time.time()+3, t0+budget_s*0.72), 40)

    # Phase 4: Final high-T L-BFGS polish
    for T in [5000, 20000, 80000]:
        if time.time() > t0 + budget_s * 0.88:
            break
        try:
            hc = lbfgs_smooth(best_h, n, T, 400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 5: Final coordinate descent
    best_h, best_c5 = cd_fast(best_h, n, t_end - 2, 500)
    best_h = proj_wf(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
