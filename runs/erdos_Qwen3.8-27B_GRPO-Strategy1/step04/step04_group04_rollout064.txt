```python
def soft_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = np.exp(np.clip(T * (corr - m), -500, 500))
    p = e / e.sum()
    ones_minus_h = 1.0 - h
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if p[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # (1-h)[i-lag] for i in [lag, n): contributes (1-h[i-lag])
            # h[i+lag] for i in [0, n-lag): contributes -h[i+lag]
            if n - lag > 0:
                grad[lag:] += p[k] * ones_minus_h[:n-lag]
                grad[:n-lag] -= p[k] * h[lag:]
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += p[k] * ones_minus_h[:n-nl]
                grad[:n-nl] -= p[k] * h[nl:]
    return grad * dx
```

                if v.max() * dx < best_val:
                    hi_b = mid
                else:
                    lo = mid
                if hi_b - lo < 1e-11:
                    break
            t_new = 0.5 * (lo + hi_b)
            d = t_new - hi
            if abs(d) > 1e-12:
                v = corr + d * coeff
                v[n - 1] -= d * d
                new_c5 = float(v.max()) * dx
                if new_c5 < best_c5 - 1e-15:
                    best_h[i] = t_new
                    corr += d * coeff
                    corr[n - 1] -= d * d
                    best_c5 = new_c5
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = np.exp(np.clip(T * (corr - m), -500, 500))
    p = e / e.sum()
    omh = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            L = n - lag
            grad[lag:] += pk * omh[:L]
            grad[:L] -= pk * h[lag:]
        else:
            nl = -lag
            L = n - nl
            grad[nl:] += pk * omh[:L]
            grad[:L] -= pk * h[nl:]
    return grad * dx

def proj_grad_descent(h, n, t_end, T_init=200, lr_init=0.003):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    T = T_init
    for it in range(5000):
        if time.time() > t_end:
            break
        g = grad_smooth(best_h, T)
        step = lr
        for _ in range(30):
            trial = proj(best_h - step * g, n)
            c5_t = compute_c5(trial)
            if c5_t < best_c5:
                best_h = trial
                best_c5 = c5_t
                T = min(T * 1.04, 50000)
                break
            step *= 0.5
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    for T in [30, 150, 600, 3000, 15000]:
        if time.time() - t0 > budget_s * 0.05:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    if time.time() - t0 < budget_s * 0.12:
        best_h, best_c5 = proj_grad_descent(best_h, n, t0 + budget_s * 0.12, T_init=200)

    if time.time() - t0 < budget_s * 0.38:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.38, max_sweeps=500)

    for T in [20000, 80000, 300000]:
        if time.time() - t0 > budget_s * 0.48:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=600, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    if time.time() - t0 < budget_s * 0.62:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.62, max_sweeps=400)

    work_h, work_c5 = best_h.copy(), best_c5
    for it in range(60):
        if time.time() - t0 > budget_s * 0.78:
            break
        cool = 1.0 - 0.88 * (it / 60.0)
        step_sz = 0.03 + 0.14 * cool
        h_p = proj(work_h + step_sz * np.random.randn(n), n)
        try:
            hc = slsqp_opt(h_p, 3000, n, maxiter=250)
            c5v = compute_c5(hc)
            temp = 0.0015 * cool + 0.0002
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h, work_c5 = hc, c5v
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
        except Exception:
            pass
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h

    for n_try in [120, 100, 90, 140, 160]:
        if time.time() - t0 > budget_s * 0.90:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = proj(np.interp(x_new, x_old, best_h), n_try)
        for T in [3000, 30000]:
            if time.time() - t0 > budget_s * 0.92:
                break
            try:
                hc = slsqp_opt(h_int, T, n_try, maxiter=400)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h, n = c5v, hc, n_try
                    h_int = hc
            except Exception:
                pass
        if time.time() - t0 < budget_s * 0.94:
            hc, c5v = coord_descent(h_int, n_try, t0 + budget_s * 0.94, max_sweeps=60)
            if c5v < best_c5:
                best_c5, best_h, n = c5v, hc, n_try

    nf = len(best_h)
    if time.time() - t0 < budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, nf, t0 + budget_s * 0.96, max_sweeps=100)
    if time.time() - t0 < budget_s * 0.99:
        try:
            hc = slsqp_opt(best_h, 200000, nf, maxiter=500, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass
    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
