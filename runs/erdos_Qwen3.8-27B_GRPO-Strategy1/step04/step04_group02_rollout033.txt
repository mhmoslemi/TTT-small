    n = len(h)
    grad = np.zeros(n)
    for j in range(n):

            if 0 <= j + sigma < n:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

    m = np.max(corr)
    if T * m > 700:
        return m * dx
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project(h, n):

    target = n / 2.0

    diff = target - s

def compute_smooth_grad(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T * (m - np.min(corr)) > 700:
        # Very peaked, just use the max
        p = np.zeros(len(corr))
        p[np.argmax(corr)] = 1.0
    else:
        p = np.exp(T * (corr - m))
        p /= p.sum()
    grad = np.zeros(n)

        if w < 1e-12:

import numpy as np
from scipy.optimize import minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        tot = s
        if tot < 1e-13:
            return h
        h = h - h * (abs(diff) / tot)
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-13:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-13:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-13 and s2 > 1e-13:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def smooth_val(corr, T, dx):
    m = np.max(corr)
    if T * m > 700:
        return m * dx
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def coord_opt_fixed(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += (1.0 - h[i - k])
        if 0 <= i + k < n:
            dk += (-h[i + k])
        d[n - 1 + k] = dk
        if 0 <= i + k < n:
            dk2 = (1.0 - h[i + k])
        if 0 <= i - k < n:
            dk2 += (-h[i - k])
        d[n - 1 - k] = dk2
    c0 = corr_base[n - 1]

    best_h = h.copy()

def nesterov_grad_descent(h, n, t_end, T_start=20, T_end=2000, max_iter=3000):
    dx = 2.0 / n
    h_y = h.copy()  # momentum position
    v = np.zeros(n)  # velocity
    lr = 0.5
    best_h = h.copy()
    best_c5 = compute_c5(best_h)[0]
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        frac = it / max_iter
        T = T_start * (T_end / T_start) ** (frac * 0.7)  # slow increase
        
        grad, corr = smooth_grad(h_y, T, dx)
        # Project gradient: remove mean component
        grad -= grad.mean()
        
        # Nesterov update
        momentum = 0.9
        v = momentum * v - lr * grad
        h_new = project(h_y + v, n)
        
        # Line search (simple)
        new_s = smooth_val(corr, T, dx)  # old value (approx)
        for _ in range(10):
            test_corr = np.correlate(h_new, 1-h_new, mode='full')
            test_s = smooth_val(test_corr, T, dx)
            if test_s <= new_s + 1e-15:
                break
            v *= 0.5
            h_new = project(h_y + v, n)
            new_s = smooth_val(test_corr, T, dx)
        
        h_y = h_new
        c5 = compute_c5(h_y)[0]
        if c5 < best_c5:
            best_h = h_y.copy()
            best_c5 = c5
        
        # Decay learning rate
        lr = max(0.01, lr * 0.999)
    
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)
