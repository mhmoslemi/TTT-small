                return np.max(corr + s * a)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-13})
            s_new = res.x
            if abs(s_new) > 1e-14:
                best_h[i] += s_new
                corr += s_new * a
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved and sweep > 3:
            break
    return best_h, best_c5

def sin_perturb(h, amp, freq, phase):
    n = len(h)
    x = np.linspace(0, 2 * np.pi * freq, n)
    return h + amp * np.sin(x + phase)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(128)
        best_h[:64] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup (4% budget)
    for T in [40, 400, 2000]:
        if elapsed() > budget_s * 0.04:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=400, ftol=1e-13)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 2: Intensive bidirectional coordinate descent (35% budget)
    t_cd = t0 + budget_s * 0.38
    best_h, best_c5 = coord_descent_bidir(best_h, n, t_cd, n_sweeps=800)

    # Phase 3: Multi-start with diverse perturbations + CD (27% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    n_starts = 50
    for it in range(n_starts):
        if elapsed() > budget_s * 0.65:
            break
        cool = max(1.0 - 0.85 * (it / float(n_starts)), 0.05)
        r = np.random.rand()
        if r < 0.30:
            step = 0.02 + 0.10 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.50:
            h_p = sin_perturb(work_h, 0.02 + 0.04 * cool,
                               np.random.randint(1, 5), np.random.uniform(0, 2 * np.pi))
        elif r < 0.70:
            shift = np.random.randint(-5, 6)
            h_sh = np.roll(work_h, shift)
            blend = 0.3 + 0.4 * np.random.rand()
            h_p = blend * h_sh + (1 - blend) * work_h
        elif r < 0.85:
            mask = np.random.rand(n) < 0.25
            h_p = work_h.copy()
            step = 0.03 + 0.12 * cool
            h_p[mask] += step * np.random.randn(mask.sum())
        else:
            h_sym = (work_h + work_h[::-1]) / 2.0
            h_p = h_sym + (0.01 + 0.04 * cool) * np.random.randn(n)
        h_p = project_to_sum(h_p, n / 2)
        try:
            T_val = max(80, int(1500 * cool + 50))
            hc = slsqp_polish(h_p, T_val, n, maxiter=250, ftol=1e-12)
            hc, c5v = coord_descent(hc, n, time.time() + 1.2, n_sweeps=15)
            accept = c5v < work_c5
            if not accept and c5v - work_c5 < 0.001:
                temp = 0.002 * cool + 0.00005
                accept = np.random.rand() < np.exp(-(c5v - work_c5) / temp)
            if accept:
                work_h, work_c5 = hc.copy(), c5v
                if c5v < best_c5:
                    best_h, best_c5 = hc.copy(), c5v
        except Exception:
            pass

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 4: Multi-resolution (8% budget)
    for n_try in [128, 160, 200, 256]:
        if n_try == n or elapsed() > budget_s * 0.74:
            continue
        try:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_i = np.interp(x_new, x_old, best_h)
            h_i = project_to_sum(h_i, n_try / 2)
            hc = slsqp_polish(h_i, 800, n_try, maxiter=300, ftol=1e-13)
            hc, c5v = coord_descent(hc, n_try, time.time() + 2.5, n_sweeps=25)
            if c5v < best_c5:
                best_c5, best_h, n = c5v, hc, n_try
        except Exception:
            pass

    # Phase 5: Final long coordinate descent (18% budget)
    t_final = t0 + budget_s * 0.93
    best_h, best_c5 = coord_descent_bidir(best_h, n, t_final, n_sweeps=400)

    # Phase 6: High-T SLSQP + final CD (7% budget)
    for T in [8000, 40000, 150000]:
        if elapsed() > budget_s * 0.96:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=600, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    if elapsed() < budget_s * 0.995:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, n_sweeps=40)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
