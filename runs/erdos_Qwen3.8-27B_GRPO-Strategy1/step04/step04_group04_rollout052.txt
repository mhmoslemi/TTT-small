import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, T):
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / T

def smooth_max_grad(corr, h, T, dx):
    n = len(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    g = np.zeros(n)
    for l in range(-(n-1), n):
        pk = p[l + (n-1)]
        if pk < 1e-15:
            continue
        js = max(0, l)
        je = min(n, n+l)
        if je > js:
            g[js:je] += pk * (1.0 - h[js-l:je-l])
        js2 = max(0, -l)
        je2 = min(n, n-l)
        if je2 > js2:
            g[js2:je2] += -pk * h[js2+l:je2+l]
    return g * dx

def lbfgs_smooth(h, T, mu, n, maxiter=500):
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        f = smooth_max_val(corr, T) * (2.0/n)
        pen = mu * (x.sum() - n/2.0)**2
        return f + pen
    def grad(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        g = smooth_max_grad(corr, x, T, 2.0/n)
        g += 2.0 * mu * (x.sum() - n/2.0)
        return g
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            coef[k] += (1.0 - h[j1])
        if 0 <= j2 < n:
            coef[k] += (-h[j2])
    k0 = n - 1
    h_save = h[i]
    def f1d(t):
        cv = corr_base + t * coef
        cv[k0] += t * (1.0 - t)
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            hi_new = coord_opt(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, T):
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / T

def smooth_max_grad(corr, h, T, dx):
    n = len(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        pk = p[l + (n - 1)]
        if pk < 1e-15:
            continue
        js = max(0, l)
        je = min(n, n + l)
        if je > js:
            g[js:je] += pk * (1.0 - h[js - l:je - l])
        js2 = max(0, -l)
        je2 = min(n, n - l)
        if je2 > js2:
            g[js2:je2] += -pk * h[js2 + l:je2 + l]
    return g * dx

def lbfgs_smooth(h, T, mu, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        f = smooth_max_val(corr, T) * dx
        pen = mu * (x.sum() - n / 2.0) ** 2
        return f + pen
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        g = smooth_max_grad(corr, x, T, dx)
        g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            coef[k] += (1.0 - h[j1])
        if 0 <= j2 < n:
            coef[k] += (-h[j2])
    k0 = n - 1
    def f1d(t):
        cv = corr_base + t * coef
        cv[k0] += t * (1.0 - t)
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            hi_new = coord_opt(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def pair_swap_opt(h, n, t_end, max_tries=2000):
    """Try swapping pairs of coordinates near boundaries."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

            h_try = best_h.copy()

            if c5_try < best_c5 - 1e-15:

                best_h = h_try
    return best_h, best_c5
