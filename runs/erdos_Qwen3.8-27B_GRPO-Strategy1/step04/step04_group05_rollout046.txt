        # new_corr[k] = corr[k] - (1-h[p-k])*[valid1_p] + h[p+k]*[valid2_p] + (1-h[q-k])*[valid1_q] - h[q+k]*[valid2_q]
        # = corr[k] + D[p,k] + (-D_neg[q,k])
        # where D_neg is the "0->1" change = -(1-h[q-k])*[v1_q] + h[q+k]*[v2_q]... 
        # Wait: 0->1 is the negative of 1->0 change.
        # 1->0 change at p: D[p,k] = -(1-h_p1)*v1 + h_p2*v2
        # 0->1 change at q: -D[q,k] (negated)
        # So new_corr[k] = corr[k] + D[p,k] - D[q,k]
        
        # To find best (p,q): minimize max_k of (corr[k] + D[p,k] - D[q,k])
        new_cands = corr[None, :, :] + D[ones, :][:, None, :] - D[zeros, :][None, :, :]
        peaks = new_cands.max(axis=2)
        best_pi = int(peaks.argmin())
        p_i = ones[best_pi // len(zeros)]
        q_i = zeros[best_pi % len(zeros)]
        new_pk = float(peaks[p_i % len(ones), q_i]) if False else float(peaks[best_pi // len(zeros), best_pi % len(zeros)])
        
        if new_pk < peak - 1e-10:
            h[p_i] = 0.0
            h[q_i] = 1.0
            corr = corr + D[p_i, :] - D[q_i, :]
            peak = new_pk
            tabu_counter += 1
            tabu[p_i] = tabu_counter
            tabu[q_i] = tabu_counter
            if peak < best_pk:
                best_pk = peak
                best_h = h.copy()
            sweep += 1
        else:
            break
    return best_h, best_pk * (2.0 / N)

def anneal_step(h, N, rng, T, deadline):
    h_b = (h >= 0.5).astype(float)
    nh = N // 2
    n1 = int(h_b.sum())
    if n1 > nh:
        idx = np.where(h_b == 1)[0]
        h_b[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:nh - n1]] = 1
    best = h_b.copy()
    best_pk = float(np.max(np.correlate(h_b, 1.0 - h_b, mode='full')))
    cur_pk = best_pk
    ones = np.where(h_b == 1)[0].tolist()
    zeros = np.where(h_b == 0)[0].tolist()
    it = 0
    while time.time() < deadline and it < 50000:
        it += 1
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        p, q = int(i), int(j)
        h_b[p], h_b[q] = 0.0, 1.0
        new_pk = float(np.max(np.correlate(h_b, 1.0 - h_b, mode='full')))
        delta = new_pk - cur_pk
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
            ones.remove(p)
            zeros.remove(q)
            ones.append(q)
            zeros.append(p)
        else:
            h_b[p], h_b[q] = 1.0, 0.0
        if it % 2000 == 0:
            T *= 0.85
    return best, best_pk * (2.0 / N)

def adam_continuous(h, N, target, deadline, k=60, lr0=0.03, decay=0.994):
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    center = N - 1
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wk * (1.0 - h[:N - L])
                grad[:N - L] -= wk * h[L:]
            else:
                M = -L
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_continuous(h, target)
        if it % 300 == 0:
            lr *= decay
    return best_h, best_peak * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    if N0 % 2 != 0:
        N0 -= 1
        h0 = h0[:N0]
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Steepest descent from initial
    d1 = deadline - 400
    h_bin = make_binary(h0, N0)
    h_s, c5_s = steepest_binary(h_bin, N0, d1, rng, max_sweeps=2000)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Iterated anneal + steepest descent
    d2 = deadline - 150
    h_cur = best_h.copy()
    for restart in range(30):
        if time.time() > d2:
            break
        rem = d2 - time.time()
        if rem < 15:
            break
        ann_deadline = time.time() + min(12.0, rem * 0.45)
        h_a, c5_a = anneal_step(h_cur, N0, rng, T=2.5, deadline=ann_deadline)
        sd_deadline = d2 - 3
        if time.time() < sd_deadline - 5:
            h_s, c5_s = steepest_binary(h_a, N0, sd_deadline, rng, max_sweeps=800)
        else:
            h_s, c5_s = h_a, c5_a
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
            h_cur = h_s.copy()

    # Phase 3: Continuous Adam polish
    if time.time() < deadline - 20:
        h_pol = best_h.astype(float) + rng.randn(N0) * 0.05
        h_pol = project_continuous(h_pol, N0 / 2.0)
        h_ad, c5_ad = adam_continuous(h_pol, N0, N0 / 2.0, deadline - 15, k=60, lr0=0.02, decay=0.995)
        if c5_ad < best_c5:
            best_c5, best_h = c5_ad, h_ad.copy()

    # Phase 4: Try different N values
    for dN in [4, -4, 8, -8, 12, 16]:
        N_try = N0 + dN
        if N_try < 200 or N_try % 2 != 0:
            continue
        if time.time() > deadline - 25:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_b = make_binary(h_interp, N_try)
        h_s, c5_s = steepest_binary(h_b, N_try, deadline - 8, rng, max_sweeps=600)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s
        if time.time() < deadline - 12 and c5_s <= best_c5 + 0.001:
            h_pol = h_s.astype(float) + rng.randn(N_try) * 0.04
            h_pol = project_continuous(h_pol, N_try / 2.0)
            h_ad, c5_ad = adam_continuous(h_pol, N_try, N_try / 2.0, deadline - 6, k=50, lr0=0.015, decay=0.996)
            if c5_ad < best_c5:
                best_c5, best_h = c5_ad, h_ad

    # Phase 5: Final Adam polish
    if time.time() < deadline - 4:
        hF = best_h.astype(float) + rng.randn(len(best_h)) * 0.02
        hF = project_continuous(hF, len(best_h) / 2.0)
        hF, c5F = adam_continuous(hF, len(best_h), len(best_h) / 2.0, deadline - 1, k=70, lr0=0.006, decay=0.998)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
