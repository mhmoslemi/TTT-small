    diff = target - s

    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    nc = len(corr)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            coef = np.zeros(nc)
            for k in range(-(n - 1), n):
                c = 0.0
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[k + n - 1] = c
            if top_k == 1:
                def f1d(s):
                    return np.max(corr + s * coef)
            else:
                def f1d(s):
                    nc2 = corr + s * coef
                    idx = np.argpartition(nc2, -top_k)[-top_k:]
                    return np.sum(nc2[idx])
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-13})
            s_new = res.x
            if abs(s_new) > 1e-14:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5 - 1e-16:
                    best_c5 = c5_new
                    improved += 1
        if improved == 0:
            break
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter=500, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def fourier_perturb(h, rng):
    n = len(h)
    K = rng.randint(2, 10)
    k = rng.randint(1, K)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.01, 0.06)
    x = np.arange(n)
    if rng.rand() < 0.5:
        pert = amp * np.cos(2 * np.pi * k * x / n + phase)
    else:
        pert = amp * np.sin(2 * np.pi * k * x / n + phase)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(128)
        best_h[:64] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup (5%)
    for T in [30, 100, 500, 2000]:
        if elapsed() > budget_s * 0.05:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=400, ftol=1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 2: Intensive CD (20%)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.18, n_sweeps=500, top_k=1)
    # top-2 smoothing pass
    best_h, _ = coord_descent(best_h, n, t0 + budget_s * 0.22, n_sweeps=100, top_k=2)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.25, n_sweeps=300, top_k=1)

    # Phase 3: Basin hopping (40%)
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 100
    t_bh_end = t0 + budget_s * 0.65
    rng = np.random.RandomState(seed)

    for it in range(n_bh):
        if elapsed() > t_bh_end:
            break
        cool = max(1.0 - 0.96 * (it / float(n_bh)), 0.02)
        r = rng.rand()
        if r < 0.25:
            step = 0.015 + 0.09 * cool
            h_p = work_h + step * rng.randn(n)
        elif r < 0.45:
            h_p = fourier_perturb(work_h, rng)
        elif r < 0.60:
            center = rng.randint(0, n)
            width = rng.randint(max(3, n // 20), n // 5)
            idx = np.arange(n)
            dist = np.minimum(np.abs(idx - center), n - np.abs(idx - center))
            mask = dist <= width
            step = 0.02 + 0.07 * cool
            h_p = work_h.copy()
            h_p[mask] += step * rng.randn(mask.sum())
        elif r < 0.75:
            size = rng.randint(max(4, n // 20), n // 8)
            i1 = rng.randint(0, n - size)
            i2 = rng.randint(0, n - size)
            h_p = work_h.copy()
            delta = (h_p[i1:i1+size] - h_p[i2:i2+size]) * rng.uniform(0.1, 0.4)
            h_p[i1:i1+size] -= delta
            h_p[i2:i2+size] += delta
        elif r < 0.85:
            h_p = work_h.copy()
            shift = rng.randint(1, max(2, n // 10))
            h_mix = np.roll(h_p, shift)
            mix = 0.15 + 0.25 * cool
            h_p = (1 - mix) * h_p + mix * h_mix
        else:
            # Swap pairs
            h_p = work_h.copy()
            nswaps = rng.randint(3, max(4, n // 10))
            for _ in range(nswaps):
                a, b = rng.randint(0, n, 2)
                if a != b:
                    amt = (h_p[b] - h_p[a]) * 0.5 * rng.uniform(0.2, 0.8)
                    h_p[a] += amt
                    h_p[b] -= amt

        h_p = project_to_sum(h_p, n / 2)
        try:
            T_bh = int(300 + 8000 * cool)
            hc = slsqp_polish(h_p, T_bh, n, maxiter=250, ftol=1e-12)
            hc, c5v = coord_descent(hc, n, time.time() + 2.5, n_sweeps=10, top_k=1)
            temp = 0.0025 * cool + 0.00005
            if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h, work_c5 = hc, c5v
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc.copy()
        except Exception:
            pass

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 4: Multi-resolution (10%)
    t_mr = t0 + budget_s * 0.78
    for n_try in [256, 192, 288]:
        if elapsed() > t_mr:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2)
        for T in [500, 3000]:
            if elapsed() > t_mr:
                break
            try:
                hc = slsqp_polish(h_i, T, n_try, maxiter=500, ftol=1e-13)
                hc, c5v = coord_descent(hc, n_try, time.time() + 4.0, n_sweeps=15, top_k=1)
                if c5v < best_c5:
                    best_c5, best_h, n = c5v, hc, n_try
            except Exception:
                pass

    # Phase 5: Final intensive polish (22%)
    t_f1 = t0 + budget_s * 0.93
    best_h, best_c5 = coord_descent(best_h, n, t_f1, n_sweeps=600, top_k=1)

    for T in [5000, 30000, 150000]:
        if elapsed() > budget_s * 0.97:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=1000, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, n_sweeps=80, top_k=1)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
