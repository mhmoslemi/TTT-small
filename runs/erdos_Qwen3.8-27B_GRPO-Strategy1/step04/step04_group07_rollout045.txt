    return project_h(h_p, n)

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new)

def gen_start(n, idx):
    m = idx % 6
    if m == 0:
        t = np.arange(n)
        h = 0.5 + 0.4 * np.sin(2.0 * np.pi * t * 2.0 / n)
    elif m == 1:
        t = np.arange(n)
        h = 0.5 + 0.35 * np.cos(2.0 * np.pi * t * 3.0 / n + 0.7)
    elif m == 2:
        h = np.random.rand(n)
    elif m == 3:
        t = np.arange(n)
        h = 0.5 + 0.3 * np.sin(2.0 * np.pi * t * 5.0 / n) + 0.15 * np.sin(2.0 * np.pi * t * 8.0 / n)
    elif m == 4:
        h = np.where(np.arange(n) % 4 < 2, 0.7, 0.3)
    else:
        h = 0.5 + 0.5 * (1.0 / (1.0 + np.exp(-3.0 * (np.arange(n) / n - 0.5)))) - 0.25
    return project_h(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = gen_start(160, 0)
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent on initial (n=160)
    t1 = t0 + budget_s * 0.12
    best_h, best_c5 = fast_coord_descent(best_h, n, t1, max_sweeps=2000)

    # Phase 2: Multi-start exploration with diverse perturbations
    t2 = t0 + budget_s * 0.50
    n_starts = 16
    for s in range(n_starts):
        if time.time() > t2:
            break
        scale = 0.06 + 0.35 * (s / float(n_starts))
        h_p = perturb_h(best_h, n, scale)
        t_sub = min(t2, time.time() + budget_s * 0.022)
        h_c, c5_c = fast_coord_descent(h_p, n, t_sub, max_sweeps=800)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c

    # Phase 3: Refine best found
    t3 = t0 + budget_s * 0.62
    best_h, best_c5 = fast_coord_descent(best_h, n, t3, max_sweeps=2000)

    # Phase 4: Multi-resolution search
    t4 = t0 + budget_s * 0.80
    for n_try in [120, 140, 180, 200, 240, 280, 320, 400]:
        if time.time() > t4:
            break
        h_tr = interp_h(best_h, n_try)
        t_sub = min(t4, time.time() + budget_s * 0.018)
        h_tr, c5_tr = fast_coord_descent(h_tr, n_try, t_sub, max_sweeps=1000)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 5: Second round of perturbations on best
    t5 = t0 + budget_s * 0.90
    for s in range(10):
        if time.time() > t5:
            break
        scale = 0.04 + 0.20 * (s / 10.0)
        h_p = perturb_h(best_h, n, scale)
        t_sub = min(t5, time.time() + budget_s * 0.008)
        h_c, c5_c = fast_coord_descent(h_p, n, t_sub, max_sweeps=600)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c

    # Phase 6: Final heavy coordinate descent
    t6 = t0 + budget_s * 0.98
    best_h, best_c5 = fast_coord_descent(best_h, n, t6, max_sweeps=3000)

    # Phase 7: Tiny final perturbations
    if time.time() < t0 + budget_s * 0.995:
        for _ in range(4):
            if time.time() > t0 + budget_s * 0.995:
                break
            h_p = perturb_h(best_h, n, 0.02)
            h_c, c5_c = fast_coord_descent(h_p, n, t0 + budget_s * 0.995, max_sweeps=400)
            if c5_c < best_c5:
                best_c5, best_h = c5_c, h_c

    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
