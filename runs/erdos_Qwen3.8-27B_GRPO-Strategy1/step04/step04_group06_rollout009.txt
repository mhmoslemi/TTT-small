import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, dx

def smooth_max_val(corr, dx, T):
    vals = corr * dx
    m = vals.max()
    if T < 1e4:
        return m + np.log(np.sum(np.exp(T * (vals - m)))) / T
    return m

def grad_smooth_max(h, n, T):
    corr, dx = get_corr(h)
    vals = corr * dx
    m = vals.max()
    if T < 1e4:
        exps = np.exp(T * (vals - m))
        weights = exps / exps.sum()
    else:
        weights = np.zeros(len(vals))
        idx_max = np.where(vals >= m - 1e-15)[0]
        weights[idx_max] = 1.0 / len(idx_max)
    grad = np.zeros(n)
    center = n - 1
    for idx in range(len(vals)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - center
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            g = np.zeros(n)
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
            grad += w * g
        else:
            kk = -k
            g = np.zeros(n)
            g[kk:] += (1.0 - h[:n - kk])
            g[:n - kk] -= h[kk:]
            grad += w * g
    return grad * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h * (target / s), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def nesterov_opt(h, n, T, max_steps, t_end):
    h_curr = h.copy()
    h_prev = h.copy()
    t_k = 1.0
    base_lr = 0.04
    for step in range(max_steps):
        if time.time() > t_end:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h_ex = h_curr + (t_k - 1.0) / t_new * (h_curr - h_prev)
        h_ex = np.clip(h_ex, 0, 1)
        g = grad_smooth_max(h_ex, n, T)
        lr = base_lr / np.sqrt(step + 1)
        h_next = h_ex - lr * g
        h_next = project_h(h_next, n)
        h_prev = h_curr
        h_curr = h_next
        t_k = t_new
    return h_curr

def coord_opt_single(h, i, n):
    h_base = h.copy()
    old = h_base[i]
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        if ip < n:
            a[center + k] += (1.0 - h[ip])
            a[center - k] -= h[ip]
        if im >= 0:
            a[center + k] -= h[im]
            a[center - k] += (1.0 - h[im])
    ct = corr_base[center]
    lo, hi = 0.0, 1.0
    for _ in range(25):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = corr_base + m1 * a
        v1[center] = ct + m1 - m1 * m1
        v2 = corr_base + m2 * a
        v2[center] = ct + m2 - m2 * m2
        if v1.max() > v2.max():
            lo = m1
        else:
            hi = m2
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            x_new = coord_opt_single(best_h, i, n)
            h_try = best_h.copy()
            h_try[i] = x_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def block_peak_shave(h, n, t_end, iters=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(iters):
        if time.time() > t_end:
            break
        corr, dx = get_corr(best_h)
        vals = corr * dx
        peak = vals.max()
        peak_idx = np.argwhere(vals >= peak - 0.001 * n)[0]
        if len(peak_idx) == 0:
            peak_idx = [np.argmax(vals)]
        for pi in peak_idx[:3]:
            k = pi - (n - 1)
            if k == 0:
                targets = np.argsort(best_h)[:n // 4]
            elif k > 0:
                overlaps = best_h[:n-k] * (1.0 - best_h[k:])
                targets = np.argsort(overlaps)[::-1][:n // 3]
            else:
                kk = -k
                overlaps = best_h[kk:] * (1.0 - best_h[:n-kk])
                targets = np.argsort(overlaps)[::-1][:n // 3]
                targets = targets + kk
            for ti in targets[:10]:
                x_new = coord_opt_single(best_h, ti, n)
                h_try = best_h.copy()
                h_try[ti] = x_new
                h_try = project_h(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        x = np.linspace(0, 2, n_init, endpoint=False)
        best_h = np.clip(0.5 + 0.12 * np.cos(np.pi * x) + 0.08 * np.sin(2 * np.pi * x), 0, 1)
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    # Phase 1: Nesterov accelerated gradient (35%)
    p1_end = t0 + budget_s * 0.35
    for T in [80, 200, 500, 1500, 4000]:
        if time.time() > p1_end:

        steps = max(300, int((p1_end - time.time()) / max(0.001, (n * n * 20) / (budget_s * 0.35))))
        h_new = nesterov_opt(best_h, n, T, steps=min(steps, 3000), t_end=p1_end)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    # Phase 2: Peak shaving (15%)
    p2_end = t0 + budget_s * 0.50
    best_h, best_c5 = block_peak_shave(best_h, n, p2_end, iters=300)
    # Phase 3: Multi-restart with Nesterov (20%)
    p3_end = t0 + budget_s * 0.70
    for r in range(6):
        if time.time() > p3_end:
            break
        sigma = 0.04 * (1.0 - r * 0.12)
        h_pert = best_h + sigma * np.random.randn(n)
        h_pert = project_h(h_pert, n)
        for T in [300, 1000]:
            if time.time() > p3_end:
                break
            h_new = nesterov_opt(h_pert, n, T, 800, t_end=p3_end)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_pert
                h_pert = h_new
    # Phase 4: Coordinate descent + peak shave (20%)
    p4_end = t0 + budget_s * 0.90
    best_h, best_c5 = coord_descent(best_h, n, min(p4_end, t0 + (p4_end - time.time()) * 0.5), max_sweeps=400)
    best_h, best_c5 = block_peak_shave(best_h, n, p4_end, iters=200)
    # Phase 5: Try other resolutions (10%)
    p5_end = t_end - 10

        if time.time() > p5_end:

        if n_try == n:

        h_interp = project_h(h_interp, n_try)
        h_new = nesterov_opt(h_interp, n_try, 500, 500, t_end=p5_end)

            best_h = h_new

    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
