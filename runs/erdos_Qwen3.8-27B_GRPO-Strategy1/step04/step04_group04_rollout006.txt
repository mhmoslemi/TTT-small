    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += pk * (1.0 - h[:n - lag])
                grad[:n - lag] += -pk * h[lag:]
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += -pk * h[nl:]
    return grad * dx

    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        scale = target / s
        h = h * scale
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
