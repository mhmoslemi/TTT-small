def subgradient_step(h, n, n_active_frac=0.01, lr=0.001):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Find active set
    threshold = m - (m * n_active_frac + 1e-10)
    active = np.where(corr >= threshold)[0]
    # Compute subgradient
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        w = 1.0 / len(active)
        if lag >= 0:
            if n - lag > 0:
                grad[:n-lag] += w * (1.0 - h[lag:n])
                grad[lag:] += w * (-h[:n-lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += w * (1.0 - h[:n-nl])
                grad[:n-nl] += w * (-h[nl:])
    return grad * dx, m * dx

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h, n, t_end, max_rounds=200, n_tries=5000):

    cnt = int(np.round(np.sum(best_h)))
    target = n // 2
    if cnt > target:
        ones = np.where(best_h == 1)[0]
        fl = np.random.choice(ones, cnt - target, replace=False)
        best_h[fl] = 0
    elif cnt < target:
        zeros = np.where(best_h == 0)[0]
        fl = np.random.choice(zeros, target - cnt, replace=False)
        best_h[fl] = 1
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False

        zeros = np.where(best_h == 0)[0]

            if time.time() > t_end:
                return best_h, best_c5
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            best_h[i], best_h[j] = best_h[j], best_h[i]
            c5t = compute_c5(best_h)

    return best_h, best_c5

def subgradient_descent(h, n, t_end, lr_init=0.005, n_iter=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    lr = lr_init
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        m = np.max(corr)
        frac = 0.005
        thresh = m - (m * frac + 0.5)
        active = np.where(corr >= thresh)[0]
        w = 1.0 / len(active)
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            if lag >= 0:
                if n - lag > 0:
                    grad[:n-lag] += w * (1.0 - best_h[lag:n])
                    grad[lag:] += w * (-best_h[:n-lag])
            else:
                nl = -lag
                if n - nl > 0:
                    grad[nl:] += w * (1.0 - best_h[:n-nl])
                    grad[:n-nl] += w * (-best_h[nl:])
        step = lr
        for _ in range(20):
            trial = proj(best_h - step * grad, n)
            c5t = compute_c5(trial)
            if c5t < best_c5:
                best_h = trial
                best_c5 = c5t
                lr = min(lr * 1.02, 0.05)
                break
            step *= 0.5
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    return best_h, best_c5

def coordinate_refine(h, n, t_end, max_sweeps=50):

        if not improved:

def swap_anneal(h, n, t_end, t0=0.02, t1=0.001, max_steps=50000):
    best_h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    cnt = int(np.round(np.sum(best_h)))
    target = n // 2
    if cnt > target:
        ones = np.where(best_h == 1)[0]
        best_h[np.random.choice(ones, cnt - target, replace=False)] = 0
    elif cnt < target:
        zeros = np.where(best_h == 0)[0]
        best_h[np.random.choice(zeros, target - cnt, replace=False)] = 1
    cur_c5 = compute_c5(best_h)
    best_h2 = best_h.copy()
    best_c5 = cur_c5
    for it in range(max_steps):
        if time.time() > t_end:
            break
        frac = it / float(max_steps)
        T = t0 * (t1 / t0) ** frac
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        best_h[i], best_h[j] = best_h[j], best_h[i]
        c5t = compute_c5(best_h)
        d = c5t - cur_c5
        if d < 0 or np.random.rand() < np.exp(-d / max(T, 1e-12)):
            cur_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best_h2 = best_h.copy()
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]
    return best_h2, best_c5

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h

def golden_ratio_seq(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (np.floor(i * 0.6180339887) - np.floor((i-1) * 0.6180339887)) > 0.5 else 0.0
    return h
