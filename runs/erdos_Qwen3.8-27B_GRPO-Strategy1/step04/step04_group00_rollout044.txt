import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + room * (diff / tot), 0, 1)
        else:
            if s < 1e-14:
                break
            h = np.clip(h * (1.0 + diff / s), 0, 1)
    return h

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_fast(corr_base, h, i, n):
    shift = np.arange(-(n - 1), n)
    j1 = i - shift
    j2 = i + shift
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coef = np.zeros(2 * n - 1)
    coef[v1] += 1.0 - h[np.clip(j1[v1], 0, n - 1)]
    coef[v2] += -h[np.clip(j2[v2], 0, n - 1)]
    dx = 2.0 / n
    def f1d(t):
        return np.max(corr_base + t * coef) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return res.x

def coord_descent(h, t_end, max_sweeps=200, order=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    for sweep in range(max_sweeps):

            break
        improved = False
        if order is None:

        for i in order:
            if time.time() > t_end:

        best_h, best_c5 = coord_descent(best_h, t3_end, max_sweeps=80)
    
    # Final SLSQP polish with very high T
    if time.time() < t0 + budget_s * 0.97:
        for T in [100000, 500000]:
            if time.time() > t0 + budget_s * 0.97:
                break
            hc = slsqp_opt(best_h, T, best_n, maxiter=1500, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    
    # Final quick coord descent
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, t0 + budget_s * 0.99, max_sweeps=20)
    
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    
    # Ensure sum constraint is tight
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
