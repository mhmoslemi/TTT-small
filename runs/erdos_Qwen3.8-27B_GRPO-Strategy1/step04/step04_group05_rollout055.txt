import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_binary(h, N):
    h_b = (h > 0.5).astype(float)
    nh = N // 2
    n_ones = int(h_b.sum())
    if n_ones > nh:
        idx = np.where(h_b == 1)[0]
        h_b[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:nh - n_ones]] = 1
    return h_b

def compute_delta(h, N):
    center = N - 1
    p_idx = np.arange(N)[:, None]
    L_idx = np.arange(-(N - 1), N)[None, :]
    pL = p_idx - L_idx
    valid1 = ((pL >= 0) & (pL < N)).astype(float)
    pLc = np.clip(pL, 0, N - 1)
    pLp = p_idx + L_idx
    valid2 = ((pLp >= 0) & (pLp < N)).astype(float)
    pLpc = np.clip(pLp, 0, N - 1)
    return -(1.0 - h[pLc]) * valid1 + h[pLpc] * valid2

def steepest_desc(h, N, max_sweeps, deadline, rng):
    h = h.astype(float).copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        P = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        D = compute_delta(h, N)
        best_peak = P
        best_pq = None
        for p in ones:
            a = c + D[p, :]
            diffs = a[None, :] - D[zeros, :]
            peaks = diffs.max(axis=1)
            mi = np.argmin(peaks)
            if peaks[mi] < best_peak:
                best_peak = float(peaks[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def anneal_binary(h, N, n_iters, rng, T0=2.0, T1=0.001, deadline=None):
    h_b = smooth_binary(h, N)
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur_pk = best_pk
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = h_b[j], h_b[i]
        new_pk = float(np.max(compute_corr(h_b)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.003 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
    return best, best_pk * (2.0 / N)

def corr_gradient(h, w, N):
    N2 = 2 * N - 1
    center = N - 1
    grad = np.zeros(N)
    p_idx = np.arange(N)[:, None]
    for m in range(N2):
        wm = w[m]
        if wm < 1e-12:
            continue
        L = m - center
        if L == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N - L] += wm * (1.0 - h[L:])
            grad[L:] -= wm * h[:N - L]
        else:
            M = -L
            grad[M:] += wm * (1.0 - h[:N - M])
            grad[:N - M] -= wm * h[M:]
    return grad

def continuous_adam(h, N, deadline, beta_init=8.0, beta_max=150.0, lr0=0.03, decay=0.995):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    beta = beta_init
    lr = lr0
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        beta = min(beta * 1.015, beta_max)
        exps = np.exp(beta * (corr - corr.max()))
        s_exp = exps.sum()
        if s_exp < 1e-300:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w = exps / s_exp
        grad = corr_gradient(h, w, N)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        t += 1
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if t % 80 == 0:

            hb = smooth_binary(h, N)
            pb = float(np.max(compute_corr(hb)))
            if pb < best_peak:
                best_peak = pb
                best_h = hb.copy()
    return best_h, best_peak * (2.0 / N)

            h[i] = 1.0 if d < N // 4 else 0.0
    else:

    return smooth_binary(h, N)
