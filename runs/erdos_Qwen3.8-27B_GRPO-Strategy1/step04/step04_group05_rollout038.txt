```python
def continuous_grad(h, N, w):
    """Compute sum_k w[k] * dc[k]/dh for continuous h."""
    n2 = 2*N-1
    idx = np.arange(n2)
    L = idx - (N-1)
    j = np.arange(N)[:,None]
    l = L[None,:]
    jp = j + l
    jm = j - l
    vjp = (jp>=0)&(jp<N)
    vjm = (jm>=0)&(jm<N)
    hjp = np.where(vjp, h[np.clip(jp,0,N-1)], 0.0)
    hjm = np.where(vjm, h[np.clip(jm,0,N-1)], 0.0)
    # For L≠0: (1-h[j+L])·1[j+L valid] - h[j-L]·1[j-L valid]
    # For L=0: 1-2h[j]
    grad = (1.0-hjp)*vjp.astype(float) - hjm*vjm.astype(float)
    # Fix L=0 column
    grad[:, N-1] = (1.0 - 2.0*h)
    return grad @ w  # sum over k (columns)
```
