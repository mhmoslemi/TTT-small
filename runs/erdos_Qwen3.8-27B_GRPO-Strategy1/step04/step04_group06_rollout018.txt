from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    h = np.asarray(h, dtype=float)
    target = n / 2.0
    # Bisection on lambda: h_proj[i] = clip(h[i] - lambda, 0, 1)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = h_p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Small correction for numerical error
    diff = target - h_p.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-boundary elements
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

    """Compute correlations and softmax weights."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    if T < 0.5:
        T = 0.5
    exp_vals = np.exp(T * (corr - m))
    w = exp_vals / exp_vals.sum()
    return corr, w

    T = max(T, 1.0)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    corr = get_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = corr.max()
    T = max(T, 1.0)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    c = n - 1
    grad += w[c] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[c + k]
        if wk > 1e-14:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        wm = w[c - k]
        if wm > 1e-14:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n-k]
    return grad * dx

def slsqp_smooth(h0, T, n, maxiter=1500):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj(x, T)
    def obj_jac(x):
        return smooth_obj(x, T), smooth_grad(x, T)
    def con_eq(x):
        return x.sum() - n * 0.5
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    res = minimize(obj_jac, h0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_feasible(res.x, n)

def coord_opt(h, n, max_sweeps=100, t_end=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    center = n - 1
    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if t_end is not None and time.time() > t_end:
                return best_h, best_c5
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = get_corr(h_base)
            delta = np.zeros(2 * n - 1)
            for k in range(1, n):
                ip = i + k
                im = i - k
                val_ip = h_base[ip] if 0 <= ip < n else 0.0
                val_im = h_base[im] if 0 <= im < n else 0.0
                delta[center + k] = (1.0 - val_im) - val_ip
                delta[center - k] = (1.0 - val_ip) - val_im
            delta[center] = 1.0
            c0 = corr_base[center]
            def f1d(x):
                vals = corr_base + x * delta
                vals[center] = c0 + x - x * x
                return np.max(vals) * dx
            try:
                res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-10})
                best_h[i] = res.x
            except Exception:
                pass
            best_h = project_feasible(best_h, n)
            c5_new = compute_c5(best_h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
        if not improved:
            break
    return best_h, best_c5

def lgbfp_opt(h0, T, n, t_end, max_iter=20000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_f = smooth_obj(h, T)
    lr = 1.0 / (T + 10.0)
    mom = 0.0
    v = np.zeros(n)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        f = smooth_obj(h, T)
        g = smooth_grad(h, T)
        if f < best_f:
            best_f = f
            best_c5 = compute_c5(h)
            best_h = h.copy()
        if it % 200 == 199:
            lr = max(1e-5, lr * 0.98)
        v = 0.85 * v + lr * g
        h_new = h - v
        h_new = project_feasible(h_new, n)
        f_new = smooth_obj(h_new, T)
        if f_new > f + 1e-6:
            lr *= 0.8
            h = project_feasible(h - lr * g, n)
        else:
            h = h_new
    return best_h, compute_c5(best_h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x), 0, 1)
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with ramping temperature
    for T in [10, 30, 100, 300, 1000]:
        if time.time() > t_end - budget_s * 0.45:
            break
        try:
            h_new = slsqp_smooth(best_h, T, n, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Coordinate descent on true max
    t_ph2 = min(t_end - budget_s * 0.35, t0 + budget_s * 0.30)
    best_h, best_c5 = coord_opt(best_h, n, max_sweeps=300, t_end=t_ph2)

    # Phase 3: Multi-restart with perturbations
    n_restarts = 6
    t_ph3 = t_end - budget_s * 0.30
    for r in range(n_restarts):
        if time.time() > t_ph3:
            break
        scale = 0.04 * (1.0 + 0.2 * r)
        h_pert = best_h + scale * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        for T in [50, 200, 800]:
            if time.time() > t_ph3:
                break
            try:
                h_new = slsqp_smooth(h_pert, T, n, maxiter=800)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
        h_new, c5_new = coord_opt(best_h, n, max_sweeps=40, t_end=t_ph3)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 4: Try L-BFGS-style at higher T for polish
    if time.time() < t_end - budget_s * 0.20:
        h_new, c5_new = lgbfp_opt(best_h, 2000, n, t_end - budget_s * 0.20, max_iter=5000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 5: Upscale to higher resolution
    for n_try in [240, 320]:
        if time.time() > t_end - budget_s * 0.10:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_feasible(h_up, n_try)
        for T in [200, 800]:
            if time.time() > t_end - budget_s * 0.08:
                break
            try:
                h_new = slsqp_smooth(h_up, T, n_try, maxiter=600)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    n = n_try
            except Exception:
                pass
        h_new, c5_new = coord_opt(best_h, n, max_sweeps=60, t_end=t_end - budget_s * 0.08)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 6: Final SLSQP polish
    for T in [3000, 10000]:
        if time.time() > t_end - 15:
            break
        try:
            h_new = slsqp_smooth(best_h, T, n, maxiter=1500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Phase 7: Final coordinate descent
    best_h, best_c5 = coord_opt(best_h, n, max_sweeps=200, t_end=t_end - 2)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
