import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Safe softmax
    shifted = T * (corr - m)
    max_s = np.max(shifted)
    val = m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / T
    return val * dx

def project(h, n):

        return h
    if diff > 0:

        tot = room.sum()
        if tot < 1e-14:
            return h
        frac = diff / tot
        h2 = h + room * frac
        if h2.max() <= 1.0 + 1e-14:
            return np.clip(h2, 0, 1)
    # fallback: iterative
    h = np.clip(h, 0, 1)

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    ms = np.max(shifted)
    val = m + (ms + np.log(np.sum(np.exp(shifted - ms)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h2 = h + room * (diff / tot)
        if h2.max() <= 1.0 + 1e-12:
            return np.clip(h2, 0, 1)
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] = np.maximum(0.0, h[mask] - (-diff) / mask.sum())
    return np.clip(h, 0, 1)

def build_lin_coeffs(h, n, i):
    lin = np.zeros(2 * n - 1)
    for k in range(1, n):
        cp = 0.0
        cn = 0.0
        if i + k < n:
            cp += (1.0 - h[i + k])
            cn -= h[i + k]
        if i - k >= 0:
            cp -= h[i - k]
            cn += (1.0 - h[i - k])
        lin[n - 1 + k] = cp
        lin[n - 1 - k] = cn
    return lin

def coord_opt_fast(h, n, i, corr_curr):
    dx = 2.0 / n
    old = h[i]
    lin = build_lin_coeffs(h, n, i)
    if np.all(np.abs(lin) < 1e-15):
        return old
    def f1d(t):
        d = t - old
        if abs(d) < 1e-15:
            return np.max(corr_curr) * dx
        nc = corr_curr + d * lin
        nc[n - 1] += d * (1.0 - t - old)
        return np.max(nc) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13, 'maxiter': 200})
    t = res.x
    return t

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=50, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n)
    dx = 2.0 / n
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fast(best_h, n, i, corr)
            if abs(hi - best_h[i]) > 1e-10:
                best_h[i] = hi
                best_h = project(best_h, n)
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                break
    return best_h, best_c5

def pair_descent(h, n, t_end, rng, n_pairs=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        for eps in [0.01, 0.02, 0.005]:
            di = rng.choice([-1, 1]) * eps
            dj = -di * n / (n - 1)  # roughly preserve sum
            test = best_h.copy()
            test[i] = np.clip(test[i] + di, 0, 1)
            test[j] = np.clip(test[j] + dj, 0, 1)
            test = project(test, n)
            cv = compute_c5(test)
            if cv < best_c5 - 1e-14:
                best_h, best_c5 = test, cv
                break
    return best_h, best_c5

def basin_hop(h, n, t_end, rng, n_iters=150, base_step=0.035):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(n_iters, 1))
        step = base_step * (0.2 + 0.8 * cool)
        pert = work + step * rng.standard_normal(n)
        pert = project(pert, n)
        T_use = int(300 + 500 * cool)
        try:
            hc = slsqp_opt(pert, T_use, n, maxiter=200, ftol=1e-13)
        except Exception:
            hc = pert
        cv = compute_c5(hc)
        temp = 0.003 * cool + 0.0002
        if cv < work_c5 or (cv - work_c5 < 0.004 and rng.random() < np.exp(-(cv - work_c5) / temp)):
            work, work_c5 = hc, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def comb_push(h, n, alpha=0.05):
    """Push values away from 0.5 to encourage binary structure."""
    h2 = h.copy()
    h2 = h2 + alpha * (2.0 * h2 - 1.0) * (1.0 - np.abs(2.0 * h2 - 1.0))
    return project(np.clip(h2, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 240
        best_h = np.zeros(n0)
        best_h[:n0 // 2] = 1.0

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish (increasing T)
    for T in [200, 1000, 5000, 20000]:
        if elapsed() > budget_s * 0.10:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=800, ftol=1e-15)
            cv = compute_c5(hc)
            if cv < best_c5 - 1e-15:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.28, max_sweeps=50, rng=rng)

    # Phase 3: Pair descent
    if elapsed() < budget_s * 0.38:
        best_h, best_c5 = pair_descent(best_h, n, t0 + budget_s * 0.38, rng, n_pairs=4000)

    # Phase 4: Basin hopping
    if elapsed() < budget_s * 0.52:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.52, rng,
                                     n_iters=180, base_step=0.03)

    # Phase 5: CD + comb push to escape plateaus
    if elapsed() < budget_s * 0.62:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.58, max_sweeps=40, rng=rng)
        h_push = comb_push(best_h, n, 0.03)
        cv = compute_c5(h_push)
        if cv < best_c5:

        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.62, max_sweeps=25, rng=rng)
