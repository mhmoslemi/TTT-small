        r = 1.0 - h; t = r.sum()

def block_opt(h, n, lo, hi, t_end):
    """Optimize h[lo:hi] while fixing all other coordinates."""
    idx = np.arange(lo, hi)
    ma = len(idx)
    v = h[idx].copy()
    
    def obj(x):
        test = h.copy()
        test[idx] = np.clip(x, 0, 1)
        return compute_c5(test)
    
    # Use L-BFGS-B or Nelder-Mead
    bnds = [(0, 1)] * ma
    try:
        r = minimize(obj, v, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': 1000, 'gtol': 1e-12})
        ht = h.copy()
        ht[idx] = np.clip(r.x, 0, 1)
        ht = project(ht, n)
        return ht
    except:
        return h

def block_sweep(h, n, n_blocks, t_end, rng):
    bh = h.copy()
    bc = compute_c5(bh)
    bs = n // n_blocks
    for _ in range(20):  # outer iterations
        if time.time() > t_end: break

            lo = b * bs

            ht = block_opt(bh, n, lo, hi, t_end)
            c5 = compute_c5(ht)
            if c5 < bc - 1e-14:

The objective is max_k f_k(h) where f_k(h) = Σ_j h[j](1-h[j+k]).

def project(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-15:
        return h
    if d > 0:
        r = 1.0 - h; t = r.sum()
        if t > 1e-15: h = h + r * (d / t)
    else:
        h = np.clip(h * (1.0 - d / max(s, 1e-15)), 0, 1)
    h = np.clip(h, 0, 1)
    s2 = h.sum(); d2 = tgt - s2

    dx = 2.0 / n

        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
