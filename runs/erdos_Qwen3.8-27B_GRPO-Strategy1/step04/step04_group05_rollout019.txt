```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    """Project h onto {0≤h≤1, sum(h)=target} using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_objective(h, N, alpha):
    corr = compute_corr(h)
    shifted = alpha * corr - alpha * np.max(corr)  # numerical stability
    logsumexp = np.log(np.sum(np.exp(shifted))) + np.max(corr)
    return (1.0/alpha) * logsumexp * (2.0/N)

def smooth_gradient(h, N, alpha):
    corr = compute_corr(h)
    # Softmax weights
    shifted = alpha * corr - alpha * np.max(corr)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient of corr w.r.t. h
    # For each lag L, gradient is a vector of length N
    # We compute the weighted sum
    grad = np.zeros(N)
    center = N - 1
    for k in range(2*N-1):
        L = k - center
        w = weights[k]
        if w < 1e-10:
            continue
        if L == 0:
            grad += w * (1.0 - 2.0*h)
        elif L > 0:
            # h[p] as h[i+L] when i=p-L: contribution (1-h[p-L]) for L≤p≤N-1
            grad[L:] += w * (1.0 - h[:N-L])
            # h[p] as (1-h[i]) when i=p: contribution -h[p+L] for p+L≤N-1
            grad[:N-L] -= w * h[L:]
        else:
            M = -L
            # h[p] as h[i-M] when i=p+M: contribution (1-h[p+M]) for p+M≤N-1
            grad[:N-M] += w * (1.0 - h[M:])
            # h[p] as (1-h[i]) when i=p: contribution -h[p-M] for p≥M
            grad[M:] -= w * h[:N-M]
    return grad * (2.0/N)
```

        # Line search (backtracking)
        new_obj = smooth_objective(h_new, N, alpha)
        if new_obj > obj:

                h_try = project_sum(h_try, target)
                obj_try = smooth_objective(h_try, N, alpha)
                if obj_try < obj:
                    h_new = h_try
                    break
            else:
                h_new = h  # accept no change
        
        h = h_new
        c5 = compute_c5(h, N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
