def smooth_max_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

    def obj(x):
        return smooth_max_obj(x, T, n)
    def con(x):
        return x.sum() - n / 2.0

                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def slsqp(h, T, n, maxiter=600):
    def obj(x):
        return smooth_obj(x, T, n)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def active_coord_descent(h, n, t_end, max_sweeps=800, n_active_frac=0.15):
    dx = 2.0 / n
    best = h.copy()
    corr = np.correlate(best, 1.0 - best, mode='full').astype(np.float64)
    best_c5 = float(np.max(corr)) * dx
    offsets = np.arange(2 * n - 1) - (n - 1)
    n_active = max(3, int(len(offsets) * n_active_frac))
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        top_thresh = np.partition(corr, -n_active)[-n_active]
        active_lags = np.where(corr >= top_thresh - 1e-12)[0]
        
        coord_set = set(int(k) for k in active_lags)
        for k in active_lags:
            off = offsets[k]
            for i in range(n):
                if 0 <= i + off < n or 0 <= i - off < n:
                    coord_set.add(i)
        coords = np.array(sorted(coord_set), dtype=int)
        
        for i in coords[np.random.permutation(len(coords))]:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            if old < 1e-14 or old > 1.0 - 1e-14:
                continue
            # Compute coefficient vector (vectorized)
            j1 = i + offsets
            j2 = i - offsets
            valid1 = (j1 >= 0) & (j1 < n)
            valid2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(2 * n - 1)
            coef[valid1] += (1.0 - best[j1[valid1]])
            coef[valid2] -= best[j2[valid2]]
            
            # 1D minimax: min over delta of max(corr + delta*coef)
            # f(delta) is convex piecewise-linear; min at breakpoints or boundaries
            lo, hi = -old, 1.0 - old
            # Evaluate at boundaries and intersections of top lags
            candidates = np.array([0.0, lo, hi])
            # Intersections of top active lags
            top_idx = np.argsort(corr)[::-1][:min(8, len(corr))]
            for a in range(len(top_idx)):
                for b in range(a + 1, len(top_idx)):
                    ia, ib = top_idx[a], top_idx[b]
                    dc = coef[ia] - coef[ib]
                    if abs(dc) > 1e-14:
                        d = (corr[ib] - corr[ia]) / dc
                        if lo <= d <= hi:
                            candidates = np.append(candidates, d)
            # Add a few uniform points
            candidates = np.append(candidates, np.linspace(lo, hi, 7))
            
            best_d = 0.0
            best_mv = float(np.max(corr))
            for d in candidates:
                d = np.clip(d, lo, hi)
                mv = float(np.max(corr + d * coef))
                if mv < best_mv:
                    best_mv = mv
                    best_d = d
            
            if abs(best_d) < 1e-13:
                continue
            new_val = old + best_d
            best[i] = new_val
            corr += best_d * coef
            
            # Fix sum constraint
            s = best.sum()
            if abs(s - n / 2) > 1e-10:
                diff = n / 2 - s
                if diff > 0:
                    room = 1.0 - best
                    tot = room.sum()
                    if tot > 1e-15:
                        best += room * (diff / tot)
                else:
                    best *= (1.0 + diff / s) if s > 1e-15 else 1.0
                best = np.clip(best, 0, 1)
                corr = np.correlate(best, 1.0 - best, mode='full').astype(np.float64)
            
            c5n = float(np.max(corr)) * dx
            if c5n < best_c5 - 5e-15:
                best_c5 = c5n
                improved = True
        
        if not improved and sweep > 80:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP warm-up
    for T in [300, 2000, 10000, 50000]:
        if time.time() - t0 > budget_s * 0.08:
            break
        try:
            hc = slsqp(best_h, T, n, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass
    
    # Phase 2: Main coordinate descent
    if time.time() - t0 < budget_s * 0.40:
        best_h, best_c5 = active_coord_descent(best_h, n, t0 + budget_s * 0.40, max_sweeps=600)
    
    # Phase 3: Basin hopping
    work_c5 = best_c5
    n_bh = 25
    for it in range(n_bh):
        if time.time() - t0 > budget_s * 0.62:
            break
        cool = 1.0 - 0.88 * (it / n_bh)
        h_p = best_h.copy()
        ptype = it % 3
        if ptype == 0:
            i, j = np.random.choice(n, 2, replace=False)
            h_p[i], h_p[j] = h_p[j], h_p[i]
        elif ptype == 1:
            h_p += (0.03 + 0.08 * cool) * np.random.randn(n)
        else:
            blen = np.random.randint(3, min(12, n // 4))
            start = np.random.randint(0, n - blen)
            h_p[start:start + blen] = 1.0 - h_p[start:start + blen]
        h_p = proj(h_p, n)
        t_local = min(time.time() + 4.0, t0 + budget_s * 0.62)
        hc = active_coord_descent(h_p, n, t_local, max_sweeps=40, n_active_frac=0.2)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_c5 = c5v
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
    
    # Phase 4: Second active coord descent
    if time.time() - t0 < budget_s * 0.82:
        best_h, best_c5 = active_coord_descent(best_h, n, t0 + budget_s * 0.82, max_sweeps=400, n_active_frac=0.08)
    
    # Phase 5: High-T SLSQP polish
    for T in [200000, 500000]:
        if time.time() - t0 > budget_s * 0.93:
            break
        try:
            hc = slsqp(best_h, T, n, maxiter=700)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass
    
    # Phase 6: Tight final coord descent
    if time.time() - t0 < budget_s * 0.99:
        best_h, best_c5 = active_coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=300, n_active_frac=0.05)
    
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
