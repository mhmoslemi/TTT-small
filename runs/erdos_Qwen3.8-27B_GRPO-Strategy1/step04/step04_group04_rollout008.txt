def proj(h, n):
    """Project h onto {0≤h≤1, sum=h=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def comp_grad(h, T):

    n = len(h)

    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -400, 400)

def compute_c5(h):

def smooth_c5(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -400, 400)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-12:
        return h

        r = 1.0 - h
        if r.sum() < 1e-12:
            return h
        h += r * (d / r.sum())
    else:
        h *= (1.0 + d / s) if s > 1e-12 else 0
