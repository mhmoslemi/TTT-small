import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_cd_state = {}
_lbfgs_state = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        h *= (target / s)
        h = np.clip(h, 0, 1)
        s = h.sum()
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-13:
            h += room * (target - s) / tot
    return np.clip(h, 0, 1)

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    s_exp = exps.sum()
    val = (m + np.log(s_exp) / T) * dx
    w = exps / s_exp
    idx = np.arange(n)
    k = np.arange(2 * n - 1) - (n - 1)
    gi = idx[:, None] + k[None, :]
    hi = idx[:, None] - k[None, :]
    vg = (gi >= 0) & (gi < n)
    vh = (hi >= 0) & (hi < n)
    gv = np.zeros((n, 2 * n - 1))
    hv = np.zeros((n, 2 * n - 1))
    gv[vg] = g[gi[vg]]
    hv[vh] = h[hi[vh]]
    grad = dx * ((gv - hv) * w[None, :]).sum(axis=1)
    return val, grad

def _lbfgs_obj(x):
    x = np.clip(x, 0.0, 1.0)
    v, _ = smooth_c5_grad(x, _lbfgs_state['T'])
    n = _lbfgs_state['n']
    lam = _lbfgs_state['lam']
    return v + lam * (x.sum() - n / 2.0) ** 2

def _lbfgs_jac(x):
    x = np.clip(x, 0.0, 1.0)
    _, gr = smooth_c5_grad(x, _lbfgs_state['T'])
    n = _lbfgs_state['n']
    lam = _lbfgs_state['lam']
    return gr + 2.0 * lam * (x.sum() - n / 2.0)

def lbfgs_opt(h_in, T, n, maxiter=400, lam=1000.0):
    _lbfgs_state['T'] = T
    _lbfgs_state['n'] = n
    _lbfgs_state['lam'] = lam
    res = minimize(_lbfgs_obj, h_in.copy(), jac=_lbfgs_jac,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

def _cd_f1d(t):
    v = _cd_state['cb'] + t * _cd_state['d']
    v[_cd_state['n'] - 1] = _cd_state['c0'] + t * (1.0 - t)
    return np.max(v) * _cd_state['dx']

def coord_descent(h_in, n, t_end, max_sweeps=80, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h_in.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    _cd_state['dx'] = dx
    _cd_state['n'] = n
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
            d = np.zeros(2 * n - 1)
            for k in range(1, n):
                dk = 0.0
                if 0 <= i - k < n:
                    dk += 1.0 - best_h[i - k]
                if 0 <= i + k < n:
                    dk -= best_h[i + k]
                d[n - 1 + k] = dk
                d[n - 1 - k] = dk
            _cd_state['cb'] = corr_base
            _cd_state['d'] = d
            _cd_state['c0'] = corr_base[n - 1]
            r = minimize_scalar(_cd_f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-14})
            if abs(r.x - best_h[i]) > 1e-12:
                best_h[i] = r.x
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        stall = 0 if improved else stall + 1
        if stall >= 4:
            break
    return best_h, best_c5

def swap_descent(h_in, n, t_end, rng, n_trials=8000):
    best_h = h_in.copy()
    best_c5 = compute_c5(best_h)
    fracs = [1.0, 0.7, 0.4, 0.2]
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        for f in fracs:
            dh = f * (best_h[j] - best_h[i])
            test = best_h.copy()
            test[i] += dh
            test[j] -= dh
            np.clip(test, 0.0, 1.0, out=test)
            test = project(test, n)
            cv = compute_c5(test)
            if cv < best_c5 - 1e-15:
                best_h, best_c5 = test, cv
                break
    return best_h, best_c5

def interp_h(h_old, n_new):
    xo = np.linspace(0, 2, len(h_old), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(xn, xo, h_old), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def el():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS smooth optimization (low to high T)
    for T in [10, 50, 200, 1000, 4000]:
        if el() > budget_s * 0.15:
            break
        try:
            hc = lbfgs_opt(best_h, T, n, maxiter=500, lam=1000.0)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 2: Coordinate descent
    if el() < budget_s * 0.40:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.40, max_sweeps=100, rng=rng)

    # Phase 3: Swap descent
    if el() < budget_s * 0.50:
        best_h, best_c5 = swap_descent(best_h, n, t0 + budget_s * 0.50, rng=rng, n_trials=12000)

    # Phase 4: High-T L-BFGS
    for T in [15000, 60000]:
        if el() > budget_s * 0.60:
            break
        try:
            hc = lbfgs_opt(best_h, T, n, maxiter=400, lam=2000.0)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 5: Coordinate descent again
    if el() < budget_s * 0.72:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.72, max_sweeps=80, rng=rng)

    # Phase 6: Perturbation restarts
    for _ in range(6):
        if el() > budget_s * 0.84:
            break
        sigma = 0.015 + 0.01 * rng.random()
        pert = best_h + sigma * rng.standard_normal(n)
        pert = project(pert, n)
        try:
            hc = lbfgs_opt(pert, 300, n, maxiter=300, lam=1000.0)
            t_sub = min(t0 + el() + budget_s * 0.015, t0 + budget_s * 0.86)
            hc, cv = coord_descent(hc, n, t_sub, max_sweeps=20, rng=rng)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 7: Multi-resolution
    for n_try in [160, 200, 280, 320]:
        if el() > budget_s * 0.93:
            break
        h_i = interp_h(best_h, n_try)
        try:
            hc = lbfgs_opt(h_i, 400, n_try, maxiter=400, lam=1000.0)
            hc, cv = coord_descent(hc, n_try, t0 + budget_s * 0.94, max_sweeps=25, rng=rng)
            if cv < best_c5:
                best_h, best_c5, n = hc, cv, n_try
        except Exception:
            pass

    # Phase 8: Final polish
    if el() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=40, rng=rng)
    try:
        hc = lbfgs_opt(best_h, 300000, n, maxiter=300, lam=5000.0)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv
    except Exception:
        pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
