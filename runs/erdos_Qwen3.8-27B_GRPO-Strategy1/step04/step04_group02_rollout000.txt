- The breakpoint for pair (a,b) is at t = (corr[a]-corr[b])/(d[b]-d[a]) (where d[b]≠d[a])

1. Check t=0 and t=1
2. Find the "kink" points where the active constraint changes
3. The minimum of a convex piecewise linear function on [0,1] is either at a kink or at a boundary

# d[n-1+k] = (1-h[i+k]) * [0<=i+k<n] - h[i-k] * [0<=i-k<n]

mask_m = (im >= 0) & (im < n)
d[mask_p] += 1.0 - h[ip[mask_p]]

Actually wait, there's a subtlety. When both mask_p and mask_m are true for the same index, we add both contributions. Let me verify:

For k=1: ip = i+1, im = i-1.
d[n-1+1] = (1-h[i+1])*[valid] - h[i-1]*[valid]

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            h = h - h * (diff / max(s, 1e-14))
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, n, t_end, max_sweeps=500, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    best_c5 = np.max(corr) * dx
    stall = 0
    ks = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            hi_old = best_h[i]
            ip = i + ks
            im = i - ks
            d = np.zeros(2 * n - 1)
            mp = (ip >= 0) & (ip < n)
            mm = (im >= 0) & (im < n)
            d[mp] += 1.0 - best_h[ip[mp]]
            d[mm] -= best_h[im[mm]]
            if not np.any(np.abs(d) > 1e-15):
                continue
            base = corr
            off = hi_old
            def f1d(t):
                return np.max(base + (t - off) * d) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
            hi_new = res.x
            if abs(hi_new - hi_old) > 1e-13:
                delta = hi_new - hi_old
                corr += delta * d
                best_h[i] = hi_new
                c5 = np.max(corr) * dx
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    return best_h, np.max(corr) * dx

    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)
