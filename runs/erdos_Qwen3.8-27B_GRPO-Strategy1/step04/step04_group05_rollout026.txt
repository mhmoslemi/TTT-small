Let me also think about a completely different approach: **minimax subgradient**.

The objective f(h) = max_k corr_k(h). The subgradient at h is:
- If there's a unique argmax k*: ∂f = ∂corr_k*/∂h
- If multiple k achieve the max: ∂f = convex hull of {∂corr_k/∂h : k in argmax}

The subgradient descent would be:
h_{t+1} = project(h_t - lr * g_t)

where g_t is a subgradient. Using the "magic" subgradient (average of all active lags) tends to work well.

The key advantage of subgradient methods is they work in continuous space, so they can find solutions where h is not fully binary.

But the problem is that subgradient methods are slow for the max function (they bounce around the active set). A better approach is to use a smoothed version: f_ε(h) = ε * log(sum_k exp(corr_k/ε)), which has a smooth gradient.

The log-sum-exp approach:
f_ε(h) = ε * log(Σ_k exp(corr_k(h)/ε))

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binary_local_search(h, N, max_sweeps=500, deadline=None):
    h = h.astype(float).copy()
    L = 2 * N - 1
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h)
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        s = 2.0 * h - 1.0
        s_rev = s[::-1]
        best_nm = P
        best_p = best_q = -1
        for p in ones:
            dp = np.zeros(L)
            dp[p:p+N] = s_rev
            base = corr + dp
            pref = np.maximum.accumulate(base)
            suff = np.maximum.accumulate(base[::-1])[::-1]

                nm = max(out_val, in_val)
                if nm < best_nm:

                    best_p = p
                    best_q = q
        if best_p < 0:

def lse_gradient_step(h, N, eps, lr):
    corr = compute_corr(h)
    m = float(corr.max())
    w = np.exp((corr - m) / eps)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    # Only process lags with significant weight
    active = np.where(w > 1e-6)[0]
    for k in active:
        L = k - center
        if L == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            grad[L:] += w[k] * (1.0 - h[:N-L])
            grad[:N-L] -= w[k] * h[L:]
        elif L < 0:
            M = -L
            if M < N:
                grad[M:] += w[k] * (1.0 - h[:N-M])
                grad[:N-M] -= w[k] * h[M:]
    h_new = h - lr * grad
    return project_sum(h_new, N / 2.0), m

def simulate_anneal_binary(h, N, n_iters, rng, T0=3.0, T1=0.01, deadline=None):
    h = h.astype(float).copy()
    nh = N // 2
    n_ones = int(h.sum())
    idx1 = np.where(h == 1)[0]
    idx0 = np.where(h == 0)[0]
    if n_ones > nh:

        idx1 = np.where(h == 1)[0]
        idx0 = np.where(h == 0)[0]
    elif n_ones < nh:
        h[idx0[:nh-n_ones]] = 1
        idx1 = np.where(h == 1)[0]
        idx0 = np.where(h == 0)[0]
    best = h.copy()

    cur_pk = best_pk
    for it in range(n_iters):

            break
        T = T0 * (T1 / T0) ** (it / n_iters)

        j = idx0[rng.randint(len(idx0))]
        h[i], h[j] = h[j], h[i]
        new_pk = float(compute_corr(h).max())
