```python
def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def update_corr(corr, h, i, new_val, n):
    """Compute new corr after changing h[i] to new_val, given old corr."""
    old_val = h[i]
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return corr
    new_corr = corr.copy()
    # For each k, the change is:
    # delta * [ (1-h[j1]) if j1=i+k-(n-1) in [0,n-1] 
    #          - h[j2] if j2=i-k+(n-1) in [0,n-1] and j2!=i ]
    # Plus the -delta² term at k=n-1 (where j1=j2=i)
    for k in range(2*n - 1):
        j1 = i + k - (n - 1)
        j2 = i - k + (n - 1)
        contrib = 0.0
        if 0 <= j1 < n:
            contrib += (1.0 - h[j1])
        if 0 <= j2 < n:
            contrib -= h[j2]
        new_corr[k] += delta * contrib
        if k == n - 1:
            new_corr[k] -= delta * delta  # quadratic correction
    return new_corr
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    vec = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    if abs(vec.sum() - target) < 1e-13:
        return vec
    lo = -vec.max() - 0.5
    hi = 0.5 - vec.min()
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) / 2, 0.0, 1.0)

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_smooth(h, T, n, maxiter=300, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(res.x, n)

def coord_desc(h, n, t_end, seed=42, n_grid=40, perturb_every=8, perturb_amt=0.04):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    grid = np.linspace(0.0, 1.0, n_grid)
    no_improve = 0
    while time.time() < t_end:
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            best_v, best_c = old, best_c5
            for v in grid:
                if abs(v - old) < 1e-9:
                    continue
                hh = best_h.copy()
                hh[i] = v
                hh = project(hh, n)
                c = compute_c5(hh)
                if c < best_c:
                    best_c = c
                    best_v = v
            if best_v != old:
                best_h[i] = best_v
                best_h = project(best_h, n)
                best_c5 = best_c
                improved = True
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= perturb_every:
            idx = rng.randint(n)
            best_h[idx] += rng.uniform(-perturb_amt, perturb_amt)
            best_h = project(best_h, n)
            best_c5 = compute_c5(best_h)
            no_improve = 0
    return best_h, best_c5

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
```
