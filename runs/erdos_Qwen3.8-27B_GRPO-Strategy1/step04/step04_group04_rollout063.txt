        h_j = h[j]

        # corr[k] changes by: delta * (a_i[k] - a_j[k])

                hj = h[j]
                a_j = np.zeros(n_corr)
                vp = (j + lags >= 0) & (j + lags < n)
                vm = (j - lags >= 0) & (j - lags < n)
                a_j[vp] = 1.0 - h[j + lags[vp]]
                a_j[vm] -= h[j - lags[vm]]
                d = a_i - a_j
                lo = max(-hi, hj - 1.0)
                hi_b = min(1.0 - hi, hj)
                if lo >= hi_b - 1e-14:
                    continue
                a, b = lo, hi_b
                for _ in range(8):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    if np.max(corr + m1 * d) <= np.max(corr + m2 * d):
                        b = m2
                    else:
                        a = m1
                delta = (a + b) / 2.0
                if abs(delta) < 1e-14:
                    continue
                nm = float(np.max(corr + delta * d)) * dx
                if nm < best_val:
                    best_val, best_j = nm, j
            if best_j >= 0 and best_val < best_c5 - 1e-15:
                hj = h[best_j]
                a_j = np.zeros(n_corr)
                vp = (best_j + lags >= 0) & (best_j + lags < n)
                vm = (best_j - lags >= 0) & (best_j - lags < n)
                a_j[vp] = 1.0 - h[best_j + lags[vp]]
                a_j[vm] -= h[best_j - lags[vm]]
                d = a_i - a_j
                lo = max(-hi, hj - 1.0)
                hi_b = min(1.0 - hi, hj)
                a, b = lo, hi_b
                for _ in range(14):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    if np.max(corr + m1 * d) <= np.max(corr + m2 * d):
                        b = m2
                    else:
                        a = m1
                delta = (a + b) / 2.0
                h[i] = np.clip(hi + delta, 0, 1)
                h[best_j] = np.clip(hj - delta, 0, 1)
                corr += delta * d
                best_c5 = best_val
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed + 7)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 120
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = 0.5 + 0.3 * np.sin(np.pi * x) + 0.15 * np.sin(2 * np.pi * x)
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP refinement
    for T in [200, 1000, 5000]:
        if time.time() - t0 > budget_s * 0.04:
            break
        hc = slsqp_refine(best_h, n, T, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: Fast paired CD (main workhorse) - 50% of budget
    t2 = t0 + budget_s * 0.50
    best_h, best_c5 = paired_cd(best_h, n, t2, seed=seed, accept_rate=0.0)

    # Phase 3: Greedy sweeps
    t3 = t0 + budget_s * 0.65
    best_h, best_c5 = greedy_sweep(best_h, n, t3, rng)

    # Phase 4: More paired CD with small acceptance (annealing)
    t4 = t0 + budget_s * 0.80
    best_h, best_c5 = paired_cd(best_h, n, t4, seed=seed + 100, accept_rate=0.05)

    # Phase 5: Basin hopping with paired CD as local solver
    t5 = t0 + budget_s * 0.90
    work_h, work_c5 = best_h.copy(), best_c5
    for it in range(30):
        if time.time() > t5:
            break
        cool = 1.0 - it / 30.0
        step_sz = 0.03 + 0.08 * cool
        h_pert = work_h + step_sz * rng.randn(n)
        h_pert = project_h(h_pert, n)
        hc, c5v = paired_cd(h_pert, n, min(time.time() + 15, t5), seed=seed + 200 + it)
        temp = 0.002 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 6: Try different resolutions
    for n_try in [90, 120, 144]:
        if time.time() > t0 + budget_s * 0.95:
            break
        n_try = int(n_try // 2 * 2)
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_h(h_interp, n_try)
        hc = slsqp_refine(h_interp, n_try, 2000, maxiter=400)
        hc, c5v = paired_cd(hc, n_try, min(time.time() + 20, t0 + budget_s * 0.97), seed=seed + 500)
        if c5v < best_c5:
            best_c5, best_h, n = c5v, hc, n_try

    # Phase 7: Final polish
    best_h, best_c5 = paired_cd(best_h, n, t0 + budget_s * 0.99, seed=seed + 999, accept_rate=0.0)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
