def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    wsum = w.sum()
    val = (m + np.log(wsum) / T) * dx
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15 * wsum:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] -= wi * h[k:]
        else:
            kk = -k
            grad[:n - kk] += wi * (1.0 - h[kk:])
            grad[kk:] -= wi * h[:n - kk]
    grad *= dx / (T * wsum)
    return val, grad

def proj_grad_desc(h, T, target, n, t_end, max_iter=8000):
    h_cur = h.copy()
    val, grad = smooth_c5_grad(h_cur, T)
    momentum = np.zeros(n)
    lr = 0.02
    best_h = h_cur.copy()
    best_val = val
    for it in range(max_iter):
        if time.time() > t_end:
            break
        momentum = 0.85 * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 1e-10:
            h_try = project(h_cur - lr * momentum / norm, target)
            val_try, grad_try = smooth_c5_grad(h_try, T)
            if val_try < val * (1 - 1e-12):
                h_cur = h_try
                val = val_try
                grad = grad_try
                if val < best_val:
                    best_val = val
                    best_h = h_cur.copy()
            else:
                lr *= 0.6
                if lr < 1e-9:
                    break
        else:
            break
    return best_h, best_val

def coord_descent_fast(h, target, t_end, max_sweeps=20000):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            idx1 = i - ks
            idx2 = i + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            a = np.zeros(2 * n - 1)
            a[valid1] += 1.0 - best_h[idx1[valid1]]
            a[valid2] -= best_h[idx2[valid2]]
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if hi - lo < 1e-12:
                continue
            a_local = a
            c_local = corr
            def f_obj(d):
                return np.max(c_local + d * a_local)
            res = minimize_scalar(f_obj, bounds=(lo, hi), method='bounded', options={'xatol': 1e-11})
            if abs(res.x) > 1e-12:
                new_corr = corr + res.x * a
                new_max = np.max(new_corr)
                if new_max * dx < best_c5 - 1e-14:
                    best_h[i] += res.x
                    corr = new_corr
                    best_c5 = new_max * dx
                    improved = True
        best_h = project(best_h, target)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        c5_now = np.max(corr) * dx
        if c5_now < best_c5:
            best_c5 = c5_now
        else:
            best_c5 = c5_now
        if not improved:
            stall += 1
            if stall >= 4:
                break
        else:
            stall = 0
    return best_h, best_c5

def block_cd(h, target, t_end, block_size=3, max_sweeps=2000):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved_any = False
        num_blocks = n // block_size
        for b in range(num_blocks):
            idxs = np.arange(b * block_size, (b + 1) * block_size)
            best_block = best_h[idxs].copy()
            for _ in range(3):
                for ii in idxs:
                    i = ii - idxs[0]
                    idx1 = ii - (np.arange(2 * n - 1) - (n - 1))
                    idx2 = ii + (np.arange(2 * n - 1) - (n - 1))
                    valid1 = (idx1 >= 0) & (idx1 < n)
                    valid2 = (idx2 >= 0) & (idx2 < n)
                    a = np.zeros(2 * n - 1)
                    h_eval = best_h.copy()
                    h_eval[idxs] = best_block
                    a[valid1] += 1.0 - h_eval[idx1[valid1]]
                    a[valid2] -= h_eval[idx2[valid2]]
                    lo = -best_block[i]
                    hi = 1.0 - best_block[i]
                    c_c = np.correlate(h_eval, 1.0 - h_eval, mode='full')
                    def f_b(d):
                        return np.max(c_c + d * a)
                    res = minimize_scalar(f_b, bounds=(lo, hi), method='bounded', options={'xatol': 1e-10})
                    if abs(res.x) > 1e-10:
                        best_block[i] += res.x
                best_block = np.clip(best_block, 0.0, 1.0)
            new_h = best_h.copy()
            new_h[idxs] = best_block
            new_h = project(new_h, target)
            new_corr = np.correlate(new_h, 1.0 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            if new_c5 < best_c5 - 1e-14:
                best_h = new_h
                corr = new_corr
                best_c5 = new_c5
                improved_any = True
        if not improved_any:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    target = n / 2.0
    best_h = project(best_h, target)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Smooth gradient descent with increasing T (12% budget)
    t1 = t0 + budget_s * 0.12
    for T in [30, 100, 500]:
        if elapsed() > budget_s * 0.10:
            break
        h_g, _ = proj_grad_desc(best_h, T, target, n, t1, max_iter=3000)
        h_g = project(h_g, target)
        c5_v = compute_c5(h_g)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h_g

    # Phase 2: Intensive coordinate descent (50% budget)
    t2 = t0 + budget_s * 0.62
    best_h, best_c5 = coord_descent_fast(best_h, target, t2, max_sweeps=50000)

    # Phase 3: Block coordinate descent + diversification (18%)
    t3 = t0 + budget_s * 0.80
    best_h, best_c5 = block_cd(best_h, target, t3, block_size=3, max_sweeps=2000)
    best_h, best_c5 = coord_descent_fast(best_h, target, min(t3, time.time() + budget_s * 0.05), max_sweeps=5000)

    # Phase 4: Diversification attempts (10%)
    t4 = t0 + budget_s * 0.90
    for trial in range(8):
        if elapsed() > (t4 - t0):
            break
        noise_scale = 0.02 + 0.03 * (trial / 8.0)
        h_p = best_h + noise_scale * np.random.randn(n)
        h_p = project(h_p, target)
        h_p, c5_p = coord_descent_fast(h_p, target, min(t4, time.time() + budget_s * 0.012), max_sweeps=3000)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p
        if time.time() > t4:
            break
        # Also try from a more aggressive perturbation
        h_p2 = np.clip(best_h + 0.06 * np.random.randn(n), 0, 1)
        h_p2 = project(h_p2, target)
        h_p2, c5_p2 = coord_descent_fast(h_p2, target, min(t4, time.time() + budget_s * 0.008), max_sweeps=1500)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2

    # Phase 5: Final intensive polish (10%)
    t5 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent_fast(best_h, target, t5, max_sweeps=30000)

    best_h = project(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
