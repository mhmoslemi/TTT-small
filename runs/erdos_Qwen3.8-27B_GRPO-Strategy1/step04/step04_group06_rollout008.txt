    ws = ew.sum()
    w = ew / ws
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(w)):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - center
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < k < n:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        elif -n < k < 0:
            m2 = -k
            grad[:n-m2] += wk * (1.0 - h[m2:])
            grad[m2:] -= wk * h[:n-m2]
    return dx * grad

def project(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = h + room * (diff / tot)
    else:
        if s > 1e-14:
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_h(h0, T, n, maxiter=1000, ftol=1e-16):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h0, jac=lambda x: smooth_obj_grad(x, T),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project(res.x, n)
    except Exception:
        return h0

def coord_opt_single(h, n, i):
    """Optimize single coordinate i"""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k if i + k < n else -1
        im = i - k if i - k >= 0 else -1
        if k < n:
            if ip >= 0:
                a[center + k] += 1.0 - h[ip]
            if im >= 0:
                a[center - k] += 1.0 - h[im]
            if im >= 0:
                a[center + k] -= h[im]
            if ip >= 0:
                a[center - k] -= h[ip]
    dx = 2.0 / n
    cb = corr_base
    ct = cb[center]
    
    def f1d(x):
        vals = cb + x * a
        vals[center] = ct + x - x*x
        return np.max(vals) * dx
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_single(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5
