    dx = 2.0 / n

                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        
        if not improved and sweep > 2:
            break
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    return best_h, best_c5

def block_perturb(h, n, strength, rng):
    L = rng.randint(3, max(4, n // 8))
    start = rng.randint(0, n - L + 1)
    h2 = h.copy()
    block = h2[start:start+L].copy()
    h2[start:start+L] = np.clip(block + strength * rng.randn(L), 0, 1)
    return h2

def swap_perturb(h, n, rng, num_swaps=3):
    h2 = h.copy()
    for _ in range(num_swaps):
        a, b = rng.choice(n, 2, replace=False)
        h2[a], h2[b] = h2[b], h2[a]
    return project(h2, n)

def basis_hopping(work_h, work_c5, n, t_end, max_iters=400, rng=None):
    if rng is None:
        rng = np.random.RandomState(99)
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.03, 1.0 - 0.9 * it / max_iters)
        pert_type = rng.randint(0, 4)
        if pert_type == 0:
            s = 0.02 + 0.15 * cool
            h_p = block_perturb(work_h, n, s, rng)
        elif pert_type == 1:
            h_p = swap_perturb(work_h, n, rng, num_swaps=max(1, int(2*cool)+1))
        else:
            s = 0.03 + 0.12 * cool
            h_p = work_h + s * rng.randn(n)
            h_p = project(h_p, n)
        h_p = project(h_p, n)
        h_c, c5_c = cd_1var(h_p, n, min(time.time() + 15, t_end), max_sweeps=80)
        temp = 0.0015 * cool + 0.0001
        if c5_c < work_c5 or rng.rand() < np.exp(-(c5_c - work_c5) / temp):
            work_h, work_c5 = h_c.copy(), c5_c
            if c5_c < best_c5:
                best_c5, best_h = c5_c, h_c.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def remaining():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        best_h = np.ones(n0) * 0.5
        for i in range(0, n0, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive 1-var CD
    t1 = t0 + budget_s * 0.22
    best_h, best_c5 = cd_1var(best_h, n, t1, max_sweeps=3000)

    # Phase 2: 2-var CD (new: escapes 1-var local minima)
    t2 = t0 + budget_s * 0.38
    best_h, best_c5 = cd_2var(best_h, n, t2, max_sweeps=500)

    # Phase 3: Fine-tune 1-var CD
    t3 = t0 + budget_s * 0.46
    best_h, best_c5 = cd_1var(best_h, n, t3, max_sweeps=2000)

    # Phase 4: Basin hopping
    t4 = t0 + budget_s * 0.75
    best_h, best_c5 = basis_hopping(best_h, best_c5, n, t4, max_iters=500, rng=rng)

    # Phase 5: After hopping, 2-var then 1-var refinement
    t5 = t0 + budget_s * 0.82
    best_h, best_c5 = cd_2var(best_h, n, t5, max_sweeps=300)
    t5b = t0 + budget_s * 0.86
    best_h, best_c5 = cd_1var(best_h, n, t5b, max_sweeps=2000)

    # Phase 6: Try different resolutions
    for n_try in [128, 144, 192, 256]:
        if remaining() < budget_s * 0.04:
            break
        h_t = interp_h(best_h, n_try)
        t_a = t0 + budget_s * 0.01 + (budget_s * 0.03) * (len([x for x in [128,144,192,256] if x <= n_try]) - 1)
        t_b = min(t0 + budget_s * 0.92, time.time() + budget_s * 0.03)
        h_t, c5_t = cd_1var(h_t, n_try, t_b, max_sweeps=800)
        h_t, c5_t = cd_2var(h_t, n_try, min(t0 + budget_s * 0.93, time.time() + 10), max_sweeps=100)
        if c5_t < best_c5:
            best_c5, best_h, n = c5_t, h_t, n_try

    # Phase 7: Second basin hop round (shorter)
    t7 = min(t0 + budget_s * 0.95, time.time() + remaining() * 0.4)
    if remaining() > budget_s * 0.05:
        best_h, best_c5 = basis_hopping(best_h, best_c5, n, t7, max_iters=200, rng=rng)

    # Phase 8: Final squeeze
    t8 = t0 + budget_s * 0.98
    best_h, best_c5 = cd_2var(best_h, n, t8, max_sweeps=200)
    best_h, best_c5 = cd_1var(best_h, n, t0 + budget_s * 0.995, max_sweeps=2000)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
