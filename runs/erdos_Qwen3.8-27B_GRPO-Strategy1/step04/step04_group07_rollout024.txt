            f0 = _obj(0.0)
            f1 = _obj(1.0)
            if f0 < bf:
                bv, bf = 0.0, f0
            if f1 < bf:
                bv, bf = 1.0, f1
            if bf * dx < best_c5 - 1e-15:
                h[i] = bv
                best_c5 = bf * dx
                corr[:] = np.correlate(h, 1.0 - h, mode='full')
                improved += 1
        if improved == 0:
            break
    return h, best_c5

def slsqp_smooth(h, n, T, maxiter=400):
    def obj(x):
        v, _ = smooth_max_val_grad(x, T)
        return v
    def con_eq(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_eq}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_h(np.clip(res.x, 0, 1), n)

def basin_hop(h0, c5_0, n, t_end, n_iter=200, seed=0):
    rng = np.random.default_rng(seed)
    best_h, best_c5 = h0.copy(), c5_0
    cur_h, cur_c5 = h0.copy(), c5_0
    xs = np.arange(n, dtype=float)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.05, 1.0 - 0.85 * it / max(1, n_iter))
        r = rng.random()
        if r < 0.25:
            blen = int(rng.integers(4, max(5, n // 5)))
            start = int(rng.integers(0, n - blen))
            h_p = cur_h.copy()
            h_p[start:start + blen] = 1.0 - h_p[start:start + blen]
        elif r < 0.55:
            h_p = cur_h.copy()
            amp = 0.04 + 0.12 * cool
            for _ in range(int(rng.integers(2, 5))):
                freq = rng.uniform(1.0, 10.0)
                ph = rng.uniform(0, 2 * np.pi)
                h_p = h_p + amp * np.sin(freq * xs * 2.0 * np.pi / n + ph)
        else:
            h_p = cur_h + (0.02 + 0.10 * cool) * rng.standard_normal(n)
        h_p = project_h(h_p, n)
        T_use = int(80 + 3000 * cool)
        try:
            hc = slsqp_smooth(h_p, n, T_use, maxiter=int(150 + 200 * cool))
            c5v = compute_c5(hc)
        except Exception:
            continue
        temp = 0.0015 * cool + 0.0001
        if c5v < cur_c5 or rng.random() < np.exp(-(c5v - cur_c5) / max(temp, 1e-9)):
            cur_h, cur_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if cur_c5 < best_c5:
        best_c5, best_h = cur_c5, cur_h
    return best_h, best_c5

def interp_h(h, n_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_h(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl(f):
        return t0 + budget_s * f

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        best_h[::2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Smooth gradient descent
    best_h, best_c5 = gd_phase(best_h, n, tl(0.08), steps=2000)

    # Phase 2: Coordinate descent (main)
    best_h, best_c5 = coord_descent(best_h, n, tl(0.28), max_sweeps=600)

    # Phase 3: SLSQP smooth max schedule
    for T in [100, 400, 1500, 5000, 15000]:
        if time.time() > tl(0.40):
            break
        try:
            hc = slsqp_smooth(best_h, n, T, maxiter=600)
            c5v = compute_c5(hc)
            if c5v < best_c5 - 1e-14:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 4: Coordinate descent refinement
    best_h, best_c5 = coord_descent(best_h, n, tl(0.50), max_sweeps=400)

    # Phase 5: Basin hopping (structured perturbations)
    best_h, best_c5 = basin_hop(best_h, best_c5, n, tl(0.70), n_iter=200, seed=seed)

    # Phase 6: Post-hop coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, tl(0.80), max_sweeps=400)

    # Phase 7: Multi-resolution exploration
    for n_try in [140, 180, 200, 240, 320]:
        if time.time() > tl(0.88):
            break
        ht = interp_h(best_h, n_try)
        ht, c5t = coord_descent(ht, n_try, tl(0.89), max_sweeps=150)
        if c5t < best_c5 - 1e-14:
            best_c5, best_h, n = c5t, ht, n_try

    # Phase 8: Final SLSQP + coordinate descent polish
    best_h, best_c5 = coord_descent(best_h, n, tl(0.94), max_sweeps=300)
    if budget_s - (time.time() - t0) > 40:
        for T in [20000, 80000, 200000]:
            if budget_s - (time.time() - t0) < 20:
                break
            try:
                hc = slsqp_smooth(best_h, n, T, maxiter=1200)
                c5v = compute_c5(hc)
                if c5v < best_c5 - 1e-14:
                    best_c5, best_h = c5v, hc
            except Exception:
                pass
    best_h, best_c5 = coord_descent(best_h, n, tl(0.99), max_sweeps=100)

    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
