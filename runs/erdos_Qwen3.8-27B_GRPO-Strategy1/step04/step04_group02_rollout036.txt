```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2

def lbfgs_opt(h, T, n, t_end, maxiter=5000):
    def obj(x):
        return smooth_c5(x, T)
    def grad(x):

        eps = 1e-7
        g = np.zeros(n)
        for i in range(n):

            xp[i] += eps
            xm = x.copy()
            xm[i] -= eps
```

The smooth C5 = dx * logsumexp(T * corr) / T + dx * max(corr)

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr = np.clip(corr, -1e10, 1e10)
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(T * (corr - m)))) / T)
    return val * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.clip(T * (corr - m), -500, 500)
    w = np.exp(exps)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    g = np.zeros(n)
    for k_idx in range(2*n-1):
        wki = w[k_idx]
        if wki < 1e-14:
            continue
        k = k_idx - (n-1)
        if k >= 0:
            g[:n-k] -= wki * h[k:]
            g[k:] += wki * (1.0 - h[:n-k])
        else:
            mv = -k
            g[:n-mv] += wki * (1.0 - h[mv:])
            g[mv:] -= wki * h[:n-mv]
    return g * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if abs(diff2) > 1e-14:
            if diff2 > 0:
                room = 1.0 - h
                if room.sum() > 1e-14:
                    h = h + room * (diff2 / room.sum())
            elif s2 > 1e-14:
                h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def prox_grad_descent(h, n, T, t_end, lr0=0.01, nesterov=True):
    h = h.copy().astype(float)
    y = h.copy()
    alpha = 1.0
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = lr0
    it = 0
    while time.time() < t_end:
        g = smooth_c5_grad(y, T)
        y_new = y - lr * g
        h_new = project(y_new, n)
        # Line search (simple: just accept or halve)
        cv = compute_c5(h_new)
        if cv <= compute_c5(h) + 1e-15 or True:  # accept for now
            if nesterov:
                alpha_new = (1 + np.sqrt(1 + 4*alpha*alpha)) / 2
                y = h_new + (alpha / alpha_new - 1) * (h_new - h)
                alpha = alpha_new
            else:
                y = h_new.copy()
            h = h_new
            if cv < best_c5:
                best, best_c5 = h_new.copy(), cv
            lr *= 0.999
        else:
            lr *= 0.5
        it += 1
        if it % 500 == 0:
            cv_check = compute_c5(h)
            if cv_check < best_c5:
                best, best_c5 = h.copy(), cv_check
    if compute_c5(h) < best_c5:
        best, best_c5 = h.copy(), compute_c5(h)
    return best, best_c5

def coord_opt_fixed(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        val_p = 0.0
        val_m = 0.0
        if 0 <= i - k < n:
            val_p += (1.0 - h[i - k])
            val_m += (-h[i - k])
        if 0 <= i + k < n:
            val_p += (-h[i + k])
            val_m += (1.0 - h[i + k])
        d[n - 1 + k] = val_p
        d[n - 1 - k] = val_m
    c0 = corr_base[n - 1]
    
    def f1d(t):
        peak = c0 + t * (1.0 - t)
        # Approximate max as max of peak and the linear terms
        # For speed, only check a subset
        vals = corr_base + t * d
        vals[n-1] = c0 + t * (1.0 - t)
        return np.max(vals)
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fixed(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-13:
                best_h[i] = hi
                best_h = project(best_h, n)
                improved = True
        if improved:
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=2000, ftol=1e-16):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def interpolate_h(h_old, n_new):
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.clip(T * (corr - m), -500, 500)
    w = np.exp(exps)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for k_idx in range(2*n-1):
        wki = w[k_idx]
        if wki < 1e-14:
            continue
        k = k_idx - (n-1)
        if k >= 0:
            if n-k > 0:
                g[:n-k] -= wki * h[k:]
            if k < n:
                g[k:] += wki * (1.0 - h[:n-k])
        else:
            mv = -k
            if n-mv > 0:
                g[:n-mv] += wki * (1.0 - h[mv:])
            if mv < n:
                g[mv:] -= wki * h[:n-mv]
    return g * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        return h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            if room.sum() > 1e-14:
                h = h + room * (diff2 / room.sum())
        elif diff2 < -1e-14 and s2 > 1e-14:
            h = np.clip(h * (target / s2), 0.0, 1.0)
        return h

def nesterov_pg(h, n, T, t_end, lr0=0.03, n_iter_max=50000):
    best_h = h.copy().astype(float)
    best_c5 = compute_c5(best_h)
    x = best_h.copy()
    y = x.copy()
    alpha = 1.0
    lr = lr0
    for it in range(n_iter_max):
        if time.time() > t_end:
            break
        g = smooth_c5_grad(y, T)
        y_new = y - lr * g
        x_new = project(y_new, n)
        cv = compute_c5(x_new)
        if cv < best_c5 - 1e-15:
            best_h, best_c5 = x_new.copy(), cv
        alpha_new = (1.0 + np.sqrt(1.0 + 4.0*alpha*alpha)) / 2.0
        y = x_new + (alpha / alpha_new - 1.0) * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        alpha = alpha_new
        x = x_new
        lr *= 0.9995
        if it > 100:
            lr = max(lr, 0.001)
    return best_h, best_c5

def coord_descent_corr(h, n, t_end, max_sweeps=500):

def slsqp_opt(h, T, n, maxiter=2000, ftol=1e-16):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

            pass
    if time.time() < t0 + budget_s * 0.85:
        best_h, best_c5 = coord_descent_corr(best_h, n, t_end=t0 + budget_s*0.85)

    # Phase 6: Multi-resolution
    for n_try in [160, 180, 200, 240, 280, 320]:
        if time.time() > t0 + budget_s * 0.96:
            break
        h_int = interpolate_h(best_h, n_try)
        cv_int = compute_c5(h_int)
        if cv_int < best_c5:
            best_h, best_c5, n = h_int, cv_int, n_try
        else:
            h_try = interpolate_h(best_h, n_try)
            try:
                hc = slsqp_opt(h_try, 1000, n_try, maxiter=800, ftol=1e-14)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5, n = hc, cv, n_try
            except Exception:
                pass
        if n == n_try and time.time() < t0 + budget_s * 0.97:
            best_h, best_c5 = coord_descent_corr(best_h, n, t_end=t0 + budget_s*0.97)

    # Final: one more coord descent pass
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = coord_descent_corr(best_h, n, t_end=t0 + budget_s*0.995, max_sweeps=100)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
