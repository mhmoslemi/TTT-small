```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    wsum = exps.sum()
    logsum = m + np.log(wsum)
    val = (logsum / T) * dx
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        k = np.arange(2*n - 1) - (n - 1)  # -（n-1), ..., n-1
        # a[m] = 1 - h[j+k[m]] if 0 <= j+k[m] < n
        idx_a = j + k
        valid_a = (idx_a >= 0) & (idx_a < n)
        a = np.zeros(2*n-1)
        a[valid_a] = 1.0 - h[idx_a[valid_a]]
        
        # b[m] = -h[j-k[m]] if 0 <= j-k[m] < n
        idx_b = j - k
        valid_b = (idx_b >= 0) & (idx_b < n)
        b = np.zeros(2*n-1)
        b[valid_b] = -h[idx_b[valid_b]]
        
        grad[j] = np.dot(exps, a + b) / wsum * dx
    
    return val, grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if abs(diff2) > 1e-14:
            if diff2 > 0:
                room = 1.0 - h
                tot = room.sum()
                if tot > 1e-14:
                    h = h + room * (diff2 / tot)
            else:
                h = np.clip(h * (target / max(s2, 1e-14)), 0.0, 1.0)
    return h
```

Let me use option 3: implement a simple ternary search or golden section directly in the coordinate descent loop.

```python
def golden_min_1d(f, a, b, tol=1e-10, maxiter=200):
    """Golden section search for minimum of f on [a,b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(maxiter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return 0.5 * (a + b)
```

But f needs to be callable without closures. I can make it a function that takes (t, c_base, g, n, dx) and use it directly in the golden section... but golden_min_1d expects f(t). 

Alternative: I'll just do a vectorized grid search + refinement:

```python
def find_opt_t(c_base, g, n, dx, hi):
    """Find optimal t in [0,1] for coordinate change."""
    # Coarse grid
    ts = np.linspace(0.0, 1.0, 201)
    vals = c_base[None, :] + ts[:, None] * g[None, :]
    # Fix the n-1 entry
    vals[:, n-1] = c_base[None, n-1] + ts[:, None] - ts[:, None]**2
    maxvals = np.max(vals, axis=1) * dx
    i0 = np.argmin(maxvals)
    
    # Fine search around ts[i0]
    t_lo = max(0.0, ts[i0] - 1.0/200.0)
    t_hi = min(1.0, ts[i0] + 1.0/200.0)
    ts_fine = np.linspace(t_lo, t_hi, 201)
    vals = c_base[None, :] + ts_fine[:, None] * g[None, :]
    vals[:, n-1] = c_base[None, n-1] + ts_fine[:, None] - ts_fine[:, None]**2
    maxvals = np.max(vals, axis=1) * dx
    j0 = np.argmin(maxvals)
    return float(ts_fine[j0])
```

This is O(2 * 201 * 2n) ≈ O(n) per coordinate (the vectorized numpy operations are fast). For n=240 coordinates, that's about 240 * 201 * 480 ≈ 23M operations per sweep. In numpy, vectorized, that's about 0.1s per sweep. With 50 sweeps, 5s. 

Actually wait, the arrays are 201 × (2n-1) = 201 × 479 ≈ 96K elements. Computing vals for 201 points: 201 × 479 = 96K operations, twice (coarse + fine) = 192K per coordinate. For 240 coordinates: 46M per sweep. In numpy floating point, that's about 0.1-0.2s per sweep. 50 sweeps = 5-10s. Acceptable.

```python
import numpy as np
from scipy.optimize import minimize
import time

_OPT_T = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smooth_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return float((m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx)

def _smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    wsum = float(exps.sum())
    k_arr = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        idx_a = j + k_arr
        va = (idx_a >= 0) & (idx_a < n)
        idx_b = j - k_arr
        vb = (idx_b >= 0) & (idx_b < n)
        contrib = np.zeros(2 * n - 1)
        if va.any():
            contrib[va] = 1.0 - h[idx_a[va]]
        if vb.any():
            contrib[vb] += -h[idx_b[vb]]
        grad[j] = float(np.dot(exps, contrib) / wsum * dx)
    return grad

def _slsqp_obj(x):
    return _smooth_val(x, _OPT_T)

def _slsqp_grad(x):
    return _smooth_grad(x, _OPT_T)

def _sum_eq(x):
    return x.sum() - len(x) / 2.0

def _sum_eq_jac(x):
    return np.ones(len(x))

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(3):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            h = h * (target / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
    return h

def find_opt_t(c_base, g, n, dx):
    ts = np.linspace(0.0, 1.0, 101)
    vals = c_base[None, :] + ts[:, None] * g[None, :]
    vals[:, n - 1] = c_base[n - 1] + ts - ts * ts
    maxvals = vals.max(axis=1) * dx
    i0 = int(np.argmax(1.0 / (1.0 + np.exp(50.0 * (maxvals - maxvals.min())))))
    i0 = int(np.argmin(maxvals))
    t_lo = max(0.0, ts[i0] - 1.0 / 100.0)
    t_hi = min(1.0, ts[i0] + 1.0 / 100.0)
    tf = np.linspace(t_lo, t_hi, 101)
    vals = c_base[None, :] + tf[:, None] * g[None, :]
    vals[:, n - 1] = c_base[n - 1] + tf - tf * tf
    maxvals = vals.max(axis=1) * dx
    j0 = int(np.argmin(maxvals))
    return float(tf[j0])

def coord_descent(h, n, t_end, max_sweeps=50, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    best_c5 = float(corr.max() * dx)
    k_arr = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = best_h[i]
            g = np.zeros(2 * n - 1)
            idx_a = i + k_arr
            va = (idx_a >= 0) & (idx_a < n)
            idx_b = i - k_arr
            vb = (idx_b >= 0) & (idx_b < n)
            if va.any():
                g[va] = 1.0 - best_h[idx_a[va]]
            if vb.any():
                g[vb] += -best_h[idx_b[vb]]
            c_base = corr.copy()
            c_base[n - 1] = corr[n - 1] - hi * (1.0 - hi)
            c_base -= hi * g
            c_base[n - 1] = corr[n - 1] - hi * (1.0 - hi)
            t_opt = find_opt_t(c_base, g, n, dx)
            if abs(t_opt - hi) < 1e-12:
                continue
            best_h[i] = t_opt
            best_h = project(best_h, n)
            corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
            c5 = float(corr.max() * dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
            elif c5 > best_c5 + 1e-10:
                best_h[i] = hi
                best_h = project(best_h, n)
                corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def slsqp_with_grad(h, T, n, maxiter=1000):
    global _OPT_T
    _OPT_T = T
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def perturb(h, n, rng, scale=0.03):
    p = h + scale * rng.standard_normal(n)
    return project(p, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n = 240
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    for T in [300, 1500, 6000, 20000]:
        if elapsed() > budget_s * 0.25:
            break
        hc = slsqp_with_grad(best_h, T, n, maxiter=800)
        cv = compute_c5(hc)
        if cv < best_c5 - 1e-14:
            best_h, best_c5 = hc, cv

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.40, max_sweeps=50, rng=rng)

    for T in [20000, 60000]:
        if elapsed() > budget_s * 0.50:
            break
        hc = slsqp_with_grad(best_h, T, n, maxiter=1200)
        cv = compute_c5(hc)
        if cv < best_c5 - 1e-14:
            best_h, best_c5 = hc, cv

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.60, max_sweeps=40, rng=rng)

    for restart in range(4):
        if elapsed() > budget_s * 0.78:
            break
        rng_r = np.random.default_rng(seed + 100 + restart * 7)
        h_p = perturb(best_h, n, rng_r, scale=0.02 + 0.01 * restart)
        for T in [500, 3000]:
            if elapsed() > budget_s * 0.78:
                break
            hc = slsqp_with_grad(h_p, T, n, maxiter=500)
            cv = compute_c5(hc)
            hc, cv2 = coord_descent(hc, n, min(elapsed() + 8, t0 + budget_s * 0.78), max_sweeps=15, rng=rng_r)
            if cv2 < best_c5 - 1e-14:
                best_h, best_c5 = hc, cv2

    for n_try in [180, 280, 320]:
        if elapsed() > budget_s * 0.92:
            break
        h_interp = interpolate_h(best_h, n_try)
        for T in [1000, 8000]:
            if elapsed() > budget_s * 0.93:
                break
            hc = slsqp_with_grad(h_interp, T, n_try, maxiter=600)
            cv = compute_c5(hc)
            hc, cv = coord_descent(hc, n_try, min(elapsed() + 5, t0 + budget_s * 0.93), max_sweeps=15, rng=rng)
            if cv < best_c5 - 1e-14:
                best_h, best_c5, n = hc, cv, n_try

    if elapsed() < budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=30, rng=rng)

    if elapsed() < budget_s * 0.99:
        hc = slsqp_with_grad(best_h, 50000, n, maxiter=1500)
        cv = compute_c5(hc)
        if cv < best_c5 - 1e-14:
            best_h = hc
            best_c5 = cv

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
