import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)

def smooth_c5(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log1p(np.sum(np.exp(T * (corr - m)) - 1.0)) / T  # numerically stable
    return val * dx

def coord_descent_fast(h_in, n, t_end, max_sweeps=400):

    corr = np.correlate(h, 1.0 - h, mode='full')

    c0_idx = n - 1
    ks = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = h[i]
            # Compute coeff
            idx1 = i - ks  # where h appears in (1-h) factor: h[i-k]
            idx2 = i + ks  # where h appears in h factor: h[i+k]
            coeff = np.zeros(2*n - 1)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff[v1] += 1.0 - h[idx1[v1]]
            coeff[v2] -= h[idx2[v2]]
            
            # Find best v
            grid = np.linspace(0, 1, 201)
            delta = grid - old
            all_c = corr[:, None] + delta[None, :] * coeff[:, None]
            all_c[c0_idx, :] = corr[c0_idx] + delta * (1.0 - grid - old)
            f_vals = np.max(all_c, axis=0)
            
            i_min = np.argmin(f_vals)
            lo = grid[max(0, i_min-3)]
            hi = grid[min(200, i_min+3)]
            
            def f1d(v):
                d = v - old
                cn = corr + d * coeff
                cn[c0_idx] = corr[c0_idx] + d * (1.0 - v - old)
                return np.max(cn)

            if lo < hi:

                if res.fun < best_f:

def slsqp_sym(x, T, n, maxiter=400):

        h[:half] = x
        h[half:] = 1.0 - x[::-1]
        return smooth_c5(h, T)

        return obj_and_build(x)
    res = minimize(obj, x, method='SLSQP', bounds=[(0,1)]*half,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

    # Phase 7: basin hopping round 2 (different perturbations)
    best_h, best_c5 = basin_hop_sym(best_h, best_c5, n, t0 + budget_s * 0.78, max_iters=250, seed_off=seed + 999)

    # Phase 8: coordinate descent
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.85, max_sweeps=1000)

    # Phase 9: try higher resolution
    best_n = n
    for n_try in [200, 240, 320]:
        if elapsed() > budget_s * 0.93:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent_fast(h_tr, n_try, t0 + budget_s * 0.935, max_sweeps=150)
        if c5_tr < best_c5:
            best_c5, best_h, best_n = c5_tr, h_tr, n_try

    # Phase 10: final SLSQP + coordinate descent
    if elapsed() < budget_s * 0.96:
        try:
            h_s, c5_s = try_slq(best_h, best_n, 30000, 1000)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s
        except Exception:
            pass
    best_h, best_c5 = coord_descent_fast(best_h, best_n, t0 + budget_s * 0.99, max_sweeps=500)
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
