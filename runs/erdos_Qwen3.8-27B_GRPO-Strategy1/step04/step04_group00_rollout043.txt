import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-14:
            return h
        h = h + rm * (d / t)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def soft_c5(x, T):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    return float((m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx)

def sum_con(x, n):
    return x.sum() - n / 2.0

    try:
        r = minimize(soft_c5, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                     args=(T,), options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(r.x, 0, 1))
    except Exception:
        return h

    return float(np.max(cb + hi * cf) * dx)

def coord_best(h, n, i):
    sv = h[i]
    h[i] = 0.0

    h[i] = sv

                           args=(cb, cf, dx), options={'xatol': 1e-12}).x

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        rm = 1.0 - h
        tt = rm.sum()
        if tt < 1e-14:
            return h
        h = h + rm * (d / tt)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def soft_c5(x, T):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    return float((m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx)

def sum_con(x, n):
    return x.sum() - n / 2.0

def slsqp_round(h, T, n, maxiter=800):
    try:
        r = minimize(soft_c5, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                     args=(T,), options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(r.x, 0, 1))
    except Exception:
        return h

def f1d_c(hi, cb, cf, dx):
    return float(np.max(cb + hi * cf) * dx)

def coord_best(h, n, i):
    sv = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full')
    h[i] = sv
    cf = np.zeros(2 * n - 1)
    sa = np.arange(-(n - 1), n)
    ia = sa + n - 1
    m1 = (i - sa >= 0) & (i - sa < n)
    cf[ia[m1]] += 1.0 - h[i - sa[m1]]
    m2 = (i + sa >= 0) & (i + sa < n)
    cf[ia[m2]] -= h[i + sa[m2]]
    dx = 2.0 / n
    return minimize_scalar(f1d_c, bounds=(0.0, 1.0), method='bounded',
                           args=(cb, cf, dx), options={'xatol': 1e-12}).x

def sg_step(h, alpha, n, T):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    w = np.exp(T * (c - m))
    ws = w.sum()
    if ws < 1e-30:
        return h
    w /= ws
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        a = w[idx]
        if a < 1e-12:
            continue
        s = idx - n + 1
        l1 = max(0, s)
        h1 = min(n, n + s)
        if h1 > l1:
            g[l1:h1] += a * (1.0 - h[l1 - s:h1 - s])
        l2 = max(0, -s)
        h2 = min(n, n - s)
        if h2 > l2:
            g[l2:h2] += -a * h[l2 + s:h2 + s]
    return project_h(h - alpha * g * dx)

def cd_loop(h, n, t_end, rng, max_sw=500):
    best = h.copy()
    bc = compute_c5(best)
    for _ in range(max_sw):
        if time.time() > t_end:
            break
        imp = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best, bc
            hn = coord_best(best, n, i)
            if abs(hn - best[i]) < 1e-13:
                continue
            bt = best.copy()
            bt[i] = hn
            bt = project_h(bt)
            c5 = compute_c5(bt)
            if c5 < bc - 1e-15:
                bc = c5
                best = bt
                imp = True
        if not imp:
            break
    return best, bc

        T = 5.0 + 200.0 * (1.0 - cool)
