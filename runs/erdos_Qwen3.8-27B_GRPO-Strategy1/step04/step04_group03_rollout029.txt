def fast_coord_descent(h, t_end, n_sweeps=10000):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        n_improved = 0
        for i in range(n):
            h_old = best_h[i]
            # Compute coefficient
            coef = np.zeros(2*n-1)
            # (1-h[i+s]) for valid s
            s_lo = max(0, -i)
            s_hi = min(n-1-i, n-1)
            if s_lo <= s_hi:
                coef[s_lo+n-1:s_hi+n] += (1.0 - best_h[i+s_lo:i+s_hi+1])
            # -h[i-s] for valid s
            s_lo2 = max(-(n-1-i), -(n-1))
            s_hi2 = min(i, n-1)
            if s_lo2 <= s_hi2:
                idx = i - np.arange(s_hi2, s_lo2-1, -1)
                coef[s_lo2+n-1:s_hi2+n] -= best_h[idx]
            
            # Golden section
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-13:
                continue
            a, b = lo, hi
            c_val = b - gr*(b-a)
            d_val = a + gr*(b-a)
            fc = np.max(corr + c_val * coef)
            fd = np.max(corr + d_val * coef)
            for _ in range(30):
                if fc <= fd:
                    b = d_val; d_val = c_val; fd = fc
                    c_val = b - gr*(b-a)
                    fc = np.max(corr + c_val * coef)
                else:
                    a = c_val; c_val = d_val; fc = fd
                    d_val = a + gr*(b-a)
                    fd = np.max(corr + d_val * coef)
                if b - a < 1e-11:
                    break
            s_new = (a+b) * 0.5
            if abs(s_new) > 1e-13:
                corr += s_new * coef
                best_h[i] = np.clip(h_old + s_new, 0.0, 1.0)
                new_max = np.max(corr) * dx
                if new_max < best_c5 - 1e-15:
                    best_c5 = new_max
                    n_improved += 1
        
        if n_improved == 0:
            break
    
    return best_h, best_c5
