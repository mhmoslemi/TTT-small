    dx = 2.0 / n
    corr = compute_corr(h)
    # Softmax weights
    corr_norm = corr - corr.max()
    weights = np.exp(T * corr_norm)
    weights /= weights.sum()
    # Gradient of weighted sum: Σ w[m] * ∂corr[m]/∂h
    # ∂corr[m]/∂h[i] = (1-h[i-d]) * [0<=i-d<n] - h[i+d] * [0<=i+d<n] where d = m-(n-1)
    # For each i: grad[i] = Σ_m w[m] * ((1-h[i-d_m]) - h[i+d_m])

        return h
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            coef = np.zeros(2 * n - 1)
            for d in range(-(n - 1), n):
                c = 0.0
                j1 = i - d
                j2 = i + d
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[d + n - 1] = c
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if hi - lo < 1e-14:
                continue
            res = minimize_scalar(lambda s: np.max(corr + s * coef),
                                  bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-13})
            s_new = res.x
            if abs(s_new) > 1e-14:
                best_h[i] = np.clip(best_h[i] + s_new, 0, 1)
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5 - 1e-16:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def subgradient_phase(h, n, t_end):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    step = 0.03
    t0 = time.time()
    n_iter = 0
    while time.time() < t_end:
        n_iter += 1
        progress = min(1.0, (time.time() - t0) / max(t_end - t0, 1e-6))
        T = 50 + 4950 * progress
        corr = compute_corr(h)
        cs = corr - corr.max()
        w = np.exp(T * cs)
        w /= w.sum()
        g = grad_weighted_corr(h, w, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        for _ in range(30):
            h_new = project_to_sum(h - step * g / g_norm, n / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) * (1 + 1e-10):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                step = min(step * 1.05, 0.15)
                break
            step *= 0.6
        if step < 1e-7:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(224)
        best_h[:112] = 1.0
    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with gradient, T schedule (12% budget)
    t1 = t0 + budget_s * 0.12
    for T in [50, 200, 1000, 5000, 20000]:
        if time.time() > t1:
            break
        try:
            hc = slsqp_grad(best_h, T, n, maxiter=600, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 2: Coordinate descent (30% budget)
    t2 = t0 + budget_s * 0.42
    best_h, best_c5 = coord_descent(best_h, n, t2)

    # Phase 3: Subgradient with adaptive T (20% budget)
    t3 = t0 + budget_s * 0.62
    hs, cs = subgradient_phase(best_h, n, t3)
    if cs < best_c5:
        best_h, best_c5 = hs, cs

    # Phase 3b: SLSQP after subgradient
    if time.time() < t0 + budget_s * 0.68:
        for T in [10000, 50000]:
            if time.time() > t0 + budget_s * 0.68:
                break
            try:
                hc = slsqp_grad(best_h, T, n, maxiter=800, ftol=1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_h, best_c5 = hc, c5v
            except Exception:
                pass

    # Phase 4: Basin hopping with diverse perturbations (15% budget)
    t4 = t0 + budget_s * 0.83
    work_h = best_h.copy()
    n_bh = 40
    for it in range(n_bh):
        if time.time() > t4:
            break
        cool = max(1.0 - 0.9 * (it / float(n_bh)), 0.05)
        r = np.random.rand()
        if r < 0.35:
            step = 0.01 + 0.08 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.65:
            shift = int(np.random.uniform(-0.03, 0.03) * n)
            h_p = 0.7 * np.roll(work_h, shift) + 0.3 * work_h
        else:
            mask = np.random.rand(n) < 0.2
            h_p = work_h.copy()
            h_p[mask] += 0.05 * cool * np.random.randn(int(mask.sum()))
        h_p = project_to_sum(h_p, n / 2.0)
        try:
            T_bh = int(200 + 3000 * cool)
            hc = slsqp_grad(h_p, T_bh, n, maxiter=300, ftol=1e-13)
            hc, c5v = coord_descent(hc, n, time.time() + 1.0, max_sweeps=10)
            if c5v < work_c5 if 'work_c5' in dir() else c5v < compute_c5(work_h):
                work_h, c5v = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = hc.copy(), c5v
        except Exception:
            pass
    # Fix: track work_c5 properly
    # (re-do basin hopping with proper tracking)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(n_bh):
        if time.time() > t4:
            break
        cool = max(1.0 - 0.9 * (it / float(n_bh)), 0.05)
        r = np.random.rand()
        if r < 0.35:
            step = 0.01 + 0.08 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.65:
            shift = int(np.random.uniform(-0.03, 0.03) * n)
            h_p = 0.7 * np.roll(work_h, shift) + 0.3 * work_h
        else:
            mask = np.random.rand(n) < 0.25
            h_p = work_h.copy()
            h_p[mask] += (0.02 + 0.06 * cool) * np.random.randn(int(mask.sum()))
        h_p = project_to_sum(h_p, n / 2.0)
        try:
            hc = slsqp_grad(h_p, int(500 + 3000 * cool), n, maxiter=250, ftol=1e-13)
            hc, c5v = coord_descent(hc, n, time.time() + 1.2, max_sweeps=8)
            temp = 0.003 * cool + 0.00005
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h, work_c5 = hc, c5v
                if c5v < best_c5:
                    best_h, best_c5 = hc.copy(), c5v
        except Exception:
            pass
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Multi-resolution (8% budget)
    t5 = t0 + budget_s * 0.91
    for n_try in [192, 256, 288]:
        if n_try == n or time.time() > t5:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2.0)
        try:
            hc = slsqp_grad(h_i, 2000, n_try, maxiter=400, ftol=1e-14)
            hc, c5v = coord_descent(hc, n_try, time.time() + 3.0, max_sweeps=30)
            if c5v < best_c5:
                best_h, best_c5, n = hc, c5v, n_try
        except Exception:
            pass

    # Phase 6: Final polish (9% budget)
    t6 = t0 + budget_s * 0.995
    best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=1000)
    try:
        hc = slsqp_grad(best_h, 100000, n, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v
    except Exception:
        pass
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.998, max_sweeps=200)

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
