import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, T):
    m = np.max(corr)
    # Shift for numerical stability
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / T

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    return smooth_max(corr, T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    # p has length 2n-1
    # grad[i] = dx * sum_k p[k] * dcorr[k]/dh[i]
    # dcorr[k]/dh[i] = (1-h[i+lag])*1[i+lag in range] - h[i-lag]*1[i-lag in range]
    # where lag = k-(n-1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # h[i] contributes (1-h[i+lag]) for i in [0, n-1-lag]
            # h[i] contributes -h[i-lag] for i in [lag, n-1]
            if n - 1 - lag >= 0:
                grad[:n - lag] += pk * (1.0 - h[lag:n])
            if lag < n:
                grad[lag:] += pk * (-h[:n - lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += pk * (-h[nl:])
    return grad * dx

def slsqp_smooth(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_max(compute_corr(x), T) * (2.0/n)
    def obj_grad(x):
        return smooth_grad(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', jac=obj_grad, bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smax(corr, T):
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(sh))) / T

def smax_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    ex = np.exp(sh)
    p = ex / ex.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n-lag] += pk * (1.0 - h[lag:n])
            grad[lag:] -= pk * h[:n-lag]
        else:
            nl = -lag
            grad[nl:] += pk * (1.0 - h[:n-nl])
            grad[:n-nl] -= pk * h[nl:]
    return grad * dx

def lbfgsb_solve(h, T, n, mu, maxiter=500):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        return smax(c, T) * dx + mu * (x.sum() - target)**2
    def jac(x):
        g = smax_grad(x, T, n)
        g += 2.0 * mu * (x.sum() - target)
        return g
    res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 2000})
    return np.clip(res.x, 0, 1)

def coord_h(i, h, corr, n):
    # Compute B_k = (1-h[i+lag])*[valid] - h[i-lag]*[valid] for this i
    B = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = k - (n-1)
        b = 0.0
        if 0 <= i+lag < n:
            b += (1.0 - h[i+lag])
        if 0 <= i-lag < n:
            b -= h[i-lag]
        B[k] = b
    return B

def coord_solve(corr, B, i, h, n, mu=0.0):
    # f(t) = max_k (corr[k] + t*B[k]) * dx [+ penalty]
    dx = 2.0 / n
    def f1d(t):
        c = corr + t * B
        val = float(np.max(c)) * dx
        if mu > 0:
            s = h.sum() - h[i] + t  # new sum
            val += mu * (s - n/2.0)**2
        return val
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500, mu=10.0):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            B = coord_h(i, best_h, n)
            # corr with h[i]=0:
            corr_base = corr - old * B  # since corr = corr_base + old*B
            t_new = coord_solve(corr_base, B, i, best_h, n, mu)
            if abs(t_new - old) < 1e-10:
                continue
            # Update corr
            corr += (t_new - old) * B
            best_h[i] = t_new
            # Check improvement (every few updates to save time)
            if i % 10 == 0:
                c5_try = float(np.max(corr)) * dx
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    improved = True
        # Full evaluation at end of sweep
        best_h = np.clip(best_h, 0, 1)
        c5_end = compute_c5(best_h)
        if c5_end < best_c5:
            best_c5 = c5_end
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        else:
            # Recompute corr to avoid drift
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        if not improved and sweep > 10:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smax(corr, T):
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(sh))) / T

def grad_smax(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    ex = np.exp(sh)
    p = ex / ex.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += pk * (1.0 - h[lag:n])
            grad[lag:] -= pk * h[:n - lag]
        else:
            nl = -lag
            grad[nl:] += pk * (1.0 - h[:n - nl])
            grad[:n - nl] -= pk * h[nl:]
    return grad * dx

def lbfgsb_solve(h, T, n, mu, maxiter, t_end):
    target = n / 2.0
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full')
        return smax(c, T) * dx + mu * (x.sum() - target) ** 2
    def jac(x):
        g = grad_smax(x, T, n)
        return g + 2.0 * mu * (x.sum() - target)
    res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-11})
    if time.time() > t_end:
        return None
    return np.clip(res.x, 0, 1)

def coord_h_coeffs(i, h, n):
    B = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        k = n - 1 + lag
        b = 0.0
        if 0 <= i + lag < n:
            b += (1.0 - h[i + lag])
        if 0 <= i - lag < n:
            b -= h[i - lag]
        B[k] = b
    return B

def coord_descent(h, n, t_end, max_sweeps=300, mu=5.0):
    dx = 2.0 / n
    best_h = proj(h.copy(), n)
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    target = n / 2.0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            old = best_h[i]
            B = coord_h_coeffs(i, best_h, n)
            corr_base = corr - old * B
            sum_other = best_h.sum() - old
            def f1d(t):
                c = corr_base + t * B
                val = float(np.max(c)) * dx
                if mu > 0:
                    val += mu * (sum_other + t - target) ** 2
                return val
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            t_new = res.x
            if abs(t_new - old) < 1e-10:
                continue
            corr += (t_new - old) * B
            best_h[i] = t_new
            improved = True
        best_h = np.clip(best_h, 0, 1)
        c5_end = compute_c5(best_h)
        if c5_end < best_c5:
            best_c5 = c5_end
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        if not improved:
            break
    return best_h, best_c5

def basin_step(h, n, t_sub, rng):
    perturb = rng.choice(['noise', 'swap', 'block'], p=[0.4, 0.35, 0.25])
    if perturb == 'noise':
        h_p = h + 0.08 * rng.randn(n)
    elif perturb == 'swap':
        h_p = h.copy()
        n_swaps = rng.randint(2, 6)
        for _ in range(n_swaps):
            i, j = rng.choice(n, 2, replace=False)
            h_p[i], h_p[j] = h_p[j], h_p[i]
    else:
        h_p = h.copy()
        s = rng.randint(0, n - 10)
        length = rng.randint(5, 20)
        length = min(length, n - s)
        h_p[s:s + length] = 1.0 - h_p[s:s + length]
    return h_p

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    n = len(best_h)

    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B from initial
    for T in [50, 300, 1500, 8000]:

        res = lbfgsb_solve(best_h, T, n, mu=50.0, maxiter=300, t_end=t0 + budget_s * 0.07)
        if res is not None:

            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 2: Coordinate descent (long)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.22, max_sweeps=400, mu=5.0)

    # Phase 3: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(200):

            break
        cool = 1.0 - 0.9 * (it / 200.0)
        h_pert = basin_step(work_h, n, t0, rng)
        h_pert = proj(h_pert, n)
        res = lbfgsb_solve(h_pert, 1000, n, mu=30.0, maxiter=150, t_end=t0 + budget_s * 0.63)

        temp = 0.004 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):

            work_c5 = c5v

        best_h = work_h.copy()

    # Phase 4: Coordinate descent again on best
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.72, max_sweeps=300, mu=3.0)

                    best_h = hc
                    n = n_try
        if time.time() > t0 + budget_s * 0.89:
            break
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=200, mu=3.0)

    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
