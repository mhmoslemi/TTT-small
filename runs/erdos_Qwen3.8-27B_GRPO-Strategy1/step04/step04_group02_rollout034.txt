import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    # Numerically stable log-sum-exp
    shifted_max = np.max(shifted)
    val = m + shifted_max / T + np.log(np.sum(np.exp(shifted - shifted_max))) / T
    return val * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    shifted_max = np.max(shifted)
    exp_shifted = np.exp(shifted - shifted_max)
    w = exp_shifted / exp_shifted.sum()  # softmax weights, sum to 1
    val = (m + shifted_max / T + np.log(exp_shifted.sum()) / T) * dx
    
    # Gradient: sum_k w[k] * d(corr[k])/d(h[i]) * dx
    lags = np.arange(-(n-1), n)  # length 2n-1
    part1 = np.zeros(n)
    part2 = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        lag = int(k) - (n - 1)
        if lag >= 0:
            part1[:n - lag] += wk * (1.0 - h[lag:])
            part2[lag:] += wk * (-h[:n - lag])
        else:
