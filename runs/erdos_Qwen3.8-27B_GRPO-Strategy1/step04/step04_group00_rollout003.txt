import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def eval_1d(hi, corr_base, coefs, dx):
    return float(np.max(corr_base + hi * coefs) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, T, n, maxiter=1000, ftol=1e-15):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    dx = 2.0 / n
    try:
        res = minimize_scalar(eval_1d, args=(corr_base, coefs, dx),
                              bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def binarize_push(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(200):
        if time.time() > t_end:
            break
        idx = np.random.randint(n)
        if best_h[idx] < 0.15 or best_h[idx] > 0.85:
            continue
        flip_to = 0.0 if best_h[idx] > 0.5 else 1.0
        h_try = best_h.copy()
        h_try[idx] = flip_to
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def make_pert(h, n, cool, rng):
    mod = rng.randint(0, 5)
    if mod == 0:
        bs = rng.randint(4, max(5, n // 8))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.04 + 0.12 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 15))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(4, max(5, n // 8))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.08 + 0.25 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.1 * cool * rng.standard_normal(half)
        p[half:] = -0.1 * cool * rng.standard_normal(n - half)
    return p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation
    for T in [30, 100, 400, 1600, 6000, 24000, 100000, 400000]:
        if elapsed() > budget_s * 0.10:
            break
        hc = slsqp_step(best_h, T, n, maxiter=1200, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Basin hopping with structured perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 50
    for it in range(n_iter):
        if elapsed() > budget_s * 0.35:
            break
        cool = 1.0 - 0.85 * (it / n_iter)
        pert = make_pert(work_h, n, cool, rng)
        h_pert = project_h(work_h + pert)
        T_bh = 400 + 8000 * cool
        hc = slsqp_step(h_pert, T_bh, n, maxiter=700, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0002
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 3: Coordinate descent + binarize push
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.55, max_sweeps=150)
    best_h, best_c5 = binarize_push(best_h, n, t0 + budget_s * 0.62)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.70, max_sweeps=100)

    # Phase 4: More basin hopping (finer)
    work_h = best_h.copy()
    work_c5 = best_c5

        hc = slsqp_step(h_pert, T_bh, n, maxiter=500, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.92, max_sweeps=100)

    # Phase 6: High-T SLSQP polish
