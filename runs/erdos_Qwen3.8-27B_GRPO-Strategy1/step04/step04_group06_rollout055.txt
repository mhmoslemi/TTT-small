def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:

Let me also think about a completely different approach: **binary search on the threshold**.

Given a target C, we want to find h such that:
- 0 ≤ h[i] ≤ 1 for all i
- sum(h) = n/2
- For all k: sum_j h[j+k]*(1-h[j]) ≤ C*n/2 (i.e., corr[k] ≤ C*n/2)

This is a feasibility problem. For a fixed C, we can try to find a feasible h. If we can't, increase C. The feasible set is defined by:
- Box constraints: 0 ≤ h[i] ≤ 1
- Linear equality: sum(h) = n/2
- Quadratic inequalities: for k=0: sum h[i](1-h[i]) ≤ C*n/2, which is sum h[i] - sum h[i]² ≤ C*n/2, i.e., n/2 - sum h[i]² ≤ C*n/2
- For k≠0: sum h[j+k](1-h[j]) ≤ C*n/2, which is a bilinear (not convex) constraint

The bilinear constraints make this non-convex. So we can't directly use a convex solver.

However, for a FIXED value of some variables, the constraints become linear/quadratic and we can solve. This suggests a block coordinate approach: fix all variables except one, and solve the 1D problem. Which is exactly coordinate descent!

So coordinate descent is the right approach, and the key is to make it fast and thorough.

def update_corr(corr, h, n, i, old_v, new_v):
    center = n - 1
    d = new_v - old_v
    corr[center] += (new_v - new_v * new_v) - (old_v - old_v * old_v)
    for k in range(1, n):
        val = 0.0
        if k <= i:
            val += 1.0 - h[i - k]
        if i + k < n:
            val -= h[i + k]
        if val != 0.0:
            corr[center + k] += d * val
        val = 0.0
        if i + k < n:
            val += 1.0 - h[i + k]
        if k <= i:
            val -= h[i - k]
        if val != 0.0:
            corr[center - k] += d * val

def coord_descent(h, n, t_end, max_sweeps=10000):
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * 2.0 / n
    center = n - 1
    
    for sweep in range(max_sweeps):
        now = time.time()
        if now > t_end:
            break
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            v = h[i]
            a_vec = build_a_vec(h, n, i)
            c_center = corr[center]
            t_opt = minimize_1d(corr, a_vec, center, c_center, v)
            if abs(t_opt - v) > 1e-8:
                h[i] = t_opt
                update_corr(corr, h, n, i, v, t_opt)
                new_max = np.max(corr)
                if new_max * 2.0 / n < best_c5:
                    best_c5 = new_max * 2.0 / n
                improved = True
        if not improved:
            break
    
    final_c5 = compute_c5(h)
    return h, min(best_c5, final_c5)

def interpolate_h(h_old, n_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project(h_new, n_new)
