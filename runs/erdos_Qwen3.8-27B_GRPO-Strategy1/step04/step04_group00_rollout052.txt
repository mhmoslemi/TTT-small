```python
def min_max_affine(base, coefs):
    # min_{x in [0,1]} max_k (base[k] + x*coefs[k])
    # Returns the optimal x
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = base + mid * coefs
        idx = np.argmax(vals)
        if coefs[idx] > 0:
            hi = mid
        elif coefs[idx] < 0:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) / 2
```

This works because if the active constraint has positive slope, we need to go left (decrease x) to reduce its value. If negative slope, go right.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(corr, T):
    m = np.max(corr)
    shifted = corr - m
    if T < 1e6:
        return m + np.log(np.sum(np.exp(T * shifted))) / T
    top = shifted > shifted.max() - 0.1 / T
    return m + np.log(np.sum(np.exp(T * shifted[top]))) / T

def slsqp_opt(h, T, n, maxiter=1500, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    def eq_con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def min_max_affine_1d(base, coefs):
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        idx = np.argmax(base + mid * coefs)
        c = coefs[idx]
        if c > 1e-15:
            hi = mid
        elif c < -1e-15:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=20000, seed=None):
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    m = 2 * n - 1
    k_range = np.arange(m) - (n - 1)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            valid1 = (i + k_range >= 0) & (i + k_range < n)
            valid2 = (i - k_range >= 0) & (i - k_range < n)
            coefs = np.zeros(m)
            coefs[valid1] += 1.0 - best_h[i + k_range[valid1]]
            coefs[valid2] += -best_h[i - k_range[valid2]]
            base = corr - best_h[i] * coefs
            x_new = min_max_affine_1d(base, coefs)
            delta = x_new - best_h[i]
            if abs(delta) < 1e-12:
                continue
            best_h[i] = x_new
            corr += delta * coefs
            improved = True
        if improved:
            c5_new = compute_c5(best_h)
            if c5_new < best_c5:
                best_c5 = c5_new
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def slsqp_anneal(h, n, T_list, t_end):
    best = h.copy()
    best_c5 = compute_c5(best)
    for T in T_list:
        if time.time() > t_end:
            break
        hc = slsqp_opt(best, T, n, maxiter=2000, ftol=1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best = hc
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = compute_c5(best_h)

    t1 = t0 + budget_s * 0.12

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.18, max_sweeps=5000, seed=seed)

    # Phase 2: Multi-start basin exploration
    t_phase2_end = t0 + budget_s * 0.72
    n_starts = 10
    per_start = (t_phase2_end - time.time()) / n_starts
    for s in range(n_starts):
        t_start = time.time()
        if t_start > t_phase2_end:
            break
        # Generate perturbation
        ptype = s % 4
        if ptype == 0:
            scale = 0.03 + 0.04 * rng.random()
            h_pert = project_h(best_h + scale * rng.standard_normal(n))
        elif ptype == 1:
            freq = rng.integers(1, 12)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.02 + 0.06 * rng.random()
            h_pert = project_h(best_h + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase))
        elif ptype == 2:
            mask = rng.random(n) < 0.25
            h_pert = best_h.copy()
            h_pert[mask] = np.clip(h_pert[mask] + 0.08 * rng.standard_normal(mask.sum()), 0, 1)
            h_pert = project_h(h_pert)
        else:
            # Block swap perturbation
            b = rng.integers(5, 20)
            i0 = rng.integers(0, n - b)
            h_pert = best_h.copy()
            seg1 = h_pert[i0:i0 + b].copy()
            seg2 = h_pert[(i0 + b)//2:(i0 + b)//2 + b].copy()
            h_pert[i0:i0 + b] = np.clip(seg2, 0, 1)
            h_pert = project_h(h_pert)

        # SLSQP on perturbed
        T_use = 1000 + 5000 * rng.random()
        hc = slsqp_opt(h_pert, T_use, n, maxiter=800, ftol=1e-15)
        hc = project_h(hc)
        c5_try = compute_c5(hc)

    # Phase 3: Extended coordinate descent on best

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
