def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n

    #   As first factor at lag k: j = i + k - (n-1), contributes (b-a)*(1-h[j])
    #   As second factor at lag k: j = i - k + (n-1), contributes h[j]*(a-b) = (b-a)*(-h[j])
    coef = np.zeros(2 * n - 1)

    return res.x, f(res.x)  # return optimal value and the max corr at that value

def compute_influence(i, h, n):
    coef = np.zeros(2 * n - 1)
    # For each valid j1 = i + k - (n-1) in [0, n-1]:
    #   k = j1 - i + (n-1), contribution (1-h[j1])
    # For each valid j2 = i - k + (n-1) in [0, n-1]:
    #   k = (n-1) - j2 + i, contribution -h[j2]
    
    # Valid j1: 0 ≤ i + k - (n-1) ≤ n-1 → n-1-i ≤ k ≤ 2n-2-i
    k_lo = n - 1 - i
    k_hi = 2 * n - 2 - i
    k_lo = max(k_lo, 0)
    k_hi = min(k_hi, 2 * n - 2)
    if k_lo <= k_hi:
        js = np.arange(k_lo, k_hi + 1)
        j1s = i + js - (n - 1)
        coef[js] += (1.0 - h[j1s])
    
    # Valid j2: 0 ≤ i - k + (n-1) ≤ n-1 → i-(n-1) ≤ k ≤ i+(n-1)
    k_lo2 = i - (n - 1)
    k_hi2 = i + (n - 1)
    k_lo2 = max(k_lo2, 0)
    k_hi2 = min(k_hi2, 2 * n - 2)
    if k_lo2 <= k_hi2:
        js2 = np.arange(k_lo2, k_hi2 + 1)
        j2s = i - js2 + (n - 1)
        coef[js2] += (-h[j2s])
    
    return coef

For a checkerboard (h = [1,0,1,0,...]):
- Zero lag: 0 (binary)
- Lag 1: h[i]*(1-h[i+1]) = 1*(1-0) = 1 for even i, 0 for odd i. Count ≈ n/2.
- Lag 2: h[i]*(1-h[i+2]) = h[i]*h[i] (since pattern repeats). For binary, this is h[i]. Count = n/2.
- Lag 3: similar to lag 1, count ≈ n/2 - 1.
- Lag 4: similar to lag 2.

So the max is about n/2, giving C₅ ≈ 1. Also bad!

What about a "golden ratio" type pattern? Or a pattern where the 1s are at positions that are spread out?

Actually, I think the key insight is that the problem is equivalent to: find a 0-1 sequence of length n with n/2 ones that minimizes the maximum cross-correlation with its complement.

This is related to "Optimal Arrays" and "Costas arrays" in radar/signal processing, but with a different objective.

Since there are 180 ones and 180 zeros, the "random" expectation for corr at any lag is about (n/2)*(n/2)/n ≈ n/4 = 45. So the optimal is significantly below the random expectation, which makes sense.

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_influence(i, h, n):
    m = 2 * n - 1
    coef = np.zeros(m)
    klo = max(n - 1 - i, 0)
    khi = min(2 * n - 2 - i, m - 1)
    if klo <= khi:
        ks = np.arange(klo, khi + 1)
        partners = i + ks - (n - 1)
        coef[ks] += (1.0 - h[partners])
    klo2 = max(i - (n - 1), 0)
    khi2 = min(i + (n - 1), m - 1)
    if klo2 <= khi2:
        ks2 = np.arange(klo2, khi2 + 1)
        partners2 = i - ks2 + (n - 1)
        coef[ks2] += (-h[partners2])
    return coef

def smooth_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx
