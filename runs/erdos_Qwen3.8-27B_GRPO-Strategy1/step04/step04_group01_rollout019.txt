    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    target = n / 2.0
    vc = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    diff = target - vc.sum()
    if abs(diff) < 1e-14:

        if s > target:

        else:
            lo = mid
    return np.clip(vc + (lo + hi) * 0.5, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def update_corr_after_change(corr, old_h, new_h, n):
    """Update corr after changing all of old_h to new_h. Expensive but correct."""
    return np.correlate(new_h, 1.0 - new_h, mode='full')

def golden_1d(f, a, b, tol=1e-12, max_iter=60):
    """Golden section search for min of f on [a,b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) * 0.5

def fast_coord_descent(h, n, t_end, max_sweeps=10000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = compute_corr(best_h)
    dx = 2.0 / n
    M = 2 * n - 1
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() > t_end:
                return best_h, best_c5
            hj = best_h[j]
            # Compute gradient w.r.t. h[j]
            # g[m] for m = 0,...,M-1, lag = m-(n-1)
            g = np.zeros(M)
            for m in range(M):
                lag = m - (n - 1)
                lp1 = j + lag
                lm1 = j - lag
                if 0 <= lp1 < n:
                    g[m] += 1.0 - best_h[lp1]
                if 0 <= lm1 < n:
                    g[m] -= best_h[lm1]
            
            # For m = n-1 (lag=0), the contribution is different (quadratic)
            # g[n-1] = (1-h[j]) - h[j] = 1-2*h[j] (already computed above)
            # But the actual function is c[n-1] + (v-h[j]) - v^2 + h[j]^2
            # NOT c[n-1] + (v-h[j])*(1-2*h[j])
            
            # Define f(v)
            def f_of_v(v):
                # Linear terms (all m except n-1)
                lin = corr + (v - hj) * g
                lin[n-1] = -1e18  # mask out the quadratic term
                # Quadratic term
                quad = corr[n-1] + (v - hj) - v * v + hj * hj
                return max(np.max(lin), quad)
            
            # Golden section
            v_opt = golden_1d(f_of_v, 0.0, 1.0, tol=1e-12, max_iter=50)
            
            if abs(v_opt - hj) < 1e-13:
                continue
            
            # Update
            new_hj = v_opt
            dh = new_hj - hj
            best_h[j] = new_hj
            # Update corr
            corr += dh * g
            corr[n-1] = best_h.sum() - np.sum(best_h**2)  # recompute the quadratic one
            
            # Check improvement
            c5_new = np.max(corr) * dx
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break
    
    # Final projection
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5
