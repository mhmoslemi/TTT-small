            h_pert = project(best_h + noise, n)
            h_pert, _ = coord_descent_sweep(h_pert, n, order, t_end)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 - 1e-14:
                best_c5 = c5_p
                best_h = h_pert
            else:
                if rnd > 30:
                    break
    return best_h, best_c5

def block_perturb(h, n, strength):
    """Perturb a random contiguous block."""
    blen = np.random.randint(3, max(4, n // 8))
    start = np.random.randint(0, n - blen)
    h_p = h.copy()
    h_p[start:start + blen] += strength * np.random.randn(blen)
    return project(h_p, n)

def swap_perturb(h, n, strength):
    """Decrease one coordinate, increase another (preserves sum approx)."""
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j:
        return h
    h_p = h.copy()
    h_p[i] -= strength
    h_p[j] += strength
    return project(h_p, n)

def perturbation_search(h, n, best_c5, t_end):
    best_h = h.copy()
    attempts = 0
    while time.time() < t_end:
        time_per_attempt = max(1.0, (t_end - time.time()) / 20.0)
        a_end = min(time.time() + time_per_attempt, t_end)
        # Generate diverse perturbation
        r = np.random.random()
        if r < 0.3:
            h_p = block_perturb(best_h, n, 0.08)
        elif r < 0.6:
            h_p = swap_perturb(best_h, n, 0.04)
        elif r < 0.8:
            h_p = project(best_h + 0.05 * np.random.randn(n), n)
        else:
            # Shift a segment
            slen = np.random.randint(5, max(6, n // 5))
            sstart = np.random.randint(0, max(1, n - slen))
            h_p = best_h.copy()
            seg = h_p[sstart:sstart + slen].copy()
            np.random.shuffle(seg)
            h_p[sstart:sstart + slen] = seg
            h_p = project(h_p, n)
        # Coordinate descent polish
        order = np.random.permutation(n)
        h_p, _ = coord_descent_sweep(h_p, n, order, a_end)
        if time.time() > t_end:
            break
        c5_p = compute_c5(h_p)
        attempts += 1
        if c5_p < best_c5 - 1e-14:
            best_c5 = c5_p
            best_h = h_p
    return best_h, best_c5

def smooth_c5_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    m = corr.max()
    # Avoid overflow
    shifted = T * (corr - m)
    max_shifted = shifted.max()
    logsum = m + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / T
    return logsum * dx

def slsqp_refine(h, n, T, maxiter):
    def obj_fn(x):
        return smooth_c5_obj(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_fn, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project(np.clip(res.x, 0, 1), n)

def multi_resolution(h_best, n_best, c5_best, t_end):
    for n_new in [128, 192, 240, 320]:
        if time.time() > t_end:
            break
        if n_new == n_best:
            continue
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_best, period=2.0)
        h_interp = project(h_interp, n_new)
        # Quick CD at new resolution
        order = np.random.permutation(n_new)
        h_try, _ = coord_descent_sweep(h_interp, n_new, order, min(time.time() + 5, t_end))
        c5_try = compute_c5(h_try)
        if c5_try < c5_best:
            # SLSQP polish
            if time.time() < t_end - 10:
                try:
                    h_try2 = slsqp_refine(h_try, n_new, 200, 300)
                    c5_try2 = compute_c5(h_try2)
                    if c5_try2 < c5_try:
                        h_try = h_try2
                        c5_try = c5_try2
                except Exception:
                    pass
            c5_best = c5_try
            h_best = h_try
            n_best = n_new
    return h_best, c5_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.2 * np.sin(2 * np.pi * x) + 0.05 * np.sin(4 * np.pi * x), 0, 1)

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Extended coordinate descent
    cd1_end = t0 + budget_s * 0.30
    best_h, best_c5 = iterated_cd(best_h, n, cd1_end, max_rounds=800)

    # Phase 2: SLSQP with moderate T
    if time.time() < t0 + budget_s * 0.38:
        for T in [100, 500, 2000]:
            if time.time() > t0 + budget_s * 0.38:
                break
            try:
                hc = slsqp_refine(best_h, n, T, 500)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 3: Perturbation-based search (main exploration)
    p_end = min(t0 + budget_s * 0.65, t_end - 30)
    best_h, best_c5 = perturbation_search(best_h, n, best_c5, p_end)

    # Phase 4: Multi-resolution
    mr_end = min(t0 + budget_s * 0.85, t_end - 20)
    best_h, best_c5, n = multi_resolution(best_h, len(best_h), best_c5, mr_end)

    # Phase 5: Final high-T SLSQP
    if time.time() < t_end - 15:
        for T in [5000, 30000]:
            if time.time() > t_end - 10:
                break
            try:
                hc = slsqp_refine(best_h, n, T, 800)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 6: Final coordinate descent
    best_h, best_c5 = iterated_cd(best_h, n, max(t0 + 5, t_end - 3), max_rounds=300)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
