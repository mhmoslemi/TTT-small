def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14:
            return h
        h -= h * (abs(diff) / tot)
    np.clip(h, 0.0, 1.0, out=h)
    return h

def coord_opt(h, n, i):
    hb = h.copy()
    hb[i] = 0.0
    cb = np.correlate(hb, 1.0 - hb, mode='full')
    k = np.arange(2 * n - 1)

    j2 = i - k + n - 1

    n = len(best_h)
    target = n / 2.0
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: Thorough coordinate descent
    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = cd_sweeps(best_h, n, target, t1, 5000)

    # Phase 2: Coarse basin hopping
    t2 = t0 + budget_s * 0.42
    best_h, best_c5 = basin_hop(best_h, best_c5, n, target, t2, 120, 60, rng)

    # Phase 3: Coordinate descent
    t3 = t0 + budget_s * 0.52
    best_h, best_c5 = cd_sweeps(best_h, n, target, t3, 5000)

    # Phase 4: Fine basin hopping with smaller steps
    t4 = t0 + budget_s * 0.74
    best_h, best_c5 = basin_hop(best_h, best_c5, n, target, t4, 200, 40, rng)

    # Phase 5: Try higher resolution
    t5 = t0 + budget_s * 0.87
    if time.time() < t5 and t5 - time.time() > 30:
        n2 = n + 40
        x1 = np.arange(n) * (2.0 / n)
        x2 = np.arange(n2) * (2.0 / n2)
        h2 = np.interp(x2, x1, best_h)
        h2 = project_h(h2, n2 / 2.0)
        h2, c5_2 = cd_sweeps(h2, n2, n2 / 2.0, t5, 3000)
        if c5_2 < best_c5:
            best_h, best_c5, n, target = h2, c5_2, n2, n2 / 2.0

    # Phase 6: Final coordinate descent polish
    best_h, best_c5 = cd_sweeps(best_h, n, target, t0 + budget_s * 0.97, 8000)

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
