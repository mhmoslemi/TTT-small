import cvxpy as cp
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    vec = np.array(h, dtype=float)
    vc = np.clip(vec, 0.0, 1.0)
    diff = vc.sum() - target
    if abs(diff) < 1e-14:
        return vc
    # Bisection for shift
    lo = -max(np.max(vec), 1.0) - 1.0
    hi = 1.0 - max(np.min(vec), 0.0) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(vec + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid

def build_A(v, n):
    """A[k_idx, j] = v[j+k] if valid, else 0. overlap[k] = A[k,:] @ h"""
    A = np.zeros((2*n-1, n))
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)
        j_start = max(0, -k)
        j_end = min(n, n-k)
        if j_start < j_end:
            A[k_idx, j_start:j_end] = v[j_start+k:j_end+k]
    return A

def lp_alternate(h, n, max_iter=200):

        best_c5 = c5_lp
        best_h = h_lp

    # Phase 2: SLSQP smooth polish (10%)
    t2_end = t0 + budget_s * 0.40
    for T in [200, 1000, 5000, 20000]:
        if time.time() > t2_end:
            break
        hc = slsqp_smooth(best_h.copy(), T, n, maxiter=200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Multiple LP restarts with perturbations (40%)
    t3_end = t0 + budget_s * 0.80
    n_restarts = 0
    while time.time() < t3_end:
        n_restarts += 1
        # Generate perturbation
        scale = 0.05 + 0.10 * np.exp(-0.05 * n_restarts)
        h_pert = best_h + scale * np.random.randn(n)
        h_pert = project(h_pert, n)
        # LP from perturbed start
        t_sub = min(t3_end, time.time() + budget_s * 0.025)
        h_rp, c5_rp = run_lp_alternate(h_pert, n, t_sub, max_iter=150, reg=1e-5)
        # Quick SLSQP polish
        hc = slsqp_smooth(h_rp, 5000, n, maxiter=100)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Coordinate descent polish (10%)
    t4_end = t0 + budget_s * 0.92
    best_h, best_c5 = coord_descent(best_h, n, t4_end, max_sweeps=300)

    # Phase 5: Final SLSQP + CD (8%)
    t5_end = t0 + budget_s * 0.99
    for T in [10000, 50000, 200000]:
        if time.time() > t5_end:
            break
        hc = slsqp_smooth(best_h.copy(), T, n, maxiter=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if time.time() < t5_end:
        best_h, best_c5 = coord_descent(best_h, n, t5_end, max_sweeps=100)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
