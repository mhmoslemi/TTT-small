# Helper functions (top-level, no closures)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project(h, n):
    """Project to {0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    vec = np.asarray(h, dtype=np.float64)
    vc = np.clip(vec, 0.0, 1.0)
    diff = target - vc.sum()
    if abs(diff) < 1e-12:
        return vc
    # Bisection for shift
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.clip(vec + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + 0.5*(lo+hi), 0.0, 1.0)

def smooth_obj(h, T, k):
    """Smooth max via log-sum-exp on top-k lags"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    if k < len(corr):
        idx = np.argpartition(-corr, k)[:k]
        corr = corr[idx]
    m = float(np.max(corr))
    return (m + np.log(np.sum(np.exp(T*(corr - m))))/T) * (2.0/n)

def slsqp_run(h, T, n, maxiter=400, k=6):
    """SLSQP with smooth top-k objective"""
    try:
        def f(x): return smooth_obj(x, T, k)
        def g(x): return x.sum() - n/2.0
        res = minimize(f, h, method='SLSQP', bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':g}],

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    target = n / 2.0
    vec = np.asarray(h, dtype=np.float64)
    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + 0.5 * (lo + hi), 0.0, 1.0)

def smooth_obj(h, T, k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    if k < len(corr):
        idx = np.argpartition(-corr, k)[:k]
        corr = corr[idx]
    m = float(np.max(corr))
    return (m + float(np.log(np.sum(np.exp(T * (corr - m)))) / T)) * (2.0 / n)

def slsqp_run(h, T, n, maxiter=400, k=5):

            return smooth_obj(x, T, k)

            return x.sum() - n / 2.0

                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(res.x, n)

        return project(h, n)
