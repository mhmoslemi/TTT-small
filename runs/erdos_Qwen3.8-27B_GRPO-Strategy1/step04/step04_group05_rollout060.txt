import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    return h

def steepest_binary(h, N, max_sweeps, deadline):
    h = h.astype(float).copy()
    center = N - 1
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        d_lags = np.arange(2 * N - 1) - center
        p_idx = np.arange(N)[:, None]
        d = d_lags[None, :]
        pd = p_idx + d
        pmd = p_idx - d
        vpd = (pd >= 0) & (pd < N)
        vpmd = (pmd >= 0) & (pmd < N)
        hpd = np.where(vpd, h[np.clip(pd, 0, N - 1)], 0.0)
        hpmd = np.where(vpmd, h[np.clip(pmd, 0, N - 1)], 0.0)
        D = -(1.0 - hpd) * vpd.astype(float) + hpmd * vpmd.astype(float)
        best_peak = P
        best_pq = None
        for p in ones:
            base = corr - D[p]
            cand = base[None, :] + D[zeros, :]
            new_peaks = cand.max(axis=1)
            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_peak:
                best_peak = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def cont_grad(corr, h, N, T):
    center = N - 1
    c_shift = corr - np.max(corr)
    w = np.exp(T * c_shift)
    w_sum = w.sum() + 1e-30
    w = w / w_sum
    thresh = w.max() * 0.005
    active = np.where(w > thresh)[0]
    grad = np.zeros(N)
    for ki in active:
        wk = w[ki]
        k = ki - center
        if k >= 0:
            if N - k > 0:
                grad[:N - k] += wk * (1.0 - h[k:])
            if k < N:
                grad[k:] -= wk * h[:N - k]
        else:
            m_ = -k
            if N - m_ > 0:
                grad[m_:] += wk * (1.0 - h[:N - m_])
                grad[:N - m_] -= wk * h[m_:]
    return grad

def continuous_opt(h, N, deadline, T0=60.0, T1=8.0, lr0=0.03, decay=0.9985):
    target = N / 2.0
    h = h.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    total_steps_est = max(1, int((deadline - time.time()) * 30))
    it = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(np.max(corr))
        if peak < best_peak - 1e-12:
            best_peak = peak
            best_h = h.copy()
        it += 1
        frac = min(1.0, it / total_steps_est)
        T = T0 * (T1 / T0) ** frac
        grad = cont_grad(corr, h, N, T)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        lr = lr0 * decay ** it
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_peak * (2.0 / N)
