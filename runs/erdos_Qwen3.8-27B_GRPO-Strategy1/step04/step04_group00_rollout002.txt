from scipy.optimize import minimize_scalar, minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-12:
            h = h + room * (diff / tot)
    else:
        tot = h.sum()
        if tot > 1e-12:
            scale = max(0.0, 1.0 + diff / tot)
            h = h * scale
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_sweep(h, corr, n, dx, t_check, max_iter=1):
    """Do coordinate descent sweeps. Returns h, corr, max_corr."""
    improved_any = False
    for _ in range(max_iter):
        if time.time() > t_check:
            break
        changed = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_check:
                return h, corr, np.max(corr), improved_any
            m = np.arange(-(n-1), n)
            jp = i + m
            jm = i - m
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            coef = np.zeros(2*n-1)
            if vp.any():
                coef[vp] += (1.0 - h[np.clip(jp[vp], 0, n-1)])
            if vm.any():
                coef[vm] += (-h[np.clip(jm[vm], 0, n-1)])
            
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            
            def obj1d(d):
                return np.max(corr + d * coef)
            
            res = minimize_scalar(obj1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-13})
            delta = res.x
            if abs(delta) < 1e-13:
                continue
            corr += delta * coef
            h[i] = min(max(h[i] + delta, 0.0), 1.0)
            changed = True
        if changed:
            improved_any = True
        else:
            break
    # Rescale to fix sum constraint
    h = project_h(h)
    corr = get_corr(h)
    return h, corr, np.max(corr), improved_any

    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-12:
            h = h + room * (diff / tot)
    else:
        tot = h.sum()
        if tot > 1e-12:
            h = h * max(0.0, 1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def obj1d_wrapper(corr, coef, d):
    return np.max(corr + d * coef)

def coord_sweep(h, corr, n, t_check, max_sweeps=1):
    dx = 2.0 / n
    stopped = False
    for _ in range(max_sweeps):
        if time.time() > t_check:
            break
        changed = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_check:
                stopped = True
                break
            m = np.arange(-(n - 1), n)
            jp = i + m
            jm = i - m
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            coef = np.zeros(2 * n - 1)
            if vp.any():
                coef[vp] += (1.0 - h[np.clip(jp[vp], 0, n - 1)])
            if vm.any():
                coef[vm] += (-h[np.clip(jm[vm], 0, n - 1)])
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-15:
                continue
            res = minimize_scalar(obj1d_wrapper, args=(corr, coef),
                                  bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-14})
            delta = res.x
            if abs(delta) < 1e-14:
                continue
            corr += delta * coef
            h[i] = min(max(h[i] + delta, 0.0), 1.0)
            changed = True
        if not changed:
            break
    h = project_h(h)
    corr = get_corr(h)
    return h, corr, stopped

def slsqp_polish(h, n, T, maxiter):
    dx = 2.0 / n
    def obj_func(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    def con_func(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj_func, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_func}],
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        perm = np.random.permutation(n_init)
        best_h[perm[:n_init // 2]] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    best_corr = get_corr(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Try diverse initializations (15%)
    t1 = t0 + budget_s * 0.15
    inits = [best_h.copy()]
    for freq in [3, 5, 7, 11]:
        ph = np.random.uniform(0, 2 * np.pi)
        hs = (np.sin(2 * np.pi * freq * np.arange(n) / n + ph) > 0).astype(float)
        inits.append(project_h(hs))
    for _ in range(4):
        hb = np.zeros(n)
        pm = np.random.permutation(n)
        hb[pm[:n // 2]] = 1.0
        inits.append(project_h(hb))
    inits.append(project_h(best_h + 0.08 * np.random.randn(n)))
    for hc in inits:

            break

        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            best_corr = get_corr(best_h)
    best_h, best_corr, _ = coord_sweep(best_h, best_corr, n, t1, max_sweeps=15)
    best_c5 = float(np.max(best_corr) * 2.0 / n)

        elif mode == 2:

            h_p = work_h.copy()
            h_p[mask] += (0.05 + 0.1 * cool) * np.random.randn(mask.sum())
            h_p = project_h(h_p)
        else:
            h_p = project_h(work_h + 0.02 * np.random.randn(n))
        if elapsed() > t3:
            break
        corr_p = get_corr(h_p)
        h_p, corr_p, _ = coord_sweep(h_p, corr_p, n,
                                     min(t3, elapsed() + 20), max_sweeps=25)
        c5v = float(np.max(corr_p) * 2.0 / n)
        temp = 0.0015 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = h_p
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_p.copy()
                best_corr = corr_p.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h
        best_corr = get_corr(best_h)

    # Phase 4: SLSQP polish (8%)
    t4 = t0 + budget_s * 0.88
    for T in [3000, 15000, 60000]:
        if elapsed() > t4:
            break
        hc = slsqp_polish(best_h, n, T=T, maxiter=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            best_corr = get_corr(best_h)

    # Phase 5: Final coordinate descent (10%)
    t5 = t0 + budget_s * 0.985
    while elapsed() < t5:
        best_h, best_corr, stopped = coord_sweep(best_h, best_corr, n, t5, max_sweeps=1)
        if stopped:
            break
    best_c5 = float(np.max(best_corr) * 2.0 / n)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
