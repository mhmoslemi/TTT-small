    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_to_sum(h, target):

    s = h.sum()

    if abs(diff) < 1e-15:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:

        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k_star, n):

    idx = np.arange(n)
    g = np.zeros(n)
    m1 = (idx - lag >= 0) & (idx - lag < n)
    g[m1] += 1.0 - h[idx[m1] - lag]
    m2 = (idx + lag >= 0) & (idx + lag < n)
    g[m2] -= h[idx[m2] + lag]

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k_star, n):
    lag = k_star - (n - 1)
    idx = np.arange(n)
    g = np.zeros(n)
    m1 = (idx - lag >= 0) & (idx - lag < n)
    g[m1] += 1.0 - h[idx[m1] - lag]
    m2 = (idx + lag >= 0) & (idx + lag < n)
    g[m2] -= h[idx[m2] + lag]
    g -= g.mean()
    return g

def active_descent(h, t_end, max_iters=80000):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c5 = float(np.max(corr) * 2.0 / n)
    alphas = np.array([0.1, 0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005])
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        k_star = int(np.argmax(corr))
        g = grad_corr_k(h, k_star, n)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn
        best_f = float(np.max(corr))
        improved = False
        for a in alphas:
            h_t = np.clip(h - a * g, 0, 1)
            h_t = project_to_sum(h_t, n / 2.0)
            f_t = float(np.max(np.correlate(h_t, 1.0 - h_t, mode='full')))
            if f_t < best_f - 5e-12:
                h, corr = h_t, np.correlate(h_t, 1.0 - h_t, mode='full')
                best_f = f_t
                c5 = best_f * 2.0 / n
                if c5 < best_c5:
                    best_c5 = c5
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def coord_descent(h, n, t_end, n_sweeps=300):
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    best_c5 = float(np.max(corr) * dx)
    lag_arr = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            m1 = (i - lag_arr >= 0) & (i - lag_arr < n)
            m2 = (i + lag_arr >= 0) & (i + lag_arr < n)
            coef = np.zeros(2 * n - 1)
            coef[m1] = 1.0 - best_h[i - lag_arr[m1]]
            coef[m2] -= best_h[i + lag_arr[m2]]
            lo, hi = -h_old, 1.0 - h_old

                continue
