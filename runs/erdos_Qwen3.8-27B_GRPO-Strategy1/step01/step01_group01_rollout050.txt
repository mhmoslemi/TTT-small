    s = t * (corr - mx)

    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mx = np.max(corr)

    #             -h[j-l] valid for j in [0, n-1+l], contributes -h[-l:]

    for t_val in [800, 1500, 3000]:
        if time.time() > dl3 - 20:
            break
        h = proj_grad(h, t_val, max_iter=200, alpha0=0.3)
        c = compute_c5(project_h(h, n))
        if c < c_best:
            c_best = c
            h_best = project_h(h, n).copy()

    # Phase 4: Final SLSQP polish
    if time.time() < t0 + budget_s - 10:
        for t_val in [2000, 5000]:
            if time.time() > t0 + budget_s - 5:
                break
            try:
                r = minimize(obj_slsqp, h_best, args=(t_val,), method='SLSQP',
                             bounds=bnds, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-18})
                hc = project_h(np.clip(r.x, 0.0, 1.0), n)
                c5 = compute_c5(hc)
                if c5 < c_best:
                    c_best = c5
                    h_best = hc.copy()
            except Exception:
                pass

    # Phase 5: Final tiny projected gradient step
    if time.time() < t0 + budget_s - 3:
        h = proj_grad(h_best.copy(), 5000, max_iter=50, alpha0=0.1)
        c = compute_c5(project_h(h, n))
        if c < c_best:
            c_best = c
            h_best = project_h(h, n).copy()

    h_best = project_h(h_best, n)
    c_final = compute_c5(h_best)
    return h_best, c_final, n
