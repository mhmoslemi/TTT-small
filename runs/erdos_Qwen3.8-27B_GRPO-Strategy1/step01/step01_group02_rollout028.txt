import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:] += 1.0 - h[:N - m]
            g[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            g[l:] += 1.0 - h[:N - l]
            g[:N - l] -= h[l:]
    return g

def optimize_adam(h, N, target, dx, t0, budget, rng, lr0=0.05, tau0=20.0):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr0
    tau = tau0
    b1, b2, eps = 0.85, 0.999, 1e-8
    mb = np.zeros(N)
    vb = np.zeros(N)
    
    for it in range(8000):
        if time.time() - t0 > budget - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        
        # Smooth max weights
        sc = tau * corr
        sc = sc - sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-30:
            break
        w /= ws
        
        # Gradient: only top lags
        thresh = max(1.0 / (N * (2*N-1)), 0.002 / N)
        active = np.where(w > thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        grad = np.zeros(N)
        for idx in active:
            lag = int(idx) - (N - 1)
            grad += w[idx] * grad_lag(h, lag, N)
        
        # Adam
        mb = b1 * mb + (1.0 - b1) * grad
        vb = b2 * vb + (1.0 - b2) * grad * grad
        mh = mb / (1.0 - b1 ** (it + 1))
        vh = vb / (1.0 - b2 ** (it + 1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        
        # Schedule

            lr *= 0.82
            tau *= 2.0
        if lr < 1e-9:

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:] += 1.0 - h[:N - m]
            g[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            g[l:] += 1.0 - h[:N - l]
            g[:N - l] -= h[l:]
    return g

def optimize_adam(h, N, target, dx, t0, budget, rng, lr0=0.04, tau0=4.0, max_it=6000):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr0
    tau = tau0
    b1, b2, eps = 0.85, 0.999, 1e-8
    mb = np.zeros(N)
    vb = np.zeros(N)

    for it in range(max_it):
        if time.time() - t0 > budget - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()

        sc = tau * corr
        sc = sc - sc.max()
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-30:
            break
        w /= ws

        thresh = max(1.0 / ((2 * N - 1) * N), 0.003 / N)
        active = np.where(w > thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(N)
        for idx in active:
            lag = int(idx) - (N - 1)
            grad += w[idx] * grad_lag(h, lag, N)

        mb = b1 * mb + (1.0 - b1) * grad
        vb = b2 * vb + (1.0 - b2) * grad * grad
        mh = mb / (1.0 - b1 ** (it + 1))
        vh = vb / (1.0 - b2 ** (it + 1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)

        if (it + 1) % 250 == 0:
            lr *= 0.80
            tau *= 1.7
        if lr < 1e-9:
            break

    return best_h, best_c5

            best_h = h_try
