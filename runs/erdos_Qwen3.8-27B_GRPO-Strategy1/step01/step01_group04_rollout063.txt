    return sm * dx + penalty * (integral - 1.0) ** 2

def worst_shift_obj(h, k_star, n, smooth_temp, weight):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    shifted = c - cmax
    sm = cmax + smooth_temp * np.log(np.sum(np.exp(shifted / smooth_temp)))
    integral = np.sum(h) * dx
    return (sm + weight * c[k_star]) * dx + 500.0 * (integral - 1.0) ** 2

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

def run_lbfgs(h_in, temp, pen, maxit, t0, tlim):
    if time.time() - t0 > tlim:
        return h_in
    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    mf = maxit * (2 * n + 1) * 3
    res = minimize(smooth_c5_obj, h_in, args=(temp, pen),
                  method='L-BFGS-B', bounds=bounds,
                  options={'maxiter': maxit, 'ftol': 1e-13, 'gtol': 1e-9, 'maxfun': mf})
    nh = np.clip(res.x, 0.0, 1.0)
    return repair_h(nh)

def perturb_h(h, mode, rng):
    n = len(h)
    if mode == 0:
        k = rng.randint(1, max(2, n // 12))
        h2 = np.roll(h, k)
    elif mode == 1:
        h2 = h[::-1].copy()
    elif mode == 2:
        h2 = np.clip(h + rng.randn(n) * 0.04, 0.0, 1.0)
    elif mode == 3:
        nblocks = 8
        bs = n // nblocks
        b1, b2 = rng.randint(0, nblocks, 2)
        h2 = h.copy()
        if b1 != b2:
            s1, s2 = b1 * bs, b2 * bs
            tmp = h2[s1:s1 + bs].copy()
            h2[s1:s1 + bs] = h2[s2:s2 + bs]
            h2[s2:s2 + bs] = tmp
    elif mode == 4:
        h2 = np.clip(1.0 - h + rng.randn(n) * 0.02, 0.0, 1.0)
    else:
        h2 = h + rng.randn(n) * 0.03
    return repair_h(h2)

def do_worst_shift_refine(h, n_iters, t0, tlim):
    n = len(h)
    cur = h.copy()
    for _ in range(n_iters):
        if time.time() - t0 > tlim:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        k_star = int(np.argmax(corr))
        bounds = [(0.0, 1.0)] * n
        mf = 200 * (2 * n + 1) * 3
        res = minimize(worst_shift_obj, cur, args=(k_star, n, 0.01, 2.0),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': mf})
        cur = np.clip(res.x, 0.0, 1.0)
        cur = repair_h(cur)
    return cur

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    h0 = repair_h(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: optimize at n=360
    tlim1 = t0 + budget_s * 0.22
    cur = h0.copy()
    sched360 = [
        (0.06, 500, 250),
        (0.015, 2000, 250),
        (0.004, 5000, 250),
        (0.001, 20000, 250),
    ]
    for temp, pen, maxit in sched360:
        cur = run_lbfgs(cur, temp, pen, maxit, t0, tlim1)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cur.copy()

    # Restart at 360 with structured perturbations
    tlim1b = t0 + budget_s * 0.38
    for r in range(4):
        if time.time() > tlim1b:
            break
        hp = perturb_h(best_h, r % 4, rng)
        for temp, pen, maxit in sched360[:2]:
            hp = run_lbfgs(hp, temp, pen, maxit, t0, tlim1b)
        c5 = compute_c5(hp)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hp.copy()

    # Worst-shift refinement at 360
    tlim1c = t0 + budget_s * 0.48
    cur = do_worst_shift_refine(best_h, 6, t0, tlim1c)
    c5 = compute_c5(cur)
    if c5 < best_c5:
        best_c5 = c5
        best_h = cur.copy()

    # One more restart after refinement
    if time.time() < t0 + budget_s * 0.55:
        hp = perturb_h(best_h, 2, rng)
        for temp, pen, maxit in sched360[:2]:
            hp = run_lbfgs(hp, temp, pen, maxit, t0, t0 + budget_s * 0.55)
        c5 = compute_c5(hp)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hp.copy()

    # Phase 2: upscale to 720
    tlim2 = t0 + budget_s * 0.82
    if time.time() < tlim2:
        h720 = interp_upscale(best_h, 2)
        h720 = repair_h(h720)
        cur720 = h720.copy()

        sched720 = [
            (0.04, 500, 200),
            (0.01, 2000, 200),
            (0.003, 5000, 200),
            (0.0008, 15000, 200),
        ]
        for temp, pen, maxit in sched720:
            cur720 = run_lbfgs(cur720, temp, pen, maxit, t0, tlim2)
            c5 = compute_c5(cur720)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cur720.copy()

        # Restart at 720
        tlim2b = t0 + budget_s * 0.93
        for r in range(3):
            if time.time() > tlim2b:
                break
            hp = perturb_h(best_h, (r + 1) % 5, rng)
            for temp, pen, maxit in sched720[:2]:
                hp = run_lbfgs(hp, temp, pen, maxit, t0, tlim2b)
            c5 = compute_c5(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()

        # Worst-shift at 720
        tlim2c = t0 + budget_s * 0.98
        cur = do_worst_shift_refine(best_h, 4, t0, tlim2c)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cur.copy()

    # Final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
