        l = -m
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def optimize_topk(h, N, target, dx, t0, budget_s, iters=4000, lr0=0.04):
    best = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    best_h = h.copy()
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8

    for it in range(iters):
        if time.time() - t0 > budget_s - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best:
            best = c5
            best_h = h.copy()

        p = it / iters
        k_lags = 3 + int(4 * max(0, 1.0 - 2.0 * p))
        topk = np.argsort(corr)[-k_lags:]
        grad = np.zeros(N)
        for idx in topk:
            grad += grad_lag(h, int(idx) - (N - 1), N)

        # Small exploration via softmax (fades out)
        if p < 0.6:
            tau = 5.0 + 25.0 * (p / 0.6)
            sc = tau * (corr - corr.max())
            w = np.exp(sc)
            w /= w.sum()
            active = np.where(w > 0.01 / N)[0]
            for idx in active:
                grad += 0.05 * w[idx] * grad_lag(h, int(idx) - (N - 1), N)

        # Binary penalty (encourages 0/1 values)
        bp = 0.015 * p
        grad += bp * (1.0 - 2.0 * h)

        gn = np.linalg.norm(grad)
        if gn > 1e-12:
            grad /= gn

        lr = lr0 * 0.5 * (1.0 + np.cos(np.pi * p)) + 0.002
        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** (it + 1))
        vh = v_buf / (1.0 - b2 ** (it + 1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)

    return best_h, best

def local_search(h, N, target, dx, t0, max_time=30, seed=0):
    rng_ls = np.random.RandomState(seed)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    t_end = time.time() + max_time

    while time.time() < t_end and t0 is not None and time.time() - t0 < 1080:
        i = rng_ls.randint(N)
        j = rng_ls.randint(N)
        if i == j:
            continue
        for di in [0.04, -0.04, 0.08, -0.08, 0.15, -0.15]:
            h_try = best_h.copy()
            h_try[i] += di
            h_try[j] -= di
            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            h_try = project_sum(h_try, target)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    sizes = sorted(set([n0, 2 * n0]))

    for N in sizes:
        if time.time() - t0 > budget_s - 70:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)

        # Build starting points
        h_int = np.interp(xN, x0, best_h)
        h_int = project_sum(h_int, target)
        starts = [h_int.copy()]
        for sh in [1, -1, 2]:
            starts.append(project_sum(np.roll(h_int, sh), target))
        starts.append(project_sum(h_int + rng.randn(N) * 0.04, target))
        starts.append(project_sum(h_int + rng.randn(N) * 0.08, target))

        for si, hs in enumerate(starts):
            if time.time() - t0 > budget_s - 40:
                break
            h_opt, c5_opt = optimize_topk(hs.copy(), N, target, dx, t0, budget_s,
                                          iters=4000, lr0=0.04)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        # Local search polish
        if time.time() - t0 < budget_s - 50:
            h_cur = best_h if len(best_h) == N else project_sum(np.interp(xN, x0, best_h), target)
            h_ls, c5_ls = local_search(h_cur, N, target, dx, t0, max_time=35, seed=seed + N)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls

    # Final polish on the global best
    Nf = len(best_h)
    dxf = 2.0 / Nf
    tf = Nf / 2.0
    if time.time() - t0 < budget_s - 20:
        h_p, c5_p = optimize_topk(best_h.copy(), Nf, tf, dxf, t0, budget_s, iters=2000, lr0=0.015)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p
    if time.time() - t0 < budget_s - 8:
        h_l, c5_l = local_search(best_h, Nf, tf, dxf, t0, max_time=12, seed=seed + 999)
        if c5_l < best_c5:
            best_c5 = c5_l
            best_h = h_l

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
