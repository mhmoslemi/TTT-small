    target = n / 2.0

        rem = budget_s - 15 - (time.time() - t0)
        mi = max(80, int(rem * 4))
        h_new = repair_h(slsqp_opt(best_h, n, 3000, mi), n)
        c5 = compute_c5(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1.0
    best_n = 45

    init_45 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_45 = np.array(initial_h_values, dtype=float).copy()

    if init_45 is not None:
        h45 = repair_h(init_45, 45)
        alloc = int((budget_s - 100) * 0.50)
        h_new, c5_new = optimize_for_n(45, h45, t0, budget_s, alloc)
        if c5_new < best_c5:
            best_c5, best_h, best_n = c5_new, h_new, 45

    n_tries = [44, 50, 36, 54]
    for n_try in n_tries:
        if time.time() - t0 > budget_s - 80:
            break
        for h_init in make_init(n_try, rng):
            if time.time() - t0 > budget_s - 60:
                break
            remaining = budget_s - 60 - (time.time() - t0)
            alloc = max(10, int(remaining * 0.12))
            h_new, c5_new = optimize_for_n(n_try, h_init, t0, budget_s, alloc)
            if c5_new < best_c5:
                best_c5, best_h, best_n = c5_new, h_new, n_try

    if best_h is not None and time.time() - t0 < budget_s - 10:
        n = best_n
        rem = budget_s - 10 - (time.time() - t0)
        if rem > 5:
            mi = max(50, int(rem * 5))
            h_new = repair_h(slsqp_opt(best_h, n, 8000, mi), n)
            c5 = compute_c5(h_new, n)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new
        if time.time() - t0 < budget_s - 5:
            best_h = coord_descend(best_h, n, 4, t0, budget_s)
            best_c5 = compute_c5(best_h, n)

    best_h = repair_h(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
