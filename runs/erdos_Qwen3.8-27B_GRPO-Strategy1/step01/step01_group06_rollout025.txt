import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grad_lag(h, k):
    """Gradient of f_k with respect to h, where f_k = sum h[i+k]*(1-h[i])*dx"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad = (1.0 - 2.0 * h) * dx
    elif k > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
        grad *= dx
    else:
        l = -k
        grad[:n-l] += (1.0 - h[l:])
        grad[l:] -= h[:n-l]
        grad *= dx
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def subgradient_descent(h0, n, T_steps=2000, top_m=5, verbose=False):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for t in range(T_steps):
        corr = compute_corr(h)
        cur_max = np.max(corr)
        
        # Sparsification: use top m lags
        m_eff = min(int(top_m * (1 + t / (T_steps // 3))), len(corr))
        sorted_idx = np.argsort(-corr)
        active = sorted_idx[:m_eff]
        
        # Average gradient over active lags
        # Convert correlation index to lag
        # np.correlate mode='full' index: lag k is at index k + n - 1
        # So lag = idx - (n - 1)
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += compute_grad_lag(h, k)
        grad /= m_eff
        
        # Step size: Polyak-like
        step = 0.5 / (np.linalg.norm(grad) + 1e-8)
        step = min(step, 0.1)  # Cap step size
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        new_val = compute_c5(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_val

def slsqp_polish(h, n, T=500, maxiter=500):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = project_feasible(np.clip(res.x, 0, 1), n)
        return hc
    except:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Subgradient descent with increasing sparsification
    h_cur = best_h.copy()
    for phase_m in [3, 5, 10, 20, 50]:
        if elapsed() > budget_s * 0.4:
            break
        h_cur, val = subgradient_descent(h_cur, n, T_steps=500, top_m=phase_m)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def compute_coef_vec(h, j, n):
    """Coefficient vector: corr changes by dh * coef_vec when h[j] changes by dh"""
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for K in range(-(n - 1), n):
        idx = K + n - 1
        if K >= 0:
            c = 0.0
            if j >= K:
                c += (1.0 - h[j - K])
            if j <= n - 1 - K:
                c -= h[j + K]
        else:
            L = -K
            c = 0.0
            if j <= n - 1 - L:
                c += (1.0 - h[j + L])
            if j >= L:
                c -= h[j - L]
        coef[idx] = c * dx
    return coef

def coord_descent(h, n, max_sweeps=200, grid_pts=40):
    """Coordinate descent: minimize max correlation by cycling through coordinates"""
    h = h.copy()
    corr = compute_corr_array(h)
    best_val = float(np.max(corr))
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            coef = compute_coef_vec(h, j, n)
            # new_corr = corr + (t - old) * coef
            # Find t in [0,1] minimizing max(new_corr)
            best_t = old
            best_max = float(np.max(corr))
            # Grid search
            ts = np.linspace(0.0, 1.0, grid_pts)
            for t in ts:
                if abs(t - old) < 1e-12:
                    continue
                new_corr = corr + (t - old) * coef
                m = float(np.max(new_corr))
                if m < best_max - 1e-14:
                    best_max = m
                    best_t = t
            # Refine around best_t
            if best_t != old:
                lo = max(0.0, best_t - 1.0/grid_pts)
                hi = min(1.0, best_t + 1.0/grid_pts)
                for _ in range(10):
                    m1 = lo + (hi - lo) * 0.382
                    m2 = lo + (hi - lo) * 0.618
                    v1 = float(np.max(corr + (m1 - old) * coef))
                    v2 = float(np.max(corr + (m2 - old) * coef))
                    if v1 < v2:
                        hi = m2
                    else:
                        lo = m1
                best_t = (lo + hi) / 2.0
                best_max = float(np.max(corr + (best_t - old) * coef))
            
            if best_t != old:
                h[j] = best_t
                corr = corr + (best_t - old) * coef
                improved = True
                if best_max < best_val:
                    best_val = best_max
        if not improved:
            break
    # Enforce exact sum constraint
    h = project_feasible(h, n)
    return h, compute_c5(h)

def slsqp_polish(h, n, T=500, maxiter=600):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = project_feasible(np.clip(res.x, 0, 1), n)
        return hc
    except Exception:
        return h

def subgradient_step(h, corr, n, top_m, step_size):
    """One subgradient step using top_m most active lags"""
    sorted_idx = np.argsort(-corr)
    active = sorted_idx[:top_m]
    grad = np.zeros(n)
    for idx in active:
        K = int(idx) - (n - 1)
        # Compute gradient of f_K w.r.t. h
        if K == 0:
            grad += (1.0 - 2.0 * h)
        elif K > 0:
            g = np.zeros(n)
            g[K:] += (1.0 - h[:n-K])
            g[:n-K] -= h[K:]
            grad += g
        else:
            L = -K
            g = np.zeros(n)
            g[:n-L] += (1.0 - h[L:])
            g[L:] -= h[:n-L]
            grad += g
    grad /= top_m
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    # Initialize
    if 'initial_h_values' in dir() and initial_h_values is not None:

    elif 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        best_h = np.array(globals()['initial_h_values'], dtype=float).copy()
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent (fast, exact per-coordinate)
    if elapsed() < budget_s * 0.3:
        h_cd, c5_cd = coord_descent(best_h, n, max_sweeps=300, grid_pts=50)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    # Phase 2: SLSQP polish
    if elapsed() < budget_s * 0.45:
        for T in [200, 800, 3000]:
            if elapsed() > budget_s * 0.45:
                break
            h_pol = slsqp_polish(best_h, n, T=T, maxiter=800)
            c5v = compute_c5(h_pol)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_pol
    
    # Phase 3: More coordinate descent after polish
    if elapsed() < budget_s * 0.55:
        h_cd, c5_cd = coord_descent(best_h, n, max_sweeps=200, grid_pts=40)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    # Phase 4: Subgradient with sparsification
    if elapsed() < budget_s * 0.70:
        h_sg = best_h.copy()
        n_sg = len(h_sg)
        corr_sg = compute_corr_array(h_sg)
        for i in range(5000):
            if elapsed() > budget_s * 0.70:
                break
            m_eff = min(3 + i // 500, len(corr_sg))
            step = 0.02 * (1.0 + 0.5 * np.cos(0.01 * i))
            h_sg = subgradient_step(h_sg, corr_sg, n_sg, m_eff, step)
            corr_sg = compute_corr_array(h_sg)
            c5v = float(np.max(corr_sg))
            if c5v < best_c5:

        # Polish after subgradient
