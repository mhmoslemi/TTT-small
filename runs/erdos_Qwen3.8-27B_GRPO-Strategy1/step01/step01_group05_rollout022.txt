```python
def coordinate_step(h, n, j):
    """Find optimal h[j] given all other coordinates fixed."""
    dx = 2.0 / n
    # Compute correlation as function of h[j]
    # corr[k] = sum_i h[i] * (1-h)[i-k+(n-1)]
    # As function of t = h[j]:
    # corr[k] = A[k] + B[k] * t where
    # A[k] = corr[k] with h[j] set to 0
    # B[k] = corr[k] with h[j] set to 1 minus corr[k] with h[j] set to 0
    
    h_orig = h[j]
    h[j] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[j] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full')
    h[j] = h_orig
    
    A = corr0  # constant part
    B = corr1 - corr0  # linear coefficient
    
    # f(t) = max_k (A[k] + B[k]*t) * dx
    # Find min over t in [0,1]
    
    # Candidate points: 0, 1, and breakpoints
    candidates = [0.0, 1.0]
    n_corr = len(A)
    for i in range(n_corr):
        for k in range(i+1, n_corr):
            if B[k] != B[i]:
                t = (A[i] - A[k]) / (B[k] - B[i])
                if 0.0 < t < 1.0:
                    candidates.append(t)
    
    # Also evaluate f at current value
    candidates.append(h_orig)
    
    best_t = h_orig
    best_f = np.max((A + B * h_orig)) * dx
    for t in candidates:
        t = np.clip(t, 0.0, 1.0)
        f_t = np.max(A + B * t) * dx
        if f_t < best_f - 1e-12:
            best_f = f_t
            best_t = float(t)
    
    return best_t, best_f
```

                    db = B[i] - B[k]
                    if abs(db) > 1e-15:
                        t = (A[k] - A[i]) / db
                        if -0.01 < t < 1.01:
                            cands.append(np.clip(t, 0.0, 1.0))
            
            # Evaluate at candidates
            best_t = h_save
            best_f = np.max(A + B * h_save) * dx
            for t in cands:
                t = float(np.clip(t, 0.0, 1.0))
                f_t = np.max(A + B * t) * dx
                if f_t < best_f - 1e-13:
                    best_f = f_t
                    best_t = t
            
            if abs(best_t - h_save) > 1e-12:
                h[j] = best_t
                # Repair sum
                diff = target - np.sum(h)
                if abs(diff) > 1e-12:
                    # Adjust all free coordinates
                    free_mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                    nf = np.sum(free_mask)
                    if nf == 0:
                        h += diff / n
                        h = np.clip(h, 0.0, 1.0)
                    else:
                        h[free_mask] += diff / nf
                        h = np.clip(h, 0.0, 1.0)
                improved = True
        
        c5_now = compute_c5(h, n)
        if not improved or abs(c5_now - best_c5) < tol:
            best_c5 = min(best_c5, c5_now)
            if not improved:
                break
        best_c5 = min(best_c5, c5_now)
    
    return h, best_c5

def slsqp_optimize(h0, n, bounds, maxiter=300, temp=10):
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    res = minimize(
        smooth_obj, h0, method='SLSQP',
        bounds=bounds, constraints=cons,
        args=(n, temp),
        options={'maxiter': maxiter, 'ftol': 1e-15}
    )
    return res.x
