So for a fixed coordinate j, each c_k is at most linear in h[j] (since h[j] appears at most once in each product term). Actually, h[j] can appear in c_j (lag 0) as h[j]*(1-h[j]), which is quadratic. But for k≠0, it's linear.

So the objective as a function of h[j] is max of (one quadratic + many linear functions). This is a 1D problem that can be solved exactly!

This suggests a **cyclic coordinate descent** approach:
- For j = 0, 1, ..., n-1:
  - Fix all h[i] for i≠j
  - Optimize h[j] to minimize the max correlation
  - (This is a 1D minimization of a max of linear/quadratic functions)

The 1D problem: minimize max_k (a_k * h[j] + b_k * h[j]² + c_k) over h[j] ∈ [0,1]
where a_k, b_k, c_k depend on h[-j].

This can be solved by:
- The function is convex (max of convex functions - wait, max of linear is convex, max of one quadratic and linear is convex if the quadratic has positive coefficient)
- Actually h[j]*(1-h[j]) = h[j] - h[j]², which is concave (negative quadratic)
- So c_0 (lag 0) is concave in h[j], while c_k for k≠0 are linear in h[j]
- Max of concave and linear is... not necessarily convex or concave

def repair_h(h, n):

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)

        else:
            h[free] += diff / nf

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def obj_lse(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    val = np.log(np.sum(np.exp((corr - m) / temp))) * temp + m
    return val * dx


def obj_topk(x, n, temp, k):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(corr)[-k:]
    top = corr[idx]
    m = top[0]
    return float(np.log(np.sum(np.exp((top - m) / temp))) * temp + m)


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            h += diff / len(h)
        else:
            h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_opt(x0, n, temp, maxiter, topk=5):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]

                       constraints=cons, args=(n, temp, topk),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
    else:
        res = minimize(obj_lse, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, temp),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
    return np.clip(res.x, 0.0, 1.0)

                   options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-13})
    return np.clip(res.x, 0.0, 1.0)
