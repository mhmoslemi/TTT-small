    n = len(h)

    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(x):
    return x.sum() - len(x) / 2.0

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t > 1e-13:
                h += r * (d / t)

            if t > 1e-13:
