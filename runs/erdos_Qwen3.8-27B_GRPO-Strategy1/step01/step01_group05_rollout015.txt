import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_objective(x, n, temp, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Log-sum-exp for smooth max
    max_c = np.max(corr)
    smooth_max = (max_c + np.log(np.sum(np.exp((corr - max_c) / temp))) * temp) * dx
    # Penalty for sum constraint
    sum_penalty = penalty * (np.sum(x) - n / 2.0) ** 2
    return smooth_max + sum_penalty

def project_to_feasible(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            # Try harder: spread to nearest boundaries
            continue
        per = diff / nfree
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def lbfgsb_solve(x0, n, temp, penalty, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        smooth_objective, x0,
        method='L-BFGS-B',
        bounds=bounds,
        args=(n, temp, penalty),
        options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10, 'disp': False}
    )
    return res.x
