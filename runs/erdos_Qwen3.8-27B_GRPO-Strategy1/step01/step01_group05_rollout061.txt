        xg = np.linspace(0.0, 2.0, best_n, endpoint=False)
        if trial == 0:
            H = np.fft.fft(best_h - 0.5)
            p0 = np.zeros(1 + 2 * ncoef)
            p0[0] = 0.5
            for k in range(1, ncoef + 1):
                p0[2 * k - 1] = 2.0 * np.real(H[k]) / best_n
                p0[2 * k] = -2.0 * np.imag(H[k]) / best_n
        elif trial == 1:
            p0 = np.concatenate(([0.5], np.random.uniform(-0.06, 0.06, 2 * ncoef)))
        elif trial == 2:
            p0 = np.concatenate(([0.5], np.random.uniform(-0.12, 0.12, 2 * ncoef)))
        else:
            p0 = np.concatenate(([0.5], np.random.uniform(-0.25, 0.25, 2 * ncoef)))

        rem = budget_s - 160 - (time.time() - t_start)
        mi = max(300, min(2000, int(rem * 2.5)))
        p_opt = fourier_nm(p0, best_n, ncoef, mi)
        h_t = fourier_to_h(p_opt, best_n)
        c5_t, _ = compute_c5_full(h_t, best_n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

        # SLSQP polish from Fourier result
        if time.time() - t_start < budget_s - 80:
            h_t = slsqp_smooth(h_t, best_n, bounds, 20, 120)
            h_t = repair_h(h_t, best_n)
            h_t, c5_t = coord_descent(h_t, best_n, t_start, budget_s, 60, 1)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Phase 3: Random restarts with SLSQP + coord descent
    temps_p3 = [8, 30, 80]
    for trial in range(35):
        if time.time() - t_start > budget_s - 60:
            break
        rem = budget_s - 60 - (time.time() - t_start)
        if rem < 12:
            break
        sc = 0.05 * np.exp(-trial / 14.0) + 0.008
        cand = best_h + np.random.uniform(-sc, sc, best_n)
        cand = repair_h(cand, best_n)
        temp = temps_p3[trial % 3]
        mi = min(120, max(25, int(rem * 2.5)))
        h_t = slsqp_smooth(cand, best_n, bounds, temp, mi)
        h_t = repair_h(h_t, best_n)
        h_t, c5_t = coord_descent(h_t, best_n, t_start, budget_s, 50, 1)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 4: Try even n with bang-bang start
    for n_try in [44, 48, 52]:
        if time.time() - t_start > budget_s - 50:
            break
        if n_try == best_n:
            continue
        bb = np.zeros(n_try)
        for i in range(n_try // 2):
            bb[2 * i] = 1.0
        bb = repair_h(bb, n_try)
        bnd = [(0.0, 1.0)] * n_try
        for temp in [15, 60]:
            if time.time() - t_start > budget_s - 30:
                break
            h_t = slsqp_smooth(bb, n_try, bnd, temp, 100)
            h_t = repair_h(h_t, n_try)
            h_t, c5_t = coord_descent(h_t, n_try, t_start, budget_s, 20, 2)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_try
                bounds = bnd

    # Phase 5: Final aggressive polish
    if time.time() - t_start < budget_s - 5:
        rem = max(20, int(budget_s - 5 - (time.time() - t_start)))
        bf = [(0.0, 1.0)] * best_n
        for temp in [5, 40]:
            if time.time() - t_start > budget_s - 3:
                break
            h_t = slsqp_smooth(best_h, best_n, bf, temp, max(50, rem))
            h_t = repair_h(h_t, best_n)
            h_t, c5_t = coord_descent(h_t, best_n, t_start, budget_s, 2, 2)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t

    best_h = repair_h(best_h, best_n)
    final_c5, _ = compute_c5_full(best_h, best_n)
    return best_h, final_c5, best_n
