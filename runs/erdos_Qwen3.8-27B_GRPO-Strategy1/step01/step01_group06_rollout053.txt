    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h = h + room * (diff / total)
    else:
        total = s
        if total > 1e-14:
            h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def corr_gradient(h, lag):
    """Gradient of corr at given lag (positive or negative). Returns array of length n."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        m = lag
        if m < n:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    else:
        m = -lag
        if m < n:
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] -= h[m:]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_c5(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h = h + room * (diff / total)
    else:
        total = s
        if total > 1e-14:
            h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def corr_gradient(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        m = lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    else:
        m = -lag
        if m < n:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    return grad

    best = h.copy()
    best_c5 = compute_c5(best)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_c5}]

            break
        try:

                          bounds=bnds, constraints=cons,

            hc = project_feasible(np.clip(res.x, 0, 1), n)

                best = hc

def subgrad_iters(h, n, max_iters, t_limit, top_k=3):
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = 0.08
    momentum = np.zeros(n)
    beta = 0.7
    for it in range(max_iters):

            break
        corr = np.correlate(best, 1.0 - best, mode='full')
