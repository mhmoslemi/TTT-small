import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    h = h.copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0, 1)
    if diff > 0:
        order = np.argsort(h)
        for i in order:
            if diff <= 0: break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

            if diff >= 0: break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub

    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)

            grad[k2:n] += w * (-h[0:n-k2]) * dx
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    h = h.copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0, 1)
    if diff > 0:
        order = np.argsort(h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)[::-1]
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0, 1)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    v = np.interp(nx, ox, parent)
    return fix_sum(v, new_n)

def subgrad_step(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    k = k_star - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1.0 - 2.0 * h)
    elif k > 0:
        grad[k:n] += (1.0 - h[0:n - k])
        grad[0:n - k] += (-h[k:n])
    else:
        k2 = -k
        grad[0:n - k2] += (1.0 - h[k2:n])
        grad[k2:n] += (-h[0:n - k2])
    return h - alpha * grad * dx, k

def subgradient_descent(h, n, max_iter, elapsed_func, budget_s):
    h = h.copy()
    cur = compute_c5(h)
    alpha = 0.01
    for it in range(max_iter):
        if budget_s - elapsed_func() < 5:
            break
        h_new, k = subgrad_step(h, alpha, n)
        h_new = fix_sum(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < cur:
            h = h_new
            cur = c_new
            alpha = min(alpha * 1.05, 0.05)
        else:
            alpha *= 0.7
            if alpha < 1e-5:
                break
    return h, cur

def softmax_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    return dx * (m + beta * np.log(s))

def make_eq_fun(n):
    target = n / 2.0
    def eq(x):
        return np.sum(x) - target
    def jac(x):
        return np.ones(n)
    return eq, jac

def coord_descent(h, n, rng, max_steps, elapsed_func, budget_s):
    h = h.copy()
    cur = compute_c5(h)
    for it in range(max_steps):
        if budget_s - elapsed_func() < 3:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if rng.random() < 0.3:
            # Structured perturbation: sinusoidal window
            freq = rng.choice([1, 2, 3, 4, 5])
            amp = rng.uniform(0.005, 0.03)
            idx = np.arange(n)
            wave = amp * np.sin(2.0 * np.pi * freq * idx / n + rng.uniform(0, 2*np.pi))
            h_tr = h + wave
            j2 = (i + 1) % n
            d = (h_tr[i] - h[i])
            h_tr[i] = np.clip(h_tr[i], 0, 1)
            h_tr[j2] = np.clip(h[j2] - d, 0, 1)
        else:
            d = rng.uniform(0.002, 0.05) * rng.choice([-1.0, 1.0])
            h_tr = h.copy()
            h_tr[i] = np.clip(h[i] + d, 0, 1)
            h_tr[j] = np.clip(h[j] - d, 0, 1)
        h_tr = fix_sum(h_tr, n)
        c = compute_c5(h_tr)
        if c < cur - 1e-14:
            h = h_tr
            cur = c
    return h, cur
