import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = t * corr
    s -= np.max(s)
    return (np.max(s) / t + np.log(np.sum(np.exp(s))) / t) * dx

def softmax_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = t * corr
    s -= np.max(s)
    w = np.exp(s)
    w /= np.sum(w)
    lags = np.arange(2 * n - 1) - (n - 1)
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        lag = lags[m]
        if lag >= 0:
            term1[:n-lag] += wm * h[lag:n]
            term2[lag:] += wm * h[:n-lag]
        else:
            la = -lag
            term1[la:] += wm * h[:n-la]
            term2[:n-la] += wm * h[la:n]
    return dx * (1.0 - term1 - term2)

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

    return np.ones((1, len(x)))

        trial = proj(trial, n)
        if compute_c5(trial) < compute_c5(hc):

    return hc

        c5 = compute_c5(hc)
        if c5 > best_c5 + 1e-12:
            hc[idx] = old
            hc = proj(hc, n)
    return hc
