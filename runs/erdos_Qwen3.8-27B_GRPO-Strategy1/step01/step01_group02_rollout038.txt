    """Correct gradient of correlation at given lag w.r.t. h."""
    g = np.zeros(N)

        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)

        if (it + 1) % 250 == 0:
            lr *= 0.87
            tau *= 1.6
        if lr < 1e-9:
            break
        if stuck > 400:
            mask = rng.random(N) < 0.12
            h[mask] += rng.randn(int(np.sum(mask))) * 0.035
            h = project_sum(h, target)
            stuck = 0

    return best_h, best_c5

def optimize_lbfgs(h, N, target, dx, t0, budget_s, tau=4.0, top_k=4):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    def obj(x):
        x2 = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x2, 1.0 - x2, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        sv = tau * (corr[top_idx] - corr[top_idx].max())
        val = (corr[top_idx].max() + np.log(np.exp(sv).sum()) / tau)
        val += 200.0 * (np.sum(x2) - target) ** 2
        return val

    def grad_obj(x):
        x2 = np.clip(x, 0.0, 1.0)
        g = smooth_max_grad(x2, N, tau, top_k)
        g += 400.0 * (np.sum(x2) - target)
        return g

    result = minimize(obj, h, jac=grad_obj, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * N,
                      options={'maxiter': 200, 'maxls': 30, 'ftol': 1e-14})
    h_opt = project_sum(np.clip(result.x, 0.0, 1.0), target)
    c5_opt = compute_c5(h_opt, dx)
    if c5_opt < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5_opt
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    sizes = []
    for s in [n0 // 2, n0, 2 * n0, 3 * n0]:
        if s >= 48 and s // 2 * 2 == s and s not in sizes:
            sizes.append(s)
    sizes.sort()

    for N in sizes:
        if time.time() - t0 > budget_s - 45:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_interp = project_sum(np.interp(xN, x0, h0), target)

        # Adam with corrected gradient
        for r in range(3):
            if time.time() - t0 > budget_s - 30:
                break
            h_try = h_interp.copy()
            if r > 0:
                h_try += rng.randn(N) * (0.03 + 0.01 * r)
                h_try = project_sum(h_try, target)
            h_a, c5_a = optimize_adam(h_try, N, target, dx, t0, budget_s,
                                      lr_init=0.04, top_k=3, tau_init=6.0)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()

        # L-BFGS-B
        if time.time() - t0 < budget_s - 35:
            h_l, c5_l = optimize_lbfgs(best_h if len(best_h) == N else h_interp,
                                       N, target, dx, t0, budget_s, tau=5.0, top_k=3)
            if c5_l < best_c5:
                best_c5 = c5_l
                best_h = h_l.copy()

        # Binary init + optimize
        if time.time() - t0 < budget_s - 50:
            for r in range(2):
                h_bin = np.zeros(N)
                h_bin[rng.choice(N, size=int(target), replace=False)] = 1.0
                h_bin = project_sum(h_bin, target)
                h_b, c5_b = optimize_adam(h_bin, N, target, dx, t0, budget_s,
                                          lr_init=0.03, top_k=2, tau_init=4.0)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()

    # Final polish
    N_best = len(best_h)
    dx_best = 2.0 / N_best
    target_best = N_best / 2.0
    if time.time() - t0 < budget_s - 20:
        h_f, c5_f = optimize_adam(best_h.copy(), N_best, target_best, dx_best,
                                  t0, budget_s, lr_init=0.015, top_k=2, tau_init=10.0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
