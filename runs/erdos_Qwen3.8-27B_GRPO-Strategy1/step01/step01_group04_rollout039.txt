import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(x, temp, pen):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(x) * dx
    return sm * dx + pen * (integral - 1.0) ** 2

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h.copy()
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

def interp_to_m(h, m):
    n = len(h)
    if m <= n:
        if m == n:
            return h.copy()
        x_old = (np.arange(n) + 0.5) * (2.0 / n)
        x_new = (np.arange(m) + 0.5) * (2.0 / m)
        return np.interp(x_new, x_old, h)
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def lbfgs_step(h, temp, pen, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_smooth, h, args=(temp, pen),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return np.clip(res.x, 0.0, 1.0)

def coord_descent(h, nvals, sweeps, t0, budget_frac):
    n = len(h)
    h = h.copy()
    grid = np.linspace(0.0, 1.0, nvals)
    for sw in range(sweeps):
        if time.time() - t0 > budget_frac:
            return h
        any_imp = False
        for i in range(n):
            orig = h[i]
            best_v = orig
            best_c = compute_c5(h)
            for v in grid:
                if abs(v - orig) < 1e-7:
                    continue
                h[i] = v
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = v
            if any_imp:
                pass
            if best_v != orig:
                any_imp = True
            h[i] = best_v
        if not any_imp:
            break
    return h

def local_refine(h, steps, t0, budget_frac):
    n = len(h)
    h = h.copy()
    for _ in range(steps):
        if time.time() - t0 > budget_frac:
            break
        for i in range(n):
            orig = h[i]
            candidates = np.clip([orig - 0.1, orig - 0.03, orig + 0.03, orig + 0.1], 0.0, 1.0)
            best_v = orig

                    continue
                h[i] = v
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = v
            h[i] = best_v
    return h

def random_binary_h(n, rs):

    h = np.zeros(n)
    h[positions] = 1.0
    return h
