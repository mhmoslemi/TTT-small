import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h):
    n = len(h)
    integ = np.sum(h)
    if integ < 1e-14:
        return h
    h2 = h * (n / 2.0) / integ
    return np.clip(h2, 0.0, 1.0)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def topk_smooth(h, temp, penalty, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    L = len(corr)
    k = min(top_k, L)
    idx = np.argpartition(corr, -k)[-k:]
    top = corr[idx]
    cmax = np.max(top)
    shifted = (top - cmax) / temp
    sm = cmax + temp * np.log(np.sum(np.exp(shifted)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def optimize_stage(h_in, t0, time_limit, top_k_base, n_schedules=4):
    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    L = 2 * n - 1
    schedules = [
        (0.06, top_k_base, 500.0, 200),
        (0.02, min(top_k_base * 3, L), 2000.0, 200),
        (0.005, min(top_k_base * 6, L), 8000.0, 200),
        (0.001, L, 30000.0, 150),
    ][:n_schedules]
    cur = h_in.copy()
    best_local = compute_c5(cur)
    for si, (temp, tk, pen, maxit) in enumerate(schedules):
        elapsed = time.time() - t0
        if elapsed > time_limit:
            break
        res = minimize(
            topk_smooth, cur, args=(temp, pen, tk),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        if c5v < best_local:
            best_local = c5v
        cur = nh
    return cur, best_local

def freq_perturb(h, n_modes, amplitude):
    n = len(h)
    h2 = h.copy()
    for m_idx in range(1, n_modes + 1):
        amp = amplitude / (m_idx ** 0.7)
        phase = np.random.uniform(0, 2.0 * np.pi)
        h2 += amp * np.sin(2.0 * np.pi * m_idx * np.arange(n) / n + phase)
    return repair_h(np.clip(h2, 0.0, 1.0))

def blend_perturb(h, frac):
    n = len(h)
    shift = np.random.randint(1, n // 2)
    h_shifted = np.roll(h, shift)
    h2 = (1.0 - frac) * h + frac * h_shifted
    return repair_h(np.clip(h2, 0.0, 1.0))

def binary_push(h, width):
    n = len(h)
    h2 = h.copy()
    lo = 0.5 - width
    hi = 0.5 + width
    h2[h2 < lo] *= 0.5
    h2[h2 > hi] += (1.0 - h2[h2 > hi]) * 0.5
    return repair_h(np.clip(h2, 0.0, 1.0))

def random_start_feasible(n):
    h = np.random.uniform(0.0, 1.0, n)
    return repair_h(h)
