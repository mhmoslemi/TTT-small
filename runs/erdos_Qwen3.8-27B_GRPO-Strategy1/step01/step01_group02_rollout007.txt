    sizes = sorted(set([n0, n0 + 2, n0 - 2, 2 * n0, 2 * (n0 + 1)] +
                       [s for s in range(96, 250, 4) if s % 2 == 0]))
    sizes = [s for s in sizes if 60 <= s <= 500]

    # Phase 1: Quick evaluation at each size (interpolate + short optimize)
    size_scores = []
    for N in sizes:
        if time.time() - t0 > budget_s - 100:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        hI = project_sum(np.interp(xN, x0, h0), target)
        # Quick 200-iter optimize
        hq, cq = adam_optimize(hI, N, target, dx, t0, budget_s, rng,
                               lr_init=0.03, top_k=2, max_iter=200)
        size_scores.append((cq, N))

    size_scores.sort()
    # Focus on top 3 sizes
    focus_sizes = [s[1] for s in size_scores[:3]]
    for sc, N in size_scores:
        if sc < best_c5:
            best_c5 = sc
            # Recover the best h for this size briefly
            dx = 2.0 / N
            target = N / 2.0
            xN = np.linspace(0.0, 2.0, N, endpoint=False)
            hI = project_sum(np.interp(xN, x0, h0), target)
            hq, cq = adam_optimize(hI, N, target, dx, t0, budget_s, rng,
                                   lr_init=0.03, top_k=2, max_iter=200)
            best_h = hq

    # Phase 2: Deep optimization on focus sizes
    for N in focus_sizes:
        if time.time() - t0 > budget_s - 60:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        hI = project_sum(np.interp(xN, x0, h0), target)

        n_restarts = 4 if N == n0 else 2
        for r in range(n_restarts):
            if time.time() - t0 > budget_s - 30:
                break
            h_try = hI.copy()
            if r == 1:
                h_try += rng.randn(N) * 0.02
                h_try = project_sum(h_try, target)
            elif r == 2:
                h_try += rng.randn(N) * 0.05
                h_try = project_sum(h_try, target)
            elif r == 3:
                # Perturb only in the region of worst correlation
                corr0 = np.correlate(hI, 1.0 - hI, mode='full')
                worst_idx = int(np.argmax(corr0))
                worst_lag = worst_idx - (N - 1)
                h_try = hI.copy()
                h_try[rng.randint(N)] += 0.04
                h_try = project_sum(h_try, target)

            h_opt, c5_opt = adam_optimize(h_try, N, target, dx, t0, budget_s, rng,
                                          lr_init=0.035, top_k=2, max_iter=4000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                # Polish
                h_pol, c5_pol = polish_exact_max(h_opt, N, target, dx, t0, budget_s,
                                                  lr_init=0.002, max_iter=2000)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Phase 3: Binary local search on best
    N_best = len(best_h)
    dx_best = 2.0 / N_best
    tgt_best = N_best / 2.0
    if time.time() - t0 < budget_s - 30:
        h_bs, c5_bs = binary_local_search(best_h.copy(), N_best, tgt_best, dx_best,
                                          t0, budget_s, rng, max_iters=8000)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()

    # Phase 4: Final fine polish
    if time.time() - t0 < budget_s - 10:
        h_fin, c5_fin = polish_exact_max(best_h.copy(), N_best, tgt_best, dx_best,
                                         t0, budget_s, lr_init=0.001, max_iter=2000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
