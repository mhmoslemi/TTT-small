    # corr[m] = sum_j h[j]*(1-h[j+m-n+1]) for valid j

    # identify the terms involving h[i] and express as intercept + slope*t
    
    # Precompute full corr
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    # For each m, the terms involving h[i] are:
    # j=i: valid if 0<=i+m-n+1<n -> contribution h[i]*(1-h[i+m-n+1]) = t*(1-h[j2])
    # j+m-n+1=i -> j=i+m-n+1: valid if 0<=i+m-n+1<n -> contribution h[j2]*(1-t)
    # where j2 = i+m-n+1
    
    j2 = i + np.arange(2*n-1) - n + 1  # array of j2 for each m
    valid = (j2 >= 0) & (j2 < n)
    
    # Constant part = corr[m] - terms_involving_hi
    # terms_involving_hi[m] = [if valid: h[i]*(1-h[j2]) + h[j2]*(1-h[i])]
    #                        = [if valid: t*(1-h[j2]) + h[j2]*(1-t)]
    #                        = [if valid: h[j2] + t*(1-2*h[j2])]
    # So: corr[m] = base[m] + h[j2] + t*(1-2*h[j2]) where valid
    #            = base[m] where not valid
    
    hvals = np.where(valid, h[j2], 0)  # h[j2] where valid
    intercept = corr - np.where(valid, h[i]*(1-hvals) + hvals*(1-h[i]), 0)
    slope = np.where(valid, 1 - 2*hvals, 0)
    
    # Minimize f(t) = max_m(intercept[m] + slope[m]*t) + lam*t over t in [0,1]
    a, b = 0.0, 1.0
    for _ in range(50):
        m1 = a + (b-a)/3.0
        m2 = b - (b-a)/3.0
        f1 = np.max(intercept + slope*m1) + lam*m1
        f2 = np.max(intercept + slope*m2) + lam*m2
        if f1 > f2:
            a = m1
        else:
            b = m2
    return (a+b)/2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
    return np.clip(h, 0.0, 1.0)

def _coord_opt(h, i, n, lam):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.arange(2 * n - 1)
    j2 = i + idx - n + 1
    valid = (j2 >= 0) & (j2 < n)
    jc = np.clip(j2, 0, n - 1)
    hj2 = np.where(valid, h[jc], 0.0)
    terms = np.where(valid, h[i] * (1.0 - hj2) + hj2 * (1.0 - h[i]), 0.0)
    intercept = corr - terms
    slope = np.where(valid, 1.0 - 2.0 * hj2, 0.0)
    a, b = 0.0, 1.0
    for _ in range(60):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(intercept + slope * m1) + lam * m1
        f2 = np.max(intercept + slope * m2) + lam * m2
        if f1 > f2:
            a = m1
        else:
            b = m2
    return (a + b) / 2.0

def coord_descent(h_init, n, target_sum, n_passes=200):
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(n_passes):
        for i in range(n):
            h[i] = _coord_opt(h, i, n, 0.0)
        h = project_sum(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def pair_search(h, n, target_sum, n_rounds=50, time_limit=None):
    dx = 2.0 / n
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for rd in range(n_rounds):
        if time_limit and time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                # Try swap
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return best_h, best_c5

def slsqp_optimize(h_init, n, target_sum, maxiter=300):
    dx = 2.0 / n
    t0 = compute_c5(h_init, dx) / dx
    
    def obj(x):
        return x[n]
    
    def ineq_fun(x):
        c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return x[n] - c
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    x0 = np.concatenate([h_init, [t0 + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq_fun}, {'type': 'eq', 'fun': eq_fun}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sum(res.x[:n], target_sum)
    except Exception:
        pass
    return h_init.copy()

Idea: Maintain the correlation array incrementally. When I change h[i] from old_val to new_val, the correlation changes by:
Δcorr[m] = (new_val - old_val) * [(1-h[j2]) - h[j2]] for valid j2 = i+m-n+1

def coord_descent_fast(h_init, n, target_sum, n_passes=200):
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for _ in range(n_passes):
        for i in range(n):
            old = h[i]
            # Compute optimal new value
            j2 = i + np.arange(2*n-1) - n + 1
            valid = (j2 >= 0) & (j2 < n)
            jc = np.clip(j2, 0, n-1)
            hj2 = np.where(valid, h[jc], 0.0)
            # intercept = corr - h[i]*(1-hj2) - hj2*(1-h[i]) where valid
            terms = np.where(valid, old*(1-hj2) + hj2*(1-old), 0.0)
            intercept = corr - terms
            slope = np.where(valid, 1.0 - 2.0*hj2, 0.0)
            
            # Ternary search
            a, b = 0.0, 1.0
            for _ in range(45):
                m1 = a + (b-a)/3.0
                m2 = b - (b-a)/3.0
                if np.max(intercept + slope*m1) > np.max(intercept + slope*m2):
                    a = m1
                else:
                    b = m2
            new_val = (a+b)/2.0
            
            # Update corr incrementally
            delta = new_val - old
            if abs(delta) > 1e-15:
                corr += delta * slope  # slope = (1-2*hj2) where valid, 0 else
            h[i] = new_val
        
        h = project_sum(h, target_sum)
        # Recompute corr after projection
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
    return np.clip(h, 0.0, 1.0)

def cd_fast(h_init, n, target_sum, n_passes=500, early_stop=50):
    dx = 2.0 / n
    h = project_sum(h_init.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stale = 0
    idx_arr = np.arange(2 * n - 1)
    order = np.arange(n)
    for p in range(n_passes):
        np.random.shuffle(order)
        for i in order:
            old = h[i]
            j2 = i + idx_arr - n + 1
            valid = (j2 >= 0) & (j2 < n)
            jc = np.clip(j2, 0, n - 1)
            hj2 = np.where(valid, h[jc], 0.0)
            terms = np.where(valid, old * (1.0 - hj2) + hj2 * (1.0 - old), 0.0)
            intercept = corr - terms
            slope = np.where(valid, 1.0 - 2.0 * hj2, 0.0)
            a, b = 0.0, 1.0
            for _ in range(45):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(intercept + slope * m1) > np.max(intercept + slope * m2):
                    a = m1
                else:
                    b = m2
            new_val = (a + b) / 2.0
            delta = new_val - old
            if abs(delta) > 1e-15:
                corr += delta * slope
                h[i] = new_val
        h = project_sum(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > early_stop and p > 50:
            break
    return best_h, best_c5

def slsqp_opt(h_init, n, target_sum, maxiter=200):
    dx = 2.0 / n
    t0 = compute_c5(h_init, dx) / dx

    def _obj(x):
        return x[n]

    def _ineq(x):
        c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return x[n] - c

    def _eq(x):
        return np.sum(x[:n]) - target_sum

    x0 = np.concatenate([h_init, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': _ineq}, {'type': 'eq', 'fun': _eq}]
    try:
        res = minimize(_obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and res.x[n] < t0 + 0.01:
            return project_sum(res.x[:n], target_sum)
    except Exception:
        pass
    return h_init.copy()

def direction_search(h, n, target_sum, t_limit):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h_w = h.copy()
    for _ in range(2000):
        if time.time() > t_limit:
            break
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        delta = np.random.uniform(-0.3, 0.3)
        ni, nj = h_w[i] + delta, h_w[j] - delta
        if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
            continue
        h_w[i] = np.clip(ni, 0, 1)
        h_w[j] = np.clip(nj, 0, 1)
        h_w = project_sum(h_w, target_sum)
        c5 = compute_c5(h_w, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_w.copy()
        elif np.random.random() < 0.1:
            h_w = best_h.copy()
    return best_h, best_c5
