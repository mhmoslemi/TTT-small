import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

    cur = h_in.copy()
    best_h = cur.copy()
    best_c5 = compute_c5(cur)

        if time.time() > time_limit:
            break
        res = minimize(
            smooth_obj, cur, args=(temps[i], pens[i]),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxits_list[i], 'ftol': 1e-14, 'gtol': 1e-10}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        if c5v < best_c5:

            best_h = nh.copy()
        cur = nh
