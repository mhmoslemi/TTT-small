    return np.correlate(h, 1.0 - h, mode='full')

    vals = corr * dx

    sum_exp = np.sum(exp_shifted)
    obj = v_max + np.log(sum_exp) / T
    weights = exp_shifted / sum_exp
    return obj, weights

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        interior = (h > 1e-9) & (h < 1.0 - 1e-9)
        n_int = np.sum(interior)
        if n_int == 0:
            h += diff / n
        else:
            h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    return h

def get_weights(corr, dx, T):
    vals = corr * dx
    v_max = np.max(vals)
    if T < 1e-6:
        w = np.zeros(len(vals))
        w[np.argmax(vals)] = 1.0
        return v_max, w
    shifted = vals - v_max
    exp_s = np.exp(T * shifted)
    se = np.sum(exp_s)
    obj = v_max + np.log(se) / T
    w = exp_s / se
    return obj, w

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    _, w = get_weights(corr, dx, T)
    A = w[n-1:]
    B = w[:n-1][::-1]
    omh = 1.0 - h
    grad = np.zeros(n)
    for i in range(n):
        t1 = np.dot(A[:i+1], omh[i::-1][:i+1])
        t2 = np.dot(A[:n-i], h[i:i+n-i])
        t3 = np.dot(B[:i], omh[i-1::-1][:i]) if i >= 1 else 0.0
        t4 = np.dot(B[:n-1-i], h[i+1:i+1+n-1-i]) if n-1-i > 0 else 0.0
        grad[i] = dx * (t1 - t2 + t3 - t4)
    return grad

def obj_fc(h, T, pen, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    v_max = np.max(vals)
    if T < 1e-6:
        sm = v_max
    else:
        sm = v_max + np.log(np.sum(np.exp(T * (vals - v_max)))) / T
    return sm + pen * (np.sum(h) - target) ** 2

def grad_fc(h, T, pen, target):
    g = compute_grad(h, T)
    g += 2.0 * pen * (np.sum(h) - target)
    return g

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def try_optimize_lbfgs(h0, n, T, pen, max_iter, ftol=1e-14):
    target = n * 0.5
    res = opt.minimize(
        obj_fc, h0.copy(), args=(T, pen, target),
        jac=lambda x: grad_fc(x, T, pen, target),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10}
    )
    return res.x
