def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-15:
            return h
        h = h + room * (diff / total)

        room = h.copy()
        total = room.sum()
        if total < 1e-15:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-15:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-15:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def subgrad_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    tol = max(1e-3, 0.02 * max_c)
    active = np.where(corr >= max_c - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        for i in range(n):
            j1 = i - lag
            j2 = i + lag
            if 0 <= j1 < n:
                grad[i] += (1.0 - h[j1])
            if 0 <= j2 < n:
                grad[i] -= h[j2]
    grad /= max(1, len(active))
    return dx * grad

def coord_descent(h, n, max_sweeps=80, atol=1e-10):
    h = h.copy()
    prev_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        order = np.arange(n)
        np.random.shuffle(order)
        for i in order:
            save = h[i]
            h[i] = 0.0
            c0 = compute_c5(h)
            h[i] = 1.0
            c1 = compute_c5(h)
            # Also try interior via bisection on the piecewise-linear max
            best_val = min(c0, c1)
            best_x = 0.0 if c0 <= c1 else 1.0
            # Bisection for the minimum of the convex PL function on [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = (lo + hi) / 2
                h[i] = mid
                cm = compute_c5(h)
                if cm < best_val:
                    best_val = cm
                    best_x = mid
                h[i] = lo
                clo = compute_c5(h)
                h[i] = hi
                chi = compute_c5(h)
                if clo <= chi:
                    hi = mid
                else:
                    lo = mid
            h[i] = best_x
            c_new = compute_c5(h)
            if abs(c_new - prev_c5) < atol and abs(best_x - save) < 1e-6:
                h[i] = save
            else:
                h[i] = best_x
                improved = True
        h = project_feasible(h, n)
        prev_c5 = compute_c5(h)
        if not improved:
            break
    return h

                options={'maxiter': 600, 'ftol': 1e-14}
            )
            hc = project_feasible(np.clip(res.x, 0, 1), n)

            pass
