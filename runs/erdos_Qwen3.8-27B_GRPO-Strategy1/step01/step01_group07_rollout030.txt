    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def upsample(h, new_n):
    old_n = len(h)
    ox = np.linspace(0.5, old_n - 0.5, old_n) / old_n
    nx = np.linspace(0.5, new_n - 0.5, new_n) / new_n
    v = np.interp(nx, ox, h)
    diff = new_n / 2.0 - np.sum(v)
    v += diff / new_n
    v = np.clip(v, 0.0, 1.0)
    diff = new_n / 2.0 - np.sum(v)
    v += diff / new_n
    v = np.clip(v, 0.0, 1.0)
    return v

    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))

import numpy as np
from scipy.optimize import minimize
import time

_N = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def _eq_fun(x):
    return np.sum(x) - _N / 2.0

def _eq_jac(x):
    return np.ones(_N)

def _softmax(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    return dx * (m + beta * np.log(s))

def fix_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    ox = np.linspace(0.5, old_n - 0.5, old_n) / old_n
    nx = np.linspace(0.5, new_n - 0.5, new_n) / new_n
    v = np.interp(nx, ox, h)
    diff = new_n / 2.0 - np.sum(v)
    v += diff / new_n
    v = np.clip(v, 0.0, 1.0)
    diff = new_n / 2.0 - np.sum(v)
    v += diff / new_n
    return np.clip(v, 0.0, 1.0)

def slsqp_refine(h, n, beta, maxiter=200):
    global _N
    _N = n
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_softmax, h, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return fix_sum(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h

def local_search(h, n, rng, time_limit, base_step=0.004, max_step=0.08):
    cur = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sign = 1.0 if rng.rand() < 0.5 else -1.0
        if sign > 0:
            d_max = min(1.0 - h[i], h[j])
        else:
            d_max = min(h[i], 1.0 - h[j])
        if d_max < 1e-9:
            continue
        d = rng.uniform(base_step * 0.5, max(base_step, d_max * 0.9))
        old_i, old_j = h[i], h[j]
        h[i] += sign * d
        h[j] -= sign * d
        c_new = compute_c5(h)
        if c_new < cur - 1e-14:
            cur = c_new
        else:
            h[i], h[j] = old_i, old_j
    return h, cur

```python
def peak_shave(h, n, rng, time_limit, k_peaks=5):
    """Local search focused on reducing top peaks."""
    cur = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-k_peaks:]
        # Pick a random peak lag
        peak_idx = top_idx[rng.randint(k_peaks)]
        # Lag in terms of index offset
        lag = peak_idx - (n - 1)
        # To reduce correlation at this lag, try to modify h at positions
        # that contribute to this correlation
        # c[peak_idx] = sum_i h[i](1-h[i+lag]) (approx)
        # Random try
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sign = 1.0 if rng.rand() < 0.5 else -1.0
        if sign > 0:
            d_max = min(1.0 - h[i], h[j])
        else:
            d_max = min(h[i], 1.0 - h[j])
        if d_max < 1e-9:
            continue
        d = rng.uniform(0.002, max(0.003, d_max * 0.8))
        old_i, old_j = h[i], h[j]
        h[i] += sign * d
        h[j] -= sign * d
        c_new = compute_c5(h)
        if c_new < cur - 1e-14:
            cur = c_new
        else:
            h[i], h[j] = old_i, old_j
    return h, cur
```
