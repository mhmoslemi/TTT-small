    """Optimize h[j] given all other h fixed."""

    corr = np.correlate(h, 1.0 - h, mode='full')
    nl1 = 2 * n - 1
    
    # For coordinate j, compute a_k and c_k for each lag
    a = np.zeros(nl1)
    c = np.zeros(nl1)
    for k_idx in range(nl1):
        l = k_idx - (n - 1)
        # Contribution of h[j] to corr_k:
        # h[j] appears in h[n+l] when n+l=j → n=j-l: contributes 1-h[j-l]
        # h[j] appears in h[n] when n=j: contributes -h[j+l]
        ak = 0.0
        if 0 <= j - l < n:
            ak += (1.0 - h[j - l])
        if 0 <= j + l < n:
            ak -= h[j + l]
        a[k_idx] = ak
        c[k_idx] = corr[k_idx] - ak * h[j]
    
    # Minimize max_k (c_k + a_k * t) over t in [0,1]
    # Find candidate points: 0, 1, and all intersection points
    candidates = [0.0, 1.0]
    for i in range(nl1):
        for j2 in range(i+1, nl1):
            if abs(a[i] - a[j2]) > 1e-12:
                t_int = (c[j2] - c[i]) / (a[i] - a[j2])
                if 0.0 < t_int < 1.0:
                    candidates.append(t_int)
    
    # Evaluate at all candidates
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        t = np.clip(t, 0.0, 1.0)
        val = np.max(c + a * t)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

def f1d_eval(t, c, a):
    return np.max(c + a * t)

def coord_descent(h0, target_sum, n_sweeps=200, t_end=None):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    nl1 = 2 * n - 1
    prev_c5 = best_c5
    for sweep in range(n_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        for j in range(n):
            hj = h[j]
            a = np.empty(nl1)
            for ki in range(nl1):
                l = ki - (n - 1)
                ak = 0.0
                jl = j - l
                jr = j + l
                if 0 <= jl < n:
                    ak += 1.0 - h[jl]
                if 0 <= jr < n:
                    ak -= h[jr]
                a[ki] = ak
            corr = np.correlate(h, 1.0 - h, mode='full')
            c = corr - a * hj
            r = minimize_scalar(f1d_eval, args=(c, a), bounds=(0.0, 1.0),
                                method='bounded', options={'xatol': 1e-12, 'maxiter': 300})
            h[j] = r.x
        h = project_sr(h, target_sum)
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
        if prev_c5 - c5_now < 1e-14 and sweep > 3:
            break
        prev_c5 = c5_now
    return h, best_c5

            break
        h_try = h0 + psize * (np.random.rand(n) * 2.0 - 1.0)
        h_try = project_sr(h_try, target_sum)
        h_cd, c_cd = coord_descent(h_try, target_sum, 80, t_end)
        if c_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c_cd
    return best_h, best_c5

def pair_swap_search(h0, target_sum, n_try, t_end):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    deltas = np.array([0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9])
    for _ in range(n_try):
        if t_end is not None and time.time() > t_end:
            break
        i, j = np.random.choice(n, 2, replace=False)
        delta = deltas[np.random.randint(len(deltas))]
        sign = 1.0 if np.random.rand() < 0.5 else -1.0
        h_new = h.copy()
        h_new[i] += sign * delta
        h_new[j] -= sign * delta
        h_new = project_sr(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
    return h, best_c5
