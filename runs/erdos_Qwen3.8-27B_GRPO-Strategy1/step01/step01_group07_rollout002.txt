    return float(np.max(corr)) * dx

    """Gradient of corr[shift] w.r.t. h, where shift is the lag index offset from center."""
    n = len(h)
    grad = np.zeros(n)
    # corr at lag 'shift' (where shift = k - (n-1) for k in 0..2n-2)
    # c_shift = sum_i h[i+shift] * (1-h[i]) for valid i
    # d/dh[j]:
    #   Term from h[i+shift]=h[j] when i=j-shift: (1-h[j-shift]) if 0<=j-shift<n
    #   Term from (1-h[i])=(1-h[j]) when i=j: -h[j+shift] if 0<=j+shift<n

    lo1 = max(0, shift)

    return grad

        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def grad_single_shift(h, shift):
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, shift)
    hi1 = min(n - 1, n - 1 + shift)
    if lo1 <= hi1:
        grad[lo1:hi1 + 1] += (1.0 - h[lo1 - shift:hi1 - shift + 1])
    lo2 = max(0, -shift)
    hi2 = min(n - 1, n - 1 - shift)
    if lo2 <= hi2:
        grad[lo2:hi2 + 1] += -h[lo2 + shift:hi2 + shift + 1]
    return grad

def grad_max_corr(h, top_m=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_m]
    grad = np.zeros(n)
    for k in idx:
        shift = k - (n - 1)
        grad += grad_single_shift(h, shift)
    grad /= len(idx)
    return grad * dx

def project_feasible(z, n):
    lo, hi = z.min() - 2.0, z.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > n / 2.0:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    out = np.clip(z - t, 0.0, 1.0)
    out += (n / 2.0 - out.sum()) / n
    return np.clip(out, 0.0, 1.0)

def projected_gradient(h, n, lr_init=0.004, max_iter=4000, top_m=3, anneal=True):
    h = h.copy()
    cur_c5 = compute_c5(h)
    lr = lr_init
    best_local = cur_c5
    best_h_local = h.copy()
    for it in range(max_iter):
        g = grad_max_corr(h, top_m=top_m)
        h_new = project_feasible(h - lr * g, n)
        new_c5 = compute_c5(h_new)
        accept = new_c5 < cur_c5 - 1e-15
        if anneal and not accept:
            temp = max(0.0001, 0.001 * (1.0 - it / max_iter))
            if np.random.random() < np.exp(-max(0, (new_c5 - cur_c5)) / temp):
                accept = True
        if accept:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_local:
                best_local = cur_c5
                best_h_local = h.copy()
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-9:
                break
    return best_h_local, best_local

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0.0, 2.0, old_n, endpoint=False)
    nx = np.linspace(0.0, 2.0, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    for _ in range(4):
        v += (new_n / 2.0 - np.sum(v)) / new_n
        v = np.clip(v, 0.0, 1.0)
    return v

def fix_sum(h, n):
    h = h.copy()
    for _ in range(6):
        h += (n / 2.0 - np.sum(h)) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, rng, max_steps=2000):
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(max_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.001, 0.02) * rng.choice([-1.0, 1.0])
        hi2 = np.clip(h[i] + d, 0.0, 1.0)
        hj2 = np.clip(h[j] - d, 0.0, 1.0)
        if abs(hi2 - h[i]) < 1e-12 or abs(hj2 - h[j]) < 1e-12:
            continue
        h[i], h[j] = hi2, hj2
        c = compute_c5(h)
        if c < cur - 1e-14:
            cur = c
        else:
            h[i], h[j] = hi2 - d, hj2 + d
    return h, cur

def slsqp_topk(h, n, top_k=8, maxiter=300, ftol=1e-12):
    def obj(x):
        t = x[-1]
        return t

    def obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def cons_builder(n, top_k):
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.concatenate([np.ones(n), np.zeros(1)])}]
        def aug(x, kk=top_k):
            hv = x[:n]
            corr = np.correlate(hv, 1.0 - hv, mode="full")
            idx = np.argsort(corr)[::-1][:kk]
            vals = np.zeros(2 * n - 1)
            for k in idx:
                vals[k] = corr[k] - x[-1]
            return vals

        def aug_jac(x, kk=top_k):
            hv = x[:n]
            corr = np.correlate(hv, 1.0 - hv, mode="full")
            idx = np.argsort(corr)[::-1][:kk]
            J = np.zeros((kk, n + 1))
            for ii, k in enumerate(idx):
                shift = k - (n - 1)
                J[ii, :n] = grad_single_shift(hv, shift) * (2.0 / n)
                J[ii, -1] = -1.0
            return J

        cons.append({'type': 'ineq', 'fun': aug, 'jac': aug_jac})
        return cons

    x0 = np.concatenate([h, [compute_c5(h) + 1e-4]])
    bounds = list(zip(np.zeros(n), np.ones(n))) + [(compute_c5(h) - 0.05, None)]
    cons = cons_builder(n, top_k)
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = fix_sum(np.clip(res.x[:n], 0.0, 1.0), n)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h, compute_c5(h)
