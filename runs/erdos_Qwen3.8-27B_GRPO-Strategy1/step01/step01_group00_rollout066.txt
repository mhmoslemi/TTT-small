def project(h, n):
    """Project onto {h : sum(h) = n/2, 0 <= h <= 1}"""
    h = np.clip(h, 0, 1)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h

    dx = 2.0 / n

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
        h = np.clip(h, 0.0, 1.0)
    # Second pass if needed
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-10:
        interior = (h > 1e-6) & (h < 1.0 - 1e-6)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    return fix_sum(h, n)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    h_new = fix_sum(h_new, new_n)
    return h_new

def subgradient_update(h, n, momentum, beta, lr, n_top=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_vals = corr * dx
    # Top lags
    top_idx = np.argsort(c5_vals)[-n_top:]
    # Weights: softmax of top values
    top_vals = c5_vals[top_idx]
    w = np.exp((top_vals - np.max(top_vals)) / 0.01)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx]
        lag = k - (n - 1)
        wg = w[idx]
        if lag >= 0:
            d = lag
            m = n - 1 - d
            if m >= 0:
                grad[d:] += wg * (1 - h[:m+1])
                grad[:m+1] -= wg * h[d:d+m+1]
        else:
            d = -lag
            m = n - 1 - d
            if m >= 0:
                grad[:m+1] += wg * (1 - h[d:d+m+1])
                grad[d:d+m+1] -= wg * h[:m+1]
    
    # Momentum
    momentum[:] = beta * momentum + grad
    h_new = h - lr * momentum
    h_new = project(h_new, n)
    return h_new, np.max(c5_vals)

def lsq_objective(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    if T < 1e-6:
        return max_c
    shifted = (corr - max_c) / T
    return max_c + T * np.log(np.exp(shifted).sum() / len(corr))

def gradient_top_lags(h, n_top=20):
    """Compute subgradient using top n_top lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    # Get top lags
    top_idx = np.argsort(corr)[-n_top:]
    # Softmax weights
    vals = corr[top_idx]
    T_local = 20.0  # Local temperature
    w = np.exp((vals - np.max(vals)) * T_local)

    grad = np.zeros(n)
    for idx in range(n_top):
        k = top_idx[idx]
        lag = k - (n - 1)
        _add_lag_grad(grad, h, n, lag, w[idx])
    return grad

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    for _ in range(4):
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        n_int = np.sum(interior)
        if n_int == 0:
            break
        h[interior] += (target - np.sum(h)) / n_int
        np.clip(h, 0.0, 1.0, out=h)
        if abs(np.sum(h) - target) < 1e-12:
            break
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def add_lag_grad(grad, h, n, lag, w):
    nd = n - abs(lag)
    if nd <= 0:
        return
    if lag >= 0:
        grad[lag:] += w * (1.0 - h[:nd])
        grad[:nd] -= w * h[lag:]
    else:
        grad[:nd] += w * (1.0 - h[-lag:])
        grad[-lag:] -= w * h[:nd]

def gradient_top(h, n_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    top_idx = np.argsort(corr)[-n_top:]
    vals = (corr[top_idx] - cmax) * 50.0
    vals -= np.max(vals)
    w = np.exp(vals)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n_top):
        lag = top_idx[j] - (n - 1)
        add_lag_grad(grad, h, n, lag, w[j])
    return grad

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    if T < 1e-8:
        return cmax
    sh = (corr - cmax) * T
    sh -= np.max(sh)
    return cmax + np.log(np.sum(np.exp(sh))) / T

def sum_con(x):
    return np.sum(x) - len(x) * 0.5

def grad_descent(h, n, iters, t_start, budget_s, reserve_s):
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.08
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(iters):
        el = time.time() - t_start
        if budget_s - el < reserve_s:
            break
        if it % 3000 == 0:
            lr = 0.08 * np.exp(-it / 10000)
        nt = min(20, 1 + it // 2000)
        g = gradient_top(h, nt)
        momentum = beta * momentum + g
        h_new = h - lr * momentum
        np.clip(h_new, 0.0, 1.0, out=h_new)
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 8000 == 7999:
            noise = np.random.randn(n) * 0.015
            h = fix_sum(np.clip(h + noise, 0, 1), n)
            momentum *= 0.3
    return best_h, best_c5

def slsqp_refine(h, n, T, t_start, budget_s, reserve_s, maxiter):
    if budget_s - (time.time() - t_start) < reserve_s:
        return h, compute_c5(h)
    res = opt.minimize(
        soft_obj, h, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': sum_con}],
        options={'maxiter': maxiter, 'ftol': 1e-15}
    )
    h_c = fix_sum(np.clip(res.x, 0, 1), n)
    c5_c = compute_c5(h_c)
    if c5_c < compute_c5(h):
        return h_c, c5_c
    return h, compute_c5(h)
