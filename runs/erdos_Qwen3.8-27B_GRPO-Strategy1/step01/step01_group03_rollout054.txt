            best_c5 = c_t
            best_h = h_t.copy()
            best_n = n_try

    # Phase 6: Multi-start perturbations on best_n
    n_best = best_n
    dx_best = 2.0 / n_best
    target_best = n_best / 2.0
    for r in range(8):
        if not ok(50):
            break
        scale = 0.03 + 0.02 * r
        h_r = best_h + scale * np.random.randn(n_best)
        h_r = project_sr(h_r, target_best)
        h_r = slsqp_minmax(h_r, dx_best, target_best, maxiter=250)
        c_r = compute_c5(h_r, dx_best)
        if c_r < best_c5 and ok(30):
            h_r = slsqp_active(h_r, dx_best, target_best, maxiter=150, nphases=2)
            c_r = compute_c5(h_r, dx_best)
            if ok(20):
                dl_r = time.time() + min(budget_s - elapsed() - 10, 30)
                h_r, c_r = coord_descend(h_r, dx_best, target_best, n_sweeps=15, deadline=dl_r)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r.copy()
                best_n = n_best

    # Phase 7: Binary + fractional starts
    for r in range(4):
        if not ok(40):
            break
        h_b = np.zeros(n_best)
        pos = np.random.permutation(n_best)
        k = int(target_best)
        h_b[pos[:k]] = 1.0
        frac = target_best - k
        if frac > 0.01:
            h_b[pos[k]] = frac
        h_b = project_sr(h_b, target_best)
        h_b = slsqp_minmax(h_b, dx_best, target_best, maxiter=250)
        c_b = compute_c5(h_b, dx_best)
        if c_b < best_c5 and ok(25):
            dl_b = time.time() + min(budget_s - elapsed() - 10, 30)
            h_b, c_b = coord_descend(h_b, dx_best, target_best, n_sweeps=15, deadline=dl_b)
            if c_b < best_c5:
                best_c5 = c_b
                best_h = h_b.copy()
                best_n = n_best

    # Phase 8: Final micro-polish
    if ok(15):
        dl_mp = time.time() + min(budget_s - elapsed() - 3, 60)
        best_h, best_c5 = micro_polish(best_h, best_n and 2.0 / best_n, best_n / 2.0,
                                       n_trials=800, deadline=dl_mp)

    # Final SLSQP if time
    if ok(10):
        h_f = slsqp_minmax(best_h, 2.0 / best_n, best_n / 2.0, maxiter=200)
        c_f = compute_c5(h_f, 2.0 / best_n)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    best_h = project_sr(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
