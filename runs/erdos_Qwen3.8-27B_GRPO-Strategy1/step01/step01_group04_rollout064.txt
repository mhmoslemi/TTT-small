import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_and_grad(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_v = np.exp(shifted / temp)
    w = exp_v / np.sum(exp_v)
    smax = cmax + temp * np.log(np.sum(exp_v))
    integral = np.sum(h) * dx
    val = smax * dx + penalty * (integral - 1.0) ** 2
    
    # Gradient
    A = np.correlate(1.0 - h, w, 'full')[:n]
    B_full = np.correlate(h[::-1], w, 'full')
    B = B_full[n-1::-1]
    grad = dx * (A - B) + 2.0 * penalty * (integral - 1.0)
    return val, grad

def repair_h(h):
    n = len(h)
    s = np.sum(h)
    if s < 1e-14:
        return h
    return np.clip(h / s, 0.0, 1.0)

def interp_up(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    return np.interp(x_new, x_old, h)

def optimize_block(h0, t0, max_time, temp_schedule, n):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    for temp, penalty, maxit in temp_schedule:
        elapsed = time.time() - t0
        if elapsed > max_time:
            break
        res = minimize(
            obj_and_grad, best_h,
            args=(temp, penalty),
            method='L-BFGS-B',
            jac=True,
            bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': 2000}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        # Repair to maintain integral constraint
        s = np.sum(nh)
        target = n / 2.0
        if s > 0 and abs(s - target) > 1e-6:
            # Scale to fix integral, then clip
            nh = nh * (target / s)
            nh = np.clip(nh, 0.0, 1.0)
            # Iterate a few times to handle clipping
            for _ in range(10):
                s2 = np.sum(nh)
                if abs(s2 - target) < 1e-10:
                    break
                nh = nh * (target / s2)
                nh = np.clip(nh, 0.0, 1.0)
        
        c5v = compute_c5(nh)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = nh.copy()
    return best_h, best_c5

def fourier_perturb(h, sigma, seed_rng):
    n = len(h)
    # Perturb in Fourier space (limited to low frequencies for smoothness)
    hat = np.fft.rfft(h.astype(float))
    freqs = np.fft.rfftfreq(n)
    # Weight by frequency to make low-freq perturbations larger
    mask = np.exp(-2.0 * (freqs * n) ** 2)
    noise_hat = (seed_rng.randn(n//2+1) + 1j*seed_rng.randn(n//2+1)) * sigma * mask
    noise_hat[0] = 0  # Don't change the mean
    noise = np.fft.irfft(noise_hat, n)
    h2 = np.clip(h + noise, 0.0, 1.0)
    return repair_h(h2)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Standard optimization schedule
    sched1 = [(0.1, 1000.0, 300), (0.03, 5000.0, 300), (0.008, 10000.0, 200)]
    sched2 = [(0.05, 2000.0, 300), (0.01, 10000.0, 300), (0.003, 20000.0, 200)]
    sched3 = [(0.02, 5000.0, 200), (0.005, 20000.0, 200), (0.001, 50000.0, 150)]
    sched4 = [(0.005, 20000.0, 150), (0.001, 50000.0, 150), (0.0003, 100000.0, 100)]
    
    # Phase 1: Optimize at initial resolution (360)
    max_t1 = budget_s * 0.15
    h_cur, c5_1 = optimize_block(h_init, t0, max_t1, sched1, len(h_init))
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h_cur.copy()
    
    # Phase 2: Upscale to 720, optimize
    if time.time() - t0 < budget_s * 0.45:
        h_up = interp_up(h_cur, 2)
        h_up = repair_h(h_up)
        max_t2 = budget_s * 0.25
        h_cur, c5_2 = optimize_block(h_up, t0, max_t2, sched2, len(h_up))
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h_cur.copy()
    
    # Phase 3: Upscale to 1440, optimize
    if time.time() - t0 < budget_s * 0.75:
        h_up = interp_up(h_cur, 2)
        h_up = repair_h(h_up)
        max_t3 = budget_s * 0.35
        h_cur, c5_3 = optimize_block(h_up, t0, max_t3, sched3, len(h_up))
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h_cur.copy()
    
    # Phase 4: Restart exploration at the best resolution
    n_final = len(best_h)
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        n_restarts = min(5, int(remaining / 15))
        for ri in range(n_restarts):
            if time.time() - t0 > budget_s * 0.92:
                break
            # Diversify: Fourier perturb + small random
            if ri % 2 == 0:
                h_pert = fourier_perturb(best_h, 0.05 / (ri + 1), rng)
            else:
                h_pert = best_h + rng.randn(n_final) * 0.03 / (ri + 1)
                h_pert = repair_h(np.clip(h_pert, 0.0, 1.0))
            h_pert, c5r = optimize_block(h_pert, t0, budget_s * 0.92 - (time.time() - t0), sched4, n_final)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = h_pert.copy()
    
    # Final: Try a direct optimization of the current best with very low temp
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        final_sched = [(0.001, 50000.0, 100), (0.0003, 100000.0, 80)]
        h_fin, c5_fin = optimize_block(best_h, t0, remaining * 0.8, final_sched, len(best_h))
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
    
    # Ensure valid
    best_h = np.clip(best_h, 0.0, 1.0)
    # Fix integral
    target = len(best_h) / 2.0
    s = np.sum(best_h)
    if abs(s - target) > 1e-8:
        best_h = best_h * (target / s)
        best_h = np.clip(best_h, 0.0, 1.0)
    
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
