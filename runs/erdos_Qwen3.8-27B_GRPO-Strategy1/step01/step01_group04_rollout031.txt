```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13 * n:
            break
        if d > 0:
            mask = h < 1.0
            c = np.count_nonzero(mask)
            if c == 0:
                break
            h[mask] = np.clip(h[mask] + d / c, 0.0, 1.0)
        else:
            mask = h > 0.0
            c = np.count_nonzero(mask)
            if c == 0:
                break
            h[mask] = np.clip(h[mask] + d / c, 0.0, 1.0)
    return h

def interp_up(h, f):
    n = len(h)
    m = n * f
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(m) + 0.5) / m
    return np.interp(xn, xo, h)

def obj_grad(h, temp, K, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = 2 * n - 1
    target = n / 2.0
    eq_val = np.sum(h) - target
    
    if K is not None and K < nc:
        idx = np.argpartition(corr, -K)[-K:]
        top_c = corr[idx]
    else:
        idx = np.arange(nc)
        top_c = corr
    
    cmax = np.max(top_c)
    exps = np.exp((top_c - cmax) / max(temp, 1e-15))
    wsum = np.sum(exps)
    w = exps / wsum
    
    lse = cmax + temp * np.log(wsum)
    val = lse * dx + lam * eq_val * eq_val
    
    one_minus_2h = 1.0 - 2.0 * h
    grad = np.full(n, 2.0 * lam * eq_val)
    
    for j, wj in zip(idx, w):
        m_lo = max(0, j - (n - 1))
        m_hi = min(n - 1, j)
        if m_lo <= m_hi:
            seg = one_minus_2h[j - m_hi : j - m_lo + 1][::-1]
            grad[m_lo:m_hi + 1] += wj * seg
    
    grad *= dx
    return val, grad

def lbb_opt(h, t0, tlimit, phases):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cur = h.copy()
    best = compute_c5(cur)
    best_h = cur.copy()
    for temp, K, maxit, lam in phases:
        if time.time() >= tlimit:
            break
        try:
            res = minimize(
                obj_grad, cur, args=(temp, K, lam),
                method='L-BFGS-B', jac=True, bounds=bounds,
                options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-10})
            nh = repair_h(res.x)
            c = compute_c5(nh)
            if c < best:
                best, best_h = c, nh.copy()
            cur = nh
        except Exception:
            pass
    return best_h, best

def pushpull_refine(h, t0, tlimit, niter=200):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    best = compute_c5(cur)

    for _ in range(niter):
        if time.time() >= tlimit:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        kmax = int(np.argmax(corr))
        # Try small random perturbations
        for _ in range(3):

            h2 = np.clip(cur + rng.randn(n) * sigma, 0.0, 1.0)
