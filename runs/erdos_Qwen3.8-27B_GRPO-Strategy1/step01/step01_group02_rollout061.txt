import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        ss = float(np.clip(h - mid, 0.0, 1.0).sum())
        if ss > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:] += (1.0 - h[:N - m])
            g[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def adam_optim(h, N, target, dx, t0, time_limit, lr=0.02, K=8, tau=12.0, max_iter=5000, decay_K=False):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    cur_K = K
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        k = max(3, min(cur_K, len(corr)))
        topk = np.argpartition(corr, -k)[-k:]
        sc = tau * corr[topk]
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
        grad = np.zeros(N)
        for j in range(k):
            lag = int(topk[j]) - (N - 1)
            grad += w[j] * grad_lag(h, lag, N)
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * grad * grad
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)
        if (it + 1) % 300 == 0:
            tau *= 1.5
            lr *= 0.82
            if decay_K and cur_K > 3:
                cur_K = max(3, cur_K - 1)
    return best_h, best_c5

def sa_binary(h_init, N, target, dx, t0, time_limit, rng, T_init=0.4):
    idx = np.argsort(h_init)[::-1][:int(target)]
    h = np.zeros(N)
    h[idx] = 1.0
    cur_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    cool = 0.99985
    batch = 300
    while time.time() - t0 < time_limit:
        for _ in range(batch):
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h, dx)
            if new_c5 <= cur_c5:
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                if rng.rand() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
                    cur_c5 = new_c5
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            T *= cool
        if T < 0.002:
            break
    return best_h, best_c5

def hill_climb_binary(h_bin, N, target, dx, t0, time_limit, rng, max_rounds=200):
    cur_c5 = compute_c5(h_bin, dx)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = np.flatnonzero(h_bin)
        for i in ones:
            if time.time() - t0 > time_limit:
                break
            zeros = np.flatnonzero(1 - h_bin)
            for j_idx in range(min(30, len(zeros))):
                j = zeros[rng.permutation(len(zeros))[j_idx]]
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                new_c5 = compute_c5(h_bin, dx)
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                    improved = True
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_bin.copy()
                    break
                else:
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
        if not improved:
            break
    return best_h, best_c5
