def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr)) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_topk(h, k=5):
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    top_idx = np.argsort(corr)[-k:]
    w = 1.0 / k
    grad = np.zeros(n)
    for s in top_idx:
        tau = int(s) - n + 1
        lo1, hi1 = max(0, -tau), min(n, n - tau)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 + tau:hi1 + tau])
        lo2, hi2 = max(0, tau), min(n, n + tau)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 - tau:hi2 - tau]
