import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smax_obj_and_grad(h, target, dx, tau, penalty_w):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(corr.max())
    sc = tau * (corr - cmax)
    e = np.exp(sc)
    w = e / float(e.sum())
    smax = cmax + float(np.log(float(e.sum()))) / tau
    
    grad = np.zeros(N)
    thresh = 1e-4
    active = np.where(w > thresh)[0]
    for idx in active:
        lag = int(idx) - (N - 1)
        wi = w[int(idx)] * dx
        if lag >= 0 and lag < N:
            grad[lag:] += wi * (1.0 - h[:N - lag])
            grad[:N - lag] -= wi * h[lag:]
        elif lag < 0:
            l = -lag
            if l < N:
                grad[:N - l] += wi * (1.0 - h[l:])
                grad[l:] -= wi * h[:N - l]
    
    s = float(h.sum()) - target
    grad += 2.0 * penalty_w * s
    val = smax + penalty_w * s * s
    return val, grad

    best_h = h_init.copy()

            return val
        
        def g(z):
            val, grad = smax_obj_and_grad(z, target, dx, tau, pw)
            return grad
        
        try:
            res = minimize(f, best_h, jac=g, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
            h_new = project_sum(np.clip(res.x, 0.0, 1.0), target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

    """Create diverse initialized."""
    x = np.arange(N) * dx
    if mode == 0:
        # Sinusoidal
        p = rng.randint(1, max(2, N//10))

        h = (h > h.mean()).astype(float)
    elif mode == 2:

        h = np.zeros(N)
        block = rng.randint(N//20, max(2, N//5))
        for i in range(0, N, block):
            if rng.random() > 0.5:
                h[i:min(i+block, N)] = 1.0
    else:
        # Pure random
        h = (rng.random(N) > 0.5).astype(float)
    return project_sum(h, target)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smax_val_grad(h, target, dx, tau, pw):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(corr.max())
    sc = tau * (corr - cmax)
    e = np.exp(sc)
    esum = float(e.sum())
    w = e / esum
    smax = cmax + np.log(esum) / tau
    grad = np.zeros(N)
    thresh = 1e-4
    active = np.where(w > thresh)[0]
    for idx in active:
        lag = int(idx) - (N - 1)
        wi = w[int(idx)] * dx
        if lag >= 0 and lag < N:
            grad[lag:] += wi * (1.0 - h[:N - lag])
            grad[:N - lag] -= wi * h[lag:]
        elif lag < 0:
            l = -lag
            if l < N:
                grad[:N - l] += wi * (1.0 - h[l:])
                grad[l:] -= wi * h[:N - l]
    s = float(h.sum()) - target
    grad += 2.0 * pw * s
    return smax + pw * s * s, grad

def optimize_lbfgs(h_init, N, target, dx, t0, budget_s, tau_start=5.0, n_stages=5):
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)
    bounds = [(0.0, 1.0)] * N
    for stage in range(n_stages):
        if time.time() - t0 > budget_s - 8:
            break
        tau = tau_start * (2.5 ** stage)
        pw = 800.0 + 400.0 * stage
        maxiter = max(50, int(300 - 40 * stage))
        def f(z):
            return smax_val_grad(z, target, dx, tau, pw)[0]
        def g(z):
            return smax_val_grad(z, target, dx, tau, pw)[1]
        try:
            res = minimize(f, best_h, jac=g, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9, 'maxfun': maxiter * 3})
            h_new = project_sum(np.clip(res.x, 0.0, 1.0), target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

    if mode == 0:

def binary_local_search(h_init, N, target, dx, t0, budget_s, max_rounds=50):
    """Try flipping individual bits to reduce max overlap."""
    h = (h_init > 0.5).astype(float)
    h = project_sum(h, target)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for _ in range(max_rounds):
        if time.time() - t0 > budget_s - 5:
            break
        improved = False
        order = np.random.permutation(N)
        for i in order:
            if time.time() - t0 > budget_s - 5:
                break
            old = h[i]
            h[i] = 1.0 - old
            s = float(h.sum())
            if abs(s - target) < 1e-10:

                if c5 < best_c5 - 1e-10:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] = old
            else:
                h[i] = old
        if not improved:
            break
    return best_h, best_c5
