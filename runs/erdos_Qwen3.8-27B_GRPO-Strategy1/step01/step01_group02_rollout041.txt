import numpy as np

def compute_c5_full(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        n = N - m
        if n > 0:
            g[:n] += 1.0 - h[m:m + n]
            g[m:m + n] -= h[:n]
    else:
        l = -m
        n = N - l
        if n > 0:
            g[l:l + n] += 1.0 - h[:n]
            g[:n] -= h[l:l + n]
    return g
