            c5_new2 = compute_c5(h_new)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new
            lr *= 0.98
            if lr < 0.002:
                lr = 0.08

    # Phase 2: Focused local search (target worst lag)
    if elapsed() < budget_s * 0.38:
        for rnd in range(60):
            if elapsed() > budget_s * 0.38:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            tl = int(np.argmax(corr))
            best_h, c5_try = local_search(best_h, n, 200, target_lag=tl)
            if c5_try < best_c5:
                best_c5 = c5_try
            # Occasional random local search without target
            if rnd % 5 == 4:
                best_h2, c5_try2 = local_search(best_h, n, 100, target_lag=None)
                if c5_try2 < best_c5:
                    best_h = best_h2
                    best_c5 = c5_try2

    # Phase 3: SLSQP polish at original dimension
    if elapsed() < budget_s * 0.52:
        best_h, best_c5 = slsqp_polish(best_h, n, t0)

    # Phase 4: More local search after SLSQP
    if elapsed() < budget_s * 0.62:
        for rnd in range(40):
            if elapsed() > budget_s * 0.62:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            tl = int(np.argmax(corr))
            best_h, c5_try = local_search(best_h, n, 150, target_lag=tl)
            if c5_try < best_c5:
                best_c5 = c5_try

    # Phase 5: Try higher dimensions
    for n_try in [80, 100, 120, 160]:
        if elapsed() > budget_s * 0.85:
            break
        h_try = np.zeros(n_try)
        for i in range(n_try):
            lo = int(i * n / n_try)
            hi = max(lo + 1, int((i + 1) * n / n_try))
            hi = min(hi, n)
            h_try[i] = best_h[lo:hi].mean()
        h_try = project_feasible(h_try, n_try)
        c5_try = compute_c5(h_try)

        # Subgradient on new dimension
        lr2 = 0.08
        for it in range(80000):
            if elapsed() > budget_s * 0.85:
                break
            h_new, m_idx = subgrad_step(h_try, n_try, lr2)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                n = n_try
                h_try = best_h.copy()
                lr2 = min(lr2 * 1.005, 0.12)
            else:
                h_try = h_new
                lr2 *= 0.99
                if lr2 < 0.003:
                    lr2 = 0.06

        # Local search on best so far
        if elapsed() < budget_s * 0.85:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            tl = int(np.argmax(corr))
            best_h, c5_try = local_search(best_h, len(best_h), 200, target_lag=tl)
            if c5_try < best_c5:
                best_c5 = c5_try

        # Quick SLSQP
        if elapsed() < budget_s * 0.85:
            for T in [200, 1000]:
                if elapsed() > budget_s * 0.85:
                    break
                n_cur = len(best_h)
                cons = [{'type': 'eq', 'fun': sum_constraint}]
                bnds = [(0.0, 1.0)] * n_cur
                try:
                    res = minimize(smooth_c5, best_h, args=(T,),
                                   method='SLSQP', bounds=bnds, constraints=cons,
                                   options={'maxiter': 300, 'ftol': 1e-14})
                    hc = project_feasible(np.clip(res.x, 0, 1), n_cur)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                except Exception:
                    pass

    # Phase 6: Block perturbation + re-optimize (escape local minima)
    if elapsed() < budget_s * 0.93:
        for trial in range(8):
            if elapsed() > budget_s * 0.93:
                break
            n_cur = len(best_h)
            hp = block_perturb(best_h, n_cur, max(3, n_cur // 10))
            c5p = compute_c5(hp)
            # Quick subgradient on perturbed
            lr3 = 0.05
            for it in range(20000):
                if elapsed() > budget_s * 0.93:
                    break
                h_new, _ = subgrad_step(hp, n_cur, lr3)
                c5_new = compute_c5(h_new)
                if c5_new < c5p:
                    c5p = c5_new
                    hp = h_new
                    lr3 = min(lr3 * 1.01, 0.1)
                else:
                    lr3 *= 0.97
                    if lr3 < 0.002:
                        break
            if c5p < best_c5:
                best_c5 = c5p
                best_h = hp

    # Phase 7: Final polish
    n_final = len(best_h)
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = slsqp_polish(best_h, n_final, t0)

    # Final local search
    if elapsed() < budget_s * 0.995:
        n_final = len(best_h)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        tl = int(np.argmax(corr))
        best_h, best_c5 = local_search(best_h, n_final, 300, target_lag=tl)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
