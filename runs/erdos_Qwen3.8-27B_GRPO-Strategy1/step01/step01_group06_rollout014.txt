        for i in order:
            old = best[i]
            # Try: 0, 1, and if interior, try 0.5*old, 1-0.5*old
            candidates = [0.0, 1.0]
            if 0.01 < old < 0.99:
                candidates.append(old * 0.5)
                candidates.append(1.0 - (1.0 - old) * 0.5)
            
            best_candidate = old
            best_obj = best_max
            
            for v in candidates:
                if abs(v - old) < 1e-8:
                    continue
                best[i] = v
                # Project
                temp = project_feasible(best, n)
                c = np.correlate(temp, 1.0 - temp, mode='full')
                cm = np.max(c)
                if cm < best_obj - 1e-10:
                    best_obj = cm
                    best_candidate = v
                    best = temp
                    best_corr = c
                    best_max = cm
            
            if best_candidate != old:
                improved = True
        
        if not improved:
            break
    
    return best, compute_c5(best)

Actually, I realize there might be an issue: the projection distributes the sum error proportionally to the "room" (1-h for adding, h for removing). This means it affects all coordinates, not just the one I changed. So the "coordinate" descent isn't really coordinate-wise.

A better approach: paired exchange. Change h[i] += d and h[j] -= d simultaneously. This preserves the sum exactly without needing projection.

        order = np.random.permutation(n)
        for i in order:

                continue  # Boundary, can only go one way
            # Find a partner j with opposite incentive
            # If best[i] > 0.5, try decreasing; else try increasing
            if best[i] > 0.5:

                # Try a few j values
                for _ in range(5):
                    j = np.random.randint(n)
                    if j == i or best[j] >= 1:

                        best[j] += d

                            improved = True
                            break
                        else:
                            best[i] += d
                            best[j] -= d
            else:
                # Want to increase best[i]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = np.minimum(T * (corr - m), 500.0)
    val = m + np.log(np.sum(np.exp(scaled))) / T
    return val * dx

def top_lag_c5(h, m, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -m)[-m:]
    top = corr[idx]
    mx = np.max(top)
    val = mx + np.log(np.sum(np.exp(T * (top - mx)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + (diff / total) * room
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + (diff / total) * room
    return np.clip(h, 0.0, 1.0)

def sum_con_fun(x):
    return x.sum() - len(x) / 2.0

def coord_descent(h, n_passes, t0, time_limit):
    n = len(h)
    best = h.copy()
    best_max = np.max(np.correlate(best, 1.0 - best, mode='full'))
    for p in range(n_passes):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = best[i]
            cands = [0.0, 1.0]
            if 0.02 < old < 0.98:
                cands.append(old * 0.4)
                cands.append(1.0 - (1.0 - old) * 0.4)
            best_obj = best_max
            best_v = old
            best_arr = best.copy()
            for v in cands:
                if abs(v - old) < 1e-9:
                    continue
                best[i] = v
                temp = project_feasible(best, n)
                c = np.max(np.correlate(temp, 1.0 - temp, mode='full'))
                if c < best_obj - 1e-11:
                    best_obj = c
                    best_v = v
                    best_arr = temp.copy()
            if best_v != old:
                best = best_arr
                best_max = best_obj
                improved = True
        if not improved:
            break
    return best, compute_c5(best)

def make_binary_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    pos = rng.choice(n, n // 2, replace=False)
    h[pos] = 1.0
    return h

def make_spread_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    k = n // 2
    base = np.linspace(0, n, k + 1, endpoint=False) + 0.5
    jitter = rng.uniform(-0.45, 0.45, k)
    pos = np.round(base + jitter).astype(int) % n
    pos = np.unique(pos)
    if len(pos) < k:
        rem = np.setdiff1d(np.arange(n), pos)
        pos = np.concatenate([pos, rng.choice(rem, k - len(pos), replace=False)])
    pos = pos[:k]
    h[pos] = 1.0
    return h

def slsqp_optimize(h, n, T, maxiter, ftol, t0, time_limit):
    if time.time() - t0 > time_limit:
        return h, compute_c5(h)
    try:
        res = minimize(
            smooth_c5, h, args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con_fun}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        hc = project_feasible(np.clip(res.x, 0, 1), n)
        c5v = compute_c5(hc)
        if c5v < compute_c5(h):
            return hc, c5v
    except:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = make_spread_start(60, seed)

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm-up on parent
    for T in [8, 40, 150]:
        if time.time() - t0 > budget_s * 0.08:
            break
        best_h, best_c5 = slsqp_optimize(best_h, n, T, 150, 1e-10, t0, budget_s * 0.08)

    # Phase 2: Coordinate descent (fast, exploits per-coord structure)
    best_h, best_c5 = coord_descent(best_h, 80, t0, budget_s * 0.18)

    # Phase 3: Alternate SLSQP + coord descent for polish
    for T in [300, 1000]:
        if time.time() - t0 > budget_s * 0.30:
            break
        best_h, best_c5 = slsqp_optimize(best_h, n, T, 200, 1e-11, t0, budget_s * 0.30)
        best_h, best_c5 = coord_descent(best_h, 40, t0, budget_s * 0.33)

    # Phase 4: Multi-start exploration
    n_sizes = [n, 64, 80, 100, 120]
    best_c5_phase4 = best_c5
    for n_try in n_sizes:
        if time.time() - t0 > budget_s * 0.72:
            break
        n_tests = 3 if n_try == n else 4
        for trial in range(n_tests):
            if time.time() - t0 > budget_s * 0.72:
                break
            s = seed + n_try * 131 + trial * 9973
            if trial % 2 == 0:
                h0 = make_binary_start(n_try, s)
            else:
                h0 = make_spread_start(n_try, s)
            # Quick SLSQP
            h0, c0 = slsqp_optimize(h0, n_try, 40, 100, 1e-9, t0, budget_s * 0.72)
            # Coord descent
            h0, c0 = coord_descent(h0, 30, t0, budget_s * 0.74)
            # Second SLSQP
            h0, c0 = slsqp_optimize(h0, n_try, 300, 150, 1e-10, t0, budget_s * 0.76)
            h0, c0 = coord_descent(h0, 40, t0, budget_s * 0.78)
            if c0 < best_c5:
                best_c5 = c0
                best_h = h0
                n = len(best_h)

    # Phase 5: Final deep refinement
    if time.time() - t0 < budget_s * 0.88:
        for T in [500, 1500, 4000]:
            if time.time() - t0 > budget_s * 0.93:
                break
            best_h, best_c5 = slsqp_optimize(best_h, len(best_h), T, 300, 1e-12, t0, budget_s * 0.93)

    # Phase 6: Final coord descent sweep
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, 200, t0, budget_s * 0.97)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
