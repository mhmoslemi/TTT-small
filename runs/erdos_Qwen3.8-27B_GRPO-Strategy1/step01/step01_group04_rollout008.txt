    dx = 2.0 / n

    return float(np.max(corr) * dx)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def project_feasible(h):
    n = len(h)
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-12:
            break
        h = h + (target_sum - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def subgradient_polish(h, n_steps=500, t0=None, budget_s=None):
    """Polish using subgradient steps targeting the active max correlation"""
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    best_c5 = compute_c5(cur)
    alpha = 0.01
    
    for i in range(n_steps):
        if t0 is not None and time.time() - t0 > budget_s * 0.95:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)  # the actual shift
        
        # Compute subgradient: (grad)[m] = 1 - cur[m+lag] - cur[m-lag]
        grad = np.ones(n)
        if lag >= 0:
            grad[:n-lag] -= cur[lag:]  # 1 - cur[m+lag] for m < n-lag
            grad[lag:] -= cur[:n-lag]  # - cur[m-lag] for m >= lag
        else:
            grad[:n+lag] -= cur[-lag:]  # 1 - cur[m+lag] = 1 - cur[m-|lag|]... 
            # Wait, if lag < 0, then m+lag < m, so cur[m+lag] is cur[m-|lag|]
            # and m-lag = m+|lag|, so cur[m-lag] is cur[m+|lag|]
            # grad[m] = 1 - cur[m+lag] - cur[m-lag] = 1 - cur[m-|lag|] - cur[m+|lag|]
            grad[-lag:] -= cur[:n+lag]  # cur[m+lag] = cur[m-|lag|] for m >= |lag|
            grad[:n+lag] -= cur[-lag:]  # cur[m-lag] = cur[m+|lag|] for m < n-|lag|

        cur = project_feasible(cur)
        
        c5v = compute_c5(cur)
        if c5v < best_c5:
            best_c5 = c5v
        
        # Decrease step size
        alpha = max(0.001, alpha * 0.999)
    
    return cur, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_objective(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        frac = (target - s) / n
        h = h + frac
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def lbfgsb_step(h, temp, penalty, maxiter, t0, budget_s):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    if time.time() - t0 > budget_s * 0.93:
        return h, compute_c5(h)
    res = minimize(
        smooth_objective, h, args=(temp, penalty),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11, 'maxfun': 15000}
    )
    nh = np.clip(res.x, 0.0, 1.0)
    nh = project_feasible(nh)
    return nh, compute_c5(nh)

def coord_descent_polish(h, max_rounds, t0, budget_s):

            offsets = [-0.15, -0.08, -0.04, -0.02, 0.02, 0.04, 0.08, 0.15, 0.25, -0.25]
            for off in offsets:
                t = old + off
                t = max(0.0, min(1.0, t))

                cur[m] = t
                c5v = compute_c5(cur)
                if c5v < best_v - 1e-12:

                    best_t = t
            cur[m] = best_t
        cur = project_feasible(cur)
        c5v = compute_c5(cur)
        if c5v < best_c5:
            best_c5 = c5v
        else:
            if rnd > 2:
                break
    return cur, best_c5

def structured_perturb(h, n_modes, rng):
    n = len(h)
    h2 = h.copy()
    x = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(2, max(3, n // 4)))
        amp = rng.uniform(0.04, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        h2 = h2 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project_feasible(h2)

def gaussian_perturb(h, sigma, rng):
    n = len(h)
    h2 = h + rng.standard_normal(n) * sigma
    return project_feasible(h2)
