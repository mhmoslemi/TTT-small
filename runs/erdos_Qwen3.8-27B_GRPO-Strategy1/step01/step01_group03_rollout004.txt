def slsqp_solve(h0, dx, target_sum, maxiter=300):

        return x[n]
    def ineq(x):

    def eqf(x):
        return np.sum(x[:n]) - target_sum

        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def c5_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def slsqp_solve(h0, dx, target_sum, maxiter=300):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.001]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def smooth_obj_grad(h, beta):
    n = len(h)
    comps = c5_vec(h, 1.0)
    m = comps.max()
    smax = m + np.log(np.exp(beta * (comps - m)).sum()) / beta
    w = np.exp(beta * (comps - m))
    w /= w.sum()
    grad = np.zeros(n)
    for l in range(2 * n - 1):
        if w[l] < 1e-14:
            continue
        shift = l - (n - 1)
        js = max(0, -shift)
        je = min(n, n - shift)
        if js < je:
            grad[js:je] += w[l] * (1.0 - h[js + shift:je + shift])
        js2 = max(0, shift)
        je2 = min(n, n + shift)
        if js2 < je2:
            grad[js2:je2] -= w[l] * h[js2 - shift:je2 - shift]
    grad -= np.mean(grad)
    return smax, grad

def lbfgs_projected(h0, target_sum, beta, maxiter=200, t_deadline=None):
    n = len(h0)
    def obj_proj(p):
        h = project_sr(p, target_sum)
        v, _ = smooth_obj_grad(h, beta)
        return v
    def grad_proj(p):
        h = project_sr(p, target_sum)
        _, g = smooth_obj_grad(h, beta)
        return g
    try:
        res = minimize(obj_proj, h0, jac=grad_proj, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_sr(res.x, target_sum) if res.x is not None else h0.copy()
    except Exception:
        return h0.copy()

def multi_beta_refine(h0, dx, target_sum, t_deadline):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    betas = [10, 25, 50, 100, 200, 400]
    iters = [150, 120, 100, 80, 60, 50]
    for b, ni in zip(betas, iters):
        if time.time() > t_deadline:
            break
        h = lbfgs_projected(h, target_sum, b, ni, t_deadline)
        c5_new = compute_c5(h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
    return h, best_c5

def sa_optimize(h0, dx, target_sum, t_deadline, n_iter=4000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    T = 0.015
    T_min = 0.0005
    cool = (T_min / T) ** (1.0 / max(n_iter, 1))
    evals = 0
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        step = 0.02 + 0.06 * T
        h_new = h + step * np.random.randn(n)
        h_new = project_sr(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        evals += 1
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-max(delta, 0) / max(T, 1e-12)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        T = max(T * cool, T_min)
        if (it + 1) % 400 == 0 and time.time() < t_deadline - 5:
            h_ref = slsqp_solve(h, dx, target_sum, 80)
            c5_ref = compute_c5(h_ref, dx)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref
                h = h_ref
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    try:
        h_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h_raw = np.zeros(47)
        h_raw[:23] = 1.0
        h_raw[23] = 0.5

    n = len(h_raw)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_sr(h_raw, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def tl():
        return budget_s - (time.time() - t_start)

    # Phase 1: SLSQP from initial
    if tl() > 40:
        h1 = slsqp_solve(h0, dx, target_sum, 400)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Multi-beta L-BFGS-B refinement
    if tl() > 60:
        dl = time.time() + min(tl() - 20, 120)
        h2, c2 = multi_beta_refine(best_h, dx, target_sum, dl)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: SLSQP polish
    if tl() > 25:
        h3 = slsqp_solve(best_h, dx, target_sum, 300)
        c3 = compute_c5(h3, dx)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Structured multi-start
    n_restarts = 25
    for r in range(n_restarts):
        if tl() < 50:
            break
        if r % 5 == 0:
            h_r = best_h + 0.08 * np.random.randn(n)
        elif r % 5 == 1:
            h_r = best_h + 0.15 * np.random.randn(n)
        elif r % 5 == 2:
            h_r = best_h.copy()
            i, j = np.random.choice(n, 2, replace=False)
            h_r[i], h_r[j] = h_r[j], h_r[i]
            h_r += 0.04 * np.random.randn(n)
        elif r % 5 == 3:
            h_r = best_h.copy()
            blk = np.random.randint(0, max(n - 6, 1))
            ln = np.random.randint(3, 8)
            h_r[blk:blk + ln] = np.clip(h_r[blk:blk + ln] + 0.25 * np.random.randn(ln), 0, 1)
        else:
            mask = np.random.rand(n) < 0.3
            h_r = best_h.copy()
            h_r[mask] = np.random.rand(mask.sum())
            h_r += 0.03 * np.random.randn(n)

        h_r = project_sr(h_r, target_sum)
        h_r = slsqp_solve(h_r, dx, target_sum, 150)
        c5_r = compute_c5(h_r, dx)

        if c5_r < best_c5 and tl() > 70:
            h_r, c5_r2 = multi_beta_refine(h_r, dx, target_sum, time.time() + min(tl() - 30, 40))
            if c5_r2 < c5_r:
                c5_r = c5_r2
            h_r = slsqp_solve(h_r, dx, target_sum, 100)
            c5_r = compute_c5(h_r, dx)

        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 5: Simulated annealing
    if tl() > 120:
        sa_dl = time.time() + min(tl() - 25, 500)
        h_sa, c5_sa = sa_optimize(best_h, dx, target_sum, sa_dl, n_iter=4000)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa, c5_sa
        if tl() > 25:
            h_pol = slsqp_solve(best_h, dx, target_sum, 200)
            c5_pol = compute_c5(h_pol, dx)
            if c5_pol < best_c5:
                best_h, best_c5 = h_pol, c5_pol

    # Phase 6: Try nearby n_points
    for n_try in [45, 48, 49, 50, 51, 52, 53]:
        if tl() < 50:
            break
        if n_try == n:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_try = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                          np.linspace(0, 2, n, endpoint=False), best_h, period=2)
        h_try = project_sr(h_try, ts_try)
        h_try = slsqp_solve(h_try, dx_try, ts_try, 200)
        c5_try = compute_c5(h_try, dx_try)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try
            n, dx, target_sum = n_try, dx_try, ts_try

    # Final polish
    if tl() > 15:
        h_f = slsqp_solve(best_h, dx, target_sum, 250)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    best_h = project_sr(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, len(best_h)
