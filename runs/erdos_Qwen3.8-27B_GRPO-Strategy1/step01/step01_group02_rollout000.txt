Actually, let me reconsider the gradient computation. For topk lags, I'm computing the gradient of a weighted sum. But the max is not differentiable, so I'm using a smooth approximation. The topk approach is essentially: maximize the sum of the top-k values (with softmax weighting). This should work well.

One important thing: the gradient should be correct. Let me double-check.

f_k(h) = sum_n h[n+k] * (1-h[n]) for valid n.

This is bilinear in h. The gradient is:
∂f_k/∂h[i] = sum over terms where h[i] appears.

For k >= 0:
f_k = sum_{n=0}^{N-1-k} h[n+k] * (1-h[n])

h[i] appears as h[n+k] when n = i-k (valid if 0 <= i-k < N-k, i.e., k <= i < N)
  → contributes (1-h[i-k])

h[i] appears as h[n] in (1-h[n]) when n = i (valid if 0 <= i < N-k)
  → contributes -h[i+k]

So for k >= 0:
∂f_k/∂h[i] = (1-h[i-k]) * 1[k<=i<N] - h[i+k] * 1[0<=i<N-k]

For k < 0, let l = -k:
f_{-l} = sum_{n=l}^{N-1} h[n-l] * (1-h[n]) = sum_{m=0}^{N-1-l} h[m] * (1-h[m+l])

h[i] appears as h[m] when m = i (valid if 0 <= i < N-l)
  → contributes (1-h[i+l])

h[i] appears as h[m+l] in (1-h[m+l]) when m = i-l (valid if l <= i < N)
  → contributes -h[i-l]

So for k = -l < 0:
∂f_k/∂h[i] = (1-h[i+l]) * 1[0<=i<N-l] - h[i-l] * 1[l<=i<N]

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, k, N):
    g = np.zeros(N)
    if k >= 0:
        if k < N:
            g[k:] += (1.0 - h[:N - k])
            g[:N - k] -= h[k:]
    else:
        l = -k
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def optimize_h(h, N, target, dx, t0, budget_s, rng, lr=0.02, tau_init=5.0,
               topk=5, max_iter=5000, tau_end=60.0):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    tau = tau_init
    M = 2 * N - 1

    for it in range(max_iter):
        if time.time() - t0 > budget_s - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

        # Adaptive tau: interpolate from tau_init to tau_end
        frac = float(it) / max_iter
        tau = tau_init + (tau_end - tau_init) * frac * frac

        # Top-k softmax weights
        if topk < M:
            top_idx = np.argpartition(corr, -topk)[-topk:]
            sc = tau * corr[top_idx]
            sc -= sc.max()
            w = np.exp(sc)
            wsum = w.sum()
            if wsum < 1e-15:
                break
            w /= wsum
        else:
            top_idx = np.arange(M)
            sc = tau * corr
            sc -= sc.max()
            w = np.exp(sc)
            w /= w.sum()

        grad = np.zeros(N)
        for idx, wi in zip(top_idx, w):
            if wi < 1e-10:
                continue
            lag = int(idx) - (N - 1)
            grad += wi * grad_lag(h, lag, N)

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)

        # Anneal lr
        if (it + 1) % 500 == 0:
            lr *= 0.90
        if lr < 1e-9:
            break
    return best_h, best_c5

def local_refine(h, N, target, dx, t0, budget_s, rng, n_try=200):
    """Small random perturbations to escape local minima."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(n_try):
        if time.time() - t0 > budget_s - 5:
            break
        h_try = best_h + rng.randn(N) * 0.008
        h_try = project_sum(h_try, target)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5
