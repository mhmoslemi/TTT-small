    n = len(h)
    h = h.copy()
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            # Compute effect of changing h[i]
            old = h[i]
            corr = np.correlate(h, 1-h, mode='full')
            
            # Effect on corr when h[i] -> h[i] + delta
            # As h part: corr[m] += delta * (1-h[m-(n-1)-i]) for m in [n-1-i, 2n-2-i]
            # As (1-h) part: corr[m] += -delta * h[i-m+n-1] for m in [max(0,i), min(2n-2, i+n-1)]
            
            effect = np.zeros(2*n-1)
            # h part contribution
            lo = max(0, n-1-i)
            hi = min(2*n-2, 2*n-2-i)
            for m in range(lo, hi+1):
                j = m - (n-1) - i
                if 0 <= j < n:
                    effect[m] += (1 - h[j])
            # (1-h) part contribution  
            lo2 = max(0, i)
            hi2 = min(2*n-2, i+n-1)

                if 0 <= j < n:
                    effect[m] -= h[j]
            
            # Now find delta that minimizes max(corr + delta * effect)
            # subject to -old <= delta <= 1-old
            lo_d = -old
            hi_d = 1 - old
            
            # The function f(delta) = max_m (corr[m] + delta * effect[m]) is convex PL
            # We can find the minimum by binary search on the derivative
            # or by checking all breakpoints
            
            # Actually, let's just do a fine scan + refine
            # Binary search approach: f is convex, so check sign of subgradient
            def f_val(d):
                return np.max(corr + d * effect)
            
            # Check boundaries
            f_lo = f_val(lo_d)
            f_hi = f_val(hi_d)
            
            # Binary search for where subgradient changes sign
            # Subgradient at delta: {sum of effects for active lags}
            # At the optimum, 0 is in the subgradient
            
            lo_b = lo_d
            hi_b = hi_d
            for _ in range(60):
                mid = 0.5 * (lo_b + hi_b)
                vals = corr + mid * effect
                mx = np.max(vals)
                active = vals >= mx - 1e-12
                subgrad = np.sum(effect[active])
                if subgrad > 0:
                    hi_b = mid
                elif subgrad < 0:
                    lo_b = mid
                else:
                    break
            
            best_d = 0.5 * (lo_b + hi_b)
            # Check boundaries too
            candidates = [lo_d, hi_d, best_d, 0.0]
            best_f = np.inf

                    best_d = c

        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    """Project h onto {sum=h_target} ∩ [0,1]^n"""
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

def coord_descent(h, dx, target_sum, max_sweeps=200, time_limit=None, t0=None):
    """Coordinate descent exploiting piecewise-linear structure of C5."""
    n = len(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if t0 is not None and time.time() > time_limit:
            break
        for i in range(n):
            old = h[i]
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            # Effect of h[i] -> h[i] + delta on each corr[m]
            effect = np.zeros(2*n-1)
            # h[i] appears in corr[m] as h part: m in [n-1-i, 2n-2-i]
            lo = max(0, n-1-i)
            hi = min(2*n-2, 2*n-2-i)
            for m in range(lo, hi+1):
                j = m - (n-1) - i
                if 0 <= j < n:
                    effect[m] += (1.0 - h[j])
            # h[i] appears in corr[m] as (1-h) part: m in [i, i+n-1] (clipped to [0,2n-2])
            lo2 = max(0, i)
            hi2 = min(2*n-2, i+n-1)
            for m in range(lo2, hi2+1):
                j = i - m + n - 1
                if 0 <= j < n:
                    effect[m] -= h[j]
            
            # Minimize max(corr + delta*effect) over delta in [-old, 1-old]
            lo_d = -old
            hi_d = 1.0 - old
            
            # Binary search on subgradient
            lo_b, hi_b = lo_d, hi_d
            for _ in range(50):
                mid = 0.5*(lo_b + hi_b)
                vals = corr + mid * effect
                mx = vals.max()
                active = vals >= mx - 1e-12
                sg = effect[active].sum()
                if sg > 1e-14:
                    hi_b = mid
                elif sg < -1e-14:
                    lo_b = mid
                else:
                    break
            
            best_d = 0.5*(lo_b + hi_b)
            # Also check boundaries and 0
            cands = [lo_d, hi_d, 0.0, best_d]

                cd = max(lo_d, min(hi_d, cd))
                fv = (corr + cd * effect).max()
                if fv < best_f:
                    best_f = fv
                    best_d = cd
            
            h[i] += best_d
        
        # Re-project to sum constraint (coordinate descent may violate it slightly)
        h = project_sum(h, target_sum)
    
    return h

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0.0, 1.0)]*n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sum(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def sa_optimize(h0, dx, target_sum, t_start, t_end, max_iter=5000):
    """Simulated annealing with structured moves."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    current_c5 = best_c5
    T = 0.05  # initial temperature
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        # Choose move
        r = np.random.rand()
        if r < 0.3:
            # Single coordinate nudge
            i = np.random.randint(n)
            h[i] += np.random.randn() * 0.05
        elif r < 0.5:
            # Pair swap (preserves sum)
            i, j = np.random.choice(n, 2, replace=False)
            d = np.random.randn() * 0.1
            h[i] += d
            h[j] -= d
        elif r < 0.7:
            # Block shift
            blen = np.random.randint(2, 6)
            i = np.random.randint(n - blen)
            shift = np.random.choice([-1, 1])
            h_shifted = h.copy()
            h_shifted[i:i+blen] = 0
            j = (i + shift) % n
            # Wrap-around handling
            for k in range(blen):
                idx = i + shift + k
                if 0 <= idx < n:
                    h_shifted[idx] += h[i+k]
        elif r < 0.85:
            # Binary flip
            i = np.random.randint(n)
            h[i] = 1.0 - h[i]
            # Adjust sum by modifying another element
            j = np.random.randint(n)

            h[j] += d
        else:

            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

    return best_h, best_c5

def upsample(h, factor=2):
    """Upsample h by linear interpolation."""
    from scipy.interpolate import interp1d
    n = len(h)
    dx = 2.0 / n
    x_old = np.arange(n) * dx
    x_new = np.arange(n * factor) * (dx / factor)
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    return f(x_new)

        h4, c4 = sa_optimize(best_h, dx, target_sum, t_end, 8000)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4
        # Polish SA result
        if tok(25) and c4 < best_c5 + 0.005:
            h4 = refine(best_h, dx, target_sum, t_start + budget_s - 30)
            c4 = compute_c5(h4, dx)
            if c4 < best_c5:
                best_h, best_c5 = h4, c4

    # Phase 5: Multi-start with structured patterns
    starts = []
    for r in range(5):
        if not tok(20):
            break
        h_s = best_h + 0.05 * np.random.randn(n)
        h_s = project_sum(h_s, target_sum)
        starts.append(h_s)
    for r in range(4):
        if not tok(20):
            break
        h_b = np.zeros(n)
        pos = np.random.permutation(n)
        half = int(target_sum)
        h_b[pos[:half]] = 1.0
        frac = target_sum - half
        if frac > 0.01:
            h_b[pos[half]] = frac
        h_b = project_sum(h_b, target_sum)
        starts.append(h_b)
    # Alternating block pattern
    if tok(15):
        h_blk = np.zeros(n)
        for k in range(n):
            h_blk[k] = 1.0 if (k % 2 == 0) else 0.0
        h_blk = project_sum(h_blk, target_sum)
        starts.append(h_blk)
    # Shifted versions
    for s in [1, 2, 3]:
        if not tok(15):
            break
        h_sh = np.roll(best_h, s)
        h_sh = project_sum(h_sh, target_sum)
        starts.append(h_sh)
    # LSE-smoothed solve from best
    if tok(30):
        h_lse = lse_smooth_solve(best_h, dx, target_sum, beta=20.0, maxiter=200)
        starts.append(h_lse)

    for h_s in starts:
        if not tok(25):
            break
        h_opt = slsqp_solve(h_s, dx, target_sum, 120)
        c_opt = compute_c5(h_opt, dx)
        if c_opt < best_c5 + 0.002 and tok(30):
            h_opt = refine(h_opt, dx, target_sum, t_start + budget_s - 30)
            c_opt = compute_c5(h_opt, dx)
        if c_opt < best_c5:
            best_h, best_c5 = h_opt, c_opt

    # Phase 6: Iterative polish loop
    for _ in range(6):
        if not tok(12):
            break
        h_p = coord_descent(best_h, dx, target_sum, 40, t_start + budget_s - 15)
        h_p = project_sum(h_p, target_sum)
        h_p = slsqp_solve(h_p, dx, target_sum, 100)
        c_p = compute_c5(h_p, dx)
        if c_p < best_c5:
            best_h, best_c5 = h_p, c_p
        else:
            break

    # Final SLSQP
    if tok(6):
        h_f = slsqp_solve(best_h, dx, target_sum, 80)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    best_h = project_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
