import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def sum_eq(h, n):
    return np.sum(h) - 0.5 * n

def sum_eq_jac(h, n):
    return np.ones(n)

def fix_sum(h):
    n = len(h)
    target = 0.5 * n
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    m = np.sum(interior)
    if m > 0:
        h[interior] += diff / m
    else:
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff2 = 0.5 * n - np.sum(h)
    if abs(diff2) > 1e-12:
        interior2 = (h > 1e-10) & (h < 1.0 - 1e-10)
        m2 = np.sum(interior2)
        if m2 > 0:
            h[interior2] += diff2 / m2
    return np.clip(h, 0.0, 1.0)

def upsample_fn(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n)
    x_new = np.linspace(0.0, 2.0, new_n)
    return fix_sum(np.interp(x_new, x_old, h))

def slsqp_step(h, n, T, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(obj_smooth, h, args=(T,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
    return fix_sum(res.x)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    h_best = np.array(initial_h_values, dtype=float).copy()
    n = len(h_best)
    c5_best = compute_c5(h_best)
    
    # Phase 1: SLSQP annealing at n=280
    for T in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() - t0 > budget_s - 80:

        h_try = slsqp_step(h_best, n, T, 5000)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best:
            h_best, c5_best = h_try, c5_try
    
    # Phase 2: Basin hopping
    for i in range(10):
        if time.time() - t0 > budget_s - 300:

        sigma = 0.015 + 0.01 * (i % 4)
        h_pert = fix_sum(np.clip(h_best + np.random.randn(n) * sigma, 0, 1))
        h_try = slsqp_step(h_pert, n, 0.005, 3000)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best:

            # Refine

            if c5_try2 < c5_best:

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def sum_eq(h, n):
    return np.sum(h) - 0.5 * n

def sum_eq_jac(h, n):
    return np.ones(n)

def fix_sum(h):
    n = len(h)
    target = 0.5 * n
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    m = np.sum(interior)
    if m > 0:
        h[interior] += diff / m
    else:
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff2 = 0.5 * n - np.sum(h)
    if abs(diff2) > 1e-12:
        interior2 = (h > 1e-10) & (h < 1.0 - 1e-10)
        m2 = np.sum(interior2)
        if m2 > 0:
            h[interior2] += diff2 / m2
    return np.clip(h, 0.0, 1.0)

def upsample_fn(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n)
    x_new = np.linspace(0.0, 2.0, new_n)
    return fix_sum(np.interp(x_new, x_old, h))

def slsqp_step(h, n, T, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    res = opt.minimize(obj_smooth, h, args=(T,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
    return fix_sum(res.x)

def targeted_descent(h, n_steps=50):
    """Directly reduce the worst correlation via subgradient steps."""
    n = len(h)
    h = h.copy()
    base_c5 = compute_c5(h)
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = int(np.argmax(corr))
        k = k_worst - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            for j in range(k, n):
                grad[j] += (1.0 - h[j - k])
            for j in range(0, n - k):
                grad[j] -= h[j + k]
        else:
            kp = -k
            for j in range(0, n - kp):
                grad[j] += (1.0 - h[j + kp])
            for j in range(kp, n):
                grad[j] -= h[j - kp]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        lr = 0.5 / gnorm
        for alpha_mult in [1.0, 0.5, 0.25, 0.1]:
            h_try = np.clip(h - alpha_mult * lr * grad, 0.0, 1.0)
            h_try = fix_sum(h_try)
            if compute_c5(h_try) < compute_c5(h):
                h = h_try
                break
        if compute_c5(h) < base_c5:
            base_c5 = compute_c5(h)
    return h
