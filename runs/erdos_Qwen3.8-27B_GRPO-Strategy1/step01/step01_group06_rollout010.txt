    corr = np.correlate(h, 1.0 - h, mode='full')

    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

import time

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()

        room = h.copy()
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def sum_con(h):
    return h.sum() - len(h) / 2.0

def slsqp_opt(h0, n, T, maxiter=500):
    cons = [{'type': 'eq', 'fun': sum_con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, np.asarray(h0, dtype=float), args=(T,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = project_feasible(res.x, n)
        return hc, compute_c5(hc)
    except Exception:
        return h0, compute_c5(h0)

    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    return np.clip(h, 0.0, 1.0)

    for T in Ts:
        hc, c5 = slsqp_opt(hc, n, T, maxiter_each)
        if c5 < best:
            best = c5
        else:
            break
    return hc, best

    top_idx = np.argsort(corr)[-top_k:]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def sum_con(h):
    return h.sum() - len(h) / 2.0

def slsqp_opt(h0, n, T, maxiter=500):
    cons = [{'type': 'eq', 'fun': sum_con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, np.asarray(h0, dtype=float), args=(T,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = project_feasible(res.x, n)
        return hc, compute_c5(hc)
    except Exception:
        return h0, compute_c5(h0)

def sine_init(n, freq, amp, phase):
    i = np.arange(n)
    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    return np.clip(h, 0.0, 1.0)

def multi_T_slsqp(h0, n, Ts, maxiter_each=350):
    hc = h0.copy()
    best = compute_c5(hc)
    for T in Ts:
        hc, c5 = slsqp_opt(hc, n, T, maxiter_each)
        if c5 < best:
            best = c5
        else:
            break
    return hc, best

def active_set_opt(h0, n, top_k, T=300):
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    dx = 2.0 / n
    def top_obj(h):
        c = np.correlate(h, 1.0 - h, mode='full')
        tc = c[top_idx]
        m = np.max(tc)
        return (m + np.log(np.sum(np.exp(T * (tc - m)))) / T) * dx
    cons = [{'type': 'eq', 'fun': sum_con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(top_obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-15})
        hc = project_feasible(res.x, n)
        return hc, compute_c5(hc)
    except Exception:
        return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = sine_init(60, 1.0, 0.5, 0.0)

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-T SLSQP on initial solution (15% time)
    if elapsed() < budget_s * 0.15:
        for T in [30, 100, 400, 1500, 5000]:
            if elapsed() > budget_s * 0.14:
                break
            hc, c5 = slsqp_opt(best_h, n, T, 400)
            if c5 < best_c5:
                best_c5, best_h = c5, hc

    # Phase 2: Active-set refinement (10% time)
    if elapsed() < budget_s * 0.25:
        for k in [3, 5, 8, 12, 20]:
            if elapsed() > budget_s * 0.24:
                break
            hc, c5 = active_set_opt(best_h, n, k)
            if c5 < best_c5:
                best_c5, best_h = c5, hc

    # Phase 3: Structured multi-n search (35% time)
    for n_try in [40, 50, 60, 70, 80, 90, 100, 120, 150, 200]:
        if elapsed() > budget_s * 0.60:
            break
        starts = []
        for freq in [1.0, 0.5, 1.5, 2.0]:
            for ph in [0.0, np.pi / 2, np.pi, 3.0 * np.pi / 4]:
                starts.append(sine_init(n_try, freq, 0.5, ph))
        # Triangular wave
        i_arr = np.arange(n_try)
        i_arr2 = i_arr % (n_try // 2)
        tri = np.minimum(i_arr2, (n_try // 2) - i_arr2) / (n_try // 4.0)
        starts.append(np.clip(0.5 + 0.4 * (tri - 0.5), 0, 1))
        # Upsample from best
        if n > 4:
            xp = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            xn = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            starts.append(np.clip(np.interp(xn, xp, best_h), 0, 1))

        for h0 in starts:
            if elapsed() > budget_s * 0.60:
                break
            h0 = project_feasible(h0, n_try)
            c0 = compute_c5(h0)
            if c0 < best_c5 + 0.01:
                hc, c5 = multi_T_slsqp(h0, n_try, [50, 300, 1500, 6000], 300)
                if c5 < best_c5:
                    best_c5, best_h, n = c5, hc, n_try

    # Phase 4: Sinusoidal perturbation restarts (15% time)
    n_cur = len(best_h)
    n_restarts = 20
    for r in range(n_restarts):
        if elapsed() > budget_s * 0.78:
            break
        freq_p = np.random.uniform(0.5, 3.0)
        ph_p = np.random.uniform(0, 2 * np.pi)
        amp_p = np.random.uniform(0.02, 0.08)
        h_pert = best_h + amp_p * np.sin(2 * np.pi * freq_p * np.arange(n_cur) / n_cur + ph_p)
        h_pert = project_feasible(h_pert, n_cur)
        hc, c5 = multi_T_slsqp(h_pert, n_cur, [200, 1000, 5000], 300)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 5: Alternating active-set + high-T SLSQP (15% time)
    n_cur = len(best_h)
    for _ in range(15):
        if elapsed() > budget_s * 0.93:
            break
        hc, c5 = active_set_opt(best_h, n_cur, 6, T=500)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
        hc, c5 = slsqp_opt(best_h, n_cur, 5000, 500)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
        hc, c5 = active_set_opt(best_h, n_cur, 10, T=1000)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 6: Final ultra-high-T polish
    if elapsed() < budget_s * 0.97:
        n_cur = len(best_h)
        hc, c5 = slsqp_opt(best_h, n_cur, 15000, 800)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
        hc, c5 = active_set_opt(best_h, n_cur, 5, T=3000)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
