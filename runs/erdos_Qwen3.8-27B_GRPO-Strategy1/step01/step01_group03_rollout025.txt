    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    """Project h onto {sum(h)=target_sum, 0<=h<=1} approximately."""
    h = np.array(h, dtype=float)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def coord_descent(h0, dx, target_sum, n_iter=500, deadline=None):

        h1, c1 = coord_optimize(h0, dx, target_sum, 500, dl, rng)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: SLSQP polish
    if time_ok(10):
        h2 = slsqp_refine(best_h, dx, target_sum, 500)
        c2 = compute_c5(h2, dx)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: More coordinate descent (with damping for escape)
    if time_ok(10):
        dl = t0 + budget_s * 0.40
        h3, c3 = coord_optimize(best_h, dx, target_sum, 400, dl, rng, damping=0.7)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
        h3b, c3b = coord_optimize(best_h, dx, target_sum, 300, dl, rng, damping=1.0)
        if c3b < best_c5:
            best_h, best_c5 = h3b, c3b

    # Phase 4: Randomized restarts with coordinate descent + SLSQP
    n_restarts = 14
    for r in range(n_restarts):
        if not time_ok(25):
            break
        pert = 0.02 + 0.015 * r
        h_r = best_h + pert * rng.randn(n)
        h_r = project_sum(h_r, target_sum)
        dl = t0 + budget_s - 100 - 10 * max(0, n_restarts - 1 - r)
        h_cd, c_cd = coord_optimize(h_r, dx, target_sum, 200, dl, rng)
        cs = compute_c5(h_cd, dx)
        if cs < best_c5:
            if time_ok(20):
                h_sl = slsqp_refine(h_cd, dx, target_sum, 300)
                cs2 = compute_c5(h_sl, dx)
                if cs2 < cs:
                    h_cd, cs = h_sl, cs2
        if cs < best_c5:
            best_h, best_c5 = h_cd, cs

    # Phase 5: Binary/structured starts
    for r in range(8):
        if not time_ok(20):
            break
        h_b = np.zeros(n)
        pos = rng.permutation(n)
        n_ones = int(target_sum)
        h_b[pos[:n_ones]] = 1.0
        frac = target_sum - n_ones
        if frac > 0.01:
            h_b[pos[n_ones]] = frac
        h_b = project_sum(h_b, target_sum)
        dl = t0 + budget_s - 50
        h_cd, c_cd = coord_optimize(h_b, dx, target_sum, 150, dl, rng)
        cs = compute_c5(h_cd, dx)
        if cs < best_c5 and time_ok(10):
            h_sl = slsqp_refine(h_cd, dx, target_sum, 200)
            cs2 = compute_c5(h_sl, dx)
            if cs2 < cs:
                h_cd, cs = h_sl, cs2
        if cs < best_c5:
            best_h, best_c5 = h_cd, cs

    # Phase 6: Final combined polish
    if time_ok(10):
        h_f = slsqp_refine(best_h, dx, target_sum, 400)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    if time_ok(5):
        h_f2, c_f2 = coord_optimize(best_h, dx, target_sum, 100, t0 + budget_s - 2, rng)
        if c_f2 < best_c5:
            best_h, best_c5 = h_f2, c_f2

    if time_ok(3):
        h_f3 = slsqp_refine(best_h, dx, target_sum, 200)
        c_f3 = compute_c5(h_f3, dx)
        if c_f3 < best_c5:
            best_h, best_c5 = h_f3, c_f3

    best_h = project_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
