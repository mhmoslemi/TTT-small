def compute_grad(h, w, N):
    # w: array of length 2N-1, w[idx] = weight for lag = idx - (N-1)
    # Returns gradient of weighted sum of correlations w.r.t. h
    grad = np.zeros(N)
    k = np.arange(N)
    L = np.arange(-(N-1), N)  # shape (2N-1,)
    # For each k and L:
    # t1 = [0<=k+L<N] * (1-h[k+L])
    # t2 = [0<=k-L<N] * h[k-L]
    # grad[k] += w[L+(N-1)] * (t1 - t2)
    # 
    # Vectorize: create (N, 2N-1) arrays
    # k terms: K = k[:,None], LL = L[None,:]
    # K + LL: shape (N, 2N-1)
    # K - LL: shape (N, 2N-1)
    
    K = k[:, None]  # (N, 1)
    LL = L[None, :]  # (1, 2N-1)
    
    idx1 = K + LL  # (N, 2N-1)
    idx2 = K - LL  # (N, 2N-1)
    
    mask1 = (idx1 >= 0) & (idx1 < N)
    mask2 = (idx2 >= 0) & (idx2 < N)
    
    # h[idx1] where mask1, 0 otherwise
    # Need to be careful with out-of-bounds
    safe_idx1 = np.where(mask1, idx1, 0)
    safe_idx2 = np.where(mask2, idx2, 0)
    
    t1 = mask1 * (1.0 - h[safe_idx1])  # (N, 2N-1)
    t2 = mask2 * h[safe_idx2]  # (N, 2N-1)
    
    grad = (t1 - t2) @ w  # (N,) -- each row dotted with w
    
    return grad

    if abs(diff) < 1e-12:

        else:

    L = np.arange(-(N-1), N)[None, :]

    idx2 = K - L

    i2 = K - L

    m2 = (i2 >= 0) & (i2 < N)

            hb = np.zeros(N)
            hb[rng.choice(N, int(tgt), replace=False)] = 1.0
            inits.append(hb)

        n_init = len(inits)
        per_init = max(4.0, (alloc - 4) / n_init)

        for hs in inits:
            t_now = time.time()
            if t_now > budget_s - 10:
                break
            dl_total = min(t_now + per_init, budget_s - 6)
            dl_adam = t_now + per_init * 0.50
            dl_sa = dl_total

            # Phase 1: Adam
            h1, c1 = adam_opt(hs.copy(), N, tgt, dx, dl_adam, rng)
            if c1 < best_c5:
                best_c5, best_h = c1, h1.copy()

            # Phase 2: SA polish
            h2, c2 = sa_opt(h1, N, tgt, dx, dl_sa, rng)
            c2f = compute_c5(h2, dx)
            if c2f < best_c5:
                best_c5, best_h = c2f, h2.copy()

    # Final: ensure constraint and evaluate
    Nf = len(best_h)
    dx_f = 2.0 / Nf
    best_h = project_sum(best_h, Nf / 2.0)
    final_c5 = compute_c5(best_h, dx_f)
    return best_h, final_c5, Nf
