def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def compute_gradient(h, beta=20.0):
    """Compute gradient of max corr using softmax weighting."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(corr); w[n-1] = 1
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    # For each position j, grad[j] = sum_k w[k] * [(1-h[j-k]) - h[j+k]] (with bounds)
    # Using the structure: corr[k] = sum_i h[i+k]*(1-h[i])
    # d/dh[j]: (1-h[j-k]) if 0<=j-k<n, minus h[j+k] if 0<=j+k<n
    
    # Vectorized computation
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)  # lag from -(n-1) to (n-1)
        wv = w[k_idx]
        if wv < 1e-15:
            continue
        # (1-h[j-k]) for j such that 0<=j-k<n, i.e., k<=j<n+k
        lo1 = max(0, k)
        hi1 = min(n, n+k)
        if lo1 < hi1:

        # -h[j+k] for j such that 0<=j+k<n, i.e., -k<=j<n-k

        hi2 = min(n, n-k)

                h = h_new

    return best_h, best_c5

def slsqp_polish(h0, dx, target_sum, maxiter=100):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and res.success:
            return project_feasible(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(400):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def compute_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = corr - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wv = w[k_idx]
        if wv < 1e-14:
            continue
        k = k_idx - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += wv * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wv * h[lo2 + k:hi2 + k]
    return grad

def projected_gradient(h0, dx, target_sum, max_iter=2500, t_deadline=None, perturb_scale=0.03):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improve = 0
    for it in range(max_iter):
        if t_deadline is not None and time.time() > t_deadline:
            break
        if no_improve > 800:
            break
        beta = 8.0 + 60.0 * min(1.0, it / 600.0)
        lr = 0.025 / (1.0 + 0.002 * it)
        g = compute_gradient(h, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        improved = False
        for shrink in [1.0, 0.5, 0.2, 0.05]:
            h_new = project_feasible(h - lr * shrink * g, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                no_improve = 0
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 100:
                h = project_feasible(h + perturb_scale * np.random.randn(n), target_sum)
                no_improve = 0
    return best_h, best_c5

def slsqp_polish(h0, dx, target_sum, maxiter=120):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None:
            h_res = project_feasible(res.x[:n], target_sum)
            if compute_c5(h_res, dx) < compute_c5(h0, dx):
                return h_res
    except Exception:
        pass
    return h0.copy()

def make_smooth_init(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    a1 = rng.uniform(0.1, 0.35)
    a2 = rng.uniform(0.0, 0.12)
    a3 = rng.uniform(0.0, 0.08)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    h = 0.5 + a1 * np.cos(np.pi * x + p1) + a2 * np.cos(2 * np.pi * x + p2) + a3 * np.cos(3 * np.pi * x)
    return project_feasible(h, n / 2.0)

def make_binary_init(n, rng):
    h = np.zeros(n)
    pos = rng.permutation(n)
    h[pos[:n // 2]] = 1.0
    return project_feasible(h, n / 2.0)
