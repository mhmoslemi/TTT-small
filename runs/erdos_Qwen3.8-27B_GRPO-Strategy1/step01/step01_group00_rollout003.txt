def fix_sum(h, n):

    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h.copy()
    h = h.copy()
    diff = target - s
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += diff / n_int
    else:
        h[:] += diff / n
    h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-10:
        interior = (h > 1e-6) & (h < 1.0 - 1e-6)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += (target - s2) / n_int
        h = np.clip(h, 0.0, 1.0)
    return h

def get_top_shifts(h, m=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-m:][::-1]
    shifts = top_idx - (n - 1)
    return shifts, corr[top_idx]

def subgradient_step(h, n, shifts, step_size):
    h_new = h.copy()
    j = np.arange(n)
    grad = np.zeros(n)
    for k in shifts:
        k = int(k)
        if abs(k) < n:
            hc = np.roll(h, -k)
            hm = np.roll(h, k)
            grad += (1.0 - hm) - hc
    grad /= len(shifts)
    h_new -= step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new

def subgradient_optimize(h_init, n, max_iters=3000, m_active=3, base_step=0.008):
    h_curr = h_init.copy()
    c5_curr = compute_c5(h_curr)
    step = base_step
    best_h = h_curr.copy()
    best_c5 = c5_curr
    patience = 0

    for it in range(max_iters):
        shifts, _ = get_top_shifts(h_curr, m=m_active)
        h_cand = subgradient_step(h_curr, n, shifts, step)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_curr:
            h_curr = h_cand
            c5_curr = c5_cand
            step = min(step * 1.03, 0.04)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                patience = 0
            else:
                patience += 1
        else:
            step *= 0.6
            patience += 1
        if step < 1e-7 or patience > 500:
            break
    return best_h, best_c5

def objective_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = (corr - c_max) / T
    shifted = np.clip(shifted, -50.0, 0.0)
    s = np.sum(np.exp(shifted))
    return (c_max + T * np.log(s)) * dx

def sum_constraint(h, n):
    return np.sum(h) - n * 0.5

def try_slsqp(h_start, n, T, max_iter=600, ftol=1e-13):
    try:
        res = opt.minimize(
            objective_smooth, h_start,
            args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': ftol}
        )
        h_res = np.clip(res.x, 0.0, 1.0)
        h_res = fix_sum(h_res, n)
        if compute_c5(h_res) < compute_c5(h_start):
            return h_res
    except Exception:
        pass
    return h_start

def try_cobyla(h_start, n, maxiter=200):
    try:
        res = opt.minimize(
            compute_c5, h_start,
            method='COBYLA',
            constraints=[
                {'type': 'ineq', 'fun': lambda h: h[i] for i in range(n)},
                {'type': 'ineq', 'fun': lambda h: 1.0 - h[i] for i in range(n)},
                {'type': 'eq', 'fun': lambda h: np.sum(h) - n * 0.5}
            ],
            options={'maxiter': maxiter, 'rhobeg': 0.01}
        )
        h_res = np.clip(res.x, 0.0, 1.0)
        h_res = fix_sum(h_res, n)
        if compute_c5(h_res) < compute_c5(h_start):
            return h_res
    except Exception:
        pass
    return h_start

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    target = new_n * 0.5
    s = np.sum(h_new)
    if s > 1e-15:
        h_new *= target / s
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, new_n)

def binary_shake(h, n, noise_level=0.3):
    h_b = (h > 0.5).astype(float)
    s = np.sum(h_b)
    if s > 0:
        h_b *= (n * 0.5) / s
    h_b = np.clip(h_b, 0.0, 1.0)
    h_shake = h_b + np.random.randn(n) * noise_level
    h_shake = np.clip(h_shake, 0.0, 1.0)
    return fix_sum(h_shake, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t_start

    # Phase 1: Subgradient at initial resolution (aggressive)
    if budget_s - elapsed() > 250:
        h_sg, c5_sg = subgradient_optimize(h_best, n_best, max_iters=4000, m_active=3, base_step=0.01)
        if c5_sg < c5_best:
            h_best = h_sg
            c5_best = c5_sg

    # Phase 2: SLSQP with descending temperature
    for T in [0.05, 0.02, 0.005, 0.001, 0.0003]:
        if budget_s - elapsed() < 100:
            break
        h_opt = try_slsqp(h_best.copy(), n_best, T, max_iter=800, ftol=1e-14)
        h_opt = fix_sum(h_opt, n_best)
        c5_c = compute_c5(h_opt)
        if c5_c < c5_best:
            h_best = h_opt
            c5_best = c5_c

    # Phase 3: More subgradient with more active shifts
    if budget_s - elapsed() > 180:
        h_sg, c5_sg = subgradient_optimize(h_best, n_best, max_iters=5000, m_active=5, base_step=0.004)
        if c5_sg < c5_best:
            h_best = h_sg
            c5_best = c5_sg

    # Phase 4: Binary shake to escape local optima
    n_shakes = 4
    for i in range(n_shakes):
        if budget_s - elapsed() < 200:
            break
        h_sh = binary_shake(h_best, n_best, noise_level=0.5)
        h_sg2, c5_sg2 = subgradient_optimize(h_sh, n_best, max_iters=2000, m_active=3, base_step=0.008)
        if c5_sg2 < c5_best:
            h_opt2 = try_slsqp(h_sg2, n_best, 0.01, max_iter=500)
            h_opt2 = fix_sum(h_opt2, n_best)
            c5_c2 = compute_c5(h_opt2)
            if c5_c2 < c5_best:
                h_best = h_opt2
                c5_best = c5_c2

    # Phase 5: Upsample to higher resolution
    for n_try in [420, 560, 700]:
        if budget_s - elapsed() < 180:
            break
        h_up = upsample(h_best, n_try)
        h_sg3, c5_sg3 = subgradient_optimize(h_up, n_try, max_iters=1500, m_active=3, base_step=0.006)
        h_opt3 = try_slsqp(h_sg3, n_try, 0.005, max_iter=500)
        h_opt3 = fix_sum(h_opt3, n_try)
        c5_c3 = compute_c5(h_opt3)
        if c5_c3 < c5_best:
            h_best = h_opt3
            c5_best = c5_c3
            n_best = n_try

    # Phase 6: Final fine optimization
    if budget_s - elapsed() > 80:
        h_sg4, c5_sg4 = subgradient_optimize(h_best, n_best, max_iters=6000, m_active=2, base_step=0.002)
        if c5_sg4 < c5_best:
            h_best = h_sg4
            c5_best = c5_sg4
        h_final = try_slsqp(h_best.copy(), n_best, 0.0002, max_iter=1000, ftol=1e-15)
        h_final = fix_sum(h_final, n_best)
        c5_final_test = compute_c5(h_final)
        if c5_final_test < c5_best:
            h_best = h_final
            c5_best = c5_final_test

    # Ensure validity
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, n_best)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n_best)
