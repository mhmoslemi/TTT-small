    dc = c[:, None] - c[None, :]
    dq = q[None, :] - q[:, None]
    valid = np.abs(dc) > 1e-12
    t_mat = np.full((m, m), np.nan)
    t_mat[valid] = dq[valid] / dc[valid]
    t_flat = t_mat.ravel()
    valid_t = np.isfinite(t_flat) & (t_flat > 1e-12) & (t_flat < 1.0 - 1e-12)
    candidates.extend(t_flat[valid_t].tolist())
    
    if len(candidates) <= 2:
        best_t = 0.0
        best_val = np.inf
        for t in candidates:
            val = np.max(q + c * t)
            if val < best_val:
                best_val = val
                best_t = t
        return best_t
    
    candidates_arr = np.array(candidates)
    # Evaluate max(q + c*t) for all candidates
    vals = np.max(q[None, :] + c[None, :] * candidates_arr[:, None], axis=1)
    best_idx = np.argmin(vals)
    return candidates_arr[best_idx]

def gauss_seidel_pass(h, target_sum):
    """One full pass of Gauss-Seidel coordinate descent"""
    n = len(h)
    n_lags = 2 * n - 1
    h = h.copy()
    
    for j in range(n):
        L = np.arange(-(n-1), n)  # lags
        jpL = j + L
        jmL = j - L
        
        # c_L = 1 - h[j+L]*1_{valid} - h[j-L]*1_{valid}
        c_coeff = np.ones(n_lags)
        mask_jp = (jpL >= 0) & (jpL < n)
        c_coeff[mask_jp] -= h[jpL[mask_jp]]
        mask_jm = (jmL >= 0) & (jmL < n)
        c_coeff[mask_jm] -= h[jmL[mask_jm]]
        
        # f(L) = correlate(h, 1-h, mode='full')
        corr = np.correlate(h, 1.0 - h, mode='full')
        q_vals = corr - c_coeff * h[j]
        
        t_opt = solve_1d_max_lines(q_vals, c_coeff)
        h[j] = t_opt
    
    return h

def coordinate_descent(h0, dx, target_sum, max_passes=200, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improve = 0
    
    for it in range(max_passes):
        if t_deadline is not None and time.time() > t_deadline:
            break
        h = gauss_seidel_pass(h, target_sum)
        h = project_simplex(h, target_sum)
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                break
    return best_h, best_c5

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None and res.success:
            return project_simplex(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()
