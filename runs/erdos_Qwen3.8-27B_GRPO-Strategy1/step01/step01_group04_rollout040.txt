def smooth_objective(h, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

    cmax = np.max(corr)
    shifted = corr - cmax
    smax = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    # Constraint penalty
    s = np.sum(h) - n / 2.0
    return smax * dx + penalty * s * s

    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def repair(h):

    target = len(h) / 2.0

from scipy.optimize import minimize

    return float(np.max(corr))

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    val = np.sum(corr ** p) ** (1.0 / p)
    s = np.sum(h) - n / 2.0
    return val + pen * s * s

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(corr)
    sm = cm + temp * np.log(np.sum(np.exp((corr - cm) / temp)))

def obj_lp(h, p, pen):
    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def obj_lp(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    val = np.sum(corr ** p) ** (1.0 / p)
    s = np.sum(h) - n / 2.0
    return val + pen * s * s

def obj_lse(h, tmp, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(corr)
    sm = cm + tmp * np.log(np.sum(np.exp((corr - cm) / tmp)))
    s = np.sum(h) - n / 2.0
    return sm + pen * s * s

def upsc(h, f):
    n, m = len(h), len(h) * f
    xo = (np.arange(n) + 0.5) * 2.0 / n
    xn = (np.arange(m) + 0.5) * 2.0 / m
    return np.interp(xn, xo, h)

def fix(h):
    h = np.clip(h, 0.0, 1.0)
    t = len(h) / 2.0
    s = np.sum(h)
    if s > 1e-15:
        h = np.clip(h * (t / s), 0.0, 1.0)
    d = t - np.sum(h)
    if abs(d) > 1e-13:
        k = max(1, min(len(h), int(abs(d)) + 1))
        idx = np.argsort(h) if d > 0 else np.argsort(-h)
        h[idx[:k]] = np.clip(h[idx[:k]] + d / k, 0, 1)
    return h

def bproj(h):
