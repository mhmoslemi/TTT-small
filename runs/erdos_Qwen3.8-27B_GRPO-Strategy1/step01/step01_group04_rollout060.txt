def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

def optimize_l_bfgsb(h_in, t0, time_limit, schedules):
    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    cur = h_in.copy()
    best_local = compute_c5(cur)
    for temp, pen, maxit in schedules:
        if time.time() - t0 > time_limit:
            break
        res = minimize(
            smooth_c5, cur, args=(temp, pen),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-9, 'maxfun': maxit * 3}
        )
        nh = np.clip(res.x, 0.0, 1.0)
        nh = repair_h(nh)
        c5v = compute_c5(nh)
        if c5v < best_local:
            best_local = c5v
        cur = nh
    return cur, best_local

def structured_perturb(h, seed_rng):
    n = len(h)
    r = seed_rng.random()
    if r < 0.4:
        # Gaussian noise
        sigma = seed_rng.uniform(0.05, 0.2)
        h2 = h + seed_rng.randn(n) * sigma
    elif r < 0.7:
        # Shift
        k = seed_rng.randint(-n // 4, n // 4 + 1)
        h2 = np.roll(h, k)
    elif r < 0.85:
        # Inversion
        h2 = 1.0 - h
    else:
        # Multi-scale: blend with a shifted version
        k = seed_rng.randint(1, n // 3 + 1)
        h_shifted = np.roll(h, k)
        alpha = seed_rng.uniform(0.3, 0.7)
        h2 = alpha * h + (1 - alpha) * h_shifted
    return repair_h(np.clip(h2, 0, 1))

def coordinate_polish(h, t0, time_limit):
    """Try to reduce C5 by adjusting individual coordinates."""
    n = len(h)

    target_sum = n / 2.0

        active_lags = np.where(corr >= threshold)[0]

                best_c5 = lb
                best_h = h_p.copy()
                cur_h = h_p.copy()

    # Phase 4: Coordinate polish on best solution
    if time.time() - t0 < t0 + budget_s * 0.88:
        polish_limit = t0 + budget_s * 0.09
        best_h, pc5 = coordinate_polish(best_h, t0, polish_limit)
        if pc5 < best_c5:
            best_c5 = pc5

    # Phase 5: One more short L-BFGS polish at very low temperature
    if time.time() - t0 < t0 + budget_s * 0.93:
        sched_final = [(0.0005, 30000, 80)]
        best_h, lb = run_lbfgs(best_h, t0, t0 + budget_s * 0.05, sched_final)
        if lb < best_c5:
            best_c5 = lb
            best_h = cur_h.copy() if lb < compute_c5(best_h) else best_h

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    elapsed = time.time() - t0
    if elapsed > budget_s:
        pass  # return best found

    return best_h, best_c5, len(best_h)
