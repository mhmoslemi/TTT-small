import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(1000):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_pen(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = temp * (corr - mx)
    val = (mx + np.log(np.sum(np.exp(s))) / temp) * dx
    dev = np.sum(h) - n / 2.0
    val += penalty * dev * dev
    return val

def grad_pen(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = temp * (corr - mx)
    w = np.exp(s) / np.sum(np.exp(s))
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk < 1e-16:
            continue
        grad[k:] += wk * (1.0 - h[:n - k])
        grad[:n - k] += -wk * h[k:]
    grad *= dx
    dev = np.sum(h) - n / 2.0
    grad += 2.0 * penalty * dev
    return grad

def active_set_step(h, n, lr, n_top):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = np.argsort(corr)[-n_top:]
    grad = np.zeros(n)
    for m in active:
        k = int(m) - (n - 1)
        ak = abs(k)
        if ak == 0:
            grad += (1.0 - 2.0 * h)
        else:
            grad[ak:] += (1.0 - h[:n - ak])
            grad[:n - ak] += -h[ak:]
    grad /= len(active)
    h_new = project(h - lr * grad, n)
    return h_new, compute_c5(h_new), compute_c5(h)

def binary_local_search(h, n, max_passes=30):
    h = h.copy()
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:n // 2]] = 1.0
    c_cur = compute_c5(h_bin)
    ones = np.where(h_bin == 1.0)[0]
    for p in range(max_passes):
        improved = False
        for i_idx in range(len(ones)):
            j = ones[i_idx]
            for k in range(n):
                if h_bin[k] != 0.0:
                    continue
                h_bin[j] = 0.0
                h_bin[k] = 1.0
                c_test = compute_c5(h_bin)
                if c_test < c_cur - 1e-14:
                    c_cur = c_test
                    ones[i_idx] = k
                    improved = True
                    break
                else:
                    h_bin[j] = 1.0
                    h_bin[k] = 0.0
        if not improved:
            break
    return h_bin, c_cur

def xinterp(h, nd):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return project(np.interp(xd, xs, h), nd)
