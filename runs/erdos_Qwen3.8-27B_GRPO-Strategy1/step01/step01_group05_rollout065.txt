def compute_c5(h, n):

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(100):

corr[k] = sum_n h[n+k] * (1-h[n])

If I change h[i] from 1 to 0 and h[j] from 0 to 1:
- For each shift k: corr[k] changes by the terms involving h[i] and h[j]
- h[i] appears as h[n+k] when n = i-k, and as h[n] when n = i
- h[j] appears as h[n+k] when n = j-k, and as h[n] when n = j

So for each k:
Δcorr[k] = (contribution of j as h[n+k]) + (contribution of i as h[n]) - (contribution of i as h[n+k]) - (contribution of j as h[n])

Specifically:

def compute_c5(h, n):

    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h

def smooth_obj(x, n, temp):
    corr = np.asarray(np.correlate(x, 1.0 - x, mode='full'), dtype=float)
    s = corr * temp
    m = np.max(s)
    return (m + np.log(np.sum(np.exp(s - m)))) / temp * (2.0 / n)

def obj_w(x, n, temp):
    return smooth_obj(x, n, temp)

def con_w(x, n):
    return np.sum(x) - n / 2.0

def slsqp_step(x0, n, temp, maxiter):

    res = minimize(obj_w, x0.copy(), args=(n, temp), method='SLSQP',
                   constraints=[{'type': 'eq', 'fun': con_w, 'args': (n,)}],
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x

            stall = 0

        temp = max(temp * cooling, min_temp)
    return best_h, best_c5

def hill_climb(h, n, rng, t_end):
    h = h.copy()
    c5_cur = compute_c5(h, n)
    best_c5 = c5_cur
    best_h = h.copy()
    n1 = n // 2
    while time.time() < t_end:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(min(500, len(ones) * len(zeros))):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h, n)
            if c5_new < c5_cur:
                c5_cur = c5_new
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                improved = True
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    init_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float).copy()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    n_list = [44, 48, 52, 56, 64, 80]
    margin = 40
    time_available = budget_s - margin

    weights = np.array([1.0 / np.sqrt(ni) for ni in n_list])
    weights = weights / weights.sum() * time_available
    weights = np.clip(weights, 12.0, 220.0)
    weights = weights / weights.sum() * time_available

    for idx_n in range(len(n_list)):
        n = n_list[idx_n]
        elapsed = time.time() - t_start
        if elapsed > budget_s - margin:
            break
        alloc = min(weights[idx_n], budget_s - margin - elapsed)
        if alloc < 8:
            continue

        h_ini_n = None
        if init_h is not None:
            h_ini_n = resample_to_n(init_h, n, rng)

        # SA phases
        sa_time = alloc * 0.55
        n_sa = max(2, int(sa_time / 25))
        per_sa = sa_time / n_sa
        h_sa_best = h_ini_n if h_ini_n is not None else get_binary(n, rng)
        c5_sa_best = compute_c5(h_sa_best, n)

        for r in range(n_sa):
            t_end_sa = time.time() + min(per_sa, budget_s - 50 - (time.time() - t_start))
            if time.time() > budget_s - 50:
                break
            if r == 0 and h_ini_n is not None:
                h_r, c5_r = sa_run(n, rng, t_end_sa, init_h=h_ini_n)
            else:
                h_r, c5_r = sa_run(n, rng, t_end_sa)
            if c5_r < c5_sa_best:
                c5_sa_best = c5_r
                h_sa_best = h_r

        # Hill climb polish on best SA result
        if time.time() < budget_s - 40:
            hc_time = min(15, budget_s - 40 - (time.time() - t_start))
            if hc_time > 2:
                h_hc, c5_hc = hill_climb(h_sa_best, n, rng, time.time() + hc_time)
                if c5_hc < c5_sa_best:
                    c5_sa_best = c5_hc
                    h_sa_best = h_hc

        if c5_sa_best < best_c5:
            best_c5 = c5_sa_best
            best_h = h_sa_best.copy()
            best_n = n

        # SLSQP refinement
        if time.time() < budget_s - 30:
            h_s = repair_h(h_sa_best, n)
            for temp in [20, 50, 120]:
                if time.time() > budget_s - 20:
                    break
                maxiter = min(350, max(80, int((budget_s - 25 - (time.time() - t_start)) * 5)))
                h_t = slsqp_step(h_s, n, temp, maxiter)
                h_t = repair_h(h_t, n)
                c5_t = compute_c5(h_t, n)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t.copy()
                    best_n = n
                    h_s = h_t.copy()

        # COBYLA from best binary
        if time.time() < budget_s - 25:
            res = minimize(obj_w, repair_h(h_sa_best, n), args=(n, 40),
                           method='COBYLA',
                           options={'maxiter': 500, 'rhobeg': 0.02, 'disp': False})
            h_cb = repair_h(res.x, n)
            c5_cb = compute_c5(h_cb, n)
            if c5_cb < best_c5:
                best_c5 = c5_cb
                best_h = h_cb.copy()
                best_n = n

    # Final SLSQP polish on overall best
    if best_h is not None and time.time() < budget_s - 10:
        remaining = budget_s - 10 - (time.time() - t_start)
        if remaining > 15:
            maxiter = max(100, int(remaining * 6))
            h_t = slsqp_step(best_h, best_n, 60, maxiter)
            h_t = repair_h(h_t, best_n)
            c5_t = compute_c5(h_t, best_n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t

    if best_h is None:
        best_h = get_binary(44, rng)
        best_n = 44
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair_h(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
