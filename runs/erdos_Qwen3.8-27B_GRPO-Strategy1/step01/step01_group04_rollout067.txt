from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_and_lag(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    return float(corr[max_idx] * dx), max_idx

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h
    h2 = h / integ
    h2 = np.clip(h2, 0.0, 1.0)
    # Re-normalize after clipping
    new_integ = np.sum(h2) * dx
    if new_integ > 0:
        h2 = h2 / new_integ
    return np.clip(h2, 0.0, 1.0)

def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    return sm * dx

def optimize_lfgsb(h, temp, maxiter, bounds):
    res = minimize(smooth_c5, h, args=(temp,),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
    return np.clip(res.x, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return np.ones(n) * 0.5
    h2 = h / integ
    h2 = np.clip(h2, 0.0, 1.0)
    return h2

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    return sm * dx

def lfgs_optimize(h, temp, maxiter, bounds):
    res = minimize(smooth_obj, h, args=(temp,),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
    return np.clip(res.x, 0.0, 1.0)

def binarize(h):
    """Convert to 0/1 with exactly n/2 ones"""
    n = len(h)
    target = n // 2  # exactly n/2 ones
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_local_search(h, max_iters, time_limit):
    """Greedy swap local search on 0/1 sequence"""
    n = len(h)
    current_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    stagnation = 0
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        if stagnation > 500:
            break

        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        
        if new_c5 <= current_c5 + 1e-15:
            current_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stagnation = 0
        else:
            h[i], h[j] = h[j], h[i]
            stagnation += 1
    
    return h, current_c5

    target = n // 2
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

c[k] = sum_m a[m] * conj(v[m - k + len(v) - 1])

So for a=h, v=1-h, len(v)=n:
c[k] = sum_m h[m] * (1-h[m - k + n - 1])

For k = n-1 (the center): c[n-1] = sum_m h[m] * (1-h[m]), which is 0 for 0/1 h.
For k = n-1+j: c[n-1+j] = sum_m h[m] * (1-h[m-j]) = sum_{m=j}^{n-1} h[m] * (1-h[m-j]) = sum_{i=0}^{n-1-j} h[i+j] * (1-h[i])

This is the correlation at "positive lag" j.
For k = n-1-j: c[n-1-j] = sum_m h[m] * (1-h[m+j]) = sum_{i=0}^{n-1-j} h[i] * (1-h[i+j])

This is the correlation at "negative lag" j.

So the full output contains correlations for lags from -(n-1) to +(n-1).

The C5 computation: max of all these, times dx. This includes lag 0 (which is sum h(1-h) = 0 for binary) and all other lags.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    return sm * dx

def binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def repair_h(h):
    n = len(h)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, 0.5)
    h2 = h * (target / s)
    return np.clip(h2, 0.0, 1.0)

def greedy_swap(h, max_rounds, time_limit):
    n = len(h)
    dx = 2.0 / n
    c5_now = compute_c5(h)
    for _ in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = -1e-12
        best_ij = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5_t = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
                d = c5_t - c5_now
                if d < best_delta:
                    best_delta = d
                    best_ij = (i, j)
                h[i], h[j] = 1.0, 0.0
        if best_ij is not None:
            i, j = best_ij
            h[i], h[j] = 0.0, 1.0
            c5_now += best_delta
            improved = True
        if not improved:
            break
    return h, c5_now

def sa_swap(h, n_iter, t0_val, t1_val, time_limit):

    dx = 2.0 / n
    c5_now = compute_c5(h)
    c5_best = c5_now
    h_best = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        T = t0_val * (t1_val / t0_val) ** (it / max(n_iter - 1, 1))
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5_t = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        d = c5_t - c5_now
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-15)):
            c5_now = c5_t
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return h_best, c5_best
