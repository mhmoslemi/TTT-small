import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection."""
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    # Numerical stability
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sm = cmax + temp * np.log(np.sum(np.exp((corr - cmax) / temp)))
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def subgradient_step(h, dx, step_size):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    l = abs(lag)
    if l == 0:
        grad = 1.0 - 2.0 * h
    elif lag > 0:
        ne = n - l
        grad = np.zeros(n)
        grad[:ne] = (1.0 - h[:ne]) - h[l:n]
    else:
        ne = n - l
        grad = np.zeros(n)
        grad[:ne] = (1.0 - h[l:n])
        grad[l:n] -= h[:ne]
    gn = np.linalg.norm(grad)
    if gn > 1e-12:
        h_new = h - step_size * grad / gn
    else:
        h_new = h.copy()
    return project_h(h_new, n)

def binary_local_search(h, t0, time_limit):
    n = len(h)
    b = (h > 0.5).astype(float)
    n_ones = int(np.sum(b))
    target = n // 2
    idx = np.random.permutation(n)
    p = 0
    while n_ones != target and p < n:
        if n_ones > target and b[idx[p]] == 1:
            b[idx[p]] = 0; n_ones -= 1
        elif n_ones < target and b[idx[p]] == 0:
            b[idx[p]] = 1; n_ones += 1
        p += 1
    best_c5 = compute_c5(b)
    one_idx = np.where(b == 1)[0]
    zero_idx = np.where(b == 0)[0]
    for _ in range(20000):
        if time.time() - t0 > time_limit:
            break
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        i = one_idx[np.random.randint(len(one_idx))]
        j = zero_idx[np.random.randint(len(zero_idx))]
        b[i] = 0; b[j] = 1
        c5_new = compute_c5(b)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            one_idx = np.where(b == 1)[0]
            zero_idx = np.where(b == 0)[0]
        else:
            b[i] = 1; b[j] = 0
    return b, best_c5

def optimize_smooth(h, t0, total_time):
    n = len(h)
    cur = h.copy()
    best = compute_c5(cur)
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n
    schedules = [
        (0.05, 500.0, 200),
        (0.02, 2000.0, 150),
        (0.005, 8000.0, 150),
        (0.001, 20000.0, 100),
    ]
    for temp, pen, maxit in schedules:
        if time.time() - t0 > total_time * 0.85:
            break
        res = minimize(smooth_objective, cur, args=(temp, pen),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10})
        nh = project_h(np.clip(res.x, 0.0, 1.0), n)
        c5v = compute_c5(nh)
        if c5v < best:
            best = c5v; cur = nh
        # Subgradient polish
        for sg in range(200):
            if time.time() - t0 > total_time * 0.9:
                break
            step = 0.1 / (sg + 10)
            cur = subgradient_step(cur, dx, step)
            c5v = compute_c5(cur)
            if c5v < best:
                best = c5v
    return cur, best

def golden_start(n):
    """Create a spread-out binary pattern using golden ratio."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    k = n // 2
    positions = np.zeros(k, dtype=int)
    for i in range(k):
        positions[i] = int((i * phi * n) % n)
    # Deduplicate
    positions = np.unique(positions)
    # If we have fewer than k, add more
    while len(positions) < k:
        extra = np.random.randint(0, n)
        if extra not in positions:
            positions = np.append(positions, extra)
    positions = positions[:k]
    h = np.zeros(n)
    h[positions] = 1.0
    return h

    n = len(h_init)

    cur_h, c5_1 = optimize_smooth(h_init, t0, budget_s * 0.3)
    if c5_1 < best_c5:

    # Phase 2: Binary local search

        b1, c5_b1 = binary_local_search(cur_h, t0, budget_s * 0.12)
        if c5_b1 < best_c5:
            best_c5 = c5_b1; best_h = b1.copy()
        b2, c5_b2 = binary_local_search(h_init, t0, budget_s * 0.08)
        if c5_b2 < best_c5:
            best_c5 = c5_b2; best_h = b2.copy()
    
    # Phase 3: Try golden ratio start
    if time.time() - t0 < budget_s * 0.65:
        h_gold = golden_start(n)
        h_gold_smooth = h_gold + np.random.randn(n) * 0.1
        h_gold_smooth = project_h(h_gold_smooth, n)
        _, c5_g = optimize_smooth(h_gold_smooth, t0, budget_s * 0.1)
        if c5_g < best_c5:
            best_c5 = c5_g; best_h = h_gold_smooth.copy()
        b3, c5_b3 = binary_local_search(h_gold, t0, budget_s * 0.06)
        if c5_b3 < best_c5:
            best_c5 = c5_b3; best_h = b3.copy()
    
    # Phase 4: Refine best with smooth + binary
    if time.time() - t0 < budget_s * 0.82:
        cur = best_h + np.random.randn(n) * 0.03
        cur = project_h(cur, n)
        _, c5_r = optimize_smooth(cur, t0, budget_s * 0.12)
        if c5_r < best_c5:
            best_c5 = c5_r; best_h = cur.copy()
    
    # Phase 5: Final binary polish
    if time.time() - t0 < budget_s * 0.92:

        if c5_f < best_c5:
            best_c5 = c5_f; best_h = b5.copy()

    best_h = project_h(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, n

    # Phase 4: Block pattern start (8% budget)
    d4 = t0 + budget_s * 0.65
    if time.time() < d4:
        h_blk = block_start(n)
        b4, c5_b4 = binary_local_search(h_blk, d4)
        if c5_b4 < best_c5:
            best_c5 = c5_b4; best_h = b4.copy()

    # Phase 5: Refine current best (12% budget)
    d5 = t0 + budget_s * 0.77
    if time.time() < d5:
        cur = best_h + np.random.randn(n) * 0.04
        cur = project_h(cur, n)
        _, c5_r = optimize_smooth(cur, t0, d5)
        if c5_r < best_c5:
            best_c5 = c5_r; best_h = cur.copy()

    # Phase 6: Extended binary polish (10% budget)
    d6 = t0 + budget_s * 0.87
    if time.time() < d6:
        b6, c5_f = binary_local_search(best_h, d6)
        if c5_f < best_c5:
            best_c5 = c5_f; best_h = b6.copy()

    # Phase 7: Final subgradient polish (5% budget)
    d7 = t0 + budget_s * 0.94
    if time.time() < d7:
        dx = 2.0 / n
        cur = best_h.copy()
        for i in range(800):
            if time.time() > d7:
                break
            step = 0.05 / (i + 5)
            cur = subgradient_step(cur, dx, step)
            c5v = compute_c5(cur)
            if c5v < best_c5:
                best_c5 = c5v; best_h = cur.copy()

    best_h = project_h(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
