import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T > 1e-10:
        shifted = corr - cmax
        logsum = np.log(np.sum(np.exp(shifted / T)))
        obj = (cmax + T * logsum) * dx
    else:
        obj = cmax * dx
    obj += pen * (np.sum(h) - n * 0.5) ** 2
    return obj

def fix_sum(h, n):
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    h = h.copy()
    lo, hi = -0.5, 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        ht = np.clip(h + mid, 0.0, 1.0)
        if ht.sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = interior.sum()
        if cnt > 0:
            h[interior] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def lbfhs_optimize(h0, n, T, pen, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(
        soft_max_obj, h0.copy(), args=(T, pen),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': maxiter * 3}
    )
    return res.x

def coordinate_polish(h, n, iterations=200):
    """Try small adjustments to pairs of coordinates to reduce max correlation."""
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h)
    for _ in range(iterations):
        improved = False
        for _ in range(n):
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            for delta in [0.01, 0.005, 0.002]:
                h2 = h.copy()
                h2[i] += delta
                h2[j] -= delta
                if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                    c2 = compute_c5(h2)
                    if c2 < best_c - 1e-9:
                        h = h2
                        best_c = c2
                        improved = True
                        break
                for delta2 in [-0.01, -0.005, -0.002]:
                    h2 = h.copy()
                    h2[i] += delta2
                    h2[j] -= delta2
                    if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                        c2 = compute_c5(h2)
                        if c2 < best_c - 1e-9:
                            h = h2
                            best_c = c2
                            improved = True
                            break
                    if improved:
                        break
        if not improved:
            break
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)
