import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_simplex(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_smooth_max(h, dx, beta):
    n = len(h)
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    w = np.exp(beta * (vals - m))
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(n_corr):
        wi = w[idx]
        if wi < 1e-13:
            continue
        s = idx - (n - 1)
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[0:n-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[0:n-s]
        else:
            sn = -s
            grad[sn:] += wi * (1.0 - h[0:n-sn])
            grad[0:n-sn] -= wi * h[sn:]
    return grad * dx

def lbfgs_smooth(h0, dx, target_sum, beta, lam, maxiter=5000):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    def f(x):
        sm = compute_smooth_max(x, dx, beta)
        pen = lam * (x.sum() - target_sum) ** 2
        return sm + pen
    def g(x):
        gx = grad_smooth_max(x, dx, beta)
        gp = 2.0 * lam * (x.sum() - target_sum)
        return gx + gp * np.ones(n)
    try:
        res = minimize(f, h0, jac=g, method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0.copy()

def compute_smooth_max(h, dx, beta):

    vals = corr * dx
    m = vals.max()
    return m + np.log(np.sum(np.exp(vals - m))) / beta

def slsqp_exact(h0, dx, target_sum, maxiter=400):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        h = x[:n]
        t = x[n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr * dx
    def eq(x):
        return np.sum(x[:n]) - target_sum
    t_init = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t_init + 0.02]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 5.0)]
    cons = ({'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq})
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            return project_simplex(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()
