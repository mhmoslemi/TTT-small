import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_pnorm(h, dx, p):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = np.maximum(corr * dx, 1e-14)
    F = float(np.power(np.sum(g ** p), 1.0 / p))
    dF_dg = np.power(g, p - 1) / np.power(F, p - 1)
    grad = np.zeros(N)
    for k in range(-(N - 1), N):
        w = dF_dg[k + N - 1] * dx
        if w < 1e-16:
            continue
        if k >= 0:
            grad[:N - k] += w * (1.0 - h[k:N])
            grad[k:] -= w * h[:N - k]
        else:
            l = -k
            grad[l:] += w * (1.0 - h[:N - l])
            grad[:N - l] -= w * h[l:]
    return F, grad

    h = project_onto_constraint(h_init.copy(), target)
    h_prev = h.copy()
    t = 1.0
    step = step_init
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(max_iter):
        if time.time() - t0 > budget - 5:
            break
        frac = min(1.0, it / (0.3 * max_iter))
        p_cur = int(round(p_start + (p_end - p_start) * frac))
        alpha = (t - 2.0) / t if t > 1 else 0.0
        y = np.clip(h + alpha * (h - h_prev), 0.0, 1.0)
        F, grad = grad_pnorm(y, dx, p_cur)
        s = step
        z = project_onto_constraint(y - s * grad, target)
        Fz, _ = grad_pnorm(z, dx, p_cur)

            z = project_onto_constraint(y - s * grad, target)

        h = z
        t += 1.0

            best_h = h.copy()
