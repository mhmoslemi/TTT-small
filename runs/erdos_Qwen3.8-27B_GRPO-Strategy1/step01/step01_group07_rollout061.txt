def compute_grad_pnorm(x, n, p):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = len(corr)  # 2n-1
    abs_corr = np.maximum(corr, 1e-30)

    factor = dx * S**(1.0/p - 1.0) / m
    w = abs_corr**(p-1)  # length 2n-1
    
    # For each j in [0,n-1], grad[j] = factor * sum_k w[k] * ((1-x[j-m_k])_valid - x[j+m_k]_valid)

    # This is (n, 2n-1) which for n=280 is 280*559 ≈ 156K entries. Fine.
    
    js = np.arange(n)[:, None]  # (n, 1)
    ms = (np.arange(m) - (n-1))[None, :]  # (1, m)
    
    # term1: (1-x[j-m_k]) where j-m_k in [0,n-1]
    idx1 = js - ms  # (n, m)
    valid1 = (idx1 >= 0) & (idx1 < n)
    # Need to gather x[idx1] safely
    idx1_clipped = np.clip(idx1, 0, n-1)
    term1 = np.where(valid1, 1.0 - x[idx1_clipped], 0.0)  # (n, m)
    
    # term2: x[j+m_k] where j+m_k in [0,n-1]
    idx2 = js + ms  # (n, m)

        if abs(diff) < 1e-10:
            break
        if diff > 0:

            room = 1.0 - h
            mask = room > 1e-6

        else:

            room = h
            mask = room > 1e-6
            h[mask] += diff * room[mask] / room[mask].sum()
    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def obj_pnorm(x, n, p, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = 2 * n - 1
    c = np.maximum(corr, 1e-28)
    val = dx * (np.sum(c**p) / m) ** (1.0 / p)
    val += pen * (np.sum(x) - n / 2.0) ** 2
    return val

def grad_pnorm(x, n, p, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = 2 * n - 1
    c = np.maximum(corr, 1e-28)
    S = np.sum(c**p) / m
    w = c ** (p - 1)
    factor = dx * S ** (1.0 / p - 1.0) / m
    js = np.arange(n)[:, None]
    ms = (np.arange(m) - (n - 1))[None, :]
    idx1 = js - ms
    v1 = (idx1 >= 0) & (idx1 < n)
    t1 = np.where(v1, 1.0 - x[np.clip(idx1, 0, n - 1)], 0.0)
    idx2 = js + ms
    v2 = (idx2 >= 0) & (idx2 < n)
    t2 = np.where(v2, x[np.clip(idx2, 0, n - 1)], 0.0)
    g = factor * ((t1 - t2) @ w)
    g += 2.0 * pen * (np.sum(x) - n / 2.0)
    return g

def upsample(h, nn):
    on = len(h)
    if on == nn:
        return h.copy()
    ox = np.arange(on, dtype=np.float64) / on
    nx = np.arange(nn, dtype=np.float64) / nn
    v = np.interp(nx, ox, h)
    v += (nn / 2.0 - np.sum(v)) / nn
    v = np.clip(v, 0.0, 1.0)
    v += (nn / 2.0 - np.sum(v)) / nn
    return np.clip(v, 0.0, 1.0)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-8
            if not mask.any():
                break
            h[mask] += diff * room[mask] / room[mask].sum()
        else:
            room = h
            mask = room > 1e-8
            if not mask.any():
                break
            h[mask] += diff * room[mask] / room[mask].sum()
    return np.clip(h, 0.0, 1.0)

    rng = np.random.RandomState(seed)

    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_c5 = compute_c5(parent)
    best_h = parent.copy()
    best_n = pn

    # Resolutions to try
    ns = []
    for m in [1, 2, 3, 4]:
        ns.append(pn * m)
    for extra in [70, 100, 140, 200, 300, 400, 500]:
        ns.append(extra)
    ns = sorted(set(x for x in ns if 50 <= x <= 600))

    penalty = 1e6
    p_seq = [5, 10, 20, 50, 100, 200]

    for n in ns:
        if time.time() - t0 > budget_s - 80:
            break
        h0 = parent.copy() if n == pn else upsample(parent, n)

        # Build diverse initializations
        inits = [h0]
        for s in [0.005, 0.012, 0.025]:
            hp = h0 + rng.randn(n) * s
            inits.append(fix_sum(np.clip(hp, 0, 1), n))

                    h_new = fix_sum(np.clip(res.x, 0, 1), n)

                h[j] += d

            valid_b = (idx_b >= 0) & (idx_b < n)

            h_try = fix_sum(np.clip(h_try, 0, 1), n)
            c_try = compute_c5(h_try)
            if c_try < cur:
                h = h_try
                cur = c_try
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()

    return best_h, best_c5, best_n
