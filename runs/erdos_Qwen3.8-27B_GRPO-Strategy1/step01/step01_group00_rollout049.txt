import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def objective_soft(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max
    shifted = np.clip((corr - c_max) / T, -60, 60)
    s = np.sum(np.exp(shifted))
    if s < 1e-300:
        return c_max
    return c_max + T * np.log(s)

def sum_constraint(h, n):
    return np.sum(h) - n * 0.5

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) > 1e-10:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = h[idx] + (target - np.sum(h))
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def do_slsqp(h0, n, T, max_iter, ftol):
    try:
        hv = h0.copy()
        res = opt.minimize(
            objective_soft, hv, args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': ftol}
        )
        h_out = np.clip(res.x, 0.0, 1.0)
        h_out = fix_sum(h_out, n)
        if np.isfinite(compute_c5(h_out)):
            return h_out
    except Exception:
        pass
    return h0

def perturb_h(h, n, rng, mode):
    hp = h.copy()
    if mode == 0:
        hp = np.roll(hp, rng.randint(1, n))
    elif mode == 1:
        hp = hp[::-1].copy()
    elif mode == 2:
        c = max(2, n // 5)
        s = rng.randint(0, n - c)
        hp[s:s+c] = 1.0 - hp[s:s+c]
    elif mode == 3:
        hp += rng.normal(0, 0.06, n)
    elif mode == 4:
        c = max(2, n // 7)
        s1 = rng.randint(0, n - c)
        s2 = (s1 + c + rng.randint(1, max(2, n - 2*c))) % (n - c)
        blk = hp[s1:s1+c].copy()
        hp[s1:s1+c] = hp[s2:s2+c]
        hp[s2:s2+c] = blk
    elif mode == 5:
        hp = np.clip(hp + rng.normal(0, 0.12, n), 0, 1)
    else:
        m = max(2, n // 4)
        idx = rng.choice(n, size=m, replace=False)
        hp[idx] = 1.0 - hp[idx]
    hp = np.clip(hp, 0.0, 1.0)
    return fix_sum(hp, n)

def make_template(n, typ, rng):
    if typ == 0:
        h = (np.arange(n) < n // 2).astype(float)
    elif typ == 1:

    else:
        h = np.zeros(n)
        h[:n//3] = 1.0
        h[2*n//3:] = 1.0
        h = fix_sum(h, n)
    return h
