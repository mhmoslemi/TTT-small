from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.sum(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xs = np.arange(n_old) * 2.0 / n_old
    xd = np.arange(n_new) * 2.0 / n_new
    return proj(np.interp(xd, xs, h, left=h[0], right=h[-1]), n_new)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = c.max()
    s = tv * (c - mx)
    return (mx + np.log(np.sum(np.exp(s))) / tv) * dx
