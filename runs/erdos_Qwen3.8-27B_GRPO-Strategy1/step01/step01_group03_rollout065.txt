                    tc = (f[top5[a]] - f[top5[b]]) / denom
                    if t_lo < tc < t_hi:
                        cands.append(tc)
        
        best_t = 0.0
        best_val = f.max()
        for t in cands:
            t = max(t_lo, min(t_hi, t))
            v = (f + t * e).max()
            if v < best_val:
                best_val = v
                best_t = t
        
        if abs(best_t) > 1e-10:
            h[i] += best_t
            h[j] -= best_t
            h = np.clip(h, 0, 1)
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
            else:
                # Revert
                h[i] -= best_t
                h[j] += best_t
    
    return project_h(h, target_sum), best_c5
