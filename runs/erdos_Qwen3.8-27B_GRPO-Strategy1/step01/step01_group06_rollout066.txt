```python
    corr = np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    log_sum = np.log(np.sum(np.exp(T * (corr - m))))
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = np.minimum(T * (corr - m), 500)
    log_sum = np.log(np.sum(np.exp(vals)))
    return (m + log_sum / T) * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14: return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14: return h
        h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def build_influence(h, n):
    omh = 1.0 - h
    infl = np.zeros((n, 2*n-1))
    for i in range(n):
        lo = max(0, n-1-i)
        hi = min(2*n-2, 2*n-2-i)

            if np.any(v):

    return infl

    """After h[changed_idx] += delta, omh[changed_idx] -= delta"""
    for t in range(n):
        k = changed_idx - t + n - 1
        if 0 <= k <= 2*n-2:
            infl[t, k] -= delta

def cd_sweep(h, n, n_cand, rng, max_sweeps_per_call=1):

    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    infl = build_influence(h, n)
    cur_max = np.max(corr)
    order = rng.permutation(n)
    
    for ii in range(n):
        i = order[ii]
        best_val = cur_max
        best_delta = 0.0
        best_j = -1
        
        for s in range(1, min(n_cand, n)):
            j = order[(ii + s) % n]
            if j == i:
                continue
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-13:
                continue
            diff_c = infl[i] - infl[j]
            # 1D convex minimization: min max(corr + diff_c*δ)
            lo, hi = d_lo, d_hi
            for _ in range(45):

                mx = vals.max()
                active = vals >= mx - 1e-10

            if new_max < best_val - 1e-11:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    v = T * (corr - m)
    v = np.clip(v, -500, 500)
    return (m + np.log(np.sum(np.exp(v))) / T) * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    d = tgt - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-14: return h
        h += r * (d / t)
    else:
        t = h.sum()
        if t < 1e-14: return h
        h += h * (d / t)
    return np.clip(h, 0.0, 1.0)

def build_infl(h, n):
    omh = 1.0 - h
    infl = np.zeros((n, 2*n - 1))
    for i in range(n):
        lo = max(0, n - 1 - i)
        hi = min(2*n - 2, 2*n - 2 - i)
        idxs = i + np.arange(lo, hi + 1) - (n - 1)
        v = (idxs >= 0) & (idxs < n)
        if np.any(v):
            infl[i, lo:hi+1][v] = omh[idxs[v]]
    return infl

def update_infl(infl, n, idxs_changed, deltas):
    for idx, delta in zip(idxs_changed, deltas):
        for t in range(n):
            k = idx - t + n - 1
            if 0 <= k <= 2*n - 2:
                infl[t, k] -= delta

def cd_sweep(h, n, n_cand, rng):
    corr = np.correlate(h, 1.0 - h, mode='full')
    infl = build_infl(h, n)
    cur_max = corr.max()
    order = rng.permutation(n)
    
    for ii in range(n):
        i = order[ii]
        best_val = cur_max
        best_delta = 0.0
        best_j = -1
        
        for s in range(1, min(n_cand, n)):
            j = order[(ii + s) % n]
            if j == i:
                continue
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-13:
                continue
            dc = infl[i] - infl[j]
            lo, hi = d_lo, d_hi
            for _ in range(40):
                mid = (lo + hi) * 0.5
                vals = corr + dc * mid
                mx = vals.max()
                act = vals >= mx - 1e-9
                sg = dc[act]
                if sg.min() > 1e-7:
                    hi = mid
                elif sg.max() < -1e-7:
                    lo = mid
                else:
                    break
            delta = (lo + hi) * 0.5
            nm = (corr + dc * delta).max()
            if nm < best_val - 1e-11:
                best_val = nm
                best_delta = delta
                best_j = j
        
        if best_j >= 0:
            h[i] += best_delta
            h[best_j] -= best_delta
            np.clip(h, 0.0, 1.0, out=h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_max = corr.max()
            update_infl(infl, n, [i, best_j], [best_delta, -best_delta])
    
    return h, cur_max

def slsqp(h, n, T, mx=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        r = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': mx, 'ftol': 1e-13})
        return project(np.clip(r.x, 0, 1), n)
    except:
        return h

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

                c = compute_c5(best_h)

    # Phase 2: SLSQP polish

    best_c5 = compute_c5(best_h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    v = T * (corr - m)
    v = np.clip(v, -500, 500)
    return (m + np.log(np.sum(np.exp(v))) / T) * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    d = tgt - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-14:
            return h
        h += r * (d / t)
    else:
        t = h.sum()
        if t < 1e-14:
            return h
        h += h * (d / t)
    return np.clip(h, 0.0, 1.0)

def build_infl(h, n):
    omh = 1.0 - h
    infl = np.zeros((n, 2 * n - 1))
    for i in range(n):
        lo = max(0, n - 1 - i)
        hi = min(2 * n - 2, 2 * n - 2 - i)
        idxs = i + np.arange(lo, hi + 1) - (n - 1)
        v = (idxs >= 0) & (idxs < n)
        if np.any(v):
            infl[i, lo:hi + 1][v] = omh[idxs[v]]
    return infl

def update_infl(infl, n, idx, delta_h):
    for t in range(n):
        k = idx - t + n - 1
        if 0 <= k <= 2 * n - 2:
            infl[t, k] -= delta_h

def cd_sweep(h, n, n_cand, rng):
    corr = np.correlate(h, 1.0 - h, mode='full')
    infl = build_infl(h, n)
    cur_max = corr.max()
    order = rng.permutation(n)
    for ii in range(n):
        i = order[ii]
        best_val = cur_max
        best_delta = 0.0
        best_j = -1
        for s in range(1, min(n_cand, n)):
            j = order[(ii + s) % n]
            if j == i:
                continue
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-13:
                continue
            dc = infl[i] - infl[j]
            lo, hi = d_lo, d_hi
            for _ in range(45):
                mid = (lo + hi) * 0.5
                vals = corr + dc * mid
                mx = vals.max()
                act = vals >= mx - 1e-9
                sg = dc[act]
                if sg.min() > 1e-7:
                    hi = mid
                elif sg.max() < -1e-7:
                    lo = mid
                else:
                    break
            delta = (lo + hi) * 0.5
            nm = (corr + dc * delta).max()
            if nm < best_val - 1e-11:
                best_val = nm
                best_delta = delta
                best_j = j
        if best_j >= 0:
            old_i = h[i]
            old_j = h[best_j]
            h[i] = min(max(old_i + best_delta, 0.0), 1.0)
            h[best_j] = min(max(old_j - best_delta, 0.0), 1.0)
            ai = h[i] - old_i
            aj = h[best_j] - old_j
            if abs(ai) > 1e-15 or abs(aj) > 1e-15:
                corr = np.correlate(h, 1.0 - h, mode='full')
                cur_max = corr.max()
                update_infl(infl, n, i, ai)
                update_infl(infl, n, best_j, aj)
    return h, cur_max

def slsqp(h, n, T, mx=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        r = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': mx, 'ftol': 1e-13})
        return project(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(int(seed))
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) > 0:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 0: Quick SLSQP warm-start
    for T in [30, 150, 500]:
        if elapsed() > budget_s * 0.05:
            break
        hc = slsqp(best_h, n, T, 200)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5 = c
            best_h = hc

    # Phase 1: Coordinate descent (main workhorse)
    n_cand_opts = [20, 35, 50]
    ci = 0
    stall = 0
    sweep_count = 0
    while elapsed() < budget_s * 0.72:
        n_cand = n_cand_opts[ci % len(n_cand_opts)]
        best_h, _ = cd_sweep(best_h, n, n_cand, rng)
        best_h = project(best_h, n)
        c = compute_c5(best_h)
        if c < best_c5 - 1e-13:

            stall = 0

            if stall % 15 == 0:

                perturbed = project(best_h + rng.uniform(-p, p, n), n)

                c2 = compute_c5(best_h)
                if c2 > best_c5 + 0.005:

                stall = 0

                sweep_count = 0

    # Phase 2: SLSQP multi-temperature polish
    if elapsed() < budget_s * 0.88:
        for T in [200, 600, 1500, 4000]:
            if elapsed() > budget_s * 0.90:
                break
            hc = slsqp(best_h, n, T, 500)

        best_h, _ = cd_sweep(best_h, n, 45, rng)

        c = compute_c5(best_h)

    # Final cleanup
```
