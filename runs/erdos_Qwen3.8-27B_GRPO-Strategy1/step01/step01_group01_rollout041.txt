    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def pnorm_obj(h, p, lam, mu, n):
    """Augmented Lagrangian with p-norm objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = (np.sum(corr**p))**(1.0/p) * dx
    constraint = np.sum(h) - n/2.0
    val += lam * constraint + 0.5 * mu * constraint**2
    return val

def pnorm_grad(h, p, lam, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sum_p = np.sum(corr**p)
    base = sum_p**(1.0/p - 1.0)
    w = p * corr**(p-1)  # shape (2n-1,)
    
    # Compute gradient of p-norm part
    j = np.arange(n)[:, None]
    k = np.arange(-(n-1), n)[None, :]
    
    m1 = j - k
    valid1 = (m1 >= 0) & (m1 < n)
    m2 = j + k
    valid2 = (m2 >= 0) & (m2 < n)
    
    h1 = np.where(valid1, 1.0 - h[m1.clip(0, n-1)], 0.0)
    h2 = np.where(valid2, h[m2.clip(0, n-1)], 0.0)
    
    grad_pn = (w[None, :] * (h1 - h2)).sum(axis=1) * dx * base
    
    # Gradient of augmented Lagrangian
    constraint = np.sum(h) - n/2.0
    grad_al = (lam + mu * constraint) * np.ones(n)
    
    return grad_pn + grad_al

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return np.interp(xd, xs, h)

def obj_func(h, p, lam, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = (np.sum(corr**p))**(1.0/p) * dx
    c = np.sum(h) - n / 2.0
    val += lam * c + 0.5 * mu * c * c
    return val

import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = 0.5 * n
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def xinterp(h, nd):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return np.interp(xd, xs, h)

def obj_fn(h, p, lam, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = (np.sum(corr**p))**(1.0 / p) * dx
    c = np.sum(h) - 0.5 * n
    return val + lam * c + 0.5 * mu * c * c

def grad_fn(h, p, lam, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sp = np.sum(corr**p)
    if sp < 1e-300:
        return np.zeros(n)
    base = sp**(1.0 / p - 1.0)
    w = p * corr**(p - 1)
    j = np.arange(n)[:, None]
    k = np.arange(-(n - 1), n)[None, :]
    m1 = j - k
    v1 = (m1 >= 0) & (m1 < n)
    m2 = j + k
    v2 = (m2 >= 0) & (m2 < n)
    h1 = np.where(v1, 1.0 - h[m1.clip(0, n - 1)], 0.0)
    h2 = np.where(v2, h[m2.clip(0, n - 1)], 0.0)
    gp = (w[None, :] * (h1 - h2)).sum(axis=1) * dx * base
    c = np.sum(h) - 0.5 * n
    return gp + (lam + mu * c)

def auglag_run(h0, n, p, max_outer=40, time_limit=60):
    ts = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lam, mu = 0.0, 5.0
    h_cur = h0.copy()
    for _ in range(max_outer):
        if time.time() - ts > time_limit:
            break
        res = fmin_l_bfgs_b(obj_fn, h_cur, fprime=grad_fn,
                            args=(p, lam, mu, n),
                            pgtol=1e-13, maxiter=400, maxfun=100000, iprint=0)
        h_new = np.clip(res[0], 0.0, 1.0)
        viol = abs(np.sum(h_new) - 0.5 * n)
        if viol < 1e-9:
            h_new = proj(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            break
        else:
            lam += mu * (np.sum(h_new) - 0.5 * n)
            mu = min(mu * 2.0, 1e6)
            h_cur = proj(h_new, n)
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
    return best_h, best_c5

def make_fourier_perturb(n, rng, amp):
    x = np.arange(n) * 2.0 / n
    k = rng.randint(1, n // 2)
    phase = rng.uniform(0, 2 * np.pi)
    return amp * np.sin(2 * np.pi * k * x + phase)

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    n = 280
    h0 = None
    try:
        if len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float).copy()
            h0 = proj(h0, n)
    except Exception:
        pass
    if h0 is None:
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Coarse exploration at n=140
    if rem() > 200:
        n2 = 140
        h2 = proj(xinterp(h0, n2), n2)
        for p in [50, 200]:
            if rem() < 40:
                break
            h2, _ = auglag_run(h2, n2, p, max_outer=30, time_limit=40)
            h_back = proj(xinterp(h2, n), n)
            c5_b = compute_c5(h_back)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_back.copy()

    # Phase 2: Progressive p-refinement at n=280
    p_schedule = [50, 100, 200, 500, 1000]
    for p in p_schedule:
        if rem() < 40:

                    break

                    h_p = proj(best_h + make_fourier_perturb(n, rng, 0.03), n)

            best_h = h_new.copy()

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
