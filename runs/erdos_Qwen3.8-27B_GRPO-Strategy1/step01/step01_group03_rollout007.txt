import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

def sa_optimize(h0, dx, target_sum, t_end, seed=0):

def active_slsqp(h0, dx, target_sum, maxiter=120, n_active=6):
    n = len(h0)
    best = h0.copy()
    best_c5 = compute_c5(h0, dx)
    for round in range(3):
        corr = np.correlate(best, 1.0 - best, mode='full')
        t0 = np.max(corr)
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        slack = 0.003 / dx
        def ineq(x):
            c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
            return x[n] - c[active]
        def eqf(x):
            return np.sum(x[:n]) - target_sum
        x0 = np.concatenate([best, [t0 + slack]])
        bnds = [(0.0, 1.0)] * n + [(0.0, float(n) * 2)]
        cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
        try:
            res = minimize(lambda x: x[n], x0, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None:
                h_try = project_sr(res.x[:n], target_sum)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5 - 1e-10:

                    best_c5 = c5_try

                    x0[:n] = best
                    x0[n] = np.max(np.correlate(best, 1 - best, mode='full')) + slack
        except Exception:
