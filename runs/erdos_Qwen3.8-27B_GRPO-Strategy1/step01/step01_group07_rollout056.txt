                    res = minimize(sm_obj, h0, args=(beta, n),
                                   jac=sm_grad, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 500, 'ftol': 1e-14})
                    h0 = fix_sum(np.clip(res.x, 0, 1))
                    c5 = compute_c5(h0)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h0.copy()
                        best_n = n
                except Exception:
                    pass
                if restart == 1:
                    h0 = fix_sum(np.clip(h0 + rng.randn(n) * 0.01, 0, 1))

        # Binary projection
        hb = binary_proj(h0, n)
        cb = compute_c5(hb)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
            best_n = n

    # Coordinate descent with backtracking
    if tl() > 10:
        h = best_h.copy()
        n = best_n
        cur = compute_c5(h)
        steps = min(50000, int(tl() * 40))
        for it in range(steps):
            if tl() < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.002, 0.06) * rng.choice([-1.0, 1.0])
            hi, hj = h[i], h[j]
            h[i] = np.clip(hi + d, 0, 1)
            h[j] = np.clip(hj - d, 0, 1)
            if abs(np.sum(h) - n / 2.0) > 1e-10:
                adj = (n / 2.0 - np.sum(h))
                h[i] = np.clip(h[i] + adj, 0, 1)
            c = compute_c5(h)
            if c < cur - 1e-14:
                cur = c
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
            else:
                h[i] = hi
                h[j] = hj

    # Nelder-Mead on interior variables
    if tl() > 20:
        n = best_n
        h = best_h.copy()
        interior = np.where((h > 0.08) & (h < 0.92))[0]
        if len(interior) >= 10:
            nsel = min(40, len(interior))
            idx = rng.choice(interior, size=nsel, replace=False)
            h0_sub = h[idx].copy()

            def nm_obj(y):
                h_t = h.copy()
                h_t[idx] = np.clip(y, 0, 1)
                return compute_c5(h_t)

            try:
                res = minimize(nm_obj, h0_sub, method='Nelder-Mead',
                               options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-13})
                h[idx] = np.clip(res.x, 0, 1)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            except Exception:
                pass

    return best_h, best_c5, best_n
