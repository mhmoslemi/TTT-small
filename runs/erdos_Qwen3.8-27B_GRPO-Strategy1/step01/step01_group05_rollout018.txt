def greedy_local(h, n, steps=200):
    h = h.copy()
    cur = compute_c5(h, n)
    for _ in range(steps):
        best_i, best_v, best_c = -1, None, cur
        for _ in range(30):
            i = np.random.randint(n)
            v = np.random.uniform(0, 1)
            h2 = h.copy()
            h2[i] = v
            h2 = repair(h2, n)
            c = compute_c5(h2, n)
            if c < best_c - 1e-10:
                best_i, best_v, best_c = i, v, c
        if best_i >= 0:
            h = h.copy()
            h[best_i] = best_v
            h = repair(h, n)
            cur = best_c
        else:
            break
    return h, cur

def golden_h(n):
    phi = (1 + np.sqrt(5)) / 2
    h = np.array([1.0 if (i * phi) % 1 < 0.5 else 0.0 for i in range(n)])
    return repair(h, n)
