def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sm = cmax + temp * np.log(np.sum(np.exp((corr - cmax) / temp)))
    dx = 2.0 / n
    return sm * dx + penalty * (np.sum(h) * dx - 1.0) ** 2

    n = len(h)
    target = n / 2.0

    if s < 1e-12:
        return h
    return np.clip(h * (target / s), 0.0, 1.0)

    m = n * f
    xo = (np.arange(n) + 0.5) * (2.0 / n)
