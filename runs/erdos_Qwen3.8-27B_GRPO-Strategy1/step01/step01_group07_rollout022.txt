    return np.correlate(h, 1.0 - h, mode="full")

    n = len(h)

    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    else:

            grad[:(n-s)] -= h[s:]

                    hi = mid
                else:
                    lo = mid

            if time.time() - t0 > budget_s - 5:
                return h, cur_c5
            h[j] = 0.0
            c0 = corr_values(h)
            h[j] = 1.0
            c1 = corr_values(h)
            diff = c1 - c0
            h[j] = 0.0
            # Golden section search: minimize f(t)=max_k(c0[k]+t*diff[k]) on [0,1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, 1.0
            m1 = lo + (1.0 - gr) * (hi - lo)
            m2 = lo + gr * (hi - lo)
            f1_v = np.max(c0 + m1 * diff)
            f2_v = np.max(c0 + m2 * diff)
            for _ in range(20):
                if f1_v < f2_v:
                    hi = m2
                    m2 = m1
                    f2_v = f1_v
                    m1 = lo + (1.0 - gr) * (hi - lo)
                    f1_v = np.max(c0 + m1 * diff)
                else:
                    lo = m1
                    m1 = m2
                    f1_v = f2_v
                    m2 = lo + gr * (hi - lo)
                    f2_v = np.max(c0 + m2 * diff)
            h[j] = (lo + hi) * 0.5
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < cur_c5 - 1e-14:
            cur_c5 = c5
    return h, cur_c5

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64) / old_n
    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    v = project_feasible(v, new_n)
    return v

def pair_swap_search(h, n, n_trials, t0, budget_s):
    rng = np.random.RandomState(12345 + n_trials)
    cur = compute_c5(h)
    for _ in range(n_trials):
        if time.time() - t0 > budget_s - 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c >= cur - 1e-15:
            h[i], h[j] = h[j], h[i]
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    # Phase 1: Gradient descent polishing at native resolution
    h = parent.copy()
    betas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    h, c5 = project_grad_descent(h, pn, betas, 150, t0, budget_s)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Coordinate descent (major refinement)
    best_h, best_c5 = coord_descent(best_h, 300, t0, budget_s)

    # Phase 3: Multi-start with perturbations
    for trial in range(8):
        if time.time() - t0 > budget_s - 60:
            break
        h_try = project_feasible(best_h + rng.randn(pn) * 0.002, pn)
        h_try, c5 = project_grad_descent(h_try, pn, [0.002, 0.001, 0.0005], 100, t0, budget_s)
        h_try, c5 = coord_descent(h_try, 80, t0, budget_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 4: Further coordinate descent on best
    best_h, best_c5 = coord_descent(best_h, 400, t0, budget_s)

    # Phase 5: Try higher resolution
    if time.time() - t0 < budget_s - 90:
        for mult in [2, 3]:
            if time.time() - t0 > budget_s - 50:
                break
            nn = min(pn * mult, 840)
            if nn <= pn:
                continue
            h_hi = upsample(best_h, nn)
            h_hi, c5 = project_grad_descent(h_hi, nn, [0.01, 0.005, 0.002, 0.001], 200, t0, budget_s)
            h_hi, c5 = coord_descent(h_hi, 100, t0, budget_s)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = nn

    # Final coordinate descent pass
    best_h, best_c5 = coord_descent(best_h, 200, t0, budget_s)

    return best_h, best_c5, best_n
