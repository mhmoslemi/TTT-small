        if shift >= 0:

            # j-shift valid: j in [shift, n-1]

            grad[s:] += wm * (1.0 - h[:n-s])
            grad[:n-s] -= wm * h[s:]

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(n + 10):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = t * corr
    c_shift = c - np.max(c)
    exp_c = np.exp(c_shift)
    sum_exp = np.sum(exp_c)
    val = (np.max(c) + np.log(sum_exp)) / t * dx
    w = exp_c / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += wk * (1.0 - h[:n - m])
            grad[:n - m] -= wk * h[m:]
        else:
            s = -m
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]
    return val, dx * grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(n + 10):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = t * corr
    c_shift = c - np.max(c)
    exp_c = np.exp(c_shift)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += wk * (1.0 - h[:n - m])
            grad[:n - m] -= wk * h[m:]
        else:
            s = -m
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]
    return dx * grad

def obj_fn(h, t_val):

    xd = np.arange(nd) * 2.0 / nd
    return project(np.interp(xd, xs, h, left=0, right=0), nd)

def thue_morse(n):
    a = np.zeros(n)

        a[i] = 0.5 if bin(i).count('1') % 2 == 0 else 0.0
    return a

def proj_grad_loop(h, n, t_list, steps_total, t0_time, time_limit):

    best_c = compute_c5(h)
    total_steps = steps_total
    steps_per_t = total_steps // len(t_list)
    stuck = 0
    prev_best = best_c
    for t_i, t_val in enumerate(t_list):

            if step % 3000 == 0 and step > 0:

                if cur > prev_best + 1e-10:
                    stuck += 1
                    if stuck >= 2:

                        stuck = 0
                else:
                    stuck = 0
                prev_best = min(prev_best, cur)
            grad = smooth_max_grad(h, t_val)

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
