**Creative idea: Use the problem structure to decompose**

The key observation is that C₅ = max_k (Σ_i h[i](1-h[i+k])) * 2/n.

Let's define g_k(h) = Σ_i h[i](1-h[i+k]). We want min_h max_k g_k(h) * 2/n.

This is a min-max problem. By von Neumann's minimax theorem (applied in a generalized sense), we can think of:
min_h max_k g_k(h) = max_{p: p_k≥0, Σp_k=1} min_h Σ_k p_k g_k(h)

The inner problem min_h Σ_k p_k g_k(h) is a quadratic minimization (since each g_k is quadratic in h, and a weighted sum of quadratics is quadratic). Subject to 0≤h≤1, Σh=n/2.

This is a QP (quadratically constrained... no, the objective is quadratic, constraints are linear and box). This is a convex QP if the Hessian is PSD!

Let's check: g_k(h) = Σ_i h[i] - Σ_i h[i]h[i+k]. The first term is linear. The second term -Σ h[i]h[i+k] has Hessian -2A_k where A_k is a symmetric banded matrix with A_k[i, i+k] = A_k[i+k, i] = 1. The eigenvalues of A_k range from -(n/2) to +(n/2) approximately. So the Hessian of -Σ h[i]h[i+k] has eigenvalues ranging from -n to +n. NOT PSD in general.

So the QP is nonconvex. But for a specific p, it might be convex if the weights are right.

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def sum_fn(h, n):
    return np.sum(h) - n * 0.5

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h.copy()
    h = h.copy()
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    ni = np.sum(interior)
    if ni > 0:
        h[interior] += (target - s) / ni
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-14:
        return cmax * dx
    shifted = (corr - cmax) / T
    shifted = np.clip(shifted, -500, 0)
    s = np.sum(np.exp(shifted))
    return (cmax + T * np.log(s)) * dx

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = fix_sum(h_new, new_n)
    return h_new

def slsqp_optimize(h_start, n, T, max_iter, ftol):
    res = opt.minimize(
        soft_max_obj, h_start, args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': sum_fn, 'args': (n,)}],
        options={'maxiter': max_iter, 'ftol': ftol}
    )
    return np.clip(res.x, 0.0, 1.0)

def co_by_la_optimize(h_start, n, T, max_iter):
    res = opt.minimize(
        soft_max_obj, h_start, args=(T,),
        method='COBYLA',
        constraints=['0 <= h[i] <= 1' for i in range(n)],
        options={'maxiter': max_iter, 'rho': 0.1}
    )
    return np.clip(res.x, 0.0, 1.0)

def coordinate_polish(h, n, iterations=50):
    h = h.copy()
    dx = 2.0 / n
    for _ in range(iterations):
        improved = False
        for i in range(n):
            old_val = h[i]
            # Try small perturbations
            for delta in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                new_val = old_val + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                # Fix sum by adjusting a random interior point
                target = n * 0.5
                s = np.sum(h)
                diff = target - s
                if abs(diff) < 1e-15:
                    break
                # Adjust another point
                j = (i + n // 2) % n
                if j != i:
                    adj = h[j] + diff
                    if 0 <= adj <= 1:
                        h[j] = adj
                c = compute_c5(h)
                if c < compute_c5(fix_sum(h, n)):
                    h = fix_sum(h, n)
                    improved = True
                    break
            if not improved and delta == -0.01:
                h[i] = old_val
        h = fix_sum(h, n)
    return h
