- h[i] appears in (1-h[m-j]) when m-j=i, i.e., j=m-i: contributes -h[m-i] if 0≤m-i<n and m-i≠i (i.e., m≠2i)

So ∂c_m/∂h[i] = (1-h[m-i])·[0≤m-i<n] - h[m-i]·[0≤m-i<n, m≠2i]

The gradient of the softmax objective:
grad[i] = dx · Σ_m w[m] · ∂c_m/∂h[i]
        = dx · Σ_m w[m] · [(1-h[m-i])·[0≤m-i<n] - h[m-i]·[0≤m-i<n, m≠2i]]

Let j = m-i (so m = i+j):
= dx · Σ_j w[i+j] · [(1-h[j])·[0≤j<n] - h[j]·[0≤j<n, i+j≠2i]]
= dx · Σ_j w[i+j] · [(1-h[j]) - h[j]·[j≠i]]  (since 0≤j<n always in the sum)
= dx · Σ_j w[i+j] · [1-h[j] - h[j] + h[j]·[j=i]]
= dx · Σ_j w[i+j] · [1-2h[j] + h[j]·δ_{j,i}]
= dx · [Σ_j w[i+j]·(1-2h[j]) + h[i]·w[2i]·[0≤2i<2n-1]]

So the formula is:
grad[i] = dx · [Σ_{j=0}^{n-1} w[i+j]·(1-2h[j]) + h[i]·w[2i]·(2i<2n-1)]

where w[m] = 0 for m outside [0, 2n-2].

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_value(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    w = np.exp(shifted / T)
    w /= np.sum(w)
    
    g = 1.0 - 2.0 * h  # length n
    
    # result[i] = sum_j w[i+j] * g[j], with w[m]=0 for m outside [0,2n-2]
    # Use convolution: result[i] = conv(g, w_rev)[2n-2-i]
    w_rev = w[::-1]
    conv = np.convolve(g, w_rev, mode='full')
    # conv has length n + (2n-1) - 1 = 3n-2
    # result[i] = conv[2n-2-i] for i=0..n-1
    idx = 2*n-2 - np.arange(n)  # [2n-2, 2n-3, ..., n-1]
    result = conv[idx]
    
    # Correction term: h[i]*w[2i] if 2i < 2n-1
    corr_idx = 2 * np.arange(n)
    valid = corr_idx < (2*n-1)
    correction = np.zeros(n)
    correction[valid] = h[valid] * w[corr_idx[valid]]
    
    return (result + correction) * dx

def project_to_constraint(h, S):
    """Project h onto {0<=h<=1} ∩ {sum(h)=S}"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

        else:

        if np.sum(interior) > 0:
            h_proj[interior] += diff / np.sum(interior)
            h_proj = np.clip(h_proj, 0.0, 1.0)

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    w = np.exp(shifted / T)
    w /= np.sum(w)
    g = 1.0 - 2.0 * h
    w_rev = w[::-1]
    conv = np.convolve(g, w_rev, mode='full')
    idx = 2 * n - 2 - np.arange(n)
    return conv[idx] * dx

def project_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - S) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s_try = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_try > S:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) * 0.5
    h_p = np.clip(h + lam, 0.0, 1.0)
    diff = S - np.sum(h_p)
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def upsample_to(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h, [h[0]]]))
    h_new = project_h(h_new, new_n * 0.5)
    return h_new

def gd_optimize(h0, n, T_schedule, lr, max_steps, time_limit, t_start):
    h = h0.copy()
    S = n * 0.5
    for i, T in enumerate(T_schedule):
        if i % 100 == 0 and time.time() - t_start > time_limit:
            break
        for step in range(max_steps):
            if time.time() - t_start > time_limit:
                break
            grad = softmax_grad(h, T)
            # Backtracking line search
            alpha = 1.0
            c5_curr = compute_c5(h)
            improved = False
            for _ in range(8):
                h_try = project_h(h - alpha * lr * grad, S)
                c5_try = compute_c5(h_try)
                if c5_try < c5_curr - 1e-15:
                    h = h_try
                    improved = True
                    break
                alpha *= 0.5
            if not improved:
                break
    return h

def sum_constraint(x, n):
    return np.sum(x) - n * 0.5

import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max * dx
    shifted = corr - c_max
    return (c_max + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def softmax_grad_fn(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    w = np.exp(shifted / T)
    w /= np.sum(w)
    g = 1.0 - 2.0 * h
    w_rev = w[::-1]
    conv = np.convolve(g, w_rev, mode='full')
    idx = 2 * n - 2 - np.arange(n)
    return conv[idx] * dx

def project_h(h, S):

        if s_try > S:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) * 0.5
    h_p = np.clip(h + lam, 0.0, 1.0)
    diff = S - np.sum(h_p)

def upsample_to(h, new_n):

    h_ext = np.concatenate([h, [h[0]]])
    h_new = np.interp(x_new, x_old, h_ext)
    return project_h(h_new, new_n * 0.5)

def gd_optimize(h0, n, stages, time_limit, t_start):
    h = h0.copy()
    S = n * 0.5
    for (T_s, T_e, steps, lr) in stages:
        for step in range(steps):
            if time.time() - t_start > time_limit:
                return h
            T = T_s * (T_e / T_s) ** (step / max(steps - 1, 1))
            grad = softmax_grad_fn(h, T)
            gnorm = np.max(np.abs(grad))

                break
            alpha = 1.0

                c5_try = compute_c5(h_try)

                    break

def sum_con(x, n):
    return np.sum(x) - n * 0.5

def slsqp_obj(x, T):
    return softmax_obj(x, T)

def slsqp_jac(x, T):
    return softmax_grad_fn(x, T)

    h = np.zeros(n)
    period = n // 2
    for i in range(n):
        h[i] = 1.0 if (i % period) < period // 2 else 0.0

    i = np.arange(n, dtype=float)
    h = 0.5 + 0.45 * np.cos(2.0 * np.pi * freq * i / n)
