import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, t, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s = np.clip(s, -100, 0)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(500):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def ceq(x):
    return np.array([x.sum() - len(x) * 0.5])

def ceqj(x):
    return np.ones((1, len(x)))

def fourier_start(n, K, rng):
    x = 2.0 * np.arange(n) / n
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        amp = 0.10 / np.sqrt(j)
        a = rng.randn() * amp
        b = rng.randn() * amp
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    return proj(h, n)

def pattern_start(n, k, amp, phase, noise, rng):
    x = 2.0 * np.arange(n) / n
    h = 0.5 + amp * np.sin(np.pi * k * x + phase) + noise * rng.randn(n)
    return proj(h, n)

def subgrad_poly(h, n, n_iter, alpha0):
    h = h.copy()
    best = compute_c5(h)
    alpha = alpha0
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        tol = 0.02 * mx + 1e-8
        active = np.where(corr >= mx - tol)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for m in active:
            s = m - (n - 1)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += -h[:n - s]
            else:
                p = -s
                grad[p:] += (1.0 - h[:-p])
                grad[:-p] += -h[p:]
        grad /= len(active)
        h_new = proj(h - alpha * grad, n)
        c = compute_c5(h_new)
        if c < best:
            best = c
            h = h_new
        else:
            alpha *= 0.7
        alpha = max(alpha, 0.001)
    return h, best

def coord_polish(h, n, n_pass, base_step, rng):
    h = h.copy()
    best = compute_c5(h)
    step = base_step
    for p in range(n_pass):
        improved = False
        idx = rng.permutation(n)
        for i in idx:
            old = h[i]
            for delta in [step, -step]:
                nv = old + delta
                if nv < 0 or nv > 1:
                    continue
                h2 = h.copy()
                h2[i] = nv
                dh = nv - old
                if dh > 0:
                    j = int(np.argmax(h))
                    if j != i and h[j] >= dh:
                        h2[j] -= dh
                    else:
                        h2 = proj(h2, n)
                elif dh < 0:
                    j = int(np.argmin(h))
                    if j != i and (1.0 - h[j]) >= -dh:
                        h2[j] -= dh
                    else:
                        h2 = proj(h2, n)
                c = compute_c5(h2)
                if c < best - 1e-12:
                    h = h2
                    best = c
                    improved = True
                    break
        if not improved:
            break
        step *= 0.88
        if step < 0.0008:
            break
    return h, best

    rng = np.random.RandomState(seed)

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ceq, 'jac': ceqj}]
    t_vals = [20, 50, 100, 200, 400]

    # Phase 1: Multi-start SLSQP with warm-up t schedule
    for si, s in enumerate(starts):

            try:

                             bounds=bnds, constraints=cons,
                             options={'maxiter': 500, 'ftol': 1e-12})
                hc = proj(np.clip(r.x, 0, 1), n)
                c = compute_c5(hc)
                if c < bc:
                    bc = c
                    bh = hc.copy()
                cur = hc.copy()
            except Exception:

        try:

            hc = proj(np.clip(r.x, 0, 1), n)
            c = compute_c5(hc)

                bc = c
                bh = hc.copy()
        except Exception:

    # Phase 3: Subgradient polishing

    if time.time() - t0 < budget_s * 0.97:
        bh, c = coord_polish(bh, n, 300, 0.012, rng)
        if c < bc:
            bc = c

    bh = proj(bh, n)
    c5 = compute_c5(bh)
    return bh, c5, n
