conv2[i+n-1] = sum_j h[j] * weights[i+n-1-j]
For j = v: weight index = i+n-1-v

We want weights[i-v+n-1] = weights[n-1+i-v]. ✓

So second_term[i] = conv2[i+n-1]. ✓

For the first term:
conv1[k] = sum_j (1-h)[j] * weights[::-1][k-j] = sum_j (1-h)[j] * weights[2n-2-(k-j)] = sum_j (1-h)[j] * weights[2n-2-k+j]
We want: sum_u (1-h)[u] * W[u-i] = sum_u (1-h)[u] * weights[u-i+n-1]
Set k such that 2n-2-k+j = u-i+n-1 with j=u: 2n-2-k+u = u-i+n-1 → 2n-2-k = -i+n-1 → k = 2n-2-n+1+i = n-1+i

conv1[n-1+i] = sum_j (1-h)[j] * weights[2n-2-(n-1+i)+j] = sum_j (1-h)[j] * weights[n-1-i+j]
For j=u: weights[n-1-i+u] = weights[u-i+n-1] ✓

So first_term[i] = conv1[n-1+i]. ✓

Great! So the gradient is:
grad[i] = dx * (conv1[n-1+i] - conv2[n-1+i]) for i=0,...,n-1

where:
conv1 = np.convolve(1-h, weights[::-1], 'full')  [length 3n-2]
conv2 = np.convolve(h, weights, 'full')  [length 3n-2]

Both can be computed via FFT in O(n log n)!

grad_penalty[i] = 2 * lam * (sum(h) - n/2)

import numpy as np
import scipy.fft
import scipy.optimize as opt
import time

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    N = 1
    while N < 2 * n - 1:
        N *= 2
    a = h.astype(np.float64)
    b = (1.0 - h[::-1]).astype(np.float64)
    corr = np.real(scipy.fft.ifft(scipy.fft.fft(a, N) * scipy.fft.fft(b, N)))[:2*n-1]
    return float(np.max(corr)) * dx

def softmax_c5_and_weights(h, T):
    n = len(h)
    dx = 2.0 / n
    N = 1
    while N < 2 * n - 1:
        N *= 2
    a = h.astype(np.float64)
    b = (1.0 - h[::-1]).astype(np.float64)
    corr = np.real(scipy.fft.ifft(scipy.fft.fft(a, N) * scipy.fft.fft(b, N)))[:2*n-1]
    c_max = np.max(corr)
    if T < 1e-12:
        weights = np.zeros(2*n-1)
        weights[c_max == corr] = 1.0
        weights /= np.sum(weights)
        val = c_max * dx
    else:
        exp_vals = np.exp((corr - c_max) / T)
        weights = exp_vals / np.sum(exp_vals)
        val = (c_max + T * np.log(np.sum(exp_vals))) * dx
    return val, weights, corr

def compute_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    N = 1
    while N < 2 * n - 1:
        N *= 2
    a = h.astype(np.float64)
    b = (1.0 - h[::-1]).astype(np.float64)
    corr = np.real(scipy.fft.ifft(scipy.fft.fft(a, N) * scipy.fft.fft(b, N)))[:2*n-1]
    c_max = np.max(corr)
    if T < 1e-12:
        weights = np.zeros(2*n-1)
        weights[corr == c_max] = 1.0
        weights /= np.sum(weights)
    else:
        exp_vals = np.exp((corr - c_max) / T)
        weights = exp_vals / np.sum(exp_vals)
    
    # Gradient via convolution
    N2 = 1
    while N2 < 3 * n - 1:
        N2 *= 2
    bh = 1.0 - h
    w_rev = weights[::-1]
    conv1 = np.real(scipy.fft.ifft(scipy.fft.fft(bh, N2) * scipy.fft.fft(w_rev, N2)))[:2*n-1+n-1]
    conv2 = np.real(scipy.fft.ifft(scipy.fft.fft(h, N2) * scipy.fft.fft(weights, N2)))[:2*n-1+n-1]
    
    n1 = 2 * n - 2  # len(w_rev) - 1 = (2n-1)-1 = 2n-2
    offset1 = 2 * n - 2  # offset for cross-correlation
    offset2 = n - 1  # offset for second convolution
    
    # first_term[i] = conv1[offset1 + i]
    # second_term[i] = conv2[offset2 + i]
    grad = dx * (conv1[offset1:offset1+n] - conv2[offset2:offset2+n])
    return grad

def project_fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) > 1e-12:
        diff = target - s
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
            h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_step(h, n, T, lam, lr, n_steps=100):
    """Nesterov accelerated projected gradient"""
    m = n
    x = h.copy()
    y = h.copy()
    beta = 0.5
    best_x = x.copy()
    best_c5 = compute_c5_fft(x)
    
    for step in range(n_steps):
        grad = compute_gradient(y, T)
        s = np.sum(y) - m * 0.5
        grad += 2.0 * lam * s  # penalty gradient
        
        # Nesterov update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = x + (beta / beta_new) * (x - y)
        # Gradient step
        x_new = y_new - lr * grad
        # Project
        x_new = project_fix(x_new, m)
        
        # Check improvement
        c5_new = compute_c5_fft(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_x = x_new.copy()
        
        y = x_new
        x = x_new
    
    return best_x

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_fix(h_new, new_n)

def lbfgs_obj(x, T, lam):
    n = len(x)
    c5, weights, corr = softmax_c5_and_weights(x, T)
    s = np.sum(x) - n * 0.5
    return c5 + lam * s * s

def lbfgs_grad(x, T, lam):
    n = len(x)
    grad = compute_gradient(x, T)
    s = np.sum(x) - n * 0.5
    grad += 2.0 * lam * s
    return grad

def lbfgs_optimize(h_start, n, T, lam, max_iter):
    res = opt.minimize(
        lbfgs_obj, h_start,
        jac=lbfgs_grad,
        args=(T, lam),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12}
    )
    return res.x
