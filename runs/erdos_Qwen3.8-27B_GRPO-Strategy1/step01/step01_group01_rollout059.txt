import numpy as np
from scipy.optimize import minimize
import time

_T = 200.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(2000):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_soft_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def c5_soft_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s -= np.max(s)
    e = np.exp(s)
    w = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        m1lo = max(0, lo - n + 1 + k)
        m1hi = min(n - 1, hi - n + 1 + k)
        if m1lo <= m1hi:
            i_vals = np.arange(m1lo, m1hi + 1) - n + 1 + k
            grad[m1lo:m1hi + 1] += wk * (1.0 - h[i_vals])
        m2lo = max(0, lo)
        m2hi = min(n - 1, hi)
        if m2lo <= m2hi:
            idx = np.arange(m2lo, m2hi + 1) + (n - 1 - k)
            grad[m2lo:m2hi + 1] += wk * (-h[idx])
    grad *= dx
    return grad

def _obj(x):
    return c5_soft_obj(x, _T)

def _grad(x):
    return c5_soft_grad(x, _T)

def ceq_fun(x):
    return np.sum(x) - len(x) / 2.0

def ceq_jac(x):
    return np.ones(len(x))

def time_left(t0, budget_s, frac):
    return t0 + frac * budget_s - time.time()

def slsqp_optimize(h, n, t_val, bnds, cons, maxiter, ftol):
    global _T
    _T = t_val
    r = minimize(_obj, h.copy(), jac=_grad, method='SLSQP',
                 bounds=bnds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(r.x, 0.0, 1.0), n)

def coord_search(h, n, t0, budget_s, frac, num_passes, delta_list):
    c5 = compute_c5(h)
    for _ in range(num_passes):
        if time_left(t0, budget_s, frac) < 5:
            break
        improved = False
        for i in range(n):
            if time_left(t0, budget_s, frac) < 2:
                break
            orig = h[i]
            for d in delta_list:
                h[i] = np.clip(orig + d, 0.0, 1.0)
                h2 = proj(h, n)
                c5_try = compute_c5(h2)
                if c5_try < c5 - 1e-12:
                    c5 = c5_try
                    h = h2
                    improved = True
                    break
                h[i] = orig
    return h, c5
