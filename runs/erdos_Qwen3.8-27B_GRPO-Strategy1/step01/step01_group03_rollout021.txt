import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:

    return h

def subgrad_step(h, n, top_k=5):
    """Compute subgradient of max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    idx = np.argsort(-corr)[:top_k]
    g = np.zeros(n)
    for li in idx:
        k = abs(li - center)
        for j in range(n):
            c = 0.0
            if j >= k:
                c += (1.0 - h[j - k])
            if j <= n - 1 - k:
                c -= h[j + k]
            g[j] += c
    g /= top_k
    return g, np.max(corr)

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def coordinate_descent(h, dx, target_sum, n_passes=3, t_deadline=None):

    best_c5 = compute_c5(h, dx)
    for p in range(n_passes):

            break
        for j in range(n):
            if t_deadline and time.time() > t_deadline:

                    c5_new = compute_c5(h_new, dx)

                        best_c5 = c5_new
                        break
    return h, best_c5

def subgrad_descent(h0, dx, target_sum, n_iter=2000, t_deadline=None):

For the log-sum-exp approach:
f_beta(h) = (1/beta) * log(sum_k exp(beta * corr_k * dx))

This is a smooth approximation to max_k corr_k * dx. As beta → ∞, it approaches the max.

The gradient of f_beta with respect to h[j] is:
sum_k p_k * d(corr_k * dx)/dh[j]
where p_k = exp(beta * corr_k * dx) / sum_l exp(beta * corr_l * dx)

This is exactly the softmax gradient that was in the previous code! So the previous approach was already using this. The issue is that it was slow (O(n²) due to the explicit loop over j).

```python
def lse_gradient(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    
    # For each lag k (0 to n-1), the gradient of corr[center+k] w.r.t. h is:
    # grad_k[j] = (1-h[j-k]) * I[j>=k] - h[j+k] * I[j<=n-1-k]
    # 
    # We can vectorize this:
    g = np.zeros(n)
    for k in range(n):
        if w[n-1+k] > 1e-15:  # only for significant lags
            wk = w[n-1+k]
            # (1-h[j-k]) for j >= k
            if k > 0:
                g[k:] += wk * (1.0 - h[:n-k])
            # -h[j+k] for j <= n-1-k
            g[:n-k] -= wk * h[k:]
    g *= dx
    return g
```

This is O(n²) but with a simple loop over k (n iterations) and vector ops inside. For n=47, that's 47 vector ops of length ~47. Very fast.

def lse_lbfgsb(h0, dx, target_sum, beta=20, maxiter=2000):
    n = len(h0)
    # Fix h[0] = target_sum - sum(h[1:])
    
    def obj_and_grad(y):
        h = np.empty(n)
        h[1:] = y
        h[0] = target_sum - y.sum()
        if h[0] < -0.01 or h[0] > 1.01:
            return 1e10, np.zeros(n-1)
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1.0 - h_clip, mode='full')
        m = corr.max()
        w = np.exp(beta * (corr - m) * dx)
        w /= w.sum()
        val = np.log(w.sum()) / beta  # = log-sum-exp / beta
        
        # Gradient
        g = np.zeros(n)
        for k in range(n):
            wk = w[n-1+k] * dx
            if wk > 1e-18:
                if k > 0:
                    g[k:] += wk * (1.0 - h_clip[:n-k])
                g[:n-k] -= wk * h_clip[k:]
        # For h[0] = target_sum - sum(y), d/dy[j] h[0] = -1
        # So grad_y[j] = g[j+1] - g[0]
        gy = g[1:] - g[0]
        return val, gy
    
    y0 = h0[1:].copy()
    bnds = [(0.0, 1.0)] * (n-1)
    res = minimize(obj_and_grad, y0, jac=True, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-14})
    h = np.empty(n)
    h[1:] = res.x
    h[0] = target_sum - res.x.sum()
    return project_sr(h, target_sum)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def lse_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(beta * (corr - m) * dx)
    ws = w.sum()
    if ws < 1e-300:
        return 0.0, np.zeros(n)
    w /= ws
    val = m * dx + np.log(ws) / beta
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k] * dx
        if wk > 1e-18:
            if k > 0:
                g[k:] += wk * (1.0 - h[:n - k])
            g[:n - k] -= wk * h[k:]
    return val, g

def lse_lbfgsb(h0, dx, target_sum, beta, maxiter=3000):
    n = len(h0)
    def obj_grad(y):
        h = np.empty(n)
        h[1:] = y
        h[0] = target_sum - y.sum()
        if h[0] < -0.05 or h[0] > 1.05:
            return 1e6, np.zeros(n - 1)
        hc = np.clip(h, 0.0, 1.0)
        val, g = lse_grad(hc, dx, beta)
        gy = g[1:] - g[0]
        return val, gy
    y0 = h0[1:].copy()
    bnds = [(0.0, 1.0)] * (n - 1)
    try:
        res = minimize(obj_grad, y0, jac=True, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h = np.empty(n)
        h[1:] = res.x
        h[0] = target_sum - res.x.sum()
        return project_sr(h, target_sum)
    except Exception:
        return h0.copy()

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def informed_sa(h0, dx, target_sum, max_iter=20000, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    T = 0.008
    stall = 0
    for it in range(max_iter):
        if t_deadline and time.time() > t_deadline:
            break
        if it % 400 == 0 and it > 0:
            T *= 0.92
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr))
        center = n - 1
        top_idx = np.argsort(-corr)[:4]
        h_new = h.copy()
        r = np.random.rand()
        if r < 0.65:
            li = top_idx[np.random.randint(len(top_idx))]
            k = abs(li - center)
            if np.random.rand() < 0.5:
                pos = np.random.randint(0, max(1, n - k))
                h_new[pos] -= 0.03
                h_new[pos + k] += 0.03
            else:
                pos = np.random.randint(k, n) if k > 0 else np.random.randint(0, n)
                h_new[pos] += 0.03
                h_new[pos - k] -= 0.03
        elif r < 0.85:
            j = np.random.randint(n)
            h_new[j] += np.random.randn() * 0.04
        else:
            j1, j2 = np.random.randint(0, n, 2)
            h_new[j1] += 0.05
            h_new[j2] -= 0.05
        h_new = project_sr(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        if c5_new <= c5 or np.random.rand() < np.exp(-(c5_new - c5) / max(T, 1e-6)):
            h = h_new
            if c5_new < c5:
                stall = max(0, stall - 3)
        if stall > 2000:
            h = best_h.copy()
            T = 0.015
            stall = 0
    return best_h, best_c5

def swap_search(h0, dx, target_sum, n_passes=3, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    for p in range(n_passes):
        if t_deadline and time.time() > t_deadline:
            break
        for j in range(n):
            if t_deadline and time.time() > t_deadline:
                break
            improved = False
            for direction in [1, -1]:
                for step in [0.1, 0.05, 0.02, 0.01]:
                    h_new = h.copy()
                    h_new[j] += direction * step
                    h_new = project_sr(h_new, target_sum)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < best_c5 - 1e-12:
                        h = h_new
                        best_c5 = c5_new
                        improved = True
                        break
                if improved:
                    break
    return h, best_c5

def make_interp_init(h47, n_new):
    x_old = np.linspace(0, 2, 48)[:47]
    x_new = np.linspace(0, 2, n_new + 1)[:-1]
    h_new = np.interp(x_new, x_old, h47)
    return h_new
