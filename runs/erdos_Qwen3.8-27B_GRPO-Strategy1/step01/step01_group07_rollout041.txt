def project_constraint(x, target):
    """Project x onto {y: sum(y)=target, 0<=y<=1} via bisection on lambda"""

        s = y.sum()
        if s > target:
            lo = mid

def upsample(parent, new_n):

    v = np.interp(nx, ox, parent)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    return dx * (m + beta * np.log(np.sum(np.exp((corr - m) / beta))))

def smooth_max_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    exp_c = np.exp((corr - m) / beta)
    sum_exp = np.sum(exp_c)
    obj = dx * (m + beta * np.log(sum_exp))
    w = exp_c / sum_exp
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wt = w[k]
        if wt < 1e-15:
            continue
        if s >= 0:
            A[:n - s] += wt * x[s:]
            B[s:] += wt * x[:n - s]
        else:
            ss = -s
            A[ss:] += wt * x[:n - ss]
            B[:n - ss] += wt * x[ss:]
    w_cum = np.concatenate([[0.0], np.cumsum(w)])
    C = np.zeros(n)
    for i in range(n):
        lo = max(0, n - 1 - i)
        hi = min(2 * n - 2, 2 * n - 2 - i)
        C[i] = w_cum[hi + 1] - w_cum[lo]
    grad = dx * (C - A - B)
    return obj, grad

def project_constraint(x, target):
    lo = x.min() - target / len(x) - 1.0
    hi = x.max() - target / len(x) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_fn(parent, new_n):
    old_n = len(parent)
    if new_n == old_n:
        return parent.copy()
    ox = np.linspace(0.0, 1.0, old_n, endpoint=False)
    nx = np.linspace(0.0, 1.0, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    return project_constraint(v, new_n / 2.0)

def slsqp_obj_wrapper(x, n, beta):
    return smooth_max_obj(x, n, beta)

def slsqp_eq_wrapper(x, n):
    return np.sum(x) - n / 2.0

def slsqp_jac_wrapper(x, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    # Phase 1: Smooth max projected gradient with annealing at parent resolution
    h = best_h.copy()
    n = pn
    target = n / 2.0

    betas = [0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0006, 0.0003]
    for beta in betas:
        if budget_s - elapsed() < 15:
            break
        max_steps = int(min(200, (budget_s - elapsed() - 10) / 2.0))
        max_steps = max(30, max_steps)
        alpha = 0.5
        for step in range(max_steps):
            if budget_s - elapsed() < 8:
                break
            obj, grad = smooth_max_grad(h, n, beta)
            for _ in range(25):
                h_trial = project_constraint(h - alpha * grad, target)
                obj_trial, _ = smooth_max_grad(h_trial, n, beta)
                if obj_trial < obj - 1e-15:
                    break
                alpha *= 0.5
            else:
                alpha = max(alpha, 1e-6)
                break
            h = h_trial
            obj = obj_trial
            alpha = min(alpha * 1.1, 2.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 2: Random restarts with perturbation
    n_restarts = 6
    for r in range(n_restarts):
        if budget_s - elapsed() < 50:
            break
        scale = 0.015 + 0.01 * r
        h_r = project_constraint(best_h + rng.randn(n) * scale, target)
        for beta in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if budget_s - elapsed() < 8:
                break
            max_steps = 80
            alpha = 0.3
            for step in range(max_steps):
                if budget_s - elapsed() < 5:
                    break
                obj, grad = smooth_max_grad(h_r, n, beta)
                for _ in range(20):
                    h_trial = project_constraint(h_r - alpha * grad, target)
                    obj_trial, _ = smooth_max_grad(h_trial, n, beta)
                    if obj_trial < obj - 1e-15:
                        break
                    alpha *= 0.5
                else:
                    break
                h_r = h_trial
                obj = obj_trial
                alpha = min(alpha * 1.05, 1.5)
            c5 = compute_c5(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()
                best_n = n

    # Phase 3: Try different resolutions
    for n_try in [int(pn * 0.5), int(pn * 1.5), 400]:
        if n_try < 50 or n_try > 500 or budget_s - elapsed() < 40:
            continue
        h_try = upsample_fn(best_h, n_try)
        target_try = n_try / 2.0
        for beta in [0.01, 0.005, 0.002, 0.001]:
            if budget_s - elapsed() < 8:
                break
            max_steps = 60
            alpha = 0.4
            for step in range(max_steps):
                if budget_s - elapsed() < 5:
                    break
                obj, grad = smooth_max_grad(h_try, n_try, beta)
                for _ in range(20):
                    h_trial = project_constraint(h_try - alpha * grad, target_try)
                    obj_trial, _ = smooth_max_grad(h_trial, n_try, beta)
                    if obj_trial < obj - 1e-15:
                        break
                    alpha *= 0.5
                else:
                    break
                h_try = h_trial
                obj = obj_trial
                alpha = min(alpha * 1.05, 1.5)
            c5 = compute_c5(h_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()
                best_n = n_try

    # Phase 4: Coordinate descent polishing
    if budget_s - elapsed() > 20:
        h = best_h.copy()
        n = best_n
        target = n / 2.0
        cur = best_c5
        n_iter = int(min(8000, (budget_s - elapsed() - 5) * 40))
        for it in range(max(100, n_iter)):
            if budget_s - elapsed() < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.002, 0.025) * rng.choice([-1.0, 1.0])
            hi_new = h[i] + d
            hj_new = h[j] - d
            if hi_new < 0.0 or hi_new > 1.0 or hj_new < 0.0 or hj_new > 1.0:
                continue
            h[i] = hi_new
            h[j] = hj_new
            c = compute_c5(h)
            if c < cur - 1e-14:
                cur = c
            else:
                h[i] -= d
                h[j] += d
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()

    # Phase 5: SLSQP refinement
    if budget_s - elapsed() > 40:
        n = best_n
        beta_final = 0.001
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': slsqp_eq_wrapper, 'jac': slsqp_jac_wrapper, 'args': (n,)}]
        try:
            res = minimize(slsqp_obj_wrapper, best_h.copy(), args=(n, beta_final),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-12})
            h_new = project_constraint(res.x, n / 2.0)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    # Final feasibility check
    best_h = project_constraint(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
