import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _lse_obj(h, t):
    corr = get_corr(h)
    mx = np.max(corr)
    s = np.clip(t * (corr - mx), -700.0, 0.0)
    return mx + np.log(np.sum(np.exp(s))) / t

    return _lse_obj(h, 50.0)

    return _lse_obj(h, 100.0)

    return _lse_obj(h, 200.0)

def obj_t1000(h):

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def pair_swap_descent(h, n, rng, max_iters=8000):
    bh = h.copy()
    bc = compute_c5(bh)
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.1, -0.1]
    for _ in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            ni = bh[i] + d
            nj = bh[j] - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            bh[i] = ni
            bh[j] = nj
            c = compute_c5(bh)
            if c < bc - 1e-15:
                bc = c
                break
            else:
                bh[i] -= d
                bh[j] += d
    return bh, bc
