def softmax_obj(x, n, beta, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    # Softmax of (corr - m) / beta
    e = np.exp((corr - m) / beta)
    val = dx * (m + beta * np.log(e.sum()))
    # Penalty for sum constraint
    val += penalty * (np.sum(x) - n / 2.0) ** 2
    return val
