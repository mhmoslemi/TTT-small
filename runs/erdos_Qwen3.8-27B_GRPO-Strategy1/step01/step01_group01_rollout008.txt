import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def obj_power(h, p):
    n = len(h)
    dx = 2.0 / n
    c = corr_raw(h) * dx
    c_pos = np.maximum(c, 0.0)
    return np.sum(c_pos ** p) ** (1.0 / p)

def obj_softplus(h, t):
    n = len(h)
    dx = 2.0 / n
    c = corr_raw(h)
    mx = np.max(c)
    s = t * (c - mx)
    s -= np.max(s)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def grad_softplus(h, t):
    n = len(h)
    dx = 2.0 / n
    c = corr_raw(h)
    mx = np.max(c)
    s = t * (c - mx)
    s -= np.max(s)
    w = np.exp(s) / np.sum(np.exp(s))
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    k = m_idx - (n - 1)
    a = j_idx - k
    b = j_idx + k
    a_valid = (a >= 0) & (a < n)
    b_valid = (b >= 0) & (b < n)
    a_clip = np.clip(a, 0, n - 1)
    b_clip = np.clip(b, 0, n - 1)
    term1 = np.where(a_valid, 1.0 - h[a_clip], 0.0)
    term2 = np.where(b_valid, -h[b_clip], 0.0)
    return dx * ((term1 + term2) @ w)

def grad_power(h, p):
    n = len(h)
    dx = 2.0 / n
    c = corr_raw(h) * dx
    c_pos = np.maximum(c, 0.0)
    S = np.sum(c_pos ** p)
    if S < 1e-30:
        return np.zeros(n)
    base = S ** (1.0 / p - 1.0)
    wc = p * c_pos ** (p - 1) * dx * base / p
    wc = wc / (S ** (1.0 / p))
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    k = m_idx - (n - 1)
    a = j_idx - k
    b = j_idx + k
    a_valid = (a >= 0) & (a < n)
    b_valid = (b >= 0) & (b < n)
    a_clip = np.clip(a, 0, n - 1)
    b_clip = np.clip(b, 0, n - 1)
    term1 = np.where(a_valid, 1.0 - h[a_clip], 0.0)
    term2 = np.where(b_valid, -h[b_clip], 0.0)
    return (term1 + term2) @ wc

def proj_sum(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.sum(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def ceq_fun(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceq_jac(x):
    return np.ones((1, len(x)))

def penalty_obj_t1(h, p, pw):
    c = corr_raw(h) * (2.0 / len(h))
    c_pos = np.maximum(c, 0.0)
    return np.sum(c_pos ** p) ** (1.0 / p) + pw * (np.sum(h) - len(h) / 2.0) ** 2

def penalty_obj_t2(h, t, pw):
    n = len(h)
    dx = 2.0 / n
    c = corr_raw(h)
    mx = np.max(c)
    s = t * (c - mx)
    s -= np.max(s)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx + pw * (np.sum(h) - n / 2.0) ** 2

def penalty_obj_t3(h, t, pw):
    return obj_softplus(h, t) + pw * (np.sum(h) - len(h) / 2.0) ** 2

def penalty_grad_t1(h, p, pw):
    g = grad_power(h, p) + 2.0 * pw * (np.sum(h) - len(h) / 2.0)
    return g

def penalty_grad_t2(h, t, pw):
    return grad_softplus(h, t) + 2.0 * pw * (np.sum(h) - len(h) / 2.0)

    rng = np.random.RandomState(seed)

    h_init = None

    h_init = proj_sum(h_init, n)
    h_best = h_init.copy()
    c_best = compute_c5(h_best)
    bnds = [(0.0, 1.0)] * n
    t1 = t0 + 0.35 * budget_s
    t2 = t0 + 0.65 * budget_s
    t3 = t0 + budget_s - 5
    # Phase 1: L-BFGS-B with power-sum + penalty

                hr = proj_sum(np.clip(res.x, 0, 1), n)

            except Exception:

    h_best = proj_sum(h_best, n)
