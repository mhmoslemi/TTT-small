    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    log_w = T * (corr - m) - np.log(np.sum(np.exp(T * (corr - m))))
    w = np.exp(log_w)
    
    grad = np.zeros(n)
