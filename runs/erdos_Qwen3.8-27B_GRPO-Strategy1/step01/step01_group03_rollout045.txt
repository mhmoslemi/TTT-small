        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_corr_lag(h, l):
    """Gradient of correlation at lag l w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= j - l < n:
            val += 1.0 - h[j - l]
        if 0 <= j + l < n:
            val -= h[j + l]
        g[j] = val
    return g

def lse_gradient(h, dx, beta):
    """Gradient of log-sum-exp approximation to max correlation"""
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    
    g = np.zeros(n)
    for j in range(n):
        for l in range(-(n-1), n):
            m_idx = (n-1) + l
            if w[m_idx] < 1e-15:
                continue
            val = 0.0
            if 0 <= j - l < n:
                val += 1.0 - h[j - l]
            if 0 <= j + l < n:
                val -= h[j + l]
            g[j] += w[m_idx] * val
    return g * dx

def slsqp_solve(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return x[n] - c
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.005]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def subgrad_optimize(h0, dx, target_sum, n_iter=5000, deadline=None):
    """Subgradient method with smoothing"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        beta = 5.0 + 20.0 * (it / n_iter)**0.5
        lr = 0.5 / (it + 10)**0.7
        
        g = lse_gradient(h, dx, beta)
        
        # Try different step sizes
        improved = False
        for shrink in [1.0, 0.5, 0.2, 0.1, 0.05]:
            h_new = project_sr(h - lr * shrink * g, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
        
        if not improved:
            # Try random perturbation
            h_try = h + 0.02 * np.random.randn(n)
            h_try = project_sr(h_try, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
    
    return best_h, best_c5
