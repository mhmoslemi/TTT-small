def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return np.full(n, target / n)
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return np.full(n, target / n)
        h += h * (diff / total)
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            for idx in np.argsort(1.0 - h):
                add = min(diff2, 1.0 - h[idx])
                h[idx] += add
                diff2 -= add
                if abs(diff2) < 1e-14:
                    break
        else:
            for idx in np.argsort(h):
                rem = min(-diff2, h[idx])
                h[idx] -= rem
                diff2 += rem
                if abs(diff2) < 1e-14:
                    break
    return h

def interpolate_h(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Multi-temperature SLSQP on the initial solution
    T_list = [5, 15, 50, 200, 800, 3000, 10000]
    for T in T_list:
        if elapsed() > budget_s * 0.30:
            break
        try:
            res = minimize(smooth_c5, best_h, args=(T,),
                          method='SLSQP', bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}],
                          options={'maxiter': 1000, 'ftol': 1e-15, 'iprint': 0})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Aggressive random coordinate local search (exact C5)
    for it in range(50000):
        if elapsed() > budget_s * 0.50:
            break
        j = np.random.randint(n)
        old = best_h[j]
        if old < 1e-12 or old > 1.0 - 1e-12:
            continue
        scale = 0.01 + 0.04 * np.random.random()
        delta = np.random.randn() * scale
        nv = old + delta
        if nv < -0.01 or nv > 1.01:
            continue
        nv = np.clip(nv, 0.0, 1.0)
        if abs(nv - old) < 1e-12:
            continue
        test = best_h.copy()
        test[j] = nv
        test = project_feasible(test, n)
        c5t = compute_c5(test)
        if c5t < best_c5 - 1e-14:
            best_h = test
            best_c5 = c5t

    # Phase 3: Pair transfer local search (preserves sum exactly)
    for it in range(80000):
        if elapsed() > budget_s * 0.70:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if best_h[i] < 1e-12 or best_h[j] > 1.0 - 1e-12:
            continue
        max_amt = min(best_h[i], 1.0 - best_h[j])
        if max_amt < 1e-12:
            continue
        amt = np.random.random() * max_amt * 0.4
        if amt < 1e-14:
            continue
        test = best_h.copy()
        test[i] -= amt
        test[j] += amt
        c5t = compute_c5(test)
        if c5t < best_c5 - 1e-14:
            best_h = test
            best_c5 = c5t

    # Phase 4: Try alternative n values (50, 70)
    if elapsed() < budget_s * 0.82:
        for n_alt in [50, 70]:
            if elapsed() > budget_s * 0.82:
                break
            h_alt = interpolate_h(best_h, n, n_alt)
            h_alt = project_feasible(h_alt, n_alt)
            for T in [100, 2000]:
                if elapsed() > budget_s * 0.82:
                    break
                try:
                    res = minimize(smooth_c5, h_alt, args=(T,),
                                  method='SLSQP', bounds=[(0.0, 1.0)] * n_alt,
                                  constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}],
                                  options={'maxiter': 600, 'ftol': 1e-14, 'iprint': 0})
                    hc = project_feasible(np.clip(res.x, 0, 1), n_alt)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                        n = n_alt
                except Exception:
                    pass

    # Phase 5: Final high-T SLSQP polish
    for T in [5000, 30000, 100000]:
        if elapsed() > budget_s * 0.94:
            break
        try:
            res = minimize(smooth_c5, best_h, args=(T,),
                          method='SLSQP', bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}],
                          options={'maxiter': 5000, 'ftol': 1e-18, 'iprint': 0})
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 6: Ultra-fine coordinate search on final solution
    if elapsed() < budget_s * 0.98:
        for sweep in range(3):
            for j in range(n):
                if elapsed() > budget_s * 0.98:
                    break
                base = best_h[j]
                for delta in np.linspace(-0.001, 0.001, 7):
                    nv = base + delta
                    if nv < 0 or nv > 1:
                        continue
                    test = best_h.copy()
                    test[j] = nv
                    test = project_feasible(test, n)
                    c5t = compute_c5(test)
                    if c5t < best_c5 - 1e-15:
                        best_h = test
                        best_c5 = c5t

    # Final: ensure exact sum constraint
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
