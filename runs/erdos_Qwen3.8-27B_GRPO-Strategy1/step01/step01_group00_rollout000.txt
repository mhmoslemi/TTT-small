        res = opt.minimize(
            objective_lse, h, args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def improve_cycle(h, n, t_start, deadline):
    h = snap_binary(h, 0.15)
    h, c5 = coord_descent(h, n, 20)
    if time.time() < deadline - 60:
        h = try_slsqp(h, n, 0.005, 300)
        h = fix_sum(h, n)
    h = snap_binary(h, 0.1)
    h, c5 = coord_descent(h, n, 15)
    c5 = compute_c5(h)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP at initial resolution
    for T in [0.05, 0.015, 0.005, 0.002]:
        if time.time() > deadline - 250:
            break
        h_opt = try_slsqp(h_best, n_best, T, 400)
        h_opt = fix_sum(h_opt, n_best)
        c5t = compute_c5(h_opt)
        if c5t < c5_best:
            h_best = h_opt
            c5_best = c5t

    # Phase 2: Discrete refinement cycle
    if time.time() < deadline - 200:
        h_c, c5c = improve_cycle(h_best, n_best, t_start, deadline)
        if c5c < c5_best:
            h_best, c5_best = h_c, c5c

    # Phase 3: SA on binary structure
    if time.time() < deadline - 150:
        h_bin = snap_binary(h_best, 0.1)
        h_sa, c5_sa = simulated_annealing(h_bin, n_best, 0.04, 0.0003, 4000)
        if c5_sa < c5_best:
            h_best, c5_best = h_sa.copy(), c5_sa
        # Second SA with different params
        if time.time() < deadline - 100:
            h_sa2, c5_sa2 = simulated_annealing(h_best, n_best, 0.02, 0.0001, 3000)
            if c5_sa2 < c5_best:
                h_best, c5_best = h_sa2.copy(), c5_sa2

    # Phase 4: SLSQP + discrete cycle again
    if time.time() < deadline - 100:
        for T in [0.003, 0.001]:
            if time.time() > deadline - 80:
                break
            h_opt = try_slsqp(h_best, n_best, T, 350)
            h_opt = fix_sum(h_opt, n_best)
            c5t = compute_c5(h_opt)
            if c5t < c5_best:
                h_best, c5_best = h_opt, c5t
        if time.time() < deadline - 60:
            h_c, c5c = improve_cycle(h_best, n_best, t_start, deadline)
            if c5c < c5_best:
                h_best, c5_best = h_c, c5c

    # Phase 5: Try higher resolutions
    for n_try in [400, 560, 700]:
        if time.time() > deadline - 80:
            break
        h_hi = upsample(h_best, n_try)
        h_opt = try_slsqp(h_hi, n_try, 0.02, 200)
        h_opt = fix_sum(h_opt, n_try)
        c5t = compute_c5(h_opt)
        if c5t < c5_best:
            h_best, c5_best, n_best = h_opt, c5t, n_try

    # Phase 6: Final polish
    if time.time() < deadline - 20:
        h_opt = try_slsqp(h_best, n_best, 0.001, 500)
        h_opt = fix_sum(h_opt, n_best)
        c5t = compute_c5(h_opt)
        if c5t < c5_best:
            h_best, c5_best = h_opt, c5t
        h_best = snap_binary(h_best, 0.08)
        h_best, c5_best = coord_descent(h_best, len(h_best), 12)

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    return (h_best, compute_c5(h_best), len(h_best))
