import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_obj(h, T):
    cv = corr_raw(h) * (2.0 / len(h))
    c_max = np.max(cv)
    if T < 1e-15:
        return c_max
    shifted = cv - c_max
    return c_max + T * np.log(np.sum(np.exp(shifted / T)))

def sum_con(h, n):
    return np.sum(h) - n * 0.5

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    ni = np.sum(interior)
    if ni > 0:
        h[interior] += (target - s) / ni
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argsort(h) if diff > 0 else np.argsort(-h)
        for i in idx[:20]:
            adj = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-12:
                break
    return h

def upsample(h, new_n):
    old_n = len(h)
    xo = np.arange(old_n) * (2.0 / old_n)
    xn = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(xn, xo, h)
    h_new *= (new_n * 0.5) / max(np.sum(h_new), 1e-15)
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, new_n)

def slsqp(h0, n, T, iters):
    r = opt.minimize(softmax_obj, h0, args=(T,), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                     options={'maxiter': iters, 'ftol': 1e-15})
    if hasattr(r, 'x') and np.all(np.isfinite(r.x)):
        return fix_sum(np.clip(r.x, 0.0, 1.0), n)
    return h0

def subgrad_step(h, n, alpha, nk=5):
    cv = corr_raw(h)
    idx = np.argsort(-cv)[:nk]
    grad = np.zeros(n)
    for ki in idx:
        lag = ki - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            p = -lag
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
        grad += g
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, n)
