def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, n):
    h = h.astype(np.float64).copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def project_feasible(h, n):
    h = h.astype(np.float64).copy()
    target = n / 2.0
    lo, hi = -h.max() - 0.1, 1.0 - h.min() + 0.1
    for _ in range(70):

        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid

    free = (h > 1e-12) & (h < 1.0 - 1e-12)
    if np.any(free) and abs(diff) > 1e-15:
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)

    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        idx = np.arange(0, n - k)
        grad[idx] += (1.0 - h[idx + k])
        grad[idx + k] += -h[idx]
    else:
        k2 = -k
        idx = np.arange(0, n - k2)
        grad[idx + k2] += (1.0 - h[idx])
        grad[idx] += -h[idx + k2]
    return grad

    corr = compute_corr(h)
    cmax = float(np.max(corr))
    thresh = cmax - 0.1 * np.std(corr)
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > top_k:
        order = np.argsort(corr[active])[::-1][:top_k]
        active = active[order]
    gs = np.zeros(n)
    for idx in active:
        lag = int(idx) - (n - 1)
        gs += grad_for_lag(h, lag)
    gs /= len(active)
    gs -= np.mean(gs)
    return gs, cmax

    m = np.max(corr)
    dx = 2.0 / n
    return dx * (m + beta * np.log(np.sum(np.exp((corr - m) / beta))))
