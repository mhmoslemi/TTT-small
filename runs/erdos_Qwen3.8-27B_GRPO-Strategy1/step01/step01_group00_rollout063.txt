    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2N-1

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_weights(corr, T):
    c_max = np.max(corr)
    w = np.exp((corr - c_max) / T)
    s = np.sum(w)
    if s > 0:
        w /= s
    else:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    return w

def compute_gradient(h, T):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = softmax_weights(corr, T)
    
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        m = k - (N - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        if m >= 0:
            if N - m > 0:
                grad[m:] += wk * (1.0 - h[:N - m])
                grad[:N - m] -= wk * h[m:]
        else:
            p = -m
            if N - p > 0:
                grad[:N - p] += wk * (1.0 - h[p:])
                grad[p:] -= wk * h[:N - p]
    
    return dx * grad

def project_onto_constraint(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    N = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    
    # Bisection on shift lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_try = np.clip(h - mid, 0.0, 1.0)
        s_try = np.sum(h_try)
        if s_try > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    
    # Fine correction
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        interior = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        n_int = np.sum(interior)
        if n_int > 0:
            h_proj[interior] += diff / n_int
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = new_n * 0.5
    h_new = project_onto_constraint(h_new, target)
    return h_new

def gradient_descent(h, T_schedule, lr_schedule, n_iters):
    """Run gradient descent with given schedules"""
    N = len(h)
    target_sum = N * 0.5
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(n_iters):
        T = T_schedule[i] if i < len(T_schedule) else T_schedule[-1]
        lr = lr_schedule[i] if i < len(lr_schedule) else lr_schedule[-1]
        
        grad = compute_gradient(h, T)
        g_norm = np.max(np.abs(grad))
        if g_norm < 1e-14:
            break
        
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            h = h_new
        else:

            h_new2 = project_onto_constraint(h_new2, target_sum)
            c5_2 = compute_c5(h_new2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                h = h_new2
            # else: no improvement, continue with same h
    
    return h, best_c5

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_weights(corr, T):
    c_max = np.max(corr)
    if T < 1e-15:
        w = np.zeros_like(corr)
        w[c_max == corr] = 1.0
        w /= np.sum(w)
        return w
    w = np.exp((corr - c_max) / T)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def compute_gradient(h, T):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = softmax_weights(corr, T)
    grad = np.zeros(N)
    threshold = 1e-10
    for k in range(2 * N - 1):
        m = k - (N - 1)
        wk = w[k]
        if wk < threshold:
            continue
        if m >= 0:
            if N - m > 0:
                grad[m:] += wk * (1.0 - h[:N - m])
                grad[:N - m] -= wk * h[m:]
        else:
            p = -m
            if N - p > 0:
                grad[:N - p] += wk * (1.0 - h[p:])
                grad[p:] -= wk * h[:N - p]
    return dx * grad

def project_constraint(h, target_sum):
    N = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        h_try = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_try) > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        interior = (h_proj > 1e-9) & (h_proj < 1.0 - 1e-9)
        n_int = int(np.sum(interior))
        if n_int > 0:
            h_proj[interior] += diff / n_int
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def upsample_fn(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=old_n)
    return project_constraint(h_new, new_n * 0.5)

def gradient_descent(h, n_iters, T0, T1, lr0, lr1, time_limit=None, t_start=None):
    N = len(h)
    target = N * 0.5
    h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    for i in range(n_iters):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        frac = i / max(n_iters - 1, 1)
        T = T0 * (T1 / T0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        grad = compute_gradient(h, max(T, 1e-6))
        gmax = np.max(np.abs(grad))
        if gmax < 1e-14:
            break
        h_new = h - lr * (grad / gmax) * 0.5
        h_new = project_constraint(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                h_new2 = h - lr * 0.1 * (grad / gmax)
                h_new2 = project_constraint(h_new2, target)
                c5_2 = compute_c5(h_new2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    h = h_new2
                    no_improve = 0
    return h, best_c5
