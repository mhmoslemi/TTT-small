import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = float(np.sum(h)) - target
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_lag(h, lag, N):
    """Correct gradient of correlation at given lag w.r.t. h."""
    g = np.zeros(N)
    if lag >= 0:
        if lag < N:
            g[lag:] += 1.0 - h[:N - lag]
            g[:N - lag] -= h[lag:]
    else:
        l = -lag
        if l < N:
            g[:N - l] += 1.0 - h[l:]
            g[l:] -= h[:N - l]
    return g

def optimize_nesterov(h, N, target, dx, t0, deadline, rng, top_k=12, lr=0.05, momentum=0.85):
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    v = np.zeros(N)
    it = 0
    top_idx = np.arange(2 * N - 1)
    w = np.ones(2 * N - 1)
    while time.time() < deadline:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_val - 1e-13:
            best_val = c5
            best_h = h.copy()
        if it % 40 == 1 or it == 1:
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
            tv = corr[top_idx]
            w_local = (tv / (tv.max() + 1e-12)) ** 2.0
        else:
            w_local = w  # reuse
        if it % 40 == 1:
            w = w_local
        grad = np.zeros(N)
        for idx, wi in zip(top_idx, w_local):
            lag = int(idx) - (N - 1)
            grad += wi * grad_lag(h, lag, N)
        v = momentum * v - lr * grad
        h = project_sum(h + v, target)
        if it % 250 == 0:
            lr *= 0.9
            v *= 0.5
            h = project_sum(h + rng.randn(N) * 0.02, target)
        if lr < 1e-10:
            break
    return best_h, best_val

def coord_descent(h, N, target, dx, t0, deadline, rng, top_k=8):
    h = h.copy()
    best_val = compute_c5(h, dx)
    step = dx * 0.3
    n_inner = 0
    while time.time() < deadline:
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        thresh = np.partition(corr, -top_k)[-top_k]
        c5_now = float(np.max(corr) * dx)
        for i in range(N):
            if time.time() > deadline:
                break
            for delta in (step, -step):
                old = h[i]
                h[i] = np.clip(old + delta, 0.0, 1.0)
                h = project_sum(h, target)
                c2 = compute_c5(h, dx)
                if c2 < c5_now - 1e-12:
                    c5_now = c2
                    improved = True
                else:
                    h[i] = np.clip(old, 0.0, 1.0)
                    h = project_sum(h, target)
        if c5_now < best_val:
            best_val = c5_now
        n_inner += 1
        if not improved or n_inner > 3:
            break
    h = project_sum(h, target)
    return h, compute_c5(h, dx)
