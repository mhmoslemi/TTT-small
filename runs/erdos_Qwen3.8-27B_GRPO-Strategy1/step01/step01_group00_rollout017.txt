    if abs(s - target) < 1e-13:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return fix_sum(h_new, new_n)

def bipolarize(h, p):
    h_new = np.where(h >= 0.5, h ** p, 1.0 - (1.0 - h) ** p)
    return fix_sum(h_new, len(h))

def coord_sweep(h, T, n_sweeps, deadline):
    n = len(h)
    m = 2 * n - 1
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            if time.time() > deadline:
                break
            h[j] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full')
            d = corr1 - corr0
            base = corr0
            lo, hi = 0.0, 1.0
            for _ in range(35):
                mid = (lo + hi) * 0.5
                val = base + mid * d
                mx = np.max(val)
                sh = val - mx
                w = np.exp(sh / T) / np.sum(np.exp(sh / T))
                g = np.sum(w * d)
                if g > 0:
                    hi = mid
                else:
                    lo = mid
            h[j] = (lo + hi) * 0.5
        h = fix_sum(h, n)
    return h

def try_slsqp(h, n, T, max_iter, deadline):
    elapsed = time.time()
    if elapsed > deadline:
        return h
    try:
        res = opt.minimize(
            smooth_max_obj, h,
            args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        if time.time() < deadline:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 10

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP with decreasing T at base resolution
    for T in [0.08, 0.03, 0.01, 0.004]:
        if time.time() > deadline - 200:
            break
        h_try = try_slsqp(h_best.copy(), n_best, T, 500, deadline)
        c = compute_c5(h_try)
        if c < c5_best:
            h_best, c5_best = h_try, c

    # Phase 2: Coordinate descent (fast local refinement)
    if time.time() < deadline - 150:
        h_cd = coord_sweep(h_best.copy(), 0.003, 25, deadline - 150)
        h_cd = fix_sum(h_cd, n_best)
        c = compute_c5(h_cd)
        if c < c5_best:
            h_best, c5_best = h_cd, c

    # Phase 3: Bipolarize + re-optimize (push toward {0,1})
    for p in [4, 7, 12]:
        if time.time() > deadline - 150:
            break
        h_bip = bipolarize(h_best, p)
        h_try = try_slsqp(h_bip, n_best, 0.005, 300, deadline)
        h_cd = coord_sweep(h_try, 0.002, 15, deadline)
        h_cd = fix_sum(h_cd, n_best)
        c = compute_c5(h_cd)
        if c < c5_best:
            h_best, c5_best = h_cd, c

    # Phase 4: Fine SLSQP refinement
    for T in [0.002, 0.001, 0.0005]:
        if time.time() > deadline - 100:
            break
        h_try = try_slsqp(h_best.copy(), n_best, T, 800, deadline)
        c = compute_c5(h_try)
        if c < c5_best:
            h_best, c5_best = h_try, c

    # Phase 5: Final coordinate descent
    if time.time() < deadline - 60:
        h_cd = coord_sweep(h_best.copy(), 0.001, 20, deadline - 60)
        h_cd = fix_sum(h_cd, n_best)
        c = compute_c5(h_cd)
        if c < c5_best:
            h_best, c5_best = h_cd, c

    # Phase 6: Random perturbations to escape local optima
    for scale in [0.06, 0.03, 0.015]:
        if time.time() > deadline - 80:
            break
        for _ in range(3):
            if time.time() > deadline - 60:
                break
            noise = np.random.randn(n_best) * scale
            h_pert = np.clip(h_best + noise, 0.0, 1.0)
            h_pert = fix_sum(h_pert, n_best)
            h_try = try_slsqp(h_pert, n_best, 0.008, 300, deadline)
            h_cd = coord_sweep(h_try, 0.002, 10, deadline)
            h_cd = fix_sum(h_cd, n_best)
            c = compute_c5(h_cd)
            if c < c5_best:
                h_best, c5_best = h_cd, c

    # Phase 7: Higher resolution attempt
    if time.time() < deadline - 300:
        n_up = 420
        h_up = upsample(h_best, n_up)
        for T in [0.03, 0.008]:
            if time.time() > deadline - 150:
                break
            h_try = try_slsqp(h_up, n_up, T, 400, deadline)
            h_cd = coord_sweep(h_try, 0.003, 10, deadline)
            h_cd = fix_sum(h_cd, n_up)
            c = compute_c5(h_cd)
            if c < c5_best:
                h_best, c5_best = h_cd, c
                n_best = n_up

    # Phase 8: Ultra-fine final polish
    if time.time() < deadline - 30:
        h_cd = coord_sweep(h_best.copy(), 0.0005, 10, deadline - 5)
        h_cd = fix_sum(h_cd, n_best)
        c = compute_c5(h_cd)
        if c < c5_best:
            h_best, c5_best = h_cd, c

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    n_final = len(h_best)
    return (h_best, c5_final, n_final)
