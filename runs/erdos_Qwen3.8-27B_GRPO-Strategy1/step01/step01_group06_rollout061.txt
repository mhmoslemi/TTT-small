Actually, coordinate descent for max of linear functions is interesting. For fixed h[-i], the objective is:

where a_l and b_l depend on the other coordinates. This is a convex piecewise-linear function of h_i, minimized at a breakpoint or boundary.

The breakpoints are where a_l1 * h_i + b_l1 = a_l2 * h_i + b_l2, i.e., h_i = (b_l2 - b_l1)/(a_l1 - a_l2).

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def grad_smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        l = idx - (n - 1)
        ws = w[idx]
        if ws < 1e-15:
            continue
        js = max(0, -l)
        je = min(n, n - l)
        if js < je:
            grad[js:je] += ws * (1.0 - h[js + l:je + l])
        js2 = max(0, l)
        je2 = min(n, n + l)
        if js2 < je2:
            grad[js2:je2] -= ws * h[js2 - l:je2 - l]
    return dx * grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(20):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            room = np.where(mask, 1.0 - h, 0.0)
            total = room.sum()
            if total < 1e-13:
                break
            h += room * (diff / total)
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            avail = h.copy()
            avail[~mask] = 0.0
            total = avail.sum()
            if total < 1e-13:
                break
            h -= avail * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_fn(h):
    return h.sum() - len(h) / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(int(seed))
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient descent with annealed temperature
    for T in [5, 20, 60, 200, 700, 2500, 8000, 30000]:
        if elapsed() > budget_s * 0.30:
            break
        h = best_h.copy()
        eta = 0.04
        decay = 0.9997
        n_iter = 3000
        for it in range(n_iter):
            if elapsed() > budget_s * 0.30:
                break
            g = grad_smooth_c5(h, T)
            h = h - eta * g
            h = project_feasible(h, n)
            eta *= decay
            if (it + 1) % 300 == 0:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: SLSQP with increasing temperature
    for T in [30, 100, 400, 1500, 5000, 20000]:
        if elapsed() > budget_s * 0.55:
            break
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_fn}]
        try:
            res = minimize(
                smooth_c5, best_h, args=(T,),
                method='SLSQP', bounds=bnds, constraints=cons,
                options={'maxiter': 600, 'ftol': 1e-13, 'iprint': 0}
            )
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 3: Nelder-Mead multi-start
    n_starts = 40
    for i in range(n_starts):
        if elapsed() > budget_s * 0.80:
            break
        scale = 0.12 * np.exp(-0.06 * i) + 0.005
        h0 = best_h + np.random.uniform(-scale, scale, size=n)
        h0 = project_feasible(h0, n)
        try:
            res = minimize(
                compute_c5, h0,
                method='Nelder-Mead',
                options={'maxiter': 1000, 'xatol': 1e-10, 'fatol': 1e-12, 'adapt': True}
            )
            hc = project_feasible(np.clip(res.x, 0, 1), n)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 4: Coordinate descent (exact 1D minimization per coordinate)
    if elapsed() < budget_s * 0.92:
        for sweep in range(5):
            if elapsed() > budget_s * 0.92:
                break
            h = best_h.copy()
            corr_full = 2 * n - 1
            for j in range(n):
                a = np.zeros(corr_full)
                for idx in range(corr_full):
                    l = idx - (n - 1)
                    term = 0.0
                    jp = j + l
                    jm = j - l
                    if 0 <= jp < n:
                        term += (1.0 - h[jp])
                    if 0 <= jm < n:
                        term -= h[jm]
                    a[idx] = term
                corr = np.correlate(h, 1.0 - h, mode='full')
                b = corr - a * h[j]
                # Convex minimization over [0,1]: ternary search
                lo, hi2 = 0.0, 1.0
                for _ in range(40):
                    m1 = lo + (hi2 - lo) / 3.0
                    m2 = hi2 - (hi2 - lo) / 3.0
                    f1 = np.max(a * m1 + b)
                    f2 = np.max(a * m2 + b)
                    if f1 < f2:
                        hi2 = m2
                    else:
                        lo = m1
                h[j] = (lo + hi2) / 2.0
            h = project_feasible(h, n)
            c5v = compute_c5(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 5: Final SLSQP polish
    if elapsed() < budget_s * 0.98:
        for T in [50000, 200000]:
            if elapsed() > budget_s * 0.985:
                break
            bnds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_fn}]
            try:
                res = minimize(
                    smooth_c5, best_h, args=(T,),
                    method='SLSQP', bounds=bnds, constraints=cons,
                    options={'maxiter': 800, 'ftol': 1e-14, 'iprint': 0}
                )
                hc = project_feasible(np.clip(res.x, 0, 1), n)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
