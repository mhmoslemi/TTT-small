```
def project(h):
    h = np.clip(h, 0, 1)
    # Adjust for sum constraint
    target = len(h) / 2
    current = np.sum(h)
    if abs(current - target) < 1e-10:
        return h
    # Binary search on offset
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        h_test = np.clip(h + mid, 0, 1)
        s = np.sum(h_test)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo+hi)/2, 0, 1)
```

```python
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    return np.interp(x_new, x_old, h)

def lbfgs_opt(h, temps, t0, time_limit, penalty=5000.0, maxiter=400):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for temp in temps:
        if time.time() - t0 > time_limit:
            break
        res = minimize(
            obj_fn, best_h, args=(temp, penalty),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        nh = repair(np.clip(res.x, 0.0, 1.0))
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh
    return best_h, best_c

def grad_polish(h, t0, time_limit, n_steps=200):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h)
    lr = 0.05
    temp = 0.03
    stuck = 0
    for it in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        cmax = np.max(corr)
        shifted = corr - cmax
        p = np.exp(shifted / temp)
        p /= np.sum(p)
        active_mask = p > 1e-6
        active_idx = np.where(active_mask)[0]
        A = np.zeros(n)
        B = np.zeros(n)
        for ai in active_idx:
            lag = ai - (n - 1)
            w = p[ai]
            if lag >= 0:
                A[:n - lag] += w * best_h[lag:]
                B[lag:] += w * best_h[:n - lag]
            else:
                ll = -lag
                A[ll:] += w * best_h[:n - ll]
                B[:n - ll] += w * best_h[ll:]
        g = 1.0 - A - B
        h_new = best_h - lr * g
        h_new = repair(h_new)
        c = compute_c5(h_new)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h_new
            stuck = 0
            lr = min(lr * 1.05, 0.3)
        else:
            stuck += 1
            lr *= 0.8
            if lr < 0.001:
                break
        if it % 40 == 39:
            temp *= 0.85
    return best_h, best_c

def random_perturb(h, sigma):
    n = len(h)
    h2 = h + np.random.randn(n) * sigma
    return repair(h2)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    h_init = repair(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    cur_h = h_init.copy()

    # Level 0: optimize at initial n
    temps0 = [0.1, 0.05, 0.02]
    cur_h, c = lbfgs_opt(cur_h, temps0, t0, budget_s * 0.10, maxiter=300)
    if c < best_c5:
        best_c5 = c
        best_h = cur_h.copy()

    # Level 1: upscale to 2x
    if time.time() - t0 < budget_s * 0.25:
        cur_h = repair(interp_up(cur_h, 2))
        temps1 = [0.05, 0.02, 0.01, 0.005]
        cur_h, c = lbfgs_opt(cur_h, temps1, t0, budget_s * 0.40, maxiter=400)
        if c < best_c5:
            best_c5 = c
            best_h = cur_h.copy()
        # Gradient polish
        cur_h, c = grad_polish(cur_h, t0, budget_s * 0.50)
        if c < best_c5:
            best_c5 = c
            best_h = cur_h.copy()

    # Level 2: upscale to 4x (total 4x original)
    if time.time() - t0 < budget_s * 0.45:
        cur_h = repair(interp_up(cur_h, 2))
        temps2 = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
        cur_h, c = lbfgs_opt(cur_h, temps2, t0, budget_s * 0.70, maxiter=500)
        if c < best_c5:
            best_c5 = c
            best_h = cur_h.copy()
        # Gradient polish at high resolution
        if time.time() - t0 < budget_s * 0.80:
            cur_h, c = grad_polish(cur_h, t0, budget_s * 0.80, n_steps=300)
            if c < best_c5:
                best_c5 = c
                best_h = cur_h.copy()

    # Random restarts at current resolution
    if time.time() - t0 < budget_s * 0.82:
        cur_n = len(cur_h)
        for ri in range(4):
            if time.time() - t0 > budget_s * 0.88:
                break
            sigma = 0.08 / (1.0 + ri * 0.5)
            h_p = random_perturb(best_h, sigma)
            h_p, c = lbfgs_opt(h_p, [0.01, 0.005, 0.002], t0, budget_s * 0.90, maxiter=250)
            if c < best_c5:
                best_c5 = c
                best_h = h_p.copy()

    # Final ultra-fine polish
    if time.time() - t0 < budget_s * 0.92:
        cur_h, c = lbfgs_opt(best_h, [0.003, 0.001, 0.0005], t0, budget_s * 0.96, maxiter=600)
        if c < best_c5:
            best_c5 = c
            best_h = cur_h.copy()
        if time.time() - t0 < budget_s * 0.97:
            cur_h, c = grad_polish(best_h, t0, budget_s * 0.98, n_steps=100)
            if c < best_c5:
                best_c5 = c
                best_h = cur_h.copy()

    # Final repair and verify
    best_h = repair(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
```
