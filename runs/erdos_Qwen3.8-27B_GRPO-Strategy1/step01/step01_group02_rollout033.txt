import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def weighted_gradient(h, w, N):
    """Compute gradient of weighted sum of correlations."""
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        wm = w[idx]
        if wm < 1e-8:
            continue
        m = idx - (N - 1)
        if m >= 0:
            if m < N:
                grad[:N - m] += wm * one_minus_h[m:]
                grad[m:] -= wm * h[:N - m]
        else:
            l = -m
            if l < N:
                grad[l:] += wm * one_minus_h[:N - l]
                grad[:N - l] -= wm * h[l:]
    return grad

def l2_gradient(h, N):
    """Gradient of sum of squared correlations (over all lags)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # d(sum corr^2)/d(h[j]) = 2 * sum_m corr[m] * dC_m/dh[j]
    w = 2.0 * corr
    return weighted_gradient(h, w, N)

def optimize_adam(h, N, target, dx, t0, budget_s, rng, lr_init=0.02, tau_init=5.0):
    """Adam optimizer with smooth max."""
    best_local = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    best_h = h.copy()
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    tau = tau_init
    patience = 0
    best_for_patience = best_local
    
    for it in range(8000):
        if time.time() - t0 > budget_s - 8:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_local:
            best_local = c5_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        # Softmax weights
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
        
        # Gradient
        grad = weighted_gradient(h, w, N)
        
        # Nesterov look-ahead
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        
        # Nesterov: compute gradient at look-ahead point
        h_lookahead = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        corr_la = np.correlate(h_lookahead, 1.0 - h_lookahead, mode='full')
        sc_la = tau * corr_la
        sc_la -= sc_la.max()
        w_la = np.exp(sc_la)
        w_la /= w_la.sum()
        grad_la = weighted_gradient(h_lookahead, w_la, N)
        
        # Adam update with look-ahead gradient
        m_buf2 = beta1 * m_buf + (1.0 - beta1) * grad_la
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad_la * grad_la)
        m_hat2 = m_buf2 / (1.0 - beta1 ** (it + 1))
        # v_hat already updated
        h = h - lr * m_hat2 / (np.sqrt(v_buf / (1.0 - beta2 ** (it + 1))) + eps)
        h = project_sum(h, target)
        
        # Adaptive tau and lr
        if (it + 1) % 200 == 0:
            tau *= 1.5
            lr *= 0.9
        if patience > 500:
            tau *= 2.0
            lr = max(lr * 0.8, 1e-6)
            patience = 0
        if lr < 1e-8:
            break
    
    return best_h, best_local

def optimize_l2_proxy(h, N, target, dx, t0, budget_s, max_iter=500):
    """Minimize ||corr||_2 as a proxy."""
    best_h = h.copy()
    best_val = float(np.sum(np.correlate(h, 1.0 - h, mode='full')**2) * dx**2)
    lr = 0.01
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iter):
        if time.time() - t0 > budget_s - 30:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.sum(corr**2))
        if val < best_val:
            best_val = val
            best_h = h.copy()
        
        grad = l2_gradient(h, N)
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)
        
        if (it + 1) % 100 == 0:
            lr *= 0.85
    
    return best_h

def local_search(h, N, target, dx, max_iters=2000):
    """Try local perturbations."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng_local = np.random.RandomState(12345)
    
    for it in range(max_iters):
        # Random perturbation: pick 1-3 random indices and nudge them
        n_perturb = rng_local.choice([1, 1, 1, 2, 2, 3])
        indices = rng_local.choice(N, size=n_perturb, replace=False)
        deltas = rng_local.uniform(-0.1, 0.1, size=n_perturb)
        
        h_try = h.copy()
        h_try[indices] += deltas
        h_try = project_sum(h_try, target)
        
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5:
            h = h_try.copy()
            best_c5 = c5_try
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def weighted_gradient(h, w, N):
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        wm = w[idx]
        if wm < 1e-9:
            continue
        m = idx - (N - 1)
        if m >= 0:
            if m < N:
                grad[:N - m] += wm * one_minus_h[m:]
                grad[m:] -= wm * h[:N - m]
        else:
            l = -m
            if l < N:
                grad[l:] += wm * one_minus_h[:N - l]
                grad[:N - l] -= wm * h[l:]
    return grad

def l2_gradient(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return 2.0 * weighted_gradient(h, corr, N)

def optimize_adam(h, N, target, dx, t0, budget_s, rng, lr_init=0.02, tau_init=5.0, max_iter=6000):
    best_local = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    best_h = h.copy()
    lr = lr_init
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    tau = tau_init
    stall = 0
    stall_threshold = best_local - 1e-10

    for it in range(max_iter):
        if time.time() - t0 > budget_s - 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_local:
            best_local = c5_val
            best_h = h.copy()
            stall = 0
            stall_threshold = best_local - 1e-12
        if c5_val > stall_threshold:
            stall += 1
        else:
            stall = 0

        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
        grad = weighted_gradient(h, w, N)

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h_new, target)

        if (it + 1) % 300 == 0:
            lr *= 0.88
            tau *= 1.6
        if stall > 400:
            tau *= 2.5
            lr = max(lr * 0.7, 1e-7)
            stall = 0
        if lr < 1e-9:
            break
    return best_h, best_local

def optimize_l2(h, N, target, dx, t0, budget_s, max_iter=200):
    best_h = h.copy()
    best_val = float(np.sum(np.correlate(h, 1.0 - h, mode='full')**2))
    lr = 0.005
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    for it in range(max_iter):
        if time.time() - t0 > budget_s - 40:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.sum(corr**2))
        if val < best_val:
            best_val = val
            best_h = h.copy()
        grad = l2_gradient(h, N)
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        h = project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        if (it + 1) % 80 == 0:
            lr *= 0.8
    return best_h

def local_search(h, N, target, dx, max_iters=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng_ls = np.random.RandomState(999)
    step = 0.05
    for it in range(max_iters):
        n_p = rng_ls.choice([1, 1, 1, 2, 2, 3])
        idx = rng_ls.choice(N, size=n_p, replace=False)
        h_try = h.copy()
        h_try[idx] += rng_ls.uniform(-step, step, size=n_p)
        h_try = project_sum(h_try, target)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5:
            h = h_try.copy()
            best_c5 = c5_try
            best_h = h.copy()
        elif it % 200 == 0:
            step *= 0.92
    return best_h, best_c5

def make_3val_init(N, c_val, rng):
    h = np.zeros(N)
    need = N / 2.0
    nc = int(round(need / (1.0 - c_val)))
    nc = max(1, min(N - 1, nc))
    n1 = N - nc
    # Adjust to hit exact sum
    h[:nc] = c_val
    h[nc:] = 1.0
    # Shuffle with clustering of 1s
    perm = np.arange(N)
    # Put 1s in blocks
    block_size = max(1, n1 // 2)
    positions_1 = []
    pos = 0
    while len(positions_1) < n1 and pos < N:
        end = min(pos + block_size, N - (n1 - len(positions_1)))
        positions_1.extend(range(pos, end))
        pos = end + 1
    positions_1 = positions_1[:n1]
    h2 = np.full(N, c_val)
    h2[positions_1] = 1.0
    h2 = project_sum(h2, N / 2.0)
    return h2
