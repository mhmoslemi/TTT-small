def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = 2.0 * n  # Wait, sum(h)*dx = 1, dx = 2/n, so sum(h) = n/2
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.astype(float)
    # Binary search for lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_new = np.clip(h - mid, 0.0, 1.0)
        s_new = np.sum(h_new)
        if s_new > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-14:
        interior = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h_new[interior] += diff / n_int
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def compute_grad(h, w):
    n = len(h)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        shift = k_idx - (n - 1)
        # First term: (1-h[m-shift]) for valid m
        m_lo = max(0, shift)
        m_hi = min(n, n + shift)
        if m_lo < m_hi:
            j_lo = m_lo - shift
            j_hi = m_hi - shift
            grad[m_lo:m_hi] += wk * (1.0 - h[j_lo:j_hi])
        # Second term: -h[m+shift] for valid m
        m_lo2 = max(0, -shift)
        m_hi2 = min(n, n - shift)
        if m_lo2 < m_hi2:
            j_lo2 = m_lo2 + shift
            j_hi2 = m_hi2 + shift
            grad[m_lo2:m_hi2] -= wk * h[j_lo2:j_hi2]
    return grad

def projected_step(h, grad, step, n):
    """Take a gradient step and project onto constraints."""
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new

def soft_max_obj(corr, T):
    c_max = np.max(corr)
    if T < 1e-14:
        return c_max
    shifted = corr - c_max
    max_shift = np.max(shifted)
    if max_shift < -700:
        return c_max
    exps = np.exp(shifted - max_shift)
    s = np.sum(exps)
    return c_max + T * (max_shift + np.log(s))

        return h

            lo = mid
        else:
            hi = mid

    h = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = np.sum(mask)
        if cnt > 0:

def sm_weights(corr, T):
    cmax = np.max(corr)
    if T < 1e-14:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
        return w
    sh = corr - cmax
    mx = np.max(sh)
    e = np.exp(sh - mx)
    return e / np.sum(e)

    cmax = np.max(corr)
    if T < 1e-14:
        return cmax
    sh = corr - cmax
    mx = np.max(sh)
    e = np.exp(sh - mx)

def grad_flux(h, w):
    n = len(h)
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-14:

        sh = ki - (n - 1)
        a = max(0, sh)
        b = min(n, n + sh)

        b2 = min(n, n - sh)
        if a2 < b2:
            g[a2:b2] -= wk * h[a2 + sh:b2 + sh]
    return g

    return cmax + T * (mx + np.log(np.sum(e)))

def grad_flux(h, w):
    n = len(h)
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-14:
            continue
        sh = ki - (n - 1)
        a = max(0, sh)
        b = min(n, n + sh)
        if a < b:
            g[a:b] += wk * (1.0 - h[a - sh:b - sh])
        a2 = max(0, -sh)
        b2 = min(n, n - sh)
        if a2 < b2:
            g[a2:b2] -= wk * h[a2 + sh:b2 + sh]
    return g

def nesterov_opt(h0, n, T, niter, t0, budget, min_rem=50):
    dx = 2.0 / n
    h = h0.copy()
    hp = h.copy()
    beta = 0.85
    alpha = 1.0 / n
    best_c5 = compute_c5(h)
    stagnant = 0
    for it in range(niter):
        if time.time() - t0 > budget - min_rem:
            break
        y = h + beta * (h - hp)
        y = np.clip(y, 0.0, 1.0)
        y = fix_sum(y, n)
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        w_y = sm_weights(corr_y, T)
        g_y = grad_flux(y, w_y) * dx
        h_new = y - alpha * g_y
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        c5_cur = compute_c5(h)
        if c5_new <= c5_cur + 1e-14:
            hp = h.copy()
            h = h_new
            alpha = min(alpha * 1.04, 3.0 / n)
            stagnant = 0
        else:
            alpha *= 0.65
            hp = h.copy()
            stagnant += 1
        if c5_new < best_c5:
            best_c5 = c5_new
        if stagnant > 100:
            break
    return h, best_c5

def slqp_polish(h0, n, T, maxiter, t0, budget, min_rem=40):
    dx = 2.0 / n
    def obj_f(vec):
        c = np.correlate(vec, 1.0 - vec, mode='full')
        return sm_value(c, T) * dx
    def con_f(vec):
        return np.sum(vec) - n * 0.5
    if time.time() - t0 > budget - min_rem:
        return h0
    res = opt.minimize(obj_f, h0.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_f}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
    hx = np.clip(res.x, 0.0, 1.0)
    hx = fix_sum(hx, n)
    return hx

def target_attack(h, n, t0, budget, n_attacks=30, min_rem=40):
    c5b = compute_c5(h)
    h_cur = h.copy()
    for _ in range(n_attacks):
        if time.time() - t0 > budget - min_rem:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_max = np.argmax(corr)
        sh = k_max - (n - 1)
        a = max(0, sh)
        b = min(n, n + sh)
        contrib = np.zeros(n)
        if a < b:
            contrib[a:b] = h_cur[a:b] * (1.0 - h_cur[a + sh:b + sh])
        contrib += contrib.copy()
        top = max(2, n // 8)
        idx = np.argsort(-contrib)[:top]
        h_try = h_cur.copy()
        h_try[idx] -= 0.005
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = fix_sum(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5b:
            h_cur = h_try
            c5b = c5_try
    return h_cur, c5b

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    hb = np.array(initial_h_values, dtype=float).copy()
    n = len(hb)
    c5b = compute_c5(hb)

    # Phase 1: Nesterov with T annealing (main workhorse)
    schedule = [(0.06, 500), (0.025, 600), (0.01, 700), (0.004, 600),
                (0.0015, 500), (0.0006, 400), (0.0002, 300)]
    for T, niter in schedule:
        if time.time() - t0 > budget_s - 150:
            break
        hb, c5b = nesterov_opt(hb, n, T, niter, t0, budget_s)

    # Phase 2: Targeted attack on worst shifts
    if time.time() - t0 < budget_s - 100:
        hb, c5b = target_attack(hb, n, t0, budget_s, n_attacks=40)

    # Phase 3: SLSQP polish
    if time.time() - t0 < budget_s - 80:
        hx = slqp_polish(hb, n, 0.001, 600, t0, budget_s, min_rem=50)
        c5x = compute_c5(hx)
        if c5x < c5b:
            hb, c5b = hx, c5x

    # Phase 4: More Nesterov at very small T
    if time.time() - t0 < budget_s - 60:
        hb, c5b = nesterov_opt(hb, n, 0.0002, 500, t0, budget_s, min_rem=30)

    # Phase 5: Another targeted attack
    if time.time() - t0 < budget_s - 40:
        hb, c5b = target_attack(hb, n, t0, budget_s, n_attacks=30, min_rem=25)

    # Phase 6: Final SLSQP
    if time.time() - t0 < budget_s - 25:
        hx = slqp_polish(hb, n, 0.0001, 1200, t0, budget_s, min_rem=15)
        c5x = compute_c5(hx)
        if c5x < c5b:
            hb, c5b = hx, c5x

    # Final cleanup
    hb = np.clip(hb, 0.0, 1.0)
    hb = fix_sum(hb, n)
    c5f = compute_c5(hb)
    return (hb, c5f, n)
