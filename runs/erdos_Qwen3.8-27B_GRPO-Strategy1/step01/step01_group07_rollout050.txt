    if abs(d) > 1e-12:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smoothmax(corr, beta, h):
    n = len(h)
    mc = np.max(corr)
    p = np.exp((corr - mc) / beta)
    p /= p.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        # Term 1: +w*(1-h[j-lag]) for 0<=j-lag<n
        a = max(0, lag)
        b = min(n - 1, n - 1 + lag)
        if a <= b:
            grad[a:b+1] += w * (1.0 - h[a-lag:b+1-lag])
        # Term 2: -w*h[j+lag] for 0<=j+lag<n
        a2 = max(0, -lag)
        b2 = min(n - 1, n - 1 - lag)
        if a2 <= b2:
            grad[a2:b2+1] -= w * h[a2+lag:b2+1+lag]
    return grad

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0, 1, old_n, endpoint=False)
    nx = np.linspace(0, 1, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - v.sum()) / new_n
    v = np.clip(v, 0.0, 1.0)
    v += (new_n / 2.0 - v.sum()) / new_n
    return np.clip(v, 0.0, 1.0)

def projected_gradient_step(h, beta, step_init=0.05, max_backtrack=30):
    n = len(h)
    target = n / 2.0
    corr = compute_corr(h)
    obj_cur = np.max(corr)
    grad = grad_smoothmax(corr, beta, h)
    # Backtracking
    step = step_init
    for _ in range(max_backtrack):
        h_new = project_feasible(h - step * grad, target)
        corr_new = compute_corr(h_new)
        obj_new = np.max(corr_new)
        if obj_new < obj_cur or step < 1e-10:
            return h_new, obj_new
        step *= 0.5
    return h_new, obj_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(z, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    d = target_sum - h.sum()
    if abs(d) > 1e-12:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smoothmax(corr, beta, h):
    n = len(h)
    mc = np.max(corr)

            obj_cur = np.max(corr)
            grad = grad_smoothmax(corr, beta, h)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            step = min(step, 0.5 / max(gnorm, 1e-8))

            if obj_new < obj_cur + 1e-14:

                step = min(step * 1.05, 0.5)
            else:

            if it % 100 == 99:
                step *= 0.7
    return h
