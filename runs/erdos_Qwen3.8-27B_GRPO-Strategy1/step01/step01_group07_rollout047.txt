    h = np.clip(h, 0, 1)
    # Iterative projection to satisfy both 0<=h<=1 and sum=n/2
    for _ in range(100):
        s = np.sum(h)
        if abs(s - n/2) < 1e-12:
            break
        delta = (n/2 - s) / n
        h2 = h + delta
        if np.min(h2) >= 0 and np.max(h2) <= 1:

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_box_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0, 1, old_n, endpoint=False)
    nx = np.linspace(0, 1, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    v = project_box_sum(v, new_n)
    return v

def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    return dx * (m + beta * np.log(s))

def eq_fun(x):
    return np.sum(x) - N_GLOBAL[0] / 2.0

def eq_jac(x):
    return np.ones(N_GLOBAL[0])

N_GLOBAL = [0]

def gradient_smooth_max(h, beta):
    """Compute gradient of dx * soft_max(corr, beta) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp((corr - m) / beta)
    s = np.sum(e)
    p = e / s
    val = dx * (m + beta * np.log(s))
    grad = np.zeros(n)
    for idx in range(len(corr)):
        pi = p[idx]
        if pi < 1e-13:
            continue
        lag = idx - (n - 1)
        js = max(0, -lag)
        je = min(n, n - lag)
        if je > js:
            grad[js:je] += pi * (1.0 - h[js+lag:je+lag])
        js2 = max(0, lag)
        je2 = min(n, n + lag)
        if je2 > js2:
            grad[js2:je2] += pi * (-h[js2-lag:je2-lag])
    grad *= dx
    return val, grad

def nesterov_optimize(h0, n, betas, t_start, t_end):
    """Nesterov accelerated projected gradient with backtracking"""
    h = h0.copy()
    v = np.zeros(n)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in betas:
        if time.time() - t_start > t_end:
            break
        v = np.zeros(n)
        step = 1.0
        it = 0
        max_it = 5000
        prev_val, _ = gradient_smooth_max(h, beta)
        stuck = 0
        for it in range(max_it):
            if time.time() - t_start > t_end:
                break
            _, g = gradient_smooth_max(h, beta)
            # Nesterov step
            y = h + v * step
            y = y - step * g
            h_new = project_box_sum(y, n)
            val_new, _ = gradient_smooth_max(h_new, beta)
            if val_new > prev_val + 1e-14:
                step *= 0.85
                if step < 1e-10:
                    step = 1e-10
                continue
            v = 0.9 * (h_new - h) + 0.1 * v
            h = h_new
            prev_val = val_new
            if it % 50 == 0:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            stuck += 1
            if stuck > 2000 and step < 1e-7:
                break
    return best_h, best_c5

def coord_descent(h, n, t_start, t_end, rng, max_iter=30000):
    cur = compute_c5(h)
    for it in range(max_iter):
        if time.time() - t_start > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.002, 0.03) * rng.choice([-1.0, 1.0])
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        s = np.sum(h)
        if abs(s - n / 2.0) > 1e-10:
            h[i], h[j] = h[i] - d * 0 + (hi - h[i]) + d, h[j] + d + (hj - h[j]) - d
            h[i], h[j] = hi, hj
            delta = n / 2.0 - np.sum(h)
            if abs(delta) > 1e-12:
                h[i] += delta * 0.5
                h[j] += delta * 0.5
                if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                    h[i], h[j] = hi, hj
                    h[:] = project_box_sum(h, n)
                    c = compute_c5(h)
                    if c <= cur + 1e-14:
                        cur = c
                    else:
                        h[i], h[j] = hi - d, hj + d
                        h[:] = project_box_sum(h, n)
                    continue
        c = compute_c5(h)
        if c < cur - 1e-13:
            cur = c
        else:
            h[i], h[j] = hi - d, hj + d
            h[:] = project_box_sum(h, n)
    return h, cur
