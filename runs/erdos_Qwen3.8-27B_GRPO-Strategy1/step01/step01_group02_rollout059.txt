            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:] += (1.0 - h[:N - m])
            g[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def binary_local_search(h0, N, t0, deadline, rng, max_iters=200000):
    dx = 2.0 / N
    target_int = N // 2
    h = (h0 > 0.5).astype(float)
    ones = np.where(h == 1)[0]
    if len(ones) > target_int:
        h[ones[target_int:]] = 0
        ones = ones[:target_int]
    elif len(ones) < target_int:
        zeros = np.where(h == 0)[0]
        need = target_int - len(ones)
        h[zeros[:need]] = 1
        ones = np.where(h == 1)[0]
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    zeros = np.where(h == 0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        oi = rng.randint(len(ones))
        zi = rng.randint(len(zeros))
        pi, pj = ones[oi], zeros[zi]
        h[pi] = 0.0
        h[pj] = 1.0
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best = h.copy()
            ones[oi] = pj
        else:
            h[pi] = 1.0
            h[pj] = 0.0
    return best, best_c5

def grad_opt(h, N, t0, deadline, rng, lr_init=0.012, mom_init=0.7, top_k=6, max_iters=4000):
    dx = 2.0 / N
    target = N / 2.0
    lags = np.arange(-(N - 1), N)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr = lr_init
    mom = mom_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = get_corr(h)
        c5_val = float(corr.max() * dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        top_idx = np.argsort(corr)[-top_k:]
        top_vals = corr[top_idx]
        spread = max(top_vals.max() - top_vals.min(), 1e-8)
        w = np.exp(-8.0 * (top_vals.max() - top_vals) / spread)
        w /= w.sum()
        grad = np.zeros(N)
        for i in range(top_k):
            lag = int(lags[top_idx[i]])
            grad += w[i] * grad_lag(h, lag, N)
        v = mom * v - lr * grad
        h = h + v
        h = project_sum(h, target)
        if (it + 1) % 400 == 0:
            lr *= 0.87
            mom = min(mom + 0.012, 0.94)
        if lr < 1e-10:
            break
    return best_h, best_c5

def smooth_binary(h_bin, N, t0, deadline, rng):
    h = h_bin.copy()
    h = h + 0.03 * rng.randn(N)
    h = project_sum(h, N / 2.0)
    return grad_opt(h, N, t0, deadline, rng, lr_init=0.01, mom_init=0.65, top_k=5, max_iters=2500)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / N0)

    # Phase 1: Gradient optimize parent directly (N0)
    dl = t0 + budget_s * 0.25
    h1, c5_1 = grad_opt(h0.copy(), N0, t0, dl, rng, lr_init=0.012, mom_init=0.7, top_k=6, max_iters=3000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary local search on N0
    dl2 = t0 + budget_s * 0.40
    h_b, c5_b = binary_local_search(h0, N0, t0, dl2, rng, max_iters=150000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    # Smooth and optimize the binary solution
    if c5_b < best_c5 + 0.001 and time.time() < dl2 + budget_s * 0.1:
        dl3 = min(dl2 + budget_s * 0.12, t0 + budget_s * 0.55)
        h_s, c5_s = smooth_binary(h_b, N0, t0, dl3, rng)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Upscale to 2*N0 and optimize
    N1 = 2 * N0
    if time.time() < t0 + budget_s * 0.80:
        dl4 = t0 + budget_s * 0.78
        x0 = np.linspace(0, 2, N0, endpoint=False)
        x1 = np.linspace(0, 2, N1, endpoint=False)
        h_interp = np.interp(x1, x0, best_h)
        h_interp = project_sum(h_interp, N1 / 2.0)
        h2, c5_2 = grad_opt(h_interp, N1, t0, dl4, rng, lr_init=0.009, mom_init=0.7, top_k=8, max_iters=4000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

        # Binary at N1
        if time.time() < t0 + budget_s * 0.92:
            dl5 = t0 + budget_s * 0.92
            h_b1 = (best_h[:N0] > 0.5).astype(float)
            h_b1_full, c5_b1 = binary_local_search(h_b1, N1, t0, dl5, rng, max_iters=80000)
            if c5_b1 < best_c5:
                best_c5 = c5_b1
                best_h = h_b1_full.copy()
            elif c5_b1 < best_c5 + 0.0005:
                h_s1, c5_s1 = smooth_binary(h_b1_full, N1, t0, dl5, rng)
                if c5_s1 < best_c5:
                    best_c5 = c5_s1
                    best_h = h_s1.copy()

    # Phase 4: Restart with perturbation on best solution
    if time.time() < t0 + budget_s - 25:
        dl6 = t0 + budget_s - 15
        N_best = len(best_h)
        h_pert = best_h + rng.randn(N_best) * 0.04
        h_pert = project_sum(h_pert, N_best / 2.0)
        h_r, c5_r = grad_opt(h_pert, N_best, t0, dl6, rng, lr_init=0.01, mom_init=0.72, top_k=5, max_iters=2000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Another binary restart
    if time.time() < t0 + budget_s - 10:
        dl7 = t0 + budget_s - 8
        N_best = len(best_h)
        h_rand_bin = np.zeros(N_best)
        h_rand_bin[rng.choice(N_best, size=N_best // 2, replace=False)] = 1.0
        h_rb, c5_rb = binary_local_search(h_rand_bin, N_best, t0, dl7, rng, max_iters=80000)
        if c5_rb < best_c5:
            best_c5 = c5_rb
            best_h = h_rb.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
