    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # softmax weights
    shifted = corr - np.max(corr)
    exps = np.exp(shifted / temp)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        if weights[k_idx] < 1e-8:

        r = k_idx - n + 1  # lag
        # ∂c[m]/∂h[x] = (1-h[x-r]) - h[x+r] for valid indices

            xr = x + r

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_simplex(vec, target_sum):
    """Project onto {0 <= x_i <= 1, sum(x) = target_sum}"""
    lo, hi = vec.min() - 1, vec.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(vec - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - (lo + hi) / 2, 0, 1)

def smooth_obj(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sm = cmax + temp * np.log(np.sum(np.exp((corr - cmax) / temp)))
    integ = np.sum(h) * dx
    return sm * dx + penalty * (integ - 1.0)**2

def subgradient_step(h, temp, step_size, n_active=100):
    """One subgradient step for the max-correlation objective"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights over correlations
    shifted = corr - np.max(corr)
    exps = np.exp(shifted / temp)
    weights = exps / np.sum(exps)
    
    # Only use top active constraints
    n_active = min(n_active, 2*n - 1)
    
    grad = np.zeros(n)
    # For efficiency, only compute for lags with significant weight
    active_mask = weights > np.max(weights) * 1e-4
    active_idx = np.where(active_mask)[0]
    
    for m in active_idx:
        r = m - n + 1
        w = weights[m]
        # Compute lag gradient for this r
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if r >= 0:
            term1[r:] = 1.0 - h[:n-r]
            term2[:n-r] = h[r:]
        else:
            r2 = -r
            term1[:n-r2] = 1.0 - h[r2:]
            term2[r2:] = h[:n-r2]
        grad += w * (term1 - term2)
    
    grad *= dx
    return grad

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def repair_h(h):
    n = len(h)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return h
    h2 = np.clip(h, 0.0, 1.0)
    h2 = project_box_simplex(h2, target)
    return h2

def subgradient_optimize(h, n_iterations, t0, budget_s):
    """Run subgradient method with decreasing step size"""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    target_sum = n / 2.0
    
    # Adaptive: start with larger temp, decrease
    for it in range(n_iterations):
        if time.time() - t0 > budget_s * 0.85:
            break
        temp = max(0.001, 0.05 * (0.995 ** it))
        step_size = 0.1 / (1 + 0.01 * it)
        
        grad = subgradient_step(h, temp, step_size)
        h_new = h - step_size * grad
        h_new = project_box_simplex(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

    n = len(h)
    bounds = [(0.0, 1.0)] * n
    schedule = [
        (0.02, 5000.0, 300),
        (0.005, 20000.0, 200),
        (0.001, 50000.0, 150),

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_simplex(vec, target_sum):
    n = len(vec)
    if target_sum <= 0 or target_sum >= n:
        return np.clip(vec, 0.0, 1.0)
    lo, hi = vec.min() - 1.0, vec.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(vec - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    sm = cmax + temp * np.log(np.sum(np.exp((corr - cmax) / temp)))
    integ = np.sum(h) * dx
    return sm * dx + penalty * (integ - 1.0) ** 2

def subgradient_step(h, temp=0.03, n_active=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - np.max(corr)
    exps = np.exp(shifted / temp)
    weights = exps / np.sum(exps)
    thresh = np.max(weights) * 1e-3
    active = np.where(weights > thresh)[0]
    if len(active) > n_active:
        top_idx = np.argsort(weights)[::-1][:n_active]
        active = top_idx
    grad = np.zeros(n)
    for m in active:
        r = m - n + 1
        w = weights[m]
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if r >= 0:
            t1[r:] = 1.0 - h[:n - r]
            t2[:n - r] = h[r:]
        else:
            r2 = -r
            t1[:n - r2] = 1.0 - h[r2:]
            t2[r2:] = h[:n - r2]
        grad += w * (t1 - t2)
    return grad * dx

def subgradient_optimize(h, n_iters, t0, budget_s, frac=0.8):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    target = n / 2.0
    for it in range(n_iters):
        if time.time() - t0 > t0 + budget_s * frac:
            break
        temp = max(0.005, 0.04 * 0.99 ** it)
        alpha = 0.15 / (1.0 + 0.005 * it)
        grad = subgradient_step(h, temp, 80)
        h_new = h - alpha * grad
        h_new = project_box_simplex(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def lbgfbs_optimize(h, t0, budget_s, frac=0.15):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    schedules = [(0.03, 3000.0, 200), (0.008, 10000.0, 200), (0.002, 30000.0, 150)]
    cur = h.copy()
    best_local = compute_c5(cur)
    for temp, pen, maxit in schedules:
        if time.time() - t0 > t0 + budget_s * frac:
            break
        res = minimize(smooth_obj, cur, args=(temp, pen), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10})
        nh = np.clip(res.x, 0.0, 1.0)
        nh = project_box_simplex(nh, n / 2.0)
        cv = compute_c5(nh)
        if cv < best_local:
            best_local = cv
        cur = nh
    return cur, best_local

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def repair_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    return project_box_simplex(h, n / 2.0)

def walsh_perturb(h, freq, amp):
    n = len(h)
    x = np.arange(n) / n
    w = np.sign(np.sin(np.pi * freq * x))
    h2 = h + amp * w * (1.0 - 2.0 * h)
    return repair_h(h2)

def block_swap(h, rng):
    n = len(h)
    blen = rng.randint(n // 10, n // 4)
    i = rng.randint(0, n - 2 * blen + 1)
    j = rng.randint(0, n - blen + 1)
    if abs(i - j) <= blen:
        j = (i + blen + rng.randint(blen, n - blen)) % (n - blen)
    h2 = h.copy()
    h2[j:j + blen], h2[i:i + blen] = h2[i:i + blen].copy(), h2[j:j + blen].copy()
    return repair_h(h2)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    n0 = len(h_init)

    # Phase 1: Subgradient at n0
    h_s = repair_h(h_init)
    h_s, c5_s = subgradient_optimize(h_s, 600, t0, budget_s, frac=0.12)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()

    # Phase 2: L-BFGS-B polish at n0
    h_p = h_s
    h_p, c5_p = lbgfbs_optimize(h_p, t0, budget_s, frac=0.10)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 3: Upscale to 2x
    if time.time() - t0 < t0 + budget_s * 0.30:
        h2 = repair_h(interp_upscale(h_p, 2))
        h2, c5_2 = subgradient_optimize(h2, 400, t0, budget_s, frac=0.10)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        h2, c5_2 = lbgfbs_optimize(h2, t0, budget_s, frac=0.15)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        cur_h, cur_n = h2, len(h2)
    else:
        cur_h, cur_n = h_p, n0

    # Phase 4: Upscale to 4x (or 2x more)
    if time.time() - t0 < t0 + budget_s * 0.55:
        h3 = repair_h(interp_upscale(cur_h, 2))
        h3, c5_3 = lbgfbs_optimize(h3, t0, budget_s, frac=0.20)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        cur_h, cur_n = h3, len(h3)

    # Phase 5: Multiple perturbation restarts
    n_restarts = 6
    for ri in range(n_restarts):
        if time.time() - t0 > t0 + budget_s * 0.88:
            break
        np.random.seed(seed + 1000 + ri)
        if ri % 3 == 0:
            sigma = 0.08 / (ri // 3 + 1)
            h_pert = np.clip(cur_h + np.random.randn(cur_n) * sigma, 0, 1)
            h_pert = repair_h(h_pert)
        elif ri % 3 == 1:
            hf = 1 + (ri % 5)
            h_pert = walsh_perturb(cur_h, hf, 0.12 / (ri // 3 + 1))
        else:
            h_pert = block_swap(cur_h, np.random.RandomState(seed + 2000 + ri))
        hf2, c5_hf = lbgfbs_optimize(h_pert, t0, budget_s, frac=0.04)
        if c5_hf < best_c5:
            best_c5 = c5_hf
            best_h = hf2.copy()

    # Final repair and evaluate
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
