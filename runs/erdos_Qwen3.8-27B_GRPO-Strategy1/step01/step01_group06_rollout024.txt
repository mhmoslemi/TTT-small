    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def blended_grad(h, T):
    """Compute gradient of smooth max (softmax-weighted) at given T."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        for i in range(n):
            v = 0.0
            if 0 <= i - s < n:
                v += (1.0 - h[i - s])
            if 0 <= i + s < n:
                v -= h[i + s]
            grad[i] += w[k] * v
    return grad

def slsqp_smooth(h, n, T, maxit=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxit, 'ftol': 1e-14})
    return project_h(np.clip(res.x, 0, 1), n)

def subgrad_line_search(h, n, base_alpha):
    """Subgradient step with backtracking line search on true max."""
    c_curr = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i - s < n:
            grad[i] += (1.0 - h[i - s])
        if 0 <= i + s < n:
            grad[i] -= h[i + s]
    for _ in range(10):
        h_try = project_h(h - base_alpha * grad, n)
        c_try = compute_c5(h_try)
        if c_try < c_curr - 1e-14:
            return h_try, c_try
        base_alpha *= 0.5
    return h, c_curr

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        nn = 60
        best_h = np.zeros(nn)
        best_h[:nn // 2] = 1.0
    best_h = project_h(best_h, len(best_h))
    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP smooth max with increasing temperature
    for T in [10, 50, 200, 800, 3000, 10000, 30000, 80000]:
        if elapsed() > budget_s * 0.22:
            break
        hc = slsqp_smooth(best_h, n, T, maxit=800)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-15:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Blended subgradient with line search (top-lag weighted)
    if elapsed() < budget_s * 0.50:
        alpha = 0.003
        stagnation = 0
        for it in range(4000):
            if elapsed() > budget_s * 0.50:
                break
            h_try, c_try = subgrad_line_search(best_h, n, alpha)
            if c_try < best_c5 - 1e-14:
                best_c5 = c_try
                best_h = h_try
                stagnation = 0
                alpha = min(alpha * 1.3, 0.01)
            else:
                stagnation += 1
                if stagnation > 200:
                    alpha *= 0.7
                    stagnation = 0
                if alpha < 1e-6:
                    break

    # Phase 3: SLSQP polish at very high T
    if elapsed() < budget_s * 0.70:
        for T in [10000, 50000, 150000, 400000]:
            if elapsed() > budget_s * 0.70:
                break
            hc = slsqp_smooth(best_h, n, T, maxit=600)
            c5v = compute_c5(hc)
            if c5v < best_c5 - 1e-15:
                best_c5 = c5v
                best_h = hc

    # Phase 4: Multi-start perturbations with SLSQP
    n_perturb = 25
    for i in range(n_perturb):
        if elapsed() > budget_s * 0.90:
            break
        scale = 0.04 * np.exp(-0.1 * i) + 0.003
        h0 = best_h + np.random.uniform(-scale, scale, size=n)
        h0 = project_h(h0, n)
        for T in [2000, 20000]:
            if elapsed() > budget_s * 0.90:
                break
            hc = slsqp_smooth(h0, n, T, maxit=300)
            c5v = compute_c5(hc)
            if c5v < best_c5 - 1e-15:
                best_c5 = c5v
                best_h = hc
                h0 = hc

    # Phase 5: Fine subgradient + final SLSQP
    if elapsed() < budget_s * 0.98:
        for it in range(3000):
            if elapsed() > budget_s * 0.97:
                break
            h_try, c_try = subgrad_line_search(best_h, n, 0.0008)
            if c_try < best_c5 - 1e-15:
                best_c5 = c_try
                best_h = h_try
        hc = slsqp_smooth(best_h, n, 200000, maxit=400)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-15:
            best_c5 = c5v
            best_h = hc

    # Final feasibility check
    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
