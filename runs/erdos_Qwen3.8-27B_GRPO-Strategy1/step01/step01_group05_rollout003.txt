        if abs(d) < 1e-13:

    bounds = [(0.0, 1.0)] * n
    res = minimize(make_obj(n, target, temp), h0, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(res.x, target)

def simulated_annealing(h, n, target, n_iter=50000, t_start=10.0, t_end=0.01):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    cur = h.copy()
    cur_c5 = best_c5
    for i in range(n_iter):
        T = t_start * (t_end / t_start) ** (i / n_iter)
        # Pick random coordinate to modify
        idx = np.random.randint(n)
        delta = np.random.uniform(-0.1, 0.1)
        cur_modified = cur.copy()
        cur_modified[idx] = np.clip(cur[idx] + delta, 0.0, 1.0)
        cur_modified = project(cur_modified, target)
        new_c5 = compute_c5(cur_modified, n)
        if new_c5 < cur_c5 or np.random.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10)):
            cur = cur_modified
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best = cur.copy()
                best_c5 = new_c5
    return best, best_c5

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * dx


def project(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            h = np.clip(h + d / len(h), 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + d / nf, 0.0, 1.0)
    return h


def make_obj(n, target, temp):
    def obj(x):
        h = project(x, target)
        return smooth_obj(h, n, temp)
    return obj


def lbgfb(h0, n, target, temp, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    res = minimize(make_obj(n, target, temp), h0, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(res.x, target)


def alt_pattern(n, period=2, hi=0.95, lo=0.05):
    x = np.arange(n)
    h = np.where(x % period == 0, hi, lo)
    return project(h, n / 2.0)


def structured_start(n, rng):
    x = np.arange(n)
    freqs = [1, 2, 3, 5]
    best = None
    best_c = 1e9
    for f in freqs:
        for phase in np.linspace(0, 2 * np.pi, 5, endpoint=False):
            h = 0.5 + 0.4 * np.sin(2 * np.pi * f * x / n + phase)
            h = project(h, n / 2.0)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best = h.copy()
    return best


def coordinate_polish(h, n, target, n_sweeps=30, step=0.01):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    for _ in range(n_sweeps):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            for delta in [-step, -step/2, step/2, step]:
                trial = best_h.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                # Adjust a random other coordinate to fix sum
                j = (i + 1 + np.random.randint(n - 1)) % n
                if j == i:
                    continue
                diff = trial[i] - best_h[i]
                trial[j] = np.clip(trial[j] - diff, 0.0, 1.0)
                trial = project(trial, target)
                c = compute_c5(trial, n)
                if c < best_c5 - 1e-10:
                    best_c5 = c
                    best_h = trial
                    improved = True
                    break
        if not improved:
            step *= 0.5
            if step < 1e-4:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    DEADLINE = budget_s - 15

    best_h = None
    best_c5 = 1e9
    best_n = 45

    # Start from initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n0 = len(h0)
    h0 = project(h0, n0 / 2.0)
    c0 = compute_c5(h0, n0)
    best_h, best_c5, best_n = h0, c0, n0

    def try_update(h, n):
        nonlocal best_h, best_c5, best_n
        h = project(h, n / 2.0)
        c = compute_c5(h, n)
        if c < best_c5:
            best_c5, best_h, best_n = c, h, n

    # Phase 1: L-BFGS-B temperature annealing on n0
    for temp in [2000, 500, 150, 50, 20, 7, 2]:
        if time.time() > DEADLINE - 80:
            break
        h_t = lbgfb(best_h, best_n, best_n / 2.0, temp, maxiter=600)
        try_update(h_t, best_n)

    # Phase 2: Differential evolution (global search on n0)
    if time.time() < DEADLINE - 120:
        def de_obj(x):
            h = project(x, best_n / 2.0)
            return smooth_obj(h, best_n, 30)
        try:
            de_res = differential_evolution(de_obj, [(0, 1)] * best_n,
                                            seed=seed, maxiter=60, popsize=40,
                                            tol=1e-5, mutation=(0.5, 1.0),
                                            recombination=0.8, polish=False)
            try_update(de_res.x, best_n)
            h_de = lbgfb(de_res.x, best_n, best_n / 2.0, 10, 400)
            try_update(h_de, best_n)
        except Exception:
            pass

    # Phase 3: Try even n values (44, 46) with alternating starts
    for n_try in [44, 46, 50]:
        if time.time() > DEADLINE - 60:
            break
        for period in [2, 4, 6]:
            h_alt = alt_pattern(n_try, period)
            h_alt = lbgfb(h_alt, n_try, n_try / 2.0, 100, 400)
            h_alt = lbgfb(h_alt, n_try, n_try / 2.0, 10, 400)
            try_update(h_alt, n_try)

    # Phase 4: Structured starts on best_n
    if time.time() < DEADLINE - 50:
        h_struct = structured_start(best_n, rng)
        if h_struct is not None:
            h_s = lbgfb(h_struct, best_n, best_n / 2.0, 50, 500)
            h_s = lbgfb(h_s, best_n, best_n / 2.0, 5, 400)
            try_update(h_s, best_n)

    # Phase 5: Random restarts with local optimization
    n_trials = 0
    while time.time() < DEADLINE - 40 and n_trials < 25:
        n_trials += 1
        scale = 0.15 * np.exp(-n_trials / 15.0) + 0.02
        h_rand = project(best_h + rng.uniform(-scale, scale, best_n), best_n / 2.0)
        h_r = lbgfb(h_rand, best_n, best_n / 2.0, 20, 300)
        try_update(h_r, best_n)

    # Phase 6: Nelder-Mead refinement
    if time.time() < DEADLINE - 30:
        nm_obj = make_obj(best_n, best_n / 2.0, 8)
        res = minimize(nm_obj, best_h, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-11, 'fatol': 1e-13})
        try_update(res.x, best_n)

    # Phase 7: Coordinate descent polish
    if time.time() < DEADLINE - 5:
        best_h, best_c5 = coordinate_polish(best_h, best_n, best_n / 2.0,
                                            n_sweeps=40, step=0.008)

    # Final L-BFGS-B polish
    if time.time() < DEADLINE:
        h_f = lbgfb(best_h, best_n, best_n / 2.0, 3, 300)
        try_update(h_f, best_n)

    best_h = project(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
