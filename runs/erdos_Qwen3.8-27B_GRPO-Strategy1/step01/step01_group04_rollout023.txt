    # Bisection on lambda
    lo, hi = y.min() - 1 - target_sum, y.max() + 1

    corr = np.correlate(h, 1.0 - h, mode='full')

    term2 = sw[n-1::-1] @ h

k=0, shift=-2: result[0] = sum_i a[i]*v[i+2]. i=0: a[0]*v[2]=1*0=0. Total=0. ✓

k=2, shift=0: result[2] = sum_i a[i]*v[i]. = 1*0+2*1+3*0=2. ✓

k=4, shift=2: result[4] = sum_i a[i]*v[i-2]. i=2: a[2]*v[0]=3*0=0. Total=0. ✓

So the formula is:
result[k] = sum_i a[i] * v[i - k + n - 1]  (for valid indices)

In our case: a = h, v = 1-h, both length n.
corr[k] = sum_i h[i] * (1-h)[i - k + n - 1]
= sum_i h[i] * (1 - h[i-k+n-1])

Now, d corr[k]/d h[m]:
h[m] appears in two places:
1. In h[i] when i=m: contributes (1-h[m-k+n-1]) if 0 <= m-k+n-1 < n
2. In (1-h[i-k+n-1]) when i-k+n-1 = m, i.e., i = m+k-n+1: contributes -h[m+k-n+1] if 0 <= m+k-n+1 < n

So d corr[k]/d h[m] = (1-h[m-k+n-1])*I[0<=m-k+n-1<n] - h[m+k-n+1]*I[0<=m+k-n+1<n]

Let j = m-k+n-1, so k = m-j+n-1
j ranges: 0 <= j < n, and 0 <= m-j+n-1 < 2n-1
For m in [0,n-1]: m-j+n-1 ranges from m (when j=n-1) to m+n-1 (when j=0).
All in [0, 2n-2] since m >= 0 and m+n-1 <= 2n-2. ✓

term1[m] = sum_{j=0}^{n-1} W[m-j+n-1] * (1-h[j])

Second term (term2[m]):
= sum_k W[k] * h[m+k-n+1] for 0 <= m+k-n+1 < n
Let j = m+k-n+1, so k = j-m+n-1
j ranges: 0 <= j < n, and 0 <= j-m+n-1 < 2n-1
For m in [0,n-1]: j-m+n-1 ranges from n-1-m (when j=0) to 2n-2-m (when j=n-1).
All in [0, 2n-2]. ✓

term2[m] = sum_{j=0}^{n-1} W[j-m+n-1] * h[j]

Now, using sliding_window_view:
sw = sliding_window_view(W, n)  # sw[m, i] = W[m+i] for m in [0, n-1], i in [0, n-1]

import numpy as np
from scipy.optimize import minimize
from numpy.lib.stride_tricks import sliding_window_view
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        nm = mask.sum()
        if nm > 0:
            x[mask] += diff / nm
        x = np.clip(x, 0.0, 1.0)
    return x

def obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_exact = float(np.max(corr) * dx)
    cmax = corr.max()
    shifted = (corr - cmax) / temp
    max_exp = np.max(shifted)
    exp_shifted = np.exp(shifted - max_exp)
    Z = np.sum(exp_shifted)
    W = exp_shifted / Z
    smax = (cmax + temp * np.log(Z) + temp * max_exp) * dx
    wh = 1.0 - h
    sw = sliding_window_view(W, n)
    term1 = sw @ wh[::-1]
    term2 = sw[n - 1 :: -1] @ h
    grad = dx * (term1 - term2)
    return smax, c5_exact, grad

def pg_optimize(h0, temps, iters_per_temp, t0, t_max):
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_v = compute_c5(h)
    for temp in temps:
        step = 1.0
        for it in range(iters_per_temp):
            if time.time() - t0 > t_max:
                return best_h, best_v
            smax, c5, grad = obj_grad(h, temp)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            # Armijo backtracking
            accepted = False
            for trial in range(8):
                y = h - step * grad
                hp = project_feasible(y, tgt)
                smax_new, _, _ = obj_grad(hp, temp)
                if smax_new < smax - 1e-8 * step * np.dot(grad, y - h):
                    h = hp
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                # Try a smaller fixed step
                step = 0.01 * temp
                y = h - step * grad
                h = project_feasible(y, tgt)
            # Track best using exact C5
            if it % 20 == 0 or it == iters_per_temp - 1:
                v = compute_c5(h)
                if v < best_v:
                    best_v = v
                    best_h = h.copy()
        # Reset step for next temperature
        step = 1.0
    final_v = compute_c5(h)
    if final_v < best_v:
        best_v = final_v
        best_h = h.copy()
    return best_h, best_v

def lbfgs_polish(h0, temp, t0, t_max):
    n = len(h0)
    tgt = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def penalized_obj(x, tp):
        smax, _, _ = obj_grad(x, tp)
        penalty = 1000.0 * (np.sum(x) - tgt) ** 2
        return smax + penalty

    def penalized_grad(x, tp):
        _, _, g = obj_grad(x, tp)
        penalty_g = 2000.0 * (np.sum(x) - tgt) * np.ones(n)
        return g + penalty_g

    res = minimize(
        penalized_obj, h0.copy(), args=(temp,),
        method='L-BFGS-B', bounds=bounds,
        jac=penalized_grad,
        options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-8}
    )
    if time.time() - t0 > t_max:
        return h0, compute_c5(h0)
    hh = np.clip(res.x, 0.0, 1.0)
    hh = project_feasible(hh, tgt)
    v = compute_c5(hh)
    return hh, v

def interp_up(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def random_perturb(h, sigma, rng):
    noise = rng.standard_normal(len(h)) * sigma
    h2 = np.clip(h + noise, 0.0, 1.0)
    return project_feasible(h2, len(h) / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_max = t0 + budget_s * 0.92

    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient at n=360 with multi-temperature schedule
    temps_main = [0.15, 0.06, 0.025, 0.01, 0.004, 0.0015, 0.0006]
    iters_main = [800, 1000, 1200, 1500, 1500, 2000, 3000]
    cur_h, cur_v = pg_optimize(h_init, temps_main, iters_main, t0, t_max)
    if cur_v < best_c5:
        best_c5 = cur_v
        best_h = cur_h.copy()

    # Phase 2: L-BFGS polish at smallest temp
    if time.time() - t0 < t_max * 0.7:
        cur_h, cur_v = lbfgs_polish(cur_h, 0.0005, t0, t_max)
        if cur_v < best_c5:
            best_c5 = cur_v
            best_h = cur_h.copy()

    # Phase 3: Upscale to 720 and refine
    if time.time() - t0 < t_max * 0.5:
        h720 = interp_up(best_h, 2)
        h720 = project_feasible(h720, 360.0)
        temps_up = [0.03, 0.01, 0.004, 0.001, 0.0004]
        iters_up = [600, 800, 1000, 1200, 1500]
        h720, v720 = pg_optimize(h720, temps_up, iters_up, t0, t_max)
        if v720 < best_c5:
            best_c5 = v720
            best_h = h720.copy()

    # Phase 4: Random restarts at n=360
    if time.time() - t0 < t_max * 0.55:
        for ri in range(5):
            if time.time() - t0 > t_max * 0.75:
                break
            sigma = 0.12 / (ri + 1)
            h_pert = random_perturb(best_h, sigma, rng)
            temps_r = [0.03, 0.01, 0.003, 0.001]
            iters_r = [400, 600, 800, 1000]
            h_r, v_r = pg_optimize(h_pert, temps_r, iters_r, t0, t_max)
            if v_r < best_c5:
                best_c5 = v_r
                best_h = h_r.copy()

    # Phase 5: Final L-BFGS polish
    if time.time() - t0 < t_max * 0.85:
        best_h, best_c5 = lbfgs_polish(best_h, 0.0003, t0, t_max)

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
