def lse_gradient(x, n, temp, penalty):

    corr = np.correlate(x, 1.0 - x, mode='full')
    Tc = temp * corr
    mx = np.max(Tc)
    p = np.exp(Tc - mx) / np.sum(np.exp(Tc - mx))
    
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    # Vectorized computation
    # For each j: grad[j] = dx * sum_m p[m] * ((1-x[j-s_m])*I - x[j+s_m])*I)
    # First term: sum_m p[m] * (1-x[j-s_m])*I[0<=j-s_m<n]
    # This is a correlation of (1-x) with p, but with the shift convention...
    
    # Let me just do the loop, n is small
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            sh = shifts[m]
            t1 = 0.0
            a = j - sh
            if 0 <= a < n:
                t1 = 1.0 - x[a]
            t2 = 0.0
            b = j + sh
            if 0 <= b < n:
                t2 = x[b]
            s += p[m] * (t1 - t2)
        grad[j] = dx * s + 2.0 * penalty * (np.sum(x) - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, n, iters=1000):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(iters):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h

def lse_obj(x, n, temp, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    Tc = temp * corr - temp * corr.max()
    e = np.exp(Tc)
    lse = (np.log(e.sum()) / temp + corr.max()) * dx
    return lse + pen * (x.sum() - n / 2.0) ** 2

def lse_grad(x, n, temp, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    Tc = temp * corr - temp * corr.max()
    e = np.exp(Tc)
    p = e / e.sum()
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            sh = shifts[m]
            a = j - sh
            b = j + sh
            t1 = (1.0 - x[a]) if 0 <= a < n else 0.0
            t2 = x[b] if 0 <= b < n else 0.0
            s += p[m] * (t1 - t2)
        grad[j] = dx * s + 2.0 * pen * (x.sum() - n / 2.0)
    return grad

    xc = np.clip(x, 0.0, 1.0)
    corr = np.correlate(xc, 1.0 - xc, mode='full')

    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(iters):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(free.sum())
        if nf == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def lse_obj(x, n, temp, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    Tc = temp * (corr - corr.max())
    e = np.exp(Tc)
    val = (np.log(np.sum(e)) / temp + corr.max()) * dx
    val += pen * (x.sum() - n / 2.0) ** 2

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    Tc = temp * (corr - corr.max())

    shifts = np.arange(2 * n - 1) - (n - 1)
