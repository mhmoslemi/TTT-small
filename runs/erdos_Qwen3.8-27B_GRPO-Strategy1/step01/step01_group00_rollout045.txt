    # Only use top_k significant weights
    sm_obj = (c_max + T * np.log(np.sum(weights))) * (2.0 / n)
    
    # Sort by weight, keep top_k
    k = min(top_k, len(corr))
    top_idx = np.argsort(weights)[-k:]
    weights_top = np.zeros(len(corr))
    weights_top[top_idx] = weights[top_idx]
    # Renormalize
    w_sum = np.sum(weights_top)
    if w_sum > 0:
        weights_top /= w_sum
    
    # Compute gradient
    offset = n - 1
    grad = np.zeros(n)
    for m in top_idx:
        w = weights_top[m]
        if w > 1e-12:
            lag = m - offset
            grad += w * grad_at_lag(h, lag, n)
    
    return sm_obj, grad

def project_onto_simplex_constraints(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) < 1e-14:
        return h
    # Distribute to interior points
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)

        step = diff / n_int
        step = np.clip(step, -h[interior] + 1e-12, 1.0 - h[interior] - 1e-12) if diff != 0 else 0
        # Actually just add evenly and reclip
        h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_at_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        L = -lag
        g[:n - L] += 1.0 - h[L:]
        g[L:] -= h[:n - L]
    return g

def smooth_grad(h, T, top_k=8):
    n = len(h)
    corr = corr_full(h)
    c_max = np.max(corr)
    dx = 2.0 / n
    if T < 1e-14:

        lag = idx - (n - 1)
        return c_max * dx, dx * grad_at_lag(h, lag, n)
    shifted = (corr - c_max) / T
    w = np.exp(shifted)
    w /= np.sum(w)

    wt = np.zeros(len(corr))
    wt[top] = w[top]
    ws = np.sum(wt)
    if ws < 1e-30:
        return obj, np.zeros(n)
    wt /= ws
    offset = n - 1

            grad += wt[m] * grad_at_lag(h, lag, n)
