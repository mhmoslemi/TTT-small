def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_of_coord(h, j, v):
    """Compute C5 when h[j] is set to v."""
    h_old = h[j]
    h[j] = v
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = np.max(corr) * dx
    h[j] = h_old
    return val

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coord_descent_step(h, j, lam):
    """Find optimal h[j] for fixed other coordinates."""
    n = len(h)
    target_sum = n / 2.0
    sum_others = h.sum() - h[j]

        h[j] = v
        n = len(h)

    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = n / 2.0 - h.sum()
    if diff > 1e-13:
        room = 1.0 - h
        h += room * (diff / room.sum())
    elif diff < -1e-13:
        h += h * (diff / h.sum())
    return np.clip(h, 0.0, 1.0)

def _cd_step(h, j, m, s, lo, hi, iters=30):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    h[j] = c; h[m] = s - c; fc = compute_c5(h)
    h[j] = d; h[m] = s - d; fd = compute_c5(h)
    for _ in range(iters):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            h[j] = c; h[m] = s - c; fc = compute_c5(h)
        else:
            lo, c, fc = c, d, fc
            d = lo + gr * (hi - lo)
            h[j] = d; h[m] = s - d; fd = compute_c5(h)
    h[j] = (lo + hi) / 2.0
    h[m] = s - h[j]

def pair_cd(h, n, sweeps, rng, iters=30):
    h = h.copy()
    for _ in range(sweeps):
        perm = rng.permutation(n)
        moved = False
        for i in range(n):
            j, m = perm[i], perm[(i+1) % n]
            s = h[j] + h[m]
            lo, hi = max(0.0, s-1.0), min(1.0, s)
            if hi - lo < 1e-12: continue
            a0 = h[j]
            _cd_step(h, j, m, s, lo, hi, iters)
            if abs(h[j] - a0) > 1e-10:
                moved = True
        if not moved:
            break
    return h

def interp_h(h, n_new):
    x = np.arange(len(h)) * 2.0 / len(h)
    y = np.arange(n_new) * 2.0 / n_new
    return np.interp(y, x, h)

def smooth_c5(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T*(corr-m))))/T) * 2.0/n

def sum_fun(h):
    return h.sum() - len(h)/2.0
