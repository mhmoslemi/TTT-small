    m = np.max(c)
    return m + T * np.log(np.exp((c - m) / T).sum())

def repair_h(h, n):

    if time.time() - t_start < budget_s - 120:
        for T in [3, 5, 10, 20]:
            if time.time() - t_start > budget_s - 120:
                break
            maxiter = min(200, max(40, int((budget_s - 120 - (time.time() - t_start)) * 4)))
            x_t = slsqp_opt(best_h, n, T, maxiter)
            x_t = repair_h(x_t, n)
            c5_t = compute_c5(x_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = x_t

    # Phase 4: Multi-start random perturbations
    if time.time() - t_start < budget_s - 100:
        for trial in range(40):
            if time.time() - t_start > budget_s - 100:
                break
            scale = 0.08 * np.exp(-trial / 15.0) + 0.01
            cand = best_h + rng.uniform(-scale, scale, n)
            cand = repair_h(cand, n)
            c5_c = compute_c5(cand, n)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = cand
            # Quick SLSQP from this start
            if time.time() - t_start > budget_s - 100:
                break
            x_t = slsqp_opt(cand, n, 10, 80)
            x_t = repair_h(x_t, n)
            c5_t = compute_c5(x_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = x_t

    # Phase 5: Try n=90 (finer resolution)
    if time.time() - t_start < budget_s - 80:
        n2 = 90
        h2 = interpolate_h(best_h, n2)
        c5_2 = compute_c5(h2, n2)
        for T in [20, 8, 4, 2]:
            if time.time() - t_start > budget_s - 80:
                break
            maxiter = min(250, max(50, int((budget_s - 80 - (time.time() - t_start)) * 3)))
            x_t = slsqp_opt(h2, n2, T, maxiter)
            x_t = repair_h(x_t, n2)
            c5_t = compute_c5(x_t, n2)
            if c5_t < c5_2:
                c5_2 = c5_t
                h2 = x_t
        # Subgradient on n=90
        if time.time() - t_start < budget_s - 40:
            for it in range(4000):
                if time.time() - t_start > budget_s - 40:
                    break
                alpha = 0.06 / (1.0 + it * 0.002)
                h2 = subgrad_step(h2, n2, alpha, rng)
                c5_t = compute_c5(h2, n2)
                if c5_t < c5_2:
                    c5_2 = c5_t
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            best_n = n2

    # Phase 6: Final polish - coordinate descent on best
    if time.time() - t_start < budget_s - 15:
        h_cd, c5_cd = coord_descent(best_h, best_n, 2, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Final SLSQP polish
    if time.time() - t_start < budget_s - 5:
        rem = budget_s - 5 - (time.time() - t_start)
        if rem > 5:
            maxiter = max(30, int(rem * 6))
            x_t = slsqp_opt(best_h, best_n, 3, maxiter)
            x_t = repair_h(x_t, best_n)
            c5_t = compute_c5(x_t, best_n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = x_t

    best_h = repair_h(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, final_c5, best_n
