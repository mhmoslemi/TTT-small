import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def topk_smooth_max(corr, k, temp):
    """Smooth max using top-k values with log-sum-exp"""
    k = min(k, len(corr))
    # Get top k values
    idx = np.argpartition(corr, -k)[-k:]
    top = corr[idx]
    cmax = np.max(top)
    shifted = top - cmax
    return cmax + temp * np.log(np.sum(np.exp(shifted / temp)))

def obj_fn(h, temp, topk, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    smax = topk_smooth_max(corr, topk, temp)
    integral = np.sum(h) * dx
    return smax * dx + penalty * (integral - 1.0) ** 2

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Project to sum = target
    for _ in range(100):

        if abs(diff) < 1e-10:

        h = h + diff / n

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def topk_smooth_max(corr, k, temp):
    k = min(k, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    top = corr[idx]
    cmax = np.max(top)
    shifted = top - cmax
    return cmax + temp * np.log(np.sum(np.exp(shifted / temp)))


def obj_fn(h, temp, topk, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    smax = topk_smooth_max(corr, topk, temp)
    integral = np.sum(h) * dx
    return smax * dx + penalty * (integral - 1.0) ** 2


def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
