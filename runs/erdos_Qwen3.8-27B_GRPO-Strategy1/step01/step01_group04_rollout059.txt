    n = len(h)

        # Term 2: j appears as i+s, so i=j-s, term is h[j-s]*(1-h[j]), gradient w.r.t. h[j] is -h[j-s]
        #   valid if j-s in [0,n-1]

    """Project h onto {0<=h<=1, sum(h)=target_sum} approximately."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > target_sum + 1e-10:
        # Need to reduce: lower values at positions where h is smallest
        excess = current_sum - target_sum
        # Reduce uniformly among non-clipped-at-0 elements
        order = np.argsort(h)
        for idx in order:
            if excess <= 0:
                break
            reduce = min(h[idx], excess)
            h[idx] -= reduce
            excess -= reduce
        # If still above (all were 1), clip
        h = np.clip(h, 0.0, 1.0)

        order = np.argsort(-h)

            if deficit <= 0:
                break
            add = min(1.0 - h[idx], deficit)
            h[idx] += add
            deficit -= add
        h = np.clip(h, 0.0, 1.0)
    return h

        return h, compute_c5(h)
    nh = np.clip(res.x, 0.0, 1.0)
    nh = repair_h(nh)
    return nh, compute_c5(nh)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Binary local search at initial resolution
    max_t1 = t0 + budget_s * 0.12
    h_bin, c5_bin = binary_local_search(h_init, t0, max_t1, max_flips=100000)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Phase 2: Chase optimization at initial resolution
    max_t2 = t0 + budget_s * 0.30
    h_work = best_h.copy()
    h_work, c5_work = chase_optimize(h_work, t0, max_t2, top_k=12, n_iter=20000, base_step=0.03)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()
    
    # Phase 3: L-BFGS refinement at initial resolution
    max_t3 = t0 + budget_s * 0.45
    h_ref, c5_ref = l_bfgs_refine(best_h, t0, max_t3, max_iter=300)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref.copy()
    
    # Phase 4: Upscale x2 and optimize
    max_t4 = t0 + budget_s * 0.70
    h_up = interp_upscale(best_h, 2)
    h_up = repair_h(h_up)
    h_up, c5_up = chase_optimize(h_up, t0, max_t4, top_k=20, n_iter=25000, base_step=0.02)
    if c5_up < best_c5:
        best_c5 = c5_up
        best_h = h_up.copy()
    
    # Phase 5: Second upscale x2 and optimize
    max_t5 = t0 + budget_s * 0.88
    h_up2 = interp_upscale(best_h, 2)
    h_up2 = repair_h(h_up2)
    h_up2, c5_up2 = chase_optimize(h_up2, t0, max_t5, top_k=25, n_iter=20000, base_step=0.015)
    if c5_up2 < best_c5:
        best_c5 = c5_up2
        best_h = h_up2.copy()
    
    # Phase 6: Multi-start with structured perturbations on best so far
    if time.time() < t0 + budget_s * 0.82:
        n_cur = len(best_h)
        for trial in range(5):
            if time.time() > t0 + budget_s * 0.93:
                break
            h_try = structured_perturb(best_h, rng)
            remaining = max(1.0, (t0 + budget_s * 0.95) - time.time())
            h_try, c5_try = chase_optimize(h_try, t0, t0 + budget_s * 0.95,
                                           top_k=12, n_iter=int(remaining * 3000),
                                           base_step=0.02)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    # Final L-BFGS polish
    if time.time() < t0 + budget_s * 0.90:
        final_h, final_c5 = l_bfgs_refine(best_h, t0, t0 + budget_s * 0.96, max_iter=150)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    # Final repair
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
