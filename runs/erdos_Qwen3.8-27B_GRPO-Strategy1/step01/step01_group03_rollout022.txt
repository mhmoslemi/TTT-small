import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

def slsqp_active(h0, dx, target_sum, maxiter=200, n_active=15):
    """SLSQP focusing on active (near-max) lag constraints only."""
    n = len(h0)
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    cmax = corr.max()
    # Select active lags: those within some fraction of max
    thresh = cmax - 0.05 * cmax  # or absolute threshold
    active_idx = np.where(corr >= thresh)[0]
    if len(active_idx) > n_active:
        # Keep only the top n_active
        order = np.argsort(corr)[::-1][:n_active]
        active_idx = order
    
    n_cons = len(active_idx)
    
    def obj(x):
        return x[n]
    
    def ineq(x):
        h = x[:n]
        c = np.correlate(h, 1.0 - h, mode='full')
        return x[n] - c[active_idx]
    
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    
    t0 = cmax + 0.01
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.x is not None:
            h_new = project_sr(res.x[:n], target_sum)
            return h_new
    except Exception:
        pass
    return h0.copy()

def slsqp_full(h0, dx, target_sum, maxiter=300):
    """SLSQP with all constraints."""
    n = len(h0)
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    cmax = corr.max()
    
    def obj(x):
        return x[n]
    
    def ineq(x):
        h = x[:n]
        c = np.correlate(h, 1.0 - h, mode='full')
        return x[n] - c
    
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    
    x0 = np.concatenate([h0, [cmax + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def projected_gradient(h0, dx, target_sum, n_iter=2000, deadline=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        beta = 20.0 + 40.0 * min(1.0, it / max(1, n_iter // 2))
        lr = 0.05 / (1.0 + 0.01 * it)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = corr.max()
        w = np.exp(beta * (corr - cmax))
        w /= w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for j in range(n):
            s = 0.0
            for m_idx in range(2 * n - 1):
                m = m_idx - (n - 1)
                val = 0.0
                if 0 <= j - m < n:
                    val += (1.0 - h[j - m])
                if 0 <= j + m < n:
                    val -= h[j + m]
                if w[m_idx] > 1e-15:
                    s += w[m_idx] * val
            grad[j] = s
        grad *= dx
        
        # Line search
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = project_sr(h - lr * shrink * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                break
    
    return best_h, best_c5

    return best_h, best_c5, best_n
