import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, temp, mult):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    if temp < 1e-10:
        sm = mx
    else:
        sm = mx + temp * np.log(np.sum(np.exp((corr - mx) / temp)))
    c5 = sm * dx
    s = np.sum(h) - n * 0.5
    return c5 + mult * s * s

def repair(h):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = len(h) * 0.5
    if s > 0:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    return h

def upscale(h, f):
    n = len(h)
    m = n * f
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(m) + 0.5) / m
    return np.interp(xn, xo, h)

def opt_phase(h, t0, budget, max_iters=500):
    """Run L-BFGS-B for max_iters iterations."""
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_smooth, h, args=(0.01, 1e6), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iters, 'ftol': 1e-14, 'gtol': 1e-10})
    return np.clip(res.x, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_fn(h, temp, mult):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    if temp > 1e-10:
        sm = mx + temp * np.log(np.sum(np.exp((corr - mx) / temp)))
    else:
        sm = mx
    s = np.sum(h) - 0.5 * n
    return sm * dx + mult * s * s

def repair(h):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    tgt = 0.5 * h.size
    if s > 1e-12:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
    return h

def up(h, f):
    n = h.size
    m = n * f
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(m) + 0.5) / m
    return np.interp(xn, xo, h)

def do_lbfgs(h, temp, maxit):
    n = h.size
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_fn, h, args=(temp, 1e5), method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-11})
    return repair(np.clip(res.x, 0.0, 1.0))

def local_search(h, t0, deadline, best_ref):
    n = h.size
    c_cur = compute_c5(h)
    best_ls = c_cur
    best_h_ls = h.copy()
    T = 5e-5
    stall = 0
    while time.time() < deadline:
        roll = np.random.rand()
        if roll < 0.6:
            bl = np.random.choice([5, 10, 20, 40])
            if bl >= n:
                bl = max(5, n // 10)
            st = np.random.randint(0, n - bl + 1)
            h2 = h.copy()
            sig = 0.02 + 0.03 * (bl / 40.0)
            h2[st:st + bl] += np.random.randn(bl) * sig
        elif roll < 0.8:
            bl = np.random.choice([10, 20, 50])
            if bl >= n:
                bl = max(5, n // 10)
            st = np.random.randint(0, n - bl + 1)
            h2 = h.copy()
            direction = np.random.choice([-1, 1])
            h2[st:st + bl] += direction * 0.05
        else:
            bl = np.random.randint(40, max(50, n // 5))
            st = np.random.randint(0, n - bl + 1)
            h2 = h.copy()
            h2[st:st + bl] = 1.0 - h2[st:st + bl]
        h2 = repair(h2)
        c2 = compute_c5(h2)
        if c2 < c_cur:
            h, c_cur = h2, c2
            stall = 0
            if c2 < best_ls:
                best_ls = c2
                best_h_ls = h2.copy()
        else:
            stall += 1
            if np.random.rand() < np.exp(-(c2 - c_cur) / T):
                h, c_cur = h2, c2
        T *= 0.9995
        if T < 5e-8:
            T = 5e-8
        if stall > 2000:
            h = repair(best_h_ls + np.random.randn(n) * 0.03)
            c_cur = compute_c5(h)
            stall = 0
            T = 5e-5
    return h, c_cur, best_h_ls, best_ls
