        saved = h[j]

        h[j] = 0.0
        c0 = np.correlate(h, 1.0 - h, mode='full')
        
        # Compute correlation with h[j] = 1
        h[j] = 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        
        # a = c0, b = c1 - c0
        # Minimize max(a + t*b) for t in [0,1]
        a = c0
        b = c1 - c0
        
        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(50):
            m1 = lo + (hi - lo) * 0.333
            m2 = hi - (hi - lo) * 0.333
            if np.max(a + m1 * b) < np.max(a + m2 * b):
                hi = m2
            else:
                lo = m1
        h[j] = (lo + hi) * 0.5
    
    # Project to maintain sum constraint
    h = proj(h, n)
    return h

                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 500, 'ftol': 1e-16,
                                        'maxfun': 30000, 'maxls': 30})
                h2 = proj(np.clip(res.x, 0.0, 1.0), n)
                c5 = compute_c5(h2)
                if c5 < bc:
                    bc = c5
                    bh = h2
            except Exception:
                pass

    # Phase 3: Final coordinate sweeps
    while time.time() < t0 + budget_s - 5:
        h = bh.copy()
        order = rng.permutation(n)
        h = sweep(h, n, order)
        c5 = compute_c5(h)
        if c5 < bc:
            bc = c5
            bh = h

    return bh, compute_c5(bh), n
