    one_minus_h = 1.0 - h

        # Try each coordinate (or random subset)
        i = rng.randint(n)
        # Try delta up and down
        for d in [0.001, 0.005, 0.01, 0.02, 0.05]:
            for sign in [1, -1]:
                delta = sign * d
                new_val = h[i] + delta
                if new_val < 0 or new_val > 1:
                    continue
                # Adjust another coordinate to maintain sum
                # Or adjust all others proportionally
                # Simpler: pick j != i and adjust
                j = (i + 1 + rng.randint(n-1)) % n
                h[i] = new_val
                h[j] = clip(h[j] - delta, 0, 1)
                if abs(sum(h) - n/2) > 1e-6:
                    h[i] -= delta  # revert
                    continue
                new_c5 = compute_c5(h)
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                else:
                    h[i] -= delta
                    h[j] += delta  # revert
        # Renormalize
        h += (n/2 - sum(h)) / n
        h = clip(h, 0, 1)
    
    return h, cur_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    return dx * (m + beta * np.log(s))

def project_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.linspace(0, 1, old_n, endpoint=False)
    nx = np.linspace(0, 1, new_n, endpoint=False)
    v = np.interp(nx, ox, parent, period=1)  # periodic interp
    v = project_sum(v, new_n)
    return v

import numpy as np
from scipy.optimize import minimize
import time

_Ng = 280

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    return dx * (m + beta * np.log(s))

def eq_fun(x):
    return np.sum(x) - _Ng / 2.0

def eq_jac(x):
    return np.ones_like(x)

def project_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64) / old_n
    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    v = project_sum(v, new_n)
    return v

def sa_phase(h, n, rng, t_end, t0, budget_s):
    global best_h_ref, best_c5_ref
    cur = compute_c5(h)
    T = 0.004
    patience = 0
    best_local = (h.copy(), cur)
    for step in range(500000):
        if time.time() > t_end:
            break
        T *= 0.9997
        n_mod = rng.randint(1, 4)
        idx = rng.randint(0, n, n_mod)
        deltas = rng.uniform(-0.015, 0.015, n_mod)
        h_new = h.copy()
        for k in range(n_mod):
            h_new[idx[k]] += deltas[k]
        h_new = project_sum(h_new, n)
        new_c5 = compute_c5(h_new)
        delta = new_c5 - cur
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            h = h_new
            cur = new_c5
            patience = 0
            if cur < best_c5_ref:
                best_c5_ref = cur
                best_h_ref = h.copy()
            if cur < best_local[1]:
                best_local = (h.copy(), cur)
        else:
            patience += 1
        if patience > 3000:
            h = best_local[0].copy()
            cur = best_local[1]
            patience = 0
    return h, cur

def coord_descent(h, n, rng, t_end):
    cur = compute_c5(h)
    step_sizes = [0.002, 0.005, 0.01, 0.02, 0.04]
    for sweep in range(500000):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            j = (i + 1 + rng.randint(n - 1)) % n
            for d in step_sizes:
                for sign in (1.0, -1.0):
                    delta = sign * d
                    v1 = h[i] + delta
                    v2 = h[j] - delta
                    if v1 < -1e-12 or v1 > 1 + 1e-12 or v2 < -1e-12 or v2 > 1 + 1e-12:
                        continue
                    h[i] = max(0.0, min(1.0, v1))
                    h[j] = max(0.0, min(1.0, v2))
                    c = compute_c5(h)
                    if c < cur - 1e-14:
                        cur = c
                        best_c5_ref = min(best_c5_ref, cur)
                        best_h_ref = h.copy()
                    else:
                        h[i] -= delta
                        h[j] += delta
                        break
    return h, cur

def slsqp_pass(h, n, beta, maxiter, t0, budget_s):
    global _Ng, best_h_ref, best_c5_ref
    _Ng = n
    if time.time() > t0 + budget_s - 20:
        return h
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_obj, h, args=(n, beta),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_sum(np.clip(res.x, 0, 1), n)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
        return h
    except Exception:
        return h

best_h_ref = None
best_c5_ref = 1e9

    t0 = time.time()

    for beta in betas:

        h = slsqp_pass(h, n, beta, 600, t0, budget_s)
        _Ng = n

    if time.time() < t0 + budget_s * 0.32:
        h, _ = coord_descent(h, n, rng, t0 + budget_s * 0.32)

    # Phase 2: Simulated annealing
    sa_end = t0 + budget_s * 0.6
    h, _ = sa_phase(h, n, rng, sa_end, t0, budget_s)

    # Phase 3: Coordinate descent (systematic)
    cd_end = t0 + budget_s * 0.82
    h, _ = coord_descent(h, n, rng, cd_end)

    # Phase 4: Final SLSQP polish at very low beta
    if time.time() < t0 + budget_s * 0.9:
        h = slsqp_pass(h, n, 0.00005, 400, t0, budget_s)
        _Ng = n

    # Phase 5: Final fine CD
    h, _ = coord_descent(h, n, rng, t0 + budget_s - 3)

    # Try a random-perturbation restart if time allows
    if time.time() < t0 + budget_s - 30:
        h2 = best_h_ref.copy()
        pert = rng.randn(n) * 0.005
        h2 = project_sum(h2 + pert, n)
        h2 = slsqp_pass(h2, n, 0.0005, 300, t0, budget_s)
        _Ng = n
        h2, _ = coord_descent(h2, n, rng, t0 + budget_s - 5)

    return best_h_ref, best_c5_ref, n
