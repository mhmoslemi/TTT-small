def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / temp
    shifted = np.clip(shifted, -100, 0)
    sm = cmax + temp * np.log(np.sum(np.exp(shifted)))
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0)**2

def grad_topk(h, topk_indices, dx):
    """Gradient of max of specific correlations (subgradient at argmax)."""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for idx in topk_indices:
        l = idx - (n - 1)
        # d corr_l / d h[i] = (1-h[i-l])*I(0<=i-l<n) - h[i+l]*I(0<=i+l<n)
        lo1 = max(0, l)
        hi1 = min(n, n + l)
        if lo1 < hi1:
            grad[lo1:hi1] += one_minus_h[lo1-l:hi1-l]
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2+l:hi2+l]
    return grad * dx / len(topk_indices)

    n = len(h)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h, args=(temp, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
    return project_feasible(np.clip(res.x, 0.0, 1.0))

def active_set_opt(h, t0, t_end, K=5, temp=0.001, lr0=0.04):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(5000):
        if time.time() - t0 > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        topk = np.argsort(corr)[-K:]
        tv = corr[topk]
        lw = np.exp(np.clip((tv - tv.max()) / temp, -100.0, 0.0))
        lw /= lw.sum()
        g = grad_topk_weighted(h, topk, lw, dx)
        lr = lr0 / (1.0 + 0.002 * it)
        h_new = project_feasible(h - lr * g)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new.copy()
        h = h_new
    return best_h

def coord_search(h, t0, t_end, dx_scale=0.05):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.default_rng(0)
    for it in range(3000):
        if time.time() - t0 > t_end:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            j = (j + 1) % n
        delta = rng.uniform(-dx_scale, dx_scale)
        h_new = h.copy()
        h_new[i] = np.clip(h[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h[j] - delta, 0.0, 1.0)
        if abs(np.sum(h_new) - np.sum(h)) < 1e-10:
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new.copy()
                h = h_new
    return best_h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float).copy()
    h = project_feasible(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: L-BFGS at n=360 with decreasing temperature
    temps_p1 = [0.04, 0.015, 0.005, 0.002]
    for temp in temps_p1:
        if time.time() - t0 > budget_s * 0.30:
            break
        h = lgbfs_step(h, temp, 200.0, maxiter=200)
        c5v = compute_c5(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Random restarts at n=360
    for sr in range(3):
        if time.time() - t0 > budget_s * 0.45:
            break
        sigma = 0.12 / (sr + 1)
        h_r = project_feasible(h + np.random.randn(len(h)) * sigma)
        h_r = lgbfs_step(h_r, 0.01, 200.0, maxiter=150)
        h_r = lgbfs_step(h_r, 0.003, 500.0, maxiter=100)
        c5v = compute_c5(h_r)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_r.copy()
            h = h_r.copy()

    # Phase 3: Upscale to 720 and optimize
    if time.time() - t0 < budget_s * 0.60:
        h_up = project_feasible(interp_upscale(h, 2))
        for temp in [0.015, 0.005, 0.002]:
            if time.time() - t0 > budget_s * 0.72:
                break
            h_up = lgbfs_step(h_up, temp, 200.0, maxiter=150)
            c5v = compute_c5(h_up)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_up.copy()
        h = h_up
    else:
        h = best_h.copy()

    # Phase 4: Active-set (top-K) optimization
    if time.time() - t0 < budget_s * 0.88:
        t_end = t0 + budget_s * 0.90
        h = active_set_opt(h, t0, t_end, K=5, temp=0.001, lr0=0.04)
        c5v = compute_c5(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Coordinate search polish
    if time.time() - t0 < budget_s * 0.95:
        t_end = t0 + budget_s * 0.96
        h = coord_search(best_h, t0, t_end, dx_scale=0.04)
        c5v = compute_c5(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Final L-BFGS polish
    if time.time() - t0 < budget_s * 0.98:
        h_f = lgbfs_step(best_h, 0.001, 1000.0, maxiter=80)
        c5v = compute_c5(h_f)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_f.copy()

    best_h = project_feasible(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
