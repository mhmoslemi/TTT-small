            j_ord = valid_j[np.argsort(-g[valid_j])]
            for ii in range(min(5, len(i_ord))):
                if moved:
                    break
                for jj in range(min(5, len(j_ord))):
                    if moved:
                        break
                    i = i_ord[ii]
                    j = j_ord[jj]
                    if i == j:
                        continue
                    for d in [0.02, 0.05, 0.1, 0.2]:
                        hi = min(1.0, h[i] + d)
                        hj = h[j] - (hi - h[i])
                        if hj < 0.0:
                            continue
                        h_tr = h.copy()
                        h_tr[i] = hi
                        h_tr[j] = hj
                        c = compute_c5(h_tr)
                        if c < cur - 1e-14:
                            h = h_tr
                            cur = c
                            moved = True
                            break
        if not moved:
            for _ in range(4):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = rng.uniform(0.005, 0.25)
                hi = np.clip(h[i] + d, 0.0, 1.0)
                hj = h[j] - (hi - h[i])
                if hj < 0.0 or hj > 1.0:
                    continue
                h_tr = h.copy()
                h_tr[i] = hi
                h_tr[j] = hj
                c = compute_c5(h_tr)
                if c < cur - 1e-14:
                    h = h_tr
                    cur = c
                    moved = True
                    break
        if cur < best - 1e-14:
            best = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000 and it % 1000 == 999:
            k = rng.randint(1, 8)
            idx = rng.choice(n, size=k, replace=False)
            shifts = rng.uniform(-0.15, 0.15, size=k)
            h_tr = h.copy()
            for m in range(k):
                h_tr[idx[m]] = np.clip(h[idx[m]] + shifts[m], 0.0, 1.0)
            h_tr = fix_sum(h_tr, n)
            c = compute_c5(h_tr)
            if c < cur + 0.003:
                h = h_tr
                cur = c
                stall = 0
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    t0 = time.time()
    rng = np.random.RandomState(seed)

    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP multi-beta at parent resolution
    if tl() > 120:
        bounds = list(zip(np.zeros(pn), np.ones(pn)))
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        for beta in [0.012, 0.004, 0.0015]:
            if tl() < 30:
                break
            _N = pn
            try:
                res = minimize(soft_max_obj, best_h, args=(pn, beta),
                               method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-12})
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), pn)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 2: Gradient local search
    if tl() > 60:
        h_gs, c5_gs = gradient_local_search(best_h, best_n, 80000, rng, tl() - 25)
        if c5_gs < best_c5:
            best_c5 = c5_gs
            best_h = h_gs.copy()

    # Phase 3: SLSQP refinement
    if tl() > 50:
        bounds = list(zip(np.zeros(best_n), np.ones(best_n)))
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        _N = best_n
        for beta in [0.002, 0.0007]:
            if tl() < 20:
                break
            try:
                res = minimize(soft_max_obj, best_h, args=(best_n, beta),
                               method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-13})
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), best_n)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 4: Try upsampling to 2x and optimize
    if tl() > 100:
        n2 = pn * 2
        if n2 <= 800:
            h_up = upsample(best_h, n2)
            c5_up = compute_c5(h_up)
            _N = n2
            bounds2 = list(zip(np.zeros(n2), np.ones(n2)))
            cons2 = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
            for beta in [0.008, 0.002]:
                if tl() < 30:
                    break
                try:
                    res = minimize(soft_max_obj, h_up, args=(n2, beta),
                                   method='SLSQP', bounds=bounds2,
                                   constraints=cons2,
                                   options={'maxiter': 300, 'ftol': 1e-12})
                    h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n2)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        best_n = n2
                except Exception:
                    pass
            if tl() > 40 and best_n == n2:
                h_gs, c5_gs = gradient_local_search(best_h, best_n, 40000, rng, tl() - 15)
                if c5_gs < best_c5:
                    best_c5 = c5_gs
                    best_h = h_gs.copy()

    # Phase 5: Final gradient local search at best resolution
    if tl() > 20:
        h_gs, c5_gs = gradient_local_search(best_h, best_n, 100000, rng, tl() - 5)
        if c5_gs < best_c5:
            best_c5 = c5_gs
            best_h = h_gs.copy()

    # Final fix
    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
