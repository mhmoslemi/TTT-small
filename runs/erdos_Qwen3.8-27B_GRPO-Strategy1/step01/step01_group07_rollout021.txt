def soft_max_grad(x, n, beta, penalty):
    dx = 2.0 / n
    nm1 = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / beta
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    one_x = 1.0 - x
    grad = np.zeros(n)
    for j in range(n):
        # d(corr[k])/dx[j] = (1-x[j-shift]) * I[valid] - x[j+shift] * I[valid]
        # where shift = k - (n-1), k = 0..2n-2
        # First part: sum over k where 0 <= j - shift < n, i.e., 0 <= j - (k-nm1) < n
        #   = sum over k where nm1-j < k-nm1 <= nm1-j... 
        #   shift = k - nm1, so j - shift = j - k + nm1
        #   valid when 0 <= j - k + nm1 <= n-1, i.e., 1-nm1+j <= k <= n-1+j
        #   k ranges from max(0, 1-nm1+j) to min(2n-2, n-1+j)
        # Second part: j + shift = j + k - nm1
        #   valid when 0 <= j + k - nm1 <= n-1, i.e., nm1-j <= k <= nm1-j+n-1 = 2nm1-j
        #   k ranges from max(0, nm1-j) to min(2n-2, 2nm1-j)
        
        # Vectorized:
        shifts = np.arange(2*n-1) - nm1  # shift values: -(n-1), ..., n-1
        # d(corr[k])/dx[j] = one_x[j-shifts] - x[j+shifts] (with masking)
        idx1 = j - shifts  # for one_x part
        idx2 = j + shifts  # for x part
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        term1 = weights * np.where(mask1, one_x[np.clip(idx1, 0, n-1)], 0)
        term2 = weights * np.where(mask2, x[np.clip(idx2, 0, n-1)], 0)
        grad[j] = dx * (np.sum(term1) - np.sum(term2))
    grad += 2.0 * penalty * (np.sum(x) - n / 2.0)
    return grad

        h = np.clip(h + diff * w, 0.0, 1.0)
    return h

def flat_spectrum_init(n, rng):
    best = None
    best_c = 1e9
    for _ in range(30):
        g = rng.randn(n)
        G = np.fft.rfft(g)
        phases = np.angle(G)
        G_flat = np.exp(1j * phases)
        G_flat[0] = 0.0
        h = np.fft.irfft(G_flat, n=n).real
        std = h.std()
        if std < 1e-10:
            continue
        h = (h - h.mean()) / (2.0 * std) + 0.5
        h = np.clip(h, 0, 1)
        h = fix_sum(h, n)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best = h.copy()
    return best if best is not None else np.full(n, 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    global _OPT_N, _OPT_BETA, _OPT_MU
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    n_list = []
    for m in range(1, 5):
        nc = pn * m
        if nc <= 600:
            n_list.append(nc)
    for extra in [50, 70, 100, 120, 140, 175, 200, 250, 280, 320, 350, 400, 450, 500, 560]:
        if extra not in n_list and extra <= 600:
            n_list.append(extra)
    n_list = sorted(set(n_list))

    # Multi-resolution L-BFGS-B optimization
    for n in n_list:
        if remaining() < 80:
            break
        h0 = upsample(best_h, n) if n != best_n else best_h.copy()
        inits = [h0]
        if n <= 400 and remaining() > 120:
            inits.append(flat_spectrum_init(n, rng))

        for h_init in inits:
            if remaining() < 25:
                break
            _OPT_N = n
            x = fix_sum(h_init.copy(), n)
            bounds = [(0.0, 1.0)] * n
            mu_schedule = [50.0, 200.0, 1000.0]

            for mu in mu_schedule:
                _OPT_MU = mu
                betas = [0.04, 0.015, 0.006, 0.0025, 0.001, 0.0004, 0.00015]
                for beta in betas:
                    if remaining() < 12:
                        break
                    _OPT_BETA = beta
                    maxiter = 300 if beta >= 0.003 else 500
                    try:
                        res = minimize(_obj, x, jac=_obj_grad, method='L-BFGS-B',
                                       bounds=bounds,
                                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9})
                        x = np.clip(res.x, 0, 1)
                        x = fix_sum(x, n)
                    except Exception:
                        pass

            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()
                best_n = n

    # Directed coordinate descent on best solution
    if remaining() > 30:
        h = best_h.copy()
        n = best_n
        cur = compute_c5(h)
        dx = 2.0 / n
        no_improve = 0
        max_no = 300

        while remaining() > 5 and no_improve < max_no:
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = int(np.argmax(corr))
            shift = k_star - (n - 1)

            js = np.arange(n)
            js1 = js - shift
            js2 = js + shift
            m1 = (js1 >= 0) & (js1 < n)
            m2 = (js2 >= 0) & (js2 < n)
            g1 = np.where(m1, 1.0 - h[np.clip(js1, 0, n - 1)], 0.0)
            g2 = np.where(m2, h[np.clip(js2, 0, n - 1)], 0.0)
            dcorr = g1 - g2

            dec = np.where(dcorr > 0.05)[0]
            inc = np.where(dcorr < -0.05)[0]

            if len(dec) > 0 and len(inc) > 0:
                i = dec[rng.randint(len(dec))]
                j = inc[rng.randint(len(inc))]
            else:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue

            best_step = None
            best_c = cur
            for step in [0.03, 0.015, 0.007, 0.003]:
                h_try = h.copy()
                h_try[i] = np.clip(h[i] - step, 0, 1)
                h_try[j] = np.clip(h[j] + step, 0, 1)
                if abs(h_try.sum() - n / 2.0) > 1e-6:
                    h_try = fix_sum(h_try, n)
                c = compute_c5(h_try)
                if c < best_c - 1e-14:
                    best_c = c
                    best_step = h_try.copy()

            if best_step is not None:
                h = best_step
                cur = best_c
                no_improve = 0
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
            else:
                no_improve += 1
                if no_improve % 40 == 0:
                    a = rng.randint(n)
                    b = rng.randint(n)
                    if a != b:
                        h[a] = np.clip(h[a] + rng.uniform(-0.04, 0.04), 0, 1)
                        h[b] = np.clip(h[b] - rng.uniform(-0.04, 0.04), 0, 1)
                        h = fix_sum(h, n)

        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
            best_n = n

    return best_h, best_c5, best_n
