def slsqp_polish(h0, dx, target_sum, maxiter=400):
    n = len(h0)
    t0 = np.max(np.correlate(h0, 1 - h0, mode='full'))
    
    def obj_func(x):
        return x[n]
    
    corr0 = np.correlate(h0, 1 - h0, mode='full')
    top_idx = np.argsort(corr0)[-15:]
    
    cons = []
    for k in top_idx:
        def make_con(kk=k):
            def con_fun(x):
                c = np.correlate(x[:len(x)-1], 1.0 - x[:len(x)-1], mode='full')
                return x[len(x)-1] - c[kk]
            return con_fun
        cons.append({'type': 'ineq', 'fun': make_con()})
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    cons.append({'type': 'eq', 'fun': eq_fun})
    
    def ineq_lo(x):
        return x[:n]
    def ineq_hi(x):
        return -x[:n] + 1.0
    
    bnds = [(0.0, 1.0)] * n + [(0.0, (2 * n))]
    x0 = np.concatenate([h0, [t0 + 0.005]])
    try:
        res = minimize(obj_func, x0, method='SLSQP', bounds=bnds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and res.success or (res.x is not None and res.fun < t0):
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def softmax_gradient_step(h, dx, target_sum, beta=40.0, lr=0.02, deadline=None):
    n = len(h)
    k_vals = np.arange(2 * n - 1) - (n - 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(2000):
        if deadline is not None and time.time() > deadline + 999999:
            break
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = corr.max()
        w = np.exp(beta * (corr - m))
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(n):
            g_j = gradient_fk(h, j, k_vals, n)
            grad[j] = np.dot(w, g_j)
        new_beta = 20.0 + 40.0 * (it / 500.0)
        new_lr = lr / (1.0 + 0.002 * it)
        h_new = project_sr(h - new_lr * grad, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t_start = time.time()
    
    def time_left(margin=5):
        return (t_start + budget_s - margin) - time.time()
    
    try:
        h_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h_raw = np.zeros(47)
        h_raw[:23] = 1.0
        h_raw[23] = 0.5
    
    n = len(h_raw)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_sr(h_raw, target_sum)
    best_h = h0.copy()
    best_n = n
    best_c5 = compute_c5(best_h, dx)
    
    # Phase 1: Paired coordinate descent (main workhorse)
    if time_left(50) > 0:
        deadline = t_start + budget_s - 50
        h1, corr1 = paired_cd(h0, target_sum, 400, deadline, rng)
        c1 = corr1 * dx
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    
    # Phase 2: SLSQP polish
    if time_left(30) > 0:
        h2 = slsqp_polish(best_h, 2.0 / best_n, best_n / 2.0, 300)
        c2 = compute_c5(h2, 2.0 / best_n)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
    
    # Phase 3: More cd from SLSQP result
    if time_left(30) > 0:
        deadline = t_start + budget_s - 30
        h3, corr3 = paired_cd(best_h, best_n / 2.0, 300, deadline, rng)
        c3 = corr3 * (2.0 / best_n)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
    
    # Phase 4: SLSQP again
    if time_left(20) > 0:
        h4 = slsqp_polish(best_h, 2.0 / best_n, best_n / 2.0, 200)
        c4 = compute_c5(h4, 2.0 / best_n)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4
    
    # Phase 5: Softmax gradient from perturbed starts
    for restart in range(5):
        if time_left(40) <= 0:
            break
        pert = best_h + 0.05 * rng.standard_normal(best_n)
        pert = project_sr(pert, best_n / 2.0)
        deadline = t_start + budget_s - 40
        h5, c5 = softmax_gradient_step(pert, 2.0 / best_n, best_n / 2.0,
                                         beta=30.0, lr=0.03, deadline=deadline)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5
        if time_left(30) <= 0:
            break
        deadline2 = t_start + budget_s - 30
        h5b, corr5b = paired_cd(h5, best_n / 2.0, 150, deadline2, rng)
        c5b = corr5b * (2.0 / best_n)
        if c5b < best_c5:
            best_h, best_c5 = h5b, c5b
    
    # Phase 6: Try different n with binary starts
    for trial_n in [49, 51, 47, 53]:
        if time_left(30) <= 0:
            break
        tn = trial_n
        tdx = 2.0 / tn
        tsum = tn / 2.0
        h_t = np.zeros(tn)
        pos = rng.permutation(tn)
        n_ones = int(tsum)
        h_t[pos[:n_ones]] = 1.0
        frac = tsum - n_ones
        if frac > 0.01:
            h_t[pos[n_ones]] = frac
        h_t = project_sr(h_t, tsum)
        if time_left(25) > 0:
            deadline = t_start + budget_s - 25
            h_t2, corr_t = paired_cd(h_t, tsum, 150, deadline, rng)
            c_t = corr_t * tdx
            if c_t < best_c5:
                best_h, best_c5, best_n = h_t2, c_t, tn
        if time_left(15) > 0:
            h_t3 = slsqp_polish(best_h, tdx, tsum, 100)
            c_t3 = compute_c5(h_t3, tdx)
            if c_t3 < best_c5:
                best_h, best_c5 = h_t3, c_t3
    
    # Final projection and evaluation
    final_n = len(best_h)
    final_dx = 2.0 / final_n
    best_h = project_sr(best_h, final_n / 2.0)
    best_c5 = compute_c5(best_h, final_dx)
    
    return best_h, best_c5, final_n
