import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = 0.5 * n
    h = h.copy()
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += diff / n_int
    h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    diff2 = target - s2
    if abs(diff2) > 1e-12:
        idx = np.argmax(np.minimum(h, 1.0 - h))
        if idx >= 0:
            h[idx] += diff2
    h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        val = c_max * dx
    else:
        shifted = np.clip((corr - c_max) / T, -100, 100)
        weights = np.exp(shifted)
        s = np.sum(weights)
        weights = weights / s
        val = (c_max + T * np.log(s)) * dx
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx] if T >= 1e-14 else (1.0 if corr[idx] == c_max else 0.0)

            m = n - k
            if m > 0:
                grad[0:m] += w * (-h[k:k+m])
                grad[k:k+m] += w * (1.0 - h[0:m])
        else:

            m = n - kk
            if m > 0:
                grad[0:m] += w * (-h[kk:kk+m])
                grad[kk:kk+m] += w * (1.0 - h[0:m])
    
    grad *= dx
    
    # Penalty gradient
    s_h = np.sum(h)
    diff = s_h - 0.5 * n
    val += penalty * diff * diff
    grad += 2.0 * penalty * diff * np.ones(n)
    
    return val, grad

import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = 0.5 * n
    h = h.copy()
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += diff / n_int
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff2 / n_int
    h = np.clip(h, 0.0, 1.0)
    return h

def _lse_value(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)

        return c_max * dx
    shifted = np.clip((corr - c_max) / T, -100, 100)
    s = np.sum(np.exp(shifted))
    return (c_max + T * np.log(s)) * dx

def _lse_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:

        mask = corr >= c_max - 1e-14
        weights[mask] = 1.0

        weights = np.exp(shifted)
