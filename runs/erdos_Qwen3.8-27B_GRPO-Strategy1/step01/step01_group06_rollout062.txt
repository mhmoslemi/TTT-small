from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')

        else:
            t = -s
            sg[:n-t] += (1.0 - h[t:])
            sg[t:] += (-h[:n-t])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def penalized_obj(x):
    base = compute_c5(x)
    sp = abs(x.sum() - len(x) / 2.0) * 500.0
    bp = (np.sum(np.maximum(0.0, x - 1.0)) + np.sum(np.maximum(0.0, -x))) * 500.0
    return base + sp + bp

def project_exact(h, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sg_single(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = int(np.argmax(corr))
    s = k - (n - 1)
    sg = np.zeros(n)
    if s == 0:
        sg = 1.0 - 2.0 * h
    elif s > 0:
        sg[s:] += (1.0 - h[:n - s])
        sg[:n - s] += (-h[s:])
    else:
        t = -s
        sg[:n - t] += (1.0 - h[t:])
        sg[t:] += (-h[:n - t])
    return dx * sg

def sg_multi(h, n, Tw):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(Tw * (corr - m))
    w /= w.sum()
    top = np.argsort(w)[-8:][::-1]
    sg = np.zeros(n)
    for idx in top:
        wi = w[idx]
        if wi < 1e-7:
            continue
        s = idx - (n - 1)
        if s == 0:
            sg += wi * (1.0 - 2.0 * h)
        elif s > 0:
            sg[s:] += wi * (1.0 - h[:n - s])
            sg[:n - s] += wi * (-h[s:])
        else:
            t = -s
            sg[:n - t] += wi * (1.0 - h[t:])
            sg[t:] += wi * (-h[:n - t])
    return dx * sg

def do_slsqp(h, n, T, maxiter):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = project_exact(np.clip(res.x, 0, 1), n)
        return hc, compute_c5(hc)
    except Exception:
        return h, compute_c5(h)

def coord_descent(h, n, deadline, best_c5):
    best_c5_cur = best_c5
    h_cur = h.copy()
    for sweep in range(500):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            candidates = [0.0, 1.0]
            v = h_cur[j]
            for dv in [0.1, 0.05, 0.02, -0.1, -0.05, -0.02]:
                nv = v + dv
                if 0.0 <= nv <= 1.0:
                    candidates.append(nv)
            for val in candidates:
                ht = h_cur.copy()
                ht[j] = val
                ht = project_exact(ht, n)
                c5v = compute_c5(ht)
                if c5v < best_c5_cur - 1e-12:
                    best_c5_cur = c5v
                    h_cur = ht
                    improved = True
                    break
        if not improved:
            break
    return h_cur, best_c5_cur
