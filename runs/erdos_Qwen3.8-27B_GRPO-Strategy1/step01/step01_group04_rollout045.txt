import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, target_sum):
    n = len(h)
    def f(c):
        return np.sum(np.clip(h + c, 0.0, 1.0)) - target_sum
    lo = -float(np.max(h))
    hi = float(np.max(1.0 - h))
    if f(lo) < 0 or f(hi) > 0:
        return np.clip(h, 0.0, 1.0)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_single_shift(h, L):
    """Gradient of c_L = sum_i h[i]*(1-h[i+L])*dx with respect to h, without dx factor"""
    n = len(h)
    g = 1.0 - h
    grad = np.zeros(n)
    if L >= 0:
        grad[:n-L] += g[:n-L]
        grad[L:] -= h[L:]
    else:
        Ll = -L
        grad[Ll:] += g[Ll:]
        grad[:n-Ll] -= h[:n-Ll]
    return grad

def topk_subgradient(h_init, t0, budget_s, k=15, max_iter=3000):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.003
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - t0 > budget_s:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_vals = corr * dx
        
        # Top-k shifts
        if k < len(c_vals):
            topk_idx = np.argpartition(c_vals, -k)[-k:]
        else:
            topk_idx = np.arange(len(c_vals))
        
        grad = np.zeros(n)
        for idx in topk_idx:
            L = idx - (n - 1)
            grad += grad_single_shift(h, L)
        grad /= (len(topk_idx))  # Average gradient
        
        # Try step
        h_new = project_to_feasible(h - alpha * grad, n / 2.0)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                alpha *= 0.8
                no_improve = 0
        
        # Adaptive: increase alpha occasionally
        if it > 0 and it % 500 == 0 and alpha < 0.01:
            alpha *= 1.3
    
    return best_h, best_c5

def smooth_c5(h, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    integral = np.sum(h) * dx
    return sm * dx + penalty * (integral - 1.0) ** 2

def lbfgs_polish(h_in, t0, budget_s):
    n = len(h_in)
    bounds = [(0.0, 1.0)] * n
    cur = h_in.copy()
    best_local = compute_c5(cur)
    
    schedules = [(0.01, 500.0, 300), (0.003, 2000.0, 200), (0.001, 10000.0, 150)]
    for temp, pen, maxit in schedules:
        if time.time() - t0 > budget_s:

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-13, 'gtol': 1e-10})
        nh = np.clip(res.x, 0.0, 1.0)
        nh = project_to_feasible(nh, n / 2.0)
        c5v = compute_c5(nh)
        if c5v < best_local:
            best_local = c5v
        cur = nh
    
    return cur, best_local

def fourier_perturb(h, n_modes=5):
    n = len(h)
    x = np.linspace(0, 2.0, n, endpoint=False)
    h2 = h.copy()

        h2 += coeff * np.sin(2 * np.pi * m * x / 2.0)
        h2 += coeff * np.cos(2 * np.pi * m * x / 2.0)
    return project_to_feasible(h2, n / 2.0)

def interp_upscale(h, factor):

    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)
