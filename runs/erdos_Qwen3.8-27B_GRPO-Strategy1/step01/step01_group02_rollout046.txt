def coord_descent(h_init, N, target, dx, t0, budget_s, rng, n_passes=30):
    """Coordinate descent with individual h[i] updates."""
    h = h_init.copy()
    best_c5 = compute_c5(h, dx)[0]
    best_h = h.copy()
    
    for p in range(n_passes):
        if time.time() - t0 > budget_s - 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(N)
        
        for i in order:
            # Try several values for h[i]
            candidates = [h[i]]
            if h[i] > 0.05:
                candidates += [h[i] * 0.5, 0.0]
            if h[i] < 0.95:
                candidates += [h[i] + (1 - h[i]) * 0.5, 1.0]
            # Also try small perturbations
            for d in [-0.05, -0.1, 0.05, 0.1]:
                v = h[i] + d
                if 0 <= v <= 1:
                    candidates.append(v)
            
            best_val = h[i]
            best_c5_i = best_c5  # current value
            
            for cv in set(candidates):
                old = h[i]
                h[i] = cv
                h = project_sum(h, target)  # re-project
                c5_new, _ = compute_c5(h, dx)
                if c5_new < best_c5_i - 1e-12:
                    best_c5_i = c5_new
                    best_val = cv
                    improved = True
                h[i] = old
            
            h[i] = best_val
            if best_c5_i < best_c5:
                best_c5 = best_c5_i
                best_h = h.copy()
                h = project_sum(h, target)
        
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if m >= 0:
        if m < N:
            g[m:] += (1.0 - h[:N - m])
            g[:N - m] -= h[m:]
    else:
        l = -m
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def optimize_adam(h_init, N, target, dx, t0, budget_s, rng, lr_init=0.035, tau_init=5.0, top_k=5, patience=600):
    h = h_init.copy()
    best_local = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr_init
    tau = tau_init
    beta1, beta2, eps = 0.88, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    no_imp = 0
    n_corr = 2 * N - 1

    for it in range(12000):
        if time.time() - t0 > budget_s - 12:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_local - 1e-14:
            best_local = c5_val
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > patience:
            h = best_h + rng.randn(N) * 0.025
            h = project_sum(h, target)
            m_buf *= 0.3
            v_buf *= 0.3
            no_imp = 0
            lr = max(lr, lr_init * 0.4)

        if top_k and top_k < n_corr:
            idx = np.argsort(corr)[-top_k:]
            sc = tau * corr[idx] - tau * corr[idx].max()
            w = np.exp(sc)
            w /= w.sum()
        else:
            idx = np.arange(n_corr)
            sc = tau * corr
            sc -= sc.max()
            w = np.exp(sc)
            w /= w.sum()

        thresh = 0.008 / max(top_k if top_k else N, 1)
        active_mask = w > thresh
        if not active_mask.any():
            active_mask = np.zeros(len(idx), dtype=bool)
            active_mask[np.argmax(idx)] = True
        active = idx[active_mask]
        w_norm = w[active_mask].sum()

        grad = np.zeros(N)
        for i in active:
            lag = int(i) - (N - 1)
            grad += (w[active_mask[list(active).index(i)]] / w_norm) * grad_lag(h, lag, N)

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** (it + 1))
        v_hat = v_buf / (1.0 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)

        if (it + 1) % 400 == 0:
            lr *= 0.82
            tau *= 1.7
        if (it + 1) % 800 == 0:
            no_imp = min(no_imp, patience // 2)
        if lr < 1e-9:
            break

    return best_h, best_local

def coord_descent(h_init, N, target, dx, t0, budget_s, rng, n_passes=25):
    h = h_init.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    for p in range(n_passes):
        if time.time() - t0 > budget_s - 10:
            break
        improved = False
        order = rng.permutation(N)
        for i in order:
            cur = h[i]
            cands = [cur]
            if cur > 0.03:
                cands += [cur * 0.5, 0.0]
            if cur < 0.97:
                cands += [cur + (1.0 - cur) * 0.5, 1.0]
            cands += [max(0, cur - 0.08), min(1, cur + 0.08), max(0, cur - 0.03), min(1, cur + 0.03)]
            best_v = cur
            best_c5_i = compute_c5(h, dx)
            for cv in cands:
                old = h[i]
                h[i] = cv
                h2 = project_sum(h, target)
                c5_t = compute_c5(h2, dx)
                if c5_t < best_c5_i - 1e-13:
                    best_c5_i = c5_t
                    best_v = cv
                    h = h2
                    improved = True
                else:
                    h[i] = old

            if best_v != cur:
                h[i] = best_v
                h = project_sum(h, target)
                if compute_c5(h, dx) < best_c5:
                    best_c5 = compute_c5(h, dx)
                    best_h = h.copy()

        if time.time() - t0 > budget_s - 10:
            break

    return best_h, best_c5

def fourier_perturb(h, N, rng, n_modes=3, amp=0.06):
    hf = np.fft.rfft(h)
    freqs = np.arange(len(hf))
    if N > 10:
        in_range = (freqs >= 2) & (freqs <= N // 3)
    else:
        in_range = freqs >= 2
    idxs = np.where(in_range)[0]
    if len(idxs) == 0:
        return h
    picks = rng.choice(idxs, size=min(n_modes, len(idxs)), replace=False)
    hf2 = hf.copy()
    for fi in picks:
        hf2[fi] += amp * (rng.randn() + 1j * rng.randn())
        if fi < N - fi:
            hf2[N - fi] = np.conj(hf2[fi])
    h2 = np.fft.irfft(hf2, n=N)
    return project_sum(h2, N / 2.0)
