```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_obj_and_grad(h, T):
    n = len(h)
    corr = corr_full(h)
    c_max = np.max(corr)
    grad = np.zeros(n)
    
    if T < 1e-14:
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        _add_grad(grad, h, n, lag)
        return c_max, grad
    
    vals = corr / T
    vals -= np.max(vals)
    exp_vals = np.exp(vals)
    w = exp_vals / np.sum(exp_vals)
    obj = c_max + T * np.log(np.sum(exp_vals))
    
    thresh = 1e-12
    for k in range(len(corr)):
        if w[k] < thresh:
            continue
        lag = k - (n - 1)
        _add_grad_weighted(grad, h, n, lag, w[k] / T)
    
    return obj, grad

def _add_grad(grad, h, n, lag):
    if lag >= 0:
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] -= h[i]
    else:
        p = -lag
        for i in range(p, n):
            grad[i] += (1.0 - h[i - p])
            grad[i - p] -= h[i]

def _add_grad_weighted(grad, h, n, lag, wt):
    if lag >= 0:
        for i in range(n - lag):
            grad[i] += wt * (1.0 - h[i + lag])
            grad[i + lag] -= wt * h[i]
    else:
        p = -lag
        for i in range(p, n):
            grad[i] += wt * (1.0 - h[i - p])
            grad[i - p] -= wt * h[i]

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h.copy()
    h = h.copy()
    h += (target - s) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def sum_constraint(h, n):
    return np.sum(h) - n * 0.5

def slsqp_optimize(h_start, n, T, max_iter=500):
    def fun(x):
        obj, _ = lse_obj_and_grad(x, T)
        return obj
    
    def jac(x):
        _, g = lse_obj_and_grad(x, T)
        return g
    
    res = opt.minimize(
        fun, h_start, jac=jac,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    if res.fun < lse_obj_and_grad(h_start, T)[0]:
        return res.x
    return h_start

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = fix_sum(h_new, new_n)
    return h_new

def downsample(h, new_n):
    old_n = len(h)
    if new_n >= old_n:
        return upsample(h, new_n)
    # Average blocks
    ratio = old_n / new_n
    h_new = np.zeros(new_n)
    for i in range(new_n):
        start = int(i * ratio)
        end = int((i + 1) * ratio)
        if end > old_n:
            end = old_n
        if start < end:
            h_new[i] = np.mean(h[start:end])
    h_new = fix_sum(h_new, new_n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    t_start = time.time()

        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_best:

            c5_best = c5_cand
    
    # Phase 2: Perturbations at current resolution
    n_pert = 8
    for i in range(n_pert):
        if time_left() < 200:
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _add_grad_vec(grad, h, n, lag, wt):
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += wt * (1.0 - h[lag:lag + m])
            grad[lag:] -= wt * h[:m]
    else:
        p = -lag
        m = n - p
        if m > 0:
            grad[p:] += wt * (1.0 - h[:m])
            grad[:m] -= wt * h[p:]

def lse_obj_and_grad(h, T):
    n = len(h)
    corr = corr_full(h)
    c_max = np.max(corr)
    grad = np.zeros(n)
    if T < 1e-14:
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)
        _add_grad_vec(grad, h, n, lag, 1.0)
        return c_max, grad
    vals = (corr - c_max) / T
    exp_vals = np.exp(vals)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = c_max + T * np.log(s)
    thresh = 1e-10
    for k in range(len(corr)):
        if w[k] < thresh:
            continue
        lag = k - (n - 1)
        _add_grad_vec(grad, h, n, lag, w[k] / T)
    return obj, grad

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h.copy()
    h = h.copy()
    h += (target - s) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

class _OptCtx:
    def __init__(self, T):
        self.T = T
    def fun(self, x):
        o, _ = lse_obj_and_grad(x, self.T)
        return o
    def jac(self, x):
        _, g = lse_obj_and_grad(x, self.T)
        return g

def slsqp_optimize(h_start, n, T, max_iter=500):
    ctx = _OptCtx(T)
    res = opt.minimize(
        ctx.fun, h_start, jac=ctx.jac,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    obj_before, _ = lse_obj_and_grad(h_start, T)
    obj_after, _ = lse_obj_and_grad(res.x, T)
    if obj_after < obj_before:
        return res.x
    return h_start

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    if new_n >= old_n:
        return upsample(h, new_n)
    h_new = np.zeros(new_n)
    ratio = old_n / new_n

        if e > old_n:
            e = old_n
        if s < e:
            h_new[i] = np.mean(h[s:e])
    return fix_sum(h_new, new_n)
```
