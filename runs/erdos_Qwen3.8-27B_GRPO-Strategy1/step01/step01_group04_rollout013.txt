    sm = (cmax + temp * np.log(np.sum(np.exp(shifted)))) * dx
    integral = np.sum(h) * dx - 1.0
    return sm + penalty * integral * integral

def project_onto_box_and_sum(h, target_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h_clipped - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h_clipped - (lo + hi) / 2.0, 0.0, 1.0)

def run_lbfgs(h, t0, budget_s, max_frac):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cur = h.copy()
    best = cur.copy()
    best_c5 = compute_c5(best)
    
    schedules = [(0.05, 500, 200), (0.02, 2000, 200), (0.005, 10000, 200)]
    for temp, pen, maxit in schedules:
        if time.time() - t0 > budget_s * max_frac:
            break
        res = minimize(objective, cur, args=(temp, pen),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10})
        cur = np.clip(res.x, 0.0, 1.0)
        cur = project_onto_box_and_sum(cur, n / 2.0)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_c5 = c5
            best = cur.copy()
        cur = best.copy()  # Always continue from best
    
    return best, best_c5

def random_perturb_structural(h, rng):
    """Apply a structural perturbation: shift a segment."""
    n = len(h)
    h2 = h.copy()
    # Choose a random segment
    i0 = rng.randint(0, n // 2)
    length = rng.randint(n // 10, n // 3)
    shift = rng.randint(-n // 5, n // 5 + 1)
    if shift != 0:
        seg = h2[i0:i0+length].copy()
        j0 = i0 + shift
        if j0 >= 0 and j0 + length <= n:
            h2[i0:i0+length] = h2[j0:j0+length]
            h2[j0:j0+length] = seg
        # Redistribute to maintain integral
    # Add small Gaussian noise
    h2 += rng.randn(n) * 0.02
    return project_onto_box_and_sum(h2, n / 2.0)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Optimize at 360
    cur, c5_360 = run_lbfgs(h_init, t0, budget_s, 0.15)
    if c5_360 < best_c5:
        best_c5, best_h = c5_360, cur.copy()
    
    # Phase 2: Upscale to 720 and optimize
    if time.time() - t0 < budget_s * 0.5:
        cur = interp_upscale(cur, 2)
        cur = project_onto_box_and_sum(cur, len(cur) // 2)
        cur, c5_720 = run_lbfgs(cur, t0, budget_s, 0.45)
        if c5_720 < best_c5:
            best_c5, best_h = c5_720, cur.copy()
    
    # Phase 3: Random restarts at 720
    n_restarts = 6
    for i in range(n_restarts):
        if time.time() - t0 > budget_s * 0.85:
            break
        h_pert = random_perturb_structural(best_h, rng)
        h_pert, c5_r = run_lbfgs(h_pert, t0, budget_s, 0.88)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_pert.copy()
    
    # Phase 4: Final polish at best resolution
    if time.time() - t0 < budget_s * 0.95:
        best_h, c5_final = run_lbfgs(best_h, t0, budget_s, 0.98)
        if c5_final < best_c5:
            best_c5 = c5_final
    
    best_h = project_onto_box_and_sum(best_h, len(best_h) // 2)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)

Wait, let me look this up properly. NumPy's correlate:
`np.correlate(a, v, mode)` computes:
`(a * v)[k] = sum_n a[n+k] * conj(v[n])` (for valid n)

For mode='full', the result has length n+m-1 where n=len(a), m=len(v).
For same length n: result has length 2n-1.
result[k] = sum_n a[n + k - (m-1)] * conj(v[n]) for k=0,...,2n-2

With m=n: result[k] = sum_n a[n + k - (n-1)] * v[n]

So result[n-1] = sum_n a[n] * v[n] (lag 0, the inner product)
result[n-1+1] = sum_n a[n+1] * v[n] (a shifted right by 1)
result[n-1-1] = sum_n a[n-1] * v[n] (a shifted left by 1)

In our case, a = h, v = 1-h:

result[n-1+lag] = sum h[n+lag] * (1-h[n]) for valid n

This corresponds to ∫ h(x+k) * (1-h(x)) dx ≈ sum h[i+lag]*(1-h[i])*dx

Which is ∫ h(x+k)(1-h(x)) dx = ∫ h(u)(1-h(u-k)) du (substituting u=x+k)

So the max over all lags gives us max_k ∫ h(x)(1-h(x-k)) dx, which by renaming k→-k is the same as max_k ∫ h(x)(1-h(x+k)) dx.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_func(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / temp
    sm = (cmax + temp * np.log(np.sum(np.exp(shifted)))) * dx
    integral = np.sum(h) * dx - 1.0
    return sm + penalty * integral * integral

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def run_lbfgs(h, t0, deadline_frac, budget_s):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cur = h.copy()
    best = cur.copy()
    best_c5 = compute_c5(best)
    target = n / 2.0
    schedules = [(0.06, 500.0, 300), (0.02, 2000.0, 300), (0.005, 10000.0, 250)]
    for temp, pen, maxit in schedules:
        if time.time() > t0 + budget_s * deadline_frac:
            break
        res = minimize(obj_func, cur, args=(temp, pen),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10})
        cur = np.clip(res.x, 0.0, 1.0)
        cur = project_h(cur, target)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_c5 = c5
            best = cur.copy()
        else:
            cur = best.copy()
    return best, best_c5

def structural_perturb(h, rng):
    n = len(h)
    h2 = h.copy()
    mode = rng.randint(3)
    if mode == 0:
        i0 = rng.randint(0, max(1, n - n // 4))
        length = rng.randint(n // 15, n // 4)
        shift = rng.randint(-n // 6, n // 6 + 1)
        j0 = i0 + shift
        if 0 <= j0 and j0 + length <= n and shift != 0:
            seg = h2[i0:i0 + length].copy()
            h2[i0:i0 + length] = h2[j0:j0 + length]
            h2[j0:j0 + length] = seg
    elif mode == 1:
        i0 = rng.randint(0, max(1, n - n // 5))
        length = rng.randint(n // 10, n // 3)
        scale = 0.5 + rng.random() * 1.0
        h2[i0:i0 + length] *= scale
    else:
        i0 = rng.randint(0, max(1, n - n // 5))
        length = rng.randint(n // 10, n // 3)
        h2[i0:i0 + length] += rng.randn(length) * 0.15
    h2 += rng.randn(n) * 0.01
    return project_h(h2, n / 2.0)

def cosine_init(n, freq, amp, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    h += amp * 0.5 * np.cos(4.0 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    return project_h(np.clip(h, 0, 1), n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 2: upscale to 720, optimize
    if time.time() < t0 + budget_s * 0.55:
        cur = interp_upscale(best_h, 2)
        cur = project_h(cur, len(cur) / 2.0)
        cur, c5 = run_lbfgs(cur, t0, 0.45, budget_s)

    # Phase 3: multiple restarts at 720
    n = len(best_h)
    if time.time() < t0 + budget_s * 0.88:
        for i in range(8):
            if time.time() > t0 + budget_s * 0.88:
                break
            h_pert = structural_perturb(best_h, rng)
            h_pert, c5_r = run_lbfgs(h_pert, t0, 0.90, budget_s)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_pert.copy()
                n = len(best_h)

    # Phase 4: try different initializations
    if time.time() < t0 + budget_s * 0.92:
        for freq in [1, 2, 3]:

                break
            h_cos = cosine_init(n, float(freq), 0.25, rng)
            h_cos, c5_c = run_lbfgs(h_cos, t0, 0.93, budget_s)
            if c5_c < best_c5:
                best_c5, best_h = c5_c, h_cos.copy()

    if time.time() < t0 + budget_s * 0.98:
        best_h, c5_f = run_lbfgs(best_h, t0, 0.99, budget_s)
        if c5_f < best_c5:
            best_c5 = c5_f

    best_h = project_h(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
