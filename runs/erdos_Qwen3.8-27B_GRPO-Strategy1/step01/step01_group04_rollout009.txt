def binarize(h):
    """Project h to the nearest binary vector with sum = n/2."""
    n = len(h)
    target = n // 2
    # Keep the top n/2 values as 1
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0

def smooth_objective(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n

    shifted = corr - cmax
    smax = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integral = np.sum(h) * dx
    return smax * dx + penalty * (integral - 1.0)**2

    s = np.sum(nh)
    target = n / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, penalty, pnorm):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if pnorm > 0:
        cmax = np.max(corr)
        shifted = corr - cmax
        smax = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    else:
        p = -pnorm
        smax = (np.sum(np.abs(corr) ** p)) ** (1.0 / p)
    integral = np.sum(h) * dx
    return smax * dx + penalty * (integral - 1.0) ** 2

def pnorm_obj(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    smax = (np.sum(np.abs(corr) ** p)) ** (1.0 / p)
    integral = np.sum(h) * dx
    return smax * dx + penalty * (integral - 1.0) ** 2

def repair_h(h):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = len(h) / 2.0
    if s > 1e-15:
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_local_search(h_bin, t0, max_time, seed):
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    max_stall = 25000
    iters = 0
    
    while stall < max_stall and (time.time() - t0) < max_time:
        iters += 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def lbfgs_pass(h, temp, penalty, maxiter, pnorm=None):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    if pnorm is not None and pnorm > 0:
        res = minimize(pnorm_obj, h, args=(pnorm, penalty),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-9})
    else:
        res = minimize(smooth_obj, h, args=(temp, penalty, 1),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-9})
    return np.clip(res.x, 0.0, 1.0)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def optimize_stage(h, t0, time_limit):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cur = repair_h(h.copy())
    best = compute_c5(cur)
    best_h = cur.copy()
    
    schedules = [
        (0.15, 500.0, 80, None),
        (0.04, 1500.0, 70, None),
        (0.01, 5000.0, 60, None),
        (0.003, 20000.0, 50, None),
        (0, 0, 60, 8),
        (0, 0, 50, 16),
        (0.001, 50000.0, 40, None),
    ]
    for temp, pen, maxit, pn in schedules:
        if time.time() - t0 > time_limit:
            break
        if pn is not None:
            cur = lbfgs_pass(cur, 0, pen, maxit, pnorm=pn)
        else:
            cur = lbfgs_pass(cur, temp, pen, maxit)
        cur = repair_h(cur)
        c5v = compute_c5(cur)
        if c5v < best:
            best = c5v
            best_h = cur.copy()
    return best_h, best
