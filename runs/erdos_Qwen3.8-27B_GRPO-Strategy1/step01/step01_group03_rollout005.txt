def compute_c5(h, dx):

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()

        else:
            m = h > 1e-15
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
    return h

def slsqp_minmax(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.001]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except:
        pass
    return h0.copy()

def simulated_anneal(h0, dx, target_sum, n_steps=50000, T_start=0.05, T_end=0.001):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h, dx)
    h_best = h.copy()
    c5_best = c5_cur
    
    for step in range(n_steps):
        T = T_start * (T_end / T_start) ** (step / n_steps)
        # Move: pick 2 indices, transfer mass
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        delta = np.random.uniform(-0.5, 0.5)
        hi, hj = h[i], h[j]
        hi_new = hi + delta
        hj_new = hj - delta
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue
        h[i] = hi_new
        h[j] = hj_new
        c5_new = compute_c5(h, dx)
        if c5_new < c5_cur or np.random.rand() < np.exp(-(c5_new - c5_cur) / T):
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return h_best, c5_best

    # Final polish
    if time_left(20):
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        h_f = slsqp_minmax(best_h, dx, target_sum, maxiter=500)
        c_f = compute_c5(h_f, dx)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
        if time_left(10):
            h_f2 = smooth_optimize(best_h, dx, target_sum, beta=60, maxiter=400)
            c_f2 = compute_c5(h_f2, dx)
            if c_f2 < best_c5:
                best_c5, best_h = c_f2, h_f2

    best_h = project_sr(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, best_n
