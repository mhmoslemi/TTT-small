    return float(np.max(corr) * dx)

def obj_lse(h, temp, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    integ = np.sum(h) * dx
    return sm * dx + pen * (integ - 1.0) ** 2

def obj_mixed(h, temp, pen, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    lse = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    p90 = np.percentile(corr, 90)
    sm = alpha * lse + (1.0 - alpha) * p90
    integ = np.sum(h) * dx

    xn = (np.arange(m) + 0.5) * (2.0 / m)
