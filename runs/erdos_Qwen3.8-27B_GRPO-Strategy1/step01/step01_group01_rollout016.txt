from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, t, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # smooth max = (1/t) * log(sum(exp(t*corr)))
    mx = np.max(corr)
    exps = np.exp(t * (corr - mx))
    logsum = mx + np.log(np.sum(exps)) / t
    # softmax weights
    w = exps / np.sum(exps)
    # gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        # A part: (1-h[j-k]) for 0<=j-k<n
        js = max(0, k)
        je = min(n, k + n)
        if js < je:
            grad[js:je] += wk * (1.0 - h[js-k:je-k])
        # B part: -h[j+k] for 0<=j+k<n
        js = max(0, -k)
        je = min(n, n - k)
        if js < je:
            grad[js:je] -= wk * h[js+k:je+k]
    # penalty gradient
    grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    obj = logsum * dx + lam * (np.sum(h) - n / 2.0) ** 2
    return obj, grad

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0

            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
