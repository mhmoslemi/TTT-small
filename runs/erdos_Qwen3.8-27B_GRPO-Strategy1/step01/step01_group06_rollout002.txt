            if time.time() > t_phase4_end:
                improved = False
                break
            for j in range(i + 1, n_cur):
                if h_disc[i] == h_disc[j]:
                    continue
                h_disc[i], h_disc[j] = h_disc[j], h_disc[i]
                nc = binary_c5(h_disc)
                if nc < disc_cost:
                    disc_cost = nc
                    improved = True
                else:
                    h_disc[i], h_disc[j] = h_disc[j], h_disc[i]
                if time.time() > t_phase4_end:
                    break
        if disc_cost < best_c5:
            best_c5 = disc_cost
            best_h = h_disc.copy()

    # Phase 5: Final high-temperature polish
    if remaining() > 10:
        n_fin = len(best_h)
        for T_s in [3000, 8000]:
            if remaining() < 5:
                break
            try:
                hc, c5v = slsqp_optimize(best_h, n_fin, T_s, maxiter=700, ftol=1e-14)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
