def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        nf = int(free.sum())
        if nf == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def _sum_con(x, n):
    return x.sum() - n / 2.0


def _sum_con_jac(x, n):
    return np.ones(n)


def _corr_grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        if shift >= 0:
            cnt = n - shift
            if cnt > 0:
                idx = np.arange(cnt)
                G[k, idx + shift] += (1.0 - h[idx])
                G[k, idx] -= h[idx + shift]
        else:
            s = -shift
            cnt = n - s
            if cnt > 0:
                idx = np.arange(cnt)
                G[k, idx] += (1.0 - h[idx + s])
                G[k, idx + s] -= h[idx]
    return G


def _smooth_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * (2.0 / n)


def _smooth_grad(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    e = np.exp(beta * (corr - m))
    p = e / e.sum()
    G = _corr_grad_matrix(x, n)
    return (2.0 / n) * (G.T @ p)


def slsqp_solve(x0, n, beta, maxiter):
    bnds = [(0.0, 1.0)] * n
    res = minimize(
        fun=_smooth_obj, jac=_smooth_grad, x0=x0,
        method='SLSQP', args=(n, beta),
        bounds=bnds,
        constraints=[{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return res.x
