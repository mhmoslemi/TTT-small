import numpy as np
from scipy.optimize import minimize
import time

_cur_t = 1.0
_cur_n = 1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def xinterp(h, nd):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(nd) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def obj_smooth(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = _cur_t * (corr - mx)
    return (mx + np.log(np.sum(np.exp(s))) / _cur_t) * (2.0 / _cur_n)

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def do_optimize(h0, n, t, maxiter, ftol):
    global _cur_t, _cur_n
    _cur_t = float(t)
    _cur_n = n
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    r = minimize(obj_smooth, h0, method='SLSQP', bounds=bnds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    hc = proj(np.clip(r.x, 0.0, 1.0), n)
    return hc, compute_c5(hc)
