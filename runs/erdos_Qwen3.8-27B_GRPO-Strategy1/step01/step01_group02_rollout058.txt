def weighted_gradient(h, w, N):
    """Compute gradient of sum_l w[l]*C_l w.r.t. h."""
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        wi = w[idx]
        if wi < 1e-7:
            continue

            grad[:n] += wi * (1.0 - h[lag:lag+n])
            grad[lag:lag+n] -= wi * h[:n]

            lr = lr0 * (0.87 ** phase)
            tau = tau0 * (1.6 ** phase)
        if lr < 1e-9:
            break
    
    return best_h, best_c5

def targeted_local_search(h, N, target, dx, t0, budget, rng, n_iters=5000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iters):
        if time.time() - t0 > budget - 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        thresh = max_val - 1.0
        worst = np.where(corr >= thresh)[0]
        if len(worst) == 0:
            break
        idx = worst[rng.randint(len(worst))]
        lag = idx - (N - 1)
        
        h_new = h.copy()
        if lag >= 0:
            n = N - lag
            contrib = h[:n] * (1.0 - h[lag:lag+n])
            pos = np.where(contrib > 0.01)[0]
            if len(pos) == 0:
                continue
            j = pos[rng.randint(len(pos))]
            if rng.rand() < 0.5:
                h_new[j] = max(0.0, h_new[j] - 0.2)
            else:
                h_new[lag + j] = min(1.0, h_new[lag + j] + 0.2)
        else:
            l = -lag
            n = N - l
            contrib = h[l:l+n] * (1.0 - h[:n])
            pos = np.where(contrib > 0.01)[0]
            if len(pos) == 0:
                continue
            j = pos[rng.randint(len(pos))]
            if rng.rand() < 0.5:
                h_new[l + j] = max(0.0, h_new[l + j] - 0.2)
            else:
                h_new[j] = min(1.0, h_new[j] + 0.2)
        
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
    return best_h, best_c5

def general_local_search(h, N, target, dx, t0, budget, rng, n_iters=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iters):
        if time.time() - t0 > budget - 10:
            break
        j = rng.randint(N)
        h_new = h.copy()
        delta = (rng.rand() - 0.5) * 0.4
        h_new[j] = np.clip(h_new[j] + delta, 0.0, 1.0)
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    dx_par = 2.0 / n_par
    target_par = n_par / 2.0
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, dx_par)
    
    # Phase 1: Intensive optimization at parent size (multiple restarts)
    inits = [h_par.copy()]
    inits.append(project_sum(h_par + rng.randn(n_par) * 0.06, target_par))
    tm = np.zeros(n_par)
    for i in range(n_par):
        tm[i] = bin(i).count('1') % 2
    inits.append(project_sum(tm + rng.randn(n_par) * 0.08, target_par))
    
    for ii, h_init in enumerate(inits):
        if time.time() - t0 > budget_s - 200:
            break
        h_opt, c5_opt = optimize_smoothmax(h_init.copy(), n_par, target_par, dx_par,
                                           t0, budget_s, rng, lr0=0.03, tau0=3.5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 2: Targeted local search at parent size
    if time.time() - t0 < budget_s - 120:
        h_ls, c5_ls = targeted_local_search(best_h.copy(), n_par, target_par, dx_par,
                                            t0, budget_s, rng, n_iters=8000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
        # Optimize again from local search result
        if time.time() - t0 < budget_s - 100:
            h_opt2, c5_opt2 = optimize_smoothmax(best_h.copy(), n_par, target_par, dx_par,
                                                 t0, budget_s, rng, lr0=0.01, tau0=15.0, max_iter=4000)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2
        if time.time() - t0 < budget_s - 80:
            h_ls2, c5_ls2 = general_local_search(best_h.copy(), n_par, target_par, dx_par,
                                                  t0, budget_s, rng, n_iters=3000)
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2
    
    # Phase 3: Try upsampling to 2x and optimize
    N2 = 2 * n_par
    if time.time() - t0 < budget_s - 150:
        dx2 = 2.0 / N2
        target2 = N2 / 2.0
        x1 = np.linspace(0.0, 2.0, n_par, endpoint=False)
        x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h2_init = np.interp(x2, x1, best_h)
        h2_init = project_sum(h2_init, target2)
        
        h2_opt, c5_2 = optimize_smoothmax(h2_init, N2, target2, dx2,
                                          t0, budget_s, rng, lr0=0.02, tau0=5.0)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt
        
        if time.time() - t0 < budget_s - 60:
            h2_ls, c5_2ls = targeted_local_search(best_h.copy(), N2, target2, dx2,
                                                  t0, budget_s, rng, n_iters=4000)
            if c5_2ls < best_c5:
                best_c5 = c5_2ls
                best_h = h2_ls
            if time.time() - t0 < budget_s - 40:
                h2_opt2, c5_2o2 = optimize_smoothmax(best_h.copy(), N2, target2, dx2,
                                                     t0, budget_s, rng, lr0=0.008, tau0=20.0, max_iter=3000)
                if c5_2o2 < best_c5:
                    best_c5 = c5_2o2
                    best_h = h2_opt2
                if time.time() - t0 < budget_s - 25:
                    h2_ls2, c5_2ls2 = general_local_search(best_h.copy(), N2, target2, dx2,
                                                            t0, budget_s, rng, n_iters=2000)
                    if c5_2ls2 < best_c5:
                        best_c5 = c5_2ls2
                        best_h = h2_ls2
    
    # Phase 4: Final polish at best resolution
    Nf = len(best_h)
    dxf = 2.0 / Nf
    tf = Nf / 2.0
    if time.time() - t0 < budget_s - 20:
        h_f, c5_f = general_local_search(best_h.copy(), Nf, tf, dxf,
                                          t0, budget_s, rng, n_iters=1500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
