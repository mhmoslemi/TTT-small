                    lr *= 0.5
                    if lr < 1e-6:
                        break
                else:
                    lr = min(lr * 1.1, 0.1)
                h_p = h_new
                
            c5 = compute_c5(h_p)
            if c5 < best_c:
                best_c = c5
                best_h = h_p.copy()
    
    # Final C5
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
