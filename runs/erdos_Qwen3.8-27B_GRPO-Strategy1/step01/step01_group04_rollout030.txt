import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_obj_grad(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exps = np.exp(shifted / temp)
    w = exps / np.sum(exps)
    S = cmax + temp * np.log(np.sum(exps))
    conv1 = np.convolve(g, w, 'full')[n - 1: 2 * n - 1]
    conv2 = np.correlate(h, w, 'full')[n - 1: 2 * n - 1]
    grad_S = conv1 - conv2
    integral_dev = np.sum(h) * dx - 1.0
    f = dx * S + penalty * integral_dev * integral_dev
    grad = dx * grad_S + 2.0 * penalty * dx * integral_dev
    return f, grad

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return h.copy()
    h2 = h / integ
    return np.clip(h2, 0.0, 1.0)

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def lbfgs_pass(h, temp, penalty, maxiter, t0, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_c5_obj_grad, h, args=(temp, penalty),
            method='L-BFGS-B', bounds=bounds, jac=True,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12,
                     'maxls': 30, 'iprint': -1}
        )
    except Exception:
        return h, compute_c5(h)
    nh = np.clip(res.x, 0.0, 1.0)
    nh = repair_h(nh)
    return nh, compute_c5(nh)

def deadline_check(t0, budget_s, frac):
    return time.time() - t0 >= budget_s * frac

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float).copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    n = len(h_init)

    # Phase 1: Multi-temperature L-BFGS-B at original resolution
    cur_h = h_init.copy()
    schedules_n0 = [
        (0.1, 50.0, 500),
        (0.03, 200.0, 500),
        (0.01, 1000.0, 600),
        (0.003, 5000.0, 500),
    ]
    for temp, pen, mi in schedules_n0:

            break
        cur_h, c5v = lbfgs_pass(cur_h, temp, pen, mi, t0, budget_s)
        if c5v < best_c5:

            best_h = cur_h.copy()

    # Phase 2: Random restarts at n
    for ri in range(6):
        if deadline_check(t0, budget_s, 0.50):
            break
        sigma = 0.10 / (ri + 1)
        h_pert = np.clip(best_h + np.random.randn(n) * sigma, 0.0, 1.0)
        h_pert = repair_h(h_pert)
        h_pert, c5v = lbfgs_pass(h_pert, 0.01, 500.0, 300, t0, budget_s)
