            break
        beta = 10.0 + 60.0 * min(1.0, p / max(1, n_passes * 0.4))
        lam_step = 0.03 * (1.0 + p / max(1, n_passes * 0.5))
        order = np.random.permutation(n)
        for i in order:
            b = np.zeros(2 * n - 1)
            lo1 = max(0, -i)
            hi1 = min(n - 1, n - 1 - i)
            if lo1 <= hi1:
                kr = np.arange(lo1, hi1 + 1)
                b[kr + n - 1] += (1.0 - h[i + kr])
            lo2 = max(-(n - 1), i - n + 1)
            hi2 = min(n - 1, i)
            if lo2 <= hi2:
                kr = np.arange(lo2, hi2 + 1)
                b[kr + n - 1] -= h[i - kr]
            a = corr - h[i] * b
            t_new = find_optimal_t(a, b, lam, beta)
            delta = t_new - h[i]
            if abs(delta) > 1e-14:
                h[i] = t_new
                corr += delta * b
        lam += lam_step * (h.sum() - target_sum)
        lam = max(-20.0, min(20.0, lam))
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def binary_local_search(h_init, dx, target_sum, n_iter=200000, t_deadline=None):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_c5 = best_c5
    for it in range(n_iter):
        if t_deadline is not None and time.time() > t_deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h, dx)
        if c5 < cur_c5:
            cur_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def slsqp_polish(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            cand = project_feasible(res.x[:n], target_sum)
            if compute_c5(cand, dx) <= compute_c5(h0, dx) + 1e-6:
                return cand
    except Exception:
        pass
    return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def rem():
        return budget_s - (time.time() - t_start)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = np.zeros(47)
        h_parent[:24] = 1.0
        h_parent[23] = 0.5

    n_p = len(h_parent)
    dx_p = 2.0 / n_p
    tgt_p = n_p / 2.0
    h_parent = project_feasible(h_parent, tgt_p)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h, dx_p)
    best_n = n_p
    best_dx = dx_p

    # Phase 1: Coordinate descent on parent
    if rem() > 30:
        h1, c1 = coord_descent(best_h, best_dx, tgt_p, 3000, t_start + budget_s * 0.30)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: SLSQP polish
    if rem() > 20:
        h2 = slsqp_polish(best_h, best_dx, tgt_p, 300)
        c2 = compute_c5(h2, best_dx)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: More coord descent after SLSQP
    if rem() > 30:
        h3, c3 = coord_descent(best_h, best_dx, tgt_p, 2000, t_start + budget_s * 0.45)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Try n=46 binary
    if rem() > 50:
        n46 = 46
        dx46 = 2.0 / n46
        tgt46 = n46 / 2.0
        grid_orig = np.linspace(0, 2, n_p, endpoint=False) + dx_p / 2
        grid46 = np.linspace(0, 2, n46, endpoint=False) + dx46 / 2
        h46 = np.interp(grid46, grid_orig, h_parent)
        h46 = project_feasible(h46, tgt46)
        dl = t_start + budget_s * 0.58
        h46b, c46b = binary_local_search(h46, dx46, tgt46, 400000, dl)
        if c46b < best_c5:
            best_h, best_c5, best_n, best_dx = h46b, c46b, n46, dx46
        if rem() > 25:
            h46c, c46c = coord_descent(h46b, dx46, tgt46, 1500, t_start + budget_s * 0.65)
            if c46c < best_c5:
                best_h, best_c5 = h46c, c46c

    # Phase 5: Try n=48 binary
    if rem() > 50:
        n48 = 48
        dx48 = 2.0 / n48
        tgt48 = n48 / 2.0
        grid48 = np.linspace(0, 2, n48, endpoint=False) + dx48 / 2
        h48 = np.interp(grid48, grid_orig, h_parent)
        h48 = project_feasible(h48, tgt48)
        dl = t_start + budget_s * 0.75
        h48b, c48b = binary_local_search(h48, dx48, tgt48, 400000, dl)
        if c48b < best_c5:
            best_h, best_c5, best_n, best_dx = h48b, c48b, n48, dx48
        if rem() > 25:
            h48c, c48c = coord_descent(h48b, dx48, tgt48, 1500, t_start + budget_s * 0.82)
            if c48c < best_c5:
                best_h, best_c5 = h48c, c48c

    # Phase 6: Multi-start coord descent on best_n
    if rem() > 40:
        tgt_best = best_n / 2.0
        for r in range(5):
            if rem() < 25:
                break
            h_r = best_h + 0.1 * np.random.randn(best_n)
            h_r = project_feasible(h_r, tgt_best)
            dl = t_start + budget_s - 50
            h_r2, c_r2 = coord_descent(h_r, best_dx, tgt_best, 2000, dl)
            if c_r2 < best_c5:
                best_h, best_c5 = h_r2, c_r2

    # Phase 7: Final SLSQP + coord descent
    if rem() > 15:
        tgt_best = best_n / 2.0
        hf = slsqp_polish(best_h, best_dx, tgt_best, 200)
        cf = compute_c5(hf, best_dx)
        if cf < best_c5:
            best_h, best_c5 = hf, cf
    if rem() > 10:
        tgt_best = best_n / 2.0
        hz, cz = coord_descent(best_h, best_dx, tgt_best, 2000, t_start + budget_s - 5)
        if cz < best_c5:
            best_h, best_c5 = hz, cz

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_dx)
    return best_h, best_c5, best_n
