from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def corr_gradient(h, lag_idx):
    """Gradient of correlate(h, 1-h, 'full')[lag_idx] w.r.t. h"""
    n = len(h)
    offset = lag_idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        j2 = j + offset
        if 0 <= j2 < n:
            grad[j] += (1.0 - h[j2])
        j3 = j - offset
        if 0 <= j3 < n:
            grad[j] -= h[j3]
    return grad

    n = len(h0)
    
    def full_obj(x):
        return x[n]

        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode='full')

        return np.sum(x[:n]) - target_sum

    """Return (smooth_max_value, gradient) of smooth max of correlate(h,1-h,'full')"""
    n = len(h)

One more idea: **alternating minimization**. Fix the active set A (the lags that achieve the max), and solve:
min t s.t. corr[k] = t for k in A, sum(h) = n/2, 0<=h<=1

This is a system of |A| + 1 equations in n + 1 unknowns (h_1,...,h_n, t). If |A| < n, it's underdetermined, and we want the solution that minimizes t. We can use SLSQP or NM on this reduced problem.

After solving, check if the new corr values are consistent (i.e., no other lag exceeds t). If not, add that lag to A and re-solve.

def active_set_slsqp(h0, dx, target_sum, max_iter=50):
    n = len(h0)
    h = h0.copy()
    t = np.max(np.correlate(h, 1.0 - h, mode='full'))
    
    for outer in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        active = np.where(corr > t - 1e-6)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Solve: min t s.t. corr[k] = t for k in active, sum(h) = n/2, 0<=h<=1
        def obj(x):
            return x[n]
        
        def ineq(x):
            corr_x = np.correlate(x[:n], 1.0 - x[:n], mode='full')
            return x[n] - corr_x[active]
        
        def eq(x):
            return x[:n].sum() - target_sum
        
        x0 = np.concatenate([h, [t + 0.01]])
        bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
        cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]
        
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-10})
        
        h_new = project_simplex(res.x[:n], target_sum)
        t_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))
        
        if t_new >= t - 1e-8:
            break
        h = h_new
        t = t_new
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_simplex(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def smooth_max_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sm_val = m + np.log(np.sum(np.exp(corr - m)) / len(corr)) / beta
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for lag in range(len(corr)):
        wl = w[lag]
        if wl < 1e-12:
            continue
        offset = lag - (n - 1)
        for j in range(n):
            j2 = j + offset
            if 0 <= j2 < n:
                grad[j] += wl * (1.0 - h[j2])
            j3 = j - offset
            if 0 <= j3 < n:
                grad[j] -= wl * h[j3]
    return sm_val, grad

def lbfgs_smooth(h0, dx, target_sum, beta=100, maxiter=3000):
    n = len(h0)
    mu = 500.0
    def fun(x):
        sm, g = smooth_max_grad(x, beta)
        s = x.sum() - target_sum
        return sm * dx + mu * s * s, g * dx + 2.0 * mu * s
    try:
        res = minimize(fun, h0, jac=True, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_simplex(res.x, target_sum)
    except Exception:
        return h0.copy()

def slsqp_active(h0, target_sum, maxiter=200, tol_gap=1e-5):
    n = len(h0)
    h = h0.copy()
    t_cur = np.max(np.correlate(h, 1.0 - h, mode='full'))
    for _ in range(12):
        corr = np.correlate(h, 1.0 - h, mode='full')
        active = np.where(corr > t_cur - tol_gap)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        if len(active) > 15:
            active = np.argsort(corr)[-15:]
        def obj(x):
            return x[n]
        def ineq(x):
            cx = np.correlate(x[:n], 1.0 - x[:n], mode='full')
            return x[n] - cx[active]
        def eq(x):
            return x[:n].sum() - target_sum
        x0 = np.concatenate([h, [t_cur + 0.01]])
        bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
        cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_simplex(res.x[:n], target_sum)
        except Exception:
            break
        t_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))
        if t_new >= t_cur - 1e-9:
            h = h_new
            break
        h = h_new
        t_cur = t_new
    return h

def sa_search(h0, dx, target_sum, T_init, T_final, n_iter, deadline):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h, dx)
    c5 = best_c5
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T_init * (T_final / T_init) ** (it / float(max(n_iter - 1, 1)))
        n_mod = max(1, int(n * (0.15 - 0.1 * it / n_iter)))
        idx = np.random.choice(n, n_mod, replace=False)
        h_new = h.copy()
        h_new[idx] += np.random.uniform(-0.35, 0.35, n_mod)
        h_new = project_simplex(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        d = c5_new - c5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-10)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5
