    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    if abs(s - target) < 1e-10:

    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)

    if n_int > 0:
        h[interior] += (target - s) / n_int
        h = np.clip(h, 0.0, 1.0)
    return h

Let shift = k - (n-1), ranging from -(n-1) to (n-1).
Let w_shift[s] = w[s + n - 1] for s in [-(n-1), (n-1)], and 0 otherwise.

Then:
∑_k w[k] * h[j+k-(n-1)] = ∑_s w_shift[s] * h[j+s]  (valid correlation)
∑_k w[k] * h[j-k+(n-1)] = ∑_s w_shift[s] * h[j-s]  (valid correlation with flipped shifts)

The first is a "valid" cross-correlation of h with w_shift.
The second is a "valid" cross-correlation of h with w_shift reversed.

We can compute these using FFT:
- Pad h and w_shift to length ≥ 2n
- Compute: ifft(fft(h) * conj(fft(w_shift))) gives the full correlation
- Extract the valid part

def subgradient_step(h, n, step_size, n_active=10):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get n_active largest corr values
    top_idx = np.argpartition(corr, -n_active)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        shift = k - (n - 1)
        for j in range(n):
            jp = j + shift
            jm = j - shift
            g = 0.0
            if 0 <= jp < n:
                g += 1.0 - h[jp]
            if 0 <= jm < n:
                g -= h[jm]
            grad[j] += g
    grad *= dx / n_active
    # Gradient step
    h_new = h - step_size * grad
    # Project to [0,1] with sum constraint
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new

- L-BFGS-B might converge to a point where the sum constraint isn't exactly satisfied. I fix this with fix_sum.

import numpy as np
import scipy.optimize as opt
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    h = h.copy()
    interior = (h > 1e-6) & (h < 1.0 - 1e-6)
    ni = int(np.sum(interior))
    if ni > 0:
        h[interior] += (target - s) / ni
    return np.clip(h, 0.0, 1.0)


def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, new_n)


def obj_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = (corr - c_max) / T
    exp_terms = np.exp(shifted)
    total = np.sum(exp_terms)
    val = (c_max + T * np.log(total)) * dx
    if penalty > 0:
        sd = np.sum(h) - n * 0.5
        val += penalty * sd * sd
    weights = exp_terms / total
    j_idx = np.arange(n)[:, None]
    shifts = np.arange(-(n - 1), n)
    jp = j_idx + shifts
    jm = j_idx - shifts
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    hp = np.where(vp, h[np.clip(jp, 0, n - 1).astype(int)], 0.0)
    hm = np.where(vm, h[np.clip(jm, 0, n - 1).astype(int)], 0.0)
    A = (hp * weights).sum(axis=1)
    B = (hm * weights).sum(axis=1)
    grad = dx * (np.sum(weights) - A - B)
    if penalty > 0:
        sd = np.sum(h) - n * 0.5
        grad += 2.0 * penalty * sd
    return val, grad


def obj_only(h, T, penalty):
    return obj_grad(h, T, penalty)[0]


def jac_only(h, T, penalty):
    return obj_grad(h, T, penalty)[1]


def subgradient_polish(h, n, n_steps=200, n_active=20):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step_size = 0.01
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_active = np.argpartition(corr, -n_active)[-n_active:]
        grad = np.zeros(n)
        for k in k_active:
            shift = k - (n - 1)
            for j in range(n):
                jp = j + shift
                jm = j - shift
                g = 0.0
                if 0 <= jp < n:
                    g += 1.0 - h[jp]
                if 0 <= jm < n:
                    g -= h[jm]
                grad[j] += g
        grad *= dx / n_active
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            step_size = min(step_size * 1.05, 0.05)
        else:
            step_size *= 0.9
            if step_size < 0.0001:
                break
    return best_h, best_c5
