        if time_exhausted(0.60):
            break
        scale = 0.25 * np.exp(-0.025 * i) + 0.02
        h0 = best_h + np.random.uniform(-scale, scale, size=n)
        h0 = project_feasible(h0, n)
        for T in [100, 500, 2000]:
            if time_exhausted(0.60):
                break
            h_new = slsqp_optimize(h0, T, maxiter=350)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
                n = len(best_h)
                h0 = h_new.copy()

    # Phase 3: Try different n values (60% - 80% budget)
    for n_try in [46, 50, 54, 56, 60, 64, 70, 80, 100]:
        if time_exhausted(0.80):
            break
        if n_try == n:
            h0 = best_h.copy()
        else:
            x_old = np.linspace(0.0, 2.0, n)
            x_new = np.linspace(0.0, 2.0, n_try)
            h0 = np.interp(x_new, x_old, best_h)
            h0 = project_feasible(h0, n_try)
        for T in [50, 300, 1500]:
            if time_exhausted(0.80):
                break
            h_new = slsqp_optimize(h0, T, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
                n = n_try
                h0 = h_new.copy()

    # Phase 4: Coordinate descent polish (80% - 93% budget)
    if not time_exhausted(0.93):
        h_cd, c5_cd = coord_descent(best_h, n_sweeps=300,
                                    time_limit_fn=lambda: time_exhausted(0.93))
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd
            n = len(best_h)

    # Phase 5: Final high-T SLSQP (93% - 99% budget)
    if not time_exhausted(0.99):
        for T in [5000, 15000, 50000]:
            if time_exhausted(0.99):
                break
            h_new = slsqp_optimize(best_h, T, maxiter=1200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
                n = len(best_h)

    # Phase 6: One more coordinate descent pass
    if not time_exhausted(0.995):
        h_cd, c5_cd = coord_descent(best_h, n_sweeps=100,
                                    time_limit_fn=lambda: time_exhausted(0.995))
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd

    # Final sanity check
    best_h = project_feasible(best_h, len(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
