import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    scaled = np.clip((corr - m) / beta, -100, 0)
    exps = np.exp(scaled)
    total = np.sum(exps)
    weights = exps / total
    f = dx * (m + beta * np.log(total))
    grad = np.zeros(n)
    one_minus_x = 1.0 - x
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        if s >= 0:
            grad[:n - s] += w * one_minus_x[s:]
            grad[s:] -= w * x[:n - s]
        else:
            t = -s
            grad[t:] += w * one_minus_x[:n - t]
            grad[:n - t] -= w * x[t:]
    grad *= dx
    return f, grad

def subgrad_corr_grad(h, n, k_star):
    s = k_star - (n - 1)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += one_minus_h[s:]
        grad[s:] -= h[:n - s]
    else:
        t = -s
        grad[t:] += one_minus_h[:n - t]
        grad[:n - t] -= h[t:]
    return grad

def fix_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n

    v = np.interp(new_x, old_x, parent)
    v = np.clip(v + (new_n / 2.0 - np.sum(v)) / new_n, 0.0, 1.0)
    v = np.clip(v + (new_n / 2.0 - np.sum(v)) / new_n, 0.0, 1.0)
    return v

        k_star = int(np.argmax(corr))

            c = compute_c5(h_new)
            if c <= best + 1e-14:

                best = c
                break
