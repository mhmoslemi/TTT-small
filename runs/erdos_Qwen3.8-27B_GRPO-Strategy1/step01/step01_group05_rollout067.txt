import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    if T < 1e-6:
        return m
    shifted = (corr - m) * T
    return m + np.log(np.sum(np.exp(shifted))) / T

def sum_constr_fun(x, n):
    return np.sum(x) - n / 2.0

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / np.sum(free), 0.0, 1.0)
    return h

def slsqp_run(x0, n, T, maxiter):
    bnds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': sum_constr_fun, 'args': (n,)}
    res = minimize(smooth_obj, x0, method='SLSQP', bounds=bnds,
                   constraints=[con], args=(n, T),
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def subgrad_refine(h, n, steps, alpha0):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    h = h.copy()
    target = n / 2.0
    for step in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        # k_star corresponds to shift: for full mode, index i corresponds to shift (i - (n-1))
        k_shift = k_star - (n - 1)
        # gradient of g_k wrt h[j]

        h = h - alpha * grad

        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.cos(2 * np.pi * mode * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    return repair_h(h, n)

def bangbang_init(n, block_size, offset):
    h = np.zeros(n)
    for i in range(n):
        if (i + offset) % (2 * block_size) < block_size:
            h[i] = 1.0
    return repair_h(h, n)

    x = np.linspace(0, 2, n, endpoint=False)
    h = np.maximum(0.0, 1.0 - np.abs(x - center) / width)
    h = np.clip(h, 0.0, 1.0)
    return repair_h(h, n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, T):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    if T < 1e-3:
        return m
    return m + np.log(np.sum(np.exp((c - m) * T))) / T

def eq_constr(x, n):
    return np.sum(x) - n * 0.5

def repair(h, n):
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) == 0:
            h = np.clip(h + d / n, 0, 1)
        else:
            h[free] = np.clip(h[free] + d / np.sum(free), 0, 1)
    return h

                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':eq_constr,'args':(n,)}],
                   args=(n,T), options={'maxiter':iters,'ftol':1e-15})
    return res.x

    mom = np.zeros(n)
    beta = 0.6
    for s in range(steps):
        tau = int(np.argmax(np.correlate(h, 1.0 - h, mode='full'))) - (n - 1)
        g = np.empty(n)
        for j in range(n):
            v = 1.0
            if 0 <= j + tau < n:
                v -= h[j + tau]
            if 0 <= j - tau < n:
                v -= h[j - tau]
            g[j] = v * dx
        a = a0 * (1.0 + a0) / (1.0 + 0.08 * s)
        mom = beta * mom + (1.0 - beta) * g
        hn = repair(np.clip(h - a * mom, 0.0, 1.0), n)
        cn = compute_c5(hn, n)
        if cn < bc:
            bc, best, h = cn, hn.copy(), hn
    return best, bc

def make_starts(n, rng):
    out = []
    x = np.linspace(0, 2, n, False)
    out.append(np.full(n, 0.5))
    for m in range(1, 5):
        for a in (0.3, 0.6, 0.85):
            out.append(np.clip(0.5 + a * np.cos(2 * np.pi * m * x / 2.0 + rng.uniform(0, 6.28)), 0, 1))
    for bs in (2, 3, 4, 5, 7):
        off = int(rng.integers(0, 2 * bs))
        h = ((np.arange(n) + off) % (2 * bs) < bs).astype(float)
        out.append(h)
    out.append(np.clip(0.4 + 0.5 * np.sin(np.pi * x), 0, 1))
    out.append(np.clip(0.5 + 0.3 * np.cos(np.pi * x / 2.0) + 0.2 * np.sin(3 * np.pi * x), 0, 1))
    out.append(np.clip(0.5 + 0.35 * np.sin(2 * np.pi * x) * np.cos(np.pi * x), 0, 1))
    out.append(np.clip(0.5 + 0.25 * np.sin(4 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x), 0, 1))
    return [repair(s, n) for s in out]

def do_optimize(h0, n, t0, tb, margin):
    best = h0.copy()
    bc = compute_c5(h0, n)
    for T in (200, 80, 30, 10, 3, 1):
        el = time.time() - t0
        if el > tb - margin:
            break
        rem = tb - margin - el
        mi = min(300, max(30, int(rem * 8)))
        if mi < 20:
            break
        hn = repair(do_slsqp(best, n, T, mi), n)
        cn = compute_c5(hn, n)
        if cn < bc:
            bc, best = cn, hn
    el = time.time() - t0
    if el < tb - margin:
        st = min(400, max(100, int((tb - margin - el) * 300)))
        best, bc = do_subgrad(best, n, st, 0.008)
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    gb, gc = None, 1.0

    try:
        hp = repair(np.array(initial_h_values, dtype=float), int(len(initial_h_values)))
        np_ = len(hp)
    except Exception:
        hp, np_ = None, 45

    if hp is not None:
        gb, gc = hp.copy(), compute_c5(hp, np_)
        gb, gc = do_optimize(hp, np_, t0, t0 + 200, 60)
        el = time.time() - t0
        if el < budget_s - 250:
            hp2 = repair(hp + 0.03 * rng.randn(hp.size), np_)
            h2, c2 = do_optimize(hp2, np_, t0, min(t0 + 280, budget_s - 250), 50)
            if c2 < gc:
                gc, gb = c2, h2
        el = time.time() - t0
        if el < budget_s - 200:
            hp3 = repair(hp + 0.015 * rng.randn(hp.size), np_)
            h3, c3 = do_optimize(hp3, np_, t0, min(t0 + 350, budget_s - 200), 50)
            if c3 < gc:
                gc, gb = c3, h3

    for n in (45, 54, 60, 72, 90):
        if hp is not None and n == np_:
            continue
        if time.time() > budget_s - 200:
            break
        ss = make_starts(n, rng)
        sc = sorted((compute_c5(s, n), i) for i, s in enumerate(ss))
        t_n = time.time()
        for rank, (_, idx) in enumerate(sc[:3]):
            if time.time() > budget_s - 80:
                break
            allot = min(t_n + 90 - rank * 20, budget_s - 60 - rank * 20)
            hn, cn = do_optimize(ss[idx], n, t0, allot, 30)
            if cn < gc:
                gc, gb = cn, hn

    if gb is not None and time.time() < budget_s - 15:
        ng = len(gb)
        rem = budget_s - 15 - time.time()
        gb, gc = do_subgrad(gb, ng, min(3000, max(200, int(rem * 600))), 0.003)
        if time.time() < budget_s - 5:
            mi = max(50, int((budget_s - 5 - time.time()) * 20))
            hn = repair(do_slsqp(gb, ng, 1, mi), ng)
            cn = compute_c5(hn, ng)
            if cn < gc:
                gc, gb = cn, hn

    if gb is None:
        gb = np.full(45, 0.5)
        gc = compute_c5(gb, 45)
    gb = repair(gb, len(gb))
    gc = compute_c5(gb, len(gb))
    return gb, gc, len(gb)
