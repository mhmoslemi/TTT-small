        penalty = max(0, corr[idx_lag] * dx - compute_c5(x) + 0.001) * 50
        return base + penalty
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': 400, 'ftol': 1e-13})
    return project_feasible(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP with increasing temperature
    temps1 = [30, 100, 300, 800]
    for T in temps1:
        if elapsed() > budget_s * 0.20:
            break
        try:
            hc = slsqp_step(best_h, n, T, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Coordinate descent
    if elapsed() < budget_s * 0.40:
        hd, cd = coordinate_descent(best_h, n, n_sweeps=25, n_trials=4)
        if cd < best_c5:
            best_c5 = cd
            best_h = hd

    # Phase 3: Lag-focused optimization on worst lags
    if elapsed() < budget_s * 0.55:
        for _ in range(3):
            if elapsed() > budget_s * 0.55:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            worst_idx = np.argmax(corr)
            try:
                hf = lag_focus_optimize(best_h, n, worst_idx)
                cf = compute_c5(hf)
                if cf < best_c5:
                    best_c5 = cf
                    best_h = hf
            except Exception:
                pass

    # Phase 4: SLSQP with top-k objective via log-sum-exp at high T
    temps2 = [1500, 4000, 10000]
    for T in temps2:
        if elapsed() > budget_s * 0.70:
            break
        try:
            hc = slsqp_step(best_h, n, T, maxiter=800, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 5: Another coordinate descent round
    if elapsed() < budget_s * 0.82:
        hd, cd = coordinate_descent(best_h, n, n_sweeps=20, n_trials=3)
        if cd < best_c5:
            best_c5 = cd
            best_h = hd

    # Phase 6: Try different n values by interpolation
    if elapsed() < budget_s * 0.90:
        for n_try in [46, 54, 70, 80]:
            if elapsed() > budget_s * 0.90:
                break
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            hi = np.interp(x_new, x_old, best_h)
            hi = project_feasible(hi, n_try)
            for T in [200, 1000, 3000]:
                if elapsed() > budget_s * 0.90:
                    break
                try:
                    hc = slsqp_step(hi, n_try, T, maxiter=400)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                        n = n_try
                except Exception:
                    pass

    # Phase 7: Final polish with very high T
    if elapsed() < budget_s * 0.97:
        n_cur = len(best_h)
        for T in [15000, 40000]:
            if elapsed() > budget_s * 0.97:
                break
            try:
                hc = slsqp_step(best_h, n_cur, T, maxiter=1000, ftol=1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Final coordinate nudge
    if elapsed() < budget_s * 0.99:
        hd, cd = coordinate_descent(best_h, len(best_h), n_sweeps=8, n_trials=3)
        if cd < best_c5:
            best_c5 = cd
            best_h = hd

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
