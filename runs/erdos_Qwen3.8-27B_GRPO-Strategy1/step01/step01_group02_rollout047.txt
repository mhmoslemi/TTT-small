        vb = b2 * vb + (1.0 - b2) * (grad * grad)
        mhat = mb / (1.0 - b1**(it+1))
        vhat = vb / (1.0 - b2**(it+1))
        
        h = h - current_lr * mhat / (np.sqrt(vhat) + eps)
        h = project_sum(h, target)
        
        if (it + 1) % tau_every == 0:
            current_lr *= 0.75
            tau *= tau_mult
        if current_lr < 1e-9:
            break
    
    return best_h, best_c5

def local_search(h, N, dx, target, t0, deadline, rng, n_rounds=50):
    """Simple coordinate descent: try small perturbations."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)[0]
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try random coordinate adjustments

            h_new[i] = np.clip(h_new[i] + delta, 0, 1)

                best_h = h.copy()

        if not improved:

    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, lag, N):
    g = np.zeros(N)
    if lag >= 0:
        if lag < N:
            g[lag:] += (1.0 - h[:N - lag])
            g[:N - lag] -= h[lag:]
    else:
        l = -lag
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def optimize(h, N, dx, target, deadline, rng,
             lr=0.02, tau0=6.0, tau_mult=1.5, tau_every=150, max_iter=4000):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)[0]
    b1, b2, eps = 0.8, 0.999, 1e-7
    mb = np.zeros(N)
    vb = np.zeros(N)
    tau = tau0
    cur_lr = lr

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        sc = tau * (corr - corr.max())
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-300:
            break
        w /= ws

        thresh = max(0.003, 3.0 / ws)
        grad = np.zeros(N)
        active = np.where(w > thresh)[0]
        for idx in active:
            lag = int(idx) - (N - 1)
            grad += w[idx] * grad_lag(h, lag, N)

        mb = b1 * mb + (1.0 - b1) * grad
        vb = b2 * vb + (1.0 - b2) * (grad * grad)
        mhat = mb / (1.0 - b1 ** (it + 1))
        vhat = vb / (1.0 - b2 ** (it + 1))

        h = h - cur_lr * mhat / (np.sqrt(vhat) + eps)
        h = project_sum(h, target)

        if (it + 1) % tau_every == 0:
            cur_lr *= 0.72
            tau *= tau_mult
        if cur_lr < 1e-10:
            break

    return best_h, best_c5
