2. Take a gradient step to reduce corr[k*]
3. But also monitor the second-worst lag and ensure it doesn't increase too much
4. If the second-worst becomes the new worst, switch to reducing that one

This "cycling" approach might converge better than the smooth-max when there are only 2-3 active lags.

Actually, the smooth-max already does this implicitly when tau is large (it focuses on the single worst lag). The issue is when there are multiple lags at nearly the same level - then we need to balance them.

Let me use a hybrid: 
- If the top lag is significantly above the second (gap > epsilon), focus on the top lag only
- If they're close, use equal weights on the top few

This is essentially what the adaptive tau does, but let me make it more explicit.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lag_grad(h, lag):
    N = len(h)
    g = np.zeros(N)
    if lag >= 0:
        if lag < N:
            g[lag:] += (1.0 - h[:N - lag])
            g[:N - lag] -= h[lag:]
    else:
        l = -lag
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def optimize_adam(h_init, N, target, dx, deadline, rng, lr_init=0.02, tau_init=5.0, max_iter=8000):
    h = h_init.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr_init
    tau = tau_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    Nf = 2 * N - 1
    thresh = 0.005 / Nf
    no_imp = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = get_corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1

        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()

        grad = np.zeros(N)
        active = np.where(w > thresh)[0]
        for idx in active:
            lag = int(idx) - (N - 1)
            grad += w[idx] * lag_grad(h, lag)

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** (it + 1))
        vh = v_buf / (1.0 - b2 ** (it + 1))
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)

        if (it + 1) % 250 == 0:
            lr *= 0.87
            na = len(active)
            if na < 4:
                tau = max(3.0, tau * 1.4)
            elif na > 25:
                tau = max(2.0, tau * 0.85)
            else:
                tau *= 1.12

        if no_imp > 500 and it % 300 == 150:
            strat = rng.randint(3)
            if strat == 0:
                ctr = rng.randint(N)
                wid = rng.randint(5, max(6, N // 4))
                for i in range(N):
                    d = abs(i - ctr)
                    if d > N // 2:
                        d = N - d
                    if d < wid:
                        h[i] += rng.randn() * 0.04 * (1.0 - d / wid)
            elif strat == 1:
                st = rng.randint(0, max(1, N - 20))
                ln = rng.randint(6, max(7, N // 5))
                h[st:st + ln] = 1.0 - h[st:st + ln]
            else:
                idxs = rng.choice(N, size=max(5, N // 8), replace=False)
                h[idxs] = 0.5 + rng.randn(len(idxs)) * 0.12
            h = project_sum(h, target)
            m_buf *= 0.3
            v_buf *= 0.3
            no_imp = 0

        if lr < 3e-9:
            break
    return best_h, best_c5

def polish_lags(h, N, target, dx, deadline, top_k=3):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for phase in range(5):
        corr = get_corr(h)
        top_idx = np.argsort(corr)[::-1][:top_k]
        avg_g = np.zeros(N)
        for idx in top_idx:
            lag = int(idx) - (N - 1)
            avg_g += lag_grad(h, lag)
        avg_g /= top_k
        ng = np.linalg.norm(avg_g)
        if ng < 1e-10:
            break
        for step in range(200):
            if time.time() > deadline:
                return best_h, best_c5
            improved = False
            for tlr in [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                ht = project_sum(h - tlr * avg_g, target)
                ct = compute_c5(ht, dx)
                if ct < best_c5 - 1e-11:
                    h = ht
                    best_c5 = ct
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                break
        corr = get_corr(h)
        top_idx = np.argsort(corr)[::-1][:top_k]
        avg_g = np.zeros(N)
        for idx in top_idx:
            lag = int(idx) - (N - 1)
            avg_g += lag_grad(h, lag)
        avg_g /= top_k
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)

        best_h = h_opt.copy()

        best_c5 = c5_f

    # Phase 3: Anti-periodic binary restart (250s)
    d3 = min(deadline, t0 + 680)
    if time.time() < d3:
        h_anti = np.zeros(n0)
        half = n0 // 2
        tgt_half = int(round(target0 * half / n0))
        ones_f = rng.choice(half, size=tgt_half, replace=False)
        h_anti[ones_f] = 1.0
        h_anti[half:] = 1.0 - h_anti[:half]
        if n0 % 2 == 1:
            h_anti[-1] = h_anti[-2]
        h_anti = project_sum(h_anti, target0)
        h_opt, c5_opt = optimize_adam(h_anti, n0, target0, dx0, d3, rng, lr_init=0.018, tau_init=7.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Fourier-based restart (180s)
    d4 = min(deadline, t0 + 860)
    if time.time() < d4:
        x = np.linspace(0.0, 2.0, n0, endpoint=False)
        ph = rng.randn(4) * 0.5
        h_ft = 0.5 + 0.3 * np.sin(np.pi * x + ph[0]) + 0.15 * np.sin(2 * np.pi * x + ph[1]) + 0.08 * np.sin(3 * np.pi * x + ph[2])
        h_ft = project_sum(h_ft, target0)
        h_opt, c5_opt = optimize_adam(h_ft, n0, target0, dx0, d4, rng, lr_init=0.015, tau_init=6.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Final polish + small-lr Adam (remaining ~200s)
    if time.time() < deadline:
        Nb = len(best_h)
        dxb = 2.0 / Nb
        tgtb = Nb / 2.0
        d_pol = min(deadline, t0 + budget_s - 40)
        h_pol, c5_pol = polish_lags(best_h.copy(), Nb, tgtb, dxb, d_pol, top_k=3)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        if time.time() < deadline - 5:
            h2, c5_2 = optimize_adam(best_h.copy(), Nb, tgtb, dxb, deadline, rng, lr_init=0.004, tau_init=12.0, max_iter=4000)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
        h_pol2, c5_pol2 = polish_lags(best_h.copy(), Nb, tgtb, dxb, deadline + 1, top_k=2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
