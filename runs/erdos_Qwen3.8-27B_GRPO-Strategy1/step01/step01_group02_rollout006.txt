This IS a convolution! Specifically, it's the convolution of W with (1-h), evaluated at position i, with appropriate indexing.

Similarly: sum_k w[k] * h[i - s_k] = sum_s W[s] * h[i-s] = sum_s W[-s] * h[i+s]

So:
grad = convolve(W_padded, 1-h) - convolve(W_padded_reversed, h) (with appropriate indexing)

This would be O(N²) in the worst case but can be done with FFT in O(N log N).

For N=190, O(N²) = 36k which is trivial. Let me just use the direct approach:

```python
def full_grad(h, w, N):
    """Compute full gradient given weights w over all lags."""
    # w has 2N-1 elements, w[k] for k=0..2N-2
    # lag_k = k - (N-1)
    grad = np.zeros(N)
    for i in range(N):
        # First term: sum_k w[k] * (1-h[i+lag_k]) for valid i+lag_k
        # Second term: sum_k w[k] * h[i-lag_k] for valid i-lag_k
        for k in range(2*N-1):
            lag = k - (N-1)
            j1 = i + lag
            j2 = i - lag
            if 0 <= j1 < N:
                grad[i] += w[k] * (1.0 - h[j1])
            if 0 <= j2 < N:
                grad[i] -= w[k] * h[j2]
    return grad
```

```python
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def corr_grad_lag(h, k, N):
    lag = k - (N - 1)
    g = np.zeros(N)
    n_lo = max(0, -lag)
    n_hi = N - 1 - abs(lag)
    if n_lo > n_hi:
        return g
    n = np.arange(n_lo, n_hi + 1)
    m = n + lag
    g[n] += (1.0 - h[m])
    g[m] += (-h[n])
    return g

def smooth_max_val_grad(h, dx, tau):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sc = tau * corr
    sc -= sc.max()
    e = np.exp(sc)
    w = e / e.sum()
    val = (sc.max() + np.log(e.sum())) / tau
    grad = np.zeros(N)
    active = np.where(w > 1e-10)[0]
    for k in active:
        grad += w[int(k)] * corr_grad_lag(h, int(k), N)
    return val, grad

def active_set_descent(h, dx, target, k_top, t0, budget_s, rng, max_it=3000):
    N = len(h)
    base_lr = 0.06
    for it in range(max_it):
        if time.time() - t0 > budget_s - 20:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_val = float(corr.max())
        top_idx = np.argsort(corr)[-k_top:]
        top_corr = corr[top_idx]
        tv = top_corr - top_corr.max()
        w = np.exp(4.0 * tv)
        w /= w.sum()
        grad = np.zeros(N)
        for idx, wi in zip(top_idx, w):
            grad += wi * corr_grad_lag(h, int(idx), N)
        lr = base_lr
        improved = False
        for _ in range(20):
            h_new = project_sum(h - lr * grad, target)
            c5_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx)
            if c5_new < c5_val - 1e-14:
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            base_lr *= 0.8
            if base_lr < 1e-7:
                break
            if it % 100 == 99 and base_lr < 0.01:
                h = project_sum(h + rng.randn(N) * 0.003, target)
                base_lr = 0.03
    return h

def surgical_optimize(h, dx, target, t0, budget_s, rng, max_rounds=300):
    N = len(h)
    for _ in range(max_rounds):
        if time.time() - t0 > budget_s - 10:

        worst = int(np.argmax(corr))
        g = corr_grad_lag(h, worst, N)
```
