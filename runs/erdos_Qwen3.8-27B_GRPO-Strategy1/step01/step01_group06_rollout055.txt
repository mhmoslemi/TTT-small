import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14:
            return h
        h = h + h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, lr):
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    mx = np.max(corr)
    S = np.where(corr >= mx - 1e-10)[0]
    k_off = n - 1
    grad = np.zeros(n)
    for idx in S:
        k = idx - k_off
        if k >= 0:
            i_lo, i_hi = 0, n - 1 - k
        else:
            i_lo, i_hi = -k, n - 1
        j_arr = np.arange(i_lo, i_hi + 1)
        contrib = np.full_like(j_arr, 1.0)
        hjk = h[j_arr + k]
        contrib[j_arr + k >= n] = -h[j_arr + k - n]
        np.add.at(grad, j_arr, (-hjk).where(np.arange(i_lo, i_hi+1)+k < n, h[j_arr + k]))
    grad = grad / len(S)
    return h, grad, corr, mx, S, k_off

def subgradient_optimize(h, n, n_iters, t0, budget_s, time_frac):
    deadline = t0 + budget_s * time_frac
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.3
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        thresh = mx - max(0.01 * mx, 1e-8)
        S = np.where(corr >= thresh)[0]
        k0 = n - 1
        grad = np.zeros(n)
        for idx in S:
            k = idx - k0
            if k >= 0:
                i_arr = np.arange(0, n - k)
            else:
                i_arr = np.arange(-k, n)
            j = i_arr
           terms1 = np.zeros(n)
            terms1[j] = 1.0
            terms2 = np.zeros(n)
            ik = i_arr + k
            valid = (ik >= 0) & (ik < n)
            terms2[j[valid]] = -h[ik[valid]]
            grad += terms1 + terms2
        grad /= max(len(S), 1)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.8)
        else:
            lr *= 0.7
            h = project_feasible(h, n)
        if lr < 1e-5:
            break
    return best_h, best_c5

def slsqp_smooth(h, n, T, maxiter=400):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = project_feasible(np.clip(res.x, 0, 1), n)
        return hc, compute_c5(hc)
    except Exception:
        return h.copy(), compute_c5(h)

def make_fourier_h(n, rng):
    h = np.zeros(n)
    n_freq = max(2, n // 6)
    freqs = np.sort(rng.choice(np.arange(1, n // 2 + 1), size=n_freq, replace=False))
    amps = 0.18 * rng.uniform(0.3, 1.0, size=n_freq)
    phas = rng.uniform(0, 2 * np.pi, size=n_freq)
    xx = np.arange(n)
    for f, a, p in zip(freqs, amps, phas):
        h += a * np.sin(2 * np.pi * f * xx / n + p)
    h = (h - h.min()) / (h.max() - h.min() + 1e-12)
    return project_feasible(h, n)
