        else:
            for i in range(len(h)):

                rem = min(room, -d)
                h[idx[i]] -= rem

def grad_subgrad(h, dx, top_m=3):

    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top m indices
    idx = np.argsort(corr)[-top_m:]
    weights = np.ones(len(idx)) / len(idx)
    grad = np.zeros(n)
    for w, k in zip(weights, idx):

        #   j = i (as 1-h term) -> contributes -h[i+shift] = -h[j+shift]

            i2 = j
            if 0 <= i2 < n and 0 <= i2 + shift < n:

    return grad * dx

def grad_subgrad_vec(h, dx, top_m=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k in idx:
        s = k - (n - 1)  # shift
        if s >= 0:
            # corr[k] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            h_term = h[s:]  # h[i+s] for i=0..n-1-s
            one_minus = (1-h)[:n-s]  # (1-h[i]) for i=0..n-1-s
            # d/dh[j]:
            # as h[i+s]: j=i+s, so i=j-s, valid for s<=j<n, term is (1-h[j-s])
            # as (1-h[i]): j=i, valid for 0<=j<n-s, term is -h[j+s]
            if s > 0:
                grad[s:] += (1-h)[:-s]  # (1-h[j-s]) for j=s..n-1
            # For j in [0, n-s): term is -h[j+s]
            if n - s > 0:
                grad[:n-s] -= h[s:]  # -h[j+s] for j=0..n-1-s
        else:
            s_neg = -s  # > 0
            # corr[k] = sum_{i=s_neg}^{n-1} h[i-s_neg]*(1-h[i])
            h_term = h[:n-s_neg]  # h[i-s_neg] for i=s_neg..n-1
            one_minus = (1-h)[s_neg:]  # (1-h[i]) for i=s_neg..n-1
            # d/dh[j]:
            # as h[i-s_neg]: j=i-s_neg, i=j+s_neg, valid for 0<=j<n-s_neg, term is (1-h[j+s_neg])
            # as (1-h[i]): j=i, valid for s_neg<=j<n, term is -h[j-s_neg]
            if n - s_neg > 0:
                grad[:n-s_neg] += (1-h)[s_neg:]  # (1-h[j+s_neg]) for j=0..n-s_neg-1
            if s_neg < n:
                grad[s_neg:] -= h[:n-s_neg]  # -h[j-s_neg] for j=s_neg..n-1
    return grad * dx / top_m
