    mx = corr.max()

            i2 = j + L
            if 0 <= i2 < n:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_smooth(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    shifted = beta * (corr - mx)
    w = np.exp(np.clip(shifted, 0, 500))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for L in range(-(n-1), n):
        m = n - 1 + L
        if w[m] < 1e-15:
            continue
        j_lo1 = max(0, L)
        j_hi1 = min(n-1, n-1+L)
        if j_lo1 <= j_hi1:
            idx = np.arange(j_lo1, j_hi1+1)
            grad[idx] += w[m] * (1.0 - h[idx - L])
        j_lo2 = max(0, -L)
        j_hi2 = min(n-1, n-1-L)
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2+1)
            grad[idx] -= w[m] * h[idx + L]
    return grad

def projected_gradient(h0, target_sum, n_iters, beta_start, beta_end, deadline):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/n)
    dx = 2.0/n
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(n_iters - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac**2
        lr = 1.0 / (beta + 1.0)
        g = grad_smooth(h, beta)
        # Skip the mean component (it's in the null space of the projection)
        g -= g.mean()
        # Line search
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = project_sr(h - lr * shrink * g, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                break
    return best_h, best_c5

def slsqp_epigraph(h0, dx, target_sum, maxiter=200):
    n = len(h0)
    def objective(x):
        return x[n]
    def ineq_cons(x):
        corr = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return x[n] - corr
    def eq_con(x):
        return np.sum(x[:n]) - target_sum
    t0 = float(np.max(np.correlate(h0, 1.0 - h0, mode='full'))) + 0.01
    x0 = np.concatenate([h0, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(2*n))]
    cons = [{'type': 'ineq', 'fun': ineq_cons}, {'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or (res.x is not None and res.fun < t0):
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def pair_swap(h, dx, target_sum, max_rounds=100, deadline=None):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    base_eps = 0.1
    for r in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        improved = False
        eps = base_eps * (0.5 ** min(r // 10, 5))
        for i in range(n):
            for j in range(i + 1, n):
                # Try moving from i to j and j to i
                for si, sj, d in [(1, -1, eps), (-1, 1, eps)]:
                    hi = h[i] + si * d
                    hj = h[j] + sj * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c5_new = compute_c5(h, dx)
                        if c5_new < best_c5 - 1e-14:
                            best_c5 = c5_new
                            improved = True
                        else:
                            h[i], h[j] = hi - si * d, hj - sj * d
                if improved:
                    break
            if improved:
                break
        if not improved:
            base_eps *= 0.5
            if base_eps < 1e-5:
                break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_slsqp_state = {}

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def grad_smooth(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    shifted = beta * (corr - mx)
    w = np.exp(np.clip(shifted, 0, 500))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for L in range(-(n - 1), n):
        m = n - 1 + L
        wm = w[m]
        if wm < 1e-15:
            continue
        j_lo1 = max(0, L)
        j_hi1 = min(n - 1, n - 1 + L)
        if j_lo1 <= j_hi1:
            idx = np.arange(j_lo1, j_hi1 + 1)
            grad[idx] += wm * (1.0 - h[idx - L])
        j_lo2 = max(0, -L)
        j_hi2 = min(n - 1, n - 1 - L)
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2 + 1)
            grad[idx] -= wm * h[idx + L]
    return grad

def projected_gradient(h0, target_sum, n_iters, beta_start, beta_end, deadline):

    dx = 2.0 / n
    h = h0.copy()

        lr = 2.0 / (beta + 2.0)
        g = grad_smooth(h, beta)
        g -= g.mean()
        found = False
        for shrink in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08, 0.04]:
            h_new = project_sr(h - lr * shrink * g, target_sum)
            c5_new = compute_c5(h_new, dx)

                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                found = True
                stall = 0
                break
        if not found:
            stall += 1
            if stall > 300:
                beta_start *= 1.5
                stall = 0
    return best_h, best_c5

    _slsqp_state['target_sum'] = target_sum
