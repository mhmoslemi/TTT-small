    sm = np.sum(np.exp(shifted))

        if c_new < cur - 1e-15:
            h = h_new
            cur = c_new
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
    return best_h, best_c

def random_pair_search(h, n, max_iter, rng):
    cur = compute_c5(h)
    best_h = h.copy()
    best_c = cur
    for it in range(max_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.002, 0.06) * rng.choice([-1.0, 1.0])
        h_tr = h.copy()
        h_tr[i] = np.clip(h[i] + d, 0, 1)
        h_tr[j] = np.clip(h[j] - d, 0, 1)
        if abs(np.sum(h_tr) - n / 2.0) > 1e-8:
            continue
        c = compute_c5(h_tr)
        if c < cur - 1e-14:
            h = h_tr
            cur = c
            if cur < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    parent = np.asarray(initial_h_values, dtype=np.float64)
    pn = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with beta sweep at parent resolution
    n = pn
    h = fix_sum(parent, n)
    beta_schedule = [0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0006, 0.0004, 0.0003]
    for beta in beta_schedule:
        if remaining() < 45:
            break
        maxit = 400 if beta > 0.01 else 600
        h = slsqp_optimize(h, n, beta, maxit)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 2: Projected gradient polish at small beta
    if remaining() > 60:
        for beta in [0.0003, 0.0002, 0.0001]:
            if remaining() < 35:
                break
            niter = min(8000, int(remaining() * 40))
            h, c5 = proj_grad_polish(best_h, best_n, beta, niter, lr_init=0.02)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 3: Subgradient targeting active lag
    if remaining() > 30:
        niter = min(20000, int(remaining() * 60))
        h, c5 = subgrad_polish(best_h, best_n, niter, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Try higher resolution
    if remaining() > 60:
        n2 = pn * 2
        if n2 <= 600:
            h2 = upsample(best_h, n2)
            for beta in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                if remaining() < 25:
                    break
                h2 = slsqp_optimize(h2, n2, beta, 300)
                c5 = compute_c5(h2)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h2.copy()
                    best_n = n2
            if remaining() > 20:
                h2, c5 = proj_grad_polish(h2, n2, 0.0004, 3000, 0.02)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h2.copy()
                    best_n = n2

    # Phase 5: Random pair local search
    if remaining() > 15:
        niter = min(30000, int(remaining() * 100))
        h, c5 = random_pair_search(best_h, best_n, niter, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final validation
    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
