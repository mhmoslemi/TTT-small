For lag k: sum h[i]*(1-h[i+k]) = sum h[i] - sum h[i]*h[i+k] = n/2 - sum h[i]*h[i+k]

To minimize the max, we want all the cross-correlations sum h[i]*h[i+k] to be as large and even as possible.

This is related to **difference sets** and **optimal arrangements** of ones on a circle/line.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
    return h

def _smax_obj(x, dx, beta):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ev = np.exp(beta * (c - m))
    return m + np.log(ev.sum()) / beta

def _smax_jac(x, dx, beta):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ev = np.exp(beta * (c - m))
    w = ev / ev.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        shift = k - (n - 1)
        wk = w[k] * dx
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 + shift:hi1 + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * x[lo2 - shift:hi2 - shift]
    return grad

def slsqp_opt(h0, dx, beta, maxiter=500):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return _smax_obj(x, dx, beta)
    def jac(x):
        return _smax_jac(x, dx, beta)
    def eqf(x):
        return x.sum() - target
    def eqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new, dx) < compute_c5(h0, dx):
                return h_new
    except Exception:
        pass
    return h0.copy()

def lbfgs_opt(h0, dx, beta, lam, maxiter=3000):
    n = len(h0)
    target = n / 2.0
    def obj_grad(x, *args):
        o = _smax_obj(x, dx, beta) + lam * (x.sum() - target) ** 2
        g = _smax_jac(x, dx, beta) + 2.0 * lam * (x.sum() - target)
        return o, g
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x, *a: (lambda v: (v[0], v[1]))(obj_grad(x, *a)),
                       h0, jac=True, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        h_new = project_feasible(res.x, target)
        if compute_c5(h_new, dx) < compute_c5(h0, dx):
            return h_new
    except Exception:
        pass
    return h0.copy()

def coord_descent(h0, dx, n_iters=200, seed=0):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    for it in range(n_iters):
        improved = False
        for j in rng.permutation(n):
            old = h[j]
            # h[n-1] absorbs the constraint
            if j == n - 1:
                continue
            target_j = target - h.sum() + old
            # Binary search for best h[j] in [0,1] s.t. h[n-1] in [0,1]
            lo = max(0.0, min(1.0, target_j - 1.0))
            hi = min(1.0, max(0.0, target_j))
            if lo >= hi:
                continue
            # Try a few values
            best_v = old
            best_f = compute_c5(h, dx)
            for frac in [0.0, 0.25, 0.5, 0.75, 1.0]:
                v = lo + frac * (hi - lo)
                if abs(v - old) < 1e-8:
                    continue
                h[j] = v
                h[n - 1] = target - h.sum() + old
                h[n - 1] = np.clip(h[n - 1], 0, 1)
                if abs(h[n-1] - (target - h.sum() + old)) < 1e-10:
                    f = compute_c5(h, dx)
                    if f < best_f - 1e-10:
                        best_f = f
                        best_v = v
            h[j] = best_v
            h[n - 1] = target - h.sum() + old
            h[n - 1] = np.clip(h[n-1], 0, 1)
            if best_v != old:
                improved = True
        if compute_c5(h, dx) < best_c5 - 1e-12:
            best_c5 = compute_c5(h, dx)
        if not improved:
            break
    h = project_feasible(h, target)
    return h

def interpolate_to_n(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0 / n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0 / n_dst
    h_interp = np.interp(x_dst, x_src, h_src, period=2.0)
    return project_feasible(h_interp, n_dst / 2.0)

def make_random_binary(n, rng):
    h = np.zeros(n)
    target = n / 2.0
    int_part = int(target)
    frac = target - int_part
    pos = rng.permutation(n)
    h[pos[:int_part]] = 1.0
    if frac > 1e-6:
        h[pos[int_part]] = frac
    return project_feasible(h, target)

def make_structured(n, rng, style=0):
    h = np.zeros(n)
    target = n / 2.0
    if style == 0:
        # Half-1, half-0 with random split
        n1 = int(target)
        pos = rng.permutation(n)
        h[pos[:n1]] = 1.0
        frac = target - n1
        if frac > 0.01:
            h[pos[n1]] = frac
    elif style == 1:
        # Periodic blocks
        period = rng.randint(3, 7)
        for i in range(n):
            h[i] = 1.0 if (i // period) % 2 == 0 else 0.0
    elif style == 2:
        # Sinusoidal
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return project_feasible(h, target)
