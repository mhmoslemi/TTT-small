def exact_coord_step(h, j, corr_full, n):
    """Find optimal h[j] to minimize max_k corr_full[k], given all other h fixed."""
    s = np.arange(-(n-1), n)  # shape (2n-1,)
    term1_valid = (j + s >= 0) & (j + s < n)
    term2_valid = (j - s >= 0) & (j - s < n)
    
    old_val = h[j]
    
    # For each k, express corr_full[k] = a_k + m_k*t + q_k*t^2
    # q_k is non-zero only for k = n-1 (s=0): q = -1
    # For s≠0: q_k = 0 (linear)
    # For s=0 (k=n-1): the term is t*(1-t) = t - t^2, so q = -1, and we need to adjust m
    
    a = corr_full.copy()
    m = np.zeros(2*n-1)
    q = np.zeros(2*n-1)
    
    # Remove old contribution
    for k in range(2*n-1):
        sk = s[k]
        # term1: i=j, valid if 0<=j+sk<n
        if term1_valid[k]:
            if sk == 0:  # quadratic: t*(1-t)
                a[k] -= old_val * (1 - old_val)  # remove old t*(1-t)
                q[k] = -1.0
                m[k] += 1.0
            else:
                a[k] -= old_val * (1 - h[j+sk])
                m[k] += (1 - h[j+sk])
        # term2: i=j-sk, valid if 0<=j-sk<n
        if term2_valid[k]:
            # contribution: h[j-sk]*(1-t) = h[j-sk] - h[j-sk]*t
            a[k] -= h[j-sk] * (1 - old_val)
            m[k] -= h[j-sk]
    
    # Now corr_full[k] = a[k] + m[k]*t + q[k]*t^2
    # Minimize max_k over t in [0,1]
    
    # Dense eval
    ts = np.linspace(0, 1, 100)
    vals = np.max(a + m*ts[:,None] + q*ts[:,None]**2, axis=1)
    best_idx = np.argmin(vals)
    
    # Refine with golden section around best
    lo = max(0, ts[max(0, best_idx-1)])
    hi = min(1, ts[min(len(ts)-1, best_idx+1)])

    c = hi - gr*(hi-lo)

            d = lo + gr*(hi-lo)
            fd = np.max(a + m*d + q*d**2)
    
    t_opt = (lo + hi) / 2

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode="full")

def softmax_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    if s == 0:
        return dx * m
    return dx * (m + beta * np.log(s))

def eq_fun(x, n):
    return np.sum(x) - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64)
    nx = np.linspace(0, old_n, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0.0, 1.0)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

def fix_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def subgradient_polish(h, n, n_iter=200, beta=0.01):

            #   For j in [max(0,sk), min(n, n+sk)]: subtract h[j-sk]

            if j_lo2 < j_hi2:

        h_new += (n/2.0 - h_new.sum()) / n
        h_new = np.clip(h_new, 0, 1)
        
        c5_new = compute_c5(h_new)
        if c5_new < best:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode="full")

def softmax_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / beta))
    if s < 1e-300:
        return dx * m
    return dx * (m + beta * np.log(s))

def eq_fun(x, n):
    return np.sum(x) - n / 2.0

def eq_jac(x, n):
    return np.ones(n)

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64)
    nx = np.linspace(0, old_n, new_n, endpoint=False)
    v = np.interp(nx, ox, parent)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    v = np.clip(v, 0.0, 1.0)
    v += (new_n / 2.0 - np.sum(v)) / new_n
    return np.clip(v, 0.0, 1.0)

def fix_sum(h, n):
    h = h + (n / 2.0 - np.sum(h)) / n
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=30, n_eval=80):

            t2v = (j - s_arr >= 0) & (j - s_arr < n)
            old = h[j]
            a = corr_full.copy()
            m = np.zeros(2 * n - 1)
            q = np.zeros(2 * n - 1)
            idx1 = np.where(t1v)[0]
            for k in idx1:
                sk = s_arr[k]

                    m[k] += 1.0

                    m[k] += (1.0 - h[j + sk])
            idx2 = np.where(t2v)[0]
            for k in idx2:
                sk = s_arr[k]
                a[k] -= h[j - sk] * (1.0 - old)
                m[k] -= h[j - sk]
            ts = np.linspace(0.0, 1.0, n_eval)
            vals = np.max(a[None, :] + m[None, :] * ts[:, None] + q[None, :] * ts[:, None] ** 2, axis=1)
            bidx = int(np.argmin(vals))
            if bidx <= 0:
                t_opt = 0.0
            elif bidx >= n_eval - 1:
                t_opt = 1.0
            else:
                lo = ts[max(0, bidx - 2)]
                hi = ts[min(n_eval - 1, bidx + 2)]
                ts2 = np.linspace(lo, hi, 30)
                vals2 = np.max(a[None, :] + m[None, :] * ts2[:, None] + q[None, :] * ts2[:, None] ** 2, axis=1)
                t_opt = ts2[int(np.argmin(vals2))]
            if abs(t_opt - old) > 1e-13:
                h[j] = t_opt
                corr_full = a + m * t_opt + q * t_opt ** 2
                improved = True
        h = fix_sum(h, n)
        corr_full = corr_vec(h)
        if not improved:
            break
    return h, corr_full.max() * (2.0 / n)

def subgradient_polish(h, n, n_iter=400, beta=0.008):
    h = h.copy()
    s_arr = np.arange(-(n - 1), n)
    best = compute_c5(h)
    for it in range(n_iter):
        corr_full = corr_vec(h)
        w = np.exp((corr_full - corr_full.max()) / beta)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            sk = s_arr[k]
            wl = max(0, -sk)
            wh = min(n, n - sk)
            if wl < wh:
                grad[wl:wh] += (1.0 - h[wl + sk:wh + sk]) * w[k]
            wl2 = max(0, sk)
            wh2 = min(n, n + sk)
            if wl2 < wh2:
                grad[wl2:wh2] -= h[wl2 - sk:wh2 - sk] * w[k]
        alpha = 0.008 / (1.0 + 0.005 * it)
        h_new = h - alpha * grad
        h_new += (n / 2.0 - h_new.sum()) / n
        h_new = np.clip(h_new, 0.0, 1.0)
        c5_new = compute_c5(h_new)
        if c5_new < best:
            best = c5_new
            h = h_new
    return h, best
