        return h

def subgradient(h, n, weights):
    """Compute subgradient given weights on correlations."""
    d_vals = np.arange(2*n-1) - (n-1)
    j = np.arange(n)
    jd = j[:, None] - d_vals[None, :]
    mask1 = (jd >= 0) & (jd < n)
    idx1 = np.clip(jd, 0, n-1)

    jd2 = j[:, None] + d_vals[None, :]
    mask2 = (jd2 >= 0) & (jd2 < n)

    vals2 = np.where(mask2, h[idx2], 0.0)
    term2 = vals2 @ weights

    ni = np.sum(interior)
    if ni > 0:
        h[interior] += diff / ni
    h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-10:
        h += (target - s2) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, n, weights):
    d_vals = np.arange(2*n-1) - (n-1)
    j = np.arange(n)
    jd = j[:, None] - d_vals[None, :]
    mask1 = (jd >= 0) & (jd < n)
    idx1 = np.clip(jd, 0, n-1)
    vals1 = np.where(mask1, 1.0 - h[idx1], 0.0)
    term1 = vals1 @ weights
    jd2 = j[:, None] + d_vals[None, :]
    mask2 = (jd2 >= 0) & (jd2 < n)
    idx2 = np.clip(jd2, 0, n-1)
    vals2 = np.where(mask2, h[idx2], 0.0)
    term2 = vals2 @ weights
    return term1 - term2

def slsqp_step(h, n, T, max_iter):
    corr_cache = {}
    def obj(x):
        corr = get_corr(x)
        mc = np.max(corr)
        return float(mc + T * np.log(np.sum(np.exp((corr - mc) / T))))
    def con(x):
        return np.sum(x) - n * 0.5
    res = opt.minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
    return res.x

def lbfgs_step(h, n, T, pen, max_iter):

        sm = mc + T * np.log(np.sum(np.exp((corr - mc) / T)))
        s = np.sum(x) - n * 0.5

                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': max_iter, 'gtol': 1e-12})
    return res.x
