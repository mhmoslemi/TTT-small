    """Find z in [0,1] minimizing max_i(c[i] + s[i]*z)."""
    # The function is convex piecewise-linear
    # Minimum is at z=0, z=1, or at an intersection point
    # Find the upper envelope and its minimum
    
    # Quick check at boundaries
    f0 = np.max(c)
    f1 = np.max(c + s)
    best_z = 0.0 if f0 <= f1 else 1.0
    best_f = min(f0, f1)
    
    # For interior: the minimum of max of lines is where the two "active" lines cross
    # Sort by slope, then the upper envelope changes at intersections
    # Simpler: use the fact that at the minimum, at least 2 lines are tied (or it's a boundary)
    # Find the z where the max is minimized by checking all pair intersections
    
    # Actually, a simpler approach: the minimum z* satisfies:
    # z* is where the derivative of the max changes sign
    # d/dz max_i(c_i + s_i*z) = s_i* where i* is the active line
    # At the minimum, the active line's slope changes from negative to positive
    # So: find z where the line with the highest value has slope crossing 0
    
    # For numerical efficiency, let's just use a reduced ternary search (20 iterations)
    lo, hi = 0.0, 1.0

        if np.max(c+s*m1) <= np.max(c+s*m2):
            hi = m2
        else:
            lo = m1
    z = (lo+hi)/2.0

def interp_upscale(h, factor):
    n, m = len(h), len(h) * factor
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(m) + 0.5) *(2.0 / m)
    return np.interp(xn, xo, h)

def coord_sweep(h, time_cap):
    n = len(h)
    dx = 2.0 / n
    t_s = time.time()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mp = np.arange(-(n - 1), n)
    order = np.random.permutation(n)
    for ii in range(n):
        if time.time() - t_s > time_cap:
            break
        i = order[ii]
        vp = (i + mp >= 0) & (i + mp < n)
        vm = (i - mp >= 0) & (i - mp < n)
        hp = np.zeros(2 * n - 1)
        hm = np.zeros(2 * n - 1)
        hp[vp] = h[i + mp[vp]]
        hm[vm] = h[i - mp[vm]]
        s = dx * (1.0 - hp - hm)
        ho = h[i]
        c = corr - ho * s
        lo, hi = 0.0, 1.0
        for _ in range(30):
            m1 = lo + (hi - lo) * (1.0 / 3.0)
            m2 = hi - (hi - lo) * (1.0 / 3.0)
            if np.max(c + s * m1) <= np.max(c + s * m2):
                hi = m2
            else:
                lo = m1
        hn = (lo + hi) * 0.5
        d = hn - ho
        if abs(d) > 1e-15:
            h[i] = hn
            corr += d * s
    return h, float(np.max(corr) * dx)

def lfbgbs_polish(h, n, t_limit):
    dx = 2.0 / n
    def obj_func(x):
        nh = np.clip(x, 0.0, 1.0)
        corr = np.correlate(nh, 1.0 - nh, mode='full')
        im = np.sum(nh) * dx
        return np.max(corr) * dx + 500.0 * (im - 1.0) ** 2
    res = minimize(obj_func, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10})
    if time.time() > t_limit:
        return h
    return repair_h(np.clip(res.x, 0.0, 1.0))

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()

    h_p = np.array(initial_h_values, dtype=float).copy()
    best_h = h_p.copy()
    best_c5 = compute_c5(h_p)

    # Upscale parent to n=720
    h = interp_upscale(h_p, 2)
    h = repair_h(h)
    n = len(h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 1: Coordinate descent at n=720 (55% budget)
    t_p1_end = t0 + budget_s * 0.55
    while time.time() < t_p1_end:
        cap = min(45.0, t_p1_end - time.time())
        if cap < 2.0:
            break
        h, c5 = coord_sweep(h, cap)
        h = repair_h(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: L-BFGS-B polish (15% budget)
    if time.time() < t0 + budget_s * 0.70:
        try:
            t_lim = t0 + budget_s * 0.72
            h2 = lfbgbs_polish(h, n, t_lim)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5, best_h = c5_2, h2.copy()
            h = h2
        except Exception:
            pass

    # Phase 3: More coordinate descent after polish (12% budget)
    t_p3_end = t0 + budget_s * 0.84
    while time.time() < t_p3_end:
        cap = min(30.0, t_p3_end - time.time())
        if cap < 2.0:
            break
        h, c5 = coord_sweep(h, cap)
        h = repair_h(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 4: Random restarts with coordinate descent (10% budget)
    t_p4_end = t0 + budget_s * 0.95
    n_restarts = 5
    for ri in range(n_restarts):
        if time.time() > t_p4_end:
            break
        sigma = 0.12 / (ri + 1) * (1.0 + ri * 0.3)
        hr = repair_h(np.clip(h + np.random.randn(n) * sigma, 0.0, 1.0))
        for _ in range(25):
            if time.time() > t_p4_end:
                break
            cap = min(20.0, t_p4_end - time.time())
            if cap < 1.0:
                break
            hr, cr = coord_sweep(hr, cap)
            hr = repair_h(hr)
            cr = compute_c5(hr)
            if cr < best_c5:
                best_c5, best_h = cr, hr.copy()

    # Final verification
    best_h = repair_h(best_h)
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, len(best_h)
