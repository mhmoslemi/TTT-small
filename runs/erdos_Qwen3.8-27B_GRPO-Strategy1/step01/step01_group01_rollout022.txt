```python
    lags = np.arange(2*n-1) - (n-1)
    al = np.abs(lags)
    
    # For each j (0 to n-1) and each m (0 to 2n-2):
    # contribution to g[j] from m:
    #   (1-h[j-al[m]]) * [al[m] <= j] - h[j+al[m]] * [al[m] < n-j]
    
    # Create index matrices
    J = np.arange(n)  # (n,)
    AL = al  # (2n-1,)
    
    # 2D: J[:, None] and AL[None, :]
    J2 = J[:, None]  # (n, 1)
    AL2 = AL[None, :]  # (1, 2n-1)
    
    # mask1: AL2 <= J2  → (n, 2n-1)
    mask1 = AL2 <= J2
    # mask2: AL2 < n - J2  → (n, 2n-1)
    mask2 = AL2 < (n - J2)
    
    # term1[j,m] = (1-h[j-AL[m]]) * mask1[j,m]
    # We need h[j - AL[m]] for valid j-AL[m] in [0, n-1]
    idx1 = J2 - AL2  # (n, 2n-1)
    # Clip to valid range, but only use where mask1 is True
    idx1_clipped = np.clip(idx1, 0, n-1)
    term1 = (1 - h[idx1_clipped]) * mask1 * w_full[None, :]
    
    # term2[j,m] = h[j+AL[m]] * mask2[j,m]
    idx2 = J2 + AL2  # (n, 2n-1)
    idx2_clipped = np.clip(idx2, 0, n-1)
    term2 = h[idx2_clipped] * mask2 * w_full[None, :]
    
    g = (term1.sum(axis=1) - term2.sum(axis=1)) / temp
    return g
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h, 0.0, 1.0).copy()

        h = np.clip(h, 0.0, 1.0)
```
