import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def slsqp_solve(h0, dx, target_sum, maxiter=300, ftol=1e-12):
    n = len(h0)
    t_init = compute_c5(h0, dx) / dx
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    x0 = np.concatenate([h0, [t_init + 0.003]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def penalty_grad(h, dx, threshold, beta=100.0):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    viol = np.maximum(0.0, corr - threshold)
    w = beta * viol
    grad = np.zeros(N)
    for j in range(N):
        k_lo = j - (N-1)
        k_hi = j
        ks = np.arange(k_lo, k_hi+1)
        idx = ks + (N-1)
        grad[j] += np.dot(w[idx], 1.0 - h[j - ks])
        k_lo2 = -j
        k_hi2 = N-1-j
        ks2 = np.arange(k_lo2, k_hi2+1)
        idx2 = ks2 + (N-1)

        k_hi = j
        ks = np.arange(k_lo, k_hi+1)

        idx2 = ks2 + (N-1)
        grad[j] -= np.dot(w[idx2], h[j + ks2])

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def slsqp_solve(h0, dx, target_sum, maxiter=300, ftol=1e-12):
    n = len(h0)
    t_init = compute_c5(h0, dx) / dx
    def obj(x):
        return x[n]
    def ineq(x):

        return np.sum(x[:n]) - target_sum
    x0 = np.concatenate([h0, [t_init + 0.002]])

                      options={'maxiter': maxiter, 'ftol': ftol})
