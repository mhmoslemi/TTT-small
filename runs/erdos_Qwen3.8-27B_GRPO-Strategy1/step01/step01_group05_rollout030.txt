```python
        diff = target - s

            h = np.clip(h, 0.0, 1.0)
            continue

        h[free] = np.clip(h[free], 0.0, 1.0)
    return h

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = temp * corr
    m = np.max(scaled)
    return (np.log(np.sum(np.exp(scaled - m))) + m) / temp * dx

def subgrad_step(h, n, alpha, iters=50):
    """Subgradient descent on the max correlation"""
    dx = 2.0 / n
    h = h.copy()
    for it in range(iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m_idx = np.argmax(corr)
        m = m_idx - n + 1  # shift
        # Gradient of corr[m_idx] w.r.t. h[i]
        g = np.zeros(n)
        for i in range(n):
            # (1-h[i+m])*[0<=i+m<n] - h[i-m]*[0<=i-m<n]
            val = 0.0
            if 0 <= i + m < n:
                val += (1.0 - h[i + m])
            if 0 <= i - m < n:
                val -= h[i - m]
            g[i] = val
        h -= alpha * g
        h = repair_h(h, n)
        alpha *= 0.995  # slow decay
    return h

def binary_local_search(h, n, n_iter=2000):
    """Local search on the binary structure"""
    dx = 2.0 / n
    # Round to nearest bang-bang with one fractional
    h_sig = h.copy()
    # Identify which value to keep fractional
```

```python
def subgrad_polish(h, n, n_iters=200, alpha0=0.01):
    dx = 2.0 / n
    h = h.copy()
    alpha = alpha0
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m_idx = np.argmax(corr)
        m = m_idx - (n - 1)
        g = np.zeros(n)
        for i in range(n):
            if 0 <= i + m < n:
                g[i] += (1.0 - h[i + m])
            if 0 <= i - m < n:
                g[i] -= h[i - m]
        h -= alpha * g
        h = repair_h(h, n)
        alpha *= 0.99
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / len(free)
            h[free] = np.clip(h[free], 0.0, 1.0)
    return h

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def soft_max_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    s = temp * corr
    m = np.max(s)
    return (np.log(np.sum(np.exp(s - m))) + m) / temp * dx

def slsqp_opt(x0, n, bounds, temp, maxiter):
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    res = minimize(soft_max_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return repair_h(res.x, n)

def subgrad_polish(h, n, n_iters=300, alpha0=0.005):
    h = h.astype(float).copy()
    alpha = alpha0
    for _ in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m_idx = np.argmax(corr)
        m = m_idx - (n - 1)
        g = np.zeros(n)
        for i in range(n):
            if 0 <= i + m < n:
                g[i] += (1.0 - h[i + m])
            if 0 <= i - m < n:
                g[i] -= h[i - m]
        h -= alpha * g
        h = repair_h(h, n)
        alpha *= 0.995
    return h

def discrete_search(h, n, max_rounds=3000, exhaustive=True):
    h_d = np.where(h > 0.6, 1.0, np.where(h < 0.4, 0.0, 0.5)).astype(float)
    h_d = repair_h(h_d, n)
    c5 = compute_c5(h_d, n)
    for rd in range(max_rounds):
        improved = False
        ones = np.where(h_d > 0.99)[0]
        zeros = np.where(h_d < 0.01)[0]
        halves = np.where(np.abs(h_d - 0.5) < 0.01)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if exhaustive:
            o_list = ones
            z_list = zeros
        else:
            o_list = np.random.permutation(ones)[:12]
            z_list = np.random.permutation(zeros)[:12]
        for i in o_list:
            for j in z_list:
                h_d[i], h_d[j] = h_d[j], h_d[i]
                c = compute_c5(h_d, n)
                if c < c5 - 1e-12:
                    c5 = c
                    improved = True
                else:
                    h_d[i], h_d[j] = h_d[j], h_d[i]
        for pos in halves:
            for new_pos in range(n):
                if new_pos == pos:
                    continue
                h_d[pos], h_d[new_pos] = h_d[new_pos], h_d[pos]
                c = compute_c5(h_d, n)
                if c < c5 - 1e-12:
                    c5 = c
                    improved = True
                else:
                    h_d[pos], h_d[new_pos] = h_d[new_pos], h_d[pos]
        if not improved:
            break
    return h_d, c5

def block_slsqp(h, n, bounds, temp, maxiter):
    return slsqp_opt(h, n, bounds, temp, maxiter)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(45, 0.5)
    n = len(h_init)
    h_init = repair_h(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP with decreasing temperature
    for temp in [300, 100, 40, 15, 5]:
        if time.time() - t_start > budget_s - 400:
            break
        h_t = block_slsqp(best_h, n, bounds, temp, 250)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    # Phase 2: Subgradient polish
    if time.time() - t_start < budget_s - 300:
        h_t = subgrad_polish(best_h, n, 400, 0.004)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    # Phase 3: Discrete local search (exhaustive single moves)
    if time.time() - t_start < budget_s - 150:
        h_d, c5_d = discrete_search(best_h, n, max_rounds=4000, exhaustive=True)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d.copy()
        # SLSQP refinement from discrete
        h_t = block_slsqp(best_h, n, bounds, 20, 400)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
        # Another discrete pass
        h_d2, c5_d2 = discrete_search(best_h, n, max_rounds=2000, exhaustive=True)
        if c5_d2 < best_c5:
            best_c5 = c5_d2
            best_h = h_d2.copy()

    # Phase 4: Multi-start with diverse initializations
    n_starts = 50
    for trial in range(n_starts):
        if time.time() - t_start > budget_s - 80:
            break
        mod = trial % 5
        if mod == 0:
            scale = 0.12 * np.exp(-trial / 25.0) + 0.01
            cand = best_h + np.random.uniform(-scale, scale, n)
        elif mod == 1:
            cand = np.random.uniform(0, 1, n)
        elif mod == 2:
            period = int(np.random.choice([3, 4, 5, 7, 9, 11]))
            base = np.random.uniform(0.05, 0.95, period)
            cand = np.tile(base, n // period + 1)[:n]
        elif mod == 3:
            cand = np.random.choice([0.0, 0.5, 1.0], n)
        else:
            # Structured: alternating blocks
            block = int(np.random.choice([2, 3, 4, 5]))
            cand = np.array([0.0 if (i // block) % 2 == 0 else 1.0 for i in range(n)])
        cand = repair_h(cand, n)

        rem = budget_s - 80 - (time.time() - t_start)
        if rem < 15:
            break
        h_t = block_slsqp(cand, n, bounds, 40, min(200, int(rem * 0.4)))
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            # Quick discrete refinement
            h_d3, c5_d3 = discrete_search(best_h, n, max_rounds=1000, exhaustive=False)
            if c5_d3 < best_c5:
                best_c5 = c5_d3
                best_h = h_d3.copy()
            h_t2 = block_slsqp(best_h, n, bounds, 15, 200)
            c5_t2 = compute_c5(h_t2, n)
            if c5_t2 < best_c5:
                best_c5 = c5_t2
                best_h = h_t2

    # Phase 5: Final combined polish
    if time.time() - t_start < budget_s - 10:
        h_t = subgrad_polish(best_h, n, 200, 0.003)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
        h_d4, c5_d4 = discrete_search(best_h, n, max_rounds=3000, exhaustive=True)
        if c5_d4 < best_c5:
            best_c5 = c5_d4
            best_h = h_d4.copy()
        rem = budget_s - 5 - (time.time() - t_start)
        if rem > 10:
            h_t = block_slsqp(best_h, n, bounds, 10, max(100, int(rem * 3)))
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t

    best_h = repair_h(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
```
