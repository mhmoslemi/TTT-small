    best_h = h.copy()

            break
        
        c5_val, c5_vals = compute_c5(h, dx)
        if c5_val < best_c5:

        # Backtracking line search

    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    mx = vals.max()
    w = np.exp(beta * (vals - mx))

    # Final polish

        dx_f = 2.0 / best_n
        tgt_f = best_n / 2.0
        hf = slsqp_polish(best_h, dx_f, tgt_f, 150)
        cf, _ = compute_c5(hf, dx_f)
        if cf < best_c5:
            best_h, best_c5 = hf, cf
        hf2, cf2 = coordinate_refine(best_h, dx_f, tgt_f, 3)
        if cf2 < best_c5:
            best_h, best_c5 = hf2, cf2

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5, _ = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
