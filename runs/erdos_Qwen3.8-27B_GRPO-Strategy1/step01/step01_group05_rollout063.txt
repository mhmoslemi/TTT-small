import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def constraint_fun(x, n):
    return np.sum(x) - n / 2.0

def soft_max_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = temp * corr
    m = np.max(scaled)
    if m > 400:
        return float(np.max(corr) * dx) + 5.0
    return (np.log(np.sum(np.exp(scaled - m))) / temp + 1.0) * dx

def repair_h(h, n, max_iter=1000):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13 * n:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            per = diff / float(nfree)
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    k = n // 2
    h = np.zeros(n)
    h[rng.choice(n, k, replace=False)] = 1.0
    return h

def sa_binary(n, t_start, time_limit, seed):
    if n % 2 != 0:
        return None, None
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    for si in range(n_starts):
        if time.time() - t_start > time_limit:
            break
        T = 0.4 + 0.1 * si
        T_min = 0.003
        cooling = 0.9999
        if si == 0 and 'initial_h_values' in globals() and initial_h_values is not None:
            h = repair_h(initial_h_values, n)
            h[h > 0.5] = 1.0
            h[h <= 0.5] = 0.0
            h = repair_h(h, n)
            if np.sum(h) != n / 2:
                h = random_binary(n, seed + si * 1000)
        else:
            h = random_binary(n, seed + si * 7919)
        h_c5 = compute_c5(h, n)
        h_best = h.copy()
        h_best_c5 = h_c5
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        no_improve = 0
        while T > T_min and time.time() - t_start < time_limit:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, n)
            delta = new_c5 - h_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
                h_c5 = new_c5
                no_improve = 0
                if new_c5 < h_best_c5:
                    h_best = h.copy()
                    h_best_c5 = new_c5
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
                no_improve += 1
            T *= cooling
            if no_improve > 50000:
                break
        if h_best_c5 < best_c5:
            best_c5 = h_best_c5
            best_h = h_best.copy()
    return best_h, best_c5

def slsqp_solve(x0, n, temp, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        soft_max_obj, x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_fun, 'args': (n,)}],
        args=(n, temp),
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return res.x

def fourier_to_h(params, n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    nh = len(params) // 2
    for k in range(1, nh + 1):
        h += params[2 * (k - 1)] * np.cos(k * np.pi * x)
        h += params[2 * (k - 1) + 1] * np.sin(k * np.pi * x)
    return np.clip(h, 0.0, 1.0)

def fourier_obj(params, n):
    h_f = fourier_to_h(params, n)
    h_f = repair_h(h_f, n)
    return compute_c5(h_f, n)

            best_h = h_sa
            best_n = nt

    n_opts = [40, 42, 44, 45, 46, 48, 50, 54]

        rem = budget_s - 60 - (time.time() - t_start)
        if rem < 12:

        if no == best_n:
            h_st = best_h.copy()
        else:
            idx = np.round(np.linspace(0, n_parent - 1, no)).astype(int)
            h_st = h_parent[idx].copy()

        for K in [3, 5, 7]:
            if time.time() - t_start > budget_s - 10:

                c5_f = compute_c5(h_f, best_n)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f

            h_t = slsqp_solve(cand, best_n, 15, 80)
