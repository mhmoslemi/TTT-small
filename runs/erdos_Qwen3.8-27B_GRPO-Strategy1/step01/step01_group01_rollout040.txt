    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

    for rnd in range(max_rounds):
        if time_limit and time.time() > t0 + time_limit:
            break
        improved = False
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])

        rng.shuffle(ones)
        rng.shuffle(zeros)

        max_trials = 2000
        trials = 0
        for i in ones:
            trials += 1
            if trials > max_trials:
                break
            for j in zeros:
                trials += 1
                if trials > max_trials:
                    break
                # Try swap i (1->0) and j (0->1)
                h[i] = 0.0
                h[j] = 1.0
                c = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
                if c < best - 1e-12:
                    best = c
                    improved = True
                    # Restart search
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, t=80.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(c)
    s = t * (c - mx)
    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def ceq(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def ceqj(x):
    return np.ones((1, len(x)))

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def binary_ls(h, n, max_iters=30000, rng=None, t0=0, time_limit_s=45):
    h = to_binary(h, n).astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    for it in range(max_iters):
        if time.time() > t0 + time_limit_s:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = h.copy()
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c5

def sla_ls(h, n, rng, t0, time_limit_s=45):
    """Simulated annealing local search."""
    h = to_binary(h, n).astype(float)
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    T = 0.005
    
    while time.time() < t0 + time_limit_s and T > 1e-6:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        d = c - cur_c5
        if d < 0 or np.log(rng.rand()) > -d / T:
            cur_c5 = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T *= 0.99995
    
    return best_h, best_c5

def slsqp_opt(h, t0, time_limit_s, t=80.0, maxiter=2000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
    
    def obj(x):
        return obj_smooth(x, t=t)
    
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bnds,
                     constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = proj(np.clip(r.x, 0.0, 1.0), n)
        return hc, compute_c5(hc)
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    n = 280
    h0 = None

    if h0 is None:

        h0 = proj(h0, n)
    
    bh = h0.copy()
    bc5 = compute_c5(bh)
    
    # Phase 1: SLSQP optimization
    deadline1 = t0 + 0.4 * budget_s
    while time.time() < deadline1:
        n_cur = len(bh)
        hc, c5 = slsqp_opt(bh, t0, deadline1 - time.time(), t=80.0, maxiter=1500)
        if c5 < bc5:
            bc5 = c5
            bh = hc
        # Add perturbation for next round
        bh = proj(bh + rng.randn(n_cur) * 0.02, n_cur)
    
    # Phase 2: Binary local search + SLSQP iteration
    deadline2 = t0 + 0.8 * budget_s
    while time.time() < deadline2:
        n_cur = len(bh)
        # Project to binary
        bb, bc5_b = binary_ls(bh, n_cur, max_iters=20000, rng=rng, t0=t0,
                              time_limit_s=min(30, deadline2 - time.time()))
        if bc5_b < bc5:
            bc5 = bc5_b
            bh = bb.copy()
        # SLSQP from binary
        hc, c5 = slsqp_opt(bh, t0, min(60, deadline2 - time.time()), t=100.0, maxiter=2000)
        if c5 < bc5:
            bc5 = c5
            bh = hc
    
    # Phase 3: Simulated annealing on binary
    deadline3 = t0 + 0.92 * budget_s
    while time.time() < deadline3:
        n_cur = len(bh)
        bb, bc5_b = sla_ls(bh, n_cur, rng, t0,
                           time_limit_s=min(30, deadline3 - time.time()))
        if bc5_b < bc5:
            bc5 = bc5_b
            bh = bb.copy()
            # SLSQP polish
            hc, c5 = slsqp_opt(bh, t0, min(30, deadline3 - time.time()), t=200.0, maxiter=1000)
            if c5 < bc5:
                bc5 = c5
                bh = hc
    
    # Phase 4: Final polish with high-temperature smooth
    if time.time() < t0 + budget_s - 5:
        n_cur = len(bh)
        bnds = [(0.0, 1.0)] * n_cur
        cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
        
        def obj_hi(x):
            return obj_smooth(x, t=500.0)
        
        try:
            r = minimize(obj_hi, bh, method='SLSQP', bounds=bnds,
                         constraints=cons,
                         options={'maxiter': 5000, 'ftol': 1e-16})

            c5 = compute_c5(hc)
            if c5 < bc5:
                bc5 = c5
                bh = hc
        except:
            pass
    
    bh = proj(bh, len(bh))
    return bh, compute_c5(bh), len(bh)

    # Vectorize: create index matrices
    k_idx = np.arange(n)[:, None]  # (n, 1)
    j_idx = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    
    # First term indices: k+j-n+1
    idx1 = k_idx + j_idx - n + 1  # (n, 2n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    
    # Second term indices: k-j+n-1
    idx2 = k_idx - j_idx + n - 1  # (n, 2n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    # First term: (1-h[idx1]) * w[j_idx] * valid1
    # We need h[idx1_clipped] but only where valid
    idx1_c = np.clip(idx1, 0, n-1)
    idx2_c = np.clip(idx2, 0, n-1)
    
    term1 = valid1 * (1.0 - h[idx1_c]) * w[None, :]  # (n, 2n-1)
    term2 = valid2 * h[idx2_c] * w[None, :]  # (n, 2n-1)
    
    grad = dx * (term1.sum(axis=1) - term2.sum(axis=1))
    return val, grad
