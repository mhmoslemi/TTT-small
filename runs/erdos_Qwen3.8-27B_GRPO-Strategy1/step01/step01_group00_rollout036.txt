```python
    h = h.copy()
```

```python
import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-15:
        return c_max * dx
    s = np.sum(np.exp((corr - c_max) / T))
    return (c_max + T * np.log(s)) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    target = 0.5 * n
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += diff / n_int
    h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, n, T, max_iter):
    try:
        res = optimize.minimize(
            lse_obj, h, args=(T,),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': max_iter, 'ftol': 1e-16}
        )
        h_new = fix_sum(np.clip(res.x, 0, 1), n)
        if compute_c5(h_new) < compute_c5(h):
            return h_new
    except Exception:
        pass
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
```

- Then enforce 0 ≤ h[n-1] ≤ 1 as an additional constraint

```python
import numpy as np

def compute_c5(h):
```
```python
    dx = 2.0 / len(h)
```
```python
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lse_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.max(corr)
    if T < 1e-15:
        return c * dx
    return (c + T * np.log(np.sum(np.exp((corr - c) / T)))) * dx

def sum_con(h, n):
    return np.sum(h) - 0.5 * n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) < 1e-14:
        return h
    m = (h > 1e-7) & (h < 1.0 - 1e-7)
    k = np.sum(m)
    if k > 0:
        h[m] += d / k
    h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        for i in range(n):
            if 1e-5 < h[i] < 1.0 - 1e-5:
                h[i] = min(max(h[i] + d2, 0.0), 1.0)
                break
    return h

def slsqp(h, n, T, iters):
    c0 = compute_c5(h)
    try:
        r = optimize.minimize(lse_obj, h, args=(T,), method='SLSQP',
            bounds=[(0., 1.)] * n,
            constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
            options={'maxiter': iters, 'ftol': 1e-16})
        hn = fix_sum(r.x, n)
        if compute_c5(hn) < c0:
            return hn
    except Exception:
        pass
    return h

    x1 = np.arange(nn, dtype=float) * (2.0 / nn)
    return fix_sum(np.interp(x1, x0, h, period=2.0), nn)

                continue
            best_c5 = compute_c5(h)
            best_h = h.copy()

                hn = h.copy()
                hn[:] += -delta / (n - 1)
                hn[j] = min(max(h[j] + delta, 0.0), 1.0)
                hn = np.clip(hn, 0.0, 1.0)
                c = compute_c5(hn)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_h = hn
            h = best_h
        h = fix_sum(h, n)
```
