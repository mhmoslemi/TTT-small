import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    if T < 1e-6:
        return m * dx
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + float(np.log(np.exp(shifted).sum())) / T) * dx

def sum_diff(x):
    return float(x.sum() - len(x) / 2.0)

def sum_jac(x):
    return np.ones(len(x))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h = h * (1.0 + diff / total)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_sweeps, deadline):
    t0 = time.time()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old = h[i]
            cands = np.array([0.0, 0.03, 0.07, 0.12, 0.18, 0.25, 0.35,
                             0.5, 0.65, 0.75, 0.82, 0.88, 0.93, 0.97, 1.0, old])
            cands = np.unique(np.round(np.clip(cands, 0, 1), 6))
            best_cv = best_c5
            best_v = old
            for cv in cands:
                h_try = h.copy()
                h_try[i] = cv
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_cv - 1e-13:
                    best_cv = c_try
                    best_v = cv
            if best_v != old:
                h[i] = best_v
                h = project_feasible(h, n)
                best_c5 = compute_c5(h)
                improved = True
        if not improved:
            break
    return h, best_c5
