```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fft_correlate(a, v):
    n, m = len(a), len(v)
    size = n + m - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    fv = np.fft.rfft(v[::-1], fsize)
    return np.fft.irfft(fa * fv, fsize)[:size]

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    h = h.copy()
    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
    n_int = np.sum(interior)
    if n_int > 0:
        h[interior] += (target - s) / n_int
        h = np.clip(h, 0.0, 1.0)
    # If still off, distribute to all
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    c_max = np.max(corr)
    if T < 1e-12:
        return c_max * dx
    shifted = corr - c_max
    s = np.sum(np.exp(shifted / T))
    return (c_max + T * np.log(s)) * dx

def softmax_objective_grad(h, T):
    # Returns (objective, gradient)
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    c_max = np.max(corr)
    if T < 1e-12:
        # Gradient of max is not well-defined, use subgradient
        active = corr == c_max
        weights = active / np.sum(active)
    else:
        shifted = corr - c_max
        exp_vals = np.exp(shifted / T)
        weights = exp_vals / np.sum(exp_vals)
    
    # Gradient of corr[k] w.r.t. h[j]
```

One more idea: **adaptive perturbation**. After each SLSQP run, analyze which shifts k are "active" (achieve the max). Then perturb h specifically at positions that would reduce those active shifts.

```python
def directed_perturbation(h, strength):
    n = len(h)
    corr = fft_correlate(h, 1.0 - h)
    c_max = np.max(corr)
    # Active shifts (within 10% of max)
    active = corr > 0.99 * c_max
    if np.sum(active) == 0:
        return h + np.random.randn(n) * 0.01
    # For each active shift k, perturb h to reduce corr[k]
    pert = np.zeros(n)
    for k in range(2*n-1):
        if not active[k]:
            continue
        # corr[k] = sum_i h[i+k](1-h[i])
        # d(corr[k])/d(h[j]) = (1-h[j-k]) - h[j+k] (approximately)
        # To decrease corr[k], move h in direction -d(corr[k])/d(h)
        for j in range(n):
            # j contributes to corr[k] if j-k in [0,n-1] (as h[i+k] with i=j-k)
            # or if j+k in [0,n-1] (as h[i] with i=j, contributing via (1-h[i]))
            contrib = 0
            if 0 <= j-k < n:
                contrib += (1 - h[j-k])
            if 0 <= j+k < n:
                contrib -= h[j+k]
            pert[j] -= contrib  # negative gradient direction
    pert = pert / (np.max(np.abs(pert)) + 1e-10)  # normalize
    return h + strength * pert
```

```python
import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fft_correlate(a, v):
    n, m = len(a), len(v)
    size = n + m - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    fv = np.fft.rfft(v[::-1], fsize)
    return np.fft.irfft(fa * fv, fsize)[:size]

def fix_sum(h, n):
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    h = h.copy()
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    ni = np.sum(interior)
    if ni > 0:
        h[interior] += (target - s) / ni
    h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-10:
        h += (target - s2) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    c_max = np.max(corr)
    if T < 1e-14:
        obj = c_max * dx
        active = (corr >= c_max - 1e-10).astype(float)
        w = active / np.sum(active)
    else:
        sh = np.clip((corr - c_max) / T, -80, 0)
        exp_s = np.exp(sh)
        w = exp_s / np.sum(exp_s)
        obj = (c_max + T * np.log(np.sum(exp_s))) * dx
    grad = np.zeros(n)
    for j in range(n):
        k_lo1 = max(0, j)
        k_hi1 = min(2 * n - 2, j + n - 1)
        if k_lo1 <= k_hi1:
            ks = np.arange(k_lo1, k_hi1 + 1)
            idx = j - ks + n - 1
            grad[j] += np.dot(w[ks], 1.0 - h[idx])
        k_lo2 = max(0, n - 1 - j)
        k_hi2 = min(2 * n - 2, 2 * n - 2 - j)
        if k_lo2 <= k_hi2:
            ks = np.arange(k_lo2, k_hi2 + 1)
            idx = j + ks - (n - 1)
            grad[j] -= np.dot(w[ks], h[idx])
    grad *= dx
    return obj, grad

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, new_n)

def try_slsqp(h_start, n, T, max_iter):
    res = opt.minimize(
        softmax_obj_grad, h_start,
        args=(T,),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': max_iter, 'ftol': 1e-14}
    )
    h_out = np.clip(res.x, 0.0, 1.0)
    return fix_sum(h_out, n)

def fourier_h(params, n, K):
    x = np.arange(n) * 2.0 / n
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += params[k - 1] * np.cos(k * np.pi * x)
    return h
```
