                h0 = proj(bh + 0.008 * rng.randn(npoints), npoints)
            # Subgradient nudge to potentially escape
            h0 = subgrad_nudge(h0, npoints, nsteps=30, alpha0=0.002)
            hnew = slsqp_run(h0, t, npoints, maxiter=3500, ftol=1e-14)
            c5 = compute_c5(hnew)
            if c5 < bc:
                bc, bh = c5, hnew

    # Phase 4: L-BFGS-B refinement (faster per iteration)
    if time.time() < t0 + budget_s * 0.90:
        for t in [2000.0, 5000.0]:
            if time.time() > t0 + budget_s * 0.90 - 5:
                break
            hnew = lbfgsb_run(bh, t, npoints, m_penalty=10000.0, maxiter=2000)
            c5 = compute_c5(hnew)
            if c5 < bc:
                bc, bh = c5, hnew

    # Phase 5: Ultra-fine SLSQP polish
    deadline_p = t0 + budget_s - 10
    for t in [8000.0, 15000.0, 30000.0]:
        if time.time() > deadline_p - 5:
            break
        hnew = slsqp_run(bh, t, npoints, maxiter=5000, ftol=1e-15)
        c5 = compute_c5(hnew)
        if c5 < bc:
            bc, bh = c5, hnew

    # Final subgradient polish
    if time.time() < t0 + budget_s - 8:
        bh = subgrad_nudge(bh, npoints, nsteps=200, alpha0=0.0005)
        c5 = compute_c5(bh)
        if c5 < bc:
            bc = c5

    bh = proj(bh, npoints)
    final_c5 = compute_c5(bh)
    return bh, final_c5, len(bh)
