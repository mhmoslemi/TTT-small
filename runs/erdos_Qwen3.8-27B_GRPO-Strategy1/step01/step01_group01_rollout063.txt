import numpy as np
from scipy.optimize import minimize
import time

_tg = 80.0
_pg = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_og(h, t, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    e = np.exp(s)
    se = np.sum(e)
    w = e / se
    val = (mx + np.log(se) / t) * dx
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-16:
            continue
        l = m - (n - 1)
        j1s = max(0, -l)
        j1e = min(n, n - l)
        if j1s < j1e:
            g[j1s:j1e] += wm * (1.0 - h[j1s + l:j1e + l])
        j2s = max(0, l)
        j2e = min(n, n + l)
        if j2s < j2e:
            g[j2s:j2e] -= wm * h[j2s - l:j2e - l]
    g *= dx
    if pen > 0.0:
        deficit = np.sum(h) - n / 2.0
        val += pen * deficit * deficit
        g += 2.0 * pen * deficit
    return val, g

    v, _ = smooth_max_og(x, _tg, _pg)
    return v

    _, g = smooth_max_og(x, _tg, _pg)
    return g

    xd = np.arange(nd) * 2.0 / nd
    return proj_sum(np.interp(xd, xs, h), nd)

    bh = h_init.copy()

    def do_opt(h_start, t, maxiter, pen):

        _tg = t
        _pg = pen

            return h_start
