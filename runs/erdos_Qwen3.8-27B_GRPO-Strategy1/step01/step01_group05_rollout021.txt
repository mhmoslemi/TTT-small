def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def subgradient_c5(h, n):
    """Return (c5_value, subgradient)"""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    c5_val = np.max(corr) * dx
    k_star = np.argmax(corr)
    shift = k_star - (n - 1)
    
    # Gradient of corr[k_star] w.r.t. h
    # c[k_star] = sum_i h[i + shift] * (1 - h[i])
    # for valid i, i+shift
    idx_a = np.arange(n) + shift
    valid = (idx_a >= 0) & (idx_a < n)
    
    # h[i+shift] * (1-h[i])
    # d/dh[j]: if j = i+shift, contrib is (1-h[i]) = (1-h[j-shift])
    #          if j = i, contrib is -h[i+shift] = -h[j+shift]
    
    grad = np.zeros(n)
    i_valid = np.arange(n)[valid]
    j_valid = idx_a[valid]
    
    # Term 1: h[j] appears as h[i+shift] → derivative is (1-h[i])
    # For each valid pair (i, i+shift), h[i+shift] has derivative (1-h[i])
    np.add.at(grad, j_valid, 1.0 - h[i_valid])
    
    # Term 2: h[j] appears in (1-h[i]) with i=j → derivative is -h[i+shift]
    # For each valid pair (i, i+shift), h[i] has derivative -h[i+shift]
    np.add.at(grad, i_valid, -h[j_valid])
    
    return c5_val, grad * dx

            continue
        per = diff / nfree

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * temp
    m = np.max(scaled)
    return (np.log(np.sum(np.exp(scaled - m))) + m) / temp * dx

def smooth_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * temp
    m = np.max(scaled)
    p = np.exp(scaled - m)
    p = p / np.sum(p)
    # gradient of the softmax
    # d/dh[j] of softmax = sum_k p[k] * d(corr[k])/dh[j]
    grad = np.zeros(n)

        j_valid = idx_a[valid]
        grad[j_valid] += p[k] * (1.0 - h[i_valid])
        grad[i_valid] += -p[k] * h[j_valid]
    return grad * dx

def weighted_subgrad(h, n, top_k=3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k shifts
    top_indices = np.argsort(corr)[-top_k:]
    # Weights: how much each exceeds the rest
    top_values = corr[top_indices]
    rest_max = np.max(np.delete(corr, top_indices))
    weights = np.maximum(0, top_values - rest_max) + 0.1
    weights = weights / np.sum(weights)
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_indices):

        j_v = idx[valid]
        grad[j_v] += w * (1.0 - h[i_v])
        grad[i_v] += -w * h[j_v]
    
    return float(np.max(corr) * dx), grad * dx

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * temp
    m = np.max(scaled)
    return (np.log(np.sum(np.exp(scaled - m))) + m) / temp * dx


def smooth_grad(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * temp
    m = np.max(scaled)
    p = np.exp(scaled - m)
    p /= np.sum(p)
    grad = np.zeros(n)
    idx_all = np.arange(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        shift = k - (n - 1)
        idx = idx_all + shift
        valid = (idx >= 0) & (idx < n)
        iv = idx_all[valid]
        jv = idx[valid]
        grad[jv] += w * (1.0 - x[iv])
        grad[iv] += -w * x[jv]
    return grad * dx


def sum_constraint(x, n):
    return np.sum(x) - n / 2.0


def project_sum(h, n):

        grad[iv] += -w * h[jv]
    return c5_val, grad * dx
