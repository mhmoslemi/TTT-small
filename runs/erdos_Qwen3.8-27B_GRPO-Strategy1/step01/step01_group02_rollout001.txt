        # corr at lag: sum_{i=0}^{N-1-lag} h[i+lag] * (1-h[i])
        # = sum_{i=lag}^{N-1} h[i] * (1-h[i-lag])
        # d/dh[j]: if j >= lag: (1-h[j-lag]); if j < N-lag: -h[j+lag]
        g[lag:] += (1.0 - h[:N-lag])  # g[j] for j>=lag gets (1-h[j-lag])
        g[:N-lag] -= h[lag:]  # g[j] for j<N-lag gets -h[j+lag]
    elif -N < lag < 0:
        l = -lag
        # sym: corr at -l = corr of (1-h) with h at l
        # sum_{i=0}^{N-1-l} (1-h[i+l]) * h[i]
        # d/dh[j]: if j < N-l: h[j] contributes (1-h[j+l])... wait
        # Actually: sum_{i=0}^{N-1-l} h[i] * (1-h[i+l])
        # d/dh[j]: if j < N-l: (1-h[j+l]); if j >= l: -h[j-l]
        g[:N-l] += (1.0 - h[l:])  # for j in [0, N-l): (1-h[j+l])
        g[l:] -= h[:N-l]  # for j in [l, N): -h[j-l]
    return g

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, m, N):
    g = np.zeros(N)
    if 0 <= m < N:
        g[m:] += (1.0 - h[:N - m])
        g[:N - m] -= h[m:]
    elif -N < m < 0:
        l = -m
        g[:N - l] += (1.0 - h[l:])
        g[l:] -= h[:N - l]
    return g

def optimize_core(h, N, target, dx, t0, deadline, lr=0.02, tau=10.0, K_start=4, K_frac=0.15):
    best_c5 = compute_c5(h.copy(), dx)
    best_h = h.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(N)
    v_buf = np.zeros(N)
    K_end = max(K_start + 1, int(N * K_frac))
    stagnation = 0
    best_stag_c5 = best_c5

    for it in range(20000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        c5 = float(np.max(c))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Scramble restart if stuck
        if stagnation > 200 and best_c5 > best_stag_c5 - 1e-6:
            best_stag_c5 = best_c5
            h = project_sum(best_h + np.random.randn(N) * 0.08, target)
            m_buf[:] = 0.0
            v_buf[:] = 0.0
            stagnation = 0
            lr = max(lr * 3.0, 0.01)
            tau = max(tau * 0.5, 5.0)

        # Adaptive K
        frac = min(1.0, (it + 1) / 800.0)
        K = max(3, int(K_start + (K_end - K_start) * frac))

        # Top-K selection
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
        c_max = c_top.max()
        c_min = c_top.min()
        if c_max - c_min > 1e-12:
            w = np.exp(tau * (c_top - c_max))
            w /= w.sum()
        else:
            w = np.ones(K) / K

        # Gradient (skip tiny weights)
        thresh = 0.002 / N
        grad = np.zeros(N)
        for j in range(K):
            if w[j] > thresh:
                lag = int(top_idx[j]) - (N - 1)
                grad += w[j] * grad_lag(h, lag, N)

        # Adam with warmup
        warm = min(1.0, (it + 1) / 30.0)
        eff_lr = lr * warm
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * (grad * grad)
        m_hat = m_buf / (1 - beta1 ** (it + 1))
        v_hat = v_buf / (1 - beta2 ** (it + 1))
        h = h - eff_lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_sum(h, target)

        # Schedule
        if (it + 1) % 250 == 0:
            lr *= 0.82
            tau *= 2.2
        if eff_lr < 1e-10:
            break

    return best_h, best_c5

def make_inits(N, target, h_par, x_par, rng):
    inits = []
    xN = np.linspace(0.0, 2.0, N, endpoint=False)
    h_i = project_sum(np.interp(xN, x_par, h_par), target)
    inits.append(h_i.copy())
    for ns in [0.03, 0.07, 0.12, 0.20]:
        inits.append(project_sum(h_i + rng.randn(N) * ns, target))
    # Binary random
    h_r = np.zeros(N)
    h_r[rng.choice(N, int(target), replace=False)] = 1.0
    inits.append(h_r)
    # Block pattern
    nb = rng.randint(3, max(4, N // 15))
    h_b = np.zeros(N)
    bs = max(2, N // nb)
    for i in range(0, N, bs):
        h_b[i:i + max(1, bs // 2)] = 1.0
    inits.append(project_sum(h_b, target))
    # Sine
    freq = rng.randint(3, 12)
    t = np.linspace(0, 2 * np.pi * freq, N)
    h_s = np.clip(0.5 + 0.45 * np.sin(t + rng.rand() * 6.28), 0, 1)
    inits.append(project_sum(h_s, target))
    return inits
