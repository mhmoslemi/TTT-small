            # Block move (length 2-5)
            L = rng.randint(2, 6)
            i = rng.randint(0, n - L)
            j = rng.randint(0, max(1, n - L))
            if abs(i - j) < L:
                j = (j + L + rng.randint(1, 5)) % max(1, n - L)
            scale = 0.004 + 0.02 * (T / T0_temp)
            d = rng.normal(0.0, scale, L)
            h_tr = h.copy()
            h_tr[i:i + L] = np.clip(h[i:i + L] + d, 0.0, 1.0)
            h_tr[j:j + L] = np.clip(h[j:j + L] - d, 0.0, 1.0)
            h_tr = fix_sum_clip(h_tr, n)
        elif r < 0.92:
            # Larger block move (length 5-10)
            L = rng.randint(5, 11)
            if L >= n:
                L = max(2, n // 4)
            i = rng.randint(0, n - L)
            j = rng.randint(0, max(1, n - L))
            if abs(i - j) < L:
                j = (j + L) % max(1, n - L)
            scale = 0.003 + 0.015 * (T / T0_temp)
            d = rng.normal(0.0, scale, L)
            h_tr = h.copy()
            h_tr[i:i + L] = np.clip(h[i:i + L] + d, 0.0, 1.0)
            h_tr[j:j + L] = np.clip(h[j:j + L] - d, 0.0, 1.0)
            h_tr = fix_sum_clip(h_tr, n)
        else:
            # Swap two values
            i, j = rng.choice(n, 2, replace=False)
            h_tr = h.copy()
            h_tr[i], h_tr[j] = h[j], h[i]

        c_tr = compute_c5(h_tr)
        delta = c_tr - cur_c5
        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            h = h_tr
            cur_c5 = c_tr
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    # Phase 4: Final greedy coordinate descent
    if budget_s - (time.time() - t0) > 8:
        h = best_h.copy()
        c = best_c5
        while budget_s - (time.time() - t0) > 3:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.001, 0.012) * rng.choice([-1.0, 1.0])
            h_tr = h.copy()
            old_i = h_tr[i]
            h_tr[i] = np.clip(old_i + d, 0.0, 1.0)
            actual_d = h_tr[i] - old_i
            h_tr[j] = np.clip(h_tr[j] - actual_d, 0.0, 1.0)
            fix = n / 2.0 - np.sum(h_tr)
            if abs(fix) > 1e-13:
                h_tr += fix / n
                h_tr = np.clip(h_tr, 0.0, 1.0)
                h_tr += (n / 2.0 - np.sum(h_tr)) / n
                h_tr = np.clip(h_tr, 0.0, 1.0)
            c_tr = compute_c5(h_tr)
            if c_tr < c - 1e-15:
                h = h_tr
                c = c_tr
                if c < best_c5:
                    best_c5 = c
                    best_h = h.copy()

    # Final validation
    best_h = fix_sum_clip(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
