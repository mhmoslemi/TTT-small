```python
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for sweep in range(n_sweeps):
        if t_deadline and time.time() > t_deadline:
            break
        corr = np.correlate(h, 1.0-h, mode='full')
        order = np.random.permutation(n)
        for j in order:
            # Pick a partner m to maintain sum
            m = (j + 1 + np.random.randint(n-1)) % n
            # Compute B_k(j) and B_k(m)
            # B_k(j) = (1-h[j-k])*δ(0≤j-k<n) - h[j+k]*δ(0≤j+k<n)
            # for all k from -(n-1) to (n-1)
            B_j = np.zeros(2*n-1)
            B_m = np.zeros(2*n-1)
            for k_idx in range(2*n-1):
                k = k_idx - (n-1)
                Bj = 0.0
                if 0 <= j-k < n:
                    Bj += (1-h[j-k])
                if 0 <= j+k < n:
                    Bj -= h[j+k]
                Bm = 0.0
                if 0 <= m-k < n:
                    Bm += (1-h[m-k])
                if 0 <= m+k < n:
                    Bm -= h[m+k]
                B_j[k_idx] = Bj
                B_m[k_idx] = Bm
            
            # C₅(δ) = max_k [corr[k] + (B_j[k]-B_m[k])*δ] * dx
            D = B_j - B_m
            # Find δ* minimizing max_k [corr[k] + D[k]*δ]
            # over δ ∈ [δ_lo, δ_hi]
            δ_lo = max(-h[j], h[m]-1)
            δ_hi = min(1-h[j], h[m])
            
            if δ_hi - δ_lo < 1e-12:
                continue
            
            # Binary search for the point where the maximum is minimized
            # The function f(δ) = max_k [c_k + D_k*δ] is convex piecewise linear
            # Its minimum is where the subgradient contains 0
            # Binary search: f'(δ-) <= 0 <= f'(δ+)
            lo, hi = δ_lo, δ_hi
            for _ in range(50):
                mid = (lo + hi) / 2
                vals = corr + D * mid
                mx = vals.max()
                # subgradient at mid: weighted average of D at active positions
                active = vals >= mx - 1e-12
                sg = D[active].mean()
                if sg > 1e-10:
                    hi = mid
                elif sg < -1e-10:
                    lo = mid
                else:
                    break
            δ_star = (lo + hi) / 2
            
            h[j] += δ_star
            h[m] -= δ_star
            h = np.clip(h, 0, 1)
        
        # Project to satisfy sum constraint
        h = project_sr(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5
```

```python
        # First term: (1-h[j+k]) for 0<=j+k<n → j from 0 to n-1-k (if k>=0) or 0 to n-1 (if k<0)
        # Actually: 0<=j+k<n means -k <= j < n-k
        lo1 = max(0, -k)
        hi1 = min(n, n-k)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * one_minus_h[lo1+k:hi1+k]
        # Second term: -h[j-k] for 0<=j-k<n → k <= j < n+k
        lo2 = max(0, k)
        hi2 = min(n, n+k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2-k:hi2-k]
    return grad * dx
```

```python
    h1, c1 = grad_descent(best_h, dx, target, 3000, dl1)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: coordinate descent (exact 1D)
    dl2 = t0 + budget_s * 0.35
    h2, c2 = coord_descent(best_h, dx, target, 500, dl2)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: SLSQP polish
    if (time.time() - t0) < budget_s * 0.45:
        h3 = slsqp_solve(best_h, dx, target, 300)
        c3 = compute_c5(h3, dx)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: more coordinate descent
    dl4 = t0 + budget_s * 0.55
    h4, c4 = coord_descent(best_h, dx, target, 400, dl4)
    if c4 < best_c:
        best_h, best_c = h4, c2 if c4 >= best_c else (h4, c4)
    if c4 < best_c:
        best_h, best_c = h4, c4

    # Phase 5: multi-start perturbations
    dl5 = t0 + budget_s * 0.85
    for r in range(20):
        if time.time() > dl5:
            break
        if r < 10:
            h_r = best_h + 0.05 * np.random.randn(n)
        elif r < 15:
            h_r = (best_h > 0.5).astype(float) + 0.1 * np.random.randn(n)
        else:
            phi = (1.0 + 5.0**0.5) / 2.0
            h_r = (np.arange(n) * phi).astype(float) % 1.0
            h_r = (h_r < 0.5).astype(float)
        h_r = project_sr(h_r, target)
        h_r, c_r = coord_descent(h_r, dx, target, 150, min(dl5, t0 + budget_s - 30))
        if c_r < best_c and (time.time() - t0) < budget_s - 40:
            h_r = slsqp_solve(h_r, dx, target, 150)
            c_r = compute_c5(h_r, dx)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Phase 6: try higher resolution
    for n_new in [67, 89]:
        if (time.time() - t0) > budget_s * 0.88:
            break
        dx_new = 2.0 / n_new
        target_new = n_new / 2.0
        h_up = upsample_h(best_h, n_new)
        h_up = project_sr(h_up, target_new)
        h_up, c_up = coord_descent(h_up, dx_new, target_new, 200,
                                   min(t0 + budget_s - 40, t0 + (budget_s - (time.time() - t0)) * 0.5))
        if (time.time() - t0) < budget_s - 30:
            h_up = slsqp_solve(h_up, dx_new, target_new, 150)
            c_up = compute_c5(h_up, dx_new)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_new

    # Final polish
    if (time.time() - t0) < budget_s - 5:
        dx_f = 2.0 / len(best_h)
        h_f = slsqp_solve(best_h, dx_f, len(best_h) / 2.0, 100)
        c_f = compute_c5(h_f, dx_f)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_sr(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c, len(best_h)
```
