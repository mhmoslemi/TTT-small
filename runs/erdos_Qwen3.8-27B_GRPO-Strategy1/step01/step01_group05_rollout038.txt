    hi = y.max() + 2.0

        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:

        else:

    best = h.copy()

        h = h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_topk(h, n, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    g = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            ka = -k
            g[:n-ka] += (1.0 - h[ka:])
            g[ka:] -= h[:n-ka]
    return g, corr

def subgrad_loop(h, n, iters, alpha0, decay, top_k, t0, budget, margin):
    h = h.copy()
    a = alpha0
    best = h.copy()
    bc = compute_c5(h)
    stall = 0
    for i in range(iters):
        if time.time() - t0 > budget - margin:
            break
        g, _ = subgrad_topk(h, n, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h2 = project_h(h - a * g / gn, n)
        c = compute_c5(h2)
        if c < bc - 1e-13:
            bc = c
            best = h2.copy()
            stall = 0
        else:
            h = h2  # still take the step
            stall += 1
            if stall > 300:
                a = max(0.001, a * 0.5)
                stall = 0
        h = h2
        a *= decay
    return best, bc

def slsqp_lse(h, n, temp, maxiter):
    dx = 2.0 / n
    def f(x):
        c = np.correlate(x, 1.0-x, mode='full')
        s = temp * c
        m = np.max(s)
        return (m + np.log(np.sum(np.exp(s - m)))) / temp * dx
    def jac(x):
        c = np.correlate(x, 1.0-x, mode='full')
        s = temp * c
        m = np.max(s)
        w = np.exp(s - m)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(w)):
            k = ii - (n-1)
            if k >= 0:
                g[k:] += w[ii] * (1.0 - x[:n-k])
                g[:n-k] -= w[ii] * x[k:]
            else:
                ka = -k
                g[:n-ka] += w[ii] * (1.0 - x[ka:])
                g[ka:] -= w[ii] * x[:n-ka]
        return g * temp * dx
    res = minimize(f, h, jac=jac, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':lambda x: np.sum(x)-n/2.0,
                                 'jac':lambda x: np.ones(n)}],
                   options={'maxiter':maxiter,'ftol':1e-14})
    return res.x

    t0 = time.time()

            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def topk_grad(h, n, tk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-tk:]
    g = np.zeros(n)
    for ii in idx:
        k = ii - (n - 1)
        if k >= 0:
            g[k:] += 1.0 - h[:n - k]
            g[:n - k] -= h[k:]
        else:
            a = -k
            g[:n - a] += 1.0 - h[a:]
            g[a:] -= h[:n - a]
    return g

def sg_loop(h, n, it, a0, dec, tk, t0, bud, marg):
    h = h.copy()
    a = a0
    bh = h.copy()
    bc = compute_c5(h)
    stall = 0
    for _ in range(it):
        if time.time() - t0 > bud - marg:
            break
        g = topk_grad(h, n, tk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h2 = project_h(h - a * g / gn, n)
        c = compute_c5(h2)
        if c < bc - 1e-13:
            bc = c
            bh = h2.copy()
            stall = 0
        else:
            stall += 1
            if stall > 600:
                a = max(0.0003, a * 0.4)
                stall = 0
        h = h2
        a *= dec
    return bh, bc

def slsqp_step(h, n, temp, mi):
    dx = 2.0 / n

    def f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        s = temp * c
        m = s.max()
        return (m + np.log(np.sum(np.exp(s - m)))) / temp * dx

    def jf(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        s = temp * c
        m = s.max()
        w = np.exp(s - m)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(w)):
            if w[ii] < 1e-12:
                continue
            k = ii - (n - 1)
            if k >= 0:
                g[k:] += w[ii] * (1.0 - x[:n - k])
                g[:n - k] -= w[ii] * x[k:]
            else:
                a = -k
                g[:n - a] += w[ii] * (1.0 - x[a:])
                g[a:] -= w[ii] * x[:n - a]
        return g * dx

    r = minimize(f, h, jac=jf, method='SLSQP', bounds=[(0, 1)] * n,
                 constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                               'jac': lambda x: np.ones(n)}],
                 options={'maxiter': mi, 'ftol': 1e-14})
    return project_h(r.x, n)
