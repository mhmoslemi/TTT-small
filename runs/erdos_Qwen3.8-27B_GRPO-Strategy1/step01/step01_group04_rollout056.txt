    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / temp
    # Use log-sum-exp for numerical stability
    shifted_max = np.max(shifted)
    sm = cmax + temp * (shifted_max + np.log(np.sum(np.exp(shifted - shifted_max))))

        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        k2 = -k
        g[k2:] += (1.0 - h[:n - k2])
        g[:n - k2] -= h[k2:]
    return g

def project(h, n):
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            mask = proj < 1.0
            if mask.any():
                adj = min(diff, (1.0 - proj[mask]).sum())
                proj[mask] = np.minimum(1.0, proj[mask] + adj / mask.sum())
        else:
            mask = proj > 0.0
            if mask.any():
                adj = max(diff, -(proj[mask]).sum())
                proj[mask] = np.maximum(0.0, proj[mask] + adj / mask.sum())
    return proj

def lgbfgs_opt(h, temps, maxits, deadline, pen=1e5):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cur = h.copy()
    best = compute_c5(cur)
    best_h = cur.copy()
    for temp, maxit in zip(temps, maxits):
        if time.time() > deadline:
            break
        res = minimize(smooth_obj, cur, args=(temp, pen),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10})
        nh = project(np.clip(res.x, 0.0, 1.0), n)
        c5v = compute_c5(nh)
        if c5v < best:
            best = c5v
            best_h = nh.copy()
        cur = nh
    return cur, best_h, best

def subgrad_descent(h, n_steps, lr0, lr1, deadline):
    n = len(h)
    h_cur = h.copy()
    best = compute_c5(h_cur)
    best_h = h_cur.copy()
    H = [h_cur.copy()]
    for step in range(n_steps):
        if time.time() > deadline:
            break
        t = step / max(n_steps, 1)
        lr = lr0 * (lr1 / lr0) ** t
        g = subgrad_active(h_cur)
        h_new = project(h_cur - lr * g, n)
        c5v = compute_c5(h_new)
        if c5v < best:
            best = c5v
            best_h = h_new.copy()
        h_cur = h_new
        H.append(h_cur.copy())
        if len(H) > 200:
            H.pop(0)
    # AhB averaging
    H_arr = np.array(H)
    h_avg = project(H_arr.mean(axis=0), n)
    c5_avg = compute_c5(h_avg)
    if c5_avg < best:
        best = c5_avg
        best_h = h_avg.copy()
    return h_cur, best_h, best

def interp_upscale(h, factor):
    n = len(h)
    m = n * factor
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x_new, x_old, h)

def perturb_shift(h, rng):
    n = len(h)
    k = int(rng.integers(1, max(n // 3, 2)))
    sg = 1 if rng.random() < 0.5 else -1
    return project(np.roll(h, sg * k), n)

def perturb_invert(h):
    return project(1.0 - h, len(h))

def perturb_gauss(h, sigma, rng):
    return project(h + rng.standard_normal(len(h)) * sigma, len(h))

def perturb_mix(h, h2, alpha, rng):
    a = alpha + rng.uniform(-0.1, 0.1)
    a = np.clip(a, 0.15, 0.85)
    return project(a * h + (1 - a) * h2, len(h))

def full_optimize(h, deadline, n_subgrad=3000):
    n = len(h)
    h1, bh1, c1 = lgbfgs_opt(h, [0.03, 0.008, 0.002], [150, 250, 350], deadline)
    h2, bh2, c2 = subgrad_descent(h1, n_subgrad, 0.04, 0.0005, deadline)
    best_h = bh1 if c1 <= c2 else bh2
    best_c = min(c1, c2)
    if time.time() < deadline:
        h3, bh3, c3 = lgbfgs_opt(h2, [0.005, 0.001], [200, 300], deadline)
        if c3 < best_c:
            best_h, best_c = bh3, c3
        h4, bh4, c4 = subgrad_descent(h3, n_subgrad // 2, 0.02, 0.0003, deadline)
        if c4 < best_c:
            best_h, best_c = bh4, c4
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t0 = time.time()
    master_deadline = t0 + budget_s * 0.93

    h_init = np.array(initial_h_values, dtype=float).copy()
    n = len(h_init)
    h_init = project(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at initial resolution
    dl1 = t0 + budget_s * 0.35
    h_opt, c_opt = full_optimize(h_init, dl1, n_subgrad=4000)
    if c_opt < best_c5:
        best_c5, best_h = c_opt, h_opt.copy()

    # Phase 2: Structured perturbations at same resolution
    deadline_p = t0 + budget_s * 0.72
    h_cur = h_opt.copy()
    n_perts = 8
    for pi in range(n_perts):
        if time.time() > deadline_p:
            break
        r = rng.random()
        if r < 0.25:
            h_pert = perturb_gauss(h_cur, 0.08 + 0.05 * pi / n_perts, rng)
        elif r < 0.5:
            h_pert = perturb_shift(h_cur, rng)
        elif r < 0.7:
            h_pert = perturb_invert(h_cur)
        else:
            h_alt = np.roll(h_cur, int(rng.integers(10, n // 4)))
            h_pert = perturb_mix(h_cur, h_alt, 0.5, rng)
        dl_p = min(deadline_p, t0 + time.time() - t0 + budget_s * 0.06)
        h_new, c_new = full_optimize(h_pert, dl_p, n_subgrad=2000)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()
            h_cur = h_new.copy()
        else:
            h_cur = h_new.copy()

    # Phase 3: Upscale and refine
    if time.time() < t0 + budget_s * 0.7:
        n2 = n * 2
        h_up = project(interp_upscale(best_h, 2), n2)
        c_up = compute_c5(h_up)
        if c_up < best_c5 + 0.003:
            dl_up = t0 + budget_s * 0.88
            h_up2, c_up2 = full_optimize(h_up, dl_up, n_subgrad=2500)
            if c_up2 < best_c5:
                best_c5, best_h = c_up2, h_up2.copy()

    # Phase 4: Final polish at best resolution
    if time.time() < t0 + budget_s * 0.9:
        nfin = len(best_h)
        dl_fin = t0 + budget_s * 0.93
        h_fin, bh_fin, c_fin = lgbfgs_opt(best_h, [0.002, 0.0005], [200, 300], dl_fin)
        if c_fin < best_c5:
            best_c5, best_h = c_fin, bh_fin.copy()
        h_fin2, bh_fin2, c_fin2 = subgrad_descent(h_fin, 2000, 0.01, 0.0002, dl_fin)
        if c_fin2 < best_c5:
            best_c5, best_h = c_fin2, bh_fin2.copy()

    # Final validation
    best_h = project(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
