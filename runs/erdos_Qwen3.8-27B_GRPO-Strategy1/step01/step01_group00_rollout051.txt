            grad[lo:hi] += wj * (1.0 - h[lo-k:hi-k])
        # Second term: -wj * h[m+k] for m where 0<=m+k<n
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wj * h[lo2+k:hi2+k]
    return grad * dx

def obj_with_penalty(h, T, P):
    n = len(h)
    target = n * 0.5
    penalty = P * (np.sum(h) - target) ** 2
    return softmax_obj(h, T) + penalty

def grad_with_penalty(h, T, P):
    n = len(h)
    target = n * 0.5
    g = softmax_grad(h, T)
    g += 2 * P * (np.sum(h) - target)
    return g

def project_sum(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Water-filling
    for _ in range(100):
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        nm = np.sum(mask)
        if nm == 0:
            break
        step = diff / nm
        if diff > 0 and step > 0:
            step = min(step, 1.0 - 1e-12 - h[mask].max())
        elif diff < 0 and step < 0:
            step = max(step, -1e-12 + h[mask].min())
        h[mask] += step
        diff = target - np.sum(h)
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
    return np.clip(h, 0.0, 1.0)

def downsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=old_n)
    return project_sum(h_new, new_n)

def upsample(h, new_n):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, T, P):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-12:
        val = c_max * dx
    else:
        shifted = corr - c_max
        weights = np.exp(shifted / T)
        sw = np.sum(weights)
        val = (c_max + T * np.log(sw)) * dx
    return val + P * (np.sum(h) - 0.5 * n) ** 2

def softmax_grad(h, T, P):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ns = len(corr)
    c_max = np.max(corr)
    if T < 1e-12:
        weights = np.zeros(ns)
        mask = np.isclose(corr, c_max)
        weights[mask] = 1.0 / np.sum(mask)
    else:
        shifted = corr - c_max
        weights = np.exp(shifted / T)
        weights /= np.sum(weights)
    grad = np.zeros(n)
    for j in range(ns):
        wj = weights[j]
        if wj < 1e-14:

        lo = max(0, k)
        hi = min(n, n + k)
        if lo < hi:

        hi2 = min(n, n - k)
