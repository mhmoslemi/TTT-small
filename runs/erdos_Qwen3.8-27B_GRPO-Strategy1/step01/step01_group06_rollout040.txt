def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)

        total = room.sum()
        if total < 1e-14:
            return h

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h - room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sensitivity_vec(h, i):
    """Compute s[m] for all m: new_corr[m] ≈ corr[m] + s[m]*(v - h[i])"""
    n = len(h)
    s = np.zeros(2 * n - 1)
    # From h[i] in first factor: m in [i, i+n-1], s[m] += 1-h[i-m+n-1]
    for m in range(i, min(i + n, 2 * n - 1)):
        j = i - m + n - 1
        if 0 <= j < n:
            s[m] += 1.0 - h[j]
    # From h[i] in second factor: m in [n-1-i, 2n-2-i], s[m] -= h[i+m-n+1]
    lo = max(n - 1 - i, 0)
    hi = min(2 * n - 2 - i, 2 * n - 2)
    for m in range(lo, hi + 1):
        k = i + m - n + 1
        if 0 <= k < n:
            s[m] -= h[k]
    return s

def coordinate_step(h, i, corr):
    """Optimize h[i] given all other values fixed. Returns new value of h[i]."""
    n = len(h)
    s = sensitivity_vec(h, i)
    d_lo = -h[i]
    d_hi = 1.0 - h[i]
    
    def f1d(d):
        return np.max(corr + s * d)
    
    # Use minimize_scalar on bounded interval
    try:
        res = minimize_scalar(f1d, bounds=(d_lo, d_hi), method='bounded',
                             options={'xatol': 1e-10})
        d_opt = res.x
    except:
        # Fallback: grid search
        grid = np.linspace(d_lo, d_hi, 50)
        vals = [f1d(d) for d in grid]
        d_opt = grid[np.argmin(vals)]
    
    return h[i] + d_opt

def cooldown_sz(h, i, j):
    """Swap-like operation: transfer mass from i to j or vice versa."""
    n = len(h)
    # Try to move epsilon from i to j
    eps = 0.01
    for e in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
        if h[i] >= e and h[j] <= 1.0 - e:
            h_new = h.copy()
            h_new[i] -= e
            h_new[j] += e
            c = compute_c5(h_new)
            if c < 0.381:
                return h_new, c
    return h, compute_c5(h)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h - room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):

def compute_corr(h):

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h -= room * (diff / total)
    return np.clip(h, 0.0, 1.0)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h -= room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sensitivity_vec(h, i):
    n = len(h)
    s = np.zeros(2 * n - 1)
    lo1 = max(i, 0)
    hi1 = min(i + n, 2 * n - 1)
    if hi1 > lo1:
        m_range = np.arange(lo1, hi1)
        j_vals = i - m_range + n - 1
        valid = (j_vals >= 0) & (j_vals < n)
        s[m_range[valid]] += 1.0 - h[j_vals[valid]]
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * n - 2 - i, 2 * n - 2)
    if hi2 >= lo2:
        m_range2 = np.arange(lo2, hi2 + 1)
        k_vals = i + m_range2 - n + 1
        valid2 = (k_vals >= 0) & (k_vals < n)
        s[m_range2[valid2]] -= h[k_vals[valid2]]
    return s

def coord_optimize(h, i, corr):
    d_lo = -h[i]
    d_hi = 1.0 - h[i]
    if d_hi - d_lo < 1e-12:
        return
    s = sensitivity_vec(h, i)
    gr = 0.6180339887498949
    a, b = d_lo, d_hi
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)
    fc = np.max(corr + s * c_)
    fd = np.max(corr + s * d_)
    best_d = c_ if fc < fd else d_
    best_f = min(fc, fd)
    for _ in range(30):
        if b - a < 1e-11:
            break
        if fc < fd:
            b, d_, fd = d_, c_, fc
            c_ = b - gr * (b - a)
            fc = np.max(corr + s * c_)
        else:
            a, c_, fc = c_, d_, fd
            d_ = a + gr * (b - a)
            fd = np.max(corr + s * d_)
        if fc < best_f:
            best_f, best_d = fc, c_
        if fd < best_f:
            best_f, best_d = fd, d_
    h[i] += best_d
    corr += s * best_d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from parent
    for sweep in range(800):
        if elapsed() > budget_s * 0.50:
            break
        corr = compute_corr(best_h)
        order = np.random.permutation(n)
        for i in order:
            coord_optimize(best_h, i, corr)
        best_h = project_feasible(best_h, n)
        c5_now = compute_c5(best_h)
        if c5_now < best_c5:
            best_c5 = c5_now

    # Phase 2: Simulated annealing with CD refinement
    T_sa = 0.04
    no_improve = 0
    for it in range(8000):
        if elapsed() > budget_s * 0.72:
            break
        h_try = best_h.copy()
        nc = np.random.randint(1, 6)
        for _ in range(nc):
            idx = np.random.randint(n)
            h_try[idx] += np.random.uniform(-0.25, 0.25)
        # Also try a swap
        if np.random.random() < 0.3:
            i1, i2 = np.random.choice(n, 2, replace=False)
            h_try[i1], h_try[i2] = h_try[i2], h_try[i1]
        h_try = project_feasible(h_try, n)
        corr_t = compute_corr(h_try)
        for i in np.random.permutation(n)[:25]:
            coord_optimize(h_try, i, corr_t)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        delta = c5_try - best_c5
        if delta < 0:
            best_h = h_try
            best_c5 = c5_try
            no_improve = 0
        elif np.random.random() < np.exp(-delta / max(T_sa, 1e-12)):
            best_h = h_try
            no_improve += 1
        else:
            no_improve += 1
        T_sa *= 0.9996
        if T_sa < 0.002:
            T_sa = 0.002

    # Phase 3: Try different n values
    for n_try in [20, 30, 40, 50, 80, 100, 120]:
        if elapsed() > budget_s * 0.88:
            break
        if n_try == n:
            continue
        h_try = np.zeros(n_try)
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try += np.random.uniform(-0.1, 0.1, n_try)
        h_try = project_feasible(h_try, n_try)
        for sw in range(80):
            if elapsed() > budget_s * 0.88:
                break
            corr_t = compute_corr(h_try)
            for i in np.random.permutation(n_try):
                coord_optimize(h_try, i, corr_t)
            h_try2 = project_feasible(h_try, n_try)
            c5_t = compute_c5(h_try2)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_try2
                n = n_try
                break
            h_try = h_try2

    # Also try a fresh random binary start at current n
    if elapsed() < budget_s * 0.92:
        n_cur = len(best_h)
        h_rand = np.zeros(n_cur)
        ones = np.random.choice(n_cur, n_cur // 2, replace=False)
        h_rand[ones] = 1.0
        h_rand = project_feasible(h_rand, n_cur)
        for sw in range(100):
            if elapsed() > budget_s * 0.92:
                break
            corr_t = compute_corr(h_rand)
            for i in np.random.permutation(n_cur):
                coord_optimize(h_rand, i, corr_t)
            h_rand = project_feasible(h_rand, n_cur)
        c5_r = compute_c5(h_rand)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_rand

    # Phase 4: Final CD polish
    n = len(best_h)
    for sweep in range(400):
        if elapsed() > budget_s * 0.97:
            break
        corr = compute_corr(best_h)
        for i in np.random.permutation(n):
            coord_optimize(best_h, i, corr)
        best_h = project_feasible(best_h, n)
        c5_now = compute_c5(best_h)
        if c5_now < best_c5:
            best_c5 = c5_now

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
