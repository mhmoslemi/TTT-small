def evaluate_erdos_solution(h):
    """Compute the objective: max correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr_val = np.max(corr)
    return float(max_corr_val * dx)
