import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    u = t * (corr - mx)
    e = np.exp(u)
    return (mx + np.log(np.sum(e)) / t) * dx

def smooth_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    u = t * (corr - mx)
    e = np.exp(u)
    se = np.sum(e)
    w = e / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-10:
            continue
        if s >= 0:
            # (1-h[i-s]) for i in [s, n-1]
            grad[s:] += wk * (1.0 - h[:n-s])
            # -h[i+s] for i in [0, n-1-s]
            grad[:n-s] -= wk * h[s:]
        else:

            grad[:n-m] += wk * (1.0 - h[m:])
            # -h[i-m] for i in [m, n-1]
            grad[m:] -= wk * h[:n-m]
    return dx * grad

import numpy as np
from scipy.optimize import minimize
import time

_CN = 280

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14 * n:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    u = t * (corr - mx)
    e = np.exp(u)
    return (mx + np.log(np.sum(e)) / t) * dx

def smooth_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    u = t * (corr - mx)
    e = np.exp(u)
    se = np.sum(e)
    w = e / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-11:
            continue
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
        else:
            m = -s
            grad[:n - m] += wk * (1.0 - h[m:])
            grad[m:] -= wk * h[:n - m]
    return dx * grad

def _cons_fun(x):
    return np.array([np.sum(x) - _CN / 2.0])

def _cons_jac(x):
    return np.ones((1, len(x)))

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xs = np.arange(n_old) * 2.0 / n_old
    xd = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(xd, xs, h)
    return project(h_new, n_new)

    global _CN
    t0 = time.time()

    h0 = None

    n = len(h0)
    bh = project(h0.copy(), n)
    bc = compute_c5(bh)

    # Phase 1: SLSQP with analytical gradient at original n
    # Use increasing temperature for progressive sharpening
    phase1_end = t0 + 0.60 * budget_s
    temps = [20, 50, 120, 400, 800]
    for ti, temp in enumerate(temps):
        if time.time() > phase1_end - 20:
            break
        n_r = 2 if ti > 2 else 3
        for r in range(n_r):

            if r == 0:

            elif r == 1:
                h_start = project(bh + rng.randn(n) * 0.04, n)
            else:

            # Also try structured perturbation

                xg = np.arange(n) * 2.0 / n
                h_start = project(bh + 0.08 * np.sin(2 * np.pi * 3 * xg), n)

            _CN = n
            bnds = [(0.0, 1.0)] * n
            cons = {'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}
            maxit = 2000 if temp <= 50 else 4000

            try:

                               jac=smooth_grad, method='SLSQP',
                               bounds=bnds, constraints=cons,

                hc = project(np.clip(res.x, 0.0, 1.0), n)

            except Exception:
