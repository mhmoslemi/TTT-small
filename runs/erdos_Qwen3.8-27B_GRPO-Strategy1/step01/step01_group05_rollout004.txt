def smooth_max(vals, temp):
    m = np.max(vals)

    # Clip to avoid overflow
    centered = np.clip(centered, -50, 50)
    return m + temp * np.log(np.sum(np.exp(centered)))

    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, temp) * (2.0 / n)
