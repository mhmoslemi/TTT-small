                    h[i] += add
                    diff -= add
                    if diff < 1e-13:

            else:
                idx = np.argsort(-h)
                for i in idx:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
                    if diff > -1e-13:
                        break
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    return (np.log(np.sum(np.exp((corr - m) / temp))) * temp + m) * dx

def sum_diff(x, n):
    return np.sum(x) - n / 2.0

def de_obj(x, n):
    # For DE: just use the raw max (no smoothness needed)
    h = np.clip(x, 0, 1)
    # Enforce sum constraint approximately
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return compute_c5(h, n)

def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            if diff > 0:
                idx = np.argsort(h)
                rem = diff

                    add = min(rem, 1.0 - h[i])
                    h[i] += add
                    rem -= add
                    if rem < 1e-14:
                        break
            else:
                idx = np.argsort(-h)
                rem = -diff
                for i in idx:
                    sub = min(rem, h[i])
                    h[i] -= sub
                    rem -= sub
                    if rem < 1e-14:
                        break
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h

                   bounds=bounds, constraints=[cons],
                   args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

        return smooth_obj(x, n, temp)

                   options={'maxiter': maxit, 'xatol': 1e-10, 'fatol': 1e-12,
                            'adaptive': True})
    return res.x

One more thought: the correlation has 2n-1 = 89 terms. The maximum is taken over all of them. For the shift 0 term: sum(h[i]*(1-h[i]))*dx = (sum(h) - sum(h^2))*dx. With sum(h) = n/2 = 22.5, this is (22.5 - sum(h^2))*dx. To minimize this, we want to maximize sum(h^2), which means making h as binary as possible (0 or 1). But for shifted terms, we want h[i] and h[i+k] to be correlated (both 0 or both 1) to maximize h[i]*h[i+k] and minimize h[i]*(1-h[i+k]).

This is a tension: k=0 wants h to be binary, while k>0 wants h to be locally constant (so h[i] ≈ h[i+k]). The optimal solution balances these.

For a block structure: if h is constant on blocks of size m, alternating between high and low values, then:
- For k=0: the h*(1-h) term is minimized when h is binary
- For k = m: h[i]*(1-h[i+m]) = 0 (since h[i+m] is the "opposite" value)
- For k = 2m: h[i]*(1-h[i+2m]) = h[i]*(1-h[i]) (back to the k=0 problem)

So a two-block structure with period 2m gives:
- C₅ ≈ max of {0, h*(1-h)*overlap, ...}
