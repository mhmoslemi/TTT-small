    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h = h.copy()
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_lag_grad(h, k):
    N = len(h)
    g = np.zeros(N)
    ak = abs(k)
    if ak == 0:
        return 1.0 - 2.0 * h
    if k >= 0:
        g[ak:] += (1.0 - h[:N-ak])
        g[:N-ak] -= h[ak:]
    else:
        g[:N-ak] += (1.0 - h[ak:])
        g[ak:] -= h[:N-ak]

    return (mx + np.log(np.sum(np.exp(s))) / t) * dx

def proj(h, n):
