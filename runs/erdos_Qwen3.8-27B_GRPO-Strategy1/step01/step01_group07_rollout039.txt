import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def corr_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def obj_smooth(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    scaled = np.clip((corr - m) / beta, -50.0, 0.0)
    return dx * (m + beta * np.log(np.sum(np.exp(scaled))))

def fix_sum(h, n):
    h = np.asarray(h, dtype=np.float64)
    h = h + (n / 2.0 - np.sum(h)) / float(n)
    return np.clip(h, 0.0, 1.0)

def resample(arr, new_n):
    old_n = len(arr)
    if new_n == old_n:
        return arr.copy()
    ox = np.arange(old_n, dtype=np.float64)
    nx = (np.arange(new_n) * old_n) / float(new_n)
    v = np.interp(nx, ox, arr)
    return fix_sum(v, new_n)

_N = 280

def eq_fun(x):
    return np.sum(x) - _N / 2.0

    return np.ones(_N)

def grad_lag(h, n, k):
    shift = k - (n - 1)
    if shift == 0:
        return 1.0 - 2.0 * h
    s = abs(shift)
    grad = np.zeros(n)
    if shift > 0:
        for j in range(s, n):
            grad[j] += (1.0 - h[j - s])
        for j in range(n - s):
            grad[j] -= h[j + s]
    else:
        for j in range(n - s):
            grad[j] += (1.0 - h[j + s])
        for j in range(s, n):
            grad[j] -= h[j - s]
    return grad

def subgrad_hunt(h, n, rng, max_iters):
    h = h.copy()
    best = compute_c5(h)
    lr = 0.08
    stall = 0
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top = np.argsort(corr)[-4:]
        grad = np.zeros(n)
        for k in top:
            grad += grad_lag(h, n, k)
        grad /= float(len(top))
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c = compute_c5(h_new)
        if c < best - 1e-16:
            h = h_new
            best = c
            lr = min(lr * 1.03, 0.6)
            stall = 0
        else:
            stall += 1
            lr *= 0.9
            if stall > 200:
                break
    return h, best

def targeted_cd(h, n, rng, max_iters):
    h = h.copy()
    best = compute_c5(h)
    deltas = [0.005, 0.01, 0.015, 0.02, 0.025, 0.03, 0.04, 0.003]
    for it in range(max_iters):
