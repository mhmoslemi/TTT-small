def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_scrambled(n, m):
    idx = np.arange(n)
    return ((idx * m) % n < n // 2).astype(float)

def to_binary_balanced(h, n):
    idx = np.argsort(-h)[:n // 2]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def greedy_all_swaps(h, deadline):
    n = len(h)
    dx = 2.0 / n
    best_val = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_d = 1e10
        best_ij = None
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                v = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                h[i] = 1; h[j] = 0

                if d < best_d:

        h[i] = 0; h[j] = 1
