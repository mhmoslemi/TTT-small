    best = compute_c5(h, n)
    epsilons = [0.002, 0.01, 0.03, 0.08]
    for p in range(max_passes):
        improved = False
        for i in range(n):
            for j in [(i + 1) % n, (i + 2) % n, (i + n // 3) % n, (i + n // 2) % n]:
                if i == j:
                    continue
                for eps in epsilons:
                    for sign in (1, -1):
                        h[i] += sign * eps
                        h[j] -= sign * eps
                        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                            h[i] -= sign * eps
                            h[j] += sign * eps
                            continue
                        c = compute_c5(h, n)
                        if c < best - 1e-12:
                            best = c
                            improved = True
                        else:
                            h[i] -= sign * eps
                            h[j] += sign * eps
        if not improved:
            break
    return h, best


def slsqq_optimize(h, n, temp, maxiter, bnds, cns):
    try:
        r = minimize(smooth_max_c5, h, method='SLSQP', bounds=bnds,
                     constraints=cns, args=(n, temp),
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(r.x, n)
    except Exception:
        return h.copy()


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(45, 0.5)
    n0 = len(h_init)
    h_init = repair(h_init, n0)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, n0)
    best_n = n0

    n_list = sorted(set([n0, 40, 44, 46, 48]))

    for n in n_list:
        if time.time() - t0 > budget_s - 90:
            break

        if n == n0:
            h = h_init.copy()
        else:
            xo = np.linspace(0, 2, n0)
            xn = np.linspace(0, 2, n)
            h = repair(np.interp(xn, xo, h_init), n)

        bnds = [(0.0, 1.0)] * n
        cns = [{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}]

        # SLSQP cooling schedule
        for temp in [80, 30, 12, 5, 2.5]:
            if time.time() - t0 > budget_s - 40:
                break
            rem = budget_s - 90 - (time.time() - t0)
            if rem < 8:
                break
            mi = int(min(400, max(80, rem * 2.5)))
            h = slsqq_optimize(h, n, temp, mi, bnds, cns)
            c = compute_c5(h, n)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n

        # Powell derivative-free
        if time.time() - t0 > budget_s - 35:
            break
        rem = budget_s - 90 - (time.time() - t0)
        if rem > 15:
            try:
                r = minimize(smooth_max_c5, h, method='Powell', bounds=bnds,
                             args=(n, 25), options={'maxiter': 120, 'ftol': 1e-12})
                hn = repair(r.x, n)
                c = compute_c5(hn, n)
                if c < best_c5:
                    best_c5, best_h, best_n = c, hn.copy(), n
            except Exception:
                pass

        # Coordinate refinement
        if time.time() - t0 > budget_s - 20:
            break
        h, c = coord_refine(h, n, max_passes=12)
        if c < best_c5:
            best_c5, best_h, best_n = c, h.copy(), n

    # Random restarts on best_n
    if time.time() - t0 < budget_s - 25:
        n = best_n
        h = best_h.copy()
        bnds = [(0.0, 1.0)] * n
        cns = [{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}]

        for trial in range(25):
            if time.time() - t0 > budget_s - 12:
                break
            scale = 0.02 + 0.05 * trial / 25.0
            cand = h + np.random.uniform(-scale, scale, n)
            cand = repair(cand, n)
            for temp in (8, 30):
                if time.time() - t0 > budget_s - 10:
                    break
                mi = int(min(150, max(30, (budget_s - 12 - (time.time() - t0)) * 2)))
                cand = slsqq_optimize(cand, n, temp, mi, bnds, cns)
                c = compute_c5(cand, n)
                if c < best_c5:
                    best_c5, best_h = c, cand.copy()
            if time.time() - t0 > budget_s - 8:
                break
            cand2, c2 = coord_refine(cand, n, max_passes=5)
            if c2 < best_c5:
                best_c5, best_h = c2, cand2.copy()

    # Final coordinate polish
    if time.time() - t0 < budget_s - 3:
        best_h, best_c5 = coord_refine(best_h, best_n, max_passes=10)

    return best_h, compute_c5(best_h, best_n), best_n
