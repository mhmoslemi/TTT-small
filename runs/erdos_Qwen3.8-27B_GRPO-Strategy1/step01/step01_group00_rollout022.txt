import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = h.copy()
    for _ in range(3):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_weights(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if T < 1e-14:
        tol = max(1e-8 * c_max, 1e-12)
        weights = (np.abs(corr - c_max) <= tol).astype(float)
        s = np.sum(weights)
        if s < 1e-30:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= s
        val = c_max
    else:
        shifted = corr - c_max
        ew = np.exp(shifted / T)
        tot = np.sum(ew)
        weights = ew / tot
        val = c_max + T * np.log(tot)
    return weights, val

def compute_grad(h, weights, n):
    grad = np.zeros(n)
    n_full = len(weights)
    for k in range(n_full):
        wk = weights[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += wk * (1.0 - h[lag:])
                grad[lag:] += wk * (-h[:n - lag])
        else:
            m = -lag
            if m < n:
                grad[m:] += wk * (1.0 - h[:n - m])
                grad[:n - m] += wk * (-h[m:])
    return grad

def make_obj_grad(T, n, mu):
    target = n * 0.5
    dx = 2.0 / n
    def obj(x):
        _, val = corr_weights(x, T, n)
        return val * dx + mu * (np.sum(x) - target) ** 2
    def grad(x):
        w, _ = corr_weights(x, T, n)
        g = dx * compute_grad(x, w, n)
        g += 2.0 * mu * (np.sum(x) - target) / n
        return g
    return obj, grad

def alt_init_1(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return fix_sum(h, n * 0.5)

def alt_init_2(n):
    x = np.arange(n)
    h = 0.5 + 0.4 * np.cos(2.0 * np.pi * x / n)
    h = np.clip(h, 0.0, 1.0)
    return fix_sum(h, n * 0.5)

def alt_init_3(n):

        end = (blk + 1) * n // 4
        h[start:end] = 1.0 if blk % 2 == 0 else 0.0
    return fix_sum(h, n * 0.5)
