        return x[n]

        return np.sum(x[:n]) - target_sum

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sr(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def slsqp_full(h0, dx, target_sum, maxiter=500):
    n = len(h0)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    
    def obj(x):
        return x[n]
    def ineq(x):
        c = np.correlate(x[:n], 1.0 - x[:n], mode='full')
        return x[n] - c * dx
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    
    t0 = float(np.max(corr0))
    x0 = np.concatenate([h0.copy(), [t0 + 0.002]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_sr(res.x[:n], target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

        pass
    return best_h, best_c5

            break
        w /= ws
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            for k in range(2 * n - 1):
                d = k - (n - 1)
                if d >= 0:
                    z = 0.0
                    if j >= d:
                        z += (1.0 - h[j - d])
                    if j + d < n:
                        z -= h[j + d]
                else:
                    e = -d
                    z = 0.0
                    if j + e < n:
                        z += (1.0 - h[j + e])
                    if j >= e:
                        z -= h[j - e]
                g += w[k] * z
            grad[j] = g * dx
        mom = 0.65 * mom + 0.35 * grad
        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.03]:
            h_new = project_sr(h - lr * shrink * mom, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
    return best_h, best_c5

def local_search(h0, dx, target_sum, n_trials=200, deadline=None):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    rng = np.random.RandomState(12345)
    for t in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        n_pert = rng.randint(1, 4)
        idx = rng.choice(n, n_pert, replace=False)
        h_new = best_h.copy()
        for i in idx:
            h_new[i] += rng.uniform(-0.08, 0.08)
        h_new = project_sr(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            h_pol, c5_pol = slsqp_solve(h_new, dx, target_sum, 150, top_k=12)
            if c5_pol < best_c5:
                best_h, best_c5 = h_pol, c5_pol
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def remaining(margin=5):
        return budget_s - (time.time() - t_start) - margin

    try:
        h_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h_raw = np.zeros(47)
        h_raw[:23] = 1.0
        h_raw[23] = 0.5

    n = len(h_raw)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_sr(h_raw, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Full SLSQP
    if remaining(100) > 30:
        h1, c1 = slsqp_solve(h0, dx, target_sum, maxiter=500)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Projected gradient with momentum
    if remaining(80) > 60:
        dl = t_start + budget_s - 80
        h2, c2 = proj_grad(best_h, dx, target_sum, 5000, dl)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Top-k constraint generation (3 rounds)
    for round_i in range(3):
        if remaining(40) < 20:
            break
        h3, c3 = slsqp_solve(best_h, dx, target_sum, maxiter=250, top_k=15)
        if c3 < best_c5:
            best_h, best_c5 = h3, c1 if False else c3

    # Phase 4: Another gradient pass
    if remaining(60) > 40:
        dl = t_start + budget_s - 60
        h4, c4 = proj_grad(best_h, dx, target_sum, 4000, dl)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Local perturbation search
    if remaining(60) > 30:
        dl = t_start + budget_s - 60
        h5, c5 = local_search(best_h, dx, target_sum, 300, dl)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Phase 6: Final SLSQP polish (full constraints)
    if remaining(10) > 15:
        h6, c6 = slsqp_solve(best_h, dx, target_sum, maxiter=400)
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    # Phase 7: Try structured alternative starts
    if remaining(200) > 100:
        candidates = []
        # Alternating pattern
        h_alt = np.tile([0.6, 0.4], n // 2)
        if n % 2 == 1:
            h_alt = np.append(h_alt, 0.5)
        candidates.append(project_sr(h_alt, target_sum))
        # Block pattern: 3-on 2-off
        h_blk = np.zeros(n)
        for i in range(0, n, 5):
            h_blk[i:i+3] = 1.0
        candidates.append(project_sr(h_blk, target_sum))
        # Smooth ramp
        h_ramp = np.linspace(0.2, 0.8, n)
        candidates.append(project_sr(h_ramp, target_sum))

        for hc in candidates:
            if remaining(80) < 30:
                break
            hc_pol, hc_c5 = slsqp_solve(hc, dx, target_sum, 200, top_k=15)
            if hc_c5 < best_c5 - 0.001 and remaining(80) > 40:
                dl = t_start + budget_s - 80
                hc_pol2, hc_c52 = proj_grad(hc_pol, dx, target_sum, 2000, dl)
                if hc_c52 < best_c5:
                    best_h, best_c5 = hc_pol2, hc_c52

    # Final polish
    if remaining(5) > 10:
        hf, cf = slsqp_solve(best_h, dx, target_sum, maxiter=300, top_k=10)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    best_h = project_sr(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
