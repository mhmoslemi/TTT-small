        if time.time() - t0 > budget_s * 0.12:
            break
        cur = l1_opt(cur, temp, pen, mi)
        cur = repair(cur)
        cv = compute_c5(cur)
        if cv < best_c5:
            best_c5 = cv
            best_h = cur.copy()

    # Phase 2: Upscale to 720, optimize
    if time.time() - t0 < budget_s * 0.30:
        cur2 = interp_up(cur, 2)
        cur2 = repair(cur2)
        for temp, pen, mi in [(0.015, 200, 180), (0.003, 1000, 120), (0.0008, 5000, 100)]:
            if time.time() - t0 > budget_s * 0.32:
                break
            cur2 = l1_opt(cur2, temp, pen, mi)
            cur2 = repair(cur2)
            cv = compute_c5(cur2)
            if cv < best_c5:
                best_c5 = cv
                best_h = cur2.copy()

    # Phase 3: Alternate initializations at n=360
    if time.time() - t0 < budget_s * 0.42:
        rng2 = np.random.RandomState(seed + 100)
        alts = [
            repair(block_init(n, 3)),
            repair(block_init(n, 5)),
            repair(block_init(n, 8)),
            rand_bin_init(n, rng2),
            rand_bin_init(n, rng2),
        ]
        for alt in alts:
            if time.time() - t0 > budget_s * 0.42:
                break
            cv = compute_c5(alt)
            if cv < best_c5:
                best_c5 = cv
                best_h = alt.copy()
            ca = l1_opt(alt, 0.01, 300, 80)
            ca = repair(ca)
            cv = compute_c5(ca)
            if cv < best_c5:
                best_c5 = cv
                best_h = ca.copy()

    # Phase 4: Simulated annealing (bulk of remaining time)
    if time.time() - t0 < budget_s * 0.88:
        t_end_anneal = t0 + budget_s * 0.88
        best_h, best_c5 = anneal(best_h, t0, t_end_anneal)

    # Phase 5: Final L-BFGS-B polish
    if time.time() - t0 < budget_s * 0.96:
        for temp, pen, mi in [(0.002, 3000, 100), (0.0005, 10000, 80)]:
            if time.time() - t0 > budget_s * 0.97:
                break
            cf = l1_opt(best_h, temp, pen, mi)
            cf = repair(cf)
            cv = compute_c5(cf)
            if cv < best_c5:
                best_c5 = cv
                best_h = cf.copy()

    # Final repair to ensure sum(h) ≈ n/2
    best_h = repair(best_h)
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, len(best_h)
