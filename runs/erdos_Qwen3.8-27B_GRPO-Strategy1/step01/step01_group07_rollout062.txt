= Σ_{m: 0≤j-m<n} p(m)*(1-h[j-m])
= (1 - something) ...

Actually let r = j - m, so m = j - r. The condition 0≤j-m<n becomes 0≤r<n.
Σ_{r=0}^{n-1} p(j-r) * (1-h[r])

The second sum: Σ_m p(m)*I[0≤j+m<n]*h[j+m]
Let s = j+m, so m = s-j. Condition 0≤s<n.
Σ_{s=0}^{n-1} p(s-j) * h[s]

This is a convolution! 
First term: convolution of p with (1-h), evaluated at position j
Second term: correlation of p with h, evaluated at position j

First: (p * (1-h))[j] where (p * f)[j] = Σ_r p[j-r] * f[r]

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def fix_sum(h, n):
    for _ in range(10):
        h = h + (n / 2.0 - np.sum(h)) / n
        h = np.clip(h, 0.0, 1.0)
        if abs(np.sum(h) - n / 2.0) < 1e-10:
            break
    return h


def upsample(parent, new_n):
    old_n = len(parent)
    v = np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, old_n), parent)
    return fix_sum(v, new_n)


def grad_corr_single(h, m, dx):
    n = len(h)
    out = np.zeros(n)
    if m >= 0:
        r = n - m
        if r > 0:
            out[m:] = 1.0 - h[:r]
            out[:r] -= h[m:m + r]
    else:
        p = -m
        r = n - p
        if r > 0:
            out[:r] = 1.0 - h[p:]
            out[p:] -= h[:r]
    return dx * out


def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = corr.max()
    s = (corr - c_max) / beta
    e = np.exp(s)
    return dx * (c_max + beta * np.log(e.sum()))


def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = corr.max()
    s = (corr - c_max) / beta

        m = k - (n - 1)
        grad += p[k] * grad_corr_single(x, m, dx)
    return grad
