```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_max_corr(h):
    """Compute gradient of max_k c_k * dx where c_k = correlate(h,1-h,'full')[k]"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    k_max = np.argmax(c)
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # c = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
        # grad[j] = [j < n-lag] * (-h[j+lag]) + [j >= lag] * (1-h[j-lag])
        g[:n-lag] += -h[lag:]
        g[lag:] += 1.0 - h[:n-lag]
    else:
        lp = -lag
        # c = sum_{i=lp}^{n-1} h[i-lp] * (1-h[i])
        # grad[j] = [j < n-lp] * (1-h[j+lp]) + [j >= lp] * (-h[j-lp])
        g[:n-lp] += 1.0 - h[lp:]
        g[lp:] += -h[:n-lp]
    return g * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_subgradient(h0, max_iters=5000, tol=1e-10):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    
    step = 0.05
    for it in range(max_iters):
        c5_vals = np.correlate(h, 1.0-h, mode='full') * dx
        c5 = np.max(c5_vals)
        if c5 < best_c - tol:
            best_c = c5
            best_h = h.copy()
        
        g = grad_max_corr(h)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        # Backtracking line search
        alpha = step
        for _ in range(30):
            h_new = project_feasible(h - alpha * g, n)
            if compute_c5(h_new) <= c5 + 1e-12:
                break
            alpha *= 0.5
        else:
            break
        
        h = h_new
        step = alpha * 1.2 if alpha > step * 0.5 else alpha * 0.7
    
    return best_h, best_c
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-15:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    s = t * (corr - mx)
    s = np.clip(s, -50.0, 50.0)
    return mx + np.log(np.sum(np.exp(s))) / t

def ceq(x):
    return np.sum(x) - len(x) / 2.0

def ceqj(x):
