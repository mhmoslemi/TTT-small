                break
            if b[idx] == 1:
                b[idx] = 0
                deficit += 1

        for i in range(n):

                trial[i] = 0
                # Find a 0 to flip to 1

                while trial[j] == 1:

            if c < best_c - 1e-15:

                best_c = c
                improved = True
                break
        if not improved:

        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(soft_max(corr, t) * (2.0 / n))

    """Projected gradient descent with BB step"""
    n = len(h)
    h = h.copy()
    prev_h = None
    prev_g = None

    for it in range(max_iter):

        g = soft_max_grad(corr, h, t)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        if prev_h is not None:
            sh = h - prev_h
            sy = g - prev_g
            ys = np.dot(sh, sy)
            ss = np.dot(sh, sh)
            yy = np.dot(sy, sy)
            if yy > 1e-16 and ys > 1e-14:
                alpha = ss / ys
            elif ss > 1e-14:
                alpha = ss / (gnorm * 0.1)
            else:
                alpha = lr_init or 0.01
        else:
            alpha = (lr_init if lr_init else 0.5) / max(gnorm, 1e-8)
        alpha = min(max(alpha, 1e-12), 5.0)
        
        h_new = project(h - alpha * g, n)
        f_new = obj_h(h_new, t)
        
        if f_new > f + 1e-12:
            # Backtracking
            for _ in range(30):
                alpha *= 0.5
                h_new = project(h - alpha * g, n)
                f_new = obj_h(h_new, t)
                if f_new <= f + 1e-12:
                    break

                prev_h = h.copy()
                prev_g = g.copy()
                continue
        
        h = h_new
        f = f_new
        prev_h = h.copy()
        prev_g = g.copy()

    return np.ones((1, len(x)))

def grad_pgd(h, t, max_iter, lr_init=None):
    n = len(h)
    h = h.copy()
    prev_h = None
    prev_g = None
    f = obj_h(h, t)
    alpha = lr_init if lr_init else 0.05
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = soft_max_grad(corr, h, t)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        if prev_h is not None:
            sh = h - prev_h
            sy = g - prev_g
            ss = np.dot(sh, sh)
            sysh = np.dot(sh, sy)
            if sysh > 1e-14:
                alpha = ss / sysh
                alpha = min(max(alpha, 1e-12), 2.0)
        else:
            alpha = 0.5 / max(gnorm, 1e-8)
        
        h_new = project(h - alpha * g, n)
        f_new = obj_h(h_new, t)
        if f_new > f + 1e-13:
            for _ in range(25):
                alpha *= 0.5
                h_new = project(h - alpha * g, n)
                f_new = obj_h(h_new, t)
                if f_new <= f + 1e-13:
                    break
            else:
                prev_h = h.copy()
                prev_g = g.copy()
                continue
        h = h_new
        f = f_new
        prev_h = h.copy()
        prev_g = g.copy()
    return h, f

def xinterp(h, n_new):
    xs = np.arange(len(h)) * 2.0 / len(h)
    xd = np.arange(n_new) * 2.0 / n_new
    return project(np.interp(xd, xs, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = project(np.full(280, 0.5) + rng.randn(280) * 0.1, 280)

    n0 = len(h0)
    h0 = project(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0)

    # Phase 1: coarse SLSQP at n=70
    if tl() > 150:
        h70 = xinterp(h0, 70)
        bnds70 = [(0.0, 1.0)] * 70
        cons70 = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
        for tv in [50.0, 200.0, 800.0]:
            if tl() < 40:
                break
            _current_t[0] = tv
            try:
                r = minimize(slsqp_fj, h70, jac=True, method='SLSQP',
                             bounds=bnds70, constraints=cons70,
                             options={'maxiter': 500, 'ftol': 1e-14})
                hp = project(r.x, 70)
                hu = xinterp(hp, n0)
                c = compute_c5(hu)
                if c < best_c:
                    best_c = c
                    best_h = hu.copy()
                    h70 = xinterp(best_h, 70)
            except Exception:
                pass

    # Phase 2: PGD refinement at n0 with increasing t
    if tl() > 80:
        for tv in [200.0, 1000.0, 5000.0]:
            if tl() < 30:
                break
            h_try, _ = grad_pgd(best_h, tv, max_iter=3000, lr_init=0.02)
            c = compute_c5(h_try)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h_try.copy()

    # Phase 3: SLSQP high-t polish
    if tl() > 50:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': ceq, 'jac': ceqj}
        for tv in [2000.0, 10000.0]:
            if tl() < 20:
                break
            _current_t[0] = tv
            try:
                r = minimize(slsqp_fj, best_h, jac=True, method='SLSQP',
                             bounds=bnds, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-16})
                hp = project(r.x, n)
                c = compute_c5(hp)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = hp.copy()
            except Exception:
                pass

    # Phase 4: PGD ultra-high t
    if tl() > 40:
        for tv in [10000.0, 50000.0]:
            if tl() < 15:
                break
            h_try, _ = grad_pgd(best_h, tv, max_iter=8000, lr_init=0.005)
            c = compute_c5(h_try)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h_try.copy()

    # Phase 5: perturbed restarts
    if tl() > 80:
        n = len(best_h)
        for amp in [0.003, 0.01, 0.02]:
            for _ in range(4):
                if tl() < 25:
                    break
                h_p = project(best_h + rng.randn(n) * amp, n)
                h_try, _ = grad_pgd(h_p, 500.0, max_iter=1500, lr_init=0.01)
                c = compute_c5(h_try)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = h_try.copy()

    # Phase 6: try higher resolution
    if tl() > 150 and len(best_h) <= 280:
        n2 = len(best_h) * 2
        h_up = xinterp(best_h, n2)
        h_try, _ = grad_pgd(h_up, 300.0, max_iter=2000, lr_init=0.01)
        c1 = compute_c5(h_try)
        if c1 < best_c - 1e-12 and tl() > 60:
            h_try, _ = grad_pgd(h_try, 3000.0, max_iter=4000, lr_init=0.005)
            c1 = compute_c5(h_try)
        if c1 < best_c - 1e-12:
            best_c = c1
            best_h = h_try.copy()

    # Final PGD polish
    if tl() > 20:
        h_try, _ = grad_pgd(best_h, 20000.0, max_iter=10000, lr_init=0.002)
        c = compute_c5(h_try)
        if c < best_c:
            best_c = c
            best_h = h_try.copy()

    best_h = project(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
