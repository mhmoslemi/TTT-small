import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def obj_with_penalty(h, T, n):
    s = h.sum() - n / 2.0
    pen = 1000.0 * s * s
    clip_pen = 500.0 * (np.sum(np.maximum(0.0, h - 1.0))**2 + np.sum(np.maximum(0.0, -h))**2)
    return smooth_c5(h, T) + pen + clip_pen

def obj_c5_penalty(h, n):
    s = h.sum() - n / 2.0
    pen = 500.0 * abs(s)
    clip_pen = 200.0 * (np.sum(np.maximum(0.0, h - 1.0)) + np.sum(np.maximum(0.0, -h)))
    return compute_c5(h) + pen + clip_pen

def project_feasible(h, n):

    target = n / 2.0

        s = h.sum()

            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += room * (diff / tr)
        else:
            tv = h.sum()
            if tv < 1e-14:
                break
            h += h * (diff / tv)
        np.clip(h, 0.0, 1.0, out=h)

    )
