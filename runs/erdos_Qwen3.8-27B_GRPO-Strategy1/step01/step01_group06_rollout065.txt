    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = np.minimum(shifted, 500)  # prevent overflow
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

            return h
        frac = diff / total
        h = h + room * frac
    else:
        total = h.sum()
        if total < 1e-15:

        frac = -diff / total

import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def smooth_c5(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    t = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(t))) / T) * 2.0 / n

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s > 1e-15:
            h += room * diff / s
    else:
        s = h.sum()
        if s > 1e-15:
            h *= 1.0 + diff / s
    return np.clip(h, 0.0, 1.0)

def _sum_eq(x):
    return x.sum() - len(x) / 2.0

def _slsqp(h0, T, n, mi=500, ft=1e-14):
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': _sum_eq}]
    try:
        r = minimize(smooth_c5, h0, args=(T,), method='SLSQP',
                     bounds=b, constraints=c, options={'maxiter': mi, 'ftol': ft})
        h = project_feasible(np.clip(r.x, 0, 1), n)
        return h, compute_c5(h)
    except Exception:
        return h0.copy(), compute_c5(h0)

def _lbfgsb(h0, T, n, mi=3000):
    def obj(x):
        return smooth_c5(x, T) + 5e4 * (x.sum() - n / 2.0) ** 2
    b = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h0, method='L-BFGS-B', bounds=b,
                     options={'maxiter': mi, 'gtol': 1e-13})
        h = project_feasible(np.clip(r.x, 0, 1), n)
        return h, compute_c5(h)
    except Exception:
        return h0.copy(), compute_c5(h0)

    best = h.copy()
    bc = compute_c5(best)
    steps = [0.06, 0.03, 0.015, 0.008, 0.004]

                    v = best[i] + s * d
                    if v < 0 or v > 1:
                        continue
                    h2 = best.copy()

                    c = compute_c5(h2)
                    if c < bc - 1e-14:
                        best, bc = h2, c
                        imp = True

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h.copy(), 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s > 1e-15:
            h += room * diff / s
    else:
        s = h.sum()
        if s > 1e-15:
            h *= 1.0 + diff / s
    return np.clip(h, 0.0, 1.0)
