def softmax_gradient(h, dx, beta):

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)

    #   - h[j-k] if j-k in [0,n-1]       (h[j] is the second factor, i.e., h[i+k]=h[j] means i=j-k)
    for k in range(-(n-1), n):

            if 0 <= j+k < n:
                grad[j] += wk * (1.0 - h[j+k])
            if 0 <= j-k < n:
                grad[j] -= wk * h[j-k]
    return grad

def compute_c5(h, dx):

def project_simplex(h, target_sum):

    w = np.exp(beta * (corr - m))

def slsqp_solve(h0, dx, target_sum, maxiter=400):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eqf(x):
        return np.sum(x[:n]) - target_sum
    t0 = compute_c5(h0, dx) / dx
    x0 = np.concatenate([h0, [t0 + 0.01]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqf}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_simplex(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

                best_n = n

    # Phase 6: Try slightly different n values with refined search
    for n_try in [45, 49, 51, 53]:
        if not time_ok(50):
            break
        if n_try == best_n:
            continue
        dx_t = 2.0 / n_try
        h_t = make_structured(n_try, n_try % 5)
        h_t = slsqp_solve(h_t, dx_t, n_try / 2.0, 300)
        c5_t = compute_c5(h_t, dx_t)
        if c5_t < best_c5 + 0.005 and time_ok(40):
            h_t2, c5_t2 = proj_grad_opt(h_t, dx_t, n_try / 2.0, 3000, t0 + budget_s - 40)
            c5_t = min(c5_t, c5_t2)
            h_t = h_t2 if c5_t2 < c5_t else h_t
            h_t = slsqp_solve(h_t, dx_t, n_try / 2.0, 200)
            c5_t = compute_c5(h_t, dx_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_try

    # Final polish
    if best_h is not None and time_ok(5):
        n = best_n
        dx = 2.0 / n
        h_f = slsqp_solve(best_h, dx, n / 2.0, 200)
        c5_f = compute_c5(h_f, dx)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    best_h = project_simplex(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n / 2.0 * best_h - best_h * 0 + best_h * 0 + best_h * 0 + best_h * 0 + dx * 0 + np.correlate(best_h, 1-best_h, mode='full').max() * (2.0/best_n) * 0 + compute_c5(best_h, 2.0/best_n))
    # Simpler final computation
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    if best_h is None:
        best_h = project_simplex(make_structured(47, 0), 47 / 2.0)
        best_c5 = compute_c5(best_h, 2.0 / 47)
        best_n = 47
    return best_h, best_c5, best_n
