            if i == j:
                continue
            h_bin[i] = 0
            h_bin[j] = 1
            c5_try = compute_c5(h_bin)
            if c5_try < c5_best - 1e-12:
                c5_best = c5_try
                h_best = h_bin.copy()
                improved = True
            else:
                h_bin[i] = 1
                h_bin[j] = 0
        if not improved and sweep > 15:
            break

    # Phase 3: SLSQP polish starting from best
    if elapsed() < budget_s - 150:
        h_start = h_best.copy()
        T_sl = 0.003
        def obj_sl(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cm = corr.max()
            sh = corr - cm
            return (cm + T_sl * np.log(np.sum(np.exp(sh / T_sl)))) * (2.0 / len(x))
        def con_sl(x):
            return np.sum(x) - len(x) * 0.5
        res = opt.minimize(obj_sl, h_start, method='SLSQP',
                           bounds=[(0, 1)] * n_best,
                           constraints=[{'type': 'eq', 'fun': con_sl}],
                           options={'maxiter': 500, 'ftol': 1e-14})
        h_c = project_capped_simplex(res.x, n_best * 0.5)
        c5_c = compute_c5(h_c)
        if c5_c < c5_best:
            h_best = h_c
            c5_best = c5_c

    # Phase 4: Try nearby resolutions
    if elapsed() < budget_s - 120:
        for n_try in [n_best - 20, n_best + 20, n_best - 40, n_best + 40]:
            if elapsed() > budget_s - 100:
                break
            if n_try < 100 or n_try > 1000:
                continue
            x_old = np.arange(n_best) * (2.0 / n_best)
            x_new = np.arange(n_try) * (2.0 / n_try)
            h_try = np.interp(x_new, x_old, h_best, period=2.0)
            h_try = project_capped_simplex(h_try, n_try * 0.5)
            T_sl2 = 0.002
            def obj_sl2(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                cm = corr.max()
                sh = corr - cm
                return (cm + T_sl2 * np.log(np.sum(np.exp(sh / T_sl2)))) * (2.0 / len(x))
            def con_sl2(x):
                return np.sum(x) - len(x) * 0.5
            res2 = opt.minimize(obj_sl2, h_try, method='SLSQP',
                                bounds=[(0, 1)] * n_try,
                                constraints=[{'type': 'eq', 'fun': con_sl2}],
                                options={'maxiter': 400, 'ftol': 1e-14})
            h_c2 = project_capped_simplex(res2.x, n_try * 0.5)
            c5_c2 = compute_c5(h_c2)
            if c5_c2 < c5_best:
                h_best = h_c2
                c5_best = c5_c2
                n_best = n_try

    # Phase 5: Final projected gradient refinement
    if elapsed() < budget_s - 30:
        h_curr2 = h_best.copy()
        v2 = np.zeros(len(h_curr2))
        step2 = 0.01
        for T_f in [0.001, 0.0005]:
            for it in range(2000):
                if elapsed() > budget_s - 10:
                    break
                h_new2, v2, _ = prox_grad_step(h_curr2, v2, T_f, step2, 0.8)
                c5_new2 = compute_c5(h_new2)
                if c5_new2 < c5_best:
                    h_best = h_new2.copy()
                    c5_best = c5_new2
                    h_curr2 = h_new2
                    step2 = min(step2 * 1.1, 0.05)
                else:
                    step2 *= 0.9

    h_best = project_capped_simplex(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
