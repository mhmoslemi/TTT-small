def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, temp, penalty, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        vals = corr
    cmax = np.max(vals)
    shifted = vals - cmax
    sm = cmax + temp * np.log(np.sum(np.exp(shifted / temp)))
    dx = 2.0 / n
    pen_term = penalty * (np.sum(h) * dx - 1.0) ** 2
    return sm * dx + pen_term

def repair_h(h):
    n = len(h)
    dx = 2.0 / n
    integ = np.sum(h) * dx
    if integ < 1e-12:
        return np.full_like(h, 0.5)

    h2 = np.clip(h2, 0.0, 1.0)
