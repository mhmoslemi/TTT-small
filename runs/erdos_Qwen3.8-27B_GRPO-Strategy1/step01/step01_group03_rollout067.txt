    a = a.copy()

    zeros = np.where(a == 0)[0]

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_max(h):
    """Return just the max correlation (without dx)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_binary(h, n_ones):
    """Project h to a binary vector with exactly n_ones ones."""
    idx = np.argsort(-h)
    a = np.zeros(len(h))
    a[idx[:n_ones]] = 1.0
    return a

def sla_binary(a, n, n_ones, t0, t_end, n_iter=200000):
    """Simulated annealing on binary vector."""
    a = a.copy()
    ones = set(np.where(a == 1)[0])
    zeros = set(np.where(a == 0)[0])
    best_a = a.copy()
    best_val = corr_max(a)
    current_val = best_val
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        T = t0 * (t_end_val / t0) ** (it / n_iter)  # geometric cooling
        # Random swap
        p1 = list(ones)[np.random.randint(len(ones))]
        p0 = list(zeros)[np.random.randint(len(zeros))]
        a[p1] = 0; a[p0] = 1
        new_val = corr_max(a)
        delta = new_val - current_val
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            current_val = new_val
            ones.discard(p1); zeros.discard(p0)
            ones.add(p0); zeros.add(p1)
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            a[p1] = 1; a[p0] = 0
    
    return best_a, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_max_val(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_sr(h, target_sum):
    h = np.array(h, dtype=float)
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def sa_binary(n, n_ones, T0, deadline, seed=42):

    best_a = None

            a[pos[:n_ones]] = 1.0

            for _ in range(nk):
                oi = np.where(a == 1)[0]
                zi = np.where(a == 0)[0]
                a[oi[rng.randint(len(oi))]] = 0
                a[zi[rng.randint(len(zi))]] = 1
        cur_val = corr_max_val(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        T = T0
        gamma = 0.9999
        stall = 0
        while time.time() < deadline:
            oi = np.where(a == 1)[0]
            zi = np.where(a == 0)[0]
            p1 = oi[rng.randint(len(oi))]
            p0 = zi[rng.randint(len(zi))]
            a[p1], a[p0] = 0, 1
            nv = corr_max_val(a)
            d = nv - cur_val
            if d < 0 or (T > 1e-8 and rng.rand() < np.exp(-d / T)):
                cur_val = nv
                if nv < best_val:
                    best_val = nv
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                a[p1], a[p0] = 1, 0
                stall += 1
            T *= gamma
            if stall > 50000:
                break
    return best_a, best_val

def local_search_binary(a, deadline):
    a = a.copy()
    cur_val = corr_max_val(a)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        best_imp = 1e-10
        best_swap = None
        oi = np.where(a == 1)[0]
        zi = np.where(a == 0)[0]
        for i in oi:
            ai = a[i]
            for j in zi:
                a[i], a[j] = 0, 1
                v = corr_max_val(a)
                a[i], a[j] = ai, 1 - ai
                imp = cur_val - v
                if imp > best_imp:
                    best_imp = imp
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            a[i], a[j] = 0, 1
            cur_val -= best_imp
            improved = True
    return a, cur_val

def slsqp_polish(h0, dx, target_sum, maxiter=400):
    n = len(h0)
    def obj(x):
        return x[n]
    def ineq_f(x):
        return x[n] - np.correlate(x[:n], 1.0 - x[:n], mode='full')
    def eq_f(x):
        return np.sum(x[:n]) - target_sum
    t_init = corr_max_val(h0) + 0.002
    x0 = np.concatenate([h0, [t_init]])
    bnds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': ineq_f}, {'type': 'eq', 'fun': eq_f}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            return project_sr(res.x[:n], target_sum)
    except Exception:
        pass
    return h0.copy()

def optimize_for_n(n, deadline_sa, deadline_ls, deadline_slsqp, seed, h_init=None):
    dx = 2.0 / n
    target_sum = n / 2.0
    n_ones = int(target_sum)
    has_frac = (n % 2 == 1)
    
    if has_frac:
        if h_init is not None and len(h_init) == n:
            h = project_sr(h_init, target_sum)
        else:
            h = np.zeros(n)
            h[:n_ones] = 1.0
            h[n_ones] = target_sum - n_ones
            h = project_sr(h, target_sum)
        return slsqp_polish(h, dx, target_sum, 300), dx, target_sum
    else:

        a, _ = local_search_binary(a, deadline_ls)
        h = slsqp_polish(a, dx, target_sum, 300)
        return h, dx, target_sum
