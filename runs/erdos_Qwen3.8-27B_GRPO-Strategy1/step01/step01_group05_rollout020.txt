import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 2.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-13:
        free = (out > 1e-13) & (out < 1.0 - 1e-13)
        nf = free.sum()
        if nf > 0:
            out[free] = np.clip(out[free] + diff / nf, 0.0, 1.0)
        else:
            out = np.clip(out + diff / n, 0.0, 1.0)
    return out


def c5_value(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def compute_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr / temp
    m = c.max()
    w = np.exp(c - m)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        wk = w[k]
        if s >= 0:
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n - s]
        else:
            s2 = -s
            g[s2:] += wk * (1.0 - h[:n - s2])
            g[:n - s2] -= wk * h[s2:]
    return g * dx


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') / temp
    m = corr.max()
    return (np.log(np.sum(np.exp(corr - m))) + m) * temp * dx


def slsqp_opt(h0, n, temp, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_obj, h0, method='SLSQP',
                   jac=lambda x: compute_grad(x, n, temp),
                   args=(n, temp),
                   bounds=[(0.0, 1.0)] * n, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x


def nesterov_subgrad(h0, n, n_iter, t_end):
    h = h0.copy()
    h_m = h0.copy()
    best_c5 = c5_value(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > t_end:
            break
        temp = max(2.0, 60.0 * np.exp(-it / 4000.0))
        tk = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        alpha = it / (it + 1 + tk) if it > 0 else 0.0
        _ = h + alpha * (h - h_m)  # extrapolation (used implicitly)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr / temp
        m = c.max()
        w = np.exp(c - m)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            s = k - (n - 1)
            wk = w[k]
            if s >= 0:
                g[:n - s] += wk * (1.0 - h[s:])
                g[s:] -= wk * h[:n - s]
            else:
                s2 = -s
                g[s2:] += wk * (1.0 - h[:n - s2])
                g[:n - s2] -= wk * h[s2:]
        g *= dx / temp
        lr = 0.03 / (1.0 + 0.002 * it)
        h_new = project_feasible(h + alpha * (h - h_m) - lr * g, n)
        c5_new = c5_value(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_m = h
        h = h_new
    return best_h, best_c5


def structured_init(n, kind):
    h = np.zeros(n)
    if kind == 0:
        h[:] = 0.5
    elif kind == 1:
        for i in range(n):
            if (i // 3) % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
    elif kind == 2:
        for i in range(n):
            if (i % 2 == 0 and (i // 4) % 2 == 0) or (i % 2 == 1 and (i // 4) % 2 == 1):
                h[i] = 1.0
    elif kind == 3:
        idx = np.sort(np.random.choice(n, size=n // 2, replace=False))
        h[idx] = 1.0
    elif kind == 4:
        h = np.sin(np.linspace(0, 4 * np.pi, n)) ** 2
        h /= h.max() if h.max() > 0 else 1.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        best_h = np.full(45, 0.5)
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = c5_value(best_h, n)

    # Phase 1: SLSQP annealing on best
    for temp in [200, 80, 40, 20, 10, 5, 3]:
        if time.time() - t0 > budget_s - 250:

        h_new = slsqp_opt(best_h, n, temp, 350)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Nesterov subgradient (long exploration)
    t_sg_end = t0 + budget_s * 0.55
    h_sg, c5_sg = nesterov_subgrad(best_h, n, 50000, t_sg_end)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()

    # Phase 3: SLSQP polish after subgradient
    for temp in [10, 5, 3, 2]:
        if time.time() - t0 > budget_s - 100:
            break
        h_new = slsqp_opt(best_h, n, temp, 300)
        h_new = project_feasible(h_new, n)
        c5_new = c5_value(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Multi-start perturbations
    n_max_starts = 30
    for trial in range(n_max_starts):
        if time.time() - t0 > budget_s - 60:

        scale = 0.08 * np.exp(-trial / 10.0) + 0.01

        h_new = slsqp_opt(h_try, n, 8, 120)

        x_old = np.linspace(0, 2, n)

        for temp in [30, 8]:
            if time.time() - t0 > budget_s - 20:
                break
            h_new = slsqp_opt(h_try, n_try, temp, 200)
            h_new = project_feasible(h_new, n_try)
            c5_new = c5_value(h_new, n_try)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 6: Structured inits + SLSQP
    if time.time() - t0 < budget_s - 20:
        for kind in range(5):
            if time.time() - t0 > budget_s - 10:
                break
            h_struct = structured_init(n, kind)
            h_new = slsqp_opt(h_struct, n, 15, 150)
            h_new = project_feasible(h_new, n)
            c5_new = c5_value(h_new, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final SLSQP polish

        h_new = slsqp_opt(best_h, n, 3, 200)
        h_new = project_feasible(h_new, n)
        c5_new = c5_value(h_new, n)

            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = c5_value(best_h, n)
    return best_h, final_c5, n
