```python
def gradient_max_corr(h, frac=0.95):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")  # length 2n-1
    max_c = np.max(corr)
    active = corr >= frac * max_c
    
    grad = np.zeros(n)
    
    # Positive lags: k = 0, 1, ..., n-1 (indices t = n-1, n, ..., 2n-2)
    for k in range(n):
        t = n - 1 + k
        if not active[t]:
            continue
        # grad[k:n] += (1 - h[0:n-k])
        # grad[0:n-k] -= h[k:n]
        grad[k:] += 1.0 - h[:n-k]
        grad[:n-k] -= h[k:]
    
    # Negative lags: k = -1, -2, ..., -(n-1) (indices t = n-2, n-3, ..., 0)
    for l in range(1, n):
        t = n - 1 - l
        if not active[t]:
            continue
        # grad[0:n-l] += (1 - h[l:n])
        # grad[l:n] -= h[0:n-l]
        grad[:n-l] += 1.0 - h[l:]
        grad[l:] -= h[:n-l]
    
    dx = 2.0 / n
    return grad * dx / max(1, np.sum(active))
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def gradient_max_corr(h, frac):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    max_c = np.max(corr)
    if max_c < 1e-15:
        return np.zeros(n)
    active = corr >= frac * max_c
    n_act = max(1, int(np.sum(active)))
    grad = np.zeros(n)
    for k in range(n):
        t = n - 1 + k
        if not active[t]:
            continue
        grad[k:] += 1.0 - h[:n-k]
        grad[:n-k] -= h[k:]
    for l in range(1, n):

        if not active[t]:
            continue
        grad[:n-l] += 1.0 - h[l:]
        grad[l:] -= h[:n-l]
```
