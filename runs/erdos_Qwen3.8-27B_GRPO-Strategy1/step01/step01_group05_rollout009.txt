import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    corr = np.correlate(np.clip(x, 0, 1), 1.0 - np.clip(x, 0, 1), mode='full')
    scaled = temp * corr
    m = np.max(scaled)
    return (np.log(np.sum(np.exp(scaled - m))) / temp + m / temp) * dx


def de_objective(params, n, target, temp):
    h = np.clip(params, 0.0, 1.0)
    penalty = 50.0 * (np.sum(h) - target) ** 2
    return smooth_obj(h, n, temp) + penalty


def repair_sum(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0).copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = np.sum(free)
        if nf == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return h


def slsqp_polish(h, n, temp, maxiter, bounds):
    res = minimize(smooth_obj, h, method='SLSQP', bounds=bounds,
                   args=(n, temp), options={'maxiter': maxiter, 'ftol': 1e-15})
    return repair_sum(res.x, n)


def coord_descent(h, n, max_sweeps, t_start, budget_s, margin=15):
    best_c5 = compute_c5(h, n)
    target = n / 2.0
    for sweep in range(max_sweeps):
        if time.time() - t_start > budget_s - margin:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            best_val = h.copy()
            best_obj = best_c5
            cur = h[i]
            candidates = np.unique(np.concatenate([
                np.linspace(0, 1, 21),
                np.clip([cur - 0.1, cur - 0.05, cur - 0.02, cur - 0.005,
                         cur + 0.005, cur + 0.02, cur + 0.05, cur + 0.1, 0.0, 1.0], 0, 1)
            ]))
            for v in candidates:
                h[i] = v
                h_rep = repair_sum(h, n)
                c5 = compute_c5(h_rep, n)
                if c5 < best_obj - 1e-10:
                    best_obj = c5
                    best_val = h_rep.copy()
            if best_obj < best_c5:
                h = best_val
                best_c5 = best_obj
                improved = True
        if not improved:
            break
    return h, best_c5


def try_nm(h, n, t_start, budget_s, temp=35):
    if time.time() - t_start > budget_s - 25:
        return h, compute_c5(h, n)
    res = minimize(smooth_obj, h, method='Nelder-Mead', args=(n, temp),
                   options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12, 'disp': False})
    h_t = repair_sum(res.x, n)
    return h_t, compute_c5(h_t, n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n = len(h0)
    h0 = repair_sum(h0, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP warm start
    for temp in [8, 20, 60, 150]:
        if time.time() - t_start > budget_s - 400:
            break
        h_t = slsqp_polish(best_h, n, temp, 400, bounds)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t

    # Phase 2: Differential Evolution (global)
    if time.time() - t_start < budget_s - 120:
        remaining = budget_s - 120 - (time.time() - t_start)
        maxiter = max(100, min(800, int(remaining * 1.5)))
        try:
            de_res = differential_evolution(
                de_objective, bounds=bounds, args=(n, target, 40),

                popsize=60, mutation=(0.4, 1.2), recombination=0.7,
                polish=False, updating='deferred', init='latinhypercube'
            )
            h_de = repair_sum(de_res.x, n)
            c5_de = compute_c5(h_de, n)
            if c5_de < best_c5:
                best_c5, best_h = c5_de, h_de
        except Exception:
            pass
        # Polish best
        h_t = slsqp_polish(best_h, n, 30, 300, bounds)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t

    # Phase 3: Coordinate descent
    if time.time() - t_start < budget_s - 80:
        best_h, best_c5 = coord_descent(best_h, n, 15, t_start, budget_s, margin=80)

    # Phase 4: Nelder-Mead
    best_h, best_c52 = try_nm(best_h, n, t_start, budget_s)
    if best_c52 < best_c5:
        best_c5 = best_c52

    # Phase 5: Multi-start local search
    for trial in range(40):
        if time.time() - t_start > budget_s - 40:

        scale = 0.12 * np.exp(-trial / 12.0) + 0.005
        cand = best_h + np.random.uniform(-scale, scale, n)
        cand = repair_sum(cand, n)
        h_t = slsqp_polish(cand, n, 50, 200, bounds)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t

    # Phase 6: Structured starts (periodic, anti-symmetric)
    if time.time() - t_start < budget_s - 30:
        for period in [3, 4, 5, 6, 9, 15]:
            if time.time() - t_start > budget_s - 15:
                break
            if n % period == 0:
                base = np.random.uniform(0.1, 0.9, period)
                pat = np.tile(base, n // period)
                pat = repair_sum(pat, n)
                h_t = slsqp_polish(pat, n, 40, 200, bounds)
                c5_t = compute_c5(h_t, n)
                if c5_t < best_c5:
                    best_c5, best_h = c5_t, h_t
        # Anti-symmetric: h[i + n//2] = 1 - h[i]
        if n % 2 == 0:
            half = n // 2
            anti = np.random.uniform(0, 1, half)
            anti_full = np.concatenate([anti, 1.0 - anti])
            anti_full = repair_sum(anti_full, n)
            h_t = slsqp_polish(anti_full, n, 40, 200, bounds)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5, best_h = c5_t, h_t

    # Phase 7: Final alternating CD + SLSQP
    for _ in range(6):
        if time.time() - t_start > budget_s - 10:
            break
        best_h, best_c5 = coord_descent(best_h, n, 4, t_start, budget_s, margin=10)
        if time.time() - t_start > budget_s - 8:
            break
        h_t = slsqp_polish(best_h, n, 80, 200, bounds)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t

    # Phase 8: Final NM polish
    best_h, best_c52 = try_nm(best_h, n, t_start, budget_s, temp=20)
    if best_c52 < best_c5:
        best_c5 = best_c52

    best_h = repair_sum(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
