def lag_gradient(h, lag, N):
    g = np.zeros(N)
    if lag >= 0:
        if lag < N:
            g[lag:] += (1.0 - h[:N - lag])
            g[:N - lag] -= h[lag:]
    else:
        l = -lag
        if l < N:
            g[:N - l] += (1.0 - h[l:])
            g[l:] -= h[:N - l]
    return g

def optimize(h, N, target, dx, t0, deadline, tau0=5.0, lr0=0.04, iters=3000):
    """Multi-phase Adam with smooth max."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    tau = tau0
    lr = lr0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_b = np.zeros(N)
    v_b = np.zeros(N)
    
    for it in range(iters):
        if time.time() > deadline:
            break
        # Adaptive phase
        phase = min(it // 700, 4)
        tau = tau0 * (4.0 ** phase)
        lr = lr0 * (0.5 ** phase)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr) * dx)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        
        # Softmax weights
        sc = tau * (corr - corr.max())
        w = np.exp(sc)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2*N-1)/(2*N-1)
        else:
            w /= ws
        
        # Gradient from top lags
        thresh = 1e-4
        active = np.where(w > thresh)[0]
        if len(active) < 3:
            active = np.argsort(w)[-5:]
        grad = np.zeros(N)
        for idx in active:
            lag = int(idx) - (N - 1)
            grad += w[idx] * lag_gradient(h, lag, N)
        
        # Adam
        m_b = beta1*m_b + (1-beta1)*grad
        v_b = beta2*v_b + (1-beta2)*grad*grad
        m_h = m_b / (1 - beta1**(it+1))
        v_h = v_b / (1 - beta2**(it+1))
        h = h - lr * m_h / (np.sqrt(v_h) + eps)
        h = project_sum(h, target)
    
    return best_h, best_c5
