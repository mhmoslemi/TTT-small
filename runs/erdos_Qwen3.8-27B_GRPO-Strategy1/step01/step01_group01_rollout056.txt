def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = t * (corr - mx)
    s_max = np.max(s)
    return (mx + (s_max + np.log(np.sum(np.exp(s - s_max)))) / t) * dx

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break

            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)

    # Constraint: corr_lin[k] * dx <= c

    # J[k,:]·h_new - c/dx <= (J[k,:]·h - corr[k]) * dx

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    c_obj[-1] = 1.0

    h_damped = (1 - damp) * h + damp * h_new

    s = t * (corr - mx)
    sm = np.max(s)
    return (mx + (sm + np.log(np.sum(np.exp(s - sm)))) / t) * dx

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, nd):
    n = len(h)
    if n == nd:
        return h.copy()
    xs = np.arange(n) * 2.0 / n
    xd = np.arange(nd) * 2.0 / nd
    return proj(np.interp(xd, xs, h), nd)

def cons_fun(x):
    return np.array([np.sum(x) - len(x) / 2.0])

def cons_jac(x):
    return np.ones((1, len(x)))

def compute_J(h, n):
    K = 2 * n - 1
    off = n - 1
    h1 = 1.0 - h
    J = np.zeros((K, n))
    for k in range(K):
        sh = k - off
        lo1 = max(0, sh)
        hi1 = min(n, sh + n)
        lo2 = max(0, -sh)
        hi2 = min(n, n - sh)
        if hi1 > lo1:
            J[k, lo1:hi1] += h1[lo1 - sh:hi1 - sh]
        if hi2 > lo2:
            J[k, lo2:hi2] -= h[lo2 + sh:hi2 + sh]
    return J

def slp_iter(h, n, damp=1.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    J = compute_J(h, n)
    K = len(corr)
    rhs = (J @ h - corr) * dx
    A_ub = np.hstack([J, -np.ones((K, 1)) / dx])
    b_ub = rhs
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0, 1)] * n + [(-1e6, 1e6)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return h, False
    h_slp = proj(res.x[:n], n)
    h_new = (1.0 - damp) * h + damp * h_slp
    return proj(h_new, n), True

_T_global = 80.0

def obj_wrapper(h):
    return soft_max_obj(h, _T_global)

def run(seed=42, budget_s=1100, **kwargs):
    global _T_global
    t0 = time.time()
    rng = np.random.RandomState(seed)
    rem = lambda: budget_s - (time.time() - t0)

    h_init = None
    try:
        if len(initial_h_values) >= 70:
            h_init = np.array(initial_h_values, dtype=float).copy()
            h_init = proj(h_init, len(initial_h_values))
    except Exception:
        pass
    if h_init is None:
        h_init = np.full(70, 0.5)

    bh = h_init.copy()
    bc = compute_c5(bh)

    # Phase 1: Coarse n=70 optimization with annealing
    n = 70
    h70 = interp_to(bh, n)
    bnds70 = [(0.0, 1.0)] * n
    cons70 = {'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}
    xg = np.arange(n) * 2.0 / n

    starts = [h70.copy()]
    for freq in [1, 2, 3, 4, 5, 7, 9, 11]:
        starts.append(proj(0.5 + 0.35 * np.sin(2.0 * np.pi * freq * xg + rng.rand() * 2.0 * np.pi), n))
    for amp in [0.05, 0.1, 0.2, 0.3]:
        starts.append(proj(0.5 + amp * rng.randn(n), n))
    starts.append(proj(np.clip(np.sin(2.0 * np.pi * 3.0 * xg) * 0.5 + 0.5, 0, 1), n))

    temps = [15, 30, 50, 80, 120, 200, 350, 500]
    for t_val in temps:
        _T_global = t_val
        for hc0 in starts:
            if rem() < 40:
                break
            try:
                r = minimize(obj_wrapper, hc0, method='SLSQP',
                           bounds=bnds70, constraints=cons70,

                c5 = compute_c5(hc)

                    bh = hc.copy()
                    starts[0] = hc.copy()
            except Exception:
                pass
        if rem() < 60:

    # Phase 2: SLP at n=70

    # Phase 3: n=140 refinement

        s140 = [h140.copy()]
        for _ in range(4):
            s140.append(proj(h140 + rng.randn(n2) * 0.02, n2))
        for t_val in [80, 150, 300, 500]:
            _T_global = t_val
            if rem() < 25:
                break
            for hc0 in s140:
                if rem() < 15:
                    break
                try:
                    r = minimize(obj_wrapper, hc0, method='SLSQP',
                               bounds=bnds140, constraints=cons140,

                    hc = proj(r.x, n2)

                    if c5 < bc:

                        bh = hc.copy()

                except Exception:

        if rem() > 50:

                if rem() < 15:

                h140, ok = slp_iter(s140[0], n2, damp=0.6)
