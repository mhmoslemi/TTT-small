import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def lse_core(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    C = raw * dx
    m = float(np.max(C))
    shifted = (C - m) / beta
    shifted = np.clip(shifted, -200, 200)
    exp_s = np.exp(shifted)
    total = float(exp_s.sum())
    obj = m + beta * np.log(total)
    w = exp_s / total
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - (n - 1)
        js1 = max(0, lag)
        je1 = min(n, n + lag)
        if js1 < je1:
            idx = np.arange(js1, je1)
            grad[idx] += wk * (1.0 - h[idx - lag])
        js2 = max(0, -lag)
        je2 = min(n, n - lag)
        if js2 < je2:
            idx2 = np.arange(js2, je2)
            grad[idx2] -= wk * h[idx2 + lag]
    return obj, dx * grad

def lse_obj(x, beta, pw):
    n = len(x)
    o, _ = lse_core(x, beta)
    s = np.sum(x) - n / 2.0
    return o + pw * s * s

def lse_grad(x, beta, pw):
    n = len(x)
    _, g = lse_core(x, beta)
    s = np.sum(x) - n / 2.0
    g += (2.0 * pw * s / n) * np.ones(n)
    return g

    h = h.copy()
    for _ in range(100):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(parent, new_n):
    old_n = len(parent)
    ox = np.arange(old_n, dtype=np.float64) / old_n
    nx = np.arange(new_n, dtype=np.float64) / new_n
    v = np.interp(nx, ox, parent)
    v = fix_sum(v, new_n)
    return v

def subgrad_step(h, n, top_k=5, lr=0.002):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = float(np.max(corr))
    thresh = cmax - 1e-6
    active = np.where(corr >= thresh)[0]
    if len(active) > top_k:
        top_idx = np.argsort(corr[active])[::-1][:top_k]
        active = active[top_idx]
    sg = np.zeros(n)
    for m in active:
        lag = m - (n - 1)
        js1 = max(0, lag)
        je1 = min(n, n + lag)
        if js1 < je1:
            idx = np.arange(js1, je1)
            sg[idx] += (1.0 - h[idx - lag])
        js2 = max(0, -lag)
        je2 = min(n, n - lag)
        if js2 < je2:
            idx2 = np.arange(js2, je2)
            sg[idx2] -= h[idx2 + lag]
    sg *= dx / len(active)
    h_new = h - lr * sg
    h_new = fix_sum(np.clip(h_new, 0.0, 1.0), n)
    return h_new

def coord_descent(h, n, rng, steps, t0, budget_s):

    dx = 2.0 / n
    for it in range(steps):
        if budget_s - (time.time() - t0) < 5:

        js2 = max(0, -lag)

        pos_mask = sg_local > 0.01

            i = i_pos[rng.randint(len(i_pos))]
            j = i_neg[rng.randint(len(i_neg))]

        h_tr[i] = np.clip(h[i] - d, 0.0, 1.0)

                continue
        c = compute_c5(h_tr)
        if c < cur - 1e-15:
