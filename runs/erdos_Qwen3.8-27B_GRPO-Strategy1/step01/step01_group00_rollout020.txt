def grad_correct(h, T):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')  # length 2n-1, index m = 0..2n-2
    # corr[m] = overlap[s] where s = (n-1) - m
    Tc = T * dx
    c_max = np.max(corr)
    exp_vals = np.exp(Tc * (corr - c_max))
    w = exp_vals / np.sum(exp_vals)  # w[m] for m = 0, ..., 2n-2
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-16:
            continue
        s = (n - 1) - m  # shift
        # Term 1: (1-h[j+s]) for 0 ≤ j+s ≤ n-1, i.e., -s ≤ j ≤ n-1-s

        hi1 = min(n, n - s)  # exclusive

        # Term 2: -h[j-s] for 0 ≤ j-s ≤ n-1, i.e., s ≤ j ≤ n-1+s

    return grad * dx
