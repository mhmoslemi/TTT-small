import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_optim(h, n, max_passes=200, time_deadline=None):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)

    for p in range(max_passes):
        if time_deadline is not None and time.time() > time_deadline:
            break
        for j in range(n):
            old_v = h[j]
            B = np.zeros(2 * n - 1)
            B[j:j+n] += (1.0 - h[::-1])
            B[n-1-j : 2*n-1-j] -= h
            A = corr - old_v * B
            lo, hi = 0.0, 1.0
            for _ in range(22):
                mid = (lo + hi) * 0.5
                vals = A + mid * B
                mx = np.max(vals)
                active = vals >= mx - 1e-10
                b_act = B[active]
                if np.min(b_act) > 1e-12:
                    hi = mid
                elif np.max(b_act) < -1e-12:
                    lo = mid
                else:
                    break
            v = (lo + hi) * 0.5
            f_val = np.max(A + v * B)
            f_0 = np.max(A)
            f_1 = np.max(A + B)
            if f_0 < f_val - 1e-10:
                v = 0.0
            elif f_1 < f_val - 1e-10:
                v = 1.0
            delta = v - old_v
            if abs(delta) > 1e-14:
                h[j] = v
                corr += delta * B
        h_projected = project_sum(h, n)
        if not np.allclose(h_projected, h, atol=1e-13):
            h = h_projected
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def upsample_to(h, new_n, period=2.0):
    old_n = len(h)
    x_old = np.arange(old_n) * period / old_n
    x_new = np.arange(new_n) * period / new_n
    h_new = np.interp(x_new, x_old, h, period=period)
    return h_new
