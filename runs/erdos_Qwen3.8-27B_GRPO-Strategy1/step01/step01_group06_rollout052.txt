import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    s = np.sum(exps)
    val = (m + np.log(s) / T) * dx
    weights = exps / s
    grad = np.zeros(n)
    for j in range(len(corr)):
        w = weights[j]
        if w < 1e-14:
            continue
        l = j - (n - 1)
        if l >= 0:
            grad[l:n] += w * (1.0 - h[0:n-l])
            grad[0:n-l] -= w * h[l:n]
        else:
            lp = -l
            grad[0:n-lp] += w * (1.0 - h[lp:n])
            grad[lp:n] -= w * h[0:n-lp]
    return val, dx * grad
