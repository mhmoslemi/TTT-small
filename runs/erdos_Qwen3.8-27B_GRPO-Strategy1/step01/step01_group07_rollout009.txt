def soft_max_grad(h, n, beta, top_k=10):

    shifted = corr - np.max(corr)
    weights = np.exp(shifted / beta)
    weights /= weights.sum()
    # Use top_k
    idx = np.argsort(weights)[-top_k:]

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    if np.abs(np.sum(h) - target) < 1e-12 and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_for_shift(h, n, k_idx):
    m = k_idx - (n - 1)
    j = np.arange(n)
    jm = j - m
    jp = j + m
    h_m = np.zeros(n)
    vm = (jm >= 0) & (jm < n)
    h_m[vm] = h[jm[vm]]
    h_p = np.zeros(n)
    vp = (jp >= 0) & (jp < n)
    h_p[vp] = h[jp[vp]]
    return 1.0 - h_m - h_p

def slsqp_eq(x, n):
    return np.sum(x) - n / 2.0

def slsqp_eq_jac(x, n):
    return np.ones(n)

def slsqp_obj(x, n, beta):

    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    s = np.exp((corr - m) / beta).sum()
    return dx * (m + beta * np.log(s))

def round_robin_step(h, n, top_idx, lr):

    for ki in top_idx:
        g = grad_for_shift(h_new, n, ki)
        h_new = project_feasible(h_new - lr * g, n)
    return h_new
