    m = np.max(corr)
    vals = np.exp(T * (corr - m))
    return (m + np.log(vals.sum()) / T) * dx

                g[:n+s] -= h[-s:]
                g[-s:] -= h[:n+s]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        r = room.sum()
        if r > 1e-14:
            h += room * (diff / r)
    else:
        r = h.sum()
        if r > 1e-14:
            h += h * (diff / r)
    return np.clip(h, 0.0, 1.0)

def softmax_c5(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    vals = np.exp(T * (corr - m))
    return (m + np.log(vals.sum()) / T) * dx

def softmax_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        g = np.ones(n)
        if s > 0:
            g[s:] -= x[:n - s]
            g[:n - s] -= x[s:]
        elif s < 0:
            g[:n + s] -= x[-s:]
            g[-s:] -= x[:n + s]
        else:
            g -= 2.0 * x
        grad += w[k] * g
    return dx * grad

def sum_con(x):
    return x.sum() - len(x) / 2.0

def active_grad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    s = k_star - (n - 1)
    g = np.ones(n)
    if s > 0:
        g[s:] -= h[:n - s]
        g[:n - s] -= h[s:]
    elif s < 0:
        g[:n + s] -= h[-s:]
        g[-s:] -= h[:n + s]
    else:
        g -= 2.0 * h
    return g, corr, k_star

def active_set_descent(h, n, t0, budget_s, max_iter=2000):
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = 0.02
    for it in range(max_iter):
        if time.time() - t0 > budget_s:
            break
        g, corr, k_star = active_grad(best, n)
        h_new = best - lr * g
        h_new = proj(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-10:
                lr = 0.01
    return best, best_c5

def slsqp_optimize(h0, n, T, maxiter, t0, budget_s):
    if time.time() - t0 > budget_s - 30:
        return h0, compute_c5(h0)
    cons = [{'type': 'eq', 'fun': sum_con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            softmax_c5, h0, args=(T,),
            jac=softmax_grad,
            method='SLSQP', bounds=bnds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        hc = proj(np.clip(res.x, 0, 1), n)
        return hc, compute_c5(hc)
    except Exception:
        return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(60)
        best_h[:30] = 1.0

    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T
    T_schedule = [10, 30, 80, 200, 500, 1200, 3000, 8000]
    for T in T_schedule:
        if time.time() - t0 > budget_s * 0.35:
            break
        best_h, c5v = slsqp_optimize(best_h, n, T, 600, t0, budget_s * 0.35)
        if c5v < best_c5:
            best_c5 = c5v

    # Phase 2: Active-set projected gradient (many small steps)
    best_h, c5v = active_set_descent(best_h, n, t0, budget_s * 0.65, max_iter=3000)
    if c5v < best_c5:
        best_c5 = c5v

    # Phase 3: Multi-start with perturbations + SLSQP
    n_starts = 60
    for i in range(n_starts):
        if time.time() - t0 > budget_s * 0.85:
            break
        scale = 0.08 * np.exp(-0.025 * i) + 0.005
        h0 = best_h + np.random.uniform(-scale, scale, size=n)
        h0 = proj(h0, n)
        T_use = 2000 if i % 2 == 0 else 5000
        h0, c5v = slsqp_optimize(h0, n, T_use, 400, t0, budget_s * 0.85)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h0

    # Phase 4: Active-set gradient again on updated best
    if time.time() - t0 < budget_s * 0.93:
        best_h, c5v = active_set_descent(best_h, n, t0, budget_s * 0.93, max_iter=2000)
        if c5v < best_c5:
            best_c5 = c5v

    # Phase 5: Final SLSQP polish with very high T
    if time.time() - t0 < budget_s * 0.98:
        best_h, c5v = slsqp_optimize(best_h, n, 15000, 1500, t0, budget_s * 0.98)
        if c5v < best_c5:
            best_c5 = c5v

    # Final feasibility
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
