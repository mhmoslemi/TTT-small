import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def soft_max_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = temp * corr
    m = np.max(scaled)
    return (np.log(np.sum(np.exp(scaled - m))) / temp + m / temp) * dx


def obj_pen(h, n, temp, mu):
    return soft_max_obj(h, n, temp) + mu * (np.sum(h) - n / 2.0) ** 2


def sum_eq(h, n):
    return np.sum(h) - n / 2.0


def sum_eq_jac(h, n):
    return np.ones(n)


def repair_h(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] = np.clip(h[free] + diff / nf, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def make_cons(n):
    return {'type': 'eq', 'fun': sum_eq, 'args': (n,), 'jac': sum_eq_jac}


def do_lbfgs(h0, n, temp, mu, maxiter, bounds):
    res = minimize(obj_pen, h0, method='L-BFGS-B', bounds=bounds,
                   args=(n, temp, mu), options={'maxiter': maxiter, 'ftol': 1e-16})
    return repair_h(res.x, n)


def do_slbqp(h0, n, temp, maxiter, bounds):
    cons = [make_cons(n)]
    res = minimize(soft_max_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(n, temp),
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return repair_h(res.x, n)


def local_search_binary(n, t0, budget_s, margin):
    n_half = n // 2
    n_ones = n_half
    rem = n - 2 * n_half  # 1 if n odd, 0 if even
    h = np.zeros(n)
    perm = np.random.permutation(n)
    h[perm[:n_ones]] = 1.0
    if rem == 1:
        h[perm[n_ones]] = 0.5
    best_local = compute_c5(h, n)
    best_h = h.copy()
    T = 0.008
    for step in range(80000):
        if budget_s - margin - (time.time() - t0) < 5:
            break
        if step % 20000 == 19999:
            T *= 0.4
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h, n)
        delta = c5_new - best_local
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-6)):
            best_local = c5_new
            best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_local


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(45, 0.5)
    n = len(h_init)
    h_init = repair_h(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    def tleft(margin):
        return budget_s - margin - (time.time() - t0)

    # Phase 1: L-BFGS-B from initial (fast)
    for temp in [80, 150, 300]:
        if tleft(250) < 8:
            break
        for mu in [500, 2000, 10000]:
            if tleft(250) < 4:
                break
            h_t = do_lbfgs(best_h.copy(), n, temp, mu, 5000, bounds)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t

    # Phase 2: SLSQP exact constraint polish
    for temp in [30, 60, 120, 250]:
        if tleft(200) < 10:
            break
        maxit = min(800, max(200, int(tleft(200) * 4)))
        h_t = do_slbqp(best_h.copy(), n, temp, maxit, bounds)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    # Phase 3: Binary local search (global exploration)
    if tleft(120) > 30:
        h_ls, c5_ls = local_search_binary(n, t0, budget_s, 120)
        h_ls = repair_h(h_ls, n)
        c5_ls = compute_c5(h_ls, n)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
        # Polish the binary solution
        if tleft(100) > 10:
            h_t = do_lbfgs(h_ls, n, 50, 5000, 3000, bounds)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t
            if tleft(80) > 8:
                h_t = do_slbqp(h_t, n, 40, 400, bounds)
                c5_t = compute_c5(h_t, n)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t

    # Phase 4: Diverse structured starts
    if tleft(100) > 20:
        x_grid = np.linspace(0, 1, n)
        starts = []
        # Sinusoidal
        for freq in [1, 2, 3, 4]:
            for amp in [0.3, 0.5, 0.7]:
                for ph in [0, np.pi / 3, 2 * np.pi / 3]:
                    s = 0.5 + amp * np.sin(2 * np.pi * freq * x_grid + ph)
                    starts.append(repair_h(s, n))
        # Block patterns
        for bl in [3, 4, 5, 6, 9, 15]:
            for off in range(min(bl, 4)):
                s = np.zeros(n)
                for i in range(n):
                    if (i + off) % (2 * bl) < bl:
                        s[i] = 1.0
                starts.append(repair_h(s, n))
        # Random bang-bang
        for _ in range(8):
            s = np.zeros(n)
            idx = np.random.choice(n, int(n / 2), replace=False)
            s[idx] = 1.0
            rem = n - 2 * int(n // 2)
            if rem == 1:
                idx2 = np.random.choice(n, 1, replace=False)
                if s[idx2[0]] == 0:
                    s[idx2[0]] = 0.5
                else:
                    s[idx] = 0.5
            starts.append(repair_h(s, n))

        for s in starts:
            if tleft(60) < 5:
                break
            h_t = do_lbfgs(s, n, 50, 3000, 2000, bounds)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5 + 0.001:
                h_t = do_slbqp(h_t, n, 40, 300, bounds)
                c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t

    # Phase 5: Iterative perturbation + refine
    for trial in range(40):
        if tleft(50) < 5:
            break
        scale = 0.08 * (1.0 + 0.5 * np.exp(-trial / 10.0))
        s = best_h + np.random.randn(n) * scale
        s = repair_h(s, n)
        h_t = do_slbqp(s, n, 25, 150, bounds)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
            if tleft(50) > 10:
                h_t = do_lbfgs(h_t, n, 80, 5000, 2000, bounds)
                c5_t = compute_c5(h_t, n)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t

    # Phase 6: Second binary search with different seed
    if tleft(80) > 40:
        np.random.seed(seed + 7)
        h_ls2, c5_ls2 = local_search_binary(n, t0, budget_s, 80)
        h_ls2 = repair_h(h_ls2, n)
        c5_ls2 = compute_c5(h_ls2, n)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2
        if tleft(60) > 8:
            h_t = do_lbfgs(h_ls2, n, 40, 5000, 3000, bounds)
            c5_t = compute_c5(h_t, n)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t
            if tleft(50) > 5:
                h_t = do_slbqp(h_t, n, 20, 300, bounds)
                c5_t = compute_c5(h_t, n)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t

    # Final polish
    if tleft(10) > 5:
        h_t = do_slbqp(best_h.copy(), n, 15, 500, bounds)
        c5_t = compute_c5(h_t, n)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    best_h = repair_h(best_h, n)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
