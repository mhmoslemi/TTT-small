def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def logsumexp_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n

    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / e.sum()

        s = idx - (n - 1)
        wi = w[idx]

        hi1 = min(n, s + n)
        if lo1 < hi1:

        return h

        if t < 1e-12: return h
        h += r * (d / t)
    else:
        r = h.copy()
        t = r.sum()
        if t < 1e-12: return h
        h += r * (d / t)
    return np.clip(h, 0, 1)

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP initial polish
    if time.time() - t0 < budget_s * 0.08:
        for T in [100, 1000]:
            hc = slsqp_run(best_h, n, T, 600, False, 0)
            c = compute_c5(hc)
            if c < best_c5:
                best_c5, best_h = c, hc

    # Phase 2: Aggressive gradient descent with temperature schedule
    T_sched = [20, 50, 150, 500, 1500, 5000]
    for T in T_sched:
        tlim = min(budget_s * 0.42, t0 + 50)
        if time.time() - t0 > budget_s * 0.42:
            break
        for lr_init in [0.015, 0.008, 0.004]:
            if time.time() - t0 > budget_s * 0.42:
                break
            hc, c = grad_desc(best_h, n, T, lr_init, 3000, 0.85, t0, budget_s * 0.42)
            if c < best_c5:
                best_c5, best_h = c, hc

    # Phase 3: Top-k SLSQP and gradient descent
    if time.time() - t0 < budget_s * 0.58:
        for k in [3, 5, 8]:
            if time.time() - t0 > budget_s * 0.58:
                break
            hc = slsqp_run(best_h, n, 0, 800, True, k)
            c = compute_c5(hc)
            if c < best_c5:
                best_c5, best_h = c, hc
        for T in [500, 3000]:
            if time.time() - t0 > budget_s * 0.58:
                break
            hc = slsqp_run(best_h, n, T, 800, False, 0)
            c = compute_c5(hc)
            if c < best_c5:
                best_c5, best_h = c, hc

    # Phase 4: Basin hopping with gradient polish
    n_bh = 50
    for it in range(n_bh):
        if time.time() - t0 > budget_s * 0.80:
            break
        step = 0.07 * (1.0 - it / float(n_bh)) + 0.012
        r = np.random.rand()
        if r < 0.35:
            hp = best_h + step * np.random.randn(n)
        elif r < 0.65:
            hp = best_h.copy()
            bl = np.random.randint(4, n // 3 + 1)
            st = np.random.randint(0, n - bl)
            hp[st:st + bl] += step * (np.random.rand(bl) - 0.5) * 2.0
        else:
            hp = best_h.copy()
            i1, i2 = np.random.randint(0, n, 2)
            hp[i1] += step * 0.6
            hp[i2] -= step * 0.6
        hp = project(hp, n)
        hc, c = grad_desc(hp, n, 800, 0.008, 400, 0.75, t0, budget_s * 0.80)
        temp = 0.003 * (1.0 - it / float(n_bh)) + 0.0002
        if c < best_c5 or (c < best_c5 + 0.005 and np.random.rand() < np.exp(-(c - best_c5) / temp)):
            if c < best_c5:
                best_c5, best_h = c, hc
            else:
                best_h = hc

    # Phase 5: Multi-resolution
    for n_try in [80, 120, 200]:
        if time.time() - t0 > budget_s * 0.90:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        for trial in range(2):
            if time.time() - t0 > budget_s * 0.91:
                break
            h_try = h_interp + np.random.uniform(-0.03, 0.03, n_try)
            h_try = project(h_try, n_try)
            for T in [100, 500, 2000]:
                if time.time() - t0 > budget_s * 0.91:
                    break
                hc, c = grad_desc(h_try, n_try, T, 0.01, 1500, 0.85, t0, budget_s * 0.91)
                if c < best_c5:
                    x_n2 = np.linspace(0, 2, n_try, endpoint=False)
                    x_o2 = np.linspace(0, 2, n, endpoint=False)
                    best_h = project(np.interp(x_o2, x_n2, hc), n)
                    best_c5 = compute_c5(best_h)

    # Phase 6: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.97:
        for T in [5000, 20000, 50000]:
            if time.time() - t0 > budget_s * 0.98:
                break
            hc = slsqp_run(best_h, n, T, 1200, False, 0)
            c = compute_c5(hc)
            if c < best_c5:
                best_c5, best_h = c, hc
        for k in [3, 5]:
            if time.time() - t0 > budget_s * 0.985:
                break
            hc = slsqp_run(best_h, n, 0, 1000, True, k)
            c = compute_c5(hc)
            if c < best_c5:
                best_c5, best_h = c, hc

    # Phase 7: Coordinate descent on worst shifts
    for _ in range(300):
        if time.time() - t0 > budget_s * 0.995:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top5 = np.argsort(corr)[-5:]
        improved = False
        for k_sh in top5:
            s = k_sh - (n - 1)
            for j in range(max(0, s), min(n, n - s)):
                for d in [0.02, -0.02, 0.05, -0.05]:
                    ht = best_h.copy()
                    ht[j] = np.clip(ht[j] + d, 0, 1)
                    ht = project(ht, n)
                    ct = compute_c5(ht)
                    if ct < best_c5 - 1e-13:
                        best_h = ht
                        best_c5 = ct
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
