import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_objgrad(h, N, T, topk):
    corr = compute_corr(h)
    dx = 2.0 / N
    lenc = len(corr)
    k = min(topk, lenc)
    idx = np.argpartition(corr, -k)[-k:]
    cmax = float(corr[idx].max())
    exps = np.exp((corr[idx] - cmax) / T)
    w = exps / exps.sum()
    val = (cmax + T * np.log(exps.sum())) * dx
    grad = np.zeros(N)
    center = N - 1
    for ii in range(k):
        wi = w[ii]
        if wi < 1e-12:
            continue
        s = int(idx[ii]) - center
        if s >= 0 and s < N:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        elif s < 0:
            t = -s
            if t < N:
                grad[:N - t] += wi * (1.0 - h[t:])
                grad[t:] -= wi * h[:N - t]
    return val, grad * dx

def adam_opt(h, N, target, deadline, T_start, T_end, topk, lr0, decay, decay_every):
    best_h = h.copy()
    best_val = compute_c5(best_h, N)
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 3000.0)
        T = T_start * (T_end / T_start) ** frac
        val, grad = softmax_objgrad(h, N, T, topk)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % decay_every == 0:
            lr *= decay
    return best_h, best_val

def pair_swap_search(h, N, target, deadline, rng):
    best_h = h.copy()
    best_val = compute_c5(best_h, N)
    h = h.copy()
    while time.time() < deadline:
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j:
            continue
        delta = 0.01 + 0.04 * rng.random()
        hi_val, lo_val = h[i], h[j]
        if hi_val >= lo_val:
            h[i] = hi_val - delta
            h[j] = lo_val + delta
        else:
            h[i] = hi_val + delta
            h[j] = lo_val - delta
        h = project_sum(h, target)
        val = compute_c5(h, N)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        else:
            h[i], h[j] = hi_val, lo_val
            h = project_sum(h, target)
    return best_h, best_val

def targeted_nudge(h, N, target, deadline, rng):
    best_h = h.copy()
    best_val = compute_c5(best_h, N)
    while time.time() < deadline:
        corr = compute_corr(best_h)
        center = N - 1
        peak_idx = int(np.argmax(corr))
        s = peak_idx - center
        h_try = best_h.copy()
        if s >= 0 and s < N:
            overlap = h_try[:N - s] * h_try[s:]
            top_pos = np.argsort(overlap)[-8:]

                h_try[p] -= 0.015
                if p + s < N:
                    h_try[p + s] += 0.008
        else:
            t = -s

                overlap = h_try[t:] * h_try[:N - t]
                top_pos = np.argsort(overlap)[-8:]
                for p in top_pos:
                    h_try[p + t] -= 0.015

        h_try = project_sum(h_try, target)

            val2 = compute_c5(h_try2, N)
            if val2 < best_val:
                best_val = val2
                best_h = h_try2.copy()
    return best_h, best_val
