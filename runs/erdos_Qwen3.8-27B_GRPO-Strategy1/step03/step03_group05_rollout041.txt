import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy().astype(np.float64)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14: break
            h += room * (diff / tr)
        else:
            tr = h.sum()
            if tr < 1e-14: break
            h += h * (diff / tr)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_shift(h, s):
    """Gradient of corr at shift s with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] += (-h[:n-s])
    else:
        t = -s
        g[t:] += (1.0 - h[:n-t])
        g[:n-t] += (-h[t:])
    return g

def weighted_grad(h, top_indices, weights, n):
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        s = int(idx - (n - 1))
        grad += w * grad_shift(h, s)
    return grad

def nesterov_subgrad(h0, n, n_iter, top_k, c_target, alpha0, momentum):
    """Accelerated projected subgradient."""
    h = h0.copy()
    y = h0.copy()
    best = h0.copy()
    best_c5 = compute_c5(h0)
    t = 1
    alpha = alpha0
    
    for it in range(n_iter):
        corr = compute_corr(y)
        # Find top-k shifts
        # Use hinge-based weights: weight = max(0, corr - c_target)
        excess = np.maximum(corr - c_target, 0)
        if excess.sum() < 1e-15:
            # All below target, use top-k
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
            w = corr[top_idx].astype(float)
            ws = w.sum()
            w = w / ws if ws > 0 else np.ones(top_k)/top_k
        else:
            # Weight by excess
            top_idx = np.where(excess > 0)[0]
            if len(top_idx) > top_k:
                order = np.argsort(excess[top_idx])[::-1][:top_k]
                top_idx = top_idx[order]
            w = excess[top_idx]
            ws = w.sum()
            w = w / ws if ws > 0 else np.ones(len(top_idx))/len(top_idx)
        
        grad = weighted_grad(y, top_idx, w, n)
        
        # Nesterov update
        z = y + momentum * (y - h)
        y = project_feasible(z - alpha * grad, n)
        h = y.copy()
        t += 1
        alpha = alpha0 / np.sqrt(t)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best, best_c5

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += room * (diff / tr)
        else:
            tr = h.sum()
            if tr < 1e-14:
                break
            h += h * (diff / tr)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_shift(h, s, n):
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] += (-h[:n-s])
    else:
        t = -s
        g[t:] += (1.0 - h[:n-t])
        g[:n-t] += (-h[t:])
    return g

def obj_and_grad(h, top_k, n):
    corr = compute_corr(h)
    m = np.max(corr)
    # softmax weights over top-k
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c = corr[idx]
    T = 50.0
    w = np.exp(T * (c - c.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for i, wi in zip(idx, w):
        s = int(i - (n - 1))
        grad += wi * grad_shift(h, s, n)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val, grad, corr

def nesterov_opt(h0, n, n_iter, top_k, alpha0, budget_frac):
    t0 = time.time()
    limit = budget_frac
    h = h0.copy()
    y = h0.copy()
    best = h0.copy()
    best_c5 = compute_c5(h0)
    tk = top_k
    for t in range(1, n_iter + 1):
        if time.time() - t0 > limit:
            break
        # Adaptive top-k: increase as we converge
        tk_cur = min(top_k, max(1, int(top_k * t / max(1, n_iter // 3))))
        val, grad, corr = obj_and_grad(y, tk_cur, n)
        m = np.max(corr)
        alpha = alpha0 / np.sqrt(t)
        momentum = (t - 1) / (t + 1.0)
        z = y + momentum * (y - h)
        y = project_feasible(z - alpha * grad, n)
        h = y.copy()
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def binary_push(h, n, frac=0.3):
    """Push a fraction of values toward 0 or 1."""
    h = h.copy()
    mask = (h > 0.1) & (h < 0.9)
    idx = np.where(mask)[0]
    if len(idx) == 0:
        return h
    k = max(1, int(frac * len(idx)))
    chosen = np.random.choice(idx, k, replace=False)
    for i in chosen:
        if h[i] < 0.5:
            h[i] = max(0.0, h[i] - 0.15)
        else:
            h[i] = min(1.0, h[i] + 0.15)
    return project_feasible(h, n)

def coord_descent(h, n, n sweeps=10):
    """Coordinate descent: for each j, try h[j]=0, h[j]=1, current, optimize."""
    h = h.copy()
    for _ in range(n_sweeps):
        for j in range(n):
            old = h[j]
            # Try a few candidate values
            cands = [0.0, 1.0, old, max(0, old-0.1), min(1, old+0.1)]
            cands = [c for c in cands if 0 <= c <= 1]
            best_c = compute_c5(h)
            best_v = old
            for cv in cands:
                h[j] = cv
                h = project_feasible(h, n)  # This changes other coords too
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = cv
            h[j] = best_v
            h = project_feasible(h, n)
    return h

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += room * (diff / tr)
        else:
            tr = h.sum()
            if tr < 1e-14:
                break
            h += h * (diff / tr)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_shift(h, s, n):
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n - s] += (1.0 - h[s:])
        g[s:] += (-h[:n - s])
    else:
        t = -s
        g[t:] += (1.0 - h[:n - t])
        g[:n - t] += (-h[t:])
    return g

def obj_grad_softmax(h, top_k, n, T):
    corr = compute_corr(h)
    m = corr.max()
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c = corr[idx]
    w = np.exp(T * (c - c.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for i, wi in zip(idx, w):
        s = int(i - (n - 1))
        grad += wi * grad_shift(h, s, n)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val, grad, corr

def anneal_opt(h0, n, n_iter, top_k_start, top_k_end, alpha0, T_start, T_end, t_deadline, shake_every=5000, shake_mag=0.02):
    h = h0.copy()
    y = h0.copy()
    best = h0.copy()
    best_c5 = compute_c5(h0)
    for t in range(1, n_iter + 1):
        if time.time() > t_deadline:
            break
        frac = (t - 1) / max(1, n_iter - 1)
        tk = max(1, int(round(top_k_start + (top_k_end - top_k_start) * frac)))
        T = T_start * (T_end / max(T_start, 1e-9)) ** frac
        val, grad, corr = obj_grad_softmax(y, tk, n, T)
        alpha = alpha0 / (1.0 + 0.003 * t)
        momentum = min(0.9, 0.75 * (1.0 - 0.5 * frac))
        z = y + momentum * (y - h)
        y = project_feasible(z - alpha * grad, n)
        h = y.copy()
        if shake_every and t % shake_every == 0 and frac < 0.7:
            h = project_feasible(h + np.random.randn(n) * shake_mag * (1.0 - frac), n)
            y = h.copy()
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def binary_nudge(h, n, strength=0.2):
    h = h.copy()
    h = np.where(h > 0.6, 1.0, np.where(h < 0.4, 0.0, h))
    return project_feasible(h, n)

def gen_structured_inits(n, k_inits=6):
    inits = []
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n)
    for iv in range(k_inits):
        h = np.zeros(n)
        if iv == 0:
            h[np.where(np.sin(x * 2.0 * np.pi * 0.31) > 0)[0][:n // 2]] = 1.0
        elif iv == 1:
            h[np.where(np.sin(x * 2.0 * np.pi * 0.23 + 0.7) > 0)[0][:n // 2]] = 1.0
        elif iv == 2:
            positions = np.sort((np.arange(n) * phi) % n).astype(int)
            h[positions[:n // 2]] = 1.0
        elif iv == 3:
            positions = np.full(n // 2, -1)
            step = n / (n // 2)
            for i in range(n // 2):
                positions[i] = int((i * step) % n)
            h[positions] = 1.0
        elif iv == 4:
            h = ((x % 7) < 3.5).astype(float)
        else:
            for _ in range(200):
                mask = np.random.rand(n) < 0.5
                if mask.sum() == n // 2:
                    h = mask.astype(float)
                    break
                h = mask.astype(float)
            h = project_feasible(h, n)
        h = project_feasible(h, n)
        inits.append(h)
    return inits
