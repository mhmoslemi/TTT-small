```python
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

        if abs(diff) < 1e-13:

        if total < 1e-14:
```

```python
    """Compute d(corr)/d(h[i]) vector."""
    n = len(h)
    c = np.zeros(2*n-1)
    for p in range(2*n-1):

        if s >= 0:
            # First position: j=i, valid if i+s < n
            if i + s < n:
                c[p] += (1 - h[i+s])
            # Second position: j+s=i → j=i-s, valid if i-s >= 0 and i-s < n-s
            if i - s >= 0 and i - s < n - s:
                c[p] += (-h[i-s])

            # First position: j=i, valid if i >= sp and i < n
            if i >= sp and i < n:
                c[p] += (1 - h[i-sp])
            # Second position: j-sp=i → j=i+sp, valid if i+sp < n
            if i + sp < n:
                c[p] += (-h[i+sp])
    return c
```

```python

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corrr_coeff(h, i):
    """d(corr)/d(h[i]) for all correlation positions."""
    n = len(h)
    c = np.zeros(2 * n - 1)
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            if i + s < n:
                c[p] += 1.0 - h[i + s]
            if i - s >= 0 and i - s < n - s:
                c[p] -= h[i - s]
        else:
            sp = -s
            if i >= sp:
                c[p] += 1.0 - h[i - sp]
            if i + sp < n:
                c[p] -= h[i + sp]
    return c

def c5_at_delta(corr, coeff, delta):
    """C5 when h[i] changes by delta (from current)."""
    new_corr = corr + delta * coeff
    return float(np.max(new_corr))

    """Fast coordinate descent using precomputed correlation structure."""
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)

    best_c5 = float(np.max(corr) * dx)

        for i in range(n):
            coeff = corrr_coeff(h, i)
            a = h[i]
            # Find best b in [0,1] minimizing max(corr + (b-a)*coeff)
            # Subject to: after setting h[i]=b, re-project for sum
            # For now, ignore sum constraint and handle later

            # Find min of max over t in [-a, 1-a]
            
            t_min, t_max = -a, 1.0 - a
            # Evaluate at endpoints
            best_t = 0  # default: no change
            best_val = np.max(corr)
            best_val = min(best_val, np.max(corr + t_min * coeff))
            if np.max(corr + t_min * coeff) < best_val:
                best_val = np.max(corr + t_min * coeff)
                best_t = t_min
            eval_tm = np.max(corr + t_max * coeff)
            if eval_tm < best_val:
                best_val = eval_tm
                best_t = t_max
            
            # Check crossing points (only for lines that could be on upper envelope)
            # Simplification: check top 20 lines
            m = np.max(corr)
            active = np.argsort(corr)[::-1][:40]
            for idx_a in range(len(active)):
                for idx_b in range(idx_a + 1, len(active)):
                    pa, pb = active[idx_a], active[idx_b]
                    da = coeff[pa]
                    db = coeff[pb]
                    if abs(da - db) < 1e-15:
                        continue
                    t_cross = (corr[pb] - corr[pa]) / (da - db)
                    if t_min < t_cross < t_max:
                        val = np.max(corr + t_cross * coeff)
                        if val < best_val:

                            best_t = t_cross
            
            new_val = a + best_t
            new_val = np.clip(new_val, 0.0, 1.0)
            if abs(new_val - a) > 1e-8:
                h[i] = new_val
                h = project_feasible(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5 = float(np.max(corr) * dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return h, best_c5
```

Wait, I'm re-projecting inside the inner loop, which changes all coordinates. This means the "grid" for coordinate i is being evaluated with different other-coordinates each time (due to projection). This is actually fine - it's a valid optimization method (like a projected greedy method).

But it's slow because project_feasible is called 7800 times per sweep. Let me make it faster:

Actually, the projection is: add (target - sum)/n to all free coordinates, clip. For small changes (one coordinate changed by at most 1), the correction is at most 1/n ≈ 0.013. So typically one iteration of projection suffices.

Let me optimize the projection:
```python
def project_fast(h, target):
    """Fast projection onto {0≤h≤1, sum=h_target}."""
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Distribute diff evenly, respecting bounds
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    n_mask = mask.sum()
    if n_mask == 0:
        return h
    step = diff / n_mask
    h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # If still not exact, do one more pass
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-14) if diff > 0 else (h > 1e-14)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_desc(h, n_sweeps=150, grid_n=120):
    n = len(h)
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    grid = np.linspace(0, 1, grid_n)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            best_v = old
            best_c = best_c5
            best_state = h.copy()
            for v in grid:
                if abs(v - old) < 1e-4:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = v
                    best_state = h.copy()
            h = best_state
            if best_c < best_c5 - 1e-11:
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def simulated_anneal(h, t_start, t_end, n_iters=5000, eta=0.05):
    n = len(h)
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    for it in range(n_iters):
        T = t_start * (t_end / t_start) ** (it / max(n_iters - 1, 1))
        i = np.random.randint(n)
        h[i] = np.clip(h[i] + np.random.randn() * eta, 0, 1)
        h = project_feasible(h, n)
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
        else:
            h = best_h.copy()
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_direct(h, maxiter=500):
    n = len(h)
    dx = 2.0 / n
    _, corr = (None, np.correlate(h, 1.0 - h, mode='full'))
    t0 = float(np.max(corr)) + 0.02
    def erdos_obj(x):
        return x[-1]
    def erdos_eq(x):
        return x[:-1].sum() - n / 2.0
    def erdos_ineq(x):
        h_ = x[:-1]
        t = x[-1]
        corr_ = np.correlate(h_, 1.0 - h_, mode='full')
        return t - corr_
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
    cons = [{'type': 'eq', 'fun': erdos_eq}, {'type': 'ineq', 'fun': erdos_ineq}]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def random_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    best_h, best_n, best_c5 = None, None, 1.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        best_h = project_feasible(h0, len(h0))
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    
    if best_h is None:
        best_n = 78
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent on initial
    if elapsed() < budget_s * 0.15:
        best_h, best_c5 = coord_desc(best_h, n_sweeps=100, grid_n=100)

    # Phase 2: SLSQP smooth polish
    if elapsed() < budget_s * 0.25:
        for T in [2000, 10000, 50000]:
            if elapsed() > budget_s * 0.25:
                break
            h_try = slsqp_smooth(best_h, T, maxiter=300)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 3: Simulated annealing to escape local minima
    if elapsed() < budget_s * 0.40:
        remain = budget_s * 0.40 - elapsed()
        n_it = int(remain * 3000)
        n_it = max(1000, min(n_it, 8000))
        h_sa, c5_sa = simulated_anneal(best_h, 0.03, 0.002, n_iters=n_it, eta=0.08)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 4: Multi-start with different n and inits
    n_candidates = sorted(set([best_n, best_n - 4, best_n - 2, best_n + 2, best_n + 4, 72, 84]))
    n_candidates = [x for x in n_candidates if 50 <= x <= 120 and x % 2 == 0]
    for n_try in n_candidates:
        if elapsed() > budget_s * 0.60:
            break
        if n_try == best_n:
            continue
        inits = [
            interpolate_h(best_h, n_try),
            random_binary(n_try, seed=seed + n_try),
        ]
        for h_init in inits:
            if elapsed() > budget_s * 0.60:
                break
            h_init = project_feasible(h_init, n_try)
            h_cd, c5_cd = coord_desc(h_init, n_sweeps=60, grid_n=80)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd
                best_n = n_try
            h_sl = slsqp_smooth(h_cd, 20000, maxiter=200)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl
                best_n = n_try

    # Phase 5: More coordinate descent on best
    if elapsed() < budget_s * 0.72:
        best_h, best_c5 = coord_desc(best_h, n_sweeps=200, grid_n=150)

    # Phase 6: SLSQP direct formulation
    if elapsed() < budget_s * 0.78:
        h_d = slsqp_direct(best_h, maxiter=800)
        c5_d = compute_c5(h_d)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d

    # Phase 7: Fine coordinate descent + short SA
    if elapsed() < budget_s * 0.85:
        best_h, best_c5 = coord_desc(best_h, n_sweeps=150, grid_n=200)
        if elapsed() < budget_s * 0.85:
            h_sa2, c5_sa2 = simulated_anneal(best_h, 0.01, 0.001, n_iters=3000, eta=0.03)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2

    # Phase 8: Final SLSQP polish with escalating T
    if elapsed() < budget_s * 0.95:
        for T in [30000, 100000, 300000]:
            if elapsed() > budget_s * 0.95:
                break
            h_f = slsqp_smooth(best_h, T, maxiter=500)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    # Phase 9: Ultra-fine coordinate descent
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_desc(best_h, n_sweeps=100, grid_n=300)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
