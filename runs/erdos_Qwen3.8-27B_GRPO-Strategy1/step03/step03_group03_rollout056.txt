    #   B_k = (1-h[i+k])*[i+k<n] - h[i-k]*[i-k>=0]
    #   A_k = corr_k - h[i]*B_k (current corr minus h[i]'s contribution)

    # Minimize max(A + B*t) over t in [0,1]
    # Generate all intersection points
    n_len = len(A)
    # Compute all pairwise intersections
    # t_ij = (A_j - A_i) / (B_i - B_j)
    BA = B[:, None] - B[None, :]  # (n_len, n_len)
    AA = A[None, :] - A[:, None]  # (n_len, n_len)

            h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
        h_swap = project_feasible(h_swap, best_n)
        h_opt = cyclic_cd(h_swap, max_sweeps=250)
        c5_opt, _ = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_s = snap_and_reproject(h_opt, best_n)
        h_s = cyclic_cd(h_s, max_sweeps=100)
        c5_s, _ = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 5: Targeted kink shifts
    while time.time() - t0 < budget_s * 0.92:
        h_try = best_h.copy()
        n_kinks = rng.randint(1, 4)
        for _ in range(n_kinks):
            i = rng.randint(0, best_n)
            j = rng.randint(0, best_n)
            if i != j:
                h_try[i], h_try[j] = h_try[j], h_try[i]
        h_try = project_feasible(h_try, best_n)
        h_opt = cyclic_cd(h_try, max_sweeps=300)
        c5_opt, _ = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Final deep polish
    if time.time() - t0 < budget_s * 0.97:
        best_h = cyclic_cd(best_h, max_sweeps=600, tol=1e-14)
        best_c5, _ = compute_c5(best_h)
        h_s = snap_and_reproject(best_h, best_n, eps=0.02)
        h_s = cyclic_cd(h_s, max_sweeps=400, tol=1e-14)
        c5_s, _ = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
