    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, N, tau):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    sc = tau * F
    sc -= sc.max()
    e = np.exp(sc)
    w = e / e.sum()

            grad[:N - M] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:N - M]

    best_c5 = compute_c5(best_h, 2.0 / n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    sizes = [n0, 200, 260]
    sizes = sorted(set(s for s in sizes if s >= 80 and s % 2 == 0))

    for N in sizes:
        if time.time() > deadline - 100:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, best_h)
        h_init = project_sum(h_init, target)

        remain = deadline - time.time()

        # Phase 1: broad smooth optimization
        p1_dead = time.time() + remain * 0.45
        h1, c1 = grad_descent(h_init, N, target, dx, p1_dead,
                              tau0=25.0, tau1=150.0, lr0=0.04, lr1=0.004,
                              iters=8000)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

        # Phase 2: focused high-tau optimization
        if time.time() < deadline - 40:
            p2_dead = time.time() + (deadline - time.time()) * 0.40
            h2, c2 = grad_descent(h1, N, target, dx, p2_dead,
                                  tau0=150.0, tau1=800.0, lr0=0.008, lr1=0.0004,
                                  iters=6000)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
            cur_h = h2
        else:
            cur_h = h1

        # SA refinement
        if time.time() < deadline - 15:
            hs, cs = sa_refine(cur_h, N, target, dx, deadline - 8, n_trials=5000)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs.copy()

        # Peak shaving
        if time.time() < deadline - 6:
            hp, cp = peak_shave(best_h, N, target, dx, deadline - 3, iters=400)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp

    # Random restart at parent size if time allows
    if time.time() < deadline - 80:
        Ns = n0
        dxs = 2.0 / Ns
        target_s = Ns / 2.0
        h_r = best_h + rng.randn(Ns) * 0.08
        h_r = project_sum(h_r, target_s)
        hr_dead = time.time() + (deadline - time.time()) * 0.7
        hr, cr = grad_descent(h_r, Ns, target_s, dxs, hr_dead,
                              tau0=20.0, tau1=500.0, lr0=0.03, lr1=0.002,
                              iters=5000)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()

    # Final ultra-high-tau polish
    Nb = len(best_h)
    if time.time() < deadline - 3:
        dx_b = 2.0 / Nb
        hf, cf = grad_descent(best_h, Nb, Nb // 2, dx_b, deadline - 1,
                              tau0=800.0, tau1=5000.0, lr0=0.003, lr1=0.00005,
                              iters=1500)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
