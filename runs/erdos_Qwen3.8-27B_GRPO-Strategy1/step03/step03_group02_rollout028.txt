def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_gradient(h, N, tau, K):
    """Compute weighted gradient focusing on top K correlations."""
    one_m_h = 1.0 - h
    corr = np.correlate(h, one_m_h, mode='full')
    c_max = float(np.max(corr))
    c5 = c_max * (2.0 / N)
    
    # Select top K
    if K >= len(corr):
        top_idx = np.arange(len(corr))
        c_top = corr
    else:
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    
    # Softmax weights
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5
    w /= ws
    
    # Compute gradient

    center = N - 1

def pg_step(h, N, target, dx, tau, K, lr_max, deadline, n_iters):
    """Projected gradient with backtracking line search."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        grad, c5_val = softmax_gradient(h, N, tau, K)

            best_h = h.copy()

        h_curr_c5 = c5_val

            h_try = h - lr * grad
            h_try = project_sum(h_try, target)

                break

        else:

            h = project_sum(h_try, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    sizes = [100, 120, 150, 200, 250]

    for N in sizes:
        if time.time() > deadline - 70:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)
        remain = deadline - time.time()
        if remain < 70:
            break

        # Phase 1: Gradient optimization, moderate tau
        p1d = time.time() + remain * 0.22
        K1 = min(35, 2 * N - 1)
        h1, c1 = grad_optimize(h_init.copy(), N, target, dx, p1d,
                               tau=40.0, K=K1, lr0=0.025, decay=0.998)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

        # Phase 2: Higher tau, finer steps
        if time.time() < deadline - 50:
            p2d = time.time() + (deadline - time.time()) * 0.20
            K2 = min(10, 2 * N - 1)
            h2, c2 = grad_optimize(h1.copy(), N, target, dx, p2d,
                                   tau=300.0, K=K2, lr0=0.008, decay=0.999)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

        # Phase 3: Binarize + swap search (key innovation)
        if time.time() < deadline - 30:
            h_bin = binarize(h2, N, target)
            h_sw, c_sw = swap_search(h_bin, N, target, dx, deadline=time.time() + 20)
            if c_sw < best_c5:
                best_c5 = c_sw
                best_h = h_sw.copy()
                h2 = h_sw.copy()

        # Phase 4: Micro refinement around current best
        if time.time() < deadline - 10:
            h_m, c_m = micro_refine(h2, N, target, dx,
                                    deadline=deadline - 5, scale=0.03, n_trials=15000)
            if c_m < best_c5:
                best_c5 = c_m
                best_h = h_m.copy()

        # Random restart with gradient + swap
        if time.time() < deadline - 55:
            h_r = h_init * 0.5 + 0.5 * rng.rand(N)
            h_r = project_sum(h_r, target)
            hr_d = time.time() + (deadline - time.time()) * 0.35
            hr, cr = grad_optimize(h_r, N, target, dx, hr_d,
                                   tau=50.0, K=min(20, 2 * N - 1),
                                   lr0=0.02, decay=0.999)
            if time.time() < deadline - 15:
                hr_bin = binarize(hr, N, target)
                hr_sw, cr_sw = swap_search(hr_bin, N, target, dx,
                                           deadline=deadline - 8)
                if cr_sw < cr:
                    cr, hr = cr_sw, hr_sw
            if cr < best_c5:
                best_c5 = cr
                best_h = hr.copy()

    # Final ultra-fine polish at best size
    Nb = len(best_h)
    dx_b = 2.0 / Nb
    if time.time() < deadline - 4:
        hf, cf = grad_optimize(best_h.copy(), Nb, Nb // 2, dx_b,
                               deadline - 1, tau=1500.0, K=min(5, 2 * Nb - 1),
                               lr0=0.002, decay=0.9997)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
