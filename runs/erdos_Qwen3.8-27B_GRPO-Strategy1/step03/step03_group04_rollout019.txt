import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def to_binary(h, N):
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    N2 = N // 2
    if n_ones > N2:
        idx = np.where(h_b == 1)[0]
        # Remove the ones with lowest "autocorrelation support"
        h_b[idx[N2:]] = 0
    elif n_ones < N2:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:N2 - n_ones]] = 1
    return h_b

def binary_local_search(h, N, deadline, rng, n_max_swaps=50000):
    """Greedy + annealing local search for binary sequences."""
    dx = 2.0 / N
    h_b = h.copy()
    corr = compute_corr(h_b)
    best_pk = float(np.max(corr))
    cur_pk = best_pk
    best = h_b.copy()
    T0, T1 = 4.0, 0.002
    n_ones = np.where(h_b == 1)[0]
    n_zeros = np.where(h_b == 0)[0]
    
    for it in range(n_max_swaps):
        if time.time() > deadline:
            break
        # Adaptive temperature
        frac = it / n_max_swaps
        T = T0 * (T1 / T0) ** frac
        
        # Pick random swap
        i = n_ones[rng.randint(len(n_ones))]
        j = n_zeros[rng.randint(len(n_zeros))]
        
        h_b[i], h_b[j] = 0.0, 1.0
        new_corr = compute_corr(h_b)
        new_pk = float(np.max(new_corr))
        delta = new_pk - cur_pk
        
        if delta < 0 or (T > 0.005 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            # Update sets
            n_ones = np.where(h_b == 1)[0]
            n_zeros = np.where(h_b == 0)[0]
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0, 0.0
    
    return best, best_pk * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def adam_refine(h, N, target, deadline):
