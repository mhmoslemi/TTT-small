    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / N)
    center = N - 1
    if K >= len(corr):
        idx = np.arange(len(corr))
        ct = corr.copy()
    else:

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, tgt):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - tgt) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, N, L):
    if L == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(N)
    if L > 0:
        g[L:] += 1.0 - h[:N - L]
        g[:N - L] -= h[L:]
    else:
        M = -L
        g[:N - M] += 1.0 - h[M:]
        g[M:] -= h[:N - M]
    return g

def get_grad(corr, h, N, K, tau):
    if K <= 1:
        ks = int(np.argmax(corr))
        return grad_lag(h, N, ks - (N - 1))
    idx = np.argpartition(corr, -K)[-K:] if K < len(corr) else np.arange(len(corr))
    ct = corr[idx]
    sc = tau * ct
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(N)
    c = N - 1
    for ii in range(len(w)):
        if w[ii] < 1e-6:
            continue
        g += w[ii] * grad_lag(h, N, idx[ii] - c)
    return g

def run_phase(h, N, tgt, dx, dl, ta0, ta1, lr0, lr1, K, n_est=2000):
    bh = h.copy()
    bc = compute_c5(h, dx)
    m1 = np.zeros(N)
    m2 = np.zeros(N)
    B1, B2, ep = 0.9, 0.999, 1e-8
    it = 0
    while time.time() < dl:
        it += 1
        f = min(float(it) / max(n_est, 1), 1.0)
        ta = ta0 * (ta1 / ta0) ** f
        lr = lr0 * (lr1 / lr0) ** f
        corr = np.correlate(h, 1.0 - h, mode='full')
        cv = float(np.max(corr)) * dx
        if cv < bc - 1e-13:
            bc = cv
            bh = h.copy()
        g = get_grad(corr, h, N, K, ta)
        m1 = B1 * m1 + (1 - B1) * g
        m2 = B2 * m2 + (1 - B2) * g * g
        h = project_sum(h - lr * m1 / (1 - B1 ** it) / (np.sqrt(m2 / (1 - B2 ** it)) + ep), tgt)
    return bh, bc

def run_fine(h, N, tgt, dx, dl):
    bh = h.copy()
    bc = compute_c5(h, dx)
    lr = 0.002
    while time.time() < dl:
        corr = np.correlate(h, 1.0 - h, mode='full')
        cv = float(np.max(corr)) * dx
        if cv < bc - 1e-14:
            bc = cv
            bh = h.copy()
        ks = int(np.argmax(corr))
        g = grad_lag(h, N, ks - (N - 1))
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        h = project_sum(h - lr * g / gn, tgt)
        lr *= 0.9998
    return bh, bc

def run_micro(h, N, tgt, dx, dl, seed=2024):
    bh = h.copy()

    rng = np.random.RandomState(seed)
