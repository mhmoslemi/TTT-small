import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_grad(h, T, topk, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        idx = np.sort(idx)
    else:
        idx = np.arange(len(corr))
    c = corr[idx]
    f = c * dx
    m_f = np.max(f)
    exps = np.exp(T * (f - m_f))
    p = exps / np.sum(exps)
    val = m_f + np.log(np.sum(exps)) / T
    grad = np.zeros(n)
    for ii in idx:
        w = p[ii] * dx
        s = ii - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if hi1 > lo1:
            j1 = np.arange(lo1, hi1)
            grad[j1] += w * (1.0 - h[j1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if hi2 > lo2:
            j2 = np.arange(lo2, hi2)
            grad[j2] -= w * h[j2 + s]
    return val, grad

def obj_and_jac(x, T, topk, n, lam):
    h = np.clip(x, 0.0, 1.0)
    base_val, base_grad = smooth_c5_grad(h, T, topk, n)
    s = h.sum()
    penalty = lam * (s - n / 2.0) ** 2
    pgrad = 2.0 * lam * (s - n / 2.0) * np.ones(n)
    return base_val + penalty, base_grad + pgrad

def lbfgsb_opt(h0, T, topk, n, lam, maxiter):
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_and_jac, h0, jac=True, method='L-BFGS-B',
                   bounds=bnds, args=(T, topk, n, lam),
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    h = project_feasible(np.clip(res.x, 0, 1), n)
    return h

def de_obj(x):
    n = len(x)
    h = project_feasible(x, n)
    return compute_c5(h) + abs(h.sum() - n / 2.0) * 5.0
