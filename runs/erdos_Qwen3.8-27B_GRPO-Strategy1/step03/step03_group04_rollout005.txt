```python
import numpy as np
import time

def compute_corr_full(h):
    """Compute all correlation values using FFT."""
    N = len(h)
    # Use FFT for correlation
    # c(L) = sum_i h[i] * (1-h[i+L]) for valid i
    # This is the cross-correlation of h with (1-h)
    v = 1.0 - h
    n_fft = 2 * N
    H = np.fft.rfft(h, n_fft)
    V = np.fft.rfft(v, n_fft)
    # Cross-correlation: c(k) = sum_n h[n+k] * v[n]
    # Using FFT: ifft(conj(H) * V) gives cross-correlation
    corr = np.fft.irfft(np.conj(H) * V, n_fft)
    # Reorder to get the full correlation
    # The full correlation has length 2N-1
    # corr[0] corresponds to shift N-1 (or -N+1, need to check)
    full = np.concatenate([corr[1:N], corr[N_fft-N+1:N_fft]])
    # Actually let me just use np.correlate for correctness
    full = np.correlate(h, v, mode='full')
    return full

def compute_c5(h, N):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def lsq_gradient(h, N, T):
    """Compute gradient of log-sum-exp of correlations."""
    corr = np.correlate(h, 1.0-h, mode='full')
    # Softmax weights
    corr_max = corr.max()
    exp_c = np.exp(T * (corr - corr_max))
    w = exp_c / exp_c.sum()
    
    # Gradient: for each position j,
    # grad[j] = sum_L w[L] * [(1-h[j+L])*1{valid} - h[j-L]*1{valid}]
    grad = np.zeros(N)
    center = N - 1
    for idx in range(2*N-1):
        L = idx - center
        wi = w[idx]
        if wi < 1e-12:
            continue
        if L == 0:
            grad += wi * (1.0 - 2.0*h)
        elif L > 0:
            # (1-h[j+L]) for j=0..N-L-1
            grad[:N-L] += wi * (1.0 - h[L:])
            # -h[j-L] for j=L..N-1
            grad[L:] -= wi * h[:N-L]
        else:
            M = -L
            # (1-h[j-L]) = (1-h[j+M]) for j=0..N-M-1
            grad[:N-M] += wi * (1.0 - h[M:])
            # -h[j+L] = -h[j-M] for j=M..N-1
            grad[M:] -= wi * h[:N-M]
    return grad

def lse_optimize(h0, N, target, deadline, T0, T1, lr0, steps_per_t):
    """Optimize using log-sum-exp loss."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    total_t = time.time()
    n_steps = 0
    while time.time() < deadline:
        T = T0 * (T1/T0) ** (n_steps / max(steps_per_t, 1))
        T = max(T, T0)
        grad = lse_gradient(h, N, T)
        h = h - lr0 * grad
        h = project_sum(h, target)
        n_steps += 1
        if n_steps % 200 == 0:
            c5 = compute_c5(h, N)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        lr0 *= 0.9995
    c5 = compute_c5(h, N)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def binary_swap_search(h0, N, deadline, rng):
    """Local search by swapping 1s and 0s in binary representation."""
    dx = 2.0/N
    h_b = (h0 > 0.5).astype(float)
    # Ensure exactly N/2 ones
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    while len(ones) > N//2:
        i = ones[0]
        h_b[i] = 0
        ones = ones[1:]
        zeros = np.append(zeros, i).astype(int)
    while len(ones) < N//2:
        j = zeros[0]
        h_b[j] = 1
        zeros = zeros[1:]
        ones = np.append(ones, j).astype(int)
    
    best = h_b.copy()
    best_corr = np.correlate(h_b, 1.0-h_b, mode='full').copy()
    best_pk = best_corr.max()
    cur = h_b.copy()
    cur_corr = best_corr.copy()
    cur_pk = best_pk
    
    T = 2.0
    n_total = max(int((deadline - time.time()) * 200), 1000)
    for it in range(n_total):
        if time.time() > deadline:
            break
        T = 2.0 * (0.01/2.0) ** (it/n_total)
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i], cur[j] = cur[j], cur[i]
        new_corr = np.correlate(cur, 1.0-cur, mode='full')
        new_pk = new_corr.max()
        delta = new_pk - cur_pk
        if delta < 0 or (rng.random() < np.exp(-delta/max(T,1e-6))):
            cur_pk = new_pk
            cur_corr = new_corr
            if new_pk < best_pk:
                best_pk = new_pk
                best = cur.copy()
                best_corr = new_corr
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_pk * dx
```

```python
import numpy as np
import time

def compute_c5(h, N):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0, 1).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_grad(h, N, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    wc = np.exp(T * (corr - cmax))
    s = wc.sum()
    if s < 1e-300:
        w = np.zeros(2*N-1); w[np.argmax(corr)] = 1.0
    else:
        w = wc / s
    grad = np.zeros(N)
    center = N - 1
    for idx in range(2*N - 1):
        wi = w[idx]
        if wi < 1e-10:
            continue
        L = idx - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N-L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:N-L]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, cmax

def lse_optimize(h0, N, target, deadline, T_start, T_end, lr0):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    elapsed_total = deadline - time.time()
    total_steps = int(elapsed_total * 30)
    total_steps = max(total_steps, 100)
    lr = lr0
    for it in range(total_steps):
        if time.time() > deadline:
            break
        frac = it / max(total_steps - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        T = max(T, 0.5)
        grad, peak = lse_grad(h, N, T)
        h = h - lr * grad
        h = project_sum(h, target)
        if it % 100 == 0 or it == total_steps - 1:
            pk = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if pk < best_peak:
                best_peak = pk
                best_h = h.copy()
        lr *= 0.9992
    pk = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    if pk < best_peak:
        best_peak = pk
        best_h = h.copy()
    return best_h, best_peak * (2.0 / N)

def binary_local_search(h0, N, deadline, rng):
    dx = 2.0 / N
    h_b = (h0 > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    while len(ones) > N // 2:
        i = ones[0]; h_b[i] = 0; ones = ones[1:]
        zeros = np.hstack([zeros, [i]]).astype(int)
    while len(ones) < N // 2:
        j = zeros[0]; h_b[j] = 1; zeros = zeros[1:]
        ones = np.hstack([ones, [j]]).astype(int)
    best = h_b.copy()
    best_pk = float(np.max(np.correlate(h_b, 1.0 - h_b, mode='full')))
    cur = h_b.copy()
    cur_pk = best_pk
    n_total = max(int((deadline - time.time()) * 80), 500)
    for it in range(n_total):
        if time.time() > deadline:
            break
        T = 3.0 * (0.003 / 3.0) ** (it / n_total)

        j = zeros[rng.randint(len(zeros))]
```
