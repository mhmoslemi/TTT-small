import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    sc = tau * corr * dx
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), float(np.max(corr) * dx)
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-5:
            continue
        L = ix - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    grad *= dx
    return grad, float(np.max(corr) * dx)

def smax_descent(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, n_iters):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline and it < n_iters:
        batch = min(50, n_iters - it)
        for _ in range(batch):
            if time.time() >= deadline:
                break
            frac = min(it / max(n_iters, 1), 1.0)
            tau = tau0 * (tau1 / tau0) ** frac
            lr = lr0 * (lr1 / lr0) ** frac
            grad, c5v = smax_grad(h, N, tau)
            if c5v < best_c5 - 1e-13:
                best_c5 = c5v
                best_h = h.copy()
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad * grad
            it += 1
            mh = m / (1 - b1 ** it)
            vh = v / (1 - b2 ** it)
            h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c5

def pair_cd(h, N, target, dx, n_sweeps, deadline, rng):

                best_h = project_sum(best_h, target)
                best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5
