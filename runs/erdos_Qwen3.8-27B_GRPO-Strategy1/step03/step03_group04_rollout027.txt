def project_sum(h, target):

    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid

    v = np.zeros(N)

import numpy as np
import time

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N, rng, noise=0.05):
    scores = h + rng.randn(N) * noise
    threshold = np.sort(scores)[N // 2]
    h_b = (scores >= threshold).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > N // 2:
        idx = np.where(h_b == 1)[0]
        pick = rng.choice(idx, n_ones - N // 2, replace=False)
        h_b[pick] = 0
    elif n_ones < N // 2:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:N // 2 - n_ones]] = 1
    return h_b

def local_search_binary(h_b, N, deadline, rng):

def adam_opt(h, N, target, deadline, k=50, lr0=0.04, decay=0.99):
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    center = N - 1
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wk * (1.0 - h[:N - L])
                grad[:N - L] -= wk * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wk * (1.0 - h[M:])
                    grad[M:] -= wk * h[:N - M]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 60 == 0:
            lr *= decay
    return best_h, best_peak * (2.0 / N)
