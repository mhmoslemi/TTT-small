def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    # Binary search for the shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5(h, N, topK, smooth_tau):
    """Compute gradient of C5 w.r.t. h using active-set weighting."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr)) * dx
    center = N - 1
    nc = len(corr)
    
    # Weight: combination of softmax and top-K
    if smooth_tau > 0:
        sc = smooth_tau * corr
        sc -= sc.max()
        w_sm = np.exp(sc)
        w_sm /= w_sm.sum()
    else:
        w_sm = np.ones(nc) / nc
    
    k = max(1, min(topK, nc))

    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k

def block_grad(h, N, topK, smooth_tau, block_size=4):

            e = s + block_size

        if n_blocks * block_size < N:
            g_smooth[n_blocks*block_size:] = g[n_blocks*block_size:]
        g = 0.5 * g + 0.5 * g_smooth
    return g, c5
