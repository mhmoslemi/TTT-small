```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any(): break
            room = np.zeros(n); room[mask] = 1.0 - h[mask]
            total = room.sum()
            if total < 1e-14: break
            h[mask] += room[mask] * (diff / total)
        else:
            mask = h > 0.0
            if not mask.any(): break
            total = h[mask].sum()
            if total < 1e-14: break
            h[mask] += h[mask] * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_erdos(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return (2.0 / len(h)) * val

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_erdos, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    n = len(h)
    saved = h[j]
    best_val = np.inf
    best_hj = saved
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            room = np.zeros(n)
            room[mask] = 1.0 - h[mask]
            total = room.sum()
            if total < 1e-14:
                break
            h[mask] += room[mask] * (diff / total)
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            total = h[mask].sum()
            if total < 1e-14:
                break
            h[mask] += h[mask] * (diff / abs(total))
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_erdos(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return (2.0 / len(h)) * val

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_erdos, h.copy(), args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

            for t in ts:
                h[j] = t
                corr = np.correlate(h, 1.0 - h, mode='full')
                val = corr.max()
                if val < best_val:
                    best_val = val
                    best_t = t
            h[j] = best_t
            # Maintain sum constraint by adjusting
            diff = n / 2.0 - h.sum()
            if abs(diff) > 1e-12:

        if max_change < 1e-14:
            break
```

```python
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            room = np.zeros(n)
            room[mask] = 1.0 - h[mask]
            total = room.sum()
            if total < 1e-14:
                break
            h[mask] += room[mask] * (diff / total)
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            total = h[mask].sum()
            if total < 1e-14:
                break
            h[mask] += h[mask] * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_erdos(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return (2.0 / len(h)) * val

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_erdos, h.copy(), args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, max_sweeps=5000, time_stop=None, n_grid=20):
    n = len(h)
    best_c5, _ = compute_c5(h)
    ts = np.linspace(0.0, 1.0, n_grid)
    for sweep in range(max_sweeps):
        if time_stop is not None and time.time() > time_stop:
            break
        max_change = 0.0
        for j in range(n):
            saved = h[j]
            best_val = np.inf
            best_t = saved
            for t in ts:
                h[j] = t
                corr = np.correlate(h, 1.0 - h, mode='full')
                val = corr.max()
                if val < best_val:
                    best_val = val
                    best_t = t
            h[j] = best_t
            max_change = max(max_change, abs(h[j] - saved))
        h = project_feasible(h, n)
        c5_now, _ = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
        if max_change < 1e-15:
            break
    return h, best_c5

def cd_refined(h, time_stop, n_grid=30):
    """Multiple CD passes with increasing grid resolution"""
    best_c5, _ = compute_c5(h)
    for ng in [n_grid, int(n_grid * 1.5), int(n_grid * 2)]:
        if time.time() > time_stop:
            break
        h, c5 = coord_descent(h, max_sweeps=8000, time_stop=time_stop, n_grid=ng)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Quick SLSQP warm-up
    if time.time() - t_start < budget_s * 0.05:
        h_try = slsqp_smooth(best_h, T=2000, maxiter=200)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Primary coordinate descent (handles non-smooth max exactly)
    ts_stop = t_start + budget_s * 0.25
    h_cd, c5_cd = cd_refined(best_h, ts_stop, n_grid=20)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd

    # Phase 3: SLSQP polish + CD alternation
    n_alt = 0
    ts_stop = t_start + budget_s * 0.50
    while time.time() < ts_stop:
        h_try = slsqp_smooth(best_h, T=10000 + 5000 * n_alt, maxiter=300)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        h_cd, c5_cd = coord_descent(best_h, max_sweeps=4000, time_stop=ts_stop, n_grid=25)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        n_alt += 1
        if n_alt >= 5:
            break

    # Phase 4: Try nearby resolutions
    ts_stop = t_start + budget_s * 0.78
    for dn in [-2, 2, -4, 4, -6, 6]:
        n_try = best_n + dn
        if n_try < 50 or n_try % 2 != 0:
            continue
        if time.time() > ts_stop:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_try = slsqp_smooth(h_interp, T=5000, maxiter=200)
        h_try, c5_try = coord_descent(h_try, max_sweeps=3000, time_stop=ts_stop, n_grid=22)
        h_try = slsqp_smooth(h_try, T=50000, maxiter=200)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            best_n = n_try

    # Phase 5: Final ultra-fine coordinate descent
    ts_stop = t_start + budget_s * 0.92
    h_final, c5_final = cd_refined(best_h, ts_stop, n_grid=35)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final

    # Phase 6: Final SLSQP with very high T
    if time.time() - t_start < budget_s * 0.97:
        h_try = slsqp_smooth(best_h, T=150000, maxiter=500)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        # One more CD pass at max resolution
        h_cd2, c5_cd2 = coord_descent(best_h, max_sweeps=5000,
                                       time_stop=t_start + budget_s * 0.99, n_grid=50)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
