    corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t > 1e-14:
                h += room * (diff / t)
        else:
            room = h.copy()
            t = room.sum()
            if t > 1e-14:
                h += room * (diff / t)
    return np.clip(h, 0.0, 1.0)

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def lse_pnorm_obj(h, T, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    lse = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    pn = np.sum(corr**p) ** (1.0 / p)
    return 0.5 * (lse + pn)

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def slsqp_opt(h0, T, n, maxiter, ftol, use_pnorm=False, p=50):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    if use_pnorm:
        res = minimize(lse_pnorm_obj, h0, args=(T, p), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    else:
        res = minimize(lse_obj, h0, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, max_rounds, t_start, t_limit):
    best = h.copy()
    best_c5 = compute_c5(best)
    step_sizes = [0.08, 0.04, 0.02, 0.01, 0.005]
    for rnd in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        for si in range(min(si + 1, len(step_sizes)) if False else len(step_sizes)):
            delta = step_sizes[si] * (1 if rnd % 2 == 0 else -1)
            for j in range(0, n, max(1, n // 80)):
                if time.time() > t_limit:
                    return best, best_c5
                ht = best.copy()
                ht[j] = np.clip(ht[j] + delta, 0, 1)
                ht = make_feasible(ht, n)
                c = compute_c5(ht)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    best = ht
                    improved = True
        if not improved and rnd > 40:
            break
    return best, best_c5

def basin_hopping(h, n, t_start, t_limit, seed_offset=0):
    rng = np.random.RandomState(seed_offset)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    max_iter = int((t_limit - t_start) / 4.0)
    max_iter = max(max_iter, 20)
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        step = 0.10 * (1.0 - it / max(1, max_iter)) + 0.02
        pt = it % 5
        if pt == 0:
            hp = cur + step * rng.randn(n)
        elif pt == 1:
            hp = cur.copy()
            bl = max(2, n // 10)
            i1 = rng.randint(0, n - bl)
            i2 = rng.randint(0, n - bl)
            tmp = hp[i1:i1+bl].copy()
            hp[i1:i1+bl] = hp[i2:i2+bl]
            hp[i2:i2+bl] = tmp
        elif pt == 2:
            hp = cur.copy()
            nf = max(1, n // 6)
            idx = rng.choice(n, nf, replace=False)
            hp[idx] = 1.0 - hp[idx]
        elif pt == 3:
            hp = cur.copy()
            idx = rng.choice(n, max(1, n // 4), replace=False)
            hp[idx] = hp[idx] + rng.uniform(-0.3, 0.3, len(idx))
        else:
            hp = cur + 0.25 * rng.randn(n)
        hp = make_feasible(hp, n)

            hn = hp
            cv = compute_c5(hp)
        temp = 0.004 * (1.0 - it / max(1, max_iter)) + 0.0002
        if cv < cur_c5 or rng.rand() < np.exp(-(cv - cur_c5) / temp):
            cur, cur_c5 = hn, cv
            if cv < best_c5:
                best, best_c5 = hn.copy(), cv
    return best, best_c5

    best_c5 = compute_c5(best_h)
