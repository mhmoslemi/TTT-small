        it_count += 1
        f = min(it_count / 1500.0, 1.0)
        K = max(1, int(12 - 9 * f))
        g, c5 = subgrad(best_h, N0, K)
        patience += 1 if c5 >= best_c5 - 1e-13 else 0
        m = 0.85 * m + 0.15 * g
        h_try = norm_sum(best_h - lr * m, target0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
            patience = 0
        elif patience > 80 and rng.random() < 0.005:
            h_try = norm_sum(best_h + rng.randn(N0) * 0.008, target0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try
                patience = 0
        lr *= 0.9997
        if lr < 1e-9:
            lr = 0.015

    # Phase 2: Coordinate descent (400s)
    p2_dead = t0 + min(700, budget_s - 100)
    best_h, best_c5 = coord_sweep(best_h.copy(), N0, target0, p2_dead)

    # Phase 3: Fine gradient polish (200s)
    p3_dead = t0 + min(950, budget_s - 30)
    m = np.zeros(N0)
    lr = 0.008
    while time.time() < p3_dead:
        g, c5 = subgrad(best_h, N0, 3)
        m = 0.90 * m + 0.10 * g
        h_try = norm_sum(best_h - lr * m, target0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        lr *= 0.9999
        if lr < 1e-10:
            break

    # Phase 4: Final coordinate refinement
    if time.time() < deadline - 5:
        best_h, best_c5 = coord_sweep(best_h.copy(), N0, target0, deadline - 3)

    # Phase 5: Try slightly larger grid
    if time.time() < deadline - 15 and best_c5 < 0.3810:
        N2 = N0 + 20
        x0 = np.linspace(0, 2, N0, endpoint=False)
        x2 = np.linspace(0, 2, N2, endpoint=False)
        h2 = np.interp(x2, x0, best_h)
        h2 = norm_sum(h2, N2 / 2.0)
        c5_2 = compute_c5(h2)
        m2 = np.zeros(N2)
        lr2 = 0.015
        d2 = deadline - 3
        while time.time() < d2:
            g2, c5v = subgrad(h2, N2, 4)
            if c5v < c5_2:
                c5_2 = c5v
            m2 = 0.85 * m2 + 0.15 * g2
            h2 = norm_sum(h2 - lr2 * m2, N2 / 2.0)
            lr2 *= 0.999
            if lr2 < 1e-10:
                break
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
