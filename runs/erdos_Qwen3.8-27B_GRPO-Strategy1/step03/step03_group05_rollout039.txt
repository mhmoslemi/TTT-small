import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= h * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    if T < 1:
        return float(m * dx)
    return float((m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_refine(h, n, obj, args, mi, ft):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': mi, 'ftol': ft})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h

def coord_descent(h, n, sweeps=10):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    for _ in range(sweeps):
        improved = False
        for j in np.random.permutation(n):
            ho = h[j]
            B = np.zeros(2 * n - 1)
            B[n-1-j : 2*n-1-j] += (1.0 - h)
            B[j : j+n] -= h[::-1]
            A = corr - ho * B
            bv = np.max(corr)
            bh = ho
            cands = np.array([0.0, 1.0, ho])
            t0 = np.argsort(A)[-5:]
            t1 = np.argsort(A + B)[-5:]
            for ii in t0:
                for jj in t1:
                    if ii == jj:
                        continue
                    d = B[jj] - B[ii]
                    if abs(d) > 1e-14:
                        hc = (A[jj] - A[ii]) / d
                        if -0.01 <= hc <= 1.01:
                            cands = np.append(cands, hc)
            cands = np.clip(cands, 0, 1)
            for hc in cands:
                v = np.max(A + hc * B)
                if v < bv:
                    bv = v
                    bh = hc
            if abs(bh - ho) > 1e-10:
                corr += (bh - ho) * B
                h[j] = bh
                improved = True
        if np.max(corr) * dx < best_c5:
            best_c5 = float(np.max(corr) * dx)
        if not improved:
            break
    return h, best_c5

    h0 = project_feasible(h0, n)
    
    best_h = h0.copy()

For coordinate j, the sensitivity of corr[k] (shift k, 0-indexed) to h[j]:
- If 0 ≤ j+k < n: h[j] contributes to term i=j in corr[k], derivative = (1-h[j+k])
- If 0 ≤ j-k < n: h[j] contributes to term i=j-k in corr[k] as the "1-h" part, derivative = -h[j-k]

So:
dcorr[k]/dh[j] = (1-h[j+k]) * [0≤j+k<n] - h[j-k] * [0≤j-k<n]

In array form:
- Left part: for k from -j to n-1-j, dcorr[k+n-1] += (1-h[j+k])
  j+k goes from 0 to n-1, so the values are (1-h[0]), (1-h[1]), ..., (1-h[n-1]) = (1-h)
  The indices in corr are from -j+n-1 = n-1-j to n-1-j+n-1 = 2n-2-j
  So: B[n-1-j : 2n-1-j] += (1-h)  [slice has length n]

  j-k goes from n-1 to 0, so the values are h[n-1], h[n-2], ..., h[0] = h[::-1]
  The indices in corr are from j-n+1+n-1 = j to j+n-1
  So: B[j : j+n] -= h[::-1]  [slice has length n]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= h * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    if T < 1:
        return float(m * dx)
    return float((m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_refine(h, n, obj, args, mi, ft):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': mi, 'ftol': ft})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, n, sweeps=10):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    for _ in range(sweeps):
        improved = False
        order = np.random.permutation(n)
        for j in order:
            ho = h[j]
            B = np.zeros(2 * n - 1)
            B[n - 1 - j: 2 * n - 1 - j] += (1.0 - h)
            B[j: j + n] -= h[::-1]
            A = corr - ho * B
            bv = np.max(corr)
            bh = ho
            cands = [0.0, 1.0, ho]
            t0 = np.argsort(A)[-5:]
            t1 = np.argsort(A + B)[-5:]
            for ii in t0:
                for jj in t1:
                    if ii == jj:
                        continue
                    d = B[jj] - B[ii]
                    if abs(d) > 1e-14:
                        hc = (A[jj] - A[ii]) / d
                        if -0.01 <= hc <= 1.01:
                            cands.append(hc)
            cands = np.clip(np.array(cands), 0.0, 1.0)
            for hc in cands:
                v = np.max(A + hc * B)
                if v < bv:
                    bv = v
                    bh = hc
            if abs(bh - ho) > 1e-12:
                corr += (bh - ho) * B
                h[j] = bh
                improved = True
        sc5 = float(np.max(corr) * dx)
        if sc5 < best_c5:
            best_c5 = sc5
        if not improved:
            break
    return h, best_c5

def make_block_start(n, pattern):

    if pattern == 0:

    elif pattern == 3:

            h[blk: blk + 6] = 1.0
    return project_feasible(h, n)
