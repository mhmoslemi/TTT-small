def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    cs = np.clip(corr - m, -50.0, 0.0)
    return (m + np.log(np.sum(np.exp(T * cs))) / T) * dx

def grad_lag(h, L):
    """Gradient of correlation at lag L (signed, where L=k-(n-1) in full mode)"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)

        idx1 = np.arange(0, n - L)
        if len(idx1) > 0:

        idx2 = np.arange(L, n)
        if len(idx2) > 0:
            g[idx2] -= h[idx2 - L]
    else:
        Lp = -L
        idx1 = np.arange(Lp, n)
        if len(idx1) > 0:
            g[idx1] += (1.0 - h[idx1 - Lp])
        idx2 = np.arange(0, n - Lp)
        if len(idx2) > 0:
            g[idx2] -= h[idx2 + Lp]
    return g * dx

    """Projected gradient with smooth max and momentum"""
    n = len(h)
    h_curr = h.astype(float).copy()
    v = np.zeros(n)  # momentum
    best_h = h_curr.copy()
    best_val = compute_c5(h_curr)

        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        m = np.max(corr)
        cs = np.clip(corr - m, -50.0, 0.0)
        w = np.exp(T * cs)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            w /= wsum

        grad = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-10:
                continue
            L = k - (n - 1)
            grad += w[k] * grad_lag(h_curr, L)
        
        # Nesterov momentum
        v = momentum * v - 0.05 * grad
        h_try = make_feasible(h_curr + v, n)
        
        # Check improvement
        val_try = compute_c5(h_try)
        if val_try < best_val:
            best_val = val_try
            best_h = h_try.copy()
        h_curr = h_try
    
    return best_h, best_val

def slsqp_refine(h0, T, n, maxiter=500):

                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h0
