def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_gradient(h, N, tau, K):
    """Log-sum-exp smoothed gradient with top-K focus."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)

        top_idx = np.arange(len(corr))
        c_top = corr
    else:

        c_top = corr[top_idx]
    m = c_top.max()

    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1

        if wi < 1e-8:

        shift = rng.choice([-2, -1, 1, 2])
        end = start + length
        new_start = start + shift
        new_end = end + shift
        if new_start < 0 or new_end > N:
            continue
        t = bh.copy()
        # Circular shift of block within the array
        # Actually, let's just add a triangular bump
        amp = rng.uniform(0.01, 0.05) * rng.choice([1, -1])
        for k in range(length):
            fr = 1.0 - abs(k - length/2) / (length/2)
            t[start+k] += amp * fr
        t = project_sum(t, tgt)
        c = compute_c5(t, dx)
        if c < bc - 1e-12:
            bc, bh = c, t
    return bh, bc

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_grad(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / N)
    if K >= len(corr):
        top_idx = np.arange(len(corr))
        c_top = corr
    else:
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    sc = tau * (c_top - c_top.max())
    sc = np.clip(sc, -50, 0)
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5
    w /= ws
    grad = np.zeros(N)
    c = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-8:
            continue
        L = top_idx[ix] - c
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def gd_cycle(h, N, tgt, dx, deadline, tau0, tau1, lr0, lr1, K, n_it):
    bh, bc = h.copy(), compute_c5(h, dx)
    m, v = np.zeros(N), np.zeros(N)
    b1, b2, ep = 0.9, 0.999, 1e-8
    it = 0
    while time.time() < deadline and it < n_it:
        fr = min(it / max(n_it, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** fr
        lr = lr0 * (lr1 / lr0) ** fr
        g, cv = topk_grad(h, N, tau, K)
        if cv < bc - 1e-12:
            bc, bh = cv, h.copy()
        it += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        h = project_sum(h - lr * (m / (1 - b1**it)) / (np.sqrt(v / (1 - b2**it)) + ep), tgt)
    return bh, bc

def coord_cd(h, N, tgt, dx, deadline, grid):
    bh, bc = h.copy(), compute_c5(h, dx)
    rng = np.random.RandomState(314 + N)
    no_imp = 0
    while time.time() < deadline:
        imp = False
        for i in rng.permutation(N):
            if time.time() > deadline:
                break
            for d in grid:
                t = bh.copy()
                t[i] += d
                t = project_sum(t, tgt)
                c = compute_c5(t, dx)
                if c < bc - 1e-12:
                    bc, bh, imp = c, t.copy(), True
                    break
        no_imp = no_imp + 1 if not imp else 0
        if no_imp >= 6:
            break
    return bh, bc

def pair_cd(h, N, tgt, dx, deadline, n_tr):
    bh, bc = h.copy(), compute_c5(h, dx)

        t = project_sum(t, tgt)
        c = compute_c5(t, dx)
        if c < bc - 1e-12:

    return bh, bc
