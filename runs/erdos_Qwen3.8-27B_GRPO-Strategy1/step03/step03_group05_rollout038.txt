import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_min_v(b, c, old_val):
    d = b - c * old_val
    g0 = np.max(d)
    g1 = np.max(d + c)
    best_v = 0.0
    best_g = g0
    if g1 < best_g:
        best_g = g1
        best_v = 1.0
    m = len(d)
    dd = d[None, :] - d[:, None]
    cc = c[:, None] - c[None, :]
    valid = np.abs(cc) > 1e-14
    vstar = np.full((m, m), np.nan)
    vstar[valid] = dd[valid] / cc[valid]
    vstar = vstar.flatten()
    vstar = vstar[np.isfinite(vstar) & (vstar > 1e-10) & (vstar < 1.0 - 1e-10)]
    if len(vstar) > 0:
        g_vals = np.max(d[None, :] + c[None, :] * vstar[:, None], axis=1)
        bi = np.argmin(g_vals)
        if g_vals[bi] < best_g:
            best_g = g_vals[bi]
            best_v = vstar[bi]
    return best_v

def coord_descent(h, n_sweeps=100, tol=1e-13):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    s_range = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        max_change = 0.0
        for i in range(n):
            old = h[i]
            a = np.zeros(2 * n - 1)
            vp = (i + s_range >= 0) & (i + s_range < n)
            vm = (i - s_range >= 0) & (i - s_range < n)
            a[vp] += (1.0 - h[i + s_range[vp]])
            a[vm] -= h[i - s_range[vm]]
            new_v = coord_min_v(corr, a, old)
            delta = new_v - old
            if abs(delta) > 1e-15:
                h[i] = new_v
                corr += a * delta
                max_change = max(max_change, abs(delta))
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        if max_change < tol:
            break
    return h

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_optimize(h, n, T=500, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(softmax_obj, h, args=(T, n), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

        if abs(diff) < 1e-14:

def coord_min_v(d, c):
    """Minimize max_s(d[s] + c[s]*v) over v in [0,1]"""
    # Grid search + ternary refinement
    grid = np.linspace(0, 1, 33)
    g_grid = np.maximum.reduce(d[None,:]*np.ones((1,33)) + c[None,:]*grid[:,None])
    i = np.argmin(g_grid)
    lo = max(0.0, grid[max(0, i-1)])

        m2 = hi - (hi - lo) / 3.0
        if np.max(d + c * m1) <= np.max(d + c * m2):

        else:

    s_range = np.arange(-(n-1), n)

        for i in range(n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_min_v(d, c):
    grid = np.linspace(0, 1, 33)
    g_grid = np.max(d[None, :] + c[None, :] * grid[:, None], axis=1)
    i = np.argmin(g_grid)
    lo = max(0.0, grid[max(0, i - 1)])
    hi = min(1.0, grid[min(32, i + 1)])
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(d + c * m1) <= np.max(d + c * m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n_sweeps=100, tol=1e-12):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    s_range = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        max_change = 0.0
        for i in range(n):
            old = h[i]
            a = np.zeros(2 * n - 1)
            vp = (i + s_range >= 0) & (i + s_range < n)
            vm = (i - s_range >= 0) & (i - s_range < n)
            a[vp] += (1.0 - h[i + s_range[vp]])
            a[vm] -= h[i - s_range[vm]]
            d = corr - a * old
            new_v = coord_min_v(d, a)
            delta = new_v - old
            if abs(delta) > 1e-14:
                h[i] = new_v
                corr += a * delta
                max_change = max(max_change, abs(delta))
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        if max_change < tol:
            break
    return h

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp_opt(h, n, T=500, maxiter=200, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(softmax_obj, h, args=(T, n), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def structured_init(n, kind, rng):
    if kind == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif kind == 1:
        h = np.zeros(n)
        h[n // 4:3 * n // 4] = 1.0
    elif kind == 2:
        h = np.zeros(n)
        h[:n // 3] = 1.0
        h[2 * n // 3:] = 1.0
    elif kind == 3:
        h = np.zeros(n)
        h[::2] = 1.0
    else:
        h = np.ones(n) * 0.5
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed + 7)

    # Phase 1: Try structured inits + CD polish, keep best

            h_cd = coord_descent(h_init, n_sweeps=150)
            c = compute_c5(h_cd)

    # Phase 2: Intensive CD on best

        elif r < 0.88:

        co = compute_c5(ho)

        cf = compute_c5(hf)
        if cf < best_c5:

        hf = slsqp_opt(best_h, best_n, 30000, 400, 1e-15)
        hf = coord_descent(hf, n_sweeps=200, tol=1e-14)
        cf = compute_c5(hf)
        if cf < best_c5:

        # Extra micro-CD

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
