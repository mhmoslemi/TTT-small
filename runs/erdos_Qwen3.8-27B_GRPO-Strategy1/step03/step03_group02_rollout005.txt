from scipy.optimize import linprog

def lp_step(h, N, dx, target, K):
    """One LP-based step to reduce C5."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    top_idx = np.argsort(corr)[-K:]
    
    # Compute gradients for top K lags
    grads = []
    for k in top_idx:
        L = int(k) - center
        g = np.zeros(N)
        if L == 0:
            g = 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += (1.0 - h[:N-L])
            g[:N-L] -= h[L:]
        else:
            M = -L
            g[:N-M] += (1.0 - h[M:])
            g[M:] -= h[:N-M]
        grads.append(g)
    
    # LP: min t s.t. g_k·h - t/dx <= c_k, 0<=h<=1, sum(h)=target
    # Variables: [h(0), ..., h(N-1), t]
    # Objective: [0, ..., 0, 1]
    
    c_obj = np.zeros(N + 1)
    c_obj[-1] = 1.0
    
    A_ineq = []
    b_ineq = []
    
    for k_idx in range(len(top_idx)):
        g = grads[k_idx]
        c_k = corr[top_idx[k_idx]] - np.sum(g * h)
        # g·h - t/dx <= c_k
        row = np.zeros(N + 1)
        row[:N] = g
        row[-1] = -1.0 / dx
        A_ineq.append(row)
        b_ineq.append(c_k)
    
    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    
    # Bounds: 0 <= h[j] <= 1, t unbounded below (but will be positive)
    bounds = [(0, 1)] * N + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, 
                  A_eq=A_eq, b_eq=[target], 
                  bounds=bounds, method='highs')
    
    if res.success:
        h_lp = res.x[:N]
        return h_lp
    return None

                tmp = trial[i1:i1 + L].copy()
                trial[i1:i1 + L] = trial[i2:i2 + L]
                trial[i2:i2 + L] = tmp
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_h = trial
    return best_h, best_c5

def optimize_size(h_init, N, target, deadline, rng, n_lp):
    dx = 2.0 / N
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    it = 0
    while time.time() < deadline - 30:
        # LP step
        h_lp = do_lp_step(h, dx, target, min(n_lp, 2 * N - 1))
        if h_lp is not None and time.time() < deadline - 10:
            h_new, c_new = line_search(h, h_lp, dx, target)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
            h = h_new
        # Gradient step
        tau = 30.0 + 50.0 * (it / 50.0)
        K = min(25, 2 * N - 1)
        lr = 0.03 / (1.0 + 0.02 * it)
        h_g = softmax_grad(h, tau, K, lr, target)
        c_g = compute_c5(h_g, dx)
        if c_g < best_c5:
            best_c5 = c_g
            best_h = h_g.copy()
        h = h_g
        it += 1
        # Periodic local search
        if it % 25 == 0 and time.time() < deadline - 60:
            h_ls, c_ls = local_search(h, dx, target, 300, time.time() + 20, rng)
            if c_ls < best_c5:
                best_c5 = c_ls
                best_h = h_ls.copy()
            h = h_ls
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)

    # Phase 1: Main optimization at original size
    h1, c1 = optimize_size(h0, n0, target0, deadline - 200, rng, 80)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: Intensive local search at best solution
    hn, cx = local_search(best_h, 2.0 / len(best_h), len(best_h) / 2.0,
                          8000, deadline - 80, rng)
    if cx < best_c5:
        best_c5 = cx
        best_h = hn

    # Phase 3: Try size 200
    N2 = 200
    if time.time() < deadline - 60:
        dx2 = 2.0 / N2
        target2 = N2 / 2.0
        xN = np.linspace(0.0, 2.0, N2, endpoint=False)
        x0 = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        h2 = np.interp(xN, x0, best_h)
        h2 = project_sum(h2, target2)
        h2, c2 = optimize_size(h2, N2, target2, deadline - 30, rng, 60)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2
        h2, c2 = local_search(h2, dx2, target2, 2000, deadline - 5, rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 4: Final LP polish at best size
    Nb = len(best_h)
    dxb = 2.0 / Nb
    tb = Nb / 2.0
    if time.time() < deadline - 3:
        for _ in range(20):
            if time.time() > deadline - 2:
                break
            h_lp = do_lp_step(best_h, dxb, tb, 50)
            if h_lp is not None:
                h_new, c_new = line_search(best_h, h_lp, dxb, tb)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, min(best_c5, final_c5), len(best_h)
