    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_grad(h, T):
    """Compute smoothed objective value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    Z = np.sum(exps)
    val = (m + np.log(Z) / T) * dx
    weights = exps / Z  # length 2n-1
    
    # Group by |k|
    # corr index j corresponds to shift k = j - (n-1)
    W = np.zeros(n)  # W[p] for p = 0, 1, ..., n-1
    W[0] = weights[n-1]  # k=0
    for p in range(1, n):
        W[p] = weights[n-1+p] + weights[n-1-p]
    
    grad = np.zeros(n)
    for p in range(n):
        wp = W[p]
        if wp < 1e-12:
            continue
        if p == 0:
            grad += wp * (1.0 - 2.0 * h)
        else:
            grad[p:] += wp * (1.0 - h[:n-p])
            grad[:n-p] -= wp * h[p:]
    grad *= dx
    return val, grad

def projected_grad_descent(h, T, n_iters, lr_init=0.01):
    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_val, _ = compute_c5(h_cur)
    val_smooth, grad = smooth_grad(h_cur, T)
    lr = lr_init
    for it in range(n_iters):
        # Backtracking line search
        alpha = lr
        for bs in range(30):
            h_trial = project_feasible(h_cur - alpha * grad, n)
            val_trial, _ = compute_c5(h_trial)
            if val_trial < best_val:

                best_h = h_trial.copy()
                h_cur = h_trial

            lr *= 0.7
            if lr < 1e-8:

        # Recompute gradient
        val_smooth, grad = smooth_grad(h_cur, T)
        # Increase T gradually
        T = min(T * 1.01, 100000)
    return best_h, best_val

        else:
