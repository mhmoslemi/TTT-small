import numpy as np
from scipy.optimize import minimize, differential_evolution

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def fourier_to_h(params, n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    nm = len(params) // 2
    for m in range(nm):
        h += params[2*m] * np.cos(np.pi * (m+1) * x) + params[2*m+1] * np.sin(np.pi * (m+1) * x)
    return h

    penalty = np.sum(np.maximum(0, -h)**2 + np.maximum(0, h-1)**2) * 10
    h = project_feasible(h, n)
    return compute_c5(h) + penalty

                    ht[j] = 1

                    c = compute_c5(ht)
                    if c < best_c5:
                        best = ht
                        best_c5 = c
                        improved = True
    return best, best_c5

def slsqp_smooth(h0, T, n, maxiter=500, ftol=1e-15):
    cons = [{'type': 'eq', 'fun': sum_eq}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(soft_max_c5, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def subgradient_step(h, n, step_size=0.01):
    """One subgradient step on the true max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    k = k_star - (n - 1)
    # Subgradient: d(corr[k_star])/dh[j]
    j_lo = max(0, -k)
    j_hi = min(n, n - k)
    grad = np.zeros(n)
    idx1 = np.arange(j_lo, j_hi)
    grad[idx1] += (1.0 - h[idx1 + k])
    j0_lo = max(0, j_lo + k)
    j0_hi = min(n, j_hi + k)
    idx2 = np.arange(j0_lo, j0_hi)
    grad[idx2] -= h[idx2 - k]
    
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h0, T, n, maxiter=500, ftol=1e-15):
    cons = [{'type': 'eq', 'fun': sum_eq}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(soft_max_c5, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def fourier_to_h(params, n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    nm = len(params) // 2
    for m in range(nm):
        h += params[2*m] * np.cos(np.pi * (m+1) * x)
        h += params[2*m+1] * np.sin(np.pi * (m+1) * x)
    return h

def fourier_obj(params):
    n = 160
    h = fourier_to_h(params, n)
    penalty = np.sum(np.maximum(0, -h)**2 + np.maximum(0, h-1)**2) * 5.0
    h = project_feasible(h, n)
    return compute_c5(h) + penalty

def subgradient_step(h, n, step_size):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    k = k_star - (n - 1)
    grad = np.zeros(n)
    j1_lo = max(0, -k)
    j1_hi = min(n, n - k)
    idx1 = np.arange(j1_lo, j1_hi)
    if len(idx1) > 0:
        grad[idx1] += (1.0 - h[idx1 + k])
    j2_lo = max(0, k)
    j2_hi = min(n, n + k)
    idx2 = np.arange(j2_lo, j2_hi)
    if len(idx2) > 0:
        grad[idx2] -= h[idx2 - k]
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

def binary_local_search(h0, n, max_iters=500):
    hb = (h0 > 0.5).astype(float)
    hb = project_feasible(hb, n)
    best = hb.copy()
    best_c5 = compute_c5(best)
    for _ in range(max_iters):
        improved = False
        for _ in range(n * 2):
            j = np.random.randint(n)
            if best[j] > 0.5 and best.sum() > n/2 + 0.01:
                ht = best.copy()
                ht[j] = 0.0
                ht = project_feasible(ht, n)
                c = compute_c5(ht)
                if c < best_c5 - 1e-14:
                    best = ht
                    best_c5 = c
                    improved = True
            elif best[j] < 0.5 and best.sum() < n/2 - 0.01:

                ht[j] = 1.0

                c = compute_c5(ht)
                if c < best_c5 - 1e-14:
                    best = ht
                    best_c5 = c
                    improved = True
        if not improved:
            break
    return best, best_c5

def multi_slq(h0, n, t_list, maxiter, ftol):
    best = h0.copy()
    best_c5 = compute_c5(best)
    for T in t_list:
        try:
            hc = slsqp_smooth(best, T, n, maxiter, ftol)
            c5v = compute_c5(hc)

                best_c5 = c5v
                best = hc
        except Exception:
            pass
    return best, best_c5
