        else:

                pk = float(np.max(np.correlate(h_b, 1.0 - h_b, mode='full')))
                if pk < best_pk:
                    best_pk = pk
                    improved = True
                else:
                    h_b[i], h_b[j] = 1, 0
        if not improved:
            break
    return h_b, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)

    for N in [550, 680]:
        if time.time() > deadline - 70:
            break
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)
        remain = deadline - time.time()
        # Phase 1: LSE-Adam (35% of remaining)
        d1 = time.time() + remain * 0.35
        h1, c1 = adam_lse(h_init.copy(), N, target, d1, lr0=0.015)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        # Phase 2: Annealing (35% of remaining)
        if time.time() < deadline - 40:
            d2 = time.time() + (deadline - time.time()) * 0.35
            h2, c2 = swap_anneal(h1, N, d2, rng, use_pairs=0.4)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()
        # Phase 3: Perturb + re-anneal (25% of remaining)
        if time.time() < deadline - 25:
            h3 = h2.copy()
            ones = np.where(h3 == 1)[0]
            zeros = np.where(h3 == 0)[0]
            n_flip = max(N // 12, 15)
            fl_orig = n_flip
            for _ in range(fl_orig):
                if len(ones) > 1 and len(zeros) > 1:
                    oi = rng.randint(len(ones))
                    zi = rng.randint(len(zeros))
                    h3[ones[oi]], h3[zeros[zi]] = 0, 1
                    ones = np.where(h3 == 1)[0]
                    zeros = np.where(h3 == 0)[0]
            d3 = time.time() + (deadline - time.time()) * 0.60
            h3, c3 = swap_anneal(h3, N, d3, rng, use_pairs=0.5)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()
        # Phase 4: Second LSE-Adam from binary optimum
        if time.time() < deadline - 20:
            h4 = (best_h if len(best_h) == N else h3).astype(float)
            h4 = project_sum(h4, target)
            h4 += rng.randn(N) * 0.02
            h4 = project_sum(h4, target)
            d4 = time.time() + (deadline - time.time()) * 0.45
            h4, c4 = adam_lse(h4, N, target, d4, lr0=0.008)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()
            # Quick anneal on LSE result
            if time.time() < deadline - 10:
                h5, c5v = swap_anneal(h4, N, deadline - 3, rng, use_pairs=0.45)
                if c5v < best_c5:
                    best_c5, best_h = c5v, h5.copy()

    # Final: try a random restart at N=680 if time permits
    if time.time() < deadline - 30:
        N2 = 680
        target2 = N2 / 2.0
        h_rand = np.zeros(N2)
        h_rand[rng.choice(N2, N2 // 2, replace=False)] = 1.0
        d_r = time.time() + (deadline - time.time()) * 0.7
        h_r, c_r = swap_anneal(h_rand, N2, d_r, rng, use_pairs=0.5)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
