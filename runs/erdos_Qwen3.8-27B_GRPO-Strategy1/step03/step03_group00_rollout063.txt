```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        tot = s
        if tot < 1e-14:
            return h
        h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fast_coord_descent(h, n, t_end, max_sweeps=1000):
    """Coordinate descent using precomputed correlations."""
    best_h = h.copy()
    C = compute_correlation(best_h)
    dx = 2.0 / n
    best_c5 = np.max(C) * dx
    center = n - 1  # index for k=0
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            
            # Compute effect of h[i] on correlation
            k = np.arange(2 * n - 1) - center  # lags from -(n-1) to (n-1)
            B = np.zeros(2 * n - 1)
            
            idx_plus = i + k
            valid_plus = (idx_plus >= 0) & (idx_plus < n)
            B[valid_plus] += (1.0 - best_h[np.clip(idx_plus[valid_plus], 0, n-1)])
            
            idx_minus = i - k
            valid_minus = (idx_minus >= 0) & (idx_minus < n)
            B[valid_minus] -= best_h[np.clip(idx_minus[valid_minus], 0, n-1)]
            
            # For k=0 (center): the term is h[i]*(1-h[i]) = h[i] - h[i]^2
            # The "linear" coefficient would be (1-h[i]) from the h[i]*... part
            # and -h[i] from the ...*(1-h[i]) part
            # Total: 1 - 2*h[i] for the linear part, plus -h[i]*h[i] (quadratic)
            # Actually: h[i]*(1-h[i]) = h[i] - h[i]^2
            # If we write this as h[i]*B[center] + quadratic_correction:
            # h[i] - h[i]^2 = h[i]*(1-h[i])
            # B[center] as computed above = (1-h[i]) - h[i] = 1-2*h[i]
            # And the actual contribution is h[i]*(1-h[i]) not h[i]*(1-2*h[i])
            # Difference: h[i]*(1-h[i]) - h[i]*(1-2*h[i]) = h[i] - h[i]^2 - h[i] + 2*h[i]^2 = h[i]^2
            # So: actual = h[i]*B[center] + h[i]^2
            # Or: new_val = C_base + t*B, where C_base already excludes h[i]'s contribution
            # C_base[center] = C[center] - (h[i] - h[i]^2)
            # new_C[center] = C_base[center] + t - t^2
            
            hi = best_h[i]
            C_base = C - hi * B
            # Fix center: C_base[center] should be C[center] - (hi - hi^2), not C[center] - hi*B[center]
            # We already subtracted hi*B[center], but the true subtraction should be hi - hi^2
            # So add back the correction: C_base[center] += hi - hi^2 - hi*B[center]
            # = hi - hi^2 - hi*(1-2*hi) = hi - hi^2 - hi + 2*hi^2 = hi^2
            C_base[center] += hi * hi
            
            # Now the 1D problem: minimize max(t in [0,1]) of:
            #   C_base[m] + t*B[m] for m != center
            #   C_base[center] + t - t^2 for m = center
            
            # Use numerical 1D optimization
            t_1d = solve_1d(C_base, B, center)
            
            # Evaluate
            C_new = C_base + t_1d * B
            C_new[center] = C_base[center] + t_1d - t_1d * t_1d
            c5_new = np.max(C_new) * dx
            
            if c5_new < best_c5 - 1e-14:
                best_h[i] = t_1d
                C = C_new
                best_c5 = c5_new
                improved = True
        
        if not improved and sweep > 5:
            break
    
    return best_h, best_c5
```

```python
def solve_1d(C_base, B, center):
    """Solve min_t max(C_base + t*B, C_base[center]+t-t^2) for t in [0,1]."""
    # First solve ignoring the quadratic center term
    mask = np.ones(len(C_base), dtype=bool)
    mask[center] = False
    t0 = _solve_1d_linear(C_base[mask], B[mask])
    
    # Refine with the quadratic term included
    # Evaluate at candidate points
    candidates = [t0, 0.0, 1.0]
    # Symmetric points around t0
    for d in [0.005, 0.02, 0.05, 0.1]:
        candidates.append(max(0.0, t0 - d))
        candidates.append(min(1.0, t0 + d))
    
    best_t = t0
    best_val = np.inf
    for t in candidates:
        val = _eval_1d(C_base, B, center, t)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Local refinement around best_t with golden section
    # (the function is not necessarily convex, but near the min it should be smooth)
    a = max(0.0, best_t - 0.02)
    b = min(1.0, best_t + 0.02)
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(30):
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = _eval_1d(C_base, B, center, c1)
        f2 = _eval_1d(C_base, B, center, c2)
        if f1 < f2:
            b = c2
        else:
            a = c1
    t_opt = (a + b) / 2
    if _eval_1d(C_base, B, center, t_opt) < best_val:
        best_t = t_opt
    
    return best_t

def _eval_1d(C_base, B, center, t):
    val = np.max(C_base + t * B)  # includes center
    center_val = C_base[center] + t - t * t
    if center_val > val:
        val = center_val
    return val

def _solve_1d_linear(A, B, t_lo=0.0, t_hi=1.0):
    """Minimize max(A + t*B) for t in [t_lo, t_hi]."""
    # Binary search on the threshold value
    lo_bounds = np.where(B >= 0, A + B * t_lo, A + B * t_hi)
    v_lo = np.max(lo_bounds)
    v_hi = min(np.max(A + B * t_lo), np.max(A + B * t_hi))
    
    if v_hi - v_lo < 1e-14:
        return (t_lo + t_hi) / 2
    
    pos = B > 1e-15
    neg = B < -1e-15
    has_pos = np.any(pos)
    has_neg = np.any(neg)
    
    for _ in range(40):
        v_mid = (v_lo + v_hi) / 2
        t_min = t_lo
        t_max = t_hi
        if has_pos:
            constraint = (v_mid - A[pos]) / B[pos]
            t_min = max(t_min, np.max(constraint))
        if has_neg:
            constraint = (v_mid - A[neg]) / B[neg]
            t_max = min(t_max, np.min(constraint))
        if t_min <= t_max + 1e-14:
            v_hi = v_mid
        else:
            v_lo = v_mid
    
    # Find optimal t
    t_min = t_lo
    t_max = t_hi
    if has_pos:
        t_min = max(t_min, np.max((v_hi - A[pos]) / B[pos]))
    if has_neg:
        t_max = min(t_max, np.min((v_hi - A[neg]) / B[neg]))
    if t_min > t_max:
        return (t_lo + t_hi) / 2
    return np.clip((t_min + t_max) / 2, t_lo, t_hi)
```

```python
def coord_descent(h, n, t_end, max_sweeps=5000):

    dx = 2.0 / n

    center = n - 1

        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            k = np.arange(2 * n - 1) - center
            B = np.zeros(2 * n - 1)
            ip = i + k
            vp = (ip >= 0) & (ip < n)
            B[vp] += 1.0 - best_h[ip[vp]]
            im = i - k
            vm = (im >= 0) & (im < n)
            B[vm] -= best_h[im[vm]]
            hi = best_h[i]
            Cb = C - hi * B
            Cb[center] += hi * hi
            t_new = solve_1d_full(Cb, B, center)
            C_new = Cb + t_new * B
            C_new[center] = Cb[center] + t_new - t_new * t_new
            c5_new = np.max(C_new) * dx

                best_h[i] = t_new
                C = C_new
                best_c5 = c5_new
                improved = True
        if not improved and sweep > 10:
            break
    return best_h, best_c5

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    def obj(x):
        return smooth_c5(x, T)

                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)
```

```python
def coord_descent(h, n, t_end, max_sweeps=5000):
    best_h = h.copy()
    C = np.correlate(best_h, 1.0 - best_h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(C)) * dx
    center = n - 1
    k_arr = np.arange(2 * n - 1) - center
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            B = np.zeros(2 * n - 1)
            im = i - k_arr
            vm = (im >= 0) & (im < n)
            B[vm] += 1.0 - best_h[im[vm]]
            ip = i + k_arr
            vp = (ip >= 0) & (ip < n)
            B[vp] -= best_h[ip[vp]]
            hi = best_h[i]
            Cb = C - hi * B
            Cb[center] -= hi * hi
            t_new = solve_1d_full(Cb, B, center)
            if abs(t_new - hi) < 1e-12:
                continue
            C_new = Cb + t_new * B
            C_new[center] = Cb[center] + t_new - t_new * t_new
            c5_new = float(np.max(C_new)) * dx
            if c5_new < best_c5 - 1e-14:
                best_h[i] = t_new
                C = C_new
                best_c5 = c5_new
                improved = True
        if not improved and sweep > 5:
            break
    return best_h, best_c5

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=300):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def perturb(h, n, mode=0):
    if mode == 0:
        out = h + 0.1 * np.random.randn(n)
    elif mode == 1:
        out = h.copy()
        idx = np.random.choice(n, size=n // 4, replace=False)
        out[idx] = 1.0 - h[idx]
    elif mode == 2:
        out = h.copy()
        blen = n // 6
        start = np.random.randint(0, n - blen)
        out[start:start+blen] = np.roll(h[start:start+blen], np.random.randint(1, blen))
    else:
        out = h + 0.05 * np.random.randn(n)
    return project(out, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
        best_h = project(best_h, 160)

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial CD polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.10, max_sweeps=3000)

    # Phase 2: Basin hopping with CD
    work_h = best_h.copy()
    work_c5 = best_c5
    max_iter = 80
    for it in range(max_iter):
        if time.time() > t0 + budget_s * 0.50:
            break
        cool = 1.0 - 0.92 * (it / max_iter)
        mode = it % 4
        h_pert = perturb(work_h, n, mode)
        t_cd = min(time.time() + 8.0, t0 + budget_s * 0.50)
        hc, c5v = coord_descent(h_pert, n, t_cd, max_sweeps=300)
        if time.time() < t0 + budget_s * 0.50 and cool > 0.3:
            try:
                hs = slsqp_opt(hc, 150, n, maxiter=150)
                hs, c5s = coord_descent(hs, n, min(time.time()+3, t0+budget_s*0.50), max_sweeps=100)
                if c5s < c5v:
                    hc, c5v = hs, c5s
            except Exception:
                pass
        temp = 0.005 * cool + 0.0003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 3: Multi-resolution
    for n_try in [120, 200, 240]:
        if time.time() > t0 + budget_s * 0.72:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n_try)
        t_cd2 = min(time.time() + 15, t0 + budget_s * 0.75)
        hc, c5v = coord_descent(h_interp, n_try, t_cd2, max_sweeps=800)
        # SLSQP polish at new resolution
        if time.time() < t0 + budget_s * 0.75:
            try:
                hs = slsqp_opt(hc, 500, n_try, maxiter=300)
                hs, c5s = coord_descent(hs, n_try, min(time.time()+5, t0+budget_s*0.75), max_sweeps=200)
                if c5s < c5v:
                    hc, c5v = hs, c5s
            except Exception:
                pass
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: More basin hops from best (tighter)
    work_h = best_h.copy()
    work_c5 = best_c5
    nn = len(work_h)
    for it in range(50):
        if time.time() > t0 + budget_s * 0.88:
            break
        cool = 1.0 - 0.85 * (it / 50.0)
        mode = it % 4
        h_pert = perturb(work_h, nn, mode)
        t_cd3 = min(time.time() + 5, t0 + budget_s * 0.90)
        hc, c5v = coord_descent(h_pert, nn, t_cd3, max_sweeps=200)
        temp = 0.003 * cool + 0.0002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Final SLSQP + CD polish
    nn = len(best_h)
    if time.time() < t0 + budget_s * 0.94:
        try:
            hc = slsqp_opt(best_h, 3000, nn, maxiter=600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h = hc
                best_c5 = c5v
        except Exception:
            pass

    best_h, best_c5 = coord_descent(best_h, len(best_h), t0 + budget_s * 0.98, max_sweeps=5000)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
