def grad_lse(h, N, T, k=50):

    peak = float(np.max(corr))
    # Softmax weights
    tc = T * corr
    tc -= tc.max()
    wc = np.exp(tc)
    sc = wc.sum()
    if sc < 1e-300:
        wc = np.zeros_like(corr)
        wc[peak_idx] = 1
    else:
        wc /= sc
    # Use top-k weighted indices
    top_idx = np.argpartition(wc, -k)[-k:]
    top_idx = top_idx[np.argsort(wc[top_idx])[::-1]]
    grad = np.zeros(N)
    center = N - 1
    for ai in top_idx:
        w = wc[ai]
        if w < 1e-10:
            continue
        L = ai - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += w * (1.0 - h[M:])
                grad[M:] -= w * h[:N - M]
    return grad, peak

        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lse(h, N, T, k=80):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    tc = T * corr
    tc -= tc.max()
    wc = np.exp(tc)
    sc = wc.sum()
    if sc < 1e-300:
        return np.zeros(N), peak
    wc /= sc
    top_idx = np.argpartition(wc, -k)[-k:]
    top_idx = top_idx[np.argsort(wc[top_idx])[::-1]]
    grad = np.zeros(N)

        w = wc[ai]

                grad[M:] -= w * h[:N - M]
    return grad, peak

def block_search(h_b, N, deadline, rng, block_size=2):
    dx = 2.0 / N
    h_b = h_b.copy()
    cur_pk = float(np.max(compute_corr(h_b)))
    best = h_b.copy()
    best_pk = cur_pk
    while time.time() < deadline:
        # Find consecutive blocks of 1s and 0s
        ones_blocks = []
        zeros_blocks = []
        i = 0
        while i < N:
            if h_b[i] == 1:
                j = i
                while j < N and h_b[j] == 1:
                    j += 1
                if j - i >= block_size:
                    for s in range(i, j - block_size + 1):
                        ones_blocks.append(s)
                i = j
            else:
                j = i
                while j < N and h_b[j] == 0:
                    j += 1
                if j - i >= block_size:
                    for s in range(i, j - block_size + 1):
                        zeros_blocks.append(s)
                i = j
        if not ones_blocks or not zeros_blocks:
            break
        improved = False
        for ob in ones_blocks:
            for zb in zeros_blocks:
                # Save
                save_o = h_b[ob:ob+block_size].copy()
                save_z = h_b[zb:zb+block_size].copy()
                h_b[ob:ob+block_size] = 0
                h_b[zb:zb+block_size] = 1
                new_pk = float(np.max(compute_corr(h_b)))
                if new_pk < cur_pk:
                    cur_pk = new_pk
                    improved = True
                    if cur_pk < best_pk:
                        best_pk = cur_pk
                        best = h_b.copy()
                else:
                    h_b[ob:ob+block_size] = save_o
                    h_b[zb:zb+block_size] = save_z
        if not improved:
            # Perturb
            i = rng.randint(N)
            while h_b[i] == 0:
                i = rng.randint(N)
            j = rng.randint(N)
            while h_b[j] == 1:
                j = rng.randint(N)
            h_b[i], h_b[j] = h_b[j], h_b[i]
            cur_pk = float(np.max(compute_corr(h_b)))
    return best, best_pk * dx
