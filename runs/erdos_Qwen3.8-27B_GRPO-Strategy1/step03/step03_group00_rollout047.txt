The previous code handled this by re-projecting, but that changes other coordinates. A better approach for the exact 1D optimization is to handle the constraint within the 1D problem.

Another idea: **Benders decomposition / cutting plane**. At each iteration:
1. Find the current h
2. Identify the active overlap (k* = argmax overlap)
3. Add a linear constraint: overlap_{k*}(h) ≤ C (where C is the current value)
4. Solve: minimize C subject to all added constraints + 0≤h≤1 + sum(h)=n/2

This turns it into a sequence of linear programs (LPs)! Because each constraint is linear in h (overlap_k is linear in h when we view it as h[i]*(1-h[i+k])... wait, no, it's NOT linear in h because h appears in both factors.

Hmm, unless we linearize: at the current h, overlap_k ≈ overlap_k(h) + sum_i ∂overlap_k/∂h[i] * (h[i] - h[i]). Then the constraint is linear. This is the cutting plane / linearization approach.

So:
1. Start with h_0
2. For iteration t:
   a. k* = argmax_k overlap_k(h_t)
   b. Solve LP: minimize C s.t. C ≥ overlap_k(h_t) + sum_i sg_k[h_t][i] * (h[i] - h_t[i]) for all previously active k, 0≤h≤1, sum(h)=n/2
   c. h_{t+1} = solution of LP
   d. If overlap_k*(h_{t+1}) ≤ C + ε, stop (optimality)
   e. Else, add constraint for k*

This is the "outer approximation" method for the min-max problem! It's exact in the limit (if we add all constraints) and each LP is small.

But implementing this requires an LP solver. We have cvxpy available! Let me use it.

Actually wait - the linearization of overlap_k at h_t:
overlap_k(h) = sum_i h[i]*(1-h[i+k])
≈ sum_i [h_t[i]*(1-h_t[i+k]) + (h[i]-h_t[i])*(1-h_t[i+k]) + h_t[i]*(h_t[i+k]-h[i+k])]
= overlap_k(h_t) + sum_i (h[i]-h_t[i])*(1-h_t[i+k]) - sum_i h_t[i]*(h[i+k]-h_t[i+k])

Reindex the second sum: -sum_j h_t[j-k]*(h[j]-h_t[j]) (where j=i+k)
= overlap_k(h_t) + sum_i [(1-h_t[i+k]) - h_t[i-k]] * (h[i]-h_t[i])

So the linearization is: overlap_k(h) ≈ overlap_k(h_t) + sum_i c_k[i] * (h[i]-h_t[i])
where c_k[i] = (1-h_t[i+k]) - h_t[i-k] (with boundary handling).

This is indeed linear in h! So the cutting plane LP works.

Let me implement this with cvxpy:

```python
import cvxpy as cp

def cutting_plane_lp(h_t, active_k_list, n):
    """Solve LP: min C s.t. C >= linearized overlap_k for k in active_k_list, 0<=h<=1, sum(h)=n/2"""
    h_var = cp.Variable(n, bounds=(0, 1))
    C_var = cp.Variable()
    
    constraints = [cp.sum(h_var) == n / 2.0]
    
    for k_idx in active_k_list:
        k = k_idx - (n - 1)  # actual lag
        c = np.zeros(n)
        # c[i] = (1 - h_t[i+k]) - h_t[i-k] (with boundary handling)
        for i in range(n):
            if 0 <= i + k < n:
                c[i] += (1.0 - h_t[i + k])
            if 0 <= i - k < n:
                c[i] += (-h_t[i - k])
        base = np.correlate(h_t, 1.0 - h_t, mode='full')[k_idx]
        # Linearized: base + c @ (h_var - h_t) >= ... 
        # We want: base + c @ (h_var - h_t) <= C
        constraints.append(base + c @ (h_var - h_t) <= C_var)
    
    obj = cp.Minimize(C_var)
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if h_var.value is None:
        return None, None
    return np.clip(h_var.value, 0, 1), C_var.value

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        mask = h > 1e-12
        if not mask.any():
            return h
        h[mask] *= (1.0 + diff / h[mask].sum())
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(np.clip(T * (corr - m), -700, 700)))) / T
    return val * dx

def overlap_subgrad_at(h, k_idx, n):
    """Subgradient of overlap at lag index k_idx w.r.t. h"""
    k = k_idx - (n - 1)
    sg = np.zeros(n)
    if k >= 0:
        sg[:n - k] += (1.0 - h[k:n])
        sg[k:] += (-h[:n - k])
    else:
        kk = -k
        sg[kk:] += (1.0 - h[:n - kk])
        sg[:n - kk] += (-h[kk:])
    return sg

def overlap_grad_coeff(h_t, k_idx, n):
    """Coefficient vector for linearization of overlap_k at h_t"""
    k = k_idx - (n - 1)
    c = np.zeros(n)
    if k >= 0:
        c[:n - k] += (1.0 - h_t[k:n])
        c[k:] += (-h_t[:n - k])
    else:
        kk = -k
        c[kk:] += (1.0 - h_t[:n - kk])
        c[:n - kk] += (-h_t[kk:])
    return c

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_sum(np.clip(res.x, 0, 1), n)
    except:
        return h

def coord_opt_1d(h, n, i):
    """Exact 1D optimization of h[i]"""
    h_old = h[i]
    overlaps = compute_overlaps(h)
    k_lags = np.arange(-(n - 1), n)
    coef = np.zeros(len(k_lags))
    for idx, k in enumerate(k_lags):
        c = 0.0
        if 0 <= i + k < n:
            c += (1.0 - h[i + k])
        if 0 <= i - k < n:
            c += (-h[i - k])
        coef[idx] = c
    dx = 2.0 / n
    def f1d(v):
        vals = overlaps + (v - h_old) * coef
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100, random_order=True):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n) if random_order else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_1d(best_h, n, i)
            if abs(hi_new - best_h[i]) > 1e-13:
                best_h[i] = hi_new
                improved = True
        if improved:
            best_h = project_sum(best_h, n)
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
        else:
            # Try one more ordered sweep
            for i in range(n):
                hi_new = coord_opt_1d(best_h, n, i)
                if abs(hi_new - best_h[i]) > 1e-13:
                    best_h[i] = hi_new
                    improved = True
            if improved:
                best_h = project_sum(best_h, n)
                c5_try = compute_c5(best_h)
                if c5_try < best_c5:
                    best_c5 = c5_try
            else:
                break
    return best_h, best_c5

def cutting_plane_solve(h_t, n, max_iters=50, solver='CLARABEL'):
    """Solve min max_k overlap_k(h) via cutting plane (outer approximation) LPs"""
    h_work = h_t.astype(float).copy()
    n_cuts = 0
    C_upper = compute_c5(h_work)
    
    h_var = cp.Variable(n, bounds=(0, 1))
    C_var = cp.Variable()
    constraints = [cp.sum(h_var) == n / 2.0]
    
    active_coeffs = []  # list of (c_vector, base_value)
    
    for it in range(max_iters):
        # Find active constraint
        overlaps = compute_overlaps(h_work)
        k_star = np.argmax(overlaps)
        
        c_vec = overlap_grad_coeff(h_work, k_star, n)
        base = overlaps[k_star]
        
        # Check if this cut is new (not too similar to existing)
        is_new = True
        for (c_old, b_old) in active_coeffs:
            if np.linalg.norm(c_vec - c_old) < 1e-6:
                is_new = False
                break
        if is_new:
            active_coeffs.append((c_vec, base))
            constraints.append(base + c_vec @ (h_var - h_work) <= C_var)
        
        # Solve LP
        try:
            prob = cp.Problem(cp.Minimize(C_var), constraints)
            prob.solve(solver=cp.CLARABEL)
            if h_var.value is None:
                break
            h_new = np.clip(h_var.value, 0, 1)
            h_new = project_sum(h_new, n)
        except:
            break
        
        c5_new = compute_c5(h_new)
        if c5_new < C_upper - 1e-12:
            C_upper = c5_new
            h_work = h_new
        else:
            # Check optimality: is the current point nearly optimal?
            # The LP optimum gives a lower bound; if it's close to C_upper, we're done
            if abs(C_var.value - C_upper) < 1e-7:
                break
            h_work = h_new
    
    return h_work, C_upper
