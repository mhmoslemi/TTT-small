def soft_gradient(h, N, tau, K=None):
    """Compute gradient using soft-max (LSE) or top-K weighting."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    
    if K is not None and K < len(corr):
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        c_top = corr
    
    # Soft weights
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-8:
            continue
        k = top_idx[ix]
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

    best_c5 = compute_c5(h, dx)

    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:

        tau = tau0 * (tau1 / tau0) ** frac

def binary_refine(h, N, target, dx, deadline, seed=55555):
    """Local search refining near-binary values via targeted perturbations."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    # Push toward binary
    h_bin = np.where(h > 0.5, 0.95, 0.05).astype(float)
    h_bin = project_sum(h_bin, target)
    for _ in range(200):
        if time.time() > deadline:
            break
        trial = best_h.copy()
        n_mod = rng.randint(1, 6)
        idx = rng.choice(N, size=n_mod, replace=False)
        trial[idx] += rng.randn(n_mod) * 0.015
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_h = trial.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    sizes = [150, 200, 300]
    sizes = [s for s in sizes if s % 2 == 0]

    for N in sizes:
        if time.time() > deadline - 90:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)

        remain = deadline - time.time()
        if remain < 70:
            break

        # Phase A: Broad soft optimization (all lags, low tau)
        pA_dead = time.time() + remain * 0.30
        hA, cA = run_soft(h_init.copy(), N, target, dx, pA_dead,
                          tau_s=[5.0, 50.0, 150.0],
                          lr_s=[0.035, 0.015, 0.006],
                          K_s=[None, 60, 30],
                          iters=400)
        if cA < best_c5:
            best_c5 = cA
            best_h = hA.copy()

        # Phase B: Continuous SA for diversification
        if time.time() < deadline - 50:
            pB_dead = time.time() + (deadline - time.time()) * 0.25
            hB, cB = continuous_sa(hA.copy(), N, target, dx, pB_dead, seed=seed + N)
            if cB < best_c5:
                best_c5 = cB
                best_h = hB.copy()

        # Phase C: Sharp gradient (high tau, small K) to polish max
        if time.time() < deadline - 35:
            pC_dead = time.time() + (deadline - time.time()) * 0.30
            hC, cC = run_soft(hB.copy(), N, target, dx, pC_dead,
                              tau_s=[100.0, 500.0, 2000.0],
                              lr_s=[0.008, 0.003, 0.001],
                              K_s=[20, 10, 5],
                              iters=350)
            if cC < best_c5:
                best_c5 = cC
                best_h = hC.copy()

        # Phase D: Binary refine / micro-perturbation
        if time.time() < deadline - 15:
            hD, cD = binary_refine(hC.copy(), N, target, dx, time.time() + (deadline - time.time()) * 0.3)
            if cD < best_c5:
                best_c5 = cD
                best_h = hD.copy()

        # Phase E: Another SA round from best
        if time.time() < deadline - 20:
            pE_dead = time.time() + (deadline - time.time()) * 0.35
            hE, cE = continuous_sa(hD.copy(), N, target, dx, pE_dead, seed=seed + N + 100)
            if cE < best_c5:
                best_c5 = cE
                best_h = hE.copy()

    # Final ultra-sharp polish at best size
    Nb = len(best_h)
    if time.time() < deadline - 4:
        dx_b = 2.0 / Nb
        hf, cf = run_soft(best_h.copy(), Nb, Nb // 2, dx_b, deadline - 1,
                          tau_s=[2000.0, 5000.0],
                          lr_s=[0.001, 0.0001],
                          K_s=[5, 3],
                          iters=100)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
