def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_max(vals, tau):
    """Log-sum-exp approximation to max."""
    m = vals.max()
    if tau < 1e-4:
        return m
    shifted = tau * (vals - m)
    shifted -= shifted.max()  # numerical stability
    return m + np.log(np.exp(shifted).sum()) / tau

def c5_gradient(h, N, tau):
    """Gradient of smooth C5 approximation w.r.t. h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sc = tau * (corr - m)
    sc -= sc.max()

        else:

        tau = tau_start * (tau_end / tau_start) ** frac

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def topk_weights(corr, tau, K):
    n = len(corr)
    if K >= n:
        idx = np.arange(n)
        cv = corr
    else:
        idx = np.argpartition(corr, -K)[-K:]
        cv = corr[idx]
    m = cv.max()
    sc = tau * (cv - m)
    sc -= sc.max()
    e = np.exp(sc)
    w = e / e.sum()
    return idx, w

def c5_grad(h, N, tau, K):

    idx, w = topk_weights(corr, tau, K)
    dx = 2.0 / N
    grad = np.zeros(N)

            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])

    t_s, t_e = cfg['tau0'], cfg['tau1']

    m1 = np.zeros(N)
    m2 = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    while time.time() < deadline and t < n_it:
        f = t / max(n_it, 1)
        tau = t_s * (t_e / t_s) ** f
        lr = l_s * (l_e / l_s) ** (0.5 * f)
        K = max(2, int(k_s + (k_e - k_s) * f))
        K = min(K, 2 * N - 1)
        g = c5_grad(h, N, tau, K)
        c5 = compute_c5(h, dx)

import numpy as np
import time
from scipy.optimize import minimize as sp_minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def topk_sel(corr, tau, K):
    n = len(corr)
    if K >= n:
        idx = np.arange(n)
        cv = corr
    else:
        idx = np.argpartition(corr, -(K), axis=0)[:-(K) if K < n else :]
        idx = np.argpartition(corr, -K)[-K:]
        cv = corr[idx]
    m = cv.max()
    sc = tau * (cv - m)
    sc -= sc.max()
    e = np.exp(sc)
    w = e / e.sum()
    return idx, w

def c5_grad(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    idx, w = topk_sel(corr, tau, K)
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(w)):
        wi = w[ii]
        if wi < 1e-10:
            continue
        L = int(idx[ii]) - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad * dx

def adam_run(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K0, K1, iters):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m1 = np.zeros(N)
    m2 = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    while time.time() < deadline and t < iters:
        f = t / max(iters, 1)
        tau = tau0 * (tau1 / tau0) ** f
        lr = lr0 * (lr1 / lr0) ** (0.5 * f)
        K = max(2, int(K0 + (K1 - K0) * f))
        K = min(K, 2 * N - 1)
        g = c5_grad(h, N, tau, K)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h.copy()
        t += 1
        m1 = b1 * m1 + (1.0 - b1) * g
        m2 = b2 * m2 + (1.0 - b2) * g * g
        mh = m1 / (1.0 - b1 ** t)
        vh = m2 / (1.0 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    c5_end = compute_c5(best_h, dx)
    if c5_end < best_c5:
        best_c5 = c5_end
    return best_h, best_c5

def lse_smoothed(corr, tau):
    m = corr.max()
    sc = tau * (corr - m)
    sc -= sc.max()
    return m + np.log(np.exp(sc).sum()) / tau

def lse_obj(h, N, tau, dx, mu):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    val = lse_smoothed(corr, tau)
    val += mu * (h.sum() - N / 2.0) ** 2
    return val

def lse_grad(h, N, tau, dx, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sc = tau * (corr - m)
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        L = k - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            grad[:N - M] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:N - M]
    grad *= dx
    grad += 2.0 * mu * (h.sum() - N / 2.0)
    return grad

def scipy_polish(h, N, dx, tau, mu, maxiter=300):
    target = N / 2.0
    args = (N, tau, dx, mu)
    bounds = [(0.0, 1.0)] * N
    res = sp_minimize(
        lambda x: lse_obj(x, *args), h,
        jac=lambda x: lse_grad(x, *args),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'maxfun': maxiter * 3, 'ftol': 1e-14}
    )
    h_new = project_sum(res.x, target)
    c = compute_c5(h_new, dx)
    c0 = compute_c5(h, dx)
    if c <= c0:
        return h_new, c
    return h, c0

def micro_perturb(h, N, target, dx, deadline, rng, n_trials, step):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        trial = best_h.copy()
        i = rng.randint(N)
        j = rng.randint(N)
        d = rng.randn() * step
        trial[i] += d
        trial[j] -= d
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-12:
            best_c5 = c
            best_h = trial
    return best_h, best_c5

def try_binarize(h, N, target, dx, thresholds):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for t in thresholds:
        b = np.where(h > t, 1.0, 0.0)
        s = b.sum()
        if s != target:
            diff = target - s
            if diff > 0:
                order = np.argsort(-h)[:int(diff)]
                b[order] = 1.0
            elif diff < 0:
                order = np.argsort(h)[:int(-diff)]
                b[order] = 0.0
        b = project_sum(b, target)
        c = compute_c5(b, dx)
        if c < best_c5 - 1e-12:
            best_c5 = c
            best_h = b.copy()
    return best_h, best_c5
