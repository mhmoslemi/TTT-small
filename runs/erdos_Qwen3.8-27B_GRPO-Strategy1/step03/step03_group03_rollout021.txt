from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T, lam):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(corr.max())
    scaled = T * (corr - m)
    exp_w = np.exp(scaled)
    log_sum = float(np.log(exp_w.sum()))
    val = (m + log_sum / T) * dx
    s = h.sum() - N / 2.0
    val += 0.5 * lam * s * s
    w = exp_w / exp_w.sum()
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        wj = w[idx]
        if wj < 1e-15:
            continue
        k = idx - (N - 1)
        if k > 0:
            grad[k:] += wj * (1.0 - h[:N-k])
            grad[:N-k] -= wj * h[k:]
        elif k < 0:
            q = -k
            grad[:N-q] += wj * (1.0 - h[q:])
            grad[q:] -= wj * h[:N-q]
        else:
            grad += wj * (1.0 - 2.0 * h)
    grad *= dx
    grad += lam * s
    return val, grad
