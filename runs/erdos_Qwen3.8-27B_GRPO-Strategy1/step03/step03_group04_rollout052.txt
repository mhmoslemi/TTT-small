import numpy as np
import time

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, N, tau):
    """Compute gradient of smoothed max of correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    exp_c = np.exp(tau * (corr - max_c))
    w = exp_c / exp_c.sum()
    grad = np.zeros(N)
    center = N - 1
    for k_idx in range(2 * N - 1):
        wk = w[k_idx]
        if wk < 1e-8:
            continue
        k = k_idx - center
        if k >= 0:
            n_valid = N - k
            if k < N:
                grad[k:] += wk * (1.0 - h[:n_valid])
                grad[:n_valid] += wk * (-h[k:])
        else:
            j = -k
            n_valid = N - j
            grad[:n_valid] += wk * (1.0 - h[j:])
            grad[j:] += wk * (-h[:n_valid])
    return grad, float(np.max(corr))

def adam_smooth(h, N, target, deadline, tau, lr0, decay, k_select):
    """Adam optimization using smoothed max gradient."""
    best_h = h.copy()
    best_val = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = smooth_grad(h, N, tau)
        if peak < best_val:
            best_val = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= decay
            # Adaptively increase tau
            # tau *= 1.01
    return best_h, best_val * (2.0 / N)

- c[k] changes for lags k where i or j appears in the sum
- Position i contributes to c[k] when: i+k is a valid index (for h[i+k] term) or i is in the (1-h[i]) term
  - Specifically, c[k] includes h[n+k]*(1-h[n]) for valid n
  - i appears in h[n+k] when n+k=i, i.e., n=i-k. So c[k] includes h[i]*(1-h[i-k]) for k where 0≤i-k<N
  - i appears in (1-h[n]) when n=i. So c[k] includes h[i+k]*(1-h[i]) for k where 0≤i+k<N
  - When h[i] goes from 1 to 0:
    - First term: h[i]*(1-h[i-k]) goes to 0, change = -(1-h[i-k]) for valid k
    - Second term: h[i+k]*(1-h[i]) goes to h[i+k]*1=h[i+k], change = +h[i+k] for valid k

- Position j contributes similarly but in reverse (0→1):
  - First term: h[j]*(1-h[j-k]) goes from 0 to (1-h[j-k]), change = +(1-h[j-k])
  - Second term: h[j+k]*(1-h[j]) goes from h[j+k]*1=h[j+k] to h[j+k]*0=0, change = -h[j+k]

So the total change in c[k] from the swap (i→0, j→1):
Δc[k] = [-(1-h[i-k])] + [h[i+k]] + [(1-h[j-k])] + [-h[j+k]]
       = -(1-h[i-k]) + h[i+k] + (1-h[j-k]) - h[j+k]

(Where we only include terms that are valid, i.e., the indices are in [0,N).)

This means for each proposed swap (i,j), we need to update c[k] for O(N) lags. With precomputation, we can do this in O(N) per swap evaluation.

But the total number of possible swaps is O(N²), so a full "best improving swap" search is O(N³) per iteration. For N=380, that's about 55M operations per iteration. Too slow for Python.

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binary_from_h(h, target):
    """Convert to binary maintaining sum."""
    N = len(h)
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > int(target):
        # Need to turn off some 1s - prefer those with lowest h value
        ones = np.where(h_b == 1)[0]
        order = np.argsort(h[ones])
        for idx in ones[order[:n_ones - int(target)]]:
            h_b[idx] = 0
    elif n_ones < int(target):
        zeros = np.where(h_b == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in zeros[order[:int(target) - n_ones]]:
            h_b[idx] = 1
    return h_b

def smooth_max_grad(h, N, tau):
    """Gradient of log-sum-exp approximation to max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    shift = corr.max()
    exp_c = np.exp(tau * (corr - shift))
    w = exp_c / exp_c.sum()
    grad = np.zeros(N)
    center = N - 1
    # Only process significant weights
    thresh = 1e-6
    for k_idx in np.where(w > thresh)[0]:
        wk = w[k_idx]
        k = int(k_idx) - center
        if k >= 0:
            nv = N - k
            grad[k:] += wk * (1.0 - h[:nv])
            grad[:nv] += -wk * h[k:]
        else:
            j = -k
            nv = N - j
            grad[:nv] += wk * (1.0 - h[j:])
            grad[j:] += -wk * h[:nv]
    return grad, float(np.max(corr))

def smooth_adam(h, N, target, deadline, tau0, lr0, decay):
    best_h = h.copy()
    best_val = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = smooth_max_grad(h, N, tau0)
        if peak < best_val:
            best_val = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        h = h - lr * (m / (1 - b1**it)) / (np.sqrt(v / (1 - b2**it)) + eps)
        h = project_sum(h, target)
        if it % 300 == 0:
            lr *= decay
    return best_h, best_val * (2.0 / N)

def binary_local_search(h_b, N, deadline, rng, max_iters=20000):
    """Local search with random pair swaps."""
    best = h_b.copy()
    best_pk = float(np.max(np.correlate(best, 1-best, mode='full')))
    cur = h_b.copy()
    cur_pk = best_pk
    T0, T1 = 4.0, 0.002
    n_max = max_iters
    for it in range(n_max):

            break
        T = T0 * (T1 / T0) ** (it / n_max)
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i], cur[j] = cur[j], cur[i]
        new_pk = float(np.max(np.correlate(cur, 1-cur, mode='full')))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.005 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_pk

def targeted_swaps(h_b, N, rng, n_tries=5000):
    """Targeted: swap positions contributing to peak lags."""
    best = h_b.copy()
    best_pk = float(np.max(np.correlate(best, 1-best, mode='full')))
    cur = h_b.copy()
    cur_pk = best_pk
    for _ in range(n_tries):

        peak_idx = np.argmax(corr)

        # Find positions contributing to this peak

            nv = N - k
            contrib = cur[k:] * (1 - cur[:nv])
            ones_in_peak = np.where(contrib > 0)[0] + k
            zeros_in_peak = np.where(contrib > 0)[0]
        else:

            nv = N - j
            contrib = cur[:nv] * (1 - cur[j:])

            zeros_in_peak = np.where(contrib > 0)[0] + j

            break
        # Try to swap a contributing 1 with a random 0
        i = ones_in_peak[rng.randint(len(ones_in_peak))]
        zeros_all = np.where(cur == 0)[0]
        j = zeros_all[rng.randint(len(zeros_all))]
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_pk = float(np.max(np.correlate(cur, 1-cur, mode='full')))
        if new_pk < cur_pk:
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]

        corr = np.correlate(cur, 1 - cur, mode='full')
        peak_idx = int(np.argmax(corr))
        center = N - 1
        k = peak_idx - center
        if k >= 0:
            nv = N - k
            contrib = cur[k:] * (1 - cur[:nv])
            idx = np.where(contrib > 0)[0]
            one_pos = idx + k
            zero_pos = idx
        else:
            j = -k
            nv = N - j
            contrib = cur[:nv] * (1 - cur[j:])
            idx = np.where(contrib > 0)[0]
            one_pos = idx
            zero_pos = idx + j
        if len(one_pos) == 0:
            break
        i = one_pos[rng.randint(len(one_pos))]
        zeros_all = np.where(cur == 0)[0]
        j2 = zeros_all[rng.randint(len(zeros_all))]
        if i == j2:
            continue
        cur[i], cur[j2] = cur[j2], cur[i]
        new_pk = float(np.max(np.correlate(cur, 1 - cur, mode='full')))
        if new_pk <= cur_pk:
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = cur.copy()
        else:
            cur[i], cur[j2] = cur[j2], cur[i]
    return best, best_pk

def multi_start_smooth(h, N, target, deadline, rng, tau, n_starts=3):
    best_h = h.copy()
    best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / N)
    per_start = (deadline - time.time()) / n_starts
    for s in range(n_starts):
        t_end = t0_global + (time.time() - t0_global + per_start * (s + 1))
        t_end = min(t_end, deadline)
        if s == 0:
            h_s = h.copy()
        else:
            h_s = h.copy() + rng.randn(N) * (0.05 + 0.03 * s)
            h_s = project_sum(h_s, target)
        d = t_end - 3
        if d > time.time() + 5:
            h_r, c_r = smooth_adam(h_s, N, target, d, tau, 0.012, 0.985)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r
    return best_h, best_c5

t0_global = 0.0

def run(seed=42, budget_s=1100, **kwargs):
    global t0_global
    t0_global = time.time()
    deadline = t0_global + budget_s - 10
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)

    # Phase 1: Smoothed-max Adam at base resolution
    d1 = time.time() + (deadline - time.time()) * 0.18
    if d1 > t0_global + 60:
        h1, c1 = smooth_adam(h0.copy(), n0, n0 / 2.0, d1, 0.15, 0.018, 0.98)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary local search at base resolution
    if time.time() < deadline - 150:
        h_b = binary_from_h(best_h, n0 / 2.0)
        d2 = time.time() + (deadline - time.time()) * 0.2
        h2, c2 = targeted_binary(h_b, n0, rng, n_rounds=4000)
        c2s = c2 * (2.0 / n0)
        if c2s < best_c5:
            best_c5, best_h = c2s, h2
        if time.time() < deadline - 100:
            d3 = time.time() + (deadline - time.time()) * 0.25
            h3, c3 = binary_sa(h2, n0, d3, rng)
            c3s = c3 * (2.0 / n0)
            if c3s < best_c5:
                best_c5, best_h = c3s, h3

    # Phase 3: Multi-resolution refinement
    for N in [460, 550, 680]:
        if time.time() > deadline - 100:
            break
        target = N / 2.0
        Nb = len(best_h)
        x_old = np.linspace(0.0, 2.0, Nb, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_sum(h_init, target)

        remaining = deadline - time.time()
        if remaining < 80:
            break

        # Smoothed Adam with adaptive tau
        tau_val = 0.12 + 0.02 * (N // 100)
        d4 = time.time() + remaining * 0.35
        h4, c4 = smooth_adam(h_init.copy(), N, target, d4, tau_val, 0.014, 0.983)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

        # Targeted binary
        if time.time() < deadline - 50:
            h_b4 = binary_from_h(best_h if len(best_h) == N else h4, target)
            nt = max(2000, int((deadline - time.time()) * 8))
            h5, c5v = targeted_binary(h_b4, N, rng, n_rounds=nt)
            c5s = c5v * (2.0 / N)
            if c5s < best_c5:
                best_c5, best_h = c5s, h5

            # Binary SA
            if time.time() < deadline - 30:
                d6 = deadline - 15
                h6, c6 = binary_sa(h5, N, d6, rng)
                c6s = c6 * (2.0 / N)
                if c6s < best_c5:
                    best_c5, best_h = c6s, h6

            # Perturbation + re-optimize
            if time.time() < deadline - 25:
                h_p = h5.copy() + rng.randn(N) * 0.04
                h_p = project_sum(h_p, target)
                d7 = deadline - 12
                h7, c7 = smooth_adam(h_p, N, target, d7, tau_val * 1.5, 0.008, 0.99)
                if c7 < best_c5:
                    best_c5, best_h = c7, h7

    # Phase 4: Final polish
    Nb = len(best_h)
    if time.time() < deadline - 8:
        hF = np.clip(best_h + rng.randn(Nb) * 0.015, 0.0, 1.0)
        hF = project_sum(hF, Nb / 2.0)
        hF, cF = smooth_adam(hF, Nb, Nb / 2.0, deadline - 3, 0.2, 0.005, 0.992)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    # Ensure valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
