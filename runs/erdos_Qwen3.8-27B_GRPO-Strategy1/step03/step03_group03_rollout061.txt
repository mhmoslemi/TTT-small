            break
        h = h_new
    return best_h, best_c5

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_polish(h, T, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def se(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': se}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def binary_project(h, n):
    h_bin = np.zeros(n)
    h_bin[np.argsort(h)[n // 2:]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: CCP at native resolution
    if time.time() - t0 < budget_s * 0.30:
        for topK in [10, 15, 20]:
            if time.time() - t0 > budget_s * 0.30:
                break
            h_c, c5_c = ccp_loop(best_h, best_n, max_iter=250, top_K=topK)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c

    # Phase 2: SLSQP polish with escalating T
    if time.time() - t0 < budget_s * 0.42:
        for T in [3000, 15000, 60000, 200000]:
            if time.time() - t0 > budget_s * 0.42:
                break
            h_p = slsqp_polish(best_h, T, maxiter=700)
            c5_p, _ = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p

    # Phase 3: Multi-resolution
    for n_try in [76, 80, 82, 84]:
        if time.time() - t0 > budget_s * 0.65:
            break
        if n_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        h_c, c5_c = ccp_loop(h_int, n_try, max_iter=200, top_K=12)
        h_c = slsqp_polish(h_c, 50000, maxiter=400)
        c5_c, _ = compute_c5(h_c)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c
            best_n = n_try

    # Phase 4: Binary structure exploration + random restarts
    if time.time() - t0 < budget_s * 0.85:
        h_bin = binary_project(best_h, best_n)
        h_c, c5_c = ccp_loop(h_bin, best_n, max_iter=200, top_K=15)
        h_c = slsqp_polish(h_c, 80000, maxiter=500)
        c5_c, _ = compute_c5(h_c)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c

        for trial in range(10):
            if time.time() - t0 > budget_s * 0.85:
                break
            scale = 0.02 + 0.05 * np.random.rand()
            h_pert = best_h + np.random.randn(best_n) * scale
            h_pert = project_feasible(h_pert, best_n)
            h_c, c5_c = ccp_loop(h_pert, best_n, max_iter=120, top_K=12)
            h_c = slsqp_polish(h_c, 40000, maxiter=300)
            c5_c, _ = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c

    # Phase 5: Rotation-based diversification
    if time.time() - t0 < budget_s * 0.92:
        for shift in [3, 7, 13]:
            if time.time() - t0 > budget_s * 0.92:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_c, c5_c = ccp_loop(h_rot, best_n, max_iter=150, top_K=15)
            c5_c, _ = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c

    # Phase 6: Final fine polish
    if time.time() - t0 < budget_s * 0.98:
        h_c, _ = ccp_loop(best_h, best_n, max_iter=300, top_K=25)
        h_c = slsqp_polish(h_c, 300000, maxiter=1000)
        c5_c, _ = compute_c5(h_c)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c
        # Tiny perturbation + final CCP
        h_tiny = best_h + np.random.randn(best_n) * 0.005
        h_tiny = project_feasible(h_tiny, best_n)
        h_c2, _ = ccp_loop(h_tiny, best_n, max_iter=200, top_K=20)
        h_c2 = slsqp_polish(h_c2, 200000, maxiter=600)
        c5_c2, _ = compute_c5(h_c2)
        if c5_c2 < best_c5:
            best_c5 = c5_c2
            best_h = h_c2

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
