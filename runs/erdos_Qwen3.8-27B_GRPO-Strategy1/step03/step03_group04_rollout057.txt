def swap_delta(h, i, j, N):
    """Compute the change in correlation vector when h[i]:1→0 and h[j]:0→1."""
    L = np.arange(-(N-1), N)  # shape (2N-1,)
    # For position i (1→0):
    # As h[n+L] at n=i-L: valid if 0<=i-L<N
    #   old contribution: 1 * (1-h[i-L]), new: 0 → delta: -(1-h[i-L])
    # As h[n] at n=i, affecting (1-h[n]) factor: 
    #   old: h[i+L] * (1-1) = 0, new: h[i+L] * (1-0) = h[i+L] → delta: +h[i+L]
    ip = i + L  # positions i+L for the second term
    im = i - L  # positions i-L for the first term
    jp = j + L
    jm = j - L
    
    delta = np.zeros(2*N-1)
    
    # -(1-h[i-L]) where 0<=i-L<N
    m1 = (im >= 0) & (im < N)
    delta[m1] -= (1.0 - h[im[m1]])
    
    # +h[i+L] where 0<=i+L<N
    m2 = (ip >= 0) & (ip < N)
    delta[m2] += h[ip[m2]]
    
    # +(1-h[j-L]) where 0<=j-L<N
    m3 = (jm >= 0) & (jm < N)
    delta[m3] += (1.0 - h[jm[m3]])
    
    # -h[j+L] where 0<=j+L<N
    m4 = (jp >= 0) & (jp < N)
    delta[m4] -= h[jp[m4]]
    
    return delta

    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)

        peak = float(np.max(corr))
        if peak < best_peak:

            best_h = h.copy()

        max_c = peak
        exp_c = np.exp(T * (corr - max_c))
        w = exp_c / exp_c.sum()

        center = N - 1
        grad = np.zeros(N)
        # Use top few for efficiency
        top_k = max(5, N // 8)
        top_idx = np.argpartition(corr, -top_k)[-top_k:]

                grad += wa * (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                grad[L:] += wa * (1.0 - h[:N-L])
                grad[:N-L] -= wa * h[L:]
            elif L < 0 and L > -N:
                M = -L
                grad[:N-M] += wa * (1.0 - h[M:])
                grad[M:] -= wa * h[:N-M]

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    tgt = N // 2
    if n1 > tgt:
        idx = np.where(h_b == 1.0)[0]
        rng.shuffle(idx)
        h_b[idx[tgt:]] = 0.0
    elif n1 < tgt:
        idx = np.where(h_b == 0.0)[0]

def adam_run(h, N, target, deadline, lr0, decay):

    best_pk = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.85, 0.999, 1e-8

    v = np.zeros(N)
    it = 0

        corr = compute_corr(h)
