    idx = np.argpartition(corr, -top_k)[-top_k:]
    wts = corr[idx].copy()
    ws = wts.sum()
    if ws > 0: wts = wts / ws
    else: wts = np.ones(top_k) / top_k
    grad = np.zeros(n)
    for w, ki in zip(wts, idx):
        s = int(ki - (n - 1))
        for j in range(n):
            if 0 <= j + s < n:
                grad[j] += w * (1.0 - h[j + s])
            if 0 <= j - s < n:
                grad[j] -= w * h[j - s]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            room = h.copy()
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_constraint(x, n):
    return float(x.sum() - n / 2.0)

def soft_max_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def flat_top_obj(h, T, k, sw, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    soft = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    top = np.sort(corr)[::-1][:k]
    spread = top[0] - top[-1]
    return (soft + sw * spread) * dx

def topk_mean_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.mean(top) * dx)

def power_max_obj(h, p, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return float((np.sum(corr ** p)) ** (1.0 / p) * dx)

def slsqp(h, obj, args, mn):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': mn, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def targeted_gradient(h, n, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    wts = corr[idx].copy()
    ws = wts.sum()
    if ws > 0:
        wts = wts / ws
    else:
        wts = np.ones(top_k) / top_k
    grad = np.zeros(n)
    for w, ki in zip(wts, idx):
        s = int(ki - (n - 1))
        if s > 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        elif s < 0:
            grad[-s:] += w * (1.0 - h[:n + s])
            grad[:n + s] -= w * h[-s:]
        else:
            grad[:] += w * (1.0 - 2.0 * h)
    return grad

def nesterov_subgrad(h, n, steps, alpha, beta, top_k):
    best = h.copy()
    best_c = compute_c5(h)
    v = np.zeros(n)
    x = h.copy()
    for _ in range(steps):
        g = targeted_gradient(x, n, top_k)
        v = beta * v + (1.0 - beta) * g
        x_new = project_feasible(x - alpha * v, n)
        c = compute_c5(x_new)
        if c < best_c:
            best_c = c
            best = x_new.copy()
        x = x_new
        alpha *= 0.9995
    return best, best_c

def downsample_upsample_perturb(h, n, lo_res, scale):
    x_old = np.linspace(0, 2, n)
    x_lo = np.linspace(0, 2, lo_res)
    h_lo = np.interp(x_lo, x_old, h)
    h_lo = project_feasible(h_lo, lo_res)
    h_lo = project_feasible(h_lo + np.random.randn(lo_res) * scale, lo_res)
    h_new = np.interp(x_old, x_lo, h_lo)
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(100)
        best_h[:50] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Warm-up
    if time.time() - t0 < budget_s * 0.06:
        h1 = slsqp(best_h, soft_max_obj, (500, best_n), 300)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
        h2 = slsqp(best_h, flat_top_obj, (300, 5, 0.08, best_n), 300)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 2: Nesterov subgradient with multiple params
    if time.time() - t0 < budget_s * 0.15:
        for tk in [3, 5]:
            for a0 in [0.02, 0.008]:
                hn, cn = nesterov_subgrad(best_h, best_n, 400, a0, 0.85, tk)
                if cn < best_c5:
                    best_c5, best_h = cn, hn
                    hp = slsqp(hn, flat_top_obj, (300, 3, 0.05, best_n), 200)
                    cp = compute_c5(hp)
                    if cp < best_c5:
                        best_c5, best_h = cp, hp

    # Phase 3: Multi-objective SLSQP
    if time.time() - t0 < budget_s * 0.32:
        objs = [
            (soft_max_obj, (1000, best_n), 400),
            (soft_max_obj, (3000, best_n), 400),
            (flat_top_obj, (500, 3, 0.04, best_n), 400),
            (flat_top_obj, (500, 5, 0.1, best_n), 400),
            (flat_top_obj, (2000, 3, 0.015, best_n), 400),
            (topk_mean_obj, (3, best_n), 400),
            (topk_mean_obj, (5, best_n), 350),
            (power_max_obj, (30, best_n), 350),
        ]
        for obj_fn, args, mi in objs:
            if time.time() - t0 > budget_s * 0.32:
                break
            h = slsqp(best_h, obj_fn, args, mi)
            c = compute_c5(h)
            if c < best_c5:
                best_c5, best_h = c, h
                best_n = len(best_h)

    # Phase 4: Basin hopping with diverse moves
    max_trials = max(10, int(budget_s * 0.28 / 12))
    for trial in range(min(max_trials, 120)):
        if time.time() - t0 > budget_s * 0.62:
            break
        r = np.random.random()
        if r < 0.20:
            sc = 0.035 * (1.0 - 0.5 * trial / max_trials) + 0.004
            hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.38:
            g = targeted_gradient(best_h, best_n, 3)
            hp = project_feasible(best_h - 0.015 * g + np.random.randn(best_n) * 0.015, best_n)
        elif r < 0.55:
            lo = max(20, best_n // 4)
            hp = downsample_upsample_perturb(best_h, best_n, lo, 0.06)
        elif r < 0.70:
            L = np.random.randint(5, max(6, best_n // 6))
            start = np.random.randint(0, max(1, best_n - L))
            hp = best_h.copy()
            hp[start:start + L] = 1.0 - hp[start:start + L]
            hp = project_feasible(hp, best_n)
        elif r < 0.85:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            s = int(np.argmax(corr) - (best_n - 1))
            hp = best_h.copy()
            affected = set()
            for j in range(best_n):
                if 0 <= j + s < best_n:
                    affected.add(j)
                    affected.add(j + s)
                if 0 <= j - s < best_n:
                    affected.add(j)
                    affected.add(j - s)
            for j in affected:
                hp[j] += np.random.randn() * 0.08
            hp = project_feasible(hp, best_n)
        else:
            hp = best_h.copy()
            for _ in range(np.random.randint(2, 5)):
                i, j = np.random.choice(best_n, 2, replace=False)
                hp[i], hp[j] = hp[j], hp[i]
            hp = project_feasible(hp, best_n)

        h1 = slsqp(hp, soft_max_obj, (400, best_n), 180)
        c1 = compute_c5(h1)
        h2 = slsqp(hp, flat_top_obj, (300, 3, 0.04, best_n), 180)
        c2 = compute_c5(h2)
        h_best = h1 if c1 <= c2 else h2
        c_best = min(c1, c2)
        if c_best < best_c5:
            best_c5, best_h = c_best, h_best
            best_n = len(best_h)

    # Phase 5: Multi-resolution exploration
    for nr in [100, 150, 200, 250]:
        if time.time() - t0 > budget_s * 0.80:
            break
        if nr == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        for pert in [0.0, 0.012, 0.025]:
            if time.time() - t0 > budget_s * 0.80:
                break
            hu = np.interp(x_new, x_old, best_h)
            hu = project_feasible(hu, nr)
            if pert > 0:
                hu = project_feasible(hu + np.random.randn(nr) * pert, nr)
            h_up = slsqp(hu, soft_max_obj, (400, nr), 250)
            c_up = compute_c5(h_up)
            h_up2 = slsqp(hu, flat_top_obj, (300, 3, 0.03, nr), 250)
            c_up2 = compute_c5(h_up2)
            c_new = min(c_up, c_up2)
            h_new = h_up if c_up <= c_up2 else h_up2
            if c_new < best_c5:
                x_dn = np.linspace(0, 2, best_n)
                h_dn = np.interp(x_dn, x_new, h_new)
                h_dn = project_feasible(h_dn, best_n)
                c_dn = compute_c5(h_dn)
                if c_dn < best_c5:
                    best_c5, best_h = c_dn, h_dn
                else:
                    best_c5, best_h, best_n = c_new, h_new, nr
                    best_n = nr

    # Phase 6: Intensive final optimization
    if time.time() - t0 < budget_s * 0.94:
        final_objs = [
            (flat_top_obj, (1000, 3, 0.01, best_n), 500),
            (flat_top_obj, (2000, 3, 0.005, best_n), 500),
            (flat_top_obj, (2000, 5, 0.02, best_n), 400),
            (soft_max_obj, (5000, best_n), 500),
            (topk_mean_obj, (3, best_n), 500),

        for obj_fn, args, mi in final_objs:
            if time.time() - t0 > budget_s * 0.94:

            h = slsqp(best_h, obj_fn, args, mi)
            c = compute_c5(h)

    if time.time() - t0 < budget_s * 0.98:

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
