def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_softmax(h, N, beta, topM):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    s = corr - peak
    w = np.exp(beta * s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(corr)
        w[N - 1] = 1.0
    else:
        w = w / ws
    if topM < len(w):
        top_idx = np.argpartition(-w, topM)[:topM]
        mask = np.zeros(len(w), dtype=bool)
        mask[top_idx] = True
        w = w * mask
        w2 = w.sum()
        if w2 > 1e-30:
            w = w / w2
    grad = np.zeros(N)
    center = N - 1
    active = np.where(w > 1e-7)[0]
    for ai in active:
        wk = w[ai]
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_opt(h, N, target, deadline, lr0, decay, beta, topM):
    best_h = h.copy()

            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1

        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= decay
