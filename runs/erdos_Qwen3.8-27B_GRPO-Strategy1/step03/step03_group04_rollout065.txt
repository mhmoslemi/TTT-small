The key operation is: given the current correlation vector, find a swap (1→0 at position i, 0→1 at position j) that reduces the maximum.

Instead of trying random swaps, we can be more systematic:
1. Find the lag L* with maximum correlation C(L*)
2. The "contribution" of position i to C(L*) is:
   - If L* > 0: h[i]*(1-h[i+L*]) (if i+L* < N)
   - If L* < 0: h[i+|L*|]*(1-h[i]) (if i+|L*| < N)
3. Flipping h[i] from 1 to 0 reduces C(L*) by its contribution
4. Flipping h[j] from 0 to 1 increases C(L*) by (1-h[j+L*]) (or appropriate term)

So the net change in C(L*) is: -(contribution of i) + (new contribution of j)

To reduce C(L*), we want: contribution of i > new contribution of j.

But we also need to check other lags. The best swap might reduce C(L*) but increase C(L***) for another lag L***.

A practical approach:
1. Compute the full correlation vector
2. For the top T lags, compute the effect of each potential swap
3. Choose the swap that minimizes the maximum over all lags

C(L) = sum_n h[n] * (1 - h[n+L]) for L > 0 (with bounds)

The contribution of h[i] to C(L):
- As the "first" factor: h[i] * (1 - h[i+L]) if 0 ≤ i < N and 0 ≤ i+L < N
- As the "second" factor: h[i-L] * (1 - h[i]) if 0 ≤ i-L < N and 0 ≤ i < N and L < N

For binary h, (1-h[i]) is 1 if h[i]=0 and 0 if h[i]=1.

If h[i] = 1:
- As first factor: contributes (1-h[i+L]) to C(L)
- As second factor: contributes 0 (since 1-h[i] = 0)

If h[i] = 0:
- As first factor: contributes 0 (since h[i] = 0)
- As second factor: contributes h[i-L] to C(L)

So for a swap h[i]: 1→0 and h[j]: 0→1:
ΔC(L) = [new contribution of i as 0] - [old contribution of i as 1] + [new contribution of j as 1] - [old contribution of j as 0]

Old contribution of i (h[i]=1): (1-h[i+L]) [as first] + 0 [as second]
New contribution of i (h[i]=0): 0 [as first] + h[i-L] [as second]

Old contribution of j (h[j]=0): 0 [as first] + h[j-L] [as second]
New contribution of j (h[j]=1): (1-h[j+L]) [as first] + 0 [as second]

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def weighted_topk_gradient(h, N, k, alpha):
    corr = compute_corr(h)
    center = N - 1
    if k >= len(corr):
        top_idx = np.arange(len(corr))
        wts = np.ones(len(corr)) / len(corr)
    else:
        top_idx = np.argpartition(corr, -k)[-k:]
        tv = corr[top_idx]
        if alpha > 0:
            sh = tv - tv.max()
            wts = np.exp(alpha * sh)
            wts /= wts.sum()
        else:
            wts = np.full(k, 1.0 / k)
    grad = np.zeros(N)
    for idx, w in zip(top_idx, wts):
        L = int(idx) - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
        else:
            M = -L
            grad[:N - M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:N - M]
    return grad, float(corr.max())

def adam_loop(h, N, target, deadline, k_init, k_final, alpha, lr0):
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    mv = np.zeros(N)
    vv = np.zeros(N)
    it = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    total_est = max(deadline - time.time(), 1.0) * 25
    while time.time() < deadline:
        frac = min(1.0, it / total_est)
        k = max(2, int(k_init + (k_final - k_init) * frac))
        grad, _ = weighted_topk_gradient(h, N, k, alpha)
        mv = b1 * mv + (1 - b1) * grad
        vv = b2 * vv + (1 - b2) * grad * grad
        it += 1
        mh = mv / (1.0 - b1 ** it)
        vh = vv / (1.0 - b2 ** it)
        lr = max(lr0 * (0.9995 ** it), lr0 * 0.005)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 80 == 0:
            c = compute_c5(h, N)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    c = compute_c5(h, N)
    if c < best_c5:
        best_c5, best_h = c, h.copy()
    return best_h, best_c5

def binary_local_search(h_init, N, deadline, rng):
    h = (h_init > 0.5).astype(float)
    n_ones = int(h.sum())
    tgt = N // 2
    if n_ones > tgt:
        idx = np.where(h == 1)[0]
        h[idx[tgt:]] = 0
    elif n_ones < tgt:
        idx = np.where(h == 0)[0]
        h[idx[:tgt - n_ones]] = 1
    corr = compute_corr(h)
    cur_pk = float(corr.max())
    best_pk = cur_pk
    best_h = h.copy()
    stuck = 0
    while time.time() < deadline:
        max_lag = int(np.argmax(corr))
        L = max_lag - (N - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        found = False
        T = 8
        top_lags = np.argpartition(corr, -T)[-T:]
        # Precompute contribution vectors for top lags
        lags = (top_lags - (N - 1)).astype(int)
        def contrib(pos, lag):
            a = pos - lag
            b = pos + lag
            c = 0.0
            if 0 <= a < N:
                c += h[a]
            if 0 <= b < N:
                c += h[b]
            return c
        # Try targeted swaps: flip a 1 that contributes to worst lag
        if abs(L) < N:
            if L > 0:
                contrib_pos = np.where(h[:N - L] == 1)[0]
            else:
                M = -L
                contrib_pos = np.where(h[M:] == 1)[0] + M
            candidates = contrib_pos[:min(40, len(contrib_pos))]
        else:
            candidates = ones[:20]
        for i in candidates:
            hi = h[i]
            hi_l = np.array([h[max(0, i - l)] if i - l < N else 0.0 + h[max(0, i + l)] if i + l < N else 0.0 for l in lags])
            for j in zeros[rng.permutation(len(zeros))[:15]]:
                if j == i:
                    continue
                hj_l = np.array([h[max(0, j - l)] if j - l < N else 0.0 + h[max(0, j + l)] if j + l < N else 0.0 for l in lags])
                deltas = hi_l - hj_l
                new_top = corr[top_lags] + deltas
                if float(new_top.max()) < cur_pk - 1e-12:
                    h[i], h[j] = 0.0, 1.0
                    corr = compute_corr(h)
                    cur_pk = float(corr.max())
                    if cur_pk < best_pk:
                        best_pk = cur_pk
                        best_h = h.copy()
                    found = True
                    stuck = 0
                    break
            if found:
                break
        if not found:
            for _ in range(20):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                if i == j:
                    continue
                h[i], h[j] = 0.0, 1.0
                new_corr = compute_corr(h)
                new_pk = float(new_corr.max())
                if new_pk <= cur_pk + 1e-12:
                    corr = new_corr
                    cur_pk = new_pk
                    if new_pk < best_pk:
                        best_pk = new_pk
                        best_h = h.copy()
                    found = True
                    stuck = 0
                    break
                h[i], h[j] = 1.0, 0.0
            if not found:
                stuck += 1
                if stuck > 15:
                    break
    return best_h, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Phase 3: Adam refine around binary

        h3 = project_sum(h3, target)
        d3 = time.time() + (deadline - time.time()) * 0.30
