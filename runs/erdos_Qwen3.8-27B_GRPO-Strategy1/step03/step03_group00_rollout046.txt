import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    for _ in range(20):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * min(1.0, diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= h * min(1.0, -diff / tot)
        diff = target - h.sum()
    return h

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp(h, T, n, maxiter, ftol):
    def obj(x):
        return soft_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':con}],
                   options={'maxiter':maxiter,'ftol':ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_solve(h, n, i):

    dx = 2.0 / n
    # Remove contribution of h[i] from correlation
    h0 = h.copy()
    h0[i] = 0.0
    corr_base = np.correlate(h0, 1.0 - h0, mode='full')

    A = np.zeros(2*n-1)

        if shift != 0 and 0 <= i - shift < n:
            A[k] += (-h[i-shift])  # (1-t) * h[i-shift] → linear in t with coef -h[i-shift]
            # Wait: h[i-shift]*(1-t) = h[i-shift] - t*h[i-shift]
            # The constant part h[i-shift] is already in corr_base (since h0[i]=0)
            # Actually no: corr_base has h0[i]=0, so the term h[i-shift]*(1-h0[i]) = h[i-shift]*1
            # Wait, corr_base = correlate(h0, 1-h0). The term for index (i-shift) at lag shift is:
            # h0[i-shift] * (1-h0[i]) = h[i-shift] * 1 = h[i-shift] (since h0[i]=0)
            # For the new corr: h[i-shift] * (1-t)
            # So the change is: h[i-shift]*(1-t) - h[i-shift] = -t*h[i-shift]
            # This is already captured by A[k] += -h[i-shift]
            pass
        if shift == 0:
            # h[i]*(1-h[i]) = t*(1-t) = t - t²
            # corr_base has 0 here (h0[i]=0)
            A[k] += 1.0
            B[k] += -1.0
    
    def f1d(t):
        corr_new = corr_base + t * A + t*t * B
        return np.max(corr_new) * dx
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',

def coord_descent(h, n, t_end, max_sweeps=50, rng=None):

        improved = False

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * min(1.0, diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= h * min(1.0, -diff / tot)
    return h

def proj_box_sum(v, target):
    n = len(v)
    lo, hi = -v.max() - 2.0, -v.min() + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def soft_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ex = np.exp(T * (corr - m))
    w = ex / ex.sum()
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        wk = w[k]
        if wk < 1e-16:
            continue
        lag = k - (n - 1)
        j_plus = np.arange(n) + lag
        j_minus = np.arange(n) - lag
        mask_p = (j_plus >= 0) & (j_plus < n)
        mask_m = (j_minus >= 0) & (j_minus < n)
        g = np.zeros(n)
        g[mask_p] += (1.0 - h[j_plus[mask_p]]) * wk
        g[mask_m] -= h[j_minus[mask_m]] * wk
        grad += g
    return grad * dx

def prox_grad(h, T, n, n_iter, step_init):
    h_curr = h.copy()
    f_curr = soft_c5(h_curr, T)
    step = step_init
    for it in range(n_iter):
        g = soft_c5_grad(h_curr, T)
        for _ in range(10):
            h_new = proj_box_sum(h_curr - step * g, n / 2.0)
            f_new = soft_c5(h_new, T)
            if f_new < f_curr:

            step *= 0.5
        else:
            step = max(step, 1e-8)
            h_new = proj_box_sum(h_curr - step * g, n / 2.0)
            f_new = soft_c5(h_new, T)
        h_curr = h_new
        f_curr = min(f_curr, f_new)
    return h_curr

def slsqp_opt(h, T, n, maxiter=800, ftol=1e-14):
    def obj(x):
        return soft_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_solve(h, n, i):
    dx = 2.0 / n
    h0 = h.copy()
    h0[i] = 0.0
    corr_base = np.correlate(h0, 1.0 - h0, mode='full')
    A = np.zeros(2 * n - 1)
    B = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lap := (0 <= i + lag < n):
            A[k] += (1.0 - h[i + lag])
        if lap := (0 <= i - lag < n):
            A[k] += (-h[i - lag])
        if lag == 0:
            A[k] += 1.0
            B[k] += -1.0
    def f1d(t):
        return np.max(corr_base + t * A + t * t * B) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            t_new = coord_solve(best_h, n, i)
            trial = best_h.copy()
            trial[i] = t_new
            trial = project(trial, n)
            c5 = compute_c5(trial)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = trial
                improved = True
        if not improved:
            break
    return best_h, best_c5
