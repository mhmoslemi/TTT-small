def rng_escape(norm, size, scale):
    return np.zeros(size)

def polish_coords(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for i in range(N):
        if time.time() > deadline:
            break
        orig = best_h[i]
        for trial in (0.0, 1.0):
            if abs(trial - orig) < 1e-8:
                continue
            test = best_h.copy()
            test[i] = trial
            test = project_sum(test, target)
            c5 = compute_c5(test, dx)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = test
    for i in range(N):
        if time.time() > deadline:
            break
        orig = best_h[i]
        if orig < 0.05 or orig > 0.95:
            continue
        for delta in (-0.03, 0.03, -0.01, 0.01):
            trial = orig + delta
            if trial < 0.0 or trial > 1.0:
                continue
            test = best_h.copy()
            test[i] = trial
            test = project_sum(test, target)
            c5 = compute_c5(test, dx)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = test
                orig = trial
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    target0 = N0 / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / N0)

    # Phase 1: broad exploration (35%)
    p1_dead = t0 + budget_s * 0.35
    h1, c1 = nesterov_loop(best_h, N0, target0, p1_dead,
                           5.0, 100.0, 40, 8, 0.025, 0.90)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: narrowing focus (30%)
    if time.time() < deadline - 60:
        p2_dead = min(time.time() + budget_s * 0.30, deadline - 40)
        h2_start = best_h.copy()
        h2, c2 = nesterov_loop(h2_start, N0, target0, p2_dead,
                               50.0, 800.0, 8, 3, 0.008, 0.92)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 3: very focused fine-tuning (20%)
    if time.time() < deadline - 40:
        p3_dead = min(time.time() + budget_s * 0.20, deadline - 20)
        h3, c3 = nesterov_loop(best_h, N0, target0, p3_dead,
                               200.0, 5000.0, 3, 2, 0.002, 0.94)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3

    # Phase 4: retry from perturbed start if time allows (10%)
    if time.time() < deadline - 30:
        p4_dead = min(time.time() + budget_s * 0.08, deadline - 15)
        h4 = best_h + rng.randn(N0) * 0.005
        h4 = project_sum(h4, target0)
        h4, c4 = nesterov_loop(h4, N0, target0, p4_dead,
                               100.0, 1000.0, 5, 2, 0.004, 0.93)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4

    # Final polish
    if time.time() < deadline - 5:
        best_h, best_c5 = polish_coords(best_h, N0, target0, deadline - 2)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
