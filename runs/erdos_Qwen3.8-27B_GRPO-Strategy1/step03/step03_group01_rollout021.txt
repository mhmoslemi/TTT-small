import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    nc = len(corr)
    sc = corr / tau
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    grad = np.zeros(N)
    thresh = max(1e-5, 0.01 / nc)
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, float(np.max(corr)) * (2.0 / N)

def active_subgrad(h, N, eps_frac):
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    nc = len(corr)
    cmax = np.max(corr)
    span = corr.max() - corr.min()
    threshold = cmax - eps_frac * span
    active = np.where(corr >= threshold)[0]
    w = np.zeros(nc)
    w[active] = 1.0 / len(active)
    grad = np.zeros(N)
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, cmax * (2.0 / N)

        frac = min(it / float(n_iter_target), 1.0)

            tau = tau0 * (tau1 / tau0) ** frac
            grad, c5v = softmax_grad(h, N, tau)
        else:

            best_c5 = c5v

        h = project_sum(h - lr * v, target)
    return best_h, best_c5
