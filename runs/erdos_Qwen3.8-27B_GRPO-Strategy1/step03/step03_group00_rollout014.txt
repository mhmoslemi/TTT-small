def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:

    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=300)

    t3 = t0 + budget_s * 0.60
    h_bin, c5_bin = binary_local_search(best_h, n, t3, max_sweeps=50)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin
    # Unfreeze from binary: optimize a few random coordinates
    if time.time() < t3 + budget_s * 0.05:
        for _ in range(8):
            h_try = best_h + 0.05 * np.random.randn(n)
            h_try = project(h_try, n)
            hc = slsqp_opt(h_try, 800, n, maxiter=300, ftol=1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 4: Basin hopping with asymmetric escape (20%)
    t4 = t0 + budget_s * 0.80
    work_h = best_h.copy()
    work_c5 = best_c5
    it_count = 0
    while time.time() < t4:
        it_count += 1
        cool = max(0.0, 1.0 - it_count / 100.0)
        if np.random.rand() < 0.5:
            step = 0.015 + 0.10 * cool
            h_pert = work_h + step * np.random.randn(n)
        else:
            h_pert = asym_escape(work_h, n, t4)
        h_pert = project(h_pert, n)
        T_bh = 300 + 3000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=350, ftol=1e-13)
        # Quick coordinate polish
        hc, _ = coord_descent(hc, n, min(time.time() + 1.0, t4), max_sweeps=10)
        c5v = compute_c5(hc)
        temp = 0.001 + 0.005 * cool
        if c5v < work_c5 or np.random.rand() < np.exp(-max(0.0, c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Multi-resolution try (10%)
    t5 = t0 + budget_s * 0.90
    for n_try in [320, 200]:
        if time.time() > t5:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n_try)
        hc = slsqp_opt(h_interp, 1000, n_try, maxiter=400, ftol=1e-14)
        hc, c5v = coord_descent(hc, n_try, t5, max_sweeps=60)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            n = n_try

    # Phase 6: Final intensive coordinate descent (7%)
    t6 = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=300)

    # Final binary check
    if time.time() < hard_end - 5:
        h_b, c5_b = binary_local_search(best_h, n, hard_end - 2, max_sweeps=10)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Final SLSQP polish
    if time.time() < hard_end - 30:
        hc = slsqp_opt(best_h, 8000, n, maxiter=800, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, hard_end - 2, max_sweeps=30)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
