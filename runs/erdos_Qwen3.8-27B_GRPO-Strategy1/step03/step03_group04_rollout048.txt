def adam_step(h, N, target, deadline, k=20, lr0=0.05, decay=0.995, b1=0.9, b2=0.999):
    """Adam optimization on continuous h."""
    best_h = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    eps = 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(np.max(corr))
        if peak < best_pk:
            best_pk = peak
            best_h = h.copy()
        
        # Top-k gradient
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif 0 < L < N:
                grad[L:] += wk * (1.0 - h[:N-L])
                grad[:N-L] -= wk * h[L:]
            elif -N < L < 0:
                M = -L
                grad[:N-M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N-M]
        
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = h - lr0 * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr0 *= decay
    
    return best_h, best_pk * (2.0 / N)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    h_b = (h >= 0.5).astype(float)
    n_ones = int(h_b.sum())
    half = N // 2
    if n_ones > half:
        idx = np.where(h_b == 1)[0]
        order = idx[np.argsort(h[idx])]
        h_b[order[n_ones - half:]] = 0
    elif n_ones < half:
        idx = np.where(h_b == 0)[0]
        order = idx[np.argsort(-h[idx])]
        h_b[order[:half - n_ones]] = 1
    return h_b

    cur_pk = best_pk

        ones = np.where(cur == 1)[0]

        j = zeros[rng.randint(len(zeros))]
