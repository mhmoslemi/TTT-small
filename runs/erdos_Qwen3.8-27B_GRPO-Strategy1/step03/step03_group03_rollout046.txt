            grad[:n-kp] += -h[kp:]
    return grad / max(1, len(active)), len(active)

def nesterov_opt(h, n_iters, tol=1e-4, lr0=0.06):
    n = len(h)
    x = h.copy()
    y = h.copy()
    best_c5, _ = compute_c5(x)
    best_x = x.copy()
    lr = lr0
    for it in range(n_iters):
        if it % 15 == 0 and lr < 0.002:
            break
        _, corr = compute_c5(y)
        g, _ = compute_grad_active(y, corr, tol)
        x_new = project_feasible(y - lr * g, n)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        beta = it / t_k if it > 0 else 0.0
        y = x_new + beta * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        x = x_new
        if it % 25 == 0:
            c5, _ = compute_c5(x)
            if c5 < best_c5 - 1e-11:
                best_c5 = c5
                best_x = x.copy()
                lr = min(lr * 1.04, 0.25)
            else:
                lr *= 0.96
    c5, _ = compute_c5(x)
    if c5 < best_c5:
        best_c5 = c5
        best_x = x.copy()
    return best_x, best_c5

def smooth_erdos(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def spread_erdos(h, T, alpha, topm=15):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    srt = np.sort(corr)[-topm:]
    val += alpha * np.var(srt)
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=500, spread_alpha=0.0, topm=15):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        if spread_alpha > 0:
            obj = lambda x: spread_erdos(x, T, spread_alpha, topm)
            res = minimize(obj, h, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        else:
            res = minimize(smooth_erdos, h, args=(T,), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    return x[:-1].sum() - len(x[:-1]) / 2.0

def erdos_in(x):
    h = x[:-1]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - np.max(corr)

def slsqp_direct(h, maxiter=600):
    n = len(h)
    _, corr = compute_c5(h)
    t0 = float(np.max(corr)) + 0.01
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': erdos_eq}, {'type': 'ineq', 'fun': erdos_in}]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def binarize_closest(h, n):
    order = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[order[n // 2:]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: SLSQP warm-up
    for T in [300, 2000, 10000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=400)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Nesterov with wide active set
    if time.time() - t0 < budget_s * 0.30:
        h_new, c5_new = nesterov_opt(best_h, n_iters=4000, tol=2e-3, lr0=0.07)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: Spread objective SLSQP (encourage flat top of correlations)
    if time.time() - t0 < budget_s * 0.40:
        for alpha in [0.5, 2.0, 1.0]:
            if time.time() - t0 > budget_s * 0.40:
                break
            h_try = slsqp_smooth(best_h, 5000, maxiter=500, spread_alpha=alpha, topm=12)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 4: Nesterov with tighter active set
    if time.time() - t0 < budget_s * 0.50:
        h_new, c5_new = nesterov_opt(best_h, n_iters=3000, tol=5e-5, lr0=0.03)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 5: Direct SLSQP (minimax formulation)
    if time.time() - t0 < budget_s * 0.58:
        h_try = slsqp_direct(best_h, maxiter=800)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 6: Multi-n exploration
    for dn in [2, -2, 4, -4, 6]:
        if time.time() - t0 > budget_s * 0.74:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new_arr = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new_arr, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt = slsqp_smooth(h_interp, 5000, maxiter=300)
        h_opt, c5_opt = nesterov_opt(h_opt, n_iters=1500, tol=1e-3, lr0=0.05)
        c5_opt, _ = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try

    # Phase 7: Perturbation restarts with Nesterov + SLSQP
    for trial in range(10):
        if time.time() - t0 > budget_s * 0.88:
            break
        scale = 0.02 + 0.05 * np.random.rand()
        h_p = best_h + np.random.randn(best_n) * scale
        h_p = project_feasible(h_p, best_n)
        h_new, _ = nesterov_opt(h_p, n_iters=2000, tol=1e-3, lr0=0.05)
        h_new = slsqp_smooth(h_new, 15000, maxiter=400)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 8: Rotation exploration
    if time.time() - t0 < budget_s * 0.93:
        for shift in [1, 2, 3, 5, 13]:
            if time.time() - t0 > budget_s * 0.93:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_new, c5_new = nesterov_opt(h_rot, n_iters=2000, tol=1e-4, lr0=0.04)
            h_new = slsqp_smooth(h_new, 20000, maxiter=400)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 9: Final polish - ultra-high T SLSQP + direct
    if time.time() - t0 < budget_s * 0.98:
        h_f = slsqp_smooth(best_h, 100000, maxiter=500)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        h_d = slsqp_direct(best_h, maxiter=600)
        c5_d, _ = compute_c5(h_d)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d
        # Tiny Nesterov step at the end
        h_new, c5_new = nesterov_opt(best_h, n_iters=500, tol=1e-6, lr0=0.01)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
