- n=j: h[j] was 0, now 1. If 0<=j+L<N: gain (1-h[j+L])
- n=i-L (i.e., n+L=i): h[i] appears as h[n+L]. Was 1, now 0. If 0<=i-L<N: gain h[i-L] (since 1-h[i] goes from 0 to 1)

  Change: +h[i-L] if 0<=i-L<N

  Before: h[j-L]*(1-0) = h[j-L]
  After: h[j-L]*(1-1) = 0
  Change: -h[j-L] if 0<=j-L<N

def swap_delta(h_b, N, i, j):
    center = N - 1
    delta = np.zeros(2 * N - 1)
    
    # i: 1→0. As "first" (n=i): valid when 0<=i+L<N → L in [-i, N-1-i] → m in [center-i, center+N-1-i]
    m_lo = max(0, center - i)
    m_hi = min(2*N-2, center + N - 1 - i)
    if m_lo <= m_hi:
        Ls = np.arange(m_lo, m_hi+1) - center
        delta[m_lo:m_hi+1] -= (1.0 - h_b[i + Ls])
    
    # i: 1→0. As "second" (n+L=i → n=i-L): valid when 0<=i-L<N → L in [i-N+1, i] → m in [center+i-N+1, center+i]
    m_lo = max(0, center - (N - 1 - i))  # center + i - N + 1
    m_hi = min(2*N-2, center + i)
    if m_lo <= m_hi:
        Ls = np.arange(m_lo, m_hi+1) - center
        delta[m_lo:m_hi+1] += h_b[i - Ls]
    
    # j: 0→1. As "first" (n=j): valid when 0<=j+L<N → L in [-j, N-1-j] → m in [center-j, center+N-1-j]
    m_lo = max(0, center - j)
    m_hi = min(2*N-2, center + N - 1 - j)
    if m_lo <= m_hi:
        Ls = np.arange(m_lo, m_hi+1) - center
        delta[m_lo:m_hi+1] += (1.0 - h_b[j + Ls])
    
    # j: 0→1. As "second" (n+L=j → n=j-L): valid when 0<=j-L<N → L in [j-N+1, j] → m in [center+j-N+1, center+j]
    m_lo = max(0, center - (N - 1 - j))
    m_hi = min(2*N-2, center + j)
    if m_lo <= m_hi:
        Ls = np.arange(m_lo, m_hi+1) - center
        delta[m_lo:m_hi+1] -= h_b[j - Ls]
    
    return delta
