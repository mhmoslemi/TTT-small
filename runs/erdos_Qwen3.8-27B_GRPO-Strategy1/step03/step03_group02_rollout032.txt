import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s2 = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_max_grad(h, N, beta, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    n_tot = 2 * N - 1
    true_c5 = float(np.max(corr))
    if K < n_tot:
        idx = np.argpartition(corr, -K)[-K:]
        c_top = corr[idx]
    else:
        idx = np.arange(n_tot)
        c_top = corr
    m = c_top.max()
    e = np.exp(beta * (c_top - m))
    w = e / e.sum()
    grad = np.zeros(N)
    for i in range(len(idx)):
        wi = w[i]
        if wi < 1e-10:
            continue
        L = idx[i] - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

def optimize(h, N, target, dx, deadline, beta0, beta1, lr0, lr1, K):
    best_h = h.copy()

    v_buf = np.zeros(N)
    it = 0
    patience = 0
    min_improve = 1e-13
    while time.time() < deadline:
        it += 1
        frac = min(it / 500.0, 1.0)
        beta = beta0 * (beta1 / beta0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        grad, c5_val = soft_max_grad(h, N, beta, K)
        if c5_val < best_c5 - min_improve:
            best_c5 = c5_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        if patience > 150:
            break
        b1, b2, eps = 0.9, 0.999, 1e-8
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * grad * grad
        mh = m_buf / (1.0 - b1 ** it)
        vh = v_buf / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_c5

def pair_swap_search(h, N, target, dx, deadline, n_trials=8000, seed=7777):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i, j = rng.choice(N, size=2, replace=False)
        trial = best_h.copy()
        delta = rng.uniform(-0.03, 0.03)
        trial[i] += delta
        trial[j] -= delta
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_h = trial
    return best_h, best_c5

def random_perturb_search(h, N, target, dx, deadline, n_trials=5000, seed=4321):
