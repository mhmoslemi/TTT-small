        h_interp = project_feasible(np.interp(x_new, x_old, best_h), nc)
        h_try = slsqp_smooth(h_interp, 3000, 300, 20)
        h_try = slsqp_smooth(h_try, 40000, 300, 25)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            best_n = nc

    # Phase 4: Multi-start with diverse patterns
    starts = [best_h.copy(), make_thue_morse(best_n),
              make_random_binary(best_n, rng), make_block_pattern(best_n, rng)]
    starts.append(np.roll(best_h, 7))
    starts.append(np.roll(best_h, 19))
    pert = best_h + rng.randn(best_n) * 0.12
    starts.append(project_feasible(pert, best_n))
    pert2 = best_h + rng.randn(best_n) * 0.06
    starts.append(project_feasible(pert2, best_n))

    for h_start in starts:
        if elapsed() > budget_s * 0.72:
            break
        h_try = slsqp_smooth(h_start, 1500, 400, 20)
        h_try = slsqp_smooth(h_try, 20000, 400, 25)
        h_try = slsqp_smooth(h_try, 150000, 300, 30)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 5: High-T polishing
    if elapsed() < budget_s * 0.84:
        for T in [300000, 800000]:
            if elapsed() > budget_s * 0.84:
                break
            h_try = slsqp_smooth(best_h, T, 900, 35)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 6: Pair swap polish
    if elapsed() < budget_s * 0.92:
        best_h, best_c5 = pair_swap_optimize(best_h, 15)

    # Phase 7: Final refinement
    if elapsed() < budget_s * 0.97:
        h_try = slsqp_smooth(best_h, 500000, 700, 40)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        best_h, best_c5 = pair_swap_optimize(best_h, 6)

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
