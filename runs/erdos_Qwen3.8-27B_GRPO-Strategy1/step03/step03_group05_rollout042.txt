import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15: break
            h += r * (d / s)
        else:
            m = h > 1e-15
            s = h[m].sum()
            if s < 1e-15: break
            h[m] *= (1.0 + d / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_obj(h, k):
    corr = corr_full(h)
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top))

def grad_topk(h, k):
    """Gradient of sum of top-k correlations w.r.t. h."""
    n = len(h)
    corr = corr_full(h)
    idx = np.argsort(corr)[::-1][:k]
    w = np.full(k, 1.0 / k)
    grad = np.zeros(n)
    for ki, wi in zip(idx, w):
        shift = int(ki - (n - 1))
        if shift >= 0:
            if n - shift > 0:
                grad[:n-shift] += wi * (1.0 - h[shift:])
                grad[shift:] -= wi * h[:n-shift]
        else:
            sa = -shift
            if n - sa > 0:
                grad[sa:] += wi * (1.0 - h[:n-sa])
                grad[:n-sa] -= wi * h[sa:]
    return grad

def slsqp_obj_topk(h, k):
    n = len(h)
    return topk_obj(h, k) / n  # normalize

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_run(h, k, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(slsqp_obj_topk, h, args=(k,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent(h, k, n_iter):
    n = len(h)
    best = h.copy()
    best_val = topk_obj(h, k)
    for it in range(n_iter):
        improved = False
        for i in range(n):
            for delta in [0.01, 0.02, 0.05, -0.01, -0.02, -0.05]:
                if best[i] + delta < -0.01 or best[i] + delta > 1.01:
                    continue
                h_try = best.copy()
                h_try[i] = np.clip(h_try[i] + delta, 0, 1)
                h_try = project_feasible(h_try, n)
                v = topk_obj(h_try, k)
                if v < best_val - 1e-10:
                    best = h_try
                    best_val = v
                    improved = True
                    break
        if not improved:
            break
    return best, best_val

    t0 = time.time()

            c5n = compute_c5(best_h)
            if c5n < best_c5:

def grad_step(h, k, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:k]
    grad = np.zeros(n)
    for ki in idx:
        shift = int(ki - (n - 1))
        w = 1.0 / k
        if shift >= 0:
            if n - shift > 0:
                grad[:n-shift] += w * (1.0 - h[shift:])
                grad[shift:] -= w * h[:n-shift]
        else:
            sa = -shift
            if n - sa > 0:
                grad[sa:] += w * (1.0 - h[:n-sa])
                grad[:n-sa] -= w * h[sa:]
    h_new = project_feasible(h - alpha * grad, n)
    return h_new

                if v < -1e-12 or v > 1.0 + 1e-12:
                    continue
                best[i] = np.clip(v, 0.0, 1.0)
                best = project_feasible(best, n)
                val = topk_sum(best, k)
                if val < bv - 1e-13:
                    bv = val
                    imp = True
                    break
                best[i] = np.clip(orig, 0.0, 1.0)
                best = project_feasible(best, n)
        if not imp:
            break
    return best

def snap_binary(h, tol):
    h2 = h.copy()
    h2[h2 < tol] = 0.0
    h2[h2 > 1.0 - tol] = 1.0
    return project_feasible(h2, len(h2))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # P1: SLSQP equality-constrained, top-3
    if _elapsed(t0) < budget_s * 0.08:
        hs = slsqp_eq(best_h, 3, 600, 1e-13)
        cs = compute_c5(hs)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # P2: Analytic gradient descent (fast, many steps)
    if _elapsed(t0) < budget_s * 0.20:
        alpha = 2.0 / best_n
        for it in range(8000):
            if _elapsed(t0) > budget_s * 0.20:
                break
            if it % 1000 == 999:
                alpha *= 0.82
            g = grad_topk(best_h, 3)
            hn = project_feasible(best_h - alpha * g, best_n)
            if compute_c5(hn) < best_c5:
                best_c5 = compute_c5(hn)
                best_h = hn

    # P3: SLSQP top-5 (broader active set)
    if _elapsed(t0) < budget_s * 0.30:
        hs5 = slsqp_eq(best_h, 5, 500, 1e-13)
        cs5 = compute_c5(hs5)
        if cs5 < best_c5:
            best_c5, best_h = cs5, hs5

    # P4: Penalty SLSQP (different constraint handling)
    if _elapsed(t0) < budget_s * 0.38:
        for lam in [50.0, 500.0]:
            if _elapsed(t0) > budget_s * 0.38:
                break
            hp = slsqp_pen(best_h, 3, lam, 400, 1e-13)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_c5, best_h = cp, hp

    # P5: Coordinate descent coarse
    if _elapsed(t0) < budget_s * 0.50:
        dc = [0.03, 0.015, -0.03, -0.015]
        hc = coord_descent(best_h, 3, 60, dc)
        cc = compute_c5(hc)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    # P6: SLSQP intensive polish
    if _elapsed(t0) < budget_s * 0.58:
        hf = slsqp_eq(best_h, 3, 1200, 1e-14)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    # P7: Coordinate descent fine
    if _elapsed(t0) < budget_s * 0.68:
        df = [0.01, 0.005, 0.002, -0.01, -0.005, -0.002]
        hfd = coord_descent(best_h, 3, 80, df)
        cfd = compute_c5(hfd)
        if cfd < best_c5:
            best_c5, best_h = cfd, hfd

    # P8: Snap near-binary values and re-optimize
    if _elapsed(t0) < budget_s * 0.75:
        for tol in [0.03, 0.06]:
            if _elapsed(t0) > budget_s * 0.75:
                break
            hsnap = snap_binary(best_h, tol)
            hsr = slsqp_eq(hsnap, 3, 500, 1e-13)
            csr = compute_c5(hsr)
            if csr < best_c5:
                best_c5, best_h = csr, hsr

    # P9: Basin hopping with small perturbations
    for tr in range(35):
        if _elapsed(t0) > budget_s * 0.86:
            break
        sc = 0.012 * (1.0 + 0.5 * np.cos(0.4 * tr))
        hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        hr = slsqp_eq(hp, 3, 250, 1e-13)
        cr = compute_c5(hr)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    # P10: Multi-resolution search
    for nr in [150, 250]:
        if _elapsed(t0) > budget_s * 0.93:
            break
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, nr)
        hu = project_feasible(np.interp(xn, xo, best_h), nr)
        hh = slsqp_eq(hu, 3, 400, 1e-13)
        c5h = compute_c5(hh)
        xd = np.linspace(0, 2, best_n)
        hd = project_feasible(np.interp(xd, xn, hh), best_n)
        c5d = compute_c5(hd)
        if c5h < best_c5:
            best_c5, best_h, best_n = c5h, hh, nr
        elif c5d < best_c5:
            best_c5, best_h = c5d, hd

    # P11: Final micro-refinement
    if _elapsed(t0) < budget_s * 0.98:
        hf2 = slsqp_eq(best_h, 3, 800, 1e-15)
        cf2 = compute_c5(hf2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2
        dm = [0.005, 0.002, -0.005, -0.002]
        hm = coord_descent(best_h, 3, 50, dm)
        cm = compute_c5(hm)
        if cm < best_c5:
            best_c5, best_h = cm, hm
        g = grad_topk(best_h, 3)
        hn = project_feasible(best_h - 0.5 / best_n * g, best_n)
        cn = compute_c5(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
