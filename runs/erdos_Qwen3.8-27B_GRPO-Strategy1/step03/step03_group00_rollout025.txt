Let me also think about coordinate descent more carefully. The exact 1D subproblem for coordinate i is:
min_{h_i ∈ [0,1]} max_k (corr_base[k] + h_i * coef[k])

where corr_base is the correlation with h[i]=0, and coef[k] is the coefficient.

This is a 1D min-max problem. The function max_k (a_k + h_i * b_k) is the maximum of linear functions, hence convex in h_i. We can solve it with:
- Binary search on the value c: find the smallest c such that a_k + h_i * b_k ≤ c for all k, for some h_i ∈ [0,1]
- Or: the minimum of max of linear functions occurs where two (or more) lines cross, at a vertex of the upper envelope

For numerical solution, since it's 1D convex, minimize_scalar with bounded method works well.

But the key optimization: precomputing corr_base and coef for each i takes O(n²) time (computing the full correlation once). If we update one coordinate, we can update the correlation in O(n) time (since only n terms change).

Actually, for coordinate descent, after changing h[i], the correlation changes at n lags (the ones involving index i). So we can maintain the correlation array incrementally.

```python
def compute_coef(h, i, n):
    """Compute d(corr)/dh[i] as a vector of length 2n-1."""
    coef = np.zeros(2*n-1)
    # For lag k (indexed k+n-1 in coef):
    # coef[k+n-1] = (1-h[i+k]) if 0<=i+k<n else 0
    #              + (-h[i-k]) if 0<=i-k<n else 0
    # Vectorized:
    k_vals = np.arange(-(n-1), n)
    j1 = i + k_vals
    j2 = i - k_vals
    mask1 = (j1 >= 0) & (j1 < n)
    mask2 = (j2 >= 0) & (j2 < n)
    coef[mask1] += 1.0 - h[j1[mask1]]
    coef[mask2] += -h[j2[mask2]]
    return coef
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_to_feasible(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    k_vals = np.arange(-(n-1), n)
    
    sweeps = 0
    while time.time() < t_end:
        improved = False
        order = list(range(n))
        np.random.shuffle(order)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            j1 = i + k_vals
            j2 = i - k_vals
            mask1 = (j1 >= 0) & (j1 < n)
            mask2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(2*n-1)
            coef[mask1] += 1.0 - best_h[j1[mask1]]
            coef[mask2] += -best_h[j2[mask2]]
            
            hi_old = best_h[i]
            corr = np.correlate(best_h, 1.0 - best_h, 'full')
            base = corr - hi_old * coef
            
            # Minimize max(base + hi*coef) over hi in [0,1]
            # This is convex, use bounded minimization
            def f1d(hi):
                return np.max(base + hi * coef)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11})
            new_hi = res.x
            
            if abs(new_hi - hi_old) > 1e-10:
                best_h[i] = new_hi
                best_h = project_to_feasible(best_h, n)
                new_c5 = compute_c5(best_h)
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    improved = True
        sweeps += 1
        if not improved and sweeps > 3:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
```

**Another idea**: Instead of `minimize_scalar` for the 1D subproblem, I can solve it analytically. The function f(hi) = max_k (base_k + hi * coef_k) is the upper envelope of lines. The minimum occurs at:
- An endpoint (hi=0 or hi=1), or
- A point where two lines cross: hi* = (base_j - base_k) / (coef_k - coef_j) for some pair (j,k)

The number of pairs is O(n²) = O(100k) for n=160 (319 lags). But we can be smarter: the minimum of the upper envelope of lines occurs at a vertex of the convex hull of the lines (in the (coef, base) plane). We can compute this in O(n log n) time.

Actually, for 319 lines, even brute force O(n²) = ~50k comparisons is fast in NumPy.

```python
def solve_1d_minmax(base, coef, lo=0.0, hi=1.0):
    """Minimize max_k (base[k] + x*coef[k]) for x in [lo, hi]."""
    # Evaluate at endpoints
    f_lo = np.max(base + lo * coef)
    f_hi = np.max(base + hi * coef)
    best_x = lo if f_lo <= f_hi else hi
    best_f = min(f_lo, f_hi)
    
    # Check all pairwise intersections
    # x* = (base[j] - base[k]) / (coef[k] - coef[j])
    c = coef[:, np.newaxis]
    b = base[:, np.newaxis]
    dc = c - c.T  # coef[k] - coef[j] for all pairs (j,k)
    db = b.T - b  # base[j] - base[k]
    # x* = db / dc where dc != 0
    mask = np.abs(dc) > 1e-14
    with np.errstate(divide='ignore', invalid='ignore'):
        x_cross = db / dc
    x_cross = x_cross[mask]
    x_cross = x_cross[(x_cross > lo + 1e-12) & (x_cross < hi - 1e-12)]
    
    if len(x_cross) > 0:
        f_at_cross = np.max(base + np.tile(x_cross[:, np.newaxis], (1, len(base))) * coef[np.newaxis, :])
        # Hmm this is O(n * |cross_points|) which could be large
        # Alternative: for each x_cross, compute max(base + x_cross * coef)
        # Vectorize: f_cross = max over k for each x_cross
        # f_cross[m] = max_k (base[k] + x_cross[m]*coef[k])
```

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, 'full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp(h, T, n, iters=2000):
    def f(x):
        return smooth_max(x, T)
    def c(x):
        return x.sum() - n / 2.0
    r = minimize(f, h, method='SLSQP', bounds=[(0,1)]*n,
                 constraints=[{'type':'eq','fun':c}],
                 options={'maxiter':iters,'ftol':1e-15})
    return project(np.clip(r.x, 0, 1), n)

            m2 = (j2 >= 0) & (j2 < n)
            ci = np.zeros(2*n-1)
            ci[m1] += 1.0 - bh[j1[m1]]
            ci[m2] += -bh[j2[m2]]
            ho = bh[i]
            base = corr - ho * ci
            rr = minimize_scalar(lambda x: np.max(base + x*ci),
                                 bounds=(0,1), method='bounded',
                                 options={'xatol':1e-10})

                bh[i] = rr.x
                imp = True

        bh = project(bh, n)

        nc5 = np.max(corr) * dx
        if nc5 < bc5:
            bc5 = nc5
        elif not imp:
            break
    return bh, bc5
