import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return np.clip(np.full(n, target/n), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return np.clip(np.full(n, target/n), 0, 1)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=800, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_step(h, n, i):
    """Exact 1D optimization of h[i]."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        c = 0.0
        j1 = i + k
        j2 = i - k
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            if j2 != j1:
                c += (-h[j2])
            else:
                c += (-h[j2])  # k=0, j2==j1==i
        coef[k + n - 1] = c
    dx = 2.0 / n
    def f1d(hi):
        return np.max(corr_base + hi * coef) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200, verbose=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            hi_new = coord_step(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if verbose:
            print(f"  sweep {sweep}: C5={best_c5:.8f}, improved={improved}")
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # midpoints
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0/n_try
        h_interp = np.interp(x_new, x_old, best_h)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return np.clip(np.full(n, 0.5), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return np.clip(np.full(n, 0.5), 0, 1)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=800, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_step(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        c = 0.0
        j1 = i + k
        j2 = i - k
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            if j2 != j1:
                c += (-h[j2])
            else:
                c += (-h[j2])
        coef[k + n - 1] = c
    dx = 2.0 / n
    def f1d(hi):
        return np.max(corr_base + hi * coef) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            hi_new = coord_step(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def gradient_step(h, n, T, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if wk < 1e-10:
            continue
        for i in range(n):
            j1 = i + k
            j2 = i - k
            if 0 <= j1 < n:
                grad[i] += wk * (1.0 - h[j1])
            if 0 <= j2 < n and j2 != j1:
                grad[i] += wk * (-h[j2])
            elif k == 0 and j2 == j1:
                grad[i] += wk * (-h[j2])
    h_new = h - lr * grad
    h_new = project(h_new, n)
    return h_new

        return time.time() - t0
