    center = N - 1
    for ii, ai in enumerate(idx):
        wk = w[ii]
        if wk < 1e-12:
            continue
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_opt(h, N, target, deadline, k0, k1, beta, lr0, lr1):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.9, 0.999, 1e-9
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 800.0)
        k = max(2, int(k0 + (k1 - k0) * frac))
        lr = lr0 * (lr1 / lr0) ** (it / 2500.0)
        grad, peak = grad_smooth(h, N, beta, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_peak * (2.0 / N)

def fine_descent(h, N, target, deadline, k, lr_init, lr_min):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    it = 0
    while time.time() < deadline:
        it += 1
        lr = max(lr_init * (0.998 ** it), lr_min)
        if lr <= lr_min * 1.01:
            break
        grad, peak = grad_smooth(h, N, 40.0, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        h = h - lr * grad
        h = project_sum(h, target)
    return best_h, best_peak * (2.0 / N)

def anneal_binary(h, N, deadline, rng, T0, T1, rate):
    dx = 2.0 / N
    h_b = (h > 0.5).astype(float)
    target = N // 2
    n_ones = int(h_b.sum())
    if n_ones > target:
        idx = np.where(h_b == 1)[0]
        h_b[idx[target:]] = 0
    elif n_ones < target:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:target - n_ones]] = 1
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur_pk = best_pk
    cur = h_b.copy()
    n_steps = max(int((deadline - time.time()) * rate), 400)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / n_steps)
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i], cur[j] = cur[j], cur[i]
        new_pk = float(np.max(compute_corr(cur)))
        delta = new_pk - cur_pk
        if delta < 0 or T > 0.01 and rng.random() < np.exp(-delta / T):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_pk * dx

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True

def qr_pattern(N, rng):
    p = N
    while not is_prime(p):
        p += 1
    h = np.zeros(N)
    for i in range(N):
        h[i] = 1.0 if (i == 0 or pow(i, (p - 1) // 2, p) == 1) else 0.0
    n_ones = int(h.sum())
    tgt = N // 2
    if n_ones > tgt:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[tgt:]] = 0
    elif n_ones < tgt:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:tgt - n_ones]] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    target = N / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N)
    # Phase 1: multi-start Adam (35%)
    starts = [h0.copy(), qr_pattern(N, rng),
              project_sum(h0 + rng.randn(N) * 0.07, target),
              project_sum(h0 + rng.randn(N) * 0.14, target)]
    t1_total = (deadline - t0) * 0.35
    t_per = t1_total / len(starts)
    for s_idx in range(len(starts)):
        d = min(time.time() + t_per, deadline - 55)
        if time.time() > deadline - 55:
            break
        h_o, c_o = adam_opt(starts[s_idx].copy(), N, target, d, 5, 12, 20.0, 0.04, 0.003)
        if c_o < best_c5:
            best_c5, best_h = c_o, h_o.copy()
    # Phase 2: kick + optimize + anneal (35%)
    t2_total = (deadline - t0) * 0.35
    n_kicks = 4
    t_kick = t2_total / n_kicks
    for kk in range(n_kicks):
        if time.time() > deadline - 30:
            break
        kick = project_sum(best_h + rng.randn(N) * (0.06 - 0.01 * kk), target)
        d_a = time.time() + t_kick * 0.45
        h_k, c_k = adam_opt(kick, N, target, d_a, 3, 8, 25.0, 0.025, 0.002)
        d_b = time.time() + t_kick * 0.40
        h_k, c_k = anneal_binary(h_k, N, d_b, rng, 2.0, 0.004, 280)
        # Continuous refinement after binary
        if time.time() < deadline - 25:
            d_c = time.time() + t_kick * 0.14
            h_k = project_sum(h_k + rng.randn(N) * 0.03, target)
            h_k, c_k = adam_opt(h_k, N, target, d_c, 3, 6, 30.0, 0.015, 0.001)
        if c_k < best_c5:
            best_c5, best_h = c_k, h_k.copy()
    # Phase 3: fine polish (28%)
    t3 = deadline - time.time()
    if t3 > 20:
        d_fine = time.time() + t3 * 0.50
        h_f, c_f = fine_descent(best_h, N, target, d_fine, 3, 0.003, 1e-5)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
        if time.time() < deadline - 8:
            h_ann, c_ann = anneal_binary(best_h, N, deadline - 5, rng, 1.0, 0.002, 350)
            if c_ann < best_c5:
                best_c5, best_h = c_ann, h_ann
            if time.time() < deadline - 3:
                h_fin = project_sum(best_h + rng.randn(N) * 0.015, target)
                h_fin, c_fin = adam_opt(h_fin, N, target, deadline - 1, 2, 4, 40.0, 0.004, 0.0005)
                if c_fin < best_c5:
                    best_c5, best_h = c_fin, h_fin
    final_c5 = compute_c5(best_h, N)
    return best_h, final_c5, N
