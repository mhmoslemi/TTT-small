from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def sum_con(x):
    return x.sum() - len(x) / 2.0

def obj_topk(h, T, idx):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top = corr[idx]
    m = top.max()
    return m + np.log(np.sum(np.exp(T * (top - m)))) / T

def obj_all(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

                       bounds=bnds, constraints=cons,

    return project_feasible(np.clip(res.x, 0, 1), n)
