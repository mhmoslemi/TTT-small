        return x[-1]

        return x[:-1].sum() - len(x[:-1]) / 2.0
    def ineq(x):
        h_v = x[:-1]
        t = x[-1]
        c = np.correlate(h_v, 1.0 - h_v, mode='full')
        return t - c
    cons = [{'type': 'eq', 'fun': eq}, {'type': 'ineq', 'fun': ineq}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    def el():
        return time.time() - t0

    # Phase 1: Escalating T with analytic gradient SLSQP
    for T in [800, 4000, 15000, 60000, 250000]:
        if el() > budget_s * 0.18:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=1000)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Backtracking subgradient (long run)
    if el() < budget_s * 0.38:
        remain = budget_s * 0.20 - el()
        nit = max(2000, int(remain * 6))
        h_sub, c5_sub = backtrack_subgrad(best_h, max_iter=nit)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub

    # Phase 3: Another SLSQP polish at high T
    if el() < budget_s * 0.44:
        h_try = slsqp_smooth(best_h, 600000, maxiter=1200)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 4: Direct SLSQP
    if el() < budget_s * 0.52:
        h_try = direct_slsqp(best_h, maxiter=1000)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 5: Resolution sweep
    res_list = sorted(set(r for r in [best_n-4, best_n-2, best_n, best_n+2, best_n+4, best_n+6]
                         if r >= 50 and r % 2 == 0))
    for n_try in res_list:
        if el() > budget_s * 0.74:
            break
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_base = np.interp(x_new, x_old, best_h)
            h_base = project_feasible(h_base, n_try)
        h_try = slsqp_smooth(h_base, 100000, maxiter=800)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            best_n = n_try
        if el() < budget_s * 0.70:
            h_sub, c5_sub = backtrack_subgrad(h_base, max_iter=2000)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub
                best_n = n_try

    # Phase 6: Structured perturbations (boundary shifts)
    if el() < budget_s * 0.86:
        for i in range(10):
            if el() > budget_s * 0.86:
                break
            scale = 0.015 + 0.025 * np.random.rand()
            h_pert = best_h + np.random.randn(best_n) * scale
            h_pert = project_feasible(h_pert, best_n)
            h_opt = slsqp_smooth(h_pert, 80000, maxiter=500)
            h_opt, _ = backtrack_subgrad(h_opt, max_iter=1500)
            h_opt = slsqp_smooth(h_opt, 400000, maxiter=600)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 7: Cyclic shifts + optimization
    if el() < budget_s * 0.93:
        for shift in [1, 2, 3, 5, 7, 11]:
            if el() > budget_s * 0.93:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_opt, _ = backtrack_subgrad(h_rot, max_iter=2500)
            h_opt = slsqp_smooth(h_opt, 500000, maxiter=600)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 8: Final ultra-high T polish + direct
    if el() < budget_s * 0.98:
        h_try = slsqp_smooth(best_h, 2000000, maxiter=1500)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        h_try = direct_slsqp(best_h, maxiter=1200)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
