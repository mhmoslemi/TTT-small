def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
        best_h = project_feasible(best_h, 200)

    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: Progressive L-BFGS-B from initial
    for T in [20, 50, 100, 300, 800, 2000, 6000, 20000, 80000]:
        if time.time() - t0 > budget_s * 0.22:
            break
        h_new = lbfgs_opt(best_h.copy(), T, maxiter=4000, gtol=1e-10)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    # Phase 2: Periodic structure exploration
    t_limit2 = budget_s * 0.42
    for p in [2, 3, 4, 5, 6, 7, 8, 10, 12, 14, 15, 16, 20, 24, 25, 30, 32, 40, 50]:
        if time.time() - t0 > t_limit2:
            break
        if n % p != 0:
            continue
        for trial in range(3):
            if time.time() - t0 > t_limit2:
                break
            if trial == 0:
                xp = np.array(np.resize(best_h, p), dtype=float)
                xp = xp / max(xp.sum(), 1e-10) * (p / 2.0)
            elif trial == 1:
                freq = 1 + (trial - 1)
                xp = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * np.arange(p) / p + 0.5)
                xp = project_feasible(xp, p)
            else:
                xp = np.random.uniform(0.1, 0.9, p)
                xp = project_feasible(xp, p)
            for T in [80, 400]:
                if time.time() - t0 > t_limit2:
                    break
                h_try = periodic_opt(xp.copy(), p, n, T, maxiter=1500)
                c5v = compute_c5(h_try)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_try

    # Phase 3: Basin hopping
    n_bh = len(best_h)
    bh_h = best_h.copy()
    bh_c5 = best_c5
    t_limit3 = budget_s * 0.70
    max_it = 300
    for it in range(max_it):
        if time.time() - t0 > t_limit3:
            break
        step = 0.18 * (1.0 - it / float(max_it)) + 0.03
        ptype = it % 4
        if ptype == 0:
            h_pert = bh_h + step * np.random.randn(n_bh)
        elif ptype == 1:
            h_pert = bh_h.copy()
            blen = max(2, n_bh // 8)
            s1 = np.random.randint(0, n_bh - blen)
            s2 = np.random.randint(0, n_bh - blen)
            tmp = h_pert[s1:s1+blen].copy()
            h_pert[s1:s1+blen] = h_pert[s2:s2+blen]
            h_pert[s2:s2+blen] = tmp
        elif ptype == 2:
            h_pert = bh_h.copy()
            nflip = max(1, n_bh // 5)
            idx = np.random.choice(n_bh, nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        else:
            h_pert = bh_h.copy()
            mask = np.random.rand(n_bh) < 0.3
            h_pert[mask] = 0.5 + 0.4 * np.random.randn(mask.sum())

        h_pert = project_feasible(h_pert, n_bh)
        T_use = 200 + 300 * (it // 20)
        try:
            h_new = lbfgs_opt(h_pert, T_use, maxiter=800, gtol=1e-7)
            c5v = compute_c5(h_new)
        except Exception:
            h_new = h_pert
            c5v = compute_c5(h_pert)

        temp = 0.005 * (1.0 - it / float(max_it)) + 0.0002
        if c5v < bh_c5 or (c5v > bh_c5 and np.random.rand() < np.exp(-(c5v - bh_c5) / max(temp, 1e-10))):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 4: Multi-resolution with different n
    t_limit4 = budget_s * 0.87
    for n_try in [100, 120, 150, 160, 240, 300, 400]:
        if time.time() - t0 > t_limit4:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        for T in [80, 400, 2000]:
            if time.time() - t0 > t_limit4:
                break
            h_new = lbfgs_opt(h_interp.copy(), T, maxiter=2500, gtol=1e-9)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
                x_old = x_new
                break

    # Phase 5: Aggressive coordinate descent
    n_cur = len(best_h)
    t_limit5 = budget_s * 0.94
    if time.time() - t0 < t_limit5:
        max_cd = int((t_limit5 - (time.time() - t0)) / 0.3)
        max_cd = max(max_cd, 200)
        best_h, best_c5 = coord_descent(best_h, n_cur, max_iter=max_cd)

    # Phase 6: Final high-T polish
    for T in [20000, 100000, 500000]:
        if time.time() - t0 > budget_s * 0.995:
            break
        try:
            h_new = lbfgs_opt(best_h.copy(), T, maxiter=8000, gtol=1e-12, pen=1000.0)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
