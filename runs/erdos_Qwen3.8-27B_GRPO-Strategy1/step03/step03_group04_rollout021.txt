def fft_correlate(h):
    """Compute np.correlate(h, 1-h, mode='full') using FFT."""
    N = len(h)
    b = 1.0 - h
    size = 2 * N - 1
    nfft = 1
    while nfft < size:
        nfft *= 2
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(b, nfft)
    # correlate(h, b) = ifft(conj(H) * B) but we need the right alignment
    # Actually np.correlate(a, v) = ifft(conj(FFT(a)) * FFT(v)) with zero-padding
    # But the alignment... let me just use np.correlate for simplicity since N is small
    return np.correlate(h, b, mode='full')

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_to_binary(h, N):
    """Project h to binary with exactly N/2 ones."""
    h = np.clip(h, 0, 1)
    # Keep top N/2 values as 1
    thresh = np.partition(h, N//2)[N//2]
    h_b = (h >= thresh).astype(float)
    # Fix if we have more or fewer than N/2 ones due to ties
    n_ones = int(h_b.sum())
    if n_ones > N//2:

        order = np.argsort(-h[idx])

        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        h_b[idx[order[:N//2-n_ones]]] = 1
    assert int(h_b.sum()) == N//2
    return h_b

import numpy as np
import time

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N):
    """Convert continuous h to binary with exactly N/2 ones."""
    order = np.argsort(-h)
    h_b = np.zeros(N)
    h_b[order[:N//2]] = 1.0
    return h_b

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def build_G(h, N):
    """Build G matrix: G[m, b] = 2h[m-b]-1 for valid, 0 at m=2b, 0 otherwise."""
    g = 2.0 * h - 1.0
    G = np.zeros((2*N-1, N))
    for b in range(N):
        hi = min(b + N, 2*N - 1)
        G[b:hi, b] = g[:hi-b]
        if 2*b < 2*N - 1:
            G[2*b, b] = 0.0
    return G

def binary_steepest(h_b, N, deadline, max_rounds=100):
    """Steepest descent on binary swaps."""
    h = h_b.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = float(np.max(corr))
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        G = build_G(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_new_max = best_max
        best_a, best_b = -1, -1
        
        for a in zeros:
            g = 2.0 * h - 1.0
            va = np.zeros(2*N - 1)
            hi = min(a + N, 2*N - 1)
            va[a:hi] = -g[:hi-a]
            if 2*a < 2*N - 1:
                va[2*a] = 0.0
            
            base = corr + va
            new_max_all = np.max(base[:, None] + G, axis=0)
            ones_vals = new_max_all[ones]
            min_idx = np.argmin(ones_vals)
            min_val = float(ones_vals[min_idx])
            if min_val < best_new_max - 1e-12:
                best_new_max = min_val
                best_a = a
                best_b = ones[min_idx]
        
        if best_a < 0:
            break
        
        h[best_a], h[best_b] = h[best_b], h[best_a]
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    
    return best_h, best_max * (2.0 / N)

def binary_anneal(h_b, N, deadline, rng, max_iters=5000):
    """Annealing on binary swaps, with steepest descent restarts."""
    h = h_b.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_max = float(np.max(corr))
    cur_max = best_max
    T0, T1 = 5.0, 0.01
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1/T0) ** (it / max_iters)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            corr = new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_max * (2.0 / N)

def adam_refine(h, N, target, deadline, lr0=0.05, k=20):
    """Adam optimization on continuous h."""
    h = h.copy()
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1-h, mode='full')))
    dx = 2.0 / N
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    lr = lr0
    it = 0
    center = N - 1
    decay = 0.995
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        # Top-k lags
        if k >= len(corr):
            top = np.arange(len(corr))
        else:
            top = np.argpartition(corr, -k)[-k:]
        
        grad = np.zeros(N)
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                grad[L:] += wk * (1.0 - h[:N-L])
                grad[:N-L] -= wk * h[L:]
            elif L < 0 and -L < N:
                M = -L
                grad[:N-M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N-M]
        
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
    
    return best_h, best_peak * dx
