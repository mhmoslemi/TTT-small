        idx_top = np.argsort(corr)[-k:]
        wt = np.zeros(nc)
        wt[idx_top] = 1.0 / k
        w = 0.6 * wp + 0.4 * wt
        thresh = 0.002
        active = np.where(w > thresh)[0]
        grad_h = np.zeros(N)
        for ai in active:
            L = ai - center
            grad_h += w[ai] * lag_gradient(h, N, L)
        grad_f = grad_h[:n2] - grad_h[n2:]
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_f = f.copy()
        v = mom * v + (1.0 - mom) * grad_f
        f_new = f - lr * v
        f = np.clip(f_new, 0.0, 1.0)
        if it % 80 == 0:
            lr *= decay
            if lr < 1e-13:
                break
    return best_f, best_c5

def polish_push(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        h = best_h.copy()
        order = np.argsort(h)
        for i in order:
            if time.time() > deadline:
                break
            val = h[i]
            if val < 0.02 or val > 0.98:
                continue
            trial = 0.0 if val < 0.5 else 1.0
            h[i] = trial
            h = project_sum(h, target)
            c5 = compute_c5(h, dx)
            if c5 < best_c5 - 1e-11:
                best_c5 = c5
                best_h = h.copy()
                improved = True
                h = best_h.copy()
    return best_h, best_c5

def swap_polish(h, N, target, deadline):
    """Try swapping pairs to maintain sum while reducing C5."""
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        h = best_h.copy()
        n = len(h)
        order_lo = np.argsort(h)[:n // 4]
        order_hi = np.argsort(h)[-n // 4:]
        for i in order_lo:
            if time.time() > deadline:
                break
            for j in order_hi:
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-11:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break
            h = best_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / N0)

    # Phase 1: Full optimization on initial grid
    t1 = min(250, budget_s * 0.28)
    h1, c1 = optimize_h(h0.copy(), N0, N0 / 2.0, t0 + t1,
                        2.0, 60.0, 18, 3, 0.035, 0.97, 0.85)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: Antisymmetric optimization (reduces dim, no sum constraint)
    n2 = N0 // 2
    t2 = min(220, budget_s * 0.25)
    f_init = best_h[:n2].copy()
    fA, cA = optimize_antisym(f_init, N0, t0 + t2,
                              3.0, 100.0, 14, 3, 0.025, 0.97, 0.88)
    hA = np.concatenate([fA, 1.0 - fA])
    if cA < best_c5:
        best_c5 = cA
        best_h = hA

    # Phase 2b: Random antisymmetric start
    if time.time() < deadline - 180:
        tt = min(150, deadline - 200 - time.time())
        x = np.linspace(0, 1, n2, endpoint=False)
        fr = np.clip(0.5 + 0.35 * np.sin(4.7 * np.pi * x + 0.8) + rng.randn(n2) * 0.15, 0, 1)
        fB, cB = optimize_antisym(fr, N0, time.time() + tt,
                                  3.0, 80.0, 12, 3, 0.028, 0.97, 0.88)
        hB = np.concatenate([fB, 1.0 - fB])
        if cB < best_c5:
            best_c5 = cB
            best_h = hB

    # Phase 3: Upscale and refine
    for Ns in [480, 580]:
        remains = deadline - 100 - time.time()
        if remains < 40:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, Ns, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_sum(h_up, Ns / 2.0)
        tt = min(remains * 0.6, 150)
        h_ref, c_ref = optimize_h(h_up, Ns, Ns / 2.0, time.time() + tt,
                                  8.0, 200.0, 8, 2, 0.012, 0.98, 0.90)
        if c_ref < best_c5:
            best_c5 = c_ref
            best_h = h_ref

    # Phase 4: Downscale back to smaller N for better precision
    if len(best_h) > N0 and time.time() < deadline - 50:
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_ds = np.linspace(0, 2, N0, endpoint=False)
        h_ds = np.interp(x_ds, x_best, best_h)
        h_ds = project_sum(h_ds, N0 / 2.0)
        tt = min(60, deadline - 50 - time.time())
        h_ds, c_ds = optimize_h(h_ds, N0, N0 / 2.0, time.time() + tt,
                                50.0, 500.0, 4, 2, 0.004, 0.985, 0.92)
        if c_ds < best_c5:
            best_c5 = c_ds
            best_h = h_ds

    # Phase 5: Polish
    if time.time() < deadline - 8:
        Nb = len(best_h)
        best_h, best_c5 = polish_push(best_h, Nb, Nb / 2.0, deadline - 3)

    # Final feasibility check
    best_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
