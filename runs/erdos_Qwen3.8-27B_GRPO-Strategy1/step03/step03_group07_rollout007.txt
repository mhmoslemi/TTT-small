def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t > 1e-12:
            h = h + room * (diff / t)
    else:
        t = h.sum()
        if t > 1e-12:
            h = h + h * (diff / t)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr) + 1e-30
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

        return x.sum() - n / 2.0

    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)
