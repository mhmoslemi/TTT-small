from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < 0: m = 0
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14: return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14: return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_analytical(h, n, j):
    """Find optimal h[j] in [0,1] by solving min max of linear functions."""
    # Compute alpha_m and beta_m for each correlation index m
    # c[m](x) = alpha_m + beta_m * x
    g = 1.0 - h
    corr_full = np.correlate(h, g, mode='full')
    # For coordinate j, compute beta_m (d c[m] / d x)
    # c[m] = sum_i h[i]*g[i-lag] where lag = m-(n-1)
    # h[j] appears when i=j (term: x*g[j-lag]) or i-lag=j i.e. i=j+lag (term: h[j+lag]*(1-x))
    beta = np.zeros(2*n-1)
    for m in range(2*n-1):
        lag = m - (n-1)
        b = 0.0
        # Term i=j: valid if j in valid range for this lag
        # valid range: max(0,lag) <= i < min(n, n+lag)
        i_lo = max(0, lag)
        i_hi = min(n, n + lag)
        if i_lo <= j < i_hi:

            idx = j - lag
            if 0 <= idx < n:
                b += (1.0 - h[idx])
        # Term i=j+lag: valid if j+lag in valid range
        i2 = j + lag
        if i_lo <= i2 < i_hi:
            idx2 = i2 - lag  # = j
            # h[i2] * g[i2-lag] = h[j+lag] * (1-h[j])
            # coefficient of x is -h[j+lag]
            if 0 <= i2 < n:
                b -= h[i2]
        beta[m] = b
    # alpha_m = c[m] - beta_m * h[j] (current value minus the x contribution)
    alpha = corr_full - beta * h[j]
    
    # Now minimize max_m (alpha_m + beta_m * x) for x in [0,1]
    # This is a convex 1D problem. Find all pairwise intersections in [0,1].
    # Evaluate at 0, 1, and all intersections.
    candidates = [0.0, 1.0]
    # Find intersections: alpha_i + beta_i*x = alpha_j + beta_j*x
    # x = (alpha_j - alpha_i) / (beta_i - beta_j)
    for i in range(len(beta)):
        for j in range(i+1, len(beta)):

            if abs(db) > 1e-12:
                x = (alpha[j] - alpha[i]) / db
                if -0.5 <= x <= 1.5:  # slight tolerance

        v = np.max(vals)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_beta(h, n, j):
    beta = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        lag = m - (n - 1)
        b = 0.0
        i_lo = max(0, lag)
        i_hi = min(n, n + lag)
        if i_lo <= j < i_hi:
            idx = j - lag
            if 0 <= idx < n:
                b += (1.0 - h[idx])
        i2 = j + lag
        if 0 <= i2 < n and i_lo <= i2 < i_hi:
            b -= h[i2]
        beta[m] = b
    return beta

def coord_solve_1d(alpha, beta):
    lo, hi = 0.0, 1.0
    for _ in range(45):

        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(alpha + beta * m1)
        f2 = np.max(alpha + beta * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return 0.5 * (lo + hi)

    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    order = rng.permutation(n)
    for j in order:
        beta = coord_beta(h, n, j)
        alpha = corr - h[j] * beta
        x_new = coord_solve_1d(alpha, beta)
        if abs(x_new - h[j]) > 1e-14:
            corr += (x_new - h[j]) * beta
            h[j] = x_new
    return h, np.max(corr) * (2.0 / n)

def slsqp_polish(h, n, T, maxiter=600):
    def obj(x):
        return smooth_c5(x, T)
    def confun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': confun}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
