from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_topk(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = get_correlation(h)
    if k is None or k >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argpartition(corr, -k)[-k:]
    topk = corr[idx]
    m = np.max(topk)
    val = m + np.log(np.sum(np.exp(T * (topk - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t < 1e-12: return h
        h += room * (diff / t)
    else:
        room = h.copy()
        t = room.sum()
        if t < 1e-12: return h
        h += room * (diff / t)
    return np.clip(h, 0.0, 1.0)

def slsqp(h0, T, k, n, maxiter=800, ftol=1e-15):
    def obj(x):
        return smooth_topk(x, T, k)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, n_iter=100):
    """Reduce worst shift by adjusting individual coordinates."""
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iter):
        corr = get_correlation(best)
        worst = np.argmax(corr)
        shift = worst - (n - 1)
        improved = False
        # Try adjusting each coordinate
        # Gradient of C_shift w.r.t. h[j]:
        # = (1 - h[j+shift]) * I[j+shift < n] - h[j-shift] * I[j-shift >= 0]
        for j in range(n):
            g = 0.0
            if j + shift < n:
                g += (1.0 - best[j + shift])
            if j - shift >= 0:
                g -= best[j - shift]
            if g == 0:
                continue
            # Step in direction -g (to reduce C_shift)
            for delta in [0.05 * (-np.sign(g)), 0.1 * (-np.sign(g)), 0.02 * (-np.sign(g))]:
                h_try = best.copy()

                h_try = project(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_c5

    corr = np.correlate(h, 1.0 - h, mode='full')
    worst_m = int(np.argmax(corr))
    s = worst_m - (n - 1)  # shift
    g = np.zeros(n)
    for j in range(n):
        if s >= 0:
            if j < n - s:
                g[j] += (1.0 - h[j + s])
            if j >= s:
                g[j] -= h[j - s]
        else:
            ss = -s
            if j < n - ss:
                g[j] += (1.0 - h[j + ss])
            if j >= ss:
                g[j] -= h[j - ss]
    return g, s

def coord_search(h, n, n_iter=150):

def interval_start(K, n, seed):

C5 = max_k ∫h(x)(1-h(x+k))dx

If h is a "block" function (1 on a set of measure 1, 0 elsewhere), then for a shift k where the blocks don't overlap at all:
∫h(x)(1-h(x+k))dx = ∫h(x)dx = 1.0 (if h(x+k)=0 wherever h(x)>0)

For a shift where h(x) and h(x+k) are the same (blocks align perfectly):
∫h(x)(1-h(x+k))dx = 0.0

So the question is: can we arrange the blocks so that for ALL shifts, the overlap ∫h(x)h(x+k)dx is close to ∫h(x)dx = 1?

This is essentially asking for a function whose autocorrelation is as flat as possible. The autocorrelation R(τ) = ∫h(x)h(x+τ)dx should be close to 1 for all τ.

But R(0) = ∫h(x)²dx. For a 0/1 function, R(0) = ∫h = 1. And R(τ) ≤ R(0) = 1 for all τ (by Cauchy-Schwarz or rearrangement).

So C5 = max_k (1 - R(k)) = 1 - min_k R(k).

To minimize C5, we want to maximize min_k R(k). This is a "maximin autocorrelation" problem!

For a 0/1 function with ∫h = 1 on [0,2], the autocorrelation R(τ) = ∫h(x)h(x+τ)dx measures the overlap of h with its τ-shift.

The best case: R(τ) ≈ 1 for all τ ≠ 0. But this is impossible because R(0) = 1 and R is continuous (for nice h), so R must drop somewhere.

For a "random" 0/1 function, R(τ) ≈ 1/2 for τ ≠ 0 (by concentration of measure). So C5 ≈ 1/2.

For a carefully designed function, we can do better. The key insight is that we want h to be "as stationary as possible" - its overlap with shifted versions should be as uniform as possible.

This is related to the concept of "good sets" in harmonic analysis, or "low-discrepancy sequences."

One construction: h(x) = 1 for x in a "V兰德" set (or similar equidistributed set). For example, the set of x where the fractional part of x·α (for irrational α) falls in [0, 1/2]. But this gives R(τ) ≈ 1/2 for most τ, which is the random baseline.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_topk_obj(h, args):
    T, k = args[0], args[1]
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k is None or k >= len(corr):
        topk = corr
    else:
        topk = np.sort(corr, kind='stable')[-k:]
    m = float(np.max(topk))
    return (m + float(np.log(np.sum(np.exp(T * (topk - m))))) / T) * dx

def sum_con(x, args):
    return x.sum() - args[2] / 2.0

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t < 1e-12:
            return h
        h += room * (diff / t)
    else:
        t = h.sum()
        if t < 1e-12:
            return h
        h += h * (diff / t)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h0, T, k, n, maxiter=600, ftol=1e-15):
    args = (T, k, n)
    cons = [{'type': 'eq', 'fun': sum_con, 'args': args}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_topk_obj, h0, method='SLSQP',
                   bounds=bnds, constraints=cons, args=args,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(np.clip(res.x, 0.0, 1.0), n)

def _grad_for_shift(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        n1 = n - s
        g[:n1] += (1.0 - h[s:])
        g[s:] -= h[:n1]
    else:
        ss = -s
        n1 = n - ss
        g[ss:] += (1.0 - h[:n1])
        g[:n1] -= h[ss:]
    return g

def grad_descent_topk(h, n, n_steps=500, T=30, k_top=5):
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = 0.03
    for step in range(n_steps):
        corr = np.correlate(best, 1.0 - best, mode='full')
        idx = np.argpartition(corr, -k_top)[-k_top:]
        topk = corr[idx]
        m = float(np.max(topk))
        w = np.exp(T * (topk - m))
        w /= w.sum()
        g = np.zeros(n)
        for i in range(k_top):
            s = idx[i] - (n - 1)
            g += w[i] * _grad_for_shift(best, s, n)
        h_new = project_h(best - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best = h_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.6
            if lr < 1e-7:
                break
    return best, best_c5

def coord_search(h, n, n_iter=150):
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iter):
        corr = np.correlate(best, 1.0 - best, mode='full')
        mt = int(np.argmax(corr))
        s = mt - (n - 1)
        g = _grad_for_shift(best, s, n)
        improved = False
        order = np.argsort(-np.abs(g))
        for j in order[:40]:
            if abs(g[j]) < 1e-8:
                continue
            d = -np.sign(g[j])
            for st in (0.05, 0.025, 0.01, 0.08):
                ht = best.copy()
                ht[j] += d * st
                ht = project_h(ht, n)
                cv = compute_c5(ht)
                if cv < best_c5 - 1e-14:
                    best, best_c5 = ht, cv
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_c5

def interval_start(K, n, sd):

    for i in range(K):
        pos += gp[i]
        h[(x >= pos) & (x < pos + w[i])] = 1.0
        pos += w[i]
    return project_h(h, n)

            g[s:] -= h[:n1]
    else:
        ss = -s
        n1 = n - ss
        if n1 > 0:
            g[ss:] += (1.0 - h[:n1])
            g[:n1] -= h[ss:]
    return g

def grad_descent_topk(h, n, n_steps=400, T=20, k_top=5):
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = 0.03
    stall = 0
    for step in range(n_steps):
        corr = np.correlate(best, 1.0 - best, mode='full')
        idx = np.argpartition(corr, -k_top)[-k_top:]
        topk = corr[idx]
        m = float(np.max(topk))
        w = np.exp(T * (topk - m))
        w /= w.sum()
        g = np.zeros(n)
        for i in range(k_top):
            s = int(idx[i]) - (n - 1)
            g += w[i] * _grad_for_shift(best, s, n)
        h_new = project_h(best - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5, best = c5_new, h_new
            lr = min(lr * 1.04, 0.08)
            stall = 0
        else:
            lr *= 0.65
            stall += 1
            if stall > 30 or lr < 1e-8:
                break
    return best, best_c5

def coord_search(h, n, n_iter=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iter):
        corr = np.correlate(best, 1.0 - best, mode='full')
        mt = int(np.argmax(corr))
        s = mt - (n - 1)
        g = _grad_for_shift(best, s, n)
        improved = False
        order = np.argsort(-np.abs(g))
        for j in order[:45]:
            if abs(g[j]) < 1e-8:
                continue
            d = -np.sign(g[j])
            for st in (0.05, 0.025, 0.01, 0.08, 0.015):
                ht = best.copy()
                ht[j] += d * st
                ht = project_h(ht, n)
                cv = compute_c5(ht)
                if cv < best_c5 - 1e-14:
                    best, best_c5 = ht, cv
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_c5

def interval_start(K, n, sd):
    rng = np.random.RandomState(sd)
    w = rng.dirichlet(np.ones(K) * 1.5)
    gp = rng.dirichlet(np.ones(K + 1) * 1.5)
    h = np.zeros(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    pos = 0.0
    for i in range(K):
        pos += gp[i]
        h[(x >= pos) & (x < pos + w[i])] = 1.0
        pos += w[i]
    return project_h(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP T-sweep (28%)
    t1 = t0 + budget_s * 0.28
    for T in (20, 60, 200, 600, 2000, 8000):
        for k in (2, 3, 5, 10, None):
            if time.time() > t1:
                break
            try:
                hn = slsqp_opt(best_h, T, k, n, 600, 1e-15)
                cv = compute_c5(hn)
                if cv < best_c5:
                    best_c5, best_h = cv, hn
            except Exception:
                pass

    # Phase 2: Weighted top-k gradient descent (15%)
    t2 = t0 + budget_s * 0.43
    best_h, best_c5 = grad_descent_topk(best_h, n, 500, 15, 5)
    if time.time() < t2:
        best_h, best_c5 = grad_descent_topk(best_h, n, 400, 40, 3)

    # Phase 3: Coordinate search (12%)
    t3 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_search(best_h, n, 200)

    # Phase 4: Multi-restart with SLSQP + GD (25%)
    t4 = t0 + budget_s * 0.82
    for trial in range(35):
        if time.time() > t4:
            break
        r = np.random.rand()
        if r < 0.4:
            hp = best_h + 0.035 * np.random.randn(n)
        elif r < 0.7:
            hp = interval_start(np.random.randint(3, 9), n, seed + 300 + trial)
        else:
            hp = best_h.copy()
            p = np.random.permutation(n)
            ns = max(10, n // 6)
            hp[:ns] = best_h[p[:ns]]
        hp = project_h(hp, n)
        for T, k in ((300, 3), (1000, 5), (4000, 2)):
            if time.time() > t4:
                break
            try:
                hn = slsqp_opt(hp, T, k, n, 400, 1e-14)
                hn = grad_descent_topk(hn, n, 120, 50, 3)[0]
                cv = compute_c5(hn)
                if cv < best_c5:
                    best_c5, best_h = cv, hn
            except Exception:
                pass

    # Phase 5: Final polish with very high T (17%)
    t5 = t0 + budget_s * 0.985
    for T, k in ((3000, 3), (10000, 2), (30000, 3), (100000, 5)):
        if time.time() > t5:
            break
        try:
            hn = slsqp_opt(best_h, T, k, n, 1200, 1e-16)
            cv = compute_c5(hn)
            if cv < best_c5:
                best_c5, best_h = cv, hn
        except Exception:
            pass

    # Final micro-refinement
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = grad_descent_topk(best_h, n, 200, 80, 3)
        best_h, best_c5 = coord_search(best_h, n, 80)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
