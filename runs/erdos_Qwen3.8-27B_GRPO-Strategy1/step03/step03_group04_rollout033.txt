import numpy as np
import time
import math

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N-L])
            grad[:N-L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N-M]
    return grad, peak

def continuous_optimize(h, N, target, deadline, k, lr0, decay, ms=200):
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1-h, mode='full')))
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    mv = np.zeros(N)  # momentum velocity
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        # Nesterov + Adam
        it += 1
        mv = 0.9 * mv + 0.1 * grad
        m = b1 * m + (1-b1) * mv
        v = b2 * v + (1-b2) * mv * mv
        mh = m / (1-b1**it)
        vh = v / (1-b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % ms == 0:
            lr *= decay
            # Periodic restart
            if it % (ms * 10) == 0:
                m[:] = 0
                v[:] = 0
                mv[:] = 0
    return best_h, best_peak * (2.0/N)

def to_binary(h, N):
    """Convert to binary preserving sum"""
    h_b = np.zeros(N)
    # Large values first
    order = np.argsort(-h)
    h_b[order[:N//2]] = 1
    return h_b

import numpy as np

def compute_c5(h, N):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if float(np.clip(h-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0.0, 1.0)

def grad_topk(h, N, k):
    corr = np.correlate(h, 1.0-h, mode='full')
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N-1
    wk = 1.0/k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk*(1.0-2.0*h)
        elif L > 0:
            grad[L:] += wk*(1.0-h[:N-L])
            grad[:N-L] -= wk*h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += wk*(1.0-h[M:])
                grad[M:] -= wk*h[:N-M]
    return grad, peak

def cont_opt(h, N, target, deadline, k, lr0, decay, ms):
    best_h = h.copy()
    best_pk = float(np.max(np.correlate(h,1-h,mode='full')))
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N); v = np.zeros(N); it = 0
    while time.time() < deadline:
        grad, pk = grad_topk(h, N, k)
        if pk < best_pk:
            best_pk = pk; best_h = h.copy()
        it += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m/(1-b1**it); vh = v/(1-b2**it)
        h = h - lr*mh/(np.sqrt(vh)+eps)
        h = project_sum(h, target)
        if it % ms == 0:
            lr *= decay
    return best_h, best_pk*(2.0/N)

def compute_c5(h, N):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    peak = float(np.max(corr))
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def cont_opt(h, N, target, deadline, k, lr0, decay, ms):

    v = np.zeros(N)

        hN = project_sum(hN, N2 / 2.0)
        d3 = time.time() + (deadline - time.time()) * 0.12
        hN, cN = cont_opt(hN, N2, N2 / 2.0, d3, 30, 0.035, 0.996, 100)
        h_bN = to_binary(hN, N2)
        d4 = time.time() + (deadline - time.time()) * 0.40
        h4, c4 = bin_search_fast(h_bN, N2, d4, rng, n_cand=100)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Final binary polish on best solution
    Nb = len(best_h)
    if time.time() < deadline - 10:
        h_bF = to_binary(best_h, Nb)
        d5 = time.time() + (deadline - time.time()) * 0.55
        h5, c5v = bin_search_fast(h_bF, Nb, d5, np.random.RandomState(seed + 7), n_cand=130)
        if c5v < best_c5:
            best_c5, best_h = c5v, h5

    # Phase 6: Tiny continuous refinement (allow slight non-binary)
    Nb2 = len(best_h)
    if time.time() < deadline - 4:
        hF = best_h.astype(float).copy()
        hF += rng.randn(Nb2) * 0.01
        hF = project_sum(hF, Nb2 / 2.0)
        hF, cF = cont_opt(hF, Nb2, Nb2 / 2.0, deadline - 2, 50, 0.008, 0.999, 30)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
