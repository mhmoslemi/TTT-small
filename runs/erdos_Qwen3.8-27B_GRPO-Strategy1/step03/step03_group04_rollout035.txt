import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_at_lag(h, N, lag_idx):
    """Exact gradient of corr[lag_idx] w.r.t. h."""
    center = N - 1
    L = lag_idx - center  # shift
    grad = np.zeros(N)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        # corr[center+L] = sum_{n=L}^{N-1} h[n] * (1-h[n-L])
        # d/dh[i]: from first factor: (1-h[i-L]) for i in [L, N-1]
        #          from second factor: -h[i+L] for i in [0, N-1-L]
        grad[L:] += (1.0 - h[:N-L])
        grad[:N-L] -= h[L:]
    else:
        M = -L
        if M < N:
            # corr[center-M] = sum_{n=0}^{N-1-M} h[n] * (1-h[n+M])
            # d/dh[i]: from first factor: (1-h[i+M]) for i in [0, N-1-M]
            #          from second factor: -h[i-M] for i in [M, N-1]
            grad[:N-M] += (1.0 - h[M:])
            grad[M:] -= h[:N-M]
    return grad

def grad_topk(h, N, k):
    corr = compute_corr(h)
    top = np.argpartition(corr, -k)[-k:]
    peak = float(np.max(corr))
    if k == 1:
        return grad_at_lag(h, N, int(np.argmax(corr))), peak
    grad = np.zeros(N)
    wk = 1.0 / k
    center = N - 1
    for ai in top:
        L = ai - center
        g = grad_at_lag(h, N, ai)
        grad += wk * g
    return grad, peak

def adam_optimize(h, N, target, deadline, k_init, k_final, lr0, lr_decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    total_iters_est = 5000  # rough estimate
    
    while time.time() < deadline:
        # Adaptive k
        k = max(1, int(k_final + (k_init - k_final) * (1 - min(it/total_iters_est, 1.0))))
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= lr_decay
    return best_h, best_peak * dx

    T = 2.0

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def grad_at_lag(h, N, k):
    """Gradient of corr[k] with respect to h."""
    center = N - 1
    s = k - center
    grad = np.zeros(N)
    if s == 0:
        grad[:] = 1.0 - 2.0*h
    elif s > 0:
        grad[s:] += 1.0 - h[:N-s]
        grad[:N-s] -= h[s:]
    else:
        m = -s
        if m < N:
            grad[:N-m] += 1.0 - h[m:]
            grad[m:] -= h[:N-m]
    return grad

def adam_run(h, N, target, deadline, k_init, k_min, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0/N
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        progress = min(it / 3000.0, 1.0)
        k = max(k_min, int(k_init * (1.0 - progress) + k_min * progress))
        corr = compute_corr(h)
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        if k == 1:
            grad = grad_at_lag(h, N, int(np.argmax(corr)))
        else:
            top = np.argpartition(corr, -k)[-k:]
            grad = np.zeros(N)
            wk = 1.0 / k
            for ai in top:
                grad += wk * grad_at_lag(h, N, int(ai))
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 80 == 0:
            lr *= decay
    return best_h, best_peak * dx

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_at_lag(h, N, k):
    center = N - 1
    s = k - center
    grad = np.zeros(N)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        grad[s:] += 1.0 - h[:N - s]
        grad[:N - s] -= h[s:]
    else:
        m = -s
        if m < N:
            grad[:N - m] += 1.0 - h[m:]
            grad[m:] -= h[:N - m]
    return grad

def adam_run(h, N, target, deadline, k_init, k_min, lr0, decay, b1=0.85, b2=0.999):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    eps = 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        progress = min(it / 4000.0, 1.0)
        k = max(k_min, int(k_init * (1.0 - progress) + k_min * progress))
        corr = compute_corr(h)
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        if k <= 1:
            grad = grad_at_lag(h, N, int(np.argmax(corr)))
        else:
            top = np.argpartition(corr, -k)[-k:]
            grad = np.zeros(N)
            wk = 1.0 / k
            for ai in top:
                grad += wk * grad_at_lag(h, N, int(ai))
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
    return best_h, best_peak * dx

        for i in order:
