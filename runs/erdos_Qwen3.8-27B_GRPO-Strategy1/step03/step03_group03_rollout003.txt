def slsqp_smooth(h, T, maxiter):

    n = len(h)
    h_cur = h.copy()
    y = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = lr_init
    mom = mom_init
    stall = 0
    for it in range(n_iters):
        _, corr = compute_c5(y)
        g = subgrad_active(y, corr, thresh_frac=0.982, top_m=15)
        h_new = project_feasible(y - lr * g, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.015, 0.35)
            mom = min(mom + 0.0003, 0.93)
            stall = 0
        else:
            stall += 1
            lr *= 0.96
            if stall > 400:
                break
        y = h_new + mom * (h_new - h_cur)
        y = project_feasible(y, n)
        h_cur = h_new
    return best_h, best_c5

def binary_local_search(h_bin, max_sweeps=200):
    n = len(h_bin)
    h_cur = h_bin.copy()
    best_c5, _ = compute_c5(h_cur)
    for _ in range(max_sweeps):
        improved = False
        for i in range(n):
            h_trial = h_cur.copy()
            h_trial[i] = 1.0 - h_trial[i]
            c5_try, _ = compute_c5(h_trial)
            if c5_try < best_c5 - 1e-14:
                h_cur = h_trial.copy()
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            break
    return h_cur, best_c5

def slsqp_smooth(h, T, maxiter):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    def eq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                      constraints=[{'type': 'eq', 'fun': eq}],
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_direct(h, maxiter):
    n = len(h)
    _, corr = compute_c5(h)
    t0 = float(np.max(corr)) + 0.02
    def obj(x):
        return x[-1]
    def eq_c(x):
        return x[:-1].sum() - n / 2.0
    def ineq_c(x):
        h_v = x[:-1]
        t_v = x[-1]
        corr_v = np.correlate(h_v, 1.0 - h_v, mode='full')
        return t_v - corr_v
    bnds = [(0.0, 1.0)] * n + [(0.01, 2.0)]
    x0 = np.concatenate([h, [t0]])
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds,
                      constraints=[{'type': 'eq', 'fun': eq_c}, {'type': 'ineq', 'fun': ineq_c}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def make_binary(h):
    n = len(h)
    order = np.argsort(h)
    h_b = np.zeros(n)
    h_b[order[n // 2:]] = 1.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Nesterov subgradient
    if time.time() - t0 < budget_s * 0.15:
        h_o, c5_o = nesterov_subgrad(best_h, n_iters=5000, lr_init=0.06, mom_init=0.80)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Phase 2: Binary projection + greedy local search
    if time.time() - t0 < budget_s * 0.25:
        h_b = make_binary(best_h)
        h_b, c5_b = binary_local_search(h_b, max_sweeps=500)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b.copy()

    # Phase 3: SLSQP from binary (high T)
    if time.time() - t0 < budget_s * 0.35:
        h_b = make_binary(best_h)
        h_o = slsqp_smooth(h_b, 15000, maxiter=800)
        c5_o, _ = compute_c5(h_o)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Phase 4: Direct SLSQP formulation
    if time.time() - t0 < budget_s * 0.45:
        h_o = slsqp_direct(best_h, maxiter=600)
        c5_o, _ = compute_c5(h_o)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Phase 5: Multi-resolution Nesterov
    for n_try in [76, 80, 82, 84, 86]:
        if time.time() - t0 > budget_s * 0.62:
            break
        if n_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        h_o, c5_o = nesterov_subgrad(h_int, n_iters=3000, lr_init=0.05, mom_init=0.80)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o
            best_n = n_try
        h_b2 = make_binary(h_o)
        h_b2, c5_b2 = binary_local_search(h_b2, max_sweeps=200)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2
            best_n = n_try

    # Phase 6: Random perturbations + Nesterov (diversity)
    for trial in range(15):
        if time.time() - t0 > budget_s * 0.80:
            break
        scale = 0.03 + 0.05 * np.random.rand()
        h_p = best_h + np.random.randn(best_n) * scale
        h_p = project_feasible(h_p, best_n)
        h_o, c5_o = nesterov_subgrad(h_p, n_iters=2000, lr_init=0.04, mom_init=0.78)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Phase 7: Alternate binary/continuous
    if time.time() - t0 < budget_s * 0.88:
        for _ in range(3):
            if time.time() - t0 > budget_s * 0.88:
                break
            h_b = make_binary(best_h)
            h_b, c5_b = binary_local_search(h_b, max_sweeps=300)
            h_o = slsqp_smooth(h_b, 20000, maxiter=600)
            c5_o, _ = compute_c5(h_o)
            h_n, c5_n = nesterov_subgrad(h_o, n_iters=2000, lr_init=0.03, mom_init=0.82)
            c5_best_local = min(c5_b, c5_o, c5_n)
            h_pick = h_b if c5_best_local == c5_b else (h_o if c5_best_local == c5_o else h_n)
            if c5_best_local < best_c5:
                best_c5, best_h = c5_best_local, h_pick

    # Phase 8: Final SLSQP polish (very high T)
    if time.time() - t0 < budget_s * 0.93:
        for T in [80000, 300000]:
            if time.time() - t0 > budget_s * 0.96:
                break
            h_o = slsqp_smooth(best_h, T, maxiter=700)
            c5_o, _ = compute_c5(h_o)
            if c5_o < best_c5:
                best_c5, best_h = c5_o, h_o

    # Phase 9: Final Nesterov with small steps
    if time.time() - t0 < budget_s * 0.98:
        h_o, c5_o = nesterov_subgrad(best_h, n_iters=4000, lr_init=0.015, mom_init=0.88)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
