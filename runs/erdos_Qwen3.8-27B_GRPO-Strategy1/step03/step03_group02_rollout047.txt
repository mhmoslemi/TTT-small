    if K >= len(corr):
        top_idx = np.arange(len(corr))
        c_top = corr
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
        c_top = corr[top_idx]
    sc = tau * c_top * dx
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-8:
            continue
        k = top_idx[ix]
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, true_c5

def optimize(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K, iters_target, smooth_all=False):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / max(iters_target, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        if smooth_all:
            grad, c5_val = grad_smooth_max(h, N, tau)
        else:
            grad, c5_val = grad_topk(h, N, tau, K)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        b1, b2, eps = 0.9, 0.999, 1e-8
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c5

def local_search(h, N, target, dx, n_trials, deadline, scale, seed):

If I change h[j] by δ, the change in overlap(k) depends on which lags k involve h[j].

Actually, let me think about a different approach entirely.

**Spectral approach**: The overlap function is related to the Fourier transform. Specifically:
overlap(k) = ∫h(x)(1-h(x+k))dx = 1 - R(k)

where R(k) = ∫h(x)h(x+k)dx is the autocorrelation.

By the Wiener-Khinchin theorem, R(k) = F⁻¹(|F(h)|²)(k).

So we want |F(h)|² to be as "flat" as possible (so that R(k) is peaked at k=0 and decays uniformly).

In the discrete setting, if we take the FFT of h, then:
R(k) ≈ (1/N) * IFFT(|FFT(h)|²)[k]

We want IFFT(|FFT(h)|²) to be as flat as possible away from k=0.

This suggests working in the frequency domain! We can:
1. Compute FFT(h)
2. Modify |FFT(h)|² to be flatter
3. Inverse FFT to get new h
4. Project to [0,1] and sum constraint

This is like a "spectral flattening" approach.

Let me think about this more carefully. In the discrete setting:
- h has N values in [0,1]
- FFT(h) has N frequency components
- The autocorrelation R(k) for k=0,...,N-1 is IFFT(|FFT(h)|²) (up to scaling)

We want to minimize max_k (1 - R(k)) for k ≠ 0, i.e., maximize min_k R(k) for k ≠ 0.

This is a "maximin" problem in the frequency domain.

Let me try a different angle. What if I use **simulated annealing** on a subset of coordinates?

Or better yet, what about **alternating direction method**:
1. Fix all but a few coordinates
2. Optimize those coordinates analytically
3. Cycle through blocks

For a single coordinate h[j], the objective is C₅ = max_k overlap(k). The overlap as a function of h[j] is:
overlap(k) = c_k + a_k * h[j] (for each k that involves h[j])

where c_k and a_k are constants (given other coordinates).

So C₅(h[j]) = max_k (c_k + a_k * h[j]) for k in the set of lags involving j.

This is the max of linear functions, which is convex in h[j]. We can find the optimal h[j] by finding the minimum of this convex piecewise-linear function on [0,1].

The minimum of max of linear functions occurs where two (or more) of the linear functions intersect, or at a boundary.

This is a linear programming problem in 1D! We can solve it by:
1. Computing all (c_k, a_k) pairs
2. Finding the intersection points
3. Evaluating C₅ at each intersection and at 0, 1
4. Picking the best

For N=150, each coordinate is involved in at most 2N-1 lags (probably fewer). So this is manageable.
