```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target_sum):
    h = np.where(h > 1.0, 1.0, np.where(h < 0.0, 0.0, h))
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        tot = s
        if tot < 1e-14:
            return h
        h *= (1.0 + diff / tot)
    return np.where(h > 1.0, 1.0, np.where(h < 0.0, 0.0, h))

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_step(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0, 1)

def coord_descent(h, n, t_limit, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(n):
            # Compute corr with h[i]=0
            h0 = best_h.copy()
            h0[i] = 0.0
            corr0 = np.correlate(h0, 1.0 - h0, mode='full')
            # Linear coefficients
            k_range = np.arange(-(n-1), n)
            j1 = i - k_range
            j2 = i + k_range
            m1 = (j1 >= 0) & (j1 < n)
            m2 = (j2 >= 0) & (j2 < n)
            d = np.zeros(2*n-1)
            d[m1] += (1.0 - best_h[j1[m1]])
            d[m2] += (-best_h[j2[m2]])
            # Override k=0 (handled quadratically)
            d[n-1] = 0.0
            
            def f1d(t):
                v = corr0 + t * d
                v[n-1] = corr0[n-1] + t - t*t
                return np.max(v) * dx
            try:
                res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-9})
            except:
                continue
            best_h[i] = res.x
            # Re-project
            best_h = project(best_h, n/2.0)
            c5 = compute_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5
                improved = True
        if not improved:
            break
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        tot = s
        if tot < 1e-13:
            return h
        h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_step(h, T, n, maxiter=400, ftol=1e-14):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0, 1)

def coord_sweep(h, n, dx):
    """One full sweep of exact coordinate descent. Returns (h, c5, improved)."""
    best_c5 = compute_c5(h)
    improved = False
    for i in range(n):
        h0 = h.copy()
        h0[i] = 0.0
        corr0 = np.correlate(h0, 1.0 - h0, mode='full')
        k_range = np.arange(-(n - 1), n)
        j1 = i - k_range
        j2 = i + k_range
        m1 = (j1 >= 0) & (j1 < n)
        m2 = (j2 >= 0) & (j2 < n)
        d = np.zeros(2 * n - 1)
        d[m1] += (1.0 - h[j1[m1]])
        d[m2] += (-h[j2[m2]])
        d[n - 1] = 0.0
        c0_val = corr0[n - 1]
        def f1d(t):
            v = corr0 + t * d
            v[n - 1] = c0_val + t - t * t
            return np.max(v) * dx
        try:
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
        except Exception:
            continue
        h[i] = res.x
        h = project(h, n / 2.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            improved = True
    return h, best_c5, improved

def coord_descent(h, n, t_limit, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_limit:
            break
        best_h, best_c5, improved = coord_sweep(best_h, n, dx)
        if not improved:
            break
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
```
