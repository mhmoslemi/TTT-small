```python
import time
from scipy.optimize import minimize

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj(h, N, s):
    corr = compute_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(corr - m))) / s) * (2.0 / N)

def smooth_grad(h, N, s):
    corr = compute_corr(h)
    m = np.max(corr)
    exp_c = np.exp(corr - m)
    weights = exp_c / (np.sum(exp_c) * s)
    dx = 2.0 / N

        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N-L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:N-L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:N-M])
            grad[:N-M] -= w * h[M:]
    return grad * dx

def lbfgs_run(h, N, target, deadline, s=50.0):
    """Run L-BFGS-B with smooth max objective."""
    h_copy = h.copy()
    best_h = h_copy.copy()
    best_val = smooth_obj(h_copy, N, s)
    
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_obj(x, N, s)
    
    def grad(x):
        x = np.clip(x, 0, 1)
        return smooth_grad(x, N, s)
    
    # Use L-BFGS-B with bounds
    bounds = [(0, 1)] * N
    result = minimize(obj, h_copy, method='L-BFGS-B', jac=smooth_grad,
                      bounds=bounds, options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-7})
    h_opt = project_sum(result.x, target)
    val = smooth_obj(h_opt, N, s)
    if val < best_val:
        best_h = h_opt
        best_val = val
    return best_h, compute_c5(best_h, N)

def anneal_binary(h, N, deadline, rng, block_len=3):
    """Binary annealing with block swaps."""
    dx = 2.0 / N
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    target_ones = N // 2
    if n_ones != target_ones:
        # Adjust
        if n_ones > target_ones:
            idx = np.where(h_b == 1)[0]
            rng.shuffle(idx)
            h_b[idx[:n_ones - target_ones]] = 0
        else:
            idx = np.where(h_b == 0)[0]
            rng.shuffle(idx)
            h_b[idx[:target_ones - n_ones]] = 1
    
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur_pk = best_pk
    T0, T1 = 5.0, 0.002
    elapsed0 = time.time()
    
    while time.time() < deadline:
        T_frac = (deadline - time.time()) / (deadline - elapsed0 + 1e-8)
        T = T0 * (T1 / T0) ** (1.0 - T_frac)
        
        # Choose move type: single swap or block swap
        if rng.random() < 0.3:
            # Block swap
            bl = rng.randint(1, block_len + 1)
            start1 = rng.randint(0, N - bl)
            start2 = rng.randint(0, N - bl)
            if abs(start1 - start2) < bl:
                continue
            if h_b[start1] == h_b[start1 + bl - 1]:

                break

    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8

    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:

        h = project_sum(h, target)
        if it % 100 == 0:

    return best_h, best_peak * dx
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj(x, N, s, target):
    corr = compute_corr(x)
    m = np.max(corr)
    base = (m + np.log(np.sum(np.exp(corr - m))) / s) * (2.0 / N)
    base += 50.0 * (np.sum(x) - target) ** 2
    return base

def smooth_grad(x, N, s, target):
    corr = compute_corr(x)
    m = np.max(corr)
    exp_c = np.exp(corr - m)
    weights = exp_c / (np.sum(exp_c) * s)
    dx = 2.0 / N
    grad = np.zeros(N)
    center = N - 1
    for j in range(len(corr)):
        w = weights[j]
        if w < 1e-12:
            continue
        L = j - center
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[:N - L] += w * (1.0 - x[L:])
            grad[L:] -= w * x[:N - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - x[:N - M])
            grad[:N - M] -= w * x[M:]
    grad += 100.0 * (np.sum(x) - target) * (1.0 / N) * np.ones(N)
    return grad * dx

def lbfgs_refine(h, N, target, s, maxiter):
    def obj_fn(x):
        return smooth_obj(x, N, s, target)
    def grad_fn(x):
        return smooth_grad(x, N, s, target)
    bounds = [(0.0, 1.0)] * N
    res = minimize(obj_fn, h.copy(), method='L-BFGS-B', jac=grad_fn,
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14,
                                           'gtol': 1e-9, 'maxfun': maxiter * 8})
    return project_sum(res.x, target)

def topk_adam(h, N, target, deadline, k, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(np.max(corr))
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[:N - L] += wk * (1.0 - h[L:])
                grad[L:] -= wk * h[:N - L]
            else:
                M = -L
                if M < N:
                    grad[M:] += wk * (1.0 - h[:N - M])
                    grad[:N - M] -= wk * h[M:]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 80 == 0:
            lr *= decay
    return best_h, best_peak * (2.0 / N)

def binary_anneal(h, N, deadline, rng, max_block=5):
    dx = 2.0 / N
    h_b = (h > 0.5).astype(float).copy()
    n_ones = int(h_b.sum())
    target_ones = N // 2
    if n_ones != target_ones:
        if n_ones > target_ones:
            idx = np.where(h_b == 1)[0]
            rng.shuffle(idx)
            h_b[idx[:n_ones - target_ones]] = 0
        else:
            idx = np.where(h_b == 0)[0]
            rng.shuffle(idx)
            h_b[idx[:target_ones - n_ones]] = 1
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur_pk = best_pk
    t_start = time.time()
    t_total = max(deadline - t_start, 1.0)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = it / max(it + (deadline - time.time()) * 40, 1.0)
        T = 4.0 * 0.003 ** frac
        if rng.random() < 0.25 and N > 10:
            bl = rng.randint(1, max_block + 1)
            s1 = rng.randint(0, N - bl)
            s2 = rng.randint(0, N - bl)
            if abs(s1 - s2) < bl:
                continue
            block1 = h_b[s1:s1 + bl].copy()
            block2 = h_b[s2:s2 + bl].copy()
            h_b[s1:s1 + bl] = block2
            h_b[s2:s2 + bl] = block1
            new_pk = float(np.max(compute_corr(h_b)))
            if new_pk - cur_pk < 0 or rng.random() < np.exp(-(new_pk - cur_pk) / max(T, 1e-8)):
                cur_pk = new_pk
                if new_pk < best_pk:
                    best_pk = new_pk
                    best = h_b.copy()
            else:
                h_b[s1:s1 + bl] = block1
                h_b[s2:s2 + bl] = block2
        else:
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_b[i], h_b[j] = 0.0, 1.0
            new_pk = float(np.max(compute_corr(h_b)))
            if new_pk - cur_pk < 0 or rng.random() < np.exp(-(new_pk - cur_pk) / max(T, 1e-8)):
                cur_pk = new_pk
                if new_pk < best_pk:
                    best_pk = new_pk
                    best = h_b.copy()
            else:
                h_b[i], h_b[j] = 1.0, 0.0
    return best, best_pk * dx

def local_search_binary(h_b, N, deadline, rng, n_rounds=200):
    dx = 2.0 / N
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur = h_b.copy()
    cur_pk = best_pk
    for round_i in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        idx = np.arange(N)
        rng.shuffle(idx)
        for i in idx:
            if time.time() > deadline:
                break
            for j in [(i + 1) % N, (i - 1) % N, (i + 2) % N, rng.randint(N)]:
                if j == i or cur[i] == cur[j]:
                    continue
                cur[i], cur[j] = cur[j], cur[i]
                new_pk = float(np.max(compute_corr(cur)))
                if new_pk < cur_pk:
                    cur_pk = new_pk
                    improved = True
                    if new_pk < best_pk:
                        best_pk = new_pk
                        best = cur.copy()
                else:
                    cur[i], cur[j] = cur[j], cur[i]
        if not improved:
            break
    return best, best_pk * dx
```
