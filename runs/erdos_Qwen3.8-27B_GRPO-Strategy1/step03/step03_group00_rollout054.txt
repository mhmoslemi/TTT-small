    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:

    if diff > 0:

        except Exception:
            pass

    # Phase 3: Block coordinate optimization
    if elapsed() < budget_s * 0.55:
        best_h, best_c5 = block_optimize(best_h, n, block_size=4,
                                         t_end=t0 + budget_s * 0.55, max_rounds=15)

    # Phase 4: Basin hopping with coordinate descent inner
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 80
    for it in range(n_bh):
        if elapsed() > budget_s * 0.78:
            break
        cool = 1.0 - it / float(n_bh)
        step = 0.02 + 0.18 * cool * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = project(h_pert, n)
        h_cd, c5_cd = coord_descent(h_pert, n, time.time() + budget_s / 300.0, max_sweeps=12)
        temp = 0.008 * cool + 0.0002
        if c5_cd < work_c5 or (c5_cd - work_c5 < temp * 3 and
           np.random.rand() < np.exp(-(c5_cd - work_c5) / temp)):
            work_h = h_cd
            work_c5 = c5_cd
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 5: Second CD + block pass
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82, max_sweeps=200)
    if elapsed() < budget_s * 0.88:
        best_h, best_c5 = block_optimize(best_h, n, block_size=6,
                                         t_end=t0 + budget_s * 0.88, max_rounds=10)

    # Phase 6: Multi-resolution exploration
    for n_try in [192, 224, 128]:
        if elapsed() > budget_s * 0.94:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project(h_interp, n_try)
        h_cd, c5_cd = coord_descent(h_interp, n_try, t0 + budget_s * 0.94, max_sweeps=60)
        try:
            hc = slsqp_opt(h_cd, 2000, n_try, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < c5_cd:
                h_cd, c5_cd = hc, c5v
                h_cd, c5_cd = coord_descent(h_cd, n_try, t0 + budget_s * 0.94, max_sweeps=30)
        except Exception:
            pass
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            n = n_try

    # Final polish
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=100)
    try:
        hc = slsqp_opt(best_h, 10000, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    except Exception:
        pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
