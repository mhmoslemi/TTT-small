- sum_i h[i]*(1-h[i+L]) <= t for all L
- 0 <= h[i] <= 1
- sum h[i] = N/2

The constraints sum_i h[i]*(1-h[i+L]) = sum h[i] - sum h[i]*h[i+L] = N/2 - sum h[i]*h[i+L]

So we want to maximize min_L sum h[i]*h[i+L] (with appropriate index ranges).

This is a non-convex problem (quadratic with max). But we can use it for a penalty-based optimization.

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk_soft(h, N, k, temp):
    corr = compute_corr(h)
    top_idx = np.argpartition(corr, -k)[-k:]
    top_vals = corr[top_idx]
    m = float(np.max(top_vals))
    w = np.exp((top_vals - m) / temp)
    ws = float(np.sum(w))
    if ws < 1e-15:
        w = np.zeros(k); w[0] = 1.0
    else:
        w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(top_idx)):
        kidx = int(top_idx[ii])
        L = kidx - center
        wk = w[ii]
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            n = N - L
            grad[:n] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n]
        else:
            M = -L
            if M < N:
                n = N - M
                grad[:n] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:n]
    return grad, float(np.max(corr))

def adam_run(h, N, target, deadline, k, lr0, decay, temp):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m_vec = np.zeros(N)
    v_vec = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk_soft(h, N, k, temp)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m_vec = b1 * m_vec + (1.0 - b1) * grad
        v_vec = b2 * v_vec + (1.0 - b2) * grad * grad
        it += 1
        mh = m_vec / (1.0 - b1 ** it)
        vh = v_vec / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 80 == 0:
            lr *= decay
    return best_h, best_peak * (2.0 / N)

def binarize(h, N):
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    target = N // 2
    if n_ones > target:
        idx = np.where(h_b == 1)[0]
        h_b[idx[target:]] = 0
    elif n_ones < target:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:target - n_ones]] = 1
    return h_b

def local_search(h_b, N, rng, time_limit):
    deadline = time.time() + time_limit
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(best)))
    cur = h_b.copy()
    cur_pk = best_pk
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    passes = 0
    while time.time() < deadline and passes < 40:
        passes += 1
        improved = False
        ones_list = ones.tolist()
        for i in ones_list:
            for dj in range(-10, 11):
                if dj == 0:
                    continue
                j = i + dj
                if 0 <= j < N and cur[j] == 0:
                    cur[i] = 0; cur[j] = 1
                    pk = float(np.max(compute_corr(cur)))
                    if pk < best_pk - 1e-14:
                        best_pk = pk
                        best = cur.copy()
                        improved = True
                    else:
                        cur[j] = 0; cur[i] = 1
        for _ in range(300):
            if time.time() > deadline:
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i == j:
                continue
            cur[i] = 0; cur[j] = 1
            pk = float(np.max(compute_corr(cur)))
            if pk < best_pk - 1e-14:
                best_pk = pk
                best = cur.copy()
                improved = True
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
            else:
                cur[j] = 0; cur[i] = 1
        if not improved:
            break
    return best, best_pk

def anneal_binary(h_b, N, deadline, rng, reheat_cycles=3):
    dx = 2.0 / N
    cur = h_b.copy()
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(cur)))
    cur_pk = best_pk
    total_time = max(deadline - time.time(), 5.0)
    time_per_cycle = total_time / reheat_cycles
    for cycle in range(reheat_cycles):
        cyc_start = time.time()
        cyc_deadline = cyc_start + time_per_cycle
        T0, T1 = 6.0 * (0.7 ** cycle), 0.003
        n_steps = int(cyc_deadline - cyc_start) * 8000
        n_steps = max(n_steps, 2000)
        for it in range(n_steps):
            if time.time() > cyc_deadline:
                break
            T = T0 * (T1 / T0) ** (it / max(n_steps, 1))
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            cur[i], cur[j] = cur[j], cur[i]
            new_pk = float(np.max(compute_corr(cur)))
            delta = new_pk - cur_pk
            if delta < 0 or (T > 0.005 and rng.random() < np.exp(-delta / T)):
                cur_pk = new_pk
                if new_pk < best_pk:
                    best_pk = new_pk
                    best = cur.copy()
            else:
                cur[i], cur[j] = cur[j], cur[i]
    return best, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)

    for N in [380, 384]:
        if time.time() > deadline - 120:
            break
        target = N / 2.0
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)

        # Phase 1: Coarse Adam
        if time.time() < deadline - 200:
            d1 = time.time() + 120
            h1, c1 = adam_run(h_init.copy(), N, target, d1, 30, 0.08, 0.992, 2.0)
            if c1 < best_c5:
                best_c5, best_h = c1, h1.copy()

        # Phase 2: Fine Adam
        if time.time() < deadline - 150:
            d2 = time.time() + 100
            h2, c2 = adam_run(h1.copy(), N, target, d2, 8, 0.03, 0.995, 1.0)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

        # Phase 3: Binarize + Local Search
        h_bin = binarize(h2, N)
        if time.time() < deadline - 100:
            h_ls, pk_ls = local_search(h_bin.copy(), N, rng, 80)
            c_ls = pk_ls * (2.0 / N)
            if c_ls < best_c5:
                best_c5, best_h = c_ls, h_ls

        # Phase 4: Annealing
        if time.time() < deadline - 60:
            d4 = time.time() + 120
            h_an, c_an = anneal_binary(h_ls, N, d4, rng, 3)
            if c_an < best_c5:
                best_c5, best_h = c_an, h_an.copy()

        # Phase 5: Perturb + refine
        if time.time() < deadline - 80:
            h_p = h_ls.copy()
            n_flip = max(5, N // 40)
            flip_idx = rng.choice(N, size=n_flip * 2, replace=False)
            for i in flip_idx:
                h_p[i] = 1.0 - h_p[i]
            h_p = project_sum(h_p, target)
            d5 = time.time() + 60
            h5, c5 = adam_run(h_p, N, target, d5, 5, 0.015, 0.996, 0.5)
            if c5 < best_c5:
                best_c5, best_h = c5, h5.copy()
            h5b = binarize(h5, N)
            h5s, pk5s = local_search(h5b, N, rng, 30)
            c5s = pk5s * (2.0 / N)
            if c5s < best_c5:
                best_c5, best_h = c5s, h5s

    # Final polish: local search on best solution
    Nb = len(best_h)
    if time.time() < deadline - 10:
        h_final = binarize(best_h, Nb)
        h_fin, pk_fin = local_search(h_final, Nb, rng, 30)
        c_fin = pk_fin * (2.0 / Nb)
        if c_fin < best_c5:
            best_c5, best_h = c_fin, h_fin

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
