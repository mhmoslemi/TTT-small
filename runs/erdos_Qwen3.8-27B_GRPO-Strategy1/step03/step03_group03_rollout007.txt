def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:

            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:

            h += room * (diff / total)

    return h

def slsqp_lp(h, p, maxiter=500, ftol=1e-14):

    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return lp_norm(x, p)
    def obj_grad(x):
        return lp_grad(x, p)
    def con_eq(x):
        return x.sum() - n / 2.0
    def con_eq_grad(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_grad}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def lp_value(h, p):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cp = np.maximum(corr, 1e-15)
    return float((np.sum(cp**p))**(1.0/p) * dx)

def lp_gradient(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cp = np.maximum(corr, 1e-15)
    pw = cp / cp**p  # = cp^(1-p) proportional to gradient weights
    # Normalize: grad = (sum cp^p)^(1/p - 1) * sum p*cp^(p-1)*dcord/dh * dx
    # Simplified: w[k] = cp[k]^p / sum(cp^p) are the softmax-like weights
    sp = np.sum(cp**p)
    if sp < 1e-30:
        return np.zeros(n)
    w = cp**p / sp  # normalized weights summing to 1
    # grad[i] = dx * norm_p^(1-p) * sum_k cp[k]^(p-1) * dcorr[k]/dh[i]
    # = dx * (sum cp^p)^(1/p - 1) * sum_k p*cp[k]^(p-1)*dcord/dh[i]
    # Let's use: grad[i] = dx * (1/norm_p) * sum_k w[k]*p*cp[k]*dcord/dh[i]
    # where norm_p = sp^(1/p). Actually let me just use the subgradient of the max
    # For large p, this approximates the gradient of the max.
    # Simplification: use w[k] as importance weights
    wc = cp**p / sp  # sums to 1, concentrates on max for large p
    grad = np.zeros(n)
    # w_pos[k] for lag k>=0, index n-1+k
    w_pos = wc[n-1:]  # lags 0,1,...,n-1
    w_neg = wc[:n-1][::-1]  # lags -1,-2,...,-(n-1)
    one_minus_h = 1.0 - h
    for i in range(n):
        g = 0.0
        # Positive lags k=0..n-1, index n-1+k
        for k in range(n):
            wv = w_pos[k]
            if k == 0:
                g += wv * (one_minus_h[i] - h[i])  # (1-h[i])*1 + (-h[i])*1...
                # Actually at k=0: dcorr/dh[i] = (1-h[i]) - h[i]
            else:
                if k <= i:
                    g += wv * one_minus_h[i-k]
                if i + k < n:
                    g += wv * (-h[i+k])
            # Negative lag -k (k>=1), index n-1-k
            if k >= 1:
                wn = w_neg[k-1]
                if i + k < n:
                    g += wn * one_minus_h[i+k]
                if i >= k:
                    g += wn * (-h[i-k])
        grad[i] = g
    return grad * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += room * (diff / s)
        else:
            room = h.copy()
            s = room.sum()
            if s < 1e-14:
                break
            h += room * (diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_lp(h, p, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x): return lp_value(x, p)
    def obj_g(x): return lp_gradient(x, p)
    def c_eq(x): return x.sum() - n / 2.0
    def c_eq_g(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_eq_g}]
    try:
        res = minimize(obj, h, jac=obj_g, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_momentum(h, n_iters=2000, p0=50, p1=300):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    v = np.zeros(n)
    momentum = 0.75
    lr = 0.02
    stall = 0
    for it in range(n_iters):
        p_eff = int(p0 + (p1 - p0) * it / max(n_iters, 1))
        g = lp_gradient(h_cur, min(p_eff, 600))
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        v = momentum * v - lr * g / gn
        # Backtracking
        lr_eff = lr
        for _ in range(30):
            h_new = project_feasible(h_cur + v, n)
            c5_n, _ = compute_c5(h_new)
            if c5_n < best_c5 + 1e-9:
                break
            lr_eff *= 0.5
            v *= 0.5
        else:
            h_new = h_cur
        c5_cur, _ = compute_c5(h_cur)
        if c5_n < best_c5 - 1e-12:
            best_c5 = c5_n
            best_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.05, 0.3)
            stall = 0
        else:
            stall += 1
            lr *= 0.92
            momentum = max(0.5, momentum * 0.995)
            if stall > 400:
                break
    return best_h, best_c5

def coord_descent(h, n_passes=6, n_steps=7):
    n = len(h)
    h_cur = h.copy().astype(float)
    deltas = np.array([0.02, -0.02, 0.01, -0.01, 0.04, -0.04, 0.005, -0.005])
    for path in range(n_passes):
        for i in range(n):
            cur_c5 = compute_c5(h_cur)[0]
            for d in deltas:
                h_try = h_cur.copy()
                hi = h_cur[i] + d
                if hi < 0 or hi > 1:
                    continue
                h_try[i] = hi
                diff = h_try.sum() - n / 2.0
                if abs(diff) > 1e-6:
                    if diff > 0:
                        cands = np.where(h_try > 0.02)[0]
                    else:
                        cands = np.where(h_try < 0.98)[0]
                    cands = cands[cands != i]
                    if len(cands) == 0:
                        continue
                    j = cands[np.argmin(np.abs(h_try[cands] - 0.5))]
                    h_try[j] = np.clip(h_try[j] - diff, 0, 1)
                c5_t = compute_c5(h_try)[0]
                if c5_t < cur_c5:
                    h_cur = h_try
                    cur_c5 = c5_t
        h_cur = project_feasible(h_cur, n)
    return h_cur
