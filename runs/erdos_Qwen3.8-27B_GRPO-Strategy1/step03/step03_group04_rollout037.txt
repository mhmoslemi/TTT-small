    max_c = float(np.max(corr))
    exp_c = np.exp(beta * (corr - max_c))
    w_sum = float(exp_c.sum())
    w = exp_c / w_sum
    
    grad = np.zeros(N)
    center = N - 1

    active_idx = np.where(w > 1e-8)[0]

            M = -s
            grad[:N-M] += wi * (1.0 - h[M:])

import numpy as np
import time

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_gradient(h, N, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    exp_c = np.exp(beta * (corr - max_c))
    w_sum = float(exp_c.sum())
    if w_sum < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = exp_c / w_sum
    grad = np.zeros(N)
    center = N - 1
    idx = np.where(w > 1e-9)[0]
    for ki in idx:
        wi = float(w[ki])
        s = int(ki) - center
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:N - s])
            grad[:N - s] -= wi * h[s:]
        else:
            M = -s
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, max_c

def adam_lse(h, N, target, deadline, beta0, beta1, lr0, halve_every):
    best_h = h.copy()
    best_mc = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    lr = lr0
    while time.time() < deadline:
        t += 1
        frac = min(1.0, t / 2000.0)
        beta = beta0 + (beta1 - beta0) * frac
        grad, mc = lse_gradient(h, N, beta)
        if mc < best_mc:
            best_mc = mc
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        t1 = 1.0 - b1 ** t
        t2 = 1.0 - b2 ** t
        h = h - lr * (m / t1) / (np.sqrt(v / t2) + eps)
        h = project_sum(h, target)
        if t % halve_every == 0:
            lr *= 0.75
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    if c < best_mc:
        best_mc = c
        best_h = h.copy()
    return best_h, best_mc

def quick_improve(h, N, target, time_budget, rng):
    best_h = h.copy()
    best_mc = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    end_t = time.time() + time_budget
    while time.time() < end_t:
        noise = rng.randn(N) * 0.015
        h_t = h + noise
        h_t = project_sum(h_t, target)
        sub_deadline = time.time() + min(15.0, (end_t - time.time()) * 0.4)
        h_t, mc = adam_lse(h_t, N, target, sub_deadline, 50 + rng.randint(50), 300, 0.02, 300)
        if mc < best_mc:
            best_mc = mc
            best_h = h_t.copy()
            h = best_h.copy()
    return best_h, best_mc

    return best_h, final_c5, N0
