import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T, topk):
    corr = get_corr(h)
    n = len(h)
    dx = 2.0 / n
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        corr = corr[idx]
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(np.clip(T * (corr - m), -30, 30)))) / T
    return val * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def sum_constraint_fn(x):
    return x.sum() - len(x) / 2.0

def slsqp_opt(h0, T, n, topk, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_constraint_fn}]
    bnds = [(0.0, 1.0)] * n
    if topk is not None:
        res = minimize(smooth_c5, h0, args=(T, topk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    else:
        res = minimize(smooth_c5, h0, args=(T, None),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

        center = n - 1

        any_improved = False

            shift = si - center
            if shift == 0:
                continue
            if shift > 0:
                idx_a = np.arange(0, n - shift)
                idx_b = idx_a + shift
            else:
                idx_a = np.arange(0, n + shift)
                idx_b = idx_a - shift
            weights = best[idx_a] * (1.0 - best[idx_b])
            nw = len(weights)
            if nw == 0:

            n_try = min(8, nw)
