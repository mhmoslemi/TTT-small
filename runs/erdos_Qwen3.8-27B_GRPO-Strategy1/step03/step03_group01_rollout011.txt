import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_topk(h, N, topK):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr)) * dx
    center = N - 1
    nc = len(corr)
    idx_top = np.argsort(corr)[-topK:]
    w = 1.0 / float(topK)
    grad = np.zeros(N)
    for ai in idx_top:
        L = ai - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
        else:
            M = -L
            grad[:N - M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:N - M]
    return grad, c5

def grad_softmax(h, N, tau, topK):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr)) * dx
    center = N - 1
    nc = len(corr)
    sc = tau * corr
    sc -= sc.max()
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    k = max(1, min(topK, nc))
    idx_top = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 8.0) / 25.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    grad = np.zeros(N)
    thresh = 0.002
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def opt_phase1(h, N, tgt, deadline, seed):

    np.random.seed(seed)
    best_h = h.copy()

    it = 0
    stall = 0
    while time.time() < deadline:

def coord_polish(h, N, tgt, deadline, seed):
    """Coordinate descent with multiple step sizes and binarization."""
    np.random.seed(seed)
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        order = np.random.permutation(N)
        for i in order:
            if time.time() > deadline:
                break
            orig = best_h[i]
            cands = [0.0, 1.0]
            for d in (0.03, -0.03, 0.06, -0.06, 0.12, -0.12, 0.01, -0.01):
                c = orig + d
                if 0.0 <= c <= 1.0:
                    cands.append(c)
            for c in cands:
                if abs(c - orig) < 1e-9:
                    continue
                trial = best_h.copy()
                trial[i] = c
                trial = project_sum(trial, tgt)
                c5 = compute_c5(trial, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = trial.copy()
                    improved = True
                    break
    return best_h, best_c5

    x = np.linspace(0.0, 2.0, N, endpoint=False)

    blend = project_sum(blend, tgt)
    h_opt, c5 = opt_phase2(blend, N, tgt, deadline, seed + 1)
