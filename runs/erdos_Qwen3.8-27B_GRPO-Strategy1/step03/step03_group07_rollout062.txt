    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    Z = np.sum(exps)
    val = m + np.log(Z) / T
    w = exps / Z
    a = 1.0 - 2.0 * h
    corr_g = np.correlate(a, w, 'full')
    grad = corr_g[n-1 : 2*n-1] * dx
    return val, grad

    h_new = make_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    Z = np.sum(exps)
    val = m + np.log(Z) / T
    w = exps / Z
    a = 1.0 - 2.0 * h
    grad = np.empty(n)
    for j in range(n):
        grad[j] = dx * np.dot(a, w[j:j+n])
    return val, grad

def proj_grad(h, T, lr, n):
    _, grad = smooth_obj_and_grad(h, T)
    return make_feasible(h - lr * grad, n)

def local_optimize(h0, n, T, lr_init, max_iter, t_end):
    h_cur = h0.copy()
    lr = lr_init
    best_val = compute_c5(h_cur)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        h_new = proj_grad(h_cur, T, lr, n)
        c5v = compute_c5(h_new)
        if c5v < best_val:
            best_val = c5v
            h_cur = h_new
            lr = min(lr * 1.01, lr_init * 1.5)
        else:
            lr *= 0.85
            if lr < 1e-6:
                break
    return h_cur, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        n_init = 200
        best_h = np.zeros(n_init)
        best_h[::2] = 1.0
    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient descent (30% budget)
    t_lim1 = t0 + budget_s * 0.30
    for T in [200, 800, 3000, 10000]:
        if time.time() > t_lim1:
            break
        h_cur, c5v = local_optimize(best_h, n, T, 0.03, 3000, t_lim1)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_cur

    # Phase 2: Basin hopping with diverse perturbations (30% budget)
    t_lim2 = t0 + budget_s * 0.60
    bh_h = best_h.copy()
    bh_c5 = best_c5
    max_bh = int((t_lim2 - time.time()) / 1.2)
    max_bh = max(max_bh, 100)
    for it in range(max_bh):
        if time.time() > t_lim2:
            break
        ptype = it % 5

            h_pert = bh_h + 0.06 * (1 + 0.5 * np.random.rand()) * np.random.randn(n)
        elif ptype == 1:
            h_pert = bh_h.copy()
            nswap = max(1, n // 25)
            for _ in range(nswap):
                i, j = np.random.randint(0, n, 2)
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
        elif ptype == 2:
            h_pert = bh_h.copy()
            blk = max(3, n // 15)
            s1 = np.random.randint(0, n - blk)
            s2 = np.random.randint(0, n - blk)
            tmp = h_pert[s1:s1+blk].copy()
            h_pert[s1:s1+blk] = h_pert[s2:s2+blk]
            h_pert[s2:s2+blk] = tmp
        elif ptype == 3:
            h_pert = bh_h.copy()
            nk = max(2, n // 12)
            idx = np.random.choice(n, nk, replace=False)
            h_pert[idx] = np.clip(h_pert[idx] + 0.3 * np.random.randn(nk), 0, 1)
        else:

            h_pert = make_feasible(h_pert, n)

    zeros = np.where(h_bin <= 0.001)[0]
    no_imp = 0

    while no_imp < max_no and time.time() < t_lim3:

        zi = zeros[np.random.randint(len(zeros))]
        if oi == zi:
            no_imp += 1
            continue
        h_test = h_bin.copy()
        h_test[oi] = 0.0
        h_test[zi] = 1.0
        c5_test = compute_c5(h_test)
        if c5_test < c5_bin - 1e-14:
            c5_bin = c5_test
            h_bin = h_test
            no_imp = 0
            ones = np.where(h_bin >= 0.999)[0]
            zeros = np.where(h_bin <= 0.001)[0]
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                # Small random kick to escape
                h_bin = h_bin + 0.01 * np.random.randn(n)
                h_bin = make_feasible(h_bin, n)
                h_bin[(h_bin > 0.4) & (h_bin < 0.6)] = 0.5
                # Re-threshold
                h_bin = (h_bin > 0.5).astype(float)
                h_bin = make_feasible(h_bin, n)
                c5_bin = compute_c5(h_bin)
                ones = np.where(h_bin >= 0.999)[0]
                zeros = np.where(h_bin <= 0.001)[0]

    # Phase 4: Multi-granularity exploration (10% budget)
    t_lim4 = t0 + budget_s * 0.95
    for n_try in [140, 250, 300, 160]:
        if time.time() > t_lim4:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for T in [500, 3000]:
            if time.time() > t_lim4:
                break
            h_cur, c5v = local_optimize(h_interp, n_try, T, 0.02, 800, t_lim4)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_cur
                n = n_try

    # Phase 5: Final high-T polish (5% budget)
    t_lim5 = t0 + budget_s * 0.995
    n = len(best_h)
    for T in [20000, 100000]:
        if time.time() > t_lim5:
            break
        h_new, c5v = local_optimize(best_h, n, T, 0.005, 500, t_lim5)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
