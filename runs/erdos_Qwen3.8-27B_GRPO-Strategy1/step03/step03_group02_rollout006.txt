                if scale < 0.001:

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    sizes = [150, 200, 180, 240]

    for si, N in enumerate(sizes):
        if time.time() > deadline - 40:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)

        remain = deadline - time.time()
        if remain < 50:
            break

        # Allocate more time to first size
        time_frac = 0.55 if si == 0 else (0.25 if si == 1 else 0.15)
        phase_time = remain * time_frac

        # Phase 1: Softmax gradient, broad K
        p1_dead = time.time() + phase_time * 0.40
        h1, c1 = opt_adam(h_init.copy(), N, target, dx, p1_dead,
                          'softmax', tau0:=80.0, tau1:=400.0,
                          lr0:=0.04, lr1:=0.004,
                          K0:=60, K1:=20, t0v:=0.05, t1v:=0.02)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

        # Phase 2: Direct subgradient, focused
        if time.time() < deadline - 20:
            p2_dead = time.time() + phase_time * 0.35
            h2, c2 = opt_adam(h1.copy(), N, target, dx, p2_dead,
                              'direct', tau0:=1.0, tau1:=1.0,
                              lr0:=0.015, lr1:=0.001,
                              K0:=20, K1:=4, t0v:=0.02, t1v:=0.003)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

        # Phase 3: Ultra-fine direct subgradient
        if time.time() < deadline - 10:
            p3_dead = time.time() + phase_time * 0.15
            h3, c3 = opt_adam(h2.copy(), N, target, dx, p3_dead,
                              'direct', tau0:=1.0, tau1:=1.0,
                              lr0:=0.003, lr1:=0.0001,
                              K0:=5, K1:=2, t0v:=0.003, t1v:=0.0005)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()

        # Phase 4: Local search
        if time.time() < deadline - 5:
            ls_dead = min(deadline - 2, time.time() + phase_time * 0.10)
            h4, c4 = local_search(h3.copy(), N, target, dx, ls_dead, rng)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()

    # Final polish on best solution
    Nb = len(best_h)
    if time.time() < deadline - 3:
        dx_b = 2.0 / Nb
        hF, cF = opt_adam(best_h.copy(), Nb, Nb // 2, dx_b, deadline - 1,
                          'direct', 1.0, 1.0, 0.001, 0.00002,
                          3, 2, 0.002, 0.0003)
        if cF < best_c5:
            best_c5 = cF
            best_h = hF

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
