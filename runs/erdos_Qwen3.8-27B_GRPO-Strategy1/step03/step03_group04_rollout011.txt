import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    h_b = (h > 0.5).astype(float)
    half = N // 2
    n_ones = int(h_b.sum())
    if n_ones > half:
        idx = np.where(h_b == 1)[0]
        h_b[idx[half:]] = 0
    elif n_ones < half:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:half - n_ones]] = 1
    return h_b

def binary_anneal(h, N, deadline, rng, T0=5.0, T1=0.001, batch=40):

    h_b = make_binary(h, N)
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))

    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]

            i = ones[rng.randint(len(ones))]

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    if k >= len(corr):
        k = len(corr)
    top = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_run(h, N, target, deadline, lr0, decay, k):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= decay
    return best_h, best_peak * (2.0 / N)
