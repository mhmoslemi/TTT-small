import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def sum_con(h):
    return h.sum() - len(h) / 2.0

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-12:
            continue
        k = j - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += wj * (1.0 - h[lo1-k:hi1-k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wj * h[lo2+k:hi2+k]
    grad *= dx
    return grad

def proj_grad_descent(h, T, n_steps, lr_init, t_end):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    for step in range(n_steps):
        if time.time() > t_end:
            break
        g = grad_smooth(h, T)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        alpha = lr
        for _ in range(12):
            h_new = make_feasible(h - alpha * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                improved = True
                break
            alpha *= 0.4
        if not improved:
            lr *= 0.7
            stall += 1
            if stall > 30:
                break
        else:
            stall = 0
    return h, best_c5

def slsqp_opt(h0, T, n, maxiter=600, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': sum_con}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

def targeted_perturb(h, rng, strength):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-3:]
    contrib = np.zeros(n)
    for j in top_k:
        k = j - (n - 1)
        for i in range(n):
            if 0 <= i + k < n:
                contrib[i] += h[i] * (1.0 - h[i + k])
    n_adj = max(2, n // 12)
    idx = np.argsort(contrib)[-n_adj:]
    h_p = h.copy()
    for i in idx:
        if h[i] > 0.5:
            h_p[i] = max(0.0, h[i] - strength * (1.0 + rng.randn() * 0.3))
        else:
            h_p[i] = min(1.0, h[i] + strength * (1.0 + rng.randn() * 0.3))
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float).copy()
    else:
        n = 200
        best_h = np.zeros(n)
        best_h[0::2] = 1.0
        best_h = make_feasible(best_h, n)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T (15% budget)
    t1 = t0 + budget_s * 0.15
    for T in [80, 300, 1500, 8000]:
        if time.time() > t1:
            break
        try:
            h_new = slsqp_opt(best_h.copy(), T, len(best_h), 600, 1e-15)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Projected gradient descent (20% budget)
    t2 = t0 + budget_s * 0.35
    for T in [300, 1500, 8000, 30000]:
        if time.time() > t2:
            break
        h_new, c5_new = proj_grad_descent(best_h, T, 4000, 0.06, t2)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: Basin hopping with diverse strategies (40% budget)
    t3 = t0 + budget_s * 0.75
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    temp = 0.004
    it = 0
    while time.time() < t3:
        it += 1
        ptype = it % 5
        step_sz = 0.06 + 0.05 * rng.rand()
        if ptype == 0:
            h_pert = bh_h + step_sz * rng.randn(n_bh)
        elif ptype == 1:
            h_pert = bh_h.copy()
            blen = max(3, n_bh // 12)
            s1 = rng.randint(0, n_bh - blen)
            s2 = rng.randint(0, n_bh - blen)
            tmp = h_pert[s1:s1+blen].copy()
            h_pert[s1:s1+blen] = h_pert[s2:s2+blen]
            h_pert[s2:s2+blen] = tmp
        elif ptype == 2:
            h_pert = bh_h.copy()
            nflip = max(2, n_bh // 8)
            idx = rng.choice(n_bh, nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif ptype == 3:
            h_pert = targeted_perturb(bh_h, rng, 0.15 + 0.1 * rng.rand())
        else:
            h_pert = bh_h + step_sz * 1.8 * rng.randn(n_bh)

        h_pert = make_feasible(h_pert, n_bh)
        T_use = 400 + 400 * min(it // 15, 12)
        try:
            h_loc, c5_loc = proj_grad_descent(h_pert, T_use, 150, 0.04, t3)
        except Exception:
            h_loc = h_pert
            c5_loc = compute_c5(h_pert)

        if c5_loc < bh_c5 or rng.rand() < np.exp(-(c5_loc - bh_c5) / temp):
            bh_c5 = c5_loc
            bh_h = h_loc
            if c5_loc < best_c5:
                best_c5 = c5_loc
                best_h = h_loc
        temp *= 0.9995
        if temp < 0.0001:
            temp = 0.0001

    # Phase 4: Multi-resolution exploration (12% budget)
    t4 = t0 + budget_s * 0.87
    for n_try in [100, 130, 160, 240]:
        if time.time() > t4:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for T in [200, 1000]:
            if time.time() > t4:
                break
            try:
                h_new = slsqp_opt(h_interp.copy(), T, n_try, 400, 1e-14)
                h_new2, c5_new2 = proj_grad_descent(h_new, 5000, 1500, 0.04, t4)
                c5v = compute_c5(h_new2)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new2
            except Exception:
                pass

    # Phase 5: Final high-T polish (8% budget)
    t5 = t0 + budget_s * 0.995
    if time.time() < t5:
        for T in [20000, 80000, 300000]:
            if time.time() > t5:
                break
            try:
                h_new = slsqp_opt(best_h.copy(), T, len(best_h), 1500, 1e-16)
                h_new2, c5_new2 = proj_grad_descent(h_new, 200000, 3000, 0.01, t5)
                c5v = compute_c5(h_new2)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new2
            except Exception:
                pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
