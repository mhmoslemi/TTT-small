    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5(h, N, tau, topK, alpha_mix):
    """Compute gradient of smooth C5."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    center = N - 1
    nc = len(corr)
    sc = tau * corr * dx
    sc -= sc.max()
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    k = max(1, min(topK, nc))
    idx_top = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    w = (1.0 - alpha_mix) * w_sm + alpha_mix * w_top
    grad = np.zeros(N)
    thresh = max(1e-5, 1.0 / (nc * 5))
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, float(np.max(corr)) * dx

def fit_fourier(h, K, N, x):
    A = np.zeros((N, 2*K))
    for m in range(1, K+1):
        A[:, 2*(m-1)] = np.cos(m * np.pi * x)
        A[:, 2*(m-1)+1] = np.sin(m * np.pi * x)
    b = h - 0.5
    coeffs, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
    return coeffs

def fourier_obj(coeffs, K, N, x, tau):
    h = np.full(N, 0.5)
    for m in range(1, K+1):
        h += coeffs[2*(m-1)] * np.cos(m * np.pi * x)
        h += coeffs[2*(m-1)+1] * np.sin(m * np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, N/2.0)
    dx = 2.0/N
    corr = np.correlate(h, 1.0-h, mode='full')
    sc = tau * corr * dx
    sc -= sc.max()
    return np.log(np.sum(np.exp(sc))) / tau

def fourier_opt(h_init, N, deadline, K, tau):
    x = np.linspace(0.0, 2.0, N, endpoint=False)
    coeffs0 = fit_fourier(h_init, K, N, x)
    
    def obj(c):
        return fourier_obj(c, K, N, x, tau)
    
    result = minimize(obj, coeffs0, method='L-BFGS-B',
                      options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10,
                               'maxls': 50, 'fail_on_maxIteration': False})
    h = np.full(N, 0.5)
    for m in range(1, K+1):
        h += result.x[2*(m-1)] * np.cos(m * np.pi * x)
        h += result.x[2*(m-1)+1] * np.sin(m * np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, N/2.0)
    return h, compute_c5(h, 2.0/N)

def momentum_opt(h, N, target, deadline, tau_s, tau_e, lr0, decay, period, mom, tk_s, tk_e, alpha):
    best_h = h.copy()
    dx = 2.0/N
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr = lr0
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 3000.0, 1.0)
        tau = tau_s + (tau_e - tau_s) * frac
        topK = int(tk_s + (tk_e - tk_s) * frac)
        a = alpha
        grad, c5_val = grad_c5(h, N, tau, topK, a)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        v = mom * v + grad
        h = h - lr * v
        h = project_sum(h, target)
        if it % period == 0:
            lr *= decay
            if lr < 1e-12:
                break
    return best_h, best_c5

def polish(h, N, target, deadline):
    dx = 2.0/N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = h[i]
            for trial in (0.0, 1.0, max(0.0, orig-0.01), min(1.0, orig+0.01)):

                if c5 < best_c5 - 1e-11:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break
            h = best_h.copy()
