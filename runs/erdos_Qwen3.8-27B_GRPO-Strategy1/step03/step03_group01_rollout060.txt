    h = best_h.copy()
    h, c = gradient_phase(h, N, target, t_p3,
                          40.0, 250.0, 10, 3, 0.006, 0.97, 80, 0.89, 0.02, 0.06)
    if c < best_c5:
        best_c5, best_h = c, h.copy()

    # Phase 4: Binary threshold + local search
    h_bin = (best_h > 0.5).astype(float)
    h_bin = project_sum(h_bin, target)
    c5_b = compute_c5(h_bin, dx)
    if c5_b < best_c5:
        best_c5, best_h = c5_b, h_bin.copy()
    best_h, best_c5 = binary_refine(best_h, N, target, t_p4)

    # Phase 5: Ultra-fine gradient on best
    h = best_h.copy()
    h, c = gradient_phase(h, N, target, deadline,
                          150.0, 500.0, 5, 2, 0.003, 0.98, 70, 0.91, 0.03, 0.05)
    if c < best_c5:
        best_c5, best_h = c, h

    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, N
