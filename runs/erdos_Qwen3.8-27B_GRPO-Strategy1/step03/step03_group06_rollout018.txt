def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T, k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k is not None and k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        corr = corr[idx]
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13: return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-13: return h
        h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h0, T, k, maxiter, ftol=1e-14):
    n = len(h0)
    def obj(x): return smooth_c5(x, T, k)
    def con_fun(x): return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

    idx = np.argpartition(corr, -K_active)[-K_active:]

            s = -shift

            grad[:ns] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:ns]
    grad -= grad.mean()
    h_new = h - step_size * grad
    return project_feasible(h_new, n)
