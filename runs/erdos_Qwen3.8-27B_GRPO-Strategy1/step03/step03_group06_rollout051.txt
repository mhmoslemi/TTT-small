from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def grad_c5_shifted(h, delta):
    n = len(h)
    g = np.zeros(n)
    if delta == 0:
        g[:] = 1.0 - 2.0 * h
    elif delta > 0:
        g[:n-delta] += (1.0 - h[delta:])
        g[delta:] -= h[:n-delta]
    else:
        d = -delta
        g[d:] += (1.0 - h[:n-d])
        g[:n-d] -= h[d:]
    return g

def grad_topk_c5(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top_idx:
        delta = idx - (n - 1)
        g += grad_c5_shifted(h, delta)
    return g * dx / K

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def proj_grad_desc(h, K_schedule, step_init, n_max_iter, tol):
    n = len(h)
    h = h.copy()
    c5, _ = compute_c5(h)
    step = step_init
    for it in range(n_max_iter):
        g = grad_topk_c5(h, K_schedule)
        gn = np.linalg.norm(g)
        if gn < tol:
            break
        # Backtracking line search
        alpha = step
        for bt in range(30):
            h_new = project_feasible(h - alpha * g, n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < c5 - 1e-15:
                h = h_new
                c5 = c5_new
                break
            alpha *= 0.5

            h = project_feasible(h - alpha * g, n)
            c5, _ = compute_c5(h)
        step = alpha * 1.5 if alpha > step * 0.5 else max(alpha, step * 0.8)
        step = min(step, 0.1)
    return h, c5

def slsqp_polish(h, n, T, maxiter=1000):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        T_ = T
        m = np.max(corr)
        exps = np.exp(T_ * (corr - m))
        weights = exps / exps.sum()
        g = np.zeros(n)
        for idx, w in enumerate(weights):
            if w < 1e-15:
                continue
            delta = idx - (n - 1)
            g += w * grad_c5_shifted(x, delta)
        return g * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def grad_c5_shifted(h, delta):
    n = len(h)
    g = np.zeros(n)
    if delta == 0:
        g[:] = 1.0 - 2.0 * h
    elif delta > 0:
        g[:n - delta] += 1.0 - h[delta:]
        g[delta:] -= h[:n - delta]
    else:
        d = -delta
        g[d:] += 1.0 - h[:n - d]
        g[:n - d] -= h[d:]
    return g

def grad_topk_c5(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top_idx:
        g += grad_c5_shifted(h, idx - (n - 1))
    return g * dx / K

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h += room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        h += h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def proj_grad_desc(h, K, step_init, n_iter, tol, t_end):
    n = len(h)
    h = h.copy()
    c5, _ = compute_c5(h)
    step = step_init
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = grad_topk_c5(h, K)
        gn = np.linalg.norm(g)
        if gn < tol:
            break
        alpha = step
        for _ in range(25):
            h_new = project_feasible(h - alpha * g, n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h, c5 = h_new, c5_new
                break
            alpha *= 0.5
        if alpha < 1e-12:
            break
        step = min(max(alpha * 1.3, 1e-6), 0.05)
    return h, c5

def _slsqp_obj(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def _slsqp_jac(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    g = np.zeros(n)
    for i in range(len(corr)):
        if w[i] < 1e-10:
            continue
        g += w[i] * grad_c5_shifted(x, i - (n - 1))
    return g * (2.0 / n)

def _cons_fun(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(len(x))

def slsqp_polish(h, T, maxiter=800):
    n = len(h)
    cons = [{'type': 'eq', 'fun': lambda x: _cons_fun(x, n),
             'jac': lambda x: _cons_jac(x, n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, h, jac=lambda x: _slsqp_jac(x, n, T),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       args=(), options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descent_polish(h, n, n_pass, t_end):
    h = h.copy()
    c5, _ = compute_c5(h)
    for _ in range(n_pass):
        if time.time() > t_end:
            break
        improved = False
        corr, _ = compute_c5(h)
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        top5 = np.argsort(corr_full)[-5:]
        for idx in top5:
            delta = idx - (n - 1)
            for j in range(n):
                if time.time() > t_end:
                    break
                for dv in [0.02, -0.02, 0.05, -0.05]:
                    h[j] += dv
                    h = project_feasible(h, n)
                    c5_new, _ = compute_c5(h)
                    if c5_new < c5 - 1e-13:
                        c5 = c5_new
                        improved = True
                        break
                    h[j] -= dv
                if not improved:
                    h = project_feasible(h, n)
                    c5, _ = compute_c5(h)
        if not improved:
            break
    return h, c5
