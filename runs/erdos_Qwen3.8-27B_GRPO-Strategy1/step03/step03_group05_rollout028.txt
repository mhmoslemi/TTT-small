            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += room * (diff / tot)
        else:
            room = h.copy()
            tot = room.sum()
            if tot < 1e-14: break
            h += room * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_pen_obj(x, T, n, lam):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(T*(corr-m))))/T) * dx
    val += lam * (x.sum() - n/2.0)**2
    return val

def subgrad_step(h, n, top_k, T2):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    vals = corr[idx]
    wts = np.exp(T2*(vals - vals.max()))
    wts /= wts.sum()
    grad = np.zeros(n)
    for w, ki in zip(wts, idx):
        s = int(ki - (n-1))
        if s >= 0:
            grad[s:] += w*(1 - h[:n-s])
            grad[:n-s] += -w*h[s:]
        else:
            m = -s
            grad[:n-m] += w*(1 - h[m:])
            grad[m:] += -w*h[:n-m]
    return grad

def nesterov_sg(h, n_steps, top_k, alpha0, decay, T2, seed):
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    for it in range(n_steps):
        grad = subgrad_step(h_cur, n, top_k, T2)
        v = 0.9*v + alpha*grad
        h_new = project_feasible(h_cur - v, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
        h_cur = h_new
        alpha *= decay
    return best, best_c5

def pairwise_cd(h, n_iters, seed, n_evals=9):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        i, j = rng.choice(n, 2, replace=False)
        lo = max(-h[i], h[j]-1.0)
        hi = min(1.0-h[i], h[j])
        if hi - lo < 1e-10:
            continue
        best_d = 0.0
        best_v = compute_c5(h)
        for d in np.linspace(lo, hi, n_evals):
            ht = h.copy()
            ht[i] += d
            ht[j] -= d
            c = compute_c5(ht)
            if c < best_v:
                best_v = c
                best_d = d
        h[i] += best_d
        h[j] -= best_d
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def targeted_repair(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    s_star = int(np.argmax(corr) - (n-1))
    h_new = h.copy()
    if s_star >= 0:
        for i in range(s_star, n):
            if h_new[i] > 0.5 and h_new[i-s_star] < 0.5:
                d = min(h_new[i]*0.2, 1-h_new[i-s_star])
                h_new[i] -= d

            elif h_new[i] < 0.5 and h_new[i-s_star] > 0.5:
                d = min((1-h_new[i])*0.2, h_new[i-s_star])
                h_new[i] += d
                h_new[i-s_star] -= d
    else:
        m = -s_star
        for i in range(0, n-m):

                d = min(h_new[i+m]*0.2, 1-h_new[i])

                h_new[i] += d

                d = min((1-h_new[i+m])*0.2, h_new[i])
                h_new[i+m] += d
                h_new[i] -= d

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

                d = min(a * 0.15, 1.0 - b)
                h_new[i] -= d
                h_new[i - s_star] += d
            elif a < 0.6 and b > 0.4:
                d = min((1.0 - a) * 0.15, b)
                h_new[i] += d
                h_new[i - s_star] -= d
    else:
        m = -s_star
        for i in range(0, n - m):
            a, b = h_new[i + m], h_new[i]
            if a > 0.4 and b < 0.6:
                d = min(a * 0.15, 1.0 - b)
                h_new[i + m] -= d
                h_new[i] += d
            elif a < 0.6 and b > 0.4:
                d = min((1.0 - a) * 0.15, b)
                h_new[i + m] += d
                h_new[i] -= d
    return project_feasible(h_new, n)

def lsq_refine(h, T, n, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    res = minimize(lse_pen_obj, h, args=(T, n, 800.0),
                   method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 40})
    return project_feasible(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with varying temperature
    for T in [150, 800, 4000, 25000]:
        if remaining() < 25:
            break
        hc = lsq_refine(best_h, T, best_n, 500, 1e-15)
        cc = compute_c5(hc)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    # Phase 2: Nesterov subgradient (multiple configs)
    for tk, al, dec, T2 in [(3, 0.006, 0.9997, 40), (5, 0.005, 0.9996, 30), (8, 0.004, 0.9995, 20)]:
        if remaining() < 15:
            break
        hs, cs = nesterov_sg(best_h, 400, tk, al, dec, T2, seed + tk * 7)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # Phase 3: Undirected pairwise CD
    if remaining() > 60:
        n_cd = min(40000, int(remaining() * 20))
        hc, cc = pairwise_cd(best_h, n_cd, seed)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    # Phase 4: Directed pairwise CD (targets worst shift)
    if remaining() > 60:
        n_dcd = min(40000, int(remaining() * 20))
        hc, cc = pairwise_cd(best_h, n_dcd, seed + 111, directed=True)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    # Phase 5: Basin hopping
    n_trials = min(25, int(remaining() / 40))
    for trial in range(n_trials):
        if remaining() < 30:
            break
        r = np.random.random()
        if r < 0.35:
            sc = 0.025 + 0.015 * np.random.random()
            hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.60:
            hp = best_h.copy()
            n_flip = min(8, best_n // 25)
            idx = np.random.choice(best_n, n_flip, replace=False)
            hp[idx] = 1.0 - hp[idx]
            hp = project_feasible(hp, best_n)
        elif r < 0.80:
            hp = targeted_repair(best_h, best_n)
        else:
            # Block swap
            hp = best_h.copy()
            blen = min(15, best_n // 15)
            s1 = np.random.randint(0, best_n - blen)
            s2 = np.random.randint(0, best_n - blen)
            if abs(s1 - s2) > blen:
                hp[s1:s1+blen], hp[s2:s2+blen] = hp[s2:s2+blen].copy(), hp[s1:s1+blen].copy()
                hp = project_feasible(hp, best_n)
            else:
                hp = project_feasible(best_h + np.random.randn(best_n) * 0.02, best_n)
        # Refine
        if remaining() < 15:
            break
        ho = lsq_refine(hp, 2000, best_n, 250, 1e-14)
        ho, co = pairwise_cd(ho, 4000, seed + 200 + trial, directed=True)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho

    # Phase 6: Multi-resolution
    for nr in [150, 250, 350]:
        if remaining() < 50:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hu = project_feasible(hu + np.random.randn(nr) * 0.015, nr)
        hu = lsq_refine(hu, 1500, nr, 350, 1e-14)
        hu, cu = pairwise_cd(hu, 8000, seed + 500, directed=True)
        cu = compute_c5(hu)
        # Downsample back
        x_dn = np.linspace(0, 2, best_n)
        hd = np.interp(x_dn, x_new, hu)
        hd = project_feasible(hd, best_n)
        cd = compute_c5(hd)
        if cd < best_c5:
            best_c5, best_h = cd, hd
        elif cu < best_c5:
            best_c5, best_h, best_n = cu, hu, nr

    # Phase 7: Intensive directed CD
    if remaining() > 40:
        n_final = min(50000, int(remaining() * 18))
        hc, cc = pairwise_cd(best_h, n_final, seed + 777, directed=True)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    # Phase 8: High-T L-BFGS polish + final pairwise
    if remaining() > 20:
        hf = lsq_refine(best_h, 80000, best_n, 600, 1e-16)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        hc, cc = pairwise_cd(best_h, 15000, seed + 999, directed=True)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    # Phase 9: Micro Nesterov
    if remaining() > 8:
        hs, cs = nesterov_sg(best_h, 300, 3, 0.002, 0.9992, 80, seed + 1)
        if cs < best_c5:
            best_c5, best_h = cs, hs
        hf = lsq_refine(best_h, 100000, best_n, 400, 1e-16)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
