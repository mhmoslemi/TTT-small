def project_feasible(h, n):

            if tot < 1e-15:

        if c < best_c:
            best_c = c
            best = h.copy()
        alpha *= 0.9985
    return best, best_c

def micro_coord_descent(h, n, n_rounds=30):
    h = h.copy()
    best_c = compute_c5(h)
    for rnd in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argpartition(corr, -4)[-4:]
        improved = False
        for ki in top_idx:
            s = int(ki - (n - 1))
            for j in range(n):
                if s >= 0:
                    if 0 <= j+s < n or 0 <= j-s < n:
                        old = h[j]
                        for delta in [-0.02, -0.005, 0.005, 0.02]:
                            h[j] = np.clip(old + delta, 0, 1)
                        trial = project_feasible(h, n)
                        ct = compute_c5(trial)
                        if ct < best_c - 1e-9:
                            best_c = ct
                            improved = True
                            break
                        else:
                            h[j] = old
                        h = project_feasible(h, n)
                else:
                    if 0 <= j+s < n or 0 <= j-s < n:
                        old = h[j]
                        for delta in [-0.02, -0.005, 0.005, 0.02]:
                            h[j] = np.clip(old + delta, 0, 1)
                        trial = project_feasible(h, n)
                        ct = compute_c5(trial)
                        if ct < best_c - 1e-9:
                            best_c = ct
                            improved = True
                            break
                        else:
                            h[j] = old
                        h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 200
        best_h = np.zeros(n0)
        best_h[:n0//2] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP quick polish (multiple temperatures)
    for T in [150, 500, 1500]:
        if time.time() - t0 > budget_s * 0.08:
            break
        ht = slsqp_wrap(best_h, softmax_obj, (T, best_n), 400, 1e-14)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 2: Fourier re-parameterization (explore smooth manifold)
    for nm in [8, 12, 16, 22, 28]:
        if time.time() - t0 > budget_s * 0.32:
            break
        nf = 2 * nm
        vars_init = grid_to_fourier(best_h, nm)
        bnds_f = [(-0.48, 0.48)] * nf
        try:
            res = minimize(fourier_eval, vars_init, args=(nm, best_n),
                          method='SLSQP', bounds=bnds_f,
                          options={'maxiter': 600, 'ftol': 1e-15})
            hf = fourier_to_grid(res.x, nm, best_n)
            hf = project_feasible(hf, best_n)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5, best_h = cf, hf
        except Exception:
            pass

    # Phase 3: Vectorized targeted gradient descent
    if time.time() - t0 < budget_s * 0.50:
        for tk, ns, al in [(3, 200, 0.006), (5, 150, 0.004), (8, 120, 0.003)]:
            if time.time() - t0 > budget_s * 0.50:
                break
            ht, ct = targeted_grad_step(best_h, best_n, top_k=tk, n_steps=ns, alpha0=al)
            if ct < best_c5:
                best_c5, best_h = ct, ht
            ht = slsqp_wrap(ht, topk_obj, (tk, best_n), 300, 1e-14)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht

    # Phase 4: Multi-start with structured perturbations
    n_starts = max(8, int((budget_s * 0.22) / 12))
    for si in range(n_starts):
        if time.time() - t0 > budget_s * 0.74:
            break
        r = np.random.random()
        if r < 0.35:
            hp = best_h + np.random.randn(best_n) * 0.012
            hp = np.convolve(hp, np.ones(7)/7, mode='same')
        elif r < 0.65:
            freq = np.random.choice([3, 5, 7, 9, 11, 17])
            phase = np.random.uniform(0, 2*np.pi)
            t = (np.arange(best_n) + 0.5) * 2.0 / best_n
            hp = best_h + 0.018 * np.sin(np.pi * freq * t + phase)
        else:
            hp = best_h.copy()
            i = np.random.randint(0, best_n - 6)
            j = np.random.randint(0, best_n - 6)
            hp[i:i+4], hp[j:j+4] = hp[j:j+4], hp[i:i+4]
        hp = project_feasible(hp, best_n)
        hp = slsqp_wrap(hp, softmax_obj, (500, best_n), 250, 1e-13)
        hp = slsqp_wrap(hp, topk_obj, (3, best_n), 250, 1e-14)
        cp = compute_c5(hp)
        if cp < best_c5:
            best_c5, best_h = cp, hp

    # Phase 5: Coordinate-level micro-optimization
    if time.time() - t0 < budget_s * 0.84:
        hm, cm = micro_coord_descent(best_h, best_n, n_rounds=20)
        if cm < best_c5:
            best_c5, best_h = cm, hm
        hf = slsqp_wrap(best_h, topk_obj, (2, best_n), 400, 1e-15)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    # Phase 6: Resolution exploration
    if time.time() - t0 < budget_s * 0.92:
        for nr in [180, 250]:
            if time.time() - t0 > budget_s * 0.92:
                break
            x_old = (np.arange(best_n) + 0.5) * 2.0 / best_n
            x_new = (np.arange(nr) + 0.5) * 2.0 / nr
            hu = np.interp(x_new, x_old, best_h)
            hu = project_feasible(hu, nr)
            hh = slsqp_wrap(hu, topk_obj, (3, nr), 400, 1e-13)
            c5_hi = compute_c5(hh)
            x_dn = (np.arange(best_n) + 0.5) * 2.0 / best_n
            hd = np.interp(x_dn, x_new, hh)
            hd = project_feasible(hd, best_n)
            c5_dn = compute_c5(hd)
            if c5_dn < best_c5:
                best_c5, best_h = c5_dn, hd
            elif c5_hi < best_c5:
                best_c5, best_h, best_n = c5_hi, hh, nr

    # Phase 7: Final intensive polish
    if time.time() - t0 < budget_s * 0.985:
        for T in [80, 300, 800, 3000]:
            if time.time() - t0 > budget_s * 0.985:
                break
            hf = slsqp_wrap(best_h, softmax_obj, (T, best_n), 700, 1e-15)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5, best_h = cf, hf
        hf = slsqp_wrap(best_h, topk_obj, (2, best_n), 800, 1e-16)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        ht, ct = targeted_grad_step(best_h, best_n, top_k=2, n_steps=80, alpha0=0.001)
        if ct < best_c5:
            best_c5, best_h = ct, ht
        hf = slsqp_wrap(best_h, topk_obj, (3, best_n), 400, 1e-16)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
