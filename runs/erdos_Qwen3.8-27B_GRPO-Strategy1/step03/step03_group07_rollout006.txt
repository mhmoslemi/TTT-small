    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    time_limit_ph3 = budget_s * 0.58
    for it in range(500):
        if elapsed() > time_limit_ph3:
            break
        r = np.random.rand()
        if r < 0.25:
            scale = 0.08 + 0.08 * (1.0 - it / 500.0)
            h_pert = bh_h + scale * np.random.randn(n_bh)
        elif r < 0.45:
            blen = max(3, n_bh // (8 + 8 * (it // 50)))
            s1 = np.random.randint(0, n_bh - blen)
            s2 = np.random.randint(0, n_bh - blen)
            h_pert = bh_h.copy()
            h_pert[s1:s1+blen] = bh_h[s2:s2+blen]
            h_pert[s2:s2+blen] = bh_h[s1:s1+blen]
        elif r < 0.65:
            h_pert = bh_h.copy()
            nflip = max(2, n_bh // (5 + it // 40))
            idx = np.random.choice(n_bh, nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif r < 0.85:
            # Gaussian-bump perturbation
            h_pert = bh_h.copy()
            center = np.random.randint(0, n_bh)
            width = np.random.randint(3, n_bh // 5)
            kernel = np.exp(-0.5 * ((np.arange(n_bh) - center) / width) ** 2)
            h_pert += np.random.uniform(0.1, 0.3) * kernel * np.random.choice([-1, 1])
        else:
            # Random reweighting
            h_pert = bh_h.copy()
            h_pert = h_pert ** np.random.uniform(0.5, 1.5)
            h_pert = make_feasible(h_pert, n_bh)

        h_pert = make_feasible(h_pert, n_bh)
        T_use = 300 + 400 * (it // 10)
        try:
            h_new = lbgfsopt(h_pert, T_use, n_bh, 250)
            c5_new = compute_c5(h_new)
        except Exception:
            h_new = h_pert
            c5_new = compute_c5(h_pert)

        temp = 0.0025 * (1.0 - it / 500.0) + 0.00015
        if c5_new < bh_c5 or np.random.rand() < np.exp(-(c5_new - bh_c5) / temp):
            bh_c5 = c5_new
            bh_h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Multi-resolution exploration
    for n_try in [140, 160, 240, 280]:
        if elapsed() > budget_s * 0.76:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for trial in range(2):
            if elapsed() > budget_s * 0.76:
                break
            h0 = h_interp + np.random.uniform(-0.03, 0.03, n_try)
            h0 = make_feasible(h0, n_try)
            for T in [500, 8000]:
                if elapsed() > budget_s * 0.76:
                    break
                try:
                    h_new = lbgfsopt(h0, T, n_try, 600)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new
                except Exception:
                    pass

    # Phase 5: Targeted coordinate descent guided by worst shift
    n_cur = len(best_h)
    if elapsed() < budget_s * 0.93:
        stuck = 0
        for it in range(3000):
            if elapsed() > budget_s * 0.93:
                break
            improved = False
            dx_val = 2.0 / n_cur
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            worst_k = np.argmax(corr)
            # Indices affected by worst shift
            if worst_k < n_cur:
                lo = max(0, worst_k - 15)
                hi = min(n_cur, worst_k + 15)
            else:
                lo = max(0, worst_k - n_cur - 15)
                hi = min(n_cur, worst_k - n_cur + 15)
            targeted = np.random.randint(lo, max(lo+1, hi), size=8)
            random_c = np.random.choice(n_cur, 6, replace=False)
            all_c = np.unique(np.concatenate([targeted, random_c]))

            for j in all_c:
                for delta in [0.005, -0.005, 0.015, -0.015, 0.04, -0.04]:
                    h_try = best_h.copy()
                    val = best_h[j] + delta
                    if val < 0 or val > 1:
                        continue
                    h_try[j] = val
                    h_try = make_feasible(h_try, n_cur)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                stuck += 1
            else:
                stuck = 0
            if stuck > 150:
                break

    # Phase 6: Final high-precision SLSQP
    n_cur = len(best_h)
    for T in [100000, 500000]:
        if elapsed() > budget_s * 0.995:
            break
        try:
            h_new = slsqpopt(best_h, T, n_cur, 5000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
