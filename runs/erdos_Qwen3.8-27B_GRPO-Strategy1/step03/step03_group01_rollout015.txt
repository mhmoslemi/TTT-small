import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-10 and h2.max() <= 1.0 + 1e-10:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def weighted_grad(h, N, w):
    center = N - 1
    nc = len(w)
    grad = np.zeros(N)
    thresh = 1e-8
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def make_weights(corr, center, beta, topK):
    nc = len(corr)
    if topK >= nc:
        sc = beta * corr
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
    else:
        idx = np.argsort(corr)[-topK:]
        w = np.zeros(nc)
        vals = corr[idx]
        sc = beta * vals
        sc -= sc.max()
        wk = np.exp(sc)
        wk /= wk.sum()
        for i2, j in enumerate(idx):
            w[j] = wk[i2]
    return w

def optimize_phase(h, N, target, deadline, beta_lo, beta_hi, tk_lo, tk_hi, lr0, mu0):

    best_h = h.copy()

    mu = mu0
    it = 0
    stall = 0

        topK = max(1, int(tk_lo + (tk_hi - tk_lo) * frac))
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = float(np.max(corr))
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        w = make_weights(corr, N - 1, beta, topK)
        grad = weighted_grad(h, N, w)
        v = mu * v + grad
        h_new = project_sum(h - lr * v, target)
        h = h_new
        if it % 100 == 0:
            lr *= 0.96
            mu = min(0.95, mu + 0.002)
        if stall > 300:

            lr = max(lr * 1.5, lr0 * 0.5)
            stall = 0
        if lr < 1e-11:

    return best_h, best_c5

def polish_grad(h, N, target, deadline, n_rounds=20):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = float(np.max(corr))
        w = make_weights(corr, N - 1, 500.0, 2)
        grad = weighted_grad(h, N, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:

        trial_count = 0
        for i in order[:40]:

                    break
