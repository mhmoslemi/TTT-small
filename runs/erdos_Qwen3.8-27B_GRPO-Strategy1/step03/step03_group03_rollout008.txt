def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):

    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:

            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0, 1)
    return h

def smooth_erdos(h, T):

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0, 1)
    return h

def smooth_erdos(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter, maxtime):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_erdos, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'maxtime': maxtime})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_1d(h, i, T):
    n = len(h)
    def f(v):
        h2 = h.copy()
        h2[i] = v
        h2 = project_feasible(h2, n)
        return smooth_erdos(h2, T)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 50})
    return res.x

def coordinate_descent(h, T, n_sweeps=30, maxtime=60):
    n = len(h)
    t0 = time.time()
    h_cur = h.copy()
    best_c5, _ = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if time.time() - t0 > maxtime:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > maxtime:
                break
            v_new = coord_1d(h_cur, i, T)
            h_new = h_cur.copy()
            h_new[i] = v_new
            h_new = project_feasible(h_new, n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                h_cur = h_new
                best_c5 = c5_new
                improved = True
        if not improved:
            break
    return h_cur, best_c5

def momentum_subgradient(h, n_iters=5000, maxtime=60):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    t0 = time.time()
    lr = 0.05
    mom = 0.7
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        if time.time() - t0 > maxtime:
            break
        _, corr = compute_c5(h_cur)
        m = np.max(corr)
        thresh = m - 1e-9
        active = np.where(corr >= thresh)[0][:8]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                grad[:n-k] += (1.0 - h_cur[k:])
                grad[k:] += -h_cur[:n-k]
            else:
                kp = -k
                grad[kp:] += (1.0 - h_cur[:n-kp])
                grad[:n-kp] -= h_cur[kp:]
        grad /= len(active)
        v = mom * v - lr * grad
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 0.35)
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr *= 0.95
            if stall > 200:
                break
    return best_h, best_c5

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    return x[:-1].sum() - len(x[:-1]) / 2.0

def erdos_in(x):
    h = x[:-1]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr

def slsqp_direct(h, maxiter=500, maxtime=60):
    n = len(h)
    _, corr = compute_c5(h)
    t_init = float(np.max(corr)) + 0.05
    x0 = np.concatenate([h, [t_init]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': erdos_eq}, {'type': 'ineq', 'fun': erdos_in}]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'maxtime': maxtime})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def binary_swap_search(h, n_iters=6000, maxtime=40):
    n = len(h)
    t0 = time.time()
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    order = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[order[n//2:]] = 1.0
    best_c5, _ = compute_c5(h_bin)
    current_c5 = best_c5
    rng = np.random.default_rng(123)
    for it in range(n_iters):
        if time.time() - t0 > maxtime:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c5_new, _ = compute_c5(h_bin)
        if c5_new < current_c5 - 1e-14:
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return h_bin, best_c5

def interpolate_h(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)
