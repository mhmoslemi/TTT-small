import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(g, target):
    n = len(g)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(g - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(g - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            g[:m] += (1.0 - h[s:s+m])
            if s > 0:
                g[s:] -= h[:m]
    else:
        sa = -s
        m = n - sa
        if m > 0:
            g[sa:] += (1.0 - h[:m])
            if sa > 0:
                g[:m] -= h[sa:sa+m]
    return g

def obj_summax(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx]
    m = c.max()
    exps = np.exp(T * (c - m))
    w = exps / exps.sum()
    val = m + np.log(exps.sum()) / T
    return val * dx, w, idx

def obj_sm(h, T, k):
    v, _, _ = obj_summax(h, T, k)
    return v

def grad_sm(h, T, k):
    n = len(h)
    dx = 2.0 / n
    v, w, idx = obj_summax(h, T, k)
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        s = int(ki - (n - 1))
        g += wi * lag_grad(h, s)
    return g * dx

def obj_pen(h, lam, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    val = (m + np.log(exps.sum()) / T) * dx
    val += lam * (h.sum() - n / 2.0) ** 2
    return val

def grad_pen(h, lam, T, k):
    n = len(h)
    g = grad_sm(h, T, k)
    g += 2.0 * lam * (h.sum() - n / 2.0)
    return g

def sum_con(h):
    return float(h.sum() - len(h) / 2.0)

def sum_con_jac(h):
    return np.ones(len(h))

def slsqp_analytic(h, T, k, budget_s):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj_sm, h, args=(T, k), jac=grad_sm,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-15})
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        return h

def lbfgs_penalty(h, lam, T, k):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, args=(lam, T, k), jac=grad_pen,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': 500, 'gtol': 1e-10})
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        return h

def nesterov_grad(h, steps, k, alpha0, decay):
    n = len(h)
    target = n / 2.0
    v = np.zeros(n)
    beta = 0.9
    best = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    for it in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -k)[-k:]
        m_c = corr[idx].max()
        exps = np.exp(500.0 * (corr[idx] - m_c))
        w = exps / exps.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, idx):
            s = int(ki - (n - 1))
            g += wi * lag_grad(h, s)
        g *= (2.0 / n)
        v = beta * v + (1.0 - beta) * g
        h_prev = h
        h = project_box_sum(h - alpha * v, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        else:
            h = h_prev
            alpha *= 0.9
        alpha = max(alpha * decay, 1e-6)
    return best, best_c5

def coord_descent(h, n_iter, n):
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h_old = h[i]
            # Try moving h[i] by small amounts
            for delta in [-0.05, -0.02, 0.02, 0.05, -0.1, 0.1]:
                h_new_i = h_old + delta
                if h_new_i < 0 or h_new_i > 1:
                    continue
                # Check sum constraint: adjust one other coordinate
                # For simplicity, just try if within bounds
                h[i] = h_new_i
                # Re-center: find a free coordinate to adjust
                diff = target - h.sum()
                if abs(diff) > 1e-10:
                    h[i] = h_old
                    continue
                # Actually we need sum to stay at target
                # Instead, swap: increase h[i] by delta, decrease h[j] by delta
                j = np.random.randint(n)
                if j == i:
                    continue
                h[j] = h[j] - delta
                if h[j] < 0 or h[j] > 1:
                    h[j] = h[j] + delta
                    h[i] = h_old
                    continue
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best = h.copy()
                    improved = True
                    break
                else:
                    h[i] = h_old
                    h[j] = h[j] + delta
            if not improved and it > n_iter * 0.7:
                break
        h = best.copy()
    return best, best_c5

def swap_coord(h, i, j, delta, n):
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    if 0 <= h_new[i] <= 1 and 0 <= h_new[j] <= 1:
        return h_new, True
    return h, False
