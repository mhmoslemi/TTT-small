import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_gradient(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    n_corr = len(corr)
    if K >= n_corr:
        top_idx = np.arange(n_corr)
        c_top = corr
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
        c_top = corr[top_idx]
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-7:
            continue
        k = int(top_idx[ix])
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, true_c5

    best_h = h.copy()

    b1, b2, eps = 0.85, 0.999, 1e-8

    v = np.zeros(N)
    it = 0
    stall = 0
    plateau = best_c5
    rng = np.random.RandomState(2024)
    while time.time() < deadline:
        if best_c5 < plateau - 1e-10:
            plateau = best_c5
            stall = 0
        else:
            stall += 1
        frac = min(it / 300.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** min(it / 200.0, 1.0)
        K = max(2, int(K0 - (K0 - K1) * min(it / 400.0, 1.0)))
        if stall > patience:
            h = h + rng.randn(N) * 0.015
            h = project_sum(h, target)
            m[:] = 0.0
            v[:] = 0.0
            it = 0
            stall = 0
        grad, c5_val = topk_gradient(h, N, tau, K)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_c5

def local_refine(h, N, target, dx, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(999)
    while time.time() < deadline:
        h_curr = best_h.copy()
