    corr = np.correlate(h, 1.0 - h, mode='full')

    return grad / len(active), c5

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -0.5, 0.5
    for _ in range(55):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_grad(h, N, K, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * dx
    nc = len(corr)
    if K >= nc:

        vals = corr

        idx = np.argpartition(corr, -K)[-K:]
        vals = corr[idx]
