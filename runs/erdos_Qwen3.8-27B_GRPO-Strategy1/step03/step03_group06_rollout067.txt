import numpy as np
from scipy.optimize import minimize, minimize_scalar, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, max_sweeps=500, seed=42, deadline=None):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    c5_init = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    ks = np.arange(2 * n - 1)
    best_c5 = c5_init
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            h0 = h[j]
            mask1 = ks <= j
            mask2 = ks <= n - 1 - j
            slopes = np.zeros(2 * n - 1)
            slopes[mask1] = 1.0 - h[j - ks[mask1]]
            if mask2.any():
                slopes[mask2] -= h[j + ks[mask2]]
            k0_base = h0 * (1.0 - h0)
            
            # Evaluate at boundaries and a few interior points
            # f(t) = max(corr + (t-h0)*slopes + [t(1-t)-k0_base] at k=0)
            # At t=0: corr - h0*slopes, k=0: corr[0] - k0_base
            # At t=1: corr + (1-h0)*slopes, k=0: corr[0] + 1-k0_base
            
            f_curr = np.max(corr)
            v0 = corr - h0 * slopes
            v0[0] -= k0_base
            f0 = np.max(v0)
            v1 = corr + (1.0 - h0) * slopes
            v1[0] += 1.0 - k0_base
            f1 = np.max(v1)
            
            best_t = h0
            best_f = f_curr
            if f0 < best_f:
                best_f = f0
                best_t = 0.0
            if f1 < best_f:
                best_f = f1
                best_t = 1.0
            
            # Check a few interior points if current is not at boundary
            if best_t > 0.01 and best_t < 0.99:
                for t in [0.25, 0.5, 0.75, best_t * 0.5, best_t * 1.5]:
                    t = np.clip(t, 0.0, 1.0)
                    v = corr + (t - h0) * slopes
                    v[0] += t * (1.0 - t) - k0_base
                    f = np.max(v)
                    if f < best_f:
                        best_f = f
                        best_t = t
            
            # Fine local search around best_t
            if 0.0 < best_t < 1.0:
                lo = max(0.0, best_t - 0.1)
                hi = min(1.0, best_t + 0.1)
                ts = np.linspace(lo, hi, 9)
                for t in ts:
                    v = corr + (t - h0) * slopes
                    v[0] += t * (1.0 - t) - k0_base
                    f = np.max(v)
                    if f < best_f:
                        best_f = f
                        best_t = t
            
            if abs(best_t - h0) > 1e-10:
                h[j] = best_t
                corr += (best_t - h0) * slopes
                corr[0] += best_t * (1.0 - best_t) - k0_base
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                improved = True
        
        if not improved:
            break
    
    best_c5 = compute_c5(h)
    return h, best_c5

def slsqp_smooth(h0, n, T, maxiter=1000):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    
    def obj_grad(x):
        # Numerical gradient
        x = np.clip(x, 0, 1)
        g = np.zeros(n)
        eps = 1e-5
        for i in range(n):
            xp = x.copy()
            xp[i] = min(1.0, x[i] + eps)
            xm = x.copy()
            xm[i] = max(0.0, x[i] - eps)
            g[i] = (obj(xp) - obj(xm)) / (xp[i] - xm[i] + 1e-15)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

from scipy.optimize import minimize

def lbfgsb_opt(h0, n, T):
    dx = 2.0 / n
    ks = np.arange(2*n-1)
    
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    
    def grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        w = np.exp(T * (corr - m)) / T
        g = np.zeros(n)
        for j in range(n):
            dcorr = np.zeros(2*n-1)
            m1 = ks <= j
            m2 = ks <= n-1-j
            dcorr[m1] = 1.0 - x[j - ks[m1]]
            dcorr[m2] -= x[j + ks[m2]]
            g[j] = dx * np.dot(w, dcorr)
        # Handle bounds: zero out gradient at bounds
        g[x <= 1e-10] = 0
        g[x >= 1 - 1e-10] = 0
        return g
    
    cons_sum = n / 2.0
    # Can't use equality constraint easily with L-BFGS-B
    # Use penalty instead
    def obj_pen(x):
        x = np.clip(x, 0, 1)
        s = x.sum()
        pen = 100.0 * (s - cons_sum) ** 2
        return obj(x) + pen
    
    def grad_pen(x):
        x = np.clip(x, 0, 1)
        g = grad(x) + 200.0 * (x.sum() - cons_sum)
        g[~((x > 1e-10) & (x < 1-1e-10))] = 0
        return g
    
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_pen, h0, jac=grad_pen, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})
    return project(np.clip(res.x, 0, 1), n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, max_sweeps=500, seed=42, deadline=None):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    ks = np.arange(2 * n - 1)

    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            h0 = h[j]
            m1 = ks <= j
            m2 = ks <= n - 1 - j
            slopes = np.zeros(2 * n - 1)
            slopes[m1] = 1.0 - h[j - ks[m1]]
            slopes[m2] -= h[j + ks[m2]]
            k0b = h0 * (1.0 - h0)
            
            def f_of_t(t):
                v = corr + (t - h0) * slopes
                v[0] += t * (1.0 - t) - k0b
                return np.max(v)
            
            # Evaluate candidates
            f_curr = np.max(corr)
            best_t, best_f = h0, f_curr
            
            f0 = f_of_t(0.0)
            if f0 < best_f:
                best_f, best_t = f0, 0.0

                    best_c5 = c5_curr
        
        if not improved:
            break

    return h, best_c5

def lbfgs_opt(h0, n, T, maxiter=300):
    dx = 2.0 / n
    ks = np.arange(2 * n - 1)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        s = x.sum() - n / 2.0
        return val * dx + 50.0 * s * s
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        w = np.exp(T * (corr - m)) / T
        g = np.zeros(n)
        for j in range(n):
            d = np.zeros(2 * n - 1)
            m1 = ks <= j
            m2 = ks <= n - 1 - j
            d[m1] = 1.0 - x[j - ks[m1]]
            d[m2] -= x[j + ks[m2]]
            g[j] = dx * np.dot(w, d)
        s = x.sum() - n / 2.0
        g += 100.0 * s
        g[x <= 1e-8] = 0
        g[x >= 1.0 - 1e-8] = 0
        return g
    
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return project(np.clip(res.x, 0, 1), n)
