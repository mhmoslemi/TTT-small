**Key idea: After the main optimization, do a "coordinate descent" pass where we optimize one point at a time, specifically targeting the worst shift.**

For each i, given all other h[j] (j≠i) fixed, what value of h[i] minimizes the max overlap? This is a 1D optimization that can be done analytically.

The overlap at shift k contributes:
- If i is in the "left" part of shift k: h[i] * (1-h[i+k]) → linear in h[i]
- If i is in the "right" part: h[i-k] * (1-h[i]) → linear in h[i] (with negative sign)

So for a fixed shift k, the overlap as a function of h[i] is:
f(h[i]) = h[i] * A_i - h[i] * B_i + C
where A_i = sum of (1-h[i+k]) over shifts where i is on the left
and B_i = sum of h[i-k] over shifts where i is on the right

For the MAX over shifts, it's the max of several such linear functions (one per active shift). The minimum of the max of linear functions is at a breakpoint.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_gradient(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / N)
    if K >= len(corr):
        top_idx = np.arange(len(corr))
        c_top = corr.copy()
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
        c_top = corr[top_idx].copy()
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-9:
            continue
        k = int(top_idx[ix])
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def optimize(h, N, target, dx, t_end, tau0, tau1, lr0, lr1, K0, K1):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    stall = 0
    while time.time() < t_end:
        it += 1
        frac = min(it / 500.0, 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        K = max(2, int(round(K0 + (K1 - K0) * frac)))
        if stall > 120:
            lr *= 0.55
            stall = 0
        grad, c5v = topk_gradient(h, N, tau, K)
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c5

def target_perturb(h, N, target, dx, ntrials, t_end, rng):

                elif t[i] < 0.5 and t[i + L_worst] > 0.5:
                    t[i] += 0.3
            elif L_worst < 0:

                i = rng.randint(M, N)
                if t[i - M] > 0.5 and t[i] < 0.5:
