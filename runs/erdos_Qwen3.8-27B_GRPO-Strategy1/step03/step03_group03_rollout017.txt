def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sc = T * (corr - m)
    sm = np.max(sc)
    return (m + (sm + np.log(np.sum(np.exp(sc - sm)))) / T) * dx

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sc = T * (corr - m)
    sm = np.max(sc)
    return (m + (sm + np.log(np.sum(np.exp(sc - sm)))) / T) * dx

def eq_fun(x):
    return x.sum() - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def active_smooth(x, idx, T, dx):
    cc = np.correlate(x, 1.0 - x, mode='full')
    sub = cc[idx]
    m = np.max(sub)
    sc = T * (sub - m)
    sm = np.max(sc)
    return (m + (sm + np.log(np.sum(np.exp(sc - sm)))) / T) * dx

def aug_obj(x, T, lam, rho, n):
    s = x.sum() - n / 2.0
    return smooth_obj(x, T) + lam * s + 0.5 * rho * s * s

def slsqp_opt(h, T, maxiter=3000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def aug_lag_solve(h0, T, max_inner=150, rho0=5000.0, max_outer=20):
    n = len(h0)
    lam = 0.0
    rho = rho0
    h = h0.copy()
    bnds = [(0.0, 1.0)] * n
    for _ in range(max_outer):
        res = minimize(aug_obj, h, args=(T, lam, rho, n), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': max_inner, 'ftol': 1e-15})
        h = np.clip(res.x, 0, 1)
        s = h.sum() - n / 2.0
        if abs(s) < 1e-10:
            break
        lam += rho * s
        rho *= 2.0
        if rho > 1e10:
            break
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Augmented Lagrangian with escalating T
    for T in [500, 5000, 50000, 500000]:
        if elapsed() > budget_s * 0.15:
            break
        h_t = aug_lag_solve(best_h, T, max_inner=200, rho0=5000.0, max_outer=15)
        c5_t, _ = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t

    # Phase 2: SLSQP with high T (targeted polynomial-time)
    for T in [100000, 500000, 2000000, 10000000]:
        if elapsed() > budget_s * 0.40:
            break
        h_t = slsqp_opt(best_h, T, maxiter=5000)
        c5_t, _ = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t

    # Phase 3: Active-set targeted optimization
    if elapsed() < budget_s * 0.52:
        _, corr = compute_c5(best_h)
        for k in [2, 3, 5]:
            if elapsed() > budget_s * 0.52:
                break
            idx = np.argsort(corr)[::-1][:k]
            dx = 2.0 / best_n
            bnds = [(0.0, 1.0)] * best_n
            cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
            try:
                res = minimize(active_smooth, best_h, args=(idx, 100000, dx),
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-16})
                h_t = project_feasible(np.clip(res.x, 0, 1), best_n)
                c5_t, _ = compute_c5(h_t)
                if c5_t < best_c5:
                    best_c5, best_h = c5_t, h_t
            except Exception:
                pass

    # Phase 4: n-sweep (try nearby discretizations)
    for dn in [-2, 2, 4, -4, 6, -6, 8, -8, 10]:
        if elapsed() > budget_s * 0.72:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try % 2 != 0:
            continue
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, n_try)
        hi = project_feasible(np.interp(xn, xo, best_h), n_try)
        h_t = aug_lag_solve(hi, 100000, max_inner=150)
        h_t = slsqp_opt(h_t, 1000000, maxiter=4000)
        c5_t, _ = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_h, best_n = c5_t, h_t, n_try

    # Phase 5: Multi-start perturbations
    for scale in [0.04, 0.02, 0.01, 0.005]:
        if elapsed() > budget_s * 0.87:
            break
        for _ in range(3):
            if elapsed() > budget_s * 0.87:
                break
            hp = project_feasible(best_h + np.random.randn(best_n) * scale, best_n)
            h_t = aug_lag_solve(hp, 200000, max_inner=150)
            h_t = slsqp_opt(h_t, 2000000, maxiter=4000)
            c5_t, _ = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5, best_h = c5_t, h_t

    # Phase 6: Final ultra-high T polish
    if elapsed() < budget_s * 0.97:
        h_t = slsqp_opt(best_h, 50000000, maxiter=5000)
        c5_t, _ = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t
        h_t = slsqp_opt(best_h, 200000000, maxiter=3000)
        c5_t, _ = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
