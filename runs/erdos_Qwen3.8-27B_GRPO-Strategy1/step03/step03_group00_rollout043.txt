**Approach 1: Use the structure of the max more directly**

The objective is max_k c[k] where c[k] = sum_i h[i+k](1-h[i]) * dx.

We can write this as: min max_k c[k] subject to constraints.

This is a min-max problem. By max-min duality (von Neumann), this equals:
min_h max_{p ∈ Δ} sum_k p[k] c[k]

where Δ is the simplex. But this doesn't directly help computationally.

**Approach 2: Block coordinate descent with smarter updates**

Instead of optimizing one coordinate at a time, optimize a block of coordinates. For example, optimize pairs (i, i+m) for some m. This is still 1D per pair if we fix the sum.

**Approach 3: Use the duality to identify which lags are "active"**

At the optimum, typically only a few lags are at the maximum. We can:
1. Compute c[k] for all k
2. Identify the top few lags
3. Optimize h to minimize the max of just those lags (a convex problem if we use the smooth approximation, or a small LP)

**Approach 4: Exploit the Toeplitz structure**

The correlation c[k] for all k can be computed as a single FFT. But more importantly, the gradient of c[k] w.r.t. h[j] has a Toeplitz structure.

**Approach 5: Direct LP/SOCP formulation (for small n)**

        if tot < 1e-13:
            return h.copy()

            return h.copy()

def solve_slp(h, n, t_end, max_iter=200, damping_init=1.0):
    """Successive Linear Programming for the bilinear min-max problem."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    damping = damping_init
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        
        # Build LP matrices
        m = 2 * n - 1
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        
        for k in range(-(n-1), n):
            row = k + n - 1
            i_lo = max(0, -k)
            i_hi = min(n-1, -k)  # n-1-k ≥ 0 → i ≤ n-1-k
            b_const = 0.0
            for i in range(i_lo, i_hi + 1):
                j1 = i + k
                A_ub[row, j1] += (1.0 - best_h[i])
                A_ub[row, i] += (-best_h[j1])
                b_const += best_h[j1] * (1.0 - best_h[j1])
            A_ub[row, :n] *= dx
            A_ub[row, n] = -1.0
            b_ub[row] = -dx * b_const
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n * 0.5])
        
        bounds = [(0,1)] * n + [(None, None)]
        c_obj = np.zeros(n+1)
        c_obj[n] = 1.0
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        h_lp = res.x[:n]
        h_new = (1.0 - damping) * best_h + damping * h_lp
        h_new = project(h_new, n)
        c5_new = compute_c5(h_new)

            best_c5 = c5_new
            best_h = h_new

            if damping < 0.01:
                break

        if np.max(np.abs(h_new - best_h)) < 1e-8:
            break

from scipy.optimize import linprog, minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slp_step(h, n, t, damping=1.0):
    """One SLP iteration: build and solve LP, return new h and success flag."""
    dx = 2.0 / n
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    hh = h  # current linearization point
    for k in range(-(n-1), n):
        row = k + n - 1
        i_lo = max(0, -k)
        i_hi = min(n-1, n-1-k)
        b_const = 0.0
        for i in range(i_lo, i_hi + 1):
            j1 = i + k
            A_ub[row, j1] += (1.0 - hh[i])
            A_ub[row, i] += (-hh[j1])
            b_const += hh[j1] * (1.0 - hh[j1])
        A_ub[row, :n] *= dx
        A_ub[row, n] = -1.0
        b_ub[row] = -dx * b_const
    
    A_eq = np.ones((1, n+1))
    A_eq[0, n] = 0.0
    b_eq = np.array([n * 0.5])
    
    bounds = [(0, 1)] * n + [(None, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True})
    except Exception:
        return None, False
    
    if not res.success:
        return None, False
    
    h_new = (1.0 - damping) * h + damping * res.x[:n]
    h_new = project(h_new, n)
    return h_new, True

def solve_slp(h, n, t_end, max_iter=500, damping_init=1.0):
    best_h = project(h.copy(), n)
    best_c5 = compute_c5(best_h)
    damping = damping_init

            break
        h_new, ok = slp_step(best_h, n, damping)
        if not ok:
            break
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new
            damping = min(1.0, damping * 1.02)
            stall = 0
        else:
            damping *= 0.5
            stall += 1
            if stall > 20 or damping < 0.005:
                break
    
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            orig = best_h[i]
            best_h[i] = 0.0
            c_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            
            # Coefficient vector
            n1 = 2 * n - 1
            coef = np.zeros(n1)
            k_lo1, k_hi1 = -i, n - 1 - i
            if k_lo1 <= k_hi1:
                idx = np.arange(k_lo1, k_hi1 + 1) + n - 1
                coef[idx] += 1.0 - best_h[i + k_lo1 : i + k_hi1 + 1]
            k_lo2, k_hi2 = i - n + 1, i
            if k_lo2 <= k_hi2:
                idx2 = np.arange(k_lo2, k_hi2 + 1) + n - 1
                coef[idx2] += -best_h[i - k_lo2 : i - k_hi2 + 1]
            
            mid = n - 1  # index of k=0 in the corr array
            def f1d(t_val):
                vals = c_base + t_val * coef
                vals[mid] -= t_val * t_val
                return np.max(vals) * (2.0 / n)
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-9})
            best_h[i] = res.x
            best_h = project(best_h, n)
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                improved = True
        
        if not improved:
            break
    return best_h, best_c5

def smooth_c5(h, T):
    n = len(h)

    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

```python
import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def build_slp_lp(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    q = h * (1.0 - h)
    cumq = np.concatenate([[0.0], np.cumsum(q)])
    for k in range(-(n - 1), n):
        row = k + n - 1
        if k >= 0:
            b_const = cumq[n] - cumq[k]
        else:
            b_const = cumq[n + k]
        j1_lo, j1_hi = max(0, k), min(n - 1, n - 1 + k)
        if j1_lo <= j1_hi:
            A_ub[row, j1_lo:j1_hi + 1] += (1.0 - h[j1_lo - k:j1_hi - k + 1]) * dx
        j2_lo, j2_hi = max(0, -k), min(n - 1, n - 1 - k)
        if j2_lo <= j2_hi:
            A_ub[row, j2_lo:j2_hi + 1] += (-h[j2_lo + k:j2_hi + k + 1]) * dx
        A_ub[row, n] = -1.0
        b_ub[row] = -dx * b_const
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    bounds = [(0, 1)] * n + [(None, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    return c_obj, A_ub, b_ub, A_eq, b_eq, bounds

def slp_step(h, n, damping):
    c_obj, A_ub, b_ub, A_eq, b_eq, bounds = build_slp_lp(h, n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
    except Exception:
        return None, False
    if not res.success:
        return None, False
    h_new = (1.0 - damping) * h + damping * res.x[:n]
    return project(h_new, n), True

def solve_slp(h, n, t_end, max_iter=500, damping_init=1.0):
    best_h = project(h.copy(), n)
    best_c5 = compute_c5(best_h)
    damping = damping_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        h_new, ok = slp_step(best_h, n, damping)
        if not ok:
            break
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-11:
            best_c5 = c5_new
            best_h = h_new
            damping = min(1.0, damping * 1.03)
            stall = 0
        else:
            damping *= 0.5
            stall += 1
            if stall > 25 or damping < 0.003:
                break
    return best_h, best_c5

def coord_opt_1d(h, n, i):
    orig = h[i]
    h[i] = 0.0
    c_base = np.correlate(h, 1.0 - h, mode='full')
    n1 = 2 * n - 1
    coef = np.zeros(n1)
    k_lo1, k_hi1 = -i, n - 1 - i
    if k_lo1 <= k_hi1:
        idx = np.arange(k_lo1, k_hi1 + 1) + n - 1
        coef[idx] += 1.0 - h[i + k_lo1:i + k_hi1 + 1]
    k_lo2, k_hi2 = i - n + 1, i
    if k_lo2 <= k_hi2:
        idx2 = np.arange(k_lo2, k_hi2 + 1) + n - 1
        coef[idx2] += -h[i - k_lo2:i - k_hi2 + 1]
    mid = n - 1
    dx = 2.0 / n
    def f1d(t):
        vals = c_base + t * coef
        vals[mid] -= t * t
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    h[i] = res.x
    return res.x

def coord_descent(h, n, t_end, max_sweeps=60):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            orig = best_h[i]
            best_h[i] = 0.0
            c_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            n1 = 2 * n - 1
            coef = np.zeros(n1)
            k_lo1, k_hi1 = -i, n - 1 - i
            if k_lo1 <= k_hi1:
                idx = np.arange(k_lo1, k_hi1 + 1) + n - 1
                coef[idx] += 1.0 - best_h[i + k_lo1:i + k_hi1 + 1]
            k_lo2, k_hi2 = i - n + 1, i
            if k_lo2 <= k_hi2:
                idx2 = np.arange(k_lo2, k_hi2 + 1) + n - 1
                j_idx = i - np.arange(k_lo2, k_hi2 + 1)
                coef[idx2] += -best_h[j_idx]
            mid = n - 1
            def f1d(t):
                vals = c_base + t * coef
                vals[mid] -= t * t
                return np.max(vals) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            best_h[i] = res.x
            best_h = project(best_h, n)
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                improved = True
        if not improved:
            break
    return best_h, best_c5
