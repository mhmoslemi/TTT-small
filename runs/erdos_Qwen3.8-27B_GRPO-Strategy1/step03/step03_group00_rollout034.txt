    # Find lambda such that sum(clamp(h + lambda, 0, 1)) = S
    lo, hi = -1.0 - h.min(), 1.0 - h.max()

        if s < S:

So: corr[k] = corr_base[k] + h_i*(1-h[i+k])*[0≤i+k<n] - h_i*h[i-k]*[0≤i-k<n]
= corr_base[k] + h_i * ((1-h[i+k])*[0≤i+k<n] - h[i-k]*[0≤i-k<n])
= corr_base[k] + h_i * B_k

where B_k = (1-h[i+k])*[0≤i+k<n] - h[i-k]*[0≤i-k<n]

And the C₅ candidate is max_k (corr_base[k] + h_i * B_k) * dx.

So we need to minimize f(t) = max_k (a_k + t*b_k) for t ∈ [0,1] where a_k = corr_base[k] and b_k = B_k.

This is a standard 1D convex piecewise-linear minimization. The minimum is at:
- t=0, t=1, or at an intersection point of two lines (a_i + t*b_i = a_j + t*b_j).

For the exact minimum, we can:
1. Sort by slope b_k
2. Use a rotating calipers or convex hull approach
3. Or just use minimize_scalar (which is what the previous code does)

For n=160, there are 2n-1 = 319 lines. Finding the exact minimum of the upper envelope requires O(n²) in worst case (checking all pairs), but in practice minimize_scalar with bounded method will find it in ~30-50 evaluations.

Actually, for the exact minimum of max of linear functions over [0,1], we can do better:
- The minimum of the max is where the maximum is minimized.
- This happens at a point where at least two lines tie for the maximum.
- We can binary search on the value: for a given value v, check if all lines are ≤ v at t, i.e., t ≥ (v - a_k)/b_k for b_k > 0 and t ≤ (v - a_k)/b_k for b_k < 0.
- The feasible t exists iff max_{b_k>0} (v-a_k)/b_k ≤ min_{b_k<0} (v-a_k)/b_k and 0≤t≤1.

This is O(n log n) per bisection step, and O(log(1/ε)) steps. Very efficient.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = T * (corr - m)
    return float((m + np.log(np.sum(np.exp(e))) / T) * dx)

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = T * (corr - m) - (T * (corr - m)).max()
    w = np.exp(e)
    w /= w.sum()
    # grad[j] = dx * (sum_p w[j+p]*(1-h[n-1-p]) - sum_q w[n-1-j+q]*h[q])
    # Vectorize: first term is correlation-like

    W1 = w[idx1]  # (n, n)
    W2 = w[idx2]  # (n, n)
    first = W1 @ (1.0 - h_rev)
    second = W2 @ h

        v2h = v2 / (1 - b2 ** t)

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    e = T * (corr - m)
    e -= e.max()  # numerical stability
    w = np.exp(e)
    w /= w.sum()
    
    h_rev = h[::-1]
    # Build index matrices
    js = np.arange(n)
    ps = np.arange(n)
    idx1 = js[:, None] + ps[None, :]  # shape (n,n), values j+p
    idx2 = (n - 1 - js)[:, None] + ps[None, :]  # shape (n,n), values n-1-j+q
    
    first = w[idx1].dot(1.0 - h_rev)
    second = w[idx2].dot(h)
    return dx * (first - second)
