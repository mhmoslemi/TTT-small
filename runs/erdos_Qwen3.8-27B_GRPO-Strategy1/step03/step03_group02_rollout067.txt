```python
def smoothed_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr) * dx)
    sc = tau * corr
    sc_shifted = sc - sc.max()
    w = np.exp(sc_shifted)
    wsum = w.sum()
    w /= wsum
    
    grad = np.zeros(N)
    center = N - 1
    
    # d = 0
    w0 = w[center]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    
    # d > 0
    for d in range(1, N):
        wd = w[center + d]
        if wd < 1e-12:
            continue
        # + part: j in [d, N-1], (1-h[j-d])
        grad[d:] += wd * (1.0 - h[:N-d])
        # - part: j in [0, N-1-d], -h[j+d]
        grad[:N-d] -= wd * h[d:]
    
    # d < 0, M = -d > 0
    for M in range(1, N):
        wM = w[center - M]
        if wM < 1e-12:
            continue
        # + part: j in [0, N-1-M], (1-h[j+M])
        grad[:N-M] += wM * (1.0 - h[M:])
        # - part: j in [M, N-1], -h[j-M]
        grad[M:] -= wM * h[:N-M]
    
    return grad, c5
```

```python
import numpy as np
import time
import math

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def full_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr) * dx)
    sc = tau * corr - (tau * corr).max()
    w = np.exp(sc)
    ws = w.sum()
    w /= ws
    grad = np.zeros(N)
    c = N - 1
    w0 = w[c]
    if w0 > 1e-10:
        grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, N):
        wp = w[c + d]
        if wp > 1e-10:
            grad[d:] += wp * (1.0 - h[:N-d])
            grad[:N-d] -= wp * h[d:]
        wn = w[c - d]
        if wn > 1e-10:
            grad[:N-d] += wn * (1.0 - h[d:])
            grad[d:] -= wn * h[:N-d]
    return grad, c5

def run_grad(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, iters):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    while time.time() < deadline:
        frac = min(it / max(iters, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        grad, c5v = full_grad(h, N, tau)
        if c5v < best_c5 - 1e-12:
            best_c5 = c5v
            best_h = h.copy()
        it += 1
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m / (1-b1**it)
        vh = v / (1-b2**it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c5

def coord_polish(h, N, target, dx, n_passes, deadline):
    best = h.copy()
    best_c = compute_c5(h, dx)
    for p in range(n_passes):
        if time.time() > deadline:
            break
        for i in range(N):
            best_v = h[i]
            best_cv = 0.0
            # Try a few values around current
            for dv in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05, -0.1, 0.1]:
                v = h[i] + dv
                if v < 0 or v > 1:
                    continue
                trial = h.copy()
                trial[i] = v
                trial = project_sum(trial, target)
                c = compute_c5(trial, dx)
                if c < best_cv or best_cv == 0:
                    if best_cv == 0 or c < best_cv:
                        best_cv = c
                        best_v = trial.copy()
            if best_v is not h and best_cv < compute_c5(h, dx):
                h = best_v
        c_now = compute_c5(h, dx)
        if c_now < best_c - 1e-12:
            best_c = c_now
            best = h.copy()
    return best, best_c
```

```python
def compute_c5(h, dx):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_lich(h, N, tau, topk=0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr) * dx)
    if topk > 0 and topk < len(corr):
        idx = np.argsort(corr)[-topk:]
        cl = np.zeros_like(corr)
        cl[idx] = corr[idx]
        sc = tau * cl
    else:
        sc = tau * corr
    scm = sc.max()
    w = np.exp(sc - scm)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(N), c5
    w /= wsum
    grad = np.zeros(N)
    c = N - 1
    w0 = w[c]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, N):
        wp = w[c + d]
        if wp > 1e-12:

            grad[:N-d] -= wp * h[d:]
        wn = w[c - d]
        if wn > 1e-12:
            grad[:N-d] += wn * (1.0 - h[d:])
            grad[d:] -= wn * h[:N-d]
    return grad, c5

def grad_descent(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, topk, iters):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0

    while time.time() < deadline:
        frac = min(1.0 * it / max(iters, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        grad, c5v = grad_lich(h, N, tau, topk)
        if c5v < best_c5 - 1e-13:
            best_c5 = c5v
            best_h = h.copy()
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        if not np.all(np.isfinite(h)):
            h = best_h.copy()
    return best_h, best_c5
```
