```python
def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)

        total = room.sum()
        if total < 1e-14:
            return h
        h = h + room * (diff / total)
```

If I want a term -h[i]^2, I set (1/2)Q[i,i] = -1, so Q[i,i] = -2.

Let me build Q:
For each active shift s > 0:
  For i = s to n-1: Q[i, i-s] -= 1, Q[i-s, i] -= 1
For each active shift s < 0, a = |s|:
  For i = 0 to n-1-a: Q[i, i+a] -= 1, Q[i+a, i] -= 1
For shift s = 0:
  Q[i, i] -= 2 for all i

And c:
For each active shift s > 0: c[i] += 1 for i in [s, n-1]
For each active shift s < 0, a = |s|: c[i] += 1 for i in [0, n-1-a]
For shift s = 0: c[i] += 1 for all i

Then the QP is:
min (1/2) h^T Q h + c^T h
s.t. sum(h) = n/2
     0 <= h <= 1

But Q is negative semidefinite (it's a sum of -h[i]h[j] terms), so (1/2)h^TQh is concave. We're minimizing a concave function, which is a NON-CONVEX QP! This won't work directly with standard QP solvers.

Hmm. So the approach of fixing the active set gives a non-convex QP. That's why the problem is hard.

OK so I'll stick with the gradient method. Let me think about what makes it fast and effective:

1. The gradient computation is O(n*K) per step - very fast
2. The correlation computation is O(n^2) via np.correlate - this is the bottleneck
   For n=160: 160*319 ≈ 51k operations - actually np.correlate is quite fast for this size
3. The projection is O(n)

So each iteration is dominated by np.correlate, which for n=160 should be microseconds. We can do thousands of iterations easily.

Let me write the final solution. I'll focus on:
1. High-quality active-set gradient descent with adaptive step size
2. Multiple K values (from 1 to 50)
3. Structured multi-start
4. Resolution changes
5. A "vibration" step: at the end, try small random perturbations and accept if they help

Let me also think about the learning rate schedule. For subgradient methods, the standard is lr_t = c / sqrt(t). But since the problem is smooth (bilinear), a decreasing polynomial might work better: lr_t = lr0 / (1 + t/tau).

Actually, since we have a projection onto a convex set (box + hyperplane), projected gradient with backtracking should work well.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def grad_shift(h, s, n):
    a = abs(s)
    g = np.zeros(n)
    if a == 0:
        return 1.0 - 2.0 * h
    if a >= n:
        return g
    if s > 0:
        g[0:a] = -h[a:2*a]
        mid = n - 2 * a
        if mid > 0:
            g[a:a+mid] = 1.0 - h[0:mid] - h[2*a:2*a+mid]
        g[n-a:n] = 1.0 - h[n-2*a:n-a] if n-2*a >= 0 else 0.0
    else:
        g[0:a] = 1.0 - h[a:2*a]
        mid = n - 2 * a
        if mid > 0:
            g[a:a+mid] = 1.0 - h[0:mid] - h[2*a:2*a+mid]
        g[n-a:n] = -h[n-2*a:n-a] if n-2*a >= 0 else 0.0
    return g

def active_set_grad(h, n, K, topk_frac=1.0):
    corr = compute_corr(h)
    T = len(corr)
    k_eff = max(1, int(K * topk_frac))
    # Soft logsumexp weighting among top K
    k_idx = np.argpartition(corr, -k_eff)[-k_eff:]
    m = corr[k_idx].max()
    w = np.exp(corr[k_idx] - m)
    w /= w.sum()
    grad = np.zeros(n)
    for i, k in enumerate(k_idx):
        s = k - (n - 1)
        grad += w[i] * grad_shift(h, s, n)
    return grad, corr, k_idx

def gd_optimize(h, n, budget, K_schedule=None, lr0=0.5, decay=0.9995):
    if K_schedule is None:
        K_schedule = [1, 3, 5, 10, 20, 40, 80]
    best_c5 = compute_c5(h)
    h = h.copy()
    lr = lr0
    no_improve = 0
    max_no_improve = 200
    while time.time() < budget:
        grad, corr, k_idx = active_set_grad(h, n, K_schedule[0], 1.0)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.85
            if no_improve > max_no_improve:
                break
        if no_improve % 50 == 0:
            K_schedule = K_schedule[1:] + [K_schedule[-1]]
    return h, best_c5

def multi_K_gd(h, n, budget, K_list, lr0):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for K in K_list:
        if time.time() > budget:
            break
        lr = lr0 / (1.0 + 0.3 * K)
        h_cur = best_h.copy()
        no_imp = 0
        while time.time() < budget:
            grad, corr, _ = active_set_grad(h_cur, n, K)
            h_try = project_feasible(h_cur - lr * grad, n)
            c5_try = compute_c5(h_try)
            if c5_try < compute_c5(best_h) - 1e-12:
                best_h = h_try
                best_c5 = c5_try
                h_cur = h_try
                no_imp = 0
            else:
                no_imp += 1
                lr *= 0.9
                if no_imp > 300:
                    break
    return best_h, best_c5

def perturb_h(h, n, scale, rng):
    r = rng.random()
    if r < 0.35:
        return project_feasible(h + scale * rng.standard_normal(n), n)
    elif r < 0.65:
        blen = rng.randint(3, max(4, n // 5))
        start = rng.randint(0, n - blen)
        h_p = h.copy()
        h_p[start:start+blen] += scale * 1.5 * (2.0 * (rng.random(blen) < 0.5) - 1.0)
        return project_feasible(h_p, n)
    else:
        h_p = h.copy()
        i1, i2 = rng.choice(n, size=2, replace=False)
        h_p[i1] += scale * 0.5
        h_p[i2] -= scale * 0.5
        return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n_init = len(best_h)

    # Phase 1: Active-set gradient at initial resolution
    t_phase1_end = t0 + budget_s * 0.30
    best_h, best_c5 = multi_K_gd(best_h, n_init, t_phase1_end, [1, 3, 5, 8, 12, 20, 40], 1.0)

    # Phase 2: Multi-start with structured perturbations
    n_starts = 15
    t_phase2_end = t0 + budget_s * 0.55
    for si in range(n_starts):
        if elapsed() > t_phase2_end:
            break
        scale = 0.08 * (1.0 - si / n_starts) + 0.02
        h_p = perturb_h(best_h, n_init, scale, rng)
        sub_budget = min(t_phase2_end, t0 + elapsed() + budget_s * 0.02)
        h_try, c5_try = multi_K_gd(h_p, n_init, sub_budget, [3, 8, 20], 0.8)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: Try different resolutions
    t_phase3_end = t0 + budget_s * 0.80
    for n_try in [120, 140, 180, 200, 240]:
        if elapsed() > t_phase3_end:
            break
        x_old = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        # Also try a "snap" version: round to 0.5 grid then project
        h_snap = np.round(h_interp * 2.0) / 2.0
        h_snap = project_feasible(h_snap, n_try)
        for h_start in [h_interp, h_snap]:
            if elapsed() > t_phase3_end:
                break
            h_try, c5_try = multi_K_gd(h_start, n_try, min(t_phase3_end, t0 + elapsed() + budget_s * 0.03), [1, 5, 15, 30], 0.6)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                n_init = n_try

    # Phase 4: Fine-tuning with small K and many iterations
    t_phase4_end = t0 + budget_s * 0.92
    h_cur = best_h.copy()
    n_cur = len(h_cur)
    lr = 0.3
    for _ in range(50000):
        if elapsed() > t_phase4_end:
            break
        for K in [1, 2, 3, 5]:
            grad, corr, _ = active_set_grad(h_cur, n_cur, K)
            h_try = project_feasible(h_cur - lr * grad, n_cur)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                h_cur = h_try
                break
        else:
            lr *= 0.95
            if lr < 1e-5:
                break

    # Phase 5: Final local poke - try tiny adjustments at worst shifts
    t_phase5_end = t0 + budget_s * 0.98
    n_cur = len(best_h)
    for _ in range(20000):
        if elapsed() > t_phase5_end:
            break
        corr = compute_corr(best_h)
        worst = np.argmax(corr)
        s_worst = worst - (n_cur - 1)
        grad_w = grad_shift(best_h, s_worst, n_cur)
        for step in [0.01, 0.02, 0.05, 0.1]:
            h_try = project_feasible(best_h - step * grad_w, n_cur)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                break

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
