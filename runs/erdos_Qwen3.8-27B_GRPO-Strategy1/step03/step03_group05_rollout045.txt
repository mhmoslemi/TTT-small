            if 0 <= j < n:
                if h[i] > 0.5 and h[j] < 0.5:
                    d = min(h[i] * 0.4, 1.0 - h[j]) * 0.6
                    h[i] -= d
                    h[j] += d
                elif h[i] < 0.5 and h[j] > 0.5:
                    d = min((1.0 - h[i]) * 0.4, h[j]) * 0.6
                    h[i] += d
                    h[j] -= d
        h = project_feasible(h, n)
    return h

def cheb_perturb(h, K, scale):
    c = cheb_fit(h, K)
    c += np.random.randn(K + 1) * scale
    return project_feasible(cheb_eval(c, K, len(h)), len(h))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200); best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Chebyshev low-dim optimization (global-ish search)
    for K in [10, 15, 20, 25, 30]:
        if time.time() - t0 > budget_s * 0.20:
            break
        c0 = cheb_fit(best_h, K)
        for trial in range(4):
            if time.time() - t0 > budget_s * 0.20:
                break
            if trial == 0:
                cs = c0.copy()
            else:
                cs = c0 + np.random.randn(K + 1) * 0.08 * trial
            def cheb_obj(c):
                hc = cheb_eval(c, K, best_n)
                hc = project_feasible(hc, best_n)
                dx = 2.0 / best_n
                corr = np.correlate(hc, 1.0 - hc, mode='full')
                m = np.max(corr)
                val = m + np.log(np.sum(np.exp(150.0 * (corr - m)))) / 150.0
                pen = 0.005 * (hc.sum() - best_n / 2.0) ** 2
                return val * dx + pen
            try:
                res = minimize(cheb_obj, cs, method='L-BFGS-B',
                             options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-8})
                hc = project_feasible(cheb_eval(res.x, K, best_n), best_n)
                cc = compute_c5(hc)
                if cc < best_c5:
                    best_c5, best_h = cc, hc
            except Exception:
                pass

    # Phase 2: L-BFGS-B full-dim with LSE (fast, no equality constraint overhead)
    if time.time() - t0 < budget_s * 0.35:
        for T in [80, 200, 500, 1200]:
            if time.time() - t0 > budget_s * 0.35:
                break
            def lbfgs_obj(hf, Tv=T):
                hp = project_feasible(hf, best_n)
                return lse_obj_full(hp, Tv)
            try:
                res = minimize(lbfgs_obj, best_h, method='L-BFGS-B',
                             bounds=[(0.0, 1.0)] * best_n,
                             options={'maxiter': 250, 'ftol': 1e-16, 'gtol': 1e-10})
                hc = project_feasible(res.x, best_n)
                cc = compute_c5(hc)
                if cc < best_c5:
                    best_c5, best_h = cc, hc
            except Exception:
                pass

    # Phase 3: SLSQP top-k (focus on worst shifts)
    if time.time() - t0 < budget_s * 0.50:
        for k in [2, 3, 5]:
            if time.time() - t0 > budget_s * 0.45:
                break
            hc = slsqp(best_h, topk_obj_full, (k,), 350, 1e-14)
            cc = compute_c5(hc)
            if cc < best_c5:
                best_c5, best_h = cc, hc
        for T in [300, 1000, 4000]:
            if time.time() - t0 > budget_s * 0.50:
                break
            hc = slsqp(best_h, lse_obj_full, (T,), 350, 1e-14)
            cc = compute_c5(hc)
            if cc < best_c5:
                best_c5, best_h = cc, hc

    # Phase 4: Basin hopping (diverse mutations + refine)
    elapsed = time.time() - t0
    remaining = budget_s * 0.72 - elapsed
    n_trials = max(int(remaining / 10), 0)
    for trial in range(n_trials):
        if time.time() - t0 > budget_s * 0.72:
            break
        r = np.random.random()
        if r < 0.30:
            hp = cheb_perturb(best_h, 20, 0.04 + 0.02 * (1 - trial / max(n_trials, 1)))
        elif r < 0.50:
            hp = best_h.copy()
            st = np.random.randint(0, max(1, best_n - 10))
            ln = np.random.randint(4, max(5, best_n // 5))
            en = min(st + ln, best_n)
            sh = np.random.randint(1, max(2, ln // 2 + 1))
            hp[st:en] = np.roll(hp[st:en], sh)
            hp = project_feasible(hp, best_n)
        elif r < 0.70:
            sc = 0.02 + 0.03 * np.random.random()
            hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.85:
            hp = targeted_fix(best_h, best_n, 2)
        else:
            # Swap-based
            hp = best_h.copy()
            for _ in range(np.random.randint(3, 8)):
                i, j = np.random.choice(best_n, 2, replace=False)
                hp[i], hp[j] = hp[j], hp[i]
            hp = project_feasible(hp, best_n)
        # Refine
        ho = slsqp(hp, topk_obj_full, (4,), 180, 1e-12)
        ho = slsqp(ho, lse_obj_full, (600,), 180, 1e-13)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho

    # Phase 5: Multi-resolution
    for nr in [240, 280, 320]:
        if time.time() - t0 > budget_s * 0.85:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hh = slsqp(hu, topk_obj_full, (4,), 250, 1e-13)
        c5_hi = compute_c5(hh)
        if c5_hi < best_c5:
            best_c5, best_h, best_n = c5_hi, hh, nr
        else:
            x_dn = np.linspace(0, 2, best_n)
            hd = np.interp(x_dn, x_new, hh)
            hd = project_feasible(hd, best_n)
            c5_dn = compute_c5(hd)
            if c5_dn < best_c5:
                best_c5, best_h = c5_dn, hd

    # Phase 6: Targeted final repair + very high-T LSE
    if time.time() - t0 < budget_s * 0.93:
        for _ in range(4):
            if time.time() - t0 > budget_s * 0.92:
                break
            hr = targeted_fix(best_h, best_n, 3)
            hr = slsqp(hr, topk_obj_full, (2,), 300, 1e-15)
            cr = compute_c5(hr)
            if cr < best_c5:
                best_c5, best_h = cr, hr
        hf = slsqp(best_h, lse_obj_full, (10000,), 400, 1e-16)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    # Phase 7: Ultra-fine local search
    if time.time() - t0 < budget_s * 0.985:
        for _ in range(15):
            if time.time() - t0 > budget_s * 0.985:
                break
            hp = project_feasible(best_h + np.random.randn(best_n) * 0.004, best_n)
            ho = slsqp(hp, topk_obj_full, (2,), 120, 1e-16)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho
        # One more targeted fix
        hr = targeted_fix(best_h, best_n, 5)
        hr = slsqp(hr, lse_obj_full, (20000,), 200, 1e-16)
        cr = compute_c5(hr)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
