```python
    p = exp_adj / exp_adj.sum()

    grad = np.zeros(n)
    for k in range(2*n-1):

            continue
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        tot = s
        if tot < 1e-14:
            return h
        h -= h * (-diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1.0:
        return m * dx
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        tot = s
        if tot < 1e-14:
            return h
        h = h * (target / tot)
    return np.clip(h, 0.0, 1.0)

def corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_of(h, top_k=10):
    """Subgradient of max(corr) w.r.t. h, using top_k lags."""
    n = len(h)
    dx = 2.0 / n
    corr = corr_of(h)
    m = np.max(corr)
    # Get indices of top_k lags
    top_idx = np.argsort(corr)[-top_k:]
    active = corr[top_idx] - m
    if active[0] < -1e-12:
        # Only the max is active
        weights = np.zeros(len(top_idx))
        weights[np.argmax(active)] = 1.0
    else:
        # Softmax with small T
        T = 200.0
        exp_a = np.exp(T * active)
        weights = exp_a / exp_a.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        w = weights[i]
        if w < 1e-10:

        l = k - (n - 1)
        if l == 0:

        else:

            if l > 0:

                grad[al:] += -w * h[:n-al]
            else:
                grad[al:] += w * (1.0 - h[:n-al])
                grad[:n-al] += -w * h[al:]
    return grad * dx

            break
        g = grad_of(h_curr, top_k=15)
        gnorm = np.linalg.norm(g, ord=np.inf)
        if gnorm < 1e-10:

        # Step size: moderate, with backtracking
        step = lr / (gnorm * n + 1.0)
        h_new = project(h_curr - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_curr + 1e-12:
            h_curr = h_new
            c5_curr = c5_new
            if c5_new < best_c5:

                best_h = h_new.copy()
            lr *= 1.01
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    return best_h, best_c5

def solve_1d_subprob(corr0, b_vec, quad_idx):
    """Solve min_{x in [0,1]} max(corr0 + b*x, with quad correction at quad_idx)."""
    # Evaluate on grid
    xs = np.linspace(0.0, 1.0, 101)
    vals = np.empty(101)
    for i, x in enumerate(xs):
        v = np.max(corr0 + b_vec * x)
        if x > 0:
            quad_val = corr0[quad_idx] + x - x * x
            v = max(v, quad_val)
        vals[i] = v
    best_i = np.argmin(vals)
    # Refine
    lo = max(0, best_i - 3)
    hi = min(100, best_idx + 3)
    xs_f = np.linspace(xs[lo], xs[hi], 201)

        if x > 0:
            quad_val = corr0[quad_idx] + x - x * x
            v = max(v, quad_val)
        vals_f[i] = v
    return float(xs_f[np.argmin(vals_f)])

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    quad_idx = n - 1  # index of k=0 in full correlate
    
    for sweep in range(max_sweeps):

            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            # Temporarily set h[i] = 0
            old_val = best_h[i]
            best_h[i] = 0.0
            corr0 = corr_of(best_h)
            
            # Compute b[k] = d(corr[k])/d(h[i])
            b = np.zeros(2 * n - 1)
            l = 0  # We'll compute for each lag
            # For each lag l in -(n-1)..(n-1):
            for k in range(2 * n - 1):
                ll = k - (n - 1)
                c = 0.0
                j1 = i + ll

                if 0 <= j2 < n and j2 != i:

            # Solve 1D

            else:
                best_h[i] = old_val  # revert
                best_h = project(best_h, n)
        
        if not improved:
            break
    return best_h, best_c5
```

- corr0[quad_idx] + x - x²  [this is the true quadratic term]

```python
def solve_1d_subprob(corr0, b_vec, quad_idx):
    xs = np.linspace(0.0, 1.0, 101)
    vals = np.empty(101)
    for i, x in enumerate(xs):
        # Linear terms (excluding quad_idx)
        linear_vals = corr0 + b_vec * x
        linear_vals[quad_idx] = -np.inf  # exclude
        v_lin = np.max(linear_vals)
        # Quadratic term
        v_quad = corr0[quad_idx] + x - x * x
        vals[i] = max(v_lin, v_quad)
    best_i = np.argmin(vals)
    # Refine
    lo = max(0, best_i - 3)
    hi = min(100, best_i + 3)
    xs_f = np.linspace(xs[lo], xs[hi], 101)
    vals_f = np.empty(len(xs_f))

        v_lin = np.max(linear_vals)
        v_quad = corr0[quad_idx] + x - x * x
        vals_f[i] = max(v_lin, v_quad)
```

```python
def solve_1d_subprob(corr0, b_vec, quad_idx):
    xs = np.linspace(0.0, 1.0, 101)
    # Linear part: shape (101, 2n-1)
    linear = corr0[None, :] + b_vec[None, :] * xs[:, None]
    linear[:, quad_idx] = -np.inf
    v_lin = np.max(linear, axis=1)
    v_quad = corr0[quad_idx] + xs - xs**2
    vals = np.maximum(v_lin, v_quad)
    best_i = np.argmin(vals)
    # Refine
    lo = max(0, best_i - 3)
    hi = min(100, best_i + 3)
    if hi - lo < 5:
        return float(xs[best_i])
    xs_f = np.linspace(xs[lo], xs[hi], 101)
    linear_f = corr0[None, :] + b_vec[None, :] * xs_f[:, None]
    linear_f[:, quad_idx] = -np.inf
    v_lin_f = np.max(linear_f, axis=1)
    v_quad_f = corr0[quad_idx] + xs_f - xs_f**2
    vals_f = np.maximum(v_lin_f, v_quad_f)
    return float(xs_f[np.argmin(vals_f)])
```
