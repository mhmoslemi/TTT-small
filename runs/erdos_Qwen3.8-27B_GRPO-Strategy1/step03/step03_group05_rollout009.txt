    grad = np.zeros(n)
    if s > 0:
        grad[:n-s] += (1 - h[s:])
        grad[s:] -= h[:n-s]
    elif s < 0:
        s = -s
        grad[s:] += (1 - h[:n-s])
        grad[:n-s] -= h[s:]
    return grad

        g[s:] -= h[:n-s]

        g[:n-s2] -= h[s2:]
    return g

    return project_feasible(h_new, n)

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def targeted_repair(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    s_star = int(np.argmax(corr) - (n - 1))
    hp = h.copy()
    for i in range(n):
        j = i + s_star
        if 0 <= j < n:
            if hp[i] > 0.5 and hp[j] < 0.5:
                d = min(hp[i] * 0.2, 1.0 - hp[j])
                hp[i] -= d; hp[j] += d
            elif hp[i] < 0.5 and hp[j] > 0.5:
                d = min((1.0 - hp[i]) * 0.2, hp[j])
                hp[i] += d; hp[j] -= d
    return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 200
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = c5_val(best_h)

    # Phase 1: Intensive active-set gradient (vectorized O(n) per shift)
    for k in [2, 3, 4, 5, 7, 10]:
        if time.time() - t0 > budget_s * 0.10:
            break
        for a0 in [0.0005, 0.002, 0.005, 0.01]:
            if time.time() - t0 > budget_s * 0.10:
                break
            h_o, c_o = active_set_opt(best_h, k, 400, a0, 0.9995, best_n)
            if c_o < best_c5:
                best_c5, best_h = c_o, h_o

    # Phase 2: SLSQP multi-objective
    if time.time() - t0 < budget_s * 0.25:
        for k in [2, 3, 5]:
            h_t = slsqp_solve(best_h, topk_obj, (k, best_n), 400, 1e-14, best_n)
            c_t = c5_val(h_t)
            if c_t < best_c5:
                best_c5, best_h = c_t, h_t
        for T in [100, 500, 2000, 8000]:
            h_s = slsqp_solve(best_h, softmax_obj, (T, best_n), 350, 1e-14, best_n)
            c_s = c5_val(h_s)
            if c_s < best_c5:
                best_c5, best_h = c_s, h_s

    # Phase 3: Basin hopping with diverse perturbations
    n_trials = int(min(100, budget_s * 20))
    for trial in range(n_trials):
        if time.time() - t0 > budget_s * 0.62:
            break
        r = np.random.random()
        if r < 0.35:
            sc = 0.003 + 0.025 * np.random.random()
            hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.55:
            hp = best_h.copy()
            i = np.random.randint(0, max(1, best_n - 20))
            j = np.random.randint(0, max(1, best_n - 20))
            L = np.random.randint(2, max(3, best_n // 8))
            seg_i = hp[i:i+L].copy(); seg_j = hp[j:j+L].copy()
            hp[i:i+L] = seg_j; hp[j:j+L] = seg_i
        elif r < 0.72:
            hp = targeted_repair(best_h, best_n)
        else:
            # Spectral perturbation: smooth noise
            freqs = np.fft.rfftfreq(best_n)
            noise = np.random.randn(best_n)
            spec = np.fft.rfft(noise)
            spec *= np.exp(-4.0 * np.abs(freqs))
            noise_s = np.fft.irfft(spec, best_n)
            hp = project_feasible(best_h + 0.01 * noise_s, best_n)
        h_r, c_r = active_set_opt(hp, 3, 150, 0.002, 0.999, best_n)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r
        h_s = slsqp_solve(h_r, topk_obj, (3, best_n), 200, 1e-13, best_n)
        c_s = c5_val(h_s)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s

    # Phase 4: Multi-resolution optimization
    for nr in [90, 120, 150, 250, 300]:
        if time.time() - t0 > budget_s * 0.80:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, nr, endpoint=False)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        h_r, c_r = active_set_opt(hu, 3, 200, 0.004, 0.999, nr)
        h_r = slsqp_solve(h_r, topk_obj, (3, nr), 300, 1e-13, nr)
        c_high = c5_val(h_r)
        x_dn = np.linspace(0, 2, best_n, endpoint=False)
        hd = np.interp(x_dn, x_new, h_r)
        hd = project_feasible(hd, best_n)
        c_dn = c5_val(hd)
        if c_dn < best_c5:
            best_c5, best_h = c_dn, hd
        elif c_high < best_c5:
            best_c5, best_h, best_n = c_high, h_r, nr

    # Phase 5: Targeted repair + intensive local
    if time.time() - t0 < budget_s * 0.90:
        for _ in range(6):
            if time.time() - t0 > budget_s * 0.90:
                break
            hr = targeted_repair(best_h, best_n)
            hr, cr = active_set_opt(hr, 3, 200, 0.001, 0.9995, best_n)
            h_s = slsqp_solve(hr, topk_obj, (3, best_n), 350, 1e-14, best_n)
            cr = min(cr, c5_val(h_s))
            if cr < best_c5:
                best_c5, best_h = cr, hr if cr <= c5_val(h_s) else h_s

    # Phase 6: Ultra-fine final polish
    if time.time() - t0 < budget_s * 0.97:
        h_f, c_f = active_set_opt(best_h, 2, 500, 0.0003, 0.9999, best_n)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
        h_f2 = slsqp_solve(best_h, softmax_obj, (10000, best_n), 500, 1e-16, best_n)
        c_f2 = c5_val(h_f2)
        if c_f2 < best_c5:
            best_c5, best_h = c_f2, h_f2
        h_f3, c_f3 = active_set_opt(best_h, 2, 400, 0.0002, 0.99995, best_n)
        if c_f3 < best_c5:
            best_c5, best_h = c_f3, h_f3

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = c5_val(best_h)
    return best_h, best_c5, len(best_h)
