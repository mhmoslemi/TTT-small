Actually, I can be smarter. The correlation c[k] = sum_n h[n+L]*(1-h[n]). If I change h[j] to h[j]+δ:
- c[k] changes by δ*(1-h[n]) where n+L = j (i.e., n = j-L, if valid)
- c[k] changes by -δ*h[n+L] where n = j (if valid)

Δc[k] = δ * (1-h[j-L_k]) * [j-L_k valid] - δ * h[j+L_k] * [j valid, j+L_k valid]

def weighted_gradient(h, N, w):
    N = len(h)
    grad = np.zeros(N)
    center = N - 1
    nc = len(w)
    for k in range(nc):
        wi = w[k]
        if wi < 1e-7:
            continue

        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:

                grad[L:] += wi * (1.0 - h[:N-L])
                grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
    return grad

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-13:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def interp_to(h, N_new, N_old):
    x_old = np.linspace(0.0, 2.0, N_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def weighted_grad(h, N, w):
    grad = np.zeros(N)
    center = N - 1
    nc = len(w)
    for k in range(nc):
        wi = w[k]
        if wi < 1e-8:
            continue
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        elif -N < L < 0:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def lse_optimize(h, N, target, t_end, rng, T0, T1, lr0, mom0):

    mom = mom0
    it = 0
    pat = 0
    while time.time() < t_end:
        it += 1
        frac = min(it / 5000.0, 1.0)
        T = T0 * (T1 / T0) ** frac

        c5h = float(corr.max()) * dx

        es = np.exp(sc)

        w_top = np.zeros(len(corr))
        w_top[top_idx] = 1.0 / k_top
        a = min(1.0, 1.5 * frac)
        w = (1.0 - a) * w_sm + a * w_top
        g = weighted_grad(h, N, w)
        v = mom * v + g
        h = project_feasible(h - lr * v, target)

            lr *= 0.97

            v *= 0.2

def coord_polish(h, N, target, t_end):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    while time.time() < t_end:
        any_improve = False
        for j in range(N):
            if time.time() > t_end:
                break
            orig = h[j]
            trials = [0.0, 1.0, max(0.0, orig - 0.005), min(1.0, orig + 0.005),
                      max(0.0, orig - 0.02), min(1.0, orig + 0.02),
                      max(0.0, orig - 0.08), min(1.0, orig + 0.08)]
            for tv in trials:

                    best_h = h.copy()
                    any_improve = True
