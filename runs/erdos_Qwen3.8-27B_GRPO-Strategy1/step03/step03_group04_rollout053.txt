    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr0
    it = 0
    while time.time() < deadline:
        k = int(k0 + (k1 - k0) * min(1.0, it / 350.0))
        g, peak = subgrad_topk(h, N, k)
        if peak < best_v:
            best_v = peak
            best = h.copy()
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        if it % 200 == 0:
            lr *= decay
    return best, best_v * (2.0 / N)

def binary_search(h, N, deadline, rng):
    h_b = (h > 0.5).astype(float)
    n1 = int(h_b.sum())
    if n1 > N // 2:
        idx = np.where(h_b == 1)[0]
        h_b[idx[N // 2:]] = 0
    elif n1 < N // 2:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:N // 2 - n1]] = 1
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(h_b)))
    cur_pk = best_pk
    T = 4.0
    cool = 0.9997
    while time.time() < deadline:
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = h_b[j], h_b[i]
        new_pk = float(np.max(compute_corr(h_b)))
        d = new_pk - cur_pk
        if d < 0 or (T > 0.005 and rng.random() < np.exp(-d / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        T *= cool
    return best, best_pk * (2.0 / N)

def fine_polish(h, N, target, deadline):
    best = h.copy()
    best_v = float(np.max(compute_corr(h)))
    h_c = h.copy()
    lr = 0.0005
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    it = 0
    while time.time() < deadline:
        g, peak = subgrad_topk(h_c, N, 3)
        if peak < best_v:
            best_v = peak
            best = h_c.copy()
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h_c = project_sum(h_c - lr * mh / (np.sqrt(vh) + eps), target)
        if it % 100 == 0:
            lr *= 0.99
    return best, best_v * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Nesterov on initial
    remain = deadline - time.time()
    if remain > 120:
        d1 = time.time() + remain * 0.18
        h1, c1 = nesterov_opt(h0.copy(), N0, N0 / 2.0, d1, 0.006, 0.96, 6, 35)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: Adam refinement
    remain = deadline - time.time()
    if remain > 100:
        d2 = time.time() + remain * 0.22
        h2, c2 = adam_opt(best_h.copy(), N0, N0 / 2.0, d2, 0.004, 0.985, 8, 45)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Try nearby N values (interpolate and optimize)
    for N in [382, 384, 386, 388]:
        remain = deadline - time.time()
        if remain < 120:
            break
        target = N / 2.0
        x_old = np.linspace(0, 2, N0, endpoint=False)
        x_new = np.linspace(0, 2, N, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_sum(h_init, target)
        d3 = time.time() + remain * 0.30
        h3, c3 = adam_opt(h_init, N, target, d3, 0.004, 0.98, 6, 40)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Multi-start perturbations at best N
    Nb = len(best_h)
    target_b = Nb / 2.0
    for trial in range(4):
        remain = deadline - time.time()
        if remain < 80:
            break
        sigma = 0.025 + 0.015 * trial
        h_m = project_sum(best_h + rng.randn(Nb) * sigma, target_b)
        d4 = time.time() + remain * 0.28
        if trial % 2 == 0:
            h4, c4 = adam_opt(h_m, Nb, target_b, d4, 0.003, 0.985, 5, 30)
        else:
            h4, c4 = nesterov_opt(h_m, Nb, target_b, d4, 0.004, 0.97, 5, 30)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Phase 5: Binary local search
    remain = deadline - time.time()
    if remain > 60:
        d5 = time.time() + remain * 0.5
        h5, c5b = binary_search(best_h, Nb, d5, rng)
        if c5b < best_c5:
            best_c5, best_h = c5b, h5

    # Phase 6: Final fine polish
    remain = deadline - time.time()
    if remain > 15:
        hF, cF = fine_polish(best_h.copy(), Nb, Nb / 2.0, deadline - 1)
        if cF < best_c5:
            best_c5, best_h = cF, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
