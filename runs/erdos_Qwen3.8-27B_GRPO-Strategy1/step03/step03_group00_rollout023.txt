from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    raw = T * (corr - m)
    S = np.sum(np.exp(raw))
    return (m + np.log(S) / T) * dx

def project_exact(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def coord_opt_fast(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save
    c = np.zeros(2 * n - 1)
    ko = n - 1
    for k in range(-(n - 1), n):
        v = 0.0
        j1 = i + k
        if 0 <= j1 < n:
            v += (1.0 - h[j1])
        j2 = i - k
        if 0 <= j2 < n:
            v += (-h[j2])
        c[k + ko] = v
    dx = 2.0 / n
    k0 = n - 1
    def f1d(t):
        vals = corr_base + t * c
        vals[k0] -= t * t
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200, tol=1e-15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.arange(n)
        np.random.shuffle(order)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_fast(best_h, n, int(i))
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_exact(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - tol:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def nest_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    raw = T * (corr - m)
    S = np.sum(np.exp(raw))
    w = np.exp(raw) / S
    smooth_val = (m + np.log(S) / T) * dx
    grad = np.zeros(n)
    ko = n - 1
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-16:
            continue
        k = k_idx - ko
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] += wk * (-h[lo2 - k:hi2 - k])
    return smooth_val, grad * dx

def nesterov_opt(h0, n, T_schedule, t_end, momentum_decay=0.98):
    h = h0.copy()
    h_prev = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = 0.01 / (n * T_schedule[0] * 2.0 / n)
    momentum = 0.0
    for it in range(5000):
        if time.time() > t_end:
            break
        T = T_schedule[min(it, len(T_schedule) - 1)]
        val, grad = nest_grad(h, T, n)
        y = h + momentum * (h - h_prev)
        h_new = project_exact(y - step * grad, n)
        momentum = momentum_decay * momentum + (1 - momentum_decay) * momentum
        h_prev = h
        h = h_new
        c5v = compute_c5(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_exact(np.clip(res.x, 0, 1), n)
