import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:

        else:

def comp_grad(h, N, tau, topK):
    """Compute gradient with softmax + top-K weighting."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr)) * dx
    center = N - 1
    nc = len(corr)
    sc = tau * corr - tau * corr.max()
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    k = max(1, min(topK, nc))
    idx_top = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 15.0) / 50.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    grad = np.zeros(N)
    thresh = 0.002
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, c5

def adam_optimize(h, N, target, deadline, tau_start, tau_end, topK_start, topK_end, lr, beta1, beta2):

    best_h = h.copy()

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def comp_grad(h, N, tau, topK):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr)) * dx
    center = N - 1
    nc = len(corr)
    sc = tau * corr
    sc -= sc.max()
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    k = max(1, min(topK, nc))
    idx_top = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 12.0) / 40.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    grad = np.zeros(N)
    thresh = 0.002
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
