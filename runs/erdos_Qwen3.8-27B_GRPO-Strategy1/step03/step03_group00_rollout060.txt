from scipy.optimize import minimize, minimize_scalar

def project(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(h)
    # Bisection for tau
    lo, hi = -1.0, 1.0
    def f(tau):
        return np.sum(np.clip(h - tau, 0.0, 1.0)) - target
    if f(lo) < 0:
        return np.clip(h, 0.0, 1.0)  # all clipped to 1, sum >= target
    if f(hi) > 0:
        return np.clip(h, 0.0, 1.0)
    for _ in range(50):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0.0, 1.0)

    dx = 2.0 / n

def smooth_obj_and_grad(h, T):
    """Compute smooth log-sum-exp objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Stable log-sum-exp
    log_sum = m + np.log(np.sum(np.exp(T * (corr - m))))
    val = log_sum / T * dx
    
    # Gradient: dx * sum_k p_k * dcorr_k/dh_i
    p = np.exp(T * (corr - m)) / np.sum(np.exp(T * (corr - m)))
    
    # dcorr_k/dh_i for k = m_idx - (n-1)
    # For the 'full' correlation, index m_idx corresponds to lag k = m_idx - (n-1)
    # dcorr_full[n-1+k]/dh[i] = (1-h[i+k]) * [0<=i+k<n] - h[i-k] * [0<=i-k<n]
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if p[k_idx] < 1e-15:
            continue
        k = k_idx - (n - 1)
        # For each i, the contribution is:
        # (1-h[i+k]) if 0<=i+k<n, minus h[i-k] if 0<=i-k<n
        # But we need to be careful: the sum over j of h[j]*(1-h[j+k])
        # d/dh[i] of this sum:
        # Term 1: j=i, gives (1-h[i+k]) if 0<=i+k<n
        # Term 2: j+k=i i.e. j=i-k, gives -h[i-k] if 0<=i-k<n
        
        # Vectorized:
        # For term 1: i ranges over [0, n-1-k] if k>=0, or [0, n-1] if k<0
        # Actually: 0 <= i+k < n means -k <= i < n-k
        # For term 2: 0 <= i-k < n means k <= i < n+k
        
        # Let's just do it element-wise for clarity (n is at most 512)
        # Term 1: (1-h[i+k]) for valid i+k
        if k >= 0:
            # i+k < n => i < n-k, and i >= 0
            idx1 = np.arange(0, n - k)
            grad[idx1] += p[k_idx] * (1.0 - h[idx1 + k])
            # i-k >= 0 => i >= k, and i-k < n => i < n+k (always true since i < n)
            idx2 = np.arange(k, n)
            grad[idx2] -= p[k_idx] * h[idx2 - k]
        else:

            # i+k < n => i < n-k = n+kk (always true), i >= 0, i+k >= 0 => i >= -k = kk
            idx1 = np.arange(kk, n)

            grad[idx2] -= p[k_idx] * h[idx2 - k]

    n = len(h)

    corr = np.correlate(h, 1.0-h, mode='full')  # length 2n-1

from time import time

def project_box_sum(h, target):
    n = len(h)
    def f_of_tau(tau):
        return np.sum(np.clip(h - tau, 0.0, 1.0))
    # Find tau such that f_of_tau(tau) = target
    # f_of_tau is decreasing in tau
    # At tau = min(h) - 1: all h_i - tau > 1, so clip gives 1, sum = n
    # At tau = max(h) + 1: all h_i - tau < 0, so clip gives 0, sum = 0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    if f_of_tau(lo) <= target:
        return np.clip(h - lo, 0.0, 1.0)
    if f_of_tau(hi) >= target:
        return np.clip(h - hi, 0.0, 1.0)
    for _ in range(60):

        if f_of_tau(mid) > target:
            lo = mid
        else:
            hi = mid

    # Build matrix efficiently
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]  # idx[i,r] = i+r
    P = p[idx]  # P[i,r] = p[i+r]

    best_val = smooth_obj(best_h, T)
    lr = 1.0 / T  # Initial learning rate

            break
        obj = smooth_obj(best_h, T)
        grad = smooth_grad(best_h, T)

        for _ in range(30):

            obj_new = smooth_obj(h_new, T)

        best_h = h_new
        best_val = obj_new
        lr = max(alpha * 1.2, 1e-8)  # Increase LR slightly
    
    return best_h

    best_h = h.copy()

        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5

            # For m where 0<=m-i<n: corr_base[m] = corr[m] - h[i]*(1-2*h[m-i]) (if m!=2i)
            # For m=2i: corr_base[m] = corr[m] - h[i] + h[i]²

            hi_old = best_h[i]
            if hi_old < 1e-14:

            #   r = m - i
            #   if 0 <= r < n:

            #       # quadratic term
            #       corr_base[m] = corr_full[m] - hi_old + hi_old**2
            #       # f(t) = corr_base[m] + t - t² for this component
            #     else:
            #       corr_base[m] = corr_full[m] - hi_old * (1 - 2*h[r])
            #       # f(t) = corr_base[m] + t * (1-2*h[r]) for this component

            #     corr_base[m] = corr_full[m] (no change)

                    best_c5 = c5_try
                    improved = True
        
        if not improved:
            break

from scipy.optimize import minimize_scalar
from time import time

def project_box_sum(h, target):
    n = len(h)
    def f_tau(tau):
        return np.sum(np.clip(h - tau, 0.0, 1.0))
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if f_tau(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ec = np.exp(T * (corr - m))
    s = np.sum(ec)
    obj = (m + np.log(s) / T) * dx
    p = ec / s
    q = 1.0 - 2.0 * h
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    grad = dx * (p[idx] @ q)
    return obj, grad

    best_h = h.copy()
    n_iter = 0
    alpha = step_size if step_size else 4.0 / T
    for _ in range(20000):
        if time.time() > t_end:
            break
        obj, grad = smooth_obj_grad(best_h, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:

        step = alpha
        success = False

                success = True

            step *= 0.5

        n_iter += 1

    dx = 2.0 / n
