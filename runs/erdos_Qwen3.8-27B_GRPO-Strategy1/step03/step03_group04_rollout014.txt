import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_grad(h, N, T):
    corr = compute_corr(h)
    dx = 2.0 / N
    c = corr * dx
    cmax = float(c.max())
    exps = np.exp(T * (c - cmax))
    weights = exps / exps.sum()
    grad = np.zeros(N)
    center = N - 1
    for ki in range(len(corr)):
        w = weights[ki] * dx
        if w < 1e-10:
            continue
        L = ki - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            n_ov = N - L
            grad[L:L+n_ov] += w * (1.0 - h[:n_ov])
            grad[:n_ov] -= w * h[L:L+n_ov]
        else:
            M = -L
            n_ov = N - M
            grad[:n_ov] += w * (1.0 - h[M:M+n_ov])
            grad[M:M+n_ov] -= w * h[:n_ov]
    sm = cmax + float(np.log(exps.sum())) / T
    return grad, sm

def adam_softmax(h, N, target, deadline, T_start, T_end, lr0):
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    lr = lr0
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 2000.0)
        T = T_start + (T_end - T_start) * frac
        grad, sm = softmax_grad(h, N, T)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        mh = m / (1.0 - 0.9 ** it)
        vh = v / (1.0 - 0.999 ** it)
        h = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h = project_sum(h, target)
        if it % 100 == 0:
            c = compute_c5(h, N)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
            lr *= 0.99
    return best_h, best_c5

def binary_hill_climb(h, N, deadline, rng, n_restarts=5):
    dx = 2.0 / N
    best = (h > 0.5).astype(float)
    n1 = int(best.sum())
    if n1 > N // 2:
        idx = np.where(best == 1)[0]
        best[idx[-(n1 - N//2):]] = 0
    elif n1 < N // 2:
        idx = np.where(best == 0)[0]
        best[idx[:N//2 - n1]] = 1
    best_c5 = compute_c5(best, N)
    best_int = int(np.max(compute_corr(best)))
    cur_int = best_int
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        T0, T1 = 5.0 / (restart + 1), 0.01
        n_iter = int((deadline - time.time()) * 30 / (n_restarts - restart))
        n_iter = max(n_iter, 500)
        stuck = 0
        for it in range(n_iter):
            if time.time() > deadline:
                break
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best[i], best[j] = best[j], best[i]
            new_int = int(np.max(compute_corr(best)))
            delta = new_int - cur_int
            T = T0 * (T1 / T0) ** (it / n_iter)
            if delta <= 0:
                cur_int = new_int
                if new_int < best_int:
                    best_int = new_int
                    best_c5 = best_int * dx
                    stuck = 0
                else:
                    stuck += 1
            else:
                if T > 0.05 and rng.random() < np.exp(-delta / T):
                    cur_int = new_int
                else:
                    best[i], best[j] = best[j], best[i]
                stuck += 1
            if stuck > 2000 and it > 500:
                break
        if stuck > 2000:
            idx = np.where(best == 1)[0]
            idx2 = np.where(best == 0)[0]
            for _ in range(5):
                a = idx[rng.randint(len(idx))]
                b = idx2[rng.randint(len(idx2))]
                best[a], best[b] = best[b], best[a]
            cur_int = int(np.max(compute_corr(best)))
    return best, best_c5

def block_move_hill(h, N, deadline, rng):
    best = (h > 0.5).astype(float)
    n1 = int(best.sum())
    if n1 > N // 2:
        idx = np.where(best == 1)[0]
        best[idx[-(n1 - N//2):]] = 0
    elif n1 < N // 2:
        idx = np.where(best == 0)[0]
        best[idx[:N//2 - n1]] = 1
    best_int = int(np.max(compute_corr(best)))
    cur_int = best_int
    dx = 2.0 / N
    it = 0
    n_iter = int((deadline - time.time()) * 20)
    while it < n_iter and time.time() < deadline:
        it += 1
        blk = rng.randint(2, 8)
        start1 = rng.randint(0, N - blk)
        pos2 = rng.randint(0, N - blk)
        if abs(start1 - pos2) < blk:
            continue
        seg = best[start1:start1+blk].copy()
        if seg.sum() == 0 or seg.sum() == blk:
            continue
        best[start1:start1+blk] = best[pos2:pos2+blk]
        best[pos2:pos2+blk] = seg
        n1_new = int(best.sum())
        if n1_new != N // 2:
            best[start1:start1+blk] = seg
            best[pos2:pos2+blk] = seg
            continue
        new_int = int(np.max(compute_corr(best)))
        if new_int < cur_int:
            cur_int = new_int
            if new_int < best_int:
                best_int = new_int
        else:
            best[start1:start1+blk] = seg
            best[pos2:pos2+blk] = seg
    return best, best_int * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)

    sizes = [n0, 500, 600]
    for N in sizes:
        if time.time() > deadline - 180:
            break
        if N not in [n0, 500, 600]:
            continue
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, best_h)
        h_init = project_sum(h_init, target)
        c_init = compute_c5(h_init, N)
        if c_init < best_c5:
            best_c5, best_h = c_init, h_init.copy()

        # Phase 1: Soft-max continuous optimization
        if time.time() < deadline - 150:
            d1 = time.time() + (deadline - time.time()) * 0.20
            h1, c1 = adam_softmax(h_init.copy(), N, target, d1, 5.0, 40.0, 0.08)
            if c1 < best_c5:
                best_c5, best_h = c1, h1.copy()
            h_work = h1

        # Phase 2: Binary optimization
        if time.time() < deadline - 100:
            d2 = time.time() + (deadline - time.time()) * 0.25
            h2, c2 = binary_hill_climb(h_work, N, d2, rng, n_restarts=4)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()
            h_work = h2

        # Phase 3: Block moves
        if time.time() < deadline - 70:
            d3 = time.time() + (deadline - time.time()) * 0.20
            h3, c3 = block_move_hill(h_work, N, d3, rng)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()
            h_work = h3

        # Phase 4: Un-round and continuous refine
        if time.time() < deadline - 50:
            h4 = h_work.astype(float) + rng.randn(N) * 0.05
            h4 = project_sum(h4, target)
            d4 = time.time() + (deadline - time.time()) * 0.25
            h4, c4 = adam_softmax(h4, N, target, d4, 20.0, 80.0, 0.03)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()

        # Phase 5: Final binary polish
        if time.time() < deadline - 20:
            d5 = deadline - 10
            h5, c5v = binary_hill_climb(h_work, N, d5, rng, n_restarts=3)
            if c5v < best_c5:
                best_c5, best_h = c5v, h5

    # Final evaluate
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
