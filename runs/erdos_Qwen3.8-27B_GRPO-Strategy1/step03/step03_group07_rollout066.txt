import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t > 1e-14:
                h += (diff / t) * room
        else:
            t = h.sum()
            if t > 1e-14:
                h += diff * h / t
    return np.clip(h, 0.0, 1.0)

def smo_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode='full') * dx
    m = cv.max()
    return m + np.log(np.sum(np.exp(T * (cv - m)))) / T

def smo_grad(x, T, top_k=20):
    n = len(x)
    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode='full') * dx
    m = cv.max()
    e = np.exp(T * (cv - m))
    p = e / e.sum()
    idx = np.argsort(-p)[:top_k]
    pt = p[idx].copy()
    pt /= pt.sum()
    g = np.zeros(n)
    for ki, pi in zip(idx, pt):
        k = int(abs(ki - (n - 1)))
        if k == 0:
            g += pi * dx * (1.0 - 2.0 * x)
        else:
            g[k:] += pi * dx * (1.0 - x[:n-k])
            g[:n-k] -= pi * dx * x[k:]
    return g

def slsqp_opt(h0, T, n, maxiter=500):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smo_obj, h0, args=(T,),
                       jac=lambda x: smo_grad(x, T),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t > 1e-14:
                h += (diff / t) * room
        else:
            t = h.sum()
            if t > 1e-14:
                h += diff * h / t
    return np.clip(h, 0.0, 1.0)

def smo_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode='full') * dx
    m = cv.max()
    return float(m + np.log(np.sum(np.exp(T * (cv - m)))) / T)

def smo_grad(x, T, top_k=20):
    n = len(x)
    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode='full') * dx
    m = cv.max()
    e = np.exp(T * (cv - m))
    p = e / e.sum()
    idx = np.argsort(-p)[:top_k]
    pt = p[idx].copy()
    pt /= pt.sum()
    g = np.zeros(n)
    for ki, pi in zip(idx, pt):
        k = int(abs(ki - (n - 1)))
        if k == 0:
            g += pi * dx * (1.0 - 2.0 * x)
        else:
            g[k:] += pi * dx * (1.0 - x[:n-k])
            g[:n-k] -= pi * dx * x[k:]
    return g

def pen_obj(x, T, mu):
    n = len(x)
    s = x.sum() - n / 2.0
    return smo_obj(x, T) + mu * s * s

def pen_grad(x, T, mu):
    n = len(x)
    g = smo_grad(x, T)
    s = x.sum() - n / 2.0
    g += 2.0 * mu * s
    return g

def lbfgs_opt(h0, T, n, mu, maxiter=2000):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(pen_obj, h0, args=(T, mu),
                       jac=pen_grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        h = np.clip(res.x, 0, 1)
        return make_feasible(h, n)
    except Exception:
        return h0
