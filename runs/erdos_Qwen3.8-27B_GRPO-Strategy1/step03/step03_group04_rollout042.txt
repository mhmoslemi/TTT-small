def project_sum_binary(h_b, N):
    """Ensure exactly N//2 ones in binary h_b."""
    n_ones = int(h_b.sum())
    target = N // 2
    if n_ones > target:
        ones = np.where(h_b == 1)[0]
        h_b[ones[n_ones - target:]] = 0
    elif n_ones < target:
        zeros = np.where(h_b == 0)[0]
        h_b[zeros[:target - n_ones]] = 1
    return h_b

def precompute_flip_effects(h_b):
    """Precompute Δ[p] for each position p: the change in all correlations when h[p] flips 0→1."""
    N = len(h_b)
    corr_len = 2*N - 1
    # Δ[p] has the effect on corr_full[0..2N-2]
    # corr_full[k] with d = k - (N-1)
    # For d >= 1: corr[d] = sum_{n=0}^{N-1-d} h[n+d]*(1-h[n])
    #   Δ at p (0→1): +[p>=d]*(1-h[p-d]) - [p+d<=N-1]*h[p+d]
    # For d < 0, d' = -d >= 1: corr[-d'] = sum_{n=d'}^{N-1} h[n-d']*(1-h[n])
    #   Δ at p: +[p+d'<=N-1]*(1-h[p+d']) - [p>=d']*h[p-d']
    # For d=0: corr[0] = sum h[n]*(1-h[n]), no change for binary flip
    
    D = np.zeros((N, corr_len))
    center = N - 1
    for p in range(N):
        for d in range(1, N):
            # Effect on corr at lag d (index center+d):
            delta_d = 0.0
            if p >= d:

    N = len(h_b)
    center = N - 1
    D = np.zeros((N, 2*N - 1))
    one_minus_h = 1.0 - h_b
    for d in range(1, N):
        # corr at lag d (index center+d):
        # D[p, center+d] = [p>=d]*(1-h[p-d]) - [p+d<=N-1]*h[p+d]
        if d < N:
            # p ranges 0 to N-1
            # First term: p >= d means p in [d, N-1], value is 1-h[p-d]
            D[d:, center+d] += one_minus_h[:N-d]
            # Second term: p+d <= N-1 means p in [0, N-1-d], value is -h[p+d]
            D[:N-d, center+d] -= h_b[d:]

        # D[p, center-d] = [p+d<=N-1]*(1-h[p+d]) - [p>=d]*h[p-d]
        D[:N-d, center-d] += one_minus_h[d:]
        D[d:, center-d] -= h_b[:N-d]
    return D

```python
def corr_gradient(h):
    """Returns gradient of each correlation value w.r.t. h.
    Shape: (2N-1, N)
    """
    N = len(h)
    center = N - 1
    grad = np.zeros((2*N-1, N))
    one_minus_h = 1.0 - h
    
    # Lag 0: corr[0] = sum h[n]*(1-h[n]), grad = 1-2h
    grad[center, :] = 1.0 - 2.0*h
    
    # Positive lags d = 1, ..., N-1:
    for d in range(1, N):
        # corr at lag d (index center+d): sum_{n=0}^{N-1-d} h[n+d]*(1-h[n])
        # grad w.r.t. h[p]:
        #   +[p>=d]*(1-h[p-d])  (p appears as h[n+d] with n=p-d)
        #   -[p+d<=N-1]*h[p+d]  (p appears as h[n] with n=p)
        if d < N:
            grad[center+d, d:] += one_minus_h[:N-d]  # p from d to N-1
            grad[center+d, :N-d] -= h[d:]  # p from 0 to N-1-d
        
        # corr at lag -d (index center-d): sum_{n=d}^{N-1} h[n-d]*(1-h[n])
        # grad w.r.t. h[p]:
        #   +[p+d<=N-1]*(1-h[p+d])  (p appears as h[n-d] with n=p+d)
        #   -[p>=d]*h[p-d]  (p appears as h[n] with n=p)
        grad[center-d, :N-d] += one_minus_h[d:]  # p from 0 to N-1-d
        grad[center-d, d:] -= h[:N-d]  # p from d to N-1
    
    return grad
```

This is O(N²) with a loop over N lags. For N=380, it's 379 iterations, each doing O(N) vector ops. Should be fast.

import numpy as np
import time

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def corr_and_grad(h):
    N = len(h)
    center = N - 1
    one_minus_h = 1.0 - h
    corr = np.zeros(2 * N - 1)
    # Lag 0
    corr[center] = float(np.sum(h * one_minus_h))
    for d in range(1, N):
        ov = N - d
        corr[center + d] = float(np.sum(h[d:] * one_minus_h[:ov]))
        corr[center - d] = float(np.sum(h[:ov] * one_minus_h[d:]))
    return corr

def smooth_max_grad(h, beta):
    N = len(h)
    center = N - 1
    one_minus_h = 1.0 - h
    corr = np.zeros(2 * N - 1)
    corr[center] = float(np.sum(h * one_minus_h))
    for d in range(1, N):
        ov = N - d
        corr[center + d] = float(np.sum(h[d:] * one_minus_h[:ov]))
        corr[center - d] = float(np.sum(h[:ov] * one_minus_h[d:]))
    # Softmax weights
    corr_shifted = corr - corr.max()
    w = np.exp(beta * corr_shifted)
    w /= w.sum()
    # Gradient: sum_k w_k * grad_k
    grad = np.zeros(N)
    grad[:] += w[center] * (1.0 - 2.0 * h)
    for d in range(1, N):
        ov = N - d
        wd = w[center + d]
        if abs(wd) > 1e-15:
            grad[d:] += wd * one_minus_h[:ov]
            grad[:ov] -= wd * h[d:]
        wn = w[center - d]
        if abs(wn) > 1e-15:
            grad[:ov] += wn * one_minus_h[d:]
            grad[d:] -= wn * h[:ov]
    return grad, float(corr.max())

def adam_smooth(h, N, target, deadline, beta, lr, steps=None):
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1-h, mode='full')))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    n_steps = steps if steps else int((deadline - time.time()) * 80)
    n_steps = max(n_steps, 50)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        grad, peak = smooth_max_grad(h, beta)
        step = (it % 100) + 1
        if step % 100 == 0 and peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mc = m / (1 - b1 ** (it + 1))
        vc = v / (1 - b2 ** (it + 1))
        h = h - lr * mc / (np.sqrt(vc) + eps)
        h = project_sum(h, target)
    grad, peak = smooth_max_grad(best_h, beta)
    return best_h, best_peak * (2.0 / N)

def binary_flip_effects(h_b):
    N = len(h_b)
    center = N - 1
    D = np.zeros((N, 2 * N - 1))
    omh = 1.0 - h_b
    for d in range(1, N):
        ov = N - d
        D[d:, center + d] += omh[:ov]
        D[:ov, center + d] -= h_b[d:]
        D[:ov, center - d] += omh[d:]
        D[d:, center - d] -= h_b[:ov]
    return D

def binary_greedy(h_b, N, deadline, rng):
    h_b = h_b.copy().astype(float)
    n_ones = int(h_b.sum())
    if n_ones > N // 2:
        idx = np.where(h_b == 1)[0]
        rng.shuffle(idx)
        h_b[idx[n_ones - N // 2:]] = 0
    elif n_ones < N // 2:
        idx = np.where(h_b == 0)[0]
        rng.shuffle(idx)
        h_b[idx[:N // 2 - n_ones]] = 1
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    best_c5v = float(corr.max()) * 2.0 / N
    best_h = h_b.copy()
    D = binary_flip_effects(h_b)
    stall = 0
    while time.time() < deadline:
        if stall > 200:
            break
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        cur_max = float(corr.max())
        # Focus on top lags
        n_top = min(30, 2 * N - 1)
        top_idx = np.argsort(corr)[-n_top:]
        best_new = cur_max
        best_swap = None
        # Score: for swap (i,j), new_corr = corr + D[j] - D[i]
        # Want to minimize max(new_corr), especially at top lags
        score_i = D[ones, top_idx].sum(axis=1)
        score_j = D[zeros, top_idx].sum(axis=1)
        # Try top candidates
        ci = np.argsort(-score_i)[:15]
        cj = np.argsort(score_j)[:15]
        for ii in ci:
            for jj in cj:
                i, j = int(ones[ii]), int(zeros[jj])
                nc = corr + D[j] - D[i]
                nm = float(nc.max())
                if nm < best_new:
                    best_new = nm
                    best_swap = (i, j)
        # Also try some random swaps
        for _ in range(50):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i == j:
                continue
            nc = corr + D[j] - D[i]
            nm = float(nc.max())
            if nm < best_new:
                best_new = nm
                best_swap = (i, j)
        if best_swap is not None and best_new < cur_max - 1e-12:
            i, j = best_swap
            h_b[i], h_b[j] = 1 - h_b[i], 1 - h_b[j]
            corr = corr + D[j] - D[i]
            D = binary_flip_effects(h_b)
            stall = 0
            if best_new < best_c5v / (2.0 / N):
                best_c5v = float(corr.max()) * 2.0 / N
                best_h = h_b.copy()
        else:
            stall += 1
            # Perturb: try random sequence of small swaps
            n_pert = rng.randint(3, 8)
            for _ in range(n_pert):
                oi = np.where(h_b == 1)[0]
                oz = np.where(h_b == 0)[0]
                i = int(oi[rng.randint(len(oi))])
                j = int(oz[rng.randint(len(oz))])
                h_b[i], h_b[j] = 1 - h_b[i], 1 - h_b[j]
                corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
            D = binary_flip_effects(h_b)
            if float(corr.max()) > best_c5v / (2.0 / N):
                h_b = best_h.copy()
                corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
                D = binary_flip_effects(h_b)
    return best_h, best_c5v

def soft_local(h, N, deadline, rng):
    """Local search allowing values in {0, 0.3, 0.5, 0.7, 1}."""
    h = h.copy()
    vals = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    best_c5v = compute_c5(h, N)
    best_h = h.copy()
    target = N / 2.0
    while time.time() < deadline:
        improved = False
        p = rng.randint(N)
        old = h[p]
        for v in vals:
            if abs(v - old) < 1e-9:
                continue
            h_tmp = h.copy()
            h_tmp[p] = v
            h_tmp = project_sum(h_tmp, target)
            c = compute_c5(h_tmp, N)
            if c < best_c5v - 1e-10:
                h = h_tmp
                best_c5v = c
                best_h = h.copy()
                improved = True
                break
        if not improved:
            h[p] = old
    return best_h, best_c5v
