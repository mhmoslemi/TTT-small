        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        val, w, c5 = smooth_max_and_weights(corr, dx, tau)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        grad = compute_gradient(h, w)
        t += 1
        m_acc = b1 * m_acc + (1.0 - b1) * grad
        v_acc = b2 * v_acc + (1.0 - b2) * grad * grad
        m_hat = m_acc / (1.0 - b1 ** t)
        v_hat = v_acc / (1.0 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h_new, target)
        h = h_new
    c_final = compute_c5(h, dx)
    if c_final < best_c5:
        best_c5 = c_final
        best_h = h.copy()
    return best_h, best_c5

def topk_pinch(h, N, target, dx, deadline, tau0, tau1, K, lr0, lr1, max_iters):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m_acc = np.zeros(N)
    v_acc = np.zeros(N)
    b1, b2, eps = 0.85, 0.99, 1e-8
    t = 0
    while time.time() < deadline:
        frac = min(t / max(max_iters, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        val, w_all, c5 = smooth_max_and_weights(corr, dx, tau)
        if K < len(w_all):
            idx = np.argsort(w_all)[-K:]
            w = np.zeros_like(w_all)
            w[idx] = w_all[idx]
            w /= w.sum()
        else:
            w = w_all
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        grad = compute_gradient(h, w)
        t += 1
        m_acc = b1 * m_acc + (1.0 - b1) * grad
        v_acc = b2 * v_acc + (1.0 - b2) * grad * grad
        m_hat = m_acc / (1.0 - b1 ** t)
        v_hat = v_acc / (1.0 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h_new, target)
        h = h_new
    c_final = compute_c5(h, dx)
    if c_final < best_c5:
        best_c5 = c_final
        best_h = h.copy()
    return best_h, best_c5

def local_perturb(h, N, target, dx, deadline, n_max=8000, scale=0.01):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(777)
    trials = 0
    while time.time() < deadline and trials < n_max:
        trial = best_h.copy()
        n_mod = rng.randint(1, 4)
        idx = rng.choice(N, size=n_mod, replace=False)
        trial[idx] += rng.randn(n_mod) * scale
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_h = trial
        trials += 1
    return best_h, best_c5

def nudge_top_lag(h, N, target, dx, deadline, n_trials=200):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(42)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        corr = np.correlate(best_h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        center = N - 1
        m_lag = k_max - center
        trial = best_h.copy()
        n_shift = rng.randint(1, 5)
        if m_lag >= 0:
            src_range = np.arange(m_lag, min(N, m_lag + n_shift))
            dst_range = src_range + m_lag
            valid = dst_range < N
            src_range = src_range[valid]
            dst_range = dst_range[valid]
            amt = 0.02 * rng.rand(len(src_range))
            trial[src_range] -= amt
            trial[dst_range] += amt
        else:
            p = -m_lag
            dst_range = np.arange(p, min(N, p + n_shift))
            src_range = dst_range + p
            valid = src_range < N
            dst_range = dst_range[valid]
            src_range = src_range[valid]
            amt = 0.02 * rng.rand(len(src_range))
            trial[dst_range] -= amt
            trial[src_range] += amt
        if len(src_range) == 0:
            continue
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_h = trial
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)
    h_parent = np.array(initial_h_values, dtype=float)
    n_par = len(h_parent)
    x_par = np.linspace(0, 2, n_par, endpoint=False)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_par)
    sizes = [150, 100, 200, 250]
    for N in sizes:
        if time.time() > deadline - 90:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0, 2, N, endpoint=False)
        h_init = np.interp(xN, x_par, h_parent)
        h_init = project_sum(h_init, target)
        remain = deadline - time.time()
        if remain < 70:
            break
        p1_dead = time.time() + remain * 0.20
        h1, c1 = optimize_smooth(h_init.copy(), N, target, dx, p1_dead,
                                  tau0=5.0, tau1=40.0, lr0=0.04, lr1=0.008,
                                  max_iters=600)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()
        if time.time() < deadline - 50:
            p2_dead = time.time() + (deadline - time.time()) * 0.25
            h2, c2 = optimize_smooth(h1.copy(), N, target, dx, p2_dead,
                                      tau0=40.0, tau1=300.0, lr0=0.008, lr1=0.001,
                                      max_iters=600)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
        if time.time() < deadline - 30:
            p3_dead = time.time() + (deadline - time.time()) * 0.20
            K3 = min(15, 2 * N - 1)
            h3, c3 = topk_pinch(h2.copy(), N, target, dx, p3_dead,
                                 tau0=300.0, tau1=3000.0, K=K3,
                                 lr0=0.001, lr1=0.0001, max_iters=500)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()
        if time.time() < deadline - 15:
            hn, cn = nudge_top_lag(h3.copy(), N, target, dx, deadline - 8, n_trials=500)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn.copy()
        if time.time() < deadline - 10:
            hl, cl = local_perturb(h3.copy(), N, target, dx, deadline - 5,
                                    n_max=5000, scale=0.008)
            if cl < best_c5:
                best_c5 = cl
                best_h = hl.copy()
        # Try binary projection
        if time.time() < deadline - 20:
            hb = binary_project(h3, target)
            cb = compute_c5(hb, dx)
            if cb < best_c5:
                best_c5 = cb
                best_h = hb.copy()
                if time.time() < deadline - 15:
                    hbr, cbr = local_perturb(hb.copy(), N, target, dx, deadline - 5,
                                             n_max=3000, scale=0.01)
                    if cbr < best_c5:
                        best_c5 = cbr
                        best_h = hbr.copy()
    # Final ultra-fine polish at best size
    Nb = len(best_h)
    if time.time() < deadline - 4:
        dx_b = 2.0 / Nb
        Kf = min(8, 2 * Nb - 1)
        hf, cf = topk_pinch(best_h.copy(), Nb, Nb // 2, dx_b, deadline - 1,
                             tau0=2000.0, tau1=10000.0, K=Kf,
                             lr0=0.0005, lr1=0.00003, max_iters=200)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
