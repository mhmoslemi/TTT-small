def project_feasible(h, n):

def slsqp_smooth(h, T, maxiter):

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:

def nesterov_smooth(h, n_iters=2000, T=5000):

    h_cur = h.copy()
    v = np.zeros(n)

    best_val = smooth_obj(h_cur, T)
    true_best_h = h_cur.copy()
    true_best_c5, _ = compute_c5(h_cur)

    momentum = 0.75
    stall = 0
    for it in range(n_iters):

        c5_new, _ = compute_c5(h_new)

            h_cur = h_new

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = np.clip(T * (corr - m), -700, 700)
    weights = np.exp(vals)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = abs(k_idx - (n - 1))
        if k == 0:

        else:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] += w * (-h[k:])
    return dx * grad

    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            total = (1.0 - h[mask]).sum()
            if total < 1e-14:
                break
            h[mask] += (1.0 - h[mask]) * (diff / total)
        else:
            mask = h > 1e-12
            total = h[mask].sum()
            if total < 1e-14:
                break
            h[mask] += h[mask] * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = np.clip(T * (corr - m), -700, 700)
    return float((m + np.log(np.sum(np.exp(vals))) / T) * dx)

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = np.clip(T * (corr - m), -700, 700)
    weights = np.exp(vals)
    sw = weights.sum()
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw
    grad = np.zeros(n)
    center = n - 1
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = abs(k_idx - center)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] += w * (-h[k:])
    return dx * grad

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_obj, h.copy(), args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def nesterov_smooth(h, n_iters=2000, T=10000):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    v = np.zeros(n)
    lr = 0.04
    momentum = 0.75
    stall = 0
    for it in range(n_iters):
        g = smooth_grad(h_cur, T)
        g -= g.sum() / n
        v = momentum * v - lr * g
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.02, 0.6)
            stall = 0
        else:
            stall += 1
            lr *= 0.96
            if stall > 200:
                break
    return best_h, best_c5

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    return x[:-1].sum() - len(x[:-1]) / 2.0

def erdos_in(x):
    h = x[:-1]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - np.max(corr)

def slsqp_direct(h, maxiter=400):
    n = len(h)
    _, corr = compute_c5(h)
    t0 = float(np.max(corr)) + 0.1
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
    cons = [{'type': 'eq', 'fun': erdos_eq}, {'type': 'ineq', 'fun': erdos_in}]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: SLSQP warm-start chain with escalating T
    for T in [500, 2000, 10000, 50000, 200000]:
        if time.time() - t0 > budget_s * 0.14:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=600)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Nesterov with vectorized gradient
    if time.time() - t0 < budget_s * 0.30:
        remain = budget_s * 0.16 - (time.time() - t0)
        nit = max(800, int(remain * 4))
        h_new, c5_new = nesterov_smooth(best_h, n_iters=nit, T=10000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        # Second Nesterov pass with different T
        if time.time() - t0 < budget_s * 0.35:
            h_new2, c5_new2 = nesterov_smooth(best_h, n_iters=1000, T=30000)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2

    # Phase 3: Direct SLSQP epigraph
    if time.time() - t0 < budget_s * 0.42:
        h_try = slsqp_direct(best_h, maxiter=600)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 4: High-T SLSQP refinement
    for T in [100000, 300000, 500000]:
        if time.time() - t0 > budget_s * 0.52:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=800)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 5: Multi-resolution exploration
    res_list = sorted(set([best_n - 2, best_n, best_n + 2, best_n + 4, 72, 84, 90, 96]))
    for n_try in res_list:
        if time.time() - t0 > budget_s * 0.70:
            break
        if n_try < 40 or n_try % 2 != 0:
            continue
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_base = np.interp(x_new, x_old, best_h)
            h_base = project_feasible(h_base, n_try)
        n_trials = 4 if n_try == best_n else 3
        for trial in range(n_trials):
            if time.time() - t0 > budget_s * 0.70:
                break
            scale = 0.03 + 0.05 * np.random.rand()
            h_pert = h_base + np.random.randn(n_try) * scale
            h_pert = project_feasible(h_pert, n_try)
            h_opt = slsqp_smooth(h_pert, 20000, maxiter=700)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try
                h_base = h_opt.copy()

    # Phase 6: Rotation + Nesterov
    if time.time() - t0 < budget_s * 0.82:
        for shift in [1, 2, 3, 5, 8, 13, 20]:
            if time.time() - t0 > budget_s * 0.82:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_opt, c5_opt = nesterov_smooth(h_rot, n_iters=1200, T=20000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            h_final = slsqp_smooth(best_h, 100000, maxiter=500)
            c5_final, _ = compute_c5(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final

    # Phase 7: Fine local polish with tiny perturbations
    if time.time() - t0 < budget_s * 0.93:
        for scale in [0.012, 0.006, 0.003]:
            if time.time() - t0 > budget_s * 0.93:
                break
            h_p = best_h + np.random.randn(best_n) * scale
            h_p = project_feasible(h_p, best_n)
            h_opt = slsqp_smooth(h_p, 400000, maxiter=1000)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 8: Final direct + ultra-high-T SLSQP
    if time.time() - t0 < budget_s * 0.98:
        h_f = slsqp_direct(best_h, maxiter=800)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        h_s = slsqp_smooth(best_h, 600000, maxiter=600)
        c5_s, _ = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
