def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, T, topk=None):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

        corr_use = corr[idx]

        corr_use = corr
    mx = np.max(corr_use)
    val = mx + np.log(np.sum(np.exp(T * (corr_use - mx)))) / T
    return val * dx

def smooth_max_grad(h, T, n, topk=None, penalty=100.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    num_shifts = len(corr)  # = 2n-1
    
    if topk is not None and topk < num_shifts:
        m = max(topk, 1)
        idx = np.argpartition(corr, -m)[-m:]
        corr_use = corr[idx]
        shifts = idx - (n - 1)  # shift s = k - (n-1)
    else:
        corr_use = corr
        shifts = np.arange(num_shifts) - (n - 1)
    
    # Softmax weights
    mx = np.max(corr_use)
    exp_t = np.exp(T * (corr_use - mx))
    p = exp_t / exp_t.sum()
    
    sm_val = (mx + np.log(exp_t.sum()) / T) * dx
    
    # Gradient
    grad_sm = np.zeros(n)
    for i in range(len(shifts)):
        if p[i] < 1e-14:
            continue
        s = shifts[i]
        pi = p[i]
        
        # First term: p_i * (1 - h[m-s]) for valid m
        if s >= 0:
            # m in [s, n), src = m - s in [0, n-s)
            grad_sm[s:n] += pi * (1.0 - h[0:n-s])
        else:
            # m in [0, n+s), src = m - s = m + |s| in [|s|, n)
            grad_sm[0:n+s] += pi * (1.0 - h[-s:n])
        
        # Second term: -p_i * h[m+s] for valid m
        if s >= 0:
            # m in [0, n-s), dst = m + s in [s, n)
            grad_sm[0:n-s] -= pi * h[s:n]
        else:
            # m in [-s, n), dst = m + s in [0, n+s)
            grad_sm[-s:n] -= pi * h[0:n+s]
    
    # Penalty for sum constraint
    sum_dev = h.sum() - n / 2.0
    val = sm_val + penalty * sum_dev * sum_dev
    grad = grad_sm + 2.0 * penalty * sum_dev
    
    return val, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14:
            return h
        h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def _smax_val(h, T, topk):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        cu = corr[idx]
    else:
        cu = corr
    mx = np.max(cu)
    return (mx + np.log(np.sum(np.exp(T * (cu - mx)))) / T) * (2.0 / len(h))

def _smax_fg(h, T, n, topk, pen):
    corr = np.correlate(h, 1.0 - h, mode='full')
    ns = len(corr)
    if topk is not None and topk < ns:
        idx = np.argpartition(corr, -topk)[-topk:]
        cu = corr[idx]
        sh = idx.astype(int) - (n - 1)
    else:
        cu = corr
        sh = np.arange(ns) - (n - 1)
    mx = np.max(cu)
    et = np.exp(T * (cu - mx))
    p = et / et.sum()
    fval = (mx + np.log(et.sum()) / T) * (2.0 / n)
    g = np.zeros(n)
    for i in range(len(sh)):
        w = p[i]
        if w < 1e-14:
            continue
        s = sh[i]
        if s >= 0:
            g[s:] += w * (1.0 - h[:n - s])
            g[:n - s] -= w * h[s:]
        else:
            g[:n + s] += w * (1.0 - h[-s:])
            g[-s:] -= w * h[:n + s]
    sd = h.sum() - n / 2.0
    return fval + pen * sd * sd, g + 2.0 * pen * sd

def _lbfgs(h0, T, n, topk, mi, pen):
    bnd = [(0.0, 1.0)] * n
    r = minimize(_smax_fg, h0, args=(T, n, topk, pen),
                 method='L-BFGS-B', bounds=bnd, jac=True,
                 options={'maxiter': mi, 'ftol': 1e-16, 'maxls': 100,
                          'gtol': 1e-9, 'maxfun': mi * 18})
    return np.clip(r.x, 0.0, 1.0)

def _slq_obj(x, T, topk):
    return _smax_val(x, T, topk)

def _slq_eq(x, n):
    return x.sum() - n / 2.0
