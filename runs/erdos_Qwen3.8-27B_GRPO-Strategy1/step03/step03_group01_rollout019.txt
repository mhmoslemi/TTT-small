import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = float(h.sum()) - target
    if abs(diff) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, N, corr, K, use_sofmax, tau):
    nc = 2 * N - 1
    center = N - 1
    grad = np.zeros(N)
    if K <= 10 and not use_sofmax:
        idx = np.argpartition(corr, -K)[-K:]
        for ai in idx:
            L = ai - center
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += (1.0 - h[:N-L])
                grad[:N-L] -= h[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - h[M:])
                grad[M:] -= h[:N-M]
        grad /= K
    else:
        sc = tau * corr
        sc -= sc.max()
        w = np.exp(sc)
        w /= w.sum()
        thresh = 0.002
        active = np.where(w > thresh)[0]
        for ai in active:
            L = ai - center
            wi = w[ai]
            if L == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wi * (1.0 - h[:N-L])
                grad[:N-L] -= wi * h[L:]
            else:
                M = -L
                grad[:N-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N-M]
    return grad

        h = project_sum(h - lr * v, target)

        if it % perturb_every == 0 and t_frac < 0.6:
            h += perturb_mag * np.random.randn(N)
            h = project_sum(h, target)
    return best_h, best_c5
