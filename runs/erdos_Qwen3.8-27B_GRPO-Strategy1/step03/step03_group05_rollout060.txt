def project_feasible(h, n):

            best_c5, best_h = cr, hr
        hr2, cr2 = targeted_repair(best_h, best_n, top_k=5)
        if cr2 < best_c5:
            best_c5, best_h = cr2, hr2
        hc2, cc2 = coord_descent(best_h, best_n, n_sweeps=40, delta=0.002)
        if cc2 < best_c5:
            best_c5, best_h = cc2, hc2

    # Phase 5: Basin hopping with SLSQP polish
    n_trials = 50
    for trial in range(n_trials):
        if elapsed() > budget_s * 0.72:
            break
        sc = 0.025 * np.exp(-0.025 * trial) + 0.004
        hp = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        hp = slsqp_topk(hp, best_n, 3, maxiter=250, ftol=1e-14)
        hp = slsqp_topk(hp, best_n, 5, maxiter=250, ftol=1e-14)
        hg2, cg2 = gradient_descent(hp, best_n, lr0=0.005, n_steps=80)
        c5_trial = compute_c5(hg2)
        if c5_trial < best_c5:
            best_c5, best_h = c5_trial, hg2

    # Phase 6: Multi-resolution exploration
    for nr in [150, 220, 260, 300]:
        if elapsed() > budget_s * 0.86:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hu = slsqp_topk(hu, nr, 3, maxiter=300, ftol=1e-14)
        hu = slsqp_topk(hu, nr, 5, maxiter=300, ftol=1e-14)
        c5_hi = compute_c5(hu)
        x_dn = np.linspace(0, 2, best_n)
        hd = np.interp(x_dn, x_new, hu)
        hd = project_feasible(hd, best_n)
        hd = slsqp_topk(hd, best_n, 3, maxiter=200, ftol=1e-14)
        c5_dn = compute_c5(hd)
        if c5_dn < best_c5:
            best_c5, best_h = c5_dn, hd
        elif c5_hi < best_c5:
            best_c5, best_h, best_n = c5_hi, hu, nr

    # Phase 7: Final intensive polish
    if elapsed() < budget_s * 0.95:
        for k in [3, 4, 5]:
            h_new = slsqp_topk(best_h, best_n, k, maxiter=600, ftol=1e-16)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new
        hg3, cg3 = gradient_descent(best_h, best_n, lr0=0.004, n_steps=200)
        if cg3 < best_c5:
            best_c5, best_h = cg3, hg3
        hc3, cc3 = coord_descent(best_h, best_n, n_sweeps=60, delta=0.003)
        if cc3 < best_c5:
            best_c5, best_h = cc3, hc3
        h_new = slsqp_soft(best_h, best_n, 5000, maxiter=500, ftol=1e-15)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Phase 8: Micro refinement
    if elapsed() < budget_s * 0.99:
        hc4, cc4 = coord_descent(best_h, best_n, n_sweeps=80, delta=0.001)
        if cc4 < best_c5:
            best_c5, best_h = cc4, hc4
        hg4, cg4 = gradient_descent(best_h, best_n, lr0=0.002, n_steps=100)
        if cg4 < best_c5:
            best_c5, best_h = cg4, hg4

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
