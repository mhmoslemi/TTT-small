import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * (diff / tot)
    else:
        s = h.sum()
        if s > 1e-14:
            h *= target / s
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, T, active):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if active is not None and len(active) < len(corr):
        sub = corr[active]
    else:
        sub = corr
    m = sub.max()
    val = m + np.log(np.sum(np.exp(T * (sub - m)))) / T
    return val * dx

def sum_con(h):
    return h.sum() - len(h) / 2.0

def do_slsqp(h0, T, n, maxiter, ftol, active):
    cons = [{'type': 'eq', 'fun': sum_con}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(soft_max_obj, h0.copy(), args=(T, active), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-K:]
    return np.sort(idx)
