def project_sum(h, target):
    """Project h to have exact sum = target while keeping [0,1] bounds."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    m = np.max(corr)

    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1:
        T = 1
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:

        tot = room.sum()

        h = h + room * (diff / tot)

            return h

    best_x = xs[best_j]
    best_f = f[best_j]
