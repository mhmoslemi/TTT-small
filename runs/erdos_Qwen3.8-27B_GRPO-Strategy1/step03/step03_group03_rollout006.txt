def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, T):
    """Compute smooth max value and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    coeffs = corr * dx
    m = np.max(coeffs)
    ex = np.exp(T * (coeffs - m))
    Z = ex.sum()
    f_val = m + np.log(Z) / T
    w = ex / Z  # softmax weights, sum to 1
    
    # Gradient computation
    grad = np.zeros(n)
    for s in range(0, n):
        idx = n - 1 + s
        wk = w[idx]
        if wk < 1e-12:
            continue
        if s == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        else:
            # corr[n-1+s] = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            # d/dh[j] = (1-h[j+s]) for j<n-s, minus h[j-s] for j>=s
            if n - s > 0:
                grad[:n-s] += wk * dx * (1.0 - h[s:])
                grad[s:] -= wk * dx * h[:n-s]
    for s in range(1, n):
        idx = n - 1 - s
        wk = w[idx]
        if wk < 1e-12:

        if n - s > 0:
            grad[s:] += wk * dx * (1.0 - h[:n-s])
            grad[:n-s] -= wk * dx * h[s:]
    
    return f_val, grad

def pgrad_loop(h, T_start, T_end, max_iters, time_budget):
    """Projected gradient with momentum and adaptive T."""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    v = np.zeros(n)
    mom = 0.8
    lr = 0.05
    start_t = time.time()
    for it in range(max_iters):
        T = T_start * (T_end / T_start) ** (it / max(max_iters, 1))
        f_val, grad = smooth_grad(h_cur, T)
        v = mom * v - lr * grad
        h_new = project_feasible(h_cur + v, n)
        f_new, _ = smooth_grad(h_new, T)
        if f_new < f_val - 1e-12:
            h_cur = h_new
            c5_cur, _ = compute_c5(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
            lr = min(lr * 1.01, 0.3)
            mom = min(mom + 0.0005, 0.93)
        else:
            lr *= 0.7
            if lr < 1e-7:
                break
        if time.time() - start_t > time_budget:
            break
    return best_h, best_c5

def slsqp_polish(h, T, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        dx2 = 2.0 / n
        coeffs = c * dx2
        m = np.max(coeffs)
        return m + np.log(np.sum(np.exp(T * (coeffs - m)))) / T
    
    def obj_g(x):
        _, g = smooth_grad(x, T)
        return g
    
    def eq(x):
        return x.sum() - n / 2.0
    
    def eq_g(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_g}]
    try:
        res = minimize(obj, h, jac=obj_g, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'eps': 1e-7})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
