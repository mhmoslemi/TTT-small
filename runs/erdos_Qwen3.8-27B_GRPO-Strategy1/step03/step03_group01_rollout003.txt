    # Also try from perturbed start
    return best_h, best_c5

def polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for _pass in range(8):
        if time.time() > deadline:
            break
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = best_h[i]
            trials = (0.0, 1.0, orig + 0.005, orig - 0.005, orig + 0.02, orig - 0.02, orig + 0.08, orig - 0.08)
            for trial in trials:
                if abs(trial - orig) < 1e-10 or trial < -0.01 or trial > 1.01:
                    continue
                test = best_h.copy()
                test[i] = trial
                test = project_sum(test, target)
                c5 = compute_c5(test, dx)
                if c5 < best_c5 - 5e-12:
                    best_c5 = c5
                    best_h = test
                    improved = True
                    break
        if not improved and _pass > 2:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_par)
    x_par = np.linspace(0.0, 2.0, n_par, endpoint=False)
    # Sizes to try
    sizes = [n_par, 400, 462]
    sizes = [s for s in sizes if s != n_par]
    sizes.insert(0, n_par)
    remaining = deadline - time.time() - 40
    per_size = remaining / len(sizes)
    for N in sizes:
        size_dead = time.time() + per_size
        if size_dead > deadline - 30:
            size_dead = deadline - 30
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_base = np.interp(xN, x_par, best_h)
        h_base = project_sum(h_base, target)
        c5_base = compute_c5(h_base, dx)
        # Multiple starting points
        starts = []
        starts.append(h_base.copy())
        starts.append(project_sum(h_base + rng.randn(N) * 0.04, target))
        starts.append(project_sum(h_base + rng.randn(N) * 0.08, target))
        n_starts = min(len(starts), max(1, int((size_dead - time.time()) / 45)))
        for st in range(n_starts):
            if time.time() > size_dead - 15:
                break
            h_cur = starts[st].copy()
            t_rem = size_dead - time.time() - 5
            if t_rem < 30:
                break
            # Phase 1: broad exploration
            d1 = time.time() + t_rem * 0.35
            h_cur, c1 = optimize_phase(h_cur, N, target, d1, 5.0, 60.0, 40, 12, 0.05, 0.95, 40, 0.85, rng, 150)
            # Phase 2: medium focus
            if time.time() < size_dead - 10:
                d2 = time.time() + (size_dead - time.time() - 5) * 0.35
                h_cur, c2 = optimize_phase(h_cur, N, target, d2, 60.0, 400.0, 12, 4, 0.015, 0.97, 60, 0.88, rng, 200)
            # Phase 3: sharp focus
            if time.time() < size_dead - 5:
                d3 = size_dead - 2
                h_cur, c3 = optimize_phase(h_cur, N, target, d3, 400.0, 2000.0, 4, 2, 0.004, 0.985, 80, 0.90, rng, 300)
            # Fourier refinement
            if time.time() < size_dead - 3:
                h_ft, c_ft = fourier_refine(h_cur, N, target, max(time.time(), size_dead - 2), rng)
                c3 = min(c3, c_ft)
            c_cur = compute_c5(h_cur, dx)
            c_ft_check = compute_c5(h_ft if time.time() < size_dead else h_cur, dx)
            if c_cur < c5_base:
                c5_base = c_cur
                h_base = h_cur.copy()
            if c_ft_check < c5_base:
                c5_base = c_ft_check
                h_base = h_ft.copy()
        if c5_base < best_c5:
            best_c5 = c5_base
            best_h = h_base.copy()
    # Final fine-tuning
    if time.time() < deadline - 15:
        Nb = len(best_h)
        h_fin, c_fin = optimize_phase(best_h.copy(), Nb, Nb / 2.0, deadline - 10, 500.0, 3000.0, 3, 1, 0.002, 0.99, 100, 0.92, rng, 400)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin
    # Polish
    if time.time() < deadline - 4:
        Nb = len(best_h)
        best_h, best_c5 = polish(best_h, Nb, Nb / 2.0, deadline - 1)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
