    """Power-based smooth max"""
    m = np.max(corr)
    if m <= 0:
        return 0.0
    return m * (np.sum((corr / m) ** p)) ** (1.0 / p)

    """Log-sum-exp smooth max"""
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-13:
            return h
        h = h - h * (-diff / tot)
    return np.clip(h, 0.0, 1.0)

def sum_con(x):
    return x.sum() - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

def obj_lse(x, T, k):
    n = len(x)
    h = np.clip(x, 0, 1)
    s = h.sum()
    diff = n / 2.0 - s
    if diff > 1e-10:
        room = 1.0 - h
        h = h + room * (diff / max(room.sum(), 1e-15))
    elif diff < -1e-10:
        h = h - h * (-diff / max(h.sum(), 1e-15))
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    if k is not None and k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        c = corr[idx]
    else:
        c = corr
    m = c.max()
    if m <= 1e-15:
        return 0.0
    return (m + np.log(np.sum(np.exp(T * (c - m) + 1e-30))) / T) * dx

def do_slsqp(h0, T, k, maxit):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        r = minimize(obj_lse, h0, args=(T, k), method='SLSQP',
                    bounds=bnds, constraints=cons,
                    options={'maxiter': maxit, 'ftol': 1e-18})
        return project_h(np.clip(r.x, 0, 1), n)
    except Exception:
        return h0.copy()

def fine_tune(h, n, nits):
    best = h.copy()
    bc = compute_c5(best)
    dlist = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.02, -0.02, 0.008, -0.008]
    for it in range(nits):
        corr = np.correlate(best, 1.0 - best, mode='full')
        top = np.argsort(corr)[-4:]
        found = False
        for lag in top:
            shift = lag - (n - 1)
            if shift >= 0:
                rng = range(0, n - shift)
            else:
                s2 = -shift
                rng = range(s2, n)
            for i in rng:
                vi = best[i]
                if vi < 0.03 or vi > 0.97:
                    continue
                for d in dlist:
                    ht = best.copy()
                    ht[i] = min(1.0, max(0.0, vi + d))
                    ht = project_h(ht, n)
                    ct = compute_c5(ht)
                    if ct < bc - 1e-15:
                        best, bc = ht, ct
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            break
    return best, bc

def pair_tune(h, n, nits):
    best = h.copy()
    bc = compute_c5(best)
    for it in range(nits):
        idx1 = np.where((best > 0.93) & (best < 0.999))[0]
        idx0 = np.where((best > 0.001) & (best < 0.07))[0]
        if len(idx1) == 0 or len(idx0) == 0:
            break
        improved = False
        for i in idx1[:15]:
            for j in idx0[:15]:
                for amt in [0.01, 0.005, 0.002]:
                    ht = best.copy()
                    ht[i] = max(0.0, ht[i] - amt)
                    ht[j] = min(1.0, ht[j] + amt)
                    ht = project_h(ht, n)
                    ct = compute_c5(ht)
                    if ct < bc - 1e-15:
                        best, bc = ht, ct
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best, bc

def try_interval_start(n, K, rng):
    widths = rng.dirichlet(np.ones(K) * 2.0)
    centers = np.sort(rng.uniform(0.3, 1.7, K))
    h = np.zeros(n)
    x = np.linspace(0, 2, n, endpoint=False)
    for i in range(K):
        c, w = centers[i], widths[i]
        h[(x >= c - w/2) & (x < c + w/2)] = 1.0
    return project_h(h, n)
