    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):

            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def softmax_c5(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_c5(x, k, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.mean(top) * dx)

def compute_sens(h, i, n):
    sens = np.zeros(2 * n - 1)
    m_a = np.arange(i - n + 1, i + 1)
    n_a = i - m_a
    idx_a = n - 1 + m_a
    sens[idx_a] += (1.0 - h[n_a])
    m_b = np.arange(-i, n - i)
    n_b = i + m_b
    idx_b = n - 1 + m_b
    sens[idx_b] += (-h[n_b])
    return sens

def coord_descent(h, n, n_passes=20, n_grid=41):
    best = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(best, 1.0 - best, mode='full')
    cur_max = np.max(corr)
    for it in range(n_passes):
        changed = False
        for i in range(n):
            old = best[i]
            sens = compute_sens(best, i, n)
            v_grid = np.linspace(0.0, 1.0, n_grid)
            best_v, best_t = old, cur_max
            for v in v_grid:
                d = v - old
                if abs(d) > 1e-15:
                    t = np.max(corr + d * sens)
                else:
                    t = cur_max
                if t < best_t - 1e-13:
                    best_t = t
                    best_v = v
            if best_v != old:
                span = 1.0 / (n_grid - 1)
                lo = max(0.0, best_v - span)
                hi = min(1.0, best_v + span)
                v_fine = np.linspace(lo, hi, 21)
                for v in v_fine:
                    d = v - old
                    if abs(d) > 1e-15:
                        t = np.max(corr + d * sens)
                    else:
                        t = cur_max
                    if t < best_t - 1e-13:
                        best_t = t
                        best_v = v
                if best_v != old:
                    best[i] = best_v
                    corr = np.correlate(best, 1.0 - best, mode='full')
                    cur_max = np.max(corr)
                    changed = True
        best = project_feasible(best, n)
        corr = np.correlate(best, 1.0 - best, mode='full')
        cur_max = np.max(corr)
        c5_new = compute_c5(best)
        if c5_new < best_c5:
            best_c5 = c5_new
        if not changed:
            break
    best = project_feasible(best, n)
    return best, compute_c5(best)

def slsqp_opt(h, obj, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def binary_round_and_fix(h, n, threshold=0.3):
    """Round values close to 0 or 1, then redistribute."""
    h2 = h.copy()
    mask_lo = h2 < threshold
    mask_hi = h2 > 1.0 - threshold
    h2[mask_lo] = 0.0
    h2[mask_hi] = 1.0
    return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive coordinate descent
    if elapsed() < budget_s * 0.12:
        best_h, best_c5 = coord_descent(best_h, best_n, n_passes=40, n_grid=51)

    # Phase 2: SLSQP with top-k and softmax objectives
    if elapsed() < budget_s * 0.30:
        for k in [3, 5, 8, 12]:
            if elapsed() > budget_s * 0.28:
                break
            h2 = slsqp_opt(best_h, topk_c5, (k, best_n), 400, 1e-13)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5, best_h = c2, h2
            h2, c2 = coord_descent(h2, best_n, n_passes=20, n_grid=41)
            if c2 < best_c5:
                best_c5, best_h = c2, h2
    if elapsed() < budget_s * 0.38:
        for T in [200, 1000, 5000, 20000]:
            if elapsed() > budget_s * 0.38:
                break
            h2 = slsqp_opt(best_h, softmax_c5, (T, best_n), 350, 1e-13)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5, best_h = c2, h2
            h2, c2 = coord_descent(h2, best_n, n_passes=15, n_grid=41)
            if c2 < best_c5:
                best_c5, best_h = c2, h2

    # Phase 3: Basin hopping with structured perturbations
    n_basins = 20
    for trial in range(n_basins):
        if elapsed() > budget_s * 0.55:
            break
        r = np.random.random()
        scale = 0.035 * (1.0 - 0.015 * trial) + 0.008
        if r < 0.4:
            hp = project_feasible(best_h + np.random.randn(best_n) * scale, best_n)
        elif r < 0.7:
            hp = binary_round_and_fix(best_h, best_n, threshold=0.25 + 0.1 * np.random.random())
            hp = project_feasible(hp + np.random.randn(best_n) * scale * 0.5, best_n)
        else:
            i1, i2 = np.random.choice(best_n, 2, replace=False)
            lo = min(i1, i2)
            hi = max(i1, i2)
            hp = best_h.copy()
            hp[lo:hi+1] = np.random.permutation(hp[lo:hi+1])
            hp = project_feasible(hp, best_n)
        hp = slsqp_opt(hp, topk_c5, (5, best_n), 120, 1e-12)
        hp, c_hp = coord_descent(hp, best_n, n_passes=12, n_grid=41)
        if c_hp < best_c5:
            best_c5, best_h = c_hp, hp

    # Phase 4: Multi-resolution exploration
    for nr in [150, 180, 220, 250, 300]:
        if elapsed() > budget_s * 0.78:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hu = project_feasible(hu + np.random.randn(nr) * 0.01, nr)
        hh = slsqp_opt(hu, topk_c5, (5, nr), 200, 1e-12)
        hh, c_hi = coord_descent(hh, nr, n_passes=15, n_grid=41)
        x_dn = np.linspace(0, 2, best_n)
        hd = np.interp(x_dn, x_new, hh)
        hd = project_feasible(hd, best_n)
        hd, c_dn = coord_descent(hd, best_n, n_passes=10, n_grid=41)
        if c_dn < best_c5:
            best_c5, best_h = c_dn, hd
        elif c_hi < best_c5:
            best_c5, best_h, best_n = c_hi, hh, nr

    # Phase 5: Intensive local polish loop
    if elapsed() < budget_s * 0.88:
        for rnd in range(4):
            if elapsed() > budget_s * 0.88:
                break
            best_h, best_c5 = coord_descent(best_h, best_n, n_passes=50, n_grid=61)
            h2 = slsqp_opt(best_h, topk_c5, (3, best_n), 500, 1e-14)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5, best_h = c2, h2
            best_h, best_c5 = coord_descent(best_h, best_n, n_passes=30, n_grid=61)
            h2 = slsqp_opt(best_h, softmax_c5, (10000, best_n), 400, 1e-14)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5, best_h = c2, h2
            best_h, best_c5 = coord_descent(best_h, best_n, n_passes=20, n_grid=61)

    # Phase 6: Final ultra-fine polish
    if elapsed() < budget_s * 0.96:
        h2 = slsqp_opt(best_h, topk_c5, (2, best_n), 600, 1e-15)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
        best_h, best_c5 = coord_descent(best_h, best_n, n_passes=50, n_grid=81)
        h2 = slsqp_opt(best_h, softmax_c5, (50000, best_n), 500, 1e-15)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
        best_h, best_c5 = coord_descent(best_h, best_n, n_passes=30, n_grid=81)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
