def sum_con(x):

def slsqp_opt(h, obj_func, args, maxiter, ftol):
    n = len(h)

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n)}]
    try:

                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})

        return project_feasible(h_new, n)
    except Exception:
        return h

    if time.time() - t0 < budget_s * 0.82:
        for _ in range(8):
            if time.time() - t0 > budget_s * 0.82:
                break
            hp = slsqp_opt(best_h, topk_obj, (0, 2, best_n), 500, 1e-15)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_c5, best_h = cp, hp
            # Subgradient micro-steps
            hs, cs = subgrad_ls(best_h, 150, 2, 0.005, 0.9995)
            if cs < best_c5:
                best_c5, best_h = cs, hs
    
    # Phase 6: Final coordinate descent refinement
    if time.time() - t0 < budget_s * 0.90:
        hc, cc = coordinate_descent(best_h, 40, 3)
        if cc < best_c5:
            best_c5, best_h = cc, hc
        hf = slsqp_opt(best_h, topk_obj, (0, 1, best_n), 600, 1e-15)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    
    # Phase 7: Very high T smooth max polish
    if time.time() - t0 < budget_s * 0.96:
        hf2 = slsqp_opt(best_h, smooth_max_obj, (0, 50000, best_n), 500, 1e-15)
        cf2 = compute_c5(hf2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, hf2
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
