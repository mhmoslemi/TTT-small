import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def minmax_1d(A, B):
    """Find t in [0,1] minimizing max_m(A[m] + B[m]*t)."""
    # Check endpoints
    f0 = np.max(A)
    f1 = np.max(A + B)
    # Bisection on convex function
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = A + B * mid
        fmax = np.max(vals)
        tol = 1e-14
        active = vals >= (fmax - tol)
        if active.any():
            b_min = np.min(B[active])
            b_max = np.max(B[active])
            if b_min > 1e-15:
                hi = mid
            elif b_max < -1e-15:
                lo = mid
            else:
                break
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, max_sweeps=200, tol=1e-13):
    """Exact coordinate descent for the min-max overlap problem."""
    best = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            old_j = h[j]
            # Compute B[m] for all m
            B = np.zeros(2 * n - 1, dtype=np.float64)
            shifts = np.arange(-(n-1), n)
            jm = j + shifts  # j + shift for each m
            js = j - shifts  # j - shift for each m
            valid_m = (jm >= 0) & (jm < n)
            valid_s = (js >= 0) & (js < n)
            B[valid_m] += (1.0 - h[jm[valid_m]])
            B[valid_s] -= h[js[valid_s]]
            # Current correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            A = corr - old_j * B
            # Solve 1D min-max
            t_star = minmax_1d(A, B)
            h[j] = t_star
            change = abs(t_star - old_j)
            if change > max_change:
                max_change = change
        # Enforce constraint
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        if max_change < tol:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 200
        best_h = np.zeros(n_init)
        best_h[:n_init//2] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent on initial solution
    t_end = t0 + budget_s * 0.25
    while time.time() < t_end:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=50)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new
        else:
            break
    
    # Phase 2: Try different resolutions
    resolutions = [150, 200, 250, 300, 400]
    for nr in resolutions:
        if time.time() > t0 + budget_s * 0.70:
            break
        if nr == best_n:
            continue
        # Interpolate
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, nr)
        # Coordinate descent
        h_c, c5_c = coord_descent(h_new, nr, max_sweeps=80)
        # Also try with perturbation
        h_p = project_feasible(h_new + np.random.randn(nr) * 0.03, nr)
        h_cp, c5_cp = coord_descent(h_p, nr, max_sweeps=50)
        c5_new = min(c5_c, c5_cp)
        h_best = h_c if c5_c <= c5_cp else h_cp
        if c5_new < best_c5:
            best_c5, best_h, best_n = c5_new, h_best, nr
    
    # Phase 3: SLSQP polish
    if time.time() < t0 + budget_s * 0.85:
        def objective(h, n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            dx = 2.0 / n
            vals = corr * dx
            m = np.max(vals)
            T = 500.0
            return m + np.log(np.sum(np.exp(T * (vals - m)))) / T
        
        def sum_con(x):
            return float(x.sum() - len(x) / 2.0)
        
        bnds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': sum_con}]
        try:
            res = minimize(objective, best_h, args=(best_n,), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-14})
            h_s = project_feasible(np.clip(res.x, 0, 1), best_n)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s
        except Exception:
            pass
    
    # Phase 4: Final coordinate descent polish
    if time.time() < t0 + budget_s * 0.95:
        h_f, c5_f = coord_descent(best_h, best_n, max_sweeps=200, tol=1e-14)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def minmax_1d(A, B):
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        vals = A + B * mid
        fmax = np.max(vals)
        active = vals >= (fmax - 1e-14)
        b_min = np.min(B[active])
        b_max = np.max(B[active])
        if b_min > 1e-15:
            hi = mid
        elif b_max < -1e-15:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, max_sweeps=200, tol=1e-13, seed=0):
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_c5 = compute_c5(h)
    shifts = np.arange(-(n - 1), n)
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if sweep > 0 and sweep % 5 == 0:
            order = rng.permutation(n)
        max_change = 0.0
        for j in order:
            old_j = h[j]
            jm = j - shifts
            jp = j + shifts
            B = np.zeros(2 * n - 1)
            vm = (jm >= 0) & (jm < n)
            vp = (jp >= 0) & (jp < n)
            B[vm] += (1.0 - h[jm[vm]])
            B[vp] -= h[jp[vp]]
            corr = np.correlate(h, 1.0 - h, mode='full')
            A = corr - old_j * B
            t_star = minmax_1d(A, B)
            h[j] = t_star
            ch = abs(t_star - old_j)
            if ch > max_change:
                max_change = ch
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        if max_change < tol:
            break
    return best, best_c5

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqq_polish(h, n, maxiter=300):
    def obj(x):
        return softmax_obj(x, 800.0, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 200
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent on initial solution
    t_phase1 = t0 + budget_s * 0.20
    while time.time() < t_phase1:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=80, tol=1e-13, seed=seed)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new
        else:
            break

    # Phase 2: Multi-resolution exploration
    t_phase2 = t0 + budget_s * 0.60
    resolutions = [120, 150, 200, 250, 300, 350, 400]
    for nr in resolutions:
        if time.time() > t_phase2:
            break
        if nr == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, nr)
        h_c, c5_c = coord_descent(h_try, nr, max_sweeps=60, seed=seed)
        # Perturbed start
        h_p = project_feasible(h_try + np.random.randn(nr) * 0.04, nr)
        h_cp, c5_cp = coord_descent(h_p, nr, max_sweeps=40, seed=seed + 1)
        c5_best_r = min(c5_c, c5_cp)
        h_best_r = h_c if c5_c <= c5_cp else h_cp
        if c5_best_r < best_c5:
            best_c5, best_h, best_n = c5_best_r, h_best_r, nr
        # Try downsampled version too
        x_dn = np.linspace(0, 2, best_n)
        h_d = np.interp(x_dn, x_new, h_best_r)
        h_d = project_feasible(h_d, best_n)
        h_dd, c5_dd = coord_descent(h_d, best_n, max_sweeps=40, seed=seed + 2)
        if c5_dd < best_c5:
            best_c5, best_h = c5_dd, h_dd

    # Phase 3: SLSQP refinement
    t_phase3 = t0 + budget_s * 0.75
    if time.time() < t_phase3:
        h_s = slsqq_polish(best_h, best_n, 400)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s

    # Phase 4: More coordinate descent rounds with random restarts
    t_phase4 = t0 + budget_s * 0.90
    for restart in range(5):
        if time.time() > t_phase4:
            break
        h_r = best_h.copy()
        if restart > 0:
            scale = 0.02 * (restart - 0)
            h_r = project_feasible(h_r + np.random.randn(best_n) * scale, best_n)
        h_r2, c5_r = coord_descent(h_r, best_n, max_sweeps=60, tol=1e-13, seed=seed + 10 + restart)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r2

    # Phase 5: Final intensive coordinate descent
    if time.time() < t0 + budget_s * 0.97:
        h_f, c5_f = coord_descent(best_h, best_n, max_sweeps=300, tol=1e-15, seed=seed + 99)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        h_s2 = slsqq_polish(best_h, best_n, 200)
        c5_s2 = compute_c5(h_s2)
        if c5_s2 < best_c5:
            best_c5, best_h = c5_s2, h_s2
        h_f2, c5_f2 = coord_descent(best_h, best_n, max_sweeps=100, tol=1e-15, seed=seed + 99)
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_f2

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
