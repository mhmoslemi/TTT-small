def compute_grad(h, n):
    """Compute gradient matrix G of shape (2n-1, n) where G[p,i] = partial corr[p]/partial h[i]."""
    G = np.zeros((2*n-1, n))
    # k=0 row (p = n-1)
    G[n-1, :] = 1.0 - 2.0*h
    # k≠0
    for k in range(-(n-1), n):
        if k == 0: continue
        p = n-1+k
        for i in range(n):
            g = 0.0
            j1 = i + k
            if 0 <= j1 < n:
                g += (1.0 - h[j1])
            j2 = i - k
            if 0 <= j2 < n:
                g -= h[j2]
            G[p, i] = g
    return G

def compute_corr(h):

    for _ in range(30):

            break

        else:

            h = np.clip(h * (1.0 + diff / s), 0, 1)
    return h

def eval_1d(v, cb, cf, n):
    """C5 when h[i]=v, given corr_base (h[i]=0) and coeff vector."""
    c = cb + v * cf
    c[n - 1] = cb[n - 1] + v - v * v
    return np.max(c) * (2.0 / n)

    ov = h[i]
    if ov < 1e-15 or ov > 1.0 - 1e-15:
        # Still try to optimize
        pass
    h[i] = 0.0
    cb = compute_corr(h)

    # Compute coefficients

        j1 = i + k

        if 0 <= j1 < n:  # i+k
            c_val -= h[j1]
        cf[p] = c_val

    res = minimize_scalar(eval_1d, args=(cb, cf, n), bounds=(0.0, 1.0),
