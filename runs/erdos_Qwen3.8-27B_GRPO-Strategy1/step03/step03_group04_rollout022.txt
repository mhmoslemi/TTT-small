def compute_corr(h):

def compute_c5(h, N):

    if abs(s - target) < 1e-12:

    peak = float(np.max(corr))

            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

def adam_optimize(h, N, target, deadline, k_start, k_end, lr0):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    total_iters = max(int((deadline - time.time()) * 8), 100)  # estimate
    while time.time() < deadline:
        k = max(k_end, int(k_start * (k_end / k_start) ** (it / total_iters)))
        grad, peak = grad_topk(h, N, k)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        # Cosine decay
        t = min(it / total_iters, 1.0)
        lr = lr0 * 0.5 * (1 + np.cos(np.pi * t))
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_peak * (2.0 / N)

        elif n_ones < N // 2:

            best_max = local_best_max
            best_h = local_best.copy()

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    if k >= len(corr):
        top = np.arange(len(corr))
    else:
        top = np.argpartition(corr, -k)[-k:]
    center = N - 1
    grad = np.zeros(N)
    wk = 1.0 / k
    for ai in top:
        L = ai - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N - L])
            grad[:N - L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N - M]
    return grad, peak

        idx = np.where(h_b == 1)[0]
        # Remove the ones with lowest h values
        order = np.argsort(-h[idx])
        h_b[idx[order[N // 2:]]] = 0
    elif n_ones < N // 2:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        h_b[idx[order[:N // 2 - n_ones]]] = 1
    return h_b
