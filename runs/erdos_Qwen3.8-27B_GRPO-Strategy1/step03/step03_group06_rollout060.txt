def project(h, n):
    h = np.clip(h, 0, 1)
    target = n/2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Need to add mass; add to positions with most room
        room = 1.0 - h
        total = room.sum()
        if total < 1e-15: return h
        h = h + room * (diff/total)
    else:
        # Need to remove mass
        total = h.sum()
        if total < 1e-15: return h
        h = h + h * (diff/total)
    return np.clip(h, 0, 1)

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        total = h.sum()
        if total < 1e-13:
            return h
        h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)

    n = len(x)

    corr = np.correlate(x, 1.0 - x, mode='full')

            pert[i1] = step * 0.8
            pert[i2] = -step * 0.8
        h_pert = project_feasible(bh_h + pert, len(bh_h))
        T_k = np.random.choice([1000, 5000, 10000])
        h_new = do_slsqp(h_pert, len(bh_h), T_k, None, 250, 1e-14)
        c5n = compute_c5(h_new)
        temp = 0.003 * (1.0 - it / n_bh) + 0.0001
        if c5n < bh_c5 or (c5n > bh_c5 and np.random.rand() < np.exp(-(c5n - bh_c5) / temp)):
            bh_c5 = c5n
            bh_h = h_new
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Multi-resolution
    for n_try in [100, 120, 140, 200, 240]:
        if time.time() - t0 > budget_s * 0.88:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n_try)
        for T in [500, 5000]:
            if time.time() - t0 > budget_s * 0.90:
                break
            h_new = do_slsqp(h_int, n_try, T, None, 400, 1e-14)
            c5n = compute_c5(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new
                h_int = h_new

    # Phase 6: Coordinate descent polish
    if time.time() - t0 < budget_s * 0.93:
        best_h, best_c5 = coord_polish(best_h, len(best_h), 80)

    # Phase 7: Final SLSQP polish at very high T
    n_cur = len(best_h)
    for T in [20000, 80000, 200000]:
        if time.time() - t0 > budget_s * 0.975:
            break
        h_new = do_slsqp(best_h, n_cur, T, None, 800, 1e-16)
        c5n = compute_c5(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new
    for k in [3, 5]:
        if time.time() - t0 > budget_s * 0.99:
            break
        h_new = do_slsqp(best_h, n_cur, 50000, k, 600, 1e-16)
        c5n = compute_c5(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Final subgradient + coord polish
    if time.time() - t0 < budget_s * 0.995:
        best_h = subgrad_steps(best_h, len(best_h), 500)
        best_h, best_c5 = coord_polish(best_h, len(best_h), 30)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
