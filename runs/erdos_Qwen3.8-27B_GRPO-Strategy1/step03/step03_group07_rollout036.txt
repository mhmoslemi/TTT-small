    part2 = np.correlate(w, h, mode='valid')[::-1]

    obj = np.dot(w, corr) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h = h + room * (diff / total)
    else:
        total = s
        if total > 1e-14:
            h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    cmax = np.max(corr)
    ev = np.exp(T * (corr - cmax))
    w = ev / (ev.sum() + 1e-30)
    part1 = np.correlate(w, omh[::-1], mode='valid')
    part2 = np.correlate(w, h, mode='valid')[::-1]
    grad = dx * (part1 - part2)
    obj = np.dot(w, corr) * dx
    return obj, grad

def proj_grad_desc(h0, T, n_steps, lr, momentum=0.9):
    n = len(h0)
    h = h0.copy().astype(float)
    v = np.zeros(n)
    best_obj, _ = softmax_obj_grad(h, T)
    patience = 0
    for step in range(n_steps):
        obj, grad = softmax_obj_grad(h, T)
        v = momentum * v - lr * grad
        h_new = np.clip(h + v, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        h = h_new
        if obj < best_obj - 1e-10:
            best_obj = obj
            patience = 0
        else:
            patience += 1
        if patience > 200:
            break
    return h

    np.random.seed(seed)
    t0 = time.time()
    
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        best_h = np.zeros(120)
        best_h[0::2] = 1.0
        best_h = make_feasible(best_h, 120)
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Periodic search (12% budget)
    t_p1 = budget_s * 0.12
    n_base = max(len(best_h), 200)
    
    for p in range(3, 41):
        if time.time() - t0 > t_p1:
            break
        m = n_base // p
        if m < 2:
            continue
        n_p = p * m
        for init in range(3):
            if time.time() - t0 > t_p1:
                break
            if init == 0:
                xp = np.clip(0.5 + 0.45 * np.cos(2*np.pi*np.arange(p)/p), 0.01, 0.99)
            elif init == 1:
                xp = np.resize(best_h, p).astype(float)
                xp = np.clip(xp * (p / max(xp.sum(), 1e-10)) * 0.5, 0.01, 0.99)
            else:
                xp = np.random.uniform(0.1, 0.9, p)
                xp = np.clip(xp * (p / max(xp.sum(), 1e-10)) * 0.5, 0.01, 0.99)
            
            def pobj(xp_in):
                ht = make_feasible(np.tile(xp_in, m), n_p)
                return compute_c5(ht)
            
            try:
                res = minimize(pobj, xp.copy(), method='Nelder-Mead',
                             options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11})
                xp_opt = np.clip(res.x, 0.0, 1.0)
                ht = make_feasible(np.tile(xp_opt, m), n_p)
                c5v = compute_c5(ht)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = ht
            except Exception:
                pass
    
    # Phase 2: Gradient descent with increasing T (38% budget)
    t_p2 = budget_s * 0.50
    n_cur = len(best_h)
    Ts = [30, 100, 500, 2000, 10000]
    for T in Ts:
        if time.time() - t0 > t_p2:
            break
        lr = 0.8 / (T + 20)
        n_steps = min(5000, int((t_p2 - (time.time()-t0)) * n_cur / (n_cur * T * 0.001 + 1)))
        n_steps = max(n_steps, 500)
        try:

            c5v = compute_c5(h_new)

                best_h = h_new
        except Exception:

        # Quick verify

    # Phase 3: Basin hopping (30% budget)
    t_p3 = budget_s * 0.80
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    it = 0

        ptype = it % 5

        step_sz = 0.12 * (1.0 - frac) + 0.03

        elif ptype == 1:
            hp = bh_h.copy()
            bl = max(2, n_bh // 8)
            s1 = np.random.randint(0, n_bh - bl)
            s2 = np.random.randint(0, n_bh - bl)
            tmp = hp[s1:s1+bl].copy()
            hp[s1:s1+bl] = hp[s2:s2+bl]
            hp[s2:s2+bl] = tmp
        elif ptype == 2:

            idx = np.random.choice(n_bh, nf, replace=False)
            hp[idx] = 1.0 - hp[idx]
        elif ptype == 3:
            pr = np.random.choice([4, 5, 6, 8, 10, 12, 15])
            mr = n_bh // pr
            xpr = np.random.uniform(0.15, 0.85, pr)
            xpr = xpr * (pr / max(xpr.sum(), 1e-10)) * 0.5
            hp = 0.4 * bh_h[:mr*pr] + 0.6 * np.tile(xpr, mr)
        else:
            hp = bh_h + 0.25 * np.random.randn(n_bh)
        
        hp = make_feasible(hp, n_bh)
        T_use = 300 + 500 * (it // 25)
        lr_use = 0.5 / (T_use + 20)
        try:
            h_new = proj_grad_desc(hp, T_use, 400, lr_use, momentum=0.85)
            c5v = compute_c5(h_new)
        except Exception:
            h_new = hp
            c5v = compute_c5(hp)
        
        temp = 0.003 * (1.0 - frac) + 0.0002
        if c5v < bh_c5 or (c5v - bh_c5 > -1e-12 and np.random.rand() < np.exp(-(c5v - bh_c5) / max(temp, 1e-10))):
            bh_c5 = c5v
            bh_h = h_new.copy()
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new.copy()
    
    # Phase 4: Multi-resolution (8% budget)
    t_p4 = budget_s * 0.93
    for n_try in [100, 150, 200, 250]:
        if time.time() - t0 > t_p4:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = make_feasible(hi, n_try)
        T_use = 2000
        lr_use = 0.4 / (T_use + 20)
        try:
            h_new = proj_grad_desc(hi, T_use, 3000, lr_use, momentum=0.9)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass
    
    # Phase 5: Final polish (7% budget)
    if time.time() - t0 < budget_s * 0.99:
        n_cur = len(best_h)
        for T in [50000, 200000]:
            if time.time() - t0 > budget_s * 0.99:

                    best_h = h_new

                pass
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h = h + room * (diff / total)
    else:
        total = s
        if total > 1e-14:
            h = h + h * (diff / total)
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    cmax = np.max(corr)
    ev = np.exp(T * (corr - cmax))
    w = ev / (ev.sum() + 1e-30)
    part1 = np.correlate(w, omh[::-1], mode='valid')
    part2 = np.correlate(w, h, mode='valid')[::-1]
    grad = dx * (part1 - part2)
    obj = float(np.dot(w, corr) * dx)
    return obj, grad

def proj_grad_desc(h0, T, n_steps, lr, momentum=0.9):
    n = len(h0)
    h = h0.copy().astype(float)
    v = np.zeros(n)
    for step in range(n_steps):
        obj, grad = softmax_obj_grad(h, T)
        v = momentum * v - lr * grad
        h = np.clip(h + v, 0.0, 1.0)
        h = make_feasible(h, n)
    return h

    if ihv is not None:

    else:

        best_h[0::2] = 1.0
        best_h = make_feasible(best_h, 120)

    best_c5 = compute_c5(best_h)

    # Phase 1: Periodic pattern search (10% budget)
    t_p1 = budget_s * 0.10
    n_base = max(len(best_h), 200)
    periods = list(range(3, 26))
    for p in periods:
        if time.time() - t0 > t_p1:
            break
        m = n_base // p
        if m < 2:
            continue
        n_p = p * m
        for init in range(3):
            if time.time() - t0 > t_p1:
                break
            if init == 0:
                xp = np.clip(0.5 + 0.45 * np.cos(2 * np.pi * np.arange(p) / p), 0.01, 0.99)
            elif init == 1:
                xp = np.resize(best_h, p).astype(float)
                xp = np.clip(xp * (p / max(xp.sum(), 1e-10)) * 0.5, 0.01, 0.99)
            else:
                xp = np.random.uniform(0.1, 0.9, p)
                xp = np.clip(xp * (p / max(xp.sum(), 1e-10)) * 0.5, 0.01, 0.99)

            def pobj(xp_in):
                ht = make_feasible(np.tile(xp_in, m), n_p)
                return compute_c5(ht)

                              options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-11})

            bh_c5 = c5v

            if c5v < best_c5:

    # Phase 4: Multi-resolution exploration (8% budget)
