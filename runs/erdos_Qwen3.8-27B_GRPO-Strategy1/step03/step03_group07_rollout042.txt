```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(h):
    return h.sum() - len(h) / 2.0

def slsqp_step(h0, T, n, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def subgrad_step(h, alpha, n, n_active=3):
    """Subgradient descent step for max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top n_active lags
    top_idx = np.argsort(corr)[::-1][:n_active]
    weights = np.zeros(len(corr))
    for idx in top_idx:
        weights[idx] = corr[idx]
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for j in top_idx:
        w = weights[j]
        # j is in [0, 2n-2], lag = j - (n-1)
        # grad[m] += w * (1 - 2*h[j-m]) if 0<=j-m<n
        for m in range(n):
            jm = j - m
            if 0 <= jm < n:
                grad[m] += w * (1.0 - 2.0 * h[jm])
    
    h_new = h - alpha * grad
    return make_feasible(h_new, n)
```

For the coordinate descent, instead of random single-coordinate changes, let me do:
- For each coordinate i, find the optimal h[i] given all others fixed (this is a 1D optimization)
- The optimal h[i] minimizes the max over k of corr[k], where only corr[k] for k involving i changes

Actually, for a given i, changing h[i] affects corr[j] for j in [i, i+n-1] (the n lags that involve index i). So we can do a 1D minimization of max_{j in [i,i+n-1]} corr[j](h[i]).

This is a piecewise-linear 1D optimization! We can solve it exactly.

For each j in the affected range:
corr[j] as a function of h[i] is linear: corr[j] = a_j * h[i] + b_j

The max of these linear functions is piecewise linear, and we can find its minimum.

This is a much better local search than random coordinate changes!

```python
    # corr[j] = sum_{k: valid} h[k]*(1-h[j-k])
    # The terms involving h[i]:
    # corr[j] includes h[i]*(1-h[j-i]) if 0<=j-i<n (i.e., j in [i, i+n-1])
    # and h[j-i]*(1-h[i]) if 0<=j-i<n and j-i=i, i.e., j=2i...

    # Wait, the second term: k=j-i, so h[k]=h[j-i], and (1-h[j-k])=(1-h[j-(j-i)])=(1-h[i])
    # So: h[j-i]*(1-h[i]) when 0<=j-i<n and j-i≠i
    
    # If j-i = i (i.e., j=2i): the term is h[i]*(1-h[i]), which appears once.
    # If j-i ≠ i: two separate terms.
    # If j-i < 0 or >= n: no involvement of h[i] in corr[j].
    
    # So corr[j] = c_j + a_j * h[i] + b_j * h[i]^2 (for j in [i, i+n-1] where j-i != i)
    # Actually, let me expand:
    # For j where 0 <= j-i < n and j-i != i:
    #   corr[j] = (const) + h[i]*(1-h[j-i]) + h[j-i]*(1-h[i])
    #   = const + h[i] - h[i]*h[j-i] + h[j-i] - h[j-i]*h[i]
    #   = const + h[i] + h[j-i] - 2*h[i]*h[j-i]
    #   So coefficient of h[i]: 1 - 2*h[j-i]
    #   (linear in h[i])
    
    # For j where j-i = i (j = 2i, if 2i < 2n-1):
    #   corr[j] = (const) + h[i]*(1-h[i]) = const + h[i] - h[i]^2
    #   (quadratic in h[i])
    
    # For j outside [i, i+n-1]: corr[j] doesn't depend on h[i].
    
    # So the objective max_j corr[j] as a function of h[i] is the max of:
    # - Several linear functions (for j in [i, i+n-1], j≠2i)
    # - Possibly one quadratic (for j=2i if in range)
    # - Constants (for j outside range)
    
    # We can find the minimum by:
    # 1. Computing the "base" corr without h[i]: corr_base[j]
    # 2. For each affected j: corr[j] = corr_base[j] + delta_j * h[i] (linear case)
    # 3. Finding the minimum of the max
    
    # This is a 1D convex optimization (max of affine functions is convex)
    # We can use bisection or just evaluate at critical points.
    
    # Simpler approach: just do a golden section search on [0,1]
    # Since it's convex in h[i], the minimum is unique.
    
    h_i_old = h[i]
    corr_old = np.correlate(h, 1.0 - h, mode='full')
    
    # Affected lags: j from i to i+n-1 (clamped to [0, 2n-2])
    j_min = max(0, i)
    j_max = min(2*n-2, i+n-1)
    
    # For bisection: the function f(t) = max_j corr[j](t) is convex in t
    # We can find where the subgradient changes sign
    
    def f(t):
        h[i] = t
        corr = np.correlate(h, 1.0 - h, mode='full')
        h[i] = h_i_old  # restore
        return np.max(corr)
    
    # Golden section search
    a, b = 0.0, 1.0
    gr = (np.sqrt(5) - 1) / 2
    h1 = b - gr * (b - a)
    h2 = a + gr * (b - a)
    f1, f2 = f(h1), f(h2)
    for _ in range(40):
        if f1 < f2:
            b = h2
            h2 = h1
            f2 = f1
            h1 = b - gr * (b - a)
            f1 = f(h1)
        else:
            a = h1
            h1 = h2
            f1 = f2
            h2 = a + gr * (b - a)
            f2 = f(h2)
    
    opt = (a + b) / 2
    h[i] = opt
    return h
```

Wait, but make_feasible changes ALL entries to fix the sum, which breaks the "all other coordinates fixed" assumption. This is a problem.

Alternative: don't fix the sum during coordinate descent, and fix it at the end. But then the constraint isn't satisfied during the search.

Better alternative: when we change h[i], we also adjust h[i'] (some other coordinate) to maintain the sum. For example, adjust the coordinate with the most "room" (closest to 0 or 1).

Or: use a Lagrange multiplier approach. Add λ to the objective where λ is chosen to maintain the sum.

Actually, the simplest approach: after each coordinate update, redistribute the sum constraint over all with the "proportional room" method. This is what make_feasible does. It's a small perturbation and should be fine.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot > 1e-14:
            h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def sum_con(h):
    return h.sum() - len(h) / 2.0

def slsqp(h0, T, n, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                     bounds=bnds, constraints=[{'type':'eq','fun':sum_con}],
                     options={'maxiter':maxiter, 'ftol':ftol})
        return make_feasible(np.clip(r.x, 0, 1), n)
    except:
        return h0

def subgrad(h, n, n_iter, a0, a1, n_act=3):
    h = h.copy()
    for it in range(n_iter):
        f = it / max(1, n_iter-1)
        a = a0 * (a1/a0)**f
        corr = np.correlate(h, 1.0-h, mode='full')
        top = np.argsort(corr)[::-1][:n_act]
        wts = corr[top].astype(float); wts /= wts.sum()
        grad = np.zeros(n)
        for j, w in zip(top, wts):
            lo, hi = max(0,j-n+1), min(n-1,j)
            if hi >= lo:
                grad[lo:hi+1] += w * (1.0 - 2.0*h[j - np.arange(lo,hi+1)])
        h = make_feasible(h - a*grad, n)
    return h

def coord_desc(h, n, sweeps=5):
    h = h.copy()
    gr = (np.sqrt(5)-1)/2
    for _ in range(sweeps):
        for i in range(n):
            a, b = 0.0, 1.0
            h1, h2 = b-gr*(b-a), a+gr*(b-a)
            h[i] = h1
            f1 = np.max(np.correlate(h,1.0-h,mode='full'))
            h[i] = h2
            f2 = np.max(np.correlate(h,1.0-h,mode='full'))
            for _ in range(25):

                else:

                    h2 = a+gr*(b-a); h[i]=h2; f2=np.max(np.correlate(h,1.0-h,mode='full'))
            h[i] = (a+b)/2
            h = make_feasible(h, n)
    return h
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    d = n / 2.0 - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h; t = r.sum()
        if t > 1e-14: h += r * (d / t)
    else:
        t = h.sum()
        if t > 1e-14: h += h * (d / t)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(h)

def sum_con(h):
    return h.sum() - len(h) / 2.0

def periodic_smooth(xp, p, n, T):
    return smooth_obj(np.tile(xp, n // p), T)

def slsqp(h0, T, n, mi, ft):
    try:
        r = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': sum_con}],
                     options={'maxiter': mi, 'ftol': ft})
        return make_feasible(np.clip(r.x, 0, 1), n)
    except Exception:
        return h0

def subgrad(h, n, ni, a0, a1, na):
    h = h.copy()
    for it in range(ni):
        a = a0 * (a1 / a0) ** (it / max(1, ni - 1))
        c = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(c)[::-1][:na]
        w = c[top].astype(float); w /= w.sum()
        g = np.zeros(n)
        for j, wi in zip(top, w):
            lo, hi = max(0, j - n + 1), min(n - 1, j)
            if hi >= lo:
                g[lo:hi+1] += wi * (1.0 - 2.0 * h[j - np.arange(lo, hi + 1)])
        h = make_feasible(h - a * g, n)
    return h

            h[i] = h2; f2 = np.max(np.correlate(h, 1.0 - h, mode='full'))
