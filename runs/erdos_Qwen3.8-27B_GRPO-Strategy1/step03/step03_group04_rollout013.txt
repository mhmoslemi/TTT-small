def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    corr = compute_corr(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_grad(h, N, top_k, temp):
    """Soft-max weighted gradient using top_k peaks."""
    corr = compute_corr(h)
    peak = float(np.max(corr))
    # Get top_k indices
    top = np.argpartition(corr, -top_k)[-top_k:]
    # Soft-max weights over top_k
    tv = corr[top]
    w = np.exp((tv - tv.max()) / temp)
    w /= w.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for idx in range(top_k):
        wi = w[idx]
        if wi < 1e-10:
            continue
        L = top[idx] - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        elif -N < L < 0:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, peak

    stall = 0

        if peak < best_peak - 1e-10:
            best_peak = peak
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad

        vh = v / (1.0 - b2**it)

            h = project_sum(h, target)
            m[:] = 0
            v[:] = 0
            stall = 0

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_grad(h, N, top_k, temp):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    top = np.argpartition(corr, -top_k)[-top_k:]
    tv = corr[top]
    w = np.exp((tv - tv.max()) / temp)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for idx in range(top_k):
        wi = w[idx]
        if wi < 1e-12:
            continue
        L = int(top[idx]) - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        elif -N < L < 0:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, peak

def adam_opt(h, N, target, deadline, top_k, temp0, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    temp = temp0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    stall = 0
    while time.time() < deadline:
        grad, peak = soft_grad(h, N, top_k, temp)
        if peak < best_peak - 1e-10:
            best_peak = peak
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 180 == 0:
            lr *= decay
            temp = max(0.5, temp * 0.96)
        if stall > 1500:
            h = best_h + np.random.randn(N) * 0.012
            h = project_sum(h, target)
            m[:] = 0.0
            v[:] = 0.0
            stall = 0
    return best_h, best_peak * dx

def binary_refine(h_b, N, deadline, rng, n_cand=10):
    dx = 2.0 / N
    best = h_b.copy()
    cur = h_b.copy()
    best_pk = float(np.max(compute_corr(cur)))
    cur_pk = best_pk
    while time.time() < deadline:
        corr = compute_corr(cur)
        peak_idx = int(np.argmax(corr))
        L_star = peak_idx - (N - 1)
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        candidates = []
        if L_star > 0:
            to = [i for i in ones if i + L_star < N and cur[i + L_star] == 0]
        elif L_star < 0:
            to = [i for i in ones if i - L_star < N and cur[i - L_star] == 0]
        else:
            to = list(ones)
        n_target = max(1, n_cand // 2)
        n_random = n_cand - n_target
        for _ in range(n_target):
            if len(to) > 0:
                candidates.append((to[rng.randint(len(to))], zeros[rng.randint(len(zeros))]))
            else:
                candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        for _ in range(n_random):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        best_cand_pk = cur_pk
        best_ci = best_cj = -1
        for (ci, cj) in candidates:
            cur[ci], cur[cj] = 0.0, 1.0
            npk = float(np.max(compute_corr(cur)))
            if npk < best_cand_pk:
                best_cand_pk = npk
                best_ci, best_cj = ci, cj
            cur[ci], cur[cj] = 1.0, 0.0
        if best_ci >= 0:
            cur[best_ci], cur[best_cj] = 0.0, 1.0
            cur_pk = best_cand_pk
            if cur_pk < best_pk:
                best_pk = cur_pk
                best = cur.copy()
        if rng.random() < 0.02:
            for _ in range(3):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                cur[i], cur[j] = cur[j], cur[i]
                npk = float(np.max(compute_corr(cur)))
                if npk < cur_pk + 2:
                    cur_pk = npk
                    if cur_pk < best_pk:
                        best_pk = cur_pk
                        best = cur.copy()
                else:
                    cur[i], cur[j] = cur[j], cur[i]
    return best, best_pk * dx
