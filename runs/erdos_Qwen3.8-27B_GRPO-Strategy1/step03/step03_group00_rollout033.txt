from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_opt(h, n, i):
    """Optimize h[i] with all other coords fixed. Returns optimal value in [0,1]."""
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    
    dx = 2.0 / n
    idx0 = n - 1  # index of lag 0 in full correlation
    
    # For lag m (index m + n - 1 in corr array):
    # When h[i] goes from 0 to t:
    # m=0: add t*(1-t)
    # m≠0: add t*(1-h[i+m])*[0≤i+m<n] + h[i-m]*(1-t)*[0≤i-m<n]
    
    lin = np.zeros(2*n-1)
    cst = np.zeros(2*n-1)
    for m in range(-(n-1), n):
        if m == 0:
            continue
        idx = m + n - 1
        a = 0.0
        if 0 <= i + m < n:
            a = 1.0 - h[i+m]
        b = 0.0
        if 0 <= i - m < n:
            b = h[i-m]
        lin[idx] = a - b
        cst[idx] = b
    
    def f_of_t(t):
        v = corr_base + t * lin + cst
        v[idx0] = corr_base[idx0] + t - t*t
        return np.max(v) * dx
    
    res = minimize_scalar(f_of_t, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11})
    return res.x

    best_h = h.copy()

                improved = True
            else:

        if not improved:
