import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_obj(x, k, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.mean(top) * dx)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_solve(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def topk_subgrad(h, n_steps, top_k, alpha0, decay, t0, t_max):
    n = len(h)
    dx = 2.0 / n
    alpha = alpha0
    best = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_steps):
        if time.time() - t0 > t_max:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        wts = corr[idx].astype(float)
        ws = wts.sum()
        if ws > 0:
            wts = wts / ws
        else:
            wts = np.ones(top_k) / top_k
        grad = np.zeros(n)
        for w, ki in zip(wts, idx):
            s = int(ki - (n - 1))
            if s > 0:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] += -w * h[:n - s]
            elif s < 0:
                s2 = -s
                grad[s2:] += w * (1.0 - h[:n - s2])
                grad[:s2] += -w * h[s2:]
        h = project_feasible(h - alpha * dx * grad, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        alpha *= decay
    return best, best_c5

def active_repair(h, t0, t_max):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(30):
        if time.time() - t0 > t_max:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        worst = int(np.argmax(corr) - (n - 1))
        hp = best.copy()

            for i in range(n - worst):
                if best[i] > 0.4 and best[i + worst] < 0.6:
                    d = 0.04 * min(best[i], 1.0 - best[i + worst])
                    hp[i] -= d
                    hp[i + worst] += d
        elif worst < 0:
            s2 = -worst
            for i in range(s2, n):
                if best[i] > 0.4 and best[i - s2] < 0.6:
                    d = 0.04 * min(best[i], 1.0 - best[i - s2])
                    hp[i] -= d
                    hp[i - s2] += d
        hp = project_feasible(hp, n)
        cp = compute_c5(hp)
        if cp < best_c5:
            best_c5 = cp
            best = hp
    return best, best_c5

def cheb_smooth(h, n_coeffs):
    n = len(h)
    x = np.linspace(-1, 1, n)
    c = np.polynomial.chebyshev.chebfit(x, h, min(n_coeffs, n - 1))
    h_s = np.polynomial.chebyshev.chebval(x, c)
    return project_feasible(h_s, n)

def perturb_block(h, n):
    hp = h.copy()
    start = np.random.randint(0, max(1, n - 10))
    length = np.random.randint(3, max(4, n // 4))
    end = min(start + length, n)
    shift = np.random.randint(1, 3)
    hp[start:end] = np.roll(hp[start:end], shift)
    return project_feasible(hp, n)

def perturb_targeted(h, n):
    hp = h.copy()
    corr = np.correlate(hp, 1.0 - hp, mode='full')
    worst = int(np.argmax(corr) - (n - 1))

        for i in range(n - worst):
            if hp[i] > 0.5 and hp[i + worst] < 0.5:

        s2 = -worst
        for i in range(s2, n):
            if hp[i] > 0.5 and hp[i - s2] < 0.5:
                hp[i] -= 0.08
                hp[i - s2] += 0.08
    return project_feasible(hp, n)

def perturb_gaussian(h, n, sc):
    return project_feasible(h + np.random.randn(n) * sc, n)
