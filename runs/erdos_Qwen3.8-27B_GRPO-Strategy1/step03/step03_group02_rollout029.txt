Let me also think about a "dual" approach. We can write:
C₅(h) = max_k ∫h(x)(1-h(x+k))dx

By the max-min (von Neumann) theorem, we can write:
C₅(h) = min_p max_h Σ_k p_k ∫h(x)(1-h(x+k))dx

where p is a probability distribution over k. At the optimum, p is concentrated on the active lags.

The inner maximization over h (with constraints 0≤h≤1, ∫h=1) can be solved as a convex problem for fixed p. So we can alternate:
1. Given h, find p (concentrate on max lags)
2. Given p, solve the convex optimization for h

This is essentially what the gradient method does, but the convex subproblem could be solved more exactly.

Actually, for fixed p, the objective is:
F(h) = Σ_k p_k ∫h(x)(1-h(x+k))dx = Σ_k p_k [∫h(x)dx - ∫h(x)h(x+k)dx]
= 1 - Σ_k p_k ∫h(x)h(x+k)dx

Minimizing F is equivalent to maximizing G(h) = Σ_k p_k ∫h(x)h(x+k)dx, subject to 0≤h≤1, ∫h=1.

This is a convex maximization of a quadratic form (not convex in general), so it's non-convex. The problem is NP-hard in general.

        return np.zeros(N), true_c5
    w /= wsum
    # Scale to approximate the gradient of the power mean
    center = N - 1
    grad = np.zeros(N)
    for ix in range(len(w)):
        if w[ix] < 1e-10:
            continue
        k = ix - center
        wi = w[ix]
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            if k < N:
                gk = np.zeros(N)
                gk[k:] = 1.0 - h[:N-k]
                if N-k > 0:
                    gk[:N-k] -= h[k:]
                grad += wi * gk
        else:
            m = -k
            if m < N:
                gm = np.zeros(N)
                gm[:N-m] = 1.0 - h[m:]
                if N-m > 0:
                    gm[m:] -= h[:N-m]
                grad += wi * gm
    return grad, true_c5

def run_phase(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K, iters_target, use_power=False, p_val=20):
    best_h = h.copy()
    best_c5 = compute_c5_only(h, dx)
    b1, b2, eps = 0.9, 0.999, 1e-8
    mvec = np.zeros(N)
    vvec = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / max(iters_target, 1), 1.0)
        lr = lr0 * (lr1 / lr0) ** frac
        if time.time() > deadline:
            break
        if use_power:
            tau = p_val  # not used
            grad, c5_val = gradient_power(h, N, p_val)
        else:
            tau = tau0 * (tau1 / tau0) ** frac
            grad, c5_val = gradient_topk(h, N, tau, K)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        mvec = b1 * mvec + (1.0 - b1) * grad
        vvec = b2 * vvec + (1.0 - b2) * grad * grad
        it += 1
        mh = mvec / (1.0 - b1 ** it)
        vh = vvec / (1.0 - b2 ** it)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = project_sum(h_new, target)
        h = h_new
    return best_h, best_c5

def local_search(h, N, target, dx, n_trials, deadline, scale):
    best_h = h.copy()
    best_c5 = compute_c5_only(h, dx)
    rng = np.random.RandomState(777)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        trial = best_h.copy()
        n_mod = rng.randint(1, 6)
        idx = rng.choice(N, size=n_mod, replace=False)
        trial[idx] += rng.randn(n_mod) * scale
        trial = project_sum(trial, target)
        c = compute_c5_only(trial, dx)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_h = trial
    return best_h, best_c5
