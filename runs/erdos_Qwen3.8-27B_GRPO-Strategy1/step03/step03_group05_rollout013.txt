            grad[i1] += (1.0 - h[i1 - s])
            i2 = np.arange(-s, n)
            grad[i2] -= h[i2 + s]
    grad *= dx / k
    return obj, grad

def pgd_h(h, steps, k, lr0, decay, t_deadline):
    n = len(h)
    best, best_c5 = h.copy(), compute_c5(h)
    lr = lr0
    for _ in range(steps):
        if time.time() > t_deadline:
            break
        _, grad = topk_and_grad(h, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        alpha = min(lr, 0.5 / gnorm)
        for _ in range(40):
            h_new = project_feasible(h - alpha * grad, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 + 1e-12:
                break
            alpha *= 0.4
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= decay
    return best, best_c5

def sum_eq(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_topk(h, k, maxiter, ftol, t_deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    def obj(g):
        v, _ = topk_and_grad(g, k)
        return v
    def obj_grad(g):
        _, gr = topk_and_grad(g, k)
        return gr
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > t_deadline:
            return h
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def softmax_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def make_perturb(h, n, scale, seed):
    rng = np.random.RandomState(seed)
    hp = h + rng.randn(n) * scale
    return project_feasible(hp, n)

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False) + 1.0 / len(h_old)
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Rapid projected gradient (k=8,5,3,1)
    t_phase1 = t0 + budget_s * 0.08
    for k in [8, 5, 3, 1]:
        if elapsed() > t_phase1:
            break
        h1, c1 = pgd_h(best_h, 300, k, 0.02, 0.999, t_phase1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: SLSQP with exact gradient at various k
    t_phase2 = t0 + budget_s * 0.25
    for k in [3, 5, 1, 2]:
        if elapsed() > t_phase2:
            break
        h2 = slsqp_topk(best_h, k, 500, 1e-15, t_phase2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
        h2b, c2b = pgd_h(best_h, 200, max(1, k), 0.005, 0.9995, t_phase2)
        if c2b < best_c5:
            best_c5, best_h = c2b, h2b

    # Phase 3: Basin hopping with diverse perturbations
    t_phase3 = t0 + budget_s * 0.60
    trial = 0
    while elapsed() < t_phase3 and trial < 80:
        trial += 1
        r = np.random.random()
        if r < 0.25:
            hp = make_perturb(best_h, best_n, 0.08, seed + trial * 7)
        elif r < 0.45:
            # Block shift
            hp = best_h.copy()
            s = np.random.randint(0, best_n - 10)
            L = np.random.randint(5, best_n // 4)
            sh = np.random.randint(2, L)
            hp[s:s+L] = np.roll(hp[s:s+L], sh)
            hp = project_feasible(hp, best_n)
        elif r < 0.60:
            # Targeted: reduce worst shift conflict
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            si = int(np.argmax(corr) - (best_n - 1))
            hp = best_h.copy()
            if si >= 0:
                for i in range(si, best_n):
                    if best_h[i] > 0.5 > best_h[i - si]:
                        d = min(best_h[i] * 0.2, 1.0 - best_h[i - si])
                        hp[i] -= d
                        hp[i - si] += d
                    elif best_h[i] < 0.5 < best_h[i - si]:
                        d = min((1.0 - best_h[i]) * 0.2, best_h[i - si])
                        hp[i] += d
                        hp[i - si] -= d
            else:
                for i in range(0, best_n + si):
                    if best_h[i] > 0.5 > best_h[i - si]:
                        d = min(best_h[i] * 0.2, 1.0 - best_h[i - si])
                        hp[i] -= d
                        hp[i - si] += d
            hp = project_feasible(hp, best_n)
        elif r < 0.75:
            # Different resolution
            nr = [80, 100, 130, 160, 250][trial % 5]
            hi = interpolate_h(best_h, nr)
            hi = make_perturb(hi, nr, 0.03, seed + trial * 13)
            hi = slsqp_topk(hi, 3, 200, 1e-13, t_phase3)
            c_hi = compute_c5(hi)
            if c_hi < best_c5:
                best_c5, best_h, best_n = c_hi, hi, nr
            else:
                hd = interpolate_h(hi, best_n)
                c_d = compute_c5(hd)
                if c_d < best_c5:
                    best_c5, best_h = c_d, hd
                continue
        else:
            # Smoothed logistic perturbation
            hp = best_h.copy()
            center = np.random.randint(0, best_n)
            width = np.random.randint(5, best_n // 3)
            x = np.linspace(0, 2, best_n)
            bump = 0.1 / (1.0 + np.exp(-((x - x[center]) / (width * dx))))
            hp = project_feasible(best_h + bump * (np.random.random() - 0.5), best_n)
        
        # Refine candidate
        ho = slsqp_topk(hp, 3, 250, 1e-13, t_phase3)
        ho, co = pgd_h(ho, 150, 1, 0.005, 0.9995, t_phase3)
        if co < best_c5:
            best_c5, best_h = co, ho
        elif compute_c5(ho) < best_c5:
            best_c5, best_h = compute_c5(ho), ho

    # Phase 4: Multi-resolution fine-tuning
    t_phase4 = t0 + budget_s * 0.85
    for nr in [150, 180, 220, 250, 300]:
        if elapsed() > t_phase4:
            break
        hi = interpolate_h(best_h, nr)
        hi = make_perturb(hi, nr, 0.015, seed + 999 + nr)
        hi = slsqp_topk(hi, 3, 300, 1e-14, t_phase4)
        hi, ci = pgd_h(hi, 200, 1, 0.003, 0.9997, t_phase4)
        if ci < best_c5:
            best_c5, best_h, best_n = ci, hi, nr
        # Also try without perturbation
        hi2 = interpolate_h(best_h, nr)
        hi2 = slsqp_topk(hi2, 5, 250, 1e-14, t_phase4)
        hi2, ci2 = pgd_h(hi2, 150, 1, 0.003, 0.9997, t_phase4)
        if ci2 < best_c5:
            best_c5, best_h, best_n = ci2, hi2, nr

    # Phase 5: Intensive local polish
    t_phase5 = t0 + budget_s * 0.96
    for k in [1, 2, 3]:
        if elapsed() > t_phase5:
            break
        for _ in range(3):
            if elapsed() > t_phase5:
                break
            hp = slsqp_topk(best_h, k, 800, 1e-16, t_phase5)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_c5, best_h = cp, hp
            hp2, cp2 = pgd_h(best_h, 300, k, 0.002, 0.9998, t_phase5)
            if cp2 < best_c5:
                best_c5, best_h = cp2, hp2
    # Softmax polish for global max
    if elapsed() < t_phase5 + 20:
        try:
            hsm = slsqp_topk(best_h, 1, 400, 1e-15, t_phase5)
            csm = compute_c5(hsm)
            if csm < best_c5:
                best_c5, best_h = csm, hsm
        except Exception:
            pass

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
