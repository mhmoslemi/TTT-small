        if not improved:
            deltas = [d * 0.5 for d in deltas]
            if min(deltas) < 0.001:
                break
    return best_h, best_c5

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_grad(h, N, beta, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    n_corr = len(corr)
    if K >= n_corr:
        top_idx = np.arange(n_corr)
        c_top = corr
    else:
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    sc = beta * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        if w[ix] < 1e-7:
            continue
        k = top_idx[ix]
        L = k - center
        if L == 0:
            grad += w[ix] * (1.0 - 2.0 * h)
