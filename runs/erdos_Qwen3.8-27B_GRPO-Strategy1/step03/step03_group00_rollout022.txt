import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    corr_b = get_corr(h0)
    # Compute derivative
    m = 2 * n - 1
    coef = np.zeros(m)
    for k in range(m):
        lag = k - (n - 1)
        c = 0.0
        j1 = i - lag
        if 0 <= j1 < n:
            c += 1.0 - h[j1]
        j2 = i + lag
        if 0 <= j2 < n:
            c -= h[j2]
        coef[k] = c
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = corr_b + mid * coef
        mv = vals.max()
        active = vals >= mv - 1e-14
        sl = coef[active]
        if sl.min() > 1e-15:
            hi = mid
        elif sl.max() < -1e-15:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def full_coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            x_new = coord_opt(best_h, n, i)
            if abs(x_new - best_h[i]) > 1e-10:
                old = best_h[i]
                best_h[i] = x_new
                best_h = project(best_h, n)
                c5v = compute_c5(best_h)
                if c5v < best_c5 - 1e-14:
                    best_c5 = c5v
                    improved = True
                else:
                    best_h[i] = project(np.concatenate([best_h[:i], [old], best_h[i+1:]]), n)
                    # Actually this is getting complicated. Let me just keep the new value if it's close enough
                    best_h[i] = x_new
                    best_h = project(best_h, n)
        if not improved:
            break
    return best_h, best_c5
