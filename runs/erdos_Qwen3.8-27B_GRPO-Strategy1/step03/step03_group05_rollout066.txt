```python
def lse_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_max = corr.max()
    shifted = np.clip(T * (corr - m_max), -500, 500)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    # For each output index m (0 to 2n-2), lag k = m - (n-1)
    # d corr[m] / d h[i] = (1-h[i-k]) * [0<=i-k<n] - h[i+k] * [0<=i+k<n]
    
    for m in range(2*n-1):
        w = weights[m]
        if w < 1e-10:
            continue

        # (1-h[i-k]) for i-k in [0,n-1], i.e., i in [k, k+n-1] intersect [0,n-1]

            grad[i_lo2:i_hi2+1] -= w * h[i_lo2+k:i_hi2+k+1]
    
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += room * (diff / s)
        else:
            s = h.sum()
            if s < 1e-14: break
            h += h * (diff / s)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_obj_grad(h, T, n, lam_sum=1e5):
    dx = 2.0 / n
    corr = get_corr(h)
    m = corr.max()
    shifted = np.clip(T * (corr - m), -700, 700)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    obj = (m + np.log(exps.sum()) / T) * dx
    # Add penalty
    sum_err = h.sum() - n / 2.0
    obj += lam_sum * sum_err * sum_err
    
    # Gradient
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        k = m_idx - (n - 1)
        # (1-h[i-k]) term: i in [max(0,k), min(n-1, n-1+k)]
        i1_lo, i1_hi = max(0, k), min(n - 1, n - 1 + k)
        if i1_lo <= i1_hi:
            grad[i1_lo:i1_hi+1] += w * (1.0 - h[i1_lo-k:i1_hi-k+1])
        # -h[i+k] term: i in [max(0,-k), min(n-1, n-1-k)]
        i2_lo, i2_hi = max(0, -k), min(n - 1, n - 1 - k)
        if i2_lo <= i2_hi:
            grad[i2_lo:i2_hi+1] -= w * h[i2_lo+k:i2_hi+k+1]
    grad *= dx
    # Penalty gradient
    grad += 2.0 * lam_sum * sum_err * np.ones(n)
    return obj, grad

def lbfgs_optimize(h, T, n, maxiter=300, lam_sum=1e5):
    bnds = [(0.0, 1.0)] * n
    res = minimize(lambda x: lse_obj_grad(x, T, n, lam_sum)[0],
                   h,
                   jac=lambda x: lse_obj_grad(x, T, n, lam_sum)[1],
                   method='L-BFGS-B',
                   bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
    return project_feasible(res.x, n)

def weighted_lse_obj_grad(h, T, n, top_k, lam_sum=1e5):
    """LSE with extra weight on top-k worst shifts."""
    dx = 2.0 / n
    corr = get_corr(h)
    # Boost top-k
    idx = np.argpartition(corr, -top_k)[-top_k:]
    boost = np.ones(2*n-1)
    boost[idx] = 5.0  # 5x weight on top-k
    m = np.max(corr)
    shifted = np.clip(T * boost * (corr - m), -700, 700)
    exps = np.exp(shifted)
    total_w = exps.sum()
    weights = exps / total_w
    obj = (m + np.log(total_w) / T) * dx
    sum_err = h.sum() - n / 2.0
    obj += lam_sum * sum_err * sum_err
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        k = m_idx - (n - 1)
        i1_lo, i1_hi = max(0, k), min(n-1, n-1+k)
        if i1_lo <= i1_hi:
            grad[i1_lo:i1_hi+1] += w * (1.0 - h[i1_lo-k:i1_hi-k+1])
        i2_lo, i2_hi = max(0, -k), min(n-1, n-1-k)
        if i2_lo <= i2_hi:
            grad[i2_lo:i2_hi+1] -= w * h[i2_lo+k:i2_hi+k+1]
    grad *= dx
    grad += 2.0 * lam_sum * sum_err * np.ones(n)
    return obj, grad

def targeted_repair(h, n):
    """Reduce overlap at worst shift by moving mass."""
    corr = get_corr(h)
    m_idx = np.argmax(corr)
    k = m_idx - (n - 1)
    h_new = h.copy()
    # For shift k, overlap = sum_j h[j] * (1-h[j-k]) (adjusted for valid range)
    # Reduce: where h[j] is high and h[j-k] is low, move mass
    for j in range(n):
        jk = j - k
        if 0 <= jk < n:
            if h_new[j] > 0.6 and h_new[jk] < 0.4:
                delta = min((h_new[j] - 0.5) * 0.1, (0.5 - h_new[jk]) * 0.1)
                h_new[j] -= delta
                h_new[jk] += delta
            elif h_new[j] < 0.4 and h_new[jk] > 0.6:
                delta = min((0.5 - h_new[j]) * 0.1, (h_new[jk] - 0.5) * 0.1)
                h_new[j] += delta
                h_new[jk] -= delta
    return project_feasible(h_new, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += room * (diff / s)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += h * (diff / s)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def lse_obj_grad(h, T, n, lam_sum, top_k_boost=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    boost = np.ones(2 * n - 1)
    if top_k_boost is not None:
        idx = np.argpartition(corr, -top_k_boost)[-top_k_boost:]
        boost[idx] = 4.0
    shifted = np.clip(T * boost * (corr - m), -700, 700)
    exps = np.exp(shifted)
    total = exps.sum()
    weights = exps / total
    obj = (m + np.log(total) / T) * dx
    sum_err = h.sum() - n / 2.0
    obj += lam_sum * sum_err * sum_err
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        k = m_idx - (n - 1)
        i1lo, i1hi = max(0, k), min(n - 1, n - 1 + k)
        if i1lo <= i1hi:
            grad[i1lo:i1hi + 1] += w * (1.0 - h[i1lo - k:i1hi - k + 1])
        i2lo, i2hi = max(0, -k), min(n - 1, n - 1 - k)
        if i2lo <= i2hi:
            grad[i2lo:i2hi + 1] -= w * h[i2lo + k:i2hi + k + 1]
    grad *= dx
    grad += 2.0 * lam_sum * sum_err
    return obj, grad

def lbfgs_step(h, T, n, maxiter=250, lam_sum=1e5, top_k_boost=None):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: lse_obj_grad(x, T, n, lam_sum, top_k_boost)[0],
            h,
            jac=lambda x: lse_obj_grad(x, T, n, lam_sum, top_k_boost)[1],
            method='L-BFGS-B', bounds=bnds,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return h

def targeted_repair(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_idx = np.argmax(corr)
    k = m_idx - (n - 1)
    h_new = h.copy()
    for j in range(n):
        jk = j - k
        if 0 <= jk < n:
            if h_new[j] > 0.55 and h_new[jk] < 0.45:
                d = min((h_new[j] - 0.5) * 0.08, (0.5 - h_new[jk]) * 0.08)
                h_new[j] -= d
                h_new[jk] += d
            elif h_new[j] < 0.45 and h_new[jk] > 0.55:
                d = min((0.5 - h_new[j]) * 0.08, (h_new[jk] - 0.5) * 0.08)
                h_new[j] += d
                h_new[jk] -= d
    return project_feasible(h_new, n)

def slsqp_polish(h, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return (m + np.log(np.sum(np.exp(np.clip(200*(c - m), -700, 700)))) / 200) * dx
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = np.clip(200*(c - m), -700, 700)
        ex = np.exp(sh)
        wt = ex / ex.sum()
        g = np.zeros(n)
        for mi in range(2 * n - 1):
            w = wt[mi]
            if w < 1e-12:
                continue
            k = mi - (n - 1)
            il, ih = max(0, k), min(n-1, n-1+k)
            if il <= ih:
                g[il:ih+1] += w * (1.0 - x[il-k:ih-k+1])
            il2, ih2 = max(0, -k), min(n-1, n-1-k)
            if il2 <= ih2:
                g[il2:ih2+1] -= w * x[il2+k:ih2+k+1]
        return g * dx
    def sumfun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': sumfun}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
```
