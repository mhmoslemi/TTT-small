```python
def fast_coord_descent(h, n, t_budget, max_sweeps=200):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * 2.0 / n
    t_end = time.time() + t_budget
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return h, best_c5
            old_hi = h[i]
            # Compute dC
            dC = np.zeros(2*n - 1)
            # (1-h[i+k]) term: valid when 0 <= i+k < n, k from -(n-1) to n-1
            # In corr index: k corresponds to index k + n - 1
            k_min = max(-(n-1), -i)
            k_max = min(n-1, n-1-i)
            if k_min <= k_max:
                ks = np.arange(k_min, k_max+1)
                idx_h = i + ks  # indices of h that are 1-h[j+k]
                dC[ks + n - 1] += (1.0 - h[idx_h])
            # -h[i-k] term: valid when 0 <= i-k < n, k != 0 (or k==0 for the double count)
            k_min2 = max(-(n-1), i - (n-1))  # i-k >= 0 => k <= i
            k_max2 = min(n-1, i)  # i-k <= n-1 => k >= i-(n-1)
            # Wait: 0 <= i-k < n => i-n+1 <= k <= i
            k_min2 = max(-(n-1), i - (n-1))
            k_max2 = min(n-1, i)
            if k_min2 <= k_max2:
                ks2 = np.arange(k_min2, k_max2+1)
                idx_h2 = i - ks2
                dC[ks2 + n - 1] += (-h[idx_h2])
            
            # Solve min_{t in [0,1]} max_k (corr[k] + (t - old_hi) * dC[k])
            # Evaluate on a grid
            T_grid = np.linspace(0.0, 1.0, 41)
            G = np.max(corr[np.newaxis,:] + (T_grid[:,np.newaxis] - old_hi) * dC[np.newaxis,:], axis=1)
            t_star = T_grid[np.argmin(G)]
            # Refine
            if 0 < t_star < 1:
                T_ref = np.linspace(max(0, t_star-0.025), min(1, t_star+0.025), 21)
                G_ref = np.max(corr[np.newaxis,:] + (T_ref[:,np.newaxis] - old_hi) * dC[np.newaxis,:], axis=1)
                t_star = T_ref[np.argmin(G_ref)]

                corr += delta * dC
                h[i] = new_hi
                c5_val = np.max(corr) * 2.0 / n
                if c5_val < best_c5 - 1e-14:
                    best_c5 = c5_val
                    improved = True
        
        if not improved:
            break
    return h, best_c5
```

```python
def solve_1d_max_linear(a, b, lo=0.0, hi=1.0):
    """Minimize max_k (a[k]*t + b[k]) over t in [lo, hi].
    a, b are 1D numpy arrays. Returns optimal t."""
    def g(t):
        return np.max(a * t + b)
    
    # Check endpoints
    g_lo = g(lo)
    g_hi = g(hi)
    
    # Find active line at lo and hi
    idx_lo = np.argmax(a * lo + b)
    idx_hi = np.argmax(a * hi + b)
    
    # Subgradient at lo (any active line's slope works for convex max)
    # For convex max, the subgradient is [min active slope, max active slope]
    # But for finding minimum, we just need: if all active slopes > 0, go right; if all < 0, go left
    active_lo = np.where(np.isclose(a * lo + b, g_lo, atol=0))
    active_hi = np.where(np.isclose(a * hi + b, g_hi, atol=0))
    
    # Simpler: just use ternary/golden section since it's 1D convex
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = g(c)
    fd = g(d)
    for _ in range(50):  # 50 iterations gives ~1e-10 precision
        if fc < fd:
            hi = d; d = c; fd = fc; c = hi - gr*(hi-lo); fc = g(c)
        else:
            lo = c; c = d; fc = fd; d = lo + gr*(hi-lo); fd = g(d)
    return 0.5 * (lo + hi)
```

```python
import time

    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h

    else:
        deficit = target - s
        room = 1.0 - h
        tot_room = room.sum()
        if tot_room < 1e-12:
            return h
        h = h + room * (deficit / tot_room)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-12:
                break
            h = h + room * (target - s) / tr
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_polish(h, n, T, maxiter, ftol):
    corr = np.correlate(h, 1.0 - h, mode='full')
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def fast_coord_descent(h, n, t_budget, n_sweeps=2000, perturb_every=80):
    h = project(h.copy(), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * 2.0 / n)
    t_end = time.time() + t_budget
    rng = np.random.default_rng(12345)
    
    T_grid = np.linspace(0.0, 1.0, 51)
    
    sweep = 0
    while time.time() < t_end and sweep < n_sweeps:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            old = h[i]
            # Compute dC vector
            n2 = 2 * n - 1
            dC = np.zeros(n2)
            # Term 1: (1-h[i+k]) for 0<=i+k<n, k in [-(n-1), n-1]
            klo = max(-(n-1), -i)
            khi = min(n-1, n-1-i)
            if klo <= khi:
                ks = np.arange(klo, khi+1)
                dC[ks + n - 1] += 1.0 - h[i + ks]
            # Term 2: -h[i-k] for 0<=i-k<n
            klo2 = max(-(n-1), i-n+1)
            khi2 = min(n-1, i)
            if klo2 <= khi2:
                ks2 = np.arange(klo2, khi2+1)
                dC[ks2 + n - 1] -= h[i - ks2]
            
            # Solve min_{t in [0,1]} max_k (corr[k] + (t-old)*dC[k])
            vals = corr[np.newaxis,:] + (T_grid[:,np.newaxis] - old) * dC[np.newaxis,:]
            G = np.max(vals, axis=1)
            t_star = T_grid[np.argmin(G)]
            # Refine
            if 0.01 < t_star < 0.99:
                T_ref = np.linspace(t_star-0.02, t_star+0.02, 21)
                T_ref = np.clip(T_ref, 0, 1)
                vals2 = corr[np.newaxis,:] + (T_ref[:,np.newaxis] - old) * dC[np.newaxis,:]
                G2 = np.max(vals2, axis=1)
                t_star = T_ref[np.argmin(G2)]
            
            if abs(t_star - old) > 1e-10:
                delta = t_star - old
                corr += delta * dC
                h[i] = t_star
                # Compensate sum: distribute -delta to others
                # Actually, let's do it at the end of the sweep to avoid O(n) per coordinate
                improved = True
        
        # End of sweep: project
        h = project(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * 2.0 / n)
        if c5_val < best_c5:
```
