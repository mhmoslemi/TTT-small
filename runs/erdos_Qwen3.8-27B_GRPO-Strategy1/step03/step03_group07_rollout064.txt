import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp((corr - m) * T)) / T)) * dx

def gradient_softmax(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = T * np.exp((corr - m) * T) / np.sum(np.exp((corr - m) * T))
    # gradient of corr[k] w.r.t. h[i]
    # corr[k] = sum_j h[j] * (1-h[j+k])
    # dc/dh[i] = (1-h[i+k]) - h[i-k] for valid indices
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-10:
            continue
        # k is the lag index in 'full' mode
        # lag = k - (n-1)
        lag = k - (n - 1)
        if lag >= 0:
            # corr = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            for i in range(n - lag):
                grad[i] += w * (1 - h[i + lag])
                if lag > 0:
                    grad[i + lag] += w * (-h[i])
        else:
            lag2 = -lag
            for i in range(n - lag2):
                grad[i + lag2] += w * (1 - h[i])
                grad[i] += w * (-h[i + lag2])
    return grad * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s

        if total > 1e-14:

        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def softmax_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # soft max: m + (1/T) * log(sum(exp(T*(corr-m))))
    shifted = T * (corr - m)
    max_s = np.max(shifted)
    log_sum = max_s + np.log(np.sum(np.exp(shifted - max_s)))
    return (m + log_sum / T) * dx

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def slsqp_opt(h0, T, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    res = minimize(softmax_c5, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def _sum_eq(h):
    return h.sum() - len(h) / 2.0

def _softobj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    msh = np.max(shifted)
    log_sum = msh + np.log(np.sum(np.exp(shifted - msh)))
    return (m + log_sum / T) * dx

def _active_obj(h, T, act_idx, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    msh = np.max(shifted)
    log_sum = msh + np.log(np.sum(np.exp(shifted - msh)))
    base = (m + log_sum / T) * dx
    act_pen = lam * np.mean(corr[act_idx]) * dx
    return base + act_pen

def _do_slsqp(h0, T, n, maxiter, ftol, act_idx=None, lam=0.0):

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
