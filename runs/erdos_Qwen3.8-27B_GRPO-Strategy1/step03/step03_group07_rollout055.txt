    """One subgradient step to reduce max correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)  # lag index
    k = max_idx - (n - 1)  # actual lag (can be negative)

    grad = np.zeros(n)
    if k >= 0:
        # corr[k] = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        for i in range(n - k):
            grad[i] += -h[i + k]  # from (1-h[i]) term
            grad[i + k] += 1 - h[i]  # from h[i+k] term
    else:
        # corr[k] = sum_{i=-k}^{n-1} h[i+k] * (1-h[i]) = sum_{i=0}^{n-1+k} h[i] * (1-h[i-k])
        # Wait, for k<0: corr[k] = sum h[i+k] * (1-h[i]) where i+k >= 0, i.e., i >= -k
        # Let j = -k > 0: corr[-j] = sum_{i=j}^{n-1} h[i-j] * (1-h[i])
        j = -k
        for i in range(j, n):
            grad[i - j] += 1 - h[i]  # from h[i-j] term
            grad[i] += -h[i - j]  # from (1-h[i]) term
    
    # Projected subgradient step
    h_new = h - lr * grad
    h_new = make_feasible(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def get_subgrad(h, n, k):
    """Subgradient of corr[k] wrt h, where k is the actual lag (negative to positive)"""
    grad = np.zeros(n)
    if k >= 0:
        # corr[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        idx = np.arange(n - k)
        grad[idx] += -h[idx + k]
        grad[idx + k] += 1.0 - h[idx]
    else:
        j = -k
        # corr[-j] = sum_{i=j}^{n-1} h[i-j]*(1-h[i])
        idx = np.arange(j, n)
        grad[idx - j] += 1.0 - h[idx]
        grad[idx] += -h[idx - j]
    return grad

def subgrad_descent(h, n, time_limit, lr_init=0.01, lr_decay=0.9995):
    """Subgradient descent targeting the max correlation lag"""
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = lr_init
    t_end = time.time() + time_limit
    
    while time.time() < t_end:
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_pos = np.argmax(corr)
        k = max_pos - (n - 1)
        grad = get_subgrad(best, n, k)
        
        best = best - lr * grad
        best = make_feasible(best, n)
        c5v = compute_c5(best)
        
        if c5v < best_c5:
            best_c5 = c5v
            lr = min(lr * 1.05, 0.1)
        else:
            lr = max(lr * lr_decay, 0.0001)
    
    return best, best_c5

def local_search(h, n, time_limit, rng):
    """Random local search with small perturbations"""
    best = h.copy()
    best_c5 = compute_c5(best)
    step = 0.015
    t_end = time.time() + time_limit
    
    while time.time() < t_end:
        n_adj = rng.integers(1, 4)
        idx = rng.integers(0, n, size=n_adj)
        h_try = best.copy()
        deltas = rng.uniform(-step, step, size=n_adj)
        h_try[idx] += deltas
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best = h_try
            best_c5 = c5_try
            step = max(0.001, step * 0.998)
        elif rng.random() < 0.01:
            step = min(0.05, step * 1.5)
    
    return best, best_c5

    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_diff(x):
    return x.sum() - len(x) / 2.0

def ones_jac(x):
    return np.ones(len(x))

def get_subgrad(h, n, k):
    grad = np.zeros(n)
    if k >= 0:
        idx = np.arange(n - k)
        grad[idx] += -h[idx + k]
        grad[idx + k] += 1.0 - h[idx]
    else:
        j = -k
        idx = np.arange(j, n)
        grad[idx - j] += 1.0 - h[idx]
        grad[idx] += -h[idx - j]
    return grad

def subgrad_descent(h, n, time_limit, lr_init=0.03):
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = lr_init
    t_end = time.time() + time_limit
    while time.time() < t_end:
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_pos = np.argmax(corr)
        k = max_pos - (n - 1)
        grad = get_subgrad(best, n, k)
        best = best - lr * grad
        best = make_feasible(best, n)
        c5v = compute_c5(best)
        if c5v < best_c5 - 1e-15:
            best_c5 = c5v
            lr = min(lr * 1.02, 0.08)
        else:
            lr = max(lr * 0.998, 0.0005)
    return best, best_c5

def local_search(h, n, time_limit, rng):
    best = h.copy()
    best_c5 = compute_c5(best)
    step = 0.012
    t_end = time.time() + time_limit
    while time.time() < t_end:
        n_adj = int(rng.integers(1, 4))
        idx = rng.integers(0, n, size=n_adj)
        h_try = best.copy()
        h_try[idx] += rng.uniform(-step, step, size=n_adj)
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best = h_try
            best_c5 = c5_try
            step = max(0.0005, step * 0.997)
        else:
            if rng.random() < 0.02:
                step = min(0.04, step * 1.3)
    return best, best_c5

def targeted_pair_opt(h, n, time_limit, rng):
    best = h.copy()
    best_c5 = compute_c5(best)
    t_end = time.time() + time_limit
    while time.time() < t_end:
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_pos = np.argmax(corr)
        k = max_pos - (n - 1)
        if k >= 0:
            idx_range = np.arange(n - k)
        else:
            idx_range = np.arange(-k, n)
        top_i = idx_range[np.argmax(best[idx_range] * (1.0 - best[np.mod(idx_range + np.mod(k, n), n)]))]
        partner = (top_i + k) % n
        for _ in range(50):
            di = rng.uniform(-0.05, 0.05)
            dj = rng.uniform(-0.05, 0.05)
            h_try = best.copy()
            h_try[top_i] += di
            h_try[partner] += dj
            h_try = make_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best = h_try
                best_c5 = c5_try
    return best, best_c5

def slsqp_refine(h, n, T, maxiter):
    try:
        res = minimize(smooth_max_obj, h.copy(), args=(T,),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_diff, 'jac': ones_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        n = 200
        best_h = np.empty(n)
        for i in range(n):
            best_h[i] = 0.82 if (i % 5) < 3 else 0.22
        best_h = make_feasible(best_h, n)

    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with smooth max, increasing T (15% budget)
    t1 = budget_s * 0.15
    for T in [100, 500, 2000, 8000, 30000]:
        if elapsed() > t1:
            break
        h_new = slsqp_refine(best_h, n, T, 4000)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    # Phase 2: Subgradient descent (15% budget)
    t2 = budget_s * 0.30
    if elapsed() < t2:
        remain = min(t2 - elapsed(), budget_s * 0.12)
        best_h, best_c5 = subgrad_descent(best_h, n, remain, lr_init=0.025)

    # Phase 3: Alternating SLSQP + local search (35% budget)
    t3 = budget_s * 0.65
    it = 0
    while elapsed() < t3:
        it += 1
        T = 500 * (1 + it)
        if elapsed() < t3 - 20:
            h_new = slsqp_refine(best_h, n, T, 2000)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        if elapsed() < t3 - 10:
            sub_t = min(12.0, t3 - elapsed())
            best_h, best_c5 = local_search(best_h, n, sub_t, rng)
        if elapsed() < t3 - 5:
            sub_t = min(8.0, t3 - elapsed())
            best_h, best_c5 = targeted_pair_opt(best_h, n, sub_t, rng)

    # Phase 4: Multi-resolution exploration (15% budget)
    t4 = budget_s * 0.85
    if elapsed() < t4:
        for n_try in [150, 180, 220, 250]:
            if elapsed() > t4:
                break
            if n_try == n:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = make_feasible(h_interp, n_try)
            for T in [3000, 15000]:
                if elapsed() > t4:
                    break
                h_new = slsqp_refine(h_interp, n_try, T, 2500)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
                    n = n_try
            if elapsed() < t4 - 5:
                sub_t = min(8.0, t4 - elapsed())
                best_h, best_c5 = local_search(best_h, n, sub_t, rng)

    # Phase 5: Final targeted subgradient + SLSQP polish (10% budget)
    t5 = budget_s * 0.99
    if elapsed() < t5:
        remain = min(t5 - elapsed(), budget_s * 0.05)
        best_h, best_c5 = subgrad_descent(best_h, n, remain, lr_init=0.01)
    if elapsed() < t5:
        for T in [50000, 200000]:
            if elapsed() > t5:
                break
            h_new = slsqp_refine(best_h, n, T, 8000)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Final micro local search
    if elapsed() < budget_s * 0.999:
        sub_t = min(10.0, budget_s * 0.999 - elapsed())
        best_h, best_c5 = local_search(best_h, n, sub_t, rng)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
