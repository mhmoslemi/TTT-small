    if topk is not None:
        res = minimize(smooth_max_obj, h0, args=(T, topk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    else:
        res = minimize(smooth_max_obj, h0, args=(T, None),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(np.clip(res.x, 0, 1), n)

def grad_c5_weighted(h, w):
    n = len(h)
    g = np.zeros(n)
    omh = 1.0 - h
    for l in range(2 * n - 1):
        wl = w[l]
        if wl < 1e-14:
            continue
        s = l - (n - 1)
        if s >= 0:
            g[0:n-s] += wl * omh[s:n]
            g[s:n] -= wl * h[0:n-s]
        else:
            m = -s
            g[m:n] += wl * omh[0:n-m]
            g[0:n-m] -= wl * h[m:n]
    return g

def proj_grad_step(h, n, T, lr, topk=None, steps=500):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(steps):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        if topk is not None and topk < len(corr):
            idx = np.argpartition(corr, -topk)[-topk:]
            w = np.exp(T * (corr[idx] - np.max(corr[idx])))
            w /= w.sum()
            grad = np.zeros(n)
            for j, li in enumerate(idx):
                s = li - (n - 1)
                omh = 1.0 - h
                if s >= 0:
                    grad[0:n-s] += w[j] * omh[s:n]
                    grad[s:n] -= w[j] * h[0:n-s]
                else:
                    m = -s
                    grad[m:n] += w[j] * omh[0:n-m]
                    grad[0:n-m] -= w[j] * h[m:n]
        else:
            w = np.exp(T * (corr - np.max(corr)))
            w /= w.sum()
            grad = grad_c5_weighted(h, w)
        h_new = h - lr * grad
        h_new = project_h(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return h, best_c5

def interpolate_h(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)

def flatten_top(h, n, k=5, iters=20):
    for _ in range(iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        topk = np.argpartition(corr, -k)[-k:]
        tops = corr[topk]
        hi, lo = np.max(tops), np.min(tops)
        if hi - lo < 1e-8:
            break
        spread = hi - lo
        for j, li in enumerate(topk):
            s = li - (n - 1)
            if s >= 0:
                idx_valid = np.arange(n - s)
                overlap_mat = h[idx_valid] * (1 - h[idx_valid + s])
                max_idx = idx_valid[np.argmax(overlap_mat)]
                delta = 0.01 * spread * (tops[j] - lo) / (hi - lo + 1e-12)
                if j == np.argmax(topk):
                    h[max_idx] -= delta
                else:
                    h[max_idx] += delta
            else:
                m = -s
                idx_valid = np.arange(m, n)
                overlap_mat = h[idx_valid] * (1 - h[idx_valid - m])
                max_idx = idx_valid[np.argmax(overlap_mat)]
                delta = 0.01 * spread * (tops[j] - lo) / (hi - lo + 1e-12)
                if j == np.argmax(topk):
                    h[max_idx] -= delta
                else:
                    h[max_idx] += delta
        h = project_h(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n_init = len(best_h)
    best_h = project_h(best_h, n_init)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coarse structural optimization at n=40
    n_coarse = 40
    h_c = interpolate_h(best_h, n_init, n_coarse)
    h_c = project_h(h_c, n_coarse)
    c5_c = compute_c5(h_c)

    if elapsed() < budget_s * 0.20:
        # Multi-start SLSQP at coarse level
        for trial in range(5):
            if elapsed() > budget_s * 0.18:
                break
            h_try = h_c + np.random.uniform(-0.05, 0.05, n_coarse)
            h_try = project_h(h_try, n_coarse)
            for T in [50, 200, 800]:
                if elapsed() > budget_s * 0.19:
                    break
                try:
                    hc = slsqp_step(h_try, T, n_coarse, None, 400, 1e-14)
                    c5v = compute_c5(hc)
                    if c5v < c5_c:
                        c5_c = c5v
                        h_c = hc
                except Exception:
                    pass
            # Topk focus
            for k in [3, 5, 8]:
                if elapsed() > budget_s * 0.19:
                    break
                try:
                    hc = slsqp_step(h_c, 500, n_coarse, k, 400, 1e-14)
                    c5v = compute_c5(hc)
                    if c5v < c5_c:
                        c5_c = c5v
                        h_c = hc
                except Exception:
                    pass

    # Interpolate coarse best back to fine
    h_fine = interpolate_h(h_c, n_coarse, n_init)
    h_fine = project_h(h_fine, n_init)
    c5_fine = compute_c5(h_fine)
    if c5_fine < best_c5:
        best_c5 = c5_fine
        best_h = h_fine

    # Phase 2: Projected gradient with exponential tilting at fine level
    if elapsed() < budget_s * 0.35:
        lr = 0.02
        for T in [50, 150, 500, 1500]:
            if elapsed() > budget_s * 0.35:
                break
            for steps in [300, 500]:
                if elapsed() > budget_s * 0.35:
                    break
                h_pg, c5_pg = proj_grad_step(best_h, n_init, T, lr, None, steps)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg
            # Topk variant
            for k in [5, 10]:
                if elapsed() > budget_s * 0.35:
                    break
                h_pg, c5_pg = proj_grad_step(best_h, n_init, T, lr, k, 300)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg

    # Phase 3: Aggressive SLSQP with topk at fine level
    if elapsed() < budget_s * 0.55:
        for T in [200, 800, 3000, 10000]:
            if elapsed() > budget_s * 0.55:
                break
            for k in [None, 3, 5, 8]:
                if elapsed() > budget_s * 0.55:
                    break
                try:
                    hc = slsqp_step(best_h, T, n_init, k, 800, 1e-15)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
                except Exception:
                    pass

    # Phase 4: Flattening iterations
    if elapsed() < budget_s * 0.65:
        for k in [3, 5, 7, 10]:
            if elapsed() > budget_s * 0.65:
                break
            h_fl = flatten_top(best_h.copy(), n_init, k, 20)
            c5_fl = compute_c5(h_fl)
            if c5_fl < best_c5:
                best_c5 = c5_fl
                best_h = h_fl
            # Polish the flattened result
            try:
                hc = slsqp_step(h_fl, 2000, n_init, k, 500, 1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 5: Basin hopping with structured perturbations
    bh_h = best_h.copy()
    bh_c5 = best_c5
    if elapsed() < budget_s * 0.85:
        for it in range(60):
            if elapsed() > budget_s * 0.85:
                break
            step = 0.08 * (1.0 - it / 60.0) + 0.015
            r = np.random.rand()
            if r < 0.35:
                h_pert = bh_h + step * np.random.randn(n_init)
            elif r < 0.6:
                # Anti-correlation perturbation: flip small blocks
                blen = np.random.randint(3, 8)
                start = np.random.randint(0, n_init - blen)
                h_pert = bh_h.copy()
                h_pert[start:start+blen] = 1.0 - h_pert[start:start+blen]
            elif r < 0.8:
                # Shift perturbation
                h_pert = bh_h.copy()
                shift = np.random.randint(2, 5) * (np.random.rand() < 0.5 and 1 or -1)
                h_pert = np.roll(h_pert, shift)
                h_pert = project_h(h_pert, n_init)
            else:
                # Scale narrow regions
                h_pert = bh_h.copy()
                lo = np.random.randint(0, n_init // 2)
                hi = lo + np.random.randint(5, n_init // 3)
                h_pert[lo:hi] *= 1.0 + step * 2.0 * (np.random.rand() - 0.5)
            h_pert = project_h(h_pert, n_init)
            try:
                k_kick = int(np.random.choice([3, 5, None]))
                T_kick = int(np.random.choice([300, 1000, 3000]))
                hc = slsqp_step(h_pert, T_kick, n_init, k_kick, 300, 1e-13)
                c5v = compute_c5(hc)
                temp = 0.004 * (1.0 - it / 60.0) + 0.0002
                if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
                    bh_c5 = c5v
                    bh_h = hc
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hc
            except Exception:
                pass

    # Phase 6: Coordinate descent on worst shifts
    if elapsed() < budget_s * 0.93:
        for outer in range(100):
            if elapsed() > budget_s * 0.93:
                break
            improved = False
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            top5 = np.argsort(corr)[-5:]
            for k_sh in top5:
                if elapsed() > budget_s * 0.93:
                    break
                s = k_sh - (n_init - 1)
                if s >= 0:
                    idx_valid = np.arange(n_init - s)
                else:
                    idx_valid = np.arange(-s, n_init)
                # Find the index with maximum contribution
                if s >= 0:
                    contrib = best_h[idx_valid] * (1.0 - best_h[idx_valid + s])
                else:
                    m = -s
                    contrib = best_h[idx_valid] * (1.0 - best_h[idx_valid - m])
                j_star = idx_valid[np.argmax(contrib)]
                for delta in [0.02, -0.02, 0.04, -0.04, 0.06, -0.06]:
                    h_try = best_h.copy()
                    h_try[j_star] += delta
                    h_try = project_h(h_try, n_init)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    # Phase 7: Final high-T SLSQP polish
    for T in [5000, 20000, 50000]:
        if elapsed() > budget_s * 0.99:
            break
        try:
            hc = slsqp_step(best_h, T, n_init, None, 1000, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    for k in [3, 5]:
        if elapsed() > budget_s * 0.995:
            break
        try:
            hc = slsqp_step(best_h, 8000, n_init, k, 800, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Final verification
    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
