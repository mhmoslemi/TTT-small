    """Solve the LP subproblem for the cutting plane method."""
    # Variables: h[0], h[1], ..., h[n-1], t
    # Total: n+1 variables
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    
    # For each active lag ℓ, the linearized constraint is:
    # Σ_i h[i+ℓ] - Σ_i (h[i+ℓ]*h[i]_old + h[i+ℓ]_old*h[i]) + Σ_i h[i+ℓ]_old*h[i]_old ≤ t
    # 
    # Rearranging: Σ_i h[i+ℓ] - Σ_i h[i+ℓ]*h[i]_old - Σ_i h[i+ℓ]_old*h[i] + Σ_i h[i+ℓ]_old*h[i]_old - t ≤ 0
    
    # Let me compute the coefficient vector for h and the constant term for each lag.
    
    n_corr = 2*n - 1  # number of correlation values
    A_rows = []
    b_vals = []
    
    for lag_idx in active_lags:
        # lag_idx is the index in the 'full' correlation array
        # The actual lag is ℓ = lag_idx - (n-1)
        ell = lag_idx - (n - 1)
        
        coeff = np.zeros(n)
        const = 0.0
        
        if ell >= 0:
            # c = sum_{i=0}^{n-1-ell} h[i+ell] * (1-h[i])
            for i in range(n - ell):
                # h[i+ell] - h[i+ell]*h[i]
                coeff[i+ell] += 1.0  # from h[i+ell]
                coeff[i+ell] -= h[i]  # from -h[i+ell]*h[i] (linearized: -h[i+ell]*h[i]_old)
                coeff[i] -= h[i+ell]  # from -h[i+ell]_old*h[i]
                const += h[i+ell] * h[i]  # from +h[i+ell]_old*h[i]_old
        else:
            # c = sum_{i=-ell}^{n-1} h[i+ell] * (1-h[i])
            # where i+ell ranges from 0 to n-1-ell
            kp = -ell  # positive offset
            for i in range(kp, n):
                j = i + ell  # j < i
                # h[j] - h[j]*h[i]
                coeff[j] += 1.0  # from h[j] = h[i+ell]
                coeff[j] -= h[i]  # from -h[j]*h[i] (linearized: -h[i+ell]*h[i]_old)
                coeff[i] -= h[j]  # from -h[j]_old*h[i]
                const += h[j] * h[i]  # from +h[j]_old*h[i]_old
        
        # The constraint is: const + coeff @ h - t ≤ 0
        # i.e., coeff @ h - 1*t ≤ -const
        row = np.concatenate([coeff, [-1.0]])
        A_rows.append(row)
        b_vals.append(-const)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Sum constraint: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 ≤ h[i] ≤ 1, 0 ≤ t ≤ 2
    bounds = [(0, 1)] * n + [(0, t_bound)]
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[-1]
    return h, t_bound

        return (m + np.log(np.sum(np.exp(T * s))) / T) * (2.0 / len(x))
    try:
        res = minimize(lse_obj, h, args=(), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def binary_swap_anneal(n, n_iters, h_init, t_start, t_end, rng):
    h = h_init.copy().astype(float)
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    target = n // 2
    n_ones = int(h.sum())
    if n_ones != target:
        idx = np.arange(n)
        rng.shuffle(idx)
        h = np.zeros(n)
        h[idx[:target]] = 1.0
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    curr_c5 = best_c5
    for it in range(n_iters):
        T = t_start * (t_end / t_start) ** (it / max(n_iters, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5, _ = compute_c5(h)
        d = new_c5 - curr_c5
        if d < 0 or rng.rand() < np.exp(-d / max(T, 1e-12)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def perturb_repair(h, n, rng, scale):
    h_p = h + rng.randn(n) * scale
    h_p = project_feasible(h_p, n)
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: SLSQP p-norm escalation (top 3 -> top 5 -> top 8 -> full LSE)
    for (tm, p) in [(3, 8), (3, 20), (5, 10), (5, 20), (8, 10), (10, 8), (15, 6)]:
        if time.time() - t0 > budget_s * 0.12:
            break
        h_try = slsqp_pnorm(best_h, tm, p, maxiter=350)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: LSE with escalating T
    for T in [2000, 10000, 50000, 200000]:
        if time.time() - t0 > budget_s * 0.22:
            break
        h_try = slsqp_lse(best_h, T, maxiter=300)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: Binary simulated annealing
    if time.time() - t0 < budget_s * 0.48:
        remaining = budget_s * 0.48 - (time.time() - t0)
        n_iters = max(2000, int(remaining * 600))
        t_start = (best_c5 - 0.30) * 5
        h_bin, c5_bin = binary_swap_anneal(best_n, n_iters, best_h, t_start, 1e-8, rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    # Phase 4: Polish best binary with SLSQP
    if time.time() - t0 < budget_s * 0.58:
        for (tm, p) in [(3, 20), (5, 20), (8, 10)]:
            if time.time() - t0 > budget_s * 0.58:
                break
            h_try = slsqp_pnorm(best_h, tm, p, maxiter=400)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        h_try = slsqp_lse(best_h, 300000, maxiter=300)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 5: Multi-resolution exploration
    if time.time() - t0 < budget_s * 0.78:
        n_candidates = sorted(set([best_n - 6, best_n - 3, best_n + 3,
                                    best_n + 6, 60, 66, 72, 84, 90, 96]))
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        for n_try in n_candidates:
            if time.time() - t0 > budget_s * 0.78:
                break
            if n_try < 40 or n_try > 120 or n_try % 2 != 0:
                continue
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            # Also try binary version
            h_bin_t = np.zeros(n_try)
            order = np.argsort(h_interp)
            h_bin_t[order[n_try // 2:]] = 1.0
            h_bin_t = project_feasible(h_bin_t, n_try)
            for h_start in [h_interp, h_bin_t]:
                if time.time() - t0 > budget_s * 0.78:
                    break
                h_try = slsqp_pnorm(h_start, 5, 15, maxiter=250)
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
                    best_n = n_try

    # Phase 6: Adversarial perturbation + re-optimize (multiple trials)
    if time.time() - t0 < budget_s * 0.90:
        for scale in [0.03, 0.02, 0.01, 0.005]:
            for trial in range(3):
                if time.time() - t0 > budget_s * 0.90:
                    break
                h_p = perturb_repair(best_h, best_n, rng, scale)
                h_opt = slsqp_pnorm(h_p, 5, 20, maxiter=300)
                h_opt = slsqp_lse(h_opt, 100000, maxiter=200)
                c5_opt, _ = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    # Phase 7: Ultra-fine final polish
    if time.time() - t0 < budget_s * 0.97:
        for T in [50000, 200000, 800000]:
            if time.time() - t0 > budget_s * 0.97:
                break
            h_try = slsqp_lse(best_h, T, maxiter=200)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        h_try = slsqp_pnorm(best_h, 3, 50, maxiter=300)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
