import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, n, t_end, max_sweeps=200, random_order=True):
    dx = 2.0 / n
    corr = corr_full(h).copy()
    best_h = h.copy()
    best_c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved_sweep = 0
        if random_order:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            # Compute coefficient vector for h[i]
            # coef[k + n - 1] = d(corr[k])/d(h[i])
            # For lag k: coef = (1-h[i+k]) [if valid] - h[i-k] [if valid and k!=0]
            # For k=0: coef = 1 - 2*h[i]
            
            # Vectorized computation
            kp = np.arange(-(n-1), n)  # all lags
            jp = i + kp  # i + k
            jm = i - kp  # i - k
            
            valid_p = (jp >= 0) & (jp < n)
            valid_m = (jm >= 0) & (jm < n)
            
            # Build coefficient
            coef = np.zeros(2*n - 1)
            # First term: (1 - h[i+k]) where valid
            coef[valid_p] += (1.0 - h[jp[valid_p]])
            # Second term: -h[i-k] where valid
            # For k=0: i-k = i, so -h[i] (this is the self term)
            # For k≠0: i-k ≠ i (as long as k≠0)
            coef[valid_m] += (-h[jm[valid_m]])
            
            # Solve 1D: min_{t in [0,1]} max_k (corr[k] + t * coef[k])
            def f1d(t):
                return np.max(corr + t * coef)
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            hi_new = res.x
            
            delta = hi_new - h[i]
            if abs(delta) > 1e-12:
                corr += delta * coef
                h[i] = hi_new
                improved_sweep += 1
        
        c5_now = np.max(corr) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        elif improved_sweep == 0:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, n, t_end, max_sweeps=300):
    dx = 2.0 / n
    corr = corr_full(h).copy()
    best_h = h.copy()
    best_c5 = np.max(corr) * dx
    lags = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        order = np.random.permutation(n)
        for i in order:
            jp = i + lags
            jm = i - lags
            valid_p = (jp >= 0) & (jp < n)
            valid_m = (jm >= 0) & (jm < n)
            coef = np.zeros(2 * n - 1)
            coef[valid_p] += (1.0 - h[jp[valid_p]])
            coef[valid_m] += (-h[jm[valid_m]])
            # 1D convex: min_{t in [0,1]} max_k(corr[k] + t*coef[k])
            res = minimize_scalar(
                lambda t: np.max(corr + t * coef),
                bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            hi_new = res.x
            delta = hi_new - h[i]
            if abs(delta) > 1e-12:
                corr += delta * coef
                h[i] = hi_new
                improved += 1
        c5_now = np.max(corr) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        elif improved == 0:
            break
    return best_h, best_c5

        m = np.max(c)
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    def con_func(x):
        return x.sum() - n / 2.0
    res = minimize(obj_func, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_func}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)
