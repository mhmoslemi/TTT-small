import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1:
        return m * dx
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_con(h):
    return h.sum() - len(h) / 2.0

def to_binary(h, n):
    order = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[order[:n // 2]] = 1.0
    return h_b

def sa_round(h0, n, t_limit, T0, T1):
    h = h0.copy()
    best = h0.copy()
    best_val = compute_c5(h0)
    ones_p = np.where(h == 1)[0].copy()
    zeros_p = np.where(h == 0)[0].copy()
    no = len(ones_p)
    nz = len(zeros_p)
    it = 0
    total_iter = max(1, int((t_limit - time.time()) / 3e-5))
    while time.time() < t_limit:
        it += 1
        frac = min(1.0, it / total_iter)
        T = T0 * (T1 / T0) ** frac
        ii = np.random.randint(no)
        jj = np.random.randint(nz)
        i = ones_p[ii]
        j = zeros_p[jj]
        h[i] = 0.0
        h[j] = 1.0
        new_val = compute_c5(h)
        if new_val < best_val:
            best_val = new_val
            best = h.copy()
            ones_p[ii] = j
            zeros_p[jj] = i
        elif T > 1e-15 and np.random.rand() < np.exp(-(new_val - best_val) / T):
            ones_p[ii] = j
            zeros_p[jj] = i
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best, best_val

def slsqp_step(h0, T, maxit):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': sum_con}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_max_obj, h0, args=(T,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxit, 'ftol': 1e-15})
    return make_feasible(np.clip(res.x, 0, 1), n)

def greedy_binary(h0, n, t_limit):
    h = h0.copy()
    val = compute_c5(h)
    improved = True
    while improved and time.time() < t_limit:
        improved = False
        ones_g = np.where(h == 1)[0].copy()
        zeros_g = np.where(h == 0)[0].copy()
        np.random.shuffle(ones_g)
        np.random.shuffle(zeros_g)
        for i in ones_g[:50]:
            if improved:
                break
            for j in zeros_g[:50]:
                h[i] = 0.0
                h[j] = 1.0
                v = compute_c5(h)
                if v < val - 1e-15:
                    val = v
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
    return h, val

def multi_start_binary(n, n_starts, t_limit):
    best_h = None
    best_val = np.inf
    for _ in range(n_starts):
        if time.time() > t_limit:
            break
        h_r = np.zeros(n)
        perm = np.random.permutation(n)
        h_r[perm[:n // 2]] = 1.0
        cr = compute_c5(h_r)
        if cr < best_val:
            best_val = cr
            best_h = h_r.copy()
    return best_h, best_val
