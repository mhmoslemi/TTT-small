import numpy as np
import time

def compute_corr(h, N):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h, N))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, N, rng):
    scores = h.astype(float).copy()
    scores += rng.randn(N) * 0.02
    idx = np.argsort(-scores)
    h_b = np.zeros(N)
    h_b[idx[:N // 2]] = 1.0
    return h_b

def make_golden(N):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    g = np.zeros(N)
    for i in range(N):
        g[i] = 1.0 if (i * (phi - 1.0)) % 1.0 < 0.5 else 0.0
    if g.sum() != N // 2:
        idx = np.argsort(g)
        deficit = int(N // 2 - g.sum())
        g[idx[-deficit:]] = 1.0
    return g

def make_alt(N):
    a = np.zeros(N)
    a[::2] = 1.0
    if a.sum() != N // 2:
        a = np.zeros(N)
        perm = np.arange(N)
        np.random.shuffle(perm)
        a[perm[:N // 2]] = 1.0
    return a

    best = h_b.copy()
    best_pk = cur_pk
    stall = 0
    restarts = 0
    while time.time() < deadline:
        T = max(T1, T0 * (T1 / T0) ** (stall / max_stall))
        r = rng.random()
        if r < 0.55:
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_b[i] = 0.0
            h_b[j] = 1.0
            new_pk = float(np.max(compute_corr(h_b, N)))
            d = new_pk - cur_pk
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-8)):
                cur_pk = new_pk
                stall = 0
                if cur_pk < best_pk:
                    best_pk = cur_pk
                    best = h_b.copy()
            else:
                h_b[i] = 1.0
                h_b[j] = 0.0
                stall += 1
        elif r < 0.8:

            start = rng.randint(0, N - L)

def adam_cont(h, N, deadline, k, lr0):
    best_h = h.copy()
    best_pk = float(np.max(compute_corr(h, N)))
    target = N / 2.0
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    lr = lr0
    center = N - 1
    while time.time() < deadline:
        corr = compute_corr(h, N)
        peak = float(np.max(corr))
        if peak < best_pk:
            best_pk = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        wk = 1.0 / k
        for ai in top:
            L = ai - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wk * (1.0 - h[:N - L])
                grad[:N - L] -= wk * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += wk * (1.0 - h[M:])
                    grad[M:] -= wk * h[:N - M]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 300 == 0:
            lr *= 0.993
    return best_h, best_pk * (2.0 / N)
