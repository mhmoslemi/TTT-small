import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-10:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_topk(h, N, topK):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_raw = float(np.max(corr))
    center = N - 1
    nc = len(corr)
    k = min(topK, nc)
    idx = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(N)
    for ai in idx:
        L = ai - center
        wi = 1.0 / k
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5_raw

def grad_softmax_topk(h, N, tau, topK):

    c5_raw = float(np.max(corr))
    center = N - 1
    nc = len(corr)
    sc = tau * corr
    sc -= sc.max()
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    k = min(topK, nc)
    idx_top = np.argpartition(corr, -k)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 100.0) / 500.0))

    thresh = 0.005

    grad = np.zeros(N)
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5_raw

    best_c5 = c5_cur
    best_h = h.copy()

                M = -L
