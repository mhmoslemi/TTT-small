import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-14:
            return h
        d = target - s
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            return h
        h[mask] += d / mask.sum()
        h = np.clip(h, 0, 1)
    return h

def project_quick(h, n):
    target = n / 2.0
    d = target - h.sum()
    if abs(d) < 1e-14:
        return h
    h = h + d / n
    h = np.clip(h, 0, 1)
    for _ in range(10):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:

        else:

        if not mask.any():

        h = np.clip(h, 0, 1)

                          bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return float(res.x)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        h_work = best_h.copy()

                c5_try = compute_c5(best_h)
                if c5_try < best_c5:
                    best_c5 = c5_try
                return best_h, best_c5

        c5_try = compute_c5(h_work)
        if c5_try < best_c5:
            best_h = h_work
            best_c5 = c5_try
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break
    return best_h, best_c5

def smooth_max(x, T):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(T * (x - m)))) / T

def slsqp_polish(h, n, T, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h_old), n_new)

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            return h
        d = target - s
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            return h
        h = h.copy()
        h[mask] += d / mask.sum()
        h = np.clip(h, 0, 1)
    return h

def coord_opt_fast(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    lags = np.arange(-(n - 1), n)
    coef = np.zeros(2 * n - 1)
    idx1 = i + lags
    v1 = (idx1 >= 0) & (idx1 < n)
    coef[v1] += 1.0 - h[idx1[v1]]
    idx2 = i - lags
    v2 = (idx2 >= 0) & (idx2 < n)
    coef[v2] -= h[idx2[v2]]
    res = minimize_scalar(lambda x: float(np.max(corr_base + x * coef)),
                          bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return float(res.x)

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        h_work = best_h.copy()
        for i in np.random.permutation(n):

                c5 = compute_c5(h_work)
                if c5 < best_c5:
                    best_h, best_c5 = h_work, c5
                return best_h, best_c5
            h_work[i] = coord_opt_fast(h_work, n, i)

        else:

        if stall >= 40:

    return best_h, best_c5

    t0 = time.time()
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except NameError:
        best_h = np.zeros(120); best_h[:60] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Delayed coord descent (fast)
    best_h, best_c5 = cd_delayed(best_h, n, t0+budget_s*0.28, 2000)

    # Phase 2: Per-coordinate refinement
    best_h, best_c5 = cd_percoord(best_h, n, t0+budget_s*0.42, 500)

    # Phase 3: SLSQP polish + per-coord
    t3 = t0 + budget_s*0.52
    if time.time() < t3:
        for T in [50, 200, 800]:
            if time.time() > t3-3: break
            hc = slsqp(best_h, n, T, 400)
            c = compute_c5(hc)
            if c < best_c5: best_h, best_c5 = hc, c
        best_h, best_c5 = cd_percoord(best_h, n, t3, 300)

    # Phase 4: Multi-resolution
    t4 = t0 + budget_s*0.72
    for nt in [120, 144, 200, 240]:
        if time.time() > t4 or nt == n: continue
        hi = interp_h(best_h, nt)
        ts = min(t4, time.time()+budget_s*0.04)
        hs, cs = cd_delayed(hi, nt, ts, 500)
        if cs < best_c5: best_h, best_c5, n = hs, cs, nt

    # Phase 5: Basin hopping
    t5 = t0 + budget_s*0.90
    wh, wc = best_h.copy(), best_c5
    it = 0
    while time.time() < t5:
        it += 1
        cool = max(0.0, 1.0-it/100.0)
        st = 0.01 + 0.05*cool
        hp = project(wh + st*np.random.randn(n), n)
        ts = min(t5, time.time()+max(3, budget_s*0.015))
        hs, cs = cd_percoord(hp, n, ts, 80)
        temp = 0.0005*cool + 0.0001
        if cs < wc or np.random.rand() < np.exp(-(cs-wc)/max(temp,1e-12)):
            wh, wc = hs, cs
        if cs < best_c5:
            best_h, best_c5 = hs.copy(), cs

    t6 = t0 + budget_s*0.97
