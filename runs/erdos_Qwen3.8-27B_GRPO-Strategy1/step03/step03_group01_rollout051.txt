import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, N, L):
    """Exact gradient of corr at lag L w.r.t. h, multiplied by dx."""
    dx = 2.0 / N
    g = np.zeros(N)
    if L == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif L > 0 and L < N:
        g[L:] += dx * (1.0 - h[:N - L])
        g[:N - L] -= dx * h[L:]
    elif L < 0 and -L < N:
        M = -L
        g[:N - M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:N - M]
    return g

def optimize_momentum(h, N, target, deadline, tau0, tau1, tk0, tk1, lr0, max_iter, mom0, mom1):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(N)
    it = 0
    stall = 0
    lr = lr0
    while time.time() < deadline and it < max_iter:
        it += 1
        frac = min(it / max_iter, 1.0)
        tau = tau0 + (tau1 - tau0) * frac
        topK = max(2, int(tk0 + (tk1 - tk0) * frac))
        mom = mom0 + (mom1 - mom0) * frac

        corr = np.correlate(h, 1.0 - h, mode='full')
        center = N - 1
        nc = len(corr)

        # Build weights
        sc = tau * np.asarray(corr)
        sc -= sc.max()
        w_sm = np.exp(sc)
        ss = w_sm.sum()
        if ss > 0:
            w_sm /= ss
        else:
            w_sm = np.zeros(nc)
            w_sm[np.argmax(corr)] = 1.0

        k = min(topK, nc)
        idx_top = np.argsort(corr)[-k:]
        w_top = np.zeros(nc)
        w_top[idx_top] = 1.0 / k

        alpha = min(1.0, max(0.0, (tau - 5.0) / 20.0))
        w = (1.0 - alpha) * w_sm + alpha * w_top

        thresh = 0.004
        active = np.where(w > thresh)[0]

        grad = np.zeros(N)
        for ai in active:
            L = ai - center
            wi = w[ai]
            if L == 0:
                grad += wi * dx * (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                grad[L:] += wi * dx * (1.0 - h[:N - L])
                grad[:N - L] -= wi * dx * h[L:]
            elif L < 0 and -L < N:
                M = -L
                grad[:N - M] += wi * dx * (1.0 - h[M:])
                grad[M:] -= wi * dx * h[:N - M]

        v = mom * v + grad
        h_new = h - lr * v
        h_new = project_sum(h_new, target)

        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            h = h_new.copy()
            stall += 1
            if stall > 100:
                h = best_h.copy()
                v *= 0.3
                lr *= 0.75
                stall = 0
        if lr < 1e-10:
            break
    return best_h, best_c5

def polish_coords(h, N, target, deadline):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(N):
            if time.time() >= deadline:
                break
            orig = best_h[i]
            for delta in (0.0 - orig, 1.0 - orig, -0.02, 0.02, -0.01, 0.01, -0.005, 0.005):
                trial = orig + delta
                trial = min(1.0, max(0.0, trial))
                if abs(trial - orig) < 1e-10:
                    continue
                h = best_h.copy()
                h[i] = trial
                h = project_sum(h, target)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break
    return best_h, best_c5
