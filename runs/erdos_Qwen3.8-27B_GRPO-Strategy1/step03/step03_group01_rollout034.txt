    if abs(diff) < 1e-12:
        return h
    h2 = h - diff / len(h)

def weighted_grad(h, N, tau, topK):
    """Compute gradient with softmax + topK blending."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / N)
    center = N - 1
    nc = len(corr)
    
    sc = tau * corr - tau * corr.max()
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    
    k = max(1, min(topK, nc))
    idx_top = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    
    alpha = min(1.0, max(0.0, (tau - 8.0) / 25.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    
    grad = np.zeros(N)
    thresh = 0.004
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5, corr

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    h2 = h - diff / len(h)
    if h2.min() >= -1e-12 and h2.max() <= 1.0 + 1e-12:
        return np.clip(h2, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def weighted_grad(h, N, tau, topK):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / N)
    center = N - 1
    nc = len(corr)
    sc = tau * corr - tau * corr.max()
    w_sm = np.exp(sc)
    w_sm /= w_sm.sum()
    k = max(1, min(topK, nc))
    idx_top = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 8.0) / 25.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    grad = np.zeros(N)
    thresh = 0.004
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5, corr

def optimize(h, N, target, deadline, tau0, tau1, tk0, tk1, lr0, lr_dec, lr_per, mom):
    best_h = h.copy()
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 2500.0, 1.0)
        tau = tau0 + (tau1 - tau0) * frac
        topK = max(2, int(tk0 + (tk1 - tk0) * frac))
        lr = lr0 * (lr_dec ** (it // lr_per))
        grad, c5_val, _ = weighted_grad(h, N, tau, topK)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        v = mom * v + grad
        h = h - lr * v
        h = project_sum(h, target)
        if lr < 1e-10:
            break
    return best_h, best_c5

def targeted_fix(h, N, target, deadline):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    center = N - 1
    for _ in range(200):
        if time.time() > deadline:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-4:]
        improved = False
        for ti in top_idx:
            L = ti - center
            if abs(L) < 2 or abs(L) >= N - 1:
                continue
            if L > 0:
                overlap = best_h[L:] * (1.0 - best_h[:N - L])
            else:
                M = -L
                overlap = best_h[:N - M] * (1.0 - best_h[M:])
            n_fix = min(10, len(overlap))
            if n_fix == 0:
                continue
            worst = np.argsort(overlap)[-n_fix:]
            for wpos in worst:
                if L > 0:
                    i_small = wpos
                    i_large = wpos + L
                else:
                    M = -L
                    i_large = wpos
                    i_small = wpos + M
                moves = [(0.04, 0.0), (0.0, -0.04), (0.025, -0.025), (0.06, 0.0), (0.0, -0.06)]
                for dm in moves:
                    h_try = best_h.copy()
                    h_try[i_small] = np.clip(h_try[i_small] + dm[0], 0, 1)
                    h_try[i_large] = np.clip(h_try[i_large] + dm[1], 0, 1)
                    h_try = project_sum(h_try, target)
                    c5 = compute_c5(h_try, dx)
                    if c5 < best_c5 - 1e-12:
                        best_h = h_try
                        best_c5 = c5

                if improved:

            if improved:

        if not improved:

    return best_h, best_c5

def snap_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    for _ in range(20):
        if time.time() > deadline:
            break
        near0 = np.where((best_h > 0.005) & (best_h < 0.35))[0]
        near1 = np.where((best_h > 0.65) & (best_h < 0.995))[0]
        candidates = np.concatenate([near0, near1])
        if len(candidates) == 0:
            break
        rng = np.random.RandomState(7 + len(candidates))
        if len(candidates) > 80:
            candidates = candidates[rng.permutation(len(candidates))[:80]]
        improved = False
        for i in candidates:
            if time.time() > deadline:
                break
            tv = 0.0 if best_h[i] < 0.5 else 1.0
            if abs(best_h[i] - tv) < 0.01:
                continue
            h_try = best_h.copy()
            h_try[i] = tv
            h_try = project_sum(h_try, target)
            c5 = compute_c5(h_try, dx)
            if c5 < best_c5 - 1e-12:
                best_h = h_try
                best_c5 = c5
                improved = True
        if not improved:
            break
    return best_h, best_c5

    deadline = t0 + budget_s - 3
