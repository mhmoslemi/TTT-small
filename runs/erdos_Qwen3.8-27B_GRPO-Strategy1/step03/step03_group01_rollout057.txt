            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    """Compute gradient contribution from specific lags in the correlate array."""
    center = N - 1
    grad = np.zeros(N)
    for ki in lag_indices:
        L = ki - center
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:N - L])
            grad[:N - L] -= h[L:]
        else:
            M = -L
            grad[:N - M] += (1.0 - h[M:])
            grad[M:] -= h[:N - M]

    """LSE-smoothed gradient descent with Nesterov momentum."""
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)

    nc = 2 * N - 1
    
    it = 0
    total_expected = 3000

        # Decrease learning rate
        lr = lr_start * (lr_end / lr_start) ** frac

**Faster polish**: Instead of trying 0 and 1 for each coordinate sequentially, we can:
1. Compute the gradient at the current maximum lag(s)
2. For coordinates where grad[i] < -threshold, try setting h[i] to 1
3. For coordinates where grad[i] > threshold, try setting h[i] to 0

Or better: compute the "sensitivity" for each coordinate:
- If we increase h[i] by δ, the change in corr[k] for lag L is:
  - For L ≥ 0: δ*(1-h[i+L]) (if i ≤ N-1-L) - δ*h[i-L] (if i ≥ L)
  - For L < 0 (M = -L): δ*(1-h[i-M]) (if i ≥ M) - δ*h[i+M] (if i ≤ N-1-M)

The change in the max is bounded by max_k |change in corr[k]|. We want to find moves that decrease the max.

Actually, for the "greedy" polish, the simplest correct approach is:
For each i, compute corr_new = corr + delta * sensitivity[i] (where sensitivity[i] is the vector of d(corr[k])/dh[i])
Check if max(corr_new) < current max.

The sensitivity vector for coordinate i has at most 2*(N-1) non-zero entries (one for each lag where i participates). Computing it is O(N), and checking the max of corr_new is O(N). So per coordinate it's O(N), and for all N coordinates it's O(N²). This is much faster than recomputing the full correlate!

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def weighted_grad(h, N, weights):
    center = N - 1
    nc = len(weights)
    grad = np.zeros(N)
    thresh = 1e-10
    active = np.where(weights > thresh)[0]
    for ai in active:
        L = ai - center
        wi = weights[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad
