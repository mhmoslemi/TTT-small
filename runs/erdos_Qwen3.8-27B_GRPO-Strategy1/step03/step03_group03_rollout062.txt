            if total < 1e-14:

            h = h * (target / total)
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    m = np.max(corr)
    thresh = m - 1e-10
    active = np.where(corr >= thresh)[0]
    if len(active) > top_m:
        active = active[np.argsort(corr[active])[-top_m:]]
    elif len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            # overlap_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[j] for j < n-k: (1-h[j+k])
            # d/dh[j] for j >= k: -h[j-k]
            if k == 0:
                grad[:] += 1.0 - 2.0 * h
            else:
                grad[:n-k] += (1.0 - h[k:])
                grad[k:] += -h[:n-k]
        else:

            if kp == 0:
                grad[:] += 1.0 - 2.0 * h
            else:
                grad[kp:] += (1.0 - h[:n-kp])
                grad[:n-kp] += -h[kp:]
    return grad / len(active)

The FBM method works as follows:
1. At iterate h^k, compute the active set A (correlations achieving the max)
2. Solve: min_h max_{a in A} [f_a(h^k) + ∇f_a(h^k)^T (h - h^k) + 1/2 (h-h^k)^T H (h-h^k)]
   where H is some positive definite matrix (often I)
3. This is equivalent to: min_h max_{a in A} [c_a + g_a^T (h-h^k) + 1/2 ||h-h^k||²]
4. Take a step: h^{k+1} = h^k + α * d where d solves the QP

For the QP step with active set A:
min_d max_{a in A} [g_a^T d + 1/2 ||d||²] subject to 0 ≤ h^k + d ≤ 1, sum(h^k + d) = n/2

This can be reformulated as:
min_d t
s.t. g_a^T d + 1/2 ||d||² ≤ t for all a in A
     0 ≤ h^k + d ≤ 1
     sum(d) = 0

This is a convex QP! I can solve it with scipy.

        res = minimize(smooth_powersum, h, args=(p,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h


def make_periodic(n, period, ones_in_period):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < ones_in_period:
            h[i] = 1.0
    return project_feasible(h, n)


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = make_periodic(78, 6, 3)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: FBM subgradient (18%)
    t_limit = t0 + budget_s * 0.18
    best_h, best_c5 = subgrad_fbm(best_h, 5000, t0, t_limit)

    # Phase 2: SLSQP LSE polish (12%)
    if time.time() - t0 < budget_s * 0.30:
        for T in [300, 1500, 8000, 40000]:
            if time.time() - t0 > budget_s * 0.30:
                break
            h_try = slsqp_lse(best_h, T, maxiter=350)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 3: More FBM (12%)
    if time.time() - t0 < budget_s * 0.42:
        t_limit = t0 + budget_s * 0.42
        best_h, best_c5 = subgrad_fbm(best_h, 5000, t0, t_limit)

    # Phase 4: SLSQP power-sum (8%)
    if time.time() - t0 < budget_s * 0.50:
        for p in [40, 150, 800]:
            if time.time() - t0 > budget_s * 0.50:
                break
            h_try = slsqp_ps(best_h, p, maxiter=300)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 5: Multi-n search (35%)
    if time.time() - t0 < budget_s * 0.85:
        t_limit = t0 + budget_s * 0.85
        n_candidates = sorted(set([best_n - 4, best_n + 2, best_n + 4,
                                   best_n + 6, 76, 80, 84, 90, 96, 104]))
        for n_try in n_candidates:
            if time.time() - t0 > t_limit:
                break
            if n_try < 50 or n_try % 2 != 0:
                continue
            if n_try == best_n:
                continue
            h_interp = interpolate_h(best_h, best_n, n_try)
            # Subgradient on new resolution
            h_opt, c5_opt = subgrad_fbm(h_interp, 3000, t0, t_limit)
            # SLSQP polish
            if time.time() - t0 < t_limit:
                h_opt = slsqp_lse(h_opt, 15000, maxiter=250)
                c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 6: Perturbation restarts (10%)
    if time.time() - t0 < budget_s * 0.96:
        t_limit = t0 + budget_s * 0.96
        for trial in range(12):
            if time.time() - t0 > t_limit:
                break
            scale = 0.025 + 0.05 * np.random.rand()
            h_pert = best_h + np.random.randn(best_n) * scale
            h_pert = project_feasible(h_pert, best_n)
            h_opt, c5_opt = subgrad_fbm(h_pert, 1500, t0, t_limit)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 7: Final SLQSLQ + ultra-high T (4%)
    if time.time() - t0 < budget_s * 0.99:
        h_f = slsqp_lse(best_h, 100000, maxiter=500)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        h_p = slsqp_ps(best_h, 2000, maxiter=400)
        c5_p, _ = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, best_n
