    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    half = N // 2
    it = 0
    no_imp = 0
    while time.time() < deadline:
        frac = min(1.0, it / max(1, int((deadline - (deadline - 60)))))
        T = T0 * (T1 / T0) ** min(1.0, it / 50000.0)
        if no_imp > 30000:
            T = max(T, 3.0)
            no_imp = 0
        ii = rng.randint(half)
        jj = rng.randint(half)
        i = int(ones[ii])
        j = int(zeros[jj])
        d = swap_delta(corr, h, i, j, N)
        new_peak = float((corr + d).max())
        dp = new_peak - cur_peak
        acc = dp < 0 or (T > 0.003 and rng.random() < np.exp(-dp / T))
        if acc:
            corr += d
            h[i] = 0
            h[j] = 1
            ones[ii] = j
            zeros[jj] = i
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        it += 1
    return best_h, best_peak * (2.0 / N)

def greedy_polish(h, N, deadline, rng):
    """Only accept improvements; run until no improvement found."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_peak = float(corr.max())
    best_h = h.copy()
    best_peak = cur_peak
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    half = N // 2
    stall = 0
    while time.time() < deadline and stall < 50000:
        ii = rng.randint(half)
        jj = rng.randint(half)
        i = int(ones[ii])
        j = int(zeros[jj])
        d = swap_delta(corr, h, i, j, N)
        new_peak = float((corr + d).max())
        if new_peak < cur_peak - 1e-12:
            corr += d
            h[i] = 0
            h[j] = 1
            ones[ii] = j
            zeros[jj] = i
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_peak * (2.0 / N)

def continuous_refine(h_bin, N, deadline, rng):
    """Refine binary solution with small continuous perturbation + gradient steps."""
    target = N / 2.0
    h = h_bin.astype(np.float64).copy()
    h += rng.randn(N) * 0.03
    h = project_sum(h, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = N - 1
    best_peak = float(corr.max())
    best_h = h.copy()
    lr = 0.02
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    k_top = 60
    it = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        wk = 1.0 / k_top
        for ai in top:
            L = int(ai) - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                if L < N:
                    grad[L:] += wk * (1.0 - h[:N - L])
                    grad[:N - L] -= wk * h[L:]
            else:
                M2 = -L
                if M2 < N:
                    grad[:N - M2] += wk * (1.0 - h[M2:])
                    grad[M2:] -= wk * h[:N - M2]
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 200 == 0:
            lr *= 0.95
    return best_h, best_peak * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=np.float64)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)
    
    # Try multiple resolutions
    sizes = [n0]
    # Add nearby sizes for exploration
    for extra in [n0 - 20, n0 + 20, n0 + 40, n0 + 80]:
        if extra % 2 == 0 and extra >= 200 and extra <= 900 and extra not in sizes:
            sizes.append(extra)
    sizes.sort()
    
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    
    # Allocate time: more for primary size
    time_per_size = {}
    total_w = sum(1.0 / (1.0 + 0.3 * abs(s - n0)) for s in sizes)
    for s in sizes:
        rem = deadline - time.time()
        w = 1.0 / (1.0 + 0.3 * abs(s - n0))
        time_per_size[s] = rem * (w / total_w) * 0.85
    
    for N in sizes:
        if time.time() > deadline - 40:
            break
        remain = deadline - time.time()
        alloc = min(time_per_size.get(N, remain * 0.3), remain - 20)
        if alloc < 30:
            alloc = min(remain - 10, 60)
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_init = np.interp(xN, x0, h0)
        h_init = project_sum(h_init, target)
        
        # Phase 1: Fast binary annealing (main workhorse)
        d1 = time.time() + alloc * 0.45
        h1, c1 = fast_anneal(h_init, N, d1, rng)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        
        # Phase 2: Greedy polish
        if time.time() < deadline - 30:
            d2 = min(time.time() + (deadline - time.time()) * 0.25, deadline - 25)
            h2, c2 = greedy_polish(h1, N, d2, rng)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()
        
        # Phase 3: Second anneal from pol
            if time.time() < deadline - 25:
                d3 = time.time() + (deadline - time.time()) * 0.2
                h3, c3 = fast_anneal(h2, N, d3, rng, T0=2.0, T1=0.0005)
                if c3 < best_c5:
                    best_c5, best_h = c3, h3.copy()
        
        # Phase 4: Continuous refinement
        if time.time() < deadline - 20:
            d4 = deadline - 10
            h4, c4 = continuous_refine(h3 if len(h3) == N else h2, N, d4, rng)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()
    
    # Final: best continuous refinement on the best solution found
    Nb = len(best_h)
    if time.time() < deadline - 10:
        hF, cF = continuous_refine(best_h, Nb, deadline - 2, rng)
        if cF < best_c5:
            best_c5, best_h = cF, hF
    
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
