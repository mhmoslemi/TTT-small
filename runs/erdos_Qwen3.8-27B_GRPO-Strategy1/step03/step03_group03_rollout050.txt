import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_t = np.exp(T * (corr - m))
    return (m + np.log(exp_t.sum()) / T) * dx

def smooth_c5_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_t = np.exp(T * (corr - m))
    w = exp_t / exp_t.sum()
    val = (m + np.log(exp_t.sum()) / T) * dx
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] += -wi * h[k:]
        else:
            kp = -k
            grad[:n - kp] += wi * (1.0 - h[kp:])
            grad[kp:] += -wi * h[:n - kp]
    return val, grad * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_c5_val, h.copy(), args=(T,),
                       jac=lambda x: smooth_c5_obj_and_grad(x, T)[1],
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_vec(h, corr, T_s):
    n = len(h)
    m = np.max(corr)
    exp_t = np.exp(T_s * (corr - m))
    w = exp_t / exp_t.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] += -wi * h[k:]
        else:
            kp = -k
            grad[:n - kp] += wi * (1.0 - h[kp:])
            grad[kp:] += -wi * h[:n - kp]
    return grad

def momentum_opt(h, n_iters, T_s):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = 0.06
    mom = 0.8
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        _, corr = compute_c5(h_cur)
        g = subgrad_vec(h_cur, corr, T_s)
        v = mom * v - lr * g
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.015, 0.4)
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr *= 0.96
            if stall > 600:
                break
    return best_h, best_c5

def coord_opt_1d(h, i):
    n = len(h)
    old = h[i]
    best_t = old
    best_v, _ = compute_c5(h)
    for t in np.concatenate([np.linspace(0, 1, 21), [np.clip(old + d, 0, 1) for d in [0.005, -0.005, 0.02, -0.02]]]):
        h[i] = t
        v, _ = compute_c5(h)
        if v < best_v:
            best_v = v
            best_t = t

            h[i] = t
            v, _ = compute_c5(h)
            if v < best_v:
                best_v = v
                best_t = t
    h[i] = best_t
    return best_v

def coordinate_descent(h, n_sweeps):
    n = len(h)
    h_cur = h.copy()
    best_c5, _ = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            h_cur[i] = 0
            h_cur[i] = h_cur[i]  # noop
            old = h_cur[i]
            h_cur[i] = old
            v = coord_opt_1d(h_cur, i)
            if v < best_c5 - 1e-14:
                best_c5 = v
                improved = True
        h_cur = project_feasible(h_cur, n)
        if not improved:
            break
    return h_cur, best_c5
