def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def full_softmax_gradient(h, N, tau):
    """Compute gradient using full softmax over all correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for k in range(len(w)):
        wi = w[k]
        if wi < 1e-8:
            continue
        d = k - center
        if d == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:N-d] += wi * (1.0 - h[d:])
            grad[d:] -= wi * h[:N-d]
        else:
            M = -d
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, c5_val

def topk_gradient(h, N, tau, K):

    else:

    if ws < 1e-30:

        k = top_idx[ix]

        elif d > 0:
            grad[:N-d] += wi * (1.0 - h[d:])
            grad[d:] -= wi * h[:N-d]
        else:
            M = -d
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, c5_val

def run_opt_phase(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, K, iters):

    best_c5 = compute_c5(h, dx)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline:
        frac = min(it / max(iters, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        grad, c5_val = topk_gradient(h, N, tau, K)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h_new, target)
    return best_h, best_c5

def coord_descent(h, N, target, dx, deadline, max_rounds=50):

    best_c5 = compute_c5(h, dx)
    for _ in range(max_rounds):

import numpy as np

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def sgrad(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5v = float(np.max(corr)) * (2.0 / N)
    nc = len(corr)
    if K >= nc:
        idx = np.arange(nc)
        ct = corr
    else:
        idx = np.argsort(corr)[-K:]
        ct = corr[idx]
    sc = tau * ct
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5v
    w /= ws
    ctr = N - 1
    g = np.zeros(N)
    for j in range(len(w)):
        wt = w[j]
        if wt < 1e-9:

        d = idx[j] - ctr

            g[M:] -= wt * h[:N - M]
