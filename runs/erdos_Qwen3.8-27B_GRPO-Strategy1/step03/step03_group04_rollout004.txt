```python
def full_grad(h, N, eps=0.5):
    """Compute gradient of softmax-smoothed max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    max_corr = np.max(corr)
    weights = np.exp((corr - max_corr) / eps)
    weights /= np.sum(weights)
    
    grad = np.zeros(N)
    # For each shift L (from -(N-1) to N-1):
    for L in range(-(N-1), N):
        if weights[N-1+L] < 1e-10:
            continue
        w = weights[N-1+L]
        if L >= 0:
            # corr[N-1+L] = sum_{t=0}^{N-1-L} h[t+L]*(1-h[t])
            # d/dh[j]:
            # j as t+L: [j-L in [0,N-1-L]] → (1-h[j-L]) if j>=L
            # j as t: [j in [0,N-1-L]] → -h[j+L] if j<=N-1-L
            idx1 = np.arange(L, N)  # j values where j appears as t+L
            grad[idx1] += w * (1.0 - h[idx1 - L])
            idx2 = np.arange(0, N - L)  # j values where j appears as t
            grad[idx2] -= w * h[idx2 + L]
        else:
            M = -L
            # corr[N-1-M] = sum_{t=0}^{N-1-M} h[t]*(1-h[t+M])
            # d/dh[j]:
            # j as t: [j in [0,N-1-M]] → (1-h[j+M])
            # j as t+M: [j-M in [0,N-1-M]] → +h[j-M] if j>=M
            idx1 = np.arange(0, N - M)  # j values where j appears as t
            grad[idx1] += w * (1.0 - h[idx1 + M])
            idx2 = np.arange(M, N)  # j values where j appears as t+M
            grad[idx2] += w * h[idx2 - M]
    
    return grad, max_corr
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_topk(h, N, k=20, eps=1.0):
    """Compute gradient using softmax-weighted top-k correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top-k
    idx = np.argpartition(corr, -k)[-k:]
    vals = corr[idx]
    # Softmax weights
    w = np.exp((vals - np.max(vals)) / eps)
    w /= np.sum(w)
    
    grad = np.zeros(N)
    center = N - 1
    for ii in range(k):
        ai = idx[ii]
        L = int(ai) - center
        wi = w[ii]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            # corr at shift L: sum_{t=0}^{N-1-L} h[t+L]*(1-h[t])
            # d/dh[j] = (1-h[j-L]) for j in [L,N-1] minus h[j+L] for j in [0,N-1-L]
            grad[L:] += wi * (1.0 - h[:N-L])
            grad[:N-L] -= wi * h[L:]
        else:
            M = -L
            # corr at shift -M: sum_{t=0}^{N-1-M} h[t]*(1-h[t+M])
            # d/dh[j] = (1-h[j+M]) for j in [0,N-1-M] plus h[j-M] for j in [M,N-1]
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] += wi * h[:N-M]
    peak = float(np.max(corr))
    return grad, peak

def adam_projected(h, N, target, deadline, k, eps, lr0, decay):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    lr = lr0
    b1, b2, aeps = 0.8, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    stuck = 0
    while time.time() < deadline:
        grad, peak = gradient_topk(h, N, k, eps)
        if peak < best_peak - 1e-10:
            best_peak = peak
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1**it)
        vh = v / (1.0 - b2**it)
        h_new = h - lr * mh / (np.sqrt(vh) + aeps)
        h_new = project_sum(h_new, target)
        # Accept if it doesn't increase the peak too much
        new_peak = float(np.max(compute_corr(h_new)))
        if new_peak < peak + 0.5:  # allow some increase
            h = h_new
        else:
            h = h  # reject
        if it % 100 == 0:
            lr *= decay
        if stuck > 2000:
            break
    return best_h, best_peak * (2.0 / N)

def binary_local_search(h_b, N, deadline, max_rounds=50):
    """Best-improvement local search on binary sequence."""
    best = h_b.copy()
    best_pk = float(np.max(compute_corr(best)))
    improved = True
    rounds = 0
    while improved and time.time() < deadline and rounds < max_rounds:
        improved = False
        rounds += 1
        corr = compute_corr(best)
        cur_max = float(np.max(corr))
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        best_delta = 0
        best_move = None
        # Try all moves (sample if too many)
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_candidates = n_ones * n_zeros
        if n_candidates > 20000:
            # Sample
            n_try = 20000
            for _ in range(n_try):
                i = ones[np.random.randint(n_ones)]
                j = zeros[np.random.randint(n_zeros)]
                best[i] = 0
                best[j] = 1
                new_pk = float(np.max(compute_corr(best)))
                best[i] = 1
                best[j] = 0
                delta = new_pk - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_move = (i, j)
        else:
            for i in ones:
                for j in zeros:
                    best[i] = 0
                    best[j] = 1
                    new_pk = float(np.max(compute_corr(best)))
                    best[i] = 1
                    best[j] = 0
                    delta = new_pk - cur_max
                    if delta < best_delta:
                        best_delta = delta
                        best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            best[i] = 0
            best[j] = 1
            best_pk = float(np.max(compute_corr(best)))
            improved = True
    return best, best_pk * (2.0 / N)
```
