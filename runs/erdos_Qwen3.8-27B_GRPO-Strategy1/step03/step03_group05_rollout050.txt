            if total < 1e-14:

        else:

            total = room.sum()
            if total < 1e-14:

            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)

def obj_softmax(h, T, n, pen):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = (m + np.log(np.exp(np.clip(T*(corr-m), 0, 500)).sum()) / T) * dx
    integral = h.sum() / (n/2.0)  # should be 1
    val += pen * (integral - 1.0)**2

    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    top = np.sort(corr)[::-1][:k]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_softmax(h, T, n, pen):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    logged = np.clip(T * (corr - m), 0, 500)
    val = (m + np.log(np.exp(logged).sum()) / T) * dx
    integral = h.sum() / (n / 2.0)
    val += pen * (integral - 1.0) ** 2
    return float(val)

def obj_topk(h, k, n, pen):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    val = (top.sum() / k) * dx
    integral = h.sum() / (n / 2.0)
    val += pen * (integral - 1.0) ** 2
    return float(val)

def obj_soft_topk(h, k, n, pen, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    val = np.dot(w, corr) * dx
    integral = h.sum() / (n / 2.0)
    val += pen * (integral - 1.0) ** 2
    return float(val)

def lbfgs(h, obj_func, args, maxiter=800, maxfun=20000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h, args=args, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                             'gtol': 1e-10, 'maxfun': maxfun})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def cheb_eval(x, K):
    t = 2.0 * x - 1.0
    n_pts = len(x)
    T = np.zeros((n_pts, K + 1))
    T[:, 0] = 1.0
    if K >= 1:
        T[:, 1] = t
    for k in range(2, K + 1):
        T[:, k] = 2.0 * t * T[:, k - 1] - T[:, k - 2]
    return T

def cheb_optimize(h0, n, K, T_smooth, pen, maxiter):
    x = np.linspace(0, 2, n)
    T = cheb_eval(x, K)
    coeffs, _, _, _ = np.linalg.lstsq(T, h0, rcond=None)

    def obj_c(c):
        h = T @ c
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        h = h + (target - h.sum()) / n
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        m = np.max(corr)
        val = (m + np.log(np.exp(np.clip(T_smooth * (corr - m), 0, 500)).sum()) / T_smooth) * dx
        val += pen * (h.sum() / target - 1.0) ** 2
        return float(val)

    try:
        res = minimize(obj_c, coeffs, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxfun': 5000})
        h_opt = np.clip(T @ res.x, 0, 1)
        return project_feasible(h_opt, n)
    except Exception:
        return h0

def conflict_push(h, n, top_k=5, strength=0.05):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    h_new = h.copy()
    for ki in top_idx:
        s = int(ki - (n - 1))
        if s == 0 or abs(s) >= n:
            continue
        sa = abs(s)
        limit = n - sa
        for i in range(limit):
            if s > 0:
                a, b = i, i + s
            else:
                a, b = i + sa, i
            if h_new[a] > 0.35 and h_new[b] < 0.35:
                d = min(strength, (h_new[a] - 0.15) * 0.25, (0.85 - h_new[b]) * 0.25)

                h_new[b] += d

                d = min(strength, (h_new[b] - 0.15) * 0.25, (0.85 - h_new[a]) * 0.25)
                h_new[b] -= d
                h_new[a] += d
    return project_feasible(h_new, n)

def coordinate_refine(h, n, rounds=3, delta=0.02):
    best_c = compute_c5(h)
    for _ in range(rounds):
        improved = False
        for i in range(n):
            for d in [delta, -delta]:
                h_try = h.copy()
                h_try[i] = np.clip(h[i] + d, 0, 1)
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-10:
                    h = h_try
                    best_c = c_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c
