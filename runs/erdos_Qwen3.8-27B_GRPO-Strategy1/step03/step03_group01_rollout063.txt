    wt = np.zeros(nc)
    wt[itop] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 8.0) / 60.0))
    w = (1.0 - alpha) * ws + alpha * wt
    g = np.zeros(N)
    idx = np.where(w > 1e-5)[0]
    for ai in idx:
        L = ai - center
        wi = w[ai]
        if L == 0:
            g += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            g[:N - L] += wi * (1.0 - h[L:])
            g[L:] -= wi * h[:N - L]
        elif -N < L < 0:
            M = -L
            g[M:] += wi * (1.0 - h[:N - M])
            g[:N - M] -= wi * h[M:]
    return g, c5

def adam_phase(h, N, target, deadline, tau_lo, tau_hi, k_lo, k_hi, lr0):
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    best_local = h.copy()
    best_c5_local = compute_c5(h, 2.0 / N)
    est_iters = max(20, int((deadline - time.time()) * 2.5))
    while time.time() < deadline:
        t += 1
        fr = min(1.0, t / est_iters)
        tau = tau_lo + (tau_hi - tau_lo) * fr
        topK = int(k_lo + (k_hi - k_lo) * fr)
        lr_t = lr0 * max(0.05, 0.5 * (1.0 + np.cos(np.pi * fr)))
        g, c5v = wgrad(h, N, tau, topK)
        if c5v < best_c5_local - 1e-14:
            best_c5_local = c5v
            best_local = h.copy()
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        b1c = 1.0 - 0.9 ** t
        b2c = 1.0 - 0.999 ** t
        h = h - lr_t * (m / b1c) / (np.sqrt(v / b2c) + 1e-8)
        h = project_sum(h, target)
    return best_local, best_c5_local

def focal_polish(h, N, target, deadline, lr=0.008):
    """FOCAL-style: clip gradient, take step, project."""
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / N)
    dx = 2.0 / N
    t = 0
    while time.time() < deadline:
        t += 1
        tau = 300.0 + 30.0 * t
        g, c5v = wgrad(h, N, tau, 2)
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = h.copy()
        gc = np.clip(g, -1.0, 1.0)
        lr_t = lr * max(0.1, 1.0 - t / 500.0)
        h = h - lr_t * gc
        h = project_sum(h, target)
    return best_h, best_c5

def coord_polish(h, N, target, deadline):
    """Try setting coordinates to 0/1 or small nudges, keep improvements."""
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    passes = 0
    while improved and time.time() < deadline and passes < 30:
        improved = False
        passes += 1
        for j in range(N):
            if time.time() > deadline:
                break
            h0 = best_h[j]
            tried = False
            for trial in (0.0, 1.0):
                if abs(trial - h0) < 0.01:
                    continue
                best_h[j] = trial
                best_h = project_sum(best_h, target)
                c5v = compute_c5(best_h, dx)
                if c5v < best_c5 - 1e-12:
                    best_c5 = c5v
                    improved = True
                    tried = True
                    break
                best_h[j] = h0
                best_h = project_sum(best_h, target)
            if not tried and h0 > 0.05 and h0 < 0.95:
                for nd in (0.008, -0.008):
                    tv = h0 + nd
                    if tv < 0 or tv > 1:
                        continue
                    best_h[j] = tv
                    best_h = project_sum(best_h, target)
                    c5v = compute_c5(best_h, dx)
                    if c5v < best_c5 - 1e-12:
                        best_c5 = c5v
                        improved = True
                        break
                    best_h[j] = h0
                    best_h = project_sum(best_h, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    x_par = np.linspace(0.0, 2.0, n_par, endpoint=False)
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_par)

    sizes = [400, 480, 560]
    remaining = deadline - time.time()
    opt_budget = remaining * 0.68
    per_size = opt_budget / len(sizes)

    for N in sizes:
        sz_end = min(time.time() + per_size, deadline - remaining * 0.28)
        if time.time() > sz_end - 50:
            break
        dx = 2.0 / N
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_base = np.interp(xN, x_par, h_par)
        h_base = project_sum(h_base, target)

        n_att = 3
        per_att = (sz_end - time.time() - 10) / n_att
        if per_att < 35:
            n_att = max(1, int((sz_end - time.time() - 10) / 35))

        for at in range(n_att):
            a_end = time.time() + per_att
            if a_end > sz_end - 5:
                a_end = sz_end - 5
            if time.time() > a_end - 30:
                break
            if at == 0:
                h = h_base.copy()
            else:
                h = h_base + rng.randn(N) * (0.025 + 0.015 * at)
                h = project_sum(h, target)

            # Phase 1: broad
            p1_end = time.time() + (a_end - time.time()) * 0.38
            h, c1 = adam_phase(h, N, target, p1_end, 5.0, 40.0, 25, 3, 0.022)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h.copy()

            # Phase 2: focus
            p2_end = time.time() + (a_end - time.time()) * 0.38
            h, c2 = adam_phase(h, N, target, p2_end, 40.0, 250.0, 4, 2, 0.006)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h.copy()

            # Phase 3: very focused
            h, c3 = adam_phase(h, N, target, a_end, 250.0, 1000.0, 2, 1, 0.0015)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h.copy()

    # Polishing on best solution found
    pol_budget = deadline - time.time()
    if pol_budget > 30:
        Nb = len(best_h)
        target_b = Nb / 2.0
        dx_b = 2.0 / Nb

        # FOCAL polish
        f_end = time.time() + pol_budget * 0.45
        if f_end < deadline - 10:
            best_h, best_c5 = focal_polish(best_h, Nb, target_b, f_end, lr=0.006)

        # Adam polish with very high tau
        a_end = time.time() + pol_budget * 0.30
        if time.time() < deadline - 10:
            best_h, c_adam = adam_phase(best_h, Nb, target_b, a_end, 400.0, 1500.0, 2, 1, 0.001)
            if c_adam < best_c5:
                best_c5 = c_adam

        # Coordinate polishing
        c_end = min(time.time() + pol_budget * 0.25, deadline - 3)
        if time.time() < c_end - 5:
            best_h, best_c5 = coord_polish(best_h, Nb, target_b, c_end)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
