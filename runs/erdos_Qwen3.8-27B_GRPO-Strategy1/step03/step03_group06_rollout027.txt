            except Exception:
                pass

    # Phase 3: Basin hopping with gradient-informed perturbations
    n_cur = len(best_h)
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = 70
    for it in range(n_bh):
        if elapsed() > budget_s * 0.78:
            break
        step = 0.07 * (1.0 - 2.0 * it / n_bh) + 0.01
        r = np.random.rand()
        if r < 0.35:
            h_pert = grad_informed_perturb(bh_h, step * 3, n_cur)
        elif r < 0.65:
            h_pert = bh_h + step * np.random.randn(n_cur)
        else:
            blen = np.random.randint(4, max(5, n_cur // 5))
            start = np.random.randint(0, n_cur - blen)
            h_pert = bh_h.copy()
            h_pert[start:start + blen] += step * (2.0 * np.random.rand(blen) - 1.0)
        h_pert = project_feasible(h_pert, n_cur)
        try:
            kk = np.random.choice([3, 5, None])
            hc = slsqp_opt(h_pert, 1000, kk, 250, 1e-12)
            c5v = compute_c5(hc)
            temp = 0.003 * (1.0 - it / n_bh) + 0.0001
            if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
                bh_h = hc
                bh_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass

    # Phase 4: Coordinate descent on worst shifts
    if elapsed() < budget_s * 0.88:
        h_cd = coord_targeted(best_h, len(best_h), 150, 4)
        c5v = compute_c5(h_cd)
        if c5v < best_c5:
            best_h = h_cd
            best_c5 = c5v

    # Phase 5: Nelder-Mead on free variables
    if elapsed() < budget_s * 0.93:
        h_nm = nm_optimize(best_h, len(best_h), 3000)
        c5v = compute_c5(h_nm)
        if c5v < best_c5:
            best_h = h_nm
            best_c5 = c5v

    # Phase 6: Final SLSQP polish (high T, small k)
    n_cur = len(best_h)
    final_params = [(5000, 3, 800), (8000, 3, 800), (10000, 5, 600), (20000, 3, 500), (30000, 3, 400)]
    for T, k, mi in final_params:
        if elapsed() > budget_s * 0.985:
            break
        try:
            hc = slsqp_opt(best_h, T, k, mi, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h = hc
                best_c5 = c5v
        except Exception:
            pass

    # Phase 7: Tiny perturbation + re-optimize (escape ultra-local minima)
    if elapsed() < budget_s * 0.995:
        for trial in range(4):
            h_tiny = best_h + 0.003 * np.random.randn(len(best_h))
            h_tiny = project_feasible(h_tiny, len(best_h))
            try:
                hc = slsqp_opt(h_tiny, 5000, 3, 400, 1e-14)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_h = hc
                    best_c5 = c5v
            except Exception:
                pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
