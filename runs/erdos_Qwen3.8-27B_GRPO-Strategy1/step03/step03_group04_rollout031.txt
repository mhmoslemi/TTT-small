    """Subgradient using softmax-weighted peaks."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    weights = np.exp(alpha * (corr - m))
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.zeros_like(corr)
        weights[m.argmax()] = 1.0
        wsum = 1.0
    else:
        weights /= wsum
    peak = float(m)
    grad = np.zeros(N)
    center = N - 1
    for ai in range(len(corr)):
        w = weights[ai]
        if w < 1e-6:
            continue
        L = ai - center
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            grad[L:] += w * (1.0 - h[:N - L])
            grad[:N - L] -= w * h[L:]
        elif L < 0:
            M = -L
            if M < N:
                grad[:N - M] += w * (1.0 - h[M:])
                grad[M:] -= w * h[:N - M]
    return grad, peak

def continuous_opt(h, N, target, deadline, lr0=15.0, alpha=8.0):
    """Adam-based continuous optimization with softmax subgradient."""
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1 - h, mode='full')))
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    no_improve = 0
    while time.time() < deadline:
        grad, peak = grad_smooth(h, N, alpha)
        if peak < best_peak - 1e-13:
            best_peak = peak
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if no_improve > 150:
            lr *= 0.85
            no_improve = 0
            if lr < 0.05:
                break
    return best_h, best_peak * (2.0 / N)

def binary_anneal(h, N, deadline, rng, T0=15.0):
    """Simulated annealing on binary vectors with N/2 ones."""
    dx = 2.0 / N
    target_ones = N // 2
    h_b = (h > 0.5).astype(np.float64)
    n_ones = int(h_b.sum())
    if n_ones > target_ones:
        idx = np.where(h_b == 1)[0]
        h_b[idx[target_ones:]] = 0
    elif n_ones < target_ones:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:target_ones - n_ones]] = 1

    best = h_b.copy()
    best_pk = float(np.max(np.correlate(h_b, 1 - h_b, mode='full')))
    cur_pk = best_pk
    T1 = 0.001
    it = 0
    no_improve = 0
    patience = 3000

    while time.time() < deadline:
        it += 1
        progress = min(it / 50000.0, 1.0)
        T = T0 * (T1 / T0) ** progress
        if no_improve > patience:
            T = max(T, 3.0)
            no_improve = 0
            patience = max(patience // 2, 500)

        h_save = h_b.copy()
        r = rng.random()
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if r < 0.45:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_b[i], h_b[j] = h_b[j], h_b[i]
        elif r < 0.75 and len(ones) >= 2 and len(zeros) >= 2:
            ii = rng.choice(len(ones), 2, replace=False)
            jj = rng.choice(len(zeros), 2, replace=False)
            i1, i2 = ones[ii[0]], ones[ii[1]]
            j1, j2 = zeros[jj[0]], zeros[jj[1]]
            h_b[i1], h_b[j1] = h_b[j1], h_b[i1]
            h_b[i2], h_b[j2] = h_b[j2], h_b[i2]
        else:
            k = min(rng.randint(2, 5), len(ones), len(zeros))
            ii = rng.choice(len(ones), k, replace=False)
            jj = rng.choice(len(zeros), k, replace=False)
            for a, b in zip(ones[ii], zeros[jj]):
                h_b[a], h_b[b] = h_b[b], h_b[a]

        corr = np.correlate(h_b, 1 - h_b, mode='full')
        new_pk = float(np.max(corr))
        delta = new_pk - cur_pk

        if delta < 0 or (T > 0.005 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_b[:] = h_save
            no_improve += 1

    return best, best_pk * dx

def make_binary_start(N, rng, mode=0):
    """Create diverse binary starting points."""
    h = np.zeros(N)
    if mode == 0:
        idx = rng.choice(N, N // 2, replace=False)
        h[idx] = 1.0
    elif mode == 1:
        # Alternating with noise
        h[::2] = 1.0
        for _ in range(N // 20):
            i = rng.randint(N)
            h[i] = 1 - h[i]
        # Fix count
        n1 = int(h.sum())
        if n1 > N // 2:
            idx = np.where(h == 1)[0]
            h[idx[rng.choice(N // 2 - n1, N // 2 - n1, replace=False)]] = 0
        elif n1 < N // 2:
            idx = np.where(h == 0)[0]
            h[idx[:N // 2 - n1]] = 1
    elif mode == 2:
        # Clumped
        block = max(2, N // 8)
        for s in range(0, N, block * 2):
            h[s:s + block] = 1.0
        h = (h > 0.5).astype(float)
        n1 = int(h.sum())
        if n1 > N // 2:
            idx = np.where(h == 1)[0]
            h[idx[rng.choice(n1 - N // 2, n1 - N // 2, replace=False)]] = 0
        elif n1 < N // 2:
            idx = np.where(h == 0)[0]
            h[idx[:N // 2 - n1]] = 1
    else:
        # Gaussian-ish density
        x = np.linspace(0, 1, N)
        density = 1.0 + 0.8 * np.sin(2 * np.pi * 1.5 * x + rng.rand() * 2 * np.pi)
        thr = np.percentile(density, 50.0)
        h = (density > thr).astype(float)
        n1 = int(h.sum())
        if n1 > N // 2:
            idx = np.where(h == 1)[0]
            h[idx[rng.choice(n1 - N // 2, n1 - N // 2, replace=False)]] = 0
        elif n1 < N // 2:
            idx = np.where(h == 0)[0]
            h[idx[:N // 2 - n1]] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)

    # Phase 1: Initial continuous optimization at original N (10% time)
    d1 = t0 + (deadline - t0) * 0.10
    if d1 > t0 + 5:
        h1, c1 = continuous_opt(h0.copy(), n0, n0 / 2.0, d1, lr0=20.0, alpha=6.0)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary local search at multiple N values (65% time)
    d2_end = t0 + (deadline - t0) * 0.75
    N_list = [380, 460, 540]
    for N in N_list:
        if time.time() > d2_end - 20:
            break
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        # Interpolate best solution
        h_init = np.interp(xN, x0, best_h)
        h_init = project_sum(h_init, target)

        # Quick continuous to get near-binary
        d_qs = time.time() + (d2_end - time.time()) * 0.06
        if d_qs > time.time() + 3:
            h_init, _ = continuous_opt(h_init, N, target, d_qs, lr0=40.0, alpha=3.0)

        # Multiple binary anneal restarts
        d_remain = d2_end - time.time()
        n_restarts = 3 if d_remain > 100 else 2
        for rs in range(n_restarts):
            if time.time() > d2_end - 10:
                break
            if rs == 0:
                h_start = h_init.copy()
            else:
                h_start = make_binary_start(N, np.random.RandomState(seed + rs * 17 + N), mode=rs % 4)
            d_rs = time.time() + (d_remain / (n_restarts - rs)) * 0.9
            h_b, c_b = binary_anneal(h_start, N, d_rs, rng, T0=12.0)
            if c_b < best_c5:
                best_c5, best_h = c_b, h_b

    # Phase 3: Continuous refinement of best (25% time)
    d3_start = time.time()
    d3_end = deadline - 5
    if d3_end > d3_start + 5:
        Nb = len(best_h)
        target_b = Nb / 2.0
        # Refine the best solution found
        h_ref = best_h.astype(float).copy()
        h_ref = project_sum(h_ref, target_b)
        h_ref, c_ref = continuous_opt(h_ref, Nb, target_b, d3_end, lr0=8.0, alpha=10.0)
        if c_ref < best_c5:
            best_c5, best_h = c_ref, h_ref

        # Also try perturbing and re-optimizing
        if time.time() < d3_end - 15:
            h_pert = best_h.astype(float).copy() + rng.randn(Nb) * 0.06
            h_pert = project_sum(h_pert, target_b)
            h_pert, c_pert = continuous_opt(h_pert, Nb, target_b, d3_end, lr0=5.0, alpha=12.0)
            if c_pert < best_c5:
                best_c5, best_h = c_pert, h_pert

    # Final validation
    Nf = len(best_h)
    final_c5 = compute_c5(best_h, Nf)
    return best_h, final_c5, Nf
