def compute_c5(h, dx):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau, K=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = len(corr)
    if K is not None and K < M:
        idx = np.argsort(corr)[-K:]
        c = corr[idx]
        mask = np.zeros(M, dtype=bool)
        mask[idx] = True
    else:
        idx = np.arange(M)
        c = corr
        mask = None
    cx = tau * (c - c.max())
    w = np.exp(cx)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), float(np.max(corr)) * (2.0/N)
    w = w / ws
    
    grad = np.zeros(N)
    center = N - 1
    for ii in range(len(idx)):
        wi = w[ii]
        if wi < 1e-9:
            continue

            grad += wi * tmp

        if c < best_c5 - 1e-13:
            best_c5 = c
            best_h = h2
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_gradient(h, N, weights):
    """Compute gradient given precomputed soft-max weights over correlation lags."""
    grad = np.zeros(N)
    center = N - 1
    nlags = len(weights)
    for ii in range(nlags):
        wi = weights[ii]
        if wi < 1e-9:
            continue
        L = ii - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            tmp = np.zeros(N)
            tmp[:N - L] = 1.0 - h[L:]
            tmp[L:] -= h[:N - L]
            grad += wi * tmp
        else:
            M = -L
            tmp = np.zeros(N)
            tmp[M:] = 1.0 - h[:N - M]
            tmp[:N - M] -= h[M:]
            grad += wi * tmp
    return grad

def get_weights(corr, tau, K):
    M = len(corr)
    if K is not None and K < M:
        idx = np.argsort(corr)[-K:]
        c = corr[idx]
        offset = tau * (c - c.max())
        w = np.exp(offset)
        ws = w.sum()
        if ws < 1e-30:
            return np.zeros(M)
        w /= ws
        full_w = np.zeros(M)
        full_w[idx] = w
        return full_w
    else:
        cx = tau * (corr - corr.max())
        w = np.exp(cx)
        ws = w.sum()
        if ws < 1e-30:
            return np.zeros(M)
        return w / ws

def get_weights_topk_sum(corr, K, temperature):
    """Soft uniform over top-K lags with temperature control."""
    M = len(corr)
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    cx = c / temperature
    cx -= cx.max()
    w = np.exp(cx)
    w /= w.sum()
    full_w = np.zeros(M)
    full_w[idx] = w
    return full_w

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m1 = np.zeros(N)
    v1 = np.zeros(N)
    for it in range(iters):
        if time.time() > deadline:
            break
        frac = min(float(it) / max(iters, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr)) * (2.0 / N)
        if c5v < best_c5 - 1e-13:
            best_c5 = c5v
            best_h = h.copy()
        if frac < 0.4:
            w = get_weights(corr, tau, K)
        else:
            w = get_weights(corr, tau, K2)
        g = corr_gradient(h, N, w)
        it2 = it + 1
        m1 = 0.9 * m1 + 0.1 * g
        v1 = 0.999 * v1 + 0.001 * g * g
        mh = m1 / (1.0 - 0.9 ** it2)
        vh = v1 / (1.0 - 0.999 ** it2)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + 1e-8), target)
    return best_h, best_c5
