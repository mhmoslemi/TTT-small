import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_topk(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k is not None and k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        topk = corr[idx]
    else:
        topk = corr
    m = np.max(topk)
    val = m + np.log(np.sum(np.exp(T * (topk - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-14:
            return h
        h += r * (diff / t)
    else:
        r = h.copy()
        t = r.sum()
        if t < 1e-14:
            return h
        h += r * (diff / t)
    return np.clip(h, 0.0, 1.0)

def slsqp(h0, T, k, n, maxit=500):
    def f(x):
        return smooth_topk(x, T, k)
    def g(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': g}]
    res = minimize(f, h0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxit, 'ftol': 1e-15})
    return project(np.clip(res.x, 0, 1), n)

def lbfgs_pen(h0, T, k, n, maxit=2000, pen=500.0):
    def f(x):
        h = project(x, n)
        return smooth_topk(h, T, k) + pen * (h.sum() - n / 2.0) ** 2
    bnds = [(0.0, 1.0)] * n
    res = minimize(f, h0, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-11, 'maxls': 50})
    return project(np.clip(res.x, 0, 1), n)

def interpolate_to(h_src, n_src, n_dst):
    xo = np.linspace(0, 2, n_src, endpoint=False)
    xn = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xn, xo, h_src)
