            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, peak

def adam_opt(h, N, target, deadline, k_seq, lr0, smooth_seq=None):
    """Adam with scheduled k and smoothness."""
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    n_steps_per_phase = 500
    phase = 0
    lr = lr0
    smooth = smooth_seq[0] if smooth_seq else 5.0
    while time.time() < deadline and phase < len(k_seq):
        k = k_seq[phase]
        for step in range(n_steps_per_phase):
            if time.time() > deadline:
                break
            grad, peak = subgrad_topk(h, N, k, smooth)
            if peak < best_peak - 1e-12:
                best_peak = peak
                best_h = h.copy()
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad * grad
            it += 1
            mh = m / (1.0 - b1 ** it)
            vh = v / (1.0 - b2 ** it)
            h = h - lr * mh / (np.sqrt(vh) + eps)
            h = project_sum(h, target)
            lr = lr0 * (0.5) ** (step / n_steps_per_phase)
        phase += 1
        if phase < len(k_seq) and smooth_seq:
            smooth = smooth_seq[min(phase, len(smooth_seq)-1)]
        # Binarize between phases
        h_bin = to_binary(h, N)
        bp = float(np.max(compute_corr(h_bin)))
        if bp < best_peak:
            best_peak = bp
            best_h = h_bin.copy()
        h = h_bin + 0.05 * (np.random.rand(N) - 0.5)
        h = project_sum(h, target)
        m[:] = 0
        v[:] = 0
    return best_h, best_peak * (2.0 / N)

def binaryhill(h, N, deadline, rng, n_peaks=5):
    """Binary local search focusing on peak lags."""
    dx = 2.0 / N
    cur = to_binary(h, N)
    best = cur.copy()
    corr = compute_corr(cur).copy()
    cur_pk = float(np.max(corr))
    best_pk = cur_pk
    center = N - 1
    no_improve = 0
    patience = 500
    while time.time() < deadline:
        # Get top lags
        top_idx = np.argpartition(corr, -n_peaks)[-n_peaks:]
        # Weighted random selection
        top_vals = corr[top_idx]
        probs = np.exp(3*(top_vals - top_vals.max()))
        probs /= probs.sum()
        lag_i = top_idx[rng.choice(n_peaks, p=probs)]
        L = int(lag_i) - center
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try a swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_corr = compute_corr(cur)
        new_pk = float(np.max(new_corr))
        delta = new_pk - cur_pk
        if delta <= 0:
            cur_pk = new_pk
            corr = new_corr
            no_improve = 0
            if new_pk < best_pk:
                best_pk = new_pk
                best = cur.copy()
        else:
            # Accept with probability based on "temperature"
            no_improve += 1
            T = max(0.01, 2.0 * (1.0 - no_improve / patience))
            if rng.random() < np.exp(-delta / T):
                cur_pk = new_pk
                corr = new_corr
            else:
                cur[i], cur[j] = cur[j], cur[i]
        if no_improve > patience:
            # Restart with noise
            cur = to_binary(best + 0.3*(rng.rand(N)-0.5), N)
            corr = compute_corr(cur)
            cur_pk = float(np.max(corr))
            no_improve = 0
    return best, best_pk * dx

    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N):
    target = N // 2
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > target:
        ones = np.where(h_b == 1)[0]
        h_b[ones[np.argsort(h[ones])[:n_ones - target]]] = 0
    elif n_ones < target:
        zeros = np.where(h_b == 0)[0]
        h_b[zeros[np.argsort(-h[zeros])[:target - n_ones]]] = 1
    return h_b

def subgrad_topk(h, N, k, temp=5.0):
    corr = compute_corr(h)
    peak = float(np.max(corr))
    if k >= len(corr):
        top = np.arange(len(corr))
    else:
        top = np.argpartition(corr, -k)[-k:]
    vals = corr[top]
    w = np.exp(temp * (vals - vals.max()))
    w = w / w.sum()
    grad = np.zeros(N)
    center = N - 1

    # Final validation

    best_h = project_sum(best_h, N / 2.0)
    final_c5 = compute_c5(best_h, N)
    return best_h, final_c5, N
