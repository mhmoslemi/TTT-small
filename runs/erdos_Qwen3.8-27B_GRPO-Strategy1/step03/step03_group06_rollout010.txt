                ws /= ws.sum()
            p = np.zeros(2 * K)
            p[0::2] = np.clip(cs + np.random.normal(0, 0.03, K), 0.05, 1.95)
            p[1::2] = np.clip(ws + np.random.normal(0, 0.01, K), 0.01, 0.6)
            try:
                r = nelder_mead(iv_obj, p, args=(n0,), maxiter=400,
                                xatol=1e-10, fatol=1e-12)
                hn = h_from_iv(np.clip(r.x[0::2], 0, 2),
                               np.clip(r.x[1::2], 0.005, 0.9), n0)
                hn = proj(hn, n0)
                c5n = compute_c5(hn)
                if c5n < best_c5:
                    best_c5, best_h = c5n, hn
            except Exception:
                pass

    # Phase 2: Aggressive SLSQP sweep over T and top-k
    nc = len(best_h)
    for T in [15, 40, 100, 300, 800, 2500, 8000]:
        for tk in [None, 3, 5, 8, 15]:
            if time.time() - t0 > budget_s * 0.48:
                break
            try:
                hn = slq(best_h, T, nc, tk, 900)
                c5n = compute_c5(hn)
                if c5n < best_c5:
                    best_c5, best_h = c5n, hn
            except Exception:
                pass

    # Phase 3: Basin hopping with diverse perturbations
    nb = len(best_h)
    for it in range(130):
        if time.time() - t0 > budget_s * 0.70:
            break
        st = 0.06 * (1.0 - it / 130.0) + 0.01
        r = np.random.rand()
        if r < 0.30:
            hp = best_h + st * np.random.randn(nb)
        elif r < 0.55:
            hp = best_h.copy()
            bl = np.random.randint(3, max(4, nb // 5))
            s_ = np.random.randint(0, max(1, nb - bl))
            hp[s_:s_ + bl] += st * (2.0 * np.random.rand(bl) - 1.0)
        elif r < 0.80:
            hp = best_h.copy()
            i1, i2 = np.random.randint(0, nb, 2)
            hp[i1] += st
            hp[i2] -= st
        else:
            hp = worst_shift_target(best_h, nb, 5, 0.04)
        hp = proj(hp, nb)
        try:
            tk = np.random.choice([None, 3, 5, 8])
            T_bh = np.random.choice([200, 500, 1500])
            hn = slq(hp, T_bh, nb, tk, 250)
            c5n = compute_c5(hn)
            temp = 0.004 * (1.0 - it / 130.0) + 0.0004
            if c5n < best_c5 or np.random.rand() < np.exp(-(c5n - best_c5) / temp):
                if c5n < best_c5:
                    best_c5, best_h = c5n, hn
                else:
                    best_h = hn
        except Exception:
            pass

    # Phase 4: Higher resolution
    for nt in [200, 240, 320]:
        if time.time() - t0 > budget_s * 0.87:
            break
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, nt, endpoint=False)
        hi = proj(np.interp(xn, xo, best_h), nt)
        for T in [50, 300, 1500, 5000]:
            for tk in [None, 8, 15]:
                if time.time() - t0 > budget_s * 0.90:
                    break
                try:
                    hn = slq(hi, T, nt, tk, 500)
                    c5n = compute_c5(hn)
                    if c5n < best_c5:
                        best_c5, best_h = c5n, hn
                except Exception:
                    pass

    # Phase 5: Targeted worst-shift correction
    nf = len(best_h)
    if time.time() - t0 < budget_s * 0.95:
        for _ in range(30):
            if time.time() - t0 > budget_s * 0.95:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            top5 = np.argsort(corr)[-5:]
            improved = False
            for ki in top5:
                shift = ki - (nf - 1)
                center = max(2, min(nf - 3, nf // 2 + shift))
                for d in range(-4, 5):
                    for delta in [0.03, -0.03, 0.06, -0.06]:
                        j = center + d
                        if 0 <= j < nf:
                            ht = best_h.copy()
                            ht[j] = np.clip(ht[j] + delta, 0, 1)
                            ht = proj(ht, nf)
                            c5t = compute_c5(ht)
                            if c5t < best_c5 - 1e-13:
                                best_c5, best_h = c5t, ht
                                improved = True
                                break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    # Phase 6: Final high-T polish
    for T in [3000, 10000, 30000, 80000]:
        if time.time() - t0 > budget_s * 0.99:
            break
        for tk in [None, 3, 5]:
            if time.time() - t0 > budget_s * 0.995:
                break
            try:
                hn = slq(best_h, T, nf, tk, 1200)
                c5n = compute_c5(hn)
                if c5n < best_c5:
                    best_c5, best_h = c5n, hn
            except Exception:
                pass

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
