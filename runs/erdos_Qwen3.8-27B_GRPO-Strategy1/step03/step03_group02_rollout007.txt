    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    """Project h to [0,1]^N with sum = target using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -1.0, 1.0

        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_and_grad(h, N, tau):
    """Compute correlation and smoothed gradient (corrected)."""
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full')
    c5_raw = float(np.max(corr))
    c5 = c5_raw * (2.0 / N)
    
    # Softmax weights on correlation values
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5_raw
    w /= ws

        wi = w[ix]

            grad[M:] += wi * (1.0 - h[:N-M])

    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    
    while time.time() < deadline:

        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        grad, _ = corr_and_grad(h, N, tau)
        c5_val = compute_c5(h, dx)
        if c5_val < best_c5 - 1e-14:

            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best_h, best_c5

def local_search(h, N, target, dx, n_trials, deadline, seed=42):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    scale = 0.04
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        trial_h = best_h.copy()
        n_mod = rng.randint(1, 5)
        idx = rng.choice(N, size=n_mod, replace=False)
        trial_h[idx] += rng.randn(n_mod) * scale
        trial_h = project_sum(trial_h, target)
        c = compute_c5(trial_h, dx)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = trial_h
            scale = max(0.005, scale * 0.85)
        elif rng.random() < 0.1:
            scale = min(0.1, scale * 1.15)
    return best_h, best_c5

def subgradient_step(h, N, target, dx, tau, step_size):
    corr = np.correlate(h, 1.0 - h, 'full')
    cmax = float(np.max(corr))
    active = np.where(corr >= cmax - 0.02 / tau)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(N)
    center = N - 1
    for ix in active:
        L = ix - center
        if L == 0:
            grad += (1.0 - 2.0 * h) / len(active)
        elif L > 0:
            grad[:N - L] += (1.0 - h[L:N]) / len(active)
            grad[L:] -= h[:N - L] / len(active)
        else:
            M = -L
            grad[M:] += (1.0 - h[:N - M]) / len(active)
            grad[:N - M] -= h[M:] / len(active)
    h_new = project_sum(h - step_size * grad, target)
    return h_new, compute_c5(h, dx)

def fine_subgrad(h, N, target, dx, deadline, n_iters=500):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    tau = 50.0
    step = 0.02
    for it in range(n_iters):
        if time.time() > deadline:
            break
        h_new, c5_cur = subgradient_step(h, N, target, dx, tau, step)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            step = min(0.05, step * 1.1)
        else:
            step *= 0.7
        tau *= 1.05
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / N0)
    x0 = np.linspace(0.0, 2.0, N0, endpoint=False)

    sizes = [N0, 200, 250, 300, 350]
    sizes = sorted(set(s for s in sizes if s >= 100))

    for N in sizes:
        if time.time() > deadline - 120:
            break
        dx = 2.0 / N
        target = N / 2.0

        if N == N0:
            h_init = h0.copy()
        else:
            xN = np.linspace(0.0, 2.0, N, endpoint=False)
            h_init = np.interp(xN, x0, h0)
            h_init = project_sum(h_init, target)

        rem = deadline - time.time()
        if rem < 80:
            break

        # Phase 1: Broad optimization
        p1d = time.time() + rem * 0.30
        h1, c1 = optimize_phase(h_init.copy(), N, target, dx, p1d,
                                30.0, 200.0, 0.04, 0.008)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

        # Phase 2: Sharper focus
        if time.time() < deadline - 50:
            p2d = time.time() + (deadline - time.time()) * 0.35
            h2, c2 = optimize_phase(h1.copy(), N, target, dx, p2d,
                                    200.0, 1000.0, 0.008, 0.001)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
            h1 = h2

        # Phase 3: Subgradient polishing
        if time.time() < deadline - 30:
            h3, c3 = fine_subgrad(h1.copy(), N, target, dx,
                                   deadline - 25, n_iters=800)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()
            h1 = h3

        # Phase 4: Very high tau final push
        if time.time() < deadline - 20:
            h4, c4 = optimize_phase(h1.copy(), N, target, dx, deadline - 15,
                                    2000.0, 8000.0, 0.002, 0.00005)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()
            h1 = h4

        # Local search
        if time.time() < deadline - 10:
            hls, cls = local_search(h1.copy(), N, target, dx,
                                     n_trials=8000, deadline=deadline - 5,
                                     seed=seed + N)
            if cls < best_c5:
                best_c5 = cls
                best_h = hls.copy()

        # Random restart at this size
        if time.time() < deadline - 80:
            h_r = 0.5 + 0.3 * np.sin(2 * np.pi * rng.randint(3, 8) * x0 * N / N0)
            h_r = h_r + rng.randn(N) * 0.08
            h_r = project_sum(h_r, target)
            hr_dead = time.time() + (deadline - time.time()) * 0.5
            hr, cr = optimize_phase(h_r, N, target, dx, hr_dead,
                                     50.0, 500.0, 0.03, 0.003)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr.copy()
                if time.time() < deadline - 20:
                    hr2, cr2 = fine_subgrad(hr.copy(), N, target, dx,
                                             deadline - 10, n_iters=400)
                    if cr2 < best_c5:
                        best_c5 = cr2
                        best_h = hr2.copy()

    # Ultra final polish
    Nb = len(best_h)
    if time.time() < deadline - 4:
        hf, cf = fine_subgrad(best_h.copy(), Nb, Nb // 2, 2.0 / Nb,
                               deadline - 1, n_iters=300)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
