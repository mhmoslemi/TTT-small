def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            # Need to increase sum: add to entries with room
            room = 1.0 - h
            mask = room > 1e-15
            if not mask.any():
                break
            h[mask] += diff * room[mask] / room[mask].sum()
        else:
            # Need to decrease sum: remove from entries with value
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] += diff * h[mask] / h[mask].sum()
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-15
            if not mask.any():
                break
            h[mask] += diff * room[mask] / room[mask].sum()
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] += diff * h[mask] / h[mask].sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (dx / T) * np.log(np.sum(np.exp(T * (corr - m)))) + dx * m

def grad_smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    ex = T * (corr - cm)
    weights = np.exp(ex)
    weights /= weights.sum()
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-13:
            continue
        k = m - (n - 1)
        if k >= 0:
            if k == 0:
                grad[:] += w * (1.0 - 2.0 * h)
            else:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
        else:
            kp = -k
            grad[:n - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n - kp]
    return grad * dx

def adam_iterate(h, n, T, n_iter, t_end, lr_init=0.01, betas=(0.9, 0.999), eps=1e-8):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        lr = lr_init / (1.0 + 0.002 * it)
        g = grad_smooth_c5(h, T)
        m_buf = betas[0] * m_buf + (1 - betas[0]) * g
        v_buf = betas[1] * v_buf + (1 - betas[1]) * g * g
        m_hat = m_buf / (1 - betas[0] ** (it + 1))
        v_hat = v_buf / (1 - betas[1] ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        c5_try = compute_c5(h)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            coef = np.zeros(2 * n - 1)
            for k in range(n):
                kk = k - (n - 1)
                if kk == 0:
                    coef[n - 1] += 1.0 - 2.0 * 0.0
                else:
                    if 0 <= i + kk < n:
                        coef[n - 1 + kk] += (1.0 - best_h[i + kk])
                    if 0 <= i - kk < n:
                        coef[n - 1 - kk] -= best_h[i - kk]
            def f1d(hi):
                return np.max(corr_base + hi * coef) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            h_try = best_h.copy()
            h_try[i] = res.x
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

        best_h[:n0 // 2] = 1.0

    best_c5 = compute_c5(best_h)

    # Phase 1: Adam with T-sweep (30% budget)
    t_p1 = t0 + budget_s * 0.30
    T_list = [15, 40, 100, 250, 600, 1500, 3500, 8000]
    for T in T_list:
        steps = max(500, int(3000 * (1.0 + 0.3 / (T / 100.0))))
        h_new, c5_new = adam_iterate(best_h, n, T, steps, t_p1, lr_init=0.008)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Coordinate descent (12% budget)
    t_p2 = t0 + budget_s * 0.42
    best_h, best_c5 = coord_descent(best_h, n, t_p2, max_sweeps=25)

    # Phase 3: Basin hopping with Adam (30% budget)
    t_p3 = t0 + budget_s * 0.72
    work_h = best_h.copy()
    work_c5 = best_c5
    n_restarts = 20
    for it in range(n_restarts):
        if time.time() > t_p3:
            break
        cool = 1.0 - 0.8 * (it / float(n_restarts))
        step_size = 0.05 + 0.20 * cool
        h_pert = work_h + step_size * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        T_bh = int(500 * (1.0 + 2.0 * cool))
        h_new, c5_new = adam_iterate(h_pert, n, T_bh, 2500, t_p3, lr_init=0.006)
        temp = 0.004 * cool + 0.0003
        if c5_new < work_c5 or np.random.rand() < np.exp(-(c5_new - work_c5) / temp):
            work_h = h_new
            work_c5 = c5_new
        if work_c5 < best_c5:
            best_c5 = work_c5
            best_h = work_h.copy()

    # Phase 4: Coordinate descent from basin hopping result (10% budget)
    t_p4 = t0 + budget_s * 0.82
    best_h, best_c5 = coord_descent(best_h, n, t_p4, max_sweeps=30)

    # Phase 5: Try alternative resolutions (10% budget)
    t_p5 = t0 + budget_s * 0.92
    n_current = len(best_h)
    for n_try in [120, 200]:
        if time.time() > t_p5:
            break
        x_old = np.linspace(0, 2, n_current, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        for T in [300, 2000]:
            if time.time() > t_p5:
                break
            h_new, c5_new = adam_iterate(h_interp, n_try, T, 3000, t_p5, lr_init=0.005)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                n_current = n_try

    # Phase 6: Final high-T Adam + coord descent polish (8% budget)
    t_p6 = t0 + budget_s * 0.98
    n_final = len(best_h)
    for T in [10000, 30000]:
        if time.time() > t_p6:
            break
        h_new, c5_new = adam_iterate(best_h, n_final, T, 4000, t_p6, lr_init=0.003)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    if time.time() < t_p6:
        best_h, best_c5 = coord_descent(best_h, len(best_h), t_p6, max_sweeps=20)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
