def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr - np.max(corr)
    val = np.exp(T * c).sum()
    return (np.log(val) / T + np.max(corr)) * dx

def sum_eq(h):
    return h.sum() - len(h) / 2.0

                       bounds=bnds, constraints=cons,

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr - np.max(corr)
    val = np.log(np.sum(np.exp(T * c))) / T + np.max(corr)
    return val * dx

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def slsqp_optimize(h0, T, n, maxiter=1000, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_max_obj, h0, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return make_feasible(h0.copy(), n)

def binary_local_search(h, n, max_iters=2000, t_limit=None, t_ref=None):
    h = (h > 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > n // 2:
        ones_idx = np.where(h == 1)[0]
        for i in range(n_ones - n // 2):
            j = ones_idx[np.random.randint(n_ones)]
            h[j] = 0
    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0)[0]
        for i in range(n // 2 - n_ones):
            j = zeros_idx[np.random.randint(n - n_ones)]
            h[j] = 1
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iters):
        if t_ref is not None and time.time() > t_limit:
            break
        improved = False
        n_trials = 80
        for _ in range(n_trials):
            oi = np.random.randint(n)
            if h[oi] < 0.5:
                continue
            zi = np.random.randint(n)
            if h[zi] > 0.5:
                continue
            h[oi], h[zi] = 0.0, 1.0
            c5_try = compute_c5(h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                improved = True
            else:
                h[oi], h[zi] = 1.0, 0.0
        if improved:
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break
    return make_feasible(h, n), best_c5

def periodic_init(p, n):
    x = np.zeros(n)
    half = p // 2
    for i in range(n):
        if (i % p) < half:
            x[i] = 1.0
    return make_feasible(x, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    el = lambda: time.time() - t0

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        best_h = periodic_init(10, 100)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Multi-resolution SLSQP (15% budget)
    t1 = t0 + budget_s * 0.15
    for n_try in [80, 100, 120, 160, 200, 240]:
        if el() > t1:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = make_feasible(np.interp(x_new, x_old, best_h), n_try)
        inits = [h_interp]
        for p in [6, 8, 10, 12, 14, 16, 20]:
            if n_try % p == 0:
                inits.append(periodic_init(p, n_try))
        for h_init in inits[:4]:
            if el() > t1:
                break
            for T in [500, 3000]:
                if el() > t1:
                    break
                h_new = slsqp_optimize(h_init.copy(), T, n_try, maxiter=600)
                c5v = compute_c5(h_new)
                if c5v < best_c5 - 1e-10:
                    best_c5 = c5v
                    best_h = h_new
                    n_best = n_try

    # Phase 2: Aggressive binary local search (20% budget)
    t2 = t0 + budget_s * 0.35
    h_bin, c5_bin = binary_local_search(best_h, n_best, max_iters=5000, t_limit=t2, t_ref=t0)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin
    # Second round with different seed shift
    np.random.seed(seed + 1)
    h_bin2, c5_bin2 = binary_local_search(best_h, n_best, max_iters=3000, t_limit=t2, t_ref=t0)
    if c5_bin2 < best_c5:
        best_c5 = c5_bin2
        best_h = h_bin2

    # Phase 3: High-T SLSQP polish (10% budget)
    t3 = t0 + budget_s * 0.45
    h_ind = (best_h > 0.5).astype(float)
    h_ind = make_feasible(h_ind, n_best)
    for T in [10000, 50000, 200000]:
        if el() > t3:
            break
        h_new = slsqp_optimize(h_ind.copy(), T, n_best, maxiter=2500, ftol=1e-16)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new
            h_ind = (best_h > 0.5).astype(float)
            h_ind = make_feasible(h_ind, n_best)

    # Phase 4: Basin hopping with pair swaps (40% budget)
    t4 = t0 + budget_s * 0.85
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = n_best
    it_count = 0
    while el() < t4:
        it_count += 1
        n_swaps = np.random.randint(3, max(4, n_bh // 15))
        h_pert = bh_h.copy()
        for _ in range(n_swaps):
            oi = np.random.randint(n_bh)
            if h_pert[oi] < 0.5:
                continue
            zi = np.random.randint(n_bh)
            if h_pert[zi] > 0.5:
                continue
            h_pert[oi], h_pert[zi] = 0.0, 1.0
        h_pert = make_feasible(h_pert, n_bh)

        # Quick SLSQP refinement
        T_use = 3000 + 2000 * (it_count % 5)
        h_new = slsqp_optimize(h_pert.copy(), T_use, n_bh, maxiter=300)
        c5_s = compute_c5(h_new)

        # Quick binary search
        h_b, c5_b = binary_local_search(h_pert, n_bh, max_iters=400,
                                        t_limit=min(t4, el() + 3.0), t_ref=t0)
        if c5_b < c5_s:
            h_new, c5v = h_b, c5_b
        else:
            h_new, c5v = h_new, c5_s

        # Accept with simulated annealing
        frac = it_count / max(1, int((t4 - t0) / (t4 - t0)))
        temp = 0.004 * np.exp(-1.5 * min(1.0, el() / (t4 - t0))) + 0.0001
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 5: Final multi-resolution check (15% budget)
    t5 = t0 + budget_s * 0.99
    for n_final in [100, 160, 200, 240, 300, 400]:
        if el() > t5:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_final, endpoint=False)
        h_interp = make_feasible(np.interp(x_new, x_old, best_h), n_final)
        h_b, c5_b = binary_local_search(h_interp, n_final, max_iters=2000,
                                        t_limit=min(t5, el() + 8.0), t_ref=t0)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
            n_best = n_final
            continue
        if el() < t5:
            h_new = slsqp_optimize(h_interp.copy(), 20000, n_final, maxiter=1000)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
                n_best = n_final

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
