        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr)) * dx
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = h.copy()
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(N)
        for ai in top_idx:
            g += lag_gradient(h, ai - center, N)
        g /= top_k
        v = mom * v + g
        h_new = h - lr * v
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new <= c5v + 5e-7:
            h = h_new
        else:
            h = best_h.copy()
            v *= 0.4
        if it % decay_period == 0:
            lr *= decay
            if lr < 1e-12:
                break
    return best_h, best_c5

def grad_descent_smooth(h, N, target, deadline, tau_start, tau_end, top_k_start, top_k_end, lr_init, decay, mom, decay_period):
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(N)
    lr = lr_init
    it = 0
    center = N - 1
    nc = 2 * N - 1
    while time.time() < deadline:
        it += 1
        frac = min(float(it) / 3000.0, 1.0)
        tau = tau_start + (tau_end - tau_start) * frac
        tk = int(top_k_start + (top_k_end - top_k_start) * frac)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr)) * dx
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = h.copy()
        sc = tau * corr - tau * corr.max()
        w_sm = np.exp(sc)
        w_sm /= w_sm.sum()
        top_idx = np.argsort(corr)[-tk:]
        w_top = np.zeros(nc)
        w_top[top_idx] = 1.0 / tk
        alpha = min(1.0, max(0.0, frac * 1.5))
        w = (1.0 - alpha) * w_sm + alpha * w_top
        thresh = 0.002
        active = np.where(w > thresh)[0]
        g = np.zeros(N)
        for ai in active:
            L = ai - center
            wi = w[ai]
            if L == 0:
                g += wi * (1.0 - 2.0 * h)
            elif L > 0:
                g[L:] += wi * (1.0 - h[:N - L])
                g[:N - L] -= wi * h[L:]
            else:
                M = -L
                g[:N - M] += wi * (1.0 - h[M:])
                g[M:] -= wi * h[:N - M]
        v = mom * v + g
        h_new = h - lr * v
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new <= c5v + 5e-7:
            h = h_new
        else:
            h = best_h.copy()
            v *= 0.4
        if it % decay_period == 0:
            lr *= decay
            if lr < 1e-12:
                break
    return best_h, best_c5

def pair_polish(h, N, target, deadline, n_pairs=20000):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    rng_local = np.random.RandomState(12345)
    it = 0
    while time.time() < deadline and it < n_pairs:
        it += 1
        i, j = rng_local.randint(N, 2)
        if i == j:
            continue
        oi, oj = h[i], h[j]
        if abs(oi - oj) < 1e-7:
            continue
        delta = 0.05
        h[i] = proj_i = np.clip(oi + delta, 0, 1)
        h[j] = np.clip(oj - delta, 0, 1)
        h = project_sum(h, target)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h.copy()
        else:
            h = best_h.copy()
        delta = -0.05
        h[i] = np.clip(oi + delta, 0, 1)
        h[j] = np.clip(oj - delta, 0, 1)
        h = project_sum(h, target)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h.copy()
        else:
            h = best_h.copy()
    return best_h, best_c5

def coord_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    while time.time() < deadline:
        improved = False
        for i in range(N):
            if time.time() > deadline:
                return best_h, best_c5
            orig = h[i]
            trials = (0.0, 1.0, max(0.0, orig - 0.15), min(1.0, orig + 0.15),
                      max(0.0, orig - 0.05), min(1.0, orig + 0.05),
                      max(0.0, orig * 0.5), min(1.0, orig + (1.0 - orig) * 0.5))
            for t in trials:
                if abs(t - orig) < 1e-9:
                    continue
                h[i] = t
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-11:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break
                h = best_h.copy()
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    h_par = np.array(initial_h_values, dtype=float)
    N = len(h_par)
    target = N / 2.0
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, 2.0 / N)

    # Phase 1: Smooth gradient descent with adaptive focus
    t1 = time.time() + 180
    h1, c1 = grad_descent_smooth(best_h.copy(), N, target, t1,
                                  3.0, 80.0, 12, 3, 0.035, 0.965, 0.86, 55)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Focused gradient descent (top-2)
    t2 = time.time() + 140
    h2, c2 = grad_descent_focused(best_h.copy(), N, target, t2,
                                   2, 0.012, 0.97, 0.89, 70)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: Focused gradient descent (top-1, fine)
    t3 = time.time() + 100
    h3, c3 = grad_descent_focused(best_h.copy(), N, target, t3,
                                   1, 0.004, 0.98, 0.91, 90)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: Random restarts at parent size
    for scale in (0.025, 0.05):
        if time.time() > deadline - 120:
            break
        h_alt = best_h + rng.randn(N) * scale
        h_alt = project_sum(h_alt, target)
        t_alt = time.time() + 70
        h4, c4 = grad_descent_focused(h_alt, N, target, t_alt,
                                       3, 0.02, 0.96, 0.87, 50)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Phase 5: Upsample to higher resolution, optimize, downsample
    if time.time() < deadline - 100:
        N2 = 500
        target2 = N2 / 2.0
        x2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        x1 = np.linspace(0.0, 2.0, N, endpoint=False)
        h_up = np.interp(x2, x1, best_h)
        h_up = project_sum(h_up, target2)
        t_up = min(time.time() + 90, deadline - 50)
        h_up, c_up = grad_descent_focused(h_up, N2, target2, t_up,
                                           3, 0.018, 0.965, 0.87, 55)
        if c_up < best_c5:
            best_h = np.interp(x1, x2, h_up)
            best_h = project_sum(best_h, target)
            best_c5 = compute_c5(best_h, 2.0 / N)
        # Also keep the high-res solution as candidate
        c_up_resampled = compute_c5(h_up, 2.0 / N2)
        if c_up_resampled < best_c5:
            best_h = h_up
            best_c5 = c_up_resampled
            N = N2
            target = N2 / 2.0

    # Phase 6: Pair polish
    if time.time() < deadline - 40:
        Nb = len(best_h)
        t_pair = min(time.time() + 80, deadline - 20)
        best_h, best_c5 = pair_polish(best_h, Nb, Nb / 2.0, t_pair)

    # Phase 7: Coordinate polish (multiple passes)
    if time.time() < deadline - 3:
        Nb = len(best_h)
        best_h, best_c5 = coord_polish(best_h, Nb, Nb / 2.0, deadline - 2)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
