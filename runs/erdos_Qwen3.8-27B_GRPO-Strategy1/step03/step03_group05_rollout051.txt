import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-15:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        n_adj = max(1, int(abs(diff)))
        if diff > 0:
            idx = np.argsort(1.0 - h)[-n_adj:]
        else:
            idx = np.argsort(h)[-n_adj:]
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def f_at_shift(h, shift):
    n = len(h)
    if shift >= 0:
        return float(np.sum(h[:n-shift] * (1.0 - h[shift:])))
    else:
        return float(np.sum(h[-shift:] * (1.0 - h[:n+shift])))

def active_set_cd(h, n, k_active, n_sweeps=200, update_every=5):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    active = np.argpartition(corr, -k_active)[-k_active:]
    for sweep in range(n_sweeps):
        if sweep % update_every == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            active = np.argpartition(corr, -k_active)[-k_active:]
        for i in range(n):
            slopes = np.zeros(len(active))
            f_vals = np.zeros(len(active))
            for idx, s in enumerate(active):
                shift = int(s - (n - 1))
                sl = 0.0
                if 0 <= i + shift < n:
                    sl += (1.0 - h[i + shift])
                if 0 <= i - shift < n:
                    sl -= h[i - shift]
                slopes[idx] = sl
                f_vals[idx] = f_at_shift(h, shift)
            intercepts = f_vals - slopes * h[i]
            cand_x = [0.0, 1.0, h[i]]
            for a in range(len(slopes)):
                for b in range(a + 1, len(slopes)):
                    ds = slopes[a] - slopes[b]
                    if abs(ds) > 1e-14:
                        xi = (intercepts[b] - intercepts[a]) / ds
                        if -0.01 <= xi <= 1.01:
                            cand_x.append(xi)
            best_x = h[i]
            best_v = float('inf')
            for x in cand_x:
                xc = min(max(x, 0.0), 1.0)
                v = max(sl * xc + ic for sl, ic in zip(slopes, intercepts))
                if v < best_v:
                    best_v = v
                    best_x = xc
            h[i] = best_x
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        elif sweep > 10:
            break
    return h, compute_c5(h)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def topk_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_solve(h, obj_func, obj_args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=obj_args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def lbfgsb_solve(h, obj_func, obj_args, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h, args=obj_args, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'gtol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def block_move(h, n, rng):
    h_new = h.copy()
    start = rng.randint(0, max(1, n - 2))
    length = rng.randint(2, max(3, n // 4))
    end = min(start + length, n)
    shift = rng.randint(1, max(2, length))
    h_new[start:end] = np.roll(h_new[start:end], shift)
    return project_feasible(h_new, n)

def gauss_perturb(h, n, sigma, rng):
    return project_feasible(h + rng.randn(n) * sigma, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm-up
    if time.time() - t0 < budget_s * 0.08:
        for T in [200, 1000]:
            ht = slsqp_solve(best_h, softmax_obj, (T, best_n), 300, 1e-12)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht

    # Phase 2: Active-set coordinate descent (main innovation)
    for k_act in [3, 5, 8, 12]:
        if time.time() - t0 > budget_s * 0.45:
            break
        h_cd, c_cd = active_set_cd(best_h, best_n, k_act, n_sweeps=300, update_every=4)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd

    # Phase 3: SLSQP refinement after CD
    if time.time() - t0 < budget_s * 0.55:
        for T in [2000, 5000, 10000]:
            ht = slsqp_solve(best_h, softmax_obj, (T, best_n), 500, 1e-14)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht
        ht = slsqp_solve(best_h, topk_obj, (3, best_n), 500, 1e-14)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 4: Second round of active-set CD
    for k_act in [5, 8, 15]:
        if time.time() - t0 > budget_s * 0.70:
            break
        h_cd, c_cd = active_set_cd(best_h, best_n, k_act, n_sweeps=400, update_every=3)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd

    # Phase 5: Basin hopping with CD-local refine
    n_trials = 40
    for trial in range(n_trials):
        if time.time() - t0 > budget_s * 0.85:
            break
        r = rng.random()
        sigma = 0.04 * (1.0 - 0.8 * trial / n_trials) + 0.005
        if r < 0.35:
            hp = block_move(best_h, best_n, rng)
        else:
            hp = gauss_perturb(best_h, best_n, sigma, rng)
        # Local refine: quick CD + SLSQP
        hp, _ = active_set_cd(hp, best_n, 5, n_sweeps=80, update_every=5)
        ho = slsqp_solve(hp, softmax_obj, (3000, best_n), 250, 1e-13)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho
            # Extra refine after improvement
            hp2, _ = active_set_cd(best_h, best_n, 8, n_sweeps=200, update_every=3)
            if compute_c5(hp2) < best_c5:
                best_h = hp2
                best_c5 = compute_c5(best_h)

    # Phase 6: Multi-resolution
    for nr in [150, 180, 210, 240]:
        if time.time() - t0 > budget_s * 0.92:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hu, _ = active_set_cd(hu, nr, 5, n_sweeps=100, update_every=5)
        hh = slsqp_solve(hu, topk_obj, (5, nr), 300, 1e-13)
        c5_hi = compute_c5(hh)
        if c5_hi < best_c5:
            best_c5, best_h, best_n = c5_hi, hh, nr
        x_dn = np.linspace(0, 2, best_n)
        hd = np.interp(x_dn, x_new, hh)
        hd = project_feasible(hd, best_n)
        hd, _ = active_set_cd(hd, best_n, 5, n_sweeps=100, update_every=5)
        c5_dn = compute_c5(hd)
        if c5_dn < best_c5:
            best_c5, best_h = c5_dn, hd

    # Phase 7: Final intensive polish
    if time.time() - t0 < budget_s * 0.97:
        for _ in range(3):
            if time.time() - t0 > budget_s * 0.97:
                break
            hc, cc = active_set_cd(best_h, best_n, 10, n_sweeps=500, update_every=2)
            if cc < best_c5:

    return best_h, best_c5, len(best_h)
