def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

            if s < 1e-14:
                break
            h += room * (diff / s)

def _corr_d(idx, h, n):
    k = idx - (n - 1)
    d = np.zeros(n)
    if k >= 0:
        d[:n-k] = (1.0 - h[k:])

    else:
        kp = -k
        d[kp:] = (1.0 - h[:n-kp])

    return d
