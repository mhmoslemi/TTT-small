This means subgradient methods may not converge nicely. But the smoothing + subgradient can still work as a heuristic.

The coordinate descent is actually the most reliable method here because each 1D subproblem IS convex (it's max of linear functions in one variable).

Let me optimize the coordinate descent:

Key insight: For coordinate i, given all other h[j] fixed:
corr[k] as a function of h[i] = a_k + b_k * h[i]

where a_k and b_k are computable. Then we minimize max_k (a_k + b_k * h[i]) over h[i] ∈ [0,1].

This is a 1D convex piecewise-linear minimization. The minimum is at:
- h[i] = 0, if the function is decreasing at 0
- h[i] = 1, if the function is increasing at 1  
- At a "kink" point where two lines cross

The kink points are at h[i] = (a_k2 - a_k1) / (b_k1 - b_k2) for pairs (k1, k2) where b_k1 ≠ b_k2.

For efficiency, I'll use a simple approach: evaluate at h[i]=0, h[i]=1, and a few intermediate points, then refine.

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    for i in range(n):
        h[i] = 0.0
        corr_base = np.correlate(h, 1.0 - h, mode='full')
        shifts = np.arange(-(n-1), n)
        b = np.zeros(2*n-1)
        for idx, s in enumerate(shifts):
            g = 0.0
            j1 = i + s
            if 0 <= j1 < n:
                g += (1.0 - h[j1])
            j2 = i - s
            if 0 <= j2 < n:
                g -= h[j2]

        c1 = np.max(corr_base + b)

                            bounds=(lo, hi), method='bounded', options={'xatol': 1e-10})
        h[i] = res.x

from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    shifts = np.arange(-(n - 1), n)
    for i in range(n):
        old_val = h[i]
        h[i] = 0.0

        # Compute b[k] = d corr[k] / d h[i]
        j1 = i + shifts
        j2 = i - shifts

        m2 = (j2 >= 0) & (j2 < n)
        b = np.zeros(2 * n - 1)
        b[m1] += (1.0 - h[j1[m1]])
        b[m2] -= h[j2[m2]]

        v1 = np.max(corr_base + b)
        if v0 < v1:

        else:

        res = minimize_scalar(lambda x: np.max(corr_base + x * b),
                              bounds=bounds, method='bounded',
                              options={'xatol': 1e-10})

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _f1d(x, cb, b):
    return np.max(cb + x * b)

def one_sweep(h, n):
    shifts = np.arange(-(n - 1), n)
    for i in range(n):
        h[i] = 0.0
        cb = np.correlate(h, 1.0 - h, mode='full')
        j1 = i + shifts
        j2 = i - shifts
        m1 = (j1 >= 0) & (j1 < n)
        m2 = (j2 >= 0) & (j2 < n)
        b = np.zeros(2 * n - 1)
        b[m1] += (1.0 - h[j1[m1]])
        b[m2] -= h[j2[m2]]
        mx0 = np.max(cb)
        if np.max(b[cb >= mx0 - 1e-12]) >= -1e-12:
            h[i] = 0.0
            continue
        cb1 = cb + b
        mx1 = np.max(cb1)
        if np.max(b[cb1 >= mx1 - 1e-12]) <= 1e-12:
            h[i] = 1.0
            continue
        res = minimize_scalar(_f1d, args=(cb, b), bounds=(0.0, 1.0),
                             method='bounded', options={'xatol': 1e-9})
        h[i] = res.x
    return h

def multi_cd(h, n, n_sweeps=5, t_end=None):
    for _ in range(n_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        h = one_sweep(h, n)
        h = project(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent (30% time)
    t1 = t0 + budget_s * 0.30
    plateau = 0
    while time.time() < t1:
        h_w = best_h.copy()
        h_w = one_sweep(h_w, n)
        h_w = project(h_w, n)
        c5 = compute_c5(h_w)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_w
            plateau = 0
        else:
            plateau += 1
            if plateau > 40:
                break

    # Phase 2: Basin hopping with structured exploration (30% time)
    t2 = t0 + budget_s * 0.60
    work_h = best_h.copy()
    work_c5 = best_c5
    bh_count = 0
    while time.time() < t2:
        bh_count += 1
        frac = min(1.0, bh_count / 250.0)
        cool = 1.0 - 0.92 * frac
        step = 0.02 + 0.28 * cool
        if rng.random() < 0.08:
            step *= 2.5
        h_p = work_h + step * rng.standard_normal(n)
        h_p = project(h_p, n)
        n_sw = 2 + int(6 * cool)
        h_w = h_p.copy()
        for _ in range(n_sw):
            if time.time() > t2:
                break
            h_w = one_sweep(h_w, n)
            h_w = project(h_w, n)
        c5 = compute_c5(h_w)
        temp = 0.004 * cool + 0.0003
        if c5 < work_c5 or rng.random() < np.exp(-(c5 - work_c5) / max(temp, 1e-12)):
            work_h, work_c5 = h_w, c5
            if c5 < best_c5:
                best_h, best_c5 = h_w, c5
    if work_c5 < best_c5:
        best_h, best_c5 = work_h, work_c5

    # Phase 3: Multi-resolution exploration (20% time)
    t3 = t0 + budget_s * 0.82
    for n_try in [120, 144, 200, 240]:
        if time.time() > t3:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project(h_int, n_try)
        h_int = multi_cd(h_int, n_try, 6, t3)
        c5 = compute_c5(h_int)
        if c5 < best_c5:
            best_h, best_c5, n = h_int, c5, n_try
        # Also try a random restart at this resolution
        if time.time() < t3:
            h_rnd = project(rng.standard_normal(n_try), n_try)
            h_rnd = multi_cd(h_rnd, n_try, 4, t3)
            c5r = compute_c5(h_rnd)
            if c5r < best_c5:
                best_h, best_c5, n = h_rnd, c5r, n_try

    # Phase 4: Refine with smaller perturbations (12% time)
    t4 = t0 + budget_s * 0.94
    work_h = best_h.copy()
    work_c5 = best_c5
    for _ in range(30):
        if time.time() > t4:
            break
        h_p = work_h + 0.015 * rng.standard_normal(n)
        h_p = project(h_p, n)
        h_w = multi_cd(h_p, n, 4, t4)
        c5 = compute_c5(h_w)
        if c5 < work_c5:
            work_h, work_c5 = h_w, c5
            if c5 < best_c5:
                best_h, best_c5 = h_w, c5

    # Phase 5: Final coordinate descent polish (6% time)
    t5 = t0 + budget_s * 0.99
    while time.time() < t5:
        h_w = best_h.copy()
        h_w = one_sweep(h_w, n)
        h_w = project(h_w, n)
        c5 = compute_c5(h_w)
        if c5 < best_c5:
            best_h, best_c5 = h_w, c5
        else:
            break

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
