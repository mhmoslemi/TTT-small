def compute_corr(h):

def compute_c5(h, N):

def grad_lse(h, N, beta):
    """Gradient using log-sum-exp smooth max."""
    corr = compute_corr(h)
    # Numerical stability
    cm = np.max(corr)
    exp_c = np.exp(beta * (corr - cm))
    w = exp_c / exp_c.sum()
    # w is a probability distribution over lags
    # grad = sum_k w[k] * d(corr_k)/d(h)
    grad = np.zeros(N)
    center = N - 1

        if L == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif L > 0:
