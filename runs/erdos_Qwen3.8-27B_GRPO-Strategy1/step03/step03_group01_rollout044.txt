    time_each = (budget_s - 100) / n_starts

    for s in range(n_starts):
        if time.time() > deadline - 50:
            break
        s_dead = min(time.time() + time_each, deadline - 35)
        if s == 0:
            h_cur = h_par.copy()
        elif s == 1:
            h_cur = h_par + rng.randn(N) * 0.04
            h_cur = project_sum(h_cur, target)
        elif s == 2:
            h_cur = h_par + rng.randn(N) * 0.08
            h_cur = project_sum(h_cur, target)
        elif s == 3:
            h_cur = h_par.copy()
            idx = rng.choice(N, size=N//3, replace=False)
            h_cur[idx] = np.where(h_cur[idx] > 0.5, 0.0, 1.0)
            h_cur = project_sum(h_cur, target)
        else:
            h_cur = np.zeros(N)
            h_cur[rng.randint(0, N//2-1):rng.randint(N//2, N)] = 1.0
            h_cur = project_sum(h_cur, target)

        t_rem = s_dead - time.time()
        if t_rem < 12:
            continue

        # Phase A: broad
        tA = min(t_rem * 0.38, 80)
        h_cur, cA = opt_loop(h_cur, N, target, time.time() + tA,
                             3.0, 35.0, 0.055, 0.95, 45, 0.78, 35, 12, 0)
        if cA < best_c5:
            best_c5 = cA
            best_h = h_cur.copy()

        # Phase B: focused + snap
        if time.time() < s_dead - 8:
            tB = min((s_dead - time.time()) * 0.35, 60)
            h_cur, cB = opt_loop(h_cur, N, target, time.time() + tB,
                                 35.0, 200.0, 0.018, 0.97, 55, 0.84, 10, 4, 80)
            if cB < best_c5:
                best_c5 = cB
                best_h = h_cur.copy()

        # Phase C: fine
        if time.time() < s_dead - 4:
            tC = s_dead - time.time() - 2
            h_cur, cC = opt_loop(h_cur, N, target, time.time() + tC,
                                 200.0, 900.0, 0.004, 0.982, 70, 0.90, 4, 2, 0)
            if cC < best_c5:
                best_c5 = cC
                best_h = h_cur.copy()

    # Try N=462 interpolation
    if time.time() < deadline - 70:
        N2 = 462
        xN2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h2 = np.interp(xN2, xN, best_h)
        h2 = project_sum(h2, N2 / 2.0)
        rem2 = min(deadline - time.time() - 20, 90)
        if rem2 > 20:
            h2, c2 = opt_loop(h2, N2, N2/2.0, time.time() + rem2*0.85,
                              10.0, 300.0, 0.025, 0.96, 50, 0.82, 18, 3, 60)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2

    # Final polish
    if time.time() < deadline - 6:
        Nb = len(best_h)
        bd = min(deadline - 2, time.time() + 35)
        best_h, best_c5 = polish_coords(best_h, Nb, Nb/2.0, bd)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
