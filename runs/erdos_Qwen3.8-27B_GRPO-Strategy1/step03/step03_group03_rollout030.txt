import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14: break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    val += mu * (h.sum() - n/2.0)**2
    return val

def smooth_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(T * (corr - m))
    weights /= weights.sum()
    grad = np.zeros(n)
    for idx, w in enumerate(weights):
        if w < 1e-15: continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n-k])
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:n-kp])
            grad[:n-kp] += w * (-h[kp:])
    grad *= dx
    grad += 2 * mu * (h.sum() - n/2.0)
    return grad

def l_bfgs_smooth(h, T, mu, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    res = fmin_l_bfgs_b(smooth_obj, h, fprime=smooth_grad, args=(T, mu),
                        pgtol=1e-12, maxiter=maxiter, approx_grad=False)
    return project_feasible(np.clip(res, 0, 1), n)

def binary_swa(h, t_start=1.0, t_end=0.01, n_sweeps=200):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    best_c5, _ = compute_c5(h_bin)
    best_h = h_bin.copy()
    c5_cur, _ = compute_c5(h_bin)
    indices = np.arange(n)
    for s in range(n_sweeps):
        T_sa = t_start * (t_end/t_start)**(s/n_sweeps)
        np.random.shuffle(indices)
        improved = False
        for i in indices:
            if h_bin[i] < 0.5: continue
            # find a j to swap with
            zeros = [j for j in indices if j != i and h_bin[j] > 0.5]
            if not zeros: continue
            j = np.random.choice(zeros)
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            c5_new, _ = compute_c5(h_bin)
            delta = c5_new - c5_cur
            if delta < 0 or (T_sa > 0.01 and np.random.rand() < np.exp(-delta/T_sa)):
                c5_cur = c5_new
                improved = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_bin.copy()
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return best_h, best_c5
