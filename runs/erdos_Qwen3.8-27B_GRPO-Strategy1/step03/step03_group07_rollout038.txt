from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t > 1e-14:
            h = h + room * (diff / t)
    else:
        t = h.sum()
        if t > 1e-14:
            h = h * (target / t)
    return np.clip(h, 0.0, 1.0)

def corr_w(h, j):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    K = 2 * n - 1
    ia = j - lags
    ib = j + lags
    va = (ia >= 0) & (ia < n)
    vb = (ib >= 0) & (ib < n)
    w = np.zeros(K)
    w[va] += 1.0 - h[ia[va]]
    w[vb] -= h[ib[vb]]
    return w

def coord_desc(h, sweeps=5, grid_n=50):
    n = len(h)
    h = h.copy()
    tg = np.linspace(0, 1, grid_n)
    for _ in range(sweeps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            old = h[j]
            w = corr_w(h, j)
            v = np.max(corr[:, None] + (tg - old)[None, :] * w[:, None], axis=0)
            bt = tg[np.argmin(v)]
            if abs(bt - old) > 1e-10:
                corr += (bt - old) * w
                h[j] = bt
        h = make_feasible(h, n)
    return h

def slsqp_opt(h0, T, maxiter=400):
    n = len(h0)
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)
    K = 2 * n - 1

        c = np.correlate(x, 1.0 - x, mode='full')
        M = c.max()
        ez = np.exp(T * (c - M))
        return (M + np.log(ez.sum()) / T) * dx

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        M = c.max()
        ez = np.exp(T * (c - M))
        sw = ez / ez.sum()
        g = np.zeros(n)
        for j in range(n):
            ia = j - lags
            ib = j + lags
            va = (ia >= 0) & (ia < n)
            vb = (ib >= 0) & (ib < n)
            w = np.zeros(K)
            w[va] += 1.0 - x[ia[va]]
            w[vb] -= x[ib[vb]]
            g[j] = np.dot(sw, w)
        return g * dx

    return make_feasible(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t > 1e-14:
            h = h + room * (diff / t)
    else:
        t = h.sum()
        if t > 1e-14:
            h = h * (target / t)
    return np.clip(h, 0.0, 1.0)

def corr_w(h, j):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    K = 2 * n - 1
    ia = j - lags
    ib = j + lags
    va = (ia >= 0) & (ia < n)
    vb = (ib >= 0) & (ib < n)
    w = np.zeros(K)
    w[va] += 1.0 - h[ia[va]]
    w[vb] -= h[ib[vb]]
    return w

def coord_desc(h, sweeps=5, grid_n=50):
    n = len(h)
    h = h.copy()
    tg = np.linspace(0, 1, grid_n)
    for _ in range(sweeps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            old = h[j]
            w = corr_w(h, j)
            v = np.max(corr[:, None] + (tg - old)[None, :] * w[:, None], axis=0)
            bt = tg[np.argmin(v)]
            if abs(bt - old) > 1e-10:
                corr += (bt - old) * w
                h[j] = bt
        h = make_feasible(h, n)
    return h

def slsqp_opt(h0, T, maxiter=400):
    n = len(h0)
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)
    K = 2 * n - 1

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        M = c.max()
        ez = np.exp(T * (c - M))
        return (M + np.log(ez.sum()) / T) * dx

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        M = c.max()
        ez = np.exp(T * (c - M))
        sw = ez / ez.sum()
        g = np.zeros(n)
        for j in range(n):
            ia = j - lags
            ib = j + lags
            va = (ia >= 0) & (ia < n)
            vb = (ib >= 0) & (ib < n)
            w = np.zeros(K)
            w[va] += 1.0 - x[ia[va]]
            w[vb] -= x[ib[vb]]
            g[j] = np.dot(sw, w)
        return g * dx

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return make_feasible(h0.copy(), n)

        starts.append(make_feasible(h, n))
    return starts
