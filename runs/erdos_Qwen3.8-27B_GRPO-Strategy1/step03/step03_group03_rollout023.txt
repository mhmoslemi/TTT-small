def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += room * (diff / s)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h *= 1.0 + diff / s
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-13:
            continue
        lag = idx - (n - 1)

            grad[lag:n] += wi * (1.0 - h[:n - lag])
            grad[:n - lag] -= wi * h[lag:]
        else:
            p = -lag
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:n] -= wi * h[:n - p]
    return grad * dx
