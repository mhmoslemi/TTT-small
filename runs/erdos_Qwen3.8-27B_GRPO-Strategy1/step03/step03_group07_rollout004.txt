    # Phase 2: Coordinate descent
    if remaining() > 20:
        best_h, best_c5 = coord_descent(best_h, best_n, 10, min(remaining() * 0.2, 80))

    # Phase 3: More SLSQP at high T
    for Ti in [10000, 40000, 200000]:
        if remaining() < 30:
            break
        try:
            h_new = slsqp_step(best_h.copy(), Ti, best_n, 3000, 1e-15)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass

    # Phase 4: Coordinate descent again
    if remaining() > 20:
        best_h, best_c5 = coord_descent(best_h, best_n, 8, min(remaining() * 0.15, 60))

    # Phase 5: Try different n values
    for n_try in [100, 120, 150, 240, 300]:
        if remaining() < 60:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for Ti in [200, 2000, 20000]:
            if remaining() < 40:
                break
            try:
                h_new = slsqp_step(h_interp, Ti, n_try, 1500, 1e-14)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
            except Exception:
                pass
        if remaining() < 40:
            break

    # Phase 6: Perturbation + re-optimization (basin hopping lite)
    n_cur = len(best_h)
    max_bh = int(min(remaining() * 0.4, 120) / 4.0)
    max_bh = max(max_bh, 10)
    bh_h = best_h.copy()
    bh_c5 = best_c5
    for it in range(max_bh):
        if remaining() < 30:
            break
        h_pert = bh_h + 0.06 * (1.0 - it / max(1, max_bh)) * np.random.randn(n_cur)
        h_pert = make_feasible(h_pert, n_cur)
        try:
            h_new = slsqp_step(h_pert, 3000, n_cur, 500, 1e-13)
            c5v = compute_c5(h_new)
        except Exception:
            c5v = compute_c5(h_pert)
            h_new = h_pert
        temp = 0.003 * (1.0 - it / max(1, max_bh)) + 0.0002
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_h = h_new
            bh_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 7: Final coordinate descent + SLSQP polish
    if remaining() > 10:
        best_h, best_c5 = coord_descent(best_h, len(best_h), 5, min(remaining() * 0.3, 40))

    if remaining() > 15:
        n_cur = len(best_h)
        for Ti in [50000, 200000]:
            if remaining() < 10:
                break
            try:
                h_new = slsqp_step(best_h.copy(), Ti, n_cur, 2000, 1e-16)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
            except Exception:
                pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
