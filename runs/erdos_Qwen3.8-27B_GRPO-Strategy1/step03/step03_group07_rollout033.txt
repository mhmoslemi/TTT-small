from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    # Distribute difference proportionally to available room
    if diff > 0:
        room = 1.0 - h
    else:
        room = h.copy()
    total = room.sum()
    if total > 1e-14:
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_max_topk(vals, k, T):
    """Smooth approximation of max using top-k emphasis"""
    m = np.max(vals)
    # Sort to get top k
    sorted_vals = np.sort(vals)[::-1]
    topk_mean = np.mean(sorted_vals[:k])
    # Blend between max and topk mean
    soft = m - (m - topk_mean) * (1 - np.exp(-T * (m - topk_mean)))
    # Also add log-sum-exp for the rest
    ls = m + np.log(np.sum(np.exp(T * (vals - m)))) / T
    return 0.5 * (soft + ls)

def obj_lse(h, T, k):

    ls = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return ls * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    room = (1.0 - h) if diff > 0 else h.copy()
    total = room.sum()
    if total > 1e-14:
        h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def constraint_sum(x):
    return x.sum() - len(x) / 2.0

def slsqp_refine(h0, T, n, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_lse, h0, args=(T,), method='SLSQP',
                  bounds=bnds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(np.clip(res.x, 0, 1), n)
