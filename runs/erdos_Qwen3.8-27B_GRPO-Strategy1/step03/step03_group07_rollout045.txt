        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def gradient_c5_smooth(h, T):
    """Gradient of logsumexp(T*corr)/T * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_shifted = corr - corr.max()
    exp_vals = np.exp(T * corr_shifted)
    w = exp_vals / exp_vals.sum()  # softmax weights, sum to 1
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        # d corr[k] / dh[j] = (1-h[j+lag])*I(0<=j+lag<n) - h[j-lag]*I(0<=j-lag<n)
        if lag >= 0:
            # j+lag valid when j in [0, n-1-lag]
            valid_j = np.arange(0, n - lag)
            grad[valid_j] += wk * (1.0 - h[valid_j + lag])
            # j-lag valid when j in [lag, n-1]
            valid_j2 = np.arange(lag, n)
            grad[valid_j2] -= wk * h[valid_j2 - lag]
        else:
            # lag < 0, let l = -lag
            l = -lag
            # j+lag = j-l valid when j in [l, n-1]
            valid_j = np.arange(l, n)
            grad[valid_j] += wk * (1.0 - h[valid_j - l])
            # j-lag = j+l valid when j in [0, n-1-l]
            valid_j2 = np.arange(0, n - l)
            grad[valid_j2] -= wk * h[valid_j2 + l]
    return grad * T * dx

def projected_gradient_step(h, T, lr, n_inner=1, target_sum=None):
    if target_sum is None:
        target_sum = len(h) / 2.0
    for _ in range(n_inner):
        g = gradient_c5_smooth(h, T)
        h = h - lr * g
        h = project_onto_box_sum(h, target_sum)
    return h

def bit_flip_search(h, max_iter=5000):
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        improved = False
        for _ in range(50):
            j = np.random.randint(0, n)
            for delta in [-0.15, -0.05, 0.05, 0.15, 1.0 - 2.0 * h[j]]:
                h_try = h.copy()
                h_try[j] += delta
                h_try = make_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if improved:
                break
        if not improved and it > 100:
            break
    return h, best_c5
