import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_c5(h, T, k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k is not None and k < len(corr):
        idx = np.argpartition(corr, -k)[-k:]
        corr = corr[idx]
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14: return h
        h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot < 1e-14: return h
        h += h * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def sum_con(x):
    return x.sum() - len(x) / 2.0

def slsqp(h0, T, k, n, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    res = minimize(smooth_max_c5, h0, args=(T, k), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def binary_sa(h_init, n_iter, t0, t1, rng, time_limit):
    n = len(h_init)
    h = h_init.copy().astype(float)
    cur_c = float(np.max(np.correlate(h, 1-h, mode='full')))
    best = h.copy(); best_c = cur_c
    ts = time.time()
    for it in range(n_iter):
        if time.time() - ts > time_limit:
            break
        T = t0 * (t1 / t0) ** (it / max(1, n_iter))
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.integers(len(ones))]
        i0 = zeros[rng.integers(len(zeros))]
        h[i1] = 0.0; h[i0] = 1.0
        new_c = float(np.max(np.correlate(h, 1-h, mode='full')))
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c; best = h.copy()
        else:
            h[i1] = 1.0; h[i0] = 0.0
    return best, best_c

def local_swap_refine(h, n, n_iter, rng, time_limit):
    """Try pairwise swaps to reduce max correlation."""
    h = h.copy()
    cur_c = compute_c5(h)
    ts = time.time()
    for it in range(n_iter):
        if time.time() - ts > time_limit:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.integers(len(ones))]
        i0 = zeros[rng.integers(len(zeros))]
        h[i1], h[i0] = h[i0], h[i1]
        new_c = compute_c5(proj(h, n))
        if new_c < cur_c:
            cur_c = new_c
            h = proj(h, n)
        else:
            h[i1], h[i0] = h[i0], h[i1]
    return h, cur_c
