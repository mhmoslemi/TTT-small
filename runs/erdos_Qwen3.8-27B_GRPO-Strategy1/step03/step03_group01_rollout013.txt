import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h.copy()
    lo, hi = -0.5, 0.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5(h, N, tau, topK):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr)) * dx
    center = N - 1
    nc = len(corr)
    sc = tau * corr
    sc -= sc.max()
    w_sm = np.exp(sc)
    ss = w_sm.sum()
    if ss > 0:
        w_sm /= ss
    k = max(1, min(topK, nc))
    idx_top = np.argsort(corr)[-k:]
    w_top = np.zeros(nc)
    w_top[idx_top] = 1.0 / k
    alpha = min(1.0, max(0.0, (tau - 10.0) / 30.0))
    w = (1.0 - alpha) * w_sm + alpha * w_top
    grad = np.zeros(N)
    thresh = 0.002
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad, c5

def opt_loop(h, N, target, deadline, tau0, tau1, tk0, tk1, lr0, m0, m1, perturb_every=0):
    dx = 2.0 / N
    best_h = h.copy()

    it = 0
    rng_local = np.random.RandomState(12345 + N)
    while time.time() < deadline:
        it += 1
        f = min(it / 4000.0, 1.0)
        tau = tau0 + (tau1 - tau0) * f
        topK = int(tk0 + (tk1 - tk0) * f)
        mom = m0 + (m1 - m0) * f
        grad, c5v = grad_c5(h, N, tau, topK)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        v = mom * v + grad
        for _ in range(12):
            h_new = project_sum(h - lr * v, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new <= c5v + 1e-14 * max(1.0, c5v) or lr < 1e-14:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new

                break
