            if c5 < best_c5:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h * (1.0 + diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_obj(x, p, m, n, lam):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    idx = np.argpartition(corr, -m)[-m:]
    top = np.maximum(corr[idx].astype(float), 0.0)
    if p == 1:
        val = np.sum(top)
    else:
        val = (np.sum(top ** p)) ** (1.0 / p)
    val *= dx
    val += lam * (x.sum() - n / 2.0) ** 2
    return val

def lp_grad(x, p, m, n, lam):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    idx = np.argpartition(corr, -m)[-m:]
    top = np.maximum(corr[idx].astype(float), 0.0)
    if p > 1:
        sum_p = np.sum(top ** p)
        pref = dx * (1.0 / p) * sum_p ** (1.0 / p - 1.0) * p
    else:
        pref = dx
    grad = np.zeros(n)
    for j in range(m):
        s = int(idx[j] - (n - 1))
        coeff = pref * (top[j] ** (p - 1) if p > 1 else 1.0)
        ar = np.arange(n)
        va = ar + s
        ma = (va >= 0) & (va < n)
        contrib = np.zeros(n)
        contrib[ma] += 1.0 - x[va[ma]]
        vb = ar - s
        mb = (vb >= 0) & (vb < n)
        contrib[mb] += -x[vb[mb]]
        grad += coeff * contrib
    grad += 2.0 * lam * (x.sum() - n / 2.0)
    return grad

def lbfgsb_run(h, p, m, n, lam, maxiter, gtol):
    bnds = [(0.0, 1.0)] * n
    def f(x):
        return lp_obj(x, p, m, n, lam)
    def g(x):
        return lp_grad(x, p, m, n, lam)
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': gtol})
        return res.x
    except Exception:
        return h

def softmax_slsqp(h, T, n, maxiter):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(c)
        return (mx + np.log(np.sum(np.exp(T * (c - mx)))) / T) * (2.0 / n)
    def sfun(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sfun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def worst_shift_repair(h, n, strength=0.15):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top5 = np.argpartition(corr, -5)[-5:]
    h_new = h.copy()
    for wi in top5:
        s = int(wi - (n - 1))
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                prod = h_new[i] * (1.0 - h_new[j])
                if prod > 0.05:
                    d = min(prod * strength, h_new[i] * 0.2, (1.0 - h_new[j]) * 0.2)
                    h_new[i] -= d
                    h_new[j] += d
    return project_feasible(h_new, n)

def random_perturb(h, n, scale):
    return project_feasible(h + np.random.randn(n) * scale, n)
