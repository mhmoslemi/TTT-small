    val = smooth_obj(h, T)
    return val

def block_descent(h, block_size, T, n_sweeps):
    n = len(h)
    h = h.copy()
    for _ in range(n_sweeps):
        for j in range(0, n - block_size + 1, block_size):
            idx = list(range(j, j + block_size))
            x0 = h[idx]
            def _bobj(params):
                for k, ii in enumerate(idx):
                    h[ii] = params[k]
                return smooth_obj(h, T)
            bnds = [(0.0, 1.0)] * block_size
            res = minimize_scalar(_bobj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-7}) if block_size == 1 else None
            if block_size == 1:
                h[j] = res.x
            else:
                for attempt in range(3):
                    x0 = h[idx] + np.random.uniform(-0.1, 0.1, block_size)
                    x0 = np.clip(x0, 0, 1)
                    from scipy.optimize import minimize as _min
                    r = _min(_bobj, x0, bounds=bnds, method='L-BFGS-B',
                             options={'maxiter': 100, 'ftol': 1e-12})
                    h[idx] = np.clip(r.x, 0, 1)
        h = make_feasible(h)
    return h

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return make_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        best_h = np.zeros(200)
        best_h[0::2] = 1.0
        best_h = make_feasible(best_h)

    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent with annealed T
    T_schedule = [30, 60, 100, 200, 400, 800, 1600, 3200]
    n_cur = len(best_h)
    for T in T_schedule:
        if time.time() - t0 > budget_s * 0.25:
            break
        n_sw = 3 if T < 200 else 2
        h_new = coord_descent(best_h, T, n_sw)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new

    # Phase 2: Basin hopping with coordinate descent as local solver
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    elapsed = time.time() - t0
    remaining = budget_s * 0.55 - elapsed
    max_iters = max(int(remaining / 4.0), 20)
    for it in range(max_iters):
        if time.time() - t0 > budget_s * 0.60:
            break
        ptype = it % 5
        step = 0.15 * (1.0 - it / max(1, max_iters)) + 0.03
        if ptype == 0:
            h_pert = bh_h + step * np.random.randn(n_bh)
        elif ptype == 1:
            h_pert = bh_h.copy()
            blen = max(3, n_bh // 15)
            s1 = np.random.randint(0, n_bh - blen)
            s2 = np.random.randint(0, n_bh - blen)
            tmp = h_pert[s1:s1+blen].copy()
            h_pert[s1:s1+blen] = h_pert[s2:s2+blen]
            h_pert[s2:s2+blen] = tmp
        elif ptype == 2:
            h_pert = bh_h.copy()
            nflip = max(1, n_bh // 10)
            idx = np.random.choice(n_bh, nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif ptype == 3:
            h_pert = bh_h.copy()
            h_pert[np.random.choice(n_bh, n_bh//4, replace=False)] = np.random.uniform(0, 1, n_bh//4)
        else:
            h_pert = bh_h + step * 1.5 * np.random.randn(n_bh)

        h_pert = make_feasible(h_pert)
        T_use = 200 + 200 * (it // 10)
        T_use = min(T_use, 2000)
        h_new = coord_descent(h_pert, T_use, 2)
        c5v = compute_c5(h_new)

        temp = 0.004 * (1.0 - it / max(1, max_iters)) + 0.0002
        if c5v < bh_c5 or np.random.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 3: Multi-resolution
    for n_try in [100, 150, 200, 250, 300]:
        if time.time() - t0 > budget_s * 0.82:
            break
        if n_try == len(best_h):
            continue
        h_init = resample_h(best_h, n_try)
        for trial in range(2):
            if time.time() - t0 > budget_s * 0.82:
                break
            h0 = h_init + np.random.uniform(-0.03, 0.03, n_try)
            h0 = make_feasible(h0)
            for T in [100, 400, 1000]:
                if time.time() - t0 > budget_s * 0.82:
                    break
                h_new = coord_descent(h0, T, 2)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
                    h0 = h_new

    # Phase 4: Fine coordinate descent polish on best
    if time.time() - t0 < budget_s * 0.95:
        n_cur = len(best_h)
        for T in [2000, 5000, 10000]:
            if time.time() - t0 > budget_s * 0.97:
                break
            h_new = coord_descent(best_h, T, 3)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 5: Targeted 2-point moves
    if time.time() - t0 < budget_s * 0.99:
        n_cur = len(best_h)
        for _ in range(200):
            if time.time() - t0 > budget_s * 0.99:
                break
            j1 = np.random.randint(0, n_cur)
            j2 = np.random.randint(0, n_cur)
            if j1 == j2:
                continue
            delta = np.random.choice([-0.08, -0.03, 0.03, 0.08])
            h_try = best_h.copy()
            h_try[j1] += delta
            h_try[j2] -= delta
            h_try = make_feasible(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try

    best_h = make_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
