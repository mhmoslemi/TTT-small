import time

    n = len(h)

    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        w = corr[idx]
    else:
        w = corr
    m = w.max()
    return m + np.log(np.sum(np.exp(T * (w - m)))) / T

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max_val(corr, T, topk)
    return val * dx

def obj_pen(h, T, topk, n):
    val = obj_func(h, T, topk)
    pen = 0.05 * (h.sum() - n/2.0)**2
    return val + pen

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t < 1e-14: return h
        h += room * (diff / t)
    else:
        room = h
        t = room.sum()
        if t < 1e-14: return h
        h += room * (diff / t)
    return np.clip(h, 0.0, 1.0)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_func, h0, args=(T, topk),
                  method='SLSQP', bounds=bnds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': 1e-16})
    return project(np.clip(res.x, 0, 1), n)

def lbfgs(h0, T, n, topk, maxiter=3000):
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_pen, h0, args=(T, topk, n),

        return h0

        h = fixed.copy()
        h[indices] = np.clip(x, 0, 1)
        h = project(h, n)
        return obj_pen(h, 500, 10, n)

                   options={'maxiter': n_iter, 'xatol': 1e-10, 'fatol': 1e-14})
    h_new = fixed.copy()
    h_new[indices] = np.clip(res.x, 0, 1)
    return project(h_new, n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        w = corr[idx]
    else:
        w = corr
    m = w.max()
    return m + np.log(np.sum(np.exp(T * (w - m)))) / T

def obj_slq(h, T, topk):
    n = len(h)
    return smooth_max(np.correlate(h, 1-h, mode='full'), T, topk) * (2.0/n)

def obj_lbfgs(h, T, topk, n):
    v = obj_slq(h, T, topk)
    return v + 0.1 * (h.sum() - n/2.0)**2

def proj(h, n):
    h = np.clip(h, 0, 1).copy()
    d = n/2.0 - h.sum()
    if abs(d) < 1e-14: return h
    if d > 0:
        r = 1.0 - h; t = r.sum()
        if t < 1e-14: return h
        h += r * d/t
    else:
        r = h; t = r.sum()
        if t < 1e-14: return h
        h += r * d/t
    return np.clip(h, 0, 1)

def slq(h0, T, n, topk, mi=800):
    c = [{'type':'eq', 'fun': lambda x: x.sum()-n/2.0}]
    b = [(0.0,1.0)]*n
    r = minimize(obj_slq, h0, args=(T,topk), method='SLSQP',
                 bounds=b, constraints=c, options={'maxiter':mi,'ftol':1e-16})
    return proj(np.clip(r.x,0,1), n)

def lb(h0, T, n, topk, mi=3000):
    b = [(0.0,1.0)]*n
    r = minimize(obj_lbfgs, h0, args=(T,topk,n), method='L-BFGS-B',
                 bounds=b, options={'maxiter':mi,'ftol':1e-16})
    return proj(np.clip(r.x,0,1), n)

def nm_sub(h0, n, mi=400):
    mask = (h0 > 0.02) & (h0 < 0.98)
    idx = np.where(mask)[0]
    if len(idx) < 2: return h0
    fixed = h0.copy()
    x0 = h0[idx].copy()
    def su(x):
        h = fixed.copy()
        h[idx] = np.clip(x, 0, 1)
        h = proj(h, n)
        return obj_lbfgs(h, 500, 10, n)
    r = minimize(su, x0, method='Nelder-Mead',
                 options={'maxiter':mi,'xatol':1e-10,'fatol':1e-14})
    h = fixed.copy()
    h[idx] = np.clip(r.x,0,1)

                    if cv < bc - 1e-14:
                        bc = cv; h = ht; imp = True; break
                if imp: break
            if imp: break
        if not imp: break

**Genuinely different idea: Subspace optimization**

Instead of optimizing all 160 variables, I can:

**Better different idea: Penalty method with specific shifts**
