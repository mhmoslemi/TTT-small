from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                h += diff / n
            else:
                h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                h += diff / n
            else:
                h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
        # If still off, distribute
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                adj = min(abs(diff), abs(1.0 - h[i]) if diff > 0 else abs(h[i]))
                h[i] += np.sign(diff) * adj
                diff -= np.sign(diff) * adj
                if abs(diff) < 1e-14:
                    break
    return h

def smooth_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_smooth(h, T, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_logsumexp, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    h = h.copy()
    best_c5, _ = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        h_int = np.where(h > 0.5, 1.0, 0.0)
        ones = np.where(h_int == 1.0)[0]
        zeros = np.where(h_int == 0.0)[0]
        for i in ones:
            for j in zeros:
                save = h[j].copy()
                h[j] = 1.0
                h[i] = 0.0
                saved_i = save
                c5, _ = compute_c5(h)
                if c5 < best_c5 - tol:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] = saved_i
                    h[j] = 0.0
        if not improved:
            break
    # Project to fix sum exactly
    h = project_feasible(h, n)
    c5, _ = compute_c5(h)
    return h, c5

def momentum_subgrad(h, n_iters=2000):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = 0.04
    mom = 0.8
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        _, corr = compute_c5(h_cur)
        m = np.max(corr)
        thresh = m - 1e-9
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 5:
            active = active[:5]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                grad[:n-k] += (1.0 - h_cur[k:])
                grad[k:] += -h_cur[:n-k]
            else:
                kp = -k
                grad[kp:] += (1.0 - h_cur[:n-kp])
                grad[:n-kp] += -h_cur[kp:]
        grad /= len(active)
        v = mom * v - lr * grad
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 0.3)
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr *= 0.95
            v *= 0.5
            if stall > 150:

    return best_h, best_c5

def erdos_obj(x):
    return x[-1]

def erdos_eq(x):
    return x[:-1].sum() - len(x[:-1]) / 2.0

def erdos_in(x):

def slsqp_direct(h, maxiter=500):

    _, corr = compute_c5(h)
    t0 = float(np.max(corr)) + 0.02
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': erdos_eq}, {'type': 'ineq', 'fun': erdos_in}]
    try:
        res = minimize(erdos_obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            r, t = 1.0 - h, (1.0 - h).sum()
        else:

        else: h += r * (d / t)

    t0 = c.max() + 0.02
    x0 = np.concatenate([h,[t0]])

    def ce(x): return x[:-1].sum()-len(x[:-1])/2.0
    def ci(x):
        h2=x[:-1]
        return x[-1]-np.correlate(h2,1-h2,mode='full')
    try:

                     bounds=[(0,1)]*n+[(0,2)],
                     constraints=[{'type':'eq','fun':ce},{'type':'ineq','fun':ci}],
                     options={'maxiter':mi,'ftol':1e-15})
