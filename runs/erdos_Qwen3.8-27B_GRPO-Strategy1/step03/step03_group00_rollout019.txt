def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    # Softmax-like: shifted log-sum-exp
    if T < 1e-10:
        return m * dx
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_sum(h, n):
    """Project h to sum = n/2 while keeping in [0,1]."""
    h = h.copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return np.clip(h, 0, 1)
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        s = h.sum()
        if s < 1e-14:
            return np.clip(h, 0, 1)
        h = h * (1.0 + diff / s)

    h_base = h.copy()

        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_run(h, T, n, maxiter=500):
    def fun(x):
        return smooth_obj_val(x, T)
    def jac(x):
        return smooth_grad(x, T)[1]
    def con(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=80):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            coef = np.zeros(2 * n - 1)
            for sh in range(-(n - 1), n):
                c1 = 0.0
                c2 = 0.0
                if 0 <= i - sh < n:
                    c1 = 1.0 - h_base[i - sh]
                if 0 <= i + sh < n and sh != 0:
                    c2 = h_base[i + sh]
                coef[sh + n - 1] = c1 - c2
            def f1d(t):
                ct = corr_base + t * coef
                ct[n - 1] -= t * t
                return np.max(ct) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            h_try = best_h.copy()
            h_try[i] = res.x
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_step(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    sh = k_star - (n - 1)
    g = np.zeros(n)
    for i in range(n):
        c = 0.0
        if 0 <= i - sh < n:
            c += 1.0 - h[i - sh]
        if 0 <= i + sh < n:
            c -= h[i + sh]
        g[i] = c
    h_new = h - alpha * g
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.full(160, 0.5)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP with gradient, increasing T
    for T in [15, 40, 100, 250, 500, 1200, 3000, 6000]:
        if elapsed() > budget_s * 0.22:
            break
        hc = slsqp_run(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Heavy coordinate descent
    t_cd1 = t0 + budget_s * 0.52
    best_h, best_c5 = coord_descent(best_h, n, t_cd1, max_sweeps=80)

    # Phase 3: High-T SLSQP polish
    for T in [8000, 20000, 50000]:
        if elapsed() > budget_s * 0.65:
            break
        hc = slsqp_run(best_h, T, n, maxiter=700)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Subgradient + short CD cycling
    for it in range(300):
        if elapsed() > budget_s * 0.82:
            break
        alpha = 0.008 * (1.0 - 0.6 * it / 300.0) + 0.001
        h_sg = subgrad_step(best_h, n, alpha)
        h_cd, c5_cd = coord_descent(h_sg, n, min(t0 + budget_s * 0.82, time.time() + 3.0),
                                    max_sweeps=4)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        else:
            # occasional random perturbation
            if it % 30 == 29:
                h_pert = best_h + 0.03 * np.random.randn(n)
                h_pert = project_feasible(h_pert, n)
                h_p, c5_p = coord_descent(h_pert, n, min(t0 + budget_s * 0.82, time.time() + 3.0),
                                          max_sweeps=3)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p

    # Phase 5: Multi-resolution exploration
    for n_try in [120, 144, 180, 200]:
        if elapsed() > budget_s * 0.91:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        for T in [400, 2000]:
            if elapsed() > budget_s * 0.91:
                break
            hc = slsqp_run(h_interp, T, n_try, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                n = n_try
            hc2 = coord_descent(hc, n_try, t0 + budget_s * 0.91, max_sweeps=15)[0]
            c5v2 = compute_c5(hc2)
            if c5v2 < best_c5:
                best_c5 = c5v2
                best_h = hc2
                n = n_try

    # Phase 6: Final polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=40)
    if elapsed() < budget_s * 0.99:
        hc = slsqp_run(best_h, 80000, n, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
