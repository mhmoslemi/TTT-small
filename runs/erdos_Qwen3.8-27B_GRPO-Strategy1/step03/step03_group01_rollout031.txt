For a single lag L, the function g_L(h) = sum_n h[n+L]*(1-h[n]) is bilinear. Its "gradient" (diagonal part) is (1-2h) shifted, and the off-diagonal parts are linear. The Lipschitz constant of the gradient of g_L is bounded by... well, g_L is not convex or concave (it's bilinear), so standard proximal gradient doesn't directly apply.

However, if we linearize the bilinear terms: g_L(h) ≈ g_L(h_old) + ∇g_L(h_old) · (h - h_old), then minimizing the linearized version is a linear program.

**Linearization approach (Frank-Wolfe / cutting plane):**
1. At current h, compute all correlations
2. Find the max lag L*
3. Linearize: g_{L*}(h_new) ≈ g_{L*}(h) + ∇g_{L*}(h) · (h_new - h)
4. Minimize the linearized max over active lags:
   min_t t
   s.t. ∇g_{L_i}(h) · h_new <= t - g_{L_i}(h) + ∇g_{L_i}(h) · h for each active L_i
   s.t. 0 <= h_new <= 1, sum(h_new) = N/2
   
   This is a linear program! Can solve with scipy.linprog or cvxpy.

5. Update h = h_new (or a convex combination for stability)
6. Repeat

This is theFrank-Wolfe (conditional gradient) method applied to our problem! The LP oracle is solved at each iteration.

```python
def fw_step(h, N, target, n_active=5, use_line_search=True):
    """One Frank-Wolfe step: solve LP to minimize linearized C5."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = float(np.max(corr)) * dx
    nc = len(corr)
    center = N - 1
    
    k = max(1, min(n_active, nc))
    top_idx = np.argsort(corr)[-k:]
    
    # Build LP: min t
    # s.t. grad_L(h) · h_new <= t + (g_L(h) - grad_L(h)·h) for each active L
    #      i.e., grad_L(h) · h_new - t <= g_L(h) - grad_L(h)·h
    #      0 <= h_new[i] <= 1
    #      sum(h_new) = target
    
    # For lag L, g_L(h) = corr[center+L], grad_L(h)[j] as computed above
    # Linearized: g_L(h_new) ≈ g_L(h) + grad_L(h)·(h_new - h)
    # We want g_L(h) + grad_L(h)·(h_new - h) <= t
    # i.e., grad_L(h)·h_new <= t - g_L(h) + grad_L(h)·h
    
    # variables: [h_new[0], ..., h_new[N-1], t]
    # minimize: [0, ..., 0, 1] (just t)
    # constraints:
    #   For each active lag i: grad_i · h_new - t <= g_i - grad_i · h
    #   0 <= h_new[j] <= 1
    #   sum(h_new) = target
    
    from scipy.optimize import linprog
    
    c_obj = np.zeros(N + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((k, N + 1))
    b_ub = np.zeros(k)
    
    for idx, ai in enumerate(top_idx):
        L = ai - center
        g_val = corr[ai]
        grad_i = np.zeros(N)
        if L == 0:
            grad_i = 1.0 - 2.0 * h
        elif L > 0:
            grad_i[L:] += (1.0 - h[:N-L])
            grad_i[:N-L] -= h[L:]
        else:
            M = -L
            grad_i[:N-M] += (1.0 - h[M:])
            grad_i[M:] -= h[:N-M]
        
        A_ub[idx, :N] = grad_i
        A_ub[idx, N] = -1.0
        b_ub[idx] = g_val - grad_i @ h
    
    # Bounds: 0 <= h_new[j] <= 1, t >= 0
    bounds = [(0, 1)] * N + [(0, None)]
    
    # Equality: sum(h_new) = target
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([target])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:N]
        t_new = res.x[N]
        return h_new, t_new * dx, c5
    return h, float('inf'), c5
```

```python
    while time.time() < deadline:
        it += 1
        frac = min(it / 3000.0, 1.0)
        tau = tau_lo + (tau_hi - tau_lo) * frac
        topK = int(tk_lo + (tk_hi - tk_lo) * frac)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        center = N - 1
        nc = len(corr)
        sc = tau * corr
        sc -= sc.max() + 1e-15
        w_sm = np.exp(sc)
        w_sm /= w_sm.sum()
        k = max(1, min(topK, nc))
        idx_top = np.argsort(corr)[-k:]
        w_top = np.zeros(nc)
        w_top[idx_top] = 1.0 / k
        alpha = min(1.0, max(0.0, (tau - 20.0) / 80.0))
        w = (1.0 - alpha) * w_sm + alpha * w_top
        grad = np.zeros(N)
        thresh = 0.005
        active = np.where(w > thresh)[0]
        for ai in active:
            L = ai - center
            wi = w[ai]
            grad += wi * get_lag_grad(L, h, N)
        v = mom * v + grad
        h = project_sum(h - lr * v, target)
        if it % period == 0:
            lr *= decay
            if lr < 1e-10:
                break
    return best_h, best_c5

def coord_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = h[i]
            trials = [0.0, 1.0, max(0.0, orig - 0.05), min(1.0, orig + 0.05)]
            for trial in trials:
                if abs(trial - orig) < 1e-9:
                    continue
                h[i] = trial
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-10:
                    best_c5 = c5
                    best_h = h.copy()
                    improved_any = True
                    h[i] = trial
                    break
            h = best_h.copy()
            h[i] = orig
    return best_h, best_c5

def binarize_push(h, N, target):
    blurred = h.copy()
    for _ in range(3):
        blurred = 0.7 * blurred + 0.3 * np.concatenate([blurred[-1:], blurred[:-1]])
        blurred = 0.7 * blurred + 0.3 * np.concatenate([blurred[1:], blurred[:1]])
    h_bin = (blurred > 0.5).astype(float)
    h_bin = project_sum(h_bin, target)
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h_par = np.array(initial_h_values, dtype=float)
    N = len(h_par)
    dx = 2.0 / N
    target = N / 2.0
    h_par = project_sum(h_par, target)
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h, dx)
    total_useful = budget_s - 30
    # Phase 1: FW (50%)
    t1d = t0 + total_useful * 0.50
    h1, c1 = fw_optimize(best_h.copy(), N, target, t1d, n_act_start=3, n_act_end=10)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()
    # Phase 2: Subgradient (25%)
    t2d = t0 + total_useful * 0.75
    h2, c2 = subgrad_optimize(best_h.copy(), N, target, t2d,
                              tau_lo=3.0, tau_hi=300.0, tk_lo=10, tk_hi=2,
                              lr0=0.025, decay=0.97, period=70)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()
    # Phase 3: Binary push + FW restart (15%)
    if time.time() < deadline - 60:
        h_bin = binarize_push(best_h, N, target)
        c_bin = compute_c5(h_bin, dx)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()
        t3d = min(time.time() + total_useful * 0.15, deadline - 40)
        h3, c3 = fw_optimize(best_h.copy(), N, target, t3d, n_act_start=5, n_act_end=12)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
    # Phase 4: Final subgradient + coordinate polish (10%)
    if time.time() < deadline - 30:
        t4d = time.time() + min(25, (deadline - time.time()) * 0.5)
        h4, c4 = subgrad_optimize(best_h.copy(), N, target, t4d,
                                  tau_lo=200.0, tau_hi=800.0, tk_lo=4, tk_hi=1,
                                  lr0=0.005, decay=0.985, period=100)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()
    if time.time() < deadline - 5:
        best_h, best_c5 = coord_polish(best_h, N, target, deadline - 2)
    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, N
```
