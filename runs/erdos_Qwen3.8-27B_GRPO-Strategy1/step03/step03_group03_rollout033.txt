import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    probs = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = T * probs[k]
        if w < 1e-10:
            continue
        d = abs(k - (n - 1))
        if d == 0:
            grad += (1.0 - 2.0 * h) * w
        else:
            lo = slice(0, n - d)
            hi = slice(d, n)
            grad[lo] += (1.0 - h[hi]) * w
            grad[hi] += (-h[lo]) * w
    return grad * dx

def lbfgs_smooth(h, T, time_limit=60):
    n = len(h)
    h_cur = project_feasible(h, n).astype(float)
    start = time.time()
    mu = 5.0
    bnds = [(0.0, 1.0)] * n
    for outer in range(15):
        if time.time() - start > time_limit:
            break
        def obj_pen(x):
            return smooth_val(np.clip(x, 0, 1), T) + mu * (x.sum() - n / 2.0) ** 2
        def grad_pen(x):
            xc = np.clip(x, 0, 1)
            return smooth_grad(xc, T) + 2 * mu * (x.sum() - n / 2.0)
        res = minimize(obj_pen, h_cur, jac=grad_pen, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': 300, 'ftol': 1e-17, 'gtol': 1e-14})
        h_cur = np.clip(res.x, 0.0, 1.0)
        cv = abs(h_cur.sum() - n / 2.0)
        if cv < 1e-9:
            break
        mu *= 5.0
    return project_feasible(h_cur, n)

def subgrad_top_m(h, k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    thresh = np.max(corr) - 1e-8
    idx = np.where(corr >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(corr)])
    idx = idx[:k]
    grad = np.zeros(n)
    for kk in idx:
        d = abs(kk - (n - 1))
        if d == 0:
            grad += (1.0 - 2.0 * h)
        else:
            lo = slice(0, n - d)
            hi = slice(d, n)
            grad[lo] += (1.0 - h[hi])
            grad[hi] += (-h[lo])
    return grad / len(idx)

def momentum_subgrad(h, n_iters=2000, time_limit=60):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr, mom, v = 0.08, 0.85, np.zeros(n)
    stall = 0
    start = time.time()
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        g = subgrad_top_m(h_cur, 5)
        v = mom * v - lr * g
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.04, 0.5)
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr *= 0.94
            if stall > 300:
                break
    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def binary_project(h, n):
    order = np.argsort(h)
    h_b = np.zeros(n)
    h_b[order[n // 2:]] = 1.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(78)
        best_h[:39] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: L-BFGS-B with escalating T
    for T in [100, 500, 2000, 10000, 50000]:
        if time.time() - t0 > budget_s * 0.25:
            break
        remain = budget_s * 0.25 - (time.time() - t0)
        h_new = lbfgs_smooth(best_h, T, time_limit=min(80, max(10, remain * 0.5)))
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Momentum subgradient fine-tune
    if time.time() - t0 < budget_s * 0.40:
        remain = budget_s * 0.40 - (time.time() - t0)
        h_sub, c5_sub = momentum_subgrad(best_h, n_iters=int(remain * 10), time_limit=remain * 0.8)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub

    # Phase 3: Multi-resolution search
    n_candidates = sorted(set([best_n, 70, 74, 76, 80, 82, 84, 86, 88, 90, 92, 96, 100]))
    for n_try in n_candidates:
        if time.time() - t0 > budget_s * 0.65:
            break
        if n_try % 2 != 0:
            continue
        if n_try == best_n:
            h_base = best_h.copy()
        else:
            h_base = interpolate_h(best_h, best_n, n_try)
            h_base = project_feasible(h_base, n_try)
        for trial in range(3):
            if time.time() - t0 > budget_s * 0.65:
                break
            if trial == 0:
                h_pert = h_base.copy()
            elif trial == 1:
                h_pert = binary_project(h_base, n_try).astype(float)
            else:
                h_pert = h_base + np.random.randn(n_try) * 0.08
                h_pert = project_feasible(h_pert, n_try)
            rem_t = budget_s * 0.65 - (time.time() - t0)
            h_opt = lbfgs_smooth(h_pert, 3000, time_limit=min(40, rem_t * 0.4))
            h_opt, c5_opt = momentum_subgrad(h_opt, n_iters=int(rem_t * 3), time_limit=rem_t * 0.3)
            h_opt = lbfgs_smooth(h_opt, 20000, time_limit=min(20, rem_t * 0.2))
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try

    # Phase 4: Rotation and structured perturbations
    if time.time() - t0 < budget_s * 0.80:
        for shift in [1, 2, 3, 5, 8, 13, 39]:
            if time.time() - t0 > budget_s * 0.80:
                break
            h_rot = np.roll(best_h, shift, axis=0)
            h_rot = project_feasible(h_rot, best_n)
            h_opt, c5_opt = momentum_subgrad(h_rot, n_iters=1000, time_limit=15)
            h_opt = lbfgs_smooth(h_opt, 50000, time_limit=15)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 5: Fine polish - high T smooth + very small perturbations
    if time.time() - t0 < budget_s * 0.93:
        for T in [50000, 200000]:
            if time.time() - t0 > budget_s * 0.93:

            c5_pol, _ = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
        for scale in [0.01, 0.005, 0.002]:
            if time.time() - t0 > budget_s * 0.93:
                break
            h_p = best_h + np.random.randn(best_n) * scale
            h_p = project_feasible(h_p, best_n)
            h_opt, c5_opt = momentum_subgrad(h_p, n_iters=500, time_limit=10)
            h_opt = lbfgs_smooth(h_opt, 100000, time_limit=10)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Final: binary projection + one more smooth pass

        c5_final, _ = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
