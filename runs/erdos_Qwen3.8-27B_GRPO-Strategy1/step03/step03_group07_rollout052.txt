def active_set_step(h, top_k=5, step_size=0.01, n_top_ratio=0.1):
    """Take a step to reduce the top correlations"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    
    # Find active lags
    threshold = max_corr * (1 - n_top_ratio)
    active = np.where(corr >= threshold)[0]
    
    if len(active) == 0:
        return h
    
    # Compute gradient as weighted sum over active lags
    weights = np.zeros(len(corr))
    weights[active] = 1.0 / len(active)
    
    grad = np.zeros(n)
    for j in active:
        d = j - (n - 1)  # the lag
        # grad of corr[j] w.r.t. h[m]:
        # h[m] contributes as h[i] when i=m: term is (1-h[m+d]) if valid
        # h[m] contributes as h[i+d] when i=m-d: term is (1-h[m-d]) if valid
        # Wait, let me re-derive:
        # corr[j] = sum_i h[i] * (1-h[i+d]) for valid i
        # d/dh[m]: 
        #   - h[m] in first factor (i=m): contributes (1-h[m+d]) if 0<=m<n and 0<=m+d<n
        #   - h[m] in second factor (i+d=m, i=m-d): contributes -h[m-d] if 0<=m-d<n and 0<=m<n
        if d >= 0:
            # i ranges from 0 to n-1-d
            i_valid = np.arange(0, n - d)
            m_first = i_valid  # h[i] where i is in range
            m_second = i_valid + d  # h[i+d] where i+d is in range
        else:

            d2 = -d
            i_valid = np.arange(d2, n)
            m_first = i_valid
            m_second = i_valid - d2
        
        w = weights[j]
        grad[m_first] += w * (1 - h[m_second])
        grad[m_second] += w * (-h[m_first])
    
    # Projected step
    h_new = h - step_size * grad
    h_new = make_feasible(h_new, n)
    return h_new

import time

def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = T * (corr - m)
    val = m + np.log(np.sum(np.exp(s))) / T
    return val * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = T * (corr - m)
    s_max = np.max(s)
    w = np.exp(s - s_max)
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-16:
            continue
        d = j - (n - 1)
        if d >= 0:
            grad[d:] += wj * (1.0 - h[:n-d])
            grad[:n-d] += wj * (-h[d:])
        else:
            l = -d
            grad[:n-l] += wj * (1.0 - h[l:])
            grad[l:] += wj * (-h[:n-l])
    return grad * dx

    return np.ones_like(x)

def slsqp_with_grad(h0, T, n, maxiter=2000, ftol=1e-16):

        best_h = h_ref

    # Phase 3: Multi-n exploration
    for n_try in [80, 100, 120, 144, 160, 180, 200, 240, 256]:
        if time.time() - t0 > budget_s * 0.42:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = make_feasible(h_interp, n_try)
        for T in [300, 3000]:
            if time.time() - t0 > budget_s * 0.42:
                break
            try:
                h_new = slsqp_solve(h_interp, T, n_try, 1500, 1e-15)
                h_new, c5v = active_set_refine(h_new, n_try, 100)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
            except Exception:
                pass

    # Phase 4: Random restarts at small n (thue-morse, random binary)
    for n_try in [60, 80, 100, 120]:
        if time.time() - t0 > budget_s * 0.55:
            break
        for init_type in range(3):
            if time.time() - t0 > budget_s * 0.55:
                break
            if init_type == 0:
                h0 = thue_morse(n_try).astype(float)
            elif init_type == 1:
                h0 = np.random.randint(0, 2, n_try).astype(float)
            else:
                h0 = (0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_try) / max(3, n_try // 7))).astype(float)
            h0 = make_feasible(h0, n_try)
            try:
                h_new = slsqp_solve(h0, 1000, n_try, 1000, 1e-14)
                h_new, c5v = active_set_refine(h_new, n_try, 80)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
            except Exception:
                pass

    # Phase 5: Basin hopping with SLSQP polish
    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)
    rem = budget_s * 0.80 - (time.time() - t0)
    max_itr = max(20, int(rem / 8.0))
    for it in range(max_itr):
        if time.time() - t0 > budget_s * 0.80:
            break
        ptype = it % 4
        if ptype == 0:
            h_pert = bh_h + 0.08 * np.random.randn(n_bh)
        elif ptype == 1:
            h_pert = bh_h.copy()
            blen = max(2, n_bh // 12)
            s1 = np.random.randint(0, n_bh - blen)
            s2 = np.random.randint(0, n_bh - blen)
            tmp = h_pert[s1:s1 + blen].copy()
            h_pert[s1:s1 + blen] = h_pert[s2:s2 + blen]
            h_pert[s2:s2 + blen] = tmp
        elif ptype == 2:
            h_pert = bh_h.copy()
            nflip = max(1, n_bh // 10)
            idx = np.random.choice(n_bh, nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        else:
            h_pert = bh_h + 0.15 * np.random.randn(n_bh)

        h_pert = make_feasible(h_pert, n_bh)
        try:
            h_new = slsqp_solve(h_pert, 2000, n_bh, 600, 1e-13)
            h_new, c5v = active_set_refine(h_new, n_bh, 50)
        except Exception:
            h_new = h_pert
            c5v, _ = compute_c5_vals(h_pert)

        temp = 0.004 * (1.0 - it / max(1, max_itr)) + 0.0002
        accept = c5v < bh_c5 or (bh_c5 > 0 and np.random.rand() < np.exp(-(c5v - bh_c5) / temp))
        if accept:
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 6: Coordinate-pair targeted search
    n_cur = len(best_h)
    corr_cur = np.correlate(best_h, 1.0 - best_h, mode='full')
    max_lag = np.argmax(corr_cur)
    d_max = max_lag - (n_cur - 1)
    if time.time() - t0 < budget_s * 0.92:
        for _ in range(500):
            if time.time() - t0 > budget_s * 0.92:
                break
            j = np.random.randint(0, n_cur)
            for delta in [0.03, -0.03, 0.06, -0.06, 0.1, -0.1, 0.15, -0.15]:
                h_try = best_h.copy()
                h_try[j] += delta
                h_try = make_feasible(h_try, n_cur)
                c5_try, _ = compute_c5_vals(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    break

    # Phase 7: Final high-T SLSQP + active-set polish
    n_cur = len(best_h)
    for T in [20000, 100000, 500000]:
        if time.time() - t0 > budget_s * 0.98:
            break
        try:
            h_new = slsqp_solve(best_h, T, n_cur, 4000, 1e-18)
            c5v, _ = compute_c5_vals(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass
    h_final, c5_final = active_set_refine(best_h, n_cur, 300)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final

    # Try binary rounding
    h_bin = (best_h > 0.5).astype(float)
    h_bin = make_feasible(h_bin, len(best_h))
    c5_bin, _ = compute_c5_vals(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin

    best_h = make_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5_vals(best_h)
    return best_h, best_c5, len(best_h)
