            best_h = h_bin.copy()
        frac = it / max(n_iters, 1)
        T = t_start * (t_end / t_start) ** frac
        n_swaps = 1 if it % 7 != 3 else 3
        for _ in range(n_swaps):
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_c5, _ = compute_c5(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
        else:
            for _ in range(n_swaps):
                ones = np.where(h_bin == 1.0)[0]
                zeros = np.where(h_bin == 0.0)[0]
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return best_h, best_c5

def slsqp_smooth(h, T, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_direct(h, maxiter=600):
    n = len(h)
    _, corr = compute_c5(h)
    t0 = float(np.max(corr)) + 0.03

    def obj(x):
        return x[-1]

    def eq_con(x):
        return x[:-1].sum() - len(x[:-1]) / 2.0

    def ineq_con(x):
        h_part = x[:-1]
        c = np.correlate(h_part, 1.0 - h_part, mode='full')
        return x[-1] - np.max(c)

    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': eq_con}, {'type': 'ineq', 'fun': ineq_con}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(np.clip(res.x[:n], 0, 1), n)
    except Exception:
        return h

def momentum_subgrad(h, n_iters=2000, init_lr=0.05):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = init_lr
    mom = 0.75
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        _, corr = compute_c5(h_cur)
        g = gradient_c5(h_cur, corr, top_m=12)
        v = mom * v - lr * g
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 0.35)
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr *= 0.94
            v *= 0.9
            if stall > 120:
                break
    return best_h, best_c5

def active_set_slsqp(h, maxiter=500, top_k=8):
    n = len(h)
    _, corr = compute_c5(h)
    top_idx = np.argsort(corr)[-top_k:]

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return np.sum(c[top_idx]) / top_k

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    el = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 78
        best_h = np.zeros(n_init)
        for i in range(n_init):
            best_h[i] = 1.0 if (i % 7 < 3) else 0.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Quick SLSQP warm-up
    for T in [300, 2000, 10000]:
        if el() > budget_s * 0.06:
            break
        h_try = slsqp_smooth(best_h, T, maxiter=400)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Binary SA with multiple restarts
    if el() < budget_s * 0.30:
        n_sa = int((budget_s * 0.22 - el()) * 4)
        n_sa = max(n_sa, 500)
        h_bin, c5_bin = sa_binary(best_h, best_n, n_sa, rng)
        c5_bin, _ = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
        h_pol = slsqp_smooth(best_h, 8000, maxiter=600)
        c5_pol, _ = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 3: Active-set SLSQP
    if el() < budget_s * 0.42:
        for k in [6, 10, 15]:
            if el() > budget_s * 0.42:
                break
            h_try = active_set_slsqp(best_h, maxiter=500, top_k=k)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 4: Momentum subgradient
    if el() < budget_s * 0.55:
        h_sub, c5_sub = momentum_subgrad(best_h, n_iters=3000)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub
        h_sub2, c5_sub2 = momentum_subgrad(best_h, n_iters=2000, init_lr=0.02)
        if c5_sub2 < best_c5:
            best_c5 = c5_sub2
            best_h = h_sub2

    # Phase 5: Direct SLSQP (minimize t s.t. t >= max corr)
    if el() < budget_s * 0.65:
        h_try = slsqp_direct(best_h, maxiter=800)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 6: Multi-resolution search
    n_candidates = [best_n - 6, best_n - 2, best_n + 2, best_n + 6]
    for n_try in n_candidates:
        if el() > budget_s * 0.82:
            break
        if n_try < 20 or n_try % 2 != 0:
            continue
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)
        h_bin_t = np.zeros(n_try)
        srt = np.argsort(h_interp)[::-1]
        h_bin_t[srt[:n_try // 2]] = 1.0
        h_opt = slsqp_smooth(h_bin_t, 15000, maxiter=500)
        c5_opt, _ = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try
            h_opt2, c5_opt2 = momentum_subgrad(h_opt, n_iters=1000)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2

    # Phase 7: Fragment perturbations + local polish
    if el() < budget_s * 0.92:
        for scale in [0.04, 0.02, 0.008]:
            if el() > budget_s * 0.92:
                break
            h_p = best_h + rng.standard_normal(best_n) * scale
            h_p = project_feasible(h_p, best_n)
            h_opt, c5_opt = momentum_subgrad(h_p, n_iters=1200)
            h_opt = slsqp_smooth(h_opt, 50000, maxiter=500)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 8: Ultra-high T + final direct
    if el() < budget_s * 0.98:
        h_s = slsqp_smooth(best_h, 300000, maxiter=600)
        c5_s, _ = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
        h_d = slsqp_direct(best_h, maxiter=500)
        c5_d, _ = compute_c5(h_d)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
