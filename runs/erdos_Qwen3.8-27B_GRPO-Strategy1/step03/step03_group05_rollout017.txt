def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T):
    n = len(h)
    corr = get_corr(h)
    dx = 2.0 / n
    m = float(np.max(corr))
    shifted = corr - m
    log_sum_exp = np.log(np.sum(np.exp(T * shifted)))
    return (m + log_sum_exp / T) * dx

def topk_obj(h, k):
    n = len(h)
    corr = get_corr(h)
    dx = 2.0 / n
    top = np.sort(corr)[-k:]
    return float(top.sum() * dx / k)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def grad_overlap(h, s):
    n = len(h)
    g = np.zeros(n)
    if 0 <= s < n:
        g[:n - s] += (1.0 - h)[s:n]
        g[s:] -= h[:n - s]
    elif -n < s < 0:
        d = -s
        g[d:] += (1.0 - h)[:n - d]
        g[:n - d] -= h[d:n]
    return g

def subgrad_momentum(h, steps, top_k, alpha0, decay, mom):
    n = len(h)
    alpha = alpha0
    best = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    nm1 = n - 1
    for _ in range(steps):
        corr = get_corr(h)
        med = np.median(corr)
        idx = np.argpartition(corr, -top_k)[-top_k:]
        wts = np.maximum(corr[idx] - med, 0.0)
        ws = wts.sum()
        if ws < 1e-14:
            wts = np.ones(top_k) / top_k
        else:
            wts = wts / ws
        grad = np.zeros(n)
        for w, ki in zip(wts, idx):
            s = int(ki) - nm1
            grad += w * grad_overlap(h, s)
        vel = mom * vel - alpha * grad
        h = project_feasible(h + vel, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        alpha *= decay
    return best, best_c5

    return best, best_c5

            idx = np.random.choice(best_n, size=max(1, best_n // 6), replace=False)
            hp[idx] = 1.0 - hp[idx]
        hp = project_feasible(hp, best_n)
        ho = slsqp(hp, softmax_obj, (500,), 150, 1e-12)
        ho = slsqp(ho, topk_obj, (3,), 150, 1e-13)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho

    # Phase 6: Multi-resolution exploration (10%)
    for nr in [100, 150, 250, 300]:
        if not within(0.95):
            break
        if nr == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, nr)
        hu = np.interp(x_new, x_old, best_h)
        hu = project_feasible(hu, nr)
        hu = project_feasible(hu + np.random.randn(nr) * 0.012, nr)
        hh = slsqp(hu, softmax_obj, (400,), 200, 1e-12)
        hh = slsqp(hh, topk_obj, (3,), 200, 1e-13)
        c5_hi = compute_c5(hh)
        if c5_hi < best_c5:
            best_c5, best_h, best_n = c5_hi, hh, nr
        elif nr > best_n:
            x_dn = np.linspace(0, 2, best_n)
            hd = np.interp(x_dn, x_new, hh)
            hd = project_feasible(hd, best_n)
            c5_dn = compute_c5(hd)
            if c5_dn < best_c5:
                best_c5, best_h = c5_dn, hd

    # Phase 7: Final intensive polish
    if within(0.99):
        hs, cs = subgrad_momentum(best_h, 150, 3, 0.003, 0.9997, 0.9)
        if cs < best_c5:
            best_c5, best_h = cs, hs
        for T in [20000, 100000]:
            if not within(0.99):
                break
            ht = slsqp(best_h, softmax_obj, (T,), 500, 1e-15)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht
        ht = slsqp(best_h, topk_obj, (2,), 600, 1e-15)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht
        ht = slsqp(best_h, topk_obj, (3,), 500, 1e-15)
        ct = compute_c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
