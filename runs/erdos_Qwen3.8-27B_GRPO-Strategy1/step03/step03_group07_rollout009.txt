    c1 = np.sum(weights[None, :] * valid1, axis=1)  # (n,)

    h1 = np.where(valid1, h[np.clip(idx1, 0, n-1)], 0.0)  # (n, 2n-1)

    s2 = np.sum(weights[None, :] * h2, axis=1)
    
    grad = dx * (s1 - s2)
    return grad

    h = np.clip(h, 0.0, 1.0).astype(float).copy()

    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h

        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-13:
            h += room * (diff / tot)
    else:
        room = h.copy()
        tot = room.sum()
        if tot > 1e-13:
            h += room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

    return x.sum() - len(x) / 2.0

    return np.ones(len(x))

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    kk = np.arange(2*n-1)[None, :]

    v2 = (idx2 >= 0) & (idx2 <= n-1)

    h2 = np.where(v2, h[np.clip(idx2,0,n-1)], 0.0)

    n_cur = len(best_h)
    for T in [100000, 500000]:
        if time.time() - t0 > budget_s * 0.99:
            break
        try:
            h_new = slsqp_opt(best_h, T, n_cur, 3000, 1e-16)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_new
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
