from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_topk_c5(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    topk = corr[idx]
    m = np.max(topk)
    val = m + np.log(np.sum(np.exp(T * (topk - m)))) / T
    return val * dx

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    # Sort by value, keep top n/2 as 1
    idx = np.argsort(-h)[:n//2]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total < 1e-13:
            return h
        h = h + room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def binary_energy(h, n):
    """C5 for a binary sequence."""
    return compute_c5(h)

def slsqp_smooth(h0, T, n, topk, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    bnds = [(0.0, 1.0)] * n

                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0, 1), n)

def slsqp_refine(h0, n, T, topk, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    if topk is not None:
        res = minimize(smooth_topk_obj, h0, args=(T, topk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    else:
        res = minimize(smooth_max_obj, h0, args=T,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    if res.success or res.fun < compute_c5(h0):
        return project_feasible(np.clip(res.x, 0, 1), n)
    return h0

def subgrad_step(h, n, step_size, num_worst):
    corr = np.correlate(h, 1.0 - h, mode='full')
    worst = np.argsort(corr)[-num_worst:]
    grad = np.zeros_like(h)
    for k in worst:
        shift = k - (n - 1)
        for i in range(n):
            if 0 <= i + shift < n:
                grad[i] += (1.0 - h[i + shift])
            if 0 <= i - shift < n:
                grad[i] += h[i - shift]
    grad = -grad / num_worst
    h_new = h + step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new
