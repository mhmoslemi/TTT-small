import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_grad(h, n, shifts, weights):
    """Compute exact gradient of weighted sum of correlations."""
    grad = np.zeros(n)
    for s, w in zip(shifts, weights):
        a = abs(s)
        if a == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[a:] += w * (1.0 - h[:-a])
            grad[:-a] -= w * h[a:]
    return grad

def softmax_weights(corr, T, k):
    """Get weights for top-k correlations using softmax."""
    m = np.max(corr)
    scaled = T * (corr - m)
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    return w  # full weight vector, but concentrated on top-k

def grad_line_search(h, grad, n, max_iter=100):
    """Backtracking line search with projection."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.01
    for _ in range(max_iter):
        h_new = project_feasible(h - alpha * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            alpha = min(alpha * 1.5, 0.1)
        else:
            alpha *= 0.7
            if alpha < 1e-8:
                break
    return best_h, best_c5

def multistart_slsqp(h, n, T, maxiter, ftol):
    """SLSQP with softmax-smoothed max objective."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return float((m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        exp_t = np.exp(T * (corr - m))
        w = exp_t / exp_t.sum()
        # shifts for full correlation: shift_k = k - (n-1)
        shifts = np.arange(2*n-1) - (n-1)
        g = np.zeros(n)
        for s, wi in zip(shifts, w):
            a = abs(s)
            if a == 0:
                g += wi * (1.0 - 2.0 * x)
            else:
                g[a:] += wi * (1.0 - x[:-a])
                g[:-a] -= wi * x[a:]
        return g * dx
    
    def constraint_fun(x):
        return x.sum() - n / 2.0
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def basin_hop(h, n, budget_per_trial, rng):
    """Generate a perturbed candidate and optimize."""
    r = rng.random()
    if r < 0.25:
        # Block move
        h_new = h.copy()
        start = rng.randint(0, max(1, n - 10))
        length = rng.randint(3, max(4, n // 4))
        end = min(start + length, n)
        shift = rng.randint(1, max(2, length // 2))
        h_new[start:end] = np.roll(h_new[start:end], shift)
    elif r < 0.45:
        # Gaussian perturbation
        scale = rng.uniform(0.02, 0.12)
        h_new = h + rng.randn(n) * scale
        h_new = project_feasible(h_new, n)
    elif r < 0.65:
        # Targeted repair on worst shift
        corr = compute_all_corr(h)
        s_star = int(np.argmax(corr) - (n - 1))
        h_new = h.copy()
        for i in range(n):
            j = i + s_star

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        room = (1.0 - h) if diff > 0 else h.copy()
        total = room.sum()
        if total < 1e-14:
            break
        h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_grad(h, n, shifts, weights):
    grad = np.zeros(n)
    for s, w in zip(shifts, weights):
        a = abs(s)
        if a == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[a:] += w * (1.0 - h[:-a])
            grad[:-a] -= w * h[a:]
    return grad

def softmax_w(corr, T):
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    return e / e.sum()

def line_search(h, grad, n, alpha0=0.05):
    best_h, best_c5 = h.copy(), compute_c5(h)
    alpha = alpha0
    for _ in range(40):
        h_new = project_feasible(h - alpha * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5, alpha = h_new, c5_new, min(alpha * 1.4, 0.2)
        else:
            alpha *= 0.6
            if alpha < 1e-9:
                break
    return best_h, best_c5

def slsqp_smooth(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    shifts = np.arange(2 * n - 1) - (n - 1)
    def obj(x):
        c = compute_all_corr(x)
        return float((np.max(c) + np.log(np.sum(np.exp(T * (c - np.max(c))))) / T) * dx)
    def objg(x):
        c = compute_all_corr(x)
        w = softmax_w(c, T)
        return exact_grad(x, n, shifts, w) * dx
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_topk(h, n, k, maxiter, ftol):
    dx = 2.0 / n
    shifts = np.arange(2 * n - 1) - (n - 1)
    def obj(x):
        c = compute_all_corr(x)
        top = np.partition(c, -k)[-k:]
        return float(np.sum(top) * dx / k)
    def objg(x):
        c = compute_all_corr(x)
        idx = np.partition(c, -k)[-k:]
        w = np.zeros(2 * n - 1)
        w_sorted = np.sort(c)[::-1][:k]
        ii = np.where(c == w_sorted[0])[0][0]
        mask = np.zeros(2 * n - 1, dtype=bool)
        mask[np.argpartition(c, -k)[-k:]] = True
        w[mask] = 1.0 / k
        return exact_grad(x, n, shifts[mask], w[mask]) * dx
    def cfun(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n

    try:
