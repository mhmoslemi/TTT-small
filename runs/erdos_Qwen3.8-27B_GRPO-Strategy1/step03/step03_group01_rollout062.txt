                c = compute_c5(h, dx)
                if c < best_c5 - 1e-11:
                    best_c5 = c
                    best_h = h.copy()
                    improved = True
                    h = best_h.copy()
                    break
            h[i] = best_h[i]
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    best_h = np.array(initial_h_values, dtype=float)
    Np = len(best_h)
    best_c5 = compute_c5(best_h, 2.0 / Np)
    x_par = np.linspace(0.0, 2.0, Np, endpoint=False)

    # Phase 1: N=450, broad exploration
    N1 = 450
    tgt1 = N1 / 2.0
    x1 = np.linspace(0.0, 2.0, N1, endpoint=False)
    h1 = np.interp(x1, x_par, best_h)
    h1 = project_sum(h1, tgt1)
    t1_end = t0 + 280
    h1, c1 = nesterov_loop(h1, N1, tgt1, t1_end, (1.5, 150.0), (25, 3), 0.035, 0.955, 45, 0.55)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: tighten on same grid
    if time.time() < deadline - 250:
        t2_end = min(time.time() + 180, deadline - 200)
        h1, c1 = nesterov_loop(h1, N1, tgt1, t2_end, (150.0, 800.0), (4, 1), 0.004, 0.975, 55, 0.6)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 3: N=400, different discretization
    if time.time() < deadline - 200:
        N3 = 400
        tgt3 = N3 / 2.0
        x3 = np.linspace(0.0, 2.0, N3, endpoint=False)
        # Interpolate from best (not just parent)
        x_best = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        h3 = np.interp(x3, x_best, best_h)
        h3 = project_sum(h3, tgt3)
        # Small random perturbation
        h3 += rng.randn(N3) * 0.015
        h3 = project_sum(h3, tgt3)
        t3_end = min(time.time() + 170, deadline - 150)
        h3, c3 = nesterov_loop(h3, N3, tgt3, t3_end, (3.0, 400.0), (18, 2), 0.03, 0.96, 50, 0.5)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()

    # Phase 4: N=520, finer grid from best
    if time.time() < deadline - 150:
        N4 = 520
        tgt4 = N4 / 2.0
        x4 = np.linspace(0.0, 2.0, N4, endpoint=False)
        x_best = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        h4 = np.interp(x4, x_best, best_h)
        h4 = project_sum(h4, tgt4)
        h4 += rng.randn(N4) * 0.01
        h4 = project_sum(h4, tgt4)
        t4_end = min(time.time() + 130, deadline - 100)
        h4, c4 = nesterov_loop(h4, N4, tgt4, t4_end, (5.0, 300.0), (15, 2), 0.025, 0.96, 50, 0.45)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()

    # Phase 5: Optimize best at native grid
    if time.time() < deadline - 80:
        Nb = len(best_h)
        t5_end = min(time.time() + 100, deadline - 50)
        best_h, best_c5 = nesterov_loop(best_h, Nb, Nb / 2.0, t5_end,
                                         (30.0, 600.0), (6, 1), 0.01, 0.97, 55, 0.5)

    # Phase 6: Equalize top lags
    if time.time() < deadline - 40:
        Nb = len(best_h)
        best_h, best_c5 = equalize_top(best_h, Nb, Nb / 2.0, topM=4, iters=100, step_sz=0.004)
        best_h, best_c5 = equalize_top(best_h, Nb, Nb / 2.0, topM=2, iters=60, step_sz=0.002)

    # Phase 7: Perturbation restart
    if time.time() < deadline - 30:
        Nb = len(best_h)
        h_r = best_h + rng.randn(Nb) * 0.008
        h_r = project_sum(h_r, Nb / 2.0)
        t_r_end = min(time.time() + 20, deadline - 10)
        h_r, c_r = nesterov_loop(h_r, Nb, Nb / 2.0, t_r_end, (100.0, 500.0), (3, 1), 0.003, 0.98, 40, 0.5)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()

    # Phase 8: Coordinate polish
    if time.time() < deadline - 3:
        Nb = len(best_h)
        best_h, best_c5 = polish(best_h, Nb, Nb / 2.0, deadline - 2)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
