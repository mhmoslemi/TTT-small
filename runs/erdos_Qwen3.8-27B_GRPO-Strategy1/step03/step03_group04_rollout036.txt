def anneal_binary(h, corr, N, deadline, rng, T0=3.0, T1=0.0003):
    M = 2 * N - 1
    cur_peak = float(np.max(corr))
    best_peak = cur_peak
    best_h = h.copy()
    
    # Build position lists
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    d = np.zeros(M)
    n_total = int((deadline - time.time()) * 12000)
    n_total = max(n_total, 5000)
    
    for it in range(n_total):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / n_total)
        
        oi = rng.randint(len(ones_list))
        i = ones_list[oi]
        zi = rng.randint(len(zeros_list))
        j = zeros_list[zi]
        
        h_rev = h[::-1]
        d[:] = 0.0
        d[N-1-i : 2*N-1-i] += h - 1.0
        d[i : N+i] += h_rev
        d[N-1-j : 2*N-1-j] += 1.0 - h
        d[j : N+j] -= h_rev
        
        new_peak = float(np.max(corr + d))
        delta = new_peak - cur_peak
        
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            # Accept
            corr += d
            h[i] = 0.0
            h[j] = 1.0
            cur_peak = new_peak
            # Update lists
            ones_list[oi] = ones_list[-1]
            ones_list.pop()
            ones_list.append(j)
            zeros_list[zi] = zeros_list[-1]
            zeros_list.pop()
            zeros_list.append(i)
            
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_h = h.copy()
    
    return best_h, best_peak

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def to_binary(h, N):
    nh = N // 2
    order = np.argsort(-h, kind='stable')
    h_b = np.zeros(N)
    h_b[order[:nh]] = 1.0
    return h_b

def greedy_descent(h, corr, N, deadline):
    M = 2 * N - 1
    best_peak = float(np.max(corr))
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:

        h_rev = h[::-1]
        omh = 1.0 - h
        nz = len(zeros)
        A_zeros = np.zeros((nz, M))
        for zi in range(nz):
            j = zeros[zi]
            A_zeros[zi, N-1-j : 2*N-1-j] = omh
            A_zeros[zi, j : N+j] -= h_rev
