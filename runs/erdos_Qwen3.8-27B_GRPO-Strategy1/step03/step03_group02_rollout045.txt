            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad, c5_val

def gradient_topk(h, N, tau, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr)) * (2.0 / N)
    center = N - 1
    if K >= len(corr):
        top_idx = np.arange(len(corr))
        c_top = corr.copy()
    else:
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
    sc = tau * c_top
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), c5_val
    w /= ws
    grad = np.zeros(N)
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-9:
            continue
        k = int(top_idx[ix])
        L = k - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:N - M]
    return grad, c5_val

def grad_descent(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, use_full, K, iters_max):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    it = 0
    while time.time() < deadline and it < iters_max:
        frac = min(it / max(iters_max, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        if use_full:
            grad, c5_val = gradient_full(h, N, tau)
        else:
            grad, c5_val = gradient_topk(h, N, tau, K)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        it += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c5

def coord_descent(h, N, target, dx, n_passes, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    deltas = np.array([-0.015, -0.008, -0.003, 0.003, 0.008, 0.015, -0.001, 0.001])
    for _ in range(n_passes):
        if time.time() > deadline:
            break
        for i in range(N):
            if time.time() > deadline:
                break
            for d in deltas:
                trial = best_h.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_sum(trial, target)
                c = compute_c5(trial, dx)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial.copy()
                    break
    return best_h, best_c5

def perturb_search(h, N, target, dx, n_trials, deadline, scale):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(777)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        trial = best_h.copy()
        n_mod = rng.randint(1, 5)
        idx = rng.choice(N, size=n_mod, replace=False)
        trial[idx] += rng.randn(n_mod) * scale
        trial = project_sum(trial, target)
        c = compute_c5(trial, dx)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = trial.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10.0
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    # Primary: optimize at N=150 (parent size) with full gradient
    N = 150
    dx = 2.0 / N
    target = N / 2.0
    h_init = h0.copy() if N == n0 else project_sum(np.interp(np.linspace(0, 2, N, endpoint=False), x0, h0), target)

    # Phase 1: Full gradient, moderate sweep (majority of time)
    p1_end = time.time() + (deadline - time.time()) * 0.30
    h1, c1 = grad_descent(h_init.copy(), N, target, dx, p1_end,
                          tau0=25.0, tau1=300.0, lr0=0.04, lr1=0.004,
                          use_full=True, K=50, iters_max=20000)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: Full gradient, high tau
    p2_end = time.time() + (deadline - time.time()) * 0.30
    h2, c2 = grad_descent(h1.copy(), N, target, dx, p2_end,
                          tau0=300.0, tau1=3000.0, lr0=0.008, lr1=0.0003,
                          use_full=True, K=50, iters_max=20000)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()

    # Phase 3: Top-K focused, very high tau (subgradient-like)
    if time.time() < deadline - 100:
        p3_end = time.time() + (deadline - time.time()) * 0.35
        h3, c3 = grad_descent(h2.copy(), N, target, dx, p3_end,
                              tau0=3000.0, tau1=20000.0, lr0=0.003, lr1=0.0001,
                              use_full=False, K=5, iters_max=30000)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
    else:
        h3 = h2.copy()

    # Phase 4: Coordinate descent polish
    if time.time() < deadline - 30:
        h4, c4 = coord_descent(h3.copy(), N, target, dx,
                               n_passes=4, deadline=deadline - 20)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()
    else:
        h4 = h3.copy()

    # Phase 5: Perturbation search
    if time.time() < deadline - 15:
        h5, c5v = perturb_search(h4.copy(), N, target, dx,
                                 n_trials=5000, deadline=deadline - 8, scale=0.025)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h5.copy()

    # Try N=200 for diversity (if time remains)
    if time.time() < deadline - 120:
        N2 = 200
        dx2 = 2.0 / N2
        tgt2 = N2 / 2.0
        x2 = np.linspace(0, 2, N2, endpoint=False)
        h2_init = project_sum(np.interp(x2, np.linspace(0, 2, len(best_h), endpoint=False), best_h), tgt2)
        p6_end = time.time() + (deadline - time.time()) * 0.45
        h6, c6 = grad_descent(h2_init, N2, tgt2, dx2, p6_end,
                              tau0=20.0, tau1=2000.0, lr0=0.035, lr1=0.001,
                              use_full=True, K=40, iters_max=15000)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6.copy()
            if time.time() < deadline - 40:
                h6b, c6b = coord_descent(h6.copy(), N2, tgt2, dx2,
                                         n_passes=2, deadline=deadline - 10)
                if c6b < best_c5:
                    best_c5 = c6b
                    best_h = h6b.copy()

    # Try N=120 (smaller, easier to optimize fully)
    if time.time() < deadline - 100:
        N3 = 120
        dx3 = 2.0 / N3
        tgt3 = N3 / 2.0
        x3 = np.linspace(0, 2, N3, endpoint=False)
        h3_init = project_sum(np.interp(x3, np.linspace(0, 2, len(best_h), endpoint=False), best_h), tgt3)
        p7_end = time.time() + (deadline - time.time()) * 0.5
        h7, c7 = grad_descent(h3_init, N3, tgt3, dx3, p7_end,
                              tau0=20.0, tau1=1500.0, lr0=0.04, lr1=0.001,
                              use_full=True, K=40, iters_max=20000)
        if c7 < best_c5:
            best_c5 = c7
            best_h = h7.copy()
            if time.time() < deadline - 20:
                h7b, c7b = coord_descent(h7.copy(), N3, tgt3, dx3,
                                         n_passes=3, deadline=deadline - 5)
                if c7b < best_c5:
                    best_c5 = c7b
                    best_h = h7b.copy()

    # Final micro-polish on best
    Nb = len(best_h)
    if time.time() < deadline - 5:
        dx_b = 2.0 / Nb
        hF, cF = coord_descent(best_h.copy(), Nb, Nb // 2, dx_b,
                               n_passes=2, deadline=deadline - 1)
        if cF < best_c5:
            best_c5 = cF
            best_h = hF

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
