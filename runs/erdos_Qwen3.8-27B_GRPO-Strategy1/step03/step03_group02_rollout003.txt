        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_gradient(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):
        wi = w[ix]
        if wi < 1e-9:
            continue
        k = ix - center
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:N - k] += wi * (1.0 - h[k:])
            grad[k:] += wi * (-h[:N - k])
        else:
            m = -k
            grad[m:] += wi * (1.0 - h[:N - m])
            grad[:N - m] += wi * (-h[m:])
    return grad, true_c5

def optimize(h, N, target, dx, deadline, tau0, tau1, lr0, lr1, iters, b1=0.9, b2=0.999):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    eps = 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while time.time() < deadline and it < iters:
        frac = min(it / max(iters, 1), 1.0)
        tau = tau0 * (tau1 / tau0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        grad, c5v = softmax_gradient(h, N, tau)
        if c5v < best_c5 - 1e-12:
            best_c5 = c5v
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    return best_h, best_c5

def perturb(h, N, target, scale, rng):
    t = h.copy()
    idx = rng.choice(N, size=rng.randint(1, 6), replace=False)
    t[idx] += rng.randn(len(idx)) * scale
    return project_sum(t, target)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_grad(h, N, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    true_c5 = float(np.max(corr)) * (2.0 / N)
    sc = tau * corr - tau * corr.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(N), true_c5
    w /= ws
    grad = np.zeros(N)
    center = N - 1
    for ix in range(len(w)):

        else:
            m = -k
            grad[m:] += wi * (1.0 - h[:N - m])
            grad[:N - m] += -wi * h[m:]
    return grad, true_c5

def adam_step(h, grad, N, target, m, v, it, lr, b1, b2, eps):
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * grad * grad
    it += 1
    mh = m / (1.0 - b1 ** it)
    vh = v / (1.0 - b2 ** it)
    h_new = h - lr * mh / (np.sqrt(vh) + eps)
    return project_sum(h_new, target), m, v, it

        h, m, v, it = adam_step(h, grad, N, target, m, v, it, lr, b1, b2, eps)

def contrast_push(h, target, strength):
    g = 0.5 + strength * (h - 0.5)
    g = np.clip(g, 0.0, 1.0)
