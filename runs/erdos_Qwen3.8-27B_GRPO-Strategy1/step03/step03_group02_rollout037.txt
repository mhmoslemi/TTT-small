def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_grad(h, K):
    N = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    n_c = len(corr)
    if K >= n_c:
        sel = np.arange(n_c)
        w = np.ones(n_c) / n_c
    else:
        sel = np.argsort(corr)[-K:]
        cs = corr[sel]
        m = cs[-1]
        w = np.exp(cs - m)
        ws = w.sum()
        if ws < 1e-30:
            return np.zeros(N), float(corr.max()) * (2.0/N)
        w /= ws
    center = N - 1

        idx = rng.choice(N, size=n_mod, replace=False)
        trial[idx] += rng.randn(n_mod) * scale
        trial = project_sum(trial, target)
        c = compute_c5(trial)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = trial.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)

    # Size 1: the initial size
    N1 = n0
    # Size 2: smaller (easier to optimize)
    N2 = 100
    # Size 3: larger (more resolution)
    N3 = 220

    # Allocate time: 45% N1, 20% N2, 20% N3, 15% final
    total_t = deadline - time.time()
    t1 = time.time() + total_t * 0.45
    t2 = t1 + total_t * 0.20
    t3 = t2 + total_t * 0.20

    # Optimize at N1
    target1 = N1 / 2.0
    h_w = h0.copy()
    h_a, c_a = run_phase(h_w, N1, target1, min(t1, deadline - 100),
                         15.0, 150.0, 0.05, 0.004, 50, 8, 500)
    if c_a < best_c5:
        best_c5, best_h = c_a, h_a.copy()
    # High-tau polish at N1
    if time.time() < t1:
        h_b, c_b = run_phase(h_a, N1, target1, t1,
                             200.0, 3000.0, 0.004, 0.0003, 10, 2, 500)
        if c_b < best_c5:
            best_c5, best_h = c_b, h_b.copy()
    # Multi-peel at N1
    if time.time() < deadline - 60:
        d_mp = min(time.time() + 40.0, deadline - 50)
        h_mp, c_mp = multi_peel(h_b, N1, target1, d_mp, n_targets=3)
        if c_mp < best_c5:
            best_c5, best_h = c_mp, h_mp.copy()
    # Single peel at N1
    if time.time() < deadline - 30:
        d_sp = min(time.time() + 30.0, deadline - 25)
        h_sp, c_sp = peel_phase(h_mp, N1, target1, d_sp)
        if c_sp < best_c5:
            best_c5, best_h = c_sp, h_sp.copy()

    # Optimize at N2 (smaller)
    if time.time() < t2:
        target2 = N2 / 2.0
        xN2 = np.linspace(0.0, 2.0, N2, endpoint=False)
        h_init2 = np.interp(xN2, x0, h0)
        h_init2 = project_sum(h_init2, target2)
        h_c, c_c = run_phase(h_init2, N2, target2, min(t2, deadline - 60),
                             15.0, 200.0, 0.06, 0.005, 40, 6, 600)
        if c_c < best_c5:
            best_c5, best_h = c_c, h_c.copy()
        if time.time() < t2:
            h_d, c_d = run_phase(h_c, N2, target2, t2,
                                 300.0, 5000.0, 0.003, 0.0002, 8, 2, 400)
            if c_d < best_c5:
                best_c5, best_h = c_d, h_d.copy()
        # Peel at N2
        if time.time() < deadline - 20:
            h_e, c_e = peel_phase(h_d, N2, target2, min(time.time() + 20, deadline - 15))
            if c_e < best_c5:
                best_c5, best_h = c_e, h_e.copy()
        # Local search at N2
        if time.time() < deadline - 10:
            h_f, c_f = local_search(h_e, N2, target2, deadline - 8, rng, 0.01, 1500)
            if c_f < best_c5:
                best_c5, best_h = c_f, h_f.copy()

    # Optimize at N3 (larger)
    if time.time() < t3:
        target3 = N3 / 2.0
        xN3 = np.linspace(0.0, 2.0, N3, endpoint=False)
        h_init3 = np.interp(xN3, x0, h0)
        h_init3 = project_sum(h_init3, target3)
        h_g, c_g = run_phase(h_init3, N3, target3, min(t3, deadline - 50),
                             20.0, 300.0, 0.03, 0.003, 40, 8, 400)
        if c_g < best_c5:
            best_c5, best_h = c_g, h_g.copy()
        if time.time() < t3:
            h_h, c_h = run_phase(h_g, N3, target3, t3,
                                 300.0, 3000.0, 0.003, 0.0003, 10, 3, 300)
            if c_h < best_c5:
                best_c5, best_h = c_h, h_h.copy()

    # Final polish at the best size
    if time.time() < deadline - 8:
        Nb = len(best_h)
        tb = Nb // 2
        fp_dead = deadline - 3
        h_fp, c_fp = multi_peel(best_h.copy(), Nb, tb, min(fp_dead, time.time() + 60), n_targets=2)
        if c_fp < best_c5:
            best_c5, best_h = c_fp, h_fp.copy()
        if time.time() < fp_dead - 5:
            h_fp2, c_fp2 = peel_phase(best_h.copy(), Nb, tb, fp_dead)
            if c_fp2 < best_c5:
                best_c5, best_h = c_fp2, h_fp2.copy()
        # Tiny local search
        if time.time() < fp_dead - 2:
            h_fi, c_fi = local_search(best_h.copy(), Nb, tb, fp_dead, rng, 0.008, 500)
            if c_fi < best_c5:
                best_c5, best_h = c_fi, h_fi.copy()

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
