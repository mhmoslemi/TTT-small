def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def subgradient_of_max_corr(h, n, top_k=3):
    """Compute subgradient of max correlation w.r.t. h"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top k lags
    top_idx = np.argsort(corr)[-top_k:]
    max_val = corr[top_idx[0]]
    
    grad = np.zeros(n)
    for idx in top_idx:
        L = n - 1 - idx
        if L == 0:
            grad += 1.0 - 2.0 * h
        elif L > 0:
            grad[:n-L] += (1.0 - h[L:])
            grad[L:] -= h[:n-L]
        else:
            grad[-L:] += (1.0 - h[:n+L])
            grad[:n+L] -= h[-L:]
    return grad, max_val * (2.0/n)

    # Projected gradient descent
    h_new = h - lr * grad
    h_new = make_feasible(h_new, n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

        h = np.clip(res.x, 0, 1)
        return make_feasible(h, n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * (diff / tot)
    else:
        tot = h.sum()
        if tot > 1e-14:
            h -= h * (-diff / tot)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.logaddexp.reduce(T * (corr - m)) / T  # numerically stable

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot > 1e-14:
                h -= h * min(1.0, -diff / max(tot, 1e-14))
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def sum_con_fun(x):
    return x.sum() - len(x) / 2.0

def slsqp_opt(h0, T, n, maxiter=2000, ftol=1e-16):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun}]
    try:
        r = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(np.clip(r.x, 0, 1), n)
    except Exception:
        return make_feasible(h0.copy(), n)

def check_time(t0, budget_s, frac):
    return (time.time() - t0) < budget_s * frac

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        best_h = np.array(ihv, dtype=float).copy()
    else:
        best_h = np.tile([1.0, 0.0], 100)
    n_best = len(best_h)
    best_h = make_feasible(best_h, n_best)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive T SLSQP (warm-up + converge)
    p1_end = budget_s * 0.48
    T_seq = [50, 200, 1000, 5000, 20000, 100000]
    for T in T_seq:
        if not check_time(t0, budget_s, 0.48):
            break
        h_try = slsqp_opt(best_h.copy(), T, n_best, 3000, 1e-16)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        # Second pass at same T for polish
        if check_time(t0, budget_s, 0.46):
            h_try2 = slsqp_opt(best_h.copy(), T, n_best, 2000, 1e-17)
            c5_try2 = compute_c5(h_try2)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2

    # Phase 2: Exploit h <-> 1-h symmetry for basin exploration
    p2_end = budget_s * 0.72
    # Start from complement
    h_comp = 1.0 - best_h
    h_comp = make_feasible(h_comp, n_best)
    h_comp_opt = slsqp_opt(h_comp, 10000, n_best, 3000, 1e-16)
    c5_comp = compute_c5(h_comp_opt)
    if c5_comp < best_c5:

        best_h = h_comp_opt

    bh_h = best_h.copy()
    bh_c5 = best_c5
    n_bh = len(bh_h)

            break

                h_p[s1:s1+bl] = h_p[s2:s2+bl]

        # Also try from complement

        h_opt2 = slsqp_opt(h_int2, 5000, n_try, 1000, 1e-15)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2

    # Phase 4: Final extreme-T SLSQP polish
    n_cur = len(best_h)

    return best_h, best_c5, len(best_h)
