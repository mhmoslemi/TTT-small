import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_grad(h, dx, beta, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    N = len(h)
    center = N - 1
    thresh = np.partition(corr, -K)[-K]
    mask = corr >= thresh
    idx_active = np.where(mask)[0]
    c_active = corr[mask]
    c_max = c_active.max()
    shifted = beta * (c_active - c_max)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    obj = (c_max + np.log(exps.sum()) / beta) * dx
    grad = np.zeros(N)
    for i in range(len(w)):
        wi = w[i]
        if wi < 1e-12:
            continue
        k = idx_active[i] - center
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:N - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:N - k]
        else:
            m = -k
            grad[m:] += wi * (1.0 - h[:N - m])
            grad[:N - m] -= wi * h[m:]
    grad *= dx
    return obj, grad

def optimize_lse(h, target, dx, beta, K, lr, mu, deadline, max_iter):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros_like(h)
    patience = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        h_eval = project_sum(h - mu * v, target)
        _, grad = lse_grad(h_eval, dx, beta, K)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        lr_adapt = lr * min(1.0, 0.01 / (gnorm + 1e-10))
        v = mu * v + lr_adapt * grad
        h = project_sum(h - v, target)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 500:
                lr *= 0.7
                v *= 0.5
                patience = 0
    return best_h, best_c5
