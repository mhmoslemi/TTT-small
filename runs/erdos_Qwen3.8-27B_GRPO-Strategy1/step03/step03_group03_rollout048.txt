    """Find the worst shifts and do targeted 1D optimization."""
    best = h.copy()
    best_c5, _ = compute_c5(best)
    for _ in range(iters):
        _, corr = compute_c5(best)
        # Get top shifts
        top_idx = np.argsort(corr)[-top_k:]

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            grad[:n-k] += w[idx] * (-h[k:])
            grad[k:] += w[idx] * (1.0 - h[:n-k])
        else:
            kp = -k
            grad[kp:] += w[idx] * (1.0 - h[:n-kp])
            grad[:n-kp] += w[idx] * (-h[kp:])
    return grad * dx

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def sum_eq_jac(x):
    return np.ones(len(x))

def slsqp_opt(h, T, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    try:
        res = minimize(lse_obj, h, args=(T,), jac=lambda x: lse_grad(x, T),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def targeted_refine(h, n, top_k=4, iters=800, t0=time.time(), budget=50.0):
    best = h.copy()
    best_c5, _ = compute_c5(best)
    for _ in range(iters):
        if time.time() - t0 > budget:
            break
        _, corr = compute_c5(best)
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for idx in top_idx:
            k = idx - (n - 1)
            d = np.zeros(n)
            if k >= 0:
                d[:n-k] = -best[k:]
                d[k:] += (1.0 - best[:n-k])
            else:
                kp = -k
                d[kp:] = (1.0 - best[:n-kp])
                d[:n-kp] += -best[kp:]
            d -= np.mean(d)
            for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                h_try = project_feasible(best + alpha * d, n)
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def momentum_subgrad(h, n_iters=2000, t0=None, budget=30.0):
    if t0 is None:
        t0 = time.time()
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5, _ = compute_c5(h_cur)
    lr = 0.04
    mom = 0.75
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        if time.time() - t0 > budget:
            break
        _, corr = compute_c5(h_cur)
        m = np.max(corr)
        thresh = m - 0.5
        active = np.where(corr >= thresh)[0][:6]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        g = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                g[:n-k] += -h_cur[k:]
                g[k:] += (1.0 - h_cur[:n-k])
            else:
                kp = -k
                g[kp:] += (1.0 - h_cur[:n-kp])
                g[:n-kp] += -h_cur[kp:]
        g /= len(active)
        g -= np.mean(g)
        v = mom * v - lr * g
        h_new = project_feasible(h_cur + v, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 0.3)
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr *= 0.95
            if stall > 150:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    else:
        best_n = 78
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: SLSQP with escalating T
    for T in [2000, 10000, 50000, 200000, 500000]:
        if time.time() - t0 > budget_s * 0.12:
            break
        h_try = slsqp_opt(best_h, T, maxiter=800)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: Momentum subgradient
    if time.time() - t0 < budget_s * 0.25:
        h_sub, c5_sub = momentum_subgrad(best_h, n_iters=4000, t0=t0, budget=budget_s * 0.10)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub

    # Phase 3: High-T SLSQP
    if time.time() - t0 < budget_s * 0.35:
        for T in [100000, 500000, 2000000]:
            if time.time() - t0 > budget_s * 0.35:
                break
            h_try = slsqp_opt(best_h, T, maxiter=1200)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 4: Targeted refinement
    if time.time() - t0 < budget_s * 0.45:
        h_try, c5_try = targeted_refine(best_h, best_n, top_k=4, iters=1500, t0=t0, budget=budget_s * 0.10)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 5: Multi-resolution search
    n_candidates = set()
    for dn in range(-6, 10, 2):
        n_try = best_n + dn
        if 60 <= n_try <= 120 and n_try % 2 == 0:
            n_candidates.add(n_try)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if time.time() - t0 > budget_s * 0.65:
            break
        h_base = interp_h(best_h, best_n, n_try) if n_try != best_n else best_h.copy()
        for trial in range(3):
            if time.time() - t0 > budget_s * 0.65:
                break
            if trial == 0:
                h_pert = h_base.copy()
            else:
                scale = 0.03 + 0.04 * np.random.rand()
                h_pert = project_feasible(h_base + np.random.randn(n_try) * scale, n_try)
            h_opt, c5_opt = momentum_subgrad(h_pert, n_iters=1500, t0=t0, budget=budget_s * 0.04)
            h_opt = slsqp_opt(h_opt, 200000, maxiter=600)
            c5_opt, _ = compute_c5(h_opt)
            h_opt, c5_opt2 = targeted_refine(h_opt, n_try, top_k=4, iters=500, t0=t0, budget=budget_s * 0.02)
            c5_opt = min(c5_opt, c5_opt2)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 6: Rotation search
    if time.time() - t0 < budget_s * 0.78:
        for shift in [1, 2, 3, 4, 5, 7, 10, 13]:
            if time.time() - t0 > budget_s * 0.78:
                break
            h_rot = np.roll(best_h, shift)
            h_rot = project_feasible(h_rot, best_n)
            h_opt, c5_opt = momentum_subgrad(h_rot, n_iters=2000, t0=t0, budget=budget_s * 0.02)
            h_opt = slsqp_opt(h_opt, 300000, maxiter=500)
            c5_opt, _ = compute_c5(h_opt)
            h_opt, c5_opt2 = targeted_refine(h_opt, best_n, top_k=4, iters=400, t0=t0, budget=budget_s * 0.015)
            c5_opt = min(c5_opt, c5_opt2)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 7: Fine polish - multiple small perturbations + SLSQP
    if time.time() - t0 < budget_s * 0.90:
        for scale in [0.02, 0.01, 0.005, 0.002]:
            if time.time() - t0 > budget_s * 0.90:
                break
            for trial in range(4):
                if time.time() - t0 > budget_s * 0.90:
                    break
                h_p = project_feasible(best_h + np.random.randn(best_n) * scale, best_n)
                h_opt = slsqp_opt(h_p, 500000, maxiter=800)
                h_opt, c5_opt = targeted_refine(h_opt, best_n, top_k=5, iters=600, t0=t0, budget=budget_s * 0.02)
                c5_chk, _ = compute_c5(h_opt)
                c5_opt = min(c5_opt, c5_chk)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    # Phase 8: Final ultra-high T + targeted
    if time.time() - t0 < budget_s * 0.97:
        h_f = slsqp_opt(best_h, 1000000, maxiter=800)
        h_f, c5_f = targeted_refine(h_f, best_n, top_k=6, iters=2000, t0=t0, budget=budget_s * 0.03)
        c5_f, _ = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
