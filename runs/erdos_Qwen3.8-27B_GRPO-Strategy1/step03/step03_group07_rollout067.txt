import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    tgt = n / 2.0
    d = tgt - h.sum()
    if abs(d) < 1e-12:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t > 1e-12:
            h += r * (d / t)
    else:
        t = h.sum()
        if t > 1e-12:
            h += h * (d / t)
    return np.clip(h, 0.0, 1.0)

def soft_obj(h, T):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    s = np.clip(T * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / T) * dx

def sum_eq(h):
    return h.sum() - len(h) / 2.0

def slsqp(h0, T, maxit, ftol):
    n = len(h0)
    r = minimize(soft_obj, h0, args=(T,), method='SLSQP',
                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': sum_eq}],
                 options={'maxiter': maxit, 'ftol': ftol})
    return make_feasible(np.clip(r.x, 0, 1), n)

def lag_grad(h, kidx):

        if 0 <= j - s < n:
            v += 1.0 - h[j - s]
        if 0 <= j + s < n:
            v -= h[j + s]
        g[j] = v

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    tgt = n / 2.0
    d = tgt - h.sum()
    if abs(d) < 1e-12:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t > 1e-12:
            h += r * (d / t)
    else:
        t = h.sum()
        if t > 1e-12:
            h += h * (d / t)
    return np.clip(h, 0.0, 1.0)


def soft_obj(h, T):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    s = np.clip(T * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / T) * dx


def sum_eq(h):
    return h.sum() - len(h) / 2.0


def slsqp(h0, T, maxit, ftol):
    n = len(h0)
    r = minimize(soft_obj, h0.copy(), args=(T,), method='SLSQP',
                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': sum_eq}],
                 options={'maxiter': maxit, 'ftol': ftol})
    return make_feasible(np.clip(r.x, 0.0, 1.0), n)


def lag_grad(h, kidx):
    n = len(h)
    s = kidx - n + 1

            v += 1.0 - h[j - s]
        if 0 <= j + s and j + s < n:
            v -= h[j + s]
        g[j] = v
    return g

    mask2 = (jp >= 0) & (jp < n)
    g[mask2] -= h[jp[mask2]]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    tgt = n / 2.0
    d = tgt - h.sum()
    if abs(d) < 1e-12:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t > 1e-12:
            h += r * (d / t)
    else:
        t = h.sum()
        if t > 1e-12:
            h += h * (d / t)
    return np.clip(h, 0.0, 1.0)


def soft_obj(h, T):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    s = np.clip(T * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / T) * dx


def sum_eq(h):
    return h.sum() - len(h) / 2.0


def slsqp(h0, T, maxit, ftol):
    n = len(h0)
    r = minimize(soft_obj, h0.copy(), args=(T,), method='SLSQP',
                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': sum_eq}],
                 options={'maxiter': maxit, 'ftol': ftol})
    return make_feasible(np.clip(r.x, 0.0, 1.0), n)
