def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h to satisfy sum(h)=target and 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-10:
        return h
    # Bisection for the water-filling threshold
    lo, hi = -2.0, 2.0

        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_gradient(h, N, tau, top_k):
    """Compute gradient of log-sum-exp smoothed C5."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    nc = len(corr)
    # Softmax weights
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    # Top-k blending
    if top_k < nc:
        idx_top = np.argsort(corr)[-top_k:]
        w_top = np.zeros(nc)
        w_top[idx_top] = 1.0 / top_k
        alpha = min(1.0, max(0.0, (tau - 5.0) / 30.0))
        w = (1.0 - alpha) * w + alpha * w_top
    grad = np.zeros(N)
    center = N - 1
    thresh = 0.001
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai] * dx
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

    while time.time() < deadline:

        frac = min(n_it / 1500.0, 1.0)
        tau = tau_start + (tau_end - tau_start) * frac

        c5 = compute_c5(h, 2.0 / N)
        if c5 < best_c5:

            best_h = h.copy()
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = float(h.sum())
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lse_gradient(h, N, tau, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    nc = len(corr)
    sc = tau * corr
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    if top_k < nc and top_k >= 1:
        idx_top = np.argsort(corr)[-top_k:]
        w_top = np.zeros(nc)
        w_top[idx_top] = 1.0 / top_k
        alpha = min(1.0, max(0.0, (tau - 8.0) / 25.0))
        w = (1.0 - alpha) * w + alpha * w_top
    grad = np.zeros(N)
    center = N - 1
    thresh = 0.002
    active = np.where(w > thresh)[0]
    for ai in active:
        L = ai - center
        wi = w[ai] * dx
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * (1.0 - h[:N - L])
            grad[:N - L] -= wi * h[L:]
        else:
            M = -L
            grad[:N - M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N - M]
    return grad

def grad_opt(h, N, target, deadline, tau_s, tau_e, tk_s, tk_e, lr_s, lr_e, mom):
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / N)
    v = np.zeros(N)
    it = 0
    stall = 0
    while time.time() < deadline:
        it += 1
        frac = min(it / 1200.0, 1.0)
        tau = tau_s + (tau_e - tau_s) * frac
        tk = max(1, int(tk_s + (tk_e - tk_s) * frac))
        lr = lr_s * (max(lr_e, 1e-8) / lr_s) ** frac
        grad = lse_gradient(h, N, tau, tk)
        v = mom * v + (1.0 - mom) * grad
        h = project_sum(h - lr * v, target)
        c5 = compute_c5(h, 2.0 / N)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 200 and it > 100:
            break
    return best_h, best_c5

def binary_snap(h, N):
    need = N // 2
    thresh = np.sort(h)[::-1][need - 1]
    h_bin = (h >= thresh).astype(float)
    if h_bin.sum() > need:
        excess = h_bin.sum() - need
        h_bin[np.where(h_bin == 1)[0][:excess]] = 0
    elif h_bin.sum() < need:
        deficit = need - h_bin.sum()
        h_bin[np.where(h_bin == 0)[0][:deficit]] = 1
    return h_bin

def binary_search(h_bin, N, deadline, max_pairs=60):
    dx = 2.0 / N
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin, dx)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        oi_list = ones[:max_pairs]
        zi_list = zeros[:max_pairs]
        for oi in oi_list:
            if time.time() > deadline:
                break
            for zi in zi_list:
                h_bin[oi] = 0.0
                h_bin[zi] = 1.0
                c5 = compute_c5(h_bin, dx)
                h_bin[oi] = 1.0
                h_bin[zi] = 0.0
                if c5 < best_c5 - 1e-12:
                    h_bin[oi] = 0.0
                    h_bin[zi] = 1.0
                    best_c5 = c5
                    best_h = h_bin.copy()
                    improved = True
                    break
    return best_h, best_c5

def coord_polish(h, N, target, deadline):
    dx = 2.0 / N
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            orig = h[i]
            for trial in (0.0, 1.0, max(0.0, orig - 0.03), min(1.0, orig + 0.03)):
                if abs(trial - orig) < 1e-9:
                    continue
                h[i] = trial
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-11:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break
                h[i] = orig
                h = project_sum(h, target)
    return best_h, best_c5

    p1_frac = 0.48
    p1_end = t0 + (deadline - t0) * p1_frac
    for si, N in enumerate(sizes):
        if time.time() > p1_end - 40:
            break
        seg = p1_end * (si + 1) / len(sizes) - p1_end * si / len(sizes)
        seg_dead = min(t0 + (deadline - t0) * p1_frac * (si + 1) / len(sizes), deadline - 80)
        if seg_dead - time.time() < 25:
            break
        target = N / 2.0
        xN = np.linspace(0.0, 2.0, N, endpoint=False)
        h_base = np.interp(xN, x_par, h_par) if N != n_par else h_par.copy()
        h_base = project_sum(h_base, target)
        n_att = 3 if seg_dead - time.time() > 80 else 2
        for at in range(n_att):
            att_dead = seg_dead - (n_att - at) * 8
            if time.time() > att_dead - 20:
                break
            if at == 0:
                h_cur = h_base.copy()
            elif at == 1:
                h_cur = project_sum(h_base + rng.randn(N) * 0.06, target)
            else:
                h_cur = project_sum(h_base + rng.randn(N) * 0.12, target)
            rem = att_dead - time.time()
            if rem < 25:
                break
            h1, c1 = grad_opt(h_cur, N, target, time.time() + rem * 0.45,
                              5.0, 60.0, 30, 6, 0.035, 0.012, 0.85)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()
            if time.time() < att_dead - 10:
                h2, c2 = grad_opt(h1, N, target, att_dead - 10,
                                  40.0, 400.0, 8, 2, 0.012, 0.003, 0.88)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2.copy()

    Nb = len(best_h)
    target_b = Nb / 2.0
    if time.time() < deadline - 70:
        h_bin = binary_snap(best_h, Nb)
        bs_dead = min(time.time() + 90, deadline - 40)
        h_b2, c_b2 = binary_search(h_bin, Nb, bs_dead, max_pairs=40)
        if c_b2 < best_c5:
            best_c5 = c_b2
            best_h = h_b2.copy()
            Nb = len(best_h)
            target_b = Nb / 2.0

    if time.time() < deadline - 35:
        Nb = len(best_h)
        target_b = Nb / 2.0
        h_f, c_f = grad_opt(best_h.copy(), Nb, target_b,

                            60.0, 600.0, 5, 2, 0.006, 0.001, 0.90)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()

    if time.time() < deadline - 8:
        Nb = len(best_h)
        target_b = Nb / 2.0
        best_h, best_c5 = coord_polish(best_h, Nb, target_b, deadline - 3)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
