def grad_h_from_weights(w, h):
    """Compute gradient of smoothed max correlation w.r.t. h.
    w: array of length 2n-1 (full correlation gradient weights)
    h: array of length n
    """
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for k in range(2*n-1):
            s = k - (n-1)  # shift
            if 0 <= j - s < n:
                g += w[k] * (1 - h[j-s])
            if 0 <= j + s < n:
                g -= w[k] * h[j+s]
        grad[j] = g
    return grad

    sd = h.sum() - n / 2.0
    val += pen_w * sd * sd
    grad_h = grad_h + 2.0 * pen_w * sd
    return val, grad_h
