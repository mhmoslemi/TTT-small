import numpy as np

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    h_b = (h >= 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > N // 2:
        idx_ones = np.where(h_b == 1)[0]
        order = np.argsort(h[idx_ones])  # lowest h first
        for k in range(n_ones - N // 2):
            h_b[idx_ones[order[k]]] = 0.0
    elif n_ones < N // 2:
        idx_zeros = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx_zeros])  # highest h first
        for k in range(N // 2 - n_ones):
            h_b[idx_zeros[order[k]]] = 1.0
    return h_b

def softmax_grad(h, N, T):
    corr = compute_corr(h)
    cm = corr - corr.max()
    wc = np.exp(cm / T)
    wc /= wc.sum()
    grad = np.zeros(N)
    center = N - 1
    for L in range(1, N):
        wp = wc[center + L]
        wn = wc[center - L]
        nrem = N - L
        grad[L:] += wp * (1.0 - h[:nrem])
        grad[:nrem] -= wp * h[L:]
        grad[:nrem] += wn * (1.0 - h[L:])
        grad[L:] -= wn * h[:nrem]
    grad /= T
    return grad
