                   options={'maxiter':maxiter, 'ftol':ftol})

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0, 1).astype(float)
    tgt = n / 2.0
    d = tgt - h.sum()
    if abs(d) < 1e-13:
        return h
    if d > 0:
        r = 1.0 - h; t = r.sum()
        if t > 1e-13: h += r * d / t
    else:
        r = h.copy(); t = r.sum()
        if t > 1e-13: h += r * d / t
    return np.clip(h, 0, 1)

def solve_lp(hs, active, n):
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [hv >= 0, hv <= 1, cp.sum(hv) == n / 2.0, tv >= 0]
    for ki in active:
        s = ki - (n - 1)
        i_lo = max(0, -s); i_hi = min(n - 1, n - 1 - s)

        coeff[idx + s] -= hs[idx]
        coeff[idx] -= hs[idx + s]
        const = float(np.sum(hs[idx] * hs[idx + s]))
