def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h = np.clip(h + room * (diff / total), 0, 1)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h = np.clip(h + room * (diff / total), 0, 1)

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h = np.clip(h + room * (diff / total), 0, 1)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h = np.clip(h + room * (diff / total), 0, 1)
    return h

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def topk_avg_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top = np.sort(corr)[::-1][:k]
    return float(np.sum(top) * dx / k)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    scaled = T * (corr - m)
    scaled[scaled < -500] = -500
    val = m + np.log(np.sum(np.exp(scaled))) / T
    return val * dx

def slsqp_wrap(h, obj_func, args, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h, n)

def flip_search(h, n, rounds, top_k):
    best = h.copy()
    best_c5 = compute_c5(h)
    stagnant = 0
    for _ in range(rounds):
        corr = np.correlate(best, 1.0 - best, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_d, best_pair = 0.0, None
        for ki in idx:
            s = int(ki - (n - 1))
            for i in ones:
                di = 0.0
                if 0 <= i + s < n:
                    di -= (1.0 - best[i + s])
                if 0 <= i - s < n:
                    di += best[i - s]
                for j in zeros:
                    dj = 0.0
                    if 0 <= j + s < n:
                        dj += (1.0 - best[j + s])
                    if 0 <= j - s < n:
                        dj -= best[j - s]
                    d = di + dj
                    if d < best_d:
                        best_d, best_pair = d, (i, j)
        if best_pair is not None and best_d < -0.3:
            i, j = best_pair
            best[i], best[j] = 0.0, 1.0
            c5n = compute_c5(best)
            if c5n < best_c5:
                best_c5 = c5n
                stagnant = 0
            else:
                best[i], best[j] = 1.0, 0.0
                stagnant += 1
        else:
            stagnant += 1
        if stagnant > 15:
            break
    return best, best_c5

def mass_transfer(h, n, rounds, step):
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(rounds):
        corr = np.correlate(best, 1.0 - best, mode='full')
        s = int(np.argmax(corr) - (n - 1))
        changed = False
        for i in range(n):
            j = i + s
            if 0 <= j < n and best[i] > step and best[j] < 1.0 - step:
                d = min(step, best[i], 1.0 - best[j])
                old = best[i] * (1.0 - best[j])
                new = (best[i] - d) * (1.0 - best[j] - d)
                if new < old - 1e-10:
                    best[i] -= d
                    best[j] += d
                    changed = True
        if changed:
            best = project_feasible(best, n)
            c5n = compute_c5(best)
            if c5n < best_c5:
                best_c5 = c5n
    return best, best_c5

def fft_perturb(h, n, rng, n_modes, amp):
    t = np.linspace(0, 2, n, endpoint=False)
    p = np.zeros(n)
    for _ in range(n_modes):
        f = rng.randint(1, 8)
        ph = rng.rand() * 2.0 * np.pi
        p += rng.randn() * amp * np.cos(np.pi * f * t + ph)
        p += rng.randn() * amp * np.sin(np.pi * f * t + ph)
    return project_feasible(h + p, n)
