```python
def smooth_max_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_c = np.exp(T * (corr - m))
    p = exp_c / exp_c.sum()
    F = (m + np.log(exp_c.sum()) / T) * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            # ∂c[k]/∂h[j] = (1-h[j-s]) for j in [s, n-1] minus h[j+s] for j in [0, n-1-s]
            # Part 1: j in [s, n-1], contribution (1-h[j-s])
            if s < n:
                grad[s:] += pk * (1.0 - h[:n-s])
            # Part 2: j in [0, n-1-s], contribution -h[j+s]
            if n-1-s >= 0:
                grad[:n-s] -= pk * h[s:]
        else:
            t = -s
            # ∂c[k]/∂h[j] = (1-h[j+t]) for j in [0, n-1-t] minus h[j-t] for j in [t, n-1]
            if n-1-t >= 0:
                grad[:n-t] += pk * (1.0 - h[t:])
            if t < n:
                grad[t:] -= pk * h[:n-t]
    
    grad *= dx
    return F, grad
```

```python
def smooth_max_grad(h, T, penalty=1000.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_c = np.exp(T * (corr - m))
    s = exp_c.sum()
    p = exp_c / s
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        # For each j: contribution is (1-h[j+k-n+1]) [if valid] - h[j-k+n-1] [if valid]
        # idx1[j] = j + k - n + 1, valid if 0 <= idx1 < n
        # idx2[j] = j - k + n - 1, valid if 0 <= idx2 < n
        
        # Vectorize: j ranges 0..n-1
        j = np.arange(n)
        idx1 = j + k - n + 1
        idx2 = j - k + n - 1
        
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        
        contrib = np.zeros(n)
        if np.any(valid1):
            contrib[valid1] += (1.0 - h[idx1[valid1]])
        if np.any(valid2):
            contrib[valid2] -= h[idx2[valid2]]
        
        grad += pk * contrib
    
    grad *= dx
    # Add penalty gradient for sum(h) = n/2
    grad += penalty * (h.sum() - n / 2.0)
    return grad
```

```python
            if t < 1e-12: break
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, 'full')) * (2.0 / n)

def sm_val(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    m = c.max()
    return (m + np.log(np.exp(T * (c - m)).sum()) / T) * dx

def sm_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    m = c.max()
    p = np.exp(T * (c - m))
    p /= p.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        i1 = j + k - n + 1
        i2 = j - k + n - 1
        v1 = (i1 >= 0) & (i1 < n)
        v2 = (i2 >= 0) & (i2 < n)
        if v1.any():
            g[v1] += p[k] * (1.0 - h[i1[v1]])
        if v2.any():
            g[v2] -= p[k] * h[i2[v2]]
    return g * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(30):
        d = tgt - h.sum()
        if abs(d) < 1e-11:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-12:
                break
            h += r * (d / t)
        else:
            t = h.sum()
            if t < 1e-12:
                break
            h += h * (d / t)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def lbfgs_opt(h0, T_list, n, pen=500.0, maxit=800):
    best = h0.copy()
    best_c = compute_c5(best)
    bnds = [(0.0, 1.0)] * n
    for T in T_list:
        def obj(x):
            return sm_val(x, T) + 0.5 * pen * (x.sum() - n / 2.0) ** 2
        def grad(x):
            return sm_grad(x, T) + pen * (x.sum() - n / 2.0)
        try:
            r = minimize(obj, best, jac=grad, method='L-BFGS-B',
                         bounds=bnds, options={'maxiter': maxit, 'ftol': 1e-17, 'gtol': 1e-13})
            h = proj(np.clip(r.x, 0, 1), n)
            c = compute_c5(h)
            if c < best_c:
                best, best_c = h, c
        except Exception:
            pass
    return best, best_c

def interp_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return proj(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)

        best_h = np.zeros(60)

    n_init = len(best_h)

    best_c5 = compute_c5(best_h)

    T_sched = [3, 10, 50, 200, 1000, 5000, 20000]

                def grad_top(x):

                    e = np.exp(T * (c - m))
```
