from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    target_sum = target
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.clip(h - mid, 0.0, 1.0).sum()) >= target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def harmonic_to_grid(coeffs, x, K):
    h = np.full(len(x), 0.5)
    for k in range(1, K + 1):
        h += coeffs[2*(k-1)] * np.cos(np.pi * k * x)
        h += coeffs[2*(k-1)+1] * np.sin(np.pi * k * x)
    return h

def grid_to_harmonic(h, x, K):
    N = len(x)
    coeffs = np.zeros(2*K)
    for k in range(1, K + 1):
        coeffs[2*(k-1)] = 2.0 / N * np.dot(h - 0.5, np.cos(np.pi * k * x))
        coeffs[2*(k-1)+1] = 2.0 / N * np.dot(h - 0.5, np.sin(np.pi * k * x))
    return coeffs

def smooth_max_objective(coeffs, x, N, tau, lam):
    K = len(coeffs) // 2
    h = harmonic_to_grid(coeffs, x, K)
    h = np.clip(h, 0.0, 1.0)

    target = N / 2.0
    h = project_sum(h, target)
    corr = np.correlate(h, 1.0 - h, mode='full')

    cm = corr.max()
    smax = np.log(np.exp(tau * (corr - cm)).sum() / len(corr)) / tau + cm
    obj = smax * dx
    # Penalty for bounds violation (shouldn't happen after clipping)
    return obj + lam * (np.max(np.abs(h - 0.5)) ** 2)

            c[i] -= eps

            if time.time() < tp2:
                h_cur, c2 = optimize_direct(h_cur, N, target, tp2,
                    50.0, 250.0, 0.012, 0.0008, 10, 3, 0.90)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_cur.copy()
            if time.time() < tp3:
                h_cur, c3 = optimize_direct(h_cur, N, target, tp3,
                    250.0, 800.0, 0.003, 0.0002, 3, 1, 0.93)
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = h_cur.copy()

    # Phase C: Final refinement
    Nb = len(best_h)
    if time.time() < deadline - 50:
        h_f, cf = optimize_direct(best_h.copy(), Nb, Nb / 2.0, time.time() + 30,
            80.0, 500.0, 0.004, 0.0002, 5, 1, 0.93)
        if cf < best_c5:
            best_c5 = cf
            best_h = h_f

    # Phase D: Binary coordinate polish
    if time.time() < deadline - 3:
        best_h, best_c5 = polish_binary(best_h, len(best_h), len(best_h) / 2.0, deadline - 1)

    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
