import numpy as np
from scipy.optimize import minimize, minimize_scalar
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def compute_grad_matrix(h, n):
    G = np.zeros((n, 2 * n - 1))
    omh = 1.0 - h
    for lag in range(-(n - 1), n):
        idx = lag + n - 1
        if lag >= 0:
            G[:n - lag, idx] += omh[lag:]
            G[lag:, idx] -= h[:n - lag]
        else:
            sl = -lag
            G[sl:, idx] += omh[:n + lag]
            G[:n + lag, idx] -= h[sl:]
    return G

def coord_descent_fast(h, n, t_end, max_sweeps=200, tol=1e-14):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    damp = 1.0
    for sweep in range(max_sweeps):
        if time() > t_end:
            break
        G = compute_grad_matrix(best_h, n)
        corr = compute_corr(best_h)
        improved = 0.0
        for i in range(n):
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if hi - lo < 1e-14:
                continue
            b = G[i]
            a = corr
            def f1d(d):
                return np.max(a + d * b)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-12})
            new_val = best_h[i] + res.x
            new_val = np.clip(new_val, 0.0, 1.0)
            if abs(new_val - best_h[i]) > 1e-14:
                best_h[i] = new_val
                improved += abs(new_val - best_h[i])
        best_h = project(best_h, n)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
        if improved < tol:
            break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_val(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_obj(x, T):
    return smooth_c5_val(x, T)

def slsqp_con(x):
    return x.sum() - 0.5 * len(x)

def slsqp_opt(h, T, maxiter=500, ftol=1e-14):
    n = len(h)
    cons = [{'type': 'eq', 'fun': slsqp_con}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(slsqp_obj, h, args=(T,), method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def f1d_eval(d, a, b):
    return np.max(a + d * b)

def compute_grad_matrix(h, n):
    G = np.zeros((n, 2 * n - 1))
    omh = 1.0 - h
    for lag in range(-(n - 1), n):
        idx = lag + n - 1
        if lag >= 0:
            G[lag:, idx] += omh[:n - lag]
            G[:n - lag, idx] -= h[lag:]
        else:
            sl = -lag
            G[:n - sl, idx] += omh[sl:]
            G[sl:, idx] -= h[:n - sl]
    return G

def coord_descent_fast(h, n, t_end, max_sweeps=200, tol=1e-13):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time() > t_end:
            break
        G = compute_grad_matrix(best_h, n)
        corr = compute_corr(best_h)
        improved = 0.0
        for i in range(n):
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if hi - lo < 1e-14:
                continue
            res = minimize_scalar(f1d_eval, bounds=(lo, hi), args=(corr, G[i]),
                                  method='bounded', options={'xatol': 1e-13})
            new_val = np.clip(best_h[i] + res.x, 0.0, 1.0)
            d = abs(new_val - best_h[i])
            if d > 1e-14:
                best_h[i] = new_val
                improved += d
        best_h = project(best_h, n)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new
        if improved < tol:
            break
    return best_h, best_c5

def block_perturb(h, n, step):
    lp = np.random.randint(0, n)
    w = np.random.randint(3, max(4, n // 4))
    h2 = h.copy()
    h2[lp:] = h2[lp:]
    for i in range(n):
        dist = abs(i - lp)
        if dist < w:
            h2[i] += step * np.random.randn()
    return project(h2, n)

    # Phase 1: Coordinate descent from start

    # Phase 2: SLSQP temperature sweep

            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    t2 = t0 + budget_s * 0.26
    best_h, best_c5 = coord_descent_fast(best_h, n, t2, max_sweeps=150)

    # Phase 3: Basin hopping with coord descent
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iters = 400
    for it in range(n_iters):
        if elapsed() > budget_s * 0.55:

        cool = max(0.0, 1.0 - it / float(n_iters))
