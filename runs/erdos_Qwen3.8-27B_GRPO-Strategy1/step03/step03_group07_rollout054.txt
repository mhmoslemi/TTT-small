import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-14:
            h += room * (diff / total)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 0:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    else:
        val = m
    return val * dx

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def slsqp_step(h0, T, n, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def optimize_single_coord(h, j, n):
    dx = 2.0 / n
    old_hj = h[j]
    h[j] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[j] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full')
    h[j] = old_hj
    a = (corr1 - corr0) * dx
    b = corr0 * dx
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(60):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_sweeps(h, n, n_sweeps):
    best_c5 = compute_c5(h)
    for sw in range(n_sweeps):
        idx = np.random.permutation(n)
        for j in idx:
            x_opt = optimize_single_coord(h, j, n)
            h[j] = x_opt
        h = make_feasible(h, n)
        c5v = compute_c5(h)
        if c5v < best_c5:
            best_c5 = c5v
        else:
            break
    return h, best_c5

def subgradient_descent(h, n, n_iter, t0, deadline):
    v = np.zeros(n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        alpha = 0.6 / (1.0 + it * 0.002)
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_m = max(3, 20 - it // 200)
        top_idx = np.argsort(corr)[-top_m:]
        grad = np.zeros(n)
        nhalf = n - 1
        for k_idx in top_idx:
            k = int(k_idx) - nhalf
            if k >= 0:
                grad[k:] += (1.0 - h[:n - k])
                grad[:n - k] -= h[k:]
            else:
                kn = -k
                grad[:n - kn] += (1.0 - h[kn:])
                grad[kn:] -= h[:n - kn]
        dx = 2.0 / n
        grad *= dx / top_m
        v = 0.8 * v + 0.2 * (-grad)
        h_new = h + alpha * v
        h = make_feasible(h_new, n)
    return h
