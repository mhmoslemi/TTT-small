                       constraints=[{'type':'eq','fun':sum_con}],
                       options={'maxiter':maxiter,'ftol':ftol})
        return project_feasible(np.clip(res.x,0,1), n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            m = room > 1e-14
            if not m.any():
                break
            h[m] += diff * room[m] / room[m].sum()
        else:
            m = h > 1e-14
            if not m.any():
                break
            h[m] += diff * h[m] / h[m].sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_obj(h, k, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    srt = np.sort(corr)[::-1][:k]
    return float(np.sum(srt) * dx / k)

def softmax_obj(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def sum_con(x):
    return float(x.sum() - len(x) / 2.0)

def slsqp_opt(h, obj, args, maxiter=300, ftol=1e-13):
    n = len(h)
    try:
        res = minimize(obj, h, args=args, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

    t = 2.0 * (np.arange(n) + 0.5) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2 * (k - 1)] = np.cos(np.pi * k * t)
        B[:, 2 * (k - 1) + 1] = np.sin(np.pi * k * t)
    return B

def fourier_to_h(coeffs, B, n):
    h = np.full(n, 0.5) + B @ coeffs
    return project_feasible(h, n)

def fourier_topk(coeffs, B, n, k):
    h = fourier_to_h(coeffs, B, n)
    return topk_obj(h, k, n)

def dft_coeffs(h, K):
    n = len(h)
    hf = np.fft.rfft(h - 0.5)
    c = np.zeros(2 * K)
    for k in range(1, K + 1):
        if k < len(hf):
            c[2 * (k - 1)] = 2.0 * hf[k].real / n
            c[2 * (k - 1) + 1] = -2.0 * hf[k].imag / n
    return c

def neg_corr_grad(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        sp = -s
        g[sp:] += 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g

def nesterov_h(h, n, steps, top_k, lr0, mu, decay):
    h = h.astype(float).copy()
    v = np.zeros(n)
    lr = lr0
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(steps):

        idx = np.argpartition(corr, -top_k)[-top_k:]
        w = np.exp(30.0 * (corr[idx] - corr[idx].max()))
        w /= w.sum()

        c5 = compute_c5(h_new)
        if c5 < best_c5:

        h = h_new

    if time.time() - t0 < budget_s * 0.32:
        for top_k in [3, 5, 8]:
            if time.time() - t0 > budget_s * 0.32:
                break
            for lr0 in [0.008, 0.015, 0.025]:
                if time.time() - t0 > budget_s * 0.32:
                    break
                h_n, c_n = nesterov_h(best_h, best_n, 350, top_k, lr0, 0.82, 0.9992)
                if c_n < best_c5:
                    best_c5, best_h = c_n, h_n
                h_p = slsqp_opt(h_n, topk_obj, (3, best_n), 250, 1e-14)
                c_p = compute_c5(h_p)
                if c_p < best_c5:
                    best_c5, best_h = c_p, h_p

    # Phase 3: Direct SLSQP with multiple objectives
    if time.time() - t0 < budget_s * 0.42:
        for T in [300, 1500, 6000]:
            if time.time() - t0 > budget_s * 0.42:
                break
            h_f = slsqp_opt(best_h, softmax_obj, (T, best_n), 400, 1e-14)
            c_f = compute_c5(h_f)
            if c_f < best_c5:
                best_c5, best_h = c_f, h_f
        for k in [2, 3, 5]:
            if time.time() - t0 > budget_s * 0.42:
                break
            h_f = slsqp_opt(best_h, topk_obj, (k, best_n), 400, 1e-14)
            c_f = compute_c5(h_f)
            if c_f < best_c5:
                best_c5, best_h = c_f, h_f

    # Phase 4: Multi-resolution exploration
    if time.time() - t0 < budget_s * 0.58:
        for nr in [50, 75, 100, 150, 250, 300, 400]:
            if time.time() - t0 > budget_s * 0.58:
                break
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, nr)
            h_r = np.interp(x_new, x_old, best_h)
            h_r = project_feasible(h_r, nr)
            h_r = slsqp_opt(h_r, topk_obj, (3, nr), 350, 1e-13)
            c_r = compute_c5(h_r)
            K_r = min(20, nr // 3)
            B_r = get_fourier_basis(K_r, nr)
            c0_r = dft_coeffs(h_r, K_r)
            try:
                res_r = minimize(fourier_topk, c0_r, args=(B_r, nr, 3),
                                 method='Nelder-Mead',
                                 options={'maxiter': 400, 'xatol': 1e-9, 'fatol': 1e-13})
                h_fr = fourier_to_h(res_r.x, B_r, nr)
                c_fr = compute_c5(h_fr)
            except Exception:
                c_fr, h_fr = 1e9, h_r
            if c_r < c_fr:
                h_cand, c_cand = h_r, c_r
            else:
                h_cand, c_cand = h_fr, c_fr
            if c_cand < best_c5:
                best_c5, best_h, best_n = c_cand, h_cand, nr
                # Downsample check
                if best_n > 200:
                    x_dn = np.linspace(0, 2, 200)
                    h_dn = np.interp(x_dn, np.linspace(0, 2, best_n), best_h)
                    h_dn = project_feasible(h_dn, 200)
                    c_dn = compute_c5(h_dn)
                    if c_dn < best_c5:
                        best_c5, best_h, best_n = c_dn, h_dn, 200

    # Phase 5: Basin hopping with diverse perturbation + Nesterov refine
    t_phase5_start = time.time()
    max_phase5 = min(60, int(budget_s * 0.15 / 3.5))
    for trial in range(max_phase5):
        if time.time() - t0 > budget_s * 0.75:
            break
        r = np.random.random()
        if r < 0.25:
            sc = 0.02 + 0.03 * np.random.random()
            h_p = project_feasible(best_h + np.random.randn(best_n) * sc, best_n)
        elif r < 0.40:
            h_p = best_h.copy()
            st = np.random.randint(0, max(1, best_n - 5))
            ln = np.random.randint(5, max(6, best_n // 4))
            en = min(st + ln, best_n)
            sh = np.random.randint(1, min(4, ln))
            h_p[st:en] = np.roll(h_p[st:en], sh)
            h_p = project_feasible(h_p, best_n)
        elif r < 0.55:
            K_b = min(12, best_n // 4)
            B_b = get_fourier_basis(K_b, best_n)
            c_b = dft_coeffs(best_h, K_b)
            c_b += np.random.randn(2 * K_b) * 0.06
            h_p = fourier_to_h(c_b, B_b, best_n)
        elif r < 0.70:
            h_p = targeted_repair(best_h, best_n, 0.3)
        else:
            h_p = best_h.copy()
            corr = np.correlate(h_p, 1.0 - h_p, mode='full')
            top3 = np.argpartition(corr, -3)[-3:]
            for ki in top3:
                s = int(ki - (best_n - 1))
                for i in range(best_n):
                    j = i + s
                    if 0 <= j < best_n:
                        if h_p[i] > 0.5 and h_p[j] < 0.5:
                            d = min(h_p[i] * 0.15, 1.0 - h_p[j])
                            h_p[i] -= d
                            h_p[j] += d
                        elif h_p[i] < 0.5 and h_p[j] > 0.5:
                            d = min((1.0 - h_p[i]) * 0.15, h_p[j])
                            h_p[i] += d
                            h_p[j] -= d
            h_p = project_feasible(h_p, best_n)
        h_r = nesterov_h(h_p, best_n, 80, 3, 0.01, 0.85, 0.998)
        if isinstance(h_r, tuple):
            h_r, _ = h_r
        h_s = slsqp_opt(h_r, topk_obj, (3, best_n), 200, 1e-14)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s

    # Phase 6: Final intensive polish
    if time.time() - t0 < budget_s * 0.93:
        for T in [500, 2000, 8000, 20000]:
            if time.time() - t0 > budget_s * 0.93:
                break
            h_f = slsqp_opt(best_h, softmax_obj, (T, best_n), 500, 1e-15)
            c_f = compute_c5(h_f)
            if c_f < best_c5:
                best_c5, best_h = c_f, h_f
        for k in [1, 2, 3]:
            if time.time() - t0 > budget_s * 0.93:
                break
            h_f = slsqp_opt(best_h, topk_obj, (k, best_n), 500, 1e-15)
            c_f = compute_c5(h_f)
            if c_f < best_c5:
                best_c5, best_h = c_f, h_f
        h_n, c_n = nesterov_h(best_h, best_n, 120, 3, 0.004, 0.92, 0.9997)
        if c_n < best_c5:
            best_c5, best_h = c_n, h_n
        h_f = slsqp_opt(best_h, topk_obj, (1, best_n), 400, 1e-16)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f

    # Phase 7: Second Nelder-Mead on Fourier from final solution
    if time.time() - t0 < budget_s * 0.97:
        K_f = min(15, best_n // 3)
        B_f = get_fourier_basis(K_f, best_n)
        c0_f = dft_coeffs(best_h, K_f)
        try:
            res_f = minimize(fourier_topk, c0_f, args=(B_f, best_n, 3),
                             method='Nelder-Mead',
                             options={'maxiter': 300, 'xatol': 1e-10, 'fatol': 1e-14})
            h_ff = fourier_to_h(res_f.x, B_f, best_n)
            c_ff = compute_c5(h_ff)
            if c_ff < best_c5:
                best_c5, best_h = c_ff, h_ff
                h_ff2 = slsqp_opt(h_ff, topk_obj, (3, best_n), 300, 1e-15)
                c_ff2 = compute_c5(h_ff2)
                if c_ff2 < best_c5:
                    best_c5, best_h = c_ff2, h_ff2
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
