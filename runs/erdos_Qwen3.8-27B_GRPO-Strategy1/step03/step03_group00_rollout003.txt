def coord_descent_fast(h, n, t_end, n_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h).astype(float)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            # Compute coefficient: d(corr[k])/d(h[i])
            coef = np.zeros(2*n-1)
            for k in range(-(n-1), n):
                idx = k + n - 1
                c = 0.0
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c += (-best_h[j2])
                coef[idx] = c
            # corr = (corr - h_old*coef) + t*coef = corr_base + t*coef
            # But wait, this is only valid if the linearization is correct
            # corr_new = corr + (t - h_old) * coef
            # We want to minimize max(corr + (t - h_old) * coef) over t in [0,1]
            # = minimize max(corr + s * coef) over s in [-h_old, 1-h_old]
            
            # Use scipy
            def f1d(s):
                return np.max(corr + s * coef)
            res = minimize_scalar(f1d, bounds=(-h_old, 1.0-h_old), method='bounded', options={'xatol': 1e-10})
            s_new = res.x
            t_new = h_old + s_new
            
            if abs(s_new) > 1e-12:
                best_h[i] = t_new
                corr = corr + s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

This is O(n) per coordinate update (for the coef computation and correlation update), plus the 1D optimization. Much faster than recomputing the full correlation!

But wait, there's a subtle issue. When I compute coef using the CURRENT h values, and then update h[i], the coef for FUTURE coordinates will use the updated h[i]. That's fine - it's just standard coordinate descent.

However, the incremental update `corr = corr + s_new * coef` is only exact if the correlation is truly linear in h[i]. Let me verify:

corr[k] = sum_j h[j] * (1-h[j-k])

If I change only h[i] from old to old+s:
The terms that change:
- j=i: (old+s)*(1-h[i-k]) instead of old*(1-h[i-k]) → change = s*(1-h[i-k])
- j=i+k (if valid): h[i+k]*(1-old-s) instead of h[i+k]*(1-old) → change = -s*h[i+k]

Total change in corr[k] = s*(1-h[i-k] - h[i+k]) = s * coef[k] ✓

So yes, the incremental update is exact. 

Now, for the 1D optimization, I'm minimizing max_k (a_k + s*b_k) over an interval. This is a piecewise linear function in s. The minimum occurs at a point where two "active" constraints are tied, or at a boundary.

A faster approach: the subgradient at the current point tells us the direction. If all max terms have negative slope, we can increase s; etc. But for exactness, let's keep the bounded scalar optimization.

Actually, for a piecewise linear function, the minimum over an interval is at a breakpoint. The breakpoints are where two lines cross: s = (a_p - a_q)/(b_q - b_p) for pairs where b_p ≠ b_q.

For n≈160, there are 319 correlation values, so 319*319/2 ≈ 50K potential breakpoints. We can filter to only those in the interval and find the one that minimizes the max.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def slsqp_polish(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def coord_descent_fast(h, n, t_end, n_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            coef = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                c = 0.0
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[k + n - 1] = c
            def f1d(s):
                return np.max(corr + s * coef)
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-11})
            s_new = res.x
            if abs(s_new) > 1e-12:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def cyclic_shift_perturb(h, amount):
    n = len(h)
    shift = int(amount * n)
    return np.roll(h, shift)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(128)
        best_h[:64] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polynomial T-schedule (10% budget)
    for T in [8, 30, 100, 500, 2000, 8000]:
        if elapsed() > budget_s * 0.10:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=1000, ftol=1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 2: Heavy coordinate descent (20% budget)
    t_cd = t0 + budget_s * 0.30
    best_h, best_c5 = coord_descent_fast(best_h, n, t_cd, n_sweeps=500)

    # Phase 3: Basin hopping with structured perturbations (25% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 60
    for it in range(n_bh):
        if elapsed() > budget_s * 0.55:
            break
        cool = max(1.0 - 0.92 * (it / float(n_bh)), 0.05)
        r = np.random.rand()
        if r < 0.4:
            step = 0.02 + 0.12 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.7:
            h_p = cyclic_shift_perturb(work_h, np.random.uniform(-0.05, 0.05))
            h_p = 0.7 * h_p + 0.3 * work_h
        else:
            step = 0.01 + 0.06 * cool
            mask = np.random.rand(n) < 0.3
            h_p = work_h.copy()
            h_p[mask] += step * np.random.randn(mask.sum())
        h_p = project_to_sum(h_p, n / 2)
        try:
            T_bh = int(100 + 2000 * cool)
            hc = slsqp_polish(h_p, T_bh, n, maxiter=300, ftol=1e-12)
            hc, c5v = coord_descent_fast(hc, n, time.time() + 1.5, n_sweeps=5)
            temp = 0.004 * cool + 0.0001
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h, work_c5 = hc, c5v
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc.copy()
        except Exception:
            pass

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 4: Multi-resolution search (15% budget)
    for n_try in [128, 144, 192, 224]:
        if n_try == n or elapsed() > budget_s * 0.70:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2)
        for T in [300, 2000]:
            if elapsed() > budget_s * 0.70:
                break
            try:
                hc = slsqp_polish(h_i, T, n_try, maxiter=600, ftol=1e-13)
                hc, c5v = coord_descent_fast(hc, n_try, time.time() + 4.0, n_sweeps=15)
                if c5v < best_c5:
                    best_c5, best_h, n = c5v, hc, n_try
            except Exception:
                pass

    # Phase 5: Final coordinate descent + high-T SLSQP (20% budget)
    if elapsed() < budget_s * 0.90:
        best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.93, n_sweeps=300)

    for T in [15000, 80000]:
        if elapsed() > budget_s * 0.97:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=1200, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Tiny final coord pass
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.99, n_sweeps=50)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
