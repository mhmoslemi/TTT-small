def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def topk_grad(h, N, k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    peak = float(np.max(corr))
    kk = min(k, len(corr))
    top = np.argpartition(corr, -kk)[-kk:]
    grad = np.zeros(N)
    center = N - 1
    wk = 1.0 / kk
    for ai in top:
        L = int(ai) - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wk * (1.0 - h[:N-L])
            grad[:N-L] -= wk * h[L:]
        else:
            M = -L
            if M < N:
                grad[:N-M] += wk * (1.0 - h[M:])
                grad[M:] -= wk * h[:N-M]
    return grad, peak

def adam_run(h, N, target, deadline, k0, k1, lr0, decay):

    b1, b2, eps = 0.9, 0.999, 1e-8

    v = np.zeros(N)
    it = 0
    lr = lr0
    while time.time() < deadline:
        frac = min(1.0, it / 300.0)
        k = int(k0 + frac * (k1 - k0))

        if peak < best_peak:

        hp, c4 = adam_run(hp, N, target, d4, 8, 120, 0.025, 0.997)
        if c4 < best_c5:
            best_c5, best_h = c4, hp.copy()

    # Phase 5: Second extended binary search
    if time.time() < deadline - 80:
        d5 = min(deadline - 70, t0 + 880)
        h5, c5 = bin_search(best_h, N, d5, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()
        if time.time() < deadline - 50:
            hg2, cg2 = greedy_pass(best_h, N, rng, n_trials=3000)
            if cg2 < best_c5:
                best_c5, best_h = cg2, hg2

    # Phase 6: Try higher resolution
    if time.time() < deadline - 40:
        NH = 440
        x0 = np.linspace(0, 2, N, endpoint=False)
        xH = np.linspace(0, 2, NH, endpoint=False)
        hH = np.interp(xH, x0, best_h)
        hH = project_sum(hH, NH / 2.0)
        d6 = min(deadline - 30, time.time() + 25)
        hH, c6 = adam_run(hH, NH, NH / 2.0, d6, 10, 140, 0.04, 0.995)
        if c6 < best_c5:
            best_c5, best_h = c6, hH

    # Final polish
    if time.time() < deadline - 3:
        Nf = len(best_h)
        hf = best_h.astype(float) + rng.randn(Nf) * 0.008
        hf = project_sum(hf, Nf / 2.0)
        hf, cF = adam_run(hf, Nf, Nf / 2.0, deadline - 2, 5, 80, 0.015, 0.998)
        if cF < best_c5:
            best_c5, best_h = cF, hf

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
