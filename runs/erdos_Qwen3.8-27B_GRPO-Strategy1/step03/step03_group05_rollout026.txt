import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            room = h.copy()
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, T, n, pen):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    s = h.sum() - n / 2.0
    return val + pen * s * s

def lse_grad_fast(h, T, n, pen, top_k=15):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    c_top = corr[top_idx].copy()
    m = np.max(c_top)
    e = np.exp(T * (c_top - m))
    e /= e.sum()
    grad = np.zeros(n)
    for idx, w in zip(top_idx, e):
        s = int(idx - (n - 1))
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            ss = -s
            grad[:n - ss] += w * (1.0 - h[ss:])
            grad[ss:] -= w * h[:n - ss]
    grad *= dx
    s_err = h.sum() - n / 2.0
    grad += 2.0 * pen * s_err
    return grad

def lbfgsb_opt(h, T, n, pen, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, args=(T, n, pen), method='L-BFGS-B',
                       jac=lambda x: lse_grad_fast(x, T, n, pen, top_k=20),
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def nesterov_pg(h, n_steps, T, alpha0, decay, momentum):
    n = len(h)
    v = np.zeros(n)
    y = h.copy()
    m = momentum
    best = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        alpha = alpha0 * (decay ** step)
        g = lse_grad_fast(y, T, n, 0.0, top_k=8)
        v = m * v + alpha * g
        y_new = y - v
        y_new = project_feasible(y_new, n)
        c5 = compute_c5(y_new)
        if c5 < best_c5:
            best_c5 = c5
            best = y_new.copy()
        y = y_new
        m = min(0.95, m + 0.001)
    return best, best_c5

def fourier_init(n, K=4, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = rng.uniform(-0.12, 0.12)
        b = rng.uniform(-0.12, 0.12)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return project_feasible(h, n)

def block_perturb(h, n, scale):
    h_new = h.copy()
    for _ in range(3):
        start = np.random.randint(0, n - 5)
        length = np.random.randint(5, max(6, n // 4))
        end = min(start + length, n)
        h_new[start:end] += np.random.randn(end - start) * scale
    return project_feasible(h_new, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    remaining = lambda: budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing temperature
    temps = [30, 80, 200, 500, 1500, 5000, 15000]
    for T in temps:
        if remaining() < 60:
            break
        pen = 100.0
        h_new = lbfgsb_opt(best_h, T, best_n, pen, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Phase 2: Nesterov projected gradient
    if remaining() > 80:
        for T in [200, 2000]:
            h_ng, c5_ng = nesterov_pg(best_h, 800, T, 0.008, 0.999, 0.5)
            if c5_ng < best_c5:
                best_c5, best_h = c5_ng, h_ng
            h_new = lbfgsb_opt(best_h, 5000, best_n, 200.0, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new

    # Phase 3: Random restarts with L-BFGS-B
    n_restarts = max(5, int(remaining() / 50))
    for r in range(n_restarts):
        if remaining() < 50:
            break
        scale = 0.04 + 0.03 * (r % 3)
        h_pert = block_perturb(best_h, best_n, scale)
        h_pert = lbfgsb_opt(h_pert, 1000, best_n, 100.0, maxiter=250)
        c5_p = compute_c5(h_pert)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_pert

    # Phase 4: Multi-resolution
    for n_try in [80, 120, 160]:
        if remaining() < 80:
            break
        h_low = interpolate_h(best_h, best_n, n_try)
        h_low = lbfgsb_opt(h_low, 500, n_try, 50.0, maxiter=300)
        h_low = lbfgsb_opt(h_low, 5000, n_try, 200.0, maxiter=300)
        # Interpolate back up
        h_up = interpolate_h(h_low, n_try, best_n)
        h_up = lbfgsb_opt(h_up, 5000, best_n, 200.0, maxiter=250)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5, best_h = c5_up, h_up
        # Also try keeping at lower resolution
        c5_low = compute_c5(h_low)
        if c5_low < best_c5:
            best_c5, best_h, best_n = c5_low, h_low, n_try

    # Phase 5: Fourier-based init + optimize
    if remaining() > 60:
        for fk in range(4):
            if remaining() < 40:
                break
            h_f = fourier_init(best_n, K=4 + fk, seed=seed + 100 + fk)
            h_f = lbfgsb_opt(h_f, 500, best_n, 100.0, maxiter=300)
            h_f = lbfgsb_opt(h_f, 5000, best_n, 200.0, maxiter=200)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5, best_h = c5_f, h_f

    # Phase 6: Intensive final polish
    if remaining() > 30:
        for T in [20000, 80000]:
            if remaining() < 20:
                break
            h_f = lbfgsb_opt(best_h, T, best_n, 500.0, maxiter=500)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5, best_h = c5_f, h_f
        # Nesterov micro-steps
        h_ng, c5_ng = nesterov_pg(best_h, 500, 10000, 0.002, 0.9995, 0.7)
        if c5_ng < best_c5:
            best_c5, best_h = c5_ng, h_ng
        h_f = lbfgsb_opt(best_h, 100000, best_n, 1000.0, maxiter=300)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
