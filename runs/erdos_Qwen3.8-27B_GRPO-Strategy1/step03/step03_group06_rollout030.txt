def nesterov_subgrad(h, n, steps, lr_init, topk):
    """Nesterov-accelerated projected subgradient method."""
    h = h.copy()
    h_prev = h.copy()
    beta = 0.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for step in range(steps):
        # Nesterov step
        beta = (-1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = h + beta * (h - h_prev)
        
        # Compute subgradient at y
        dx = 2.0 / n
        corr = np.correlate(y, 1.0 - y, mode='full')
        shifts = np.arange(-(n-1), n)
        topk_idx = np.argpartition(corr, -topk)[-topk:]
        # Weights proportional to values (soft top-k)
        vals = corr[topk_idx]
        weights = vals - vals.min() + 0.1
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, idx in zip(weights, topk_idx):
            s = int(shifts[idx])
            g += w * grad_corr_s(y, s, n)
        g *= dx
        
        # Learning rate
        lr = lr_init / (1.0 + step * 0.01)
        
        # Take step
        h_prev = h.copy()
        h = h - lr * g
        h = proj_feasible(h, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-10:
                h += diff * room / tot
        else:
            tot = h.sum()
            if tot > 1e-10:
                h += diff * h / max(tot, 1e-10)
    return np.clip(h, 0.0, 1.0)

def grad_corr_at_shift(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        idx1 = np.arange(0, n - s)
        idx2 = np.arange(s, n)
        g[idx1] += 1.0 - h[idx1 + s]
        g[idx2] -= h[idx2 - s]
    else:
        s2 = -s
        idx1 = np.arange(s2, n)
        idx2 = np.arange(0, n - s2)
        g[idx1] += 1.0 - h[idx1 - s2]
        g[idx2] -= h[idx2 + s2]
    return g

def subgradient_f(h, n, topk=5):
    dx = 2.0 / n
    corr = corr_array(h)
    shifts = np.arange(-(n-1), n)
    if topk < len(corr):
        topk_idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[topk_idx]
        # Soft weighting: proportional to exp(T * (vals - max))
        T_soft = 50.0
        w = np.exp(T_soft * (vals - vals.max()))
        w /= w.sum()
        g = np.zeros(n)
        for wi, idx in zip(w, topk_idx):
            g += wi * grad_corr_at_shift(h, int(shifts[idx]), n)
    else:
        idx_max = np.argmax(corr)
        g = grad_corr_at_shift(h, int(shifts[idx_max]), n)
    return g * dx

    best_h = h.copy()
    for step in range(steps):
        beta = 0.5 * (-1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
        y = h + beta * (h - h_prev)
        g = subgradient_f(y, n, topk)
        lr = lr0 / (1.0 + 0.005 * step)
        h_prev = h.copy()
        h = h - lr * g
        h = proj_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_smooth(h, T, n, topk, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        hx = proj_feasible(x, n)
        return smooth_max_obj(hx, T, topk)
    res = minimize(obj, h, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj_feasible(np.clip(res.x, 0, 1), n)

def smooth_max_obj(h, T, topk):
    n = len(h)
    dx = 2.0 / n
    corr = corr_array(h)
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        corr = corr[idx]
    m = corr.max()
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

Let me also think about the coordinate descent more carefully. The issue with the sum constraint is that changing one coordinate changes the sum. One clean approach:

"Paired coordinate descent": 
- Pick two coordinates (j, k)
- Change h[j] += δ, h[k] -= δ (preserves sum)
- Optimize over δ and the choice of (j, k)

This is like a "swap" operation. For each pair (j, k), the objective as a function of δ is:
f(δ) = max_s corr_s(δ) * dx

where corr_s(δ) depends linearly on δ (for s≠0) or quadratically (for s=0, since h[j] and h[k] both appear in the h² terms).

This is the max of O(n) linear/quadratic functions in δ, and we can find the minimum over δ ∈ [h[k]-h[j], ...] (bounded by keeping both in [0,1]).

This is a 1D optimization problem that can be solved efficiently.

But trying all O(n²) pairs is expensive. Instead, I can:
- For each j, try changing h[j] and compensating with a uniform spread (approximate)
- Or: for the top-5 worst shifts, identify which coordinates are "responsible" and swap between them

Let me implement a simpler version: for each coordinate j, try all values in a small set, project, evaluate.

One more thought: the interval parameterization from the strategy. Let me think about how to implement it:

- h is the indicator of K disjoint intervals [a_i, b_i] in [0, 2]
- Sum of (b_i - a_i) = 1 (integral constraint)
- We optimize over the 2K parameters (a_1, b_1, ..., a_K, b_K)
- Subject to: 0 ≤ a_1 < b_1 ≤ a_2 < b_2 ≤ ... ≤ a_K < b_K ≤ 2
- And: sum(b_i - a_i) = 1

The objective is still C5 = max_k ∫ h(x)(1-h(x+k)) dx, computed on a discrete grid.

For K=5-10 intervals, this is a 10-20 dimensional optimization, much lower than 160. SLSQP or even differential_evolution should work well.

The advantage: different parameterization = different basins of attraction.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.array(h, dtype=float)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid,0,1).sum() > target: lo = mid
        else: hi = mid
    h = np.clip(h - 0.5*(lo+hi), 0, 1)
    d = target - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t > 1e-10: h += d*r/t
        else:
            t = h.sum()
            if t > 1e-10: h += d*h/max(t,1e-10)
    return np.clip(h, 0, 1)

def g_shift(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(n-s)
        i2 = np.arange(s, n)
        g[i1] += 1.0 - h[i1+s]
        g[i2] -= h[i2-s]
    else:
        s2 = -s
        i1 = np.arange(s2, n)
        i2 = np.arange(n-s2)
        g[i1] += 1.0 - h[i1-s2]
        g[i2] -= h[i2+s2]
    return g

def subgrad(h, n, topk, Tsoft):
    dx = 2.0/n
    c = corr_arr(h)
    shifts = np.arange(-(n-1), n)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        v = c[idx]
        w = np.exp(Tsoft*(v - v.max()))
        w /= w.sum()
        g = sum(wi * g_shift(h, int(shifts[i]), n) for wi, i in zip(w, idx))
    else:
        g = g_shift(h, int(shifts[np.argmax(c)]), n)
    return g * dx

def nesterov(h, n, steps, lr0, topk, Tsoft):
    h = h.copy(); hp = h.copy(); beta = 0.0
    bc5 = compute_c5(h); bh = h.copy()
    for step in range(steps):
        beta = 0.5*(-1.0+np.sqrt(1.0+4.0*beta*beta))
        y = h + beta*(h-hp)
        g = subgrad(y, n, topk, Tsoft)
        lr = lr0/(1.0+0.003*step)
        hp = h.copy()
        h = proj(h - lr*g, n)
        c5 = compute_c5(h)
        if c5 < bc5: bc5 = c5; bh = h.copy()
        if step % 500 == 0 and step > 0:
            lr = lr * 0.9  # slow decay
    return bh, bc5

def smooth_obj(h, T, topk):
    n = len(h); dx = 2.0/n
    c = corr_arr(h)
    if topk and topk < len(c):
        c = c[np.argpartition(c,-topk)[-topk:]]
    m = c.max()
    return (m + np.log(np.sum(np.exp(T*(c-m))))/T)*dx

def slsqp(h, T, n, topk, maxi, ftol):
    bnds = [(0.0,1.0)]*n
    def obj(x): return smooth_obj(x, T, topk)
    def con_f(x): return x.sum() - n/2.0
    res = minimize(obj, h, method='SLSQP', bounds=bnds,
                   constraints=[{'type':'eq','fun':con_f}],
                   options={'maxiter':maxi,'ftol':ftol})
    return proj(np.clip(res.x,0,1), n)

def interval_init(K, n, seed):
    """Create h as union of K intervals, random placement."""
    rng = np.random.RandomState(seed)
    # Total width = 1 (in [0,2] domain)
    # Generate K widths that sum to 1
    w = rng.dirichlet(np.ones(K))  # sums to 1
    # Place intervals with small gaps
    total_w = w.sum()  # = 1
    gaps = 2.0 - total_w  # = 1
    gap_vals = rng.dirichlet(np.ones(K+1)) * gaps  # K+1 gaps (including ends)
    # Build interval boundaries
    h = np.zeros(n)
    x = np.linspace(0, 2, n, endpoint=False)
    pos = 0.0
    for i in range(K):
        pos += gap_vals[i]
        a = pos; b = pos + w[i]
        mask = (x >= a) & (x < b)
        h[mask] = 1.0
        pos = b
    pos += gap_vals[K]
    return proj(h, n)

def interval_opt(params, n):
    """Evaluate C5 for interval parameters."""
    K = len(params) // 2
    h = np.zeros(n)
    x = np.linspace(0, 2, n, endpoint=False)
    for i in range(K):
        center = params[2*i]
        width = params[2*i+1]
        a = center - width/2; b = center + width/2
        mask = (x >= max(0,a)) & (x < min(2,b))
        h[mask] = 1.0
    # Handle overlaps: take max (union)
    return h  # Doesn't enforce disjointness cleanly
