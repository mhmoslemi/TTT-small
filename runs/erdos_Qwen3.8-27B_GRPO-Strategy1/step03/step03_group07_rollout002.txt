        # Try to reduce each active shift by targeted modification
        for ti in top_idx[:min(10, n_active)]:
            if time.time() - t0 > budget_s * 0.40:
                break
            shift = ti - (n - 1)
            h_try = best_h.copy()
            # Small random perturbation focused near the shift
            if shift >= 0:
                lo, hi = max(0, shift - 5), min(n, n - shift + 5)
            else:
                lo, hi = max(0, -shift - 5), min(n, n + shift + 5)
            h_try[lo:hi] += rng.uniform(-0.04, 0.04, hi - lo)
            h_try = make_feasible(h_try, n)
            T_adapt = 2000 + 2000 * round_num
            try:
                h_opt = slsqp_optimize(h_try, T_adapt, maxiter=600, ftol=1e-14)
                c5v = compute_c5(h_opt)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_opt
            except Exception:
                pass

    # Phase 3: Basin hopping with mixed strategies
    bh_h = best_h.copy()
    bh_c5 = best_c5
    elapsed = time.time() - t0
    remaining = budget_s * 0.65 - elapsed
    n_iter = max(20, int(remaining / 3.5))

    for it in range(n_iter):
        if time.time() - t0 > budget_s * 0.65:
            break
        ptype = it % 6
        scale = 0.06 * (1.0 - it / max(1, n_iter)) + 0.01
        if ptype == 0:
            h_pert = bh_h + scale * rng.randn(n)
        elif ptype == 1:
            h_pert = targeted_perturb(bh_h, n, rng)
        elif ptype == 2:
            h_pert = bh_h.copy()
            blen = max(2, n // 12)
            s1 = rng.randint(0, n - blen)
            s2 = rng.randint(0, n - blen)
            if s1 != s2:
                tmp = h_pert[s1:s1+blen].copy()
                h_pert[s1:s1+blen] = h_pert[s2:s2+blen]
                h_pert[s2:s2+blen] = tmp
        elif ptype == 3:
            h_pert = bh_h.copy()
            nf = max(1, n // 12)
            idx = rng.choice(n, nf, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif ptype == 4:
            # Gaussian with larger scale for exploration
            h_pert = bh_h + scale * 2.5 * rng.randn(n)
        else:
            # Mix with a new random feasible point
            h_rand = rng.rand(n)
            h_rand = make_feasible(h_rand, n)
            h_pert = 0.7 * bh_h + 0.3 * h_rand
            h_pert = make_feasible(h_pert, n)

        h_pert = make_feasible(h_pert, n)
        T_use = 300 + 400 * (it // 8)
        try:
            h_new = slsqp_optimize(h_pert, T_use, maxiter=400, ftol=1e-13)
            c5v = compute_c5(h_new)
        except Exception:
            h_new = h_pert
            c5v = compute_c5(h_pert)

        temp = 0.0025 * (1.0 - it / max(1, n_iter)) + 0.0003
        if c5v < bh_c5 or rng.rand() < np.exp(-(c5v - bh_c5) / temp):
            bh_c5 = c5v
            bh_h = h_new
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new

    # Phase 4: Multi-resolution
    for n_try in [100, 150, 250, 300, 400]:
        if time.time() - t0 > budget_s * 0.82:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2)
        h_interp = make_feasible(h_interp, n_try)
        for T in [500, 2000, 8000]:
            if time.time() - t0 > budget_s * 0.82:
                break
            try:
                h_new = slsqp_optimize(h_interp, T, maxiter=1200, ftol=1e-15)
                c5v = compute_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
            except Exception:
                pass

    # Phase 5: Intensive final polish
    n = len(best_h)
    for T in [10000, 50000, 200000]:
        if time.time() - t0 > budget_s * 0.95:
            break
        try:
            h_new = slsqp_optimize(best_h.copy(), T, maxiter=4000, ftol=1e-18)
            c5v = compute_c5(h_new)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_new
        except Exception:
            pass

    # Phase 6: Coordinate descent micro-tuning
    if time.time() - t0 < budget_s * 0.995:
        stuck = 0
        for it in range(400):
            if time.time() - t0 > budget_s * 0.995:
                break
            improved = False
            for _ in range(30):
                j = rng.randint(0, n)
                for delta in [0.005, -0.005, 0.015, -0.015, 0.03, -0.03, 0.06, -0.06]:
                    h_try = best_h.copy()
                    h_try[j] = np.clip(best_h[j] + delta, 0.0, 1.0)
                    h_try = make_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                stuck += 1
            else:
                stuck = 0
            if stuck > 100:
                break

    # Final verification
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
