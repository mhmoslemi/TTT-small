    corr = np.conf vect(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    return np.max(corr_base + x * coefs)

def solve_1d_convex(a, b):
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = a + b * mid
        f_val = np.max(vals)
        active = vals >= f_val - 1e-14
        if np.min(b[active]) >= -1e-14:
            hi = mid
        elif np.max(b[active]) <= 1e-14:
            lo = mid
        else:
            break
    x_opt = (lo + hi) * 0.5
    if x_opt < 1e-13:
        x_opt = 0.0
    elif x_opt > 1.0 - 1e-13:
        x_opt = 1.0
    return x_opt

def get_coefs(h, i):
    n = len(h)
    h0 = h.copy()
    h1 = h.copy()
    h0[i] = 0.0
    h1[i] = 1.0
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    c1 = np.correlate(h1, 1.0 - h1, mode='full')
    return c0, c1 - c0

def coord_pass(h, n, order):
    for i in order:
        corr0, coefs = get_coefs(h, i)
        x_opt = solve_1d_convex(corr0, coefs)
        h[i] = x_opt
        s = h.sum()
        diff = n / 2.0 - s
        if abs(diff) > 1e-12:
            h = project_h(h)
    return h

def subgrad_step(h, n, alpha, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    K = min(K, n_corr)
    top_idx = np.argpartition(-corr, K)[:K]
    top_idx = top_idx[np.argsort(-corr[top_idx])]
    grad = np.zeros(n)
    for m in top_idx:
        k = m - (n - 1)
        if k >= 0:
            n_valid = np.arange(0, n - k)
        else:
            n_valid = np.arange(-k, n)
        p_left = n_valid + k
        for p, nv in zip(p_left, n_valid):
            grad[p] += (1.0 - h[nv]) * alpha
        ik = k + n_valid
        valid_ik = (ik >= 0) & (ik < n)
        for p, nvi in zip(n_valid, ik):
            if valid_ik[p]:
                grad[p] += -h[nvi] * alpha
    h_new = h + alpha * grad
    return project_h(h_new)

def make_pert(n, cool, rng, mode):
    if mode == 0:
        p = (0.03 + 0.08 * cool) * rng.standard_normal(n)
    elif mode == 1:
        bs = rng.randint(6, max(7, n // 6))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.05 + 0.15 * cool) * rng.standard_normal(bs)
    elif mode == 2:
        freq = rng.randint(1, max(2, n // 20))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.02 + 0.06 * cool) * np.sin(2.0 * np.pi * freq * np.arange(n) / n + ph)
    elif mode == 3:
        np_ = rng.randint(3, max(4, n // 10))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.1 + 0.3 * cool) * rng.standard_normal(np_)
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.06 * cool * rng.standard_normal(half)
        p[half:] = -0.06 * cool * rng.standard_normal(n - half)
    return p

def binarize_try(h, n, rng, t_end, n_iter=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        idx = rng.randint(n)
        val = best_h[idx]
        if val < 0.1 or val > 0.9:
            continue
        flip = 1.0 if val < 0.5 else 0.0
        h_try = best_h.copy()
        h_try[idx] = flip
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    work_h = best_h.copy()

    # Phase 1: Intensive coordinate descent (40% budget)
    t_phase1 = t0 + budget_s * 0.40
    passes = 0
    while time.time() < t_phase1:
        order = rng.permutation(n)
        work_h = coord_pass(work_h, n, order)
        passes += 1
        c5_cur = compute_c5(work_h)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = work_h.copy()

    # Phase 2: Subgradient equalization + coordinate polish (15% budget)
    t_phase2 = t0 + budget_s * 0.55
    alpha = 0.02
    for it in range(200):
        if time.time() > t_phase2:
            break
        K = rng.randint(2, 6)
        work_h = subgrad_step(work_h, n, alpha, K)
        work_h = coord_pass(work_h, n, rng.permutation(n))
        c5_cur = compute_c5(work_h)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = work_h.copy()
        alpha *= 0.95

    # Phase 3: Basin hopping (30% budget)
    t_phase3 = t0 + budget_s * 0.85
    work_h = best_h.copy()
    work_c5 = best_c5
    n_hop = 0
    while time.time() < t_phase3:
        n_hop += 1
        cool = max(0.15, 1.0 - 0.7 * (n_hop % 80) / 80.0)
        mode = rng.randint(5)
        pert = make_pert(n, cool, rng, mode)
        h_try = project_h(work_h + pert)
        for _ in range(3):
            h_try = coord_pass(h_try, n, rng.permutation(n))
        c5_try = compute_c5(h_try)
        temp = 0.0003 * cool + 0.00005
        if c5_try < work_c5 or (c5_try < work_c5 + 0.002 and rng.rand() < np.exp(-(c5_try - work_c5) / max(temp, 1e-12))):
            work_h = h_try
            work_c5 = c5_try
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = work_h.copy()

    # Phase 4: Binarize + coordinate descent (10% budget)
    t_phase4 = t0 + budget_s * 0.93
    work_h = best_h.copy()
    work_h, work_c5 = binarize_try(work_h, n, rng, t_phase4, n_iter=300)
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()
    for _ in range(20):
        if time.time() > t_phase4:
            break
        best_h = coord_pass(best_h, n, rng.permutation(n))
        c5_cur = compute_c5(best_h)
        if c5_cur < best_c5:
            best_c5 = c5_cur

    # Phase 5: Final polish (7% budget)
    t_end = t0 + budget_s * 0.98
    for _ in range(50):
        if time.time() > t_end:
            break
        best_h = coord_pass(best_h, n, rng.permutation(n))
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Verify constraints
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10), "Bounds violated"
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
