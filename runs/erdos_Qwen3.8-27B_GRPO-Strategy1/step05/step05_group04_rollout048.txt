```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_grad(h, n):
    """Compute correlation vector and gradient of max(corr)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Subgradient: uniform over argmax set
    active = corr >= m - 1e-10
    weights = np.zeros(len(corr))
    weights[active] = 1.0 / active.sum()
    # grad[i] = dx * sum_k weights[k] * d(corr[k])/d(h[i])
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        s = k_idx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n-a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n-a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return corr * dx, grad * dx, m * dx
```
