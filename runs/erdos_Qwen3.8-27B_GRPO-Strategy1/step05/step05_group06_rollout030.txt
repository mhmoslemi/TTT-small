```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    target = n / 2.0
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def coord_descent(h, t_end, max_sweeps=2000):
    n = len(h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    best_h = project_h(h.copy())
    best_c5 = compute_c5(best_h)
    
    grid = np.linspace(0.0, 1.0, 256)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = compute_corr(best_h)
        v_base = 1.0 - 2.0 * best_h  # (n,)
        
        for i in range(n):
            m_lo = n - 1 - i
            m_hi = 2 * n - 2 - i
            corr_aff = corr[m_lo:m_hi+1]  # (n,)
            # corr_rest_max
            if m_lo > 0:
                rest = np.concatenate([corr[:m_lo], corr[m_hi+1:]])
                corr_rest_max = rest.max()
            else:
                corr_rest_max = corr[m_hi+1:].max() if m_hi+1 < m_full else corr[:m_lo].max()
            
            # f(x) = max(corr_aff + x * v_base, corr_rest_max)
            # (The -x^2 self term never dominates for x in [0,1])
            vals = corr_aff[:, None] + grid[None, :] * v_base[:, None]  # (n, 256)
            obj = np.maximum(vals.max(axis=0), corr_rest_max)
            idx = obj.argmin()
            x_try = grid[idx]
            
            if abs(x_try - best_h[i]) < 1e-12:
                continue
            
            # Refine around x_try
            lo_b = max(0.0, grid[max(0, idx-3)])
            hi_b = min(1.0, grid[min(len(grid)-1, idx+3)])
            
            # Use minimize_scalar for refinement
            def f1d(x):
                return max(np.max(corr_aff + x * v_base), corr_rest_max)
            
            try:
                res = minimize_scalar(f1d, bounds=(lo_b, hi_b), method='bounded',
                                     options={'xatol': 1e-14})
                x_new = res.x
            except:
                x_new = x_try
            
            if abs(x_new - best_h[i]) < 1e-13:
                continue
            
            old_val = best_h[i]
            best_h[i] = x_new
            best_h = project_h(best_h)
            c5_new = compute_c5(best_h)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old_val
                best_h = project_h(best_h)
        
        if not improved:
            break
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    target = n / 2.0
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def coord_descent(h0, t_end, max_sweeps=5000):
    n = len(h0)
    dx = 2.0 / n
    best_h = project_h(h0.copy())
    best_c5 = compute_c5(best_h)
    grid = np.linspace(0.0, 1.0, 300)
    m_full = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = compute_corr(best_h)
        v_base = 1.0 - 2.0 * best_h
        for i in range(n):
            m_lo = n - 1 - i
            m_hi = 2 * n - 2 - i
            corr_aff = corr[m_lo:m_hi + 1]
            mask = np.ones(m_full, dtype=bool)
            mask[m_lo:m_hi + 1] = False
            corr_rest_max = corr[mask].max() if mask.any() else -np.inf
            vals = corr_aff[:, None] + grid[None, :] * v_base[:, None]
            obj = np.maximum(vals.max(axis=0), corr_rest_max)
            idx = obj.argmin()
            x_try = grid[idx]
            if abs(x_try - best_h[i]) < 1e-13:
                continue
            lo_b = max(0.0, grid[max(0, idx - 4)])
            hi_b = min(1.0, grid[min(len(grid) - 1, idx + 4)])
            carr = corr_aff
            vbb = v_base
            crm = corr_rest_max
            def f1d(x, _c=carr, _v=vbb, _r=crm):
                return max(float(np.max(_c + x * _v)), _r)
            try:
                res = minimize_scalar(f1d, bounds=(lo_b, hi_b), method='bounded',
                                      options={'xatol': 1e-14})
                x_new = res.x
            except Exception:
                x_new = x_try
            if abs(x_new - best_h[i]) < 1e-14:
                continue
            old_val = best_h[i]
            best_h[i] = x_new
            best_h = project_h(best_h)
            c5_new = compute_c5(best_h)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old_val
                best_h = project_h(best_h)
                corr = compute_corr(best_h)
                v_base = 1.0 - 2.0 * best_h
        if not improved:
            break
    return best_h, best_c5

def lbfgs_refine(h0, T, lam, maxiter, ftol):
    n = len(h0)
    dx = 2.0 / n
    m_full = 2 * n - 1

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = float(np.max(corr))
        Tc = T * (corr - mc)
        Tc -= np.max(Tc)
        exps = np.exp(Tc)
        w = exps / exps.sum()
        sm = mc + np.log(exps.sum()) / T
        grad = np.zeros(n)
        for idx in range(m_full):
            wi = w[idx]
            if wi < 1e-14:
                continue
            lg = idx - n + 1
            if lg == 0:
                grad[:] += wi * (1.0 - 2.0 * x)
            elif lg > 0:
                grad[lg:] += wi * (1.0 - x[:n - lg])
                grad[:n - lg] += -wi * x[lg:]
            else:
                p = -lg
                grad[:n - p] += wi * (1.0 - x[p:])
                grad[p:] += -wi * x[:n - p]
        val = dx * sm + lam * (x.sum() - n / 2.0) ** 2
        grad = dx * grad + 2.0 * lam * (x.sum() - n / 2.0)
        return val, grad

    x0 = project_h(h0)
    try:
        res = minimize(obj_grad, x0, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-13, 'maxls': 40})
        return project_h(res.x)
    except Exception:
        return x0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive coordinate descent (45% budget)
    p1_end = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, p1_end, 5000)

    # Phase 2: L-BFGS-B smooth refinement (15% budget)
    p2_end = t0 + budget_s * 0.60
    for T in [3000, 10000, 30000]:
        if time.time() > p2_end:
            break
        hc = lbfgs_refine(best_h, T, 200.0, 3000, 1e-18)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: More coordinate descent (20% budget)
    p3_end = t0 + budget_s * 0.80
    best_h, best_c5 = coord_descent(best_h, p3_end, 5000)

    # Phase 4: Multi-restart L-BFGS-b (10% budget)
    p4_end = t0 + budget_s * 0.90
    for sc in [0.02, 0.05, 0.1]:
        if time.time() > p4_end:

        h_p = project_h(best_h + sc * np.random.randn(n))
        hc = lbfgs_refine(h_p, 8000, 200.0, 2000, 1e-17)

        if c5v < best_c5:

            best_h = hc

    # Phase 5: Final coordinate descent polish (9% budget)
    p5_end = t0 + budget_s * 0.99

    # Final L-BFGS-B
```
