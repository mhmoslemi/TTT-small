                    b += 1.0 - best[i + sh]
                if 0 <= i - sh < n:
                    b -= best[i - sh]
                B[k] = b
            
            # For k = n-1 (sh=0): the formula gives B = (1-best[i]) - best[i] = 1-2*best[i]
            # But actually c_{n-1} = sum x[j]*(1-x[j]), and the term involving x[i] is x[i]*(1-x[i]).
            # c_{n-1}(new) = corr[n-1] - old*(1-old) + new*(1-new)
            # This is quadratic, not linear. My B formula gives:
            # B_{n-1} = (1-best[i]) - best[i] (from the two conditions, both true since sh=0)
            # c_{n-1} + (new-old)*B_{n-1} = corr[n-1] + (new-old)*(1-2*old)
            # But actual: corr[n-1] - old*(1-old) + new*(1-new)
            # = corr[n-1] - old + old² + new - new²
            # While linear approx: corr[n-1] + new*(1-2*old) - old*(1-2*old)
            # = corr[n-1] + new - 2*old*new - old + 2*old²
            # These are different! So for k=n-1, we need to handle it specially.
            
            # For the 1D optimization, let's handle it as:
            # For k != n-1: c_k(new) = corr[k] + (new-old)*B[k]
            # For k = n-1: c_{n-1}(new) = corr[n-1] - old*(1-old) + new*(1-new)
            
            # Define f(new) = max_k c_k(new)
            # For k != n-1: linear in new
            # For k = n-1: quadratic in new (opens upward, minimum at new=0.5)
            
            # To minimize f(new) over new in [0,1]:
            # The max of linear functions is convex, and the quadratic is also convex.
            # So f is convex in new. We can use minimize_scalar.
            
            # Actually, let me just use a simpler approach: evaluate f at a few points.
            # Or use the fact that the optimum is where two or more c_k values are equal.
            
            # For efficiency, let's use minimize_scalar with a bounded method.
            
            def f1d(nv):
                # c_k(nv) for all k
                vals = corr + (nv - old) * B
                vals[n-1] = corr[n-1] - old * (1 - old) + nv * (1 - nv)
                return vals.max()
            
            r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                               options={'xatol': 1e-14})
            nv = float(r.x)
            if abs(nv - old) < 1e-14:
                continue
            best[i] = nv
            # Update corr
            corr = corr + (nv - old) * B
            corr[n-1] = corr[n-1] - old * (1 - old) + nv * (1 - nv)
            # Wait, I need to be more careful. corr was the OLD correlation.
            # After updating best[i], the correlation changes.
            # For k != n-1: new_corr[k] = old_corr[k] + (nv-old)*B[k]
            # For k = n-1: new_corr[n-1] = old_corr[n-1] - old*(1-old) + nv*(1-nv)
            # BUT: after this update, B for OTHER coordinates changes too!
            # Because B_k depends on best[i+sh] and best[i-sh].
            # Since we only changed best[i], B_k for other coordinates i' changes only
            # if i' + sh_k = i or i' - sh_k = i, i.e., i' = i - sh_k or i' = i + sh_k.
            # For the NEXT coordinate we update, we'll recompute B from scratch anyway.
            # So we just need to keep corr up to date.
            
            best_c5_new = corr.max()
            # But wait, we also need to reproject for the integral constraint!
            # The integral constraint: sum(best) = n/2
            # Changing best[i] from old to nv changes the sum by (nv-old)
            # If the sum is off, we need to project.
            # For simplicity in the inner loop, let's just check if we need to project.
            s = best.sum()
            if abs(s - n * 0.5) > 1e-10:
                # Project
                best = proj_h(best, n)
                corr = compute_corr(best)
            best_c5 = float(corr.max() * 2.0 / n)
            imp = True
        if not imp:
            break
    return best, best_c5

def bin_sa(h, n, t_end, max_iters=300000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    nz = n // 2
    if len(ones) > nz:
        h_b[np.random.choice(ones, len(ones) - nz, replace=False)] = 0
    elif len(ones) < nz:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, nz - len(ones), replace=False)] = 1
    dx = 2.0 / n
    corr = compute_corr(h_b)
    best_c5 = float(corr.max() * dx)
    best_h = h_b.copy()
    T = 0.03
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h_b[i] == h_b[j]:
            # try again a few times
            found = False
            for _a in range(8):
                j = np.random.randint(n)
                if h_b[i] != h_b[j]:
                    found = True
                    break
            if not found:
                continue
        h_b[i], h_b[j] = h_b[j], h_b[i]
        corr = compute_corr(h_b)
        nc5 = float(corr.max() * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        T = max(0.0003, T * 0.99997)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=200):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 4
        if mod == 0:
            pert = (0.02 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.02 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.2 + 0.4 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.2 * cool) * np.random.randn(mask.sum())
        else:
            # Swap a few pairs
            pert = np.zeros(n)
            for _ in range(np.random.randint(2, 8)):
                a, b = np.random.randint(0, n, 2)
                pert[a] += 0.3 * cool
                pert[b] -= 0.3 * cool
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 8000.0 * cool
        ht = lbfgs(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.001 * cool + 0.00005
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht
    return best, best_c5
