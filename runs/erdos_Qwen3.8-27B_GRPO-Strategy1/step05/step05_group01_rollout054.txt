def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    # Distribute the correction proportionally to available room
    if s > target:
        # Need to decrease: scale down
        excess = s - target
        room = h  # can decrease by at most h[i]
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h - room * (excess / tot)
    else:
        deficit = target - s
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (deficit / tot)
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        excess = s - target
        tot = h.sum()
        if tot < 1e-14:
            return h
        h = h - h * (excess / tot)
    else:
        deficit = target - s
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (deficit / tot)
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, n_sweeps, t_end, tol=1e-13, track_corr=None):
    dx = 2.0 / n
    shifts_arr = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full') if track_corr is None else track_corr
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        any_change = False
        for i in range(n):
            if time.time() > t_end:
                break
            jp = i + shifts_arr
            jm = i - shifts_arr
            v1 = (jp >= 0) & (jp < n)
            v2 = (jm >= 0) & (jm < n)
            jpc = np.clip(jp, 0, n - 1)
            jmc = np.clip(jm, 0, n - 1)
            slopes = (1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)
            intercepts = corr - h[i] * slopes
            best_val = np.max(intercepts)
            lo, hi = 0.0, 1.0
            for _ in range(48):
                mid = (lo + hi) * 0.5
                vals_max = np.max(slopes * mid + intercepts)
                if vals_max >= best_val + 1e-15:
                    lo, hi = 0.0, hi
                    idx = np.argmax(slopes * 1.0 + intercepts)
                    if slopes[idx] > 0:
                        hi = mid
                    else:
                        lo = mid
                else:

            t_new = (lo + hi) * 0.5

    dx = 2.0 / n

        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.logaddexp.reduce(T * (corr - m) + 1e-18)) / T * dx if T < 1e8 else m * dx
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h
