import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_proj(h, n):
    h_b = (h > 0.5).astype(float)
    cnt = int(h_b.sum())
    if cnt > n // 2:
        ones = np.where(h_b == 1)[0]
        order = ones[np.argsort(h[ones])]
        h_b[order[:cnt - n // 2]] = 0
    elif cnt < n // 2:
        zeros = np.where(h_b == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        h_b[order[:n // 2 - cnt]] = 1
    return h_b

def _slide_delta(h_b, i, d, n):
    """Correlation delta when moving 1 from i to i+d (h_b[i+d]==0)."""
    j = i + d
    mf = 2 * n - 1
    delta = np.zeros(mf)
    # Remove contribution of h[i]=1: for each k, h[i] contributes (1-h[i+k-n+1])
    for k in range(mf):
        idx = i + k - n + 1
        if 0 <= idx < n:
            delta[k] -= (1.0 - h_b[idx])
    # Add contribution of h[j]=1
    for k in range(mf):
        idx = j + k - n + 1
        if 0 <= idx < n:
            delta[k] += (1.0 - h_b[idx])
    return delta

def binary_sa(h, n, t_end, max_iters=800000):
    h_b = binary_proj(h, n).copy()
    dx = 2.0 / n
    corr = compute_corr(h_b)
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    cur_c5 = best_c5
    T = 0.015
    T_min = 0.00005
    ones = np.where(h_b > 0.5)[0]
    for it in range(max_iters):
        if time.time() > t_end:
            break
        if it % 3 == 0:
            # Slide move
            i = ones[np.random.randint(len(ones))]
            d = np.random.choice([-1, 1])
            j = i + d
            if 0 <= j < n and h_b[j] == 0:
                delta = _slide_delta(h_b, i, d, n)
                new_corr = corr + delta
                new_c5 = float(np.max(new_corr) * dx)
                if new_c5 <= cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
                    h_b[i], h_b[j] = 0, 1
                    cur_c5 = new_c5
                    corr = new_corr
                    ones = np.where(h_b > 0.5)[0]
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_b.copy()
                continue
        # Swap move
        i = ones[np.random.randint(len(ones))]
        zeros_mask = h_b == 0
        j = np.random.choice(np.where(zeros_mask)[0])
        h_b[i], h_b[j] = 0, 1
        new_corr = compute_corr(h_b)
        new_c5 = float(np.max(new_corr) * dx)
        if new_c5 <= cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            corr = new_corr
            ones = np.where(h_b > 0.5)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_b.copy()
                T = max(T_min, T * 0.997)
        else:
            h_b[i], h_b[j] = 1, 0
        T = max(T_min, T * 0.99997)
    return best_h, best_c5

def coord_desc(h, n, t_end, sweeps=150):
    best = proj_h(h.copy(), n)
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = compute_corr(best)
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                idx2 = i + k - n + 1
                if 0 <= idx2 < n:
                    coef[k] += 1.0 - best[idx2]
                idx1 = i - k + n - 1
                if 0 <= idx1 < n:
                    coef[k] -= best[idx1]
            new_corr = corr - old * coef
            rib = minimize_scalar(lambda hi: float(np.max(new_corr + hi * coef)),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            nv = float(rib.x)
            if abs(nv - old) < 1e-12:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr = compute_corr(best)
                imp = True
        if not imp:
            break
    return best, best_c5

def pert_refine(h, n, t_end, n_pert=8):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(n_pert):
        if time.time() > t_end:
            break
        pert = 0.03 * np.random.randn(n)
        hp = proj_h(best + pert, n)
        hp, c5p = coord_desc(hp, n, t_end, sweeps=25)
        if c5p < best_c5:
            best = hp
            best_c5 = c5p
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    T_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate descent
    t1 = t0 + budget_s * 0.06
    best_h, best_c5 = coord_desc(best_h, n, t1, sweeps=25)

    # Phase 2: Binary SA (main exploration)
    h_bin = binary_proj(best_h, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin
    t2 = t0 + budget_s * 0.38
    best_h, best_c5 = binary_sa(best_h, n, t2)

    # Phase 3: Second SA round from perturbed binary
    h_bin2 = best_h.copy()
    if best_h.sum() == n * 0.5:
        ones = np.where(h_bin2 > 0.5)[0]
        zeros = np.where(h_bin2 < 0.5)[0]
        for _ in range(15):
            a, b = np.random.choice(ones, 2, replace=False), np.random.choice(zeros, 2, replace=False)
            h_bin2[a], h_bin2[b] = 0, 1
    h_bin2 = proj_h(h_bin2, n)
    t3 = t0 + budget_s * 0.52
    h_new, c5_new = binary_sa(h_bin2, n, t3)
    if c5_new < best_c5:
        best_h, best_c5 = h_new, c5_new

    # Phase 4: Continuous refinement
    t4 = t0 + budget_s * 0.65
    best_h, best_c5 = coord_desc(best_h, n, t4, sweeps=60)

    # Phase 5: Perturbation + refinement
    t5 = t0 + budget_s * 0.78
    best_h, best_c5 = pert_refine(best_h, n, t5, n_pert=10)

    # Phase 6: More coordinate descent
    t6 = t0 + budget_s * 0.88
    best_h, best_c5 = coord_desc(best_h, n, t6, sweeps=50)

    # Phase 7: Fine-tune binary-continuous hybrid
    if time.time() < t0 + budget_s * 0.92:
        h_hyb = best_h.copy()
        for _ in range(4):
            if time.time() > t0 + budget_s * 0.92:
                break
            h_hyb = (h_hyb > 0.3).astype(float) * 0.95 + (h_hyb <= 0.3).astype(float) * 0.05
            h_hyb = proj_h(h_hyb, n)
            h_hyb, c5h = coord_desc(h_hyb, n, t0 + budget_s * 0.92, sweeps=15)
            if c5h < best_c5:
                best_h, best_c5 = h_hyb, c5h

    # Phase 8: Final polish
    if time.time() < T_end - 5:
        best_h, best_c5 = coord_desc(best_h, n, T_end - 2, sweeps=40)

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
