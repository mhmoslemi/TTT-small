def compute_c5(h):
    n = len(h)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    Tc = T * corr
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
            grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    return dx * grad

    Tc -= np.max(Tc)

                      method='SLSQP', bounds=bnd, constraints=cons,

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s

        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)

        if s < 1e-14:
            return h.copy()
        h = h * (1.0 + diff / s)

def lse_and_grad(h, T, lam):
