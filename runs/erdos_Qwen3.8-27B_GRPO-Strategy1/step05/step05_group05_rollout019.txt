import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, T):
    n = len(h)
    corr = get_corr(h)
    w = np.exp(T * (corr - np.max(corr)))
    w /= w.sum()
    m_arr = np.arange(2 * n - 1)
    p_arr = np.arange(n)
    s1 = p_arr[:, None] + m_arr[None, :] - (n - 1)
    s2 = p_arr[:, None] - m_arr[None, :] + (n - 1)
    v1 = (s1 >= 0) & (s1 < n)
    v2 = (s2 >= 0) & (s2 < n)
    h1 = np.where(v1, 1.0 - h[np.clip(s1, 0, n - 1)], 0.0)
    h2 = np.where(v2, h[np.clip(s2, 0, n - 1)], 0.0)
    return (h1 - h2).dot(w)

def subgrad_descent(h, n, n_iter, alpha0, decay):

        T = 50.0 * (1.0 + it / 80.0)

        h_new = project_h(h - alpha * g)
        if np.allclose(h_new, h, atol=1e-14):
            break
        h = h_new
    return h

def coord_opt(h, n, i, niter=45):
    h0 = h[i]
    h[i] = 0.0
    cb = get_corr(h)
    h[i] = h0
    m_arr = np.arange(-(n - 1), n)
    v1 = (i - m_arr >= 0) & (i - m_arr < n)
    v2 = (i + m_arr >= 0) & (i + m_arr < n)
    coef = np.zeros(2 * n - 1)
    coef[v1] += (1.0 - h[i - m_arr[v1]])
    coef[v2] += (-h[i + m_arr[v2]])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        if np.max(cb + t1 * coef) < np.max(cb + t2 * coef):
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=60):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            nv = coord_opt(best_h, n, i)
            if abs(nv - best_h[i]) < 1e-12:
                continue
            ht = best_h.copy()
            ht[i] = nv
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=1500):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return m + np.log(np.sum(np.exp(T * (c - m)))) / T
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h
