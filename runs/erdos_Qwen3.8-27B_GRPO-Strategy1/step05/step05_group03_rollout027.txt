def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if T > 1:
        sh = T * (corr - m)
        sh = np.clip(sh, -500, 0)
        return m + np.log(np.sum(np.exp(sh))) / T
    return m

def optimize_periodic(h, d, n, T=1000, maxiter=500):

    m = n // d
    h_period = h[:d].copy()
    
    def obj(p):
        h_full = np.tile(p, m)
        # Re-project to maintain constraints
        h_full = project_h(h_full, n)
        # Extract period
        h_per = h_full[:d]
        return smooth_obj(h_full, T) * (2.0 / n)
    
    def cons(p):
        return p.sum() - d / 2.0
    
    try:
        res = minimize(obj, h_period, method='SLSQP',
                      bounds=[(0.0, 1.0)] * d,
                      constraints=[{'type': 'eq', 'fun': cons}],

        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h


def opt_period(h_full, d, n, T=500, maxiter=300, n_restarts=3):
    m = n // d
    if m * d != n or m < 1:
        return h_full
    best_h = h_full.copy()
    best_c5 = compute_c5(best_h)
    for r in range(n_restarts):
        if r == 0:
            p = best_h[:d].copy()
        else:
            p = np.random.rand(d)
            p = (p - p.min()) / max(p.max() - p.min(), 1e-10)
        def obj_per(x):
            hf = np.tile(x, m)
            hf = project_h(hf, n)
            corr = np.correlate(hf, 1.0 - hf, mode='full')
            mx = np.max(corr)
            sh = np.clip(T * (corr - mx), -500, 0)
            return (mx + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)
        def eq_per(x):
            return x.sum() - d / 2.0
        try:
            res = minimize(obj_per, p, method='SLSQP',
                          bounds=[(0.0, 1.0)] * d,
                          constraints=[{'type': 'eq', 'fun': eq_per}],
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_h(np.tile(np.clip(res.x, 0, 1), m), n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    return best_h


def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            best_h[i] = 0.0
            cb = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_h[i] = old
            nm1 = n - 1
            cofs = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                lag = k - nm1
                v = 0.0
                if 0 <= i + lag < n:
                    v += 1.0 - best_h[i + lag]
                if 0 <= i - lag < n:
                    v -= best_h[i - lag]
                cofs[k] = v
            def f_1d(v):
                return np.max(cb + v * cofs)
            try:
                res = minimize_scalar(f_1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-11, 'maxiter': 50})
                hi_new = float(res.x)
            except Exception:
                continue
            if abs(hi_new - old) < 1e-13:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = project_h(ht, n)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5
