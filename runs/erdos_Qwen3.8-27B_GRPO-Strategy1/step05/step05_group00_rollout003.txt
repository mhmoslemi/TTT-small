                    # But for speed, let's just do it

        # Just try random swaps and keep if improved, plus greedy best-improvement on top correlations.

                new_max = float(new_vals.max())

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_vec(y, corr, top_k, N):
    grad = np.zeros(N)
    center = N - 1
    if top_k >= len(corr):
        top = np.arange(len(corr))
    else:
        top = np.argpartition(corr, -top_k)[-top_k:]
    for ai in top:
        L = int(ai - center)
        if L == 0:
            grad += (1.0 - 2.0 * y)
        elif L > 0:
            grad[L:] += (1.0 - y[:N - L])
            grad[:N - L] -= y[L:]
        else:
            M = -L
            if M < N:
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
    grad /= len(top)
    return grad

def nesterov_opt(h, N, deadline, k=60, lr0=0.018, beta0=0.90):

    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        grad = subgrad_vec(y, corr, k, N)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.994
            beta = min(0.97, beta0 + it * 0.00015)
    return best_h, best_pk * dx

def swap_delta_top(c, h, p, q, top_idx, N):
    center = N - 1
    km = top_idx - center
    ip = p + km
    valid_ip = (ip >= 0) & (ip < N)
    dr = np.where(valid_ip, 2.0 * h[np.clip(ip, 0, N - 1)] - 1.0, 0.0)
    iq = q + km
    valid_iq = (iq >= 0) & (iq < N)
    da = np.where(valid_iq, 1.0 - 2.0 * h[np.clip(iq, 0, N - 1)], 0.0)
    return c[top_idx] + dr + da

    elif n1 < nh:

        m_idx = m_idx[np.argsort(np.random.random(len(m_idx)))][:nh - n1]
        h[m_idx] = 1
    best_c = float(np.max(fft_corr(h)))
    for sweep in range(max_sweeps):

        c = fft_corr(h)
        P = float(c.max())

        ones = np.where(h == 1)[0]
