```python
                slope = 0.0
                if 0 <= ik < n:
                    slope += (1.0 - best_h[ik])
                if 0 <= ikk < n:
                    slope -= best_h[ikk]
                base = corr[top_idx[ti]] - best_h[i] * slope
                # Breakpoint between this and another lag
                for tj in range(ti+1, n_top):
                    kj = top_idx[tj] - (n - 1)
                    ikj = i - kj
                    ikkj = i + kj
                    slopej = 0.0
                    if 0 <= ikj < n:
                        slopej += (1.0 - best_h[ikj])
                    if 0 <= ikkj < n:
                        slopej -= best_h[ikkj]
                    basej = corr[top_idx[tj]] - best_h[i] * slopej
                    # base + t*slope = basej + t*slopej
                    # t = (basej - base) / (slope - slopej)
                    ds = slope - slopej
                    if abs(ds) > 1e-12:
                        t_bp = (basej - base) / ds
                        if 0 < t_bp < 1:
                            candidates.append(t_bp)
            # Evaluate max at each candidate
            best_t = best_h[i]
            best_val = np.inf
            for t_c in candidates:
                t_c = max(0.0, min(1.0, t_c))
                # Compute max corr at this t
                mx = 0.0
                for ti in range(n_top):
                    k = top_idx[ti] - (n - 1)
                    ik = i - k
                    ikk = i + k
                    slope = 0.0
                    if 0 <= ik < n:
                        slope += (1.0 - best_h[ik])
                    if 0 <= ikk < n:
                        slope -= best_h[ikk]
                    base = corr[top_idx[ti]] - best_h[i] * slope
                    val = base + t_c * slope
                    if val > mx:
                        mx = val
                # Also need to check that no other lag exceeds mx
                # (this is approximate - the top lags should dominate)
                if mx < best_val:
                    best_val = mx
                    best_t = t_c
            if abs(best_t - best_h[i]) > 1e-10:
                best_h[i] = best_t
                best_h = project_h(best_h)
                c5_try = compute_c5(best_h)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5
```
