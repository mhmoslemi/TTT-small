import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

_T = [1.0]
_CB = [np.zeros(1)]
_CC = [np.zeros(1)]

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(x):
    T = _T[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def _sgrad(x):
    T = _T[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=300):
    _T[0] = T
    try:
        r = minimize(h, _sobj, _sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12})
        return proj_h(r.x, n)
    except Exception:
        return h

def fast_coord_desc(h, n, t_end, sweeps=100):

    best_c5 = compute_c5(best)
    dx = 2.0 / n
    mf = 2 * n - 1
    for sweep in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        cur_max = corr.max()
        order = np.random.permutation(n)
        for i in order:
            old = best[i]
            dk = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    dk[k] += (1.0 - best[j2])
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    dk[k] -= best[j1]
            lo = 0.0 - old
            hi = 1.0 - old
            a, b = lo, hi
            for _ in range(35):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(corr + m1 * dk)

                    a = m1
