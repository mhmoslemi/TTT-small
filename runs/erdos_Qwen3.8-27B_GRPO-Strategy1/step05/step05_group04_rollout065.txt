                    lin_j = np.empty(2 * n - 1)
                    lin_j[:n-1] = c_lin_j[::-1]
                    lin_j[n-1] = 1.0 - 2.0 * old_j
                    lin_j[n:] = c_lin_j
                    lin_tot = lin + lin_j
                    for delta in [0.01, -0.01, 0.005, -0.005]:
                        v_i = best_h[i] + delta
                        v_j = best_h[j] - delta
                        if 0 <= v_i <= 1 and 0 <= v_j <= 1:
                            d_i = v_i - best_h[i]
                            d_j = v_j - best_h[j]
                            cn = corr + d_i * lin + d_j * lin_j
                            cn[n-1] -= d_i * d_i + d_j * d_j
                            fm = float(np.max(cn) * dx)
                            if fm < best_c5 - 1e-14:
                                best_c5 = fm
                                corr = cn
                                best_h[i] = v_i
                                best_h[j] = v_j
                                improved = True
                                break
        if not improved:
            break
    return best_h, best_c5

def subgrad_descent(h, n, t_end, max_iters=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    mom = np.zeros(n)
    beta = 0.6
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = it / max(max_iters, 1)
        T = 80.0 * (50.0 ** frac)
        dx = 2.0 / n
        corr = np.correlate(h_c, 1.0 - h_c, mode='full')
        m = np.max(corr)
        exponents = np.clip(T * (corr - m), -50.0, 50.0)
        p = np.exp(exponents)
        p /= p.sum()
        grad = np.zeros(n)
        for m_idx in range(2 * n - 1):
            w = p[m_idx]
            if w < 1e-14:
                continue
            s = m_idx - (n - 1)
            if s > 0:
                grad[s:] += w * (1.0 - h_c[:n - s])
                grad[:n - s] -= w * h_c[s:]
            elif s < 0:
                a = -s
                grad[:n - a] += w * (1.0 - h_c[a:])
                grad[a:] -= w * h_c[:n - a]
            else:
                grad += w * (1.0 - 2.0 * h_c)
        g = dx * grad
        mom = beta * mom + g
        gm = np.max(np.abs(mom))
        if gm < 1e-14:
            break
        alpha = 0.35 / gm
        h_new = project_h(h_c - alpha * mom, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c5

def basin_hopping(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.02 + 0.08 * cool
        mode = hop % 7
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.3
            pert = np.zeros(n)
            pert[mask] = step * 2.5 * rng.randn(max(1, mask.sum()))
        elif mode == 3:
            half = n // 2
            idx1 = rng.randint(0, half, size=max(4, half // 5))
            pert = np.zeros(n)
            vals = step * rng.randn(len(idx1))
            pert[idx1] = vals
            pert[idx1 + half] = -vals
        elif mode == 4:
            # Anti-symmetry push
            pert = 0.01 * (project_antisym(cur_h, n) - cur_h)
        elif mode == 5:
            # Block shift
            bs = rng.randint(5, n // 6)
            start = rng.randint(0, n - bs)
            pert = np.zeros(n)
            shift = step * rng.randn(bs)
            pert[start:start+bs] = shift
        else:
            # Mix of random and antisym
            pert = 0.5 * step * rng.randn(n) + 0.005 * (project_antisym(cur_h, n) - cur_h)
        h_pert = project_h(cur_h + pert, n)
        hop_time = max(1.5, 6.0 * cool)
        hop_end = min(t_end, time.time() + hop_time)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=60, seed=seed + hop * 7)
        temp = 0.003 * cool + 0.0003
        if c5_opt < cur_c5:
            accept = True
        elif c5_opt > cur_c5 and (c5_opt - cur_c5) < temp:
            accept = rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)
        else:
            accept = False
        if accept:
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent (40% budget)
    t1 = t0 + budget_s * 0.38
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=1200, seed=seed)

    # Phase 2: Subgradient (15% budget)
    if time.time() < t0 + budget_s * 0.52:
        h_g, c5_g = subgrad_descent(best_h, n, t0 + budget_s * 0.52, max_iters=2000, seed=seed+5)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.55, max_sweeps=400, seed=seed+6)

    # Phase 3: Basin hopping (32% budget)
    t3_end = t0 + budget_s * 0.85
    if time.time() < t3_end:
        best_h, best_c5 = basin_hopping(best_h, n, t3_end, n_hops=500, seed=seed + 20)

    # Phase 4: Final coordinate descent (12% budget)
    t4_end = t0 + budget_s * 0.96
    if time.time() < t4_end:
        best_h, best_c5 = coord_descent(best_h, n, t4_end, max_sweeps=600, seed=seed + 500)

    # Phase 5: Quick subgradient + coord
    t5_end = t0 + budget_s * 0.99
    if time.time() < t5_end:
        h_g2, c5_g2 = subgrad_descent(best_h, n, t0 + budget_s * 0.97, max_iters=500, seed=seed+800)
        if c5_g2 < best_c5:
            best_h, best_c5 = h_g2, c5_g2
        best_h, best_c5 = coord_descent(best_h, n, t5_end, max_sweeps=200, seed=seed+900)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
