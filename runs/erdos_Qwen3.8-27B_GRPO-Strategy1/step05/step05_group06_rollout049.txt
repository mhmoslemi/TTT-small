```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_and_grad(h, k_top=3, T_w=100.0):
    """Compute correlations and weighted gradient of top-k."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_full = 2 * n - 1
    
    # Top-k indices
    k = min(k_top, m_full)
    top_idx = np.argpartition(corr, -k)[-k:][::-1]
    top_vals = corr[top_idx]
    w = np.exp((top_vals - top_vals[0]) * T_w)
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        lg = idx - (n - 1)
        if lg >= 0:
            if lg < n:
                grad[:n-lg] += wi * (1.0 - h[lg:])
            grad[lg:] += -wi * h[:n-lg]
        else:
            p = -lg
            grad[p:] += wi * (1.0 - h[:n-p])
            grad[:n-p] += -wi * h[p:]
    
    return float(np.max(corr)), grad

def projected_momentum(h, T, k_top, t_end, max_iter=50000):
    """Projected subgradient with momentum and backtracking."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m = np.zeros(n)
    lr = 0.01
    beta = 0.85
    stuck = 0
    max_stuck = 200
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        _, grad = corr_and_grad(h, k_top, T)
        m = beta * m - lr * grad
        h_new = project_h(h + m)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-15:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            stuck = 0
            lr = min(lr * 1.05, 0.1)
        elif c5_new < best_c5 + 1e-12:
            # Accept but not best
            h = h_new
            stuck += 1
        else:
            # Reject, reduce lr
            lr *= 0.7
            m *= 0.5
            stuck += 1
        
        if stuck > max_stuck:
            # Restart with smaller momentum and perturbation
            m *= 0.1
            lr = 0.005
            h = project_h(h + 0.02 * np.random.randn(n))
            stuck = 0
            max_stuck = max(50, max_stuck // 2)
    
    return best_h, best_c5

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    dx = 2.0 / n
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    x0 = project_h(h0)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        Tc = T * (corr - mc)
        Tc -= np.max(Tc)
        exps = np.exp(Tc)
        ws = exps.sum()
        sm = mc + np.log(ws) / T
        w = exps / ws
        val = dx * sm + lam * (x.sum() - n/2.0)**2
        grad = np.zeros(n)
        for idx in range(m_full):
            lg = lag_arr[idx]
            wi = w[idx]
            if wi < 1e-15:
                continue
            if lg >= 0:
                if lg < n:
                    grad[:n-lg] += wi * (1.0 - x[lg:])
                grad[lg:] += -wi * x[:n-lg]
            else:
                p = -lg
                grad[p:] += wi * (1.0 - x[:n-p])
                grad[:n-p] += -wi * x[p:]
        grad = dx * grad + 2.0 * lam * (x.sum() - n/2.0)
        return val, grad
    
    try:
        res = minimize(lambda x: obj_grad(x)[0], x0,
                       jac=lambda x: obj_grad(x)[1],
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_h(res.x)
    except:
        return x0

def coord_descent(h, t_end, max_sweeps=200):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h0r = best_h[i]
            best_h[i] = 0.0
            cb = np.correlate(best_h, 1.0 - best_h, mode='full')
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i-lg])
                    if i+lg < n:
                        lin[idx] += -best_h[i+lg]
                else:
                    p = -lg
                    if i+p < n:
                        lin[idx] += (1.0 - best_h[i+p])
                    if i >= p:
                        lin[idx] += -best_h[i-p]
            
            def f1d(x):
                return np.max(cb + x*lin + x*x*quad)
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            hi_new = res.x
            if abs(hi_new - h0r) < 1e-13:
                continue
            best_h[i] = hi_new
            best_h = project_h(best_h)
            c5t = compute_c5(best_h)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                improved = True
            else:
                best_h[i] = h0r
                best_h = project_h(best_h)
        if not improved:
            break
    return best_h, best_c5
```

```python
import time

def compute_c5(h):
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def topk_grad(h, k, T_w):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_full = 2 * n - 1
    kk = min(k, m_full)
    top = np.argpartition(-corr, kk - 1)[:kk]
    tv = corr[top]
    w = np.exp((tv - tv[0]) * T_w)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, idx in zip(w, top):
        lg = idx - (n - 1)
        if lg >= 0:
            if lg < n:
                grad[:n - lg] += wi * (1.0 - h[lg:])
            grad[lg:] += -wi * h[:n - lg]
        else:
            p = -lg
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] += -wi * h[p:]
    return float(np.max(corr)), grad

def lbfgs_smooth(h0, T, lam, maxiter, ftol):
    n = len(h0)
    dx = 2.0 / n
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    x0 = project_h(h0)

    def og(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        Tc = T * (corr - mc)
        Tc -= np.max(Tc)
        exps = np.exp(Tc)
        ws = exps.sum()
        w = exps / ws
        val = dx * (mc + np.log(ws) / T) + lam * (x.sum() - n / 2.0) ** 2
        grad = np.zeros(n)
        for idx in range(m_full):
            lg = lag_arr[idx]
            wi = w[idx]
            if wi < 1e-15:
                continue
            if lg >= 0:
                if lg < n:
                    grad[:n - lg] += wi * (1.0 - x[lg:])
                grad[lg:] += -wi * x[:n - lg]
            else:
                p = -lg
                grad[p:] += wi * (1.0 - x[:n - p])
                grad[:n - p] += -wi * x[p:]
        grad = dx * grad + 2.0 * lam * (x.sum() - n / 2.0)
        return val, grad

    try:
        res = minimize(og, x0, jac=None, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_h(res.x)
    except Exception:
        return x0

def coord_sweep(h, order):
    n = len(h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    changed = False
    for i in order:

        h[i] = 0.0
        cb = np.correlate(h, 1.0 - h, mode='full')
        lin = np.zeros(m_full)
        quad = np.zeros(m_full)
        for idx in range(m_full):
            lg = lag_arr[idx]
            if lg == 0:
                lin[idx] = 1.0
                quad[idx] = -1.0
            elif lg > 0:

                if i + lg < n:
                    lin[idx] += -h[i + lg]
            else:

                if i + p < n:

                if i >= p:
                    lin[idx] += -h[i - p]
        best_x = old
        best_f = np.max(cb + old * lin + old * old * quad)

                else:

        h[i] = best_x

        if work_c5 < best_c5:
            best_c5, best_h = work_c5, work_h

    # Phase 5: Final L-BFGS-B with very high T (10%)
    p5 = t0 + budget_s * 0.93
    for T_w in [5000, 30000, 100000]:
        if time.time() > p5:
            break
        hc = lbfgs_smooth(best_h, T_w, 500.0, 2000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 6: Final coordinate descent polish (7%)
    p6 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent(best_h, p6, 200)
    if time.time() < t0 + budget_s * 0.995:
        hc = lbfgs_smooth(best_h, 200000, 500.0, 1000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
