    """Fast subgradient descent on max correlation."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    it = 0
    while time.time() < t_end:
        corr = get_corr(h)
        max_val = np.max(corr)
        # Use top-5 correlations for subgradient
        top_k = 5
        idx = np.argsort(corr)[-top_k:]
        wts = np.exp(20.0 * (corr[idx] - max_val))
        wts /= wts.sum()
        
        grad = np.zeros(n)
        for wi, di in zip(wts, idx):
            d = di - (n - 1)
            if d >= 0:
                if d < n:
                    grad[:n-d] += wi * (1.0 - h[d:])
                    grad[d:] += -wi * h[:n-d]
            else:
                d2 = -d
                if d2 < n:
                    grad[d2:] += wi * (1.0 - h[:n-d2])
                    grad[:n-d2] += -wi * h[d2:]
        
        lr = lr0 / (1.0 + 0.001 * it)
        h_new = project_h(h - lr * grad)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Reset lr on improvement
            lr0 = max(lr * 1.1, 0.001)
        else:
            lr0 = max(lr * 0.95, 0.0005)
        h = h_new
        it += 1
    return best_h, best_c5

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.86, max_sweeps=300, niter=95)

    # Phase 6: Targeted coordinate descent on bottleneck (4% budget)
    best_h, best_c5 = coord_descent_top(best_h, n, t0 + budget_s * 0.92, niter=120)

    # Phase 7: Final SLSQP polish (3% budget)
    if elapsed() < budget_s * 0.95:
        for T in [100000, 500000, 2000000]:
            if elapsed() > budget_s * 0.97:
                break
            hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Final: one more quick coord pass if time
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=50, niter=100)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
