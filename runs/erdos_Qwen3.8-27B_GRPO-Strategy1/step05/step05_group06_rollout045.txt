def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(corr, T):
    """Compute smooth max and its gradient w.r.t. corr."""
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc = Tc - np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    return sm, w

def optimize_lbfgs(h0, T, lam, maxiter, ftol, t_end):
    n = len(h0)
    dx = 2.0 / n
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    
    x0 = project_h(h0.copy())
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm, w = smooth_max(corr, T)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_weights(corr, T):
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    s = exps.sum()
    w = exps / s
    sm = mc + np.log(s) / T
    return sm, w

def obj_and_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm, w = smooth_max_and_weights(corr, T)
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lg = lag_arr[idx]
        if 0 <= lg < n:
            grad[lg:] += wi * (1.0 - h[:n-lg])
            grad[:n-lg] -= wi * h[lg:]
        elif -n < lg < 0:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs(h0, T, lam, maxiter, ftol, t_end):
    n = len(h0)
    x0 = project_h(h0.copy())
    def og(x):
        return obj_and_grad(x, T, lam)
    tchk = time.time()
    try:
        res = minimize(og, x0, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-11,
                                'maxfun': maxiter * 3})
        return project_h(res.x)
    except Exception:
        return x0

def coord_descend(h, t_end, max_sweeps=400):
    n = len(h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h0 = best_h.copy()
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            def f1d(x):
                return np.max(cb + x * lin + x * x * quad)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            if abs(res.x - best_h[i]) < 1e-14:
                continue
            ht = best_h.copy()
            ht[i] = res.x
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def structured_perturb(h, n, scale, rng):
    """Add multi-scale structured perturbation targeting specific frequencies."""
    x = np.arange(n) / n
    h_p = h.copy()
    # Low frequency perturbation
    h_p += scale * 0.5 * np.sin(2 * np.pi * rng.choice([2, 3, 4]) * x + rng.uniform(0, 2*np.pi))

    # Small random noise
    h_p += scale * 0.1 * rng.standard_normal(n)
    return project_h(h_p)
