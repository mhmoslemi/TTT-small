    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    
    ones = np.where(cur_h == 1.0)[0]
    zeros = np.where(cur_h == 0.0)[0]
    
    t_start = time.time()
    T_init = 0.01
    T_min = 0.0001
    it = 0
    
    while time.time() < t_end and it < n_iters_max:
        it += 1
        frac = min(1.0, it / (it + 10000))  # exponential cooling
        T = T_init * (T_min / T_init) ** frac
        
        # Pick a random 1 and a random 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)
        
        if new_c5 < cur_c5 or (new_c5 > cur_c5 and rng.rand() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            # Update ones/zeros
            np.delete(ones, np.where(ones == i)[0][0])
            np.append(ones, j)
            np.delete(zeros, np.where(zeros == j)[0][0])
            np.append(zeros, i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()

    """Compute linear coefficients for changing h[i]."""
    lin = np.zeros(2*n - 1)
    old = h[i]
    lin[n-1] = 1.0 - 2.0 * old

            lin[n-1-s] += (1.0 - h[i-s])
        if i <= n-1-s:
            lin[n-1-s] -= h[i+s]
    return lin

            best_h[i] = new_val
            corr += d * lin
            corr[n-1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n-1] += d * d
        if not improved:
            break
    return best_h, best_c5

def binary_swap_sa(h, n, t_end, seed=42, max_iters=2000000):
    rng = np.random.RandomState(seed)
    cur = h.copy()
    cur = np.where(cur > 0.5, 1.0, 0.0)
    target = n // 2
    ones = int(cur.sum())
    if ones > target:
        idx = np.where(cur == 1)[0]
        rem = rng.choice(idx, ones - target, replace=False)
        cur[rem] = 0
    elif ones < target:
        idx = np.where(cur == 0)[0]
        add = rng.choice(idx, target - ones, replace=False)
        cur[add] = 1
    cur_c5 = compute_c5(cur)
    best_c5 = cur_c5
    best = cur.copy()
    T_init = 0.008
    T_min = 0.00005
    it = 0
    while time.time() < t_end and it < max_iters:
        it += 1
        frac = 1.0 - (1.0 - min(1.0, it / max_iters)) ** 1.5
        T = T_init * (T_min / T_init) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = compute_c5(cur)
        if new_c5 <= cur_c5 or rng.rand() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_c5

def basin_hopping(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.02 + 0.07 * cool
        mode = hop % 5
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 20)
            ph = rng.uniform(0, 2*np.pi)
            pert = step * np.sin(2*np.pi*f*np.arange(n)/float(n) + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.3
            pert = np.zeros(n)
            pert[mask] = step * 2.5 * rng.randn(int(mask.sum()))
        elif mode == 3:
            k = max(5, n // 6)
            idx = rng.choice(n, size=k, replace=False)
            pert = np.zeros(n)
            pert[idx] = step * rng.randn(k)
        else:
            a = rng.randint(0, n//2)
            b = rng.randint(a+1, n)
            pert = np.zeros(n)
            pert[a:b] = step * rng.randn(b-a)
        
        h_pert = project_h(cur_h + pert, n)
        hop_t = min(t_end, time.time() + 2.0 + 12.0 * cool)
        h_opt, c5_opt = fast_cd(h_pert, n, hop_t, max_sweeps=150, seed=seed+hop, maxiter_min=25)
        
        temp = 0.0025 * cool + 0.00015
        if c5_opt < cur_c5 or rng.rand() < np.exp(-(c5_opt - cur_c5) / max(temp, 1e-12)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_tm(160, seed=seed)
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: CD from initial point
    t1 = t0 + budget_s * 0.18
    best_h, best_c5 = fast_cd(best_h, n, t1, max_sweeps=4000, seed=seed, maxiter_min=35)
    
    # Phase 2: Multi-start with different initializations
    t2 = t0 + budget_s * 0.42
    if time.time() < t2:
        inits = [
            (make_tm(n, seed+7), seed+7),
            (make_random_bin(n, seed+13), seed+13),
            (make_blocks(n, seed+19), seed+19),
            (make_random_bin(n, seed+31), seed+31),
        ]
        per_start = max(5.0, (t2 - time.time()) / len(inits))
        for h_init, s in inits:
            if time.time() >= t2:
                break
            st = min(t2, time.time() + per_start)
            h_try, c5_try = fast_cd(h_init, n, st, max_sweeps=500, seed=s, maxiter_min=30)
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try
    
    # Phase 3: Basin hopping (main exploration)
    t3 = t0 + budget_s * 0.72
    if time.time() < t3:
        best_h, best_c5 = basin_hopping(best_h, n, t3, n_hops=350, seed=seed+50)
    
    # Phase 4: Final CD polish
    t4 = t0 + budget_s * 0.88
    best_h, best_c5 = fast_cd(best_h, n, t4, max_sweeps=5000, seed=seed+200, maxiter_min=40)
    
    # Phase 5: Short basin hop + binary SA + final CD
    t5 = t0 + budget_s * 0.96
    if time.time() < t5:
        best_h, best_c5 = basin_hopping(best_h, n, t5, n_hops=80, seed=seed+500)
    
    t_end_final = t0 + budget_s * 0.99
    best_h, best_c5 = fast_cd(best_h, n, t_end_final, max_sweeps=2000, seed=seed+800, maxiter_min=35)
    
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
