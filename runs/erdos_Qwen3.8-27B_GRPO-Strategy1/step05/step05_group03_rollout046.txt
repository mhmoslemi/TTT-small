import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_step(h, i, n):
    """Compute optimal h[i] given all other h[j] fixed."""
    # Compute base correlation with h[i] = 0
    save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = save
    
    # Coefficient: how corr changes when h[i] goes from 0 to 1
    # For lag k: contribution is (1-h[i+k]) if i+k valid, minus h[i-k] if i-k valid
    coefs = np.zeros(2 * n - 1)
    # First factor: h[i] * (1-h[i+k]) → for lag k, adds (1-h[i+k]) if i+k < n
    for k in range(0, n - i):
        coefs[k + n - 1] += (1.0 - h[i + k])
    # Second factor: h[i-k] * (1-h[i]) → for lag -k (i.e., h[j]*(1-h[j+k]) where j+k=i)
    # h[j] * (1-h[i]) where j = i-k, so lag = k, and the term is h[i-k]*(1-h[i])
    # When h[i] increases by 1, this term decreases by h[i-k]
    for k in range(1, min(i + 1, n)):
        coefs[k + n - 1] -= h[i - k]
    
    # So corr[lag] = corr_base[lag] + v * coefs[lag] where v = h[i]
    # Minimize max_lag (corr_base[lag] + v * coefs[lag]) over v in [0,1]
    
    # Quick check: if already at boundary
    f_at_0 = np.max(corr_base)
    f_at_1 = np.max(corr_base + coefs)
    
    res = minimize_scalar(lambda v: np.max(corr_base + v * coefs),
                          bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 50})

from scipy.optimize import minimize_scalar
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=1000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    nl = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        any_imp = False
        for i in range(n):
            save = best_h[i]
            best_h[i] = 0.0
            cb = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_h[i] = save
            cf = np.zeros(nl)
            lo = max(0, n - 1 - i)
            hi = min(nl - 1, n - 1 + i)
            ms = np.arange(lo, hi + 1)
            j1 = i - ms + n - 1
            v1 = (j1 >= 0) & (j1 < n)
            j2 = i + ms - n + 1
            v2 = (j2 >= 0) & (j2 < n)
            cf[ms[v1]] += (1.0 - best_h[j1[v1]])
            cf[ms[v2]] -= best_h[j2[v2]]
            res = minimize_scalar(lambda v: np.max(cb + v * cf),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 40})
            nv = res.x
            if abs(nv - save) > 1e-12:
                best_h[i] = nv
                best_h = project_h(best_h, n)
                c5t = compute_c5(best_h)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t
                    any_imp = True

                    best_h[i] = save
                    best_h = project_h(best_h, n)
        if not any_imp:
            break
    return best_h, best_c5

def subgrad_step(h, n, alpha, k_top=8):
    corr = np.correlate(h, 1.0 - h, mode='full')
    nl = 2 * n - 1
    idx = np.argsort(corr)[-k_top:]
    grad = np.zeros(n)
    for ki in idx:
        for i in range(n):
            j1 = i - ki + n - 1
            j2 = i + ki - n + 1
            if 0 <= j1 < n:
                grad[i] += (1.0 - h[j1])
            if 0 <= j2 < n:
                grad[i] -= h[j2]
    grad /= k_top
    return project_h(h - alpha * grad, n)

def sturmian(n, frac):
    a = np.zeros(n)
    for i in range(n):
        a[i] = 1.0 if ((i + 1) * frac) % 1.0 > (i * frac) % 1.0 else 0.0
    return a

def fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for _ in range(rng.randint(2, 5)):
        freq = rng.randint(2, n // 3)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    h = 0.5 + h
    return project_h(h, n)

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
