def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_i(h, n, i, dx):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    def obj_1d(hi):
        return float(np.max(corr_base + hi * coefs) * dx)
    try:
        res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-11})
        return float(res.x)
    except Exception:
        return h[i]

def subgrad_compute(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vmax = np.max(corr)
    vmin = np.min(corr)
    spread = vmax - vmin
    if spread < 1e-12:
        return np.zeros(n)
    threshold = vmax - 0.05 * spread
    active_idx = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    aff = np.arange(n)
    for m in active_idx:
        tau = m - (n - 1)
        vp = (aff - tau >= 0) & (aff - tau < n)
        vm = (aff + tau >= 0) & (aff + tau < n)
        grad[vp] += (1.0 - h[aff[vp] - tau])
        grad[vm] += -h[aff[vm] + tau]
    return grad / max(1, len(active_idx))

def smooth_logsumexp(corr, T):
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return m + np.log(np.sum(np.exp(shifted))) / T

def slsqp_solve(h, n, T, maxiter=800):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(smooth_logsumexp(corr, T) * dx)
    def con_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def make_perturbation(n, cool, rng):
    s = rng.randint(0, 6)
    if s == 0:
        bs = rng.randint(3, max(4, n // 10))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st+bs] = (0.03 + 0.10*cool) * rng.standard_normal(bs)
    elif s == 1:
        freq = rng.randint(1, max(2, n // 25))
        ph = rng.uniform(0, 2*np.pi)
        p = (0.02 + 0.07*cool) * np.sin(2*np.pi*freq*np.arange(n)/n + ph)
    elif s == 2:
        np_ = rng.randint(3, max(4, n // 10))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.06 + 0.20*cool) * rng.standard_normal(np_)
    elif s == 3:
        p = (0.015 + 0.05*cool) * rng.standard_normal(n)
    elif s == 4:
        b1 = rng.randint(0, n-4)
        b2 = rng.randint(0, n-4)
        if b2 == b1: b2 = (b1+7) % n
        pl = rng.randint(2, min(4, n-b1, n-b2))
        p = np.zeros(n)
        p[b1:b1+pl] = rng.uniform(-0.4, 0.4) * cool
        p[b2:b2+pl] = -p[b1:b1+pl]
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.08*cool * rng.standard_normal(half)
        p[half:] = -0.08*cool * rng.standard_normal(n-half)
    return p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 1)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Adam subgradient (15% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    b1, b2, eps_a = 0.85, 0.999, 1e-8
    lr_init = 0.03
    n_steps = 400
    t_phase1_end = t0 + budget_s * 0.15
    for t in range(n_steps):
        if time.time() > t_phase1_end:
            break
        grad = subgrad_compute(work_h, n)
        lr_t = lr_init / (1.0 + 0.005*t)
        m_a = b1*m_a + (1-b1)*grad
        v_a = b2*v_a + (1-b2)*grad*grad
        m_hat = m_a / (1 - b1**(t+1))
        v_hat = v_a / (1 - b2**(t+1))
        h_try = project_h(work_h - lr_t * m_hat / (np.sqrt(v_hat) + eps_a))
        c5_try = compute_c5(h_try)
        if c5_try < work_c5:
            work_c5 = c5_try
            work_h = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
        elif t < 60:
            lr_init *= 0.97

    # Phase 2: Intensive coordinate descent (25% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    dx = 2.0 / n
    t_ph2_end = t0 + budget_s * 0.40
    no_improve = 0
    while time.time() < t_ph2_end:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_ph2_end:
                break
            hi_new = coord_opt_i(work_h, n, i, dx)
            if abs(hi_new - work_h[i]) < 5e-11:
                continue
            work_h[i] = hi_new
            work_h = project_h(work_h)
            c5_try = compute_c5(work_h)
            if c5_try < work_c5 - 1e-14:
                work_c5 = c5_try
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if work_c5 < best_c5:
            best_c5 = work_c5
            best_h = work_h.copy()

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 3: Basin hopping (25% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    t_ph3_end = t0 + budget_s * 0.65
    n_iter = 0
    max_iters = 80
    while n_iter < max_iters and time.time() < t_ph3_end:
        n_iter += 1
        cool = max(0.0, 1.0 - n_iter / float(max_iters))
        pert = make_perturbation(n, cool, rng)
        h_pert = project_h(work_h + pert)
        # Quick SLSQP local optimization
        T_bh = 500 + 5000 * cool
        h_local = slsqp_solve(h_pert, n, T_bh, maxiter=400)
        c5_local = compute_c5(h_local)
        # Quick coord descent on the local result
        for _ in range(4):
            ii = rng.randint(n)
            hi_new = coord_opt_i(h_local, n, ii, dx)
            h_local[ii] = hi_new
            h_local = project_h(h_local)
        c5_local = compute_c5(h_local)
        temp = 0.0012 * cool + 0.0001
        accept = c5_local < work_c5 or (rng.rand() < np.exp(-(c5_local - work_c5) / max(temp, 1e-10)))
        if accept:
            work_h = h_local
            work_c5 = c5_local
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_local.copy()

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 4: SLSQP with increasing T (10% budget)
    t_ph4_end = t0 + budget_s * 0.75
    for T in [2000, 10000, 50000, 200000]:
        if time.time() > t_ph4_end:
            break
        h_try = slsqp_solve(best_h, n, T, maxiter=1500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 5: Final multi-sweep coordinate descent (15% budget)
    t_ph5_end = t0 + budget_s * 0.95
    for sweep in range(6):
        if time.time() > t_ph5_end:
            break
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_ph5_end:
                break
            hi_new = coord_opt_i(best_h, n, i, dx)
            if abs(hi_new - best_h[i]) < 5e-11:
                continue
            best_h[i] = hi_new
            best_h = project_h(best_h)

    # Phase 6: Final SLSQP polish
    if time.time() < t0 + budget_s * 0.97:
        h_try = slsqp_solve(best_h, n, 500000, maxiter=1000)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # One more coord pass
    if time.time() < t0 + budget_s * 0.99:
        for i in range(n):
            hi_new = coord_opt_i(best_h, n, i, dx)
            if abs(hi_new - best_h[i]) < 5e-11:
                continue
            best_h[i] = hi_new
            best_h = project_h(best_h)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
