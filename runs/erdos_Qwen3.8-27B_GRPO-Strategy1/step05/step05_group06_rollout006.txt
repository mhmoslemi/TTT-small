    n = len(h)

    lag = np.arange(m) - (n - 1)  # lag[k] = k - (n-1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    """Project h to [0,1]^n with sum = n/2."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, t_end, max_sweeps):
    """Fast coordinate descent exploiting linearity of C(k) in h[i]."""
    n = len(h)
    m = 2 * n - 1
    lag = np.arange(m) - (n - 1)
    best_h = h.copy()
    C = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    best_c5 = float(np.max(C) * 2.0 / n)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            ti = best_h[i]
            i_plus = i + lag
            i_minus = i - lag
            v1 = (i_plus >= 0) & (i_plus < n)
            v2 = (i_minus >= 0) & (i_minus < n)
            a = np.zeros(m)
            a[v1] = 1.0 - best_h[i_plus[v1]]
            a[v2] -= best_h[i_minus[v2]]
            
            # Minimize max_k(C[k] + (t-ti)*a[k]) over t in [0,1]
            # This is convex in t, use bounded minimization
            # Fast evaluation: just numpy ops
            res = minimize_scalar(lambda t: np.max(C + (t - ti) * a),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            hi_new = float(res.x)
            
            if abs(hi_new - ti) > 1e-11:
                delta = hi_new - ti
                C += delta * a
                best_h[i] = hi_new
                c5_new = float(np.max(C) * 2.0 / n)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        
        if not improved:
            break
    
    # Final projection (should already be feasible but just in case)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def smooth_max_obj(h, T, lam):
    """Smooth max approximation of C5 with sum constraint penalty."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    return val

def smooth_max_grad(h, T, lam):
    """Gradient of smooth_max_obj."""
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    lag_arr = np.arange(m) - (n - 1)
    for idx in range(m):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        elif lg > 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n-p]
    
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return grad

def lbfgs_opt(h0, T, lam, maxiter, t_end):
    n = len(h0)
    x0 = project_h(h0)
    
    def obj(x):
        return smooth_max_obj(x, T, lam)

        return smooth_max_grad(x, T, lam)

                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-12,
                                'maxls': 50})
        return project_h(res.x)
    except Exception:
        return x0

                delta = hi_new - ti
                C += delta * a
                best_h[i] = hi_new
                c5_new = float(np.max(C) * 2.0 / n)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5


def smooth_max_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    sm = mc + np.log(exps.sum()) / T
    return dx * sm + lam * (h.sum() - n / 2.0) ** 2


def smooth_max_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    grad = np.zeros(n)
    lag_arr = np.arange(m) - (n - 1)
    for idx in range(m):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        elif lg > 0 and lg < n:
            grad[lg:] += wi * (1.0 - h[:n - lg])
            grad[:n - lg] += -wi * h[lg:]
        elif lg < 0:
            p = -lg
            if p < n:
                grad[:n - p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n - p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return grad


def _lbfgs_obj_wrapper(x, args):
    T, lam = args
    return smooth_max_obj(x, T, lam)


def _lbfgs_grad_wrapper(x, args):
    T, lam = args
    return smooth_max_grad(x, T, lam)


def lbfgs_opt(h0, T, lam, maxiter):
    n = len(h0)
    x0 = project_h(h0)
    args = (T, lam)
    try:
        res = minimize(_lbfgs_obj_wrapper, x0, args=args,
                       jac=_lbfgs_grad_wrapper, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-17,
                                'gtol': 1e-12, 'maxls': 50})
        return project_h(res.x)
    except Exception:
        return x0


def snap_binary(h):
    """Snap values near 0 or 1, then project."""
    n = len(h)
    hs = h.copy()
    mask = (hs < 0.05) | (hs > 0.95)
    hs[mask] = np.where(hs[mask] < 0.5, 0.0, 1.0)
    return project_h(hs)


def perturbed_restart(best_h, scale, n):
    return project_h(best_h + scale * np.random.randn(n))


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end_global = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS warm-up (10%)
    p1 = t0 + budget_s * 0.10
    for T in [50, 200, 800, 3000]:
        if time.time() > p1:
            break
        hc = lbfgs_opt(best_h, T, 500.0, 2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Intensive fast coordinate descent (30%)
    p2 = t0 + budget_s * 0.40
    best_h, best_c5 = fast_coord_descent(best_h, p2, 20000)

    # Phase 3: L-BFGS high-T refinement (10%)
    p3 = t0 + budget_s * 0.50
    for T in [5000, 20000, 80000]:
        if time.time() > p3:
            break
        hc = lbfgs_opt(best_h, T, 500.0, 4000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: More coordinate descent (25%)
    p4 = t0 + budget_s * 0.75
    best_h, best_c5 = fast_coord_descent(best_h, p4, 30000)

    # Phase 5: Basin hopping with perturbations (15%)
    p5 = t0 + budget_s * 0.90
    work_h = best_h.copy()
    work_c5 = best_c5
    n_hops = 20
    for it in range(n_hops):
        if time.time() > p5:
            break
        cool = 1.0 - 0.8 * (it / float(n_hops))
        sc = 0.005 + 0.04 * cool
        h_pert = perturbed_restart(work_h, sc, n)
        hc = lbfgs_opt(h_pert, int(3000 * cool + 500), 300.0, 1500)
        hc, c5_cd = fast_coord_descent(hc, min(time.time() + 8, p5), 2000)
        c5v = compute_c5(hc)
        temp = 0.0003 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 6: Final polish (10%)
    p6 = t_end_global - 3
    if time.time() < p6 - 30:
        hc = lbfgs_opt(best_h, 300000, 800.0, 5000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if time.time() < p6 - 5:
        best_h, best_c5 = fast_coord_descent(best_h, p6 - 5, 10000)
    # Try snapping
    hs = snap_binary(best_h)
    c5s = compute_c5(hs)
    if c5s < best_c5:
        best_h = hs
        best_c5 = c5s
        if time.time() < p6:
            best_h, best_c5 = fast_coord_descent(best_h, p6, 5000)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
