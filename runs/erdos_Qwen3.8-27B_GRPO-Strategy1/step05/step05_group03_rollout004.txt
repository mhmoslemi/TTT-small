import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _cons_sum(x):
    return x.sum() - len(x) / 2.0

def slsqp_opt(h, n, T, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c = corr - np.max(corr)
        return (np.max(corr) + np.log(np.sum(np.exp(T * c))) / T) * dx
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _cons_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

def coord_opt_1d(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    lags = np.arange(-(n - 1), n)
    coefs = np.zeros(2 * n - 1)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[j1[v1]])
    coefs[v2] += (-h[j2[v2]])
    h[i] = old
    def f(t):
        return np.max(corr_base + t * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 100})
    return res.x

                    best_h = h_try
                    improved = True

            break

    top_indices = np.argsort(corr)[-top_k:]

def pair_coord_opt(h, n, i, j):
    old_i, old_j = h[i], h[j]
    h[i] = 0.0
    h[j] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    lags = np.arange(-(n - 1), n)

        coefs = np.zeros(2 * n - 1)

        j2 = idx - lags

        coefs[v2] += (-h[j2[v2]])
