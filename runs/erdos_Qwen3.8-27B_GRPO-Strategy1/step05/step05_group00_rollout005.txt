import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binary_search_full(h, N, deadline, max_sweeps=500, sample_n=None, anneal_T=0.0):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        ones_set = set(np.where(h == 1)[0].tolist())
        if n1 > nh:
            to_remove = sorted(ones_set)[nh:]
            for i in to_remove:
                h[i] = 0
        else:
            zeros_set = list(np.where(h == 0)[0])
            to_add = zeros_set[:nh - n1]
            for i in to_add:
                h[i] = 1
    best_c = float(np.max(fft_corr(h)))
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        p_list = ones if sample_n is None else (ones if len(ones) <= sample_n else np.random.choice(ones, sample_n, replace=False))
        best_delta = 0
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            delta = float(npk[mi]) - P
            if delta < best_delta:
                best_delta = delta
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if anneal_T > 0 and np.random.random() < anneal_T:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return h, best_c

def smooth_opt(h, N, deadline, T0=5.0, T1=50.0, lr0=0.05, iters_per_check=50):
    """Optimize using smooth max (log-sum-exp) with Adam-like optimizer"""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    
    # Adam state
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    T = T0
    lr = lr0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    while time.time() < deadline:
        t += 1
        # Increase temperature gradually
        frac = min(1.0, t / 500.0)
        T_cur = T0 + (T1 - T0) * frac
        
        c = fft_corr(h)
        c_safe = c - c.max()
        exp_c = np.exp(T_cur * c_safe)
        w = exp_c / exp_c.sum()
        
        # Gradient: -D.T @ w
        D = compute_D(h, N)
        grad = -(D.T @ w)
        # Scale by dx
        grad *= dx
        
        # Project gradient to account for sum constraint (remove mean)
        grad = grad - grad.mean()
        
        # Adam update
        bc1 = 1 - beta1**t
        bc2 = 1 - beta2**t
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / bc1
        v_hat = v / bc2
        h_new = h + lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h_new, target)
        
        # Check improvement
        c5_new = compute_c5(h_new, N)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Decay learning rate
        if t % 200 == 0:
            lr *= 0.95
        
        # Periodic binary projection
        if t % iters_per_check == 0:
            h_bin = (h > 0.5).astype(float)
            n1 = int(h_bin.sum())
            if n1 != N // 2:
                if n1 > N // 2:
                    idx = np.where(h_bin == 1)[0]
                    idx = idx[np.argsort(h[idx])][-((n1 - N//2)):]
                    h_bin[idx] = 0
                else:
                    idx = np.where(h_bin == 0)[0]
                    idx = idx[np.argsort(-h[idx])][:(N//2 - n1)]
                    h_bin[idx] = 1
            h = 0.5 * h + 0.5 * h_bin
    
    return best_h, best_c5

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def bin_search(h, N, deadline, max_sweeps=600, sample_n=None, anneal=0.0):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        ones_i = np.where(h > 0.5)[0]
        zeros_i = np.where(h <= 0.5)[0]
        h[:] = 0.0
        if n1 > nh:
            h[ones_i[:nh]] = 1
        else:
            h[ones_i] = 1
            extra = zeros_i[:nh - n1]
            h[extra] = 1
    best_c = float(np.max(fft_corr(h)))
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            stall = 0
        else:
            stall += 1
        if stall > 80 and anneal == 0:
            break
        if stall > 40 and anneal > 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        if sample_n and len(ones) > sample_n:
            p_list = np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        bd, bpq = 0.0, None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            d = float(npk[mi]) - P
            if d < bd:
                bd, bpq = d, (int(p), int(zeros[mi]))
        if bpq is not None and bd < -1e-12:
            h[bpq[0]] = 0.0
            h[bpq[1]] = 1.0
        elif anneal > 0 and np.random.random() < anneal:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            stall = 0
        else:
            break
    return h, best_c

def topk_smooth(h, N, deadline, k=8, lr0=0.04):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr0
    while time.time() < deadline:
        t += 1
        c = fft_corr(h)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(c, -k)[-k:]
        D = compute_D(h, N)
        grad = -(D[:, top].sum(axis=1)) * dx
        grad -= grad.mean()
        bc1 = 1 - beta1 ** t
        bc2 = 1 - beta2 ** t
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        h_new = h + lr * (m / bc1) / (np.sqrt(v / bc2) + eps)
        h = project_sum(h_new, target)
        if t % 300 == 0:
            lr *= 0.93
            hb = (h > 0.5).astype(float)
            n1 = int(hb.sum())
            if n1 != N // 2:

                zeros_i = np.where(hb == 0)[0]

                else:
                    hb[zeros_i[np.argsort(-h[zeros_i])[:(N//2 - n1)]]] = 1
            h = 0.5 * h + 0.5 * hb
    return best_h, best_pk * dx
