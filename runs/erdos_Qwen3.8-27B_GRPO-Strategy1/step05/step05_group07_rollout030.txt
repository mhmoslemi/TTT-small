from scipy.optimize import minimize, minimize_scalar

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n), corr

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    for _ in range(3):
        if s > tgt:
            scale = tgt / s
            h = h * scale
        elif s < tgt:
            rm = 1.0 - h
            t = rm.sum()
            if t > 1e-14:
                h = h + rm * ((tgt - s) / t)
            else:
                h = h * (tgt / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - tgt) < 1e-12:
            break
    # Final fix
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        mask = (h > 0.02) & (h < 0.98)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += d / cnt
        else:
            h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgs_h(h, T, n, iters=400):
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(np.clip(T * (corr - m), -700, 700)))) / T) * 2.0 / n
    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        e = np.exp(np.clip(T * (corr - m), -700, 700))
        w = e / e.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            lo1 = max(0, n - 1 - k)
            hi1 = min(n, 2 * n - 1 - k)
            if lo1 < hi1:

            if lo2 < hi2:
                idx = np.arange(lo2, hi2) - k + n - 1
                g[lo2:hi2] -= wk * x[idx]
        return g * (2.0 / n)
    try:
        r = minimize(h, _obj, _grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-11})
        return proj_h(r.x, n)
    except:
        return h

    best_c5, _ = compute_c5(best)
    mf = 2 * n - 1

        if time.time() > t_end:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5

            # coefficient of h[i] in each c_k

        return (m + np.log(np.sum(np.exp(np.clip(T * (corr - m), -700, 700)))) / T) * 2.0 / n
    def grad(x):

        m = corr.max()

    best_c5, _ = compute_c5(best)
    mf = 2 * n - 1

            break

        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):

def active_grad_step(h, n, top_m, step):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[::-1][:top_m]

            j1 = i - k + (n - 1)
            if 0 <= j1 < n:
                g[i] -= h[j1]
    g /= (top_m * n)
    return proj_h(h - step * g, n)
