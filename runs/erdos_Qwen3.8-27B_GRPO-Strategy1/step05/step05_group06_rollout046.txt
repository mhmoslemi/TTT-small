import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_full = 2 * n - 1
    mc = np.max(corr)
    # Shifted log-sum-exp for stability
    shifted = T * (corr - mc)
    m_shifted = np.max(shifted)
    exps = np.exp(shifted - m_shifted)
    logsum = m_shifted + np.log(exps.sum())
    sm_val = mc + logsum / T
    val = dx * sm_val + lam * (h.sum() - n / 2.0) ** 2
    
    # Weights
    w = exps / exps.sum()
    
    # Gradient: dc_k/dh[i]
    # corr[k] = sum_j h[j+k_offset] * (1-h[j]) where k_offset = k - (n-1)
    # Let lag = k - (n-1), so corr[k] = sum_j h[j+lag] * (1-h[j]) for valid j
    # d corr[k] / d h[i]:
    #   - From h[j+lag] term: contributes (1-h[j]) when i = j+lag, i.e., j = i-lag
    #   - From (1-h[j]) term: contributes -h[j+lag] when i = j
    #   
    #   So d corr[k]/d h[i] = (1-h[i-lag]) * [i-lag valid] - h[i+lag] * [i+lag valid]
    #   (with appropriate bounds checking)
    
    grad = np.zeros(n)
    lag_arr = np.arange(m_full) - (n - 1)
    for idx in range(m_full):

        wi = w[idx]

            continue
        if lg >= 0:

            #                    - h[i+lg] for i+lg < n (from (1-h[i]) term)

                grad[lg:] += wi * (1.0 - h[:n-lg])
            # -h[i+lg] for i = 0, ..., n-1-lg
            if lg < n:
                grad[:n-lg] += -wi * h[lg:]
        else:

            # (1-h[i+p]) for i+p < n, i.e., i = 0, ..., n-1-p
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
            # -h[i-p] for i >= p
            if p < n:
                grad[p:] += -wi * h[:n-p]
    
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_full = 2 * n - 1
    mc = np.max(corr)
    shifted = T * (corr - mc)
    m_sh = np.max(shifted)
    exps = np.exp(shifted - m_sh)
    logsum = m_sh + np.log(exps.sum())
    val = dx * (mc + logsum / T) + lam * (h.sum() - n / 2.0) ** 2
    w = exps / exps.sum()
    grad = np.zeros(n)
    lag_arr = np.arange(m_full) - (n - 1)
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
