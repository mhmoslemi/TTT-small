import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T, flat_pen=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e8:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-10:]
        val += flat_pen * np.std(top)
    return val * dx

def slsqp_opt(h, T, n, maxiter=500, flat_pen=0.0):
    try:
        res = minimize(lambda x: smooth_obj(x, T, flat_pen), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent_agg(h, n, t_end, max_sweeps=200, n_top=30):
    """Aggressive coordinate descent on top shifts."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            # Compute coefficient for each top shift
            coefs = np.empty(n_top)
            for t in range(n_top):

                c = 0.0
                jp = i + k
                jm = i - k

def subgrad_step(h, alpha, n_topk, mom_v=None, beta=0.4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if mom_v is not None:
        grad = beta * mom_v + (1.0 - beta) * grad
    return project_h(h - alpha * grad), grad

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T, flat_pen=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e8:
        val = m
    else:
        shifted = T * (corr - m)
        shifted = np.clip(shifted, -50, 50)
        val = m + np.log(np.sum(np.exp(shifted))) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-10:]
        val += flat_pen * np.std(top)
    return val * dx

def slsqp_opt(h, T, n, maxiter=400, flat_pen=0.0):
    try:
        res = minimize(lambda x: smooth_obj(x, T, flat_pen), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent_full(h, n, t_end, max_sweeps=150):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    nshifts = 2 * n - 1
    all_shifts = np.arange(-(n-1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            base = np.empty(nshifts)
            coefs = np.empty(nshifts)
            hi = best_h[i]
            for idx, k in enumerate(all_shifts):
                c_rem = 0.0
                c_coef = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c_rem += hi * (1.0 - best_h[jp])
                    c_coef += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c_rem += best_h[jm] * (1.0 - hi)
                    c_coef -= best_h[jm]
                corr_val = np.correlate(best_h, 1.0 - best_h, mode='full')[idx] if False else 0.0
                base[idx] = 0.0  # placeholder, computed below
                coefs[idx] = c_coef
            # Compute corr once
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            for idx, k in enumerate(all_shifts):
                c_rem = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c_rem += best_h[i] * (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c_rem += best_h[jm] * (1.0 - best_h[i])
                base[idx] = corr[idx] - c_rem
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-14})
            t_new = res.x
            if abs(t_new - best_h[i]) > 1e-13:
                h_try = best_h.copy()
                h_try[i] = t_new
                c5_try = np.max(base + t_new * coefs) * dx
                if c5_try < best_c5 - 1e-15:
                    best_c5 = float(c5_try)
                    best_h = h_try
                    improved = True
        best_h = project_h(best_h)
        if not improved:
            break
    return best_h, best_c5

def subgrad_step(h, alpha, n_topk, mom_v=None, beta=0.4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if mom_v is not None:
        grad = beta * mom_v + (1.0 - beta) * grad
    return project_h(h - alpha * grad), grad

def zero_sum_pert(n, scale):
    p = np.random.randn(n)
    p -= p.mean()
    return p * scale

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.3:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-6:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n-k] += (work_h[k:n] - 0.5) * 0.4
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n+k] - 0.5) * 0.4
            pert = zero_sum_pert(n, 1.0) *pert.sum/2 + pert * scale * 0.4 * cool
        elif r < 0.5:
            f = np.random.randint(1, 12)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.6 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
            pert = zero_sum_pert(n, 1.0) * pert.sum / 2 + pert
        elif r < 0.7:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.0 * cool * np.random.randn(max(int(mask.sum()), 1))
            pert = zero_sum_pert(n, 1.0) * pert.sum / 2 + pert
        else:
            pert = zero_sum_pert(n, scale * 0.5 * cool)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp_opt(h_pert, T, n, 250, 0.02 * cool)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.00008
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_at_n(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # SLSQP warmup
    for T in [50, 300, 2000, 10000]:
        if t_end - time.time() < 15:

    # Subgradient + SLSQP
