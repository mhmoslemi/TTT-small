from scipy.optimize import minimize, minimize_scalar, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    h = np.clip(h, 0.0, 1.0)
    # Fix any small residual
    s2 = h.sum()
    if abs(s2 - target) > 1e-12:
        # Water-fill adjustment
        diff2 = target - s2
        if diff2 > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx] += diff2 / len(idx)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx] += diff2 / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

            # Vary h[i] and h[j] simultaneously by delta
            # lin[idx] = [(1-h[i+lag]) - h[i-lag] + (1-h[j+lag]) - h[j-lag]] (with bounds)
            lin = np.zeros(m)
            for idx in range(m):
                lg = idx - (n-1)
                c = 0.0
                if 0 <= i+lg < n: c += 1.0 - bh[i+lg]
                if 0 <= i-lg < n: c -= bh[i-lg]
                if 0 <= j+lg < n: c += 1.0 - bh[j+lg]
                if 0 <= j-lg < n: c -= bh[j-lg]
                lin[idx] = c
            # Quadratic: -2*delta^2 at lag 0 (since i ≠ j for even n)
            lo, hi = -bh[i], 1.0-bh[i]
            if hi - lo < 1e-12: continue
            def f1(d):
                nc = corr + d*lin
                nc[n-1] -= 2.0*d*d

                imp = True
        if imp: stall = 0

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-13:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-14:
            return h
        h += r * (d / t)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + d / s)
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    if abs(s2 - target) > 1e-10:
        d2 = target - s2
        idx = np.where((h < 1.0) if d2 > 0 else (h > 0.0))[0]
        if len(idx) > 0:
            h[idx] += d2 / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj(x, T, lam):
    n = len(x)
    dx = 2.0 / n
    corr = get_corr(x)
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    e = np.exp(Tc)
    w = e / e.sum()
    sm = mc + np.log(e.sum()) / T
    val = dx * sm + lam * (x.sum() - n / 2.0) ** 2
    g = np.zeros(n)
    m = 2 * n - 1
    lags = np.arange(m) - (n - 1)
    for idx in range(m):
        lg = lags[idx]
        wi = w[idx] * dx
        if wi < 1e-15:
            continue
        if lg == 0:
            g += wi * (1.0 - 2.0 * x)
        elif lg > 0:
            g[lg:] += wi * (1.0 - x[:n - lg])
            g[:n - lg] += -wi * x[lg:]
        else:
            p = -lg
            g[:n - p] += wi * (1.0 - x[p:])
            g[p:] += -wi * x[:n - p]
    g += 2.0 * lam * (x.sum() - n / 2.0)
    return val, g

def do_lbfgs(h0, T, lam, maxiter):
    n = len(h0)
    try:
        r = minimize(soft_obj, proj(h0).copy(), jac=True, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return proj(r.x)
    except Exception:
        return proj(h0)

def cd_fast(h, t_end, max_sw=10000, sym=False):
    n = len(h)
    dx = 2.0 / n
    bh = h.copy()
    if sym:
        bh = 0.5 * (bh + bh[::-1].copy())
        bh = proj(bh)
    corr = get_corr(bh)
    bc5 = float(np.max(corr) * dx)
    m = 2 * n - 1
    half = n // 2
    stall = 0
    for sw in range(max_sw):
        if time.time() > t_end:
            break
        imp = False
        if sym:
            coords = np.random.permutation(half)
        else:
            coords = np.random.permutation(n)
        for i in coords:
            j = n - 1 - i if sym else -1
            lin = np.zeros(m)
            for idx in range(m):
                lg = idx - (n - 1)
                c = 0.0
                if 0 <= i + lg < n:
                    c += 1.0 - bh[i + lg]
                if 0 <= i - lg < n:
                    c -= bh[i - lg]
                if sym and j != i:
                    if 0 <= j + lg < n:
                        c += 1.0 - bh[j + lg]
                    if 0 <= j - lg < n:
                        c -= bh[j - lg]
                lin[idx] = c
            lo, hi = -bh[i], 1.0 - bh[i]
            if hi - lo < 1e-12:
                continue
            q = 2.0 if (sym and j != i) else 1.0
            def f1(d):
                nc = corr + d * lin
                nc[n - 1] -= q * d * d
                return np.max(nc)
            r = minimize_scalar(f1, bounds=(lo, hi), method='bounded',
                                options={'xatol': 1e-9})
            nv = bh[i] + r.x
            if abs(nv - bh[i]) < 1e-11:
                continue
            bh[i] = nv
            if sym and j != i:
                bh[j] = nv
            corr = get_corr(bh)
            nc5 = float(np.max(corr) * dx)
            if nc5 < bc5 - 1e-14:
                bc5 = nc5
                imp = True
        if imp:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj(best_h)
    best_c5 = float(np.max(get_corr(best_h)) * 2.0 / n)

    # Phase 1: L-BFGS warm-up (12%)
    t1 = t0 + budget_s * 0.12
    for T in [80, 400, 2000]:
        if time.time() > t1:

        c5v = float(np.max(get_corr(hc)) * 2.0 / n)

    # Phase 2: Main coordinate descent (45%)
    t2 = t0 + budget_s * 0.57
    best_h, best_c5 = cd_fast(best_h, t2, max_sw=15000, sym=False)

    # Phase 3: Symmetric coordinate descent (10%)
    t3 = t0 + budget_s * 0.67
    sh = 0.5 * (best_h + best_h[::-1].copy())
    sh = proj(sh)
    sh, sc5 = cd_fast(sh, t3, max_sw=8000, sym=True)
    if sc5 < best_c5:
        best_c5, best_h = sc5, sh

    # Phase 4: High-T L-BFGS refinement (12%)
    t4 = t0 + budget_s * 0.79
    for T in [8000, 30000, 100000]:
        if time.time() > t4:
            break
        hc = do_lbfgs(best_h, T, 400.0, 3000)
        c5v = float(np.max(get_corr(hc)) * 2.0 / n)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
        # Quick coord descent after each L-BFGS

    # Phase 5: More coord descent (10%)
    t5 = t0 + budget_s * 0.89
    best_h, best_c5 = cd_fast(best_h, t5, max_sw=10000, sym=False)

    # Phase 6: Basin hopping (8%)
    t6 = t0 + budget_s * 0.97
    wh, wc5 = best_h.copy(), best_c5
    for it in range(25):
        if time.time() > t6:
            break
        cool = 1.0 - 0.8 * (it / 25.0)
        sc = 0.015 + 0.04 * cool
        hp = proj(wh + sc * np.random.randn(n))
        hp, hc5 = cd_fast(hp, min(time.time() + 8, t6), max_sw=500)
        if hc5 < wc5 or np.random.rand() < np.exp(-(hc5 - wc5) / (0.0004 * cool + 0.00008)):
            wh, wc5 = hp, hc5
            if hc5 < best_c5:
                best_c5, best_h = hc5, hp

    # Final polish
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = cd_fast(best_h, t0 + budget_s * 0.99, max_sw=5000, sym=False)
        hc = do_lbfgs(best_h, 200000, 600.0, 2000)
        c5v = float(np.max(get_corr(hc)) * 2.0 / n)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_c5 = float(np.max(get_corr(best_h)) * 2.0 / n)
    return best_h, best_c5, n
