def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (d / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def coord_min_1d(corr_base, coefs, lo=0.0, hi=1.0, niter=80):
    """Ternary search for min of max(corr_base + m*coefs) over m in [lo,hi]."""
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr_base + m1 * coefs)
        f2 = np.max(corr_base + m2 * coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def compute_coefs(h, i, n):
    """Compute coefficients for 1D change at index i."""
    coefs = np.zeros(2 * n - 1)
    # h[i] as first factor: position n-1+k, need i+k in [0,n)
