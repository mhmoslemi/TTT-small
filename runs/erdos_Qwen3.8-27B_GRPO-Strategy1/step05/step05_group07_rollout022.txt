def basin_hop(h, n, t_end, n_iters=300):

            pert[j] = best[i] - work[j]

        ht = lbfgs(hp, T_val, n, iters=200)

            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return proj_h(h_new, n_new)

def downsample(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()
    # Average blocks
    block = n_old // n_new
    if block * n_new != n_old:
        block = n_old // n_new
        h_old = h_old[:block * n_new]
    h_new = h_old.reshape(n_new, block).mean(axis=1)
    return proj_h(h_new, n_new)

def proj_box_sum(v, n, target):
    """Project v onto {x: 0<=x<=1, sum x = target}"""
    # Find tau such that sum(clip(v - tau, 0, 1)) = target
    # clip(v-tau, 0, 1) = max(0, min(1, v-tau))
    # This is piecewise linear in tau, decreasing
    
    lo, hi = v.min() - 1, v.max() + 1
    
    def f(tau):
        return np.sum(np.clip(v - tau, 0.0, 1.0)) - target
    
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(v - tau, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def proj_box_sum(v, n, target):
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(v - tau, 0.0, 1.0)

def proj_h(h, n):
    return proj_box_sum(h, n, 0.5 * n)

_gT = [1.0]

def _sobj(x):
    T = _gT[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T*(c-m))))/T) * 2.0/len(x)

def _sgrad(x):
    T = _gT[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T*(c-m))
    w = e / e.sum()
    g = np.zeros(n)
    mf = 2*n - 1
    for k in range(mf):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1, hi1 = max(0, n-1-k), min(n, 2*n-1-k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2, hi2 = max(0, k-n+1), min(n, k+1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0/n)

def lbfgs_opt(h, T, n, iters=300):
    _gT[0] = T
    try:
        r = minimize(h, _sobj, _sgrad, method='L-BFGS-B',
                     bounds=[(0.0,1.0)]*n,
                     options={'maxiter':iters,'ftol':1e-18,'gtol':1e-10})
        return proj_box_sum(r.x, n, 0.5*n)
    except Exception:
        return h

def fista_opt(h, n, t_end, T=5000.0, n_iters=5000):
    """FISTA on soft-max objective with projection."""
    _gT[0] = T
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    best = h.copy()
    best_c5 = compute_c5(best)
    
    for it in range(n_iters):
        if time.time() > t_end:
            break
        grad = _sgrad(y)
        step = 1.0 / (np.linalg.norm(grad)**2 + 1e-10)
        step = min(step, 0.5)  # cap step size
        y_new = proj_box_sum(y - step * grad, n, 0.5*n)
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0*t_k*t_k)) / 2.0
        y = y_new + ((t_k - 1.0) / t_next) * (y_new - x)
        y = np.clip(y, 0.0, 1.0)
        x = y_new
        t_k = t_next
        
        # Track best (using true C5, not soft approximation)
        if it % 100 == 0:
            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best = x.copy()
    
    c5 = compute_c5(x)
    if c5 < best_c5:
        best = x.copy()
        best_c5 = c5
    return best, best_c5

                if 0 <= j1 < n:
                    coefs[mi] -= best[j1]
            
            # Minimize max(top_corr + delta*coefs) over delta in [-old, 1-old]
            lo, hi = -old, 1.0 - old
            for _ in range(50):
                m1 = lo + (hi-lo)/3.0
                m2 = hi - (hi-lo)/3.0
                if np.max(top_corr + m1*coefs) < np.max(top_corr + m2*coefs):
                    hi = m2
                else:
                    lo = m1
            delta = 0.5*(lo+hi)
            new_max = np.max(top_corr + delta*coefs)
            if target - new_max > best_impr:
                best_impr = target - new_max
                best_i = i
                best_nv = np.clip(old + delta, 0.0, 1.0)
        
        if best_i < 0 or best_impr < 1e-15:
            # Try a small random perturbation to escape
            if np.random.rand() < 0.1:
                pert = 0.05 * np.random.randn(n)
                trial = proj_box_sum(best + pert, n, 0.5*n)
                c5t = compute_c5(trial)
                if c5t < best_c5:
                    best = trial
                    best_c5 = c5t
                continue
            break
        
        trial = best.copy()
        trial[best_i] = best_nv
        trial = proj_box_sum(trial, n, 0.5*n)
        c5t = compute_c5(trial)
        if c5t < best_c5 - 1e-15:
            best = trial
            best_c5 = c5t
    
    return best, best_c5

def basin_hop(h, n, t_end, n_iters=200):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it/float(n_iters))
        r = np.random.rand()
        if r < 0.25:
            pert = (0.02+0.1*cool)*np.random.randn(n)
        elif r < 0.5:
            f = np.random.randint(1, 20)
            ph = np.random.uniform(0, 2*np.pi)
            pert = (0.02+0.08*cool)*np.sin(2*np.pi*f*np.arange(n)/n + ph)
        elif r < 0.75:
            mask = np.random.random(n) < (0.3+0.3*cool)
            pert = np.zeros(n)
            pert[mask] = (0.04+0.1*cool)*np.random.randn(int(mask.sum()))
        else:
            i, j = np.random.choice(n, 2, replace=False)
            pert = np.zeros(n)
            sv = 0.05 + 0.15*cool
            pert[i] = sv * (best[j] - work[i])
            pert[j] = sv * (best[i] - work[j])
        hp = proj_box_sum(work + pert, n, 0.5*n)
        T_val = 500.0 + 5000.0*cool
        ht = lbfgs_opt(hp, T_val, n, iters=150)
        c5t = compute_c5(ht)
        temp = 0.001*cool + 0.00003
        if c5t < work_c5 or np.random.rand() < np.exp(-max(0.0, c5t-work_c5)/temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5
