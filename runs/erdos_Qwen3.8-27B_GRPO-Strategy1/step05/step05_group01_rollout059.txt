def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.asarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def all_coefs(h, n):
    """Compute ∂corr/∂h for all (i, shift) pairs. Shape: (n, 2n-1)."""
    k = np.arange(2*n - 1) - (n - 1)
    i = np.arange(n)
    ik = i[:, None] - k[None, :]
    ipk = i[:, None] + k[None, :]
    m1 = (ik >= 0) & (ik < n)
    m2 = (ipk >= 0) & (ipk < n)
    c = np.where(m1, 1.0 - h[np.clip(ik, 0, n-1)], 0.0)
    c -= np.where(m2, h[np.clip(ipk, 0, n-1)], 0.0)
    return c

def smooth_obj(x, h0, T, topk_pen=0.0, topk=10):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    if T < 1e10:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    else:
        val = m
    if topk_pen > 0:
        srt = np.sort(corr)
        val += topk_pen * (srt[-1] - srt[-topk])
    return val * dx

def slsqp_step(h, n, T, maxiter=300, ftol=1e-15):
    def obj(x):
        return smooth_obj(x, h, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(res.x)
    except:
        return h

def coord_descent(h, n, t_end, max_sweeps=500, verbose=False):
    """Fast coordinate descent with exact 1D subgradient bisection."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = get_corr(best_h)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        coefs = all_coefs(best_h, n)
        improved = 0
        for i in range(n):
            ci = coefs[i]  # (2n-1,)
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if hi - lo < 1e-14:
                continue
            # Bisection on subgradient to find min of max_s(corr[s] + delta*ci[s])
            lo_d, hi_d = lo, hi
            for _ in range(35):
                mid = (lo_d + hi_d) / 2.0
                vals = corr + mid * ci
                kstar = np.argmax(vals)
                slope = ci[kstar]
                if slope < -1e-10:
                    lo_d = mid
                elif slope > 1e-10:
                    hi_d = mid
                else:
                    break
            delta = (lo_d + hi_d) / 2.0
            new_val = best_h[i] + delta
            new_val = np.clip(new_val, 0.0, 1.0)

            corr += delta * ci
            improved += 1

        if c5_new < best_c5 - 1e-14:

            best_c5 = np.max(corr) * (2.0 / n)

    best_c5 = compute_c5(best_h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.asarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def all_coefs(h, n):
    k = np.arange(2 * n - 1) - (n - 1)
    i = np.arange(n)
    ik = i[:, None] - k[None, :]
    ipk = i[:, None] + k[None, :]
    m_ik = (ik >= 0) & (ik < n)
    m_ipk = (ipk >= 0) & (ipk < n)
    c = np.where(m_ipk, 1.0 - h[np.clip(ipk, 0, n - 1)], 0.0)
    c -= np.where(m_ik, h[np.clip(ik, 0, n - 1)], 0.0)
    return c

def smooth_obj(x, T, topk_pen=0.0, topk=10):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    if T < 1e10:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    else:
        val = m
    if topk_pen > 0:
        srt = np.sort(corr)
        val += topk_pen * (srt[-1] - srt[-topk])
    return val * dx

def slsqp_step(h, n, T, maxiter=250, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(lambda x: smooth_obj(x, T), h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(res.x)
    except Exception:
        return h

            if t_end - time.time() < 10:
                break
            hc = slsqp_step(best_h, n, T, 200, 1e-13)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_c5 = cv
                best_h = hc
    if t_end - time.time() > 15:
        t_cd = t_end - (t_end - time.time()) * 0.15
        best_h, best_c5 = coord_descent(best_h, n, t_cd, max_sweeps=800)
    if t_end - time.time() > 35:
        t_bh = time.time() + min(55, (t_end - time.time()) * 0.35)
        bh_h, bh_c5 = basin_hop(best_h, n, t_bh, 20, 0.05)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
        best_h, best_c5 = coord_descent(best_h, n, min(time.time() + 40, t_end - 10), max_sweeps=400)
    if t_end - time.time() > 10:
        hc = slsqp_step(best_h, n, 50000, 400, 1e-15)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t_end - 2, max_sweeps=500)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=np.float64)
    else:
        h_init = np.zeros(160)
        h_init[:80] = 1.0
    n = len(h_init)
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    t_p1 = t0 + budget_s * 0.48
    best_h, best_c5 = optimize_round(best_h, n, t_p1)
    if t_end - time.time() > budget_s * 0.30:
        for n_try in [n + 20, n - 20]:
            if n_try < 40 or t_end - time.time() < budget_s * 0.12:
                continue
            h_int = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                              np.linspace(0, 2, n, endpoint=False), best_h)
            h_int = project_h(h_int)
            t_sub = min(t0 + budget_s * 0.78, t_end - budget_s * 0.10)
            h_opt, c5_opt = optimize_round(h_int, n_try, t_sub)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n = n_try
    if t_end - time.time() > 12:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, max_sweeps=600)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
