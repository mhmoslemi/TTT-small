def binary_local_search(h, n, t_end, max_iters=5000):
    h = (h > 0.5).astype(float)
    # Fix sum
    target = n // 2
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    if len(ones) > target:
        h[ones[:len(ones)-target]] = 0
    elif len(ones) < target:
        h[zeros[:target-len(ones)]] = 1
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap = None
        best_c5_trial = best_c5
        
        # Try all swaps
        for j in zeros:
            for k in ones:
                h[j] = 1
                h[k] = 0
                corr = np.correlate(h, 1.0-h, mode='full')
                c5 = np.max(corr) * dx
                h[j] = 0
                h[k] = 1
                if c5 < best_c5_trial:
                    best_c5_trial = c5
                    best_swap = (j, k)
        
        if best_swap is None:
            break
        
        j, k = best_swap
        h[j] = 1
        h[k] = 0
        best_c5 = best_c5_trial
        best_h = h.copy()
    
    return best_h, best_c5
