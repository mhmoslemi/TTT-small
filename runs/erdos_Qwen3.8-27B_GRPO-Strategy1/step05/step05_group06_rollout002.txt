            hi2 = min(m - 1, i + n - 1)
            if lo <= hi2:
                kvals = np.arange(lo, hi2 + 1) - i
                lin[lo:hi2+1] = 1.0 - 2.0 * best_h[kvals]
            quad = np.zeros(m)
            k2i = 2 * i
            if 0 <= k2i < m:
                quad[k2i] = -1.0
                if lo <= k2i <= hi2:
                    pass
            best_h[i] = old
            active = np.where(lin != 0.0)[0]
            top_idx = np.argsort(cb)[-10:]
            active = np.unique(np.concatenate([active, top_idx]))
            if len(active) == 0:
                continue
            a_c = cb[active]
            a_l = lin[active]
            a_q = quad[active]
            grid = np.linspace(0.0, 1.0, 16)
            fgrid = a_c + a_l[:, None] * grid[None, :] + a_q[:, None] * (grid[None, :] ** 2)
            fgrid = np.max(fgrid, axis=0)
            jmin = np.argmin(fgrid)
            if jmin == 0:
                lo_b, hi_b = 0.0, grid[1]
            elif jmin == 15:
                lo_b, hi_b = grid[-2], 1.0
            else:
                lo_b, hi_b = grid[jmin-1], grid[jmin+1]
            def f1d(x):
                return float(np.max(a_c + a_l * x + a_q * x * x))
            try:
                res = minimize_scalar(f1d, bounds=(lo_b, hi_b), method='bounded', options={'xatol': 1e-12})
                best_x = float(res.x)
            except Exception:
                best_x = float(grid[jmin])
            if abs(best_x - old) < 1e-14:
                continue
            ht = best_h.copy()
            ht[i] = best_x
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if improved:
            consec_no_improve = 0
        else:
            consec_no_improve += 1
            if consec_no_improve >= 5:
                break
    return best_h, best_c5

def subgrad_lbfgs(h, T, lam, maxiter):
    n = len(h)
    dx = 2.0 / n
    x0 = project_h(h)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        shifted -= np.max(shifted)
        exps = np.exp(T * shifted)
        sm = mc + np.log(expw_sum(exps)) / T
        return dx * sm + lam * (x.sum() - n / 2.0) ** 2
    def expw_sum(e):
        return e.sum()
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        shifted -= np.max(shifted)
        exps = np.exp(T * shifted)
        w = exps / exps.sum()
        sm = mc + np.log(exps.sum()) / T
        g = np.zeros(n)
        m_full = 2 * n - 1
        lag_arr = np.arange(m_full) - n + 1
        for idx in range(m_full):
            wi = w[idx]
            if wi < 1e-15:
                continue
            lg = lag_arr[idx]
            if lg >= 0:
                if lg < n:
                    g[lg:] += wi * (1.0 - x[:n-lg])
                    g[:n-lg] += -wi * x[lg:]
            else:
                p = -lg
                g[:n-p] += wi * (1.0 - x[p:])
                g[p:] += -wi * x[:n-p]
        return dx * g + 2.0 * lam * (x.sum() - n / 2.0)
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_h(res.x)
    except Exception:
        return x0

def make_candidates(n, base_h):
    cands = []
    h_step = np.zeros(n)
    h_step[:n//2] = 1.0
    cands.append(project_h(h_step))
    h_alt = np.zeros(n)
    h_alt[::2] = 1.0
    cands.append(project_h(h_alt))
    x = np.linspace(0, 2, n, endpoint=False)
    for fr in [1, 2, 3, 4, 5, 8, 12]:
        hp = 0.5 + 0.45 * np.sin(2*np.pi*fr*x/2.0)
        cands.append(project_h(hp))
    for amp in [0.2, 0.4, 0.45]:
        hw = 0.5 + amp * np.sin(2*np.pi*x/2.0) + 0.1*amp*np.sin(4*np.pi*x/2.0)
        cands.append(project_h(hw))
    for _ in range(2):
        cands.append(project_h(np.random.beta(3, 3, n)))
    cands.append(project_h(base_h + 0.05*np.random.randn(n)))
    cands.append(project_h(base_h + 0.15*np.random.randn(n)))
    return cands

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) > 0:
        base_160 = np.array(initial_h_values, dtype=float).copy()
    else:
        base_160 = np.zeros(160)
        base_160[:80] = 1.0
        base_160 = project_h(base_160)

    best_h = base_160.copy()
    best_c5 = compute_c5(best_h)

    n_target = 240
    h240 = interpolate_h(best_h, n_target)
    h240 = project_h(h240)
    c5_240 = compute_c5(h240)
    if c5_240 < best_c5:
        best_c5 = c5_240
        best_h = h240

    n = len(best_h)

    # Phase 1: Coordinate descent (40% budget)
    p1_end = t0 + budget_s * 0.40
    best_h, best_c5 = coord_descent(best_h, p1_end, max_sweeps=50000)

    # Phase 2: L-BFGS-B escape (20% budget)
    p2_end = t0 + budget_s * 0.60
    for T in [2000, 8000, 20000, 50000]:
        if time.time() > p2_end:
            break
        hc = subgrad_lbfgs(best_h, T, 200.0, 2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, t0 + budget_s * 0.55, max_sweeps=5000)

    # Phase 3: Multi-restart (20% budget)
    p3_end = t0 + budget_s * 0.80
    cands = make_candidates(n, best_h)
    for hc_init in cands:
        if time.time() > p3_end:
            break
        hc_init, c5_temp = coord_descent(hc_init, t0 + budget_s * 0.78, max_sweeps=3000)
        if c5_temp < best_c5:
            best_c5 = c5_temp
            best_h = hc_init

    # Phase 4: Final coordinate descent polish (15% budget)
    p4_end = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, p4_end, max_sweeps=100000)

    # Phase 5: Final L-BFGS-B + CD
    if time.time() < t0 + budget_s * 0.99:
        hc = subgrad_lbfgs(best_h, 80000, 300.0, 3000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        tc = t0 + budget_s * 0.995
        best_h, best_c5 = coord_descent(best_h, tc, max_sweeps=20000)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
