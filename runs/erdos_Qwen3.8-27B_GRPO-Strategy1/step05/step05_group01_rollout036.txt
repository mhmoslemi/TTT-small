import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    if T > 1e6:
        return m * dx
    e = np.exp(T * (corr - m))
    return (m + np.log(e.sum()) / T) * dx

def sum_con(x, target):
    return x.sum() - target

def slsqp(h, T, n, maxiter=500, ftol=1e-15):
    try:
        res = minimize(slsqp_obj, h, args=(T, n), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n / 2.0,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def line_search_fn(t, base, coefs):
    return np.max(base + t * coefs)

def coord_descent(h, n, t_end, max_sweeps, n_top, pair_prob=0.0):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    stalls = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        order = np.arange(n)
        np.random.shuffle(order)
        improved = 0
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.empty(n_top)
            base = np.empty(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
                base[t] = corr[top_idx[t]] - best_h[i] * c
            res = minimize_scalar(line_search_fn, args=(base, coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-14})
            t_new = res.x
            if abs(t_new - best_h[i]) > 1e-12:
                h_try = best_h.copy()
                h_try[i] = t_new
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    improved += 1
            elif pair_prob > 0.0 and np.random.random() < pair_prob:
                j = np.random.randint(n)
                if j == i:
                    continue
                offsets = np.array([-0.15, -0.05, 0.0, 0.05, 0.15])
                best_pc5 = best_c5
                best_ti = best_h[i]
                best_tj = best_h[j]
                for di in offsets:
                    ti = np.clip(best_h[i] + di, 0.0, 1.0)
                    for dj in offsets:
                        tj = np.clip(best_h[j] + dj, 0.0, 1.0)
                        ht = best_h.copy()
                        ht[i] = ti
                        ht[j] = tj
                        ht = project_h(ht)
                        c = np.correlate(ht, 1.0 - ht, mode='full')
                        cv = float(c.max() * dx)
                        if cv < best_pc5 - 1e-14:
                            best_pc5 = cv
                            best_ti = ti
                            best_tj = tj
                if best_pc5 < best_c5 - 1e-14:
                    best_h[i] = best_ti
                    best_h[j] = best_tj
                    best_h = project_h(best_h)
                    best_c5 = best_pc5
                    improved += 1
        if improved == 0:
            stalls += 1
            if stalls > 5:
                break
            n_pert = max(3, n // 40)
            pidx = np.random.choice(n, n_pert, replace=False)
            for pi in pidx:
                best_h[pi] = np.clip(best_h[pi] + 0.012 * np.random.randn(), 0.0, 1.0)
            best_h = project_h(best_h)
        else:
            stalls = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup (10%)
    t_p1 = t0 + budget_s * 0.10
    for T in [200, 2000, 20000, 200000]:
        if time.time() > t_p1 - 3:
            break
        hc = slsqp(best_h, T, n, 350, 1e-15)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc

    # Phase 2: Main coordinate descent at parent n (45%)
    t_p2 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, n, t_p2, 500, 50, pair_prob=0.04)

    # Phase 3: Try n=256 (20%)
    t_p3 = t0 + budget_s * 0.78
    if time.time() < t_p3 - 15:
        n2 = 256
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        hc2, cv2 = coord_descent(h2, n2, min(t_p3, t_end - 10), 200, 50, pair_prob=0.03)
        if cv2 < best_c5:
            best_c5 = cv2
            best_h = hc2
            n = n2

    # Phase 4: Polish - SLSQP + high-ntop coord descent (20%)
    if time.time() < t_end - 12:
        t_p4 = t_end - 8
        hc = slsqp(best_h, 500000, n, 500, 1e-16)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t_p4, 150, 80, pair_prob=0.02)

    # Final SLSQP
    if time.time() < t_end - 4:
        hc = slsqp(best_h, 2000000, n, 400, 1e-16)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc

    # Very final coord descent
    if time.time() < t_end - 1:
        best_h, best_c5 = coord_descent(best_h, n, t_end, 30, 80, pair_prob=0.0)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
