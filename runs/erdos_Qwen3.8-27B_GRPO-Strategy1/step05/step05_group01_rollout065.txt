If they DO appear at distance k, then we have a bilinear term. We can still handle it by scanning one variable and solving 1D in the other.

Let me also make sure the coordinate descent is truly exact. For each coordinate i:
- The overlap for shift k, as a function of x = h[i], is:
  overlap_k(x) = corr_k - h[i]*coef_k + x*coef_k = (corr_k - h[i]*coef_k) + x*coef_k
  where coef_k = (1-h[i+k])*[valid] - h[i-k]*[valid]

- We want: min_{x∈[0,1]} max_k [(corr_k - h[i]*coef_k) + x*coef_k]

This is the minimum of a piecewise linear convex function over [0,1]. The minimum occurs at:
- x = 0
- x = 1
- A breakeven point where two constraints are simultaneously tight

The breakeven point between constraints a and b:
(corr_a - h[i]*coef_a) + x*coef_a = (corr_b - h[i]*coef_b) + x*coef_b
x = (corr_b - corr_a + h[i]*(coef_b - coef_a)) / (coef_a - coef_b)
  = (corr_b - corr_a) / (coef_b - coef_a) + h[i]

Wait, let me redo:
A_a + x*B_a = A_b + x*B_b where A_k = corr_k - h[i]*coef_k, B_k = coef_k
x*(B_a - B_b) = A_b - A_a
x = (A_b - A_a) / (B_a - B_b) = (corr_b - h[i]*coef_b - corr_a + h[i]*coef_a) / (coef_a - coef_b)
= (corr_b - corr_a + h[i]*(coef_a - coef_b)) / (coef_a - coef_b)
= (corr_b - corr_a)/(coef_a - coef_b) + h[i]

So the breakeven x is: h[i] + (corr_b - corr_a)/(coef_a - coef_b)

For the 1D minimax of a convex piecewise linear function, the optimal x is the median of the breakeven points (in a sense). More precisely, it's the point where the "upper envelope" transitions from decreasing to increasing, which is where two lines cross at the peak.

Actually, the simplest correct approach: the minimum of max of linear functions over [0,1] is achieved at a point where at least two of the linear functions are simultaneously at their maximum (or at a boundary). So checking all breakeven points is correct.

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def vec_subgrad(h, shifts, alpha):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in shifts:
        ak = abs(k)
        if ak >= n:
            continue
        i1 = max(0, -k)
        i2 = min(n, n - k)
        grad[i1:i2] += (1.0 - h[i1+k:i2+k])
        i3 = max(0, k)
        i4 = min(n, n + k)
        grad[i3:i4] -= h[i3-k:i4-k]
    ns = max(len(shifts), 1)
    grad /= ns
    return project_h(h - alpha * grad * dx), grad

def coord_descent(h, n, t_end, n_top=20, max_sweeps=400):
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = shifts[t]
                if abs(k) >= n:
                    continue
                jp = i + k
                jm = i - k
                c = 0.0
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            A = corr[top_idx] - best_h[i] * coefs
            hi_cur = best_h[i]
            best_x = hi_cur
            best_val = float(np.max(A + hi_cur * coefs))
            cands = [0.0, 1.0, hi_cur]
            m_check = min(n_top, 10)
            for a in range(m_check):
                for b in range(a + 1, m_check):
                    if abs(coefs[b] - coefs[a]) > 1e-14:
                        xb = (A[a] - A[b]) / (coefs[b] - coefs[a])
                        if -0.01 <= xb <= 1.01:
                            cands.append(xb)
            for xc in cands:
                xc = min(max(xc, 0.0), 1.0)
                val = float(np.max(A + xc * coefs))
                if val < best_val - 1e-13:
                    best_val = val
                    best_x = xc
            if abs(best_x - hi_cur) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = best_x
            h_try = project_h(h_try)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_phase(h, n, t_end, alpha0=0.02, alpha1=0.003, n_top=10):
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    mom = np.zeros(n)
    steps = 0
    while time.time() < t_end:
        steps += 1
        frac = min(steps / 200.0, 1.0)
        alpha = alpha0 + (alpha1 - alpha0) * frac
        nt = n_top + int(10 * frac)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-nt:]
        shifts = top_idx - (n - 1)
        h_new, grad = vec_subgrad(best_h, shifts, alpha)
        mom = 0.5 * mom + 0.5 * grad
        h_mom = project_h(best_h - alpha * 0.7 * mom * (2.0 / n))
        c5_new, _ = compute_c5(h_new)
        c5_mom, _ = compute_c5(h_mom)
        if c5_mom < c5_new:
            h_new, c5_new = h_mom, c5_mom
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def swap_optimize(h, n, t_end, n_top=15, max_sweeps=100):
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        shifts = set(np.abs(top_idx - (n - 1)).tolist())
        improved = False
        for _ in range(n // 2):
            if time.time() > t_end:
                return best_h, best_c5
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            if abs(i - j) in shifts:
                continue
            coefs_i = np.zeros(n_top)
            coefs_j = np.zeros(n_top)
            for t in range(n_top):
                k = top_idx[t] - (n - 1)
                if abs(k) >= n:
                    continue
                for (pos, arr) in [(i, coefs_i), (j, coefs_j)]:
                    jp = pos + k
                    jm = pos - k
                    c = 0.0
                    if 0 <= jp < n:
                        c += (1.0 - best_h[jp])
                    if 0 <= jm < n:
                        c -= best_h[jm]
                    arr[t] = c
            A_i = corr[top_idx] - best_h[i] * coefs_i
            A_j = corr[top_idx] - best_h[j] * coefs_j
            hi = best_h[i]
            hj = best_h[j]
            xvals = np.linspace(0, 1, 21)
            best_pair = (hi, hj)
            best_v = float(np.maximum(A_i + hi * coefs_i, A_j + hj * coefs_j).max())
            for xi in xvals:
                for xj in xvals:
                    if abs(xi - xj) < 1e-6:
                        continue
                    vi = A_i + xi * coefs_i
                    vj = A_j + xj * coefs_j
                    val = float(max(vi.max(), vj.max()))
                    if val < best_v - 1e-12:
                        best_v = val
                        best_pair = (xi, xj)
            if best_pair != (hi, hj):
                h_try = best_h.copy()
                h_try[i] = best_pair[0]
                h_try[j] = best_pair[1]
                h_try = project_h(h_try)
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Subgradient descent (25%)
    t_p1 = t0 + budget_s * 0.25
    best_h, best_c5 = subgrad_phase(best_h, n, t_p1, 0.025, 0.004, 10)

    # Phase 2: Coordinate descent (25%)
    t_p2 = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, n, t_p2, n_top=20, max_sweeps=500)

    # Phase 3: Alternating subgrad + coord + swap (25%)
    t_p3 = t0 + budget_s * 0.75
    while time.time() < t_p3:
        t_sub = min(time.time() + 15, t_p3)
        best_h, best_c5 = subgrad_phase(best_h, n, t_sub, 0.01, 0.003, 8)
        t_cd = min(time.time() + 40, t_p3)
        best_h, best_c5 = coord_descent(best_h, n, t_cd, n_top=25, max_sweeps=150)
        t_sw = time.time() + 10
        if t_sw < t_p3 - 5:
            best_h, best_c5 = swap_optimize(best_h, n, t_sw, n_top=12, max_sweeps=30)

    # Phase 4: Multi-start refinement (15%)
    t_p4 = t_end - 15
    for s in range(3):
        if time.time() > t_p4:
            break
        if s == 0:
            h_try = best_h.copy()
        else:
            scale = 0.06 * (1.0 - s / 3.0)
            h_try = project_h(best_h + np.random.randn(n) * scale)
        t_sub = t_p4 - (t_p4 - time.time()) * (1 - s / 3.0) * 0.3
        best_h2, c5_2 = subgrad_phase(h_try, n, min(time.time() + 10, t_sub), 0.008, 0.002, 10)
        best_h2, c5_2 = coord_descent(best_h2, n, t_sub, n_top=20, max_sweeps=100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = best_h2

    # Final polish
    if time.time() < t_end - 3:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 2, n_top=30, max_sweeps=200)

    best_h = project_h(best_h)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, n
