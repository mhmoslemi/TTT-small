import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-14:
            h = np.clip(h * (tgt / max(s, 1e-14)), 0, 1)
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _so(x, T, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    val = m + np.log(np.sum(np.exp(T * (c - m)))) / T
    return val * 2.0 / n

def _sg(x, T, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            g[lo1:hi1] += wk * (1.0 - x[lo1 + k - n + 1:hi1 + k - n + 1])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            g[lo2:hi2] -= wk * x[lo2 - k + n - 1:hi2 - k + n - 1]
    return g * (2.0 / n)

def lbfgs_opt(h, T, iters):
    n = len(h)
    try:
        r = minimize(h, _so, _sg, args=(T, n), method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-11})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, t_end, sweeps):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        cc = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n + 1 - 2  # i-k+n-1
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            base = cc - old * coef
            r = minimize_scalar(lambda t: np.max(base + t * coef),
                                bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-13})
            nv = float(r.x)
            if abs(nv - old) < 1e-14:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                cc = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def worst_shift_grad(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    c5 = c.max()
    active = c >= c5 - 1e-6
    g = np.zeros(n)
    for k in np.where(active)[0]:
        for i in range(n):
            j2 = i + k - n + 1
            if 0 <= j2 < n:
                g[i] += 1.0 - h[j2]
            j1 = i - k + n - 1
            if 0 <= j1 < n:
                g[i] -= h[j1]
    return g, c5

def subgrad_polish(h, t_end, steps=2000):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    eta = 0.05
    for _ in range(steps):
        if time.time() > t_end:
            break
        g, c5_now = worst_shift_grad(cur, n)
        if c5_now < best_c5:
            best = cur.copy()
            best_c5 = c5_now
        g /= max(1.0, np.linalg.norm(g))
        cur = proj_h(cur - eta * g, n)
        c5_2 = compute_c5(cur)
        if c5_2 < c5_now:
            eta = min(eta * 1.05, 0.3)
        else:
            eta *= 0.92
    return best, best_c5

def binary_ls(h, t_end, max_rounds=50000):

    b = (h >= 0.5).astype(float)
    ones = np.where(b == 1)[0]
    n1 = len(ones)

    elif n1 < n // 2:

    dx = 2.0 / n

    best_c5 = float(np.max(corr) * dx)
    best_b = b.copy()
    T = 0.02

        i, j = np.random.randint(0, n), np.random.randint(0, n)

        b[i], b[j] = b[j], b[i]
        nc = np.correlate(b, 1.0 - b, mode='full')

            if nc5 < best_c5:

                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]
        T = max(0.0003, T * 0.99998)
    return best_b, best_c5
