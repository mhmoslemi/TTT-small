def nesterov_opt(h_in, N, deadline, k=40, lr0=0.02):

    target = N / 2.0
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    m = np.zeros(N)

    beta = 0.9

    while time.time() < deadline:

    return np.real(np.fft.ifft(H * G))
