def coord_step(h, i):
    """Optimal value for h[i] given all other coordinates fixed."""
    n = len(h)
    m_full = 2 * n - 1
    x = h[i]
    h[i] = 0.0
    base = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute L and Q arrays
    L = np.zeros(m_full)
    Q = np.zeros(m_full)
    lag_arr = np.arange(m_full) - (n - 1)  # shift m for each k
    
    for kk in range(m_full):
        m = lag_arr[kk]
        if m > 0:
            if i >= m:
                L[kk] += 1.0
            if m <= i <= n - 1 - m:
                Q[kk] -= h[i - m]
            if i <= n - 1 - m:
                Q[kk] -= h[i + m]
        elif m < 0:
            p = -m
            if i < n - p:
                L[kk] += 1.0
            if i + p < n:
                Q[kk] -= h[i + p]
            if i >= p:
                Q[kk] -= h[i - p]
        else:  # m == 0
            L[kk] += 1.0
            Q[kk] -= 1.0
    
    # Now minimize max_k(base[k] + x*L[k] + x^2*Q[k]) over x in [0,1]

    # So all Q[k] ≤ 0, meaning all terms are concave or linear in x
    # The max of concave/linear functions... hmm, not necessarily concave

    def f(xv):

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def min1d_fast(base, L, Q):
    m = len(base)
    xs = np.linspace(0.0, 1.0, 32)
    vals = np.empty(32)
    for j in range(32):
        x = xs[j]
        vals[j] = (base + x * L + x * x * Q).max()
    j0 = int(np.argmin(vals))
    lo = xs[max(0, j0 - 1)]
    hi = xs[min(31, j0 + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(40):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f1 = (base + m1 * L + m1 * m1 * Q).max()
        f2 = (base + m2 * L + m2 * m2 * Q).max()
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def fast_coord_descent(h, t_end, max_sweeps=100000):
    n = len(h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    m_arr = np.arange(m_full) - (n - 1)
    pos = m_arr > 0
    neg = m_arr < 0
    zero = (m_arr == 0)
    pp = -m_arr

    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = float(corr.max() * dx)

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            v = h[i]
            L = np.zeros(m_full)
            Q = np.zeros(m_full)
            c1 = pos & (m_arr <= i)
            c2 = pos & (i + m_arr <= n - 1)

            only1 = c1 & ~c2
            only2 = ~c1 & c2
            if both.any():

            if only1.any():
                L[only1] = 1.0 - h[i - m_arr[only1]]
            if only2.any():
                L[only2] = -h[i + m_arr[only2]]
            c1n = neg & (i + pp <= n - 1)
            c2n = neg & (pp <= i)
            bothn = c1n & c2n
            only1n = c1n & ~c2n
            only2n = ~c1n & c2n
            if bothn.any():

            if only1n.any():
                L[only1n] = 1.0 - h[i + pp[only1n]]
            if only2n.any():
                L[only2n] = -h[i - pp[only2n]]
            L[zero] = 1.0
            Q[zero] = -1.0

            else:

        if not improved:

    return h, best_c5

def obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-15:
            continue
        m = idx - (n - 1)
        if m >= 0:
            if m < n:
                grad[m:] += wi * (1.0 - h[:n - m])
                grad[:n - m] += -wi * h[m:]
        else:
            p = -m
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n - p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x):
        return obj_grad(x, T, lam)[0]
    def grad(x):
        return obj_grad(x, T, lam)[1]
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return x0

    best_h, best_c5 = fast_coord_descent(best_h.copy(), t4)

    # Final L-BFGS polish
    if time.time() < t0 + budget_s * 0.995:
        hc = lbfgs_opt(best_h, 200000, 600.0, 1500, 1e-18)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h = hc
            best_c5 = c5v

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
