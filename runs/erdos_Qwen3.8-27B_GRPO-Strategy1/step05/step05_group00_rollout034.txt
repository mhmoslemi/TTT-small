```python
def compute_swap_D(h, N):
    """D[p, m] = change in corr[m] when h[p] goes 1->0 (for binary h)"""
    idx = np.arange(2 * N - 1)
    s = (idx - (N - 1))[None, :]  # (1, 2N-1)
    p = np.arange(N)[:, None]  # (N, 1)
    ps = p + s  # (N, 2N-1): p + s
    pm = p - s  # (N, 2N-1): p - s
    vs = (ps >= 0) & (ps < N)
    vm = (pm >= 0) & (pm < N)
    ps_c = np.clip(ps, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    h_ps = np.where(vs, h[ps_c], 0.0)
    h_pm = np.where(vm, h[pm_c], 0.0)
    # D[p,m] = -(1-h[p+s])*v(p+s) + h[p-s]*v(p-s)
    D = -(1.0 - h_ps) * vs.astype(float) + h_pm * vm.astype(float)
    # At s=0 (m=N-1): the formula gives -(1-h[p]) + h[p] = 2h[p]-1
    # But for binary, the actual change at s=0 is 0 (h*(1-h) is always 0)
    # So we zero out the center column
    D[:, N - 1] = 0.0
    return D
```
