def softmax_subgrad(h, n, t_end, seed, max_iters=5000):

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    mom = np.zeros(n)
    beta = 0.8
    lr = 0.5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = it / max_iters
        T = 20.0 + 80.0 * (1 - frac) * (1 - frac)  # decreasing temperature
        corr = np.correlate(h_c, 1.0 - h_c, mode='full')
        tc = T * corr * dx
        tc -= tc.max()

        p /= p.sum()

            if w < 1e-14:

            w *= dx
            k = m - (n - 1)
            if k == 0:
                s += w * (1.0 - 2.0 * h_c)
            elif k > 0:
                s[:n-k] += w * (1.0 - h_c[k:])
                s[k:] -= w * h_c[:n-k]
            else:
                a = -k
                s[:n-a] += w * (1.0 - h_c[a:])
                s[a:] -= w * h_c[:n-a]
        # Nesterov: y = h + beta*mom, grad at y, mom = beta*mom + (h-y)*beta + grad

        # We want h_new = h - alpha * s to decrease c.
        # But with Nesterov: mom = beta*mom + s (extra gradient)
        # h_new = h + mom (Momentum)
        # Let me use: mom = beta*mom + s; h_new = h - lr*mom
        mom = beta * mom + s
        h_new = h_c - alpha * mom

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_c = h_new

            lr = min(lr, 5.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

        c = mask.sum()
        if c == 0:

        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
