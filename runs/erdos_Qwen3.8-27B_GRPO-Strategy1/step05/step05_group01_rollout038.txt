    grad = np.zeros(n)

    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n-1)
    jmc = np.clip(jm, 0, n-1)
    term = (1.0 - h[jpc]) * v1 - h[jmc] * v2
    grad = (p[None, :] * term).sum(axis=1) * dx
    return grad, corr

def smooth_c5_val(h, T, K=50):

import numpy as np
from scipy.optimize import minimize
import time

_N_global = 200

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_val(h, T, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    m = c.max()
    if T > 1e6:
        return m * dx
    vals = np.exp(T * (c - m))
    return (m + np.log(vals.sum()) / T) * dx

def grad_smooth(h, T, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-K:]
    c_top = corr[idx]
    p = np.exp(T * (c_top - c_top.max()))
    p /= p.sum()
    shifts = idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = ((jp >= 0) & (jp < n)).astype(np.float64)
    v2 = ((jm >= 0) & (jm < n)).astype(np.float64)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    term = (1.0 - h[jpc]) * v1 - h[jmc] * v2
    grad = (p[None, :] * term).sum(axis=1) * dx
    return grad

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fista(h_init, n, t_end, max_iter=5000, T_init=200, T_min=20):
    h = h_init.copy()
    z = h.copy()
    prev_h = h.copy()
    t_mom = 1.0
    step = 0.1
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        frac = it / max_iter
        T = T_init * (1.0 - frac) + T_min * frac
        K = int(20 + 40 * (1.0 - frac))
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z = h + (t_mom - 1.0) / t_next * (h - prev_h)
        z = np.clip(z, 0.0, 1.0)
        g = grad_smooth(z, T, K)
        h_new = project_h(z - step * g)
        # Adaptive step
        f_new = smooth_c5_val(h_new, T, min(K+20, 2*n-1))
        f_z = smooth_c5_val(z, T, min(K+20, 2*n-1))
        d2 = np.sum((h_new - z) ** 2)
        if f_new > f_z + 0.5 * step * d2 + 1e-15:
            step = max(step * 0.5, 1e-6)
            h_new = project_h(z - step * g)
        elif it > 10:
            step = min(step * 1.05, 2.0)
        prev_h = h.copy()
        h = h_new
        t_mom = t_next
        c5_v = compute_c5(h)
        if c5_v < best_c5 - 1e-8:
            best_c5 = c5_v
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 500:
            break
    return best_h, best_c5

def _obj_slsqp(x):
    n = _N_global
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    return c.max() * dx

def _cons_fun(x):
    return x.sum() - _N_global / 2.0

def slsqp_polish(h, n, t_end, maxiter=400):
    global _N_global
    _N_global = n
    try:
        res = minimize(_obj_slsqp, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_top(h, n, t_end, n_sweeps=40, n_top=12):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                break
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            # Evaluate at 11 points in [0,1]
            ts = np.linspace(0, 1, 11)
            vals = np.max(base[None, :] + ts[:, None] * coefs[None, :], axis=1)
            t_best = ts[np.argmin(vals)]
            # Refine around t_best
            if 0 < t_best < 1:
                ts2 = np.linspace(t_best - 1/10, t_best + 1/10, 11)
                ts2 = np.clip(ts2, 0, 1)
                vals2 = np.max(base[None, :] + ts2[:, None] * coefs[None, :], axis=1)
                t_best = ts2[np.argmin(vals2)]
            if abs(t_best - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = t_best
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def make_init(n, kind):
    x = np.arange(n)
    if kind == 0:
        # Block
        h = np.zeros(n)
        h[:n//2] = 1.0
    elif kind == 1:
        # Periodic
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x / n)
        h = np.clip(h, 0, 1)
    elif kind == 2:
        # Two-frequency
        h = 0.5 + 0.2 * np.sin(2 * np.pi * 2 * x / n) + 0.2 * np.sin(2 * np.pi * 5 * x / n)
        h = np.clip(h, 0, 1)
    elif kind == 3:
        # Random
        h = np.random.uniform(0, 1, n)
    else:
        h = np.random.uniform(0.2, 0.8, n)
    return project_h(h)

def optimize_one(h, n, t_end, t_fista_frac=0.5, t_coord_frac=0.25, t_slsqp_frac=0.15):
    h = project_h(h.copy())
    c5 = compute_c5(h)
    t_fista_end = time.time() + (t_end - time.time()) * t_fista_frac
    t_coord_end = t_fista_end + (t_end - time.time()) * t_coord_frac / (t_fista_frac + t_coord_frac)
    
    # FISTA phase
    if t_fista_end > time.time():
        h_f, c5_f = fista(h, n, t_fista_end)
        if c5_f < c5:
            h, c5 = h_f, c5_f
    # Coord descent phase
    t_coord_actual = t_coord_end if t_coord_end < t_end else t_end - (t_end-time.time())*0.15
    if t_coord_actual > time.time():
        h_c, c5_c = coord_top(h, n, t_coord_actual, 40, 12)
        if c5_c < c5:

    # SLSQP polish

        c5_s = compute_c5(h_s)
        if c5_s < c5:
            h, c5 = h_s, c5_s
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

_N_g = 200

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_val(corr, T, K):
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    m = c.max()
    if T > 1e6:
        return m
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def grad_smooth(h, T, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-K:]
    c_top = corr[idx]
    p = np.exp(T * (c_top - c_top.max()))
    p /= p.sum()
    shifts = idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = ((jp >= 0) & (jp < n)).astype(np.float64)
    v2 = ((jm >= 0) & (jm < n)).astype(np.float64)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    term = (1.0 - h[jmc]) * v2 - h[jpc] * v1
    grad = (p[None, :] * term).sum(axis=1) * dx
    return grad

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fista(h_init, n, t_end, max_iter=30000, T_init=300, T_min=30):
    h = h_init.copy()
    z = h.copy()
    prev_h = h.copy()
    t_mom = 1.0
    step = 0.08
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        frac = min(it / max_iter, 1.0)
        T = T_init * (1.0 - frac) + T_min * frac
        K = int(20 + 50 * (1.0 - frac))
        K = min(K, 2 * n - 1)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z = h + (t_mom - 1.0) / t_next * (h - prev_h)
        z = np.clip(z, 0.0, 1.0)
        g = grad_smooth(z, T, K)
        h_new = project_h(z - step * g)
        dx = 2.0 / n
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        corr_z = np.correlate(z, 1.0 - z, mode='full')
        f_new = smooth_val(corr_new, T, min(K + 20, 2 * n - 1)) * dx
        f_z = smooth_val(corr_z, T, min(K + 20, 2 * n - 1)) * dx
        d2 = np.sum((h_new - z) ** 2)
        if f_new > f_z + 0.5 * step * d2 + 1e-14:
            step = max(step * 0.5, 1e-7)
            h_new = project_h(z - step * g)
        elif it > 20:
            step = min(step * 1.03, 1.5)
        prev_h = h.copy()
        h = h_new
        t_mom = t_next
        c5_v = compute_c5(h)
        if c5_v < best_c5 - 1e-9:
            best_c5 = c5_v
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            break
    return best_h, best_c5

def _obj_fn(x):
    dx = 2.0 / _N_g
    c = np.correlate(x, 1.0 - x, mode='full')
    return c.max() * dx

def _cons_fn(x):
    return x.sum() - _N_g / 2.0

def slsqp_polish(h, n, t_end, maxiter=400):
    global _N_g
    _N_g = n
    try:
        res = minimize(_obj_fn, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _cons_fn}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_top(h, n, t_end, n_sweeps=60, n_top=12):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                break
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c -= best_h[jp]
                if 0 <= jm < n:
                    c += (1.0 - best_h[jm])
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            ts = np.linspace(0.0, 1.0, 21)
            grid = base[None, :] + ts[:, None] * coefs[None, :]
            vals = np.max(grid, axis=1)
            t_best = ts[np.argmin(vals)]
            if 0.0 < t_best < 1.0:
                ts2 = np.linspace(max(0, t_best - 0.05), min(1, t_best + 0.05), 11)
                grid2 = base[None, :] + ts2[:, None] * coefs[None, :]
                vals2 = np.max(grid2, axis=1)
                t_best = ts2[np.argmin(vals2)]
            if abs(t_best - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = t_best
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def make_init(n, kind, rng):
    x = np.arange(n)
    if kind == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif kind == 1:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * x / n + rng.uniform(0, 6.28))
    elif kind == 2:
        h = 0.5 + 0.2 * np.sin(4 * np.pi * x / n + 1.3) + 0.15 * np.sin(10 * np.pi * x / n + 4.1)
    elif kind == 3:
        h = rng.uniform(0, 1, n)
    else:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x / n + rng.uniform(0, 6.28))
    return project_h(np.clip(h, 0, 1))

def optimize_one(h, n, t_end):
    h = project_h(h.copy())
    c5 = compute_c5(h)
    elapsed = time.time() - t_end
    remaining = max(t_end - time.time(), 1.0)
    t_fista = time.time() + remaining * 0.55
    t_coord = t_fista + remaining * 0.25
    t_slsqp = time.time() + remaining * 0.12
    if t_fista > time.time():
        h_f, c5_f = fista(h, n, t_fista)
        if c5_f < c5:
            h, c5 = h_f, c5_f
    if t_coord > time.time():
        h_c, c5_c = coord_top(h, n, t_coord, 50, 12)
        if c5_c < c5:
            h, c5 = h_c, c5_c
    if t_slsqp > time.time() + 10:
        h_s = slsqp_polish(h, n, t_slsqp, 350)
        c5_s = compute_c5(h_s)
        if c5_s < c5:
            h, c5 = h_s, c5_s
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
