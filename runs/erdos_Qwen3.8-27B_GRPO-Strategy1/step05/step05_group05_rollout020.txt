    try:

def slsqp_opt(h, T, n, maxiter=1500):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

        if elapsed() > budget_s * 0.15:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Fast coordinate descent (25% budget)
    best_h, best_c5 = coord_descent(best_h, n, max_sweeps=400,
                                    t_end=t0 + budget_s * 0.30)

    # Phase 3: Basin hopping (25% budget)
    best_h, best_c5 = basin_hopping(best_h, n, best_c5,
                                    t_end=t0 + budget_s * 0.55, n_iters=300)

    # Phase 4: More coordinate descent after basin hopping (15% budget)
    best_h, best_c5 = coord_descent(best_h, n, max_sweeps=500,
                                    t_end=t0 + budget_s * 0.70)

    # Phase 5: Subgradient polish (10% budget)
    if elapsed() < budget_s * 0.90:
        best_h, best_c5 = subgrad_polish(best_h, n, n_iters=400)

    # Phase 6: Final SLSQP with very high T
    if elapsed() < budget_s * 0.95:
        for T in [500000, 2000000]:
            if elapsed() > budget_s * 0.96:
                break
            hc = slsqp_opt(best_h, T, n, maxiter=2000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 7: Final coordinate descent
    remaining = budget_s * 0.98 - elapsed()
    if remaining > 2.0:
        best_h, best_c5 = coord_descent(best_h, n, max_sweeps=300,
                                        t_end=t0 + budget_s * 0.98)

    # Final subgradient
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = subgrad_polish(best_h, n, n_iters=200)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
