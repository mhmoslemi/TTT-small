import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    logsum = mc + np.log(exps.sum()) / T
    w = exps / exps.sum()
    val = dx * logsum + lam * (h.sum() - n / 2.0) ** 2
    grad_c = np.zeros(n)
    lag_arr = np.arange(m) - (n - 1)
    for idx in range(m):
        wi = w[idx]
        if wi < 1e-15:
            continue
        lg = lag_arr[idx]
        if lg == 0:
            grad_c += wi * (1.0 - 2.0 * h)
        elif lg > 0:

                grad_c[p:] += -wi * h[:n - p]
    grad = dx * grad_c + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol, t_lim):
    n = len(h0)
    x0 = project_h(h0)
    def f(x):
        if time.time() > t_lim:
            raise TimeoutError()
        return obj_grad(x, T, lam)[0]
    def g(x):
        if time.time() > t_lim:
            raise TimeoutError()
        return obj_grad(x, T, lam)[1]
    try:
        res = minimize(f, x0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': 1e-12, 'maxls': 15, 'maxfun': maxiter * 30})
        return project_h(res.x)
    except Exception:
        return x0

def coord_descent(h, n, t_end, max_sweeps, eps=1e-11):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m = 2 * n - 1
    lag_arr = np.arange(m) - (n - 1)
    bbox = (0.0, 1.0)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            lin = np.zeros(m)
            quad = np.zeros(m)
            hv = best_h[i]
            for idx in range(m):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0 - 2.0 * hv
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')

                                  options={'xatol': eps})
            xi = res.x
            if abs(xi - hv) < 1e-14:

            best_h[i] = xi
            improved = True
        if not improved:
            break
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

            d[p:] = best_h[:n - p]
        else:
            d = 1.0 - 2.0 * best_h
        dn = np.linalg.norm(d)
        if dn < 1e-15:
            continue
        d /= dn
        best_step = None
        for s in [0.0005, 0.002, 0.005, 0.01, 0.02, 0.04]:
            hp = project_h(best_h + s * d)
            c5p = compute_c5(hp)
            if c5p < best_c5 - 1e-16:
                best_step = (hp, c5p)
                break
        if best_step is None:
            d = -d
            for s in [0.002, 0.005, 0.01]:
                hp = project_h(best_h + s * d)
                c5p = compute_c5(hp)
                if c5p < best_c5 - 1e-16:
                    best_step = (hp, c5p)
                    break
        if best_step is not None:
            best_h, best_c5 = best_step
            best_h, best_c5 = coord_descent(best_h, n, min(p4_end, time.time() + 8), 20, eps=1e-12)

    # Phase 5: Final polish - very tight coordinate descent
    p5_end = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, n, p5_end, 400, eps=1e-14)

    # Phase 6: Tiny random kicks + micro coord descent
    p6_end = t0 + budget_s * 0.99
    for _ in range(15):
        if time.time() > p6_end:
            break
        sc = 0.001 * np.random.randn(n)
        hp = project_h(best_h + sc)
        hc2, c52 = coord_descent(hp, n, min(p6_end, time.time() + 8), 25, eps=1e-14)
        if c52 < best_c5:
            best_c5 = c52
            best_h = hc2

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
