                h[p], h[q] = 0.0, 1.0
            elif sweep > max_sweeps * 0.7:

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

        G = (p[None,:] * ((1.0 - h[pp_c]) * vp - h[pm_c] * vm)).sum(axis=1) / beta
        
        # Gradient step with projection
        h = project_sum(h - lr * G, target)
        
        it += 1
        # Adapt lr and beta
        if it % 100 == 0:
            lr *= 0.97
            beta = min(beta_max, beta * 1.05)
        
        if it % 500 == 0 and time.time() - last_improve > 120:
            # No improvement for 2 min, perturb
            h = project_sum(h + rng.randn(N) * 0.05, target)
    
    return best_h, best_c5

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def precompute_idx(N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    return vp, vm, pp_c, pm_c

def compute_D(h, N, idx):
    vp, vm, pp_c, pm_c = idx
    return -(1.0 - h[pp_c]) * vp.astype(np.float64) + h[pm_c] * vm.astype(np.float64)

def fix_binary(h, N, rng):
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 != nh:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if n1 > nh:
            h[rng.choice(ones, n1 - nh, replace=False)] = 0
        else:
            h[rng.choice(zeros, nh - n1, replace=False)] = 1
    return h

def binary_search(h, N, deadline, max_sweeps, T=0.0, rng=None, sample_frac=1.0):
    h = fix_binary(h, N, rng)
    idx = precompute_idx(N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        D = compute_D(h, N, idx)
        if sample_frac < 1.0 and len(ones) > 30:
            n_s = max(30, int(len(ones) * sample_frac))
            p_list = rng.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = corr + D[p]
            cm = (base[None, :] - D[zeros]).max(axis=1)
            mi = int(np.argmin(cm))
            if cm[mi] < best_new_P:
                best_new_P = float(cm[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None:
            accept = (best_new_P < P)
            if not accept and T > 0:
                accept = rng.random() < np.exp(-min(50.0, (best_new_P - P) / T))
            if accept:
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
        elif T > 0:
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        else:
            break
    return best_h, best_c * (2.0 / N)

def softmax_opt(h, N, deadline, beta0=25, lr0=0.15, rng=None):
    h = h.astype(float).copy()
    target = N / 2.0
    idx = precompute_idx(N)
    vp, vm, pp_c, pm_c = idx
    best_c5 = compute_c5(h, N)
    best_h = h.copy()
    beta = beta0
    lr = lr0
    last_imp = time.time()
    it = 0
    while time.time() < deadline:
        corr = fft_corr(h)
        c5 = float(corr.max()) * (2.0 / N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_imp = time.time()
        cs = corr - corr.max()
        p = np.exp(beta * cs)
        p /= p.sum()
        G = (p[None, :] * ((1.0 - h[pp_c]) * vp - h[pm_c] * vm)).sum(axis=1) / beta
        h = project_sum(h - lr * G, target)
        it += 1
        if it % 200 == 0:
            lr *= 0.975
            beta = min(300.0, beta * 1.04)
        if it % 2000 == 0 and time.time() - last_imp > 90:
            h = project_sum(h + rng.randn(N) * 0.04, target)
            last_imp = time.time()
    return best_h, best_c5
