import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr_values(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_c5(h, T):
    cv = get_corr_values(h)
    m = np.max(cv)
    val = m + np.log(np.sum(np.exp(T * (cv - m)))) / T
    return val

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_c5(x, T)
    def con_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

def slsqp_active(h, top_k, n, maxiter=2000):

    cv = get_corr_values(h)
    idx = np.argsort(cv)[-top_k:]

    def obj(x):
        t = 0.0
        for k in idx:
            # Compute correlation at lag k
            # k is index in full correlation, convert to lag
            lag = k - (n - 1)
            v = np.correlate(x, 1.0 - x, mode='full')[k] * dx
            t = max(t, v)
        return t
    
    # Actually this won't work well because obj is not smooth

        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        vals = corr[idx]
        m = np.max(vals)
        T_act = 5000
        return m + np.log(np.sum(np.exp(T_act * (vals - m)))) / T_act
    
    def con_eq(x):
        return x.sum() - n / 2.0
    
    try:
        res = minimize(obj_smooth, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

def coord_opt_ternary(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr_base + m1 * coefs)
        f2 = np.max(corr_base + m2 * coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_ternary(best_h, n, i, niter=40)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

        p = np.zeros(n)
        p[mk] = (0.02 + 0.09 * cool) * np.random.randn(mk.sum())
        return p
    elif mode == 3:
        x = np.random.randn(n)
        spec = np.fft.rfft(x)
        spec[3:] = 0
        return np.fft.irfft(spec, n=n) * (0.02 + 0.05 * cool)
    elif mode == 4:
        a, b = np.random.choice(n, 2, replace=False)
        bl = np.random.randint(2, max(3, n // 10))
        p = np.zeros(n)
        p[a:min(a + bl, n)] = (p[a:min(a + bl, n)] + 0)
        return p
    elif mode == 5:
        return (0.005 + 0.025 * cool) * np.random.randn(n)
    elif mode == 6:
        f1 = np.random.randint(1, 6)
        f2 = np.random.randint(6, 15)
        return (0.01 + 0.03 * cool) * (
            np.sin(2 * np.pi * f1 * np.arange(n) / n) +
            np.sin(2 * np.pi * f2 * np.arange(n) / n + 0.5))
    else:
        bl = np.random.randint(3, max(4, n // 6))
        st = np.random.randint(0, n - bl)
        p = np.zeros(n)
        p[st:st + bl] = (0.03 + 0.08 * cool) * np.random.randn(bl)
        return p

def perturb_block(n, it):
    cool = 1.0 - 0.85 * (it / max(it + 1, 70))
    a, b = np.random.choice(n, 2, replace=False)
    bl = np.random.randint(2, max(3, n // 10))
    p = np.zeros(n)
    p[a:min(a + bl, n)] = (0.02 + 0.06 * cool) * np.random.randn(min(bl, n - a))
    return p

def run(seed=42, budget_s=1100, **kwargs):
    global _T_cur, _act_idx
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Warmup SLSQP with increasing temperature
    for T in [50, 300, 2000, 15000, 100000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        _T_cur = T
        hc = slsqp_solve(best_h, obj_smooth, n, 2500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Active-set focused optimization
    if time.time() - t0 < budget_s * 0.52:
        for tk in [5, 10, 20, 40, 80]:
            if time.time() - t0 > budget_s * 0.52:
                break
            cv = get_corr(best_h)
            _act_idx = np.argsort(cv)[-tk:]
            hc = slsqp_solve(best_h, obj_active, n, 2000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 3: Basin hopping with diverse perturbations
    wh = best_h.copy()
    wc = best_c5
    n_bh = 75
    for it in range(n_bh):
        if time.time() - t0 > budget_s * 0.55:
            break
        cool = 1.0 - 0.85 * (it / n_bh)
        mod = it % 8
        if mod == 4:
            p = perturb_block(n, it)
        else:
            p = perturb_gen(n, it, mod)
        hp = project_h(wh + p)
        _T_cur = 400 + 12000 * cool
        hc = slsqp_solve(hp, obj_smooth, n, 400)
        c5v = compute_c5(hc)
        temp = max(0.0005 * cool + 0.00004, 1e-9)
        if c5v < wc or np.random.rand() < np.exp(-(c5v - wc) / temp):
            wh = hc
            wc = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if wc < best_c5:
        best_c5 = wc
        best_h = wh

    # Phase 4: High-T full SLSQP
    for T in [200000, 800000, 3000000]:
        if time.time() - t0 > budget_s * 0.66:
            break
        _T_cur = T
        hc = slsqp_solve(best_h, obj_smooth, n, 3500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Second active-set pass
    if time.time() - t0 < budget_s * 0.72:
        cv = get_corr(best_h)
        _act_idx = np.argsort(cv)[-10:]
        hc = slsqp_solve(best_h, obj_active, n, 2500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Fine coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.88, 120)

    # Phase 7: Fine basin hopping
    wh2 = best_h.copy()
    wc2 = best_c5
    for it in range(25):
        if time.time() - t0 > budget_s * 0.94:
            break
        cool = 1.0 - 0.5 * (it / 25.0)
        hp = project_h(wh2 + (0.003 + 0.012 * cool) * np.random.randn(n))
        _T_cur = 5000 + 60000 * cool
        hc = slsqp_solve(hp, obj_smooth, n, 500)
        c5v = compute_c5(hc)
        temp = max(0.0002 * cool + 0.00002, 1e-9)
        if c5v < wc2 or np.random.rand() < np.exp(-(c5v - wc2) / temp):
            wh2 = hc
            wc2 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if wc2 < best_c5:
        best_c5 = wc2
        best_h = wh2

    # Phase 8: Final ultra-fine polish
    if time.time() - t0 < budget_s * 0.98:
        _T_cur = 8000000
        hc = slsqp_solve(best_h, obj_smooth, n, 5000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if time.time() - t0 < budget_s * 0.995:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, 40)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
