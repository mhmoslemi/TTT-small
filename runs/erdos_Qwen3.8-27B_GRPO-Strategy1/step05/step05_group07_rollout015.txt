import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (tgt / s)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = m.sum()
        if cnt > 0:
            h[m] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    val = m + np.log(np.sum(np.exp(T * (c - m)))) / T
    return val * 2.0 / n

def soft_grad(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    ar = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        i1 = (ar + off >= 0) & (ar + off < n)
        g[i1] += wk * (1.0 - h[ar[i1] + off])
        i2 = (ar - off >= 0) & (ar - off < n)
        g[i2] -= wk * h[ar[i2] - off]
    return g * (2.0 / n)

def cdesc(h, n, t_end, max_sweeps=300, var_w=0.0):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        c = np.correlate(best, 1.0 - best, mode='full')
        order = np.random.permutation(n)
        for i in order:
            old = best[i]
            off = np.arange(mf) - (n - 1)
            eff = np.zeros(mf)
            m1 = (i + off >= 0) & (i + off < n)
            if m1.any():
                eff[m1] += 1.0 - best[i + off[m1]]
            m2 = (i - off >= 0) & (i - off < n)
            if m2.any():
                eff[m2] -= best[i - off[m2]]
            emax = np.max(np.abs(eff))
            if emax < 1e-14:
                continue
            lo, hi = -old, 1.0 - old
            if var_w > 0:
                cm = c.mean()
                def f1d_v(d):
                    cv = c + d * eff
                    return cv.max() + var_w * np.var(cv)
            else:
                def f1d(d):
                    return np.max(c + d * eff)
            fn = f1d_v if var_w > 0 else f1d
            r = minimize_scalar(fn, bounds=(lo, hi), method='bounded',
                                options={'xatol': 1e-10})
            delta = r.x
            if abs(delta) < 1e-12:

            c += delta * eff
            improved = True
        best = proj_h(best, n)
        c5n = compute_c5(best)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        if not improved:

    return best, best_c5

def basin_hop_cd(h, n, t_end, n_iters=80):

    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (tgt / s)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = m.sum()
        if cnt > 0:
            h[m] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n

def soft_grad(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    ar = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        i1 = (ar + off >= 0) & (ar + off < n)
        g[i1] += wk * (1.0 - h[ar[i1] + off])
        i2 = (ar - off >= 0) & (ar - off < n)
        g[i2] -= wk * h[ar[i2] - off]
    return g * (2.0 / n)

def cdesc(h, n, t_end, max_sweeps=300):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    off_arr = np.arange(mf) - (n - 1)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        c = np.correlate(best, 1.0 - best, mode='full')
        order = np.random.permutation(n)
        for i in order:
            old = best[i]
            eff = np.zeros(mf)
            m1 = (i + off_arr >= 0) & (i + off_arr < n)
            if m1.any():
                eff[m1] += 1.0 - best[i + off_arr[m1]]
            m2 = (i - off_arr >= 0) & (i - off_arr < n)
            if m2.any():
                eff[m2] -= best[i - off_arr[m2]]
            if np.max(np.abs(eff)) < 1e-14:
                continue
            lo, hi = -old, 1.0 - old
            def f1d(d, _c=c, _e=eff):
                return np.max(_c + d * _e)
            r = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                options={'xatol': 1e-11})
            delta = r.x
            if abs(delta) < 1e-13:
                continue
            best[i] = old + delta
            c += delta * eff
            improved = True
        best = proj_h(best, n)
        c5n = compute_c5(best)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        if not improved:
            break
    return best, best_c5

def basin_hop_cd(h, n, t_end, n_iters=100):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - it / n_iters
        mod = it % 4
        if mod == 0:
            pert = (0.03 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(2, 20)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.3 + 0.4 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.2 * cool) * np.random.randn(mask.sum())
        else:
            nflip = max(2, int(0.08 * cool * n))
            idx = np.random.choice(n, nflip, replace=False)
            pert = np.zeros(n)
            pert[idx] = np.where(work[idx] > 0.5, -0.8, 0.8)
        hp = proj_h(work + pert, n)
        hp, c5t = cdesc(hp, n, min(t_end, time.time() + 2.5), max_sweeps=25)
        temp = 0.0012 * cool + 0.0001
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = hp
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = hp.copy()
    return best, best_c5

def lbfgs_step(h, T, n, iters=300):
    try:
        r = minimize(h, soft_obj, soft_grad, method='L-BFGS-B',
                     args=(T,), bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

    # Phase 2: Exact coordinate descent - main workhorse (30%)
    t_phase = t0 + budget_s * 0.45
