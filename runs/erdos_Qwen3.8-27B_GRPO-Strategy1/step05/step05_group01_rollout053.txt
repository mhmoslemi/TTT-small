        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h += room * (diff / tot)
    else:
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_solve(h, T, n, maxiter=600):
    """Solve with LSE-smoothed objective using SLSQP."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if T < 1:
            return m * (2.0/n)
        return (m + np.log(np.sum(np.exp(T*(corr-m))))/T) * (2.0/n)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun': lambda x: x.sum()-n/2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

                if 0 <= jm < n: c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i]*coefs
            res = minimize_scalar(lambda t: np.max(base + t*coefs),
                                  bounds=(0,1), method='bounded',

                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5
