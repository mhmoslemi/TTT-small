def slsqp_step(h, T, n, maxiter=2000):
    def obj(x):
        corr = get_corr(x)
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T*(corr-m))))/T
        return val * (2.0/n)
    def con(x):
        return x.sum() - n/2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter':maxiter,'ftol':1e-15})
        return project_h(np.clip(res.x,0,1))
    except:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_fast(h, n, i, niter=40):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    j = np.arange(2 * n - 1)
    m = j - (n - 1)
    v1 = (i - m >= 0) & (i - m < n)
    v2 = (i + m >= 0) & (i + m < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[i - m[v1]])
    coefs[v2] -= h[i + m[v2]]
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_sweep(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        idx = np.random.permutation(n)
        for i in idx:
            if time.time() > t_end:
                return best_h, best_c5
            nv = coord_opt_fast(best_h, n, i)
            if abs(nv - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = nv
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_step(h, T, n, maxiter=2000, ftol=1e-15):
    def obj(x):
        c = get_corr(x)
        mg = np.max(c)
        return (mg + np.log(np.sum(np.exp(T * (c - mg)))) / T) * (2.0 / n)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_direction(h, T=2000):
    n = len(h)
    corr = get_corr(h)
    mg = np.max(corr)
    w = np.exp(T * (corr - mg))
    w /= w.sum()
    first = np.zeros(n)
    second = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if wk < 1e-13:
            continue
        if k >= 0:
            first[:n - k] += wk * (1.0 - h[k:n])
            second[k:] += wk * h[:n - k]
        else:
            kn = -k
            first[kn:] += wk * (1.0 - h[:n - kn])
            second[:n - kn] += wk * h[kn:]
    return (first - second) * (2.0 / n)

def fourier_pert(h, n, scale):
    pert = np.zeros(n)
    nmod = np.random.randint(1, 4)
    for _ in range(nmod):
        fr = np.random.randint(1, 8)
        ph = np.random.uniform(0, 6.2832)
        amp = scale * np.random.uniform(0.5, 1.5)
        pert += amp * np.sin(6.2832 * fr * np.arange(n) / n + ph)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP warm-up
    t1 = t0 + budget_s * 0.06
    for T in [200, 2000, 20000]:
        if time.time() > t1:
            break
        hc = slsqp_step(best_h, T, n, maxiter=4000)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc

    # Phase 2: Intensive coordinate descent
    t2 = t0 + budget_s * 0.28
    best_h, best_c5 = coord_sweep(best_h, n, t2, max_sweeps=800)

    # Phase 3: Basin hopping with diverse perturbations
    t3 = t0 + budget_s * 0.62
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 300
    for it in range(n_iter):
        if time.time() > t3:
            break
        cool = 1.0 - (it / n_iter)
        step = 0.008 + 0.05 * cool
        mod = it % 6
        if mod == 0:
            h_p = project_h(work_h + step * np.random.randn(n))
        elif mod == 1:
            h_p = project_h(fourier_pert(work_h, n, step))
        elif mod == 2:
            h_p = work_h.copy()
            blk = np.random.randint(4, 18)
            st = np.random.randint(0, max(1, n - blk))
            h_p[st:st + blk] += step * np.random.randn(blk)
            h_p = project_h(h_p)
        elif mod == 3:
            gx = subgrad_direction(work_h, T=int(500 + 5000 * cool))
            h_p = project_h(work_h - step * 2.0 * gx)
        elif mod == 4:
            h_p = work_h.copy()
            a, b = np.random.choice(n, 2, replace=False)
            d = np.random.uniform(-0.08, 0.08)
            h_p[a] += d
            h_p[b] -= d
            h_p = project_h(h_p)
        else:
            h_p = project_h(work_h + step * 0.5 * np.sign(np.random.randn(n)))

        T_b = int(300 + 8000 * cool)
        hc = slsqp_step(h_p, T_b, n, maxiter=400, ftol=1e-13)
        cv = compute_c5(hc)
        hc, cv2 = coord_sweep(hc, n, t3, max_sweeps=3)
        if cv2 < cv:
            cv = cv2
        temp = 0.0004 * cool + 0.00002
        if cv < work_c5 or np.random.rand() < np.exp(-(cv - work_c5) / temp):
            work_h = hc
            work_c5 = cv
            if cv < best_c5:
                best_c5 = cv
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Try n=200
    t4 = t0 + budget_s * 0.76
    if elapsed() < budget_s * 0.60:
        n2 = 200
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_h(h2)
        c5_2 = compute_c5(h2)
        for T in [1000, 10000]:
            if time.time() > t4 - 15:
                break
            h2c = slsqp_step(h2, T, n2, maxiter=3000)
            cv2 = compute_c5(h2c)
            if cv2 < c5_2:
                c5_2 = cv2
                h2 = h2c
        h2, c5_2 = coord_sweep(h2, n2, t4, max_sweeps=300)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            n = n2

    # Phase 5: High-precision coordinate descent
    t5 = t0 + budget_s * 0.92
    best_h, best_c5 = coord_sweep(best_h, n, t5, max_sweeps=1000)

    # Phase 6: Final SLSQP polish + coordinate sweep
    t6 = t0 + budget_s * 0.97
    for T in [100000, 500000]:
        if time.time() > t6:
            break
        hc = slsqp_step(best_h, T, n, maxiter=5000, ftol=1e-16)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc
    best_h, best_c5 = coord_sweep(best_h, n, t0 + budget_s * 0.99, max_sweeps=500)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
