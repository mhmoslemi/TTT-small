    p = np.exp(corr_clipped - np.max(corr_clipped))

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h += room * min(d / tr, 1.0)
        else:
            tv = h.sum()
            if tv < 1e-15:
                break
            h -= h * min(-d / tv, 1.0)
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        rem = abs(d)
        for i in idx:
            if d > 0:
                a = min(rem, 1.0 - h[i])
                h[i] += a
            else:
                a = min(rem, h[i])
                h[i] -= a
            rem -= a
            if rem < 1e-15:
                break
    return h

def compute_lin(i, h, n):
    old = h[i]
    lin = np.zeros(2 * n - 1)
    s = np.arange(1, n)
    # mp indices: n-1+s = n+s-1 (indices n to 2n-2)
    mp = np.zeros(n - 1)
    m1 = s <= i
    m2 = (i + s) < n
    mp[m1] += (1.0 - h[i - s[m1]])
    mp[m2] -= h[i + s[m2]]
    lin[n:] = mp
    # mn indices: n-1-s (indices n-2 down to 0), stored reversed
    mn = np.zeros(n - 1)
    mn[m1] -= h[i - s[m1]]
    mn[m2] += (1.0 - h[i + s[m2]])
    lin[:n - 1] = mn[::-1]
    lin[n - 1] = 1.0 - 2.0 * old
    return lin

def fast_coord_desc(h, n, t_end, max_sweeps=20000, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    bh = h.copy()
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    bc5 = float(np.max(corr) * dx)
    s_arr = np.arange(1, n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return bh, bc5
            old = bh[i]
            lin = np.zeros(2 * n - 1)
            m1 = s_arr <= i
            m2 = (i + s_arr) < n
            if m1.any():
                lin[n:] += (1.0 - bh[i - s_arr[m1]]) * m1
                lin[:n-1] -= bh[i - s_arr[m1]][::-1] * m1[::-1]
            if m2.any():
                lin[n:] -= bh[i + s_arr[m2]] * m2
                lin[:n-1] += (1.0 - bh[i + s_arr[m2]])[::-1] * m2[::-1]
            lin[n - 1] = 1.0 - 2.0 * old
            lo = 0.0
            hi = 1.0
            base_corr = corr
            b_lin = lin
            n1 = n - 1
            def f1d(v):
                d = v - old
                val = base_corr + d * b_lin
                val[n1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-12, 'maxiter': 50})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-14:
                continue
            bh[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < bc5 - 1e-16:
                bc5 = c5_new
                improved = True
            else:
                bh[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    bh = project_h(bh, n)
    bc5 = compute_c5(bh)
    return bh, bc5

def subgrad_h(h, n, t_end, max_iters=5000, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    hc = h.copy()
    mom = np.zeros(n)
    beta = 0.8
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = min(1.0, it / max_iters)
        T = 30.0 * 50.0 ** frac
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        Tc = np.clip(T * corr, -500.0, 500.0)
        Tc -= Tc.max()
        p = np.exp(Tc)
        p /= p.sum()
        grad = np.zeros(n)
        grad += p[n - 1] * (1.0 - 2.0 * hc)
        for s in range(1, n):
            wp = p[n - 1 + s]
            wn = p[n - 1 - s]
            if wp < 1e-15 and wn < 1e-15:
                continue
            if wp > 1e-15:
                if s > 0:
                    grad[s:] += wp * (1.0 - hc[:n - s])
                if s < n:
                    grad[:n - s] -= wp * hc[s:]
            if wn > 1e-15:
                if s < n:
                    grad[:n - s] += wn * (1.0 - hc[s:])
                if s > 0:
                    grad[s:] -= wn * hc[:n - s]
        mom = beta * mom + dx * grad
        gm = np.max(np.abs(mom))
        if gm < 1e-14:
            break
        alpha = 0.3 / (gm + 1e-14)
        hn = project_h(hc - alpha * mom, n)
        c5n = compute_c5(hn)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = hn.copy()
        hc = hn
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=150, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.01 + 0.05 * cool
        mode = hop % 5
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = step * 2 * rng.randn(int(mask.sum()))
        elif mode == 3:
            idx = rng.choice(n, size=max(4, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * rng.randn(len(idx))
        else:
            pert = step * 0.4 * rng.randn(n)
        hp = project_h(cur_h + pert, n)
        hop_end = min(t_end, time.time() + max(1.5, 5.0 * cool))
        ho, c5o = fast_coord_desc(hp, n, hop_end, max_sweeps=200, seed=seed + hop)
        temp = 0.003 * cool + 0.0003
        if c5o < cur_c5 or (c5o > cur_c5 and rng.rand() < np.exp(-(c5o - cur_c5) / temp)):
            cur_h = ho
            cur_c5 = c5o
            if c5o < best_c5:
                best_c5 = c5o
                best_h = ho.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.concatenate([np.ones(n_init // 2), np.zeros(n_init // 2)])
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent
    t1 = t0 + budget_s * 0.30
    best_h, best_c5 = fast_coord_desc(best_h, n, t1, max_sweeps=50000, seed=seed)

    # Phase 2: Subgradient + coord polish
    if time.time() < t0 + budget_s * 0.48:
        hg, c5g = subgrad_h(best_h, n, t0 + budget_s * 0.44, max_iters=3000)
        if c5g < best_c5:
            best_h, best_c5 = hg, c5g
        best_h, best_c5 = fast_coord_desc(best_h, n, t0 + budget_s * 0.50, max_sweeps=20000, seed=seed + 1)

    # Phase 3: Basin hopping
    if time.time() < t0 + budget_s * 0.75:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.75, n_hops=300, seed=seed + 10)

    # Phase 4: Final heavy coordinate descent
    if time.time() < t0 + budget_s * 0.92:
        best_h, best_c5 = fast_coord_desc(best_h, n, t0 + budget_s * 0.92, max_sweeps=50000, seed=seed + 100)

    # Phase 5: Final subgradient nudge
    if time.time() < t0 + budget_s * 0.96:
        hg2, c5g2 = subgrad_h(best_h, n, t0 + budget_s * 0.95, max_iters=1000)
        if c5g2 < best_c5:
            best_h, best_c5 = hg2, c5g2

    # Phase 6: Last polish
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = fast_coord_desc(best_h, n, t0 + budget_s * 0.995, max_sweeps=20000, seed=seed + 200)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
