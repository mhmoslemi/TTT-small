def minimize_max_linear(a, b, lo, hi):
    # Minimize max_k(a[k] + b[k]*delta) over delta in [lo, hi]
    # a, b are 1D arrays, same length
    n = len(a)
    if n == 0: return 0.0
    # Start at lo, find which index is max
    def fval(d):
        return np.max(a + d*b)
    # Subgradient descent (projected)
    d = max(lo, min(hi, 0.0))
    for _ in range(200):
        val = a + d*b
        idx = np.argmax(val)
        grad = b[idx]
        if grad > 0:
            d = lo
            break
        elif grad < 0:
            d = hi
            break
        else:
            break
    # At the optimal point, the max is shared by at least 2 indices
    # Find the exact breakpoint
    # For large arrays, just do a few iterations of bisection
    lo2, hi2 = max(lo, d-1e-10), min(hi, d+1e-10)
    for _ in range(60):
        mid = (lo2+hi2)/2
        if fval(mid) <= fval(mid + (hi2-lo2)/2):
            hi2 = mid
        else:
            lo2 = mid + (hi2-lo2)/2
    return (lo2+hi2)/2

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        s = room.sum()
        if s < 1e-14:
            return h
        h = h + room * (diff / s)
    else:
        s = h.sum()
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_run(h, n, T, maxit=400, fttol=1e-14, pen_w=0.0, m_pen=5):
    dx = 2.0 / n
    def f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        v = m + np.log(np.sum(np.exp(T * (c - m)))) / T
        if pen_w > 0.0 and m_pen > 1:
            v += pen_w * np.var(np.sort(c)[-m_pen:])
        return v * dx
    try:
        r = minimize(f, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                     options={'maxiter': maxit, 'ftol': fttol})
        return project_h(np.clip(r.x, 0, 1))
    except Exception:
        return h.copy()

def coord_desc_fast(h, n, t_end, sweeps=200, ntop=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sw in range(sweeps):
        if time.time() > t_end:
            return best_h, best_c5
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        ti = np.argsort(corr)[-ntop:]
        ts = ti - (n - 1)
        tv = corr[ti].copy()
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            if time.time() > t_end:
                return best_h, best_c5
            co = np.zeros(ntop)
            for t in range(ntop):
                k = ts[t]
                jp = i + k
                jm = i - k
                co[t] = (1.0 - best_h[jp]) if 0 <= jp < n else 0.0
                co[t] -= best_h[jm] if 0 <= jm < n else 0.0
            old = best_h[i]
            # 1D minimization: min over delta of max(tv + delta*co)
            lo_d, hi_d = -old, 1.0 - old
            res = minimize_scalar(lambda d: np.max(tv + d * co),
                                  bounds=(lo_d, hi_d), method='bounded',
                                  options={'xatol': 1e-12})
            d_opt = res.x
            new_val = np.max(tv + d_opt * co)
            if new_val >= best_c5 / dx - 1e-12:
                continue
            ht = best_h.copy()
            ht[i] = old + d_opt
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                # Update corr incrementally
                delta = ht[i] - old
                for t in range(ntop):
                    k = ts[t]
                    jp = i + k
                    jm = i - k
                    tv[t] += delta * ((1.0 - ht[jp]) if 0 <= jp < n else 0.0)
                    tv[t] += delta * ht[jm] if 0 <= jm < n else 0.0
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def targeted_hue(h, n, t_end, n_iter=20):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(work_h, 1.0 - work_h, mode='full')
        m = corr.max()
        top_idx = np.where(corr > 0.95 * m)[0]
        shifts = top_idx - (n - 1)
        pert = np.zeros(n)
        for k in shifts[:15]:
            for i in range(n):
                jp = i + k
                jm = i - k
                term = 0.0
                if 0 <= jp < n:
                    term += (1.0 - work_h[jp])
                if 0 <= jm < n:
                    term -= work_h[jm]
                pert[i] += term
        norm = np.linalg.norm(pert)
        if norm < 1e-12:
            break
        alpha = 0.003 / (1.0 + 0.3 * it)
        h_try = project_h(work_h - alpha * pert / norm * n)
        c5t = compute_c5(h_try)
        if c5t < work_c5:
            work_h = h_try
            work_c5 = c5t
    return work_h, work_c5
