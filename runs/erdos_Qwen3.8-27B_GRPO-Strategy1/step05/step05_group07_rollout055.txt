import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

_T = [1.0]

def _sobj(x):
    T = _T[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def _sgrad(x):
    T = _T[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0 / n)

def lbfgs(h, T, n, iters=300):
    _T[0] = T
    try:
        r = minimize(h, _sobj, _sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h.copy()
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

_T = [1.0]

def _sobj(x):
    T = _T[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def _sgrad(x):
    T = _T[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        # Case 2: (1-h[j-s]) for j-s in [0,n), j from max(0,k-n+1) to min(n,k+1)
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            g[lo2:hi2] += wk * (1.0 - x[lo2 + k - n + 1:hi2 + k - n + 1])
        # Case 1: -h[j+s] for j+s in [0,n), j from max(0,n-1-k) to min(n,2n-k)
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - k)
        if lo1 < hi1:
            g[lo1:hi1] -= wk * x[lo1 + k - n + 1:hi1 + k - n + 1]
    return g * (2.0 / n)

def lbfgs(h, T, n, iters=300):
    _T[0] = T
    try:
        r = minimize(h, _sobj, _sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

def targeted_coord(h, n, t_end, sweeps=60, top_m=12):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    order = np.random.permutation(n)
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        top_idx = np.argsort(corr)[-top_m:]
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            s_off = i
            for k in range(mf):
                s = k - n + 1
                js = i + s
                if 0 <= js < n:
                    coef[k] -= best[js]
                jm = i - s
                if 0 <= jm < n:
                    coef[k] += (1.0 - best[jm])
            b_top = corr[top_idx] - old * coef[top_idx]
            c_top = coef[top_idx]
            cands = [0.0, 1.0]
            for a in range(top_m):
                for bb in range(a + 1, top_m):
                    denom = c_top[a] - c_top[bb]
                    if abs(denom) > 1e-14:
                        t = (b_top[bb] - b_top[a]) / denom
                        if -0.001 <= t <= 1.001:
                            cands.append(t)
            best_t, best_v = 0.0, np.inf
            for t in cands:
                t = float(np.clip(t, 0.0, 1.0))
                v = float(np.max(b_top + t * c_top))
                if v < best_v:
                    best_v, best_t = v, t
            if abs(best_t - old) > 1e-10:
                trial = best.copy()
                trial[i] = best_t
                trial = proj_h(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_c5 - 1e-15:
                    best, best_c5 = trial, c5t
                    imp = True
                    corr = np.correlate(best, 1.0 - best, mode='full')
                    top_idx = np.argsort(corr)[-top_m:]
        if not imp:
            break
    return best, best_c5

def full_coord(h, n, t_end, sweeps=30):

    for _ in range(sweeps):

            break
        imp = False
        corr = np.correlate(best, 1.0 - best, mode='full')

                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):

                js = i + s
                if 0 <= js < n:
                    coef[k] -= best[js]
                jm = i - s
                if 0 <= jm < n:
                    coef[k] += (1.0 - best[jm])
            base = corr - old * coef
            cands = [0.0, 1.0]
            for a in range(mf):
                for bb in range(a + 1, mf):
                    denom = coef[a] - coef[bb]
                    if abs(denom) > 1e-14:
                        t = (base[bb] - base[a]) / denom
                        if -0.001 <= t <= 1.001:
                            cands.append(t)
            best_t, best_v = old, np.max(base + old * coef)
            for t in cands:
                t = float(np.clip(t, 0.0, 1.0))
                v = float(np.max(base + t * coef))
                if v < best_v:
                    best_v, best_t = v, t
            if abs(best_t - old) > 1e-10:
                trial = best.copy()
                trial[i] = best_t
                trial = proj_h(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_c5 - 1e-15:
                    best, best_c5 = trial, c5t
                    imp = True
                    corr = np.correlate(best, 1.0 - best, mode='full')

def bin_sa(h, n, t_end, max_iters=500000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    if len(ones) > n // 2:
        h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T = 0.02
    for it in range(max_iters):
        if time.time() > t_end:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or h_b[i] != h_b[j]:
            continue
        og = h_b[i].copy()
        h_b[i], h_b[j] = h_b[j], h_b[i]
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            corr = nc
            if nc5 < best_c5:
                best_c5, best_h = nc5, h_b.copy()
        else:
            h_b[i], h_b[j] = og
        T = max(0.0003, T * 0.99997)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=80):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 3
        if mod == 0:
            pert = (0.03 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(2, 20)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        else:
            mask = np.random.random(n) < (0.3 + 0.4 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.2 * cool) * np.random.randn(mask.sum())
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 4000.0 * cool
        ht = lbfgs(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.001 * cool + 0.0001
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work, work_c5 = ht, c5t
            if c5t < best_c5:
                best, best_c5 = ht, c5t
    return best, best_c5

def make_thue_morse(n):
    tm = np.zeros(n)
    for i in range(1, n):
        if i % 2 == 0:
            tm[i] = tm[i // 2]
        else:
            tm[i] = 1.0 - tm[i // 2]
    return tm

def run(seed=42, budget_s=1100, **kwargs):

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
