def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            old = best_h[j]
            best_h[j] = 0.0
            c0 = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_h[j] = 1.0
            c1 = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_h[j] = old
            b = c1 - c0
            a, bb = 0.0, 1.0
            for _ in range(50):
                m1 = a + (bb - a) / 3.0
                m2 = bb - (bb - a) / 3.0
                if np.max(c0 + m1 * b) < np.max(c0 + m2 * b):
                    bb = m2
                else:
                    a = m1
            t_opt = (a + bb) / 2.0
            if abs(t_opt - old) > 1e-13:
                best_h[j] = t_opt
                best_h = project_h(best_h, n)
                c5_new = compute_c5(best_h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def struct_perturb(h, n):
    h_try = h.copy()
    r = np.random.random()
    if r < 0.3:
        i, j = np.random.choice(n, 2, replace=False)
        h_try[i], h_try[j] = h_try[j], h_try[i]
    elif r < 0.6:
        i = np.random.randint(0, max(1, n - 10))
        ln = np.random.randint(5, min(20, n - 2))
        h_try[i:i + ln] = h_try[i:i + ln][::-1]
    else:
        i = np.random.randint(0, max(1, n - 10))
        ln = np.random.randint(3, min(15, n - 2))
        sh = np.random.randint(1, 5)
        blk = h_try[i:i + ln].copy()
        h_try[i:i + ln] = np.roll(blk, sh)
    return project_h(h_try, n)

def basin_hop(h, n, t_end, n_iter=40):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.05, 1.0 - it / n_iter)
        if np.random.random() < 0.4:
            amp = 0.008 + 0.04 * cool
            h_pert = project_h(work_h + amp * np.random.randn(n), n)
        else:
            h_pert = struct_perturb(work_h, n)
            h_pert = project_h(h_pert + 0.02 * np.random.randn(n), n)
        T = 200 + 2000 * cool
        hc = slsqp_smooth(h_pert, n, T, maxiter=250)
        c5v = compute_c5(hc)
        temp = 0.0004 * cool + 0.00003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
        except Exception:
            best_h = np.zeros(160)
            best_h[:80] = 1.0
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    alts = []
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    alts.append(project_h(tm, n))
    for _ in range(3):
        rb = np.zeros(n)
        rb[np.random.choice(n, n // 2, replace=False)] = 1.0
        alts.append(rb)
    for alt in alts:
        if elapsed() > 20:
            break
        hc = slsqp_smooth(alt, n, 5000, maxiter=150)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    for T in [30000, 10000, 4000, 1500, 600, 200, 80, 30, 10]:
        if elapsed() > budget_s * 0.12:
            break
        hc = slsqp_smooth(best_h, n, T, maxiter=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.22, max_sweeps=300)

    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.52, n_iter=45)

    for T in [3000, 500, 100, 30, 10]:
        if elapsed() > budget_s * 0.68:
            break
        hc = slsqp_smooth(best_h, n, T, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
        t_sub = min(t0 + elapsed() + 15, t0 + budget_s * 0.68)
        best_h, best_c5 = coord_descent(best_h, n, t_sub, max_sweeps=80)

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.84, max_sweeps=1000)

    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.92, n_iter=15)

    if elapsed() < budget_s * 0.95:
        hc = slsqp_smooth(best_h, n, 30, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=300)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
