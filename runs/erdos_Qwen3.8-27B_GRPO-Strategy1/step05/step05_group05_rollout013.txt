```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_corr_obj(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    weights = np.exp(T * (corr - m))
    weights /= weights.sum()
    # Now compute weighted sum of gradients of each corr[k] w.r.t. h
```

Let me verify the gradient formula once more:

For k ≥ 0: corr_k = Σ_{s=0}^{n-1-k} h[s+k]·(1-h[s])
∂/∂h[m]: 
- Term h[s+k]: nonzero when s+k=m, i.e., s=m-k. Valid when 0≤m-k≤n-1-k, i.e., k≤m≤n-1. Contribution: (1-h[s]) = (1-h[m-k])
- Term (1-h[s]): nonzero when s=m. Valid when 0≤m≤n-1-k. Contribution: -h[s+k] = -h[m+k]

So grad[m] += wk * (1-h[m-k]) for k≤m≤n-1
           -= wk * h[m+k] for 0≤m≤n-1-k

In code:
grad[k:] += wk * (1 - h[:n-k])  → indices k to n-1 get (1-h[0] to 1-h[n-1-k]) ✓
grad[:n-k] -= wk * h[k:]  → indices 0 to n-1-k get -h[k] to -h[n-1] ✓

For k < 0, j = -k: corr_{-j} = Σ_{s=0}^{n-1-j} h[s]·(1-h[s+j])
∂/∂h[m]:
- Term h[s]: nonzero when s=m. Valid when 0≤m≤n-1-j. Contribution: (1-h[m+j])
- Term (1-h[s+j]): nonzero when s+j=m, i.e., s=m-j. Valid when 0≤m-j≤n-1-j, i.e., j≤m≤n-1. Contribution: -h[m-j]

So grad[m] += wk * (1-h[m+j]) for 0≤m≤n-1-j
           -= wk * h[m-j] for j≤m≤n-1

In code:
grad[:n-j] += wk * (1 - h[j:])  → indices 0 to n-1-j get (1-h[j] to 1-h[n-1]) ✓
grad[j:] -= wk * h[:n-j]  → indices j to n-1 get -h[0] to -h[n-1-j] ✓

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def grad_smooth(h, T):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-13:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[k:] += wk * (1.0 - h[:n - k])
                grad[:n - k] -= wk * h[k:]
        else:
            j = -k
            grad[:n - j] += wk * (1.0 - h[j:])
            grad[j:] -= wk * h[:n - j]
    return grad

def grad_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            g[:] = (1.0 - 2.0 * h)
        else:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
    else:
        j = -k
        g[:n - j] += (1.0 - h[j:])
        g[j:] -= h[:n - j]
    return g

def gd_loop(h, T, n, max_iter, lr0, t_end, use_subgrad_mix=0, perturb_iv=250):
    h_cur = h.copy()
    lr = lr0
    best_h = h_cur.copy()
    best_obj = smooth_obj(get_corr(best_h), T)
    stall = 0
    it = 0
    while it < max_iter:
        if time.time() > t_end:
            break
        it += 1
        corr = get_corr(h_cur)
        obj = smooth_obj(corr, T)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
        if stall > 400:
            break
        if use_subgrad_mix > 0 and it % use_subgrad_mix == 0:
            k_star = int(np.argmax(corr)) - (n - 1)
            g = grad_single(h_cur, k_star)
        else:
            g = grad_smooth(h_cur, T)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn
        accepted = False
        for _ in range(25):
            h_try = project_h(h_cur - lr * g)
            obj_try = smooth_obj(get_corr(h_try), T)
            if obj_try < obj * (1 + 1e-15):
                h_cur = h_try
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            lr *= 0.4
        if it % 60 == 0:
            lr = min(lr * 1.8, lr0)
        if perturb_iv > 0 and it % perturb_iv == perturb_iv // 2:
            h_cur = project_h(h_cur + 0.008 * np.random.randn(n))
            lr = lr0
    return best_h, best_obj

def coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(55):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=40):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            nv = coord_opt(best_h, n, i)
            if abs(nv - best_h[i]) < 1e-12:
                continue
            ht = best_h.copy()
            ht[i] = nv
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_wrap(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_obj(get_corr(x), T)
    def con_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: GD with T-annealing (30%)
    t1 = t0 + budget_s * 0.30
    for T in [20, 80, 400, 2000, 10000, 50000, 200000, 500000]:
        if elapsed() > budget_s * 0.28:
            break
        hm, _ = gd_loop(best_h, T, n, 8000, 0.04, t1, use_subgrad_mix=0, perturb_iv=400)
        c5m = compute_c5(hm)
        if c5m < best_c5:
            best_c5 = c5m
            best_h = hm

    # Phase 2: Coordinate descent (15%)
    t2 = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=25)

    # Phase 3: Basin hopping (30%)
    t3 = t0 + budget_s * 0.75
    wh = best_h.copy()
    wc5 = best_c5
    nbh = 100
    for it in range(nbh):
        if elapsed() > budget_s * 0.73:
            break
        cool = 1.0 - 0.95 * (it / float(nbh))
        mod = it % 6
        if mod == 0:
            pert = (0.01 + 0.09 * cool) * np.random.randn(n)
        elif mod == 1:
            fq = np.random.randint(1, 16)
            ph = np.random.uniform(0, 6.2832)
            pert = (0.008 + 0.07 * cool) * np.sin(6.2832 * fq * np.arange(n) / n + ph)
        elif mod == 2:
            mk = np.random.random(n) < 0.18
            pert = np.zeros(n)
            pert[mk] = (0.03 + 0.14 * cool) * np.random.randn(max(mk.sum(), 1))
        elif mod == 3:
            blk = max(1, n // 7)
            b = np.random.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            seg = min(blk, n - b * blk)
            pert[b * blk:b * blk + seg] = (0.02 + 0.07 * cool) * np.random.randn(seg)
        elif mod == 4:
            pert = np.zeros(n)
            for _ in range(3):
                a, b2 = np.random.choice(n, 2, replace=False)
                d = (best_h[b2] - best_h[a]) * (0.3 + 0.7 * cool)
                pert[a] += d
                pert[b2] -= d
        else:
            pert = (0.006 + 0.045 * cool) * np.random.randn(n)
        hp = project_h(wh + pert)
        Tb = 200 + 60000 * cool
        hn, _ = gd_loop(hp, Tb, n, 2000, 0.03, t3, use_subgrad_mix=30, perturb_iv=0)
        c5n = compute_c5(hn)
        temp = 0.001 * cool + 0.00006
        if c5n < wc5 or np.random.rand() < np.exp(-(c5n - wc5) / temp):
            wh = hn
            wc5 = c5n
        if wc5 < best_c5:
            best_c5 = wc5
            best_h = wh.copy()

    # Phase 4: Final CD + GD + SLSQP polish (25%)
    t4a = t0 + budget_s * 0.88
    best_h, best_c5 = coord_descent(best_h, n, t4a, max_sweeps=30)
    if elapsed() < budget_s * 0.93:
        hg, _ = gd_loop(best_h, 300000, n, 3000, 0.01, t0 + budget_s * 0.94, use_subgrad_mix=20, perturb_iv=0)
        c5g = compute_c5(hg)
        if c5g < best_c5:
            best_c5 = c5g
            best_h = hg
    if elapsed() < budget_s * 0.96:
        hs = slsqp_wrap(best_h, 200000, n, maxiter=2500)
        c5s = compute_c5(hs)
        if c5s < best_c5:
            best_c5 = c5s
            best_h = hs
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=12)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
