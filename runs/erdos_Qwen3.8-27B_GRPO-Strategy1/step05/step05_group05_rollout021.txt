    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / len(h))

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (d / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_eval(corr_base, B, t, p0, c0):
    vals = corr_base + t * B
    vals[p0] = c0 + t * (1.0 - t)
    return np.max(vals)

def coord_opt_i(h, n, i, niter=45):
    old_val = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    p0 = n - 1
    c0 = corr_base[p0]
    B = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        if k == 0:
            continue
        p = k + n - 1
        c = 0.0
        if 0 <= i + k < n:
            c += (1.0 - h[i + k])
        jk = i - k
        if 0 <= jk < n:
            c -= h[jk]
        B[p] = c
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = coord_eval(corr_base, B, m1, p0, c0)
        f2 = coord_eval(corr_base, B, m2, p0, c0)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    h[i] = (lo + hi) / 2.0
    return h

def coord_descent(h, n, t_end, max_sweeps=500, random_order=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        idx = np.random.permutation(n) if random_order else range(n)
        for i in idx:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            coord_opt_i(best_h, n, i, niter=40)
            if abs(best_h[i] - old) > 1e-10:
                improved = True
        best_h = project_h(best_h)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 0: Multi-resolution init (coarse to fine)
    for n_c in [32, 64]:
        if elapsed() > budget_s * 0.04:
            break
        h_c = project_h(upsample(best_h, n_c))
        for T in [50, 5000]:
            hc = slsqp_opt(h_c, T, n_c, maxiter=400)
            if compute_c5(hc) < compute_c5(h_c):
                h_c = hc
        h_c, _ = coord_descent(h_c, n_c, t0 + budget_s * 0.045, max_sweeps=80)
        h_up = project_h(upsample(h_c, n))
        if compute_c5(h_up) < best_c5:
            best_c5 = compute_c5(h_up)
            best_h = h_up

    # Phase 1: SLSQP annealing sweep
    for T in [30, 300, 3000, 30000, 200000]:
        if elapsed() > budget_s * 0.12:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=2500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Initial coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.20, max_sweeps=150)

    # Phase 3: Basin hopping with diverse perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 180
    for it in range(n_bh):
        if elapsed() > budget_s * 0.58:
            break
        cool = max(0.01, 1.0 - 0.9 * it / float(n_bh))
        mod = it % 12
        if mod == 0:
            pert = (0.02 + 0.10 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 25)
            ph = np.random.uniform(0, 6.28)
            pert = (0.02 + 0.08 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.08
            pert = np.zeros(n)
            pert[mask] = (1.0 - 2.0 * work_h[mask]) * (0.3 + 0.5 * cool)
        elif mod == 3:
            blk = max(2, n // 12)
            b1, b2 = np.random.choice(n // blk, 2, replace=False)
            pert = np.zeros(n)
            s1, s2 = b1 * blk, b2 * blk
            pert[s1:s1+blk] = work_h[s2:s2+blk] - work_h[s1:s1+blk]
            pert[s2:s2+blk] = work_h[s1:s1+blk] - work_h[s2:s2+blk]
        elif mod == 4:
            pert = (0.01 + 0.03 * cool) * np.random.randn(n)
        elif mod == 5:
            mask = np.random.random(n) < 0.06
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.15 * cool) * np.random.randn(int(mask.sum()))
        elif mod == 6:
            pert = (1.0 - 2.0 * work_h) * 0.08 + 0.015 * np.random.randn(n)
        elif mod == 7:
            freqs = np.random.randint(1, 30, 2)
            pert = sum((0.01 + 0.04 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + np.random.uniform(0, 6.28)) for f in freqs)
        elif mod == 8:
            nblk = np.random.randint(3, 8)
            pert = np.zeros(n)
            for _ in range(nblk):
                s = np.random.randint(0, n)
                ln = np.random.randint(2, max(3, n // 6))
                pert[s:min(s+ln, n)] += (0.03 + 0.08 * cool) * np.random.randn(1)
            pert = np.clip(pert, -0.3, 0.3)
        elif mod == 9:
            # Block-level noise
            blk = max(1, n // 16)
            pert = np.zeros(n)
            for b in range(n // blk):
                pert[b*blk:(b+1)*blk] += (0.02 + 0.06*cool) * np.random.randn(1)
        elif mod == 10:
            # High-freq perturbation
            pert = (0.01 + 0.04 * cool) * np.random.choice([-1, 1], n)
        else:
            pert = 0.03 * cool * np.random.randn(n) + 0.01 * np.random.randn(n)

        h_pert = project_h(work_h + pert)
        T_bh = 500 + 25000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=500, )
        c5v = compute_c5(hc)
        # Quick coord descent on the candidate
        hc, c5v = coord_descent(hc, n, t0 + min(elapsed() + 2.0, budget_s * 0.58), max_sweeps=15)
        temp = 0.0005 * cool + 0.00002
        if c5v < work_c5 or (c5v - work_c5) / temp < np.log(np.random.rand() + 1e-30):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 4: High-T SLSQP polish
    for T in [200000, 1000000, 5000000]:
        if elapsed() > budget_s * 0.68:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=3000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Extended coordinate descent (primary refinement)
    t_cd_end = min(t0 + budget_s * 0.93, t0 + budget_s * 0.68 + (budget_s * 0.25))
    best_h, best_c5 = coord_descent(best_h, n, t_cd_end, max_sweeps=3000, random_order=True)

    # Phase 6: 1-h symmetry check and final polish
    h_flip = project_h(1.0 - best_h)
    c5_flip = compute_c5(h_flip)
    if c5_flip < best_c5:
        best_c5 = c5_flip
        best_h = h_flip
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=2000, random_order=False)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
