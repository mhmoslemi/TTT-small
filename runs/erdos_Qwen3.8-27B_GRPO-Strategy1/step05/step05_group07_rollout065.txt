def sm_grad(x, T, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    N = n - 1
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        for j in range(n):
            j1 = j + k - N
            if 0 <= j1 < n:
                g[j] += wk * (1.0 - x[j1])
            j2 = j - k + N
            if 0 <= j2 < n:
                g[j] -= wk * x[j2]
    obj = (m + np.log(np.sum(e)) / T) * 2.0 / n
    return obj, g * 2.0 / n

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h *= tgt / s
    else:
        rm = 1.0 - h
        t = rm.sum()
        h = h + rm * ((tgt - s) / max(t, 1e-13))
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        mask = (h > 0.005) & (h < 0.995)
        if mask.sum() > 0:
            h[mask] += d / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def sm_obj(x, T, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * (2.0 / n)

def sm_grad(x, T, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    N = n - 1
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        for j in range(n):
            j1 = j + k - N
            if 0 <= j1 < n:
                g[j] += wk * (1.0 - x[j1])
            j2 = j - k + N
            if 0 <= j2 < n:
                g[j] -= wk * x[j2]
    return (m + np.log(e.sum()) / T) * (2.0 / n), g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=300):
    def fg(x):
        return sm_grad(x, T, n)
    try:
        r = minimize(h, fg, jac=True, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_minimax(h, n, t_end, max_sweeps=80):
    best = h.copy()
    best_c5 = compute_c5(best)
    N = n - 1
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        order = np.random.permutation(n)
        c = np.correlate(best, 1.0 - best, mode='full')
        improved = False
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            slopes = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                j1 = i + k - N
                j2 = i - k + N
                if 0 <= j1 < n:
                    slopes[k] += 1.0 - best[j1]
                if 0 <= j2 < n:
                    slopes[k] -= best[j2]
            bases = c - old * slopes
            r = minimize_scalar(lambda t: float(np.max(bases + slopes * t)),
                                bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-11})
            nv = float(r.x)
            if abs(nv - old) < 1e-13:
                continue
            best[i] = nv
            best = proj_h(best, n)
            c = np.correlate(best, 1.0 - best, mode='full')
            c5n = float(np.max(c) * 2.0 / n)
            if c5n < best_c5 - 1e-13:
                best_c5 = c5n
                improved = True
            else:
                best[i] = old
                best = proj_h(best, n)
                c = np.correlate(best, 1.0 - best, mode='full')
        if not improved:
            break
    return best, best_c5

        ht = lbfgs_opt(hp, T_val, n, iters=200)

            c5_hi = compute_c5(h_hi)
            h_hi, c5_hi = coord_minimax(h_hi, n_try, t0 + budget_s * 0.92, max_sweeps=40)
            if c5_hi < best_c5:
                best_h = h_hi
                best_c5 = c5_hi
                n = n_try
        except Exception:
            pass

    if time.time() - t0 < budget_s * 0.95:
        best_h = lbfgs_opt(best_h, 50000.0, n, iters=600)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_minimax(best_h, n, t0 + budget_s * 0.99, max_sweeps=50)

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
