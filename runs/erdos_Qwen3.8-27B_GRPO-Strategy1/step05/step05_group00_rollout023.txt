import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    n1, n0 = len(ones), len(zeros)
    D_remove = np.zeros((n1, 2*N-1))
    center = N - 1
    for ip, p in enumerate(ones):
        L_pos = np.arange(1, N - p)
        if len(L_pos) > 0:
            D_remove[ip, center + L_pos] -= (1.0 - h[p+1:N])
        L_neg = np.arange(1, p+1)
        if len(L_neg) > 0:
            D_remove[ip, center - L_neg] += h[:p]
    D_add = np.zeros((n0, 2*N-1))
    for iq, q in enumerate(zeros):
        L_pos = np.arange(1, N - q)
        if len(L_pos) > 0:
            D_add[iq, center + L_pos] += (1.0 - h[q+1:N])
        L_neg = np.arange(1, q+1)
        if len(L_neg) > 0:
            D_add[iq, center - L_neg] += -h[:q]
    return D_remove, D_add

def binary_local(h, N, deadline, max_sweeps=400, sample_n=100):
    h = h.astype(float).copy()
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            idx = idx[np.argsort(np.random.random(len(idx)))]
            h[idx[nh:]] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.argsort(np.random.random(len(idx)))]
            h[idx[:nh-n1]] = 1
    best_c = float(np.max(fft_corr(h)))
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D_r, D_a = compute_D(h, N)
        if len(ones) > sample_n:
            p_idx = np.random.choice(len(ones), sample_n, replace=False)
        else:
            p_idx = np.arange(len(ones))
        best_new_P = P
        best_pq = None
        for ip in p_idx:
            base = c - D_r[ip, :]
            cand = base[None, :] + D_a[:, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(ones[ip]), int(zeros[mi]))
        if best_pq is None:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        else:
            h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
    return h, best_c

def nesterov_soft(h, N, deadline, k0=30, temp0=0.8, lr0=0.035):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta = 0.85
    it = 0
    stale = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk - 1e-8:
            best_pk = pk
            best_h = y.copy()
            stale = 0
        else:
            stale += 1
            if stale > 2000:
                break
        k = min(k0 + it // 200, 120)
        temp = max(0.15, temp0 * (0.997 ** it))
        idx = np.argpartition(corr, -k)[-k:]
        cv = corr[idx]
        w = np.exp((cv - cv.max()) / temp)
        w /= w.sum()
        center = N - 1
        grad = np.zeros(N)
        for wi, ai in zip(w, idx):
            L = int(ai - center)
            if L > 0:
                grad[:N-L] += wi * (1.0 - y[L:])
                grad[L:] += -wi * y[:N-L]
            elif L < 0:
                M = -L
                grad[M:] += wi * (1.0 - y[:N-M])
                grad[:N-M] += -wi * y[M:]
            else:
                grad += wi * (1.0 - 2.0 * y)
        grad /= w.sum()
        h_new = project_sum(y - lr0 * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 800 == 0:
            lr0 *= 0.985
            beta = min(0.96, 0.85 + it * 0.00003)
    return best_h, best_pk * dx

def interpolate_h(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
