import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def make_coef(h, n, idx):
    m_vals = np.arange(-(n - 1), n)
    vm1 = idx - m_vals
    vm2 = idx + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coef = np.zeros(2 * n - 1)
    coef[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
    coef[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
    return coef

def pair_opt(h, n, i, j, niter=100):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-15:
        return h[i], h[j]
    h_base = h.copy()
    h_base[i] = 0.0
    h_base[j] = 0.0
    corr_base = get_corr(h_base)
    ci = make_coef(h, n, i)
    cj = make_coef(h, n, j)
    eff = corr_base + s * cj
    dc = ci - cj
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(eff + m1 * dc) < np.max(eff + m2 * dc):
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    return t, s - t

def pair_cd(h, n, t_end, max_sweeps=5000, niter_per_pair=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_pairs = n // 2
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        perm = np.random.permutation(n)
        for p in range(n_pairs):
            i, j = perm[2 * p], perm[2 * p + 1]
            hi_new, hj_new = pair_opt(best_h, n, i, j, niter=niter_per_pair)
            if hi_new != best_h[i]:
                best_h[i] = hi_new
                best_h[j] = hj_new
                improved = True
        c5_new = compute_c5(best_h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
    return best_h, best_c5

def slsqp_step(h, n, T, maxiter=500):
    def obj(x):
        corr = get_corr(x)
        mx = np.max(corr)
        val = mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T
        return val * (2.0 / n)
    def con_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

    best_h, best_c5 = pair_cd(best_h, n, t2a, max_sweeps=5000, niter_per_pair=60)
    t2b = t0 + budget_s * 0.48
    best_h, best_c5 = pair_cd(best_h, n, t2b, max_sweeps=5000, niter_per_pair=120)

    # Phase 3: Basin hopping with pair CD refinement (30% budget)
    t3 = t0 + budget_s * 0.78
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 300
    for it in range(n_bh):
        if elapsed() > t3:
            break
        cool = max(0.03, 1.0 - 0.97 * it / n_bh)
        h_pert = perturb_h(work_h, best_h, n, cool, rng)
        T_bh = int(200 + 20000 * cool)
        hc = slsqp_step(h_pert, n, T_bh, maxiter=300)
        hc, c5v = pair_cd(hc, n, min(t0 + elapsed(), t3), max_sweeps=30, niter_per_pair=80)
        temp = 0.0002 * cool + 0.00001
        accept = c5v < work_c5 or (rng.rand() < np.exp(-(c5v - work_c5) / temp))
        if accept:
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5 - 1e-14:
                best_c5 = c5v
                best_h = hc.copy()

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 4: Extended pair CD polish (15% budget)
    t4 = t0 + budget_s * 0.93
    best_h, best_c5 = pair_cd(best_h, n, t4, max_sweeps=10000, niter_per_pair=150)

    # Phase 5: Final SLSQP + CD
    if elapsed() < t0 + budget_s * 0.96:
        hc = slsqp_step(best_h, n, 100000, maxiter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if elapsed() < t0 + budget_s * 0.985:
        best_h, best_c5 = pair_cd(best_h, n, t0 + budget_s * 0.985, max_sweeps=3000, niter_per_pair=150)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
