def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    dx = 2.0 / len(h)
    c = corr * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_coefs(h, n, i):
    m = np.arange(-(n-1), n)
    coefs = np.zeros(2*n-1)
    idx_a = i + m
    va = (idx_a >= 0) & (idx_a < n)
    coefs[va] += (1.0 - h[idx_a[va]])
    idx_b = i - m
    vb = (idx_b >= 0) & (idx_b < n)
    coefs[vb] += (-h[idx_b[vb]])
    return coefs

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})

        return h
