import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def subgrad_step(c, y, N, k_top):
    """Compute subgradient at y for the max of top-k constraints."""
    grad = np.zeros(N)
    top_idx = np.argpartition(c, -k_top)[-k_top:]
    center = N - 1
    for ai in top_idx:
        k = int(ai - center)
        if k == 0:
            grad += (1.0 - 2.0 * y)
        elif k > 0:
            grad[:N-k] += (1.0 - y[k:])
            grad[k:] -= y[:N-k]
        else:
            M = -k
            grad[M:] += (1.0 - y[:N-M])
            grad[:N-M] -= y[M:]
    grad /= float(k_top)
    return grad

def nesterov_phase(h, N, deadline, k_top=30, lr=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta = 0.88
    it = 0
    stall = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        if pk < best_val * 0.9999:
            best_val = pk
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.7
                m *= 0.3
                stall = 0
                if lr < 1e-7:
                    break
        g = subgrad_step(c, y, N, k_top)
        gn = np.linalg.norm(g)

            g = g / gn
        m = beta * m - lr * g
        h = project_sum(h + m, target)
        it += 1
        if it % 200 == 0:
            lr *= 0.997
            beta = min(0.95, 0.88 + it * 0.00005)
    return best_h, float(best_val) * (2.0/N)

def precompute_effects(h, N):
    """Compute effect[p][j] for all p, j.
    effect[p][j] = change in c[j] when h[p] increases by 1 (from current value).
    For binary: effect[p][j] = (1-h[p+k])_[valid] - h[p-k]_[valid] where k = j-(N-1)
    """
    effects = np.zeros((N, 2*N-1))
    for p in range(N):
        k_vals = np.arange(2*N-1) - (N-1)  # shift values
        # As first index: p+k valid means 0 <= p+k < N
        first_valid = (p + k_vals >= 0) & (p + k_vals < N)
        k_first = p + k_vals
        # As second index: p-k valid means 0 <= p-k < N
        second_valid = (p - k_vals >= 0) & (p - k_vals < N)
        k_second = p - k_vals
        effects[p, :] = np.where(first_valid, 1.0 - h[np.clip(k_first, 0, N-1)], 0.0) \
                       - np.where(second_valid, h[np.clip(k_second, 0, N-1)], 0.0)
    return effects

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def subgrad_topk(c, y, N, k_top):
    grad = np.zeros(N)
    top_idx = np.argpartition(c, -k_top)[-k_top:]
    center = N - 1
    for ai in top_idx:
        k = int(ai - center)
        if k == 0:
            grad += (1.0 - 2.0 * y)
        elif k > 0:
            grad[:N-k] += (1.0 - y[k:])
            grad[k:] -= y[:N-k]
        else:
            M = -k
            grad[M:] += (1.0 - y[:N-M])
            grad[:N-M] -= y[M:]
    grad /= float(k_top)
    return grad
