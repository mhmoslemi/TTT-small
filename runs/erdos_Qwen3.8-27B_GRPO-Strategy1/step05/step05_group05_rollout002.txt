import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            if not mask.any():
                break
            per = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + per, 1.0)
        else:
            mask = h > 1e-13
            if not mask.any():
                break
            per = (-diff) / mask.sum()
            h[mask] = np.maximum(h[mask] - per, 0.0)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i, n_samples=500):
    h2 = h.copy()
    h2[i] = 0.0
    corr_base = get_corr(h2)
    coefs = np.zeros(2 * n - 1)
    for m in range(n - i):
        coefs[n - 1 + m] = (1.0 - h[i + m])
    for m in range(1, i + 1):
        coefs[n - 1 - m] = (-h[i - m])
    coefs[n - 1] = 1.0
    ts = np.linspace(0, 1, n_samples)
    corr_all = corr_base[None, :] + ts[:, None] * coefs[None, :]
    corr_all[:, n - 1] -= ts ** 2
    max_vals = np.max(corr_all, axis=1)
    return ts[np.argmin(max_vals)]

def coord_descent(h, n, t_end, max_sweeps=200, n_samples=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = np.random.permutation(n)
        improved = 0
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            t_new = coord_opt(best_h, n, i, n_samples)
            if abs(t_new - best_h[i]) > 1e-10:
                best_h[i] = t_new
                best_h = project_h(best_h)
                improved += 1
        c5_now = compute_c5(best_h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
        elif improved == 0:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter=80, n_cd_sweeps=15, n_samples=400):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / float(n_iter))
        step = 0.02 + 0.10 * cool
        mod = it % 6
        if mod == 0:
            pert = step * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, max(2, n // 10))
            ph = np.random.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = step * 1.5 * np.random.randn(int(mask.sum()))
        elif mod == 3:
            blk = max(2, n // 10)
            b = np.random.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            pert[b*blk:min(b*blk+blk, n)] = step * np.random.randn(blk)
        elif mod == 4:
            idx = np.random.choice(n, size=4, replace=False)
            pert = np.zeros(n)
            deltas = step * np.random.randn(4)
            for k, idx_k in enumerate(idx):
                pert[idx_k] = deltas[k]
            pert += -pert.sum() / n
        else:
            pert = step * 0.5 * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        h_cd, c5_cd = coord_descent(h_pert, n, min(time.time() + 5, t_end),
                                     max_sweeps=n_cd_sweeps, n_samples=n_samples)
        temp = 0.001 * cool + 0.00005
        if c5_cd < work_c5 or np.random.rand() < np.exp(-(c5_cd - work_c5) / temp):
            work_h = h_cd
            work_c5 = c5_cd
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h
    return best_h, best_c5

def make_binary_start(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_smooth_start(n, rng):
    t = np.linspace(0, 1, n)
    a = rng.uniform(0.2, 0.8)
    w = rng.uniform(0.5, 2.0)
    p = rng.uniform(1, 3)
    h = ((t - a) / w) ** p
    h = (h - h.min()) / (h.max() - h.min() + 1e-15)
    return project_h(h)

def make_block_start(n, rng):
    n_blocks = rng.randint(3, 7)
    h = np.zeros(n)
    for b in range(n_blocks):
        start = int(b * n / n_blocks)
        end = int((b + 1) * n / n_blocks)
        if b % 2 == 0:
            h[start:end] = rng.uniform(0.5, 1.0)
    return project_h(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_parent = len(initial_h_values)
        starts = [np.array(initial_h_values, dtype=float)]
    else:
        n_parent = 160
        starts = []

    n = n_parent
    # Add diverse starting points
    for _ in range(4):
        starts.append(make_binary_start(n, rng))
    for _ in range(3):
        starts.append(make_smooth_start(n, rng))
    starts.append(make_block_start(n, rng))

    # Phase 1: Multi-start coordinate descent
    best_h = starts[0]
    best_c5 = compute_c5(best_h)
    t_phase1_end = t0 + budget_s * 0.15
    for s_idx, h_start in enumerate(starts):
        if time.time() > t_phase1_end:
            break
        h_s = project_h(h_start)
        c5_s = compute_c5(h_s)
        # Short CD for non-parent starts, longer for parent
        ns = 30 if s_idx == 0 else 12
        h_cd, c5_cd = coord_descent(h_s, n, min(time.time() + budget_s * 0.03, t_phase1_end),
                                     max_sweeps=ns, n_samples=300)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 2: Extended coordinate descent on best
    t_phase2_end = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, t_phase2_end, max_sweeps=150, n_samples=500)

    # Phase 3: Basin hopping
    t_phase3_end = t0 + budget_s * 0.70
    best_h, best_c5 = basin_hop(best_h, n, t_phase3_end, n_iter=60, n_cd_sweeps=12, n_samples=400)

    # Phase 4: Second round of coordinate descent (finer)
    t_phase4_end = t0 + budget_s * 0.85
    best_h, best_c5 = coord_descent(best_h, n, t_phase4_end, max_sweeps=200, n_samples=800)

    # Phase 5: Fine basin hopping
    t_phase5_end = t0 + budget_s * 0.93
    best_h, best_c5 = basin_hop(best_h, n, t_phase5_end, n_iter=25, n_cd_sweeps=20, n_samples=600)

    # Phase 6: Final polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=100, n_samples=1000)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
