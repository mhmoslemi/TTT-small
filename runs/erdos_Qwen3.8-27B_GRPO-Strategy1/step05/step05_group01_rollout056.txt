import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def exact_coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            old_h = best_h[i]
            k_range = np.arange(2*n-1)
            # delta[k] = (1-h[i-k])*[0≤i-k<n] - h[i+k]*[0≤i+k<n]
            mask1 = (k_range >= i - n + 1) & (k_range <= i)
            mask2 = (k_range >= 0) & (k_range <= n - 1 - i)
            delta = np.zeros(2*n-1)
            if mask1.any():
                j1 = i - k_range[mask1]
                delta[mask1] += (1.0 - best_h[j1])
            if mask2.any():
                j2 = k_range[mask2] + i
                delta[mask2] -= best_h[j2]
            base = corr - old_h * delta
            def obj(t):
                return np.max(base + t * delta)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            t_new = res.x
            if abs(t_new - old_h) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = float(np.max(np.correlate(h_try, 1.0-h_try, mode='full')) * dx)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved and sweep > 10:
            break
    return best_h, best_c5

def SL_specific(h, n, t_end):
    """Subgradient with line search targeting the max shifts"""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for it in range(500):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        threshold = np.max(corr) - 0.05 * np.std(corr)
        active = np.where(corr >= threshold)[0]
        shifts = active - (n-1)
        i_idx = np.arange(n)[:, None]
        s_idx = shifts[None, :]
        jp = i_idx + s_idx
        jm = i_idx - s_idx
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n-1)
        jmc = np.clip(jm, 0, n-1)
        grad = ((1.0 - best_h[jpc]) * v1.astype(float) - best_h[jmc] * v2.astype(float)).mean(axis=1) * dx
        # Line search
        alpha = 0.01 / (1 + it * 0.005)
        h_new = project_h(best_h - alpha * grad)
        c5_new = compute_c5(h_new)
        trials = 0
        while c5_new > best_c5 and trials < 10:
            alpha *= 0.5
            h_new = project_h(best_h - alpha * grad)
            c5_new = compute_c5(h_new)
            trials += 1
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        else:
            alpha *= 0.1
    return best_h, best_c5

def perturb_and_optimize(h, n, t_end, scale=0.1):
    """Make a perturbation and re-optimize via coord descent"""
    i_idx = np.arange(n)
    r = np.random.random()
    if r < 0.3:
        # Block flip
        blen = np.random.randint(5, max(n//5, 8))
        bs = np.random.randint(0, n-blen)

        pert[bs:bs+blen] = (1 - 2*h[bs:bs+blen]) * scale
    elif r < 0.6:

        pert = scale * np.sin(2*np.pi*f*i_idx/n + p)

    h_pert = project_h(h + pert)
    h_opt, c5_opt = exact_coord_descent(h_pert, n, t_end - (t_end-time.time())*0.3, max_sweeps=50)
    return h_opt, c5_opt

    # Phase 3: Basin hopping with perturbations

        if t_end - time.time() < 40:

        remaining = t_end - time.time()

        scale = 0.03 + 0.12 * frac

        if c5_pert < best_c5:

            best_h = h_pert

    if t_end - time.time() > 20:
        best_h, best_c5 = exact_coord_descent(best_h, n, t_end - 5, max_sweeps=300)

    best_c5 = compute_c5(best_h)
