def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(np.clip(Tc, -50, 50))
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    # Gradient
    m_full = 2 * n - 1
    grad = np.zeros(n)
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):

        wi = w[idx]
        if wi < 1e-14:

        if lg >= 0:

            grad[p:] += -wi * h[:n-p]

    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n

    Tc = T * (corr - mc)
    Tc -= np.max(Tc)

                        hi = d
                    else:
                        lo = c
                x_opt = (lo + hi) * 0.5
            if abs(x_opt - v) < 1e-12:
                continue
            old_corr = corr.copy()
            best_h[i] = x_opt
            best_h = project_h(best_h)
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            new_raw = np.max(corr)
            if new_raw < best_raw - 1e-13:
                best_raw = new_raw
                improved += 1
            else:
                best_h[i] = v
                best_h = project_h(best_h)
                corr = old_corr
        if improved == 0 and sweep > 3:
            break
        improved_count += improved
    return best_h, best_raw * (2.0 / n)

def try_n(h_best, c5_best, n_new, t_end):
    n_old = len(h_best)
    h_new = np.interp(np.linspace(0, 2, n_new, endpoint=True),
                      np.linspace(0, 2, n_old, endpoint=True), h_best)
    h_new = project_h(h_new)
    h_new = lbfgs_step(h_new, 2000, 200.0, 1500, 1e-15)
    c5_new = compute_c5(h_new)
    if c5_new < c5_best and time.time() < t_end:
        h_new, c5_new = coord_descent_fast(h_new, 300, t_end)
        if c5_new >= c5_best:
            return h_best, c5_best, n_old
        return h_new, c5_new, n_new
    return h_best, c5_best, n_old

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    p1_end = t0 + budget_s * 0.12
    for T in [100, 500, 2000, 8000]:
        if time.time() > p1_end:
            break
        hc = lbfgs_step(best_h, T, 300.0, 3000, 1e-16)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc

    p2_end = t0 + budget_s * 0.42
    best_h, best_c5 = coord_descent_fast(best_h, 800, p2_end)

    p3_end = t0 + budget_s * 0.55
    candidates = []
    h_bip = np.zeros(n); h_bip[:n // 2] = 1.0
    candidates.append(project_h(h_bip))
    for fr in [2, 3, 5, 7, 11]:
        h_p = 0.5 + 0.4 * np.sin(2 * np.pi * fr * np.arange(n) / n)
        candidates.append(project_h(h_p))
    for sc in [0.05, 0.12, 0.2]:
        candidates.append(project_h(best_h + sc * np.random.randn(n)))
    for h_init in candidates:
        if time.time() > p3_end:
            break
        hc = lbfgs_step(h_init, 3000, 300.0, 1500, 1e-15)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc
            best_h, best_c5 = coord_descent_fast(best_h, 100, p3_end)

    p4_end = t0 + budget_s * 0.72
    for T in [10000, 30000, 100000]:
        if time.time() > p4_end:
            break
        hc = lbfgs_step(best_h, T, 500.0, 4000, 1e-17)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc
        best_h, best_c5 = coord_descent_fast(best_h, 200, p4_end)

    p5_end = t0 + budget_s * 0.85
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(25):
        if time.time() > p5_end:
            break
        cool = 1.0 - 0.85 * (it / 25.0)
        sc = 0.005 + 0.04 * cool
        h_p = project_h(work_h + sc * np.random.randn(n))
        T_bh = int(2000 * cool + 100)
        hc = lbfgs_step(h_p, T_bh, 300.0, 1000, 1e-15)
        hc, cv = coord_descent_fast(hc, 50, p5_end)
        temp = 0.0003 * cool + 0.00005
        if cv < work_c5 or np.random.rand() < np.exp(-(cv - work_c5) / temp):
            work_h = hc
            work_c5 = cv
            if cv < best_c5:
                best_c5 = cv
                best_h = hc

    if time.time() < t0 + budget_s * 0.90:
        if len(best_h) == 160:
            best_h, best_c5, n = try_n(best_h, best_c5, 200, t0 + budget_s * 0.93)
        elif len(best_h) == 200:
            best_h, best_c5, n = try_n(best_h, best_c5, 160, t0 + budget_s * 0.93)

    if time.time() < t0 + budget_s * 0.96:
        best_h, best_c5 = coord_descent_fast(best_h, 400, t0 + budget_s * 0.97)

    if time.time() < t0 + budget_s * 0.99:
        hc = lbfgs_step(best_h, 200000, 500.0, 2000, 1e-17)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc
        best_h, best_c5 = coord_descent_fast(best_h, 100, t0 + budget_s * 0.995)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
