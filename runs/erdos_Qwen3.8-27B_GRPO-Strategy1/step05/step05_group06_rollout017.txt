    dx = 2.0 / n
    corr = compute_corr(h)
    k_star = np.argmax(corr)

    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        valid1 = slice(0, n - k)
        grad[valid1] += (1.0 - h[k:])
        valid2 = slice(k, n)
        grad[valid2] -= h[:n - k]
    else:
        p = -k
        grad[p:] += (1.0 - h[:n - p])
        valid2 = slice(0, n - p)
        grad[valid2] -= h[p:]

    h_new = project_h(h_new)

def softmax_lbfgs(h, T, lam, maxiter, t_end):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    lag_arr = np.arange(m) - n + 1
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        Tc = T * (corr - mc)
        Tc -= np.max(Tc)
        exps = np.exp(Tc)
        s_exp = exps.sum()
        w = exps / s_exp
        sm = mc + np.log(s_exp) / T
        val = dx * sm + lam * (x.sum() - n / 2.0) ** 2
        grad = np.zeros(n)
        for idx in range(m):
            lg = lag_arr[idx]
            wi = w[idx]
            if wi < 1e-15:
                continue
            if lg >= 0:
                if lg < n:
                    grad[lg:] += wi * (1.0 - x[:n - lg])
                grad[:n - lg] += -wi * x[lg:]
            else:
                p = -lg
                grad[:n - p] += wi * (1.0 - x[p:])
                grad[p:] += -wi * x[:n - p]
        grad = dx * grad + 2.0 * lam * (x.sum() - n / 2.0)
        return val, grad
    try:
        res = minimize(obj_grad, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})

        return h

    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = fast_coord_descent(best_h, t0 + budget_s * 0.99,
                                             max_sweeps=100, recompute_every=10)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
