                if lg == 0:
                    lin[idx] = 1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            quad = np.zeros(m_full)
            quad[0] = -1.0  # lag 0
            # For other lags, the quadratic term in xi:
            # corr[k] = cb[k] + xi * lin[k] + xi^2 * quad[k]
            # At lag 0: h[i](1-h[i]) = xi*(1-xi) = xi - xi^2, so quad[0] = -1
            # At other lags: the cross term is xi*(1-h[i+lg]) + (-h[i-lg])*xi, no xi^2 term
            # Wait, let me recheck.
            # corr[k] = sum_j h[j](1-h[j+k])
            # When h[i] changes from 0 to xi:
            # - If i is the first index: term becomes xi*(1-h[i+k]) instead of 0
            # - If i is the second index (i+k=j, so j=i-k): term becomes h[i-k]*(1-xi) instead of h[i-k]*(1-0)=h[i-k]
            #   So the change is -h[i-k]*xi
            # Total change in corr[k]: xi*(1-h[i+k]) - h[i-k]*xi (with appropriate validity)
            # This is LINEAR in xi! No quadratic term except at k=0.
            # At k=0: xi*(1-xi) = xi - xi^2
            # So quad[0] = -1, and quad[k] = 0 for k != 0.
            
            def f1d(x):
                return np.max(cb + x * lin + x * x * quad)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved and sweep > 10:
            break
    return best_h, best_c5
