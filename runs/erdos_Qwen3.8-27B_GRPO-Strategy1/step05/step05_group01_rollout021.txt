    m = len(base_row)
    if m == 1:
        # min of base + t*coef over t in [0,1]
        if coef_row[0] < -1e-14:
            return 1.0
        elif coef_row[0] > 1e-14:
            return 0.0
        return 0.0
    candidates = [0.0, 1.0]
    for i in range(m):
        for j in range(i+1, m):
            d = coef_row[j] - coef_row[i]
            if abs(d) > 1e-14:
                t = (base_row[i] - base_row[j]) / d
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
    best_v = np.inf
    best_t = 0.0
    for t in candidates:
        v = np.max(base_row + t * coef_row)
        if v < best_v:
            best_v = v
            best_t = t
    return best_t

def coord_descent(h, n, t_end, n_top_start=8, n_top_max=50, min_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    sweep = 0
    no_improve = 0
    while time.time() < t_end:
        sweep += 1
        corr = get_corr(best_h)
        n_top = min(n_top_max, n_top_start + 2 * (sweep // 15))
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        coefs_matrix = all_coefs(best_h, n, top_shifts)
        
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            if time.time() > t_end:
                return best_h, best_c5
            base = corr[top_idx] - best_h[i] * coefs_matrix[i]
            t_new = solve_1d_minimax_vec(base, coefs_matrix[i])
            if abs(t_new - best_h[i]) > 5e-11:
                old_val = best_h[i]
                best_h[i] = t_new
                best_h = project_h(best_h)
                # Update coefs for changed coordinate (approximation: recompute)
                c5_try = compute_c5(best_h)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    improved = True
                else:
                    best_h[i] = old_val
                    best_h = project_h(best_h)
        no_improve = no_improve + 1 if not improved else 0
        if no_improve > 5 and sweep > min_sweeps:
            break
    return best_h, best_c5

def subgrad_step(h, n, top_shifts, alpha):
    coefs = all_coefs(h, n, top_shifts)
    grad = coefs.mean(axis=1)
    h_new = h - alpha * grad
    return project_h(h_new)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def all_coefs(h, n, shifts):
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = ((jp >= 0) & (jp < n)).astype(np.float64)
    v2 = ((jm >= 0) & (jm < n)).astype(np.float64)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    return (1.0 - h[jpc]) * v1 - h[jmc] * v2

def solve_1d_minimax(base, coefs):
    m = len(base)
    best_v = np.max(base)
    best_t = 0.0
    v1 = np.max(base + coefs)
    if v1 < best_v:
        best_v, best_t = v1, 1.0
    for i in range(m):
        for j in range(i + 1, m):
            d = coefs[j] - coefs[i]
            if abs(d) > 1e-14:
                t = (base[i] - base[j]) / d
                if 0.0 <= t <= 1.0:
                    v = np.max(base + t * coefs)
                    if v < best_v:
                        best_v, best_t = v, t
    return best_t

        top_shifts = top_idx - (n - 1)
        coefs_mat = all_coefs(best_h, n, top_shifts)
        improved = False
        perm = np.random.permutation(n)

            if time.time() > t_end:
                return best_h, best_c5
            base = corr[top_idx] - best_h[i] * coefs_mat[i]
            t_new = solve_1d_minimax(base, coefs_mat[i])
            if abs(t_new - best_h[i]) > 3e-11:
                old = best_h[i]
                best_h[i] = t_new
                best_h = project_h(best_h)
                c5_try = compute_c5(best_h)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    improved = True
                else:
                    best_h[i] = old
                    best_h = project_h(best_h)
