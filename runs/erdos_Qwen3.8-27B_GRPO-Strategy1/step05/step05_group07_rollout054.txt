        # Let me think of it as: reversal of v, then convolve
        # correlate(a,v)[k] = sum_i a[i+k-n+1] * v[i] where indices are valid
        # For k in [0, 2n-2]:
        #   a_idx = i + k - n + 1 must be in [0, n-1]
        #   v_idx = i must be in [0, n-1]
        #   So: 0 <= i < n AND 0 <= i+k-n+1 < n
        #   => 0 <= i < n AND n-1-k <= i <= n-1
        #   => max(0, n-1-k) <= i <= n-1
        i_lo = max(0, n - 1 - k)
        i_hi = min(n, n)  # i goes up to n-1
        if i_lo >= n:
            continue
        idx_i = np.arange(i_lo, n)
        idx_a = idx_i + k - n + 1
        # c[k] = sum_{i in range} x[idx_a] * (1 - x[idx_i])
        # dc[k]/dx[j] for j = idx_a[i]: (1 - x[idx_i[i]])
        # dc[k]/dx[j] for j = idx_i[i]: -x[idx_a[i]]
        g[idx_a] += wk * (1.0 - x[idx_i])
        g[idx_i] -= wk * x[idx_a]
    return g * (2.0 / n)

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = float(n) * 0.5
    s = h.sum()
    diff = tgt - s
    if abs(diff) > 1e-14:
        if diff > 0:
            pos = 1.0 - h
            tot = pos.sum()
            if tot > 1e-15:
                h += pos * (diff / tot)
        else:
            neg = h.copy()
            tot = neg.sum()
            if tot > 1e-15:
                h -= neg * (-diff / tot)
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-10:
            mask = (h > 1e-8) & (h < 1.0 - 1e-8)
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += d / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

_TG = [1.0]

def soft_obj(x):
    T = _TG[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sc = 2.0 / len(x)
    if T < 1e-10:
        return m * sc
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * sc

def soft_grad(x):
    T = _TG[0]
    n = len(x)
    cf = 2 * n - 1
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    es = e.sum()
    if es < 1e-300:
        return np.zeros(n)
    w = e / es
    g = np.zeros(n)
    for k in range(cf):
        wk = w[k]
        if wk < 1e-16:
            continue
        i_lo = max(0, n - 1 - k)
        if i_lo >= n:
            continue
        idx_i = np.arange(i_lo, n)
        idx_a = idx_i + k - n + 1
        g[idx_a] += wk * (1.0 - x[idx_i])
        g[idx_i] -= wk * x[idx_a]
    return g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=400):
    _TG[0] = T
    try:
        r = minimize(h, soft_obj, soft_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-11})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=60):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr_cache = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            base = corr_cache - old * coef
            def _f1d(v):
                return float(np.max(base + v * coef))
            r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-13})
            nv = float(r.x)
            if abs(nv - old) < 1e-14:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-16:
                best = trial
                best_c5 = c5t
                corr_cache = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def binary_sa(h_init, n, t_end, max_iters=300000, seed=0):
    rng = np.random.RandomState(seed)
    h_b = (h_init > 0.5).astype(float).copy()
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    if len(ones) > n // 2:
        ex = rng.choice(ones, len(ones) - n // 2, replace=False)
        h_b[ex] = 0
    elif len(ones) < n // 2:
        df = n // 2 - len(ones)
        h_b[rng.choice(zeros, df, replace=False)] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    cur_c5 = best_c5
    T = 0.06
    for it in range(max_iters):
        if time.time() > t_end:
            break
        T = max(0.0003, 0.06 * (1.0 - 0.00003 * it))
        r = rng.rand()
        if r < 0.55:
            ones_l = np.where(h_b == 1)[0]
            zeros_l = np.where(h_b == 0)[0]
            i = ones_l[rng.randint(len(ones_l))]
            j = zeros_l[rng.randint(len(zeros_l))]
            h_b[i], h_b[j] = h_b[j], h_b[i]
            reverted = False
        elif r < 0.75:
            blen = rng.randint(3, min(12, n // 4) + 1)
            start = rng.randint(0, n - blen)
            seg = h_b[start:start + blen].copy()
            rot = rng.randint(1, blen)
            h_b[start:start + blen] = np.concatenate([seg[rot:], seg[:rot]])
            reverted = False
        else:
            corr_now = np.correlate(h_b, 1.0 - h_b, mode='full')
            pk = int(np.argmax(corr_now))
            k = pk - n + 1
            if k >= 0:
                cand = [ii for ii in range(n - k) if h_b[ii] == 0 and h_b[ii + k] == 1]
            else:
                kk = -k
                cand = [ii for ii in range(kk, n) if h_b[ii] == 1 and h_b[ii - kk] == 0]
            if len(cand) > 1:
                ci = cand[rng.randint(len(cand))]
                if h_b[ci] == 1:
                    zl = np.where(h_b == 0)[0]
                    i, j = ci, zl[rng.randint(len(zl))]
                else:
                    ol = np.where(h_b == 1)[0]
                    i, j = ci, ol[rng.randint(len(ol))]
                h_b[i], h_b[j] = h_b[j], h_b[i]
                reverted = False
            else:
                ol = np.where(h_b == 1)[0]
                zl = np.where(h_b == 0)[0]
                i = ol[rng.randint(len(ol))]
                j = zl[rng.randint(len(zl))]
                h_b[i], h_b[j] = h_b[j], h_b[i]
                reverted = False
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < cur_c5 or rng.rand() < np.exp(-(nc5 - cur_c5) / T):
            cur_c5 = nc5
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            if r < 0.75 and r >= 0.55:
                h_b[start:start + blen] = seg
            else:
                h_b[i], h_b[j] = h_b[j], h_b[i]
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=150, seed=0):
    rng = np.random.RandomState(seed)
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / float(n_iters))
        mod = it % 4
        if mod == 0:
            pert = (0.02 + 0.12 * cool) * rng.randn(n)
        elif mod == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = (0.02 + 0.08 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = rng.random(n) < (0.15 + 0.35 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.16 * cool) * rng.randn(mask.sum())
        else:
            idx = rng.choice(n, size=min(20, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = (0.05 + 0.2 * cool) * rng.randn(len(idx))
        hp = proj_h(work + pert, n)
        T_val = 500.0 + 8000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.0006 * cool + 0.00004
        if c5t < work_c5 or rng.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5
