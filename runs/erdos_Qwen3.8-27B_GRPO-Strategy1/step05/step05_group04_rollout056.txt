def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    for _ in range(60):
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
        if s > t:
            m = h > 1e-15
            if m.sum() == 0:
                break
            h[m] *= t / s
        else:
            r = 1.0 - h
            tr = r.sum()
            if tr < 1e-14:
                break
            h += r * (t - s) / tr
        h = np.clip(h, 0.0, 1.0)
    return h

def _f1d(v, old, corr, lin, n, dx, buf):
    d = v - old
    buf[:] = corr
    buf += d * lin
    buf[n - 1] -= d * d
    return float(np.max(buf) * dx)

def min_1d_hyb(corr, lin, n, dx, old, buf):
    # Coarse
    grid = np.linspace(0.0, 1.0, 21)
    bv, bf = 0.0, 1e30
    for v in grid:
        f = _f1d(v, old, corr, lin, n, dx, buf)
        if f < bf:
            bf, bv = f, v
    # Golden refine
    gi = int(bv * 20)
    lo = grid[max(0, gi - 1)]
    hi = grid[min(20, gi + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _f1d(c, old, corr, lin, n, dx, buf)
    fd = _f1d(d, old, corr, lin, n, dx, buf)

            d = lo + gr * (hi - lo)
            fd = _f1d(d, old, corr, lin, n, dx, buf)
        if hi - lo < 1e-11:
