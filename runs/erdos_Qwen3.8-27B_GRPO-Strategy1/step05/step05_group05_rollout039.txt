    # For each j, grad[j] = sum_d w[d] * [(1-h[j+lag_d]) - h[j-lag_d]] (with bounds)

    for j in range(n):

        jp = j + lags
        valid_p = (jp >= 0) & (jp < n)
        # h[j-lag] terms  
        jm = j - lags
        valid_m = (jm >= 0) & (jm < n)
        a = np.sum(corr_weights[valid_p] * one_minus_h[jp[valid_p]])
        b = np.sum(corr_weights[valid_m] * h[jm[valid_m]])
        grad[j] = a - b
    return grad

        lags = np.arange(2*n-1)-(n-1)

                g = 0

            best_h = h_try
            # best_c5 updated above
        lr *= 0.999

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_pass(h, T, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = get_corr(x)
        m = np.max(corr)
        if T < 1e6:
            sh = T * (corr - m)
            val = m + np.log(np.exp(sh).sum()) / T
        else:
            val = m
        return val * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_single(h, n, i, niter=50):
    hb = h.copy(); hb[i] = 0.0
    cb = get_corr(hb)
    lags = np.arange(2 * n - 1) - (n - 1)
    co = np.zeros(2 * n - 1)
    jp = i + lags; vp = (jp >= 0) & (jp < n)
    jm = i - lags; vm = (jm >= 0) & (jm < n)
    co[vp] += 1.0 - h[jp[vp]]
    co[vm] -= h[jm[vm]]
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(cb + m1 * co) < np.max(cb + m2 * co):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_opt_pair(h, n, i, j, niter=35):
    hb = h.copy(); hb[i] = 0.0; hb[j] = 0.0
    cb = get_corr(hb)
    lags = np.arange(2 * n - 1) - (n - 1)
    ci = np.zeros(2 * n - 1); cj = np.zeros(2 * n - 1)
    jpi = i + lags; vpi = (jpi >= 0) & (jpi < n)
    jmi = i - lags; vmi = (jmi >= 0) & (jmi < n)
    ci[vpi] += 1.0 - h[jpi[vpi]]; ci[vmi] -= h[jmi[vmi]]
    jpj = j + lags; vpj = (jpj >= 0) & (jpj < n)
    jmj = j - lags; vmj = (jmj >= 0) & (jmj < n)
    cj[vpj] += 1.0 - h[jpj[vpj]]; cj[vmj] -= h[jmj[vmj]]
    li, hi, lj, hjj = 0.0, 1.0, 0.0, 1.0
    for _ in range(niter):
        mi1 = li + (hi - li) / 3; mi2 = hi - (hi - li) / 3
        mj1 = lj + (hjj - lj) / 3; mj2 = hjj - (hjj - lj) / 3
        f1 = np.max(cb + mi1*ci + mj1*cj)
        f2 = np.max(cb + mi2*ci + mj2*cj)
        f3 = np.max(cb + mi1*ci + mj2*cj)
        f4 = np.max(cb + mi2*ci + mj1*cj)
        bv = min(f1, f2, f3, f4)
        if bv == f1: hi = mi2; hjj = mj2
        elif bv == f2: li = mi1; lj = mj1
        elif bv == f3: hi = mi2; lj = mj1
        else: li = mi1; hjj = mj2
    return (li + hi) / 2.0, (lj + hjj) / 2.0

def coord_descent(h, n, t_end, sweeps=80, use_pairs=True, niter=50):
    best = h.copy(); bc = compute_c5(best)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best, bc
            nv = coord_opt_single(best, n, i, niter)
            if abs(nv - best[i]) < 1e-13:
                continue
            ht = best.copy(); ht[i] = nv; ht = project_h(ht)
            c = compute_c5(ht)
            if c < bc - 1e-15:
                bc = c; best = ht; imp = True
        if use_pairs and sw % 2 == 1:
            for _ in range(n // 5):
                if time.time() > t_end:
                    return best, bc
                i, j = np.random.choice(n, 2, replace=False)
                ni, nj = coord_opt_pair(best, n, i, j, 35)
                if abs(ni - best[i]) < 1e-13 and abs(nj - best[j]) < 1e-13:
                    continue
                ht = best.copy(); ht[i] = ni; ht[j] = nj; ht = project_h(ht)
                c = compute_c5(ht)
                if c < bc - 1e-15:
                    bc = c; best = ht; imp = True
        if not imp:
            break
    return best, bc

def basin_hopping(h_best, c_best, n, t_start, t_end, n_iter, seed=0):
    rng = np.random.default_rng(seed)
    wh = h_best.copy(); wc = c_best
    for it in range(n_iter):

            break
        cool = 1.0 - 0.92 * (it / max(n_iter, 1))

        elif m == 2:

        elif m == 3:

        elif m == 4:
