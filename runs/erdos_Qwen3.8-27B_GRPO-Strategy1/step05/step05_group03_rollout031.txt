    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    # Iterative water-filling
    for _ in range(10):
        if s > target:
            excess = s - target
            # Reduce proportionally, but not below 0
            scale = 1.0 - excess / s if s > 1e-12 else 0
            h = np.maximum(h * scale, 0.0)
        else:
            deficit = target - s
            # Add proportionally to room
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h = h + room * (deficit / tot)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(40):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = h + room * (diff / tot)
        else:
            pos = h > 1e-15
            tot = h[pos].sum()
            if tot < 1e-13:
                break
            h = np.where(pos, h * (1.0 + diff / tot), h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    sh = np.clip(T * (corr - m), -400, 400)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def slsqp_con_fun(x, n):
    return x.sum() - n / 2.0

def slsqp_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h_init, n, T, maxiter):
