    k = idx - (n - 1)  # actual lag
    grad = np.zeros(n)

        grad[:n-p] += (1.0 - h[p:])

        grad[p:] += -h[:n-p]

        grad += grad_corr_idx(h, idx)

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_corr_idx(h, idx):
    n = len(h)
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] += -h[k:]
    else:
        p = -k
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] += -h[:n-p]
    return grad

def coord_descent(h, t_end, max_sweeps=100000):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    prev_c5 = best_c5
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h0 = best_h.copy()
            h0[i] = 0.0
            cb = compute_corr(h0)
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            def f1d(x):
                return np.max(cb + x * lin + x * x * quad)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',

        if not improved:

def subgradient_polish(h, t_end, n_steps=500):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = h.copy()
    for step in range(n_steps):
        if time.time() > t_end:
            break
        corr = compute_corr(work_h)
        top_K = min(3 + (step // 100), 8)
        top_pos = np.argsort(corr)[-top_K:]
        grad = np.zeros(n)
        for idx in top_pos:
            grad += grad_corr_idx(work_h, idx)
        grad /= top_K
        alpha = 0.02 * max(0.1, 1.0 - step / n_steps)
        work_h = project_h(work_h - alpha * grad)
        c5v = compute_c5(work_h)
        if c5v < best_c5:
            best_h = work_h.copy()
            best_c5 = c5v
    return best_h, best_c5

def perturb_h(h, scale, rng):
    n = len(h)
    ptype = rng.randint(3)
    if ptype == 0:
        return project_h(h + scale * rng.randn(n))
    elif ptype == 1:
        hp = h.copy()
        n_flips = max(2, int(0.1 * n))
        idx = rng.choice(n, n_flips, replace=False)
        hp[idx] = np.clip(hp[idx] + 0.5 * scale * rng.randn(n_flips), 0, 1)
        return project_h(hp)
    else:
        seg = int(0.2 * n)
        s = rng.randint(0, n - seg)
        shift = rng.randint(3, seg)
        hp = h.copy()
        hp[s:s+seg] = h[(s+shift):(s+shift+seg)] if (s+shift+seg <= n) else h[(s+shift)%n:]
        return project_h(hp + 0.01 * rng.randn(n))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate descent (20%)
    t1 = t0 + budget_s * 0.20
    best_h, best_c5 = coord_descent(best_h, t1)

    # Phase 2: Subgradient polish (8%)
    t2 = t0 + budget_s * 0.28
    best_h, best_c5 = subgradient_polish(best_h, t2, 300)
    best_h, best_c5 = coord_descent(best_h, t0 + budget_s * 0.32)

    # Phase 3: Basin hopping (50%)
    t3 = t0 + budget_s * 0.82
    work_h = best_h.copy()
    work_c5 = best_c5
    it_count = 0
    while time.time() < t3:
        it_count += 1
        cool = max(0.15, 1.0 - 0.85 * (it_count / 200.0))
        scale = 0.015 + 0.06 * cool
        h_p = perturb_h(work_h, scale, rng)
        time_left = t3 - time.time()
        iter_time = min(8.0 * cool + 2.0, time_left * 0.3)
        if iter_time < 1.5:
            break
        hc, c5v = coord_descent(h_p, time.time() + iter_time, max_sweeps=5000)
        temp = 0.0008 * cool + 0.00005
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_h = hc
                best_c5 = c5v

    # Phase 4: Final intensive coordinate descent (12%)
    t4 = t0 + budget_s * 0.94
    best_h, best_c5 = coord_descent(best_h, t4, max_sweeps=50000)

    # Phase 5: Last subgradient + coord (4%)
    t5 = t0 + budget_s * 0.98
    best_h, best_c5 = subgradient_polish(best_h, t5, 200)
    best_h, best_c5 = coord_descent(best_h, t0 + budget_s * 0.99, max_sweeps=20000)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
