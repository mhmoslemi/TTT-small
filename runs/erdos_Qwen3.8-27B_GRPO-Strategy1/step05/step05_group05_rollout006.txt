import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    dx = 2.0 / n
    return val * dx

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_fast(h, n, i, niter=30):
    h_copy = h.copy()
    h_copy[i] = 0.0
    corr0 = get_corr(h_copy)
    coefs = np.zeros(2 * n - 1)
    offset = n - 1
    # First factor: h[i+lag] where i+lag = idx → i = idx-lag
    for lag in range(-(n - 1 - i), n - i):
        # i_valid = i - lag must be in [0, n-1]
        i_valid = i - lag
        if 0 <= i_valid < n:
            coefs[offset + lag] += (1.0 - h[i_valid])
    # Second factor: (1-h[i]) where i = idx → h[idx+lag]
    for lag in range(-i, n - i):
        i_plus_lag = i + lag
        if 0 <= i_plus_lag < n:
            coefs[offset + lag] += (-h[i_plus_lag])
    
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        mid = (lo + hi) / 2.0
        vals = corr0 + mid * coefs
        mx = vals.max()
        tol = 1e-10
        active = np.where(vals >= mx - tol)[0]
        slopes = coefs[active]
        if slopes.min() < -1e-12:
            lo = mid
        elif slopes.max() > 1e-12:
            hi = mid
        else:
            break
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            v_new = coord_opt_fast(best_h, n, i)
            if abs(v_new - best_h[i]) < 1e-13:
                continue
            best_h[i] = v_new
            # Project after each update to maintain constraint
            # Actually, let's do the projection less frequently for speed
            # Just clip for now, project at end of sweep
            best_h[i] = np.clip(v_new, 0.0, 1.0)
        # Project at end of sweep
        best_h = project_h(best_h)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            improved = True
        if not improved and sweep > 5:
            break
    return best_h, best_c5
