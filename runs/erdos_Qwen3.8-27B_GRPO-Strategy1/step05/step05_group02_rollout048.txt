    if elapsed() < budget_s * 0.97:
        for T in [800000, 3000000]:
            if elapsed() > budget_s * 0.97:
                break
            hc = slsqp_step(best_h, T, n, maxiter=1200, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Final touch-up
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=30, seed=seed + 400)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
