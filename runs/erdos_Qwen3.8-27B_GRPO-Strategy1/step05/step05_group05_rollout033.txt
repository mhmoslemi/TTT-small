def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:

            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        grad[:n-d] += (1.0 - h[d:n])

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find all lags within tol of max
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for idx in active:
        d = int(idx) - (n - 1)
        grad += corr_gradient_h(h, n, d)
    grad /= len(active)
    return h - lr * grad, max_val

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_work = h.copy()
    lr = lr0
    stuck = 0
    for it in range(100000):
        if time.time() > t_end:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    return val * (2.0 / n)

def slsqp_smooth(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_fast(h, n, i, niter=35):
    L = 2 * n - 1
    center = n - 1
    coeff = np.zeros(L)
    for idx in range(L):
        d = idx - center
        if d == 0:
            continue
        c = 0.0
        ip = i + d
        im = i - d
        if 0 <= ip < n:
            c += (1.0 - h[ip])
        if 0 <= im < n:
            c += (-h[im])
        coeff[idx] = c
    
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    
    def eval_t(t):
        c = corr_base + t * coeff
        c[center] += t * (1.0 - t)
        return np.max(c)
    
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if eval_t(m1) < eval_t(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=50, tol=1e-12):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            new_val = coord_opt_fast(best_h, n, i)
            if abs(new_val - best_h[i]) < tol:
                continue
            h_try = best_h.copy()
            h_try[i] = new_val
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hopping(h, n, t_end, n_iter=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = h.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - 0.95 * it / n_iter)
        step = 0.01 + 0.10 * cool
        pert_type = it % 5
        pert = np.zeros(n)
        if pert_type == 0:
            pert[:] = step * np.random.randn(n)
        elif pert_type == 1:
            freq = np.random.randint(1, 12)
            ph = np.random.uniform(0, 2 * np.pi)
            pert[:] = step * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif pert_type == 2:
            k = np.random.randint(n // 10, n // 3)
            start = np.random.randint(0, n - k)
            pert[start:start+k] = step * np.random.randn(k)
        elif pert_type == 3:
            mask = np.random.random(n) < (0.1 + 0.2 * cool)
            pert[mask] = (step * 2) * np.random.randn(mask.sum())
        else:
            # Swap-type perturbation
            idxs = np.random.permutation(n)
            for j in range(min(4, n // 4)):
                a, b = idxs[2*j], idxs[2*j+1]
                delta = (work_h[b] - work_h[a]) * 0.8
                pert[a] += delta
                pert[b] -= delta
        h_pert = project_h(work_h + pert)
        # Optimize from perturbed point
        hc = slsqp_smooth(h_pert, 200 + 5000 * cool, n, maxiter=200)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or (np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    return best_h, best_c5

    """Resample h from n_from points to n_to points."""
    x_from = np.linspace(0, 2, n_from, endpoint=False)
    x_to = np.linspace(0, 2, n_to, endpoint=False)
    h_new = np.interp(x_to, x_from, h, period=2.0) if False else np.interp(x_to, x_from, h)
    return project_h(h_new)

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = float(n) // 2
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)
