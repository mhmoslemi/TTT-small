    # For lag index l (0 to 2n-2), corresponding to shift m = l - (n-1)

    c = a + (1 - gr) * (b - a)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / len(h))

def slsqp_step(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def golden_opt(h, n, i, iters=40):
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    m_vals = np.arange(-(n - 1), n)
    B = np.zeros(2 * n - 1)
    vp = (i + m_vals >= 0) & (i + m_vals < n)
    vm = (i - m_vals >= 0) & (i - m_vals < n)
    B[vp] += (1.0 - h[i + m_vals[vp]])
    B[vm] -= h[i - m_vals[vm]]
    A = corr_base - h[i] * B
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = a + (1.0 - gr) * (b - a)
    d = a + gr * (b - a)
    fc = np.max(A + c * B)
    fd = np.max(A + d * B)
    for _ in range(iters):
        if fc < fd:
            b = d; d = c; fd = fc
            c = a + (1.0 - gr) * (b - a)
            fc = np.max(A + c * B)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = np.max(A + d * B)
    return (a + b) * 0.5

        h[i] = t_new

        for _ in range(3):
            h[ii] = golden_opt(h, n, ii, pair_iters)
            h[jj] = golden_opt(h, n, jj, pair_iters)
    return project_h(h)
