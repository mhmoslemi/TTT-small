            lo = mid

def smooth_c5_grad(h, T):

    corr = get_corr(h)
    m = float(np.max(corr))

def pgd_round(h, T, n, n_steps, alpha0):
    for s in range(n_steps):
        grad = smooth_c5_grad(h, T)
        alpha = alpha0 * (0.92 ** s)
        h_new = h - alpha * grad
        h_new = project_exact(h_new, n)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
    return h

def coord_descent(h, n, t_end, max_sweeps, ternary_iters=35):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old_val = best_h[i]
            best_h[i] = 0.0
            corr0 = get_corr(best_h)
            best_h[i] = 1.0
            corr1 = get_corr(best_h)
            best_h[i] = old_val
            delta = corr1 - corr0
            lo, hi = 0.0, 1.0
            for _ in range(ternary_iters):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(corr0 + m1 * delta) < np.max(corr0 + m2 * delta):
                    hi = m2
                else:
                    lo = m1

            if abs(new_val - old_val) > 1e-12:
                best_h[i] = new_val
                improved = True
        c5_now = compute_c5(best_h)
        if c5_now < best_c5:
            best_c5 = c5_now
        if not improved:
            break
    return best_h, best_c5

def pair_opt(h, n, i, j, ternary_iters=25):
    old_i, old_j = h[i], h[j]
    h[i] = 0.0; h[j] = 0.0
    corr00 = get_corr(h)
    h[i] = 1.0
    corr10 = get_corr(h)
    h[j] = 1.0
    corr11 = get_corr(h)
    h[i] = old_i; h[j] = old_j
    d1 = corr10 - corr00
    d2 = corr11 - corr10
    lo_i, hi_i, lo_j, hi_j = 0.0, 1.0, 0.0, 1.0
    for _ in range(ternary_iters):
        mi1 = lo_i + (hi_i - lo_i) / 3.0
        mi2 = hi_i - (hi_i - lo_i) / 3.0
        mj1 = lo_j + (hi_j - lo_j) / 3.0
        mj2 = hi_j - (hi_j - lo_j) / 3.0
        c = np.max(corr00 + mi1 * d1 + mj1 * d2)
        d = np.max(corr00 + mi2 * d1 + mj1 * d2)
        if c < d:
            hi_i = mi2
        else:
            lo_i = mi1
        c = np.max(corr00 + lo_i * d1 + mj1 * d2)
        d = np.max(corr00 + lo_i * d1 + mj2 * d2)
        if c < d:
            hi_j = mj2
        else:
            lo_j = mj1
    return (lo_i + hi_i) / 2.0, (lo_j + hi_j) / 2.0

def basin_hopping(h, n, c5, t_end, n_iters, use_pairs):
    best_h = h.copy()
    best_c5 = c5
    work_h = h.copy()
    work_c5 = c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(n_iters, 1))
        mod = it % 6
        if mod == 0:
            pert = (0.01 + 0.04 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 12)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.01 + 0.03 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = (0.03 + 0.08 * cool) * np.random.randn(int(mask.sum()))
        elif mod == 3:
            pert = np.zeros(n)
            for _ in range(3):
                a, b = np.random.permutation(n)[:2]
                d = (work_h[b] - work_h[a]) * (0.3 + 0.7 * cool)
                pert[a] += d; pert[b] -= d
        elif mod == 4:
            mask = np.random.random(n) < 0.06
            pert = np.zeros(n)
            pert[mask] = np.random.choice([-1.0, 1.0], size=int(mask.sum())) * (0.05 + 0.1 * cool)
        else:
            pert = (0.005 + 0.02 * cool) * np.random.randn(n)
        h_pert = project_exact(work_h + pert, n)
        h_cd, c5_cd = coord_descent(h_pert, n, t_end, max_sweeps=30, ternary_iters=25)
        temp = 3e-5 + 2e-4 * cool
        if c5_cd < work_c5 or np.random.rand() < np.exp(-(c5_cd - work_c5) / temp):

        total_ones -= put_ones
