def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)

        h = h * (1.0 + diff / s)

    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50, 50)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        sh = np.clip(T * (c - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_step(h, n, k=5, alpha=0.005, momentum=0.7):
    corr = corr_array(h)
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for ki in top_idx:
        lag = ki - (n - 1)
        idx = np.arange(n)
        j1 = idx + lag
        m1 = (j1 >= 0) & (j1 < n)
        grad[m1] += (1.0 - h[j1[m1]])
        j2 = idx - lag
        m2 = (j2 >= 0) & (j2 < n)
        grad[m2] += (-h[j2[m2]])
    return project_h(h - alpha * grad / k, n), grad / k

    """Optimize coordinate i to minimize max correlation."""
    old = h[i]
    h[i] = 0.0
    cb = corr_array(h)
    h[i] = old
    # corr[ki] = cb[ki] + h[i] * coef[ki]
    # where coef depends on lag
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n-1), n)
    for idx, lag in enumerate(lags):
        ki = n - 1 + lag
        j1 = i + lag
        j2 = i - lag
        c = 0.0
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coefs[ki] = c
    # Minimize max(cb + hi * coefs) over hi in [0,1]
    def f(hi):
        return np.max(cb + hi * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 50})
    return res.x

    """Flip a block: h[start:start+length] -> 1-h[start:start+length]"""
    h2 = h.copy()
    h2[start:start+length] = 1.0 - h2[start:start+length]
    return project_h(h2, n)

    h2 = h.copy()

    h2[s1:s1+l1] = h2[s2:s2+l2]
    h2[s2:s2+l2] = temp

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        sh = np.clip(T * (c - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_step(h, n, k=5, alpha=0.005):
    corr = corr_array(h)
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    idx = np.arange(n)
    for ki in top_idx:
        lag = ki - (n - 1)
        j1 = idx + lag
        m1 = (j1 >= 0) & (j1 < n)
        grad[m1] += (1.0 - h[j1[m1]])
        j2 = idx - lag
        m2 = (j2 >= 0) & (j2 < n)
        grad[m2] += (-h[j2[m2]])
    return project_h(h - alpha * grad / k, n), grad / k

def coord_opt(h, n, i):
    old = h[i]
    h[i] = 0.0
    cb = corr_array(h)
    h[i] = old
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n - 1), n)
    j1 = i + lags
    v1 = (j1 >= 0) & (j1 < n)
    coefs[v1] += (1.0 - h[j1[v1]])
    j2 = i - lags
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v2] += (-h[j2[v2]])
    def f(hi):
        return np.max(cb + hi * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11, 'maxiter': 40})
    return res.x

def block_flip(h, n, start, length):
    h2 = h.copy()
    h2[start:start + length] = 1.0 - h2[start:start + length]
    return project_h(h2, n)

def block_swap(h, n, s1, l1, s2, l2):
    h2 = h.copy()
    temp = h2[s1:s1 + l1].copy()
    h2[s1:s1 + l1] = h2[s2:s2 + l2]
    h2[s2:s2 + l2] = temp
    return project_h(h2, n)

def make_thue_morse(n):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    return project_h(tm, n)

    rng = np.random.RandomState(seed)

    return rb

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):

                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5
