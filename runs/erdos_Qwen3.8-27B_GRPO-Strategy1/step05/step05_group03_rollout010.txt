import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def project_period(p, d):
    target = d / 2.0
    p = np.clip(p, 0.0, 1.0)
    s = p.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return p
    if diff > 0:
        room = 1.0 - p
        tot = room.sum()
        if tot < 1e-14:
            return p
        p = p + room * (diff / tot)
    else:
        if s < 1e-14:
            return p
        p = p * (1.0 + diff / s)
    return np.clip(p, 0.0, 1.0)

def slsqp_period(p, d, n, T, maxiter=250):
    def obj(x):
        h = np.tile(np.clip(x, 0, 1), n // d)[:n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)
    def cf(x):
        return x.sum() - d / 2.0
    try:
        res = minimize(obj, p, method='SLSQP',
                       bounds=[(0.0, 1.0)] * d,
                       constraints=[{'type': 'eq', 'fun': cf}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_period(np.clip(res.x, 0, 1), d)
    except Exception:
        return p

def slsqp_full(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cf(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cf}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt_1d(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n - 1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    res = minimize_scalar(lambda hi: np.max(corr_base + hi * coefs),
                          bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11, 'maxiter': 50})
    return float(res.x)

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_1d(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def multi_start_coord(h, n, t_end, n_starts=8, amp=0.05):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc.copy()
    return best_h, best_c5
