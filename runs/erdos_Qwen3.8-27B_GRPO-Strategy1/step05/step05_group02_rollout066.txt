    n = len(h)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    t, s = len(h)*0.5, h.sum()
    d = t - s
    if abs(d) < 1e-13: return h
    if d > 0:
        r = 1.0 - h
        if r.sum() > 1e-13: h = h + r*(d/r.sum())
