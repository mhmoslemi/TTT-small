import time

def fft_corr(h):

    L = 2 * N - 1

    G = np.fft.rfft(np.flip(1.0 - h), L)

    N = len(h)
    h = (h > 0.5).astype(float)
    target = N // 2

        if n1 > target:

            h[idx[target:]] = 0

            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:target - n1]] = 1

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h[::-1], L)
    return np.fft.irfft(H * G, L)

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, seed=0):
    rng = np.random.RandomState(seed)
    N = len(h)
    hb = (h > 0.5).astype(float)
    target = N // 2
    n1 = int(hb.sum())
    if n1 != target:
        if n1 > target:
            idx = np.where(hb == 1)[0]
            rng.shuffle(idx)
            hb[idx[target:]] = 0
        else:
            idx = np.where(hb == 0)[0]
            rng.shuffle(idx)
            hb[idx[:target - n1]] = 1
    return hb

def sa_search(h, deadline, T0=0.5, Tmin=0.002, alpha=0.9985, max_sweeps=50000, n_sample=60, use_2swap_every=80):
    N = len(h)
    h = h.astype(float).copy()
    c = fft_corr(h)
    P = float(c.max())
    best_c = P
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    L = 2 * N - 1

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        T = max(Tmin, T0 * alpha ** sweep)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()

        if len(ones) < 1 or len(zeros) < 1:
            break

        do_2swap = (use_2swap_every > 0 and sweep % use_2swap_every == use_2swap_every - 1 and len(ones) >= 2 and len(zeros) >= 2)

        best_new_P = P + 1e6
        best_move = None
        n_s = min(len(ones), n_sample)
        p_list = np.random.choice(ones, n_s, replace=False) if len(ones) > n_s else ones.copy()

        if not do_2swap:
            for p in p_list:
                lo_p = N - 1 - p
                hi_p = 2 * N - 1 - p
                base = c.copy()
                base[lo_p:hi_p + 1] -= (1.0 - h[p:(N - lo_p)])
                best_q_local = -1
                best_P_local = best_new_P
                for q in zeros:
                    lo_q = N - 1 - q
                    hi_q = 2 * N - 1 - q
                    nc = base.copy()
                    nc[lo_q:hi_q + 1] += (1.0 - h[q:(N - lo_q)])
                    nP = float(nc.max())
                    if nP < best_P_local:
                        best_P_local = nP
                        best_q_local = q
                if best_q_local >= 0:
                    if best_P_local < best_new_P:
                        best_new_P = best_P_local
                        best_move = ('1swap', int(p), int(best_q_local))
        else:
            # 2-swap: pick 2 ones, 2 zeros
            if len(ones) >= 2 and len(zeros) >= 2:
                p1, p2 = np.random.choice(ones, 2, replace=False)
                q_candidates = zeros
                base2 = c.copy()
                for pp in (p1, p2):
                    lo_pp = N - 1 - pp
                    hi_pp = 2 * N - 1 - pp
                    base2[lo_pp:hi_pp + 1] -= (1.0 - h[pp:(N - lo_pp)])
                for q1_idx in range(min(len(q_candidates), 30)):
                    q1 = int(q_candidates[q1_idx])
                    lo_q1 = N - 1 - q1
                    hi_q1 = 2 * N - 1 - q1
                    nc1 = base2.copy()
                    nc1[lo_q1:hi_q1 + 1] += (1.0 - h[q1:(N - lo_q1)])
                    m1 = float(nc1.max())
                    if m1 >= best_new_P:
                        continue
                    for q2_idx in range(q1_idx + 1, min(len(q_candidates), 30)):
                        q2 = int(q_candidates[q2_idx])
                        lo_q2 = N - 1 - q2
                        hi_q2 = 2 * N - 1 - q2
                        nc2 = nc1.copy()
                        nc2[lo_q2:hi_q2 + 1] += (1.0 - h[q2:(N - lo_q2)])
                        nP = float(nc2.max())
                        if nP < best_new_P:
                            best_new_P = nP
                            best_move = ('2swap', int(p1), int(p2), int(q1), int(q2))

        if best_move is None:
            # Random 1-swap as perturbation
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c = fft_corr(h)
            continue

        delta = best_new_P - P
        if delta <= 0 or np.random.random() < min(1.0, np.exp(-delta / max(T, 1e-12))):
            if best_move[0] == '1swap':
                p, q = best_move[1], best_move[2]
                h[p] = 0.0
                h[q] = 1.0
            else:
                p1, p2, q1, q2 = best_move[1], best_move[2], best_move[3], best_move[4]
                h[p1] = 0.0
                h[p2] = 0.0
                h[q1] = 1.0
                h[q2] = 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c = fft_corr(h)

    return best_h, best_c * (2.0 / N)

def targeted_fix(h, deadline, n_iters=2000):
    """Focus on the worst shift and try to reduce it specifically."""
    N = len(h)
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h))) * (2.0 / N)
    best_h = h.copy()

    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        c5 = P * (2.0 / N)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()

        k_star = int(np.argmax(c)) - (N - 1)  # the shift achieving max
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        # Find 1s that contribute to the worst shift and try to move them
        contrib_p = []
        for p in ones:
            i_plus = p + k_star
            i_minus = p - k_star
            if 0 <= i_plus < N and h[i_plus] == 0:
                contrib_p.append(p)
            elif 0 <= i_minus < N and h[i_minus] == 0:
                contrib_p.append(p)

        if len(contrib_p) == 0:
            contrib_p = list(ones[:10])

        best_new_P = P
        best_move = None
        for p in contrib_p[:15]:
            lo_p = N - 1 - p
            hi_p = 2 * N - 1 - p
            base = c.copy()
            base[lo_p:hi_p + 1] -= (1.0 - h[p:(N - lo_p)])
            for q in zeros:
                lo_q = N - 1 - q
                hi_q = 2 * N - 1 - q
                nc = base.copy()
                nc[lo_q:hi_q + 1] += (1.0 - h[q:(N - lo_q)])
                nP = float(nc.max())
                if nP < best_new_P:
                    best_new_P = nP
                    best_move = (p, q)

        if best_move is not None and best_new_P < P:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
        else:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    N0 = len(h_init)
    h0 = make_binary(h_init, seed)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Main SA (long run)
    d1 = min(deadline - 250, t0 + 380)
    h1, c5_1 = sa_search(best_h.copy(), d1, T0=0.4, Tmin=0.002, alpha=0.9988, max_sweeps=60000, n_sample=50, use_2swap_every=60)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Targeted fix on worst shift
    if time.time() < deadline - 180:
        d2 = min(deadline - 160, time.time() + 80)
        h2, c5_2 = targeted_fix(best_h.copy(), d2, n_iters=1500)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2

    # Phase 3: Another SA round from improved
    if time.time() < deadline - 120:
        d3 = min(deadline - 100, time.time() + 80)
        h3, c5_3 = sa_search(best_h.copy(), d3, T0=0.15, Tmin=0.001, alpha=0.999, max_sweeps=40000, n_sample=40, use_2swap_every=100)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3

    # Phase 4: Try nearby N values
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [N0 - 20, N0 + 10, N0 + 20, N0 - 10, N0 + 30]:
        if time.time() > deadline - 40:

        if N_try < 200 or N_try % 2 != 0:

        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = make_binary(h_interp, seed + N_try)
        sub_dl = min(deadline - 15, time.time() + 50)
        h_rt, c5_rt = sa_search(h_bt, sub_dl, T0=0.3, Tmin=0.003, alpha=0.999, max_sweeps=20000, n_sample=40, use_2swap_every=0)
        if c5_rt < best_c5:

    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
