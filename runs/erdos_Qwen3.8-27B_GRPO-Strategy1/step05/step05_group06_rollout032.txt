def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_fg(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    ts = T * (corr - mc)
    ts -= ts.max()
    exps = np.exp(ts)
    smax = mc + np.log(exps.sum()) / T
    w = exps / exps.sum()
    val = dx * smax + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m = 2 * n - 1
    lag_arr = np.arange(m) - n + 1
    for idx in range(m):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n - lg])
                grad[:n - lg] -= wi * h[lg:]
        else:
            p = -lg
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

    x0 = project_h(h0)
    def f(x):
        return smooth_max_fg(x, T, lam)[0]

        return smooth_max_fg(x, T, lam)[1]
    try:
        res = minimize(f, x0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,

    best_c5 = compute_c5(best_h)
    # Phase 1: Moderate-T smooth max
    t1 = t_end * 0.25
    for T in [100, 500, 2000]:
        if time.time() > t1:
            break
        hc = lbfgs_step(best_h, T, 300.0, 2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Phase 2: Top-k focus
    t2 = t_end * 0.40
    for k in [3, 5, 10]:
        if time.time() > t2:
            break
        hc = topk_lbfgs(best_h, k, 200.0, 1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Phase 3: Brief restarts
    t3 = t_end * 0.55
    inits = diverse_initials(n, best_h, seed)
    for h_init in inits:
        if time.time() > t3:
            break
        hc = lbfgs_step(h_init, 500, 300.0, 1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Phase 4: Extended coordinate descent (main workhorse)
    t4 = t_end * 0.80
    best_h, best_c5 = coord_descent(best_h, t4, max_sweeps=1000)
    # Phase 5: KKT polish
    t5 = min(t_end * 0.93, t4 + (t_end - t4) * 0.5)
    best_h, best_c5 = kkt_polish(best_h, t5, max_steps=500)
    # Phase 6: Final coord descent
    best_h, best_c5 = coord_descent(best_h, t_end, max_sweeps=500)
    return best_h, best_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h160 = np.array(initial_h_values, dtype=float).copy()
    else:
        h160 = np.zeros(160)
        h160[:80] = 1.0
    h160 = project_h(h160)
    best_h = h160.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    # Primary: n=160 with generous budget
    t_160_end = t0 + budget_s * 0.60
    th, c5v, _ = optimize_one(h160, t_160_end, seed)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = th
        best_n = 160
    # Secondary: n=200
    if time.time() < t0 + budget_s * 0.75:
        h200 = interpolate_h(h160, 160, 200)
        t_200_end = t0 + budget_s * 0.85
        th, c5v, _ = optimize_one(h200, t_200_end, seed + 7)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = th
            best_n = 200
    # Tertiary: n=120
    if time.time() < t0 + budget_s * 0.90:
        h120 = interpolate_h(h160, 160, 120)
        t_120_end = t0 + budget_s * 0.93
        th, c5v, _ = optimize_one(h120, t_120_end, seed + 13)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = th
            best_n = 120
    # Final polish on global best
    t_fin = t0 + budget_s * 0.97
    if time.time() < t_fin:
        best_h, best_c5 = coord_descent(best_h, t_fin, max_sweeps=400)
        best_h, best_c5 = kkt_polish(best_h, min(t0 + budget_s * 0.99, t_fin + 5), 200)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
