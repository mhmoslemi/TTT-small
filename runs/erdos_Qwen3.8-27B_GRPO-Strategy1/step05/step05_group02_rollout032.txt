    # D(s) = base(s) + h[p-s]*b(s)
    # E(s) = (1-h[p+s])*a(s) - h[p-s]*b(s)
    
    s_range = np.arange(-(n-1), n)  # all shifts
    a = np.zeros(2*n-1)  # a(s): 1 if 0<=p+s<=n-1
    b = np.zeros(2*n-1)  # b(s): 1 if 0<=p-s<=n-1 and s!=0
    
    for idx, s in enumerate(s_range):
        if 0 <= p+s <= n-1:
            a[idx] = 1.0
        if s != 0 and 0 <= p-s <= n-1:
            b[idx] = 1.0
    
    # D(s) = corr[s_idx] - h[p]*((1-h[p+s])*a(s) - h[p-s]*b(s)) + h[p-s]*b(s)
    # Wait, I have corr = c(s) = D(s) + h[p]*E(s)
    # So D(s) = corr[s_idx] - h[p]*E(s)
    # But I don't know E(s) yet... let me compute it directly.
    
    # E(s) = (1-h[p+s])*a(s) - h[p-s]*b(s)
    E = np.zeros(2*n-1)
    D = np.zeros(2*n-1)
    D0 = 0.0  # base(0) = corr[0] - h[p]*(1-h[p]) = corr[n-1] - h[p]*(1-h[p])
    
    D0 = corr[n-1] - h[p]*(1-h[p])  # base(0)
    
    for idx, s in enumerate(s_range):
        if s == 0:
            continue

        if 0 <= p+s <= n-1:
            Es += (1.0 - h[p+s])
        if 0 <= p-s <= n-1:
            Es -= h[p-s]
        E[idx] = Es
        D[idx] = corr[idx] - h[p] * Es
    
    # Now objective: f(hp) = max(D0 + hp - hp², max_s(D[s] + hp*E[s]))
    # We want to minimize f over hp in [0,1]
    
    from scipy.optimize import minimize_scalar
    def obj(hp):
        vals = D + hp * E
        v0 = D0 + hp - hp*hp
        return max(np.max(vals), v0)
    
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    new_hp = float(res.x)
    
    # Update correlation
    delta = new_hp - h[p]
    new_corr = corr.copy()
    for idx, s in enumerate(s_range):
        if s == 0:
            new_corr[idx] = D0 + new_hp - new_hp*new_hp
        else:
            new_corr[idx] = D[idx] + new_hp * E[idx]
    
    return new_hp, new_corr

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_lse(corr, T):
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return m + np.log(np.sum(np.exp(shifted))) / T

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_update(corr, h, p, n):
    D0 = corr[n-1] - h[p] * (1.0 - h[p])
    E = np.zeros(2*n-1)
    D = np.zeros(2*n-1)
    for idx in range(2*n-1):
        s = idx - (n-1)
        if s == 0:

        Es = 0.0
        ps = p + s
        if 0 <= ps < n:
            Es += (1.0 - h[ps])
        pms = p - s
        if 0 <= pms < n:
            Es -= h[pms]

        D[idx] = corr[idx] - h[p] * Es
    def obj(hp):

        vm = np.max(D + hp * E)
        return max(v0, float(vm))

    new_corr[n-1] = D0 + new_hp - new_hp*new_hp
    return new_hp, new_corr

def slsqp_smooth(h, T, n, maxiter=500):
    def obj(x):
        return smooth_lse(np.correlate(x, 1.0-x, mode='full'), T) * (2.0/n)
    def ceq(x):
        return x.sum() - n/2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0,1.0)]*n,
                      constraints=[{'type':'eq','fun':ceq,'args':(n,)}],
                      options={'maxiter':maxiter,'ftol':1e-15})
        return project_h(np.clip(res.x,0,1))
    except:
        return h

def pair_swap_anneal(h, n, t_end, n_trials=50000):
    h = h.copy()
    c5 = compute_c5(h)
    corr = np.correlate(h, 1.0-h, mode='full')
    rng = np.random.RandomState(12345)
    T_cur = 0.002
    for trial in range(n_trials):
        if time.time() > t_end:
            break
        T_cur *= 0.9997
        # Pick a 1-ish and a 0-ish position
        ones = np.where(h > 0.6)[0]
        zeros = np.where(h < 0.4)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        # Swap
        h[i], h[j] = h[j], h[i]
        h = np.clip(h, 0, 1)
        c5_new = compute_c5(h)
        if c5_new < c5 or rng.rand() < np.exp(-(c5_new - c5)/max(T_cur, 1e-12)):
            c5 = c5_new
            corr = np.correlate(h, 1.0-h, mode='full')
        else:
            h[i], h[j] = h[j], h[i]  # undo
    h = project_h(h)
    c5 = compute_c5(h)
    return h, c5

def make_start(n, rng, mode):
    h = np.zeros(n)
    if mode == 0:  # block
        h[:n//2] = 1.0
    elif mode == 1:  # alternating with noise
        h[::2] = 1.0
        h += 0.1 * rng.standard_normal(n)
    elif mode == 2:  # random
        h = (rng.rand(n) > 0.5).astype(float)
        h += 0.05 * rng.standard_normal(n)
    elif mode == 3:  # golden ratio
        phi = (1 + np.sqrt(5)) / 2
        for i in range(n):
            h[i] = 1.0 if (i * (phi % 1)) % 1 < 0.5 else 0.0
    elif mode == 4:  # Thue-Morse like
        for i in range(n):
            h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return project_h(h)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_lse(corr, T):
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return m + np.log(np.sum(np.exp(shifted))) / T

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _slsqp_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_lse(corr, T) * (2.0 / n)

def _slsqp_con(x, n):
    return x.sum() - n / 2.0

def _cd_1d(hp, D, E, D0):
    v0 = D0 + hp - hp * hp
    vm = float(np.max(D + hp * E))
    return max(v0, vm)

def fast_coord_update(corr, h, p, n):
    D0 = corr[n - 1] - h[p] * (1.0 - h[p])
    E = np.zeros(2 * n - 1)
    D = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        if s == 0:
            continue
        Es = 0.0
        ps = p + s
        if 0 <= ps < n:
            Es += (1.0 - h[ps])
        pms = p - s
        if 0 <= pms < n:
            Es -= h[pms]
        E[idx] = Es
        D[idx] = corr[idx] - h[p] * Es
    res = minimize_scalar(_cd_1d, args=(D, E, D0), bounds=(0.0, 1.0),
                          method='bounded', options={'xatol': 1e-11, 'maxiter': 80})
    new_hp = float(res.x)
    new_corr = D.copy() + new_hp * E
    new_corr[n - 1] = D0 + new_hp - new_hp * new_hp
    return new_hp, new_corr

def cd_fast(h, n, t_end, max_sweeps=5000, seed=0):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * 2.0 / n)
    rng = np.random.RandomState(seed)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            nh, nc = fast_coord_update(corr, h, i, n)
            if abs(nh - h[i]) > 1e-12:
                h[i] = nh
                corr = nc
                imp = True
        if sw % 50 == 49:
            h = project_h(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5v = float(np.max(corr) * 2.0 / n)
            if c5v < best_c5:
                best_c5 = c5v
        if not imp:
            break
    h = project_h(h)
    best_c5 = compute_c5(h)
    return h, best_c5

def slsqp_step(h, T, n, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(T, n), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def perturb_h(h, n, scale, rng):
    p = scale * rng.standard_normal(n)
    return project_h(h + p)

def binary_refine(h, n, t_end, n_trials=80000, seed=99):
    h = h.copy()
    c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    T_cur = 0.003
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        T_cur *= 0.9998
        ones = np.where(h > 0.55)[0]
        zeros = np.where(h < 0.45)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        h = np.clip(h, 0, 1)
        c5_new = compute_c5(h)
        accept = c5_new < c5
        if not accept and c5_new - c5 < 0.01:
            accept = rng.rand() < np.exp(-(c5_new - c5) / max(T_cur, 1e-12))
        if accept:
            c5 = c5_new
        else:
            h[i], h[j] = h[j], h[i]
    h = project_h(h)
    c5 = compute_c5(h)
    return h, c5

    t0 = time.time()

    t_cd1 = t0 + budget_s * 0.43

    t_pt2 = t0 + budget_s * 0.53
