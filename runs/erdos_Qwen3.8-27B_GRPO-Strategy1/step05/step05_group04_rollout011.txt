    rng = np.random.RandomState(seed)
    best_h = h.copy()

                return best_h, best_c5

    candidates.append((old + 1.0) * 0.5)
    candidates.append(max(0.0, old - 0.1))
    candidates.append(min(1.0, old + 0.1))
    
    best_v = old
    best_f = val(old)
    for v in candidates:
        if v < 0 or v > 1:
            continue
        f = val(v)
        if f < best_f:
            best_f = f
            best_v = v
    return best_v
