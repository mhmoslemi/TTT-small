    dx = 2.0 / n

    for s in range(-(n-1), n):
        w = sm_grad[s + n - 1]

                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            a, b = 0.0, 1.0
            for _ in range(45):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(base + m1 * coefs) < np.max(base + m2 * coefs):
                    b = m2
                else:
                    a = m1
            t_new = (a + b) / 2.0
            if abs(t_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def perturb_opt(h, n, t_end, scale, n_pert=4):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    per_time = 30.0 / n_pert
    for p in range(n_pert):
        if time.time() > t_end - 2:
            break
        r = np.random.random()
        if r < 0.3:
            pert = scale * np.random.randn(n)
        elif r < 0.6:
            blen = np.random.randint(max(n // 5, 4), max(n // 2, 5))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * np.random.randn(blen)
        else:
            f = np.random.randint(2, 16)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = scale * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        h_p = project_h(h + pert)
        t_sub = min(time.time() + per_time, t_end)
        h_o, c5_o = fista_opt(h_p, n, t_sub,
                              T_start=3000, T_end=30000, n_iter=800,
                              lr_start=0.003, lr_end=0.0005)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o
    return best_h, best_c5

def optimize_n(h_init, n, t_end):
    best_h = project_h(h_init.copy())
    best_c5 = compute_c5(best_h)
    if t_end - time.time() > 25:
        for T in [100, 1000, 10000]:
            if t_end - time.time() < 15:
                break
            hc = slsqp_smooth(best_h, T, n, 150, 1e-12)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if t_end - time.time() > 35:
        t_sub = t_end - (t_end - time.time()) * 0.25
        h_f, c5_f = fista_opt(best_h, n, t_sub,
                               T_start=3000, T_end=1e8, n_iter=4000,
                               lr_start=0.004, lr_end=0.0001)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if t_end - time.time() > 20:
        t_sub = t_end - (t_end - time.time()) * 0.15
        best_h, best_c5 = coord_descent(best_h, n, t_sub, 150, 25)
    if t_end - time.time() > 25:
        t_sub = t_end - (t_end - time.time()) * 0.3
        best_h, best_c5 = perturb_opt(best_h, n, t_sub, 0.07, 4)
    if t_end - time.time() > 15:
        t_sub = t_end - (t_end - time.time()) * 0.1
        h_f, c5_f = fista_opt(best_h, n, t_sub,
                               T_start=8000, T_end=1e8, n_iter=2500,
                               lr_start=0.002, lr_end=0.0001)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if t_end - time.time() > 10:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 2, 100, 30)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0
    n_parent = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)
    t_p1 = t0 + budget_s * 0.48
    h1, c5_1 = optimize_n(best_h, n_parent, t_p1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    for n_try in [180, 250, 300]:
        if t_end - time.time() < budget_s * 0.22:
            break
        h_try = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                          np.linspace(0, 2, len(best_h), endpoint=False), best_h)
        h_try = project_h(h_try)
        t_sub = min(t0 + budget_s * 0.78, t_end - (t_end - time.time()) * 0.25)
        h_opt, c5_try = optimize_n(h_try, n_try, t_sub)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_opt
    if t_end - time.time() > 20:
        best_h, best_c5 = coord_descent(best_h, len(best_h), t_end - 3, 120, 30)
    if t_end - time.time() > 10:
        h_f, c5_f = fista_opt(best_h, len(best_h), t_end - 2,
                               T_start=20000, T_end=1e9, n_iter=1500,
                               lr_start=0.001, lr_end=0.00005)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
