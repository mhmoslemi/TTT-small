import numpy as np
from scipy.optimize import minimize_scalar, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            mask = h > 0
            if mask.sum() == 0:
                break
            h[mask] = h[mask] * (target / s)
            h = np.clip(h, 0, 1)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
            h = np.clip(h, 0, 1)
    return h

def coord_descent_fast(h, n, t_end, max_sweeps=2000, seed=42):
    # Same as before but optimized
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 3 == 2 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            # Linear in v for each correlation coefficient
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                    lin[mn] -= best_h[i - s]
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                    lin[mn] += (1.0 - best_h[i + s])
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            # Minimize max(c_base + d*lin - d²*e_n) over d = v - old, v in [0,1]
            # This is: min max_j (c_base[j] + d*lin[j] - d² * delta(j,n-1)) 
            # For j != n-1: linear in d
            # For j = n-1: c_base[n-1] + d*lin[n-1] - d²
            # Use bounded scalar minimization
            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val))
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

So total = ∫₀^{1-k} [h(x)(1-h(x+k)) + (1-h(x))h(x+k)] dx + ∫_{1-k}^{1} h(x)(1-h(x+k)) dx

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(60):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 0.0)
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h *= (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                    lin[mn] -= best_h[i - s]
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                    lin[mn] += (1.0 - best_h[i + s])
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val))
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-14:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def lns_search(h, n, t_end, seed=42, n_iter=500):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iter)
        n_flip = int(1 + 3 * cool * rng.rand())
        r = rng.rand()
        if r < 0.35:
            zeros = np.where(cur_h < 0.05)[0]
            if len(zeros) > 0:
                idx = rng.choice(zeros, size=min(n_flip, len(zeros)), replace=False)
                cur_h[idx] = 1.0
        elif r < 0.70:
            ones = np.where(cur_h > 0.95)[0]
            if len(ones) > 0:
                idx = rng.choice(ones, size=min(n_flip, len(ones)), replace=False)
                cur_h[idx] = 0.0
        elif r < 0.85:
            active = np.where((cur_h > 0.05) & (cur_h < 0.95))[0]
            if len(active) > 0:
                idx = rng.choice(active, size=min(n_flip, len(active)), replace=False)
                cur_h[idx] += rng.uniform(-0.4, 0.4, len(idx))
            else:
                idx = rng.choice(n, size=n_flip, replace=False)
                cur_h[idx] += rng.uniform(-0.3, 0.3, len(idx))
        else:
            block = rng.randint(0, n - 10)
            size = rng.randint(5, 20)
            cur_h[block:block+size] = 1.0 - cur_h[block:block+size]
        cur_h = project_h(cur_h, n)
        punch_time = min(t_end, time.time() + max(0.5, 3.0 * cool))
        h_opt, c5_opt = coord_descent(cur_h, n, punch_time, max_sweeps=50, seed=seed + it)
        temp = 0.002 * cool + 2e-5
        if c5_opt < cur_c5 or (c5_opt - cur_c5 < temp and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def basin_hop_diverse(h, n, t_end, seed=42, n_hops=300):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - hop / n_hops)
        scale = 0.02 + 0.10 * cool
        mode = hop % 7
        if mode == 0:
            pert = scale * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = scale * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 2.5 * rng.randn(mask.sum())
        elif mode == 3:
            k = rng.randint(3, n // 8)
            start = rng.randint(0, n - k)
            pert = np.zeros(n)
            pert[start:start+k] = scale * 2 * rng.randn(k)
        elif mode == 4:
            idx = rng.choice(n, size=max(4, n // 12), replace=False)
            pert = np.zeros(n)
            pert[idx] = scale * 1.5 * rng.randn(len(idx))
        elif mode == 5:
            f = rng.randint(2, 8)
            pert = scale * 0.7 * np.cos(2 * np.pi * f * np.arange(n) / n)
        else:
            pert = scale * 0.4 * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        hop_end = min(t_end, time.time() + max(1.5, 6.0 * cool))
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=40, seed=seed + hop)
        temp = 0.003 * cool + 3e-5
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def make_init_block(n, block_len=20, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    pos = 0
    while pos < n:
        if rng.rand() < 0.5:
            h[pos:pos+block_len] = 1.0
        pos += block_len
    h = project_h(h, n)
    return h

def make_init_smooth(n, freq=2, seed=0):
    rng = np.random.RandomState(seed)
    t = np.linspace(0, 1, n)
    h = np.zeros(n)
    a = rng.uniform(0.2, 0.6)
    ph = rng.uniform(0, 2 * np.pi)
    h = a + a * np.sin(2 * np.pi * freq * t + ph)
    h = np.clip(h, 0.0, 1.0)
    h = project_h(h, n)
    return h
