But actually, I can be smarter. The change in correlation from swapping h[i] (1→0) and h[j] (0→1) can be computed in O(n) instead of O(n) for the full correlate:

When we change h[i] from 1 to 0:
- For each lag k, the term h[i] * (1-h[i-k]) changes from (1-h[i-k]) to 0
- For each lag k, the term h[i+k] * (1-h[i]) changes from 0 to h[i+k]

When we change h[j] from 0 to 1:
- For each lag k, the term h[j] * (1-h[j-k]) changes from 0 to (1-h[j-k])
- For each lag k, the term h[j+k] * (1-h[j]) changes from h[j+k]*(1-0) = h[j+k] to 0

So the net change in correlation at lag k (where k = j-i or i-j) is:
- From i: remove (1-h[i-k]), add h[i+k]
- From j: add (1-h[j-k]), remove h[j+k]

This is O(1) per lag, so O(n) total for computing the new max. But we still need to check if the new max is less than the old max, which requires scanning the whole correlation vector.

        cur_c5 = best_c5
        stall = 0
        while time.time() < t_end - 0.5 and stall < 300:
            mt = rng.randint(4)
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h <= 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            if mt == 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                cur_h[i], cur_h[j] = 0.0, 1.0
                nc = compute_c5(cur_h)
                if nc < cur_c5 - 1e-15:
                    cur_c5 = nc
                    stall = 0
                else:
                    cur_h[i], cur_h[j] = 1.0, 0.0
                    stall += 1
            elif mt == 1:
                if len(ones) < 2 or len(zeros) < 2:
                    stall += 1
                    continue
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                if len(set([i1, i2, j1, j2])) < 4:
                    stall += 1
                    continue
                cur_h[i1] = 0; cur_h[i2] = 0
                cur_h[j1] = 1; cur_h[j2] = 1
                nc = compute_c5(cur_h)
                if nc < cur_c5 - 1e-15:
                    cur_c5 = nc
                    stall = 0
                else:
                    cur_h[i1] = 1; cur_h[i2] = 1
                    cur_h[j1] = 0; cur_h[j2] = 0
                    stall += 1
            elif mt == 2:
                a, b = sorted(rng.choice(n, 2, replace=False))
                if b - a < 2:
                    stall += 1
                    continue
                seg = cur_h[a:b + 1].copy()
                cur_h[a:b + 1] = seg[::-1]
                nc = compute_c5(cur_h)
                if nc < cur_c5 - 1e-15:
                    cur_c5 = nc
                    stall = 0
                else:
                    cur_h[a:b + 1] = seg
                    stall += 1
            else:
                blen = rng.randint(2, 8)
                a = rng.randint(0, max(1, n - blen))
                b = a + blen
                c = rng.randint(0, max(1, n - blen))
                d = c + blen
                if c <= b and d > a:
                    stall += 1
                    continue
                s1 = cur_h[a:b].copy()
                s2 = cur_h[c:d].copy()
                cur_h[a:b] = s2
                cur_h[c:d] = s1
                nc = compute_c5(cur_h)
                if nc < cur_c5 - 1e-15:
                    cur_c5 = nc
                    stall = 0
                else:
                    cur_h[a:b] = s1
                    cur_h[c:d] = s2
                    stall += 1
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()
        kick_count += 1
        nk = min(3 + kick_count // 3, 25)
        for _ in range(nk):
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h <= 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_h[i], best_h[j] = best_h[j], best_h[i]
    return best_h, best_c5

def best_improve_pass(h, n, max_rounds):
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    for _ in range(max_rounds):
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 1e-15
        best_i, best_j = -1, -1
        for i in ones:
            cur_h[i] = 0.0
            for j in zeros:
                cur_h[j] = 1.0
                c5v = compute_c5(cur_h)
                if cur_c5 - c5v > best_gain:
                    best_gain = cur_c5 - c5v
                    best_i, best_j = i, j
                cur_h[j] = 0.0
            cur_h[i] = 1.0
        if best_i < 0:
            break
        cur_h[best_i] = 0.0
        cur_h[best_j] = 1.0
        cur_c5 = compute_c5(cur_h)
    return cur_h, cur_c5

def upsample(h, n_new):
    n_old = len(h)
    idx = (np.arange(n_new) * n_old / n_new).astype(int)
    idx = np.clip(idx, 0, n_old - 1)
    return h[idx].copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_frac(best_h)
    best_c5 = compute_c5(best_h)

    t_p1 = t0 + budget_s * 0.38
    t_p2 = t0 + budget_s * 0.55
    t_p3 = t0 + budget_s * 0.72
    t_p4 = t0 + budget_s * 0.88
    t_final = t0 + budget_s * 0.95

    # Phase 1a: Binary projection + best-improve
    bh = to_binary(best_h)
    bh, bc5 = best_improve_pass(bh, n, max_rounds=8)
    if bc5 < best_c5:
        best_c5 = bc5
        best_h = bh.copy()

    # Phase 1b: Iterated local search (binary)
    bh = to_binary(best_h)
    bh, bc5 = iterated_local_search(bh, n, rng, t_p1)
    if bc5 < best_c5:
        best_c5 = bc5
        best_h = bh.copy()

    # Phase 2: SLSQP fractional continuation
    for T in (200, 800, 4000, 20000, 100000):
        if time.time() > t_p2:
            break
        hc = slsqp_run(best_h, n, T, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t_p3, max_sweeps=80)

    # Phase 4: More local search on re-binarized + SLSQP
    bh = to_binary(best_h)
    bh, bc5 = iterated_local_search(bh, n, rng, t_p4)
    if bc5 < best_c5:
        best_c5 = bc5
        best_h = bh.copy()
    for T in (50000, 300000, 1500000):
        if time.time() > t_final:
            break
        hc = slsqp_run(best_h, n, T, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Final CD + tiny SLSQP
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=30)
    if time.time() < budget_s * 0.98:
        hc = slsqp_run(best_h, n, 5000000, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_frac(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
