def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def soft_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if T < 1e-6:
        return m
    scaled = (corr - m) / T
    return m + T * np.log(np.mean(np.exp(scaled)))

        return project_h(np.clip(res.x, 0, 1))

def grad_top_k(h, k_shifts):
    """Compute subgradient at top-k shifts."""
    n = len(h)
    dx = 2.0 / n
    # For shift k, d/dx_i of sum_j h[j](1-h[j+k]) = (1-h[i+k]) - h[i-k] (with boundary handling)
    i_idx = np.arange(n)[:, None]
    grad = np.zeros((n, len(k_shifts)))

        jp = i_idx + k
        jm = i_idx - k
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)

        jmc = np.clip(jm, 0, n - 1)
        grad[:, t] = (1.0 - h[jpc]) * v1 - h[jmc] * v2
    return grad * dx  # shape (n, len(k_shifts))

                c = 0.0
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def freq_perturb(n, scale, cool):
    x = np.arange(n)
    r = np.random.random()
    if r < 0.35:
        f = np.random.choice([2, 3, 4, 5, 6, 8, 10])
        p = np.random.uniform(0, 2 * np.pi)
        return scale * cool * np.sin(2 * np.pi * f * x / n + p)
    if r < 0.55:
        f = np.random.choice([1, 2, 3])
        p1 = np.random.uniform(0, 2 * np.pi)
        p2 = np.random.uniform(0, 2 * np.pi)
        return scale * cool * (np.sin(2*np.pi*f*x/n + p1) + 0.5*np.sin(2*np.pi*2*f*x/n + p2))
    if r < 0.75:
        blen = np.random.randint(max(n//8, 3), max(n//3, 4))
        bs = np.random.randint(0, max(n - blen, 1))
        pert = np.zeros(n)
        pert[bs:bs+blen] = scale * cool * 1.5 * np.random.randn(blen)
        return pert
    return scale * cool * np.random.randn(n)

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(1.0 - it / max(n_iter, 1), 0.05)
        pert = freq_perturb(n, scale, cool)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp_opt(h_pert, T, n, 300, 1e-14)
        c5v = compute_c5(hc)
        temp = 0.0006 * cool + 0.00008
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_block(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # SLSQP warmup with increasing T
    for T in [200, 1000, 5000, 20000, 100000]:
        if t_end - time.time() < 12:
            break
        hc = slsqp_opt(best_h, T, n, 400, 1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Basin hopping
    if t_end - time.time() > 35:
        t_bh = t_end - (t_end - time.time()) * 0.25
        bh_h, bh_c5 = basin_hop(best_h, n, t_bh, 20, 0.1, 2000, 5000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Surgical CD
    if t_end - time.time() > 18:
        t_cd = t_end - (t_end - time.time()) * 0.3
        best_h, best_c5 = surgical_cd(best_h, n, t_cd, 70, 40)
    # Fine basin hop
    if t_end - time.time() > 22:
        t_fh = t_end - (t_end - time.time()) * 0.35
        bh_h, bh_c5 = basin_hop(best_h, n, t_fh, 15, 0.04, 10000, 30000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Final surgical with high top-k
    if t_end - time.time() > 12:
        best_h, best_c5 = surgical_cd(best_h, n, t_end - 4, 50, 60)
    # Very high T SLSQP polish
    if t_end - time.time() > 8:
        hc = slsqp_opt(best_h, 5000000, n, 800, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Penalty SLSQP on top few
    if t_end - time.time() > 5:
        hc = slsqp_opt(best_h, 1e6, n, 600, 1e-16, k_pen=3, extra_pen=0.02)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def pairwise_swap(h, n, t_end, n_sweeps=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0 / n)
    top_idx = np.argsort(corr)[-10:]
    top_shifts = (top_idx - (n - 1)).astype(int)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(n * 2):
            if time.time() > t_end:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            di = np.random.uniform(-0.1, 0.1)
            dj = -di * best_h[i] / max(best_h[j], 0.01) if best_h[j] > 0.01 else -di
            ni = np.clip(best_h[i] + di, 0, 1)
            nj = np.clip(best_h[j] + dj, 0, 1)
            if abs(ni - best_h[i]) < 1e-10 and abs(nj - best_h[j]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = ni
            h_try[j] = nj
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0
    n_parent = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)
    # Main optimization at parent n
    t_p1 = t0 + budget_s * 0.42
    best_h, best_c5 = optimize_block(best_h, n_parent, t_p1)
    # Finer resolution
    n2 = min(int(n_parent * 1.25), 320)
    if t_end - time.time() > budget_s * 0.28:
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n_parent, endpoint=False), best_h)
        h2 = project_h(h2)
        t_p2 = t0 + budget_s * 0.68
        h2o, c52 = optimize_block(h2, n2, t_p2)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2o
    # Even finer
    n3 = min(int(n_parent * 1.5), 400)
    if t_end - time.time() > budget_s * 0.18 and n3 != n2:
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, len(best_h), endpoint=False), best_h)
        h3 = project_h(h3)
        t_p3 = t0 + budget_s * 0.88
        h3o, c53 = optimize_block(h3, n3, t_p3)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3o
    # Pairwise swaps + final surgical
    n_final = len(best_h)
    if t_end - time.time() > 15:
        best_h, best_c5 = pairwise_swap(best_h, n_final, t_end - 10, 20)
    if t_end - time.time() > 8:
        best_h, best_c5 = surgical_cd(best_h, n_final, t_end - 3, 40, 80)
    if t_end - time.time() > 4:
        hc = slsqp_opt(best_h, 1e7, n_final, 500, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
