import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    c = np.fft.irfft(H * np.conj(G), L)
    return c

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D_matrix(h, N):
    """Compute D[i,k] = partial c[k-1+N] / partial h[i] for all i, k"""
    # D[i, k_offset] where k_offset = 0..2N-2 corresponds to shift (k_offset - (N-1))
    # c[shift] = sum_j h[j]*(1-h[j+shift])
    # d(c[shift])/d(h[i]) = (1-h[i+shift]) - h[i-shift] (with bounds)
    idx = np.arange(2*N-1)
    shifts = idx - (N - 1)  # -N+1 to N-1
    pos = np.arange(N)[:, None]  # (N, 1)
    sv = shifts[None, :]  # (1, 2N-1)
    ip = pos + sv  # i + shift
    im = pos - sv  # i - shift
    valid_p = (ip >= 0) & (ip < N)
    valid_m = (im >= 0) & (im < N)
    ip_c = np.clip(ip, 0, N-1)
    im_c = np.clip(im, 0, N-1)
    hp = np.where(valid_p, h[ip_c], 0.0)
    hm = np.where(valid_m, h[im_c], 0.0)
    D = (1.0 - hp) * valid_p - hm * valid_m
    return D

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * np.conj(G), L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    """D[i, k_offset] = d c[shift] / d h[i]"""
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    ip = pos + sv
    im = pos - sv
    vp = (ip >= 0) & (ip < N)
    vm = (im >= 0) & (im < N)
    ip_c = np.clip(ip, 0, N - 1)
    im_c = np.clip(im, 0, N - 1)
    return (1.0 - h[ip_c]) * vp - h[im_c] * vm

def binary_search_full(h, N, deadline, max_sweeps=5000, stall_patience=100):

def nesterov_softmax(h, N, deadline, beta=100, k_top=20, lr=0.01):

    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta_m = 0.9

        if time.time() > deadline:
            break
        
        y = h + m
        c = fft_corr(y)
        # Soft-max gradient
        beta_local = min(beta + it * 0.5, 500)
        shifted = beta_local * c - beta_local * c.max()
        w = np.exp(shifted)
        w /= w.sum()
        
        # Gradient of sum_k w[k] * c[k] wrt h
        # c[k] = sum_j h[j]*(1-h[j+k])
        # grad[i] = sum_k w[k] * ((1-h[i+k]) - h[i-k])

        # For efficiency, only use top k_top weights
        top_idx = np.argsort(w)[-k_top:]
        w_top = w[top_idx]
        w_top /= w_top.sum()
        
        D = compute_D(y, N)  # (N, 2N-1)
        grad = D.T @ w_top  # (N,) -- wait, this gives sum_k D[i,k]*w[k] for each i
        
        # Project: h_new = project(y - lr*grad)
        h_new = y - lr * grad

        h_new = np.clip(h_new, 0, 1)

        s = h_new.sum()
        if abs(s - target) > 1e-8:
            # Binary search for shift

                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo+hi)/2, 0, 1)
        
        m = beta_m * (h_new - h)
        h = h_new
        
        curr = float(np.max(fft_corr(h)))
        if curr < best_val:
            best_val = curr
            best_h = h.copy()

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * np.conj(G), L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    ip = pos + sv
    im = pos - sv
    vp = (ip >= 0) & (ip < N)
    vm = (im >= 0) & (im < N)
    ip_c = np.clip(ip, 0, N - 1)
    im_c = np.clip(im, 0, N - 1)
    return (1.0 - h[ip_c]) * vp - h[im_c] * vm

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def exhaustive_bin(h, N, deadline, max_sweeps=9999, patience=120, perturb_n=4):
    h = h.copy()
    nh = N // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) != nh:
        if len(ones) > nh:
            order = ones[np.argsort(np.random.random(len(ones)))]
            h[order[nh:]] = 0
        else:
            order = zeros[np.argsort(np.random.random(len(zeros)))]
            h[order[:nh - len(ones)]] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    c = fft_corr(h)
    D = compute_D(h, N)
    best_val = float(c.max())
    best_h = h.copy()
    stall = 0
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_val - 1e-12:
            best_val = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > patience:
            for _ in range(perturb_n):
                if len(ones) < 2 or len(zeros) < 2:
                    break
                pi = np.random.randint(len(ones))
                qi = np.random.randint(len(zeros))
                h[ones[pi]], h[zeros[qi]] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c = fft_corr(h)
            D = compute_D(h, N)
            stall = 0
            continue
        D_ones = D[ones]
        D_zeros = D[zeros]
        best_P = P + 1.0
        best_pair = None
        for pi in range(len(ones)):
            base = c - D_ones[pi]
            nm = (base[None, :] + D_zeros).max(axis=1)
            mi = int(np.argmin(nm))
            if nm[mi] < best_P:
                best_P = float(nm[mi])
                best_pair = (int(ones[pi]), int(zeros[mi]))
        if best_pair is None:
            break
        p, q = best_pair
        h[p], h[q] = 0.0, 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        c = fft_corr(h)
        D = compute_D(h, N)
    return best_h, best_val * (2.0 / N)

def nesterov_sm(h, N, deadline, beta0=60, ktop=20, lr0=0.018):
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta_mom = 0.9
    beta = beta0
    lr = lr0
    for it in range(20000):
        if time.time() > deadline:
            break
        y = h + m
        c = fft_corr(y)
        b = min(beta + it * 0.3, 400.0)
        sh = b * c - b * c.max()
        w = np.exp(sh)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(2 * N - 1) / (2 * N - 1)
        else:
            w /= wsum
        top = np.argsort(w)[-ktop:]
        wt = w[top] / w[top].sum()
        D = compute_D(y, N)
        grad = D[:, top].T @ wt
        h_new = project_sum(y - lr * grad, target)
        m = beta_mom * (h_new - h)
        h = h_new
        cv = float(np.max(fft_corr(h)))
        if cv < best_val:
            best_val = cv
            best_h = h.copy()
        if it % 400 == 0:
            lr *= 0.96
            beta_mom = min(0.97, 0.85 + it * 0.0003)
    return best_h, best_val * (2.0 / N)

def to_binary(h, N):
    target = N // 2
    n1 = int(np.sum(h >= 0.5))
    h = (h >= 0.5).astype(float)
    if n1 != target:
        if n1 > target:
            idx = np.where(h == 1)[0]
            fl = idx[np.argsort(h[idx])][:n1 - target]
            h[fl] = 0
        else:
            idx = np.where(h == 0)[0]
            fl = idx[np.argsort(-h[idx])][:target - n1]
            h[fl] = 1
    return h
