def targeted_perturb(h, n, step, rng):
    """Generate perturbation targeting the active max correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find active lags (within 5% of max)
    active = np.where(corr > 0.97 * max_val)[0]
    
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        if s > 0:
            # corr[m] = sum_i h[i] * (1-h[i+s]) for i=0 to n-1-s
            # d/dh[i] = (1-h[i+s]) for i < n-s
            # d/dh[i] = -h[i-s] for i >= s
            for i in range(n - s):
                grad[i] -= (1.0 - h[i + s])
            for i in range(s, n):
                grad[i] += h[i - s]
        elif s < 0:
            a = -s
            for i in range(n - a):
                grad[i] -= (1.0 - h[i + a])
            for i in range(a, n):
                grad[i] += h[i - a]
        else:
            grad -= (1.0 - 2.0 * h)
    
    # Normalize
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return step * rng.randn(n)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
            for i in idx:
                inc = min(1.0 - h[i], diff)
                h[i] += inc
                diff -= inc
                if abs(diff) < 1e-14:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                dec = min(h[i], -diff)
                h[i] -= dec
                diff += dec
                if abs(diff) < 1e-14:
                    break
    return h

def golden_1d(old_val, c_base, lin, n):
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    a, b = 0.0, 1.0
    c = b - (b - a) / phi
    d = a + (b - a) / phi
    dc = c - old_val
    vc = c_base + dc * lin
    vc[n - 1] -= dc * dc
    fc = float(np.max(vc))
    dd = d - old_val
    vd = c_base + dd * lin
    vd[n - 1] -= dd * dd
    fd = float(np.max(vd))
    for _ in range(45):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / phi
            dc = c - old_val
            vc = c_base + dc * lin
            vc[n - 1] -= dc * dc
            fc = float(np.max(vc))
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / phi
            dd = d - old_val
            vd = c_base + dd * lin
            vd[n - 1] -= dd * dd
            fd = float(np.max(vd))
    return (a + b) / 2.0

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    nc = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 5 == 4 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(nc)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if i - s >= 0:
                    lin[mp] += (1.0 - best_h[i - s])
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                if i + s < n:
                    lin[mn] += (1.0 - best_h[i + s])
                if i - s >= 0:
                    lin[mn] -= best_h[i - s]
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            v_new = golden_1d(old, c_base, lin, n)
            dv = v_new - old
            if abs(dv) < 1e-14:
                continue
            best_h[i] = v_new
            corr += dv * lin
            corr[n - 1] -= dv * dv
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= dv * lin
                corr[n - 1] += dv * dv
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def targeted_grad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = np.where(corr > 0.98 * mx)[0]
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        if s > 0:
            grad[:n - s] -= (1.0 - h[s:])
            grad[s:] += h[:n - s]
        elif s < 0:
            a = -s
            grad[:n - a] -= (1.0 - h[a:])
            grad[a:] += h[:n - a]
        else:
            grad -= (1.0 - 2.0 * h)
    return grad

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        base_step = 0.02 + 0.08 * cool
        mode = hop % 7
        if mode == 0:
            pert = base_step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = base_step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = base_step * 2.5 * rng.randn(int(mask.sum()))
        elif mode == 3:
            idx = rng.choice(n, size=max(6, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = base_step * 1.5 * rng.randn(len(idx))
        elif mode == 4:
            g = targeted_grad(cur_h, n)
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                pert = -base_step * 2 * g / gn + 0.2 * base_step * rng.randn(n)
            else:
                pert = base_step * rng.randn(n)
        elif mode == 5:
            i, j = rng.choice(n, 2, replace=False)
            pert = np.zeros(n)
            pert[i] = cur_h[j] - cur_h[i]
            pert[j] = cur_h[i] - cur_h[j]
            pert += 0.01 * rng.randn(n)
        else:
            blk = rng.randint(0, n - 20)
            ln = rng.randint(20, 60)
            pert = np.zeros(n)
            pert[blk:blk + ln] = base_step * rng.randn(ln)
        h_pert = project_h(cur_h + pert, n)
        hop_end = min(t_end, time.time() + max(1.5, 6.0 * cool))
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=60, seed=seed + hop)
        temp = 0.003 * cool + 0.0002
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def multi_refine(h, n, t_end, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for rep in range(30):
        if time.time() > t_end:
            break
        rep_end = min(t_end, time.time() + max(1.0, (t_end - time.time()) / (30 - rep) * 0.7))
        if rep == 0:
            h_try = best_h.copy()
        else:
            step = 0.005 + 0.02 * rng.rand()
            h_try = project_h(best_h + step * rng.randn(n), n)
        h_opt, c5_opt = coord_descent(h_try, n, rep_end, max_sweeps=300, seed=seed + rep * 7)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5

def upsample(h, factor=2):
    n = len(h)
    n2 = n * factor
    x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
    h2 = np.interp(x_new, x_old, h, period=2.0)
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial CD polish (15%)
    t1 = t0 + budget_s * 0.15
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=800, seed=seed)

    # Phase 2: Basin hopping (40%)
    t2 = t0 + budget_s * 0.55
    best_h, best_c5 = basin_hop(best_h, n, t2, n_hops=400, seed=seed + 10)

    # Phase 3: Multi-start refinement (20%)
    t3 = t0 + budget_s * 0.75
    best_h, best_c5 = multi_refine(best_h, n, t3, seed=seed + 100)

    # Phase 4: Second basin hop from best (10%)
    if time.time() < t0 + budget_s * 0.85:
        h2, c5_2 = basin_hop(best_h, n, t0 + budget_s * 0.85, n_hops=150, seed=seed + 500)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 5: Upsample to 2n and optimize (8%)
    if time.time() < t0 + budget_s * 0.98:
        h_up = upsample(best_h, 2)
        n_up = len(h_up)
        h_up = project_h(h_up, n_up)
        t5 = t0 + budget_s * 0.98
        h_up, c5_up = coord_descent(h_up, n_up, t5, max_sweeps=400, seed=seed + 999)
        if c5_up < best_c5:
            best_h, best_c5, n = h_up, c5_up, n_up

    # Final projection and evaluation
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
