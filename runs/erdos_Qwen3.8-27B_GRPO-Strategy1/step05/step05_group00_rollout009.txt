import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def logsumexp_max(corr, beta):
    """Smooth approximation of max using log-sum-exp"""
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta

def nesterov_lse(h, N, deadline, k_top=50, beta0=200, lr0=0.01, beta_decay=0.999):
    """Nesterov with log-sum-exp smoothed max objective"""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = beta0
    beta_nesterov = 0.85
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Gradient of log-sum-exp
        beta_corr = beta * corr
        m_idx = np.max(beta_corr)
        exps = np.exp(beta_corr - m_idx)
        weights = exps / np.sum(exps)
        
        # Only use top k for efficiency
        top_k = np.where(weights > np.median(weights))[0]
        if len(top_k) > k_top:
            thresh = np.sort(weights)[-k_top]
            top_k = np.where(weights >= thresh)[0]
        
        grad = np.zeros(N)
        for ai in top_k:
            L = int(ai - center)
            w = weights[ai]
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += w * (1.0 - y[:N - L])
                grad[:N - L] -= w * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += w * (1.0 - y[M:])
                    grad[M:] -= w * y[:N - M]
        
        h_new = project_sum(y - lr * grad, target)
        m = beta_nesterov * (h_new - h)
        h = h_new
        beta *= beta_decay
        if beta < 50:
            beta = 50
    return best_h, best_pk * dx
