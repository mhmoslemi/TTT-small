def smooth_grad(h, T, P):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Get weights
    if T > 1e6:
        w = np.zeros(2*n-1)
        w[np.argmax(corr)] = 1.0
    else:
        e = T * (corr - m) - T * (corr - m).max()
        w = np.exp(e)
        w /= w.sum()
    
    # Only use top shifts for efficiency
    thresh = 1e-10
    active = w > thresh
    top_idx = np.where(active)[0]
    shifts = top_idx - (n - 1)
    
    i = np.arange(n)
    jp = i[:, None] + shifts[None, :]  # (n, k)

                jp = i + s
                jm = i - s
                if 0 <= jp < n:
                    c += (1.0 - h[jp])
                if 0 <= jm < n:
                    c -= h[jm]
                coefs[t] = c
            base = corr[top_idx] - h[i] * coefs
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-13:
                continue
            cands = [lo, hi]
            for a in range(n_top):
                for b in range(a + 1, n_top):
                    den = coefs[b] - coefs[a]
                    if abs(den) > 1e-14:
                        d_int = (base[a] - base[b]) / den
                        if lo <= d_int <= hi:
                            cands.append(d_int)
            best_d = 0.0
            best_v = np.max(base)
            for dc in cands:
                dc = max(lo, min(hi, dc))
                v = np.max(base + dc * coefs)
                if v < best_v - 1e-13:
                    best_v = v
                    best_d = dc
            t_new = h[i] + best_d
            if abs(t_new - h[i]) < 1e-13:
                continue
            h2 = h.copy()
            h2[i] = t_new
            h2 = project_h(h2)
            c5n = compute_c5(h2)
            if c5n < best - 1e-15:
                best = c5n
                h = h2
                improved = True
        if not improved:
            break
    return h, best

def perturb(x, scale, n):
    r = np.random.random()
    if r < 0.3:
        f = np.random.randint(2, 20)
        p = np.random.uniform(0, 2 * np.pi)
        return scale * np.sin(2 * np.pi * f * np.arange(n) / n + p)
    elif r < 0.55:
        bl = np.random.randint(max(n // 20, 2), max(n // 4, 3))
        bs = np.random.randint(0, max(n - bl, 1))
        pt = np.zeros(n)
        pt[bs:bs + bl] = scale * np.random.randn(bl)
        return pt
    elif r < 0.8:
        mask = np.random.random(n) < 0.12
        pt = np.zeros(n)
        pt[mask] = scale * 2.0 * np.random.randn(max(int(mask.sum()), 1))
        return pt
    else:
        return scale * np.random.randn(n)

def optimize_phase(x, n, t_end):
    best_h = x.copy()
    best_c5 = compute_c5(x)
    h = x.copy()
    T_seq = [100, 300, 1000, 5000, 20000, 100000, 500000, 2000000]
    for T in T_seq:
        if time.time() > t_end - 8:
            break
        h = lbfgs_solve(h, T, 80.0, maxiter=700)
        h = lp_step(h, 10, 0.03)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() > t_end - 10:
        return best_h, best_c5
    h, c5 = coord_desc(best_h, t_end, 50, 18)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(160)
        h[:80] = 1.0
    n = len(h)
    h = project_h(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: main optimization at n (55% time)
    t1_end = t0 + budget_s * 0.52
    best_h, best_c5 = optimize_phase(h, n, t1_end)

    # Phase 2: multi-restart (28% time)
    t2_end = t0 + budget_s * 0.80
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() > t2_end - 25:
            break
        scale = 0.05 + 0.04 * (1.0 - r / n_restarts)
        h_try = project_h(best_h + perturb(best_h, scale, n))
        for T in [30000, 150000]:
            if time.time() > t2_end - 15:
                break
            h_try = lbfgs_solve(h_try, T, 100.0, maxiter=350)
            h_try = lp_step(h_try, 8, 0.02)
        c5 = compute_c5(h_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 3: refine at finer resolution (n2=280) if time allows
    if t_end - time.time() > budget_s * 0.18:
        n2 = 280
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_h(h2)
        t3_end = t0 + budget_s * 0.93
        h2, c5_2 = optimize_phase(h2, n2, t3_end)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            n = n2

    # Phase 4: final intensive polish
    t4_end = t_end - 5
    if t4_end - time.time() > 15:
        for _ in range(5):
            if time.time() > t4_end - 5:
                break
            best_h = lp_step(best_h, 14, 0.008)
        best_h, best_c5 = coord_desc(best_h, t4_end, 80, 25)
    if time.time() < t_end - 3:
        h_pol = lbfgs_solve(best_h, 5000000, 200.0, maxiter=400)
        c5p = compute_c5(h_pol)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = h_pol
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
