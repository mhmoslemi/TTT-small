```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2*n - 1
    top_idx = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for ki in top_idx:
        p = ki - (n - 1)  # lag
        # For each i, contribution to corr[ki]:
        # (1-h[i-p]) if 0 <= i-p < n
        # -h[i+p] if 0 <= i+p < n
        # Vectorized:
        i = np.arange(n)
        ip_minus = i - p
        ip_plus = i + p
        valid_minus = (ip_minus >= 0) & (ip_minus < n)
        valid_plus = (ip_plus >= 0) & (ip_plus < n)
        grad[valid_minus] += (1.0 - h[np.clip(ip_minus[valid_minus], 0, n-1)])
        grad[valid_plus] += (-h[np.clip(ip_plus[valid_plus], 0, n-1)])
    
    h_new = h - alpha * grad
    return project_h(h_new, n)
```

```python
def active_set_lp(h, n, top_k=5, max_t=None):
    """Linearized minimax step using LP."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2*n - 1
    
    # Top k lags
    top_idx = np.argsort(corr)[-top_k:]
    
    # Build LP
    # Variables: h[0..n-1], t (total n+1 variables)
    # Minimize: t
    # Subject to:
    #   sum_i grad_c_k[i] * h[i] <= t - c_k + sum_i grad_c_k[i] * h[i]  (linearized)
    #   Actually: c_k(h) ≈ c_k(h0) + sum_i ∂c_k/∂h[i] * (h[i] - h0[i])
    #             c_k(h0) + sum_i g_k[i] * (h[i]-h0[i]) <= t
    #             sum_i g_k[i]*h[i] - t <= -c_k(h0) + sum_i g_k[i]*h0[i]
    
    # Let's compute gradients for top k
    grads = np.zeros((top_k, n))
    for idx, ki in enumerate(top_idx):
        p = ki - (n-1)
        i = np.arange(n)
        ip_m = i - p
        ip_p = i + p
        vm = (ip_m >= 0) & (ip_m < n)
        vp = (ip_p >= 0) & (ip_p < n)
        grads[idx, vm] += (1.0 - h[ip_m[vm]])
        grads[idx, vp] += (-h[ip_p[vp]])
    
    c_top = corr[top_idx]
    
    # LP: min t
    # s.t. for each k in top_k: sum_i grads[k,i]*h[i] - t <= sum_i grads[k,i]*h0[i] - c_k
    #      0 <= h[i] <= 1
    #      sum h[i] = n/2
    
    # A_ub * [h, t] <= b_ub
    A_rows = []
    b_rows = []
    for idx in range(top_k):
        row = np.zeros(n + 1)
        row[:n] = grads[idx]
        row[n] = -1.0
        rhs = np.dot(grads[idx], h) - c_top[idx]
        A_rows.append(row)
        b_rows.append(rhs)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum h = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Minimize t
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def soft_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = np.clip((corr - m) * T, -700, 700)
    return (m + np.log(np.sum(np.exp(z)))) * dx / T

def soft_grad(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = np.clip((corr - m) * T, -700, 700)
    ez = np.exp(z)
    w = ez / ez.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        p = k - (n - 1)
        wk = w[k] * dx
        i = np.arange(n)
        im = i - p
        ip = i + p
        vm = (im >= 0) & (im < n)
        vp = (ip >= 0) & (ip < n)
        g[vm] += wk * (1.0 - x[im[vm]])
        g[vp] += wk * (-x[ip[vp]])
    return g

def lbfgs_smooth(h, T, n, maxiter=600, mu=200.0):
    def obj(x):
        return soft_obj(x, T, n) + mu * (x.sum() - n / 2) ** 2
    def jac(x):
        return soft_grad(x, T, n) + 2.0 * mu * (x.sum() - n / 2) * np.ones(n)
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_h(res.x, n)
    except Exception:
        return h

def slsqp_smooth(h, T, n, maxiter=400):
    def obj(x):
        return soft_obj(x, T, n)
    def jac(x):
        return soft_grad(x, T, n)
    def cons(x):
        return x.sum() - n / 2
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons, 'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def active_set_lp(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grads = np.zeros((top_k, n))
    for idx, ki in enumerate(top_idx):
        p = ki - (n - 1)
        i = np.arange(n)
        im = i - p
        ip = i + p
        vm = (im >= 0) & (im < n)
        vp = (ip >= 0) & (ip < n)
        grads[idx, vm] += (1.0 - h[im[vm]])
        grads[idx, vp] += (-h[ip[vp]])
    c_top = corr[top_idx]
    A_rows = []
    b_rows = []
    for idx in range(top_k):
        row = np.zeros(n + 1)
        row[:n] = grads[idx]
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(np.dot(grads[idx], h) - c_top[idx])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return project_h(res.x[:n], n)
    except Exception:
        pass
    return h

def coord_descent(h, n, t_end, max_sweeps=80):

    best_h = h.copy()

    for sweep in range(max_sweeps):
        if time.time() > t_end:

        improved = False

        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            coefs = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                p = k - (n - 1)
                c = 0.0
                if 0 <= i - p < n:
                    c += (1.0 - best_h[i - p])
                if 0 <= i + p < n:
                    c += (-best_h[i + p])
                coefs[k] = c
            def f(t):
                return np.max(corr + (t - old) * coefs)
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 50})
            new_t = res.x
            if abs(new_t - old) > 1e-11:
                best_h[i] = new_t
                best_h = project_h(best_h, n)
                c5v = compute_c5(best_h)
                if c5v < best_c5 - 1e-15:
                    best_c5 = c5v
                    improved = True
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        if not improved:
            break
    return best_h, best_c5

def snap_refine(h, n, threshold=0.03):
    h_s = h.copy()
    h_s[h_s < threshold] = 0.0
    h_s[h_s > 1.0 - threshold] = 1.0
    h_s = project_h(h_s, n)
    return lbfgs_smooth(h_s, 500, n, maxiter=400)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with decreasing T
    for T in [50000, 10000, 3000, 1000, 400, 150, 60]:
        if elapsed() > budget_s * 0.12:
            break
        hc = lbfgs_smooth(best_h, T, n, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: SLSQP polish
    for T in [2000, 500, 200]:
        if elapsed() > budget_s * 0.2:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Active set LP iterations
    for it in range(40):
        if elapsed() > budget_s * 0.32:
            break
        h_lp = active_set_lp(best_h, n, top_k=5)
        c5v = compute_c5(h_lp)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_lp
        hc = lbfgs_smooth(best_h, 800, n, maxiter=300, mu=100.0)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Micro basin hopping (small perturbations)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(80):
        if elapsed() > budget_s * 0.55:
            break
        cool = 1.0 - it / 80.0
        amp = 0.003 + 0.01 * cool
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        T_bh = 500 + 3000 * cool
        hc = lbfgs_smooth(h_pert, T_bh, n, maxiter=400)
        c5v = compute_c5(hc)
        temp = 0.00005 * cool + 5e-6
        if c5v < work_c5 or (np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Coordinate descent
    if elapsed() < budget_s * 0.75:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.75, max_sweeps=60)

    # Phase 6: Snap and refine (exploit near-binary structure)
    if elapsed() < budget_s * 0.82:
        hc = snap_refine(best_h, n, threshold=0.02)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        hc = snap_refine(best_h, n, threshold=0.05)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: SLSQP high-T polish
    for T in [100000, 500000]:
        if elapsed() > budget_s * 0.92:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 8: L-BFGS-B ultra-high T
    if elapsed() < budget_s * 0.96:
        hc = lbfgs_smooth(best_h, 1000000, n, maxiter=1000, mu=500.0)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 9: Final coordinate descent
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=40)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
