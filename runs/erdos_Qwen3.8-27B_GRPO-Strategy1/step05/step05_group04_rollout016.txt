import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            h *= (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def lin_coeff(h, i, n):
    lin = np.zeros(2 * n - 1)
    for s in range(1, n):
        mp = s + n - 1
        mn = n - 1 - s
        if s <= i:
            lin[mp] += (1.0 - h[i - s])
            lin[mn] -= h[i - s]
        if i + s < n:
            lin[mp] -= h[i + s]
            lin[mn] += (1.0 - h[i + s])
    lin[n - 1] = 1.0 - 2.0 * h[i]
    return lin

def cd1d(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = compute_corr(best_h)
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 3 == 2 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = lin_coeff(best_h, i, n)
            base = corr.copy()
            def f1d(v):
                d = v - old
                val = base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def cd_pairs(h, n, t_end, max_sweeps=100, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = compute_corr(best_h)
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        for _ in range(n // 2):
            if time.time() > t_end:
                return best_h, best_c5
            i, j = rng.randint(n, 2)
            if i == j:
                continue
            old = np.array([best_h[i], best_h[j]])
            lin_i = lin_coeff(best_h, i, n)
            lin_j = lin_coeff(best_h, j, n)
            base = corr.copy()
            diff = abs(i - j)
            if diff > 0:
                cross_pos = n - 1 + diff
                cross_neg = n - 1 - diff
            else:
                cross_pos, cross_neg = -1, -1

            def f2d(v):
                d = np.asarray(v, dtype=float) - old
                val = base + d[0] * lin_i + d[1] * lin_j
                if cross_pos >= 0:
                    val[cross_pos] -= d[0] * d[1]
                    val[cross_neg] -= d[0] * d[1]
                val[n - 1] -= d[0] ** 2 + d[1] ** 2
                return float(np.max(val) * dx)

            best_f = f2d(old)
            best_v = old.copy()
            for _ in range(2):

                v0 = np.clip(v0, 0, 1)
                res = minimize(f2d, v0, method='Nelder-Mead',
                               options={'maxiter': 150, 'xatol': 1e-10, 'fatol': 1e-13})
                v_opt = np.clip(res.x, 0, 1)
                f_opt = f2d(v_opt)
                if f_opt < best_f:
                    best_f = f_opt
                    best_v = v_opt
            d = best_v - old
            if np.max(np.abs(d)) < 1e-12:
                continue
            best_h[i] = best_v[0]
            best_h[j] = best_v[1]
            corr += d[0] * lin_i + d[1] * lin_j
            if cross_pos >= 0:
                corr[cross_pos] -= d[0] * d[1]
                corr[cross_neg] -= d[0] * d[1]
            corr[n - 1] -= d[0] ** 2 + d[1] ** 2

                best_c5 = c5_new
            else:

                    corr[cross_neg] += d[0] * d[1]

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=300, seed=42):

    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        frac = hop / max(n_hops, 1)
        cool = 1.0 - frac
        step = 0.02 + 0.08 * cool
        mode = hop % 7
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(2, 16)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            blk = rng.randint(3, max(4, n // 6))
            st = rng.randint(0, n - blk)
            pert = np.zeros(n)
            pert[st:st + blk] = step * 2 * rng.choice([-1, 1], blk)
        elif mode == 3:
            idx = rng.choice(n, size=max(4, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 2 * rng.randn(len(idx))
        elif mode == 4:
            c = rng.randint(n // 4, 3 * n // 4)
            w = rng.randint(5, n // 3)
            x = np.arange(n)
            pert = step * np.exp(-((x - c) / w) ** 2) * rng.choice([-1, 1])
        elif mode == 5:
            b1 = rng.randint(0, n // 3)
            b2 = rng.randint(2 * n // 3, n)
            pert = np.zeros(n)
            pert[:b1] = step
            pert[b2:] = -step
        else:
            sh = rng.randint(1, 5)
            pert = np.roll(cur_h, sh) - cur_h
            pert *= 0.3
        h_pert = project_h(cur_h + pert, n)
        hop_budget = max(1.5, 6.0 * cool)
        hop_end = min(t_end, time.time() + hop_budget)
        h_opt, c5_opt = cd1d(h_pert, n, hop_end, max_sweeps=60, seed=seed + hop)
        temp = 0.003 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h, cur_c5 = h_opt, c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def targeted_perturb(h, n, rng, strength):
    corr = compute_corr(h)
    m = np.max(corr)
    top_lags = np.argsort(corr)[-5:]
    indices_hot = set()
    for lg in top_lags:
        s = lg - (n - 1)
        if s > 0:
            indices_hot.update(range(0, n - s))
            indices_hot.update(range(s, n))
        elif s < 0:
            a = -s
            indices_hot.update(range(0, n - a))
            indices_hot.update(range(a, n))
        else:
            indices_hot.update(range(n))
    hot_list = list(indices_hot)
    if not hot_list:
        hot_list = list(range(n))
    k = min(len(hot_list), max(5, n // 5))
    idx = rng.choice(hot_list, size=k, replace=False)
    pert = np.zeros(n)
    pert[idx] = strength * rng.randn(len(idx))
    return pert
