        if len(ones) == 0 or len(zeros) == 0:
            break
        for _ in range(n_try):
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            d = compute_swap_delta(h, N, p, q)
            new_max = float(np.max(c + d)) * (2.0 / N)
            diff = new_max - P
            if diff < 0 or (T > T_min and np.random.random() < np.exp(-diff / T)):
                h[p], h[q] = 0.0, 1.0
                c = c + d
                P = new_max
        T = max(T_min, T * cool)
    return best_h, best_c

def soft_max_grad(h, N, N_top=20):
    corr = fft_corr(h)
    cmax = corr.max()
    exp_c = np.exp(4.0 * (corr - cmax))
    w = exp_c / exp_c.sum()
    top = np.argsort(corr)[-N_top:]
    grad = np.zeros(N)
    for ai in top:
        L = int(ai - (N - 1))
        wk = w[ai]
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:N - L]
        else:
            M = -L
            if M < N:
                grad[M:] += wk * (1.0 - h[:N - M])
                grad[:N - M] -= wk * h[M:]
    return grad * (2.0 / N), float(cmax) * (2.0 / N)

def nesterov_soft(h, N, deadline, k=30, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = float(np.max(fft_corr(h))) * dx
    m = np.zeros(N)
    lr = lr0
    beta = 0.8
    it = 0
    while time.time() < deadline:
        y = h + m
        grad, pk = soft_max_grad(y, N, N_top=k)
        if pk < best_c5:
            best_c5 = pk
            best_h = h.copy()
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 80 == 0:
            lr *= 0.994
            beta = min(0.95, beta + 0.0015)
    return best_h, best_c5

def try_resolution(h_best, N_best, N_try, deadline_sub, T0=0.012):
    if N_try == N_best:
        return None, None
    x_best = np.linspace(0, 2, N_best, endpoint=False)
    x_new = np.linspace(0, 2, N_try, endpoint=False)
    h_interp = np.interp(x_new, x_best, h_best)
    h_bt = (h_interp > 0.5).astype(float)
    nh = N_try // 2
    n1 = int(h_bt.sum())
    if n1 > nh:
        idx = np.where(h_bt == 1)[0]
        idx = idx[np.argsort(np.random.random(len(idx)))][nh:]
        h_bt[idx] = 0
    elif n1 < nh:
        idx = np.where(h_bt == 0)[0]
        idx = idx[np.argsort(np.random.random(len(idx)))][:nh - n1]
        h_bt[idx] = 1
    h_rt, c5_rt = binary_anneal(h_bt, N_try, deadline_sub, max_sweeps=600, T0=T0, cool=0.999)
    return h_rt, c5_rt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary annealing from initial
    d1 = min(deadline - 500, t0 + 250)
    h_s, c5_s = binary_anneal(h0, N0, d1, max_sweeps=2000, T0=0.018, T_min=0.0003, cool=0.9988, n_try=45)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Nesterov continuous
    if time.time() < deadline - 250:
        d2 = min(deadline - 200, time.time() + 130)
        hc = best_h.astype(float) + np.random.randn(len(best_h)) * 0.04
        hc = project_sum(hc, len(best_h) / 2.0)
        hn, c5n = nesterov_soft(hc, len(best_h), d2, k=35, lr0=0.012)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary annealing from continuous
    if time.time() < deadline - 200:
        d3 = min(deadline - 150, time.time() + 130)
        h_s3, c5_3 = binary_anneal(best_h, len(best_h), d3, max_sweeps=1500, T0=0.01, T_min=0.0002, cool=0.9993, n_try=50)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Multi-resolution exploration
    N_cur = len(best_h)
    offsets = [-40, -20, 20, 40, 60, 80, -60, 100]
    for off in offsets:
        if time.time() > deadline - 60:
            break
        N_try = N_cur + off
        if N_try < 200 or N_try > 1200 or N_try % 2 != 0:
            continue
        sub_dl = min(deadline - 30, time.time() + 50)
        h_rt, c5_rt = try_resolution(best_h, N_cur, N_try, sub_dl, T0=0.014)
        if c5_rt is not None and c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            N_cur = N_try

    # Phase 5: Restart with perturbation
    if time.time() < deadline - 80:
        d5 = min(deadline - 40, time.time() + 60)
        h_pert = (best_h > 0.5).astype(float)
        n_flip = max(5, len(h_pert) // 20)
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        flip_ones = np.random.choice(ones, min(n_flip, len(ones)), replace=False)
        flip_zeros = np.random.choice(zeros, min(n_flip, len(zeros)), replace=False)
        h_pert[flip_ones] = 0
        h_pert[flip_zeros] = 1
        h_rt, c5_rt = binary_anneal(h_pert, len(h_pert), d5, max_sweeps=800, T0=0.02, T_min=0.0005, cool=0.999, n_try=40)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final Nesterov polish
    if time.time() < deadline - 15:
        hF = best_h.astype(float) + np.random.randn(len(best_h)) * 0.015
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_soft(hF, len(best_h), deadline - 5, k=50, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Final binary cleanup with very low temperature
    if time.time() < deadline - 1:
        hF2, c5F2 = binary_anneal(best_h, len(best_h), deadline - 0.5, max_sweeps=300, T0=0.004, T_min=0.00005, cool=0.9998, n_try=60)
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
