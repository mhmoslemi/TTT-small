    if t_end - time.time() > 7:
        hc = slsqp_step(best_h, 5000000, 1200, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), len(best_h)
