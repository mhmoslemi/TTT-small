def nesterov_subgrad(h, n, iters, t_end):

        maxv = np.max(vals)

        ha, ta = active_set_slsqp(best_h, n, t_end - 15, max_rounds=12, n_cons=12)
        if ta < best_c5:
            best_c5, best_h = ta, ha
    # Coordinate descent
    if t_end - time.time() > 20:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 10, 60, 18)
    # Final polish
    if t_end - time.time() > 10:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 2, 40, 25)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n_parent = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent n (majority of time)
    t_p1 = t0 + budget_s * 0.5
    best_h, best_c5 = optimize_at_n(best_h, n_parent, t_p1)

    # Phase 2: Try n=150 (interpolated)
    if t_end - time.time() > budget_s * 0.25:
        n2 = 150
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n_parent, endpoint=False), best_h)
        h2 = project_h(h2)
        t_p2 = t0 + budget_s * 0.7
        h2o, c52 = optimize_at_n(h2, n2, t_p2)
        if c52 < best_c5:
            best_c5, best_h = c52, h2o

    # Phase 3: Try n=250
    if t_end - time.time() > budget_s * 0.2:
        n3 = 250
        src_n = len(best_h)
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, src_n, endpoint=False), best_h)
        h3 = project_h(h3)
        t_p3 = t0 + budget_s * 0.88
        h3o, c53 = optimize_at_n(h3, n3, t_p3)
        if c53 < best_c5:
            best_c5, best_h = c53, h3o

    # Phase 4: Try n=100 (coarser, might find different basin)
    if t_end - time.time() > budget_s * 0.15:
        n4 = 100
        src_n = len(best_h)
        h4 = np.interp(np.linspace(0, 2, n4, endpoint=False),
                       np.linspace(0, 2, src_n, endpoint=False), best_h)
        h4 = project_h(h4)
        # Add small perturbation to escape basin
        h4 = project_h(h4 + 0.05 * np.random.randn(n4))
        t_p4 = t0 + budget_s * 0.95
        h4o, c54 = optimize_at_n(h4, n4, t_p4)
        if c54 < best_c5:
            best_c5, best_h = c54, h4o

    # Final polish at best n
    if t_end - time.time() > 8:
        fn = len(best_h)
        best_h, best_c5 = coord_descent(best_h, fn, t_end - 1, 50, 30)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
