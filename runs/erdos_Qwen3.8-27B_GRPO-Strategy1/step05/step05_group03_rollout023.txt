def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

Actually for a convex piecewise-linear function, we can find the exact minimum:
- The subgradient at δ is {coefs[p] : p ∈ argmax_p (corr_base[p] + δ*coefs[p])}
- The minimum is where 0 ∈ subgradient

This is equivalent to finding the δ where the "upper envelope" has a flat spot minimum.

A simpler exact approach: The minimum of max of linear functions on [0,1] occurs at:
- δ = 0
- δ = 1  
- Or at an intersection point δ* of two lines f_p1(δ) and f_p2(δ) where f_p1(δ*) = f_p2(δ*) ≥ f_p(δ*) for all p

The intersection of f_p1(δ) = a1 + b1*δ and f_p2(δ) = a2 + b2*δ is at δ = (a2-a1)/(b1-b2) when b1≠b2.

There are at most (2n-1) choose 2 ≈ 50,000 such intersections. For each, we check if it's valid (in [0,1]) and if it's actually the minimum of the max. This is O(n³) per coordinate, which is too slow.

Better: Use the fact that we only need the minimum value, not the exact delta. We can use a "bisection on the value" approach:
- Binary search on C: is there a δ ∈ [0,1] such that corr_base[p] + δ*coefs[p] ≤ C for all p?
- For a given C, this is: δ*coefs[p] ≤ C - corr_base[p] for all p

This is a feasibility problem in C. We can binary search on C. Each feasibility check is O(n). Total: O(n log(precision)) per coordinate.

def coord_descent_fast(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            # Compute coefficients for changing h[i]
            coefs = np.zeros(2 * n - 1)
            lags = np.arange(-(n - 1), n)
            j1 = i + lags
            j2 = i - lags
            m1 = (j1 >= 0) & (j1 < n)
            m2 = (j2 >= 0) & (j2 < n)
            coefs[m1] += (1.0 - best_h[np.clip(j1[m1], 0, n - 1)])
            coefs[m2] += (-best_h[np.clip(j2[m2], 0, n - 1)])
            
            # corr_base is corr with h[i] contribution removed
            # corr[p] = corr_base[p] + best_h[i] * coefs[p]
            # So corr_base[p] = corr[p] - best_h[i] * coefs[p]
            corr_base = corr - best_h[i] * coefs
            
            # Find optimal value using binary search
            pos = coefs > 1e-12
            neg = coefs < -1e-12
            
            f0 = float(np.max(corr_base))  # value at h[i]=0
            f1 = float(np.max(corr_base + coefs))  # value at h[i]=1
            
            lo_C = min(f0, f1) - 0.001
            hi_C = max(f0, f1) + 0.001
            
            for _ in range(50):
                mid_C = (lo_C + hi_C) / 2
                # Feasibility: exists v in [0,1] s.t. corr_base[p] + v*coefs[p] <= C for all p
                feasible = True
                if np.any(corr_base[~(pos|neg)] > C + 1e-12):
                    feasible = False
                else:
                    hi_d = 1.0
                    if np.any(pos):
                        hi_d = min(hi_d, float(np.min((mid_C - corr_base[pos]) / coefs[pos])))
                    lo_d = 0.0
                    if np.any(neg):
                        lo_d = max(lo_d, float(np.max((mid_C - corr_base[neg]) / coefs[neg])))
                    feasible = lo_d <= hi_d + 1e-12
                
                if feasible:
                    hi_C = mid_C
                else:
                    lo_C = mid_C
            
            # Find optimal v
            hi_d = 1.0
            if np.any(pos):
                hi_d = min(hi_d, float(np.min((hi_C - corr_base[pos]) / coefs[pos])))
            lo_d = 0.0
            if np.any(neg):
                lo_d = max(lo_d, float(np.max((hi_C - corr_base[neg]) / coefs[neg])))
            if lo_d <= 0:
                opt_v = 0.0
            elif hi_d >= 1:
                opt_v = 1.0
            else:
                opt_v = (max(lo_d, 0) + min(hi_d, 1)) / 2
                opt_v = float(np.clip(opt_v, 0, 1))
            
            if abs(opt_v - best_h[i]) > 1e-10:
                old_v = best_h[i]
                best_h[i] = opt_v
                # Update corr incrementally
                corr += (opt_v - old_v) * coefs
                # Project
                best_h = project_h(best_h, n)
                # Recompute corr after projection (projection changed other coords)
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                c5_new = float(np.max(corr) * (2.0 / n))
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        
        if not improved:
            break
    return best_h, best_c5
