def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)

        if s < 1e-14:

        h = h * (1.0 + diff / s)

def swap_sa(h, n, t_end, n_swaps=5000000):
    """Binary swap simulated annealing."""
    best_h = h.copy()

    cur_h = h.copy()
