import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            if s < 1e-14:
                break
            h = h * (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, h_old, T, n, penalty=0.0):
    corr = get_corr(x)
    m = np.max(corr)
    if T > 1e7:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    if penalty > 0:
        top = np.sort(corr)[-10:]
        val += penalty * np.std(top)
    return val * (2.0 / n)

def slsqp_opt(h, T, n, maxiter=400, ftol=1e-15):
    try:
        res = minimize(lambda x: smooth_obj(x, h, T, n), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=500, n_top=20):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            # Minimize max(base + t*coefs) over t in [0,1]
            # This is a 1D convex piecewise-linear problem
            # Use minimize_scalar
            res = minimize_scalar(lambda t: float(np.max(base + t * coefs)),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            if s < 1e-14:
                break
            h = h * (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, T, n, maxiter=400, ftol=1e-15, pen=0.0):
    def obj(x):
        corr = get_corr(x)
        m = np.max(corr)
        if T > 1e7:
            val = m
        else:
            val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        if pen > 0:
            top = np.sort(corr)[-10:]
            val += pen * np.std(top)
        return val * (2.0 / n)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=500, n_top=25):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = 0
        for i in range(n):
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: float(np.max(base + t * coefs)),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved += 1
        if improved == 0:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter=30, scale=0.13):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.3, 1.0 - it / n_iter)
        r = np.random.random()
        if r < 0.30:
            pert = scale * cool * np.random.randn(n)
        elif r < 0.55:
            corr = get_corr(work_h)
            top_idx = np.argsort(corr)[-6:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for s in shifts:
                if 0 < s < n:
                    pert[:n - s] += 0.6 * (work_h[s:n] - work_h[:n - s])
                elif -n < s < 0:
                    pert[-s:] += 0.6 * (work_h[:n + s] - work_h[-s:])
            pert *= scale * 2.5 * cool
        elif r < 0.75:
            blen = np.random.randint(max(n // 15, 2), max(n // 4, 3))
            bs = np.random.randint(0, n - blen)
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * 1.5 * cool * np.random.randn(blen)
        else:
            f = np.random.randint(1, 10)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.8 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        h_pert = project_h(work_h + pert)
        T = 2000 + 4000 * cool
        hc = slsqp_opt(h_pert, T, n, 250, 1e-14)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0002
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_n(h_init, n, t_end):
    best_h = project_h(h_init.copy())
    best_c5 = compute_c5(best_h)
    # SLSQP warmup
    if t_end - time.time() > 40:
        for T in [100, 1000, 10000]:
            if t_end - time.time() < 15:
                break
            hc = slsqp_opt(best_h, T, n, 300, 1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Heavy coordinate descent (main workhorse)
    t_cd1 = t_end - (t_end - time.time()) * 0.30
    if t_end - time.time() > 30:
        best_h, best_c5 = coord_descent(best_h, n, t_cd1, 600, 25)
    # SLSQP refinement with penalty
    if t_end - time.time() > 30:
        hc = slsqp_opt(best_h, 30000, n, 400, 1e-15, pen=0.03)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        t_cd2 = t_end - (t_end - time.time()) * 0.20
        if t_end - time.time() > 20:
            best_h, best_c5 = coord_descent(best_h, n, t_cd2, 300, 20)
    # Basin hopping
    if t_end - time.time() > 80:
        t_bh = t_end - (t_end - time.time()) * 0.30
        bh_h, bh_c5 = basin_hop(best_h, n, t_bh, 25, 0.12)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
            best_h, best_c5 = coord_descent(best_h, n, t_end - (t_end - time.time()) * 0.15, 250, 25)
    # Final polish
    if t_end - time.time() > 15:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 5, 400, 30)
    if t_end - time.time() > 10:
        hc = slsqp_opt(best_h, 500000, n, 500, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, 200, 35)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
