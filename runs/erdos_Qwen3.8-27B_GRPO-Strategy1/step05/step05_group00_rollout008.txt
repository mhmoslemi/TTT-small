def nesterov_opt(h, N, deadline, k=60, lr0=0.02, max_iter=50000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        center = N - 1
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 500 == 0:
            lr *= 0.98
            beta = min(0.95, 0.85 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

                break
            dl_sub = min(deadline - 20, time.time() + 40)
            h_try = make_binary(h_init, N_m)
            h_rs, c5_rs = binary_search_fast(h_try, N_m, dl_sub, max_sweeps=4000, topK=120, sample_ones=100)
            if c5_rs < best_c5:
                best_c5, best_h = c5_rs, h_rs.copy()

    # Phase 6: Final Nesterov polish
    if time.time() < deadline - 20:
        hF = best_h + np.random.randn(len(best_h)) * 0.04
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_opt(hF, len(best_h), deadline - 5, k=100, lr0=0.01)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    # Phase 7: Final binary polish
    if time.time() < deadline - 1:
        hF2 = make_binary(best_h, len(best_h))
        hF2, c5F2 = binary_search_fast(hF2, len(best_h), deadline - 0.5, max_sweeps=3000, topK=150, sample_ones=150)
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
