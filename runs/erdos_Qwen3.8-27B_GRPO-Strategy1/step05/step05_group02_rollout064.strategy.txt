import numpy as np
from scipy.optimize import minimize
import time

def project_h(h):
    # 1. Clip values to [0,1]
    h_clipped = np.clip(h, 0.0, 1.0)
    # 2. Scale to meet integral constraint
    target_sum = h.shape[0] / 2.0
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-10:
        return h_clipped
    scale_factor = target_sum / current_sum
    h_scaled = h_clipped * scale_factor
    return np.clip(h_scaled, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    """Generate a double-peak Gaussian perturbation pattern"""
    centers = np.random.uniform(0.8, 1.2, 2)
