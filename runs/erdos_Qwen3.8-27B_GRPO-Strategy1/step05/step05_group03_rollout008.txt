def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary_sequence(n, seed):
    """Generate various binary sequences."""
    rng = np.random.RandomState(seed)
    # Thue-Morse
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Ensure exactly n/2 ones
    if tm.sum() != n/2:
        # Adjust
        idx_1 = np.where(tm == 1)[0]
        idx_0 = np.where(tm == 0)[0]
        diff = tm.sum() - n/2
        if diff > 0:
            to_flip = rng.choice(idx_1, int(diff), replace=False)
        else:
            to_flip = rng.choice(idx_0, -int(diff), replace=False)
        tm[to_flip] = 1 - tm[to_flip]
    return tm

def compute_flip_matrix(h, n):
    """Precompute flip vectors for each position."""
    # corr[k] = sum_i h[i+k]*(1-h[i]) for lag k
    # Full correlation has 2n-1 entries, indexed 0 to 2n-2
    # lag = idx - (n-1)
    flip = np.zeros((n, 2*n-1))
    for pos in range(n):
        for k in range(2*n-1):
            lag = k - (n-1)
            val = 0.0
            # Effect as "1-h[n]" factor: n=pos, so h[pos+lag]
            if 0 <= pos + lag < n:
                if h[pos] == 1:
                    val += h[pos + lag]  # was 0, now h[pos+lag]*1
                else:
                    val -= h[pos + lag]  # was h[pos+lag]*1, now 0
            # Effect as "h[n+k]" factor: n+lag=pos, so n=pos-lag
            if 0 <= pos - lag < n:
                if h[pos] == 1:
                    val -= (1 - h[pos - lag])  # was 1*(1-h), now 0
                else:
                    val += (1 - h[pos - lag])  # was 0, now 1*(1-h)
            flip[pos, k] = val
    return flip

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        # Scale down the excess
        excess = s - target
        # Reduce from positions furthest from boundary
        order = np.argsort(np.abs(h - 0.5))[::-1]
        remaining = excess
        for idx in order:
            red = min(h[idx], remaining)
            h[idx] -= red
            remaining -= red
            if remaining < 1e-14:
                break
    else:
        deficit = target - s
        order = np.argsort(np.abs(h - 0.5))[::-1]
        remaining = deficit
        for idx in order:
            add = min(1.0 - h[idx], remaining)
            h[idx] += add
            remaining -= add
            if remaining < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_flip_row(h, pos, n):
    lags = np.arange(-(n-1), n)
    idx1 = pos + lags
    idx2 = pos - lags
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    term1 = np.zeros(2*n-1)
    term2 = np.zeros(2*n-1)
    term1[v1] = h[idx1[v1]]
    term2[v2] = 1.0 - h[idx2[v2]]
    sign = 2.0*h[pos] - 1.0
    return sign * (term1 - term2)

def compute_flip_matrix(h, n):
    flip = np.zeros((n, 2*n-1))
    for pos in range(n):
        flip[pos] = compute_flip_row(h, pos, n)
    return flip

def project_sum(h, n):
    """Project h to have sum exactly n/2, values in [0,1]."""
    h = np.clip(h, 0, 1)
    target = n/2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        if room.sum() < 1e-14:
            return h
