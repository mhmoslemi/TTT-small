    corr = np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

For the 1D problem min_{x∈[0,1]} max_k (cb[k] + lin[k]*x + quad[k]*x²):

The function g(x) = max_k f_k(x) where each f_k is either linear or concave quadratic.
g(x) is continuous but not necessarily convex (because of the concave quadratic at lag 0).

However, g(x) is piecewise defined: in each region, the argmax is constant, and g is either linear or concave quadratic.

The minimum of g on [0,1] is at:
- x = 0 or x = 1, or
- A point where two f_k's cross and both are on the "upper envelope"

For a practical approach, I'll use:
1. Evaluate g at a grid of points (say 100 points in [0,1])
2. Find the minimum grid point
3. Refine with ternary search in a neighborhood

This is simple, robust, and avoids lambdas.

```python
def minimize_1d_max(cb, lin, quad):
    """Minimize max_k(cb[k] + lin[k]*x + quad[k]*x^2) over x in [0,1]."""
    # Coarse grid
    xs = np.linspace(0, 1, 200)
    vals = np.array([float(np.max(cb + x * lin + x * x * quad)) for x in xs])
    x_best = xs[np.argmin(vals)]
    v_best = vals.min()
    
    # Refine with ternary search in [max(0, x_best-0.01), min(1, x_best+0.01)]
    lo = max(0.0, x_best - 0.005)
    hi = min(1.0, x_best + 0.005)
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        v1 = float(np.max(cb + m1 * lin + m1 * m1 * quad))
        v2 = float(np.max(cb + m2 * lin + m2 * m2 * quad))
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    x_opt = (lo + hi) / 2
    return x_opt
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_coord_coeffs(h, i):
    n = len(h)
    m = 2 * n - 1
    lin = np.zeros(m)
    quad = np.zeros(m)
    quad[n - 1] = -1.0
    lin[n - 1] = 1.0
    upper = n - 1 - i
    lower = i
    if upper > 0:
        lin[n:n + upper] += (1.0 - h[i + 1:n])
    if lower > 0:
        lin[n:n + lower] += -h[i - 1::-1]
        lin[n - 1 - lower:n - 1] += (1.0 - h[i - 1::-1])
    if upper > 0:
        lin[i:n - 1] += -h[i + 1:n]
    return lin, quad

def minimize_coord(cb, lin, quad):
    # Coarse grid (vectorized)
    xs = np.linspace(0.0, 1.0, 160)
    X = xs[:, None]
    vals = np.max(cb[None, :] + X * lin[None, :] + X * X * quad[None, :], axis=1)
    idx = np.argmin(vals)
    x_best = xs[idx]
    # Refine
    lo = max(0.0, x_best - 1.0 / 160.0)
    hi = min(1.0, x_best + 1.0 / 160.0)
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = float(np.max(cb + m1 * lin + m1 * m1 * quad))
        v2 = float(np.max(cb + m2 * lin + m2 * m2 * quad))
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n_sweeps, t_end):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            h0 = best.copy()
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin, quad = compute_coord_coeffs(best, i)
            x_opt = minimize_coord(cb, lin, quad)
            if abs(x_opt - best[i]) > 1e-10:
                best[i] = x_opt
                improved = True
        c5_new = compute_c5(best)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
        elif not improved:
            break
    return best, best_c5

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)  # numerical stability
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm_val = mc + np.log(exps.sum()) / T
    val = dx * sm_val + lam * (h.sum() - n / 2.0) ** 2
    # Gradient
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    grad = np.zeros(n)
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-14:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n - lg])
                grad[:n - lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n - p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_pass(h0, T, lam, maxiter, t_end):
    n = len(h0)
    x0 = project_h(h0)
    def _obj(x):
        return softmax_obj_grad(x, T, lam)[0]
    def _grad(x):
        return softmax_obj_grad(x, T, lam)[1]
    try:
        res = minimize(_obj, x0, jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 5000})
        if time.time() < t_end:
            return project_h(res.x)
    except Exception:
        pass
    return x0

    np.random.seed(seed)
    t0 = time.time()

    best_c5 = compute_c5(best_h)
