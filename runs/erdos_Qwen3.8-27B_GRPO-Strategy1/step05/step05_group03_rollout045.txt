def coord_sens(h, i, n):

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0:
                break
            h[m] += diff / c
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0:
                break
            h[m] += diff / c
    return np.clip(h, 0.0, 1.0)

def coord_sens(h, i, n):
    sens = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        ki = lag + n - 1
        c = 0.0
        if 0 <= i - lag < n:
            c += (1.0 - h[i - lag])
        if 0 <= i + lag < n:
            c -= h[i + lag]
        sens[ki] = c
    return sens

def opt_coord(h, corr, i, n):
    old = h[i]
    sens = coord_sens(h, i, n)
    def fmax(d):
        return np.max(corr + d * sens)
    res = minimize_scalar(fmax, bounds=(-old, 1.0 - old), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 40})
    return np.clip(res.x + old, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        order = np.random.permutation(n)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            nv = opt_coord(best_h, corr, i, n)
            if abs(nv - best_h[i]) < 1e-11:
                continue
            ht = best_h.copy()
            ht[i] = nv
            ht = project_h(ht, n)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-14:
                best_h = ht
                best_c5 = c5t
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            if sweep > 10:
                break
            order = np.random.permutation(n)
    return best_h, best_c5

def anneal_coord(h, n, t_end, T0=0.002, T1=0.0001, max_iters=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / max_iters)
        T = T1 + (T0 - T1) * cool
        idx = np.random.randint(n)
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        nv = opt_coord(cur_h, corr, idx, n)
        # Add noise
        nv = np.clip(nv + T * 5.0 * np.random.randn(), 0.0, 1.0)
        ht = cur_h.copy()
        ht[idx] = nv
        ht = project_h(ht, n)
        c5t = compute_c5(ht)
        if c5t < cur_c5 or np.random.rand() < np.exp(-(c5t - cur_c5) / T):
            cur_h = ht
            cur_c5 = c5t
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = ht
        else:
            # revert
            cur_h[idx] = ht[idx]  # already projected, just accept rejection
            cur_h = ht  # keep projected (sum constraint satisfied)
            cur_c5 = c5t
            # Actually revert properly
            cur_h = best_h.copy() if c5t > best_c5 + 0.01 else ht
            cur_c5 = c5t if c5t < best_c5 + 0.01 else compute_c5(cur_h)
    return best_h, best_c5

def slsqp_smooth(h, n, T, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -40, 40)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def pair_opt(h, n, i, j, t_end):
    """Optimize h[i] and h[j] jointly."""
    best = h.copy()
    best_c5 = compute_c5(best)
    xor = (i + j) % 2
    for _ in range(30):
        if time.time() > t_end:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        vi = opt_coord(best, corr, i, n)
        best[i] = vi
        best = project_h(best, n)
        corr = np.correlate(best, 1.0 - best, mode='full')
        vj = opt_coord(best, corr, j, n)
        best[j] = vj
        best = project_h(best, n)
        c5 = compute_c5(best)
        if c5 >= best_c5 - 1e-14:
            break
        best_c5 = c5
    return best, best_c5
