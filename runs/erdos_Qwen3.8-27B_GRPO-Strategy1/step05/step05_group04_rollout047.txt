import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(60):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            h *= (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=200, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            # Compute linear coefficients
            for s in range(1, n):
                if s <= i:
                    lin[n - 1 - s] += (1.0 - best_h[i - s])
                    lin[n - 1 + s] -= best_h[i - s]
                if i + s < n:
                    lin[n - 1 - s] -= best_h[i + s]
                    lin[n - 1 + s] += (1.0 - best_h[i + s])
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            def f1d(v, _cbase=c_base, _lin=lin, _old=old, _dx=dx):
                d = v - _old
                val = _cbase + d * _lin
                val[n - 1] -= d * d
                return float(np.max(val) * _dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-12:
                continue
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_h[i] = v_new
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved and sweep > 5:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
