            #   lin[n-1+s] += (1-h[i+s]) if i+s < n
            #   lin[n-1+s] -= h[i-s] if i-s >= 0
            # For shift s < 0 (i.e., s' = -s > 0, index n-1+s'):
            #   lin[n-1-s] += (1-h[i+s]) if i+s < n  
            #   lin[n-1-s] -= h[i-s] if i-s >= 0
            
            s_idx = np.arange(1, n)
            # Positive shifts
            valid_p = i + s_idx < n
            valid_n = i - s_idx >= 0
            
            idx_p = n - 1 + s_idx[valid_p]
            idx_n = n - 1 + s_idx[valid_n]
            
            if np.any(valid_p):
                lin[idx_p] += (1.0 - best_h[i + s_idx[valid_p]])
            if np.any(valid_n):
                lin[idx_n] -= best_h[i - s_idx[valid_n]]
            
            # Negative shifts (mirror)
            idx_p2 = n - 1 - s_idx[valid_p]  # These are n-1-s for s where i+s<n
            idx_n2 = n - 1 - s_idx[valid_n]  # These are n-1-s for s where i-s>=0
            
            if np.any(valid_p):
                lin[idx_p2] += (1.0 - best_h[i + s_idx[valid_p]])
            if np.any(valid_n):
                lin[idx_n2] -= best_h[i - s_idx[valid_n]]
            
            lin[n - 1] = 1.0 - 2.0 * old
            
            c_base = corr.copy()
            
            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val) * dx)
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12, 'maxiter': 100})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

        if c5_bh < best_c5:
            best_c5 = c5_bh
            best_h = h_bh.copy()

    # Phase 4: Binary structure search + refine (12% time)
    t4 = t0 + budget_s * 0.70
    if time.time() < t4:
        h_bin, c5_bin = binary_search(best_h, n0, min(t4, time.time() + budget_s * 0.06), max_iters=40000, seed=seed + 300)
        h_ref, c5_ref = coord_descent(h_bin, n0, t4, max_sweeps=1000, seed=seed + 350)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 5: Try n=200 grid (13% time)
    t5 = t0 + budget_s * 0.83
    if time.time() < t5:
        n_large = 200
        h_lg = upsample(best_h, n_large)
        h_lg, c5_lg = coord_descent(h_lg, n_large, min(t5, time.time() + budget_s * 0.06), max_sweeps=800, seed=seed + 400)
        if time.time() < t5:
            h_lg, c5_lg = basin_hopping(h_lg, n_large, t5, n_hops=80, seed=seed + 500)
        if c5_lg < best_c5:
            best_c5 = c5_lg
            best_h = h_lg.copy()

    # Phase 6: Final heavy coordinate descent (12% time)
    t6 = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, len(best_h), t6, max_sweeps=5000, seed=seed + 600)

    # Phase 7: Final polish (4% time)
    t7 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent(best_h, len(best_h), t7, max_sweeps=2000, seed=seed + 700)

    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
