from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    h = np.clip(h, 0.0, 1.0)
    # Fix any residual from clipping
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        else:
            s2 = h.sum()
            if s2 > 1e-14:
                h = h * (1.0 + diff2 / s2)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-15:

        lg = idx - (n - 1)
        if lg == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lg > 0:
            if lg < n:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += room * (diff2 / tot)
        else:
            s2 = h.sum()
            if s2 > 1e-14:
                h *= (1.0 + diff2 / s2)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-15:
            continue
        lg = idx - (n - 1)
        if lg == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < lg < n:
            grad[:n - lg] += wi * (1.0 - h[lg:])
            grad[lg:] += -wi * h[:n - lg]
        elif -n < lg < 0:
            p = -lg
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n - p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_step(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def _obj(x):
        return softmax_obj_grad(x, T, lam)[0]
    def _grad(x):
        return softmax_obj_grad(x, T, lam)[1]
    try:
        res = minimize(_obj, x0, jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-13})
        return project_h(res.x)
    except Exception:
        return x0

        if time.time() > t_end:
            break
        ones = np.where(best_h > 0.4)[0]
        zeros = np.where(best_h < 0.6)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h_trial = best_h.copy()
        h_trial[i], h_trial[j] = best_h[j], best_h[i]
        h_trial = project_h(h_trial)
        c5_new = compute_c5(h_trial)
        if c5_new < best_c5 - 1e-14:
            best_h = h_trial
            best_c5 = c5_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed + 7)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS warm-start (10%)
    p1_end = t0 + budget_s * 0.10
    for T in [50, 200, 1000, 5000]:
        if time.time() > p1_end:
            break
        hc = lbfgs_step(best_h, T, 800.0, 2500, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: Main coordinate descent (35%)
    p2_end = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, n, p2_end, 5000, rng)

    # Phase 3: Binary local search + refine (15%)
    p3_end = t0 + budget_s * 0.60
    h_bin = (best_h > 0.5).astype(float)
    h_bin = project_h(h_bin)
    h_bin, c5_bin = binary_local_search(h_bin, n, p3_end, 8000, rng)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin
    p3b_end = min(t0 + budget_s * 0.72, p3_end + 0.12 * budget_s)
    best_h, best_c5 = coord_descent(best_h, n, p3b_end, 2000, rng)

    # Phase 4: Basin-hopping perturbations (20%)
    p4_end = t0 + budget_s * 0.92
    work_h = best_h.copy()
    work_c5 = best_c5
    n_hops = 25
    for it in range(n_hops):
        if time.time() > p4_end:
            break
        cool = 1.0 - 0.85 * (it / float(n_hops))
        sc = 0.004 + 0.035 * cool
        h_p = project_h(work_h + sc * rng.randn(n))
        T_bh = int(2000 * cool + 300)
        hc = lbfgs_step(h_p, T_bh, 600.0, 1500, 1e-16)
        c5v = compute_c5(hc)
        hc, c5v = coord_descent(hc, n, min(time.time() + 4.0, p4_end), 80, rng)
        temp = 0.0004 * cool + 0.00006
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc

    # Phase 5: Final polish (8%)
    p5_end = t0 + budget_s * 0.98
    best_h, best_c5 = coord_descent(best_h, n, p5_end, 3000, rng)
    if time.time() < t0 + budget_s * 0.995:
        hc = lbfgs_step(best_h, 80000, 2000.0, 4000, 1e-18)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
