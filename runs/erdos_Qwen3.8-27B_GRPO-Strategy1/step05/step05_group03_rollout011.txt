import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: increase where room exists
        order = np.argsort(1.0 - h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining < 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            sub = min(remaining, h[idx])
            h[idx] -= sub
            remaining -= sub
            if remaining < 1e-14:
                break
    return h

    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')

            break
        order = np.random.permutation(n)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
