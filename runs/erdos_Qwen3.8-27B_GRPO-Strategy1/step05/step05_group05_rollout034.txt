def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    dx = 2.0 / n

    n = len(h)

    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h

        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:

            return h

    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)

    s = exp_vals.sum()

def coord_opt_ternary(h, n, i, niter=30):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(vm1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(vm2[valid2], 0, n - 1)])
    lo, hi = 0.0, 1.0

        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):

        else:

    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:

                return best_h, best_c5

            break
    return best_h, best_c5

def make_golden_h(n):
    alpha = (np.sqrt(5.0) - 1.0) / 4.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if ((i * alpha) % 1.0) < 0.5 else 0.0
    return project_h(h)

def make_block_h(n):
    h = np.zeros(n)
    blk = max(1, n // 12)
    for b in range(n // blk + 1):
        start = b * blk
        if start >= n:
            break
        end = min(n, start + blk)
        if b % 2 == 0:
            h[start:end] = 1.0
    return project_h(h)

def quick_opt(h, n, t_end):
    for T in [500, 5000, 50000]:
        if time.time() > t_end:
            break
        hc = slsqp_grad(h, T, n, maxiter=800, ftol=1e-14)
        h = hc
    h, c5 = coord_descent(h, n, t_end, max_sweeps=30)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_golden_h(160)

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-start (3 alternating starts)
    t_ms_end = t0 + budget_s * 0.12
    starts = [best_h.copy(), make_golden_h(n), make_block_h(n)]
    for s_i, s_h in enumerate(starts):
        if time.time() > t_ms_end:
            break
        s_t_end = min(t_ms_end, t0 + (budget_s * 0.12) * (s_i + 1) / 3.0)
        s_h, s_c5 = quick_opt(s_h, n, s_t_end)
        if s_c5 < best_c5:
            best_c5 = s_c5
            best_h = s_h

    # Phase 2: SLSQP with gradient, increasing T
    for T in [1000, 10000, 100000]:
        if time.time() > t0 + budget_s * 0.20:
            break
        hc = slsqp_grad(best_h, T, n, maxiter=3000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Coordinate descent (fast)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.38, max_sweeps=300)

    # Phase 4: Basin hopping with gradient-based local opt
    work_h = best_h.copy()
    work_c5 = best_c5
    t_bh_end = t0 + budget_s * 0.68
    max_bh = 300
    for it in range(max_bh):
        if time.time() > t_bh_end:
            break
        cool = max(0.0, 1.0 - it / max_bh)
        mod = it % 7
        if mod == 0:
            pert = (0.015 + 0.06 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 20)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.01 + 0.04 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.25
            pert = np.zeros(n)
            pert[mask] = (0.03 + 0.1 * cool) * np.random.randn(mask.sum())
        elif mod == 3:
            blk = max(1, n // 10)
            b = np.random.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            pert[b * blk:min(n, (b + 1) * blk)] = (0.02 + 0.05 * cool) * np.random.randn(blk)
        elif mod == 4:
            pert = np.zeros(n)
            for _ in range(6):
                a, b2 = np.random.permutation(n)[:2]
                d = (work_h[b2] - work_h[a]) * (0.3 + 0.7 * cool)
                pert[a] += d
                pert[b2] -= d
        elif mod == 5:
            pert = np.random.randn(n)
            pert -= pert.mean()
            pert *= (0.012 + 0.035 * cool)
        else:
            mask = np.random.random(n) < 0.08
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.12 * cool) * np.sign(np.random.randn(mask.sum()))
        h_pert = project_h(work_h + pert)
        T_bh = 5000 + 80000 * cool
        hc = slsqp_grad(h_pert, T_bh, n, maxiter=600, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.00004
        if c5v < work_c5 or (c5v - work_c5 < 0.005 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: High-T SLSQP (focus on critical lags)
    for T in [200000, 1000000, 5000000]:
        if time.time() > t0 + budget_s * 0.80:
            break
        hc = slsqp_grad(best_h, T, n, maxiter=4000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Extended coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, max_sweeps=500)

    # Phase 7: Final polish
    if time.time() < t0 + budget_s * 0.97:
        hc = slsqp_grad(best_h, 5000000, n, maxiter=3000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=60)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
