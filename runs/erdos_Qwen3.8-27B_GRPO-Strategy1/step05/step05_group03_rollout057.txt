def obj_and_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    sh = np.clip(T * (vals - m), -100, 100)
    exp_sh = np.exp(sh)
    p = exp_sh / exp_sh.sum()
    f = m + np.log(np.sum(exp_sh) / len(sh)) / T
    
    # Gradient
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each j and each lag: contribution
    # grad[j] = dx * sum_k p[k] * [(1-h[j+lag_k])*I(0<=j+lag_k<n) + (-h[j-lag_k])*I(0<=j-lag_k<n)]
    grad = np.zeros(n)
    for k in range(2*n-1):
        lag = lags[k]
        # First term: i=j, need 0<=j+lag<n
        # j ranges from max(0,-lag) to min(n, n-lag)
        j_lo = max(0, -lag)
        j_hi = min(n, n - lag)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += p[k] * (1.0 - h[j_lo+lag:j_hi+lag])
        # Second term: i+lag=j, so i=j-lag, need 0<=j-lag<n
        # j ranges from max(0,lag) to min(n, n+lag)
        j_lo2 = max(0, lag)
        j_hi2 = min(n, n + lag)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] += p[k] * (-h[j_lo2-lag:j_hi2-lag])
    grad *= dx
    
    return f, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def obj_grad_al(h, T, n, lam=0.0, mu=0.0):
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    sh = np.clip(T * (vals - m), -100, 100)
    exp_sh = np.exp(sh)
    p = exp_sh / exp_sh.sum()
    sm = m + np.log(exp_sh.sum() / (2 * n - 1)) / T
    c_val = h.sum() - target
    f = sm + lam * c_val + 0.5 * mu * c_val * c_val
    grad_sm = np.zeros(n)
    for k_idx in range(2 * n - 1):
        lag = k_idx - (n - 1)

                grad_sm[m2:] += pk * (-h[:n - m2])

    grad = grad_sm + (lam + mu * c_val)

            return obj_grad_al(x, T, n, lam, mu)

    return best_h, best_c5

def coord_opt_1d(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    lags = np.arange(-(n - 1), n)
    coefs = np.zeros(2 * n - 1)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    def f_1d(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f_1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 50})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_1d(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        for i in range(n - 1, -1, -1):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_1d(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hopping(h_init, n, t0, t_end, iterations=80):
    work_h = h_init.copy()
    work_c5 = compute_c5(work_h)
    best_h = h_init.copy()
    best_c5 = work_c5
    for it in range(iterations):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(1, iterations)
        ptype = np.random.randint(4)
        if ptype == 0:
            amp = 0.03 + 0.07 * cool
            h_pert = project_h(work_h + amp * np.random.randn(n), n)
        elif ptype == 1:
            h_pert = work_h.copy()
            nflip = np.random.randint(3, max(4, n // 10))
            idx = np.random.choice(n, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = project_h(h_pert, n)
        elif ptype == 2:
            h_pert = work_h.copy()
            a = np.random.randint(0, max(1, n - 20))
            ln = np.random.randint(4, max(5, n // 6))
            b = min(a + ln, n)
            sh = np.random.randint(1, max(2, ln // 2))
            h_pert[a:b] = np.roll(h_pert[a:b], sh)
            h_pert = project_h(h_pert, n)
        else:
            h_pert = work_h.copy()
            k = np.random.randint(2, min(6, n // 8) + 1)
            idx = np.random.choice(n, size=k * 2, replace=False)
            for ii in range(0, k * 2, 2):
                h_pert[idx[ii]], h_pert[idx[ii + 1]] = h_pert[idx[ii + 1]], h_pert[idx[ii]]
            h_pert = project_h(h_pert, n)
        T_bh = 200 + 8000 * cool
        tl = max(2, min(15, (t_end - time.time()) * 0.8))
        hc, cv = run_al_lbfgsb(h_pert, T_bh, n, maxiter=1200, time_limit=tl)
        temp = 0.0004 * cool + 0.00003
        if cv < work_c5 or np.random.rand() < np.exp(-(cv - work_c5) / temp):
            work_h, work_c5 = hc, cv
            if cv < best_c5:
                best_c5, best_h = cv, hc.copy()
    return best_h, best_c5

def make_thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = float(bin(i).count('1') % 2)
    return tm

def make_block_pattern(n, period):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % (2 * period)) < period else 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    candidates = [best_h]
    candidates.append(project_h(make_thue_morse(n), n))
    for p in [3, 5, 7, 11]:
        candidates.append(project_h(make_block_pattern(n, p), n))
    for _ in range(3):
        rb = np.zeros(n)
        rb[np.random.choice(n, n // 2, replace=False)] = 1.0
        candidates.append(rb)

    for cand in candidates[1:]:
        if time.time() - t0 > budget_s * 0.04:
            break
        hc, cv = run_al_lbfgsb(cand, 3000, n, maxiter=800, time_limit=8)
        if cv < best_c5:
            best_c5, best_h = cv, hc

    for T in [20000, 8000, 3000, 1000, 500, 200, 100, 50, 20, 10, 5]:
        if time.time() - t0 > budget_s * 0.20:
            break
        hc, cv = run_al_lbfgsb(best_h, T, n, maxiter=2000, time_limit=18)
        if cv < best_c5:
            best_c5, best_h = cv, hc

    best_h, best_c5 = basin_hopping(best_h, n, t0, t0 + budget_s * 0.42, iterations=70)

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.60, max_sweeps=500)

    for T in [100, 50, 20, 10, 5, 3, 2]:
        if time.time() - t0 > budget_s * 0.78:
            break
        hc, cv = run_al_lbfgsb(best_h, T, n, maxiter=3000, time_limit=20)
        if cv < best_c5:
            best_c5, best_h = cv, hc

    best_h, best_c5 = basin_hopping(best_h, n, t0, t0 + budget_s * 0.80, iterations=40)

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.92, max_sweeps=800)

    for T in [5, 3, 2]:
        if time.time() - t0 > budget_s * 0.96:
            break
        hc, cv = run_al_lbfgsb(best_h, T, n, maxiter=4000, time_limit=15)
        if cv < best_c5:
            best_c5, best_h = cv, hc

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=300)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
