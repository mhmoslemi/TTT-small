def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

            lo = mid
        else:
            hi = mid

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def find_best_swap(c, D, ones, zeros, P, batch=60):
    n_o = len(ones)
    D_o = D[ones, :]
    best_val = P
    best_pi, best_qi = -1, -1
    for qs in range(0, len(zeros), batch):
        qb = zeros[qs:qs + batch]
        D_qb = D[qb, :]
        M = c[None, None, :] - D_o[:, None, :] - D_qb[None, :, :]
        Mmax = M.max(axis=2)
        fi = int(np.argmin(Mmax))
        pi, qi = fi // len(qb), fi % len(qb)
        if Mmax[pi, qi] < best_val:
            best_val = float(Mmax[pi, qi])
            best_pi, best_qi = pi, qs + qi
    if best_pi < 0:
        return P, None, None
    return best_val, int(ones[best_pi]), int(zeros[best_qi])

def fast_binary(h, N, deadline, max_sweeps=99999, T0=0.0, cool=0.9995):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    c = fft_corr(h)
    T = T0
    flat = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c:
            best_c = P
            flat = 0
        else:
            flat += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        bv, bp, bq = find_best_swap(c, D, ones, zeros, P)
        do_swap = False
        if bp is not None and bv < P:
            h[bp] = 0.0; h[bq] = 1.0; do_swap = True
        elif T > 0 and bp is not None:
            d = bv - P
            if d < 0.03 and np.random.random() < np.exp(-d / max(T, 1e-8)):
                h[bp] = 0.0; h[bq] = 1.0; do_swap = True
        elif T > 0 and np.random.random() < 0.08:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0; do_swap = True
        if do_swap:
            c = fft_corr(h)
        if T > 0.0001:
            T *= cool
        else:
            T = 0.0
    return h, best_c

def nesterov_opt(h, N, deadline, k=80, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    for it in range(8000):
        if time.time() > deadline:
            break
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 400 == 0:
            lr *= 0.94
            beta = min(0.98, 0.88 + it * 0.0001)
    return best_h, best_pk * dx

def upscale(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    h_bt = np.where(h_interp > 0.5, 1.0, 0.0)
    nh = N_new // 2
    n1 = int(h_bt.sum())
    if n1 > nh:
        idx = np.where(h_bt == 1)[0]
        order = np.argsort(h_interp[idx])[:n1 - nh]
        h_bt[idx[order]] = 0
    elif n1 < nh:
        idx = np.where(h_bt == 0)[0]
        order = np.argsort(-h_interp[idx])[:nh - n1]
        h_bt[idx[order]] = 1
    return h_bt
