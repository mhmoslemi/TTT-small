import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h *= (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t > 1e-13:
            h += rm * ((tgt - s) / t)
        else:
            h *= (tgt / max(s, 1e-13))
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)

def soft_obj(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n

def soft_grad(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo = max(0, n - 1 - k)
        hi = min(n, 2 * n - 1 - k)
        idx = np.arange(lo, hi)
        j1 = idx + k - n + 1
        j2 = idx - k + n - 1
        g[idx] += wk * ((1.0 - x[j1]) - x[j2])
    return g * (2.0 / n)

def lbb(h, T, n, iters):
    try:
        r = minimize(h, soft_obj, soft_grad, method='L-BFGS-B',
                     args=(T,), bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12,
                              'maxls': 50})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            if time.time() > t_end:
                break
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            base = corr - old * coef
            # minimize max(base + hi*coef) via ternary search on [0,1]
            lo_b, hi_b = 0.0, 1.0
            for _ in range(60):
                m1 = lo_b + (hi_b - lo_b) / 3.0
                m2 = hi_b - (hi_b - lo_b) / 3.0
                if np.max(base + m1 * coef) <= np.max(base + m2 * coef):
                    hi_b = m2
                else:
                    lo_b = m1
            nv = (lo_b + hi_b) / 2.0
            if abs(nv - old) < 1e-14:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def guided_basin(h, n, t_end, n_iters=150):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 4
        if mod == 0:
            pert = (0.03 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 16)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.25 + 0.35 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.06 + 0.2 * cool) * np.random.randn(mask.sum())
        else:
            # Structure-aware: target peak correlations
            corr = np.correlate(work, 1.0 - work, mode='full')
            top_k = np.argsort(corr)[-3:]
            pert = np.zeros(n)
            for k in top_k:
                for i in range(n):
                    j1 = i + k - n + 1
                    j2 = i - k + n + 1
                    if 0 <= j1 < n:
                        pert[i] += (1.0 - work[j1])
                    if 0 <= j2 < n:
                        pert[i] -= work[j2]
            pert = (0.02 + 0.08 * cool) * pert / (np.linalg.norm(pert) + 1e-10)
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 8000.0 * cool
        ht = lbb(hp, T_val, n, iters=300)
        c5t = compute_c5(ht)
        temp = 0.001 * cool + 0.00003
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def try_resolution(h_best, c5_best, n_target, t_end):
    n_curr = len(h_best)
    if n_target == n_curr:
        return h_best, c5_best
    x_old = np.linspace(0, 2, n_curr, endpoint=False) + 1.0 / n_curr
    x_new = np.linspace(0, 2, n_target, endpoint=False) + 1.0 / n_target
    h_t = np.interp(x_new, x_old, h_best, period=2.0) if n_target > n_curr else \
          np.interp(x_new, x_old, h_best)
    h_t = proj_h(h_t, n_target)
    t_rem = max(5.0, t_end - time.time())
    for T in [200, 2000, 20000]:
        if time.time() > t_end:
            break
        h_t = lbb(h_t, T, n_target, iters=min(400, int(200 * t_rem / 60)))
        c5_t = compute_c5(h_t)
    if time.time() < t_end:
        t_sub = min(time.time() + 30, t_end)
        h_t, c5_t = coord_desc(h_t, n_target, t_sub, sweeps=40)
    return h_t, c5_t

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s * 0.95

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Soft-max L-BFGS with increasing T
    for T in [100, 500, 2000, 10000, 50000]:
        if time.time() > t0 + budget_s * 0.12:
            break
        h_new = lbb(best_h, T, n, iters=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Extended coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.30, sweeps=120)

    # Phase 3: Guided basin hopping
    best_h, best_c5 = guided_basin(best_h, n, t0 + budget_s * 0.55, n_iters=150)

    # Phase 4: Post-basin coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.68, sweeps=80)

    # Phase 5: Try other resolutions
    for n_t in [140, 180, 200, 120]:
        if time.time() > t_end:
            break
        t_sub = t0 + budget_s * 0.88
        h_t, c5_t = try_resolution(best_h, best_c5, n_t, t_sub)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 6: Another basin round on best
    if time.time() < t_end:
        best_h, best_c5 = guided_basin(best_h, len(best_h), min(t_end, time.time() + budget_s * 0.12), n_iters=80)

    # Phase 7: Final polish - high-T L-BFGS + coordinate descent
    if time.time() < t_end - 30:
        best_h = lbb(best_h, 80000, len(best_h), iters=1000)
        best_c5 = compute_c5(best_h)
    if time.time() < t_end:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t_end, sweeps=50)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
