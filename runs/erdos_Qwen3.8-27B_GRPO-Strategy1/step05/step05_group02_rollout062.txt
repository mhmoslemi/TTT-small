        return h
    diff = target - s
    if diff > 0:

        tot = room.sum()
        if tot > 1e-14:
            h += room * (diff / tot)

        if s2 > 1e-14:
            h *= (1.0 + diff / s2)

Let me also think about one more optimization trick: **double loop coordinate descent**. In each sweep, for each coordinate i, I optimize h[i] given all other coordinates fixed. The objective as a function of y = h[i] is:

f(y) = max_k [corr_base[k] + y * coef[k]] * dx

where corr_base[k] is the correlation with h[i]=0, and coef[k] captures the linear dependence on y.

Actually, corr[k] = Σᵢ h[i+k](1-h[i]) is quadratic in each h[j] (because h[j] appears in both h[i+k] and (1-h[i]) terms). So as a function of y=h[i]:

corr[k] = [terms not involving h[i]] + y*(1-h[i-k]) - y*h[i+k] + y²*h[i+k]... 

Wait, let me be more careful. For a fixed lag k:
corr[k] = Σᵢ h[i+k](1-h[i])

h[i] (where the index in the sum equals our optimization variable) appears:
- As h[i'+k] when i' + k = i, i.e., i' = i-k: contributes h[i](1-h[i-k])
- As (1-h[i']) when i' = i: contributes h[i+k](1-h[i])... wait no.

Let me use different notation. Let y = h[j] be the variable. Then:
corr[k] = Σᵢ h[i+k](1-h[i])

y appears in two ways:
1. In h[i+k] when i+k = j, i.e., i = j-k: term is y*(1-h[j-k])
2. In (1-h[i]) when i = j: term is h[j+k]*(1-y)

So corr[k] = [Σᵢ≠(j-k),(j) h[i+k](1-h[i])] + y*(1-h[j-k])*(i=j-k valid) + h[j+k]*(1-y)*(i=j valid)

= constant + y*(1-h[j-k]) - y*h[j+k] + (other terms)

Wait, that's:
corr[k] = C_k + y*(1-h[j-k])_valid + h[j+k]*(1-y)_valid

where C_k is the sum over all other terms, and the subscripts indicate the terms are only present if the indices are valid.

So corr[k] as a function of y is:
corr[k](y) = C_k + y*a_k - y*b_k = C_k + y*(a_k - b_k)

where a_k = (1-h[j-k]) if j-k is valid in the sum, else 0
and b_k = h[j+k] if j is valid in the sum and j+k < n, else 0.

Wait, that's LINEAR in y! Because the cross term h[i+k]*h[i] only involves y once (either as h[i+k]=y or h[i]=y, not both, unless k=0).

For k=0: corr[0] = Σᵢ h[i](1-h[i]). The term with i=j is y(1-y) = y - y². This IS quadratic!

So for k≠0, corr[k] is linear in y. For k=0, it has a y-y² term.

This means I can solve for the optimal y more efficiently! The objective is:
f(y) = max_k [corr_base[k] + y*coef[k]] * dx  (for k≠0)
      with k=0 term: corr_base[0] + y - y²

For k≠0, this is the max of linear functions of y, which is convex in y. Adding the k=0 term (y-y² which is concave) makes it non-convex, but only slightly (the y² coefficient is -1).

In practice, I can just use minimize_scalar on this 1D function, which is what the previous code did. Let me keep that.

Actually wait, for k≠0, the dependence IS linear. So the max over k of (C_k + y*d_k) is a convex piecewise-linear function of y. The optimum is at one of the breakpoints where two lines cross, or at y=0 or y=1.

I can compute the optimal y by finding the y* that minimizes max_k (C_k + y*d_k), which is a 1D convex optimization. This can be solved exactly:
- The function g(y) = max_k (C_k + y*d_k) is convex and piecewise linear
- Its minimum is at y* where the subgradient contains 0
- Equivalently, at y* there exist k₁ and k₂ with d_{k₁} > d_{k₂} such that C_{k₁} + y*d_{k₁} = C_{k₂} + y*d_{k₂}

This is a "minimax" problem in 1D, solvable in O(n log n) by sorting the slopes.

But for simplicity and to handle the k=0 quadratic term, I'll just use minimize_scalar with the actual function evaluation.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def lse_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    a = T * (corr - m)
    a -= a.max()
    ea = np.exp(a)
    sw = ea.sum()
    w = ea / sw
    obj = (m + np.log(sw) / T) * dx
    grad = np.zeros(n)
    Tw = T * w
    for k in range(-(n-1), n):
        wm = Tw[n-1+k]
        if wm < 1e-13:
            continue
        if k == 0:
            grad += wm * (1.0 - 2.0*h)
        elif k > 0:
            grad[k:] += wm * (1.0 - h[:n-k])
            grad[:n-k] -= wm * h[k:]
        else:
            kp = -k
            grad[:n-kp] += wm * (1.0 - h[kp:])
            grad[kp:] -= wm * h[:n-kp]
    return obj, grad * dx

def projection(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = h + room * (diff / tot)
    else:
        if s > 1e-14:
            h = h * (1.0 + diff / s)
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    if abs(s2 - target) > 1e-10:
        d = target - s2
        idx = int(np.argmax(1.0 - h)) if d > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + d, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_lbfgs(h, T, maxiter=400):
    n = len(h)
    target = n / 2.0
    def fpen(x):
        o, g = lse_obj_grad(x, T)
        s = x.sum() - target
        o += 5e4 * s * s
        g += 1e5 * s
        return o, g
    try:
        res = minimize(fpen, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0,1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return projection(np.clip(res.x, 0, 1))
    except Exception:
        return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h
    old_x = (np.arange(old_n) + 0.5) / old_n * 2.0
    new_x = (np.arange(new_n) + 0.5) / new_n * 2.0
    h_new = np.interp(new_x, old_x, h)
    return projection(h_new)

def make_pert(n, cool, rng):
    r = rng.rand()
    if r < 0.25:
        bs = rng.randint(5, max(6, n // 6))
        st = rng.randint(0, n - bs)

    elif r < 0.5:
        freq = rng.randint(1, max(2, n // 12))
        ph = rng.uniform(0, 2*np.pi)
        p = (0.04 + 0.10*cool) * np.sin(2*np.pi*freq*np.arange(n)/n + ph)
    elif r < 0.7:
        p = np.zeros(n)
        np_ = rng.randint(4, max(5, n//6))
        idx = rng.choice(n, np_, replace=False)
        p[idx] = (0.1 + 0.2*cool) * rng.standard_normal(np_)
    elif r < 0.85:

    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.08*cool * rng.standard_normal(half)
        p[half:] = -0.08*cool * rng.standard_normal(n-half)
    return p

def coord_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n-1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2*n - 1)
    coefs[v1] += (1.0 - h_base[np.clip(j1[v1], 0, n-1)])
    coefs[v2] += (-h_base[np.clip(j2[v2], 0, n-1)])
    dx = 2.0 / n
    bad_idx = i - (n-1)
    corr_base[bad_idx] = corr_base[bad_idx]  # k=0 term needs y*y
    try:
        res = minimize_scalar(lambda y: np.max(corr_base + y*coefs - y*y*np.eye(1)[0,0]*0 + (y-y*y)*0)*dx,
                              bounds=(0.0,1.0), method='bounded', options={'xatol':1e-12})
        res2 = minimize_scalar(lambda y: _eval_coord(corr_base, coefs, y, dx),
                              bounds=(0.0,1.0), method='bounded', options={'xatol':1e-12})
        return float(res2.x)
    except Exception:
        return h[i]

def _eval_coord(corr_base, coefs, y, dx):
    corr = corr_base + y * coefs
    k0 = len(corr_base)//2
    corr[k0] += y - y*y
    return float(np.max(corr) * dx)

def coord_descent(h, n, t_end, max_sweeps=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = _coord_single_fast(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = projection(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def _coord_single_fast(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n-1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2*n - 1)
    coefs[v1] += (1.0 - h_base[np.clip(j1[v1], 0, n-1)])
    coefs[v2] += (-h_base[np.clip(j2[v2], 0, n-1)])
    dx = 2.0 / n
    k0 = n - 1
    try:
        res = minimize_scalar(lambda y: _eval_coord(corr_base, coefs, y, dx),
                              bounds=(0.0, 1.0), method='bounded', options={'xatol':1e-12})
        return float(res.x)
    except Exception:
        return h[i]

def basin_hop(h, n, rng, t_end, n_iter=30, T_base=500, accept_temp=0.001):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = h.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.8*(it/n_iter)
        pert = make_pert(n, cool, rng)
        h_pert = projection(work_h + pert)
        T_cur = T_base + 5000*cool
        hc = lse_lbfgs(h_pert, T_cur, maxiter=300)
        c5v = compute_c5(hc)
        temp = accept_temp * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5)/max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    return best_h, best_c5
