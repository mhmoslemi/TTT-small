def subgrad_step(h, alpha):
    """One subgradient step on the actual max of corr."""
    n = len(h)
    dx = 2.0 / n
    corr = corr_all(h)
    idx_max = np.argmax(corr)
    # Subgradient of corr[idx_max] w.r.t. h
    lg = idx_max - (n - 1)  # the lag
    grad = np.zeros(n)
    if lg >= 0:
        if lg < n:
            grad[lg:] += (1.0 - h[:n-lg])
            grad[:n-lg] += -h[lg:]
    else:
        p = -lg
        if p < n:
            grad[:n-p] += (1.0 - h[p:])
            grad[p:] += -h[:n-p]
    grad *= dx
    h_new = h - alpha * grad
    return project_h(h_new)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_all(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tr = room.sum()
        if tr < 1e-14:
            return h
        h += room * (diff / tr)
    else:
        if s < 1e-14:
            return h
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    d2 = target - s2
    if abs(d2) > 1e-12:
        m = h < 1.0 if d2 > 0 else h > 0.0
        if m.any():
            h[m] += d2 / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = corr_all(h)
    mc = np.max(corr)
    Tc = (corr - mc) * T
    Tc -= Tc.max()
    exps = np.exp(Tc)
    w = exps / exps.sum()
    grad = np.zeros(n)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-14:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n-p]
    return dx * grad

def lbfgs_step(h, T, lam, maxiter):
    n = len(h)
    def f(x):
        dx = 2.0 / n
        c = corr_all(x)
        mc = np.max(c)
        Tc = (c - mc) * T
        Tc -= Tc.max()
        return dx * (mc + np.log(np.sum(np.exp(Tc))) / T) + lam * (x.sum() - n/2.0)**2
    def g(x):
        return grad_smooth_max(x, T) + 2.0 * lam * (x.sum() - n/2.0)
    x0 = project_h(h)
    try:
        res = minimize(f, x0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-11})
        return project_h(res.x)
    except Exception:
        return x0

def coord_descent(h, t_end, max_sweeps):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            v = best_h[i]
            h0 = best_h.copy(); h0[i] = 0.0
            h1 = best_h.copy(); h1[i] = 1.0
            c0 = corr_all(h0)
            c1 = corr_all(h1)
            B = c1 - c0
            C = np.zeros(m_full)
            B[n-1] = 1.0
            C[n-1] = -1.0
            A = c0
            def f1d(x):
                return np.max(A + B * x + C * x * x)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            x_new = res.x
            if abs(x_new - v) < 1e-14:
                continue
            ht = best_h.copy()
            ht[i] = x_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def gen_candidates(n, seed):
    rng = np.random.RandomState(seed)
    cands = []
    hb = np.zeros(n); hb[:n//2] = 1.0
    cands.append(project_h(hb))
    ha = np.zeros(n); ha[::2] = 1.0
    cands.append(project_h(ha))
    for fr in [2, 3, 5, 7, 11]:
        hp = 0.5 + 0.45 * np.sin(2*np.pi*fr*np.arange(n)/n)
        cands.append(project_h(hp))
    for _ in range(4):
        cands.append(project_h(rng.beta(3, 3, n)))
    for sc in [0.02, 0.05, 0.1, 0.2]:
        cands.append(project_h(0.5 + sc * rng.randn(n)))
    return cands

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160); best_h[:80] = 1.0
        best_h = project_h(best_h)
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS warm-up (12%)
    p1_end = t0 + budget_s * 0.12

        if c5v < best_c5:

            best_h = hc

    # Phase 2: Multi-restart L-BFGS (18%)

        if time.time() > p2_end:

        hc = lbfgs_step(h_cand, 600, 200.0, 2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Coordinate descent – main workhorse (38%)

    # Phase 4: High-T L-BFGS polish (12%)
    p4_end = t0 + budget_s * 0.80
    for T in [5000, 20000, 80000, 300000]:
        if time.time() > p4_end:

        hc = lbfgs_step(best_h, T, 300.0, 4000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # CD after high-T
    if time.time() < p4_end + budget_s * 0.04:
        best_h, best_c5 = coord_descent(best_h, p4_end + budget_s * 0.04, 500)

    # Phase 5: Basin hopping (7%)
    p5_end = t0 + budget_s * 0.92

    for it in range(40):
        if time.time() > p5_end:

        T_bh = int(2000 * cool + 300)
        hc = lbfgs_step(h_p, T_bh, 200.0, 1200)
        c5v = compute_c5(hc)
        temp = 0.0003 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        # quick CD after each hop

            work_h, work_c5 = coord_descent(work_h, min(time.time()+2, p5_end), 80)
            if work_c5 < best_c5:

                best_h = work_h.copy()

    # Phase 6: Final polish (8%)
    p6_end = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent(best_h, p6_end, 1000)
    if time.time() < p6_end:

            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
