        p[idx] = (0.05 + 0.15 * cool) * rng.standard_normal(np_)
    else:
        p = (0.01 + 0.04 * cool) * rng.standard_normal(n)
    return p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n

    # Phase 1: SLSQP warm-up (4% time)
    for T in [300, 2000]:
        if time.time() - t0 > budget_s * 0.04:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Binarize + intensive swap SA (50% time)
    h_work = binarize(best_h)
    c5_work = compute_c5(h_work)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()

    box = [best_c5]
    bh_copy = best_h.copy()
    h_work, c5_work = swap_search(h_work, c5_work, t0 + budget_s * 0.30, rng, box, bh_copy, n, dx)
    best_c5 = box[0]
    best_h = bh_copy.copy()

    # Phase 3: Coordinate descent (12% time)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.42, rng, max_sweeps=150)

    # Phase 4: Binarize again + more SA (18% time)
    h_work = binarize(best_h)
    c5_work = compute_c5(h_work)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()
    box = [best_c5]
    bh_copy = best_h.copy()
    h_work, c5_work = swap_search(h_work, c5_work, t0 + budget_s * 0.60, rng, box, bh_copy, n, dx)
    best_c5 = box[0]
    best_h = bh_copy.copy()

    # Phase 5: Basin hopping (12% time)
    h_bh = best_h.copy()
    c5_bh = best_c5
    for it in range(30):
        if time.time() > t0 + budget_s * 0.72:
            break
        cool = 1.0 - 0.8 * (it / 30.0)
        pert = make_pert(h_bh, n, cool, rng)
        h_pert = project_h(h_bh + pert)
        T_bh = 500 + 5000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=400)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0002
        if c5v < c5_bh or rng.rand() < np.exp(-(c5v - c5_bh) / max(temp, 1e-9)):
            h_bh = hc
            c5_bh = c5v
            if c5_bh < best_c5:
                best_c5 = c5_bh
                best_h = h_bh.copy()

    # Phase 6: Final SA from best (10% time)
    h_work = best_h.copy()
    c5_work = compute_c5(h_work)
    box = [best_c5]
    bh_copy = best_h.copy()
    h_work, c5_work = swap_search(h_work, c5_work, t0 + budget_s * 0.82, rng, box, bh_copy, n, dx)
    best_c5 = box[0]
    best_h = bh_copy.copy()

    # Phase 7: Final coordinate descent + SLSQP polish (10% time)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, rng, max_sweeps=80)
    if time.time() < t0 + budget_s * 0.95:
        for T in [20000, 100000]:
            if time.time() > t0 + budget_s * 0.96:
                break
            hc = slsqp_opt(best_h, T, n, maxiter=1500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
