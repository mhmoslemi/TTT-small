import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smoothed_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-15:
            continue
        lg = idx - (n - 1)
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[p:] += wi * (1.0 - h[:n-p])
            grad[:n-p] += -wi * h[p:]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0.copy())
    def obj(x):
        return smoothed_obj_grad(x, T, lam)[0]
    def grad(x):
        return smoothed_obj_grad(x, T, lam)[1]
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return x0

def block_descent(h, n, block_size, t_end, T_smooth, max_sweeps):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_blocks = (n + block_size - 1) // block_size
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n_blocks)
        for bi in order:
            s = bi * block_size
            e = min(s + block_size, n)
            ba = e - s
            if ba < 2:
                continue
            rest_sum = best_h.sum() - best_h[s:e].sum()
            bt = n / 2.0 - rest_sum
            def bo(x):
                hn = best_h.copy()
                hn[s:e] = x
                corr = np.correlate(hn, 1.0 - hn, mode='full')
                mc = np.max(corr)
                Tc = T_smooth * (corr - mc)
                Tc -= Tc.max()
                return (mc + np.log(np.exp(Tc).sum()) / T_smooth) * (2.0 / n)
            def bgrad(x):
                hn = best_h.copy()
                hn[s:e] = x
                corr = np.correlate(hn, 1.0 - hn, mode='full')
                mc = np.max(corr)
                Tc = T_smooth * (corr - mc)
                Tc -= Tc.max()
                exps = np.exp(Tc)
                w = exps / exps.sum()
                g = np.zeros(ba)
                for idx in range(2 * n - 1):
                    wi = w[idx]
                    if wi < 1e-15:
                        continue
                    lg = idx - (n - 1)
                    # contribution to h[j] for j in [s, e)
                    for j in range(s, e):
                        # d corr[idx] / d h[j]
                        if lg >= 0:
                            # corr = sum_{i=0}^{n-1-lg} h[i+lg]*(1-h[i])
                            if j >= lg and (j - lg) < n - lg:
                                g[j - s] += wi * (1.0 - hn[j - lg])
                            if j < n - lg and (j + lg) < n:
                                g[j - s] += -wi * hn[j + lg]
                        else:
                            p = -lg
                            # corr = sum_{i=p}^{n-1} h[i]*(1-h[i+p])
                            if j >= p and (j + p) < n:
                                g[j - s] += wi * (1.0 - hn[j + p])
                            if (j - p) >= 0 and j < n - p:
                                g[j - s] += -wi * hn[j - p]
                return (2.0 / n) * g
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - bt,
                     'jac': lambda x: np.ones(ba)}]
            x0 = best_h[s:e].copy()
            try:
                res = minimize(bo, x0, jac=bgrad, method='SLSQP',
                               bounds=[(0.0, 1.0)] * ba,
                               constraints=cons,
                               options={'maxiter': 80, 'ftol': 1e-15})
                if not res.success or not np.all(res.x >= -1e-10):
                    continue
                hn = best_h.copy()
                hn[s:e] = np.clip(res.x, 0, 1)
                hn = project_h(hn)
                c5n = compute_c5(hn)
                if c5n < best_c5 - 1e-15:
                    best_c5 = c5n
                    best_h = hn
                    improved = True
            except Exception:
                continue
        if not improved:
            break
    return best_h, best_c5

def coord_descent_fast(h, n, t_end, max_sweeps):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h0 = best_h.copy()
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            def f1d(x):
                return np.max(cb + x * lin + x * x * quad)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            ht = best_h.copy()
            ht[i] = res.x
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

        # d/dh[q] for q in [0, n-1-p]: (1-h[q+p])
        # d/dh[q] for q in [p, n-1]: -h[q-p]
        grad[:n-p] += wi * (1.0 - h[p:])
        grad[p:] += -wi * h[:n-p]
```

Let me verify: for lg ≥ 0:
- grad[lg:] += wi * (1.0 - h[:n-lg])
  - grad[lg] += wi * (1-h[0]), grad[lg+1] += wi * (1-h[1]), ..., grad[n-1] += wi * (1-h[n-1-lg])
  - This is the contribution from h[j+lg] where j goes 0 to n-1-lg ✓
  
- grad[:n-lg] += -wi * h[lg:]
  - grad[0] += -wi*h[lg], grad[1] += -wi*h[lg+1], ..., grad[n-1-lg] += -wi*h[n-1]
  - This is the contribution from (1-h[j]) where j goes 0 to n-1-lg ✓

For lg < 0 (p = -lg):
- grad[:n-p] += wi * (1.0 - h[p:])
  - grad[0] += wi*(1-h[p]), grad[1] += wi*(1-h[p+1]), ..., grad[n-1-p] += wi*(1-h[n-1])
  - This is the contribution from h[j-p] where j goes p to n-1, so q=j-p goes 0 to n-1-p ✓
  
- grad[p:] += -wi * h[:n-p]
  - grad[p] += -wi*h[0], grad[p+1] += -wi*h[1], ..., grad[n-1] += -wi*h[n-1-p]
  - This is the contribution from (1-h[j]) where j goes p to n-1, so q=j goes p to n-1 ✓

Great, the gradient formula looks correct.

def bgrad(x):

This computes the full gradient and extracts the block. A bit wasteful but correct and simple. For n=160, computing the full gradient is O(n * m_full) = O(160 * 319) ≈ 51000 operations, which is fast.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def full_grad(h, T, lam):
    """Compute gradient of smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    grad = np.zeros(n)
    m_full = 2 * n - 1
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-15:
            continue
        lg = idx - (n - 1)
        if lg >= 0:
            grad[lg:] += wi * (1.0 - h[:n - lg])
            grad[:n - lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n - p]
    gval = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    sm = dx * (mc + np.log(exps.sum()) / T) + lam * (h.sum() - n / 2.0) ** 2
    return sm, gval

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0.copy())
    def obj(x):
        return full_grad(x, T, lam)[0]
    def grad(x):
        return full_grad(x, T, lam)[1]
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return x0

def block_descent(h, n, block_size, t_end, T_s, max_sweeps):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_blocks = (n + block_size - 1) // block_size
    m_full = 2 * n - 1
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n_blocks)
        for bi in order:
            s = bi * block_size
            e = min(s + block_size, n)
            ba = e - s
            if ba < 2:
                continue
            bt = n / 2.0 - (best_h.sum() - best_h[s:e].sum())
            
            def bo(x):
                hn = best_h.copy()
                hn[s:e] = x
                corr = np.correlate(hn, 1.0 - hn, mode='full')
                mc = np.max(corr)
                Tc = T_s * (corr - mc)
                Tc -= Tc.max()
                return (mc + np.log(np.exp(Tc).sum()) / T_s) * dx
            
            def bg(x):
                hn = best_h.copy()
                hn[s:e] = x
                corr = np.correlate(hn, 1.0 - hn, mode='full')
                mc = np.max(corr)
                Tc = T_s * (corr - mc)
                Tc -= Tc.max()
                exps = np.exp(Tc)
                w = exps / exps.sum()
                g = np.zeros(n)
                for idx in range(m_full):
                    wi = w[idx]
                    if wi < 1e-15:
                        continue
                    lg = idx - (n - 1)
                    if lg >= 0:
                        g[lg:] += wi * (1.0 - hn[:n - lg])
                        g[:n - lg] += -wi * hn[lg:]
                    else:
                        p = -lg
                        g[:n - p] += wi * (1.0 - hn[p:])
                        g[p:] += -wi * hn[:n - p]
                return dx * g[s:e]
            
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - bt,
                     'jac': lambda x: np.ones(ba)}]
            x0 = best_h[s:e].copy()
            try:
                res = minimize(bo, x0, jac=bg, method='SLSQP',
                               bounds=[(0.0, 1.0)] * ba,
                               constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-15})
                if not res.success:
                    continue
                hn = best_h.copy()
                hn[s:e] = np.clip(res.x, 0, 1)
                hn = project_h(hn)
                c5n = compute_c5(hn)
                if c5n < best_c5 - 1e-15:
                    best_c5 = c5n
                    best_h = hn
                    improved = True
            except Exception:
                continue
        if not improved:
            break
    return best_h, best_c5

def coord_desc(h, n, t_end, max_sweeps):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h0 = best_h.copy()
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            res = minimize_scalar(
                lambda x: np.max(cb + x * lin + x * x * quad),
                bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
            ht = best_h.copy()
            ht[i] = res.x
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B warm-up
    p1 = t0 + budget_s * 0.15
    for T in [50, 200, 1000, 5000]:
        if time.time() > p1:
            break
        hc = lbfgs_opt(best_h, T, 500.0, 2000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: Block coordinate descent (bs=4)
    p2 = t0 + budget_s * 0.50
    best_h, best_c5 = block_descent(best_h, n, 4, p2, 5000, 500)

    # Phase 3: High-T L-BFGS-B
    p3 = t0 + budget_s * 0.65
    for T in [20000, 100000]:
        if time.time() > p3:
            break
        hc = lbfgs_opt(best_h, T, 500.0, 3000, 1e-17)
