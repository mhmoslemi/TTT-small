import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T, n_top, var_pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1e8:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    else:
        val = m
    if var_pen > 0 and n_top > 0:
        idx = np.argsort(corr)[-n_top:]
        val += var_pen * np.var(corr[idx])
    return val * dx

def slsqp_opt(h, T, n, maxiter=400, var_pen=0.0, n_top=10):
    try:
        res = minimize(
            lambda x: smooth_c5(x, T, n_top, var_pen),
            h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def active_coord(h, n_act, t_end, max_sweeps=400):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    i_idx = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        n_a = min(n_act, len(corr))
        top_idx = np.argsort(corr)[-n_a:]
        top_shifts = top_idx - (n - 1)
        top_vals = corr[top_idx].copy()
        g = np.zeros((n, n_a))
        for t in range(n_a):
            k = top_shifts[t]
            if k >= 0:
                if n - k > 0:
                    g[:n-k, t] = 1.0 - best_h[k:n]
                    g[k:, t] -= best_h[:n-k]
            else:
                kk = -k
                if n - kk > 0:
                    g[kk:, t] = 1.0 - best_h[:n-kk]
                    g[:n-kk, t] -= best_h[kk:n]
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            gi = g[i, :]
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if hi - lo < 1e-14:
                continue
            dg = gi[None, :] - gi[:, None]
            dv = top_vals[None, :] - top_vals[:, None]
            mask = (np.abs(dg) > 1e-14)
            uu = np.triu(mask, 1)
            with np.errstate(divide='ignore', invalid='ignore'):
