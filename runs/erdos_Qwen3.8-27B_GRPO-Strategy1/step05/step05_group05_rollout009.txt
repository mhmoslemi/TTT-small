    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    # Proportional adjustment
    if s > target:
        # Need to decrease
        # Scale down values that are > 0
        pos = h > 1e-10
        h[pos] *= (1.0 - (s - target) / h[pos].sum()) if h[pos].sum() > 1e-10 else 1.0
    else:

            h[mid] += adj
        else:

    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    corr = get_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * 2.0 / len(h)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        s2 = h.sum()
        if s2 < 1e-14:

        h *= (1.0 + diff / s2)

        if nmid > 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5_val(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * 2.0 / len(h)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        mid = (h > 1e-8) & (h < 1.0 - 1e-8)
        nm = mid.sum()
        if nm > 0:
            h[mid] += diff2 / nm
    return np.clip(h, 0.0, 1.0)

def lbfgs_opt(h0, n, t_end, T=500, mu_sum=5000.0, maxiter=2000, extra_obj=None):
    def obj(x):
        v = smooth_c5_val(x, T) + mu_sum * (x.sum() - n / 2.0) ** 2
        if extra_obj is not None:
            v += extra_obj(x)
        return v
    try:
        res = minimize(obj, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h0.copy()

def slsqp_opt(h0, n, t_end, T=500, maxiter=1000):
    def obj(x):
        return smooth_c5_val(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h0.copy()

def active_set_refine(h, n, t_end, n_rounds=30):
    best = h.copy()
    best_c5 = compute_c5(best)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        corr = get_corr(best)
        order = np.argsort(corr)[::-1]
        ntop = min(3 + r, 25)
        top = order[:ntop]
        mu = 3.0 + 5.0 * r

        def extra_obj(x):
            c = get_corr(x)
            return mu * np.var(c[top]) * 2.0 / n
        new_h = lbfgs_opt(best, n, t_end, T=200, mu_sum=5000, maxiter=600, extra_obj=extra_obj)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best = new_h
            best_c5 = new_c5
        else:
            top2 = order[:max(2, ntop // 2)]
            def extra_obj2(x):
                c = get_corr(x)
                return mu * 0.5 * np.var(c[top2]) * 2.0 / n
            new_h = lbfgs_opt(best, n, t_end, T=200, mu_sum=5000, maxiter=400, extra_obj=extra_obj2)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best = new_h
                best_c5 = new_c5
    return best, best_c5

def coord_sweep(h, n, t_end, max_sweeps=60):
    best = h.copy()
    best_c5 = compute_c5(best)
    L_all = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            best[i] = 0.0
            corr_base = get_corr(best)
            I1 = (i - L_all >= 0) & (i - L_all < n)
            I2 = (i + L_all >= 0) & (i + L_all < n)
            coefs = np.zeros(2 * n - 1)
            coefs[I1] = (1.0 - best[i - L_all[I1]])
            coefs[I2] -= best[i + L_all[I2]]

                f2 = np.max(corr_base + m2 * coefs)

                else:
                    lo = m1
            new_val = (lo + hi) / 2.0

                h_try[i] = new_val

                c5_try = compute_c5(h_try)

                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best, best_c5

def basin_hop(h_start, n, t_end, n_iter=40, scale=0.06):
    best = h_start.copy()
    best_c5 = compute_c5(best)
    work = best.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(1, n_iter - 1))
        mod = it % 6
        if mod == 0:
            pert = scale * cool * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 12)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.7 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = scale * 1.5 * cool * np.random.randn(mask.sum())
        elif mod == 3:
            blk = max(2, n // 10)
            b = np.random.randint(0, max(1, n // blk - 1))
            pert = np.zeros(n)
            pert[b*blk:(b+1)*blk] = scale * cool * np.random.randn(blk)
        elif mod == 4:
            pert = np.zeros(n)
            for _ in range(3):
                a, b = np.random.choice(n, 2, replace=False)
                d = (0.05 + 0.1 * cool) * (best[b] - best[a])
                pert[a] += d
                pert[b] -= d
        else:
            pert = scale * 0.4 * cool * np.random.randn(n)
        h_pert = project_h(work + pert)
        T_l = 200 + 3000 * cool
        hc = lbfgs_opt(h_pert, n, t_end, T=T_l, mu_sum=5000, maxiter=500)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-max(0, c5v - work_c5) / temp):
            work = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best = hc
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
