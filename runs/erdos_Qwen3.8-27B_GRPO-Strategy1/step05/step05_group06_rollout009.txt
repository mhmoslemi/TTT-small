    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    # Iterative water-filling
    for _ in range(10):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
            if len(idx) == 0: break
            step = min(diff / len(idx), (1.0 - h[idx]).min())
            if step <= 0: break
            h[idx] += step
            diff -= step * len(idx)
        else:
            idx = np.where(h > 1e-15)[0]
            if len(idx) == 0: break
            step = min(-diff / len(idx), h[idx].min())
            if step <= 0: break
            h[idx] -= step
            diff += step * len(idx)
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, T, lam):
    """Entropic smoothing of max(corr)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)  # for numerical stability
    exps = np.exp(Tc)
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    return val

def obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = int(lag_arr[idx])
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lg > 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] -= wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] -= wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

    x0 = project_h(h0).copy()
    def f(x): return obj_smooth(x, T, lam)

    try:

                elif lg > 0:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_h(h):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = len(h) / 2.0
    d = tgt - h.sum()
    for _ in range(20):
        if abs(d) < 1e-15: break
        if d > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
            if len(idx) == 0: break
            s = min(d / len(idx), (1.0 - h[idx]).min())
            if s <= 0: break
            h[idx] += s; d -= s * len(idx)
        else:
            idx = np.where(h > 1e-15)[0]
            if len(idx) == 0: break
            s = min(-d / len(idx), h[idx].min())
            if s <= 0: break
            h[idx] -= s; d += s * len(idx)
    return h

def _obj_func(x, T, lam):
    dx = 2.0 / len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    tc = T * (c - mc) - T * mc  # subtract for stability... wait
    # Actually: tc = T*(c - max), then shift
    tc = T * (c - c.max())
    tc -= tc.max()
    e = np.exp(tc)
    sm = c.max() + np.log(e.sum()) / T
    return dx * sm + lam * (x.sum() - len(x) / 2.0) ** 2

def _grad_func(x, T, lam):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    tc = T * (c - mc)
    tc -= tc.max()
    e = np.exp(tc)
    w = e / e.sum()
    sm = mc + np.log(e.sum()) / T
    val = dx * sm + lam * (x.sum() - n / 2.0) ** 2
    g = np.zeros(n)
    mf = 2 * n - 1
    la = np.arange(mf) - n + 1
    for idx in range(mf):
        lg = int(la[idx])
        wi = w[idx]
        if wi < 1e-14: continue
        if lg == 0:
            g += wi * (1.0 - 2.0 * x)
        elif lg > 0:
            g[lg:] += wi * (1.0 - x[:n-lg])
            g[:n-lg] -= wi * x[lg:]
        else:
            p = -lg
            g[:n-p] += wi * (1.0 - x[p:])
            g[p:] -= wi * x[:n-p]
    return val, dx * g + 2.0 * lam * (x.sum() - n / 2.0)

def lbfgs_run(h0, T, lam, maxiter, t_end):
    x0 = project_h(h0).copy()
    try:
        r = minimize(_obj_func, x0, jac=_grad_func, args=(T, lam),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * len(x0),
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-11})
        if time.time() > t_end: return x0
        return project_h(r.x)
    except Exception:
        return x0

def _f1d(x, cb, lin, quad):
    return np.max(cb + x * lin + x * x * quad)

def coord_desc(h, t_end, sweeps=200):
    n = len(h)
    bh = h.copy()
    bc = compute_c5(bh)
    mf = 2 * n - 1
    la = np.arange(mf) - n + 1
    for _ in range(sweeps):
        if time.time() > t_end: break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end: break
            h0 = bh.copy(); h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin = np.zeros(mf); quad = np.zeros(mf)
            for idx in range(mf):
                lg = int(la[idx])
                if lg == 0:
                    lin[idx] = 1.0; quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg: lin[idx] += 1.0 - bh[i - lg]
                    if i + lg < n: lin[idx] -= bh[i + lg]
                else:
                    p = -lg
                    if i + p < n: lin[idx] += 1.0 - bh[i + p]
                    if i >= p: lin[idx] -= bh[i - p]
            r = minimize_scalar(_f1d, args=(cb, lin, quad), bounds=(0, 1),
                                method='bounded', options={'xatol': 1e-12})
            if abs(r.x - bh[i]) < 1e-13: continue
            ht = bh.copy(); ht[i] = r.x; ht = project_h(ht)
            ct = compute_c5(ht)
            if ct < bc - 1e-14:
                bc, bh, imp = ct, ht, True
        if not imp: break
    return bh, bc

def downsample(h, f):
    n = len(h); m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1)

def upsample(h, f, n):
    return np.interp(np.linspace(0, len(h)-1, n), np.arange(len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    n = len(bh)

    bc = compute_c5(bh)

    pe = t0 + budget_s * 0.10

    # Phase 2: Multi-resolution
    pe = t0 + budget_s * 0.25

        hc = project_h(downsample(bh, f))
        for T in [500, 5000]:
            if time.time() > pe: break

        hu = project_h(upsample(hc, f, n))
        hu = lbfgs_run(hu, 2000, 500, 1000, pe)
        c = compute_c5(hu)
        if c < bc: bc, bh = c, hu
    
    # Phase 3: Coord descent
    pe = t0 + budget_s * 0.45
    bh, bc = coord_desc(bh, pe, 300)
    
    # Phase 4: High-T L-BFGS
    pe = t0 + budget_s * 0.65
    for T in [5000, 20000, 100000]:
        if time.time() > pe: break
        h = lbfgs_run(bh, T, 500, 3000, pe)
        c = compute_c5(h)
        if c < bc: bc, bh = c, h
    
    # Phase 5: Basin hopping
    pe = t0 + budget_s * 0.90
    wh, wc = bh.copy(), bc
    for it in range(60):
        if time.time() > pe: break
        cool = max(0.0, 1.0 - it / 50.0)
        sc = 0.005 + 0.04 * cool
        pt = it % 4
        if pt == 0:
            hp = wh + sc * np.random.randn(n)
        elif pt == 1:
            hp = np.roll(wh, np.random.randint(1, 4)) + sc * 0.5 * np.random.randn(n)
        elif pt == 2:
            H = np.fft.rfft(wh)
            H *= (1.0 + sc * np.random.randn(1, len(H)))
            hp = np.fft.irfft(H, n=n)
        else:
            hp = wh[::-1].copy() + sc * 0.5 * np.random.randn(n)
        hp = project_h(hp)
        h = lbfgs_run(hp, 2000 + 5000 * cool, 300, 1000, pe)
        c = compute_c5(h)
        tp = max(0.0003 * cool + 0.00005, 1e-10)
        if c < wc or np.random.rand() < np.exp(-min((c - wc) / tp, 100)):
            wh, wc = h, c
            if c < bc: bc, bh = c, h
    
    # Phase 6: Final polish
    if time.time() < t0 + budget_s * 0.96:
        bh, bc = coord_desc(bh, t0 + budget_s * 0.96, 200)
    if time.time() < t0 + budget_s * 0.99:
        h = lbfgs_run(bh, 200000, 500, 2000, t0 + budget_s * 0.99)
        c = compute_c5(h)
        if c < bc: bc, bh = c, h
    
    bh = project_h(bh)
    bc = compute_c5(bh)
    return bh, bc, n

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_h(h):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = len(h) / 2.0
    d = tgt - h.sum()
    for _ in range(50):
        if abs(d) < 1e-16:
            break
        if d > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) == 0:
            break
        s = min(abs(d) / len(idx),
                (1.0 - h[idx]).min() if d > 0 else h[idx].min())
        if s <= 0:
            break
        h[idx] += s * np.sign(d)
        d -= s * np.sign(d) * len(idx)
    return h

def _obj_func(x, T, lam):
    dx = 2.0 / len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    tc = T * (c - mc) - T * mc
    e = np.exp(tc)
    sm = mc + np.log(e.sum()) / T
    return dx * sm + lam * (x.sum() - len(x) / 2.0) ** 2

def _grad_func(x, T, lam):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    tc = T * (c - mc) - T * mc
    e = np.exp(tc)
    w = e / e.sum()
    sm = mc + np.log(e.sum()) / T
    val = dx * sm + lam * (x.sum() - n / 2.0) ** 2
    g = np.zeros(n)
    mf = 2 * n - 1
    la = np.arange(mf) - n + 1
    for idx in range(mf):
        lg = int(la[idx])
        wi = w[idx]
        if wi < 1e-14:
            continue
        if lg == 0:
            g += wi * (1.0 - 2.0 * x)
        elif lg > 0:
            g[lg:] += wi * (1.0 - x[:n - lg])
            g[:n - lg] -= wi * x[lg:]
        else:
            p = -lg
            g[:n - p] += wi * (1.0 - x[p:])
            g[p:] -= wi * x[:n - p]
    return val, dx * g + 2.0 * lam * (x.sum() - n / 2.0)

def lbfgs_run(h0, T, lam, maxiter, t_end):
    x0 = project_h(h0).copy()
    try:
        r = minimize(_obj_func, x0, jac=_grad_func, args=(T, lam),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * len(x0),
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-11})
        if time.time() > t_end:
            return x0
        return project_h(r.x)
    except Exception:
        return x0

def _f1d(x, cb, lin, quad):
    return np.max(cb + x * lin + x * x * quad)

def coord_desc(h, t_end, sweeps=200):
    n = len(h)
    bh = h.copy()
    bc = compute_c5(bh)
    mf = 2 * n - 1
    la = np.arange(mf) - n + 1
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                break
            h0 = bh.copy()
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin = np.zeros(mf)
            quad = np.zeros(mf)
            for idx in range(mf):
                lg = int(la[idx])
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += 1.0 - bh[i - lg]
                    if i + lg < n:
                        lin[idx] -= bh[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += 1.0 - bh[i + p]
                    if i >= p:
                        lin[idx] -= bh[i - p]
            r = minimize_scalar(_f1d, args=(cb, lin, quad), bounds=(0, 1),
                                method='bounded', options={'xatol': 1e-12})
            if abs(r.x - bh[i]) < 1e-13:
                continue
            ht = bh.copy()
            ht[i] = r.x
            ht = project_h(ht)
            ct = compute_c5(ht)
            if ct < bc - 1e-14:
                bc, bh, imp = ct, ht, True
        if not imp:
            break
    return bh, bc

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1)

def _upsample(h, n):
    x_old = np.arange(len(h), dtype=float)
    x_new = np.linspace(0, len(h) - 1, n)
    return np.interp(x_new, x_old, h)

def _structured_perturb(wh, n, sc, it):
    pt = it % 5
    if pt == 0:
        return wh + sc * np.random.randn(n)
    elif pt == 1:
        return np.roll(wh, np.random.randint(1, 5)) + sc * 0.4 * np.random.randn(n)
    elif pt == 2:
        H = np.fft.rfft(wh)
        H *= (1.0 + sc * np.random.randn(1, len(H)))
        return np.fft.irfft(H, n=n)
    elif pt == 3:
        return wh[::-1].copy() + sc * 0.4 * np.random.randn(n)
    else:
        k = np.random.randint(2, 8)
        return wh + sc * np.random.randn(n) * np.sin(2 * np.pi * k * np.arange(n) / n)
