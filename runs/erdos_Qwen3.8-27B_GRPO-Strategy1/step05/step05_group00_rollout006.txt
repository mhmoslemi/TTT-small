import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 1
    while L < 2 * N:
        L *= 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * G, L)[:2 * N - 1]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv
    pp = pos - sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    hm = np.where(vm, h[np.clip(pm, 0, N - 1)], 0.0)
    hp = np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binarize_fixed(h, N):
    nh = N // 2
    h = h.astype(float)
    if np.all((h == 0) | (h == 1)) and int(round(h.sum())) == nh:
        return h
    order = np.argsort(-h)
    out = np.zeros(N)
    out[order[:nh]] = 1.0
    return out

def full_binary_search(h, N, deadline, max_sweeps=1000, stuck_limit=80, top_k=3):
    h = binarize_fixed(h, N)
    best_c = float(np.max(fft_corr(h)))
    stuck = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            stuck = 0
        else:
            stuck += 1
            if stuck > stuck_limit:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        base_full = c[None, :] + D[ones, :]
        target_full = D[zeros, :]
        # Use top-k shifts for more robust criterion
        top_idx = np.argpartition(-c, top_k)[:top_k]
        base_top = base_full[:, top_idx]
        target_top = target_full[:, top_idx]
        best_val = P + 1.0
        best_pq = None
        chunk = 40
        for i in range(0, len(ones), chunk):
            b = base_top[i:i + chunk, None, :]
            t = target_top[None, :, :]
            nc = b - t
            nm_max = nc.max(axis=2)
            nm_pen = nm_max + 0.1 * (nc.sum(axis=2) / top_k)
            mi = np.unravel_index(nm_pen.argmin(), nm_pen.shape)
            val = float(nm_max[mi])
            if val < best_val:
                best_val = val
                best_pq = (int(ones[i + mi[0]]), int(zeros[mi[1]]))
        if best_pq is None or best_val >= P - 1e-12:
            stuck += 1
            if stuck > stuck_limit:
                break
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return h, best_c
