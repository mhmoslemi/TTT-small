import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_val(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    if T > 1e7:
        return m
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=400, ftol=1e-15):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if T > 1e7:
            return m
        return m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(res.x)
    except Exception:
        return h

def subgrad_momentum(h, n, t_end, n_iter=80, beta=0.5):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / n_iter
        T = 300 + 10000 * cool
        alpha = 0.002 * cool + 0.0004
        corr = get_corr(best_h)
        m = np.max(corr)
        exps = np.exp(T * (corr - m))
        sw = exps.sum()
        if sw < 1e-300:
            break
        weights = exps / sw
        n_topk = min(60, len(corr))
        topk_idx = np.argsort(corr)[-n_topk:]
        shifts = (topk_idx - (n - 1)).astype(float)
        i_idx = np.arange(n)[:, None]
        s_idx = shifts[None, :]
        jp = i_idx + s_idx
        jm = i_idx - s_idx
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp.astype(int), 0, n - 1)
        jmc = np.clip(jm.astype(int), 0, n - 1)
        grad_k = (1.0 - best_h[jmc]) * v2.astype(float) - best_h[jpc] * v1.astype(float)
        grad = (weights[topk_idx][:, None] * grad_k).sum(axis=1) * dx
        v = beta * v + (1 - beta) * grad
        h_new = project_h(best_h - alpha * v)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def active_cd(h, n, t_end, max_sweeps=80, n_active=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        top_idx = np.argsort(corr)[-n_active:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_active)
            for t in range(n_active):
                k = top_shifts[t]
                c = 0.0
                jp = i - k
                jm = i + k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx].copy()
            c0 = best_h[i]
            def obj_t(t):
                return np.max(base + (t - c0) * coefs)
            res = minimize_scalar(obj_t, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            t_new = res.x
            if abs(t_new - c0) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def targeted_hop(h, n, t_end, n_iter=25, scale=0.06):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        corr = get_corr(work_h)
        top_idx = np.argsort(corr)[-8:]
        shifts = top_idx - (n - 1)
        pert = np.zeros(n)
        for k in shifts:
            if k >= 0 and k < n:
                overlap = work_h[k:n] - 0.5
                pert[:n-k] += overlap * 0.4
                pert[k:n] += (-(work_h[:n-k] - 0.5)) * 0.4
            elif k < 0 and -k < n:
                ak = -k
                overlap = work_h[:n-ak] - 0.5
                pert[ak:n] += overlap * 0.4
                pert[:n-ak] += (-(work_h[ak:n] - 0.5)) * 0.4
        pert *= scale * 0.35 * cool
        pert += scale * cool * 0.25 * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = 3000 * cool + 300
        hc = slsqp_opt(h_pert, T, n, 250, 1e-14)
        c5v = compute_c5(hc)
        temp = 0.0007 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def equalize_top(h, n, t_end, n_eq=20, n_iters=50):
    """Try to equalize the top correlations via dual variable updates."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        top_idx = np.argsort(corr)[-n_eq:]
        shifts = top_idx - (n - 1)
        # Compute gradient of max over top shifts (average)
        i_idx = np.arange(n)[:, None]
        s_idx = shifts[None, :]
        jp = i_idx + s_idx
        jm = i_idx - s_idx
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n - 1)
        jmc = np.clip(jm, 0, n - 1)
        grad = ((1.0 - best_h[jmc]) * v2.astype(float) - best_h[jpc] * v1.astype(float)).mean(axis=1) * dx
        # Backtracking line search
        alpha = 0.01
        for _ in range(8):
            h_new = project_h(best_h - alpha * grad)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                break
            alpha *= 0.5
        if alpha < 1e-6:
            break
    return best_h, best_c5
