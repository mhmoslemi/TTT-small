def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.25:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-8:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n-k] += (work_h[k:n] - 0.5)
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n+k] - 0.5)
            pert *= scale * 0.5 * cool
        elif r < 0.5:
            f = np.random.randint(2, 16)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.8 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.75:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.2 * cool * np.random.randn(max(int(mask.sum()), 1))
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp_opt(h_pert, T, n, 400, flat_w=0.02 * cool)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0002
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_at_n(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # SLSQP warmup with increasing T
    if t_end - time.time() > 50:
        for T in [50, 200, 1000, 10000, 50000]:
            if t_end - time.time() < 20:
                break
            hc = slsqp_opt(best_h, T, n, 600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Subgradient + SLSQP
    if t_end - time.time() > 60:
        mom_v = np.zeros(n)
        for it in range(50):
            if t_end - time.time() < 15:
                break
            cool = 1.0 - it / 50.0
            alpha = 0.003 + 0.006 * cool
            n_topk = 8 + int(15 * cool)
            hs, g = subgrad_top(best_h, alpha, n_topk, mom_v, beta=0.4)
            mom_v = g
            T = 200 + 8000 * cool
            hc = slsqp_opt(hs, T, n, 400, flat_w=0.05 * cool)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Basin hopping
    if t_end - time.time() > 80:
        bh_h, bh_c5 = basin_hop(best_h, n, t_end - (t_end - time.time()) * 0.35,
                                50, 0.1, 500, 8000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
            best_h = project_h(best_h)
    # Aggressive coordinate descent
    if t_end - time.time() > 40:
        best_h, best_c5 = coord_descent(best_h, n, t_end - (t_end - time.time()) * 0.3,
                                        250, 50)
    # More basin hopping at smaller scale
    if t_end - time.time() > 40:
        bh_h, bh_c5 = basin_hop(best_h, n, t_end - (t_end - time.time()) * 0.4,
                                30, 0.03, 2000, 10000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Final coordinate descent (most important for polishing)
    if t_end - time.time() > 20:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 5, 300, 60)
    # Final SLSQP polish
    if t_end - time.time() > 10:
        hc = slsqp_opt(project_h(best_h), 2000000, n, 1000, flat_w=0.01)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Final coord descent
    if t_end - time.time() > 8:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, 200, 60)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 200
        init_h = np.zeros(n0)
        init_h[:n0 // 2] = 1.0
    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)
    # Main optimization at parent n
    t_phase1_end = t0 + budget_s * 0.50
    best_h, best_c5 = optimize_at_n(best_h, n, t_phase1_end)
    # Try n=192
    if t_end - time.time() > budget_s * 0.30 and n != 192:
        n2 = 192
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        t_phase2_end = t0 + budget_s * 0.78
        h2_opt, c5_2 = optimize_at_n(h2, n2, t_phase2_end)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt
            n = n2
    # Try n=216
    if t_end - time.time() > budget_s * 0.25 and n != 216:
        n3 = 216
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = project_h(h3)
        t_phase3_end = t0 + budget_s * 0.93
        h3_opt, c5_3 = optimize_at_n(h3, n3, t_phase3_end)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3_opt
            n = n3
    # Final polish
    if t_end - time.time() > 15:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 3, 400, 60)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
