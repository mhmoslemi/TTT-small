    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50, 50)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def _sum_con(x, n):
    return x.sum() - n / 2.0

    return float(np.max(base + val * cf))

                     bounds=[(0.,1.)]*n,

                     args=(T,n),

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        s = h.sum()
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _so(x, T, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(T * (c - m), -50, 50)
    return (m + np.log(np.sum(np.exp(s))) / T) * dx

def _sc(x, n):
    return x.sum() - n / 2.0

def _co(v, b, cf):
    return float(np.max(b + v * cf))

def slsqp(h0, T, n, mi=300):
    try:
        r = minimize(_so, h0, method='SLSQP', bounds=[(0., 1.)] * n,
                     constraints=[{'type': 'eq', 'fun': _sc, 'args': (n,)}],
                     args=(T, n), options={'maxiter': mi, 'ftol': 1e-15, 'maxls': 50})
        return project_h(np.clip(r.x, 0, 1), n)
    except Exception:
        return h0

def c1d(h, n, i):
    old = h[i]
    h[i] = 0.
    b = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    cf = np.zeros(2 * n - 1)
    lg = np.arange(-(n - 1), n)
    j1, j2 = i + lg, i - lg
    v1, v2 = (j1 >= 0) & (j1 < n), (j2 >= 0) & (j2 < n)
    cf[v1] += 1.0 - h[j1[v1]]
    cf[v2] += -h[j2[v2]]
    r = minimize_scalar(_co, args=(b, cf), bounds=(0., 1.),
                        method='bounded', options={'xatol': 1e-12, 'maxiter': 50})
    return float(r.x)

def cd(h, n, t_end, ms=200):
    bh, bc = h.copy(), compute_c5(h)
    for _ in range(ms):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return bh, bc
            vn = c1d(bh, n, i)
            if abs(vn - bh[i]) < 1e-12:
                continue
            ht = bh.copy()
            ht[i] = vn
            ht = project_h(ht, n)
            ct = compute_c5(ht)
            if ct < bc - 1e-14:
                bc, bh, imp = ct, ht, True
        if not imp:
            break
    return bh, bc

def cd_anneal(h, n, t_end, ms=100):
    cur = h.copy()
    cc = compute_c5(cur)
    bc, bcur = cc, cur.copy()
    for sw in range(ms):
        if time.time() > t_end:
            break
        T = 3e-5 * (0.3) ** (sw / max(1, ms))
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return bcur, bc
            vn = c1d(cur, n, i)
            if abs(vn - cur[i]) < 1e-12:
                continue
            ht = cur.copy()
            ht[i] = vn
            ht = project_h(ht, n)
            ct = compute_c5(ht)
            if ct < cc or (T > 1e-9 and np.random.rand() < np.exp(-(ct - cc) / T)):
                cur, cc = ht, ct
            if ct < bc:
                bc, bcur = ct, ht
    return bcur, bc

def finit(n, rs):
    h = np.zeros(n)
    for _ in range(3):
        f = rs.randint(3, 40)
        ph = rs.uniform(0, 2 * np.pi)
        a = rs.uniform(0.2, 0.5)
        h += a * np.cos(2 * np.pi * f * np.arange(n) / n + ph)
    return project_h(0.5 + 0.4 * h, n)

def tpert(h, n, amp, rs):
    c = np.correlate(h, 1.0 - h, mode='full')
    tk = np.argsort(c)[-5:]
    ki = tk[rs.randint(5)]
    lag = ki - (n - 1)
    lo = max(0, -lag)
    hi_ = min(n, n - lag) if lag > 0 else n
    if hi_ <= lo:
        idx = rs.randint(n)
    else:
        idx = lo + rs.randint(hi_ - lo)
    h2 = h.copy()
    h2[idx] += amp * rs.choice([-1, 1])
    return project_h(h2, n)

    rs = np.random.RandomState(seed)
    t0 = time.time()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        s = h.sum()
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _so(x, T, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(T * (c - m), -50, 50)
    return (m + np.log(np.sum(np.exp(s))) / T) * dx
