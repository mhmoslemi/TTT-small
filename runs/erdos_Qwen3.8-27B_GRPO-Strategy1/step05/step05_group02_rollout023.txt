import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def soft_jac(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(np.clip(T * (corr - m), -50.0, 50.0))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-15:
            continue
        k = j - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            p = np.arange(lo1, hi1)
            grad[p] += wj * (1.0 - h[p - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            p = np.arange(lo2, hi2)
            grad[p] -= wj * h[p + k]
    return dx * grad

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_step(h, T, n, maxiter=1000):
    try:
        res = minimize(soft_obj, h, args=(T,), jac=lambda x, T2: soft_jac(x, T2),
                       method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def eval_1d(hi, corr_base, coefs, dx):
    return float(np.max(corr_base + hi * coefs) * dx)

def coord_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    dx = 2.0 / n
    try:
        res = minimize_scalar(eval_1d, args=(corr_base, coefs, dx),
                              bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def make_pert(n, cool, rng):
    mod = rng.randint(0, 6)
    if mod == 0:
        bs = rng.randint(4, max(5, n // 8))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.04 + 0.12 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 15))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(4, max(5, n // 8))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.08 + 0.25 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    elif mod == 4:
        # Swap perturbation
        i, j = rng.choice(n, 2, replace=False)
        p = np.zeros(n)
        p[i] = 0.3 * cool
        p[j] = -0.3 * cool
    else:
        # Block shift

        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs - 1] = 0.1 * cool
        p[st + bs - 1:st + bs] = -0.1 * cool
    return p

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def soft_jac(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(np.clip(T * (corr - m), -50.0, 50.0))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-15:
            continue
        k = j - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            p = np.arange(lo1, hi1)
            grad[p] += wj * (1.0 - h[p - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            p = np.arange(lo2, hi2)
            grad[p] -= wj * h[p + k]
    return dx * grad

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_step(h, T, n, maxiter=800):
    try:
        res = minimize(soft_obj, h, args=(T,), jac=lambda x, T2: soft_jac(x, T2),
                       method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def eval_1d(hi, corr_base, coefs, dx):
    return float(np.max(corr_base + hi * coefs) * dx)

def coord_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    dx = 2.0 / n
    try:
        res = minimize_scalar(eval_1d, args=(corr_base, coefs, dx),
                              bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=150):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def make_pert(n, cool, rng):

    elif mod == 1:

    elif mod == 2:
        np_ = rng.randint(4, max(5, n // 8))

    elif mod == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    elif mod == 4:
        i, j = rng.choice(n, 2, replace=False)
        p = np.zeros(n)

    else:
        bs = rng.randint(5, max(6, n // 6))

        p = np.zeros(n)
        p[st:st + bs - 1] = 0.1 * cool
        p[st + bs - 1:st + bs] = -0.1 * cool
    return p

def swap_polish(h, n, t_end, n_tries=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i, j = np.random.choice(n, 2, replace=False)
        if abs(best_h[i] - best_h[j]) < 0.01:
            continue
        h_try = best_h.copy()
        h_try[i], h_try[j] = best_h[j], best_h[i]
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5
