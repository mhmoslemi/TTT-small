    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):

def compute_D(h, N):

    D[:, N-1] = 0.0  # Fix center lag

def find_best_swap(corr, D, ones, zeros):

    k_star = int(np.argmax(corr))
    Dp_at_max = D[ones, k_star]
    Dq_at_max = D[zeros, k_star]
    
    # Top candidates: p with smallest D[p,k*], q with largest D[q,k*]
    p_candidates = ones[np.argsort(Dp_at_max)[:20]]
    q_candidates = zeros[np.argsort(-Dq_at_max)[:20]]

    for p in p_candidates:

        for q in q_candidates:
            new_corr = base - D[q]
            P_new = float(new_corr.max())
            if P_new < best_P:
                best_P = P_new
                best_pq = (p, q)
    return best_pq, best_P

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = (np.arange(2 * N - 1)[None, :] - (N - 1))
    pos = np.arange(N)[:, None]
    pm = pos - shifts
    pp = pos + shifts
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    D = -(1.0 - h[np.clip(pm, 0, N - 1)]) * vm.astype(float) + h[np.clip(pp, 0, N - 1)] * vp.astype(float)
    D[:, N - 1] = 0.0
    return D

def ensure_binary(h, N):
    h = h.copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h > 0.5)[0]
            order = np.argsort(h[idx])
            h[idx[order[nh - (n1 - nh):]]] = 0
        else:
            idx = np.where(h <= 0.5)[0]
            order = np.argsort(-h[idx])
            h[idx[order[:nh - n1]]] = 1
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return h

def binary_search_full(h, N, deadline, max_iters=5000, peak_candidates=30, anneal_T=0.0, full_search_frac=0.15):
    h = ensure_binary(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    nh = N // 2
    T = anneal_T
    n_full = max(1, int(max_iters * full_search_frac))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        D = compute_D(h, N)
        k_star = int(np.argmax(corr))
        Dp_at = D[ones, k_star]
        Dq_at = D[zeros, k_star]
        if it < n_full:
            p_cands = ones
            q_cands = zeros
        else:
            p_cands = ones[np.argsort(Dp_at)[:peak_candidates]]
            q_cands = zeros[np.argsort(-Dq_at)[:peak_candidates]]
        best_new_P = P
        best_pq = None
        for p in p_cands:
            base = corr + D[p]
            for q in q_cands:
                val = float((base - D[q]).max())
                if val < best_new_P:
                    best_new_P = val
                    best_pq = (int(p), int(q))
        if best_pq is None:
            if anneal_T > 0:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                if np.random.random() < np.exp(-1.0 / max(T, 1e-6)):
                    h[p], h[q] = 0.0, 1.0
                T *= 0.998
            else:
                break
        else:
            delta = best_new_P - P
            accept = delta < 0
            if not accept and anneal_T > 0:
                accept = np.random.random() < np.exp(-delta / max(T, 1e-6))
            if accept:
                h[best_pq[0]] = 0.0
                h[best_pq[1]] = 1.0
                T *= 0.999
            else:
                if anneal_T == 0:
                    break
                T *= 0.99
    return best_h, best_c

def nesterov_opt(h, N, deadline, k_peaks=50, lr0=0.03):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k_peaks)[-k_peaks:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_peaks
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.0001)
        if it % 500 == 0 and lr < 0.001:
            break
    return best_h, best_pk * dx

        else:

        h_bt = ensure_binary(h_interp, N_try)
        sub_dl = min(deadline - 20, time.time() + 70)
        h_rt, _ = binary_search_full(h_bt, N_try, sub_dl, max_iters=2000, peak_candidates=25)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)
