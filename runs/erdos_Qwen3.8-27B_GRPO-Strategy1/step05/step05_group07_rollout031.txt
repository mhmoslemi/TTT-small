        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 - d:hi2 - d]
    return g / na * (2.0 / n)

def grad_search(h, n, t_end, steps=3000):
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = 0.05
    for s in range(steps):
        if time.time() > t_end:
            break
        g = exact_max_grad(h, n)
        trial = proj_h(h - lr * g, n)
        c5t = compute_c5(trial)
        if c5t < best_c5 - 1e-15:
            h = trial
            best = trial.copy()
            best_c5 = c5t
            lr = min(lr * 1.3, 0.5)
        else:
            lr *= 0.6
            if lr < 1e-9:
                lr = 1e-6
    return best, best_c5

def lbfgs_soft(h, T, n, iters=400):
    def f_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sm = np.log(np.sum(np.exp(T * (c - m)))) / T
        return (m + sm) * 2.0 / n
    def f_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        e = np.exp(T * (c - m))
        w = e / e.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            d = k - (n - 1)
            lo1, hi1 = max(0, -d), min(n, n - d)
            if lo1 < hi1:
                g[lo1:hi1] += wk * (1.0 - x[lo1 + d:hi1 + d])
            lo2, hi2 = max(0, d), min(n, n + d)
            if lo2 < hi2:
                g[lo2:hi2] -= wk * x[lo2 - d:hi2 - d]
        return g * (2.0 / n)
    try:
        r = minimize(h, f_obj, f_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12,
                              'maxls': 50})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=80):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        cc = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            coef = np.zeros(mf)
            d_arr = np.arange(mf) - (n - 1)
            lo1, hi1 = max(0, -d_arr).clip(0, n), min(n, n - d_arr)
            mask1 = (np.arange(n) >= max(0, -d_arr).clip(0, n)) & (np.arange(n) < n - d_arr)
            for k in range(mf):
                j2 = i + k - (n - 1)
                j1 = i - (k - (n - 1))
                if 0 <= j2 < n:
                    coef[k] += (1.0 - best[j2])
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            base = cc - best[i] * coef
            def f1d(v):
                return float(np.max(base + v * coef))
            r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-13})
            nv = float(r.x)
            if abs(nv - best[i]) < 1e-14:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                cc = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def bin_sa(h, n, t_end, max_iters=500000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    if len(ones) > n // 2:
        h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, n // 2 - len(ones) if n % 2 == 0 else n // 2 + 1 - len(ones), replace=False)] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(corr.max() * dx)
    best_h = h_b.copy()
    T = 0.02
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h_b[i] != h_b[j]:
            continue
        h_b[i], h_b[j] = h_b[j], h_b[i]
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(nc.max() * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        T = max(0.0003, T * 0.99997)
    return best_h, best_c5

def interp_to_n(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return proj_h(h_new, n_new)

def basin_hop(h, n, t_end, n_iters=150):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 3
        if mod == 0:
            pert = (0.03 + 0.1 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(2, 16)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.08 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        else:
            mask = np.random.random(n) < (0.15 + 0.3 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.2 * cool) * np.random.randn(mask.sum())
        hp = proj_h(work + pert, n)
        T_val = 500.0 + 3000.0 * cool
        ht = lbfgs_soft(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / (0.001 * cool + 1e-5)):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: quick binary SA to find better binary config
    h_bin, c5_bin = bin_sa(best_h, n, t0 + budget_s * 0.06)
    if c5_bin < best_c5 * 1.005:
        h_b = h_bin.astype(float)
        h_b = lbfgs_soft(h_b, 2000.0, n, iters=300)
        h_b = grad_search(h_b, n, t0 + budget_s * 0.08, steps=500)
        if compute_c5(h_b) < best_c5:
            best_h, best_c5 = h_b, compute_c5(h_b)

    # Phase 2: L-BFGS-B soft max with increasing T
    for T in [200, 1000, 5000, 20000]:
        if time.time() > t0 + budget_s * 0.14:
            break
        hn = lbfgs_soft(best_h, T, n, iters=500)
        c5n = compute_c5(hn)
        if c5n < best_c5:
            best_h, best_c5 = hn, c5n

    # Phase 3: Exact gradient search (minimax)
    best_h, best_c5 = grad_search(best_h, n, t0 + budget_s * 0.22, steps=5000)

    # Phase 4: Coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.32, sweeps=100)

    # Phase 5: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.52, n_iters=180)

    # Phase 6: More gradient + coord descent
    best_h, best_c5 = grad_search(best_h, n, t0 + budget_s * 0.60, steps=8000)
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.68, sweeps=100)

    # Phase 7: Try nearby n values
    for dn in [-16, 16, -32, 32, 64]:
        n_try = n + dn
        if n_try < 80 or n_try > 400:
            continue
        if time.time() > t0 + budget_s * 0.82:
            break
        try:
            ht = interp_to_n(best_h, n, n_try)
            ht = lbfgs_soft(ht, 5000.0, n_try, iters=300)
            ht, c5t = grad_search(ht, n_try, min(time.time() + 20, t0 + budget_s * 0.84), steps=2000)
            ht, c5t = coord_desc(ht, n_try, min(time.time() + 15, t0 + budget_s * 0.86), sweeps=40)
            if c5t < best_c5:
                best_h, best_c5, n = ht, c5t, n_try
        except Exception:
            pass

    # Phase 8: Final high-T polish
    if time.time() < t0 + budget_s * 0.93:
        hn = lbfgs_soft(best_h, 100000.0, len(best_h), iters=800)
        c5n = compute_c5(hn)
        if c5n < best_c5:
            best_h, best_c5 = hn, c5n
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = grad_search(best_h, len(best_h), t0 + budget_s * 0.97, steps=3000)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
