    m = np.max(corr)

            top = np.sort(corr)[-min(10, len(corr)):]

        return v

    except:

    dx = 2.0/n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break

        top_idx = np.argsort(corr)[-n_top:]

        improved = False

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return float(dx * (m + np.log(np.sum(np.exp(T * (corr - m)))) / T))

def slsqp_opt(h, T, n, maxiter=500, flat_pen=0.0):
    def obj(x):
        v = soft_c5(x, T)
        if flat_pen > 0.0:
            corr = np.correlate(x, 1.0 - x, mode='full')
            top = np.sort(corr)[-8:]
            v += flat_pen * np.std(top)
        return v
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=300, n_top=50, order='random'):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        if order == 'random':
            order_seq = np.random.permutation(n)
        else:
            order_seq = np.arange(n)
        improved = False
        for i in order_seq:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.empty(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, use_targeted=True):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if use_targeted and r < 0.35:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-6:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if 0 <= k < n:
                    pert[:n - k] += 0.5 - work_h[k:n]
                elif k < 0 and (-k) < n:
                    pert[-k:] += work_h[:n + k] - 0.5
            pert *= scale * 0.3 * cool
        elif r < 0.55:
            f = np.random.randint(1, 15)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.6 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.75:
            blen = np.random.randint(max(n // 15, 2), max(n // 4, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * cool * np.random.randn(blen)
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = 200 + 8000 * cool
        hc = slsqp_opt(h_pert, T, n, 300, 0.03 * cool)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def subgrad_pass(h, n, t_end, n_iter=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    mom_v = np.zeros(n)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        alpha = 0.001 + 0.004 * cool
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        n_topk = 8 + int(15 * cool)
        topk_idx = np.argsort(corr)[-n_topk:]
        shifts = topk_idx - (n - 1)
        i_idx = np.arange(n)[:, None]
        s_idx = shifts[None, :]
        jp = i_idx + s_idx
        jm = i_idx - s_idx
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n - 1)
        jmc = np.clip(jm, 0, n - 1)
        dx = 2.0 / n
        grad = ((1.0 - best_h[jpc]) * v1.astype(np.float64) - best_h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
        mom_v = 0.35 * mom_v + 0.65 * grad
        h_new = project_h(best_h - alpha * mom_v)
        T = 500 + 5000 * cool
        hc = slsqp_opt(h_new, T, n, 250, 0.04 * cool)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup (increasing T for sharper max)
    t_p1 = t0 + 0.08 * budget_s
    for T in [50, 200, 1000, 5000, 20000]:
        if time.time() > t_p1:
            break
        hc = slsqp_opt(best_h, T, n, 400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Subgradient + SLSQP
    t_p2 = t0 + 0.22 * budget_s
    best_h, best_c5 = subgrad_pass(best_h, n, t_p2, 25)

    # Phase 3: Main coordinate descent (bulk of time)
    t_p3 = t0 + 0.72 * budget_s
    best_h, best_c5 = coord_descent(best_h, n, t_p3, 400, 50, 'random')

    # Phase 4: Basin hopping to escape local minima
    t_p4 = t0 + 0.85 * budget_s
    bh_h, bh_c5 = basin_hop(best_h, n, t_p4, 20, 0.1, use_targeted=True)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = bh_h
        best_h, best_c5 = coord_descent(best_h, n, t_end - 0.12 * budget_s, 150, 50, 'sequential')

    # Phase 5: Final polish
    t_p5 = t_end - 0.05 * budget_s
    hc = slsqp_opt(best_h, 100000, n, 800, 0.0)
    c5v = compute_c5(hc)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc
    best_h, best_c5 = coord_descent(best_h, n, t_end - 2, 80, 60, 'sequential')

    # Final SLSQP squeeze
    if time.time() < t_end - 3:
        hc = slsqp_opt(best_h, 500000, n, 1000, 0.0)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
