import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    dx = 2.0 / len(h)
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    h = project_iter(h, target, max_iter=50)
    return h

def project_iter(h, target, max_iter=50):
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Water-filling: add/subtract uniformly from non-saturated coords
        if s < target:
            room = (1.0 - h)
            mask = room > 1e-14
            if mask.sum() == 0:
                break
            add = (target - s) / mask.sum()
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-14
            if mask.sum() == 0:
                break
            rem = (s - target) / mask.sum()
            h[mask] = np.maximum(h[mask] - rem, 0.0)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=2000, ftol=1e-15):

        return smooth_c5(x, T)

        return x.sum() - n / 2.0

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],

def coord_opt_single(h, n, i, niter=80):
    """Optimize h[i] given all others fixed, minimizing max correlation."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    # coeff[m] = change in corr[m] per unit change in h[i]
    m_vals = np.arange(-(n - 1), n, dtype=float)
    vm1 = i - m_vals  # positions j where h[j] is multiplied by (1-h[i+m])
    vm2 = i + m_vals  # positions j where (1-h[j]) is multiplied by h[i+m]
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(vm1[valid1].astype(int), 0, n-1)])
    coefs[valid2] += (-h[np.clip(vm2[valid2].astype(int), 0, n-1)])
    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
