def project_h(h, target_sum):

    s = h.sum()

    if s < 1e-14:

        h = h + room * (diff / tot)

        # Term 2: j=i-off, valid if 0<=i-off<n
        t2 = 0.0
        if 0 <= i-off < n:
            t2 = h[i-off] * (1.0 - h_i)
        A[idx] = corr_full[idx] - t1 - t2
        B[idx] = 0.0
        if 0 <= i+off < n:
            B[idx] += (1.0 - h[i+off])
        if 0 <= i-off < n:
            B[idx] -= h[i-off]
    # Minimize max(A + x*B) over x in [0,1]
    val0 = np.max(A)
    val1 = np.max(A + B)
    if val0 <= val1:
        return 0.0
    # Otherwise need to search
    # Use golden section or ternary search
    # Actually, min of convex PL on [0,1]: check all breakpoints
    # For efficiency, use minimize_scalar
    def obj(x):
        return np.max(A + x * B)
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return float(res.x)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h
    if s < 1e-14:
        return np.clip(target_sum / len(h), 0.0, 1.0) * np.ones(len(h))
    diff = target_sum - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return float(m + np.log(np.sum(np.exp(shifted))) / T)

def smooth_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    exp_s = np.exp(shifted)
    wsum = exp_s.sum()
    if wsum < 1e-15:
        return np.zeros(n)
    weights = exp_s / wsum
    offsets = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for idx in range(2*n-1):
        w = weights[idx]
        if w < 1e-13:
            continue
        off = int(offsets[idx])
        g = np.ones(n)
        jlo1 = max(0, -off)
        jhi1 = min(n, n - off)
        if jlo1 < jhi1:
            g[jlo1:jhi1] -= h[jlo1+off:jhi1+off]
        jlo2 = max(0, off)
        jhi2 = min(n, n + off)
        if jlo2 < jhi2:
            g[jlo2:jhi2] -= h[jlo2-off:jhi2-off]
        grad += w * g
    return grad

def slsqp_opt(h, T, n, maxiter=500):
    def fun(x):
        return smooth_obj(x, T, n)
    def jac(x):
        return smooth_grad(x, T, n)
    def con(x):
        return x.sum() - n/2.0
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter':maxiter,'ftol':1e-15})
        return project_h(np.clip(res.x, 0, 1), n/2.0)
    except Exception:
        return h

def coord_single(h, n, i):
    dx = 2.0/n
    corr_full = np.correlate(h, 1.0-h, mode='full')
    offsets = np.arange(-(n-1), n)
    hi = h[i]
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    for idx in range(2*n-1):
        off = int(offsets[idx])
        t1 = 0.0
        if 0 <= i+off < n:
            t1 = hi*(1.0-h[i+off])
        t2 = 0.0
        if 0 <= i-off < n:
            t2 = h[i-off]*(1.0-hi)
        A[idx] = corr_full[idx] - t1 - t2
        B[idx] = 0.0
        if 0 <= i+off < n:
            B[idx] += (1.0-h[i+off])
        if 0 <= i-off < n:
            B[idx] -= h[i-off]
    val0 = np.max(A)
    val1 = np.max(A+B)
    if val0 <= val1:
        return 0.0
    if val0 >= val1 and np.min(A+B) >= np.min(A):
        return 1.0
    def obj(x):
        return np.max(A + x*B)
    try:
        res = minimize_scalar(obj, bounds=(0.0,1.0), method='bounded',
                             options={'xatol':1e-10})
        return float(res.x)
    except Exception:
        return hi

def coord_descent(h, n, t_end, max_sweeps=200, rng=None):
    if rng is None:
        rng = np.random.RandomState(7)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = 0
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n/2.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp += 1
        if imp == 0:
            break
    return best_h, best_c5

def subgrad_h(h, n, t_end, n_steps=4000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0-best_h, mode='full')
    for step in range(n_steps):
        if time.time() > t_end:
            break
        if step % 50 == 0:
            corr = np.correlate(best_h, 1.0-best_h, mode='full')
        km = int(np.argmax(corr))
        off = km - (n-1)
        grad = np.ones(n)
        jlo1 = max(0, -off)
        jhi1 = min(n, n-off)
        if jlo1 < jhi1:
            grad[jlo1:jhi1] -= best_h[jlo1+off:jhi1+off]
        jlo2 = max(0, off)
        jhi2 = min(n, n+off)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= best_h[jlo2-off:jhi2-off]
        lr = 0.03/(1.0+0.005*step)
        for mult in [1.0, 0.3, 0.1]:
            h_new = project_h(best_h - lr*mult*grad, n/2.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                corr = np.correlate(best_h, 1.0-best_h, mode='full')
                break
    return best_h, best_c5

def make_pert(n, cool, rng):
    mod = rng.randint(0, 5)
    pert = np.zeros(n)
    if mod == 0:
        bs = rng.randint(4, max(5, n//7))
        st = rng.randint(0, n-bs)
        pert[st:st+bs] = (0.04+0.14*cool)*rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n//14))
        ph = rng.uniform(0, 2*np.pi)
        pert = (0.03+0.12*cool)*np.sin(2*np.pi*freq*np.arange(n)/n+ph)
    elif mod == 2:
        ns = rng.randint(3, max(4, n//9))
        idx = rng.choice(n, ns, replace=False)
        pert[idx] = (0.08+0.28*cool)*rng.standard_normal(ns)
    elif mod == 3:
        pert = (0.02+0.06*cool)*rng.standard_normal(n)
    else:
        half = n//2
        pert[:half] = 0.1*cool*rng.standard_normal(half)
        pert[half:] = -0.08*cool*rng.standard_normal(n-half)
    return pert

def basin_hop(h, n, t_end, n_iter=80, seed=99):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.05, 1.0-0.85*it/max(n_iter,1))
        pert = make_pert(n, cool, rng)
        h_pert = project_h(work_h+pert, n/2.0)
        T = 800+6000*cool
        h_opt = slsqp_opt(h_pert, T, n, maxiter=300)
        c5_opt = compute_c5(h_opt)
        temp = 0.002*cool+0.0001
        acc = c5_opt < work_c5 or (c5_opt < work_c5+0.002 and rng.rand()<np.exp(-(c5_opt-work_c5)/max(temp,1e-9)))
        if acc:
            work_h = h_opt
            work_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
