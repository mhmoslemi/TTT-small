import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def smooth_max(corr, T):
    m = np.max(corr)
    shifted = T * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / T

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-15:
        return h
    if s > target:
        # Need to reduce
        h = h * (target / s)
    else:
        # Need to increase - add to those with most room
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-15:
            h = h + room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Push h towards binary while maintaining sum"""
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target) < 0.5:
            # Find positions to flip
            # Sort by distance from 0.5
            d = np.abs(h - 0.5)
            # Positions closest to 0 or 1 are "settled"
            # We want exactly n/2 ones
            # Count current "ones" (h > 0.5)
            ones = (h > 0.5).sum()
            if ones == target:
                # Already have right count, just push to 0/1
                h = (h > 0.5).astype(float)
                break
            elif ones > target:
                # Need to turn off some ones
                idx = np.where(h > 0.5)[0]
                # Turn off the weakest ones (closest to 0.5)
                h_idx = h[idx]
                order = np.argsort(h_idx)  # weakest first

        # Scale to correct sum
        s = h.sum()
        if s > 0:

        h = np.clip(h, 0, 1)

def slsqp_smooth(h, T, n, maxiter=500):
    """SLSQP on smoothed max"""
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, T) * dx
    def obj_grad(x):
        # Gradient of smooth max
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        w = T * (corr - m)
        p = np.exp(w) / np.sum(np.exp(w))  # softmax weights
        gm = m + np.log(np.sum(np.exp(w))) / T
        # Gradient of corr[k] w.r.t. x[j]
        # corr[k] = sum_i x[i]*(1-x[i+k_shift])
        # This is complex to compute analytically, use finite difference
        return None
    try:

def pair_swap_delta(corr, h, a, b, n):
    """Compute change in corr for all lags when swapping h[a]:1->0 and h[b]:0->1"""
    # corr has length 2n-1, indexed by lag k = -(n-1) to n-1
    # corr[n-1+k] corresponds to lag k
    # Δcorr[k] = h[a+k] + h[a-k] - h[b+k] - h[b-k] (out of range = 0)
    k_range = np.arange(-(n-1), n)
    # h[a+k] for valid a+k
    idx1 = a + k_range
    v1 = (idx1 >= 0) & (idx1 < n)
    t1 = np.zeros(2*n-1)
    t1[v1] = h[idx1[v1]]
    # h[a-k]
    idx2 = a - k_range
    v2 = (idx2 >= 0) & (idx2 < n)
    t2 = np.zeros(2*n-1)
    t2[v2] = h[idx2[v2]]
    # h[b+k]
    idx3 = b + k_range
    v3 = (idx3 >= 0) & (idx3 < n)
    t3 = np.zeros(2*n-1)
    t3[v3] = h[idx3[v3]]
    # h[b-k]
    idx4 = b - k_range
    v4 = (idx4 >= 0) & (idx4 < n)
    t4 = np.zeros(2*n-1)
    t4[v4] = h[idx4[v4]]
    return t1 + t2 - t3 - t4

def sim_anneal_swap(h, n, t_start, t_end, rng, n_ones=None):
    """Simulated annealing with pair swaps on binary h"""
    if n_ones is None:
        n_ones = int(n / 2)
    h = (h > 0.5).astype(float)  # binarize
    # Fix sum
    s = h.sum()
    if s != n_ones:
        ones_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        if s > n_ones:
            to_remove = s - n_ones
            for i in rng.choice(ones_idx, to_remove, replace=False):
                h[i] = 0
        else:
            to_add = n_ones - s
            for i in rng.choice(zero_idx, to_add, replace=False):
                h[i] = 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    curr_c5 = best_c5
    curr_corr = corr.copy()
    
    t_total = t_end - t_start
    step = 0
    n_total = 2 * n_ones  # rough estimate of pairs per full pass
    
    while time.time() < t_end:
        step += 1
        progress = min(1.0, (time.time() - t_start) / t_total)
        T = 0.02 * (1.0 - progress) ** 2 + 0.001
        
        a = rng.randint(n)
        b = rng.randint(n)
        if h[a] != 1 or h[b] != 0:
            continue
        if a == b:
            continue
        
        delta = pair_swap_delta(curr_corr, h, a, b, n)
        new_corr = curr_corr + delta
        new_c5 = float(np.max(new_corr) * dx)
        dE = new_c5 - curr_c5
        
        if dE < 0 or rng.rand() < np.exp(-dE / max(T, 1e-10)):
            # Accept
            h[a] = 0
            h[b] = 1
            curr_corr = new_corr
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()

            v2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - best_h[np.clip(j1[v1], 0, n - 1)])
            coefs[v2] += (-best_h[np.clip(j2[v2], 0, n - 1)])
            coefs[1] = 0.0  # guard
            f0 = float(np.max(corr_base) * dx)
            f05 = float(np.max(corr_base + 0.5 * coefs) * dx)
            f1 = float(np.max(corr_base + coefs) * dx)
            if f1 >= f0 - 1e-14 and f05 >= f0 - 1e-14:
                continue
            if f05 >= f1:
                hi_new = 0.5
            else:
                hi_new = 1.0
            if f05 < f0:
                hi_new = 0.5 if f05 < f1 else 1.0
            elif f1 < f0:
                hi_new = 1.0
            else:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm-up
    for T in [200, 1000, 5000, 20000]:
        if time.time() - t0 > budget_s * 0.08:
            break
        hc = slsqp_step(best_h, T, n, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Binary simulated annealing (main)
    h_bin, c5_bin = sim_anneal(best_h, n, t0, t0 + budget_s * 0.38, seed)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin

    # Phase 3: SLSQP on continuous from binary
    for T in [500, 3000, 15000, 80000]:
        if time.time() - t0 > budget_s * 0.50:
            break
        hc = slsqp_step(best_h, T, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Another SA round from refined state
    h_bin2, c5_bin2 = sim_anneal(best_h, n, t0, t0 + budget_s * 0.72, seed + 1000)
    if c5_bin2 < best_c5:
        best_c5 = c5_bin2
        best_h = h_bin2

    # Phase 5: SLSQP polish
    for T in [20000, 100000, 500000]:
        if time.time() - t0 > budget_s * 0.84:
            break
        hc = slsqp_step(best_h, T, n, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Final SA at low temperature
    h_bin3, c5_bin3 = sim_anneal(best_h, n, t0, t0 + budget_s * 0.94, seed + 2000)
    if c5_bin3 < best_c5:
        best_c5 = c5_bin3
        best_h = h_bin3

    # Phase 7: Coordinate descent polish
    best_h, best_c5 = coord_fast(best_h, n, t0 + budget_s * 0.98, max_sweeps=30)

    # Final SLSQP
    if time.time() - t0 < budget_s * 0.99:
        hc = slsqp_step(best_h, 200000, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
