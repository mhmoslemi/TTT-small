def compute_grad(h, top_idx, n):

    for k in top_idx:
        L = k - (n-1)
        # For each j, contribution is (1-h[j+L])*I(0<=j+L<n) - h[j-L]*I(0<=j-L<n)
        # j+L valid: j in [max(0,-L), min(n-1, n-1-L)]
        lo1 = max(0, -L)
        hi1 = min(n-1, n-1-L)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += (1.0 - h[lo1+L:hi1+L+1])
        # j-L valid: j-L in [0, n-1], so j in [L, n-1+L], intersect [0,n-1]
        lo2 = max(0, L)
        hi2 = min(n-1, n-1+L)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= h[lo2-L:hi2-L+1]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_obj(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / len(h))

def compute_grad(h, top_idx, n):
    grad = np.zeros(n)
    for k in top_idx:
        L = k - (n - 1)
        lo1 = max(0, -L)
        hi1 = min(n - 1, n - 1 - L)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += (1.0 - h[lo1+L:hi1+L+1])
        lo2 = max(0, L)
        hi2 = min(n - 1, n - 1 + L)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= h[lo2-L:hi2-L+1]
    return grad

def subgrad_method(h, n, t_end, lr0=0.5, beta=0.9, active_frac=0.15):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for it in range(200000):
        if time.time() > t_end:
            break
        corr = get_corr(h)
        n_active = max(5, int(active_frac * (2 * n - 1)))
        top_idx = np.argsort(corr)[-n_active:]
        grad = compute_grad(h, top_idx, n)
        lr = lr0 / (1.0 + it / 2000.0)
        v = beta * v - lr * grad
        h_new = project_h(h + v)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                active_frac = min(0.5, active_frac + 0.05)
                lr0 = lr0 * 0.7
                v *= 0.5
                no_improve = 0
        h = h_new
    return best_h, best_c5

def coord_opt_one(h, n, i, niter=45):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        c = 0.0
        if 0 <= i + L < n:
            c += (1.0 - h[i + L])
        if 0 <= i - L < n:
            c -= h[i - L]
        coef[k] = c
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coef) < np.max(corr_base + m2 * coef):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=100, use_pairs=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_one(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if use_pairs and sweep % 2 == 1 and time.time() < t_end:
            for _ in range(n // 3):
                if time.time() > t_end:
                    break
                i, j = np.random.permutation(n)[:2]
                hi, hj = best_h[i], best_h[j]
                hi_new = coord_opt_one(best_h, n, i)
                best_h[i] = hi_new
                hj_new = coord_opt_one(best_h, n, j)
                h_try = best_h.copy()
                h_try[j] = hj_new
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                else:
                    best_h[i] = hi
        if not improved:
            break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_c5_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def basin_hop(h, c5, n, t_end, max_iters=60):
    work_h = h.copy()
    work_c5 = c5
    np.random.seed(int(time.time() * 1000) % 100000)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max_iters)
        mod = it % 6
        if mod == 0:
            pert = (0.02 + 0.08 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 12)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.015 + 0.05 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.1 * cool) * np.random.randn(mask.sum())
        elif mod == 3:
            blk = max(1, n // 6)
            b = np.random.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            pert[b*blk:min(n, (b+1)*blk)] = (0.02 + 0.06 * cool) * np.random.randn(blk)
        elif mod == 4:
            pert = (0.01 + 0.03 * cool) * np.random.randn(n)
        else:
            mask = np.random.random(n) < 0.08
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.12 * cool) * np.sign(np.random.randn(mask.sum()))
        h_p = project_h(work_h + pert)
        T_bh = 500 + 10000 * cool
        hc = slsqp_opt(h_p, T_bh, n, maxiter=400)
        c5v = compute_c5(hc)
        temp = 0.0006 * cool + 0.00004
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)

    # Phase 1: Quick coordinate descent to stabilize (10%)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.10, max_sweeps=15, use_pairs=False)

    # Phase 3: Basin hopping (20%)

    if c5_bh < best_c5:

        hc = slsqp_opt(best_h, T, n, maxiter=3000)

    if elapsed() < budget_s * 0.98:

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
