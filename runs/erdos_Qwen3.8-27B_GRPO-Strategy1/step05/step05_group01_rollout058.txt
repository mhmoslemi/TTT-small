import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s < 1e-14:
            return h
        h = np.clip(h * (1.0 + diff / s), 0, 1)
    return h

def softmax_corr(h, T, topk=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argsort(corr)[-topk:]
        cv = corr[idx]
    else:
        cv = corr
    m = cv.max()
    if m < 0:
        m = 0
    if T > 1e8:
        return m
    return m + np.log(np.sum(np.exp(T * (cv - m)))) / T

def grad_softmax(h, T, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argsort(corr)[-topk:]
    else:
        idx = np.arange(len(corr))
    cv = corr[idx]
    m = cv.max()
    if T > 1e8:
        w = np.zeros(len(cv))
        w[cv >= m - 1e-10] = 1.0
        w /= w.sum()
    else:
        w = np.exp(T * (cv - m))
        w /= w.sum()
    shifts = idx - (n - 1)
    i = np.arange(n)
    jm = i[:, None] - shifts[None, :].astype(float)
    jp = i[:, None] + shifts[None, :].astype(float)
    v1 = (jm >= 0) & (jm < n)
    v2 = (jp >= 0) & (jp < n)
    jmc = np.clip(jm, 0, n - 1).astype(int)
    jpc = np.clip(jp, 0, n - 1).astype(int)
    contrib = (1.0 - h[jmc]) * v1 - h[jpc] * v2
    grad = (contrib * w[None, :]).sum(axis=1)
    return grad * dx

        x = np.clip(x, 0, 1)
        c5 = softmax_corr(x, T, topk) * (2.0 / n)
        pen = pen_lambda * (x.sum() - n / 2.0) ** 2
        return c5 + pen

        g = grad_softmax(x, T, topk)
        g += 2 * pen_lambda * (x.sum() - n / 2.0)
        return g

def slsqp_opt(h, n, T, topk, maxiter, pen_lambda=0):

def coord_descent(h, n, t_end, max_sweeps, n_top):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s < 1e-14:
            return h
        h = np.clip(h * (1.0 + diff / s), 0, 1)
    return h

def softmax_corr(h, T, topk=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argsort(corr)[-topk:]
        cv = corr[idx]
    else:
        cv = corr
    m = cv.max()
    if T > 1e8:
        return m
    return m + np.log(np.sum(np.exp(T * (cv - m)))) / T

def grad_softmax(h, T, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argsort(corr)[-topk:]
    else:
        idx = np.arange(len(corr))
    cv = corr[idx]
    m = cv.max()
    if T > 1e8:
        w = np.zeros(len(cv))
        w[cv >= m - 1e-10] = 1.0
        w /= max(w.sum(), 1e-30)
    else:
        w = np.exp(T * (cv - m))
        w /= w.sum()
    shifts = idx - (n - 1)
    i = np.arange(n)
    jm = i[:, None].astype(float) - shifts[None, :].astype(float)
    jp = i[:, None].astype(float) + shifts[None, :].astype(float)

    jpc = np.clip(jp, 0, n - 1).astype(int)
    contrib = (1.0 - h[jmc]) * v1 - h[jpc] * v2
    grad = (contrib * w[None, :]).sum(axis=1)

        x = np.clip(x, 0, 1)
        g = grad_softmax(x, T, topk)
        g += 2 * pen_lam * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_h(res.x)
    except Exception:
        return h

def slsqp_opt(h, n, T, topk, maxiter):
    dx = 2.0 / n
    def obj(x):
        return softmax_corr(x, T, topk) * dx

                       options={'maxiter': maxiter, 'ftol': 1e-16})

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n = 200
        x = np.linspace(0, 2, n, endpoint=False)
        init_h = 0.5 + 0.35 * np.sin(3 * np.pi * x) + 0.2 * np.sin(7 * np.pi * x)
        init_h = project_h(init_h)
    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)
    # Main optimization
    t_main = t0 + budget_s * 0.6
    best_h, best_c5 = optimize_phase(best_h, n, t_main)
    # Second basin hop round
    if t_end - time.time() > 120:
        bh_h, bh_c5 = basin_hop(best_h, n, t_main + 80, 25, 0.04, 5000, 30000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
        hc = lbfgs_opt(best_h, n, 20000, 6, 300, 10000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Extended coordinate descent
    if t_end - time.time() > 50:
        cd_end = time.time() + min((t_end - 50 - time.time()) * 0.6, 120)
        best_h, best_c5 = fast_coord_descent(best_h, n, cd_end, 200, 40)
    # Final SLSQP + coord
    if t_end - time.time() > 25:
        hc = slsqp_opt(best_h, n, 500000, 3, 600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if t_end - time.time() > 10:
        best_h, best_c5 = fast_coord_descent(best_h, n, t_end - 2, 80, 50)
    # Final polish
    if t_end - time.time() > 8:
        hc = slsqp_opt(best_h, n, 1000000, 2, 400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
