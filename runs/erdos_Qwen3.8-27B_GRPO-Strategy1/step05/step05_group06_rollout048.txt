import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-12:
                h += room * (diff / tot)
        elif s > 1e-12:
            h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def prep_coord(h, i):
    n = len(h)
    m = 2 * n - 1
    n1 = n - 1
    base = np.zeros(m)
    B = np.zeros(m)
    for idx in range(m):
        k = idx - n1
        val = 0.0
        b = 0.0
        lo = max(0, -k)
        hi = min(n, n - k)
        for t in range(lo, hi):
            ni = t + k
            hv_i = 0.0 if t == i else h[t]
            hv_j = 0.0 if ni == i else h[ni]
            val += hv_i * (1.0 - hv_j)
            if t == i:
                b += (1.0 - hv_j)
            if ni == i:
                b += -hv_i
        base[idx] = val
        B[idx] = b
    return base, B, n1

def f1d_wrapper(x, base, B, n1):
    m = len(base)
    if x <= 1e-14:
        c = base
    elif x >= 1.0 - 1e-14:
        c = base + B
        c[n1] -= 1.0
    else:
        c = base + x * B
        c[n1] -= x * x
    return np.max(c)

def coord_descent(h_init, t_end, max_sweeps=5000):
    n = len(h)
    best = project_h(h_init.copy())
    best_c5 = compute_c5(best)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            orig = best[i]
            base, B, n1 = prep_coord(best, i)
            res = minimize_scalar(f1d_wrapper, bounds=(0.0, 1.0), args=(base, B, n1),
                                  method='bounded', options={'xatol': 1e-10})
            nv = res.x
            if abs(nv - orig) < 1e-11:
                continue
            best[i] = nv
            best = project_h(best)
            c5n = compute_c5(best)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                improved = True
        if not improved and sweep > 0:
            break
    return best, best_c5

def softmax_smoothed_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    e = np.exp(Tc)
    return dx * (mc + np.log(e.sum()) / T)

def softmax_smoothed_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    e = np.exp(Tc)
    w = e / e.sum()
    m = 2 * n - 1
    lags = np.arange(m) - (n - 1)
    g = np.zeros(n)
    for idx in range(m):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = lags[idx]
        lo = max(0, -k)
        hi = min(n, n - k)
        for t in range(lo, hi):
            g[t] += wi * (1.0 - h[t + k])
            g[t + k] += -wi * h[t]
    g *= dx
    g += 2.0 * lam * (h.sum() - n / 2.0)
    return g

def lbfgs_phase(h, T, lam, maxit, t_end):
    from scipy.optimize import minimize
    n = len(h)
    def fo(x):
        return softmax_smoothed_obj(x, T) + lam * (x.sum() - n / 2.0) ** 2
    def go(x):
        return softmax_smoothed_grad(x, T, lam)
    x0 = project_h(h.copy())
    try:
        r = minimize(fo, x0, jac=go, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-9})
        return project_h(r.x)
    except Exception:
        return x0

def basin_hop(h_init, c5_init, t_end, n_hops=40):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = c5_init
    for it in range(n_hops):

            best_c5 = c5_c
    return best_h, best_c5

        best_h, best_c5 = coord_descent(best_h, t0 + budget_s * 0.96)

    # Phase 7: Final L-BFGS polish
    if time.time() < t0 + budget_s * 0.99:
        hc = lbfgs_phase(best_h, 100000, 600.0, 2000, t0 + budget_s * 0.99)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
