    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):

def smooth_obj(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return sm * (2.0 / len(h))

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (d / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def slsqp(h, T, n, mi=2000):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0

        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':con}],
                     options={'maxiter':mi,'ftol':1e-16})
        return project_h(np.clip(r.x,0,1))
    except:
        return h

def cstep(h, n, i, ni=45):
    hb = h.copy(); hb[i] = 0
    cb = get_corr(hb)
    co = np.zeros(2*n-1)
    for m in range(2*n-1):
        a = i + m - n + 1
        if 0 <= a < n: co[m] += 1.0 - h[a]
        b = i - m + n - 1
        if 0 <= b < n: co[m] -= h[b]
    lo, hi = 0.0, 1.0
    for _ in range(ni):
        m1 = lo + (hi-lo)/3.0
        m2 = hi - (hi-lo)/3.0
        if np.max(cb + m1*co) < np.max(cb + m2*co):
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2.0

            cv = compute_c5(ht)
            if cv < bc5 - 1e-15:

        if not imp: break
    return bh, bc5

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    t = n * 0.5 - h.sum()
    if abs(t) < 1e-14:
        return h
    if t > 0:
        r = 1.0 - h
        s = r.sum()
        if s < 1e-14:
            return h
        h = h + r * (t / s)
    else:
        s = h.sum()
        if s < 1e-14:
            return h
        h = h * (1.0 + t / s)
    return np.clip(h, 0, 1)

def smax(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

                     bounds=[(0, 1)] * n,

                     options={'maxiter': mi, 'ftol': 1e-16})

        v1 = (m1 >= 0) & (m1 < 2 * n - 1)
        if v1.any():
            grad[i] += np.dot(w[m1[v1]], 1.0 - h[b[v1]])
        a = idx
        m2 = a - i + n - 1
        v2 = (m2 >= 0) & (m2 < 2 * n - 1)
        if v2.any():
            grad[i] -= np.dot(w[m2[v2]], h[a[v2]])
    grad *= dx
    return obj, grad

def lbfgs_run(h, T, n, mi, mu=500.0):
    def fg(x):
        o, g = smax_grad(x, T)
        s = x.sum() - n * 0.5
        o += mu * s * s
        g = g + 2.0 * mu * s
        return o, g
    try:
        r = minimize(fg, h, method='L-BFGS-B', jac=True,
                     bounds=[(0, 1)] * n, options={'maxiter': mi, 'ftol': 1e-18})
        return project_h(np.clip(r.x, 0, 1))
    except Exception:
        return h

def cstep(h, n, i, ni=50):
    hb = h.copy()
    hb[i] = 0.0
    cb = get_corr(hb)
    co = np.zeros(2 * n - 1)
    ms = np.arange(2 * n - 1)
    a = i + ms - n + 1
    v1 = (a >= 0) & (a < n)
    co[v1] += 1.0 - h[a[v1]]
    b = i - ms + n - 1
    v2 = (b >= 0) & (b < n)
    co[v2] -= h[b[v2]]
    lo, hi = 0.0, 1.0
    for _ in range(ni):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(cb + m1 * co) < np.max(cb + m2 * co):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def cdescent(h, n, t_end, ms=50):
    bh = h.copy()
    bc5 = compute_c5(bh)
    for sw in range(ms):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                break
            hv = cstep(bh, n, i)
            if abs(hv - bh[i]) < 1e-12:
                continue
            ht = bh.copy()
            ht[i] = hv
            ht = project_h(ht)
            cv = compute_c5(ht)
            if cv < bc5 - 1e-15:
                bc5 = cv
                bh = ht
                imp = True
        if not imp:
            break
    return bh, bc5
