```python
    old = h[i]
    # Compute coefficients for each lag
    # corr has length 2n-1, indexed by k+n-1 for k in [-(n-1), n-1]
    coefs = np.zeros(2*n-1)
    # Term 1: h[i] contributes to corr at lag k where j=i is valid

    # Simple approach: binary search / ternary search on the piecewise linear function
    # Since it's convex, ternary search works but slowly.
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    for i in range(n):  # or random order
        old = h[i]
        if old < 1e-15 or old > 1.0 - 1e-15:

            pass
        # Compute coefficients
        coefs = np.zeros(2*n-1)
        # First term: m=i, lag = k-(n-1), need 0<=i+lag<n
        # k range: [n-1-i, 2n-2-i]
        k_start1 = n - 1 - i
        k_end1 = 2*n - 2 - i + 1
        lags1 = np.arange(k_start1, k_end1) - (n-1)  # actual lags
        idx1 = i + lags1  # indices into h

        new_val = old + res.x
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def get_coefs(n, i, h):
    coefs = np.zeros(2 * n - 1)
    ks1 = np.arange(n - 1 - i, 2 * n - 1 - i)
    coefs[ks1] += (1.0 - h)  # h_idx = 0,1,...,n-1
    ks2 = np.arange(i, i + n)
    coefs[ks2] += (-h[::-1])  # h_idx = n-1, n-2, ..., 0
    return coefs

def coord_sweep(h, corr, t_end, n_sweeps_target=1000):
    """Fast coordinate descent, returns updated h and corr."""
    n = len(h)
    dx = 2.0 / n
    sweeps = 0
    order = np.random.permutation(n)
    while True:
        if time.time() > t_end:
            break
        for i in order:
            if time.time() > t_end:
                return h, corr
            old = h[i]
            coefs = get_coefs(n, i, h)
            def f(t):
                return np.max(corr + t * coefs)
            res = minimize_scalar(f, bounds=(-old, 1.0 - old), method='bounded',
                                 options={'xatol': 1e-12, 'maxiter': 40})
            new_val = np.clip(old + res.x, 0, 1)
            delta = new_val - old
            if abs(delta) > 1e-14:
                h[i] = new_val
                corr += delta * coefs
        # Project every 10 sweeps
        sweeps += 1
        if sweeps % 10 == 0:
            h = project_h(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
        order = np.random.permutation(n)

def slsqp_smooth(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh)) / (2*n-1)) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def gen_tm(n):
    """Thue-Morse based initialization."""
    a = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    return project_h(a, n)

def gen_random_bin(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    a = np.zeros(n)
    a[idx] = 1.0

    a = np.tile(pattern, n // d)
    if n % d != 0:
        a = np.concatenate([a, pattern[:n % d]])
```

For the paired move approach: I pick (i, j) and optimize h[i] + t, h[j] - t. This preserves the sum exactly. But it's a 2D→1D reduction and needs both coefs[i] and coefs[j].

```python
def paired_move(h, corr, i, j):
    coefs_i = get_coefs(n, i, h)
    coefs_j = get_coefs(n, j, h)
    net_coefs = coefs_i - coefs_j
    # Minimize max(corr + t * net_coefs) for valid t
    # Valid t: -h[i] ≤ t ≤ 1-h[i] AND -h[j] ≤ -t ≤ 1-h[j]
    # → max(-h[i], h[j]-1) ≤ t ≤ min(1-h[i], h[j])
    t_min = max(-h[i], h[j] - 1)
    t_max = min(1 - h[i], h[j])
    if t_min >= t_max:
        return h, corr
    def f(t):
        return np.max(corr + t * net_coefs)
    res = minimize_scalar(f, bounds=(t_min, t_max), method='bounded',
                         options={'xatol': 1e-12, 'maxiter': 40})
    t_opt = res.x
    if abs(t_opt) > 1e-14:
        h[i] += t_opt
        h[j] -= t_opt
        corr += t_opt * net_coefs
    return h, corr
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def get_coefs(n, i, h):
    coefs = np.zeros(2 * n - 1)
    ks1 = np.arange(n - 1 - i, 2 * n - 1 - i)
    coefs[ks1] += (1.0 - h)
    ks2 = np.arange(i, i + n)
    coefs[ks2] += (-h[::-1])
    return coefs

def coord_sweep(h, corr, t_end):
    n = len(h)
    dx = 2.0 / n
    order = np.random.permutation(n)
    sweep_count = 0
    while True:
        if time.time() > t_end:
            break
        for i in order:
            if time.time() > t_end:
                return h, corr
            old = h[i]
            coefs = get_coefs(n, i, h)
            def f(t):
                return np.max(corr + t * coefs)
            res = minimize_scalar(f, bounds=(-old, 1.0 - old), method='bounded',
                                 options={'xatol': 1e-13, 'maxiter': 35})
            new_val = np.clip(old + res.x, 0.0, 1.0)
            delta = new_val - old
            if abs(delta) > 1e-14:
                h[i] = new_val
                corr += delta * coefs
        sweep_count += 1
        if sweep_count % 15 == 0:
            h = project_h(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
        order = np.random.permutation(n)
    return h, corr

def paired_sweep(h, corr, t_end):
    n = len(h)
    while time.time() < t_end:
        pairs = np.random.randint(0, n, size=(n, 2))
        for idx in range(n):
            if time.time() > t_end:
                return h, corr
            i, j = pairs[idx]
            if i == j:
                continue
            ci = get_coefs(n, i, h)
            cj = get_coefs(n, j, h)
            net = ci - cj
            t_min = max(-h[i], h[j] - 1.0)
            t_max = min(1.0 - h[i], h[j])
            if t_max - t_min < 1e-12:
                continue
            def f2(t):
                return np.max(corr + t * net)
            res = minimize_scalar(f2, bounds=(t_min, t_max), method='bounded',
                                 options={'xatol': 1e-13, 'maxiter': 30})
            t_opt = res.x
            if abs(t_opt) > 1e-14:
                h[i] += t_opt
                h[j] -= t_opt
                corr += t_opt * net
    return h, corr

def slsqp_smooth(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -80, 80)
        return (m + np.log(np.sum(np.exp(sh)) / (2 * n - 1)) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def gen_tm(n):
    a = np.array([bin(i % 65536).count('1') % 2 for i in range(n)], dtype=float)
    return project_h(a, n)

def gen_rand_bin(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    a = np.zeros(n)
    a[idx] = 1.0
    return a

def gen_basketball(n, rng):
    a = np.zeros(n)
    a[0] = 1.0
    for i in range(1, n):
        corr_sums = []
        best = 0
        for val in [0.0, 1.0]:
            a_test = a.copy()
            a_test[i] = val
            c = np.correlate(a_test, 1.0 - a_test, mode='full')
            score = np.sum(np.abs(c)[:n])
            if score < best or best == 0:
                best = score
                best_v = val
        a[i] = best_v
    return project_h(a, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n_best = len(best_h)
    else:
        best_h = gen_tm(160)
        n_best = 160
    best_c5 = compute_c5(best_h)

    # Generate candidates at multiple n values
    n_list = [120, 160, 200, 240]
    if n_best not in n_list:
        n_list.append(n_best)
    candidates = [(best_h.copy(), n_best, best_c5)]
    for nn in n_list:
        candidates.append((gen_tm(nn), nn, 0))
        candidates.append((gen_rand_bin(nn, rng), nn, 0))
        candidates.append((gen_rand_bin(nn, rng), nn, 0))

    # Quick coordinate descent on each candidate
    for idx in range(len(candidates)):
        if elapsed() > budget_s * 0.08:
            break
        h_c, n_c, _ = candidates[idx]
        h_c = project_h(h_c, n_c)
        corr_c = np.correlate(h_c, 1.0 - h_c, mode='full')
        t_sub = t0 + min(elapsed() + budget_s * 0.025, budget_s * 0.08)
        h_c, corr_c = coord_sweep(h_c, corr_c, t_sub)
        c5_c = compute_c5(h_c)
        candidates[idx] = (h_c, n_c, c5_c)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
            n_best = n_c
    n = n_best

    # SLSQP annealing on best
    T_list = [80000, 30000, 10000, 4000, 1500, 500, 200, 80, 30, 10]
    for T in T_list:
        if elapsed() > budget_s * 0.30:
            break
        hc = slsqp_smooth(best_h.copy(), T, n, maxiter=350)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Intensive coordinate descent (single-coordinate)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    t_cd1 = t0 + budget_s * 0.55
    best_h, corr = coord_sweep(best_h, corr, t_cd1)
    best_c5 = compute_c5(best_h)

    # Paired coordinate descent (preserves sum)
    t_cd2 = t0 + budget_s * 0.70
    best_h, corr = paired_sweep(best_h, corr, t_cd2)
    pc5 = compute_c5(best_h)
    if pc5 < best_c5:
        best_c5 = pc5

    # Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(25):
        if elapsed() > budget_s * 0.84:
            break
        cool = 1.0 - it / 25.0
        amp = 0.008 + 0.04 * cool
        h_pert = project_h(work_h + amp * rng.randn(n), n)
        T_bh = 300 + 3000 * cool
        hc = slsqp_smooth(h_pert, T_bh, n, maxiter=300)
        c5v = compute_c5(hc)
        temp = 0.0002 * cool + 0.00002
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Final intensive coordinate descent
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_h, corr = coord_sweep(best_h, corr, t0 + budget_s * 0.93)
    best_c5 = compute_c5(best_h)

    # Paired polish
    t_cd3 = t0 + budget_s * 0.96
    best_h, corr = paired_sweep(best_h, corr, t_cd3)
    pc5 = compute_c5(best_h)
    if pc5 < best_c5:
        best_c5 = pc5

    # Final SLSQP polish
    if elapsed() < budget_s * 0.98:
        hc = slsqp_smooth(best_h, 20, n, maxiter=200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
