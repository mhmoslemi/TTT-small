import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = h.copy()
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        s = float(h.sum())
        if abs(s - target) < 1e-12:
            return h
        h += (target - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binarize(h, N):
    hb = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(hb.sum())
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        hb[idx[np.random.permutation(len(idx))[:n1 - nh]]] = 0
    elif n1 < nh:
        idx = np.where(hb == 0)[0]
        hb[idx[np.random.permutation(len(idx))[:nh - n1]]] = 1
    return hb

def upscale(h, N_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h).clip(0, 1)

def binary_search(h, N, deadline, max_sweeps=400, sample_n=100, mu=0.03):

    c = fft_corr(h)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        P = float(c.max())
        if P < best_c:
            best_c = P
        D = compute_D(h, N)
        if len(ones) > sample_n:
            p_list = np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = (base[None, :] - D[zeros, :]).max(axis=1)
            mi = int(np.argmin(cand))
            if cand[mi] < best_new_P:
                best_new_P = float(cand[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None:
            p, q = best_pq
            h[p], h[q] = 0.0, 1.0
            c_new = fft_corr(h)
            if c_new.max() <= P + 1e-9:
                c = c_new
            else:
                h[p], h[q] = 1.0, 0.0
                c = fft_corr(h)
        elif mu > 0 and np.random.random() < mu:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
        elif len(ones) >= 2 and len(zeros) >= 2:
            p1, p2 = np.random.choice(ones, 2, replace=False)
            q1, q2 = np.random.choice(zeros, 2, replace=False)
            h[p1], h[p2] = 0.0, 0.0
            h[q1], h[q2] = 1.0, 1.0
            c_new = fft_corr(h)
            if c_new.max() <= P + 1e-9:
                c = c_new
            else:
                h[p1], h[p2] = 1.0, 1.0
                h[q1], h[q2] = 0.0, 0.0
                if sweep > 80:
                    break
        else:
            break
    return h, best_c

def nesterov_soft(h, N, deadline, k_top=40, lr0=0.012, T0=0.006):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        T = T0 + 0.015 * max(0, 1.0 - it / 300.0)
        sc = (corr - corr.max()) / T
        sc = np.clip(sc, -50, 0)
        w = np.exp(sc)
        w /= max(w.sum(), 1e-15)
        grad = np.zeros(N)
        center = N - 1
        thresh = w.max() * 0.005
        active_idx = np.where(w > thresh)[0]
        for ai in active_idx:
            L = int(ai - center)
            if L == 0:
                grad += w[ai] * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += w[ai] * (1.0 - y[:N - L])
                grad[:N - L] -= w[ai] * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += w[ai] * (1.0 - y[M:])
                    grad[M:] -= w[ai] * y[:N - M]
        h_new = project_sum(y - lr * grad * N, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.992
            beta = min(0.98, 0.9 + it * 0.0002)
    return best_h, best_pk * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary local search on initial
    d1 = min(deadline - 380, t0 + 130)
    h_s, _ = binary_search(h0, N0, d1, max_sweeps=350, sample_n=100, mu=0.02)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Upscale to 2x, continuous optimize
    N_up = N0 * 2
    if time.time() < deadline - 280:
        h_up = upscale(best_h, N_up)
        h_up = project_sum(h_up, N_up / 2.0)
        d2 = min(deadline - 220, time.time() + 130)
        h_cont, _ = nesterov_soft(h_up, N_up, d2, k_top=40, lr0=0.01, T0=0.005)
        c5_cont = compute_c5(h_cont, N_up)
        if c5_cont < best_c5:
            best_c5, best_h = c5_cont, h_cont.copy()

    # Phase 3: Binarize at larger N, discrete optimize
    if time.time() < deadline - 200:
        h_b3 = binarize(best_h, len(best_h))
        d3 = min(deadline - 160, time.time() + 130)
        h_s3, _ = binary_search(h_b3, len(best_h), d3, max_sweeps=400, sample_n=120, mu=0.04)
        c5_3 = compute_c5(h_s3, len(best_h))
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Try multiple N values with interpolation
    N_candidates = []
    for n_try in [480, 560, 640, 760, 880]:
        if n_try != N0 * 2:
            N_candidates.append(n_try)
    for N_try in N_candidates:
        if time.time() > deadline - 120:
            break
        N_try = N_try + (N_try + 1) % 2
        h_interp = upscale(best_h, N_try)
        h_bt = binarize(h_interp, N_try)
        sub_dl = min(deadline - 80, time.time() + 60)
        h_rt, _ = binary_search(h_bt, N_try, sub_dl, max_sweeps=180, sample_n=80, mu=0.03)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 5: Iterative refine cycle at current best N
    if time.time() < deadline - 100:
        for cycle in range(3):
            if time.time() > deadline - 50:
                break
            h_c = best_h.astype(float) + np.random.randn(len(best_h)) * 0.012
            h_c = project_sum(h_c, len(best_h) / 2.0)
            d_cyc = min(deadline - 40, time.time() + 50)
            h_nc, _ = nesterov_soft(h_c, len(best_h), d_cyc, k_top=50, lr0=0.005, T0=0.004)
            h_bf = binarize(h_nc, len(best_h))
            d_bf = min(deadline - 30, time.time() + 50)
            h_bf, _ = binary_search(h_bf, len(best_h), d_bf, max_sweeps=200, sample_n=100, mu=0.02)
            c5_cyc = compute_c5(h_bf, len(best_h))
            if c5_cyc < best_c5:
                best_c5, best_h = c5_cyc, h_bf.copy()

    # Phase 6: Final continuous polish
    if time.time() < deadline - 15:
        hF = best_h.astype(float) + np.random.randn(len(best_h)) * 0.01
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, _ = nesterov_soft(hF, len(best_h), deadline - 8, k_top=60, lr0=0.004, T0=0.003)
        c5F = compute_c5(hF, len(best_h))
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 7: Final binary cleanup
    if time.time() < deadline - 2:
        hF2 = binarize(best_h, len(best_h))
        hF2, _ = binary_search(hF2, len(best_h), deadline - 1, max_sweeps=100, sample_n=100, mu=0.01)
        c5F2 = compute_c5(hF2, len(best_h))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
