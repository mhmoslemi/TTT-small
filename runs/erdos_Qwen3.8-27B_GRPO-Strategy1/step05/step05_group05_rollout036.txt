import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    tc = T * c
    tc_max = np.max(tc)
    w = np.exp(tc - tc_max)
    w /= w.sum()
    one_minus_h = 1.0 - h
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        m = k - (n - 1)
        wk = w[k]
        i1s = max(0, m)
        i1e = min(n, n + m)
        if i1s < i1e:
            term1[i1s:i1e] += wk * one_minus_h[i1s - m:i1e - m]
        i2s = max(0, -m)
        i2e = min(n, n - m)
        if i2s < i2e:
            term2[i2s:i2e] += wk * h[i2s + m:i2e + m]
    return dx * (term1 - term2)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgs_opt(h, T, maxiter=400):
    n = len(h)
    res = minimize(
        lambda x: smooth_obj(x, T), h, method='L-BFGS-B',
        jac=lambda x: smooth_grad(x, T),
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-16, 'maxls': 60})
    return project_h(np.clip(res.x, 0, 1))

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_opt_fast(h, n, i):
    c = get_corr(h)
    m_vals = np.arange(-(n - 1), n)
    l_a = i - m_vals
    j_b = i + m_vals
    va = (l_a >= 0) & (l_a < n)
    vb = (j_b >= 0) & (j_b < n)
    g = np.zeros(2 * n - 1)
    g[va] += (1.0 - h[l_a[va]])
    g[vb] += (-h[j_b[vb]])
    lo = -h[i]
    hi = 1.0 - h[i]
    for _ in range(60):
        d1 = lo + (hi - lo) / 3.0
        d2 = hi - (hi - lo) / 3.0
        if np.max(c + d1 * g) < np.max(c + d2 * g):
            hi = d2
        else:
            lo = d1
    return h[i] + (lo + hi) / 2.0

def coord_sweep(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        for i in range(n):
            if time.time() > t_end:

        best_h = project_h(best_h)

        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
        else:
            break
    return best_h, best_c5

def subgradient_step(h, n, alpha):
    c = get_corr(h)
    k_star = np.argmax(c)
    m = k_star - (n - 1)
    g = np.zeros(n)
    i1s = max(0, m)
    i1e = min(n, n + m)
    if i1s < i1e:
        g[i1s:i1e] += (1.0 - h[i1s - m:i1e - m])
    i2s = max(0, -m)
    i2e = min(n, n - m)
    if i2s < i2e:
        g[i2s:i2e] -= h[i2s + m:i2e + m]
    h_new = h - alpha * g
    return project_h(h_new)
