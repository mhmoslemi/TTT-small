```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    out = np.clip(h, 0.0, 1.0)
    # ensure exact sum
    if abs(out.sum() - target) > 1e-12:
        out = out + (target - out.sum()) / n
        out = np.clip(out, 0.0, 1.0)
    return out

def slsqp_opt(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(np.clip(res.x, 0, 1), n)
        return h_new
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=500):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * 2.0 / n)
    lags = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = h[i]
            pm = i - lags
            pp = i + lags
            v1 = (pm >= 0) & (pm < n)
            v2 = (pp >= 0) & (pp < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - h[pm[v1]])
            coefs[v2] += (-h[pp[v2]])
            base = corr - old * coefs
            res = minimize_scalar(lambda v: np.max(base + v * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-14, 'maxiter': 60})
            v_new = res.x
            if abs(v_new - old) > 1e-13:
                h[i] = v_new
                h = project_h(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5 = float(np.max(corr) * 2.0 / n)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return h, best_c5

def pairwise_descent(h, n, t_end, max_sweeps=40):
    h = h.copy()
    best_c5 = compute_c5(h)
    grid = np.linspace(0.0, 1.0, 12)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(0, n - 1, 2):
            j = i + 1
            hi0, hj0 = h[i], h[j]
            best_pair = (hi0, hj0)
            best_v = best_c5
            for gi in grid:
                for gj in grid:
                    h[i], h[j] = gi, gj
                    h = project_h(h, n)
                    c5 = compute_c5(h)
                    if c5 < best_v - 1e-15:
                        best_v = c5
                        best_pair = (gi, gj)
            h = h.copy()
            h[i], h[j] = best_pair
            h = project_h(h, n)
            if best_v < best_c5:
                best_c5 = best_v
                improved = True
        if not improved:
            break
    return h, best_c5

def nudge_restart(h, n, t_end, n_restarts=8, amp=0.03):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for r in range(n_restarts):
        if time.time() > t_end:
            break
        idx = np.random.choice(n, size=max(2, n // 10), replace=False)
        h_try = best_h.copy()
        h_try[idx] += amp * np.random.randn(len(idx)) * np.random.choice([1, -1], size=len(idx))
        h_try = project_h(h_try, n)

        if c5_try < best_c5 + 0.001:

            if c5_opt < best_c5:

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n = len(best_h)
    else:
        n = 160
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup
    for T in [3000, 800, 200, 50, 15]:
        if time.time() - t0 > 40:
            break
        h_new = slsqp_opt(best_h, T, n, maxiter=300)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 2: Long precise coordinate descent
    t_cd2 = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, t_cd2, max_sweeps=800)

    # Phase 3: Pairwise descent
    t_pw = t0 + budget_s * 0.48
    best_h, best_c5 = pairwise_descent(best_h, n, t_pw, max_sweeps=25)

    # Phase 4: More coordinate descent
    t_cd3 = t0 + budget_s * 0.60
    best_h, best_c5 = coord_descent(best_h, n, t_cd3, max_sweeps=400)

    # Phase 5: Nudge restarts to escape local optima
    t_nr = t0 + budget_s * 0.72
    best_h, best_c5 = nudge_restart(best_h, n, t_nr, n_restarts=6, amp=0.025)

    # Phase 6: Coordinate descent after nudges
    t_cd4 = t0 + budget_s * 0.82
    best_h, best_c5 = coord_descent(best_h, n, t_cd4, max_sweeps=400)

    # Phase 7: Try n=200 if time permits
    if best_c5 > 0.38087 and time.time() - t0 < budget_s * 0.80:
        n2 = 200
        # Upsample by interpolation
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h, period=2)
        h2 = project_h(h2, n2)
        # Quick SLSQP
        h2 = slsqp_opt(h2, 300, n2, maxiter=200)
        # Coordinate descent
        h2, c5_2 = coord_descent(h2, n2, min(t0 + budget_s * 0.92, time.time() + 80), max_sweeps=100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            n = len(best_h)

    # Phase 8: Final precise coordinate descent
    t_final = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t_final, max_sweeps=300)

    # Final pairwise
    t_final2 = t0 + budget_s * 0.99
    best_h, best_c5 = pairwise_descent(best_h, n, t_final2, max_sweeps=5)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
