One more idea: **Subgradient descent on the max problem**

The objective f(h) = max_k g_k(h) where g_k(h) = corr[k] * dx.

Subgradient: ∇f(h) = (1/|A|) * sum_{k∈A} ∇g_k(h) where A = {k : g_k(h) = f(h)}

∇g_k(h)[i] = (1 - h[i+k]) - h[i-k] (with boundary handling, × dx)

This gives a direction to decrease the worst lags.

def subgrad_step(h, n, eta):
    """One subgradient step on C5"""
    dx = 2.0/n
    c = comp_corr(h)
    mx = np.max(c)
    # Active set: lags where corr is within epsilon of max
    active = np.where(c > mx - 0.5)[0]
    # Compute subgradient (average of active gradients)
    sg = np.zeros(n)
    for ak in active:
        k = ak - (n-1)  # actual lag
        # d(corr[k])/d(h[i]) = (1-h[i+k]) - h[i-k] (where valid)
        # corr[k] = sum_i h[i]*(1-h[i+k]) for 0<=i<n, 0<=i+k<n
        # ∂/∂h[j] = (1-h[j+k]) if j+k < n, else 0
        #           - h[j-k] if j-k >= 0, else 0
        i_range = np.arange(n)
        # forward: h[j] appears as h[i] with i=j, contributes (1-h[j+k]) if j+k < n
        fk = i_range + k
        valid_f = (fk >= 0) & (fk < n)
        sg[valid_f] += (1.0 - h[fk[valid_f]])
        # backward: h[j] appears as h[i+k] with i=j-k, contributes -h[j-k] if j-k >= 0
        bk = i_range - k
        valid_b = (bk >= 0) & (bk < n)
        sg[valid_b] += (-h[bk[valid_b]])
    sg /= (len(active) * dx + 1e-15)
    # Actually, the subgradient of f = max_k corr[k]*dx is:
    # ∇f = (1/|A|) * sum_{k in A} dx * ∂corr[k]/∂h
    # So sg already has the right form. Let me re-derive.
    # f(h) = max_k corr[k] * dx
    # ∇f = dx * (1/|A|) * sum_{k in A} ∂corr[k]/∂h
    # ∂corr[k]/∂h[j] = (1-h[j+k]) * 1{j+k < n} + h[j] * ∂(1-h[j+k])/∂h[j]... 
    # No, h[j] appears in two places:
    # 1. As the first factor: h[j], contributes (1-h[j+k])
    # 2. As the second factor: (1-h[j+k]), contributes -h[j-k]
    # So ∂corr[k]/∂h[j] = (1-h[j+k]) - h[j-k] (with valid range checks)
    # And ∇f = dx * (1/|A|) * sum_{k in A} [(1-h[j+k]) - h[j-k]]
    
    # Update

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def comp_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5v(h):
    n = len(h)
    return float(np.max(comp_corr(h)) * (2.0 / n))

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    d = t - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        tot = r.sum()
        if tot < 1e-14:
            return h
        h = h + r * (d / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def _e1d(hi, cb, cf, dx):
    return float(np.max(cb + hi * cf) * dx)

def coidx(h, n, i):
    old = h[i]
    hb = h.copy()
    hb[i] = 0.0
    cb = comp_corr(hb)
    kr = np.arange(-(n - 1), n)
    j1 = i + kr
    j2 = i - kr
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    cf = np.zeros(2 * n - 1)
    cf[v1] += 1.0 - h[np.clip(j1[v1], 0, n - 1)]
    cf[v2] += -h[np.clip(j2[v2], 0, n - 1)]
    dx = 2.0 / n
    try:
        r = minimize_scalar(_e1d, args=(cb, cf, dx),
                            bounds=(0.0, 1.0), method='bounded',
                            options={'xatol': 1e-14})
        return float(r.x)
    except Exception:
        return old
