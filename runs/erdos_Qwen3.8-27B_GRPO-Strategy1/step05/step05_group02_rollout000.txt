        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def lag_target_push(h, n, t_end, n_passes=300, rng=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(1)
    for it in range(n_passes):
        if time.time() > t_end:
            break
        corr = compute_corr(best_h)
        c5_val = np.max(corr) * dx
        thresh = (c5_val - 0.03 * dx) / dx
        worst_lags = np.where(corr >= thresh)[0]
        if len(worst_lags) == 0:
            worst_lags = np.argsort(corr)[-5:]
        k_idx = worst_lags[rng.randint(len(worst_lags))]
        k = k_idx - (n - 1)
        if k >= 0:
            i_range = np.arange(n - k)
        else:
            i_range = np.arange(-k, n)
        if len(i_range) < 2:
            continue
        if k >= 0:
            scores = best_h[i_range + k] * (1.0 - best_h[i_range])
        else:
            scores = best_h[i_range] * (1.0 - best_h[i_range + k])
        top_idx = np.where(scores > 0.05)[0]
        if len(top_idx) == 0:
            continue
        idx = top_idx[rng.randint(len(top_idx))]
        i_pos = i_range[idx]
        j_pos = i_range[idx] + k
        for pos in [i_pos, j_pos]:
            if not (0 <= pos < n):
                continue
            h_try = best_h.copy()
            h_try[pos] = 1.0 - h_try[pos]
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                break
    return best_h, best_c5

def make_pert(n, cool, rng):
    ptype = rng.randint(0, 6)
    if ptype == 0:
        bs = rng.randint(4, max(5, n // 8))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.04 + 0.12 * cool) * rng.standard_normal(bs)
    elif ptype == 1:
        freq = rng.randint(2, max(3, n // 15))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif ptype == 2:
        np_ = rng.randint(4, max(5, n // 8))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.08 + 0.25 * cool) * rng.standard_normal(np_)
    elif ptype == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    elif ptype == 4:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.1 * cool * rng.standard_normal(half)
        p[half:] = -0.1 * cool * rng.standard_normal(n - half)
    else:
        freq = rng.randint(1, max(2, n // 20))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.05 + 0.1 * cool) * np.cos(2 * np.pi * freq * np.arange(n) / n + ph)
    return p

def upscale(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    rng2 = np.random.RandomState(seed + 1)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(160)
        h_init[:80] = 1.0
        h_init = project_h(h_init)

    n = len(h_init)
    best_h = project_h(h_init.copy())
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation (~90s)
    t1_end = t0 + budget_s * 0.08
    for T in [50, 200, 800, 3200, 12800, 51200, 200000, 800000]:
        if time.time() > t1_end:
            break
        hc = slsqp_step(best_h, T, n, maxiter=700, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Lag-targeting + coordinate descent (~130s)
    t2_end = t0 + budget_s * 0.19
    best_h, best_c5 = lag_target_push(best_h, n, t2_end, n_passes=400, rng=rng2)
    t2b_end = t0 + budget_s * 0.28
    best_h, best_c5 = coord_descent(best_h, n, t2b_end, max_sweeps=70, rng=rng2)

    # Phase 3: Basin hopping (~180s)
    t3_end = t0 + budget_s * 0.44
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 45
    for it in range(n_iter):
        if time.time() > t3_end:
            break
        cool = 1.0 - 0.82 * (it / n_iter)
        pert = make_pert(n, cool, rng2)
        h_pert = project_h(work_h + pert)
        T_bh = 300 + 7000 * cool
        hc = slsqp_step(h_pert, T_bh, n, maxiter=500, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.0001
        if c5v < work_c5 or rng2.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Higher resolution n=320 (~250s)
    n2 = 320
    t4_end = t0 + budget_s * 0.72
    h_up = upscale(best_h, n2)
    c5_up = compute_c5(h_up)
    if c5_up < best_c5:
        best_c5 = c5_up
        best_h = h_up
        n = n2

    for T in [200, 1000, 4000, 16000, 64000]:
        if time.time() > t4_end:
            break
        hc = slsqp_step(best_h, T, n, maxiter=500, ftol=1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h, best_c5 = lag_target_push(best_h, n, t4_end, n_passes=150, rng=rng2)

    # Phase 5: Coordinate descent + basin hopping (~200s)
    t5_end = t0 + budget_s * 0.90
    best_h, best_c5 = coord_descent(best_h, n, t5_end, max_sweeps=50, rng=rng2)

    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(30):
        if time.time() > t0 + budget_s * 0.95:
            break
        cool = 0.5 - 0.4 * (it / 30)
        pert = make_pert(n, cool, rng2)
        h_pert = project_h(work_h + pert)
        T_bh = 2000 + 6000 * cool
        hc = slsqp_step(h_pert, T_bh, n, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0006 * cool + 0.0001
        if c5v < work_c5 or rng2.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Final polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=30, rng=rng2)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
