def subgrad_step(h, n, t_end, lr=0.005, n_iters=500, top_k=5):
    """Subgradient step focused on top-k correlations."""
    dx = 2.0 / n
    nc = 2 * n - 1
    
    for it in range(n_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        
        # Find top-k correlations
        top = np.argsort(corr)[-top_k:]
        # Subgradient: average of gradients of top-k
        grad = np.zeros(n)
        for m_idx in top:
            s = m_idx - (n - 1)
            w = 1.0 / top_k
            if s > 0:
                grad[s:] += w * (1.0 - h[:n - s])
                grad[:n - s] -= w * h[s:]
            elif s < 0:
                a = -s
                grad[:n - a] += w * (1.0 - h[a:])
                grad[a:] -= w * h[:n - a]
            else:
                grad += w * (1.0 - 2.0 * h)
        
        h -= lr * dx * grad
        h = project_h(h, n)
    
    return h

def basin_hop(h, n, t_end, n_hops=300, seed=42, base_step=0.01):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    t0 = time.time()
    
    for hop in range(n_hops):
        elapsed = time.time() - t0
        remaining = t_end - time.time()
        if remaining < 1.0:
            break
        frac = hop / max(n_hops, 1)
        cool = 1.0 - frac
        step = base_step * (0.3 + 0.7 * cool)
        
        mode = hop % 4
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(2, 20)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            k = rng.randint(3, 15)
            idx = rng.choice(n, size=k, replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 1.5 * rng.randn(k)
        else:
            # Block perturbation
            b = rng.randint(10, 40)
            start = rng.randint(0, n - b)
            pert = np.zeros(n)
            pert[start:start+b] = step * rng.uniform(-1, 1, b)
        
        h_pert = project_h(cur_h + pert, n)
        # Quick CD polar
        hop_time = min(remaining * 0.3, 5.0 + 10.0 * cool)
        h_opt, c5_opt = coord_descent(h_pert, n, time.time() + hop_time, 
                                        max_sweeps=50, seed=seed + hop)
        
        temp = 0.002 * cool + 0.0002
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def f1d_corr(v, c_base, lin, old, n):
    d = v - old
    val = c_base + d * lin
    val[n - 1] -= d * d
    return float(np.max(val))

def compute_lin(p, h, n):
    k_vals = np.arange(-(n - 1), n)
    idx1 = p - k_vals
    mask1 = (idx1 >= 0) & (idx1 < n)
    contrib1 = np.zeros(2 * n - 1)
    contrib1[mask1] = 1.0 - h[idx1[mask1]]
    idx2 = p + k_vals
    mask2 = (idx2 >= 0) & (idx2 < n)
    contrib2 = np.zeros(2 * n - 1)
    contrib2[mask2] = -h[idx2[mask2]]
    return contrib1 + contrib2

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 2 == 1 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = compute_lin(i, best_h, n)
            c_base = corr.copy()
            res = minimize_scalar(f1d_corr, args=(c_base, lin, old, n),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new > best_c5 + 1e-14:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
            else:
                best_h[i] = v_new
                best_c5 = c5_new
                improved = True
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def subgrad_step(h, n, t_end, lr=0.004, n_iters=300, top_k=4):
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for m_idx in top:
            s = m_idx - (n - 1)
            w = 1.0 / len(top)
            if s > 0:
                grad[s:] += w * (1.0 - h[:n - s])
                grad[:n - s] -= w * h[s:]
            elif s < 0:
                a = -s
                grad[:n - a] += w * (1.0 - h[a:])
                grad[a:] -= w * h[:n - a]
            else:
                grad += w * (1.0 - 2.0 * h)
        h = project_h(h - lr * dx * grad, n)
    return h

def basin_hop(h, n, t_end, n_hops=300, seed=42, base_step=0.012):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        remaining = t_end - time.time()
        if remaining < 1.0:
            break
        frac = hop / max(n_hops, 1)
        cool = 1.0 - frac
        step = base_step * (0.4 + 0.6 * cool)
        mode = hop % 5
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(2, 25)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            k = rng.randint(3, 12)
            idx = rng.choice(n, size=k, replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 1.5 * rng.randn(k)
        elif mode == 3:
            b = rng.randint(8, 30)
            start = rng.randint(0, max(1, n - b))
            pert = np.zeros(n)
            pert[start:start + b] = step * rng.uniform(-1, 1, b)
        else:
            pert = step * 0.5 * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        hop_time = min(remaining * 0.25, 3.0 + 8.0 * cool)
        h_opt, c5_opt = coord_descent(h_pert, n, time.time() + hop_time,
                                       max_sweeps=60, seed=seed + hop)
        temp = 0.0015 * cool + 0.0002
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and
                rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def structured_start(n, kind, rng):
    h = np.zeros(n)
    if kind == 0:
        h[:n // 2] = 1.0
    elif kind == 1:
        h[n % 2 == 0] = 1.0
    elif kind == 2:
        m = rng.randint(2, 8)
        h = 0.5 + 0.45 * np.cos(2 * np.pi * m * np.arange(n) / n)
    elif kind == 3:
        h = 0.5 + 0.4 * rng.randn(n)
    else:
        block = n // rng.randint(4, 8)
        for i in range(0, n, block):
            if rng.rand() < 0.5:
                h[i:i + block] = 1.0
    return project_h(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(160)
        h0[:80] = 1.0
    n = len(h0)
    h0 = project_h(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: CD from initial
    t1 = t0 + budget_s * 0.08
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=800, seed=seed)

    # Phase 2: Multiple structured restarts
    t2 = t0 + budget_s * 0.15
    for kind in range(5):
        if time.time() > t2:
            break
        h_s = structured_start(n, kind, rng)
        h_s, c5_s = coord_descent(h_s, n, min(t2, time.time() + budget_s * 0.02),
                                   max_sweeps=200, seed=seed + 1000 + kind)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 3: Basin hopping (moderate)
    t3 = t0 + budget_s * 0.35
    h3, c5_3 = basin_hop(best_h, n, t3, n_hops=200, seed=seed + 1, base_step=0.015)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.38, max_sweeps=400, seed=seed + 10)

    # Phase 4: Subgradient + CD
    t4 = t0 + budget_s * 0.42
    h4 = subgrad_step(best_h, n, t4, lr=0.003, n_iters=200, top_k=3)
    h4, c5_4 = coord_descent(h4, n, t0 + budget_s * 0.45, max_sweeps=400, seed=seed + 20)
    if c5_4 < best_c5:
        best_h, best_c5 = h4, c5_4

    # Phase 5: Basin hopping (smaller steps)
    t5 = t0 + budget_s * 0.62
    h5, c5_5 = basin_hop(best_h, n, t5, n_hops=250, seed=seed + 100, base_step=0.01)
    if c5_5 < best_c5:
        best_h, best_c5 = h5, c5_5
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.65, max_sweeps=500, seed=seed + 110)

    # Phase 6: Basin hopping (fine)
    t6 = t0 + budget_s * 0.80
    h6, c5_6 = basin_hop(best_h, n, t6, n_hops=300, seed=seed + 200, base_step=0.007)
    if c5_6 < best_c5:
        best_h, best_c5 = h6, c5_6
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82, max_sweeps=500, seed=seed + 210)

    # Phase 7: Heavy final CD
    t7 = t0 + budget_s * 0.93
    best_h, best_c5 = coord_descent(best_h, n, t7, max_sweeps=5000, seed=seed + 300)

    # Phase 8: Last subgrad + CD
    if time.time() < t0 + budget_s * 0.96:
        h8 = subgrad_step(best_h, n, t0 + budget_s * 0.95, lr=0.0015, n_iters=100, top_k=3)
        h8, c5_8 = coord_descent(h8, n, t0 + budget_s * 0.97, max_sweeps=400, seed=seed + 400)
        if c5_8 < best_c5:
            best_h, best_c5 = h8, c5_8

    # Phase 9: Final polish
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=1500, seed=seed + 500)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
