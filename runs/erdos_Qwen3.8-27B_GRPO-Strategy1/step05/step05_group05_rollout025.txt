                    else: lo_i = mi1; hi_j = mj2
                hi, hj = (lo_i+hi_i)/2.0, (lo_j+hi_j)/2.0
                if abs(hi - best_h[i]) < 1e-12 and abs(hj - best_h[j]) < 1e-12:
                    continue
                h_try = best_h.copy(); h_try[i] = hi; h_try[j] = hj
                h_try = project_h(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-16:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter=60):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    best_h = h.copy()
    best_c5 = work_c5
    rng = np.random
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / float(n_iter))
        mod = it % 6
        if mod == 0:
            pert = (0.02 + 0.07 * cool) * rng.randn(n)
        elif mod == 1:
            freq = rng.randint(1, 20)
            ph = rng.uniform(0, 2 * np.pi)
            pert = (0.015 + 0.05 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = rng.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.1 * cool) * rng.randn(mask.sum())
        elif mod == 3:
            blk = max(2, n // 6)
            b = rng.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            pert[b*blk:min(n,(b+1)*blk)] = (0.02 + 0.04 * cool) * rng.randn(blk)
        elif mod == 4:
            pert = np.zeros(n)
            for _ in range(5):
                a, b2 = rng.permutation(n)[:2]
                d = (work_h[b2] - work_h[a]) * (0.2 + 0.6 * cool)
                pert[a] += d; pert[b2] -= d
        else:
            pert = (0.008 + 0.025 * cool) * rng.randn(n)
        h_pert = project_h(work_h + pert, n)
        T_bh = 500 + 15000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0006 * cool + 0.00004
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Quick SLSQP
    for T in [10000, 100000]:
        if elapsed() > budget_s * 0.08:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Extended coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.30,
                                    max_sweeps=300, use_pairs=False, niter_c=100)

    # Phase 3: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.55, n_iter=70)

    # Phase 4: SLSQP high-T polish
    for T in [100000, 500000, 2000000]:
        if elapsed() > budget_s * 0.65:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=3000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Coordinate descent with pairs (long)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.88,
                                    max_sweeps=400, use_pairs=True, niter_c=110)

    # Phase 6: Final fine basin hop
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.94, n_iter=20)

    # Phase 7: Final polish
    if elapsed() < budget_s * 0.97:
        hc = slsqp_opt(best_h, 5000000, n, maxiter=4000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99,
                                        max_sweeps=50, use_pairs=False, niter_c=120)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
