import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top10 = np.sort(corr)[-10:]
    m = top10[0]
    if T > 1e10:
        return m * dx
    return (m + np.log(np.sum(np.exp(T*(top10-m))))/T) * dx

def proj(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    d = n/2.0 - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (d/tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d/s)
    return np.clip(h, 0.0, 1.0)

def top_shifts(h, k=8):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-k:]
    return idx - (n-1), corr[idx]

def grad_top(h, shifts):
    n = len(h)
    j = np.arange(n)
    g = np.zeros(n)
    for k in shifts:
        ip = j + k
        im = j - k
        m1 = (ip >= 0) & (ip < n)
        m2 = (im >= 0) & (im < n)
        g += (1.0 - h[np.clip(ip, 0, n-1)]) * m1 - (h[np.clip(im, 0, n-1)]) * m2
    return g

def slsqp_step(h, T, n, maxiter=600):
    try:
        res = minimize(lambda x: smooth_obj(x, T), h, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0}],
                      options={'maxiter':maxiter,'ftol':1e-15})
        return proj(np.clip(res.x, 0, 1))
    except:
        return h

def subgrad_step(h, alpha, n_shifts, mom, beta):
    n = len(h)
    shifts, _ = top_shifts(h, n_shifts)
    g = grad_top(h, shifts)
    new_mom = beta*mom + (1-beta)*g
    return proj(h - alpha*new_mom), new_mom

def coord_descend(h, n, t_end, max_sweeps=100, n_top=25):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        idx = np.argsort(corr)[-n_top:]
        shifts = idx - (n-1)
        improved = False
        for i in np.random.permutation(n):
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = shifts[t]
                c = 0.0
                jp, jm = i+k, i-k
                if 0 <= jp < n: c += 1.0 - best_h[jp]
                if 0 <= jm < n: c -= best_h[jm]
                coefs[t] = c
            base = corr[idx] - best_h[i]*coefs
            res = minimize_scalar(lambda t: np.max(base + t*coefs),
                                 bounds=(0,1), method='bounded', options={'xatol':1e-13})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = proj(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:

                best_h = h_try
                improved = True
        if not improved:

    return best_h, best_c5

            bl = np.random.randint(max(n // 12, 2), max(n // 4, 3))
            bs = np.random.randint(0, max(n - bl, 1))
            p = np.zeros(n)
            p[bs:bs + bl] = sc * cl * np.random.randn(bl)
        elif r < 0.90:
            # antisymmetric w.r.t. midpoint
            p = sc * 0.6 * cl * np.random.randn(n)
            p = p - p[::-1]
        else:
            p = sc * cl * np.random.randn(n)
        hp = proj(w + p)
        T = Tl + (Th - Tl) * cl
        hc = slq(hp, T, n, 300)
        c5 = compute_c5(hc)
        tp = 0.001 * cl + 0.0002
        if c5 < wc or (c5 > wc and np.random.rand() < np.exp(-(c5 - wc) / tp)):
            w, wc = hc, c5
    return w, wc

def optn(h0, n, te):
    h = proj(h0)
    bc = compute_c5(h)
    if te - time.time() > 20:
        for T in [200, 2000, 20000, 200000, 2000000]:
            if te - time.time() < 10:
                break
            hc = slq(h, T, n, 500)
            c = compute_c5(hc)
            if c < bc:
                h, bc = hc, c
    if te - time.time() > 30:
        mom = np.zeros(n)
        for it in range(50):
            if te - time.time() < 8:
                break
            cl = 1.0 - it / 50.0
            h, mom = subg(h, 0.003 + 0.008 * cl, 5 + int(15 * cl), mom)
            hc = slq(h, 1000 + 50000 * cl, n, 250)
            c = compute_c5(hc)
            if c < bc:
                h, bc = hc, c
    if te - time.time() > 40:
        h2, c2 = bhop(h, n, te - (te - time.time()) * 0.22, 30, 0.1, 500, 50000)
        if c2 < bc:
            h, bc = h2, c2
    if te - time.time() > 18:
        h, bc = cdesc(h, n, te - (te - time.time()) * 0.22, 80, 35)
    if te - time.time() > 18:
        h2, c2 = bhop(h, n, te - (te - time.time()) * 0.22, 20, 0.03, 5000, 200000)
        if c2 < bc:
            h, bc = h2, c2
    if te - time.time() > 12:
        h, bc = cdesc(h, n, te - 4, 100, 45)
    if te - time.time() > 8:
        hc = slq(h, 5000000, n, 800)
        c = compute_c5(hc)
        if c < bc:
            h, bc = hc, c
    h = proj(h)
    return h, compute_c5(h)

def fine_polish(h, n, te):
    bh, bc = h.copy(), compute_c5(h)
    while time.time() < te - 2:
        corr = np.correlate(bh, 1.0 - bh, mode='all')
        nt = 50
        idx = np.argsort(corr)[-nt:]
        sh = idx - (n - 1)
        imp = False
        for i in np.random.permutation(n):
            if time.time() > te - 2:
                break
            co = np.zeros(nt)
            for t in range(nt):
                k = sh[t]
                c = 0.0
                jp, jm = i + k, i - k
                if 0 <= jp < n:
                    c += 1.0 - bh[jp]
                if 0 <= jm < n:
                    c -= bh[jm]
                co[t] = c
            base = corr[idx] - bh[i] * co
            r = minimize_scalar(lambda t: np.max(base + t * co),
                               bounds=(0, 1), method='bounded',
                               options={'xatol': 1e-15})
            if abs(r.x - bh[i]) < 1e-16:
                continue
            ht = proj(bh.copy())
            ht[i] = r.x
            c5 = compute_c5(ht)
            if c5 < bc - 1e-16:
                bc, bh, imp = c5, ht, True
        if not imp:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    te = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 200
        h0 = 1.0 / (1.0 + np.exp(20 * (np.linspace(0, 2, n0) - 1.0)))
    h0 = proj(h0)
    bh, bc = h0.copy(), compute_c5(h0)
    bn = n0
    # Phase 1: main at n0
    bh, bc = optn(h0, n0, t0 + budget_s * 0.42)
    bn = n0
    # Phase 2: coarser n
    if te - time.time() > budget_s * 0.18:
        n2 = max(n0 - 20, 80)
        h2 = proj(np.interp(np.linspace(0, 2, n2, endpoint=False),
                            np.linspace(0, 2, n0, endpoint=False), bh))
        h2, c2 = optn(h2, n2, t0 + budget_s * 0.58)
        if c2 < bc:
            bh, bc, bn = h2, c2, n2
    # Phase 3: finer n
    if te - time.time() > budget_s * 0.16:
        n3 = n0 + 20
        h3 = proj(np.interp(np.linspace(0, 2, n3, endpoint=False),
                            np.linspace(0, 2, bn, endpoint=False), bh))
        h3, c3 = optn(h3, n3, t0 + budget_s * 0.74)
        if c3 < bc:
            bh, bc, bn = h3, c3, n3
    # Phase 4: multi-restart
    if te - time.time() > 35:
        for sr in range(4):
            if te - time.time() < 12:
                break
            sc = 0.035 * (1.0 - 0.25 * sr)
            hp = proj(bh + sc * np.random.randn(bn))
            sub_te = te - (te - time.time()) / (4 - sr)
            hh, cc = optn(hp, bn, sub_te)
            if cc < bc:
                bh, bc = hh, cc
    # Final fine polish
    if te - time.time() > 20:
        bh, bc = fine_polish(bh, bn, te - 5)
    if te - time.time() > 8:
        hc = slq(bh, 10000000, bn, 1200)
        c = compute_c5(hc)
        if c < bc:
            bh, bc = hc, c
    bh = proj(bh)
    bc = compute_c5(bh)
    return bh, bc, len(bh)
