        if flat_pen > 0 and n_pen > 0:
            top = np.sort(corr)[-n_pen:]
            val += flat_pen * (top[-1] - top[0])
        return val * dx
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, n_sweeps=500, n_top_init=20, n_top_max=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        n_top = min(n_top_max, n_top_init + sweep // 20)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            base = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
                base[t] = corr[top_idx[t]] - best_h[i] * c
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:
                break
    return best_h, best_c5

def momentum_refine(h, n, t_end, n_iters=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    mom_v = np.zeros(n)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        n_active = max(5, int(10 + 30 * cool))
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        shifts = top_idx - (n - 1)
        i_idx = np.arange(n)[:, None]
        s_idx = shifts[None, :]
        jp = i_idx + s_idx
        jm = i_idx - s_idx
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n - 1)
        jmc = np.clip(jm, 0, n - 1)
        grad = ((1.0 - best_h[jpc]) * v1.astype(np.float64)
                - best_h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
        beta = 0.5 + 0.3 * cool
        mom_v = beta * mom_v + (1.0 - beta) * grad
        alpha = 0.0008 * (0.3 + 0.7 * cool)
        h_new = project_h(best_h - alpha * mom_v)
        T = 500 + 3000 * cool
        hc = slsqp_solve(h_new, T, n, 300, 1e-14, flat_pen=0.03 * cool, n_pen=5)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    return best_h, best_c5

def diversify_restart(h, n, scale):
    pert = scale * np.random.randn(n)
    return project_h(h + pert)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup with increasing T and flat penalty for equalization
    t_p1 = t0 + budget_s * 0.15
    for T, fp in [(50, 0.1), (200, 0.08), (1000, 0.05), (5000, 0.03), (20000, 0.01)]:
        if time.time() > t_p1:
            break
        hc = slsqp_solve(best_h, T, n, 800, 1e-15, flat_pen=fp, n_pen=5)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Momentum refinement
    t_p2 = t0 + budget_s * 0.35
    best_h, best_c5 = momentum_refine(best_h, n, t_p2, n_iters=60)

    # Phase 3: Aggressive coordinate descent
    t_p3 = t0 + budget_s * 0.65
    best_h, best_c5 = coord_descent(best_h, n, t_p3, n_sweeps=400, n_top_init=20, n_top_max=50)

    # Phase 4: SLSQP with very high T (near-exact max)
    if time.time() < t_end - budget_s * 0.15:
        t_p4 = t_end - budget_s * 0.12
        for fp in [0.02, 0.005, 0.0]:
            if time.time() > t_p4:
                break
            hc = slsqp_solve(best_h, 50000, n, 800, 1e-16, flat_pen=fp, n_pen=4)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 5: Final coordinate descent with max active constraints
    if time.time() < t_end - budget_s * 0.05:
        t_p5 = t_end - budget_s * 0.03
        best_h, best_c5 = coord_descent(best_h, n, t_p5, n_sweeps=300, n_top_init=40, n_top_max=50)

    # Phase 6: Diversified restarts (try small perturbations and re-optimize)
    if time.time() < t_end - 30:
        t_p6 = t_end - 10
        for trial in range(5):
            if time.time() > t_p6:
                break
            h_try = diversify_restart(best_h, n, 0.03)
            hc = slsqp_solve(h_try, 20000, n, 400, 1e-15, flat_pen=0.01, n_pen=5)
            hc, c5v = coord_descent(hc, n, min(time.time() + 15, t_p6), n_sweeps=80, n_top_init=30, n_top_max=50)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Final polish
    if time.time() < t_end - 3:
        hc = slsqp_solve(best_h, 1e7, n, 600, 1e-16, flat_pen=0.0, n_pen=0)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t_end - 0.5, n_sweeps=50, n_top_init=50, n_top_max=50)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
