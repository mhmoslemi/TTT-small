```python
def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    # Bisection on nu: find nu such that sum(clip(h - nu, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix rounding
    diff = target - x.sum()
    if abs(diff) > 1e-15:
        # Distribute to entries not at boundary
        mid_mask = (x > 1e-10) & (x < 1 - 1e-10)
        if mid_mask.sum() > 0:
            x[mid_mask] += diff / mid_mask.sum()
        else:
            idx = np.argmax(x) if diff < 0 else np.argmin(x)
            x[idx] += diff
        x = np.clip(x, 0, 1)
    return x
```

```python
    best_c5 = compute_c5(h_init)[0]
    h = h_init.copy()
    y = h_init.copy()
    t = 1
    alpha = 0.5  # step size
    for it in range(max_iters):
        if time.time() > t_end:
            break
        y_c = np.clip(y, 0, 1)
        # Compute softmax gradient
        corr = np.correlate(y_c, 1-y_c, mode='full')
        T = 100.0
        m = corr.max()
        p = np.exp(T*(corr - m))
        p /= p.sum()
        # Gradient of F = sum p_k * corr_k * dx
        dx = 2.0/n
        grad = np.zeros(n)
        for k in range(2*n-1):
            w = p[k]
            if w < 1e-10:
                continue
            s = k - (n-1)
            if s > 0:
                grad[s:] += w * (1 - y_c[:n-s])
                grad[:n-s] -= w * y_c[s:]
            elif s < 0:
                a = -s
                grad[:n-a] += w * (1 - y_c[a:])
                grad[a:] -= w * y_c[:n-a]
            else:
                grad += w * (1 - 2*y_c)
        grad *= dx
        
        # Nesterov step
        h_new = project_feasible(y - alpha * grad, n)
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (t - 1) / t_new
        y = h_new + beta * (h_new - h)
        h = h_new
        t = t_new
        
        c5 = compute_c5(h)[0]
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    # Find nu such that sum(clip(h-nu, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    nu = (lo + hi) / 2.0
    x = np.clip(h - nu, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        mid_mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if mid_mask.sum() > 0:
            x[mid_mask] += diff / mid_mask.sum()
        else:
            idx = int(np.argmax(np.random.random(n)) * n)
            x[idx] += np.sign(diff) * abs(diff)
        x = np.clip(x, 0.0, 1.0)
    return x

def build_L(h, n):
    L = np.zeros((2*n-1, n))
    L[n-1, :] = 1.0 - 2.0*h
    for s in range(1, n):
        L[n-1+s, :n-s] += 1.0 - h[s:]
        L[n-1+s, s:] -= h[:n-s]
        L[n-1-s, s:] += 1.0 - h[:n-s]
        L[n-1-s, :n-s] -= h[s:]
    return L

def fista_opt(h_init, n, t_end, T=30.0, max_iters=5000):
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    h = h_init.copy()
    y = h_init.copy()
    t = 1
    for it in range(max_iters):
        if time.time() > t_end:
            break
        y_c = np.clip(y, 0, 1)
        # Adjust to feasible
        y_f = project_feasible(y_c, n)
        corr = np.correlate(y_f, 1-y_f, mode='full')
        m = corr.max()
        exponents = np.clip(T*(corr - m), -50, 50)
        p = np.exp(exponents)
        p /= p.sum()
        L = build_L(y_f, n)
        grad = dx * (p @ L)
        alpha = 1.0 / (4.0 * T / n + 1e-6)
        h_new = project_feasible(y_f - alpha * grad, n)
        t_new = (1 + np.sqrt(1 + 4.0*t*t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        h = h_new
        t = t_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h_init, n, t_end, max_sweeps=500, seed=42):
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1-h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return h, best_c5
            old = h[i]
            # Build lin
            lin = np.zeros(2*n-1)
            lin[n-1] = 1.0 - 2.0*old
            for s in range(1, n):
                if i+s < n:
                    lin[n-1+s] += (1.0 - h[i+s])
                    lin[n-1-s] -= h[i+s]
                if i-s >= 0:
                    lin[n-1+s] -= h[i-s]
                    lin[n-1-s] += (1.0 - h[i-s])
            # 1D minimize
            def f1d(d):
                val = corr + d * lin
                val[n-1] -= d*d
                return val.max()
            lo = -old
            hi = 1.0 - old
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 50})
            d_opt = float(res.x)
            if abs(d_opt) < 1e-12:
                continue
            new_corr = corr + d_opt * lin
            new_corr[n-1] -= d_opt * d_opt
            c5_new = float(np.max(new_corr) * dx)
            if c5_new < best_c5 - 1e-14:
                h[i] = old + d_opt
                corr = new_corr
                best_c5 = c5_new
                improved = True
            else:
                # Revert
                pass
        if not improved:
            break
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def basin_hop(h_init, n, t_end, n_hops=100, seed=42):

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    cur_h = h_init.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.01 + 0.04 * cool
        mode = hop % 4
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 10)
            ph = rng.uniform(0, 2*np.pi)
            pert = step * np.sin(2*np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            idx = rng.choice(n, size=max(3, n//8), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 2 * rng.randn(len(idx))
        else:
            # Block perturbation
            bl = rng.randint(5, 20)
            st = rng.randint(0, n - bl)
            pert = np.zeros(n)
            pert[st:st+bl] = step * 1.5 * rng.randn(bl)
        h_pert = project_feasible(cur_h + pert, n)
        # Quick CD
        hop_end = min(t_end, time.time() + max(1.0, 4.0 * cool))
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=40, seed=seed+hop)
        # Quick FISTA

            if c5_f < c5_opt:
                h_opt, c5_opt = h_f, c5_f
        temp = 0.003 * cool + 0.0002
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt-cur_c5)/temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)  # or 2*(arange+0.5)/n
    x_new = np.linspace(0, 2, n_new)

    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    nu = (lo + hi) / 2.0
    x = np.clip(h - nu, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        mid_mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mid_mask.sum() > 0:
            x[mid_mask] += diff / mid_mask.sum()
        else:
            idx = int(np.argmax(np.abs(x - 0.5)))
            x[idx] += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def build_L(h, n):
    L = np.zeros((2 * n - 1, n))
    L[n - 1, :] = 1.0 - 2.0 * h
    for s in range(1, n):
        L[n - 1 + s, :n - s] += 1.0 - h[s:]
        L[n - 1 + s, s:] -= h[:n - s]
        L[n - 1 - s, s:] += 1.0 - h[:n - s]
        L[n - 1 - s, :n - s] -= h[s:]
    return L

def fista_opt(h_init, n, t_end, T=30.0, max_iters=3000):
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    h = h_init.copy()
    y = h_init.copy()
    t = 1
    alpha = n / (6.0 * T + 2.0)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        y_f = project_feasible(y, n)
        corr = np.correlate(y_f, 1 - y_f, mode='full')
        m = corr.max()
        p = np.exp(np.clip(T * (corr - m), -50, 50))
        p /= p.sum()
        L = build_L(y_f, n)
        grad = dx * (p @ L)
        h_new = project_feasible(y_f - alpha * grad, n)
        t_new = (1 + np.sqrt(1 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        h = h_new
        t = t_new
        if it % 10 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def coord_descent(h_init, n, t_end, max_sweeps=300, seed=42):
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        L = build_L(h, n)
        improved = False
        order = rng.permutation(n) if sweep % 5 == 4 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return h, best_c5
            old = h[i]
            lin = L[:, i]
            lo = -old
            hi = 1.0 - old
            if hi - lo < 1e-14:
                continue
            def f1d(d):
                val = corr + d * lin
                val[n - 1] -= d * d
                return val.max()
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 40})
            d_opt = float(res.x)
            if abs(d_opt) < 1e-12:
                continue
            new_corr = corr + d_opt * lin
            new_corr[n - 1] -= d_opt * d_opt
            c5_new = float(np.max(new_corr) * dx)
            if c5_new < best_c5 - 1e-15:
                h[i] += d_opt
                corr = new_corr
                best_c5 = c5_new
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def interpolate_h(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def basin_hop(h_init, n, t_end, n_hops=150, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    cur_h = h_init.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.008 + 0.035 * cool
        mode = hop % 5
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 12)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            idx = rng.choice(n, size=max(3, n // 10), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 2 * rng.randn(len(idx))
        elif mode == 3:
            bl = rng.randint(5, max(6, n // 6))
            st = rng.randint(0, n - bl)
            pert = np.zeros(n)
            pert[st:st + bl] = step * 1.5 * rng.randn(bl)
        else:
            pert = step * 0.5 * rng.randn(n)
        h_pert = project_feasible(cur_h + pert, n)
        hop_end = min(t_end, time.time() + max(0.5, 3.0 * cool))
        h_opt, c5_opt = fista_opt(h_pert, n, min(hop_end, time.time() + max(0.3, 1.5 * cool)), T=25, max_iters=300)
        if time.time() < hop_end:
            h_opt, c5_opt = coord_descent(h_opt, n, hop_end, max_sweeps=30, seed=seed + hop)
        temp = 0.003 * cool + 0.0002
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5
```
