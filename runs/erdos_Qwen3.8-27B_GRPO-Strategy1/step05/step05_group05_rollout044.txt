    corr = get_corr(h)
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    weights = e / e.sum()
    dx = 2.0 / len(h)
    val = (m + np.log(e.sum()) / T) * dx
    # Gradient: (1/T) * sum_k weights[k] * df_k/dh * dx
    # f_k(h) = sum_i h[i+k]*(1-h[i]) (for valid indices)
    # df_k/dh[j] = (1-h[j-k]) if j-k valid, plus -h[j+k] if j+k valid

        w = weights[k_idx] * (1.0/T) * dx
        if w < 1e-15:

            # if j = i: contributes -h[i+k] = -h[j+k]
            # For j in [k, n-1]: h[j] appears as h[i+k] with i=j-k, contribution (1-h[j-k])
            # For j in [0, i_max]: h[j] appears as h[i] with i=j, contribution -h[j+k]

def project_h(h):

    target = n / 2.0

    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)

def coord_opt_single(h, n, i, niter=50):
    """Optimize h[i] alone via ternary search on the max correlation."""
    h_work = h.copy()
    h_work[i] = 0.0
    corr_base = get_corr(h_work)
    m_vals = np.arange(-(n - 1), n)
    v1 = i - m_vals
    v2 = i + m_vals
    ok1 = (v1 >= 0) & (v1 < n)
    ok2 = (v2 >= 0) & (v2 < n)
    coef = np.zeros(2 * n - 1)
    coef[ok1] += (1.0 - h[np.clip(v1[ok1], 0, n-1)])
    coef[ok2] += (-h[np.clip(v2[ok2], 0, n-1)])

        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coef) < np.max(corr_base + m2 * coef):

        else:

def prox_project(v, target):
    """Project v onto {0 <= h <= 1, sum(h) = target}"""
    n = len(v)
    # Binary search for lambda
    lo, hi = v.min() - target, v.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    dx = 2.0 / n
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        frac = 1.0 + diff / s
        h = np.clip(h * frac, 0.0, 1.0)
        s2 = h.sum()
        if abs(s2 - target) > 1e-12:
            h += (target - s2) / n
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_single(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    v1 = i - m_vals
    v2 = i + m_vals
    ok1 = (v1 >= 0) & (v1 < n)
    ok2 = (v2 >= 0) & (v2 < n)
    coef = np.zeros(2 * n - 1)
    coef[ok1] += (1.0 - h[np.clip(v1[ok1], 0, n - 1)])
    coef[ok2] += (-h[np.clip(v2[ok2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coef) < np.max(corr_base + m2 * coef):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_opt_pair(h, n, i, j, niter=35):
    h_base = h.copy()
    h_base[i] = 0.0
    h_base[j] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    ci = np.zeros(2 * n - 1)
    cj = np.zeros(2 * n - 1)
    for idx in [i, j]:
        v1 = idx - m_vals
        v2 = idx + m_vals
        ok1 = (v1 >= 0) & (v1 < n)
        ok2 = (v2 >= 0) & (v2 < n)
        c = np.zeros(2 * n - 1)
        c[ok1] += (1.0 - h[np.clip(v1[ok1], 0, n - 1)])
        c[ok2] += (-h[np.clip(v2[ok2], 0, n - 1)])
        if idx == i:
            ci = c
        else:
            cj = c
    lo_i, hi_i, lo_j, hi_j = 0.0, 1.0, 0.0, 1.0
    for _ in range(niter):
        mi1 = lo_i + (hi_i - lo_i) / 3.0
        mi2 = hi_i - (hi_i - lo_i) / 3.0
        mj1 = lo_j + (hi_j - lo_j) / 3.0
        mj2 = hi_j - (hi_j - lo_j) / 3.0
        f1 = np.max(corr_base + mi1 * ci + mj1 * cj)
        f2 = np.max(corr_base + mi2 * ci + mj1 * cj)
        f3 = np.max(corr_base + mi1 * ci + mj2 * cj)
        f4 = np.max(corr_base + mi2 * ci + mj2 * cj)
        best = min(f1, f2, f3, f4)
        if best == f1:
            hi_i, hi_j = mi2, mj2
        elif best == f2:
            lo_i, hi_j = mi1, mj2
        elif best == f3:
            hi_i, lo_j = mi2, mj1
        else:
            lo_i, lo_j = mi1, mj1
    return (lo_i + hi_i) / 2.0, (lo_j + hi_j) / 2.0

def coord_descent(h, n, t_end, max_sweeps=50, use_pairs=True, pair_frac=0.15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if use_pairs and time.time() < t_end:
            n_pairs = max(3, int(n * pair_frac))
            for _ in range(n_pairs):
                if time.time() > t_end:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                hi, hj = coord_opt_pair(best_h, n, i, j)
                if abs(hi - best_h[i]) < 1e-11 and abs(hj - best_h[j]) < 1e-11:
                    continue
                h_try = best_h.copy()
                h_try[i] = hi
                h_try[j] = hj
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def shift_perturb(h, n, magnitude):
    """Circular shift by 1-3 positions, blend with original."""
    k = np.random.randint(1, 4)
    h_shift = np.roll(h, k)
    frac = magnitude
    return h * (1 - frac) + h_shift * frac

def block_flip_perturb(h, n, cool):
    """Flip a contiguous block of h to 1-h."""
    blk_len = max(2, np.random.randint(3, n // 5))
    start = np.random.randint(0, n - blk_len)
    h_new = h.copy()
    # Blend to maintain sum constraint approximately
    frac = 0.3 + 0.5 * cool
    h_new[start:start + blk_len] += frac * (1.0 - 2.0 * h[start:start + blk_len])
    return h_new
