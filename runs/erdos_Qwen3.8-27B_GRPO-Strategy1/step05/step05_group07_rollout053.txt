```python
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:

    if s > tgt:

    else:

        if t < 1e-13:

        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_desc(h, n, t_end, sweeps=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            # Compute the 1D objective
            corr = np.correlate(best, 1.0 - best, mode='full')
            B = np.zeros(2*n-1)
            for k in range(2*n-1):

                if 0 <= j1 < n:
                    B[k] -= best[j1]

            # minimize max_k (A[k] + t*B[k]) for t in [0,1]
            obj_1d = lambda t: float(np.max(A + t * B))
```

```python
    # j2 ranges from i-n+1 to i+n-1
    # Valid when 0 <= j2 < n
    k_vals = np.arange(2*n-1)
    j2 = i + k_vals - n + 1
    mask2 = (j2 >= 0) & (j2 < n)
    B[mask2] += 1.0 - h[j2[mask2]]
    
    j1 = i - k_vals + n - 1
    mask1 = (j1 >= 0) & (j1 < n)
    B[mask1] -= h[j1[mask1]]
    
    return B
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h *= (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h *= (tgt / max(s, 1e-13))
        else:
            h += rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(x, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def _sgrad(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            g[lo1:hi1] += wk * (1.0 - x[lo1 + k - n + 1: hi1 + k - n + 1])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            g[lo2:hi2] -= wk * x[lo2 - k + n - 1: hi2 - k + n - 1]
    return g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=300):
    try:
        r = minimize(h, lambda x: _sobj(x, T), lambda x: _sgrad(x, T),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=80):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr_full = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            B = np.zeros(mf)
            kv = np.arange(mf)
            j2 = i + kv - n + 1
            m2 = (j2 >= 0) & (j2 < n)
            B[m2] += 1.0 - best[j2[m2]]
            j1 = i - kv + n - 1
            m1 = (j1 >= 0) & (j1 < n)
            B[m1] -= best[j1[m1]]
            A = corr_full - old * B
            res = minimize_scalar(lambda t: float(np.max(A + t * B)),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            nv = float(res.x)
            if abs(nv - old) > 1e-12:
                trial = best.copy()
                trial[i] = nv
                trial = proj_h(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_c5 - 1e-15:
                    best = trial
                    best_c5 = c5t
                    corr_full = np.correlate(best, 1.0 - best, mode='full')
                    imp = True
        if not imp:
            break
    return best, best_c5

def binary_opt(h, n, t_end, max_iters=200000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    if len(ones) > n // 2:
        h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T = 0.02
    for it in range(max_iters):
        if time.time() > t_end:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or h_b[i] == h_b[j]:
            continue
        h_b[i], h_b[j] = h_b[j], h_b[i]
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-min((nc5 - best_c5) / T, 500)):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        T = max(0.0003, T * 0.99995)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=150):
    rng = np.random
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 5
        if mod == 0:
            pert = (0.01 + 0.12 * cool) * rng.randn(n)
        elif mod == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = (0.015 + 0.09 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = rng.random(n) < (0.15 + 0.35 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.18 * cool) * rng.randn(mask.sum())
        elif mod == 3:
            pert = np.zeros(n)
            blk = rng.randint(0, n - 10)
            L = rng.randint(5, min(20, n - blk))
            pert[blk:blk+L] = (0.05 + 0.15 * cool) * rng.randn(L)
        else:
            pert = (0.005 + 0.05 * cool) * rng.randn(n)
            idx = rng.choice(n, size=n // 4, replace=False)
            pert[idx] *= 3.0
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 8000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=200)
        ht, _ = coord_desc(ht, n, min(time.time() + 2.0, t_end), sweeps=15)
        c5t = compute_c5(ht)
        temp = 0.0005 * cool + 0.00003
        if c5t < work_c5 or np.random.rand() < np.exp(-min((c5t - work_c5) / temp, 500)):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def try_resolution(h_best, c5_best, n_best, t_mid, t_end, n_try):
    if time.time() > t_end:
        return h_best, c5_best, n_best
    try:
        x_old = np.linspace(0, 2, n_best, endpoint=False) + 1.0 / n_best
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = proj_h(h_hi, n_try)
        h_hi = lbfgs_opt(h_hi, 2000.0, n_try, iters=300)
        h_hi, c5_hi = coord_desc(h_hi, n_try, t_end, sweeps=40)
        if c5_hi < c5_best:
            return h_hi, c5_hi, n_try
    except Exception:
        pass
    return h_best, c5_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

        if time.time() - t0 > budget_s * 0.06:

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Thorough coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.14, sweeps=100)

    h_bin, c5_bin = binary_opt(best_h, n, t0 + budget_s * 0.30)
    if c5_bin < best_c5:
        h_rel = lbfgs_opt(h_bin, 3000.0, n, iters=500)
        h_rel, c5_rel = coord_desc(h_rel, n, t0 + budget_s * 0.36, sweeps=40)
        if c5_rel < best_c5:
            best_h, best_c5 = h_rel, c5_rel

    # Phase 4: Basin hopping (main search)
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.58, n_iters=180)

    # Phase 5: Coordinate descent after basin hopping
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.66, sweeps=60)

    # Phase 6: Second basin hop round (finer tuning)
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.80, n_iters=80)

    # Phase 7: Try higher resolution
    best_h, best_c5, n = try_resolution(best_h, best_c5, n, t0 + budget_s * 0.82, t0 + budget_s * 0.92, 200)

    if time.time() - t0 < budget_s * 0.93:
        best_h = lbfgs_opt(best_h, 150000.0, n, iters=600)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.985, sweeps=40)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
