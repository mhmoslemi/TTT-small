def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_c5(h, T, flat_pen=0.0, n_pen=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e10:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-n_pen:]
        val += flat_pen * np.std(top)
    return val * dx

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def topk_gradient(h, n_topk):
    """Compute subgradient of max over top-k correlations"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(float) - h[jmc] * v2.astype(float)).mean(axis=1)

def slsqp_opt(h, T, n, maxiter=600, ftol=1e-15, flat_pen=0.0):

                       bounds=[(0.0, 1.0)] * n,

                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_step(h, alpha, n_topk, mom_v, beta=0.5):

    mom_v = beta * mom_v + (1.0 - beta) * grad
    h_new = project_h(h - alpha * mom_v)
    return h_new, mom_v

    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(float) - h[jmc] * v2.astype(float)).mean(axis=1)
    return grad * (2.0 / n)

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-15, flat_pen=0.0):
    try:
        res = minimize(lambda x: soft_max_c5(x, T, flat_pen), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_step(h, alpha, n_topk, mom_v, beta=0.5):
    grad = topk_gradient(h, n_topk)
    mom_v = beta * mom_v + (1.0 - beta) * grad
    h_new = project_h(h - alpha * mom_v)
    return h_new, mom_v

def coord_descent(h, n, t_end, max_sweeps=120, n_top=15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(1.0 - it / max(n_iter, 1), 0.05)
        r = np.random.random()
        if r < 0.35:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-4:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n-k] += (work_h[k:n] - 0.5)
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n+k] - 0.5)
            pert *= scale * 0.5 * cool
        elif r < 0.55:
            f = np.random.randint(1, 10)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.7 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.75:
            blen = np.random.randint(max(n // 8, 2), max(n // 3, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs+blen] = scale * cool * np.random.randn(blen)
        elif r < 0.9:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.5 * cool * np.random.randn(max(int(mask.sum()),1))
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp_opt(h_pert, T, n, 250, 1e-14)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def pair_transfer_search(h, n, t_end, n_trials=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        delta = np.random.uniform(0.02, 0.15) * np.random.choice([-1, 1])
        h_try = best_h.copy()
        h_try[i] = np.clip(h_try[i] - delta, 0, 1)
        h_try[j] = np.clip(h_try[j] + delta, 0, 1)
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def make_periodic_init(n, num_periods, duty_frac, phase=0.0):
    x = np.linspace(0, 2, n, endpoint=False)
    period_len = 2.0 / num_periods
    h = np.zeros(n)
    for i in range(n):
        pos_in_period = (x[i] + phase) % period_len
        h[i] = 1.0 if pos_in_period < duty_frac * period_len else 0.0
    return project_h(h)

def optimize_phase(h, n, t_end, tag=""):
    best_h = project_h(h)
    best_c5 = compute_c5(best_h)

    # Continuation SLSQP: smooth to sharp
    if t_end - time.time() > 25:
        for T in [30, 100, 500, 2000, 10000, 50000, 200000]:
            if t_end - time.time() < 8:
                break
            hc = slsqp_opt(best_h, T, n, 450, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Subgradient with momentum
    if t_end - time.time() > 35:
        mom_v = np.zeros(n)
        for it in range(35):
            if t_end - time.time() < 12:
                break
            cool = 1.0 - it / 35.0
            alpha = 0.002 + 0.007 * cool
            n_topk = 3 + int(12 * cool)
            hs, mom_v = subgrad_step(best_h, alpha, n_topk, mom_v, beta=0.4)
            T = 500 + 30000 * cool
            hc = slsqp_opt(hs, T, n, 300, 1e-14, flat_pen=0.04 * cool)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Basin hopping (diverse)
    if t_end - time.time() > 45:
        t_bh = t_end - (t_end - time.time()) * 0.35
        bh_h, bh_c5 = basin_hop(best_h, n, t_bh, 30, 0.18, 500, 50000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h

    # Coordinate descent
    if t_end - time.time() > 25:
        t_cd = t_end - (t_end - time.time()) * 0.3
        best_h, best_c5 = coord_descent(best_h, n, t_cd, 100, 20)

    # Pair transfer search (novel)
    if t_end - time.time() > 20:
        t_pt = t_end - (t_end - time.time()) * 0.4
        best_h, best_c5 = pair_transfer_search(best_h, n, t_pt, 3000)

    # Fine basin hopping
    if t_end - time.time() > 20:
        t_fh = t_end - (t_end - time.time()) * 0.5
        bh_h, bh_c5 = basin_hop(best_h, n, t_fh, 15, 0.06, 5000, 80000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h

    # Final coordinate descent + SLSQP polish
    if t_end - time.time() > 12:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 4, 60, 25)
    if t_end - time.time() > 8:
        hc = slsqp_opt(best_h, 8000000, n, 800, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(160)
        init_h[:80] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase A: Optimize at parent resolution (50% budget)
    t_A_end = t0 + budget_s * 0.50
    best_h, best_c5 = optimize_phase(best_h, n, t_A_end, "A")

    # Phase B: Try coarser resolution (fewer shifts → potentially lower max)
    if t_end - time.time() > budget_s * 0.22:
        n_b = 120
        h_b = np.interp(np.linspace(0, 2, n_b, endpoint=False),
                        np.linspace(0, 2, n, endpoint=False), best_h)
        h_b = project_h(h_b)
        t_B_end = t0 + budget_s * 0.70
        h_b, c5_b = optimize_phase(h_b, n_b, t_B_end, "B")
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
            n = n_b

    # Phase C: Structure-aware restart at current best n
    if t_end - time.time() > budget_s * 0.18:
        h_c = make_periodic_init(n, 3, 0.45, np.random.uniform(0, 0.5))
        # Blend with best solution
        h_c = project_h(0.5 * h_c + 0.5 * (np.interp(
            np.linspace(0, 2, n, endpoint=False),
            np.linspace(0, 2, len(best_h), endpoint=False), best_h)))
        t_C_end = t0 + budget_s * 0.87
        h_c, c5_c = optimize_phase(h_c, n, t_C_end, "C")
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c

    # Phase D: Slightly finer resolution
    if t_end - time.time() > budget_s * 0.15:
        n_d = n + 20
        h_d = np.interp(np.linspace(0, 2, n_d, endpoint=False),
                        np.linspace(0, 2, n, endpoint=False), best_h)
        h_d = project_h(h_d)
        t_D_end = t0 + budget_s * 0.95
        h_d, c5_d = optimize_phase(h_d, n_d, t_D_end, "D")
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d
            n = n_d

    # Final polish
    if t_end - time.time() > 12:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 4, 50, 25)
    if t_end - time.time() > 6:
        hc = slsqp_opt(best_h, 10000000, n, 1000, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
