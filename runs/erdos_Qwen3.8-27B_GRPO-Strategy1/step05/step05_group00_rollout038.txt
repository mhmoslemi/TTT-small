    pos = np.arange(N)[:, None]
    sv = shifts[None, :]

            to_add = np.random.choice(zeros_idx, nh-n1, replace=False)
            h[to_add] = 1

        if time.time() > deadline:

        c = fft_corr(h)

        if cur_max < best_c:

        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        D = compute_D(h, N)

            p_list = np.random.choice(ones, sample_ones, replace=False)
        else:
            p_list = ones
        
        best_new_max = cur_max
        best_pq = None

        d4 = min(deadline - 60, time.time() + 60)
        h_tw2, c_tw2 = two_swap_search(best_h, len(best_h), d4, max_iters=2000, n_pairs=30)
        c5_tw2 = float(fft_corr(h_tw2).max() * (2.0 / len(best_h)))
        if c5_tw2 < best_c5:
            best_c5, best_h = c5_tw2, h_tw2.copy()

    # Phase 5: Final local search polish
    if time.time() < deadline - 15:
        h_bin_f = (best_h > 0.5).astype(float)
        h_f, c_f = local_search(h_bin_f, len(best_h), deadline - 2, max_iters=20000, sample_ones=None, stuck_limit=25)
        c5_f = float(fft_corr(h_f).max() * (2.0 / len(best_h)))
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    final_c5 = float(fft_corr(best_h).max() * (2.0 / len(best_h)))
    return best_h, final_c5, len(best_h)
