        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            g[lo1:hi1] += wk * (1.0 - x[lo1 + k - n + 1:hi1 + k - n + 1])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            g[lo2:hi2] -= wk * x[lo2 - k + n - 1:hi2 - k + n - 1]
    return g * 2.0 / n

def lbfgs_opt(h, T, n, iters=400):
    try:
        r = minimize(h, lambda x: _sm_obj(x, T, n), lambda x: _sm_grad(x, T, n),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _s in range(sweeps):
        if time.time() > t_end:
            break
        imp = 0
        order = np.random.permutation(n)
        for idx in range(n):
            if time.time() > t_end:
                return best, best_c5
            i = int(order[idx])
            old = best[i]
            c_full = np.correlate(best, 1.0 - best, mode='full')
            coef = np.zeros(mf)
            for k in range(mf):
                j = i + k - n + 1
                if 0 <= j < n:
                    coef[k] += 1.0 - best[j]
                j2 = i - k + n - 1
                if 0 <= j2 < n:
                    coef[k] -= best[j2]
            c_base = c_full - old * coef
            r = minimize_scalar(lambda t: np.max(c_base + t * coef),
                                bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
            nv = float(r.x)
            if abs(nv - old) < 1e-14:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                imp += 1
        if imp == 0:
            break
    return best, best_c5

def fourier_perturb(h, n, t_end, n_trials=30):
    best = h.copy()
    best_c5 = compute_c5(h)
    for p in range(n_trials):
        if time.time() > t_end:
            break
        f1 = np.random.randint(1, n // 4 + 1)
        f2 = np.random.randint(1, n // 4 + 1)
        ph1 = np.random.uniform(0, 2 * np.pi)
        ph2 = np.random.uniform(0, 2 * np.pi)
        scale = 0.03 + 0.04 * (1.0 - p / max(n_trials, 1))
        pert = scale * (np.sin(2 * np.pi * f1 * np.arange(n) / n + ph1) +
                        0.5 * np.sin(2 * np.pi * f2 * np.arange(n) / n + ph2))
        pert += scale * 0.5 * np.random.randn(n)
        hp = proj_h(h + pert, n)
        c5p = compute_c5(hp)
        if c5p < best_c5 + 0.003:
            t_sub = min(t_end, time.time() + 45)
            hp, c5p = coord_desc(hp, n, t_sub, sweeps=40)
            if c5p < best_c5:
                best = hp
                best_c5 = c5p
    return best, best_c5

def bin_sa(h, n, t_end, max_iters=300000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    tgt = n // 2
    if len(ones) > tgt:
        sel = np.random.choice(ones, len(ones) - tgt, replace=False)
        h_b[sel] = 0
    elif len(ones) < tgt:
        sel = np.random.choice(zeros, tgt - len(ones), replace=False)
        h_b[sel] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T = 0.03
    for _it in range(max_iters):
        if time.time() > t_end:
            break
        ok = False
        for _a in range(8):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if h_b[i] != h_b[j]:
                ok = True
                break
        if not ok:
            continue
        h_b[i], h_b[j] = h_b[j], h_b[i]
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        T = max(0.0003, T * 0.99997)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=80):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 4
        if mod == 0:
            pert = (0.03 + 0.1 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.08 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.2 + 0.3 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.12 * cool) * np.random.randn(mask.sum())
        else:
            pert = (0.02 + 0.05 * cool) * np.random.randn(n)
            pert += 0.03 * cool * np.sort(np.random.randn(n))
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 8000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.001 * cool + 0.00003
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS at native resolution
    for T in [30, 150, 600, 3000, 15000, 60000]:
        if time.time() - t0 > budget_s * 0.07:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.16, sweeps=120)

    # Phase 3: Fourier perturbation + re-optimize
    best_h, best_c5 = fourier_perturb(best_h, n, t0 + budget_s * 0.26, n_trials=20)

    # Phase 4: Binary SA then continuous refinement
    h_b, c5_b = bin_sa(best_h, n, t0 + budget_s * 0.34, max_iters=200000)
    if c5_b < best_c5 + 0.001:
        h_r = lbfgs_opt(h_b, 500.0, n, iters=300)
        c5_r = compute_c5(h_r)
        h_r, c5_r = coord_desc(h_r, n, t0 + budget_s * 0.40, sweeps=50)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 5: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.52, n_iters=60)

    # Phase 6: Coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.60, sweeps=80)

    # Phase 7: Try higher resolutions
    for n_hi in [200, 240]:
        if time.time() - t0 > budget_s * 0.82:
            break
        try:
            n_cur = len(best_h)
            if n_cur >= n_hi:
                continue
            x_old = np.linspace(0, 2, n_cur, endpoint=False) + 1.0 / n_cur
            x_new = np.linspace(0, 2, n_hi, endpoint=False) + 1.0 / n_hi
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = proj_h(h_hi, n_hi)
            h_hi = lbfgs_opt(h_hi, 300.0, n_hi, iters=250)
            c5_hi = compute_c5(h_hi)
            h_hi = lbfgs_opt(h_hi, 5000.0, n_hi, iters=200)
            c5_hi = compute_c5(h_hi)
            h_hi, c5_hi = coord_desc(h_hi, n_hi, t0 + budget_s * 0.78, sweeps=40)
            if c5_hi < best_c5:
                best_h, best_c5 = h_hi, c5_hi
        except Exception:
            pass

    # Phase 8: Final refinement at best resolution
    n_final = len(best_h)
    if time.time() - t0 < budget_s * 0.92:
        best_h = lbfgs_opt(best_h, 100000.0, n_final, iters=600)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_desc(best_h, n_final, t0 + budget_s * 0.98, sweeps=60)

    # Phase 9: One last perturbation attempt
    if time.time() - t0 < budget_s * 0.99:
        best_h, best_c5 = fourier_perturb(best_h, n_final, t0 + budget_s * 0.995, n_trials=5)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
