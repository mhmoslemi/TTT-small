        np_ = rng.randint(3, max(4, n // 10))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.05 + 0.15 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.01 + 0.04 * cool) * rng.standard_normal(n)
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.08 * cool * rng.standard_normal(half)
        p[half:] = -0.08 * cool * rng.standard_normal(n - half)
    return p

def pair_swap(h, n, rng, t_end, n_tries=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i, j = rng.choice(n, 2, replace=False)
        h_try = best_h.copy()
        h_try[i], h_try[j] = best_h[j], best_h[i]
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T
    for T in [50, 200, 800, 3000, 12000, 50000, 200000, 800000]:
        if elapsed() > budget_s * 0.12:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Subgradient on active set
    if elapsed() < budget_s * 0.20:
        best_h, best_c5 = subgrad_step(best_h, n, 0.01, t0 + budget_s * 0.20, max_iter=300)

    # Phase 3: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.35, max_sweeps=200)

    # Phase 4: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 60
    for it in range(n_bh):
        if elapsed() > budget_s * 0.55:
            break
        cool = 1.0 - 0.9 * (it / n_bh)
        pert = make_pert(n, cool, rng)
        h_pert = project_h(work_h + pert)
        T_bh = 500 + 15000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=800, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Coordinate descent + pair swap
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.65, max_sweeps=150)
    best_h, best_c5 = pair_swap(best_h, n, rng, t0 + budget_s * 0.68, n_tries=300)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.75, max_sweeps=150)

    # Phase 6: Subgradient + coordinate
    if elapsed() < budget_s * 0.82:
        best_h, best_c5 = subgrad_step(best_h, n, 0.005, t0 + budget_s * 0.82, max_iter=200)
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.87, max_sweeps=100)

    # Phase 7: Final SLSQP polish at very high T
    for T in [2000000, 8000000, 20000000]:
        if elapsed() > budget_s * 0.94:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=3000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 8: Final coordinate descent
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.985, max_sweeps=80)
    best_h, best_c5 = pair_swap(best_h, n, rng, t0 + budget_s * 0.99, n_tries=100)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
