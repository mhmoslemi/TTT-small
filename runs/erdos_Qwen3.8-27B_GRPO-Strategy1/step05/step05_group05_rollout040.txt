def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s

    coefs = np.zeros(len(sel_idx))
    for t, k in enumerate(k_lags):
        c = 0.0
        if 0 <= i - k and i - k < n:
            c += (1.0 - h[i - k])
        if 0 <= i + k and i + k < n:
            c += (-h[i + k])

        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, T):
    corr = get_corr(x)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / len(x))

def slsqp_opt(h, T, maxiter=2000, ftol=1e-15):

        return smooth_obj(x, T)

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))

        return h

            c5v2 = compute_c5(hc2)
            if c5v2 < best_c5:
                best_c5 = c5v2
                best_h = hc2
                best_h, best_c5 = coord_desc(best_h, n, min(time.time() + 3, t3), top_k=80, niter=25)

    # Phase 4: Basin hopping
    t4 = min(t0 + budget_s * 0.58, deadline - budget_s * 0.42)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(25):
        if time.time() > t4:
            break
        cool = 1.0 - 0.85 * (it / 25.0)
        step = 0.004 + 0.015 * cool
        mod = it % 5
        if mod == 0:
            pert = step * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 10)
            ph = np.random.uniform(0, 6.28)
            pert = step * 1.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.1
            pert = np.zeros(n)
            pert[mask] = step * 2 * np.random.randn(mask.sum())
        else:
            pert = step * 0.5 * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        hc = slsqp_opt(h_pert, 1000 + 10000 * cool, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0002 * cool + 0.00002
        if c5v < work_c5 or (np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h
    best_h, best_c5 = coord_desc(best_h, n, min(time.time() + 10, t4 + 5), top_k=100, niter=35)

    # Phase 5: High-T SLSQP
    t5 = min(t0 + budget_s * 0.68, deadline - budget_s * 0.32)
    for T in [100000, 500000, 2000000]:
        if time.time() > t5:
            break
        hc = slsqp_opt(best_h, T, maxiter=3000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Extended coord + joint
    t6 = min(t0 + budget_s * 0.85, deadline - budget_s * 0.15)
    best_h, best_c5 = coord_desc(best_h, n, t6, top_k=100, niter=35)
    if time.time() < t6:
        hc = joint_refine(best_h, n, n_free=15, T=50000, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            best_h, best_c5 = coord_desc(best_h, n, min(time.time() + 8, t6 + 5), top_k=100, niter=35)

    # Phase 7: Try n=120
    if time.time() < deadline - budget_s * 0.08:
        n120 = 120
        xs = np.linspace(0, 2, n)
        x120 = np.linspace(0, 2, n120)
        h120 = project_h(np.interp(x120, xs, best_h))
        hc120 = slsqp_opt(h120, 5000, maxiter=2000, ftol=1e-15)
        c5_120 = compute_c5(hc120)
        if c5_120 < best_c5:
            best_h, best_c5, n = hc120, c5_120, n120
            best_h, best_c5 = coord_desc(best_h, n, min(time.time() + 15, deadline - 15), top_k=60, niter=30)

    # Phase 8: Final polish
    if time.time() < deadline - 5:
        hc = slsqp_opt(best_h, 5000000, maxiter=4000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if time.time() < deadline - 3:
        best_h, best_c5 = coord_desc(best_h, n, deadline - 1, top_k=120, niter=40)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
