def grad_smooth_max(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr has length 2n-1, index n-1+m corresponds to lag m
    logsumexp = np.log(np.sum(np.exp(T * corr - T * corr.max()))) + T * corr.max()
    p = np.exp(T * corr - logsumexp)  # weights, sums to 1
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        w = p[n-1+m] * T * dx  # coefficient for this lag
        if w < 1e-15:
            continue
        # First term: (1-h[j-m]) for max(0,m) <= j < min(n, n+m)
        j_lo1 = max(0, m)
        j_hi1 = min(n, n + m)
        if j_lo1 < j_hi1:
            grad[j_lo1:j_hi1] += w * (1 - h[j_lo1 - m : j_hi1 - m])
        # Second term: -h[j+m] for max(0,-m) <= j < min(n, n-m)
        j_lo2 = max(0, -m)
        j_hi2 = min(n, n - m)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w * h[j_lo2 + m : j_hi2 + m]
    return grad

Let me use the analytical approach for coordinate descent. When we change h[i] from its current value to a new value v, the correlation changes in a structured way:

For lag m, corr_m = ∑_j h[j+m]*(1-h[j])
The terms involving h[i]:
- h[i] as h[j+m]: j = i-m, contribution is h[i]*(1-h[i-m]) if valid
- h[i] in (1-h[j]): j = i, contribution is h[i+m]*(1-h[i]) if valid

So: corr_m(v) = corr_m(0) + v*(1-h[i-m]) - v*h[i+m] (for valid indices)
            = corr_m(0) + v * [(1-h[i-m]) - h[i+m]]

where corr_m(0) is the correlation with h[i] set to 0.

So the coefficient for v in corr_m is: a_m = (1-h[i-m])*I(valid) - h[i+m]*I(valid)

Then: max_m (corr_m(0) + v * a_m) is the objective, and I need to find v ∈ [0,1] that minimizes this.

This is a 1D piecewise-linear minimization! I can solve it exactly.

The function g(v) = max_m (b_m + v*a_m) where b_m = corr_m(0) and a_m are the coefficients.
g(v) is the max of linear functions, so it's convex and piecewise linear.
The minimum is at a breakpoint or where the derivative changes sign.

The derivative of g(v) (subdifferential) at v is: {a_m : b_m + v*a_m = g(v)}
For the minimum, 0 should be in the subdifferential, i.e., there should be active constraints with both positive and negative a_m, or an active constraint with a_m = 0.

In practice, I can just do ternary search on v ∈ [0,1] using the structure:
g(v) = max_m (b_m + v*a_m)

Since it's convex in v, ternary search works! And I can evaluate g(v) in O(n) (just compute max of b + v*a).

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    shifted = T * (corr - cmax)
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()
    grad = np.zeros(n)
    for m in range(-(n - 1), n):
        w = p[n - 1 + m] * dx
        if w < 1e-14:
            continue
        i1 = max(0, m)
        i2 = min(n, n + m)
        if i1 < i2:
            grad[i1:i2] += w * (1.0 - h[i1 - m:i2 - m])
        i3 = max(0, -m)
        i4 = min(n, n - m)
        if i3 < i4:
            grad[i3:i4] -= w * h[i3 + m:i4 + m]
    return grad

def nesterov_solve(h0, T, n, niter, lr0=0.08, lr1=0.0003, t_end=None):

    prev = h0.copy()
    best = h0.copy()
    best_f = compute_c5(h0)
    for it in range(niter):
        if t_end is not None and time.time() > t_end:
            break
        t = it + 1
        mu = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t)
        lr = lr0 * (lr1 / lr0) ** (it / float(niter))
        y = x + (1.0 - mu) / mu * (x - prev)
        y = project_h(y)
        g = grad_smooth_max(y, T)
        x_new = project_h(y - lr * g)
        prev = x
        x = x_new
        if it % 300 == 0:
            f = compute_c5(x)
            if f < best_f:
                best_f = f
                best = x.copy()
    f = compute_c5(x)
    if f < best_f:
        best_f = f
        best = x.copy()
    return best, best_f

def coord_sweep(h, n, t_end=None, max_sweeps=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    n_c = 2 * n - 1
    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        changed = False
        idx = np.random.permutation(n)
        for i in idx:
            if t_end is not None and time.time() > t_end:
                return best, best_c5
            h0 = best.copy()
            h0[i] = 0.0
            c0 = np.correlate(h0, 1.0 - h0, mode='full')
            a = np.zeros(n_c)
            for m in range(-(n - 1), n):
                ci = n - 1 + m
                j = i - m
                if 0 <= j < n:
                    a[ci] += 1.0 - h0[j]
                if 0 <= i + m < n:
                    a[ci] -= h0[i + m]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                else:
