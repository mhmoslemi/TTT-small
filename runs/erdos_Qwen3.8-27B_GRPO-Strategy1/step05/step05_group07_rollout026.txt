    if abs(s - tgt) < 1e-13:
        return h

            h = np.clip(h, 0.0, 1.0)

                corr = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def bin_sa(h, n, t_end, max_iters=400000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    if len(ones) > n // 2:
        h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    dx = 2.0 / n
    best_c5 = float(np.max(np.correlate(h_b, 1.0 - h_b, mode='full')) * dx)
    best_h = h_b.copy()
    T = 0.02
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        for _a in range(10):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if h_b[i] != h_b[j]:
                break
        else:
            continue
        h_b[i] = 1.0 - h_b[i]
        h_b[j] = 1.0 - h_b[j]
        nc5 = float(np.max(np.correlate(h_b, 1.0 - h_b, mode='full')) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i] = 1.0 - h_b[i]
            h_b[j] = 1.0 - h_b[j]
        T = max(0.0002, T * 0.99993)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=100):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / max(1, n_iters))
        mod = it % 4
        if mod == 0:
            pert = (0.03 + 0.1 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 16)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.02 + 0.06 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.1 + 0.4 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.15 * cool) * np.random.randn(mask.sum())
        else:
            corr = np.correlate(work, 1.0 - work, mode='full')
            cmax = corr.max()
            crit = np.where(corr > cmax * 0.95)[0]
            pert = 0.01 * cool * np.random.randn(n)
            for ki in crit[:8]:
                sh = ki - n + 1
                lo = max(0, sh)
                hi = min(n, n + sh)
                if hi > lo:
                    pert[lo:hi] += 0.04 * cool * np.random.randn(hi - lo)
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 3000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=150)
        c5t = compute_c5(ht)
        temp = 0.0005 * cool + 0.00002
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    t1 = t0 + budget_s * 0.08
    for T in [80, 400, 2000, 8000, 30000]:
        if time.time() > t1:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    t2 = t0 + budget_s * 0.20
    best_h, best_c5 = coord_desc(best_h, n, t2, sweeps=120)

    t3 = t0 + budget_s * 0.32
    _T[0] = 5000
    _LAMBDA[0] = 0.05
    try:
        h_reg = lbfgs_opt(best_h, 5000.0, n, iters=300, lam=0.05)
        c5_reg = compute_c5(h_reg)
        if c5_reg < best_c5:
            best_h = h_reg
            best_c5 = c5_reg
            best_h, best_c5 = coord_desc(best_h, n, t3, sweeps=60)
    except Exception:
        pass
    _LAMBDA[0] = 0.0

    t4 = t0 + budget_s * 0.50
    h_bin, c5_bin = bin_sa(best_h, n, t4)
    if c5_bin < best_c5:
        best_h = lbfgs_opt(h_bin, 3000.0, n, iters=350)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.55, sweeps=80)

    t5 = t0 + budget_s * 0.72
    best_h, best_c5 = basin_hop(best_h, n, t5, n_iters=100)

    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.76, sweeps=80)

    t6 = t0 + budget_s * 0.90
    for n_try in [200, 240, 280]:
        if time.time() > t6:
            break
        try:
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_try = np.interp(x_new, x_old, best_h)
            h_try = proj_h(h_try, n_try)
            h_try = lbfgs_opt(h_try, 2000.0, n_try, iters=250)
            c5_try = compute_c5(h_try)
            h_try, c5_try = coord_desc(h_try, n_try, min(time.time() + 40, t6), sweeps=50)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                n = n_try
        except Exception:
            pass

    t_end = t0 + budget_s * 0.98
    if time.time() < t_end - 5:
        best_h = lbfgs_opt(best_h, 50000.0, len(best_h), iters=500)
        best_c5 = compute_c5(best_h)
    if time.time() < t_end - 2:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t_end, sweeps=60)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
