import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + np.minimum(room, np.abs(diff) / tot * room)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    return np.clip(h, 0.0, 1.0)

def delta_vector(h, i, n):
    lags = np.arange(-(n - 1), n)
    v1 = i - lags
    v2 = i + lags
    h1 = np.where((v1 >= 0) & (v1 < n), h[np.clip(v1, 0, n - 1)], 0.0)
    h2 = np.where((v2 >= 0) & (v2 < n), h[np.clip(v2, 0, n - 1)], 0.0)
    a = 1.0 - h1 - h2
    a[n - 1] = 1.0 - 2.0 * h[i]
    return a

def coord_sweep(h, n, corr, max_delta_passes=3):
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    order = np.random.permutation(n)

        dmin = -h[i]
        dmax = 1.0 - h[i]
        if dmax - dmin < 1e-12:
            continue
        deltas = np.linspace(dmin, dmax, 25)
        new_corrs = corr[np.newaxis, :] + deltas[:, np.newaxis] * a[np.newaxis, :]
        new_corrs[:, n - 1] -= deltas ** 2

        if new_c5 < best_c5 - 1e-15:
            h[i] += bd
            corr = new_corr
            best_c5 = new_c5
    return h, corr, best_c5

            best_c5 = new_c5

        else:

    return h, best_c5

def slsqp_smooth(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -80, 80)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def block_coord_2(h, n, corr, i, j):
    dx = 2.0 / n
    ai = delta_vector(h, i, n)
    aj = delta_vector(h, j, n)
    hi_old, hj_old = h[i], h[j]
    d1s = np.linspace(-hi_old, 1.0 - hi_old, 8)
    d2s = np.linspace(-hj_old, 1.0 - hj_old, 8)
    best_val = np.max(corr)
    best_d1, best_d2 = 0.0, 0.0
    for d1 in d1s:
        c1 = corr + d1 * ai
        c1[n - 1] -= d1 * d1
        for d2 in d2s:
            c2 = c1 + d2 * aj
            c2[n - 1] -= d2 * d2
            val = np.max(c2)
            if val < best_val:
                best_val = val
                best_d1, best_d2 = d1, d2
    h[i] += best_d1
    h[j] += best_d2
    corr = corr + best_d1 * ai + best_d2 * aj
    corr[n - 1] -= best_d1 * best_d1 + best_d2 * best_d2
    return h, corr
