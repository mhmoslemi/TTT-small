def cut_plane_iter(h, n, max_iters=50, tol=1e-10):
    """Cutting-plane method for the minimax problem."""
    target = n / 2.0
    active_lags = list(range(2*n-1))  # start with all lags
    h_cur = h.copy()
    
    for it in range(max_iters):
        # Compute current corr and gradients
        corr = np.correlate(h_cur, 1.0-h_cur, mode='full') * (2.0/n)
        t_cur = np.max(corr)
        
        # Build gradient matrix: grad[k, j] = ∂(corr_k)/∂h[j] * dx
        # For simplicity, compute numerically or analytically
        offsets = np.arange(-(n-1), n)
        grad = np.zeros((2*n-1, n))
        for k, off in enumerate(offsets):
            for j in range(n):
                g = 0.0
                jp = j + off
                jm = j - off
                if 0 <= jp < n:
                    g += (1.0 - h_cur[jp])
                if 0 <= jm < n:
                    g -= h_cur[jm]
                grad[k, j] = g * (2.0/n)
        
        # Active lags
        active = np.where(corr >= t_cur - 1e-8)[0]
        
        # Solve LP: minimize t
        # s.t. corr[k] + grad[k,:] @ (h - h_cur) <= t for k in active
        #      0 <= h <= 1
        #      sum(h) = n/2
        # Variables: h (n vars) + t (1 var) = n+1 vars
        # Using scipy.optimize.linprog
        
        # Actually, let me use scipy.optimize.linprog
        from scipy.optimize import linprog
        
        # Minimize t = c @ x where x = [h[0], ..., h[n-1], t]
        c_vec = np.zeros(n+1)
        c_vec[-1] = 1.0
        
        # Inequalities: grad[k,:] @ h - t <= -corr[k] for active k
        # i.e., grad[k,:] @ h + (-1)*t <= -corr[k]
        A_ub = np.zeros((len(active), n+1))
        b_ub = np.zeros(len(active))
        for i, k in enumerate(active):
            A_ub[i, :n] = grad[k, :]
            A_ub[i, n] = -1.0
            b_ub[i] = -corr[k]
        
        # Bounds: 0 <= h[i] <= 1, t >= 0 (or some lower bound)
        bounds = [(0, 1)]*n + [(0, None)]
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        h_new = res.x[:n]
        t_new = res.x[n]
        
        # Evaluate actual C5
        actual_c5 = compute_c5(h_new)
        
        if abs(actual_c5 - t_cur) < tol:
            break
            
        h_cur = h_new
    
    return h_cur, compute_c5(h_cur)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def projection(h, target):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = (1.0 - h).sum()
        if room < 1e-14:
            return h
        h = h + diff * (1.0 - h) / room
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def grad_corr(h, n):
    """Compute gradient of all correlation values w.r.t. h. Shape: (2n-1, n)"""
    corr_all = np.zeros((2*n-1, n))
    offsets = np.arange(-(n-1), n)
    for j in range(n):
        jp = j + offsets
        jm = j - offsets
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        g = np.zeros(2*n-1)
        g[vp] = 1.0 - h[jp[vp]]
        g[vm] -= h[jm[vm]]
        corr_all[:, j] = g
    return corr_all

def cut_plane(h, n, max_iters=100, tol=1e-10):
    """Cutting-plane method."""
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    
    for it in range(max_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        t_cur = np.max(corr)
        
        # Active lags (within tol of max)
        active = np.where(corr >= t_cur - 1e-6)[0]
        if len(active) > 50:
            # Keep only top 50
            top_idx = np.argsort(corr)[-50:]
            active = top_idx
        
        # Build gradient for active lags
        grad = grad_corr(h_cur, n)[active, :] * dx  # shape (len(active), n)
        
        # LP: min t s.t. corr[k] + grad[k]·(h-h_cur) ≤ t, 0≤h≤1, sum(h)=n/2
        # Variables: [h(0), ..., h(n-1), t]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((len(active), n + 1))
        b_ub = np.zeros(len(active))
        for i, k in enumerate(active):
            A_ub[i, :n] = grad[i, :]
            A_ub[i, n] = -1.0
            b_ub[i] = -corr[k]
        
        bounds = [(0, 1)] * n + [(t_cur - 0.5, None)]  # t lower bound
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [target]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'maxiter': 10000})
        except Exception:
            break
        
        if not res.success:
            # Try with more relaxed bounds
            break
        
        h_new = np.clip(res.x[:n], 0, 1)
        h_new = projection(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_cur - 1e-12:
            h_cur = h_new
            c5_cur = c5_new
        else:
            if c5_cur - t_cur < tol:
                break
            # Stuck, break
            break
    
    return h_cur, c5_cur

    n = len(h)

def project_h(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = h.sum()
    d = target - s
    if abs(d) < 1e-15:
        return h
    if d > 0:
        r = (1.0 - h).sum()
        if r < 1e-14:
            return h
        h += d * (1.0 - h) / r
    else:
        if s < 1e-14:
            return h
        h *= target / s
