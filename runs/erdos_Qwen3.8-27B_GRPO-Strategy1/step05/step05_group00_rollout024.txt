def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_profiles(h, positions, N):
    L = 2 * N - 1
    L_vals = np.arange(-(N - 1), N)
    profiles = np.zeros((len(positions), L))
    for i, p in enumerate(positions):
        pp = p + L_vals
        pm = p - L_vals
        mp = (pp >= 0) & (pp < N)
        mm = (pm >= 0) & (pm < N)
        profiles[i, mp] = h[pp[mp]]
        profiles[i, mm] += h[pm[mm]]
    return profiles

def binary_search(h, N, deadline, max_sweeps=500, sp=25, sq=25):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(h.sum() + 0.5)
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h > 0.5)[0]
            h[idx[np.random.permutation(len(idx))[n1 - nh:]]] = 0
        else:
            idx = np.where(h < 0.5)[0]
            h[idx[np.random.permutation(len(idx))[:nh - n1]]] = 1
    h = np.where(h > 0.5, 1.0, 0.0)
    c = fft_corr(h)
    best_pk = float(np.max(c))
    T = 0.5
    stuck = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        sp_a = min(sp, len(ones))
        sq_a = min(sq, len(zeros))
        p_c = np.random.choice(ones, sp_a, replace=False)
        q_c = np.random.choice(zeros, sq_a, replace=False)
        pro_p = compute_profiles(h, p_c, N)
        pro_q = compute_profiles(h, q_c, N)
        diffs = pro_p[:, None, :] - pro_q[None, :, :]
        all_mx = (c[None, None, :] + diffs).max(axis=2)
        gi, gj = np.unravel_index(np.argmin(all_mx), all_mx.shape)
        bm = float(all_mx[gi, gj])
        cur_pk = float(np.max(c))
        if bm < cur_pk - 1e-12 or (stuck > 30 and T > 0.02):
            p, q = int(p_c[gi]), int(q_c[gj])
            h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
            pk = float(np.max(c))
            if pk < best_pk - 1e-12:
                best_pk = pk
                stuck = 0
            else:
                stuck += 1
            T *= 0.998
        else:
            stuck += 1
            if T > 0.005:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                c = fft_corr(h)
                T *= 0.99
            elif stuck > 300:
                break
            else:
                T *= 0.995
    return h

def continuous_opt(h, N, deadline, max_iter=2000, k=15):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = 0.015
    beta = 0.9
    it = 0
    stag = 0
    while time.time() < deadline and it < max_iter:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-9:
            best_pk = pk
            best_h = h.copy()
            stag = 0
        else:
            stag += 1
        if stag > 350:
            break
        center = N - 1
        thresh = pk - 0.03 * max(pk, 0.01)
        active = np.where(corr >= thresh)[0]
        if len(active) > k:
            active = np.argsort(corr)[-k:]
        grad = np.zeros(N)
        for ai in active:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= max(len(active), 1)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 120 == 0:
            lr *= 0.95
            beta = min(0.98, 0.9 + 0.0007 * it)
    return best_h, best_pk
