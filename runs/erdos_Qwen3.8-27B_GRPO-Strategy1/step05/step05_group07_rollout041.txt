```python
def proj_subgrad(h, n, t_end, lr0=0.05, decay=0.9999, momentum=0.9):
    best_h = h.copy(); best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr0
    for it in range(500000):
        if time.time() > t_end: break
        c = np.correlate(h, 1.0-h, mode='full')
        cmax = c.max()
        active = np.where(c > cmax - 1e-8)[0]
        # Subgradient of max_k c(k) = mean of gradients at active k
        g = np.zeros(n)
        for k in active:
            d = k - n + 1
            if d >= 0:
                g[:n-d] += (1.0-h[d:])/len(active)
                g[d:] -= h[:n-d]/len(active)
            else:
                dd = -d; g[dd:] += (1.0-h[:n-dd])/len(active)
                g[:n-dd] -= h[dd:]/len(active)
        v = momentum * v + lr * g
        h_new = proj_h(h - v, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5; best_h = h_new.copy()
        h = h_new
        lr *= decay
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_CB=[np.zeros(1)]; _CC=[np.zeros(1)]

def compute_c5(h):
    n=len(h); return float(np.max(np.correlate(h,1.0-h,mode='full'))*2.0/n)

def proj_h(h,n):
    h=np.clip(h,0.0,1.0); tgt=0.5*n; s=h.sum()
    if abs(s-tgt)<1e-13: return h
    if s>tgt: h*=tgt/s
    else:
        rm=1.0-h; t=rm.sum()
        if t<1e-13: h*=tgt/max(s,1e-13)
        else: h+=rm*((tgt-s)/t)
    h=np.clip(h,0.0,1.0); d=tgt-h.sum()
    if abs(d)>1e-9:
        m=(h>0.01)&(h<0.99)
        if m.sum()>0: h[m]+=d/m.sum()
    return np.clip(h,0.0,1.0)

def _sg_obj(x,T):
    n=len(x); c=np.correlate(x,1.0-x,mode='full'); m=c.max()
    return (m+np.log(np.sum(np.exp(T*(c-m))))/T)*2.0/n

def _sg_grad(x,T):
    n=len(x); c=np.correlate(x,1.0-x,mode='full'); m=c.max()
    e=np.exp(T*(c-m)); w=e/e.sum(); g=np.zeros(n)
    for k in range(2*n-1):
        wk=w[k]
        if wk<1e-14: continue
        d=k-n+1
        if d>=0: g[:n-d]+=wk*(1.0-x[d:]); g[d:]-=wk*x[:n-d]
        else:
            dd=-d; g[dd:]+=wk*(1.0-x[:n-dd]); g[:n-dd]-=wk*x[dd:]
    return g*(2.0/n)

def lbfgs(h,T,n,iters=400):
    try:
        r=minimize(h,_sg_obj,_sg_grad,method='L-BFGS-B',
                   bounds=[(0.0,1.0)]*n,args=(T,),
                   options={'maxiter':iters,'ftol':1e-16,'gtol':1e-8})
        return proj_h(r.x,n)
    except: return h

def coord_desc(h,n,t_end,sweeps=200):
    best=h.copy(); corr=np.correlate(best,1.0-best,mode='full')
    bc5=float(corr.max())*2.0/n; mf=2*n-1
    for _ in range(sweeps):
        if time.time()>t_end: break
        imp=False
        for i in range(n):
            old=best[i]; coef=np.zeros(mf)
            for k in range(mf):
                d=k-n+1; ip=i+d; im=i-d
                if 0<=ip<n: coef[k]+=1.0-best[ip]
                if 0<=im<n: coef[k]-=best[im]
            _CB[0]=corr-old*coef; _CC[0]=coef
            r=minimize_scalar(_f1d,bounds=(0.0,1.0),method='bounded',
                              options={'xatol':1e-10})
            nv=max(0.0,min(1.0,float(r.x)))
            if abs(nv-old)<1e-12: continue
            best[i]=nv
            cn=np.correlate(best,1.0-best,mode='full')
            c5n=float(cn.max())*2.0/n
            if c5n<bc5-1e-14: bc5=c5n; corr=cn; imp=True
            else: best[i]=old
        if not imp: break
    return proj_h(best,n),bc5

    v=np.zeros(n); lr=0.02
    for it in range(layers):
