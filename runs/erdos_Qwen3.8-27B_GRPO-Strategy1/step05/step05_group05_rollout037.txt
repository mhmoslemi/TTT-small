def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def subgradient_at_lag(h, lag, n):
    """Compute subgradient of correlation at given lag w.r.t. h."""
    sg = np.zeros(n)
    if lag >= 0:
        # c = sum_{j=0}^{n-1-lag} h[j+lag]*(1-h[j])
        # d/dh[i]: (1-h[i-lag]) if 0<=i-lag<=n-1-lag, minus h[i+lag] if 0<=i<=n-1-lag and 0<=i+lag<=n-1
        for i in range(n):

            if 0 <= i - lag <= n - 1 - lag:
                sg[i] += (1.0 - h[i - lag])
            # Second term: h[i] appears as h[j] when j=i
            if 0 <= i <= n - 1 - lag and 0 <= i + lag <= n - 1:

def subgradient_descent(h, n, t_end, n_iter=50000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = get_corr(best_h)
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        
        # Find active lags (within epsilon of max)
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-6)[0]
        
        # Compute average subgradient over active lags
        sg = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            if lag >= 0:
                sg[lag:n] += (1.0 - h[0:n-lag])
                sg[0:n-lag] -= h[lag:n]
            else:
                lp = -lag
                sg[0:n-lp] += (1.0 - h[lp:n])
                sg[lp:n] -= h[0:n-lp]
        sg /= len(active)
        
        # Step size: decreasing
        step = 0.5 / (it + 10) ** 0.6
        h_new = project_h(h - step * sg)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        corr = get_corr(h)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def subgrad_vectorized(h, active_lags, n):
    """Compute subgradient averaged over active lags (given as actual lag values, not indices)."""
    sg = np.zeros(n)
    for lag in active_lags:
        if lag == 0:
            sg[:] = 1.0 - 2.0 * h  # d/dh[j] of h[j](1-h[j]) = 1-2h[j]
        elif lag > 0:
            sg[lag:n] += (1.0 - h[0:n-lag])
            sg[0:n-lag] -= h[lag:n]
        else:
            lp = -lag
            sg[0:n-lp] += (1.0 - h[lp:n])
            sg[lp:n] -= h[0:n-lp]
    sg /= len(active_lags)
    return sg

def coordinate_optimize(h, n, i, niter=80):
    """Ternary search on h[i] with others fixed."""
    h_backup = h[i]
    h[i] = 0.0
    corr_base = get_corr(h)
    
    # Coefficient vector for changing h[i]
    m_vals = np.arange(-(n-1), n)
    v1 = i - m_vals  # where h[i] appears as h[j+k]
    v2 = i + m_vals  # where h[i] appears as h[j]
    valid1 = (v1 >= 0) & (v1 < n)
    valid2 = (v2 >= 0) & (v2 < n)
    coefs = np.zeros(2*n-1)
    coefs[valid1] += (1.0 - h[np.clip(v1[valid1], 0, n-1)])
    coefs[valid2] += (-h[np.clip(v2[valid2], 0, n-1)])
    
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr_base + m1 * coefs)
        f2 = np.max(corr_base + m2 * coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    h[i] = (lo + hi) / 2.0
    return h[i]

def coord_descent(h, n, t_end, max_sweeps=200, niter=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = get_corr(best_h)

            break
        improved = False
        # Process coordinates in order of "activity" (those with values in (0.01, 0.99) first)
        mid_mask = (best_h > 0.01) & (best_h < 0.99)
        order = np.concatenate([np.where(mid_mask)[0], np.where(~mid_mask)[0]])
        
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            new_val = coordinate_optimize(best_h, n, i, niter=niter)
            if abs(new_val - best_h[i]) > 1e-14:
                h_try = best_h.copy()
                old = h_try[i]
                h_try[i] = new_val
                # Rebalance
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-16:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
    
    return best_h, best_c5

    return best_h, best_c5

def subgrad_step(h, n, t_end, n_iter=50000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step_base = 0.1
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = get_corr(h)
        max_val = np.max(corr)
        # Top 3 lags
        top_idx = np.argsort(corr)[-3:]
        active_lags = top_idx - (n-1)
        sg = np.zeros(n)
        for lag in active_lags:
            if lag == 0:
                sg[:] += 1.0 - 2.0*h
            elif lag > 0:
                sg[lag:n] += (1.0 - h[0:n-lag])
                sg[0:n-lag] -= h[lag:n]
            else:
                lp = -lag
                sg[0:n-lp] += (1.0 - h[lp:n])
                sg[lp:n] -= h[0:n-lp]
        sg /= len(active_lags)
        step = step_base / (1.0 + 0.01 * it) ** 0.7
        h_new = project_h(h - step * sg)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=2000):
    def obj(x):
        corr = get_corr(x)
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T*(corr - m))))/T
        return val * (2.0/n)
    def con(x):
        return x.sum() - n/2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter':maxiter,'ftol':1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

def structured_init(n, seed):
    """Create a structured initial guess using golden ratio spacing."""
    rng = np.random.RandomState(seed)
    # Binary pattern with golden-ratio gaps
    phi = (1 + np.sqrt(5))/2
    ones = np.zeros(n, dtype=bool)
    last = 0
    for _ in range(n//2):
        gap = int(round(2.0 * (1.0/phi)))  # average gap of 2
        # Vary gaps
        r = rng.random()
        if r < 0.3:
            gap = 1
        elif r < 0.7:
            gap = 2
        else:
            gap = 3
        pos = (last + gap) % n
        if not ones[pos]:
            ones[pos] = True
            last = pos
    # Fill remaining
    count = ones.sum()
    if count < n//2:
        remaining = np.where(~ones)[0]
        rng.shuffle(remaining)
        ones[remaining[:n//2-count]] = True
    elif count > n//2:
        remaining = np.where(ones)[0]
        rng.shuffle(remaining)
        ones[remaining[n//2:]] = False
    h = ones.astype(float)
    # Add small noise
    h = 0.95 * h + 0.05 * rng.random(n)
    return project_h(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def ternary_opt(h, n, i, niter=100):
    h[i] = 0.0
    corr_base = get_corr(h)
    mv = np.arange(-(n - 1), n)
    v1 = i - mv
    v2 = i + mv
    ok1 = (v1 >= 0) & (v1 < n)
    ok2 = (v2 >= 0) & (v2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[ok1] += (1.0 - h[np.clip(v1[ok1], 0, n - 1)])
    coefs[ok2] += (-h[np.clip(v2[ok2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=300, niter=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):

            break
        improved = False

        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            nv = ternary_opt(best_h, n, i, niter)
            if abs(nv - best_h[i]) > 1e-13:
                ht = best_h.copy()
                ht[i] = nv
                ht = project_h(ht)
                cv = compute_c5(ht)
                if cv < best_c5 - 1e-16:
                    best_h = ht
                    best_c5 = cv
                    improved = True
        if not improved and sweep > 3:
            break
    return best_h, best_c5

    best_c5 = compute_c5(best_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = get_corr(h)
        mx = np.max(corr)
        top = np.argsort(corr)[-3:]

    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=2000):
