```python
    d = tgt - s

        tr = room.sum()
        if tr < 1e-14: return h
        h += room * min(d/tr, 1.0)
```

```python
            if nm_arr[bi] < best_nm - 1e-14:
                best_nm = nm_arr[bi]
                best_pair = (i, zeros[bi])
            if time.time() > t_end: break
        if best_pair is None: break
        i,j = best_pair
        h[i]=0.0; h[j]=1.0
        c = c - D[i] + D[j]
        cur_max = c.max()
        F, S = precompute_FS(h, n)
        D = F - S
    return h, cur_max
```
