def compute_D(h, N):

    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]

    pp = pos + sv

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    L = 2 * N - 1
    shifts = np.arange(L) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    D = hp * vp.astype(float) - (1.0 - hm) * vm.astype(float)
    D[:, N - 1] = 0.0
    return D

For corr[s] = sum_i h[i+s]*(1-h[i]) where s is the shift:

∂corr[s]/∂h[j]:
- h[j] appears in h[i+s] when i+s=j, i=j-s: contribution is (1-h[j-s]) [valid if 0<=j-s<N]
- h[j] appears in (1-h[i]) when i=j: contribution is -h[j+s] [valid if 0<=j+s<N]

So ∂corr[s]/∂h[j] = (1-h[j-s]) * 1[0<=j-s<N] - h[j+s] * 1[0<=j+s<N]

For the full correlation array indexed by k=0..2N-2, the shift s = k-(N-1).

    corr = fft_corr(h)
    top_idx = np.argpartition(corr, -top_k)[-top_k:]

        if s > 0:

            grad[s:] += (1.0 - h[:N-s])
            # -h[j+s] for j in [0, N-s-1]
