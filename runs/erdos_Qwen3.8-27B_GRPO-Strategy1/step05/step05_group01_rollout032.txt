C₅ = max(np.correlate(h, 1-h, mode="full") * dx)

np.correlate(h, 1-h, mode='full') has 2n-1 elements. The element at index i corresponds to:
Σ_k h[k] * (1-h)[k + (i - (n-1))]

where the sum is over valid k. Let's say lag = i - (n-1). Then:
correlate[h, 1-h]_i = Σ_m h[m] * (1-h[m+lag]) for valid m

This is the "cross-correlation" of h with (1-h) at lag `lag`.

So C5 = max_lag [Σ_m h[m](1-h[m+lag])] * dx

For lag=0: Σ_m h[m](1-h[m]) = Σ h[m] - Σ h[m]² = n/2 - Σ h[m]² (in discrete)
So C5_0 = (n/2 - Σh²) * 2/n = 1 - 2Σh²/n

To minimize C5_0, we need to maximize Σh², i.e., make h as "peaky" as possible.

For other lags, it's more complex.

    else:
        w_all = np.exp(T * (corr - m))
        f = (m + np.log(w_all.sum()) / T) * dx
    return f, grad

def lbfgs_opt(h, T, n, maxiter, t_end, K=15):
    def callback(xk):
        return time.time() > t_end
    try:
        res = minimize(_smooth_and_grad, h, args=(T, n, K), method='L-BFGS-B',
                       jac=True, bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': 50000})
        return project_h(res.x)
    except Exception:
        return project_h(h)

def proj_grad(h, n, t_end, n_iter=400):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / n_iter
        T = 300 + 30000 * (1.0 - cool)
        K = 5 + int(15 * cool)
        f, g = _smooth_and_grad(best_h, T, n, K)
        alpha = 0.008 * (0.2 + 0.8 * cool)
        v = 0.65 * v + 0.35 * g
        h_new = project_h(best_h - alpha * v)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        else:
            h_new = project_h(best_h - 0.3 * alpha * v)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    return best_h, best_c5

def coord_descent(h, n, t_end, n_top=25, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_lags = top_idx - (n - 1)
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                lag = top_lags[t]
                c = 0.0
                jp = i + lag
                jm = i - lag
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx].copy()
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if lo >= hi:
                continue
            res = minimize_scalar(lambda d: np.max(base + d * coefs),
                                  bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-13})
            d_new = res.x
            if abs(d_new) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = best_h[i] + d_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter):
    try:
        res = minimize(_smooth_val, h, args=(T, n), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sum_constraint,
                                      'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(res.x)
    except Exception:
        return project_h(h)

def _sum_constraint(x, n):
    return x.sum() - n / 2.0

def perturb_and_opt(h, n, t_end, scale, T):
    h_p = project_h(h + scale * np.random.randn(n))
    h_p = lbfgs_opt(h_p, T, n, 200, min(time.time() + 40, t_end - 20))
    return h_p, compute_c5(h_p)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with annealing T
    for T in [200, 500, 2000, 8000, 30000, 100000]:
        if time.time() > t_end - 180:
            break
        t_sub = min(time.time() + 70, t_end - 120)
        h_new = lbfgs_opt(best_h, T, n, 400, t_sub, K=15)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Projected gradient with momentum
    t_sub = min(time.time() + 120, t_end - 120)
    best_h, best_c5 = proj_grad(best_h, n, t_sub, n_iter=500)

    # Phase 3: Coordinate descent (thorough)
    t_sub = min(time.time() + 180, t_end - 80)
    best_h, best_c5 = coord_descent(best_h, n, t_sub, n_top=25, max_sweeps=200)

    # Phase 4: SLSQP polish at high T
    if time.time() < t_end - 50:
        h_new = slsqp_polish(best_h, 80000, n, 1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Perturbation restarts
    if time.time() < t_end - 120:
        for sc in [0.05, 0.1, 0.15, 0.03]:
            if time.time() > t_end - 60:
                break
            h_p, c5_p = perturb_and_opt(best_h, n, t_end, sc, 20000)
            if c5_p < best_c5:
                best_h, best_c5 = h_p, c5_p
                best_h, best_c5 = coord_descent(best_h, n, min(time.time()+40, t_end-30), n_top=20, max_sweeps=60)

    # Phase 6: Fine coordinate descent
    if time.time() < t_end - 20:
        best_h, best_c5 = coord_descent(best_h, n, min(time.time() + 60, t_end - 5), n_top=30, max_sweeps=100)

    # Phase 7: Try n=400 refinement
    if time.time() < t_end - 100:
        n2 = 400
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        for T in [1000, 10000, 50000]:
            if time.time() > t_end - 60:
                break
            h2 = lbfgs_opt(h2, T, n2, 300, min(time.time() + 50, t_end - 40), K=12)
        h2, c5_2 = coord_descent(h2, n2, min(time.time() + 60, t_end - 10), n_top=20, max_sweeps=60)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            n = n2

    # Final polish
    if time.time() < t_end - 3:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, n_top=30, max_sweeps=40)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
