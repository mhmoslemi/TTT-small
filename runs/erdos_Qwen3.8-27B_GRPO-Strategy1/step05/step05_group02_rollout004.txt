    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr, dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # For numerical stability
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        # Need to reduce sum: scale down values > 0.5 more
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        # Fix residual
        s2 = h.sum()
        if s2 > target:
            # Remove excess from largest values
            excess = s2 - target
            sorted_idx = np.argsort(h)[::-1]
            for idx in sorted_idx:
                rem = min(excess, h[idx] - 0.0)
                h[idx] -= rem
                excess -= rem
                if excess < 1e-14:
                    break
    else:
        # Need to increase sum
        h = h + (1.0 - h) * ((target - s) / (n - s))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if s2 < target:
            deficit = target - s2
            sorted_idx = np.argsort(h)
            for idx in sorted_idx:
                add = min(deficit, 1.0 - h[idx])
                h[idx] += add
                deficit -= add
                if deficit < 1e-14:
                    break

    offset = n - 1  # correlation offset
    shift = k_star - offset  # actual shift in array indices

    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)

    # d/dh[i] sum_j h[j](1-h[j+shift]) = (1-h[i+shift]) - h[i-shift] (roughly)
    grad = np.zeros(n)
    if shift >= 0:
        valid = (np.arange(n) + shift < n)
        grad[valid] += (1.0 - h[np.arange(n)[valid] + shift])
        valid2 = (np.arange(n) - shift >= 0)
        grad[valid2] -= h[np.arange(n)[valid2] - shift]

    # Perturb in direction that reduces worst lag
    pert = -scale * grad + scale * 0.1 * rng.standard_normal(n)
    return pert

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        h = np.clip(h * (target / s), 0.0, 1.0)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = np.clip(h + room * ((target - s) / tot), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(h)[:max(1, int(abs(diff)) + 1)]
            for i in idx:
                a = min(1.0 - h[i], diff)
                h[i] += a
                diff -= a
                if abs(diff) < 1e-12:
                    break
        else:
            idx = np.argsort(h)[::-1][:max(1, int(abs(diff)) + 1)]
            for i in idx:
                r = min(h[i], -diff)
                h[i] -= r
                diff += r
                if abs(diff) < 1e-12:
                    break
    return h
