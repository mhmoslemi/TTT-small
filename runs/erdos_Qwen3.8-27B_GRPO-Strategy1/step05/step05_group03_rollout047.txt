        h_new[i] += res.x

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def all_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        mask = h > 1e-10
        if mask.sum() < 1:
            return h
        h[mask] *= (1.0 + diff / h[mask].sum())
    return np.clip(h, 0.0, 1.0)

def build_grad_matrix(h):
    """Build matrix G where G[k,i] = dc_k/dh[i]."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - (n-1)
        istart = max(0, -lag)
        iend = min(n, n - lag)
        if iend > istart:
            G[k, istart:iend] = 1.0 - h[istart+lag:iend+lag]
        jstart = max(0, istart + lag)
        jend = min(n, iend + lag)
        if jend > jstart:
            G[k, jstart:jend] -= h[jstart-lag:jend-lag]
    return G

def smooth_max_grad(h, T):
    """Smoothed max objective and gradient using precomputed-style approach."""
    n = len(h)
    dx = 2.0 / n
    corr = all_corrs(h)
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0, 0.0)
    e = np.exp(T * shifted)
    w = e / e.sum()
    obj = (m + np.log(e.sum()) / T) * dx
    # Gradient = dx * sum_k w[k] * G[k,:]^T = dx * G.T @ w
    G = build_grad_matrix(h)
    grad = dx * (G.T @ w)
    return obj, grad

def lbfgs_opt(h, n, T, maxiter=200):
    try:
        res = minimize(lambda x: smooth_max_grad(x, T)[0],
                      h, jac=lambda x: smooth_max_grad(x, T)[1],
                      method='L-BFGS-B', bounds=[(0.0,1.0)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h.copy()

def pair_swap_opt(h, n, i, j, G, corr):
    """Optimize swap h[i]+=d, h[j]-=d. Returns new h or None if no improvement."""
    d_lo = max(0, h[i] - 1.0 + h[j])  # h[i]+d <= 1 => d <= 1-h[i]
    d_hi = 1.0 - h[i]
    d_lo2 = -h[j]  # h[j]-d >= 0 => d <= h[j], i.e., d_lo >= -h[j]
    d_lo = max(d_lo, d_lo2)
    if d_lo >= d_hi:
        return None
    dg = G[:, i] - G[:, j]
    # Minimize max(corr + d*dg) for d in [d_lo, d_hi]
    # This is min of max of linear functions = convex 1D problem
    def f(d):
        return np.max(corr + d * dg)
    res = minimize_scalar(f, bounds=(d_lo, d_hi), method='bounded',
                          options={'xatol': 1e-13, 'maxiter': 40})
    if res.success and res.fun < np.max(corr) - 1e-14:
        h_new = h.copy()
        h_new[i] += res.x
        h_new[j] -= res.x
        return h_new
    return None

    if abs(diff) < 1e-14:

def build_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        is_ = max(0, -lag)
        ie = min(n, n - lag)
        if ie > is_:

        js = max(0, is_ + lag)
        je = min(n, ie + lag)
        if je > js:
            G[k, js:je] -= h[js - lag:je - lag]
    return G

    n = len(h)
    dx = 2.0 / n
    corr = all_corrs(h)
    m = np.max(corr)
    sh = np.clip(corr - m, -50.0, 0.0)
    e = np.exp(T * sh)
    return (m + np.log(e.sum()) / T) * dx

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = all_corrs(h)
    m = np.max(corr)
    sh = np.clip(corr - m, -50.0, 0.0)
    e = np.exp(T * sh)
    w = e / e.sum()
    obj = (m + np.log(e.sum()) / T) * dx
    G = build_grad_matrix(h)
    grad = dx * (G.T @ w)
    return obj, grad

        res = minimize(lambda x: smooth_max_obj(x, T), h,
                       jac=lambda x: smooth_max_grad(x, T)[1],
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

def pair_swap_round(h, n, max_passes=3):
    corr = all_corrs(h)
    G = build_grad_matrix(h)
    cur_max = np.max(corr)
    for _ in range(max_passes):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            for j in np.random.choice([x for x in range(n) if x != i], size=min(3, n-1), replace=False):
                d_lo = max(-h[j], h[i] - (1.0 - h[i]) + h[i] - 1.0 + 1.0)
                d_lo = max(-h[j], 0.0)
                d_hi = 1.0 - h[i]
                if d_lo >= d_hi - 1e-14:
                    continue
                dg = G[:, i] - G[:, j]
                def f(d):
                    return np.max(corr + d * dg)
                res = minimize_scalar(f, bounds=(d_lo, d_hi), method='bounded',
                                      options={'xatol': 1e-13, 'maxiter': 30})
                if res.success and res.fun < cur_max - 1e-14:

                    h[j] -= res.x
                    corr = corr + res.x * dg
                    cur_max = np.max(corr)
                    # Update G rows affected
                    G = build_grad_matrix(h)
                    improved = True
        if not improved:

def active_set_opt(h, n, topk=5, maxiter=50):
    corr = all_corrs(h)
    for _ in range(maxiter):
        idx = np.argsort(corr)[-topk:]
        tgt = corr[idx].max()
        if tgt - corr[idx].min() < 1e-8:
            break
        for ki in idx:

                    break
