        h_alt = project_h(h_alt, alt_n)
        c5_alt = compute_c5(h_alt)
        t_d = min(t0 + budget_s * 0.92, t0 + (budget_s * 0.12))
        if c5_alt < best_c5:
            t_d = min(t0 + budget_s * 0.92, time.time() + budget_s * 0.15)
        opt_h, opt_c5 = optimize_h(h_alt, alt_n, t0, t_d, best_c5)
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_h = opt_h

    # Final polishing on best
    final_n = len(best_h)
    t_final = t0 + budget_s * 0.99
    best_h, best_c5 = optimize_h(best_h.copy(), final_n, t0, t_final, best_c5)

    best_h = project_h(best_h, final_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, final_n
