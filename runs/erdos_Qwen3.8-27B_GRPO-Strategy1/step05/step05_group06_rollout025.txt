                h[i] -= a
                diff += a
                if abs(diff) < 1e-12: break
    return h

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= Tc.max()
    exps = np.exp(Tc)
    w = exps / exps.sum()
    logsum = np.log(exps.sum())
    sm = mc + logsum / T
    val = dx * sm + lam * (h.sum() - n/2.0)**2
    
    # Gradient: d/dh_i of Σ_k w_k * corr_k
    m_full = 2 * n - 1
    grad = np.zeros(n)
    for idx in range(m_full):
        lg = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-15: continue
        if lg == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lg > 0:
            if lg < n:
                grad[:n-lg] += wi * (1.0 - h[lg:])
                # wait, need to be careful with the derivative
                # corr[idx] = sum_{i=0}^{n-1-lg} h[i] * (1-h[i+lg])
                # d/dh[k] of this:
                #   if k is in [0, n-1-lg]: contributes (1-h[k+lg])
                #   if k is in [lg, n-1]: contributes -h[k-lg]
                # So:
                # grad[k] += wi * (1-h[k+lg]) for k in [0, n-1-lg]
                # grad[k] += wi * (-h[k-lg]) for k in [lg, n-1]
                grad[:n-lg] += wi * (1.0 - h[lg:])
                grad[lg:] += -wi * h[:n-lg]
        else:
            p = -lg
            # corr[idx] = sum_{i=0}^{n-1-p} h[i+p] * (1-h[i])
            # d/dh[k]:
            #   if k in [0, n-1-p]: contributes -h[k+p]  (from (1-h[k]) term, index i=k)
            #   if k in [p, n-1]: contributes (1-h[k-p])  (from h[k] term, index i=k-p)
            grad[:n-p] += -wi * h[p:]
            grad[p:] += wi * (1.0 - h[:n-p])
    grad = dx * grad + 2.0 * lam * (h.sum() - n/2.0)
    return val, grad

def lbfgs_step(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    res = minimize(softmax_obj_grad, x0, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
    return project_h(res.x)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h += r * d / t
        else:
            m = h > 1e-14
            if not m.any():
                break
            h[m] *= 1.0 + d / h[m].sum()
            h[~m] = 0.0
        np.clip(h, 0, 1, out=h)
    d = tgt - h.sum()
    if abs(d) > 1e-11:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        if d < 0:
            idx = idx[::-1]
        for i in idx:
            a = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
            if abs(tgt - h.sum()) < 1e-12:
                break
    return h

def sfg(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = c.max()
    tc = T * (c - mc)
    tc -= tc.max()
    e = np.exp(tc)
    w = e / e.sum()
    val = dx * (mc + np.log(e.sum()) / T) + lam * (h.sum() - n / 2.0) ** 2
    g = np.zeros(n)
    mf = 2 * n - 1
    for k in range(mf):
        lg = k - (n - 1)
        wi = w[k]
        if wi < 1e-14:
            continue
        if lg == 0:
            g += wi * (1.0 - 2.0 * h)
        elif lg > 0:
            g[:n - lg] += wi * (1.0 - h[lg:])
            g[lg:] -= wi * h[:n - lg]
        else:
            p = -lg
            g[:n - p] -= wi * h[p:]
            g[p:] += wi * (1.0 - h[:n - p])
    g = dx * g + 2.0 * lam * (h.sum() - n / 2.0)
    return val, g

def lbfgs(h0, T, lam=500.0, mi=3000, ft=1e-15):
    x0 = project_h(h0)
    try:
        r = minimize(sfg, x0, jac=True, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * len(x0),
                     options={'maxiter': mi, 'ftol': ft, 'gtol': 1e-10})
        return project_h(r.x)
    except Exception:
        return x0

def to_bin(h):
    n = len(h)
    tgt = n // 2
    b = (h >= 0.5).astype(float)
    diff = tgt - int(b.sum())
    if diff > 0:
        fr = np.where(b == 0)[0]
        order = fr[np.argsort(h[fr])[::-1]]
        for i in range(int(diff)):
            b[order[i]] = 1.0
    elif diff < 0:
        fr = np.where(b == 1)[0]
        order = fr[np.argsort(h[fr])]
        for i in range(int(-diff)):
            b[order[i]] = 0.0
    return b
