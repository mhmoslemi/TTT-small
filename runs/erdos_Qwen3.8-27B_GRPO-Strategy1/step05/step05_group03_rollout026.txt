        h_try[idx] += 0.15 * kernel * rng.choice([-1, 1])

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n - 1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] -= h[np.clip(j2[v2], 0, n - 1)]
    def f(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 40})
    return res.x

    work_h = h.copy()
    work_c5 = best_c5
    stall = 0
    for sweep in range(sweeps):
        if time.time() > t_end:
            break
        temp = base_temp * (0.92 ** (sweep / 30.0))
        order = rng.permutation(n)
        improved_sweep = 0
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(work_h, n, i)
            if abs(hi_new - work_h[i]) < 1e-12:
                continue
            h_try = work_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < work_c5 or rng.random() < np.exp(-(c5_try - work_c5) / max(temp, 1e-10)):
                work_h = h_try
                work_c5 = c5_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
                    improved_sweep += 1
                    stall = 0
        if improved_sweep == 0:
            stall += 1
            if stall > 8:

        else:

    return best_h, best_c5

def structured_perturb(h, n, rng):
    h_try = h.copy()
    choice = rng.integers(5)
    if choice == 0:
        L = int(rng.integers(3, min(14, n // 5) + 1))
        start = int(rng.integers(0, n - L))
        h_try[start:start + L] = 1.0 - h_try[start:start + L]
    elif choice == 1:
        L = int(rng.integers(3, min(12, n // 6) + 1))
        i = int(rng.integers(0, n - L))
        j = int(rng.integers(0, n - L))
        if abs(i - j) > L:
            tmp = h_try[i:i + L].copy()
            h_try[i:i + L] = h_try[j:j + L]
            h_try[j:j + L] = tmp
    elif choice == 2:
        L = int(rng.integers(6, min(24, n // 4) + 1))
        start = int(rng.integers(0, n - L))
        h_try[start:start + L] = h_try[start + L - 1:start - 1:-1].copy()
    elif choice == 3:
        center = int(rng.integers(2, n - 2))
        width = int(rng.integers(3, 10))
        idx = np.arange(max(0, center - width), min(n, center + width + 1))
        kernel = np.exp(-0.5 * ((idx - center) / (width / 2.5)) ** 2)
        h_try[idx] += 0.12 * kernel * float(rng.choice([-1, 1]))
    else:
        n_pts = int(rng.integers(3, min(10, n // 10) + 1))
        pts = rng.integers(0, n, size=n_pts)
        deltas = rng.uniform(-0.15, 0.15, size=n_pts)
        for p, d in zip(pts, deltas):
            h_try[p] += d
    return project_h(h_try, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.default_rng(seed + 7)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish
    hc = slsqp_smooth(best_h, 8000, n, maxiter=200)
    c5v = compute_c5(hc)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc

    # Phase 2: Long annealed coordinate descent
    t_end2 = t0 + budget_s * 0.38
    best_h, best_c5 = anneal_coord(best_h, n, t_end2, sweeps=500, base_temp=0.002, rng=rng)

    # Phase 3: Basin hopping with structured perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    it = 0
    t_end3 = t0 + budget_s * 0.60
    while time.time() < t_end3:
        it += 1
        cool = max(0.08, 1.0 - it / 100.0)
        h_pert = structured_perturb(work_h, n, rng)
        T_sm = int(3000 * cool + 500)
        hc = slsqp_smooth(h_pert, T_sm, n, maxiter=180)
        c5v = compute_c5(hc)
        temp = 0.0006 * cool + 0.00004
        if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Refine with coordinate descent
    t_end4 = t0 + budget_s * 0.78
    best_h, best_c5 = anneal_coord(best_h, n, t_end4, sweeps=400, base_temp=0.0008, rng=rng)

    # Phase 5: Finer basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    it = 0
    t_end5 = t0 + budget_s * 0.88
    while time.time() < t_end5:
        it += 1
        cool = max(0.05, 1.0 - it / 50.0)
        amp = 0.015 + 0.025 * cool
        h_pert = project_h(work_h + amp * rng.standard_normal(n), n)
        hc = slsqp_smooth(h_pert, int(2000 * cool + 800), n, maxiter=220)
        c5v = compute_c5(hc)
        temp = 0.00025 * cool + 0.00002
        if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: Final coordinate descent
    t_end6 = t0 + budget_s * 0.95
    best_h, best_c5 = anneal_coord(best_h, n, t_end6, sweeps=300, base_temp=0.0004, rng=rng)

    # Phase 7: High-T SLSQP for final polish
    if time.time() < t0 + budget_s * 0.98:
        for T_final in [20000, 80000, 200000]:
            if time.time() > t0 + budget_s * 0.99:
                break
            hc = slsqp_smooth(best_h, T_final, n, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Final tiny coordinate pass
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = anneal_coord(best_h, n, t0 + budget_s * 0.995,
                                       sweeps=60, base_temp=0.0001, rng=rng)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
