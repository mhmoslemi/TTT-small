```python
def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    m_val = np.max(corr)
    shifted = np.clip(T * (corr - m_val), -50.0, 50.0)
    w = np.exp(shifted)
    w /= w.sum()
    
    k_vals = np.arange(-(n-1), n)  # length 2n-1, k from -(n-1) to n-1
    grad = np.zeros(n)
    for j in range(n):
        # First term: (1-h[j-k]) for 0<=j-k<n
        # j-k >= 0 => k <= j
        # j-k < n => k > j-n (always true since k >= -(n-1) > j-n)
        # So: k <= j, i.e., k in [-(n-1), j]
        # Indices in k_vals: k = i-(n-1), so i = k+(n-1)
        # k in [-(n-1), j] => i in [0, j+(n-1)]
        
        # Second term: -h[j+k] for 0<=j+k<n
        # j+k >= 0 => k >= -j
        # j+k < n => k < n-j
        # So: k in [-j, n-j-1]
        # i in [n-1-j, n-1+n-j-1] = [n-1-j, 2n-j-2]
        
        t1 = 0.0
        valid1 = (j - k_vals >= 0) & (j - k_vals < n)
        if np.any(valid1):
            t1 = np.dot(w[valid1], 1.0 - h[j - k_vals[valid1]])
        
        t2 = 0.0
        valid2 = (j + k_vals >= 0) & (j + k_vals < n)
        if np.any(valid2):
            t2 = np.dot(w[valid2], -h[j + k_vals[valid2]])
        
        grad[j] = t1 + t2
    return grad * dx
```

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_val = np.max(corr)
    shifted = np.clip(T * (corr - m_val), -50.0, 50.0)
    w = np.exp(shifted)
    w /= w.sum()
    k_vals = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for j in range(n):
        v1 = (j - k_vals >= 0) & (j - k_vals < n)
        v2 = (j + k_vals >= 0) & (j + k_vals < n)
        g = 0.0
        if np.any(v1):
            g += np.dot(w[v1], 1.0 - h[j - k_vals[v1]])
        if np.any(v2):
            g += np.dot(w[v2], -h[j + k_vals[v2]])
        grad[j] = g
    return grad * dx

        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / n * np.ones(n)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_1d(v, corr_base, coefs, dx):
    return float(np.max(corr_base + v * coefs) * dx)

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_optimize(h, T, n, maxiter=1500):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    k_vals = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hb = best_h.copy()
            hb[i] = 0.0
            corr_base = np.correlate(hb, 1.0 - hb, mode='full')
            v1 = (i - k_vals >= 0) & (i - k_vals < n)
            v2 = (i + k_vals >= 0) & (i + k_vals < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - hb[i - k_vals[v1]])
            coefs[v2] += (-hb[i + k_vals[v2]])
            try:
                res = minimize_scalar(eval_1d, args=(corr_base, coefs, dx),
                                      bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-11})
                v_new = float(res.x)
            except Exception:
                continue
            if abs(v_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = v_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def gradient_sa(h, n, t_end, rng, T_init=0.04, sigma_init=0.06, lr_init=0.008):
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    sigma = sigma_init
    lr = lr_init
    stuck = 0
    while time.time() < t_end:
        grad = smooth_c5_grad(cur_h, 3000)
        gn = np.linalg.norm(grad)
        if gn > 1e-14:
            gdir = grad / gn
        else:
            gdir = np.zeros(n)
        noise = rng.standard_normal(n)
        nn = np.linalg.norm(noise)
        if nn > 1e-14:
            noise /= nn
        direction = 0.65 * gdir + 0.35 * noise
        dn = np.linalg.norm(direction)
        if dn > 1e-14:
            direction /= dn
        step = lr * direction + sigma * rng.standard_normal(n)
        new_h = project_h(cur_h - step)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            stuck = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            stuck += 1
            if rng.rand() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
                cur_h = new_h
                cur_c5 = new_c5
        T *= 0.9997
        sigma *= 0.99995
        lr *= 0.99995
        if stuck > 800:
            cur_h = best_h + 0.02 * rng.standard_normal(n)
            cur_h = project_h(cur_h)
            cur_c5 = compute_c5(cur_h)
            T = max(T, 0.02)
            sigma = max(sigma, 0.03)
            stuck = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation (12% budget)
    p1_end = t0 + budget_s * 0.12
    for T in [80, 400, 2000, 10000, 50000, 200000]:
        if time.time() > p1_end:
            break
        hc = slsqp_optimize(best_h, T, n, maxiter=1200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Gradient-guided SA (35% budget)
    p2_end = t0 + budget_s * 0.47
    best_h, best_c5 = gradient_sa(best_h, n, p2_end, rng, T_init=0.05, sigma_init=0.08, lr_init=0.01)

    # Phase 3: Coordinate descent (20% budget)
    p3_end = t0 + budget_s * 0.67
    best_h, best_c5 = coord_descent(best_h, n, p3_end, max_sweeps=400)

    # Phase 4: Second SA round, finer (12% budget)
    p4_end = t0 + budget_s * 0.79
    best_h, best_c5 = gradient_sa(best_h, n, p4_end, rng, T_init=0.015, sigma_init=0.02, lr_init=0.004)

    # Phase 5: Another coordinate descent pass (13% budget)
    p5_end = t0 + budget_s * 0.92
    best_h, best_c5 = coord_descent(best_h, n, p5_end, max_sweeps=300)

    # Phase 6: Final SLSQP polish
    if time.time() < t0 + budget_s * 0.96:
        for T in [300000, 1000000]:
            if time.time() > t0 + budget_s * 0.96:
                break
            hc = slsqp_optimize(best_h, T, n, maxiter=2500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Final coord descent
    if time.time() < t0 + budget_s * 0.985:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=80)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
