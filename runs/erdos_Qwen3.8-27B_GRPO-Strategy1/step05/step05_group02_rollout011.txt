                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    h_test = h_base.copy()
    h_test[i] = 1.0
    corr_test = np.correlate(h_test, 1.0 - h_test, mode='full')
    coefs = corr_test - corr_base
    dx = 2.0 / n
    def eval_1d(hi):
        return float(np.max(corr_base + hi * coefs) * dx)
    try:
        res = minimize_scalar(eval_1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    if rng is None:
        rng = np.random.RandomState(0)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def sa_move(h, n, T_curr, rng):
    mt = rng.randint(0, 6)
    h_new = h.copy()
    if mt == 0:
        i = rng.randint(n)
        h_new[i] += T_curr * 0.35 * rng.standard_normal()
    elif mt == 1:
        bs = rng.randint(4, max(5, n // 10))
        st = rng.randint(0, n - bs)
        h_new[st:st+bs] += T_curr * 0.25 * rng.standard_normal(bs)
    elif mt == 2:
        i, j = rng.choice(n, 2, replace=False)
        d = T_curr * 0.35 * rng.standard_normal()
        h_new[i] += d
        h_new[j] -= d
    elif mt == 3:
        i = rng.randint(n)
        h_new[i] = 1.0 if h_new[i] > 0.5 else 0.0
    elif mt == 4:
        i1, i2 = rng.choice(n, 2, replace=False)
        h_new[i1] += T_curr * 0.2 * rng.standard_normal()
        h_new[i2] += T_curr * 0.2 * rng.standard_normal()
    else:
        bs = rng.randint(8, max(9, n // 6))
        st = rng.randint(0, n - bs)
        g = rng.standard_normal(bs).cumsum()
        h_new[st:st+bs] += T_curr * 0.15 * (g - g.mean())
    return project_h(h_new)

def sa_loop(h, n, t_end, T0, alpha, rng):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    T = T0
    while time.time() < t_end:
        h_new = sa_move(work_h, n, T, rng)
        c5_new = compute_c5(h_new)
        if c5_new < work_c5:
            work_h = h_new
            work_c5 = c5_new
        elif T > 1e-9 and rng.rand() < np.exp(-(c5_new - work_c5) / T):
            work_h = h_new
            work_c5 = c5_new
        T *= alpha
    return work_h, work_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation
    for T in [80, 300, 1500, 6000, 30000, 200000, 1200000]:
        if elapsed() > budget_s * 0.07:
            break
        hc = slsqp_step(best_h, T, n, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.14, max_sweeps=80, rng=rng)

    # Phase 3: SA exploration
    work_h, work_c5 = sa_loop(best_h, n, t0 + budget_s * 0.32, 0.06, 0.9965, rng)
    work_h, work_c5 = coord_descent(work_h, n, t0 + budget_s * 0.40, max_sweeps=70, rng=rng)
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Multi-basin SLSQP + SA
    for restart in range(4):
        if elapsed() > budget_s * 0.52:
            break
        cool = 1.0 - 0.3 * restart
        bs = rng.randint(6, max(7, n // 8))
        st = rng.randint(0, n - bs)
        h_pert = best_h.copy()
        h_pert[st:st+bs] += (0.03 + 0.10 * cool) * rng.standard_normal(bs)
        h_pert = project_h(h_pert)
        T_bh = 500 + 4000 * cool
        hc = slsqp_step(h_pert, T_bh, n, maxiter=800, ftol=1e-15)
        c5v = compute_c5(hc)
        work_h2, work_c5_2 = sa_loop(hc, n, t0 + budget_s * 0.52, 0.03, 0.997, rng)
        work_h2, work_c5_2 = coord_descent(work_h2, n, t0 + budget_s * 0.52, max_sweeps=40, rng=rng)
        if work_c5_2 < best_c5:
            best_c5 = work_c5_2
            best_h = work_h2

    # Phase 5: Upsample to 2n and optimize
    n2 = n * 2
    if elapsed() < budget_s * 0.72:
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_h(h_up)
        c5_up = compute_c5(h_up)
        for T in [200, 2000, 20000, 200000]:
            if elapsed() > budget_s * 0.62:
                break
            hc = slsqp_step(h_up, T, n2, maxiter=600, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < c5_up:
                c5_up = c5v
                h_up = hc
        h_up, c5_up = coord_descent(h_up, n2, t0 + budget_s * 0.68, max_sweeps=40, rng=rng)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up
            n = n2

    # Phase 6: Final SA + coord descent
    work_h, work_c5 = sa_loop(best_h, n, t0 + budget_s * 0.82, 0.025, 0.998, rng)
    work_h, work_c5 = coord_descent(work_h, n, t0 + budget_s * 0.88, max_sweeps=80, rng=rng)
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 7: High-T SLSQP polish
    for T in [500000, 2000000, 10000000]:
        if elapsed() > budget_s * 0.95:
            break
        hc = slsqp_step(best_h, T, n, maxiter=2000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 8: Final coord descent
    if elapsed() < budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.985, max_sweeps=50, rng=rng)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
