import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s < 1e-14:
            return h
        h = np.clip(h * (1.0 + diff / s), 0, 1)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e6:
        return m * dx
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def correct_subgrad(h, n_topk, alpha, mom_v=None, beta=0.3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_topk:]
    
    grad = np.zeros(n)
    for k in top_idx:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += (1.0 - 2.0 * h[k - idx])
    
    grad *= dx / n_topk
    
    if mom_v is not None:
        grad = beta * mom_v + (1.0 - beta) * grad
    
    h_new = project_h(h - alpha * grad)
    return h_new, grad

def coord_descent(h, n, t_end, max_sweeps=50, n_top=25):

                # h[i] appears as h[n] when n=i: contributes (1-h[k-i]) to linear part
                # h[i] appears in (1-h[k-n]) when k-n=i, n=k-i: contributes -h[k-i]
                # If k-i == i (k=2i): the term is h[i]*(1-h[i]) = h[i] - h[i]^2
                #   This is the only term involving h[i] in c[k] for this k
                #   c[k] = ... + (h[i] - h[i]^2) + ... where ... doesn't involve h[i]
                #   So c[k] = A - h[i]^2 + h[i] + B where A,B don't involve h[i]
                #   Actually: c[k] includes the term h[i]*(1-h[i]) = h[i] - h[i]^2
                #   Other terms don't involve h[i]
                #   So c[k] = (sum of other terms) + h[i] - h[i]^2
                #   As function of h[i]: -h[i]^2 + h[i] + const
                #
                # If k-i != i: h[i] appears linearly
                #   c[k] = (sum of terms not involving h[i]) + h[i]*(1-h[k-i]) + h[k-i]*(1-h[i])
                #          = const + h[i] - h[i]*h[k-i] + h[k-i] - h[i]*h[k-i]
                #   Wait no. h[k-i]*(1-h[i]) is a SEPARATE term in the sum (with n=k-i)
                #   And h[i]*(1-h[k-i]) is a term with n=i
                #   So c[k] = [other terms] + h[i]*(1-h[k-i]) + h[k-i]*(1-h[i])
                #            = [other terms] + h[i] - h[i]*h[k-i] + h[k-i] - h[i]*h[k-i]
                #   Wait, h[k-i]*(1-h[i]) = h[k-i] - h[i]*h[k-i]
                #   And h[i]*(1-h[k-i]) = h[i] - h[i]*h[k-i]
                #   Sum: h[i] + h[k-i] - 2*h[i]*h[k-i]
                #   This is QUADRATIC in h[i]! coefficient of h[i]^2 is 0, coefficient of h[i] is (1 + h[k-i] - 2*h[k-i]) = 1 - h[k-i], constant is h[k-i] - h[k-i] = ... 
                #   Wait: h[i] + h[k-i] - 2*h[i]*h[k-i]
                #   = h[i]*(1 - 2*h[k-i]) + h[k-i]
                #   This IS linear in h[i]! (The term h[i]*h[k-i] has h[k-i] as a constant since we're varying h[i])
                #   
                #   Oh wait, I was confusing myself. When we vary h[i], h[k-i] is held fixed (it's a different coordinate). So:
                #   h[i]*(1-h[k-i]) + h[k-i]*(1-h[i]) = h[i] - h[i]*h[k-i] + h[k-i] - h[i]*h[k-i]
                #   = h[i](1+h[k-i]) - 2*h[i]*h[k-i] ... no
                #   = h[i] - h[i]*h[k-i] + h[k-i] - h[i]*h[k-i]
                #   = h[i](1 - h[k-i]) + h[k-i](1 - h[i])
                #   = h[i] - h[i]*h[k-i] + h[k-i] - h[i]*h[k-i]
                #   Hmm, let me just expand: let a=h[i], b=h[k-i] (b is constant)
                #   a*(1-b) + b*(1-a) = a - ab + b - ab = a + b - 2ab
                #   As function of a: a(1-2b) + b. LINEAR in a! ✓
                #
                # For k=2i (middle term): the single term is h[i]*(1-h[i]) = h[i] - h[i]^2
                #   As function of h[i]: -h[i]^2 + h[i] + 0. QUADRATIC!
            
                if k - i == i and 0 <= k - i < n:

                    coefs_a[t_idx] = -1.0  # coefficient of h[i]^2
                    coefs_b[t_idx] = 1.0   # coefficient of h[i]
                    quad_k = t_idx
                else:
                    # Linear: (1 - 2*h[k-i]) * h[i] + h[k-i] (if both terms valid)
                    # or (1-h[k-i]) * h[i] (if only n=i term valid)
                    # or -h[k-i] * h[i] (if only n=k-i term valid)
                    # Actually, let me just use: c[k] = base + slope * h[i]
                    # where slope = d c[k] / d h[i] = (1-2*h[k-i]) if 0<=k-i<n
                    # and base = c[k] - slope * h[i]
                    if 0 <= k - i < n:
                        slope = 1.0 - 2.0 * h[k - i]
                    else:
                        slope = 0.0
                    coefs_b[t_idx] = corr[k] - slope * best_h[i]
                    coefs_a[t_idx] = slope
            
            # Now minimize max_t (coefs_a[t]*t_val + coefs_b[t]) * dx over t_val in [0,1]
            # For the quadratic case (quad_k >= 0):
            #   objective at q: max_t (coefs_a[t]*t_val^2 + ...) -- this is more complex
            # Let me just evaluate: for each t in [0,1], compute max_k (linear_or_quad) * dx
            # Use minimize_scalar with the actual evaluation
            
            def obj_at(v):
                h_try = best_h.copy()
                h_try[i] = v
                # Only need to recompute the top correlations
                # c[k] = original + delta where delta = (v - old_h[i]) * slope_k
                # For the quadratic term: delta = (v*(1-v) - old*(1-old)) = (v-old) - (v^2-old^2)
                # Actually simpler: just recompute c[k] for the top k values
                for t_idx in range(n_top):
                    k = top_idx[t_idx]
                    if quad_k == t_idx:
                        # c[k] changes by: v*(1-v) - best_h[i]*(1-best_h[i])
                        corr[t_idx] = corr[k] - best_h[i]*(1-best_h[i]) + v*(1-v)
                    else:
                        if 0 <= k - i < n:
                            slope = 1.0 - 2.0 * best_h[k - i]
                        else:
                            slope = 0.0
                        corr[t_idx] = corr[k] + slope * (v - best_h[i])
                # Also check if any OTHER correlation became the max
                # For safety, just compute the full max
                return float(np.max(corr) * (2.0/n))
            
            res = minimize_scalar(obj_at, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            t_new = res.x
            if abs(t_new - best_h[i]) > 1e-13:
                h_try = best_h.copy()
                h_try[i] = t_new
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5
