This doesn't decompose nicely, so we need to check all pairs. For N=380, that's 190*190=36100 pairs, each requiring a max over 759 elements. Total: ~27M operations. In numpy, this should be fast if vectorized.

Vectorized approach:

- zeros: indices where h=0 (size n0 ≈ N/2)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    return np.fft.irfft(H * np.conj(G), L)

def compute_c5(h, N):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / N)

def compute_flip_effects(h, N):
    corr_len = 2 * N - 1
    center = N - 1
    D = np.zeros((N, corr_len))
    for p in range(N):
        # h[p-s] for s where 0 <= p-s < N, i.e., s in [p-(N-1), p]
        s_lo = max(p - N + 1, -(N-1))
        s_hi = min(p, N-1)
        s = np.arange(s_lo, s_hi + 1)
        k = s + center
        D[p, k] += h[p - s]
        # -(1-h[p+s]) for s where 0 <= p+s < N, i.e., s in [-p, N-1-p]
        s_lo2 = max(-p, -(N-1))
        s_hi2 = min(N-1-p, N-1)
        s2 = np.arange(s_lo2, s_hi2 + 1)
        k2 = s2 + center
        D[p, k2] += -(1.0 - h[p + s2])
    return D

def quantize(h, N):
    b = np.zeros(N)
    thresh = np.partition(h, N//2)[N//2]
    b = (h >= thresh).astype(float)
    n1 = int(b.sum())
    half = N // 2
    if n1 > half:
        idx = np.where(b == 1)[0]
        # Remove extra ones at lowest values
        idx_sorted = idx[np.argsort(h[idx])]
        b[idx_sorted[:n1-half]] = 0
    elif n1 < half:
        idx = np.where(b == 0)[0]
        idx_sorted = idx[np.argsort(-h[idx])]
        b[idx_sorted[:half-n1]] = 1
    return b

                best_pq = (int(p), int(zeros[qi]))
                best_new_max = new_max

    it = 0

        h_new = h_new * (target / s)
