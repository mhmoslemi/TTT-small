from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 3 == 2 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                if i <= n - 1 - s:
                    lin[mp] -= best_h[i + s]
                if i <= n - 1 - s:
                    lin[mn] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[mn] -= best_h[i - s]
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent on parent (18%)
    t1 = t0 + budget_s * 0.18
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=5000, seed=seed)

    # Phase 2: Diverse restarts (22%)
    t2 = t0 + budget_s * 0.40
    n_div = 12
    time_per_div = max(0.5, (t2 - time.time()) / n_div)
    for d in range(n_div):
        if time.time() > t2:
            break
        h_div = diverse_start(n, rng, d % 5)
        h_div += 0.05 * rng.randn(n)
        h_div = project_h(h_div, n)
        t_end_d = min(t2, time.time() + time_per_div * 1.5)
        h_opt, c5_opt = coord_descent(h_div, n, t_end_d, max_sweeps=800, seed=seed + 100 + d)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 3: Gradient-based escape + coord (12%)
    t3 = t0 + budget_s * 0.52
    if time.time() < t3:
        h_g = best_h.copy()
        mom = np.zeros(n)
        beta = 0.7
        for it in range(2000):
            if time.time() > t3:
                break
            frac = min(1.0, it / 2000.0)
            T = 30.0 * 30.0 ** frac
            g = soft_grad(h_g, T, n)
            mom = beta * mom + g
            gm = np.max(np.abs(mom))
            if gm < 1e-13:
                break
            alpha = 0.3 / gm
            h_new = project_h(h_g - alpha * mom, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new.copy()
            h_g = h_new
        best_h, best_c5 = coord_descent(best_h, n, min(t3 + 5, t0 + budget_s * 0.55), max_sweeps=1000, seed=seed + 50)

    # Phase 4: Basin hopping (main exploration, 28%)
    t4 = t0 + budget_s * 0.83
    if time.time() < t4:
        best_h, best_c5 = basin_hopping(best_h, n, t4, n_hops=500, seed=seed + 200)

    # Phase 5: Final intensive coordinate descent (14%)
    t5 = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=8000, seed=seed + 400)

    # Phase 6: Tiny polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=500, seed=seed + 500)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
