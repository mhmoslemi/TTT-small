def compute_subgradient(h, n, T=1000):

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # Soft weights
    w = np.exp(T * (corr - max_corr))
    w /= w.sum()
    
    # Sum of weighted gradients
    grad = np.zeros(n)

        if w[k] < 1e-10:

            # c(lag) = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])

def project_feasible(x, n):

    lo = x.min() - 1.0

        if s > target:
            lo = mid
        else:

def compute_subgradient(h, n, T=100.0):

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    w = np.exp(T * (corr - max_c))

        else:

    return grad * dx

        td_cd = min(t3, time.time() + 4.0)
        hc, c5v = coord_desc(hc, n, td_cd, max_sweeps=25)
        temp = 0.0012 * cool + 0.00004
        if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 4: Extended coordinate descent (most critical phase)
    t4 = t0 + budget_s * 0.94
    best_h, best_c5 = coord_desc(best_h, n, t4, max_sweeps=300)

    # Phase 5: Final polish
    if time.time() < t0 + budget_s * 0.97:
        hc, c5v = grad_descent(best_h.copy(), n, t0 + budget_s * 0.97,
                               500.0, 8000.0, 2000)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.995, max_sweeps=60)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
