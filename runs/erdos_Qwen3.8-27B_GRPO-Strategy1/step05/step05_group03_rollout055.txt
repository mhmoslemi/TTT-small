    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        frac = diff / tot
        h = h + room * frac
        h = np.clip(h, 0.0, 1.0)
        # Re-normalize
        s2 = h.sum()
        if abs(s2 - target) > 1e-10:
            h = h + (target - s2) / n
            h = np.clip(h, 0.0, 1.0)
    else:
        if s < 1e-14:
            return np.full(n, 0.5)
        h = h * (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)

    w /= ws

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with gradient, high-T to low-T
    t1 = t0 + budget_s * 0.12
    for T in [10000, 5000, 2000, 800, 300, 100]:
        if time.time() > t1:
            break
        hc = slsqp_opt(best_h, n, T, maxiter=500)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc

    # Phase 2: Coordinate descent polish
    t2 = t0 + budget_s * 0.25
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=120)

    # Phase 3: Subgradient + SLSQP refinement
    t3 = t0 + budget_s * 0.40
    for it in range(30):
        if time.time() > t3:
            break
        cool = 1.0 - it / 30.0
        alpha = 0.002 + 0.008 * cool
        k = 3 + int(5 * cool)
        hs = subgrad_topk(best_h, n, k, alpha)
        T = 300 + 2000 * cool
        hc = slsqp_opt(hs, n, T, maxiter=400)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc

    # Phase 4: Multi-scale basin hopping
    t4 = t0 + budget_s * 0.58
    bh_h, bh_c5 = multi_scale_bh(best_h, best_c5, n, t4)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = bh_h

    # Phase 5: Coordinate descent
    t5 = t0 + budget_s * 0.70
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=100)

    # Phase 6: Try n=200 with upsampled start
    t6 = t0 + budget_s * 0.84
    if time.time() < t6:
        n2 = 200
        h2 = project_h(upsample(best_h, n, n2), n2)
        c5_2 = compute_c5(h2)
        for T in [4000, 1500, 600, 200]:
            if time.time() > t6:
                break
            h2c = slsqp_opt(h2, n2, T, maxiter=400)
            cv2 = compute_c5(h2c)
            if cv2 < c5_2:
                c5_2 = cv2
                h2 = h2c
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            n = n2
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.88, max_sweeps=40)

    # Phase 7: Final SLSQP + coordinate descent
    t7 = t0 + budget_s * 0.94
    if time.time() < t7:
        hc = slsqp_opt(best_h, n, 5000, maxiter=600)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=50)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
