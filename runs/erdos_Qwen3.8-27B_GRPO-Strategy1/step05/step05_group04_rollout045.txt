            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            top_k = np.argsort(corr)[-4:]
            scale = 0.15 * cool
            for j in top_k:
                s = j - (n - 1)
                if s > 0:
                    idx = np.arange(0, n - s)
                    pert[idx] -= scale * cur_h[idx] * 0.5
                elif s < 0:
                    idx = np.arange(-s, n)
                    pert[idx] -= scale * cur_h[idx] * 0.5
                else:
                    pert[:] -= scale * 0.3 * cur_h
        elif mode == 4:
            idx = rng.choice(n, size=max(4, n // 10), replace=False)
            pert[idx] = step * 2.0 * rng.randn(len(idx))
        elif mode == 5:
            half = n // 2
            p1 = step * rng.randn(half)
            pert[:half] = p1
            pert[half:] = -p1
        elif mode == 6:
            f1 = rng.randint(1, 5)
            f2 = rng.randint(5, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert[:] = step * (np.sin(2 * np.pi * f1 * np.arange(n) / n) +
                              0.5 * np.cos(2 * np.pi * f2 * np.arange(n) / n + ph))
        else:
            mask = rng.random(n) < 0.15
            pert[mask] = step * 2.5 * rng.randn(mask.sum())
            pert[~mask] = 0.0

        h_pert = project_h(cur_h + pert, n)
        cd_time = max(2.5, 18.0 * cool * cool + 1.0)
        hop_end = min(t_end, time.time() + cd_time)
        h_opt, c5_opt = coord_desc(h_pert, n, hop_end, max_sweeps=300, seed=seed + hop * 7)
        temp = 0.004 * cool * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def make_structured(n, seed):
    rng = np.random.RandomState(seed)
    pattern = rng.randint(0, 4)
    if pattern == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif pattern == 1:
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n) / n * 3 + rng.uniform(0, 2))
        h = np.clip(h * rng.uniform(0.8, 1.5), 0, 1)
    elif pattern == 2:
        h = np.zeros(n)
        blk = n // 5
        for i in range(5):
            h[i*blk:(i+1)*blk] = rng.uniform(0.3, 1.0)
    else:
        h = rng.uniform(0.2, 0.8, n)
    return project_h(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def remaining():
        return max(0.0, budget_s - (time.time() - t0))

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from initial (25s)
    t1 = t0 + 25
    best_h, best_c5 = coord_desc(best_h, n, t1, max_sweeps=4000, seed=seed)

    # Phase 2: Smooth gradient descent (35s)
    t2 = t0 + 60
    h_g, c5_g = smooth_descent(best_h, n, t2, max_iters=2000)
    if c5_g < best_c5:
        best_h, best_c5 = h_g, c5_g

    # Phase 3: Basin hopping (500s)
    t3 = t0 + 560
    best_h, best_c5 = basin_hop(best_h, n, t3, seed=seed + 1)

    # Phase 4: Coordinate descent refinement (180s)
    t4 = t0 + 740
    best_h, best_c5 = coord_desc(best_h, len(best_h), t4, max_sweeps=8000, seed=seed + 100)

    # Phase 5: Try other resolutions (150s)
    t5 = t0 + 890
    for n_try in [120, 200, 240, 192, 128]:
        if time.time() > t5:
            break
        h_try = make_structured(n_try, seed + n_try)
        h_try, c5_try = coord_desc(h_try, n_try, min(t5, time.time() + 30),
                                   max_sweeps=1500, seed=seed + n_try)
        if c5_try < best_c5:
            best_h, best_c5, n = h_try, c5_try, n_try

    # Phase 6: Final gradient + coord polish (120s)
    t6 = t0 + budget_s * 0.97
    if time.time() < t6:
        n_now = len(best_h)
        h_g2, c5_g2 = smooth_descent(best_h, n_now, min(t6, time.time() + 30), max_iters=1000)
        if c5_g2 < best_c5:
            best_h, best_c5 = h_g2, c5_g2
        best_h, best_c5 = coord_desc(best_h, len(best_h), t6, max_sweeps=3000, seed=seed + 500)

    # Phase 7: Final verification
    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
