    dx = 2.0 / n
    m_full = 2 * n - 1
    center = n - 1
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
    best_c5 = corr.max()

        if time.time() > t_end:
            break

            L = np.zeros(m_full)

    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx

    Tc -= Tc.max()

        wi = w[idx]

        elif k > 0:

        if cv < best_c5 or np.random.rand() < np.exp(-(cv - best_c5) / temp):
            if cv < best_c5:
                best_c5, best_h = cv, hc
            work_h = hc
        else:
            work_h = hp
        # Quick coord polish on the best
        if time.time() < p4:
            hc2, cv2 = coord_descent(best_h, min(time.time() + 5.0, p4), 50)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hc2

    # Phase 5: Final polish (10%)
    best_h, best_c5 = coord_descent(best_h, t_end - 30, 3000)
    if time.time() < t_end - 10:
        hc = lbfgs_run(best_h, 500000, 50.0, 3000, t_end - 10)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5, best_h = cv, hc
        best_h, best_c5 = coord_descent(best_h, t_end, 200)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
