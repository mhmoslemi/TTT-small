def binary_sa(h, n, t_start, t_end, seed=42):
    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    # Ensure binary
    h = (h > 0.5).astype(float)
    # Ensure sum = n/2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

        h[i] = 0
        ones = np.where(h == 1)[0]
    while len(ones) < n//2:
        i = zeros[rng.randint(len(zeros))]
        h[i] = 1
        zeros = np.where(h == 0)[0]
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    cur_h = h.copy()
    
    T_start = 0.05
    T_end = 0.001

def subgrad_polish(h, n, t_end, max_iter=2000, k=5):
    """Subgradient method focusing on top correlations."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    corr_all = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr_all)  # 2n-1
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max_iter
        alpha = 0.01 * (cool ** 1.5) + 0.001
        
        # Top k correlations
        top_k = np.argsort(corr_all)[-k:]
        
        # Subgradient: for each top correlation, compute ∂c[k]/∂h[i]
        grad = np.zeros(n)
        for ki in top_k:
            lag = ki - (n - 1)  # actual lag
            # c[k] = Σ_i h[i] (1-h[i+lag]) for valid i
            # ∂c/∂h[i] = (1-h[i+lag]) for the forward term
            #            + h[i-lag] for the backward term (where h[i] appears as h[(i-lag)+lag])
            # Wait, let me be more careful.
            # c at offset ki (0-indexed in full correlate) corresponds to lag = ki - (n-1)
            # c = Σ_m h[m] * (1 - h[m + lag]) where m and m+lag are both in [0, n-1]
            # ∂c/∂h[m] = (1 - h[m+lag])  (from the first factor)
            #                    - h[m-lag]  (from the second factor, when m = (m-lag)+lag)
            # Wait, the second term: c also includes terms where h[m+lag] appears
            # c = Σ_m h[m]*(1-h[m+lag]) = Σ_m h[m] - Σ_m h[m]*h[m+lag]
            # ∂c/∂h[m] = 1 - h[m+lag] - h[m-lag] (where both are valid)
            
            for m in range(n):
                dp = 0.0
                if 0 <= m + lag < n:
                    dp += (1.0 - h[m + lag])
                if 0 <= m - lag < n and lag != 0:
                    dp += (-h[m - lag])  # Wait, this isn't right either
                # Actually: ∂/∂h[m] of (1-h[m+lag]) = -1 if we differentiate the second factor
                # No wait. c = Σ_j h[j]*(1-h[j+lag])
                # ∂c/∂h[m] = (1-h[m+lag]) [from j=m in first factor]
                #            + Σ_j h[j]*∂(1-h[j+lag])/∂h[m] [from second factor]
                #            = (1-h[m+lag]) + h[m-lag]*(-1) [when j+lag=m, i.e., j=m-lag]
                #            = (1-h[m+lag]) - h[m-lag]
                
                if 0 <= m + lag < n:
                    dp = (1.0 - h[m + lag])
                    if lag != 0 and 0 <= m - lag < n:
                        dp -= h[m - lag]
                elif lag != 0 and 0 <= m - lag < n:
                    dp = -h[m - lag]
                else:
                    dp = 0.0
                grad[m] += dp
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad /= gnorm
        
        # Update
        h_new = h - alpha * grad
        # Project: clip to [0,1] and adjust sum
        h_new = np.clip(h_new, 0, 1)
        target = n / 2.0
        diff = target - h_new.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                room = 1.0 - h_new
                tot = room.sum()
                if tot > 1e-14:
                    h_new += room * (diff / tot)
            else:
                tot = h_new.sum()
                if tot > 1e-14:
                    h_new *= (1.0 + diff / tot)
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        corr_all = np.correlate(h, 1.0 - h, mode='full')
    
    return best_h, best_c5

```python
def compute_subgrad(h, lag, n):
    """Compute ∂c[k]/∂h for the correlation at the given lag."""
    grad = np.zeros(n)
    if lag == 0:
        # c = Σ h[i]*(1-h[i]) = 0 always, gradient = 1-2h
        grad = 1.0 - 2.0 * h
    else:
        # Forward: positions m where m+lag is valid
        forward_mask = np.zeros(n, dtype=bool)
        if lag > 0:
            forward_mask[:n-lag] = True
        else:
            forward_mask[-lag:] = True
        
        # Backward: positions m where m-lag is valid
        backward_mask = np.zeros(n, dtype=bool)
        if lag > 0:
            backward_mask[lag:] = True
        else:
            backward_mask[:n+lag] = True
        
        # ∂c/∂h[m] = (1-h[m+lag]) for m in forward_mask
        if lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
        else:
            grad[-lag:] += (1.0 - h[:n+lag])
        
        # ∂c/∂h[m] -= h[m-lag] for m in backward_mask
        if lag > 0:
            grad[lag:] -= h[:n-lag]
        else:
            grad[:n+lag] -= h[-lag:]
    return grad
```

This is O(n) per lag in vectorized numpy, so very fast.

    best_h = h.copy()
    best_c5 = cur_c5
    t_start = time.time()
    dur = max(t_end - t_start, 0.001)
    while time.time() < t_end:
        p = min((time.time() - t_start) / dur, 1.0)
        T = T0 * (T1 / T0) ** p
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0; h[j] = 1
        nc = compute_c5(h)
        d = nc - cur_c5
        if d < 0 or rng.rand() < np.exp(-d / max(T, 1e-12)):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc; best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
    return best_h, best_c5

def subgrad_polish(h, n, t_end, k=4, max_iter=10000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0
    for it in range(max_iter):
        if time.time() > t_end: break
        cool = max(1.0 - it / max_iter, 0.01)
        alpha = 0.003 * cool**1.5 + 0.0002
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-k:]
        grad = np.zeros(n)
        for ki in top_k:
            lag = ki - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif lag > 0:
                grad[:n-lag] += (1.0 - h[lag:])
                grad[lag:] -= h[:n-lag]
            else:
                nl = -lag
                grad[nl:] += (1.0 - h[:n-nl])
                grad[:n-nl] -= h[nl:]
        gn = np.linalg.norm(grad)
        if gn > 1e-14:
            h -= alpha * grad / gn
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                room = 1.0 - h
                tr = room.sum()
                if tr > 1e-14: h += room * (diff / tr)
            else:
                ts = h.sum()
                if ts > 1e-14: h *= (1.0 + diff / ts)
            h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
    return best_h, best_c5

def slsqp_smooth(h, n, T, t_end, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        x = np.clip(res.x, 0, 1)
        diff = n/2.0 - x.sum()
        if abs(diff) > 1e-10:
            x = np.clip(x + diff / n, 0, 1)
        return x
    except:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.zeros(160)
        h0[:80] = 1.0
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    def is_done(frac=1.0):
        return time.time() > t0 + budget_s * frac
    
    # Phase 1: Binary optimization
    t1 = t0 + budget_s * 0.30
    h_bin = project_binary(best_h, n)
    h_bin, c5 = local_search(h_bin, n, min(t1, t0 + budget_s*0.12), max_rounds=100)
    if c5 < best_c5: best_c5 = c5; best_h = h_bin.copy()
    
    # SA restarts
    if time.time() < t1:
        n_sa = 4
        for r in range(n_sa):
            if time.time() > t1: break
            se = seed + 500 + r * 31
            h_p = best_h.copy()
            h_p = (h_p > 0.5).astype(float)
            rng2 = np.random.RandomState(se)
            n_pert = 3 + r
            for _ in range(n_pert):
                o = np.where(h_p == 1)[0]
                z = np.where(h_p == 0)[0]
                h_p[o[rng2.randint(len(o))]] = 0
                h_p[z[rng2.randint(len(z))]] = 1
            t_sa = min(t0 + (time.time() - t0) + (t1 - time.time()) / (n_sa - r), t1)
            h_sa, c5_sa = sa_binary(h_p, n, t_sa, se)
            if c5_sa < best_c5:
                best_c5 = c5_sa; best_h = h_sa.copy()
            # Quick local search after SA
            h_ls, c5_ls = local_search(h_sa, n, t1, max_rounds=10)
            if c5_ls < best_c5:
                best_c5 = c5_ls; best_h = h_ls.copy()
    
    # Phase 2: Continuous optimization
    t2 = t0 + budget_s * 0.65
    # Subgradient from best (continuous)
    h_sg, c5_sg = subgrad_polish(best_h, n, min(t2, t0 + 30), k=4, max_iter=3000)
    if c5_sg < best_c5: best_c5 = c5_sg; best_h = h_sg.copy()
    
    # SLSQP with decreasing T
    if time.time() < t2:
        for T in [100, 50, 20, 10]:
            if time.time() > t2: break
            rem = max(1.0, (t2 - time.time()) / 4)
            h_sl = slsqp_smooth(best_h, n, T, t0 + time.time() - t0 + rem, maxiter=300)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5: best_c5 = c5_sl; best_h = h_sl.copy()
            # Quick subgrad after SLSQP
            rem2 = max(1.0, (t2 - time.time()) / 4)
            h_sg2, c5_sg2 = subgrad_polish(h_sl, n, t0 + time.time() - t0 + rem2, k=3, max_iter=2000)
            if c5_sg2 < best_c5: best_c5 = c5_sg2; best_h = h_sg2.copy()
    
    # Phase 3: Basin hopping
    t3 = t0 + budget_s * 0.85
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 30
    for it in range(n_bh):
        if time.time() > t3: break
        cool = max(0.05, 1.0 - it / n_bh)
        amp = 0.01 + 0.04 * cool
        h_pert = np.clip(work_h + amp * np.random.randn(n), 0, 1)
        # Fix sum
        diff = n/2.0 - h_pert.sum()
        h_pert = np.clip(h_pert + diff/n, 0, 1)
        # Quick SLSQP
        rem = max(1.0, (t3 - time.time()) / max(n_bh - it, 1))
        T_bh = 20 + 200 * cool
        h_b = slsqp_smooth(h_pert, n, T_bh, t0 + time.time() - t0 + rem, maxiter=200)
        c5_b = compute_c5(h_b)
        temp = 0.0003 * cool + 0.00002
        if c5_b < work_c5 or np.random.rand() < np.exp(-(c5_b - work_c5) / temp):
            work_h = h_b; work_c5 = c5_b
            if c5_b < best_c5: best_c5 = c5_b; best_h = h_b.copy()
    
    # Phase 4: Final polish
    t4 = t0 + budget_s * 0.95
    h_sg, c5_sg
