import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(s))) / T) * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    h = h + (target - s) / n
    h = np.clip(h, 0.0, 1.0)
    # Second pass for any remaining error
    s2 = h.sum()
    if abs(s2 - target) > 1e-10:
        idx = np.where((h > 1e-6) & (h < 1 - 1e-6))[0]
        if len(idx) > 0:
            h[idx] += (target - s2) / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_step(h, T, n, maxiter=500):
    try:
        res = minimize(
            lambda x: smooth_obj(x, T),
            h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
        return np.clip(res.x, 0, 1)
    except:
        return h

def coord_optimal(h, i):
    """Find optimal h[i] given all other h fixed (1D convex problem)."""
    n = len(h)
    h_base = h.copy()
    hi = h_base[i]
    h_base[i] = 0.0
    corr0 = np.correlate(h_base, 1.0 - h_base, mode='full')
    h_base[i] = 1.0
    corr1 = np.correlate(h_base, 1.0 - h_base, mode='full')
    h_base[i] = hi
    diff = corr1 - corr0
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(50):
        m1 = b - gr * (b - a)
        m2 = a + gr * (b - a)
        f1 = np.max(corr0 + m1 * diff)
        f2 = np.max(corr0 + m2 * diff)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def fast_coord_descent(h, t_end, max_sweeps=200):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = 0
        for i in range(n):
            hi_new = coord_optimal(best_h, i)
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp += 1
        if imp == 0:
            break
    return best_h, best_c5

    for x in grid:
        lin = corr_full + (x - hi) * coef
        lin[n-1] = corr_full[n-1] - hi*(1-hi) + x*(1-x)
        v = np.max(lin)
        if v < best_v:
            best_v = v
            best_x = x
    a = grid[max(0, int(grid.size * (best_x)) - 1)] if best_x > 0 else 0.0
    b = grid[min(63, int(grid.size * (best_x)) + 1)] if best_x < 1 else 1.0
    if a == 0 and b == 0: a, b = 0.0, 1.0/64
    if b == 1 and a == 1: a, b = 1.0-1.0/64, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(30):
        m1 = b - gr*(b-a)
        m2 = a + gr*(b-a)
        l1 = corr_full + (m1-hi)*coef; l1[n-1] = corr_full[n-1]-hi*(1-hi)+m1*(1-m1)
        l2 = corr_full + (m2-hi)*coef; l2[n-1] = corr_full[n-1]-hi*(1-hi)+m2*(1-m2)
        if np.max(l1) < np.max(l2):
            b = m2
        else:
            a = m1
    return (a+b)/2.0

def slsqp_step(h, T, n, maxiter=600):
    try:
        res = minimize(lambda x: smooth_obj(x, T), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return np.clip(res.x, 0, 1)
    except:
        return h

def fast_coord_descent(h, t_end, max_sweeps=2000, rng=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = 0
        corr_full = np.correlate(best_h, 1.0 - best_h, mode='full')
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for i in order:
            coef = get_coef_vec(best_h, i, n)
            x_opt = find_opt_x(corr_full, coef, best_h[i], n)
            if abs(x_opt - best_h[i]) < 1e-10:
                continue
            delta = x_opt - best_h[i]
            corr_new = corr_full + delta * coef
            corr_new[n-1] = corr_full[n-1] - best_h[i]*(1-best_h[i]) + x_opt*(1-x_opt)
            c5_try = float(np.max(corr_new) * (2.0/n))
            if c5_try < best_c5 - 1e-14:
                best_h[i] = x_opt
                best_c5 = c5_try
                corr_full = corr_new
                imp += 1
        if imp == 0:
            break
        best_h = project_h(best_h)
    return best_h, best_c5

def make_pert(n, cool, rng):
    mod = rng.randint(0, 6)
    if mod == 0:
        bs = rng.randint(4, max(5, n // 6))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st+bs] = (0.06 + 0.18 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 10))
        ph = rng.uniform(0, 2*np.pi)
        p = (0.05 + 0.12 * cool) * np.sin(2*np.pi*freq*np.arange(n)/n + ph)
    elif mod == 2:
        npts = rng.randint(3, max(4, n // 8))
        idx = rng.choice(n, npts, replace=False)
        p = np.zeros(n)
        p[idx] = (0.12 + 0.35 * cool) * rng.standard_normal(npts)
    elif mod == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    elif mod == 4:
        bs = rng.randint(6, max(7, n // 5))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st+bs] = (0.15 + 0.25 * cool) * rng.standard_normal(bs)
        p[(st+bs//2):(st+bs+bs//2) % n] -= (0.15 + 0.25 * cool) * 0.5
    else:
        half = n // 2
        p = np.zeros(n)
        amp = 0.08 * cool
        p[:half] = amp * rng.standard_normal(half)
        p[half:] = -amp * rng.standard_normal(n - half)
    return p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation (10% time)
    for T in [50, 200, 800, 3000, 12000, 50000, 200000]:
        if elapsed() > budget_s * 0.10:
            break
        hc = slsqp_step(best_h, T, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)

    # Phase 2: Fast coordinate descent (25% time)
    best_h, best_c5 = fast_coord_descent(best_h, t0 + budget_s * 0.35, max_sweeps=3000, rng=rng)
    best_h = project_h(best_h)

    # Phase 3: Basin hopping (30% time)
    work_h = best_h.copy()
    work_c5 = best_c5
    n_it = 60
    for it in range(n_it):
        if elapsed() > budget_s * 0.65:
            break
        cool = 1.0 - 0.9 * (it / n_it)
        pert = make_pert(n, cool, rng)
        h_pert = project_h(work_h + pert)
        T_bh = 500 + 15000 * cool
        hc = slsqp_step(h_pert, T_bh, n, maxiter=500)
        hc = project_h(hc)
        hc, c5v = fast_coord_descent(hc, t0 + budget_s * 0.65 - (elapsed() - (t0 + budget_s*0.35))*0.01, max_sweeps=200, rng=rng)
        temp = 0.0012 * cool + 0.0002
        if c5v < work_c5 or (rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9))):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: More coordinate descent (15% time)
    best_h, best_c5 = fast_coord_descent(best_h, t0 + budget_s * 0.82, max_sweeps=3000, rng=rng)
    best_h = project_h(best_h)

    # Phase 5: Fine basin hopping (10% time)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(30):
        if elapsed() > budget_s * 0.92:
            break
        cool = 1.0 - 0.8 * (it / 30)
        pert = make_pert(n, cool * 0.4, rng)
        h_pert = project_h(work_h + pert)
        hc = slsqp_step(h_pert, 5000 + 20000 * cool, n, maxiter=400)
        hc = project_h(hc)
        hc, c5v = fast_coord_descent(hc, t0 + budget_s * 0.92, max_sweeps=100, rng=rng)
        if c5v < work_c5:
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: Final polish (8% time)
    best_h, best_c5 = fast_coord_descent(best_h, t0 + budget_s * 0.97, max_sweeps=2000, rng=rng)
    for T in [500000, 2000000]:
        if elapsed() > budget_s * 0.97:
            break
        hc = slsqp_step(best_h, T, n, maxiter=1000)
        hc = project_h(hc)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h, best_c5 = fast_coord_descent(best_h, t0 + budget_s * 0.99, max_sweeps=500, rng=rng)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
