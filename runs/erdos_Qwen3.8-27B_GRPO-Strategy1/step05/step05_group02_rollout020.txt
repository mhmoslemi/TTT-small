        # For each idx, k = idx - n + 1
        # i-k = i - (idx-n+1) = i + n - 1 - idx
        # i+k = i + idx - n + 1
        
        pos1 = i - k_offsets  # = i + n - 1 - idx = i - (idx - n + 1)
        pos2 = i + k_offsets  # = i + idx - n + 1
        
        c1 = np.zeros(2*n-1)
        v1 = (pos1 >= 0) & (pos1 < n)
        c1[v1] += (1.0 - h_base[pos1[v1]])
        
        c2 = np.zeros(2*n-1)
        v2 = (pos2 >= 0) & (pos2 < n)
        c2[v2] += h_base[pos2[v2]]
        
        coefs = c1 - c2
        
        # Minimize max(corr_base + t*coefs) over t in [0,1]
        # This is a 1D problem
        
        def obj(t):
            return np.max(corr_base + t * coefs)
        
        res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-10})
        h[i] = float(res.x)
        
    return h

def coord_descent(h, n, rng, t_deadline, max_sweeps=200):
    """Run coordinate descent until deadline or max_sweeps."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        best_h = coord_sweep(best_h, n, rng)
        best_h = project_h(best_h)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            # No improvement, try a few more sweeps
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            if s < 1e-14:
                break
            h = h * (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_sweep(h, n, rng):
    dx = 2.0 / n
    order = rng.permutation(n)
    k_offsets = np.arange(-(n - 1), n)
    for i in order:
        h_base = h.copy()
        h_base[i] = 0.0
        corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
        pos1 = i - k_offsets
        pos2 = i + k_offsets
        c1 = np.zeros(2 * n - 1)
        v1 = (pos1 >= 0) & (pos1 < n)
        c1[v1] = 1.0 - h_base[pos1[v1]]
        c2 = np.zeros(2 * n - 1)
        v2 = (pos2 >= 0) & (pos2 < n)
        c2[v2] = h_base[pos2[v2]]
        coefs = c1 - c2
        def obj(t):
            return float(np.max(corr_base + t * coefs))
        res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-10})
        h[i] = float(res.x)
    return h

def coord_descent(h, n, rng, t_deadline, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stale = 0
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        best_h = coord_sweep(best_h, n, rng)
        best_h = project_h(best_h)
        c5 = compute_c5(best_h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            stale = 0
        else:
            stale += 1
            if stale > 5:
                break
    return best_h, best_c5

def binarize(h, n):
    """Convert to binary with exactly n/2 ones, using threshold + adjustment."""
    h_sorted_idx = np.argsort(-h)
    hb = np.zeros(n)
    for i in range(n // 2):
        hb[h_sorted_idx[i]] = 1.0
    return hb

def binary_obj(hb):
    n = len(hb)
    dx = 2.0 / n
    corr = np.correlate(hb, 1.0 - hb, mode='full')
    return float(np.max(corr) * dx)

def binary_sa(hb, n, rng, t_start, t_end, n_restarts=3):
    best_hb = hb.copy()
    best_val = binary_obj(best_hb)
    t_alloc = max(1.0, (t_end - t_start) / n_restarts)
    for restart in range(n_restarts):
        t_deadline = min(t_start + (restart + 1) * t_alloc, t_end)
        cur = best_hb.copy()
        cur_val = binary_obj(cur)
        pos_ones = np.where(cur == 1.0)[0]
        zeros_list = np.random.permutation(n)
        T = 0.02
        alpha = 0.9995
        it = 0
        while time.time() < t_deadline:
            if it % 5000 == 0 and T < 0.001:
                T = 0.005
            it += 1
            i_idx = rng.randint(len(pos_ones))
            swap_i = pos_ones[i_idx]
            # Find a zero position
            zi = rng.randint(n)
            swap_j = zeros_list[zi]
            if cur[swap_j] == 1.0:
                continue
            cur[swap_i] = 0.0
            cur[swap_j] = 1.0
            # Update tracking
            pos_ones[i_idx] = swap_j
            pos_ones = np.sort(pos_ones)
            new_val = binary_obj(cur)
            delta = new_val - cur_val
            if delta < 0 or rng.rand() < np.exp(-delta / T):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_hb = cur.copy()
            else:
                cur[swap_i] = 1.0
                cur[swap_j] = 0.0
                pos_ones[i_idx] = swap_i
                pos_ones = np.sort(pos_ones)
            T *= alpha
    return best_hb, best_val

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return float(val * dx)

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_step(h, T, n, maxiter=1000):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h.copy()

    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():

    # Phase 1: Coordinate descent polish (12%)
    t_p1 = t0 + budget_s * 0.12
    best_h, best_c5 = coord_descent(best_h, n, rng, t_p1, max_sweeps=300)

    # Phase 2: Binary SA + coordinate descent alternation (48%)
    t_p2 = t0 + budget_s * 0.60
    n_cycles = 4
    cycle_time = (t_p2 - t0) / n_cycles

        # Binarize current best
        hb = binarize(best_h, n)
        tb_deadline = t_cycle_end - (t_cycle_end - t0) * 0.4
        tb_deadline = max(tb_deadline, t_cycle_end - 30)
        hb, hb_val = binary_sa(hb, n, rng, time.time(), tb_deadline, n_restarts=2)
        # De-binarize with coordinate descent from binarized solution

            best_h = best_h2
        if elapsed() > budget_s * 0.58:

    # Phase 3: Basin hopping with diverse perturbations (25%)
    t_p3 = t0 + budget_s * 0.85
    work_h = best_h.copy()
    work_c5 = best_c5

            k = rng.randint(n // 4, n // 2)

            np.random.shuffle(block)

            h_pert = work_h.copy()
            for _ in range(nswap):
