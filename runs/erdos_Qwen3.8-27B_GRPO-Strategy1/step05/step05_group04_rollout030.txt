def cutting_plane(h, n, t_end, delta_init=0.08, max_iters=500):
    dx = 2.0 / n
    h_c = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_c)
    delta = delta_init
    stalling = 0
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h_c, 1.0 - h_c, mode='full')
        cmax = np.max(corr)
        
        # Select active lags (top 15 or within threshold)
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:15]
        # Also include all within 3*dx of max in correlation units
        thresh = cmax - 3.0
        active = np.unique(np.concatenate([active, np.where(corr >= thresh)[0]]))
        active = active[:25]  # cap
        
        # Compute gradients
        K = len(active)
        G = np.zeros((K, n))
        for idx, k in enumerate(active):
            lag = k - (n - 1)
            if lag >= 0:
                G[idx, :n-lag] += (1.0 - h_c[lag:])
                if lag > 0:
                    G[idx, lag:] -= h_c[:n-lag]
            else:
                a = -lag
                G[idx, :n-a] += (1.0 - h_c[:n-a])
                G[idx, a:] -= h_c[:n-a]
        
        # Set up LP
        # Variables: d_0,...,d_{n-1}, t (total n+1)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        # Inequality: -t + dx*(G[k,:] @ d) <= -dx*corr[active[k]]
        A_ub = np.zeros((K, n + 1))
        for idx in range(K):
            A_ub[idx, :n] = dx * G[idx, :]
            A_ub[idx, n] = -1.0
        b_ub = -dx * corr[active]
        
        # Equality: sum d_i = target - sum h_c
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0 - h_c.sum()])
        
        # Bounds
        lb = np.maximum(-h_c, -delta)
        ub = np.minimum(1.0 - h_c, delta)
        bounds = list(zip(lb, ub))
        bounds.append((None, None))  # t is free
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            delta *= 0.5
            stalling += 1
            if stalling > 10 or delta < 0.005:
                break
            continue
        
        d_star = res.x[:n]
        h_new = project_h(h_c + d_star, n)
        c5_new = compute_c5(h_new)

            delta *= 0.7

    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(40):
        s = h.sum()
        if abs(s - target) < 1e-12:
            return h
        if s > target:
            h = h * (target / max(s, 1e-15))
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h


def eval_1d(v, old, corr, lin, n, dx):
    d = v - old
    arr = corr + d * lin
    arr[n - 1] -= d * d
    return float(np.max(arr) * dx)


def minimize_1d(corr, lin, n, dx, old):
    best_v = old
    best_f = np.inf
    for v in np.linspace(0.0, 1.0, 21):
        f = eval_1d(v, old, corr, lin, n, dx)
        if f < best_f:

            best_v = v
    lo = max(0.0, best_v - 0.05)
    hi = min(1.0, best_v + 0.05)

        f = eval_1d(v, old, corr, lin, n, dx)
        if f < best_f:
            best_f = f
            best_v = v
    return best_v
