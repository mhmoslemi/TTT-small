        k = shifts[t]
        jp = i_idx + k
        jm = i_idx - k
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n - 1)
        jmc = np.clip(jm, 0, n - 1)
        grad += weights[t] * ((1.0 - x[jpc]) * v1.astype(np.float64) - x[jmc] * v2.astype(np.float64))
    grad *= dx
    return val, grad

def slsqp_opt(h, T, n, maxiter=600, ftol=1e-15):
    def obj(x):
        v, _ = smooth_obj_and_grad(x, T, n)
        return v
    def obj_grad(x):
        _, g = smooth_obj_and_grad(x, T, n)
        return g
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                                      'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def topk_grad(h, n_topk, weights=None):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    if weights is None:
        top_idx = np.argsort(corr)[-n_topk:]
        top_w = np.ones(n_topk) / n_topk
    else:
        idx = np.argsort(corr)[-n_topk:]
        top_idx = idx
        top_w = weights / weights.sum()
    shifts = top_idx - (n - 1)
    i_idx = np.arange(n)
    grad = np.zeros(n)
    for t in range(n_topk):
        k = shifts[t]
        w = top_w[t]
        jp = i_idx + k
        jm = i_idx - k
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n - 1)
        jmc = np.clip(jm, 0, n - 1)
        grad += w * ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64))
    return grad * dx

def coord_descent(h, n, t_end, max_sweeps=200, n_top=35):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = 0
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-14})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved += 1
        if improved == 0:
            break
    return best_h, best_c5

def pair_swap(h, n, t_end, n_trials=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        h_try = best_h.copy()
        h_try[i], h_try[j] = best_h[j], best_h[i]
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def basin_search(h, n, t_end, n_iter=50, scale=0.08):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.3:
            corr = get_corr(work_h)
            top_idx = np.argsort(corr)[-6:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n - k] += (work_h[k:n] - 0.5)
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n + k] - 0.5)
            pert *= scale * 0.5 * cool
        elif r < 0.55:
            f = np.random.randint(1, 16)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.6 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.75:
            blen = np.random.randint(max(n // 10, 2), max(n // 4, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * cool * 1.3 * np.random.randn(blen)
        else:
            pert = scale * cool * 0.7 * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = 2000 + 30000 * cool
        hc = slsqp_opt(h_pert, T, n, 350, 1e-14)
        c5v = compute_c5(hc)
        temp = 0.0006 * cool + 0.00008
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup
    if t_end - time.time() > 80:
        for T in [500, 3000, 20000, 100000]:
            if t_end - time.time() < 25:
                break
            hc = slsqp_opt(best_h, T, n, 700, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 2: Weighted top-k gradient + SLSQP
    if t_end - time.time() > 150:
        mom_v = np.zeros(n)
        for it in range(70):
            if t_end - time.time() < 20:
                break
            cool = 1.0 - it / 70.0
            alpha = 0.002 + 0.006 * cool
            nk = 8 + int(25 * cool)
            grad, _ = topk_grad(best_h, nk) if False else (topk_grad(best_h, nk), None)
            grad = topk_grad(best_h, nk)
            grad = 0.5 * mom_v + 0.5 * grad
            mom_v = grad
            h_new = project_h(best_h - alpha * grad)
            T = 2000 + 30000 * cool
            hc = slsqp_opt(h_new, T, n, 450, 1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 3: Basin hopping with SLSQP
    if t_end - time.time() > 120:
        bh_t = t_end - (t_end - time.time()) * 0.28
        bh_h, bh_c5 = basin_search(best_h, n, bh_t, 55, 0.1)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h

    # Phase 4: Deep coordinate descent
    if t_end - time.time() > 60:
        cd_t = t_end - (t_end - time.time()) * 0.35
        best_h, best_c5 = coord_descent(best_h, n, cd_t, 180, 40)

    # Phase 5: Pair swap local search
    if t_end - time.time() > 40:
        ps_t = t_end - (t_end - time.time()) * 0.35
        best_h, best_c5 = pair_swap(best_h, n, ps_t, n_trials=6000)

    # Phase 6: Final SLSQP polish
    if t_end - time.time() > 25:
        hc = slsqp_opt(best_h, 300000, n, 1200, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Final coordinate descent
    if t_end - time.time() > 20:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 2, 100, 50)

    # Phase 8: Final pair swap
    if t_end - time.time() > 10:
        best_h, best_c5 = pair_swap(best_h, n, t_end - 1, n_trials=3000)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
