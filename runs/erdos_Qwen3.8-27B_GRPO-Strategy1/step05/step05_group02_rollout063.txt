```python
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def grad_f_d(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        grad[:n-d] += (1.0 - h[d:])
        grad[d:] -= h[:n-d]
    else:
        dd = -d
        grad[dd:] += (1.0 - h[:-dd])
        grad[:-dd] -= h[dd:]
    return grad

def subgradient_optimize(h, n, t_end, max_iter=50000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    top_k = 10
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Get top-k shifts
        idx = np.argpartition(corr, -top_k)[-top_k:]
        # Average gradient over top-k
        avg_grad = np.zeros(n)
        for i in idx:
            d = i - (n - 1)
            avg_grad += grad_f_d(best_h, d)
        avg_grad /= top_k
        # Adaptive learning rate
        gnorm = np.linalg.norm(avg_grad)
        if gnorm < 1e-12:
            break
        lr = 0.01 / (1.0 + 0.001 * it)
        h_try = project_h(best_h - lr * avg_grad)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        # Line search if projection causes issues
    return best_h, best_c5
```

```python
def coord_single_fast(h, n, i):
    # Compute coefficients for correlation change when h[i] varies
    h[i] = 0.0  # temporarily
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    # Restore
    # coefs[k] for k = 0, ..., 2n-2 corresponds to shift d = k - (n-1)
    # Δc[k] = h[i] * ((1-h[i+d]) - h[i-d]) where valid
    coefs = np.zeros(2 * n - 1)
    for d in range(-(n-1), n):
        k = d + n - 1
        c = 0.0
        if 0 <= i + d < n:
            c += (1.0 - h[i + d])
        if 0 <= i - d < n:
            c -= h[i - d]
        coefs[k] = c
    
    # Now c_new[k] = corr_base[k] + h[i] * coefs[k]
    # Minimize max(c_new) * 2/n over h[i] ∈ [0,1]
    # This is a 1D minimax problem
    dx = 2.0 / n
    from scipy.optimize import minimize_scalar
    def obj(hi):
        return np.max(corr_base + hi * coefs) * dx
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    h[i] = res.x
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def grad_f_d(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        grad[:n-d] += (1.0 - h[d:])
        grad[d:] -= h[:n-d]
    else:
        dd = -d
        grad[dd:] += (1.0 - h[:-dd])
        grad[:-dd] -= h[dd:]
    return grad

def subgradient_phase(h, n, t_end, top_k=15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(123)
    t0 = time.time()
    it = 0
    while time.time() < t_end:
        it += 1
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Top-k shifts
        if top_k >= len(corr):
            idx = np.arange(len(corr))
        else:
            idx = np.argpartition(corr, -top_k)[-top_k:]
        # Weight by how close to max
        m = corr.max()
        weights = np.exp(-3.0 * (m - corr[idx]))
        weights /= weights.sum()
        # Average gradient
        avg_grad = np.zeros(n)
        for w, i in zip(weights, idx):
            d = i - (n - 1)
            avg_grad += w * grad_f_d(best_h, d)
        gnorm = np.linalg.norm(avg_grad)
        if gnorm < 1e-10:
            break
        # Adaptive LR
        base_lr = 0.005
        lr = base_lr / (1.0 + 0.0005 * it)
        h_try = project_h(best_h - lr * avg_grad)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        elif it % 100 == 0 and rng.rand() < 0.1:
            # Small random perturbation to escape
            h_try = project_h(best_h + 0.01 * rng.standard_normal(n))
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    rng = np.random.RandomState(456)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Evaluate at a few points
            old = best_h[i]
            # Try: 0, 1, old±0.1, old±0.01, and minimize_scalar result
            best_x = old
            best_v = best_c5  # approximate (won't be exact but good filter)
            # Use minimize_scalar for accuracy
            h_save = best_h[i]
            best_h[i] = 0.0
            c_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            # coefs
            coefs = np.zeros(2*n-1)
            for d in range(-(n-1), n):
                k = d + n - 1
                c = 0.0
                if 0 <= i+d < n:
                    c += (1.0 - best_h[i+d])
                if 0 <= i-d < n:
                    c -= best_h[i-d]
                coefs[k] = c
            best_h[i] = h_save
            
            def obj_1d(x):
                vals = c_base + x * coefs
                vals[n-1] -= x*x  # fix quadratic term for d=0
                return np.max(vals)
            
            try:
                res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11})
                new_x = float(res.x)
            except Exception:
                new_x = old
            
            if abs(new_x - old) > 1e-10:
                h_try = best_h.copy()
                h_try[i] = new_x
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    imp = True
        if not imp:
            break
    return best_h, best_c5

def slsqp_step(h, T, n, maxiter=1000):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def make_perturb(n, cool, rng):
    mod = rng.randint(0, 4)
    if mod == 0:
        bs = max(5, n // 10)
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st+bs] = (0.03 + 0.08*cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n//20))
        ph = rng.uniform(0, 2*np.pi)
        p = (0.02 + 0.06*cool) * np.sin(2*np.pi*freq*np.arange(n)/n + ph)
    elif mod == 2:
        np_ = max(5, n//10)
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.05 + 0.15*cool) * rng.standard_normal(np_)
    else:
        p = (0.015 + 0.04*cool) * rng.standard_normal(n)
    return p

def basin_hopping(h, n, t_end, n_iter=40):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(789)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.8*(it/max(1,n_iter))
        pert = make_perturb(n, cool, rng)
        h_pert = project_h(best_h + pert)
        T_bh = int(500 + 15000*cool)
        hc = slsqp_step(h_pert, T_bh, n, maxiter=600)
        c5v = compute_c5(hc)
        temp = 0.001*cool + 0.0001
        if c5v < best_c5 or rng.rand() < np.exp(-(c5v - best_c5)/max(temp, 1e-9)):
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
            else:
                best_h = hc
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def grad_f_d(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        grad[:n-d] += (1.0 - h[d:])
        grad[d:] -= h[:n-d]
    else:
        dd = -d
        grad[dd:] += (1.0 - h[:-dd])
        grad[:-dd] -= h[dd:]
    return grad

def subgradient_phase(h, n, t_end, top_k=15, lr0=0.004):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(123)
    it = 0
    while time.time() < t_end:
        it += 1
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = corr.max()
        if top_k >= len(corr):
            idx = np.arange(len(corr))
            weights = np.ones(len(corr))
            weights /= weights.sum()
        else:
            idx = np.argpartition(corr, -top_k)[-top_k:]
            raw_w = np.exp(-2.0 * (m - corr[idx]))
            if raw_w.sum() < 1e-10:
                weights = np.ones(top_k) / top_k
            else:
                weights = raw_w / raw_w.sum()
        avg_grad = np.zeros(n)
        for w, ii in zip(weights, idx):
            d = ii - (n - 1)
            avg_grad += w * grad_f_d(best_h, d)
        gnorm = np.linalg.norm(avg_grad)
        if gnorm < 1e-12:
            break
        lr = lr0 / (1.0 + 0.0003 * it)
        h_try = project_h(best_h - lr * avg_grad)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try
        elif it % 500 == 0:
            h_try = project_h(best_h + 0.02 * rng.standard_normal(n))
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
    return best_h, best_c5

def coord_descent_fast(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float).copy()
    best_c5 = float(np.max(corr) * 2.0 / n)
    dx = 2.0 / n
    two_nm1 = 2 * n - 1
    rng = np.random.RandomState(456)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if sweep % 10 == 9:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
            best_c5 = float(np.max(corr) * dx)
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            coefs = np.zeros(two_nm1)

def slsqp_step(h, T, n, maxiter=1000):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

    return best_h, best_c5, n
```
