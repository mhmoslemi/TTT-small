    if lag >= 0:
        valid = n - lag
        g[:valid] += 1.0 - h[lag:]
        g[lag:] -= h[:valid]
    else:
        l = -lag
        valid = n - l
        g[l:] += 1.0 - h[:valid]
        g[:valid] -= h[l:]
    return g

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h.copy()
    if s > target:
        mask = h > 0
        if mask.sum() == 0:
            return h
        h[mask] *= (target / s)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.sum(np.exp(np.clip((corr - m) * T, -50, 50)))
    return (m + np.log(s) / T) * dx

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-16):
    def obj(x):
        return smooth_obj(x, T)
    def con_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descend(h, n, t_end, max_sweeps=200):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Compute coefficient for each lag
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            coefs = np.zeros(2 * n - 1)
            k_range = np.arange(-(n - 1), n)
            j1 = i + k_range
            j2 = i - k_range
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coefs[v1] += (1.0 - best_h[np.clip(j1[v1], 0, n - 1)])
            coefs[v2] += (-best_h[np.clip(j2[v2], 0, n - 1)])
            # 1D minimax: min_x max_k (corr_base[k] + x * coefs[k])
            # Use golden section or bounded minimize
            def f1(x):
                return np.max(corr_base + x * coefs)
            try:
                res = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-13})
                x_new = float(res.x)
            except Exception:
                continue
            if abs(x_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = x_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binarize_steps(h, n, t_end, n_steps=500):

- This gives O(m²) where m = 2n-1, i.e., O(n²) per coordinate

For n=160, m=319, m²=102K per coordinate, 160 coords = 16M per sweep. That's more than the O(n) per coordinate from minimize_scalar. So minimize_scalar is actually better here.

Alternatively, I can use the fact that the minimum of the max of linear functions can be found by:
- Start with x = (1-x_lo-x_hi)/2 (midpoint based on where the active constraints cross)
- Or use the "bisection on the value" approach: binary search on t, check if there exists x with a_k + b_k*x <= t for all k (this is a linear feasibility problem in 1D)

The bisection approach: for a candidate value t, check if max_k (a_k + b_k*x) <= t for some x ∈ [0,1]. This means:
For all k: a_k + b_k*x <= t
If b_k > 0: x <= (t - a_k)/b_k
If b_k < 0: x >= (t - a_k)/b_k
If b_k = 0: need a_k <= t

So the feasible x range is [max(0, max_{b_k<0}(t-a_k)/b_k), min(1, min_{b_k>0}(t-a_k)/b_k)].
If lower <= upper AND all a_k <= t for b_k=0, then t is feasible.

Binary search on t gives O(m * log(1/eps)) per coordinate, which is O(n) for our purposes.

But minimize_scalar already does something similar, so let me just use it.

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h.copy()
    if s > target:
        mask = h > 0
        if mask.sum() == 0:
            return h.copy()
        h[mask] *= (target / s)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h += room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip((corr - m) * T, -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def slsqp_opt(h, T, n, maxiter=600, ftol=1e-16):
    def obj(x):
        return smooth_obj(x, T)
    def con_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h.copy()

def _make_coefs(i, h, n):
    coefs = np.zeros(2 * n - 1)
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    return coefs

def coord_descend(h, n, t_end, max_sweeps=300, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr_full = np.correlate(best_h, 1.0 - best_h, mode='full')
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = _make_coefs(i, best_h, n)
            corr_base = corr_full - best_h[i] * coefs
            def f1(x):
                return float(np.max(corr_base + x * coefs))
            try:
                res = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-13})
                x_new = float(res.x)
            except Exception:
                continue
            if abs(x_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = x_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                corr_full = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binarize_push(h, n, t_end, n_steps=600, rng=None):
    if rng is None:
        rng = np.random.RandomState(1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        idx = rng.randint(n)
        v = best_h[idx]
        if v > 0.04 and v < 0.96:
            target_val = 1.0 if v > 0.5 else 0.0
            h_try = best_h.copy()
            h_try[idx] = target_val
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
    return best_h, best_c5

def make_pert(n, sigma, rng):
    mod = rng.randint(0, 6)
    if mod == 0:
        bs = rng.randint(3, max(4, n // 5))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st+bs] = sigma * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 10))
        ph = rng.uniform(0, 2 * np.pi)
        p = sigma * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(3, max(4, n // 8))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = sigma * rng.standard_normal(np_)
    elif mod == 3:
        p = sigma * 0.25 * rng.standard_normal(n)
    elif mod == 4:
        half = n // 2
        p = np.zeros(n)
        p[:half] = sigma * 0.4 * rng.standard_normal(half)
        p[half:] = -sigma * 0.4 * rng.standard_normal(n - half)
    else:
        p = np.zeros(n)
        for _ in range(rng.randint(2, 5)):
            s = rng.randint(0, n - 3)
            l = rng.randint(2, 8)
            p[s:s+l] += sigma * 0.5 * rng.standard_normal(l)
    return p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 160
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation (12% budget)
    t1 = t0 + budget_s * 0.12
    for T in [40, 150, 600, 2500, 10000, 40000, 150000, 600000, 2500000]:
        if elapsed() > t1:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=700, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent (18% budget)
    t2 = t0 + budget_s * 0.30
    best_h, best_c5 = coord_descend(best_h, n, t2, max_sweeps=400, rng=rng)

    # Phase 3: Binarize + coord descent (8% budget)
    t3 = t0 + budget_s * 0.38
    best_h, best_c5 = binarize_push(best_h, n, t3, n_steps=500, rng=rng)
    best_h, best_c5 = coord_descend(best_h, n, t3, max_sweeps=150, rng=rng)

    # Phase 4: Basin hopping with diverse perturbations (25% budget)
    t4 = t0 + budget_s * 0.63
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 70
    for it in range(n_iter):
        if elapsed() > t4:
            break
        cool = 1.0 - 0.92 * (it / n_iter)
        sigma = 0.015 + 0.18 * cool * cool
        pert = make_pert(n, sigma, rng)
        h_pert = project_h(work_h + pert)
        T_bh = 300 + 80000 * cool * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=350, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.00015
        accept = c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12))
        if accept:
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 5: Cooldown - fine coordinate descent (12% budget)
    t5 = t0 + budget_s * 0.75
    best_h, best_c5 = coord_descend(best_h, n, t5, max_sweeps=250, rng=rng)

    # Phase 6: Second basin hopping round (12% budget)
    t6 = t0 + budget_s * 0.87
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(30):
        if elapsed() > t6:
            break
        cool = 1.0 - 0.8 * (it / 30)
        sigma = 0.01 + 0.08 * cool
        pert = make_pert(n, sigma, rng)
        h_pert = project_h(work_h + pert)
        T_bh = 5000 + 200000 * cool * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=400, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0004 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 7: Final polish (10% budget)
    t7 = t0 + budget_s * 0.94
    best_h, best_c5 = coord_descend(best_h, n, t7, max_sweeps=150, rng=rng)
    best_h, best_c5 = binarize_push(best_h, n, t7, n_steps=300, rng=rng)
    best_h, best_c5 = coord_descend(best_h, n, t7, max_sweeps=80, rng=rng)

    # Phase 8: Ultra-high-T SLSQP + final coord (5% budget)
    if elapsed() < t0 + budget_s * 0.985:
        for T in [3000000, 10000000]:
            if elapsed() > t0 + budget_s * 0.97:
                break
            hc = slsqp_opt(best_h, T, n, maxiter=1200, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        if elapsed() < t0 + budget_s * 0.99:
            best_h, best_c5 = coord_descend(best_h, n, t0 + budget_s * 0.995, max_sweeps=30, rng=rng)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
