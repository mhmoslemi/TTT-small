import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T, n_pen=3, flat_pen=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e8:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-n_pen:]
        val += flat_pen * np.std(top)
    return val * dx

def slsqp(h, T, n, maxiter=500, ftol=1e-15, flat_pen=0.0):
    try:
        res = minimize(lambda x: smooth_c5(x, T, flat_pen=flat_pen), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=200, n_top=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(len(top_shifts))
            for t_idx, k in enumerate(top_shifts):
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t_idx] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.25:
            f = np.random.randint(1, 15)

            mask = np.random.random(n) < 0.15

        elif r < 0.85:

            pert = np.zeros(n)

                    pert[k:n] += (work_h[:n-k] - 0.5)

        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)

        hc = slsqp(h_pert, T, n, 300, 1e-14)

        temp = 0.001 * cool + 0.0001

            work_c5 = c5v
    return work_h, work_c5

def subgrad_step(h, alpha, n_topk, mom_v, beta=0.4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    grad = beta * mom_v + (1.0 - beta) * grad
    proj = project_h(h - alpha * grad)
    return proj, grad

def optimize_phase(h_init, n, t_end):

    best_c5 = compute_c5(best_h)
    
    # SLSQP warmup

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T, flat_pen=0.0, n_pen=3):
    n = len(h)

        val += flat_pen * np.std(top)
    return val * dx

        return smooth_c5(x, T, flat_pen=flat_pen)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_step(h, alpha, n_topk, mom_v, beta=0.4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    # CORRECTED: d(corr[k])/d(h[i]) = (1-h[i-k])*[jm valid] - h[i+k]*[jp valid]
    grad = ((1.0 - h[jmc]) * v2.astype(np.float64) - h[jpc] * v1.astype(np.float64)).mean(axis=1) * dx
    grad = beta * mom_v + (1.0 - beta) * grad
    new_h = project_h(h - alpha * grad)
    return new_h, grad

def coord_descent(h, n, t_end, max_sweeps=200, n_top=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        ns = len(top_shifts)
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(ns)
            for t in range(ns):
                k = top_shifts[t]
                jm = i - k
                jp = i + k
                c = 0.0
                if 0 <= jm < n:
                    c += (1.0 - best_h[jm])
                if 0 <= jp < n:
                    c -= best_h[jp]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            def obj_1d(tt):
                return np.max(base + tt * coefs)
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.2:
            f = np.random.randint(1, 15)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.6 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.45:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.2 * cool * np.random.randn(max(int(mask.sum()), 1))
        elif r < 0.65:
            blen = np.random.randint(max(n // 12, 2), max(n // 3, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * 0.8 * cool * np.random.randn(blen)
        elif r < 0.82:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-8:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n - k] += (work_h[k:n] - 0.5)
                    pert[k:n] += (work_h[:n - k] - 0.5)
                elif k < 0:
                    kk = -k
                    if n - kk > 0:
                        pert[:n - kk] += (work_h[kk:n] - 0.5)
                        pert[kk:n] += (work_h[:n - kk] - 0.5)
            pert *= scale * 0.3 * cool
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp(h_pert, T, n, 300, 1e-14)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_phase(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    if t_end - time.time() > 30:
        for T in [50, 200, 1000, 5000, 50000]:
            if t_end - time.time() < 15:
                break
            hc = slsqp(best_h, T, n, 400, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if t_end - time.time() > 40:
        mom_v = np.zeros(n)
        for it in range(35):
            if t_end - time.time() < 12:
                break
            cool = 1.0 - it / 35.0
            alpha = 0.001 + 0.004 * cool
            n_topk = 8 + int(15 * cool)
            hs, g = subgrad_step(best_h, alpha, n_topk, mom_v, beta=0.4)
            mom_v = g
            T = 300 + 3000 * cool
            hc = slsqp(hs, T, n, 300, 1e-14, flat_pen=0.03 * cool)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if t_end - time.time() > 45:
        bh_h, bh_c5 = basin_hop(best_h, n, t_end - (t_end - time.time()) * 0.30,
                                35, 0.1, 200, 5000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    if t_end - time.time() > 25:
        best_h, best_c5 = coord_descent(best_h, n, t_end - (t_end - time.time()) * 0.25,
                                        150, 35)
    if t_end - time.time() > 25:
        bh_h, bh_c5 = basin_hop(best_h, n, t_end - (t_end - time.time()) * 0.35,
                                22, 0.03, 2000, 8000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    if t_end - time.time() > 15:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 5, 80, 45)
    if t_end - time.time() > 10:
        hc = slsqp(best_h, 1e6, n, 800, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(160)
        init_h[:80] = 1.0
    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)
    t_p1 = t0 + budget_s * 0.48
    best_h, best_c5 = optimize_phase(best_h, n, t_p1)
    if t_end - time.time() > budget_s * 0.24 and n != 160:
        n2 = 160
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        t_p2 = t0 + budget_s * 0.70
        h2_opt, c5_2 = optimize_phase(h2, n2, t_p2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt
            n = n2
    if t_end - time.time() > budget_s * 0.18 and n != 240:
        n3 = 240
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = project_h(h3)
        t_p3 = t0 + budget_s * 0.90
        h3_opt, c5_3 = optimize_phase(h3, n3, t_p3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3_opt
            n = n3
    if t_end - time.time() > 12:
        hc = slsqp(best_h, 5e6, n, 1200, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if t_end - time.time() > 6:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, 50, 50)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
