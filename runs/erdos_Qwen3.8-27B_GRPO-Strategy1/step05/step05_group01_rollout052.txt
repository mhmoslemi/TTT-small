import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_desc(h, n, t_end, n_top=40, max_sweeps=400, order='random'):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        ts = top_idx - (n - 1)
        improved = 0
        if order == 'random':
            perm = np.random.permutation(n)
        else:
            perm = np.arange(n)
        for i in perm:
            coefs = np.zeros(n_top)
            has_k0 = False
            for t in range(n_top):
                k = ts[t]
                c = 0.0
                jp, jm = i + k, i - k
                if 0 <= jp < n:
                    c += 1.0 - best_h[jp]
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
                if k == 0:
                    has_k0 = True
            old = best_h[i]
            base = corr[top_idx] - old * coefs
            def obj(t):
                vals = base + t * coefs
                if has_k0:
                    k0_pos = np.where(ts == 0)[0]
                    for pos in k0_pos:
                        vals[pos] = corr[top_idx[pos]] + t - t*t - old + old*old
                return np.max(vals)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            tn = res.x
            if abs(tn - old) < 1e-12:
                continue
            ht = best_h.copy()
            ht[i] = tn
            ht = project_h(ht)
            cv = compute_c5(ht)
            if cv < best_c5 - 1e-15:
                best_c5 = cv
                best_h = ht
                improved += 1
        if improved == 0:
            break
    return best_h, best_c5

def pair_search(h, n, t_end, n_iters=6000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        i, j = np.random.randint(n, 2)
        d = np.random.choice([0.01, 0.03, 0.08, 0.15, 0.3])
        if np.random.rand() < 0.5:
            d = -d
        vi, vj = best_h[i] + d, best_h[j] - d
        if vi < -1e-9 or vi > 1 + 1e-9 or vj < -1e-9 or vj > 1 + 1e-9:
            continue
        ht = best_h.copy()
        ht[i], ht[j] = vi, vj
        cv = compute_c5(ht)
        if cv < best_c5 - 1e-15:
            best_c5 = cv
            best_h = ht
    return best_h, best_c5

def targeted_perturb(h, n, scale):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-8:]
    shifts = top_idx - (n - 1)
    pert = np.zeros(n)
    for k in shifts:
        if k >= 0 and n - k > 0:
            pert[:n-k] += (h[k:n] - 0.5)
        elif k < 0 and n + k > 0:
            pert[-k:n] += (h[:n+k] - 0.5)
    return h + scale * pert

def multi_start_optimize(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t_phase = t_end - (t_end - time.time()) * 0.35
    # Broad coord descent
    best_h, best_c5 = coord_desc(best_h, n, t_phase, n_top=80, max_sweeps=150, order='random')
    # Focused coord descent
    t_focus = t_end - (t_end - time.time()) * 0.25
    best_h, best_c5 = coord_desc(best_h, n, t_focus, n_top=30, max_sweeps=200, order='random')
    # Pair search
    t_pair = t_end - (t_end - time.time()) * 0.12
    best_h, best_c5 = pair_search(best_h, n, t_pair, n_iters=4000)
    # Final coord descent
    best_h, best_c5 = coord_desc(best_h, n, t_end - 5, n_top=20, max_sweeps=100, order='sequential')
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 200
        init_h = np.zeros(n0)
        init_h[:n0 // 2] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Main optimization at parent n
    t1 = t_end - (t_end - t0) * 0.50
    best_h, best_c5 = multi_start_optimize(best_h, n, t1)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 2: Restart with perturbation from best
    if t_end - time.time() > budget_s * 0.20:
        for restart in range(3):
            pert_scale = 0.06 * (restart + 1)
            hp = project_h(targeted_perturb(best_h, n, pert_scale))
            hr, cr = coord_desc(hp, n, t_end - (t_end - time.time()) * 0.3,
                                n_top=40, max_sweeps=100, order='random')
            hr, cr = pair_search(hr, n, t_end - (t_end - time.time()) * 0.2, n_iters=2000)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr
            else:
                break

    # Phase 3: Try alternative n values
    for n2 in [160, 240, 280]:
        if t_end - time.time() < 55:
            break
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        t_sub = t_end - (t_end - time.time()) * 0.25
        h2, c5_2 = coord_desc(h2, n2, t_sub, n_top=50, max_sweeps=120, order='random')
        h2, c5_2 = pair_search(h2, n2, t_end - (t_end - time.time()) * 0.15, n_iters=2000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            n = n2

    # Phase 4: Final polish
    if t_end - time.time() > 15:
        best_h, best_c5 = coord_desc(best_h, n, t_end - 8, n_top=30, max_sweeps=80, order='sequential')
    if t_end - time.time() > 8:
        best_h, best_c5 = pair_search(best_h, n, t_end - 2, n_iters=2000)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
