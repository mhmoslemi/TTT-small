```python
def smooth_c5_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1, index 0..2n-2
    m_val = np.max(corr)
    shifted = T * (corr - m_val)
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = (m_val + np.log(sum_exp) / T) * dx
    w = exp_s / sum_exp  # softmax weights, shape (2n-1,)
    
    # Gradient
    grad = np.zeros(n)
    k_range = np.arange(-(n-1), n)  # k values
    for m in range(n):
        # term1: (1-h[m-k]) for k where 0 <= m-k < n
        valid1 = (m - k_range >= 0) & (m - k_range < n)
        # term2: -h[m+k] for k where 0 <= m+k < n
        valid2 = (m + k_range >= 0) & (m + k_range < n)
        g = np.zeros(2*n-1)
        g[valid1] = (1.0 - h[m - k_range[valid1]]) * w[valid1]
        g[valid2] += (-h[m + k_range[valid2]]) * w[valid2]
        grad[m] = np.sum(g) * dx
    return val, grad
```

This is O(n²) per gradient evaluation, which for n=160 is 25600 * some constant. Should be fine.

Now, for the paired coordinate descent, I can exploit the structure:
- Fix all variables except h[i] and h[j]
- The objective becomes max_k (base_k + x*a_k + y*b_k) where x=h[i], y=h[j]
- With constraint x+y = const (to preserve sum)
- This reduces to a 1D problem: max_k (base_k + x*(a_k+b_k) + const*(b_k))
- which is max of linear functions of x → convex piecewise linear
- Minimize over x in [0,1] with y = const - x in [0,1]

This is a 1D bounded optimization that minimize_scalar can handle, and the function is convex (max of linear functions), so it's well-conditioned.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_val = np.max(corr)
    shifted = np.clip(T * (corr - m_val), -50.0, 50.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = (m_val + np.log(sum_exp) / T) * dx
    w = exp_s / sum_exp
    k_range = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for m in range(n):
        v1 = (m - k_range >= 0) & (m - k_range < n)
        v2 = (m + k_range >= 0) & (m + k_range < n)
        g = np.zeros(2 * n - 1)
        g[v1] = (1.0 - h[m - k_range[v1]]) * w[v1]
        g[v2] += (-h[m + k_range[v2]]) * w[v2]
        grad[m] = np.sum(g) * dx
    return val, grad

def smooth_obj(h, T):
    return smooth_obj_grad(h, T)[0]

def smooth_grad(h, T):
    return smooth_obj_grad(h, T)[1]

def slsqp_step(h, T, n, maxiter=800, ftol=1e-16):
    def objfun(x):
        return smooth_obj(x, T)
    def gradfun(x):
        return smooth_grad(x, T)
    def confun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(objfun, h, jac=gradfun, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': confun}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def paired_coord(h, n, i, j, t_end):
    if time.time() > t_end:
        return h
    h_base = h.copy()
    h_base[i] = 0.0
    h_base[j] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    coef_i = np.zeros(2 * n - 1)
    v1 = (i + k_range >= 0) & (i + k_range < n)
    v2 = (i - k_range >= 0) & (i - k_range < n)
    coef_i[v1] += (1.0 - h_base[i + k_range[v1]])
    coef_i[v2] += (-h_base[i - k_range[v2]])
    coef_j = np.zeros(2 * n - 1)
    v1 = (j + k_range >= 0) & (j + k_range < n)
    v2 = (j - k_range >= 0) & (j - k_range < n)
    coef_j[v1] += (1.0 - h_base[j + k_range[v1]])
    coef_j[v2] += (-h_base[j - k_range[v2]])
    dx = 2.0 / n
    tgt = h[i] + h[j]
    if tgt < 0 or tgt > 2:
        return h
    lo = max(0.0, tgt - 1.0)
    hi = min(1.0, tgt)
    if hi - lo < 1e-14:
        return h
    combined = coef_i + coef_j
    const_part = (tgt - 1.0) * coef_j
    def obj1d(x):
        return np.max(corr_base + x * combined + const_part) * dx
    try:
        res = minimize_scalar(obj1d, bounds=(lo, hi), method='bounded',
                              options={'xatol': 1e-13})
        x_new = float(res.x)
        y_new = tgt - x_new
        if 0.0 <= x_new <= 1.0 and 0.0 <= y_new <= 1.0:
            h[i] = x_new
            h[j] = y_new
    except Exception:
        pass
    return h

def subgradient_step(h, n, step, top_m=8):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    idx = np.argpartition(-corr, top_m)[:top_m]
    wts = np.zeros(2 * n - 1)
    for k in idx:
        wts[k] = 1.0 / top_m
    k_range = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for mi in range(n):
        v1 = (mi - k_range >= 0) & (mi - k_range < n)
        v2 = (mi + k_range >= 0) & (mi + k_range < n)
        g = np.zeros(2 * n - 1)
        g[v1] = (1.0 - h[mi - k_range[v1]]) * wts[v1]
        g[v2] += (-h[mi + k_range[v2]]) * wts[v2]
        grad[mi] = np.sum(g) * dx
    h_new = h - step * grad
    return project_h(h_new)

def make_pert(n, cool, rng):
    mod = rng.randint(0, 5)
    if mod == 0:
        bs = max(4, n // 10)
        st = rng.randint(0, max(1, n - bs))
        p = np.zeros(n)
        p[st:st + bs] = (0.03 + 0.1 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 20))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.02 + 0.08 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = max(4, n // 10)
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.06 + 0.2 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.015 + 0.05 * cool) * rng.standard_normal(n)
    else:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
        p[:n//4] += 0.03 * cool
        p[3*n//4:] -= 0.03 * cool
    return p

def upscale(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_up = np.interp(x_new, x_old, h, period=2.0)
    return project_h(h_up)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h160 = np.array(initial_h_values, dtype=float)
    else:
        best_h160 = np.zeros(160)
        best_h160[:80] = 1.0
    n160 = len(best_h160)
    best_h160 = project_h(best_h160)
    best_c5 = compute_c5(best_h160)

    # Also prepare n=200 upscaling
    h200 = upscale(best_h160, 200)
    c5_200 = compute_c5(h200)

    candidates = [(best_h160, best_c5, 160, h200, c5_200)]
    best_h = best_h160
    best_c5_val = best_c5
    best_n = 160

    # Phase 1: SLSQP continuation with analytical gradient on n=160
    for T in [100, 500, 2000, 8000, 30000, 100000]:
        if time.time() > t_end * 0.12:
            break
        hc = slsqp_step(best_h160, T, n160, maxiter=600, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h160 = hc
            best_h = hc
            best_c5_val = best_c5
            best_n = 160

    # Phase 2: SLSQP continuation on n=200
    n200 = 200
    for T in [100, 500, 2000, 8000, 30000, 100000]:
        if time.time() > t_end * 0.22:
            break
        hc = slsqp_step(h200, T, n200, maxiter=600, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < c5_200:
            c5_200 = c5v
            h200 = hc
        if c5v < best_c5_val:
            best_h = hc
            best_c5_val = c5v
            best_n = 200

    # Phase 3: Basin hopping on n=160
    work_h = best_h160.copy()
    work_c5 = compute_c5(work_h)
    for it in range(30):
        if time.time() > t_end * 0.38:
            break
        cool = 1.0 - 0.8 * (it / 30.0)
        pert = make_pert(n160, cool, rng)
        h_pert = project_h(work_h + pert)
        T_bh = 500 + 5000 * cool
        hc = slsqp_step(h_pert, T_bh, n160, maxiter=400, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0002
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5_val:
                best_h = hc.copy()
                best_c5_val = c5v
                best_n = 160

    # Phase 4: Basin hopping on n=200
    work_h200 = h200.copy()
    work_c5_200 = compute_c5(work_h200)
    for it in range(20):
        if time.time() > t_end * 0.48:
            break
        cool = 1.0 - 0.8 * (it / 20.0)
        pert = make_pert(n200, cool, rng)
        h_pert = project_h(work_h200 + pert)
        T_bh = 500 + 5000 * cool
        hc = slsqp_step(h_pert, T_bh, n200, maxiter=400, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0002
        if c5v < work_c5_200 or rng.rand() < np.exp(-(c5v - work_c5_200) / max(temp, 1e-9)):
            work_h200 = hc
            work_c5_200 = c5v
            if c5v < best_c5_val:
                best_h = hc.copy()
                best_c5_val = c5v
                best_n = 200

    # Phase 5: Paired coordinate descent on best solution
    bh = best_h.copy()
    bn = len(bh)

    while time.time() < t_end * 0.68 and n_iter < 200:
        i = rng.randint(bn)
        j = rng.randint(bn)
        if i == j:
            j = (j + 1) % bn
        h_new = paired_coord(bh, bn, i, j, t_end * 0.68)

    # Phase 6: Subgradient method on best

                best_n = bn

        else:
            step *= 0.7
        bh = bh_new
        if step < 1e-6:
            step = 0.005

            best_h = hc
            best_n = len(best_h)

    c5_cur = best_c5_val
    for it in range(300):
        if time.time() > t_end * 0.97:
            break
        i = rng.randint(bn)
        j = (i + rng.randint(1, bn)) % bn
        if i == j:
            continue

            best_h = hc

    best_h = project_h(best_h)
    best_c5_val = compute_c5(best_h)
    return best_h, best_c5_val, len(best_h)
