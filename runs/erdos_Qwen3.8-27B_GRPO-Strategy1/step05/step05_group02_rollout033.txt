```python
def coord_exact(h, n, j):
    """Find optimal h[j] to minimize C5 exactly."""
    dx = 2.0 / n
    # For each lag, compute a_lag and b_lag such that overlap_lag = a_lag*h[j] + b_lag + const
    # a_lag = (1-h[j-lag])*I[valid] - h[j+lag]*I[valid]
    # b_lag = h[j+lag]*I[valid] (the constant part from (1-h[j]) expansion)
    
    a = np.zeros(2*n-1)
    b_const = np.zeros(2*n-1)
    base = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        lag = k - (n-1)
        # overlap(lag) = sum_{n'} h[n'+lag]*(1-h[n']) for valid n'
        # h[j] appears as h[n'+lag] when n' = j-lag, and as h[n'] when n' = j
        # Contribution of h[j] to overlap(lag):
        #   As h[n'+lag]: h[j]*(1-h[j-lag]) if 0<=j-lag<n and 0<=j<n
        #   As h[n']: h[j+lag]*(1-h[j]) if 0<=j<n and 0<=j+lag<n
        a_coeff = 0.0
        b_coeff = 0.0
        base_val = 0.0
        
        # Compute full overlap(lag) without h[j] contribution, treating h[j] as 0
        h_local = h.copy()
        old_j = h_local[j]
        h_local[j] = 0.0
        
        # Recompute just this lag's overlap
        # overlap(lag) = sum_{n': 0<=n'<n, 0<=n'+lag<n} h[n'+lag]*(1-h[n'])
        if lag >= 0:
            ns = np.arange(n - lag)
        else:
            ns = np.arange(-lag, n)
        
        # h_local[n'+lag] * (1-h_local[n'])
        base_val = np.sum(h_local[ns + lag] * (1 - h_local[ns]))
        
        # Now add back h[j] contribution:
        # h[j] appears as h[n'+lag] when n' = j-lag (if valid)
        if 0 <= j-lag < n and 0 <= j < n:

            if lag == 0:

                # overlap contribution: h[j]*(1-h[j])
                # a_coeff should be (1-h[j]) but we set h[j]=0 in h_local, so...
                # Actually this is getting complicated for lag=0
                pass

        if 0 <= j < n and 0 <= j+lag < n:
            a_coeff += (-h[j+lag])  # coefficient of h[j] from -(1-h[j+lag])*h[j]... 
            b_coeff += h[j+lag]  # constant part: h[j+lag]

        b_const[k] = b_coeff
        base[k] = base_val
    
    # C5(h[j]) = max_k (base[k] + b_const[k] + a[k]*h[j]) * dx

    c = base + b_const

    # This is 1D convex optimization
    # Candidate points: 0, 1, and breakpoints
    # For efficiency, just check many points and refine
    
    # Actually, let's just do a fine grid + golden section
    # Or better: the max of linear functions is convex, so ternary search works
    
    def obj_1d(t):
        vals = c + a * t
        return np.max(vals) * dx

        if obj_1d(m1) < obj_1d(m2):

        else:
            lo = m1
```

- Or at the minimum of f_lin (which is a breakpoint or boundary)
- Or at t=0 or t=1

```python
    non_aff_mask[k_min:k_max+1] = False
    max_non = np.max(cb[non_aff_mask]) if non_aff_mask.any() else -np.inf
    
    # Grid
    t_grid = np.linspace(0, 1, 101)
    
    # For each t, compute max over affected:
    # Non-lag-0: cb_aff_non0 + B_non0*t + C_non0
    # Lag-0: cb[lag0] + t - t²
    
    non0_mask = np.ones(n_aff, dtype=bool)
    if lag0_pos >= 0:
        non0_mask[lag0_pos] = False
    
    cb_non0 = cb[affected[non0_mask]]
    B_non0 = B_arr[non0_mask]
    C_non0 = C_arr[non0_mask]
    
    # Affected values: shape (len(t_grid), n_non0)
    aff_non0 = cb_non0[None, :] + (B_non0[None, :] * t_grid[:, None] + C_non0[None, :])
    max_aff_non0 = np.max(aff_non0, axis=1)
    
    if lag0_pos >= 0:
        cb_lag0 = cb[affected[lag0_pos]]
        aff_lag0 = cb_lag0 + t_grid * (1.0 - t_grid)
        max_aff = np.maximum(max_aff_non0, aff_lag0)
    else:
        max_aff = max_aff_non0
    
    max_all = np.maximum(max_aff, max_non)
    c5_vals = max_all * dx
    
    best_gi = np.argmin(c5_vals)
    return t_grid[best_gi], c5_vals[best_gi]
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (1.0 + diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_one(h, n, j):
    dx = 2.0 / n
    h_base = h.copy()
    h_base[j] = 0.0
    cb = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_min = max(0, 2 * j - n + 1)
    k_max = min(2 * n - 2, j + n - 1)
    affected = np.arange(k_min, k_max + 1)
    lags = affected - (n - 1)
    n_aff = len(affected)
    B_arr = np.zeros(n_aff)
    C_arr = np.zeros(n_aff)
    lag0 = -1
    for idx in range(n_aff):
        if lags[idx] == 0:
            lag0 = idx
            continue
        jl = j - lags[idx]
        jr = j + lags[idx]
        if 0 <= jl < n:
            B_arr[idx] += (1.0 - h[jl])
        if 0 <= jr < n:
            B_arr[idx] -= h[jr]
            C_arr[idx] += h[jr]
    non_aff = np.ones(2 * n - 1, dtype=bool)
    non_aff[k_min:k_max + 1] = False
    max_non = np.max(cb[non_aff]) if non_aff.any() else -np.inf
    t_grid = np.linspace(0.0, 1.0, 201)
    non0 = np.ones(n_aff, dtype=bool)
    if lag0 >= 0:
        non0[lag0] = False
    cb_n0 = cb[affected[non0]]
    B_n0 = B_arr[non0]
    C_n0 = C_arr[non0]
    aff = cb_n0[None, :] + (B_n0[None, :] * t_grid[:, None] + C_n0[None, :])
    max_aff = np.max(aff, axis=1)
    if lag0 >= 0:
        max_aff = np.maximum(max_aff, cb[affected[lag0]] + t_grid * (1.0 - t_grid))
    vals = np.maximum(max_aff, max_non) * dx
    gi = int(np.argmin(vals))
    return float(t_grid[gi]), float(vals[gi])

def coord_descent(h, t_end, max_sweeps=40, tlen=201):
    global _t_grid
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() > t_end:
                return best_h, best_c5
            new_t, new_val = coord_one(best_h, n, j)
            if abs(new_t - best_h[j]) > 1e-6:
                best_h[j] = new_t
                best_h = project_h(best_h)
                c5v = compute_c5(best_h)
                if c5v < best_c5 - 1e-13:
                    best_c5 = c5v
                    imp = True
        if not imp:
            break
    return best_h, best_c5

def subgrad_step(h, n, eta, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    mx = corr.max()
    if mx < 1e-15:
        return h.copy()
    w = (corr[idx] / mx) ** 6
    w /= w.sum()
    sg = np.zeros(n)
    for i, k in enumerate(idx):
        lag = k - (n - 1)
        ww = w[i]
        if lag >= 0:
            ni = np.arange(n - lag)
            sg[ni] += ww * (1.0 - h[ni + lag])
            ni2 = np.arange(lag, n)
            sg[ni2] += ww * (-h[ni2 - lag])
        else:
            la = -lag
            ni = np.arange(n - la)
            sg[ni] += ww * (-h[ni + la])
            ni2 = np.arange(la, n)
            sg[ni2] += ww * (1.0 - h[ni2 - la])
    return h - eta * dx * sg

def subgrad_phase(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    momentum = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t_step = 0.05
    eps = 1e-8
    it = 0
    patience = 0
    while time.time() < t_end:
        it += 1
        sg_raw = subgrad_step(best_h, n, 1.0, top_k=5)
        m_v = beta1 * m_v + (1 - beta1) * sg_raw
        v_v = beta2 * v_v + (1 - beta2) * sg_raw ** 2
        m_hat = m_v / (1 - beta1 ** it)
        v_hat = v_v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        best_h = best_h - t_step * update
        best_h = project_h(best_h)
        c5v = compute_c5(best_h)
        if c5v < best_c5:
            best_c5 = c5v
            patience = 0
        else:
            patience += 1
        if patience > 200:
            t_step *= 0.7
            patience = 0
        if it % 50 == 0:
            h_cd, c5_cd = coord_descent(best_h, time.time() + 5.0, max_sweeps=3)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd
                patience = 0
    return best_h, best_c5

def basin_hopping(h, n, t_end, rng, n_tries=60):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(n_tries):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.8 * (it / n_tries)
        pert = (0.03 + 0.15 * cool) * rng.standard_normal(n)
        h_pert = project_h(work_h + pert)
        T_s = 0.03 + 0.08 * cool
        sub_steps = int(30 + 40 * cool)
        hh = h_pert.copy()
        for _ in range(sub_steps):
            sg_raw = subgrad_step(hh, n, 1.0, top_k=3)
            hh = project_h(hh - T_s * sg_raw)
        hh, c5_sub = coord_descent(hh, time.time() + 3.0, max_sweeps=5)
        temp = 0.0004 * cool + 0.00005
        if c5_sub < work_c5 or rng.rand() < np.exp(-(c5_sub - work_c5) / max(temp, 1e-10)):
            work_h = hh
            work_c5 = c5_sub
        if work_c5 < best_c5:
            best_c5 = work_c5
            best_h = work_h.copy()
    return best_h, best_c5

def pair_swap(h, n, i, j):
    h2 = h.copy()
    h2[i], h2[j] = h[j], h[i]
    h2 = project_h(h2)
    return h2
```
