def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    if T < 0.01:
        return m * dx
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)

                   bounds=[(0.0, 1.0)] * n_pts,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return project_h(res.x, n_pts)

                g[:n-lag] += (1.0 - best_h[lag:n])  # h[j]*(1-h[j+lag])

            else:
                g[:n-lag] += (-best_h[-lag:n])  # -h[i] where i+lag=j
            
            grads[idx] = g

        # This is: min_C s.t. grads[idx] @ d <= C for all idx, and ||d|| <= 1
        # Approximate: find d in the span of grads that is most "negative"
        
        # Simple approach: average of negative gradients (normalized)
        g_sum = np.sum(grads, axis=0)
        norm = np.linalg.norm(grads, axis=1)
        g_avg = np.mean(grads, axis=0)
        
        # Project onto feasible direction (sum(d) ≈ 0 for sum constraint)
        g_avg -= np.mean(g_avg)
        
        g_norm = np.linalg.norm(g_avg)
        if g_norm < 1e-10:
            break
        d = -g_avg / g_norm
        
        # Clip direction to respect [0,1] bounds
        # Line search
        best_step = 0
        best_val = best_c5
        for step in np.linspace(0, 0.1, 20):
            h_try = project_h(best_h + step * d, n)
            c5_try = compute_c5(h_try)

                best_val = c5_try
                best_step = step

            best_h = project_h(best_h + best_step * d, n)
            best_c5 = best_val
        else:
            break
    
    return best_h, best_c5

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        frac = diff / tot
        h = h + room * frac
    else:
        if s < 1e-14:
            return h
        frac = 1.0 + diff / s
        h = h * frac
    return np.clip(h, 0.0, 1.0)

        accept = c5v < work_c5 or (c5v < work_c5 + 0.008 and rng.rand() < np.exp(-(c5v-work_c5)/temp))
        if accept:
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                # Quick CD to lock in improvement
                if elapsed() < budget_s * 0.60:
                    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.61, max_sweeps=25)
                    work_h = best_h.copy()
                    work_c5 = best_c5
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h
    
    # Phase 4: High-T L-BFGS-B polish
    for T in [200000, 1000000, 3000000]:
        if elapsed() > budget_s * 0.72:
            break
        hc = lbfgs_smooth(best_h, T, n, mu=300.0, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    
    # Phase 5: Extended coordinate descent (main precision phase)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, max_sweeps=500)
    
    # Phase 6: Final micro-refinement
    if elapsed() < budget_s * 0.97:
        for _ in range(4):
            if time.time() > t0 + budget_s * 0.975:
                break
            improved = False
            for i in range(n):
                if time.time() > t0 + budget_s * 0.975:
                    break
                if 0.005 < best_h[i] < 0.995:
                    h_try = best_h.copy()
                    h_try[i] += rng.uniform(-0.005, 0.005)
                    h_try = project_h(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
            if not improved:
                break
        # One more CD pass
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=30)
    
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
