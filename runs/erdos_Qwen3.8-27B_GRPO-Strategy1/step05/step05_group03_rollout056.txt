Actually, for the 1D optimization of a single variable h[i] (ignoring the constraint for now):
- corr changes linearly in h[i]
- So C₅ = max_k (a_k + b_k * h[i]) where a_k is corr[k] without the h[i] contribution and b_k is the coefficient

This makes the 1D problem a minimax of linear functions, which can be solved exactly!

The minimum of max_k (a_k + b_k * x) over x ∈ [0,1] can be found by:
- Binary search on the value V: check if ∃x ∈ [0,1] such that a_k + b_k * x ≤ V for all k
- This is a set of linear constraints on x, feasible iff max_k where b_k>0 of (V-a_k)/b_k ≤ min_k where b_k<0 of (V-a_k)/b_k ... 

Actually it's simpler: the function f(x) = max_k (a_k + b_k*x) is convex in x, so we can minimize it over [0,1] using ternary search or by finding the kink points.

But with the constraint sum(h) = n/2, changing h[i] by δ requires changing some other h[j] by -δ. This makes it 2D.

Let me just do: for each i, find the optimal h[i] assuming the sum constraint is handled by a "reservoir" variable. Or just do the 1D optimization and then projet.

Actually, the simplest approach that works well:
1. For each i, compute the coefficient vector c[k] = d(corr[k])/d(h[i])
2. The new objective is max_k (corr[k] - c[k]*h[i] + c[k]*h_new[i]) = max_k (corr[k] + c[k]*(h_new[i]-h[i]))
3. Minimize this over h_new[i] ∈ [0,1]
4. Then re-project to fix the sum

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h_init, n, T, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -400, 400)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h_init, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h_init

        T = T0 * (T1 / T0) ** progress
        i, j = np.random.choice(n, 2, replace=False)

            ni, nj = old_i + d, old_j - d

        elif r < 0.8:

            if not (0 <= ni <= 1 and 0 <= nj <= 1):

                continue

    best_c5 = compute_c5(best_h)

            if time.time() > t_end:

            # Compute effect of changing h[i]
            old = best_h[i]
            j1 = i + lags  # positions i+k for all lags k
            j2 = i - lags  # positions i-k for all lags k
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            # corr[k] changes by: (1-h[i+k]) for lag k (when i+k valid) 
            #                       - h[i-k] for lag -k (when i-k valid)
            # In the full correlation array (indexed 0 to 2n-2):
            # For lag = ki - (n-1), contribution to corr[ki] is:
            #   (1 - h[i+lag]) if i+lag in [0,n)  [from h[i]*(1-h[i+lag])]
            #   (-h[i-lag])  if i-lag in [0,n)     [from h[i-lag]*(1-h[i])]
            coefs = np.zeros(2*n-1)
            coefs[v1] += (1.0 - best_h[np.clip(j1[v1], 0, n-1)])
            coefs[v2] += (-best_h[np.clip(j2[v2], 0, n-1)])
            
            # f(new_val) = max_k (corr[k] + coefs[k]*(new_val - old))
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            
            # Minimize max_k (corr_base[k] + coefs[k]*(x - old)) over x in [0,1]
            # = minimize max_k ((corr_base[k] - coefs[k]*old) + coefs[k]*x)
            a = corr_base - coefs * old
            # f(x) = max_k (a[k] + coefs[k]*x)
            # This is convex in x. Use ternary search or find kinks.
            def f(x):
                return np.max(a + coefs * x)
            
            # Try to find minimum
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 50})
            x_new = res.x
            
            if abs(x_new - old) > 1e-10:
                best_h[i] = x_new
                best_h = project_h(best_h, n)
                c5_new = compute_c5(best_h)
                if c5_new < best_c5 - 1e-16:
                    best_c5 = c5_new
                    improved = True
        
        if not improved:
            break
    return best_h, best_c5

        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish on initial
    t_end1 = t0 + 30
    hc = slsqp(best_h, n, 5000, iters=300)
    c5v = compute_c5(hc)
    if c5v < best_c5:
        best_c5, best_h = c5v, hc

    # Phase 2: Direct SA (main exploration, no smoothing)
    t_end2 = t0 + budget_s * 0.35
    bh2, bc2 = sa_direct(best_h, n, t_end2, T0=0.003, T1=0.00003, steps=400000)
    if bc2 < best_c5:
        best_c5, best_h = bc2, bh2

    # Phase 3: SLSQP refinement at multiple temperatures
    t_end3 = t0 + budget_s * 0.50
    for T in [20000, 5000, 1000, 200, 50]:
        if time.time() > t_end3:
            break
        hc = slsqp(best_h, n, T, iters=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 4: Coordinate descent
    t_end4 = t0 + budget_s * 0.68
    best_h, best_c5 = coord_opt(best_h, n, t_end4, sweeps=120)

    # Phase 5: Multi-restart SA from perturbations of best
    t_end5 = t0 + budget_s * 0.82
    for restart in range(6):
        if time.time() > t_end5:
            break
        amp = 0.03 + 0.04 * (1.0 - restart / 6.0)
        h_pert = project_h(best_h + amp * np.random.randn(n), n)
        sub_end = t_end5
        if restart < 5:
            sub_end = t0 + budget_s * (0.70 + 0.024 * (restart + 1))
        bh5, bc5 = sa_direct(h_pert, n, sub_end, T0=0.002, T1=0.00002, steps=150000)
        if bc5 < best_c5:
            best_c5, best_h = bc5, bh5
        else:
            # Also try SLSQP on the perturbed version
            hc = slsqp(h_pert, n, 3000, iters=300)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc

    # Phase 6: Try different n values
    t_end6 = t0 + budget_s * 0.90
    for n_alt in [128, 200, 192]:
        if time.time() > t_end6:
            break
        h_alt = upsample(best_h, n_alt)
        hc = slsqp(h_alt, n_alt, 5000, iters=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h, n = c5v, hc, n_alt
        bh6, bc6 = sa_direct(h_alt, n_alt, min(t_end6, time.time()+60),
                             T0=0.003, T1=0.00002, steps=100000)
        if bc6 < best_c5:
            best_c5, best_h, n = bc6, bh6, n_alt

    # Phase 7: Final SLSQP + coordinate descent polish
    t_end7 = t0 + budget_s * 0.95
    for T in [50000, 1000000, 10]:
        if time.time() > t_end7:
            break
        hc = slsqp(best_h, n, T, iters=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 8: Final coordinate descent
    best_h, best_c5 = coord_opt(best_h, n, t0 + budget_s * 0.99, sweeps=60)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
