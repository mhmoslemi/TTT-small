            r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-9})
            xn = r.x
            if abs(xn - hi) < 1e-11:
                continue
            ht = best_h.copy()
            ht[i] = xn
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-13:
                best_h = ht
                best_c5 = c5t
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: L-BFGS warm start (8%)
    p1_end = t0 + budget_s * 0.08
    for T in [50, 200, 1000, 5000]:
        if time.time() > p1_end:
            break
        hc = lbfgs_opt(best_h, T, 200.0, 2000, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
    
    # Phase 2: Intensive coordinate descent (40%)
    p2_end = t0 + budget_s * 0.48
    best_h, best_c5 = coord_descent(best_h, p2_end, 600)
    
    # Phase 3: L-BFGS high-T refinement (15%)
    p3_end = t0 + budget_s * 0.63
    for T in [10000, 50000, 200000, 500000]:
        if time.time() > p3_end:
            break
        hc = lbfgs_opt(best_h, T, 300.0, 3000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
    
    # Phase 4: Basin hopping (20%)
    p4_end = t0 + budget_s * 0.83
    work_h, work_c5 = best_h.copy(), best_c5
    for it in range(40):
        if time.time() > p4_end:
            break
        frac = it / 40.0
        cool = 1.0 - 0.9 * frac
        sc = 0.015 + 0.06 * cool
        hp = project_h(work_h + sc * np.random.randn(n))
        T_bh = int(3000 * cool + 200)
        hc = lbfgs_opt(hp, T_bh, 300.0, 1200, 1e-15)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
    
    # Phase 5: Final coordinate descent polish (12%)
    p5_end = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, p5_end, 400)
    
    # Final L-BFGS push
    if time.time() < t0 + budget_s * 0.985:
        hc = lbfgs_opt(best_h, 1000000, 500.0, 2000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
    
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
