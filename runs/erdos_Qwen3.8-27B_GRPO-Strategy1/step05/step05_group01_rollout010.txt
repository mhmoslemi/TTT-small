                val = np.max(base + t * c)
                if val < best_val:
                    best_val = val
                    best_t = t
    return best_t

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(base, c):
    m = len(base)
    best_t = 0.0
    best_val = float(np.max(base))
    v1 = float(np.max(base + c))
    if v1 < best_val:
        best_val = v1
        best_t = 1.0
    for i in range(m - 1):
        for j in range(i + 1, m):
            dc = c[j] - c[i]
            if abs(dc) < 1e-15:
                continue
            t = (base[i] - base[j]) / dc
            if 0.0 < t < 1.0:
                val = float(np.max(base + t * c))
                if val < best_val:
                    best_val = val
                    best_t = t
    return best_t

def smooth_max_obj(x, h_shape, T, topk):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[-topk:]
    c_top = corr[top_idx]
    m = np.max(c_top)
    if T > 1e8:
        return m * dx
    return (m + np.log(np.sum(np.exp(T * (c_top - m)))) / T) * dx

def slsqp_solve(h, T, n, maxiter, topk):
    def obj(x):
        return smooth_max_obj(x, h, T, topk)
    def obj_grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        top_idx = np.argsort(corr)[-topk:]
        c_top = corr[top_idx]
        m = np.max(c_top)
        if T > 1e8:
            weights = np.zeros(topk)
            weights[np.argmax(c_top)] = 1.0
        else:
            weights = np.exp(T * (c_top - m)) / np.sum(np.exp(T * (c_top - m)))
        shifts = top_idx - (n - 1)
        grad = np.zeros(n)
        for ti in range(topk):
            if weights[ti] < 1e-8:
                continue
            s = shifts[ti]
            for i in range(n):
                jp = i + s
                jm = i - s
                g = 0.0
                if 0 <= jp < n:
                    g += (1.0 - x[jp])
                if 0 <= jm < n:
                    g -= x[jm]
                grad[i] += weights[ti] * g
        return grad * dx
    def constraint_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': constraint_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_h(np.clip(res.x, 0, 1))

def coord_descent(h, n, t_end, topk=25, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-topk:]
        top_shifts = top_idx - (n - 1)
        old_max = float(np.max(corr))
        any_change = False
        for i in range(n):
            h_i = best_h[i]
            c_arr = np.empty(topk)
            for ti in range(topk):
                s = top_shifts[ti]
                jp = i + s
                jm = i - s
                cp = (1.0 - best_h[jp]) if 0 <= jp < n else 0.0
                cm = -best_h[jm] if 0 <= jm < n else 0.0
                c_arr[ti] = cp + cm
            base_arr = corr[top_idx] - h_i * c_arr
            t_new = solve_1d_minimax(base_arr, c_arr)
            if abs(t_new - h_i) > 1e-10:
                best_h[i] = t_new
                any_change = True
        if any_change:
            best_h = project_h(best_h)
            new_max = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')))
            if new_max < best_c5 / dx - 1e-13:
                best_c5 = new_max * dx
        else:
            break
    return best_h, best_c5

def nesterov_subgrad(h, n, t_end, n_iter, topk):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    v = np.zeros(n)
    y = best_h.copy()
    beta = 0.9
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        alpha = 0.003 + 0.008 * cool
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[-topk:]
        shifts = top_idx - (n - 1)
        grad = np.zeros(n)
        for ti in range(topk):
            s = shifts[ti]
            for i in range(n):
                g = 0.0
                jp = i + s
                jm = i - s
                if 0 <= jp < n:
                    g += (1.0 - y[jp])
                if 0 <= jm < n:
                    g -= y[jm]
                grad[i] += g
        grad *= dx / topk
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        v_new = beta * v + (1.0 - beta) * grad
        y_new = y + (v - v_new) / (t_mom * alpha)
        y_new = np.clip(y_new, 0.0, 1.0)
        y_new = project_h(y_new)
        v = v_new * (1.0 + 0.3 * alpha / max(alpha, 1e-6))
        v = np.clip(v, -0.05, 0.05)
        best_h = y_new
        v = v_new
        c5v = compute_c5(best_h)
        if c5v < best_c5:
            best_c5 = c5v
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.35:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-8:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for s in shifts:
                if s >= 0 and n - s > 0:
                    pert[:n-s] += (work_h[s:] - 0.5)
                elif s < 0 and n + s > 0:
                    pert[-s:] += (work_h[:n+s] - 0.5)
            pert *= scale * 0.5 * cool
        elif r < 0.6:
            f = np.random.randint(2, 16)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.8 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.8:
            mask = np.random.random(n) < 0.25
            pert = np.zeros(n)
            cnt = max(int(mask.sum()), 1)
            pert[mask] = scale * 1.5 * cool * np.random.randn(cnt)
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = 200 + 3000 * cool
        hc = slsqp_solve(h_pert, T, n, 300, 20)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_block(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    t_now = time.time()
    if t_end - t_now > 50:
        for T in [200, 2000, 20000, 200000]:
            if t_end - time.time() < 15:
                break
            hc = slsqp_solve(best_h, T, n, 400, 25)

            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if t_end - time.time() > 80:
        t_cd = time.time() + (t_end - time.time()) * 0.45
        best_h, best_c5 = coord_descent(best_h, n, t_cd, topk=25, max_sweeps=400)
    if t_end - time.time() > 40:
        t_ns = time.time() + (t_end - time.time()) * 0.3
        nh, nc = nesterov_subgrad(best_h, n, t_ns, 100, 15)
        if nc < best_c5:
            best_c5 = nc
            best_h = nh
    if t_end - time.time() > 60:
        t_bh = time.time() + (t_end - time.time()) * 0.35
        bh_h, bh_c5 = basin_hop(best_h, n, t_bh, 35, 0.1)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    if t_end - time.time() > 40:
        t_cd2 = time.time() + (t_end - time.time()) * 0.4
        best_h, best_c5 = coord_descent(best_h, n, t_cd2, topk=35, max_sweeps=300)
    if t_end - time.time() > 20:
        hc = slsqp_solve(best_h, 500000, n, 600, 30)
        c5v = compute_c5(hc)
        if c5v < best_c5:

            best_h = hc
    if t_end - time.time() > 15:

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5
