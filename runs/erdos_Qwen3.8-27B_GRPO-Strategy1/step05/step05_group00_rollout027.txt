    #   - At i=p-k: gain h[p-k] where k=idx-(N-1), so gain h[p-idx+(N-1)]
    #     (valid if 0 <= p-idx+(N-1) < N)

    #   i = p - k_idx + (N-1)

    pk2 = p - k  # positions for the "gain" term
    valid_pk2 = (pk2 >= 0) & (pk2 < N)

    h_pk2 = np.where(valid_pk2, h[pk2_clipped], 0.0)

    # For zeros (flip 0->1):
    zeros = np.where(h == 0)[0]
    p0 = zeros[:, None]
    pk0 = p0 + k
    valid_pk0 = (pk0 >= 0) & (pk0 < N)

    h_pk02 = np.where(valid_pk02, h[np.clip(pk02, 0, N-1)], 0.0)
    
    D_zeros = 1.0 - h_pk0 - h_pk02  # (n_zeros, K)
    
    return D_ones, D_zeros, ones, zeros

def adam_opt(h, N, deadline, lr=0.05, k_top=50):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    
    m = np.zeros(N)  # First moment
    v = np.zeros(N)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        c5_val = float(corr.max()) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Subgradient: gradient of c(k_max) w.r.t. h
        # c(k) = sum_i h[i](1-h[i+k])
        # dc(k)/dh[j] = (1-h[j+k]) - h[j-k] (with bounds)
        # Use top-k peaks for subgradient
        top_idx = np.argpartition(corr, -k_top)[-k_top:]
        # Weight by how close to max
        weights = np.zeros(len(top_idx))
        for idx in top_idx:
            w = (corr[idx] - corr.min()) / (corr.max() - corr.min() + 1e-10)
            weights[corr[idx] == corr[idx]] = w ** 2  # quadratic weighting
        
        grad = np.zeros(N)
        center = N - 1
        for ai in top_idx:
            L = int(ai - center)
            w = (corr[ai] / (corr.max() + 1e-10)) ** 2
            if L == 0:
                grad += w * (1.0 - 2.0 * h)
            elif L > 0:
                # dc/dh[j] = (1-h[j+L]) - h[j-L]
                # for j+L < N: (1-h[j+L])
                # for j-L >= 0: -h[j-L]
                grad[:N - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:N - L]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += w * (1.0 - h[M:])
                    grad[M:] -= w * h[:N - M]
        
        grad /= k_top
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_sum(h_new, target)
        h = h_new
        
        # Encourage binary values periodically
        if t % 100 == 0:
            h = np.where(h > 0.7, 1.0, np.where(h < 0.3, 0.0, h))
            h = project_sum(h, target)
        
        # Decay learning rate
        if t % 200 == 0:
            lr *= 0.95
    
    return best_h, best_c5

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binarize_balanced(h, N):
    nh = N // 2
    order = np.argsort(-h)
    b = np.zeros(N)
    b[order[:nh]] = 1.0
    return b

def make_balanced_binary(h, N, rng):
    nh = N // 2
    n1 = int(np.round(h.sum()))
    b = (h >= 0.5).astype(float)
    n1 = int(b.sum())
    if n1 > nh:
        ones = np.where(b == 1)[0]
        to_remove = rng.choice(ones, n1 - nh, replace=False)
        b[to_remove] = 0
    elif n1 < nh:
        zeros = np.where(b == 0)[0]
        to_add = rng.choice(zeros, nh - n1, replace=False)
        b[to_add] = 1
    return b

def binary_search(h, N, deadline, max_sweeps=500, sample_n=150, anneal_T=0.005, restart_interval=100):

    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        h = make_balanced_binary(h, N, np.random)
    
    K = 2 * N - 1
    k_shifts = np.arange(K) - (N - 1)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > restart_interval:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            n_perturb = min(5, len(ones) // 4, len(zeros) // 4)
            if n_perturb > 0:
                p_idx = np.random.choice(ones, n_perturb, replace=False)
                q_idx = np.random.choice(zeros, n_perturb, replace=False)
                h[p_idx] = 0
                h[q_idx] = 1
            no_improve = 0
            continue
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_on = len(ones)

        if n_on < 1 or n_zer < 1:
            break
        
        # Compute D matrices
        p = ones[:, None]
        pk = p + k_shifts[None, :]
        v1 = (pk >= 0) & (pk < N)

def adam_continuous(h, N, deadline, lr=0.03, k_top=40):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    center = N - 1
    
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        c5v = float(corr.max()) * dx
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        
        top = np.argpartition(corr, -k_top)[-k_top:]
        grad = np.zeros(N)
        cmax = float(corr.max()) + 1e-10
        for ai in top:
            L = int(ai - center)
            w = (float(corr[ai]) / cmax) ** 2
            if L == 0:
                grad += w * (1.0 - 2.0 * h)
            elif L > 0:
                grad[:N-L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:N-L]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += w * (1.0 - h[M:])
                    grad[M:] -= w * h[:N-M]
        grad /= k_top
        
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*(grad**2)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        
        if t % 200 == 0:
            lr *= 0.97
        
        if t % 500 == 0:
            h = np.where(h > 0.75, 1.0, np.where(h < 0.25, 0.0, h))
            h = project_sum(h, target)
    
    return best_h, best_c5
