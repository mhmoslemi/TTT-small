import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def pair_descent(h, t_end, max_sweeps=3000, n_top=15, n_pairs=200, greedy_frac=0.3):
    n = len(h)
    dx = 2.0 / n
    bh = h.copy()
    bc5 = compute_c5(bh)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(bh, 1.0 - bh, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        shifts = top_idx - (n - 1)
        base = corr[top_idx] * dx
        
        # Compute adjacency: adj[t, i] = d(overlap_t)/d(h[i]) / dx
        adj = np.zeros((n_top, n))
        for t in range(n_top):
            k = shifts[t]
            for i in range(n):
                v = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n: v += 1.0 - bh[jp]
                if 0 <= jm < n: v -= bh[jm]
                adj[t, i] = v
        
        improved = False
        for _ in range(n_pairs):
            if time.time() > t_end:
                break
            # Pick pair
            if np.random.random() < greedy_frac:
                # Greedy: highest score vs lowest score
                scores = np.mean(adj[:min(n_top, 6)], axis=0)
                i = int(np.argmax(scores))
                j = int(np.argmin(scores))

            cands = [dlo, dhi, (dlo + dhi) * 0.5]

            im = int(np.argmin(objs))
            if objs[im] < bc5 - 1e-13:
                bh[i] += cands[im]
                bh[j] -= cands[im]
                bc5 = float(objs[im])
                improved = True
        if not improved:
            break
    return bh, bc5

def slsqp_polish(h, n, T=1e6, maxiter=800):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

        if not improved:
            break
    return bh, bc5

def slsqp_polish(h, n, T=1e6, maxiter=600):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    def obj_jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        w = np.exp(T * (corr - m))
        w /= w.sum()
        n = len(x)
        dx_ = 2.0 / n
        grad = np.zeros(n)
        for t in range(len(w)):
            if w[t] < 1e-15:
                continue
            k = t - (n - 1)
            lo1, hi1 = max(0, -k), min(n, n - k)
            lo2, hi2 = max(0, k), min(n, n + k)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - x[lo1 + k:hi1 + k])
            if hi2 > lo2:
                grad[lo2:hi2] -= x[lo2 - k:hi2 - k]
        grad *= dx_
        return grad

                       bounds=[(0.0, 1.0)] * n,
