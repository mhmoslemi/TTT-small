def subgrad_step(h, n, alpha, k_top=10):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top k shifts
    top_k = np.argsort(corr)[-k_top:]
    max_val = corr[top_k[0]]
    # Weight proportional to how close to max
    weights = np.exp(-(max_val - corr[top_k]) / (0.01 * max_val + 1e-10))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, m in zip(weights, top_k):
        s = (n - 1) - m  # shift
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += w * (1.0 - h[:n-s])  # (1-h[i-s]) for i >= s
            grad[:n-s] -= w * h[s:]  # -h[i+s] for i < n-s
        else:
            t = -s
            grad[:n-t] += w * (1.0 - h[t:])  # (1-h[i+t]) for i < n-t
            grad[t:] -= w * h[:n-t]  # -h[i-t] for i >= t
    
    return dx * grad
```

Wait, I need to be more careful. The derivative of c_s = sum_i h[i]*(1-h[i+s]) with respect to h[i]:
- From the first factor h[i]: (1-h[i+s]) if i+s < n
- From the second factor (1-h[(j)+s]) where (j+s) = i, i.e., j = i-s: -h[i-s] if i-s >= 0

So ∂c_s/∂h[i] = (1-h[i+s])·[i+s<n] - h[i-s]·[i-s≥0]

For the subgradient of max_s c_s, we use the subgradient of the active c_{s*}:
∂c_{s*}/∂h[i] = (1-h[i+s*])·[i+s*<n] - h[i-s*]·[i-s*≥0]

def subgradient_for_shift(h, n, s):
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        # (1-h[i+s]) for i in [0, n-s-1]
        grad[:n-s] += (1.0 - h[s:])
        # -h[i-s] for i in [s, n-1]
        grad[s:] -= h[:n-s]
    else:  # s < 0
        t = -s
        # (1-h[i-t]) for i in [t, n-1]  (i+s = i-t ≥ 0)
        # Wait, for s < 0: c_s = sum_i h[i]*(1-h[i+s]) = sum_i h[i]*(1-h[i-t])
        # ∂/∂h[i] = (1-h[i-t])·[i-t≥0] - h[i+t]·[i+t<n]
        # = (1-h[i-t])·[i≥t] - h[i+t]·[i<n-t]
        grad[t:] += (1.0 - h[:n-t])
        grad[:n-t] -= h[t:]
    return grad
```

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_1d_obj(corr_base, L, n_minus_1, delta):
    nc = corr_base + delta * L
    nc[n_minus_1] -= delta * delta
    return float(np.max(nc))

def obj_1d_v(corr_base, L, n_minus_1, old_val, v):
    return eval_1d_obj(corr_base, L, n_minus_1, v - old_val)

def coord_descent_fast(h, n, t_end, max_sweeps=5000, seed=42):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    nm1 = n - 1
    rng = np.random.RandomState(seed)
    improved_total = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return h, best_c5
            old = h[i]
            if old <= 1e-14 or old >= 1.0 - 1e-14:
                pass
            m_idx = np.arange(2 * n - 1)
            j1 = i + nm1 - m_idx
            j2 = i + m_idx - nm1
            L = np.zeros(2 * n - 1)
            v1 = (j1 >= 0) & (j1 < n)
            if v1.any():
                L[v1] += (1.0 - h[j1[v1]])
            v2 = (j2 >= 0) & (j2 < n)
            if v2.any():
                L[v2] -= h[j2[v2]]
            cb = corr.copy()
            res = minimize_scalar(lambda v: eval_1d_obj(cb, L, nm1, v - old),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 2e-9, 'maxiter': 30})
            v_new = float(res.x)
            delta = v_new - old
            if abs(delta) < 1e-12:
                continue
            corr_new = corr + delta * L
            corr_new[nm1] -= delta * delta
            c5_new = float(np.max(corr_new) * dx)
            if c5_new <= best_c5 + 1e-14:
                h[i] = v_new
                corr = corr_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                improved += 1
        improved_total += improved
        if improved == 0:
            break
    h = project_h(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def topk_subgrad(h, n, k_top=15, scale=0.008):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nm1 = n - 1
    kk = min(k_top, 2 * n - 1)
    top_idx = np.argpartition(corr, -kk)[-kk:]
    max_c = corr[top_idx[0]]
    spread = max(0.003, 0.5 * (max_c - corr[top_idx[-1]]))
    wts = np.exp(-(max_c - corr[top_idx]) / spread)
    wts /= wts.sum()
    grad = np.zeros(n)
    for w, m in zip(wts, top_idx):
        s = nm1 - int(m)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            t = -s
            grad[t:] += w * (1.0 - h[:n - t])
            grad[:n - t] -= w[t:] if t < n else 0
    return dx * grad

def subgrad_optimize(h, n, t_end, max_iters=50000, seed=0):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    nesterov_v = np.zeros(n)
    it = 0
    while it < max_iters and time.time() < t_end:
        batch = max(1, int(200 / (1 + it / 5000.0)))
        for _ in range(batch):
            if time.time() > t_end:
                break
            g = topk_subgrad(h_c, n, k_top=15)
            alpha = 0.08 / (1.0 + 0.0005 * it)
            nesterov_v = 0.7 * nesterov_v + g
            h_new = project_h(h_c - alpha * nesterov_v, n)
            h_c = h_new
            it += 1
        c5 = compute_c5(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()
    return best_h, best_c5

def perturb_h(h, n, rng, mag=0.04):
    mode = rng.randint(5)
    if mode == 0:
        p = mag * rng.randn(n)
    elif mode == 1:
        f = rng.randint(2, 20)
        ph = rng.uniform(0, 2 * np.pi)
        p = mag * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif mode == 2:
        p = np.zeros(n)
        mask = rng.random(n) < 0.3
        p[mask] = mag * 1.5 * rng.randn(int(mask.sum()))
    elif mode == 3:
        p = np.zeros(n)
        idx = rng.choice(n, size=max(4, n // 8), replace=False)
        p[idx] = mag * rng.randn(len(idx))
    else:
        blk = n // rng.randint(4, 12)
        p = np.tile(rng.randn(n // blk + 1)[:n // blk], n // (n // blk) + 1)[:n] * mag * 0.5
    return project_h(h + p, n)

def basin_hopping(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        mag = 0.02 + 0.07 * cool
        h_p = perturb_h(cur_h, n, rng, mag)
        remaining = max(1.0, (t_end - time.time()) / max(1, n_hops - hop - 1))
        hop_end = min(t_end, time.time() + min(remaining * 0.6, 6.0 + 10.0 * cool))
        h_opt, c5_opt = coord_descent_fast(h_p, n, hop_end, max_sweeps=150, seed=seed + hop)
        temp = 0.003 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def make_step_init(n, n_steps, rng):
    h = np.zeros(n)
    for i in range(n_steps):
        start = int(rng.randint(0, n - 1))
        width = int(rng.randint(1, n // n_steps))
        h[start:start + width] = 1.0
    return project_h(h, n)

            best_overall_h = best_h.copy()
            best_overall_n = ni

    # Final: if time remains, do one more polish on the best
    if time.time() < t0 + budget_s * 0.97 and best_overall_h is not None:
        best_overall_h, best_overall_c5 = coord_descent_fast(
            best_overall_h, best_overall_n, t0 + budget_s * 0.98, max_sweeps=2000, seed=seed + 99)

    best_overall_h = project_h(best_overall_h, best_overall_n)
    best_overall_c5 = compute_c5(best_overall_h)
    return best_overall_h, best_overall_c5, best_overall_n
