            cf[m2] += (-best_h[j2[m2]])
            try:
                res = minimize_scalar(_coord_eval, args=(cb, cf, dx),
                                      bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-13})
            except Exception:
                continue
            best_h[i] = float(res.x)
            best_h = project_h(best_h)
            c5_new = compute_c5(best_h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                imp = True
            else:
                best_h[i] = h_save
                best_h = project_h(best_h)
        if not imp:
            break
    return best_h, best_c5

def make_pert(n, cool, rng):
    mod = rng.randint(0, 5)
    if mod == 0:
        bs = rng.randint(4, max(5, n // 8))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.05 + 0.15 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 12))
        ph = rng.uniform(0.0, 6.28)
        x = np.arange(n)
        p = (0.05 + 0.15 * cool) * np.sin(6.28 * freq * x / n + ph)
    elif mod == 2:
        nn = rng.randint(5, max(6, n // 6))
        idx = rng.choice(n, nn, replace=False)
        p = np.zeros(n)
        p[idx] = (0.1 + 0.3 * cool) * rng.standard_normal(nn)
    elif mod == 3:
        p = (0.03 + 0.1 * cool) * rng.standard_normal(n)
    else:
        hf = n // 2
        p = np.zeros(n)
        p[:hf] = 0.15 * cool * rng.standard_normal(hf)
        p[hf:] = -0.15 * cool * rng.standard_normal(n - hf)
    return p
