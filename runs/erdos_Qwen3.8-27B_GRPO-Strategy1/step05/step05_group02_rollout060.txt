Actually, for the 1D problem min_δ max_k (a_k + δ*b_k), this is a linear programming problem in 1D. The optimal δ is at a point where at least two of the (a_k + δ*b_k) are equal and maximal. We can find it by:

Actually, the simplest approach: the function f(δ) = max_k (a_k + δ*b_k) is convex in δ (it's the max of linear functions). So we can use ternary search or golden section search, or just scipy.minimize_scalar.

Change: (a[0]+delta)(1-a[0]-delta) - a[0](1-a[0]) = (1-a[0])delta - a[0]delta - delta² ≈ (1-2a[0])delta (linear) ✓

Great, so the formula is:
coef[k] = (1-h[i+m]) * [0<=i+m<n] - h[i-m] * [0<=i-m<n]
where m = k-(n-1).

Now for coordinate descent, given current h and target to change h[i] to h[i]+delta:
new_corr = corr + delta * coef
new_c5 = max(new_corr) * dx

We want to find delta ∈ [-h[i], 1-h[i]] that minimizes max(corr + delta*coef).

This is a 1D convex optimization (max of linear functions is convex). We can use:
- scipy.minimize_scalar with bounds
- Or a custom bisection on the derivative (subgradient)

For efficiency, I'll precompute corr once and then for each i, compute coef and do the 1D optimization.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_and_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        h = h * (target / s)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            h = h * (target / max(s, 1e-14))
        else:
            h = h + room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def coord_coef(h, i, n):
    """Compute coefficient vector for changing h[i] by delta."""
    mm = np.arange(-(n-1), n)
    idx1 = i + mm
    idx2 = i - mm
    c = np.zeros(2*n-1)
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    c[v1] += (1.0 - h[idx1[v1]])
    c[v2] += (-h[np.clip(idx2[v2], 0, n-1)])  # Wait, I need to be careful
    # Actually idx2[v2] is already in range [0, n) by definition of v2
    c[v2] += (-h[idx2[v2]])
    return c

def coord_optimize_1d(corr, coef, h_i, n):
    """Find optimal delta for h[i], given corr and coefficient vector."""
    dx = 2.0 / n
    lo = -h_i
    hi = 1.0 - h_i
    if hi - lo < 1e-15:
        return h_i
    
    def f(delta):
        return np.max(corr + delta * coef) * dx
    
    try:
        res = minimize_scalar(f, bounds=(lo, hi), method='bounded', options={'xatol': 1e-12})
        return h_i + float(res.x)
    except:
        return h_i

def fast_coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    corr, best_c5 = corr_and_c5(best_h)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            coef = coord_coef(best_h, i, n)
            new_val = coord_optimize_1d(corr, coef, best_h[i], n)
            if abs(new_val - best_h[i]) < 1e-12:
                continue
            delta = new_val - best_h[i]
            new_corr = corr + delta * coef
            new_c5 = float(np.max(new_corr)) * dx
            if new_c5 < best_c5 - 1e-14:
                best_h[i] = new_val
                corr = new_corr
                best_c5 = new_c5
                improved = True
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_and_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h.copy()
    if s > target:
        h = h * (target / s)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h, 0, 1)
        h = h + room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_fn(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_solve(h, T, n, maxiter=500):
    try:
        res = minimize(smooth_c5_fn, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            return project_h(np.clip(res.x, 0, 1))
    except Exception:
        pass
    return h

def _max_corr_fn(d, corr, coef):
    return float(np.max(corr + d * coef))

def fast_cd(h, n, t_end, max_sweeps=50, rng=None, recompute_every=15):
    best_h = h.copy()
    corr, best_c5 = corr_and_c5(best_h)
    dx = 2.0 / n
    mm = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if sweep > 0 and sweep % recompute_every == 0:
            corr, _ = corr_and_c5(best_h)
        improved = False
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for i in order:
            if time.time() > t_end:
                break
            idx1 = i + mm
            idx2 = i - mm
            coef = np.zeros(2 * n - 1)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coef[v1] += (1.0 - best_h[idx1[v1]])
            coef[v2] += (-best_h[idx2[v2]])
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if hi - lo < 1e-14:
                continue
            best_delta = 0.0
            best_c5_local = best_c5
            try:
                res = minimize_scalar(_max_corr_fn, args=(corr, coef),
                                      bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-11})
                d = float(res.x)
                c5_d = float(np.max(corr + d * coef)) * dx
                if c5_d < best_c5_local - 1e-14:
                    best_c5_local = c5_d
                    best_delta = d
            except Exception:
                pass
            for d in [lo, hi]:
                c5_d = float(np.max(corr + d * coef)) * dx
                if c5_d < best_c5_local - 1e-14:
                    best_c5_local = c5_d
                    best_delta = d
            if abs(best_delta) > 1e-12:
                best_h[i] += best_delta
                corr += best_delta * coef
                best_c5 = best_c5_local
                improved = True
        if not improved:
            break
    return best_h, best_c5
