```python
def grad_smooth_max(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    shifted = T * (corr - np.max(corr))
    shifted = np.clip(shifted, -300, 300)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    # Only compute for significant weights
    idx_sorted = np.argsort(weights)[::-1]
    for idx in idx_sorted:
        w = weights[idx]
        if w < 1e-8:
            break
        d = idx - (n - 1)  # lag
        ad = abs(d)
        if ad == 0:
            # R = ∑ h[i]*(1-h[i]) = ∑ h[i] - ∑ h[i]^2
            # dR/dh[j] = 1 - 2*h[j]
            grad += w * (1 - 2*h)
        elif d > 0:
            # R = ∑_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # dR/dh[j] = (1-h[j+d]) for j in [0, n-1-d]
            #           + (-h[j-d]) for j in [d, n-1]
            grad[0:n-d] += w * (1 - h[d:n])
            grad[d:n] += w * (-h[0:n-d])
        else:
            # d < 0, let d' = -d > 0
            # R = ∑_{i=0}^{n-1-d'} h[i+d']*(1-h[i])
            # dR/dh[j] = (-h[j+d']) for j in [0, n-1-d']
            #           + (1-h[j-d']) for j in [d', n-1]
            dpp = -d
            grad[0:n-dpp] += w * (-h[dpp:n])
            grad[dpp:n] += w * (1 - h[0:n-dpp])
    
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.full(n, target / n)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.zeros(n)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_smooth(h, T, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    sh = T * (corr - cm)
    sh = np.clip(sh, -300, 300)
    w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w /= ws
    idx = np.argsort(w)[::-1][:top_k]
    grad = np.zeros(n)
    for ii in idx:
        wi = w[ii]
        if wi < 1e-10:
            break
        d = ii - (n - 1)
        if d == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n - d] += wi * (1.0 - h[d:])
            grad[d:] += -wi * h[:n - d]
        else:
            dp = -d
            grad[:n - dp] += -wi * h[dp:]
            grad[dp:] += wi * (1.0 - h[:n - dp])
    return grad * dx

def adam_opt(h, n, T, lr, max_iter, t_end):
    m = np.zeros(n)
    v = np.zeros(n)
    hc = h.copy()
    for it in range(max_iter):
        if time.time() > t_end:

        g = grad_smooth(hc, T)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        mh = m / (1.0 - 0.9 ** (it + 1))
        vh = v / (1.0 - 0.999 ** (it + 1))
        hc = project_h(hc - lr * mh / (np.sqrt(vh) + 1e-8), n)
    return hc

def coord_1d(h, n, i):
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h[i]  # already 0
    ol = np.zeros(2 * n - 1)
    lags = np.arange(-(n - 1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    # Need h values (with h[i]=0)
    hv = h.copy()
    ol[v1] += (1.0 - hv[np.clip(j1[v1], 0, n - 1)])
    ol[v2] += (-hv[np.clip(j2[v2], 0, n - 1)])
    def f(hi):
        return np.max(cb + hi * ol)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-8, 'maxiter': 25})
    return float(res.x)

def coord_desc(h, n, t_end, sweeps=100):
    bh = h.copy()
    bc5 = compute_c5(bh)
    for sw in range(sweeps):
        if time.time() > t_end:

        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return bh, bc5
            hi = coord_1d(bh, n, i)
            if abs(hi - bh[i]) < 1e-12:
                continue
            ht = bh.copy()
            ht[i] = hi
            ht = project_h(ht, n)
            c5 = compute_c5(ht)
            if c5 < bc5 - 1e-14:
                bc5 = c5
                bh = ht
                imp = True
        if not imp:
            break
    return bh, bc5

def pair_adjust(h, n, t_end, n_pairs=5000):
    bh = h.copy()
    bc5 = compute_c5(bh)
    remain = t_end - time.time()

                bh = ht

    return bh, bc5
```
