def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    old = h[i]
    h[i] = 0.0

    v1 = (j1 >= 0) & (j1 < n)

    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    coefs = _coord_coefs(h, n, i)
    def f(hi):
        return float(np.max(corr_base + hi * coefs))
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': xatol, 'maxiter': maxiter})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200, rng=None, ann_temp=0.0):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_1d(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
            elif ann_temp > 0 and c5_try < best_c5 + ann_temp:
                if rng.random() < np.exp(-(c5_try - best_c5) / max(ann_temp, 1e-10)):
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved and ann_temp > 0:
            ann_temp *= 0.9
        if not improved and ann_temp == 0:
            break
    return best_h, best_c5

def pair_swap_descent(h, n, t_end, rng, n_pairs=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    count = 0
    while time.time() < t_end and count < n_pairs:
        i, j = rng.choice(n, 2, replace=False)
        vi, vj = best_h[i], best_h[j]
        if abs(vi - vj) < 1e-6:
            count += 1
            continue
        h_try = best_h.copy()
        h_try[i], h_try[j] = vj, vi
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
        count += 1
    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def fourier_init(ns, rng):
    h = np.full(ns, 0.5)
    ncomp = rng.randint(2, 5)
    for _ in range(ncomp):
        freq = rng.randint(2, ns // 3)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.35)
        h += amp * np.cos(2 * np.pi * freq * np.arange(ns) / ns + phase)
    return project_h(np.clip(h, 0, 1), ns)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def remaining():
        return max(0, t0 + budget_s - time.time())

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Short SLSQP to smooth (8%)
    t1 = t0 + budget_s * 0.08
    for T in [8000, 3000]:
        if time.time() > t1:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=120)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Main coordinate descent (42%)
    t2 = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=500, rng=rng)

    # Phase 3: Targeted perturbation + coord descent (22%)
    t3 = t0 + budget_s * 0.72
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 15
    for it in range(n_iter):
        if time.time() > t3:
            break
        sub_budget = min(remaining() * 0.9, (t3 - time.time()) / max(1, n_iter - it))
        cool = 1.0 - it / n_iter
        amp = 0.03 + 0.07 * cool
        h_pert = project_h(work_h + amp * rng.randn(n), n)
        t_sub = time.time() + sub_budget * 0.6
        hc, c5v = coord_descent(h_pert, n, t_sub, max_sweeps=60, rng=rng)
        if c5v < work_c5 * (1 - 1e-5) or c5v < work_c5 - 1e-5:
            work_h = hc
            work_c5 = c5v
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Alternative starting points (12%)
    t4 = t0 + budget_s * 0.84
    for trial in range(6):
        if time.time() > t4:
            break
        sub_t = min(time.time() + (t4 - time.time()) / (6 - trial) * 0.9, t4)
        if trial % 2 == 0:
            h_alt = fourier_init(n, rng)
        else:
            h_alt = np.zeros(n)
            h_alt[rng.choice(n, n // 2, replace=False)] = 1.0
        hc = slsqp_smooth(h_alt, 2000, n, maxiter=80)
        hc, c5v = coord_descent(hc, n, sub_t, max_sweeps=40, rng=rng)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        # Binary init + coord
        h_bin = np.zeros(n)
        h_bin[rng.choice(n, n // 2, replace=False)] = 1.0
        hc2, c5v2 = coord_descent(h_bin, n, sub_t, max_sweeps=30, rng=rng)
        if c5v2 < best_c5:
            best_c5 = c5v2
            best_h = hc2

    # Phase 5: Pair swaps (5%)
    t5 = t0 + budget_s * 0.89
    best_h, best_c5 = pair_swap_descent(best_h, n, t5, rng, n_pairs=1000)

    # Phase 6: Final intensive coordinate descent (10%)
    t6 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=500, rng=rng, ann_temp=1e-6)

    # Phase 7: Ultra-fine final polish
    if remaining() > 5:
        best_h, best_c5 = coord_descent(best_h, n, time.time() + remaining() * 0.8,
                                        max_sweeps=100, rng=rng)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
