        h_f, c5_f = fista_step(best_h, t1, T_s, T_e, ni, 40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    # Phase 2: Coordinate descent
    t2 = t_end - (t_end - time.time()) * 0.25
    best_h, best_c5 = coord_descent(best_h, n, t2, 60, 25)
    # Phase 3: Perturb + refine
    t3 = t_end - (t_end - time.time()) * 0.30
    best_h, best_c5 = perturb_and_refine(best_h, n, t3, 25, 0.10)
    # Phase 4: Final coord descent
    best_h, best_c5 = coord_descent(best_h, n, t_end, 50, 30)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s * 0.97

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=np.float64)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Main optimization at parent resolution
    t_main_end = t0 + budget_s * 0.50
    best_h, best_c5 = optimize_combined(best_h, n, t_main_end)

    # Try n=160 with interpolation
    if t_end - time.time() > budget_s * 0.25 and n != 160:
        n2 = 160
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        t2_end = t0 + budget_s * 0.72
        h2o, c5_2 = optimize_combined(h2, n2, t2_end)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2o
            n = n2

    # Try n=240
    if t_end - time.time() > budget_s * 0.20 and n != 240:
        n3 = 240
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = project_h(h3)
        t3_end = t0 + budget_s * 0.90
        h3o, c5_3 = optimize_combined(h3, n3, t3_end)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3o
            n = n3

    # Final refinement at best n
    if t_end - time.time() > 20:
        n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, n, t_end - 5, 80, 35)
    if t_end - time.time() > 15:
        n = len(best_h)
        h_f, c5_f = fista_step(best_h, t_end - 3, 5000, 10, 1500, 50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
            best_h, best_c5 = coord_descent(best_h, len(best_h), t_end - 1, 40, 30)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
