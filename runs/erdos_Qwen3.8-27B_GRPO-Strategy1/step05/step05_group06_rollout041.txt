        if not active.any():
            active = (v == mv)
        sglo = A[active].min()
        sghi = A[active].max()
        if sglo <= 1e-11 and sghi >= -1e-11:
            break
        elif sglo > 0:
            b = mid
        else:
            a = mid
    y = 0.5 * (a + b)
    y = float(np.clip(y, lo, hi))
    v0 = np.max(corr + y * A)
    for yc in (lo, hi):
        vc = np.max(corr + yc * A)
        if vc < v0:
            y = yc
    return y

def cd_optimize(h0, t_end, max_sweeps=10000, stagnation_limit=80):
    n = len(h0)
    m = 2 * n - 1
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(np.max(corr) * dx)
    stag = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        order = np.random.permutation(n)
        for i in order:
            hi = h[i]
            A = np.zeros(m)
            for lag in range(n):
                if lag <= i:
                    A[n + 1 - n + lag] += 1.0 - h[i - lag]
                if i + lag < n:
                    A[n - 1 + lag] -= h[i + lag]
            for p in range(1, n):
                if i + p < n:
                    A[n - 1 - p] += 1.0 - h[i + p]
                if i >= p:
                    A[n - 1 - p] -= h[i - p]
            lo_y, hi_y = -hi, 1.0 - hi
            if hi_y - lo_y < 1e-14:
                continue
            y = solve_1d(corr, A, lo_y, hi_y)
            hn = float(np.clip(hi + y, 0.0, 1.0))
            if abs(hn - hi) < 1e-14:
                continue
            corr += (hn - hi) * A
            h[i] = hn
            c5v = float(np.max(corr) * dx)
            if c5v < best_c5 - 1e-15:
                best_c5 = c5v
                best_h = h.copy()
                improved += 1
        h = project_h(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        pc5 = float(np.max(corr) * dx)
        if pc5 <= best_c5 + 1e-13:
            best_c5 = pc5
            best_h = h.copy()
        if improved == 0:
            stag += 1
            if stag >= stagnation_limit:
                break
        else:
            stag = 0
    return best_h, best_c5

def smooth_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    Tc = T * (corr - mc)
    Tc -= Tc.max()
    w = np.exp(Tc)
    w /= w.sum()
    sm = mc + np.log(w.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    for idx in range(m):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:n] += wi * (1.0 - h[0:n - lag])
            grad[0:n - lag] += -wi * h[lag:n]
        else:
            p = -lag
            grad[0:n - p] += wi * (1.0 - h[p:n])
            grad[p:n] += -wi * h[0:n - p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter=3000, ftol=1e-17):
    n = len(h0)
    x0 = project_h(h0)
    try:
        res = minimize(lambda x: smooth_grad(x, T, lam)[0], x0,
                       jac=lambda x: smooth_grad(x, T, lam)[1],
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 5e-9})
        return project_h(res.x)
    except Exception:
        return x0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        nn = 160
        best_h = np.zeros(nn)
        best_h[:nn // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: CD (50%)
    p1_end = t0 + budget_s * 0.50
    best_h, best_c5 = cd_optimize(best_h, p1_end, 20000, 60)

    # Phase 2: L-BFGS refinement (20%)
    p2_end = t0 + budget_s * 0.70
    for T in [3000, 10000, 50000, 200000]:
        if time.time() > p2_end:
            break
        hc = lbfgs_opt(best_h, T, 500.0, 4000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Phase 3: CD again (15%)
    p3_end = t0 + budget_s * 0.85
    best_h, best_c5 = cd_optimize(best_h, p3_end, 15000, 50)

    # Phase 4: Basin hop (7%)
    p4_end = t0 + budget_s * 0.92
    wh, wc = best_h.copy(), best_c5
    for it in range(50):
        if time.time() > p4_end:
            break
        cool = max(0.05, 1.0 - 0.95 * it / 50.0)
        sc = 0.005 + 0.04 * cool
        hp = project_h(wh + sc * np.random.randn(n))
        Tbh = int(2000 * cool + 500)
        hc = lbfgs_opt(hp, Tbh, 300.0, 2000, 1e-16)
        c5v = compute_c5(hc)
        temp = 8e-5 * cool + 2e-5
        if c5v < wc or np.random.rand() < np.exp(-(c5v - wc) / temp):
            wh, wc = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v

    # Phase 5: Final CD + L-BFGS (8%)
    if time.time() < t0 + budget_s * 0.96:
        best_h, best_c5 = cd_optimize(best_h, t0 + budget_s * 0.96, 5000, 40)
    if time.time() < t0 + budget_s * 0.99:
        hc = lbfgs_opt(best_h, 1000000, 800.0, 2000, 1e-18)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
