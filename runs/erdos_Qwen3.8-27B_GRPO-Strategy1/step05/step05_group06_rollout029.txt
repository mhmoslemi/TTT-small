    dx = 2.0 / n

    mc = np.max(corr)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lbfgs_step(h, T, lam, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        Tc = T * (corr - mc)
        Tc -= np.max(Tc)
        exps = np.exp(Tc)
        w = exps / exps.sum()
        sm = mc + np.log(exps.sum()) / T
        val = dx * sm + lam * (x.sum() - n / 2.0) ** 2
        grad = np.zeros(n)
        for idx in range(m_full):
            lg = lag_arr[idx]
            wi = w[idx]
            if wi < 1e-15:
                continue
            if lg >= 0:
                if lg < n:
                    grad[lg:] += wi * (1.0 - x[:n-lg])
                    grad[:n-lg] += -wi * x[lg:]
            else:
                p = -lg
                grad[:n-p] += wi * (1.0 - x[p:])
                grad[p:] += -wi * x[:n-p]
        grad = dx * grad + 2.0 * lam * (x.sum() - n / 2.0)
        return val, grad
    
    try:
        res = minimize(obj_grad, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lbfgs_refine(h, T, lam, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        Tc = T * (corr - mc) - Tc.max()
        exps = np.exp(Tc)
        w = exps / exps.sum()
        val = dx * (mc + np.log(exps.sum()) / T) + lam * (x.sum() - n/2.0)**2
        grad = np.zeros(n)
        for idx in range(m_full):
            lg = lag_arr[idx]
            wi = w[idx]
            if wi < 1e-15:
                continue
            if lg >= 0:
                if lg < n:
                    grad[lg:] += wi * (1.0 - x[:n-lg])
                    grad[:n-lg] += -wi * x[lg:]
            else:
                p = -lg
                grad[:n-p] += wi * (1.0 - x[p:])
                grad[p:] += -wi * x[:n-p]
        grad = dx * grad + 2.0 * lam * (x.sum() - n/2.0)
        return val, grad
    
    try:
        res = minimize(obj_grad, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except:
        return h

def coord_desc(h, n, t_end):
    best = h.copy()
    dx = 2.0 / n
    m = 2*n - 1
    corr = np.correlate(best, 1.0-best, mode='full').copy()
    c5 = float(np.max(corr) * dx)
    
    while time.time() < t_end:
        imp = 0
        for j in range(n):
            old = best[j]
            coeff = np.zeros(m)
            for l in range(-(n-1), n):
                c = 0.0
                if 0 <= j+l < n:
                    c += 1.0 - best[j+l]
                if 0 <= j-l < n:
                    c -= best[j-l]
                coeff[l+n-1] = c
            
            # Candidate x values
            cands = [0.0, 1.0]
            cm = np.max(corr)
            for ti in np.argsort(corr)[-8:]:
                if ti == n-1 or abs(coeff[ti]) < 1e-12:
                    continue
                xe = old + (cm - corr[ti]) / coeff[ti]
                if 0.0 <= xe <= 1.0:
                    cands.append(xe)
            
            bx, bv = old, c5
            for x in cands:
                nc = corr + (x-old)*coeff
                nc[n-1] -= (x-old)**2
                v = float(np.max(nc)*dx)
                if v < bv:
                    bv, bx = v, x
            
            if bx != old:
                lo, hi = max(0., bx-0.05), min(1., bx+0.05)
                def _f(xx):
                    nc = corr + (xx-old)*coeff
                    nc[n-1] -= (xx-old)**2
                    return np.max(nc)
                try:

                                       options={'xatol':1e-12})

                        bx, bv = r.x, float(r.fun*dx)
                except:
                    pass

                corr += (bx-old)*coeff
                corr[n-1] -= (bx-old)**2
                best[j] = bx
                c5 = bv
                imp += 1
        if imp == 0:
            break
    best = project_h(best)
    c5 = compute_c5(best)
    return best, c5

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)
