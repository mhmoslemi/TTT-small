def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def exact_coord_opt(h, n, i):
    """Exact 1D optimization: find the value of h[i] that minimizes max of linear functions."""
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    # Effect of h[i] on correlation
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n-1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n-1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n-1)])
    # corr_base + hi * coefs, minimize max
    # The function is max of linear functions in hi, so optimum is where two are equal
    # or at a boundary (0 or 1)
    # Find the upper envelope of (corr_base + hi * coefs) for hi in [0,1]
    # The max at hi is: max_k(corr_base[k] + hi * coefs[k])
    # This is piecewise linear, convex in hi
    # Minimum at point where the slope changes sign, or at boundary
    # Evaluate at boundaries and at intersection points
    # For efficiency, just use minimize_scalar with very tight tolerance
    def f(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 100})
    return res.x

    best_h = h.copy()

        return project_h(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt_i(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n - 1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    def f(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13, 'maxiter': 120})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_i(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_topk(h, n, k, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for ki in idx:
        lag = ki - (n - 1)
        s1 = max(0, -lag)
        e1 = min(n, n - lag) if lag > 0 else n
        if s1 < e1:
            grad[s1:e1] += (1.0 - h[s1 + lag:e1 + lag])
        s2 = max(0, lag)
        e2 = min(n, n + lag) if lag < 0 else n
        if s2 < e2:
            grad[s2:e2] += (-h[s2 - lag:e2 - lag])
    return project_h(h - alpha * grad, n)

def optimize_n(n, t_start, t_end, init_h=None, seed=42):
    np.random.seed(seed)
    best_h, best_c5 = None, 1e10

    starts = []
    if init_h is not None:
        starts.append(project_h(init_h, n))
    # Random binary
    for _ in range(4):
        rb = np.zeros(n)
        rb[np.random.choice(n, n // 2, replace=False)] = 1.0
        starts.append(rb)
    # Block pattern
    blk = np.zeros(n)
    bsz, pos, val = max(3, n // 30), 0, 1.0
    while pos < n:
        blk[pos:min(pos + bsz, n)] = val
        pos += bsz
        val = 1.0 - val
    starts.append(project_h(blk, n))
    # Random smooth
    rs = 0.5 + 0.3 * np.sin(2 * np.pi * np.random.randint(2, 6) * np.arange(n) / n)
    rs += 0.1 * np.random.randn(n)
    starts.append(project_h(rs, n))

    for s in starts:
        if time.time() > t_end:
            break
        for T in [5000, 1000]:
            if time.time() > t_end:
                break
            hc = slsqp_smooth(s, T, n, maxiter=250)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
    if best_h is None:
        best_h = project_h(starts[0], n)
        best_c5 = compute_c5(best_h)

    # SLSQP annealing
    for T in [30000, 8000, 3000, 1000, 400, 150, 60, 20, 8]:
        if time.time() > t_end:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=350)
        c5v = compute_c5(hc)

    # Subgradient + SLSQP
    for it in range(15):
        if time.time() > t_end:
            break
        cool = 1.0 - it / 15.0
        hs = subgrad_topk(best_h, n, 4 + int(4 * cool), 0.005 + 0.008 * cool)
        hc = slsqp_smooth(hs, 200 + 1200 * cool, n, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:

    # Coord descent (medium)
    best_h, best_c5 = coord_descent(best_h, n, min(t_end, time.time() + 30), max_sweeps=80)

    # Basin hopping
    wh, wc5 = best_h.copy(), best_c5
    for it in range(25):
        if time.time() > t_end:
            break
        cool = max(0.1, 1.0 - it / 25.0)

        hc = slsqp_smooth(hp, 400 + 2000 * cool, n, maxiter=300)
        c5v = compute_c5(hc)
        temp = 0.0006 * cool + 0.00004
        if c5v < wc5 or np.random.rand() < np.exp(-(c5v - wc5) / temp):
            wh, wc5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
    if wc5 < best_c5:
        best_c5, best_h = wc5, wh

    # Final tight coord descent
    best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=150)
    return best_h, best_c5
