import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -0.5, 0.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos + sv
    pp = pos - sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    gm = np.where(vm, (1.0 - h)[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return gm - hp

def lse_gradient(h, N, T):
    c = fft_corr(h)
    cmax = c.max()
    sc = T * (c - cmax)
    sc -= sc.max()  # numerical stability (should be 0 already)
    w = np.exp(sc)
    w /= w.sum()
    D = compute_D(h, N)
    grad = D @ w
    return grad, c, float(cmax)

def nesterov_lse(h, N, deadline, T0=20, T1=200, lr0=0.01, iters=None):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    total_iters = iters if iters else 3000
    while time.time() < deadline and it < total_iters:
        T = T0 + (T1 - T0) * min(1.0, it / (total_iters * 0.7))
        y = h + m
        grad, c, pk = lse_gradient(y, N, T)
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        h_new = project_sum(h - lr * grad, target)
        m = beta * (h_new - h) + (1 - beta) * m  # actually standard Nesterov: m = beta*(h_new-h)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.98
            beta = 0.85 + 0.0001 * it
            beta = min(beta, 0.95)
    return best_h, best_pk * dx

    # Phase 5: Upscale to 760
    N1 = 760
    if time.time() < deadline - 150:
        h1u = upscale(best_h, N1)
        h1b = fix_binary_count(h1u, N1)
        d5 = min(deadline - 100, time.time() + 45)
        h5, c5 = binary_search_flat(h1b, N1, d5, sample_n=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5.copy()
        if time.time() < deadline - 60:
            d5b = min(deadline - 40, time.time() + 55)
            h5n = h5.astype(float) + np.random.randn(N1) * 0.03
            h5n = project_sum(h5n, N1 / 2.0)
            h5n, c5n = nesterov_lse(h5n, N1, d5b, T0=15, Tend=250, lr0=0.008)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h5n.copy()

    # Phase 6: Binary + Nesterov polish at best resolution
    Nb = len(best_h)
    if time.time() < deadline - 50:
        d6 = min(deadline - 35, time.time() + 30)
        h6, c6 = binary_search_flat((best_h > 0.5).astype(float), Nb, d6, sample_n=150)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6.copy()
        if time.time() < deadline - 15:
            d6b = min(deadline - 8, time.time() + 20)
            h6n = best_h + np.random.randn(Nb) * 0.02
            h6n = project_sum(h6n, Nb / 2.0)
            h6n, c6n = nesterov_lse(h6n, Nb, d6b, T0=40, Tend=350, lr0=0.005)
            if c6n < best_c5:
                best_c5 = c6n
                best_h = h6n.copy()

    # Phase 7: Final binary cleanup
    if time.time() < deadline - 1:
        Nf = len(best_h)
        hF = (best_h > 0.5).astype(float)
        hF, cF = binary_search_flat(hF, Nf, deadline - 0.5, sample_n=80)
        if cF < best_c5:
            best_c5 = cF
            best_h = hF

    # Ensure exact constraint and validate
    best_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
