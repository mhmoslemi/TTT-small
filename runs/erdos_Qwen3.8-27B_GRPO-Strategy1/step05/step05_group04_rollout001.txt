```python
import numpy as np
from scipy.optimize import minimize_scalar, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            m = h > 1e-12
            if m.sum() == 0:
                break
            h[m] -= (s - target) / m.sum()
        else:
            m = h < 1.0 - 1e-12
            if m.sum() == 0:
                break
            h[m] += (target - s) / m.sum()
        h = np.clip(h, 0, 1)
    return h

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep >= 2 and sweep % 5 == 4 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                if i + s < n:
                    lin[mn] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[mn] -= best_h[i - s]
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            def obj(v):
                d = v - old
                vals = c_base + d * lin
                vals[n - 1] -= d * d
                return float(np.max(vals) * dx)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-12:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    return best_h, best_c5

def periodic_obj(g, n, n_per, mode):
    h = np.tile(g, n // n_per)
    if mode == 'anti':
        h = np.zeros(n)
        reps = n // n_per
        for r in range(reps):
            start = r * n_per
            if r % 2 == 0:
                h[start:start+n_per] = g
            else:
                h[start:start+n_per] = 1.0 - g
    h = project_h(h, n)
    return compute_c5(h)

def multi_res_search(h, n, t_end, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for factor in [2, 4, 2]:
        n_coarse = max(20, n // factor)
        h_c = np.array([h[i*n//n_coarse:(i+1)*n//n_coarse].mean()
                        for i in range(n_coarse)])
        h_c = project_h(h_c, n_coarse)
        for hop in range(12):
            if time.time() > t_end:
                break
            f = rng.randint(1, 8)
            ph = rng.uniform(0, 2*np.pi)
            step = 0.03 + 0.05 * (1 - hop / 12)
            pert = step * np.sin(2*np.pi * f * np.arange(n_coarse) / n_coarse + ph)
            h_p = project_h(h_c + pert, n_coarse)
            t_sub = min(t_end, time.time() + 3.0)
            h_opt, c5_opt = coord_descent(h_p, n_coarse, t_sub, max_sweeps=60,
                                          seed=seed + hop)
            if c5_opt < compute_c5(h_c):
                h_c = h_opt
        h_up = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_coarse), h_c)
        h_up = project_h(h_up, n)
        t_sub = min(t_end, time.time() + 8.0)
        h_up, c5_up = coord_descent(h_up, n, t_sub, max_sweeps=100, seed=seed)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=500, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - hop / n_hops)
        step = 0.01 + 0.08 * cool
        mode = hop % 6
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2*np.pi)
            pert = step * np.sin(2*np.pi*f*np.arange(n)/n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = step * 3 * rng.randn(mask.sum())
        elif mode == 3:
            idx = rng.choice(n, size=max(6, n//12), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 1.5 * rng.randn(len(idx))
        elif mode == 4:
            pert = step * 0.5 * rng.randn(n)
        else:
            f1 = rng.randint(1, 6)
            f2 = rng.randint(7, 20)
            pert = step*(np.sin(2*np.pi*f1*np.arange(n)/n) +
                        0.5*np.cos(2*np.pi*f2*np.arange(n)/n))
        h_p = project_h(cur_h + pert, n)
        hop_budget = 2.0 + 6.0 * cool
        hop_end = min(t_end, time.time() + hop_budget)
        h_opt, c5_opt = coord_descent(h_p, n, hop_end, max_sweeps=60,
                                      seed=seed + hop)
        temp = 0.005 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and
                               rng.rand() < np.exp(-(c5_opt - cur_c5)/temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def fine_tune(h, n, t_end, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(200):
```
