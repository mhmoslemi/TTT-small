import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (target / s) if s > 1e-13 else h
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, t_end, max_sweeps=500):
    n = len(h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    best_c5 = float(np.max(corr) * dx)
    lag_arr = np.arange(m_full) - n + 1
    n_minus_1 = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = best_h[i]
            i_pl = i + lag_arr
            i_ml = i - lag_arr
            vp = (i_pl >= 0) & (i_pl < n)
            vm = (i_ml >= 0) & (i_ml < n)
            lin = np.zeros(m_full)
            lin[vp] += 1.0 - best_h[i_pl[vp]]
            lin[vm] -= best_h[i_ml[vm]]
            
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                d1 = m1 - old
                f1 = (corr + d1 * lin).max()
                t1 = corr[n_minus_1] + d1 * lin[n_minus_1] - d1 * d1
                if t1 > f1:
                    f1 = t1
                d2 = m2 - old
                f2 = (corr + d2 * lin).max()
                t2 = corr[n_minus_1] + d2 * lin[n_minus_1] - d2 * d2
                if t2 > f2:
                    f2 = t2
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            new_val = (lo + hi) * 0.5
            if abs(new_val - old) < 1e-13:
                continue
            d = new_val - old
            corr += d * lin
            corr[n_minus_1] -= d * d
            best_h[i] = new_val
            nc = float(np.max(corr) * dx)
            if nc < best_c5 - 1e-15:
                best_c5 = nc
                improved = True
        if not improved and sweep > 3:
            break
    corr_final = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr_final) * dx)
    return best_h, best_c5

def softmax_grad(h, T, lam):
    n = len(h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    w_dx = (w * dx).copy()
    lag_arr = np.arange(m_full) - n + 1
    grad = np.zeros(n)
    for i in range(n):
        i_pl = i + lag_arr
        i_ml = i - lag_arr
        vp = (i_pl >= 0) & (i_pl < n)
        vm = (i_ml >= 0) & (i_ml < n)
        grad[i] = np.dot(w_dx[vp], (1.0 - h[i_pl[vp]])) - np.dot(w_dx[vm], h[i_ml[vm]])
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_run(h0, T, lam, maxiter):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x):
        return softmax_grad(x, T, lam)[0]
    def grd(x):
        return softmax_grad(x, T, lam)[1]
    try:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s > 1e-13:
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, t_end, max_sweeps=1000):
    n = len(h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    best_c5 = float(np.max(corr) * dx)
    lag_arr = np.arange(m_full) - n + 1
    nm1 = n - 1
    TERN_ITERS = 25
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = best_h[i]
            i_pl = i + lag_arr
            i_ml = i - lag_arr
            vp = (i_pl >= 0) & (i_pl < n)
            vm = (i_ml >= 0) & (i_ml < n)
            lin = np.zeros(m_full)
            lin[vp] += 1.0 - best_h[i_pl[vp]]
            lin[vm] -= best_h[i_ml[vm]]
            
            # Ternary search on [0, 1]
            lo, hi = 0.0, 1.0
            for _ in range(TERN_ITERS):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                d1 = m1 - old
                f1 = (corr + d1 * lin).max()
                t1 = corr[nm1] + d1 * lin[nm1] - d1 * d1
                if t1 > f1: f1 = t1
                d2 = m2 - old
                f2 = (corr + d2 * lin).max()
                t2 = corr[nm1] + d2 * lin[nm1] - d2 * d2
                if t2 > f2: f2 = t2
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            new_val = (lo + hi) * 0.5
            if abs(new_val - old) < 1e-12:
                continue
            d = new_val - old
            corr += d * lin
            corr[nm1] -= d * d
            best_h[i] = new_val
            nc = float(np.max(corr) * dx)
            if nc < best_c5 - 1e-14:
                best_c5 = nc
                improved = True
        if not improved and sweep > 5:
            break
    
    corr_f = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr_f) * dx)
    return best_h, best_c5

def softmax_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    s = T * (corr - mc)
    s -= s.max()
    ex = np.exp(s)
    return dx * (mc + np.log(ex.sum()) / T) + lam * (h.sum() - n/2.0)**2

def softmax_jac(h, T, lam):
    n = len(h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    s = T * (corr - mc)
    s -= s.max()
    ex = np.exp(s)
    w = (ex / ex.sum()) * dx
    lag_arr = np.arange(m_full) - n + 1
    grad = np.zeros(n)
    for i in range(n):
        i_pl = i + lag_arr
        i_ml = i - lag_arr
        vp = (i_pl >= 0) & (i_pl < n)
        vm = (i_ml >= 0) & (i_ml < n)
        grad[i] = np.dot(w[vp], (1.0 - h[i_pl[vp]])) - np.dot(w[vm], h[i_ml[vm]])
    grad += 2.0 * lam * (h.sum() - n/2.0)
    return grad

                       method='L-BFGS-B', bounds=[(0.0,1.0)]*n)
        return project_h(res.x)
    except Exception:
        return x0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s > 1e-13:
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, t_end, max_sweeps=5000):
    n = len(h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    best_c5 = float(np.max(corr) * dx)
    lag_arr = np.arange(m_full) - n + 1
    nm1 = n - 1
    TI = 28

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = best_h[i]
            i_pl = i + lag_arr
            i_ml = i - lag_arr
            vp = (i_pl >= 0) & (i_pl < n)
            vm = (i_ml >= 0) & (i_ml < n)
            lin = np.zeros(m_full)
            lin[vp] += 1.0 - best_h[i_pl[vp]]
            lin[vm] -= best_h[i_ml[vm]]
            lo, hi = 0.0, 1.0
            for _ in range(TI):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                d1 = m1 - old
                f1 = (corr + d1 * lin).max()
                t1 = corr[nm1] + d1 * lin[nm1] - d1 * d1
                if t1 > f1:
                    f1 = t1
                d2 = m2 - old
                f2 = (corr + d2 * lin).max()
                t2 = corr[nm1] + d2 * lin[nm1] - d2 * d2
                if t2 > f2:
                    f2 = t2
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            new_val = (lo + hi) * 0.5
            if abs(new_val - old) < 1e-12:
                continue
            d = new_val - old
            corr += d * lin
            corr[nm1] -= d * d
            best_h[i] = new_val
            nc = float(np.max(corr) * dx)
            if nc < best_c5 - 1e-14:
                best_c5 = nc
                improved = True
        if (sweep + 1) % 20 == 0:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        if not improved and sweep > 5:
            break
    corr_f = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr_f) * dx)
    return best_h, best_c5

def _sm_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    s = T * (corr - mc)
    s -= s.max()
    ex = np.exp(s)
    return dx * (mc + np.log(ex.sum()) / T) + lam * (h.sum() - n / 2.0) ** 2

def _sm_jac(h, T, lam):
    n = len(h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    s = T * (corr - mc)
    s -= s.max()
    ex = np.exp(s)
    w = (ex / ex.sum()) * dx
    lag_arr = np.arange(m_full) - n + 1
    grad = np.zeros(n)
    for i in range(n):
        i_pl = i + lag_arr
        i_ml = i - lag_arr
        vp = (i_pl >= 0) & (i_pl < n)
        vm = (i_ml >= 0) & (i_ml < n)
        grad[i] = np.dot(w[vp], (1.0 - h[i_pl[vp]])) - np.dot(w[vm], h[i_ml[vm]])
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return grad

def lbfgs_wrap(h0, T, lam, maxiter):
    n = len(h0)
    x0 = project_h(h0)
    try:
        res = minimize(_sm_obj, x0, args=(T, lam), jac=_sm_jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_h(res.x)
    except Exception:
        return x0

def gen_candidate(n, rng):
    t = np.arange(n)
    r = rng.random()
    if r < 0.25:
        h = (t < n * 0.5).astype(float)
    elif r < 0.4:
        h = (t % 2 == 0).astype(float)
    elif r < 0.6:
        h = 0.5 + 0.45 * np.sin(2 * np.pi * rng.randint(1, 8) * t / n)
    elif r < 0.8:
        h = np.clip(0.5 + 0.3 * np.sin(2*np.pi*rng.uniform(1,10)*t/n) + 0.15*np.random.randn(n), 0, 1)
    else:
        h = np.random.beta(3, 3, n)
    return project_h(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive coordinate descent (30% budget)
    p1_end = t0 + budget_s * 0.30
    best_h, best_c5 = fast_coord_descent(best_h, p1_end, 10000)

    # Phase 2: Soft-max L-BFGS (15% budget)
    p2_end = t0 + budget_s * 0.45
    for T in [50, 200, 800, 3000, 10000]:
        if time.time() > p2_end:
            break
        hc = lbfgs_wrap(best_h, T, 500.0, 5000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Coordinate descent polish (15% budget)
    p3_end = t0 + budget_s * 0.60
    best_h, best_c5 = fast_coord_descent(best_h, p3_end, 10000)

    # Phase 4: Multi-restart with different inits (20% budget)
    p4_end = t0 + budget_s * 0.80
    work_h = best_h.copy()
    work_c5 = best_c5
    for k in range(20):
        if time.time() > p4_end:
            break
        scale = 0.02 + 0.04 * (1 - k / 20.0)
        h_p = project_h(work_h + scale * np.random.randn(n))
        hc = lbfgs_wrap(h_p, 5000 * (1.0 - k/30.0) + 100, 500.0, 2000)
        hc, c5v = fast_coord_descent(hc, min(time.time() + 15, p4_end), 3000)
        if c5v < work_c5 - 1e-6:
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 5: Try different n (12% budget)
    p5_end = t0 + budget_s * 0.92
    x_orig = np.linspace(0, 2, n, endpoint=False)
    for n_try in [128, 192, 256, 320]:
        if time.time() > p5_end:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_orig, best_h)
        h_try = project_h(h_try)
        t_sub = min(time.time() + budget_s * 0.03, p5_end)
        h_try, c5_try = fast_coord_descent(h_try, t_sub, 5000)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            n = n_try
            x_orig = x_new

    # Phase 6: Final coordinate descent polish (8% budget)
    p6_end = t0 + budget_s * 0.995
    best_h, best_c5 = fast_coord_descent(best_h, p6_end, 10000)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
