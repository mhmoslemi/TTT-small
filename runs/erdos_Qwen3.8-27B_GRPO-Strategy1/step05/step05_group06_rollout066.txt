from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n/2.0)**2
    # Gradient
    grad = np.zeros(n)
    m_full = 2*n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n/2.0)
    return val, grad

def lbfgs(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x): return softmax_obj_grad(x, T, lam)[0]
    def gr(x): return softmax_obj_grad(x, T, lam)[1]

                       bounds=[(0.0,1.0)]*n,
                       options={'maxiter':maxiter, 'ftol':ftol, 'gtol':1e-12})
        return project_h(res.x)
    except:
        return x0

            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg: lin[idx] += (1.0 - best_h[i-lg])
                    if i+lg < n: lin[idx] += -best_h[i+lg]
                else:
                    p = -lg
                    if i+p < n: lin[idx] += (1.0 - best_h[i+p])
                    if i >= p: lin[idx] += -best_h[i-p]
            def f1d(x):
                return np.max(cb + x*lin + x*x*quad)
            res = minimize_scalar(f1d, bounds=(0.0,1.0), method='bounded', options={'xatol':1e-12})
            if abs(res.x - best_h[i]) < 1e-14: continue
            ht = best_h.copy()
            ht[i] = res.x
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved: break
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-15:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h += room * (d / tot)
    else:
        if s < 1e-15:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def soft_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    a = T * (corr - mc)
    a -= a.max()
    ea = np.exp(a)
    sm = mc + np.log(ea.sum()) / T
    return dx * sm + lam * (h.sum() - n / 2.0) ** 2

def soft_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)

    a -= a.max()
    ea = np.exp(a)

    g = np.zeros(n)

                g[:n - lg] += -wk * h[lg:]

        v = corr + d * lin
        v[idx0] -= d * d
        vals[j] = np.max(v)
    # Find min region
    imin = np.argmin(vals)
    lo = max(0, xs[max(0, imin-1)])
    hi = min(1, xs[min(len(xs)-1, imin+1)])
    # Ternary search
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        d1 = m1 - old
        v1 = corr + d1 * lin
        v1[idx0] -= d1 * d1
        f1 = np.max(v1)
        d2 = m2 - old
        v2 = corr + d2 * lin
        v2[idx0] -= d2 * d2
        f2 = np.max(v2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2

The actual C₅ is about 0.38, so the bias should be much smaller than this. T=100000 gives bias ~0.00006, which is comparable to the gap we're trying to close (0.00005). So T=100000+ is needed for the final polish.

But with T=100000, the softmax weights are very peaked (only the max lag has significant weight). In this regime, L-BFGS is essentially doing gradient descent on the active lag(s), which might not be as effective.

For the final polish, coordinate descent is better because it handles the max exactly (no smoothing needed).

def compute_c5(h):
    n = len(h)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-15:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h += room * (d / tot)
    else:
        if s < 1e-15:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def soft_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    a = T * (corr - mc)
    a -= a.max()
    ea = np.exp(a)
    sm = mc + np.log(ea.sum()) / T
    return dx * sm + lam * (h.sum() - n / 2.0) ** 2

def soft_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n

    g = np.zeros(n)
