import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, n, T, k=8):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(corr)[-k:]
    top = corr[idx]
    m = top.max()
    s = np.clip(T * (top - m), -50, 50)
    return (m + np.log(np.sum(np.exp(s))) / T) * (2.0 / n)

def smooth_max_grad(x, n, T, k=8):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(corr)[-k:]
    top = corr[idx]
    m = top.max()
    s = np.clip(T * (top - m), -50, 50)
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for wi, ki in zip(w, idx):
        lag = ki - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n-lag] += wi * (1.0 - x[lag:])
                grad[lag:] += wi * (-x[:n-lag])
        else:
            L = -lag
            if n - L > 0:
                grad[L:] += wi * (1.0 - x[:n-L])
                grad[:n-L] += wi * (-x[L:])
    return grad * dx

def lbfgs_refine(h, n, T, k, maxiter):
    try:
        res = minimize(smooth_max, h, jac=lambda x: smooth_max_grad(x, n, T, k),
                       method='L-BFGS-B', bounds=[(0,1)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h

def coord_opt(h, n, i):
    # Exact 1D optimization for h[i]
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save
    # Coefficient of h[i] in each correlation
    coefs = np.zeros(2*n-1)
    lag = np.arange(-(n-1), n)
    # h[i] appears as first factor when: i + t = some index → t = ... 
    # In corr[ki], lag = ki - (n-1)
    # h[i] appears as x[i] (first factor) when the shift places i in the first array
    # corr[ki] = sum_t x[t+lag]*v[t] where v[t] = 1-x[t]
    # x[i] appears as x[t+lag] when t = i-lag, need 0<=i-lag<n and 0<=t<n
    # x[i] appears as x[t] (via 1-x[t]) when t = i, need 0<=i<n and 0<=i+lag<n
    for ki in range(2*n-1):
        L = ki - (n-1)
        c = 0.0
        # x[i] as first factor: t = i-L, need 0<=t<n and 0<=t+L<n → 0<=i-L<n and 0<=i<n
        if 0 <= i-L < n and 0 <= i < n:
            c += (1.0 - h[i-L])  # v[t] = 1-x[t] = 1-x[i-L]
        # x[i] appears in (1-x[t]) → -x[i]: t = i, need 0<=t<n and 0<=t+L<n
        if 0 <= i < n and 0 <= i+L < n:
            c += (-h[i+L])
        coefs[ki] = c
    # c5 = max(corr_base + h[i] * coefs)
    # We want to minimize this over h[i] in [0,1]
    # This is a convex piecewise-linear function, minimized at a breakpoint
    # Brute force: evaluate at critical points
    res = minimize_scalar(lambda hi: np.max(corr_base + hi * coefs),
                         bounds=(0.0, 1.0), method='bounded',
                         options={'xatol': 1e-12, 'maxiter': 100})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            hi = coord_opt(best, n, i)
            if abs(hi - best[i]) > 1e-10:
                trial = best.copy()
                trial[i] = hi
                trial = project_h(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_c5 - 1e-15:
                    best = trial
                    best_c5 = c5t
                    improved = True
        if not improved:
            break
    return best, best_c5

def pair_swap(h, n, i, j):
    """Try swapping h[i] and h[j], or adjusting both."""
    h1, h2 = h[i], h[j]
    candidates = [(h2, h1),  # swap

                  (h1 - 0.01, h2 + 0.01),
                  (min(1, h1+0.02), max(0, h2-0.02)),
                  (max(0, h1-0.02), min(1, h2+0.02))]
    return candidates
