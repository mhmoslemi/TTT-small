def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
        h = np.clip(h, 0.0, 1.0)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = np.full(n, 0.5)
        else:
            h = h + rm * ((tgt - s) / t)
            h = np.clip(h, 0.0, 1.0)
    # Fix residual
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        amt = d / n
        h = h + amt
        h = np.clip(h, 0.0, 1.0)

        if abs(d) > 1e-10:

                h = np.clip(h, 0.0, 1.0)
    return h

        t[idx] = 0

So total ΔR(k) for the swap:
= -(1-h[i+k-n+1]) - h[i-k+n-1] + (1-h[j+k-n+1]) + h[j-k+n-1]

(where indices are taken mod n... no, they're bounded. This gets complicated with the linear (non-circular) correlation.)

    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
    else:
        rm = 1.0 - h
        rm_sum = rm.sum()
        if rm_sum < 1e-14:
            h[:] = 0.5
        else:
            h += rm * ((tgt - s) / rm_sum)
            h = np.clip(h, 0.0, 1.0)

    if abs(d) > 1e-10:

    cur_c5 = float(np.max(corr) * dx)
    best_c5 = cur_c5
    best_h = h.copy()
    T = T0
    it = 0

        it += 1
        # Pick swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]

        # Actually since h[i] was 1 and h[j] was 0:
        # h[i] becomes 0, h[j] becomes 1
        nc = np.correlate(h, 1.0 - h, mode='full')
        nc5 = float(np.max(nc) * dx)
        
        if nc5 <= cur_c5 or np.random.rand() < np.exp(-(nc5 - cur_c5) / T):
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]  # undo
        
        T = max(T1, T * cool)
    
    return best_h, best_c5

    """Coordinate descent minimizing max correlation."""
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full')

            coef = np.zeros(mf)
            for k in range(mf):

                if 0 <= a < n:

                b = i - k + n - 1
                if 0 <= b < n:
                    coef[k] -= best[b]

            # Minimize max(base + t*coef) over t in [0,1]
            obj = lambda t: float(np.max(base + t * coef))
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})

                    best_c5 = c5t
                    corr = np.correlate(best, 1.0 - best, mode='full')
                    improved = True
        if not improved:

import numpy as np
from scipy.optimize import minimize_scalar
import time

_CD_BASE = [None]
_CD_COEF = [None]

def _cd_obj(t):
    return float(np.max(_CD_BASE[0] + t * _CD_COEF[0]))

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
    else:
        rm = 1.0 - h
        rm_s = rm.sum()
        if rm_s < 1e-14:
            h[:] = 0.5
        else:
            h += rm * ((tgt - s) / rm_s)
            h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if mask.sum() > 0:
            h[mask] += d / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def balance_binary(h):
    n = len(h)
    cnt = int(h.sum())
    if cnt > n // 2:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx, cnt - n // 2, replace=False)] = 0
    elif cnt < n // 2:
        idx = np.where(h == 0)[0]
        h[np.random.choice(idx, n // 2 - cnt, replace=False)] = 1
    return h

def random_binary(n):
    h = np.zeros(n)
    h[np.random.choice(n, n // 2, replace=False)] = 1.0
    return h

def structured_start(n, mode):
    x = np.arange(n, dtype=float)
    if mode == 0:
        sig = np.sin(2 * np.pi * 3 * x / n) + 0.4 * np.sin(2 * np.pi * 7 * x / n)
    elif mode == 1:
        sig = np.sin(2 * np.pi * 5 * x / n) + 0.3 * np.cos(2 * np.pi * 11 * x / n)
    elif mode == 2:
        sig = np.sin(2 * np.pi * 2 * x / n) * np.sin(2 * np.pi * 5 * x / n)
    else:
        sig = np.sin(2 * np.pi * 8 * x / n) + 0.5 * np.sin(2 * np.pi * 3 * x / n + 0.7)
    sb = (sig > np.median(sig)).astype(float)
    balance_binary(sb)
    return sb

def sa_binary(h_init, n, t_end, T0=0.05, T_min=0.0003, cool=0.99999, patience=100000):
    dx = 2.0 / n
    h = h_init.copy()
    cur_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    best_c5 = cur_c5
    best_h = h.copy()
    T = T0
    no_imp = 0
    while time.time() < t_end:
        r = np.random.rand()
        if r < 0.80:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            nc = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            if nc <= cur_c5 or np.random.rand() < np.exp(-(nc - cur_c5) / T):
                cur_c5 = nc
                no_imp = 0
                if nc < best_c5 - 1e-15:
                    best_c5 = nc
                    best_h = h.copy()
                    T = max(T_min, T * 0.998)
            else:
                h[i], h[j] = 1.0, 0.0
                no_imp += 1
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            a, b = np.random.choice(ones, 2, replace=False)
            c, d = np.random.choice(zeros, 2, replace=False)
            h[a], h[b], h[c], h[d] = 0.0, 0.0, 1.0, 1.0
            nc = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            if nc <= cur_c5 or np.random.rand() < np.exp(-(nc - cur_c5) / T):
                cur_c5 = nc
                no_imp = 0
                if nc < best_c5 - 1e-15:
                    best_c5 = nc
                    best_h = h.copy()
                    T = max(T_min, T * 0.998)
            else:
                h[a], h[b], h[c], h[d] = 1.0, 1.0, 0.0, 0.0
                no_imp += 1
        if no_imp > patience:
            h = best_h.copy()
            cur_c5 = best_c5
            T = T0 * 0.6
            no_imp = 0
        T = max(T_min, T * cool)
    return best_h, best_c5

def ils_search(h_init, n, t_end, n_outer=30, perturb_size=(4, 12)):
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    cur = best_h.copy()
    for _ in range(n_outer):
        if time.time() > t_end:
            break
        # Perturb
        ks = np.random.randint(perturb_size[0], perturb_size[1] + 1)
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        flips = np.random.choice(ones, min(ks, len(ones)), replace=False)
        fills = np.random.choice(zeros, min(ks, len(zeros)), replace=False)
        cur[flips] = 0.0
        cur[fills[:len(flips)]] = 1.0
        # Short SA (low T = near greedy)
        t_sub = min(t_end, time.time() + 8.0)
        cur, cur_c5 = sa_binary(cur, n, t_sub, T0=0.01, T_min=0.0001, cool=0.99999, patience=30000)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur.copy()
    return best_h, best_c5

def coord_desc(h, n, t_end, max_sweeps=80):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                a = i + k - (n - 1)
                if 0 <= a < n:
                    coef[k] += (1.0 - best[a])
                b = i - k + (n - 1)
                if 0 <= b < n:
                    coef[k] -= best[b]
            base = corr - old * coef
            _CD_BASE[0] = base
            _CD_COEF[0] = coef
            res = minimize_scalar(_cd_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 40})
            nv = max(0.0, min(1.0, float(res.x)))
            if abs(nv - old) > 1e-10:
                trial = best.copy()
                trial[i] = nv
                trial = proj_h(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_c5 - 1e-13:
                    best = trial
                    best_c5 = c5t
                    corr = np.correlate(best, 1.0 - best, mode='full')
                    improved = True
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = random_binary(160)
    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary SA from initial (20%)
    t1 = t0 + budget_s * 0.20
    h_bin = (best_h > 0.5).astype(float)
    balance_binary(h_bin)
    best_h, best_c5 = sa_binary(h_bin, n, t1, T0=0.04, T_min=0.0003, cool=0.99999, patience=120000)

    # Phase 2: Continuous refinement (5%)
    t2 = t0 + budget_s * 0.25
    best_h, best_c5 = coord_desc(best_h, n, t2, max_sweeps=40)

    # Phase 3: ILS with diverse perturbations (30%)
    t3 = t0 + budget_s * 0.55
    h_bin = (best_h > 0.5).astype(float)
    balance_binary(h_bin)
    best_h, best_c5 = ils_search(h_bin, n, t3, n_outer=40, perturb_size=(4, 14))
    best_h, best_c5 = coord_desc(best_h, n, min(t3, time.time() + 10), max_sweeps=20)

    # Phase 4: Multi-start SA (20%)
    t4 = t0 + budget_s * 0.75
    starts = [random_binary(n), random_binary(n), random_binary(n)]
    for m in range(4):
        starts.append(structured_start(n, m))
    starts.append((best_h > 0.5).astype(float))
    time_per = max(3.0, (t4 - time.time()) / len(starts))
    for st in starts:
        if time.time() > t4:
            break
        hs, cs = sa_binary(st, n, time.time() + min(time_per * 1.2, t4 - time.time()),
                           T0=0.03, T_min=0.0003, cool=0.99999, patience=80000)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs
            best_h, best_c5 = coord_desc(best_h, n, min(t4, time.time() + 8), max_sweeps=15)

    # Phase 5: Extended SA from best (10%)
    t5 = t0 + budget_s * 0.85
    h_bin = (best_h > 0.5).astype(float)
    balance_binary(h_bin)
    best_h, best_c5 = sa_binary(h_bin, n, t5, T0=0.015, T_min=0.0001, cool=0.999995, patience=200000)

    # Phase 6: Continuous refinement (8%)
    t6 = t0 + budget_s * 0.93
    best_h, best_c5 = coord_desc(best_h, n, t6, max_sweeps=60)

    # Phase 7: Multi-resolution
    for n_hi in [200, 240]:
        if time.time() - t0 > budget_s * 0.96:
            break
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n_hi, endpoint=False) + 1.0 / n_hi
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = proj_h(h_hi, n_hi)
        h_hi, c5_hi = coord_desc(h_hi, n_hi, min(t0 + budget_s * 0.97, time.time() + 20), max_sweeps=20)
        if c5_hi < best_c5:
            best_h, best_c5, n = h_hi, c5_hi, n_hi

    # Phase 8: Final polish
    if time.time() - t0 < budget_s * 0.99:
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.995, max_sweeps=30)

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
