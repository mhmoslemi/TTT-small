def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def vec_minimax(base, coefs, n_grid=80):
    """Minimize max(base + t*coefs) over t in [0,1].
    base and coefs are 1D arrays of same length.
    Handles the case where one entry might be quadratic (k=0)."""
    # Grid search
    ts = np.linspace(0, 1, n_grid)
    vals = np.max(base[None, :] + ts[:, None] * coefs[None, :], axis=1)
    i_best = np.argmin(vals)
    # Refine with golden section in neighborhood
    lo = max(0.0, ts[max(0, i_best-2)])
    hi = min(1.0, ts[min(n_grid-1, i_best+2)])
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(base + c * coefs)
    fd = np.max(base + d * coefs)
    for _ in range(50):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = np.max(base + c * coefs)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = np.max(base + d * coefs)
    return (a + b) / 2

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def solve_1d(base, coefs, has_quad):
    """Minimize max over t in [0,1]. If has_quad, entry 0 is base[0]+t-t^2."""
    n_g = 50
    ts = np.linspace(0.0, 1.0, n_g)
    M = base[None, :] + ts[:, None] * coefs[None, :]
    if has_quad:
        M[:, 0] = base[0] + ts - ts * ts
    vals = M.max(axis=1)
    ib = vals.argmin()
    lo = ts[max(0, ib - 1)]
    hi = ts[min(n_g - 1, ib + 1)]
    if hi - lo < 1e-10:
        return (lo + hi) / 2
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    
    def fval(t):
        v = base + t * coefs
        if has_quad:
            v[0] = base[0] + t - t * t
        return v.max()
    
    fc, fd = fval(c), fval(d)
    for _ in range(35):
        if fc < fd:
            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo); fc = fval(c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo); fd = fval(d)
    return (lo + hi) * 0.5

    """Fast coordinate descent with incremental correlation updates."""
    bh = h.copy()
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    dx = 2.0 / n
    bc5 = float(corr.max()) * dx
    off = n - 1
    
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        top_i = np.argsort(corr)[-n_top:]
        shifts = top_i - off
        # Check if shift 0 is in top
        has_0 = (shifts == 0)
        zero_pos = np.where(has_0)[0]
        
        imp = False
        order = np.random.permutation(n)
        for i in order:
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - bh[jp])
                if 0 <= jm < n:
                    c -= bh[jm]
                coefs[t] = c
            base = corr[top_i] - bh[i] * coefs
            # Handle quad term for k=0
            hq = False
            if len(zero_pos) > 0:
                hq = True
            t_new = solve_1d(base, coefs, hq)
            d = t_new - bh[i]
            if abs(d) < 1e-13:
                continue
            bh[i] = t_new
            # Incremental corr update
            m_lo = max(-(n-1), i - n + 1, -i)
            m_hi = min(n-1, i, n-1-i)
            if m_lo <= m_hi:
                ms = np.arange(m_lo, m_hi + 1)
                jms = i - ms
                jps = i + ms
                v1 = (jms >= 0) & (jms < n)
                v2 = (jps >= 0) & (jps < n)
                dd = np.zeros(len(ms))
                if v1.any():
                    dd[v1] += (1.0 - bh[jms[v1]])
                if v2.any():
                    dd[v2] -= bh[jps[v2]]
                corr[ms + off] += d * dd
            c5n = float(corr.max()) * dx
            if c5n < bc5:
                bc5 = c5n
                imp = True
        if not imp:
            break
    return bh, bc5

def slsqp_smooth(h, T, n, maxiter=600):
    """SLSQP with log-sum-exp smooth max."""
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return m + np.log(np.sum(np.exp(T * (c - m)))) / T
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        e = np.exp(T * (c - m))
        w = e / e.sum()
        g = np.zeros(2*n - 1)
        g[:] = w * T * dx  # placeholder, actually need to backprop
        # This is complex, skip gradient and let SLSQP use numerical
        return None
    try:

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
