                return h, best_c5

                grad[s:] += w * (1.0 - h_proj[:n-s])

                grad[a_s:] -= w * h_proj[:n-a_s]

        constraint = h_safe.sum() - n/2

        if it % 50 == 0:

    best_h = project_h(best_h, n)

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def f1d_wrapper(v, corr, lin, old, n, dx):
    d = v - old
    val = corr + d * lin
    val[n - 1] -= d * d
    return float(np.max(val) * dx)

def cd_pass(h, n, t_end, max_sweeps, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                h = project_h(h, n)
                return h, compute_c5(h)
            old = h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = n - 1 + s
                mn = n - 1 - s
                if i - s >= 0:
                    lin[mp] += (1.0 - h[i - s])
                    lin[mn] -= h[i - s]
                if i + s < n:
                    lin[mp] -= h[i + s]
                    lin[mn] += (1.0 - h[i + s])
            lin[n - 1] = 1.0 - 2.0 * old
            res = minimize_scalar(f1d_wrapper, bounds=(0.0, 1.0),
                                  args=(corr, lin, old, n, dx),
                                  method='bounded', options={'xatol': 1e-12})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-14:
                continue
            h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    h = project_h(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def make_perturbation(rng, n, mode, step, cur_h):
    if mode == 0:
        return step * rng.randn(n)
    elif mode == 1:
        f = rng.randint(1, 8)
        ph = rng.uniform(0, 2 * np.pi)
        return step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif mode == 2:
        c = rng.randint(0, n)
        w = int(rng.choice([6, 12, 20, 35]))
        pert = np.zeros(n)
        mask = (np.arange(n).astype(float) - c) ** 2 < w * w
        cnt = int(mask.sum())
        pert[mask] = step * 1.8 * rng.randn(cnt)
        return pert
    elif mode == 3:
        pert = np.zeros(n)
        for _ in range(3):
            f = rng.randint(1, 14)
            ph = rng.uniform(0, 2 * np.pi)
            pert += (step / 3.0) * np.cos(2 * np.pi * f * np.arange(n) / n + ph)
        return pert
    elif mode == 4:
        k = int(rng.randint(3, 12))
        idx = rng.choice(n, k, replace=False)
        pert = np.zeros(n)
        pert[idx] = step * 2.5 * rng.randn(k)
        return pert
    else:
        # Pair swap preserving sum
        i1 = rng.randint(0, n)
        i2 = rng.randint(0, n)
        if i1 == i2:
            return step * rng.randn(n)
        delta = step * rng.randn()
        pert = np.zeros(n)
        pert[i1] = delta
        pert[i2] = -delta
        return pert
