    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    converged = False
    sweep = 0
    while not converged and sweep < max_sweeps and time.time() < t_end:
        improved = False
        if sweep % 4 == 2:
            order = rng.permutation(n)
        elif sweep % 4 == 3:
            order = np.arange(n - 1, -1, -1)
        else:
            order = np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = n - 1 + s
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                    lin[mn] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[mn] -= best_h[i - s]
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-12:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        sweep += 1
        if not improved:
            converged = True
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def pair_swap_search(h, n, t_end, max_passes=20):
    """Try swapping/adjusting pairs of coordinates."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    dx = 2.0 / n
    rng = np.random.RandomState(12345)
    for p in range(max_passes):
        if time.time() > t_end:
            break
        improved = False
        # Sample pairs
        for _ in range(n * 2):
            if time.time() > t_end:
                break
            i, j = rng.choice(n, 2, replace=False)
            if abs(best_h[i] - best_h[j]) < 1e-6:
                continue
            # Try: h[i] -> h[i]+t, h[j] -> h[j]-t (preserves sum)
            # Or: swap h[i] and h[j]
            # Let's try the transfer: t ∈ [-(min(h[i],1-h[j])), min(h[i], 1-h[j])]... 
            # Actually simpler: just try swapping
            if best_h[i] > best_h[j]:
                hi, hj = best_h[i], best_h[j]
                for t in [0.0, 0.25, 0.5, 0.75, 1.0]:
                    vi = np.clip(hi - t * (hi - hj), 0, 1)
                    vj = np.clip(hj + t * (hi - hj), 0, 1)
                    # Check sum constraint approximately
                    if abs(vi - hi + vj - hj) > 1e-10:
                        continue
                    old_i, old_j = best_h[i], best_h[j]
                    di, dj = vi - old_i, vj - old_j
                    # Update corr
                    li = np.zeros(2 * n - 1)
                    lj = np.zeros(2 * n - 1)
                    for s in range(1, n):
                        mp = n - 1 + s
                        mn = n - 1 - s
                        if s <= i:
                            li[mp] += (1.0 - best_h[i - s])
                        if i + s < n:
                            li[mp] -= best_h[i + s]
                            li[mn] += (1.0 - best_h[i + s])
                        if s <= i:
                            li[mn] -= best_h[i - s]
                        li[n - 1] = 1.0 - 2.0 * old_i
                        if s <= j:
                            lj[mp] += (1.0 - best_h[j - s])
                        if j + s < n:
                            lj[mp] -= best_h[j + s]
                            lj[mn] += (1.0 - best_h[j + s])
                        if s <= j:
                            lj[mn] -= best_h[j - s]
                        lj[n - 1] = 1.0 - 2.0 * old_j
                    trial = corr + di * li + dj * lj
                    trial[n-1] -= di*di + dj*dj
                    c5_t = float(np.max(trial) * dx)
                    if c5_t < best_c5:
                        best_h[i], best_h[j] = vi, vj
                        corr = trial
                        best_c5 = c5_t
                        improved = True
                        break
                # If didn't accept, restore
            # Also try pure swap
            si, sj = best_h[i], best_h[j]
            best_h[i], best_h[j] = sj, si
            corr_sw = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
            c5_sw = float(np.max(corr_sw) * dx)
            if c5_sw < best_c5:
                corr = corr_sw
                best_c5 = c5_sw
                improved = True
            else:
                best_h[i], best_h[j] = si, sj
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5
