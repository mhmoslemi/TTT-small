At the optimum, typically 2-5 lags achieve the max. Let's say lags s1, s2, s3 achieve the max. Then we're minimizing:

This can be reformulated as:

This is a QP (quadratic constraints) for fixed active set. We could solve this with scipy or cvxpy.

But the active set changes as we optimize, so we need an outer loop to update it. This is similar to the bisection approach:
1. Bisect on t
2. For each t, check if there exists h with c(sk) ≤ t for all active k
3. If yes, reduce t; if no, increase t

The feasibility check is:
exists h: 0≤h≤1, sum(h)=n/2, c(sk) ≤ t for all k in active set

Each c(sk) constraint is bilinear (quadratic) in h. This is a non-convex feasibility problem. CVX won't help directly.

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def cd_full(h, n, t_end, max_sweeps=3000, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c5 = float(np.max(corr) * dx)
    sweep = 0
    while sweep < max_sweeps and time.time() < t_end:
        improved = False
        order = rng.permutation(n) if sweep % 3 == 2 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return h, c5
            old = h[i]
            lin = np.zeros(2*n-1)
            for s in range(1, n):
                if s <= i:
                    lin[n-1+s] += (1.0 - h[i-s])
                    lin[n-1-s] -= h[i-s]
                if i+s < n:
                    lin[n-1+s] -= h[i+s]
                    lin[n-1-s] += (1.0 - h[i+s])
            lin[n-1] = 1.0 - 2.0 * old
            base = corr.copy()
            
            def f1d(v):
                d = v - old
                c = base + d * lin
                c[n-1] -= d * d
                return float(np.max(c) * dx)
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12, 'maxiter': 40})
            vn = float(res.x)
            d = vn - old
            if abs(d) < 1e-14:
                continue
            h[i] = vn
            corr = base + d * lin
            corr[n-1] -= d * d
            c5n = float(np.max(corr) * dx)
            if c5n < c5 - 1e-15:
                c5 = c5n
                improved = True
            else:
                h[i] = old
                corr = corr - d * lin
                corr[n-1] += d * d
        sweep += 1
        if not improved:
            break
    h = project_h(h, n)
    c5 = compute_c5(h)
    return h, c5

def subgrad_anneal(h, n, t_end, max_iters=2000, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mom = np.zeros(n)
    beta = 0.75
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = it / max(max_iters, 1)
        T = 30.0 * (20.0 ** frac)
        
        # Compute gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        exponents = np.clip(T * (corr - m), -50.0, 50.0)
        p = np.exp(exponents)
        p /= p.sum()
        
        grad = np.zeros(n)
        for k in range(2*n-1):
            w = p[k]
            if w < 1e-14:
                continue
            s = k - (n-1)
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            elif s > 0:
                g = np.zeros(n)
                g[:n-s] = (1.0 - h[s:])
                g[s:] += (-h[:n-s])
                grad += w * g
            else:
                a = -s
                g = np.zeros(n)
                g[a:] = (1.0 - h[:n-a])
                g[:n-a] += (-h[a:])
                grad += w * g
        
        grad *= dx
        mom = beta * mom + grad
        gm = np.max(np.abs(mom))
        if gm < 1e-14:
            break
        alpha = 0.5 / gm
        h_new = project_h(h - alpha * mom, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    arange_n = np.arange(n)
    
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.02 + 0.08 * cool
        mode = hop % 7
        
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 20)
            ph = rng.uniform(0, 2*np.pi)
            pert = step * np.sin(2*np.pi*f*arange_n/n + ph)
        elif mode == 2:
            # Pair swaps
            pert = np.zeros(n)
            for _ in range(max(5, n//15)):
                i, j = rng.randint(0, n, 2)
                if i != j:
                    amt = step * rng.uniform(0.5, 1.5)
                    pert[i] -= amt
                    pert[j] += amt
        elif mode == 3:
            # Block shift
            blen = rng.randint(5, max(6, n//8))
            start = rng.randint(0, n - blen)
            pert = np.zeros(n)
            pert[start:start+blen] = step * rng.uniform(-1, 1, blen)
            # Compensate
            comp = -pert.sum() / n
            pert += comp
        elif mode == 4:
            # Sparse random
            mask = rng.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = step * 1.5 * rng.randn(mask.sum())
        elif mode == 5:
            # Cosine perturbation
            f = rng.uniform(0.5, 5.0)
            ph = rng.uniform(0, 2*np.pi)
            pert = step * np.cos(np.pi*f*arange_n/n + ph)
        else:
            pert = step * 0.3 * rng.randn(n)
        
        h_pert = project_h(cur_h + pert, n)
        hop_end = min(t_end, time.time() + max(3.0, 10.0 * cool))
        h_opt, c5_opt = cd_full(h_pert, n, hop_end, max_sweeps=150, seed=seed+hop)
        
        temp = 0.005 * cool + 0.0005
        accept = c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt-cur_c5)/temp))
        if accept:
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new)

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def cd_full(h, n, t_end, max_sweeps, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c5 = float(np.max(corr) * dx)
    sweep = 0
    while sweep < max_sweeps and time.time() < t_end:
        improved = False
        order = rng.permutation(n) if sweep % 3 == 2 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return h, c5
            old = h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                if s <= i:
                    lin[n - 1 + s] += (1.0 - h[i - s])
                    lin[n - 1 - s] -= h[i - s]
                if i + s < n:
                    lin[n - 1 + s] -= h[i + s]
                    lin[n - 1 - s] += (1.0 - h[i + s])
            lin[n - 1] = 1.0 - 2.0 * old
            base = corr.copy()
            
            def f1d(v):
                d = v - old
                c = base + d * lin
                c[n - 1] -= d * d
                return float(np.max(c) * dx)
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12, 'maxiter': 50})
            vn = float(res.x)
            d = vn - old
            if abs(d) < 1e-14:
                continue
            h[i] = vn
            corr = base + d * lin
            corr[n - 1] -= d * d
            c5n = float(np.max(corr) * dx)
            if c5n < c5 - 1e-15:
                c5 = c5n
                improved = True
            else:
                h[i] = old
                corr = corr - d * lin
                corr[n - 1] += d * d
        sweep += 1
        if not improved:
            break
    h = project_h(h, n)
    c5 = compute_c5(h)
    return h, c5

def subgrad_anneal(h, n, t_end, max_iters, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mom = np.zeros(n)
    beta = 0.75
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = it / max(max_iters, 1)
        T = 30.0 * (25.0 ** frac)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        exponents = np.clip(T * (corr - m), -50.0, 50.0)
        p = np.exp(exponents)
        p /= p.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = p[k]
            if w < 1e-14:
                continue
            s = k - (n - 1)
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            elif s > 0:
                g = np.zeros(n)
                g[:n - s] = (1.0 - h[s:])
                g[s:] += (-h[:n - s])
                grad += w * g
            else:
                a = -s
                g = np.zeros(n)
                g[a:] = (1.0 - h[:n - a])
                g[:n - a] += (-h[a:])
                grad += w * g
        grad *= dx
        mom = beta * mom + grad
        gm = np.max(np.abs(mom))
        if gm < 1e-14:
            break
        alpha = 0.5 / gm
        h = project_h(h - alpha * mom, n)
        c5n = compute_c5(h)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h.copy()
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops, seed):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    ar = np.arange(n)
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - hop / max(n_hops, 1))
        step = 0.02 + 0.10 * cool
        mode = hop % 8
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 20)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * ar / n + ph)
        elif mode == 2:
            pert = np.zeros(n)
            for _ in range(max(5, n // 15)):
                i, j = rng.randint(0, n, 2)
                if i != j:
                    amt = step * rng.uniform(0.5, 1.5)
                    pert[i] -= amt
                    pert[j] += amt
        elif mode == 3:
            blen = rng.randint(5, max(6, n // 8))
            start = rng.randint(0, max(1, n - blen))
            pert = np.zeros(n)
            pert[start:start + blen] = step * rng.uniform(-1, 1, blen)
        elif mode == 4:
            mask = rng.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = step * 1.5 * rng.randn(mask.sum())
        elif mode == 5:
            f = rng.uniform(0.5, 5.0)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.cos(np.pi * f * ar / n + ph)
        elif mode == 6:
            # Big jump: replace 20% of values
            pert = np.zeros(n)
            mask = rng.random(n) < 0.2
            pert[mask] = rng.uniform(0, 1, mask.sum()) - h[mask]
        else:
            pert = step * 0.3 * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        hop_end = min(t_end, time.time() + max(3.0, 12.0 * cool))
        h_opt, c5_opt = cd_full(h_pert, n, hop_end, max_sweeps=200, seed=seed + hop)
        temp = 0.006 * cool + 0.0005
        accept = (c5_opt < cur_c5 or
                  (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)))
        if accept:
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=None)
    return project_h(h_new, n_new)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        best_h = np.zeros(160)

    # Phase 1: Initial CD (12%)
