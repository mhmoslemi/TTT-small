        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_topk(h, n_topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    return grad, topk_idx, corr

def adam_steps(h, n, t_end, lr=0.004, beta1=0.9, beta2=0.999, n_topk=10, T_start=300, T_end=80000, iters=250, var_pen_start=0.05):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    for it in range(iters):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(iters, 1)
        n_top = max(3, int(n_topk * cool + 3))
        var_pen = var_pen_start * cool
        grad, topk_idx, corr = grad_topk(best_h, n_top)
        if var_pen > 1e-6:
            top_c = corr[topk_idx]
            var_grad = 2.0 * (top_c - top_c.mean()) / n_top
            shifts_vk = topk_idx - (n - 1)
            for vi in range(n):
                dp = 0.0
                for k in shifts_vk:
                    jp = vi + k
                    jm = vi - k
                    if 0 <= jp < n:
                        dp += (1.0 - best_h[jp]) * var_grad[np.where(topk_idx == jp - vi)[0][0]] if False else 0
                # simplified: just use the topk gradient direction weighted
            # Skip detailed variance grad for speed; rely on topk
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad ** 2
        m_hat = m_v / (1 - beta1 ** (it + 1))
        v_hat = v_v / (1 - beta2 ** (it + 1))
        step = lr * (0.3 + 0.7 * cool) * m_hat / (np.sqrt(v_hat) + 1e-8)
        new_h = project_h(best_h - step)
        c5v = compute_c5(new_h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = new_h
    return best_h, best_c5

def slsqp_grad(h, T, n, maxiter=400, ftol=1e-15, n_top=0):
    def obj(x):
        dx_val = 2.0 / len(x)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if T > 1e8:
            return m * dx_val
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx_val
    def obj_grad(x):
        dx_val = 2.0 / len(x)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if T > 1e8:
            idx = np.argmax(corr)
            w = np.zeros_like(corr)
            w[idx] = 1.0
        else:
            e = np.exp(T * (corr - m))
            w = e / e.sum()
        if n_top > 0:
            sorted_idx = np.argsort(w)
            w[:len(w) - n_top] *= 0.01
            w /= w.sum()
        shifts = np.arange(2 * n - 1) - (n - 1)
        i_idx = np.arange(n)[:, None]
        s_idx = shifts[None, :]
        jp = i_idx + s_idx
        jm = i_idx - s_idx
        v1 = ((jp >= 0) & (jp < n)).astype(np.float64)
        v2 = ((jm >= 0) & (jm < n)).astype(np.float64)
        jpc = np.clip(jp, 0, n - 1)
        jmc = np.clip(jm, 0, n - 1)
        w_col = w[None, :]
        g = ((1.0 - x[jpc]) * v1 * w_col - x[jmc] * v2 * w_col).sum(axis=1) * dx_val
        return g
    con_jac = np.ones(n)
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: con_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=60, n_top=18):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            a = np.zeros(n_top)
            for ti in range(n_top):
                k = top_shifts[ti]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                a[ti] = c
            b = corr[top_idx] - best_h[i] * a
            candidates = [0.0, 1.0]
            for p in range(n_top):
                for q in range(p + 1, n_top):
                    denom = a[p] - a[q]
                    if abs(denom) > 1e-12:
                        tc = (b[q] - b[p]) / denom
                        if 0.0 < tc < 1.0:
                            candidates.append(tc)
            best_t = best_h[i]
            best_val = np.max(b + best_h[i] * a)
            for tc in candidates:
                val = np.max(b + tc * a)
                if val < best_val - 1e-14:
                    best_val = val
                    best_t = tc
            if abs(best_t - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = best_t
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, rng):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(1.0 - it / max(n_iter, 1), 0.05)
        r = rng.random()
        if r < 0.3:
            f = rng.randint(1, 10)
            p = rng.uniform(0, 2 * np.pi)
            pert = scale * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.55:
            mask = rng.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = scale * 1.8 * cool * rng.randn(max(int(mask.sum()), 1))
        elif r < 0.8:
            blen = rng.randint(max(n // 12, 2), max(n // 3, 3))
            bs = rng.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * cool * rng.randn(blen)
        else:
            pert = scale * cool * 0.6 * rng.randn(n)
        h_pert = project_h(work_h + pert)
        T = 3000 * cool + 1000
        hc = slsqp_grad(h_pert, T, n, 250, 1e-14, n_top=10)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.0002
        if c5v < work_c5 or (c5v > work_c5 and rng.random() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(160)
        init_h[:80] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Adam global optimization
    t_p1 = t0 + budget_s * 0.18
    best_h, best_c5 = adam_steps(best_h, n, t_p1, lr=0.004, n_topk=10, T_start=300, T_end=80000, iters=300)

    # Phase 2: SLSQP with analytic gradient (progressive T)
    t_p2 = t0 + budget_s * 0.38
    for T in [500, 2000, 10000, 50000, 200000]:
        if time.time() > t_p2:
            break
        hc = slsqp_grad(best_h, T, n, 500, 1e-15, n_top=0)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Basin hopping
    t_p3 = t0 + budget_s * 0.55
    bh_h, bh_c5 = basin_hop(best_h, n, t_p3, 45, 0.1, rng)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = bh_h

    # Phase 4: Refine with Adam again
    t_p4 = t0 + budget_s * 0.68
    best_h, best_c5 = adam_steps(best_h, n, t_p4, lr=0.002, n_topk=12, T_start=5000, T_end=200000, iters=200, var_pen_start=0.03)

    # Phase 5: Coordinate descent (exact 1D minimax)
    t_p5 = t0 + budget_s * 0.82
    best_h, best_c5 = coord_descent(best_h, n, t_p5, max_sweeps=100, n_top=20)

    # Phase 6: SLSQP final polish with very high T
    t_p6 = t0 + budget_s * 0.92
    if time.time() < t_p6:
        hc = slsqp_grad(best_h, 1000000, n, 800, 1e-16, n_top=5)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Final coordinate descent
    if time.time() < t_end - 5:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 2, max_sweeps=50, n_top=25)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
