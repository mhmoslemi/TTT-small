def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute to non-bound elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_lin(i, h, n):
    c = n - 1
    val = np.zeros(n - 1)
    if i > 0:
        m = min(i, n - 1)
        val[:m] = 1.0 - h[i-1::-1][:m]
    if i + 1 < n:
        nr = n - 1 - i
        val[:nr] += -h[i+1:i+1+nr]
    lin = np.zeros(2 * n - 1)
    lin[c] = 1.0 - 2.0 * h[i]
    lin[c+1:] = val
    lin[:c] = val[::-1]
    return lin

def obj_1d(v):
    d = v - _state['old']
    c = _state['cb'] + d * _state['lin']
    c[_state['ctr']] -= d * d
    return float(np.max(c) * _state['dx'])

def coord_descent(h, n, t_end, max_sweeps, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if sweep % 4 == 3:
            order = rng.permutation(n)
        else:
            order = range(n)
        for i in order:
            old = best_h[i]
            if old < 1e-15 or old > 1.0 - 1e-15:
                lin = compute_lin(i, best_h, n)
                _state['old'] = old
                _state['lin'] = lin
                _state['cb'] = corr
                _state['ctr'] = n - 1
                _state['dx'] = dx
                res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-11, 'maxiter': 30})
                v_new = float(res.x)
                d = v_new - old
                if abs(d) < 1e-13:
                    continue
                corr_new = corr + d * lin
                corr_new[n-1] -= d * d
                c5_new = float(np.max(corr_new) * dx)
                if c5_new <= best_c5 + 1e-15:
                    best_h[i] = v_new
                    corr = corr_new
                    best_c5 = min(best_c5, c5_new)
                    improved = True
                continue
            lin = compute_lin(i, best_h, n)
            _state['old'] = old
            _state['lin'] = lin
            _state['cb'] = corr
            _state['ctr'] = n - 1
            _state['dx'] = dx
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11, 'maxiter': 30})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            corr_new = corr + d * lin
            corr_new[n-1] -= d * d
            c5_new = float(np.max(corr_new) * dx)
            if c5_new <= best_c5 + 1e-15:
                best_h[i] = v_new
                corr = corr_new
                best_c5 = min(best_c5, c5_new)
                improved = True
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def perturb_h(h, n, rng, mode, scale):
    if mode == 0:
        return h + scale * rng.randn(n)
    if mode == 1:
        f = rng.randint(1, 25)
        ph = rng.uniform(0, 2 * np.pi)
        return h + 2 * scale * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    if mode == 2:
        mask = rng.random(n) < 0.2
        p = np.zeros(n)
        p[mask] = 2 * scale * rng.randn(mask.sum())
        return h + p
    if mode == 3:
        g = h.copy()
        for _ in range(3):
            a = rng.randint(0, max(1, n - 30))
            b = rng.randint(a + 10, n)
            g[a:b] = h[a:b][::-1]
        return g
    f1 = rng.randint(1, 10)
    f2 = rng.randint(11, 30)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    ar = np.arange(n) / n
    return h + scale * (np.sin(2*np.pi*f1*ar + p1) + 0.5*np.sin(2*np.pi*f2*ar + p2))

def make_start(n, mode, seed):
    rng = np.random.RandomState(seed)
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 2:
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.35 * np.cos(np.pi * x)
    elif mode == 3:
        h = rng.random(n)
    elif mode == 4:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    else:
        h = 0.5 + 0.25 * np.sin(2 * np.pi * 5 * np.arange(n) / n) + 0.15 * np.cos(2 * np.pi * 11 * np.arange(n) / n)
    return project_h(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    init_h = globals().get('initial_h_values')
    if init_h is not None:
        best_h = np.array(init_h, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate descent (10%)
    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = coord_descent(best_h, n, t1, 10000, seed)

    # Phase 2: Multi-start exploration (55%)
    rng = np.random.RandomState(seed + 50)
    t2 = t0 + budget_s * 0.65
    n_starts = 40
    for r in range(n_starts):
        if time.time() > t2:
            break
        frac = r / n_starts
        if r < 8:
            # Perturb current best with various scales
            scale = 0.015 + 0.04 * (1 - frac)
            mode = r % 5
            h_start = project_h(perturb_h(best_h, n, rng, mode, scale), n)
        elif r < 16:
            # More aggressive perturbation
            scale = 0.05 + 0.05 * rng.random()
            mode = r % 5
            h_start = project_h(perturb_h(best_h, n, rng, mode, scale), n)
        elif r < 28:
            # Alternative starting constructions
            mode2 = (r - 16) % 6
            h_start = make_start(n, mode2, seed + 1000 + r)
            # Blend with current best
            blend = 0.3 + 0.4 * rng.random()
            h_start = project_h(blend * best_h + (1 - blend) * h_start, n)
        else:
            # Random diverse starts with small perturbation
            h_start = make_start(n, rng.randint(6), seed + 2000 + r)
            h_start = project_h(0.5 * h_start + 0.5 * best_h + 0.02 * rng.randn(n), n)

        sub_budget = min(t2, time.time() + max(3.0, budget_s * 0.012 * (2.0 - frac)))
        h_opt, c5_opt = coord_descent(h_start, n, sub_budget, 5000, seed + 500 + r)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Heavy refinement on best (20%)
    t3 = t0 + budget_s * 0.85
    best_h, best_c5 = coord_descent(best_h, n, t3, 20000, seed + 1000)

    # Phase 4: Targeted micro-restarts (10%)
    rng2 = np.random.RandomState(seed + 2000)
    t4 = t0 + budget_s * 0.95
    for r in range(20):
        if time.time() > t4:
            break
        scale = 0.005 + 0.015 * rng2.random()
        mode = r % 5
        h_p = project_h(perturb_h(best_h, n, rng2, mode, scale), n)
        sub_t = min(t4, time.time() + max(2.0, budget_s * 0.004))
        h_opt, c5_opt = coord_descent(h_p, n, sub_t, 3000, seed + 3000 + r)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final polish
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, 5000, seed + 5000)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
