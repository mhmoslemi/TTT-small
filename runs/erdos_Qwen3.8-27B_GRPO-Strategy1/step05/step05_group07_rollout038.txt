def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(100):
        s = h.sum()
        if abs(s - tgt) < 1e-14:
            break
        if s > tgt:
            scale = tgt / s
            h = np.clip(h * scale, 0.0, 1.0)
        else:
            deficit = tgt - s
            headroom = n - s  # total room to increase
            if headroom < 1e-14:
                h = h * (tgt / max(s, 1e-14))
            else:
                h = np.clip(h + deficit / headroom, 0.0, 1.0)
    return h

def smax_softobj(x, T, n):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def smax_softgrad(x, T, n):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)

        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g

                     options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 50})
        return proj_h(r.x, n)
    except Exception:
        return h

    """Minimize the max of top-k correlation values."""
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    # Use smooth max of top-k
    topk = np.sort(c)[-k_top:]
    T = 50.0
    m = topk.max()
    return m + np.log(np.sum(np.exp(T * (topk - m)))) / T

def topk_minimax_grad(x, n, k_top=10):

    idx = np.argsort(c)[-k_top:]
    topk_c = c[idx]
    T = 50.0
    m = topk_c.max()
    e = np.exp(T * (topk_c - m))
    w = e / e.sum()
    # Now compute gradient only for these top-k shifts
    g_full = np.zeros(2 * n - 1)

def block_cd(h, n, t_end, block_size=4, max_sweeps=200):
    """Block coordinate descent using Nelder-Mead on blocks."""
    best = h.copy()
    best_c5 = compute_c5(best)
    order = np.random.permutation(n // block_size)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for bi in order:
            start = bi * block_size
            end = min(start + block_size, n)
            sz = end - start
            if sz < 2:
                continue
            
            def blk_obj(xb):
                trial = best.copy()
                trial[start:end] = np.clip(xb, 0, 1)
                trial = proj_h(trial, n)
                return compute_c5(trial)
            
            try:
                r = minimize(best[start:end], blk_obj, method='Nelder-Mead',
                           bounds=[(0,1)]*sz,
                           options={'maxiter': 100, 'xatol': 1e-10, 'fatol': 1e-15, 'adaptive': True})
                new_blk = np.clip(r.x, 0, 1)
                trial = best.copy()
                trial[start:end] = new_blk
                trial = proj_h(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_c5 - 1e-14:
                    best = trial
                    best_c5 = c5t
                    improved = True
            except Exception:
                pass
        if not improved:
            # Shuffle and try again
            order = np.random.permutation(n // block_size)
        elif sweep > 5 and not improved:
            break
    return best, best_c5

def fourier_init(n, seed=42):

        h_full[half:] = 1.0 - hp
        c = np.correlate(h_full, 1.0 - h_full, mode='full') * 2.0 / n
        # Find which k achieves max
        # For the antiperiodic case, the gradient w.r.t. hp[i] involves
        # contributions to h_full[i] and h_full[i+half]
        # This requires the gradient of c w.r.t. h_full, then chain rule
        # Let me just use the soft-max gradient
        T = 10000.0
        m = c.max()
        e = np.exp(T * (c - m))
        w = e / e.sum()
        g_full = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            lo1 = max(0, n - 1 - k)
            hi1 = min(n, 2 * n - 1 - k)
            if lo1 < hi1:
                idx = np.arange(lo1, hi1) + k - n + 1
                g_full[lo1:hi1] += wk * (1.0 - h_full[idx])
            lo2 = max(0, k - n + 1)
            hi2 = min(n, k + 1)
            if lo2 < hi2:
                idx = np.arange(lo2, hi2) - k + n - 1
                g_full[lo2:hi2] -= wk * h_full[idx]
        # Chain rule: h_full[i] = hp[i], h_full[i+half] = 1-hp[i]
        g_anti = g_full[:half] - g_full[half:]
        return g_anti
    
    best = hp.copy()
    best_c5 = eval_anti(best)
    for _ in range(30):

            r = minimize(best, eval_anti, grad_anti, method='L-BFGS-B',
                        bounds=[(0.0, 1.0)] * half,
                        options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10})
            new_hp = np.clip(r.x, 0, 1)
            h_full = np.zeros(n)
            h_full[:half] = new_hp
            h_full[half:] = 1.0 - new_hp
            c5_new = compute_c5(h_full)
            if c5_new < best_c5:
                best = new_hp
                best_c5 = c5_new
        except:
            break
    
    h_full = np.zeros(n)
    h_full[:half] = best
    h_full[half:] = 1.0 - best
    return h_full, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(80):
        s = h.sum()
        if abs(s - tgt) < 1e-14:
            break
        if s > tgt:
            h = np.clip(h * (tgt / s), 0.0, 1.0)
        else:
            headroom = n - s
            if headroom < 1e-14:
                h = h * (tgt / max(s, 1e-14))
            else:
                h = np.clip(h + (tgt - s) / headroom, 0.0, 1.0)
    return h

def _softgrad(x, T, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0 / n)

def _softobj(x, T, n):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def lbfgs_smax(h, T, n, iters=400):
    bnd = [(0.0, 1.0)] * n
    try:
        r = minimize(h, _softobj, _softgrad, method='L-BFGS-B',
                     bounds=bnd, args=(T, n),
                     options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 40})
        return proj_h(r.x, n)
    except Exception:
        return h

def anti_smax_obj(hp, T, n):
    half = n // 2
    h = np.empty(n)
    h[:half] = hp
    h[half:] = 1.0 - hp
    return _softobj(h, T, n)

def anti_smax_grad(hp, T, n):
    half = n // 2
    h = np.empty(n)
    h[:half] = hp
    h[half:] = 1.0 - hp
    g_full = _softgrad(h, T, n)
    return g_full[:half] - g_full[half:]

def anti_opt(hp, T, n, iters=300):
    half = n // 2
    bnd = [(0.0, 1.0)] * half
    try:
        r = minimize(hp, anti_smax_obj, anti_smax_grad, method='L-BFGS-B',
                     bounds=bnd, args=(T, n),
                     options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-14})
        return np.clip(r.x, 0.0, 1.0)
    except Exception:
        return hp

def anti_to_full(hp, n):
    half = n // 2
    h = np.empty(n)
    h[:half] = hp
    h[half:] = 1.0 - hp
    return h

def block_cd(h, n, t_end, bs=4, sweeps=150):
    best = h.copy()
    best_c5 = compute_c5(best)
    nblk = n // bs
    for sweep in range(sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(nblk)
        for bi in order:
            s = bi * bs
            e = min(s + bs, n)
            sz = e - s
            if sz < 2:
                continue
            def blk_f(xb):
                tr = best.copy()
                tr[s:e] = np.clip(xb, 0, 1)
                tr = proj_h(tr, n)
                return compute_c5(tr)
            try:
                r = minimize(best[s:e], blk_f, method='Nelder-Mead',
                             options={'maxiter': 150, 'xatol': 1e-11, 'fatol': 1e-16, 'adaptive': True})
                nb = np.clip(r.x, 0, 1)
                tr = best.copy()
                tr[s:e] = nb
                tr = proj_h(tr, n)
                c5t = compute_c5(tr)
                if c5t < best_c5 - 1e-14:
                    best = tr
                    best_c5 = c5t
                    improved = True
            except Exception:
                pass
        if not improved and sweep > 3:
            break
    return best, best_c5

def coord1d(h, n, t_end, sweeps=60):
    best = h.copy()
    best_c5 = compute_c5(best)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            def f1(v):
                tr = best.copy()
                tr[i] = v
                tr = proj_h(tr, n)
                return compute_c5(tr)
            r = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                            options={'xatol': 1e-13, 'maxiter': 40})
            nv = float(r.x)
            if abs(nv - old) < 1e-12:
                continue
            tr = best.copy()
            tr[i] = nv
            tr = proj_h(tr, n)
            c5t = compute_c5(tr)
            if c5t < best_c5 - 1e-14:
                best = tr
                best_c5 = c5t
                imp = True
        if not imp:
            break
    return best, best_c5
