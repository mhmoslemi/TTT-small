```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14: return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14: return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14: return np.full(n, 0.5)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def obj_sm(h, T, lam, w=None):
    """Smoothed max objective with optional lag weights"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if w is not None:
        corr_w = corr * w
    else:
        corr_w = corr
    mc = np.max(corr_w)
    Tc = T * (corr_w - mc)
    Tc = Tc - np.max(Tc)
    exps = np.exp(Tc)
    sm = mc + np.log(exps.sum()) / T
    return dx * sm + lam * (h.sum() - n / 2.0)**2, exps / exps.sum()

def grad_sm(h, T, lam, w=None):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    if w is not None:
        corr_w = corr * w
    else:
        corr_w = corr
    mc = np.max(corr_w)

    Tc = Tc - np.max(Tc)
    exps = np.exp(Tc)
    ww = exps / exps.sum()
    grad = np.zeros(n)
    lag_arr = np.arange(m_full) - n + 1

        wi = ww[idx] * (w[idx] if w is not None else 1.0)

            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)

    n = len(h0)
    x0 = project_h(h0)
    def f(x): return obj_sm(x, T, lam, w)[0]
    def g(x): return grad_sm(x, T, lam, w)
    try:

                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except:

    """Exact 1D coordinate descent for min max_k corr[k]"""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1

            vi = best_h[i]
            best_h[i] = 0.0

            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):

                if lg == 0:

                elif lg > 0:

            if c5t < best_c5 - 1e-15:

                improved = True
            elif abs(res.x - vi) < 1e-12:
                pass
        if not improved and sweep > 3:

    return best_h, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.full(n, 0.5)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def sm_obj_and_grad(h, T, lam, lag_w=None):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cw = corr * lag_w if lag_w is not None else corr
    mc = np.max(cw)
    Tc = T * (cw - mc)
    Tc -= np.max(Tc)
    ex = np.exp(Tc)
    ws = ex / ex.sum()
    val = dx * (mc + np.log(ex.sum()) / T) + lam * (h.sum() - n/2.0)**2
    # Gradient
    grad = np.zeros(n)
    la = np.arange(m) - n + 1
    for k in range(m):
        lg = la[k]
        wk = ws[k] * (lag_w[k] if lag_w is not None else 1.0)
        if wk < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wk * (1.0 - h[:n-lg])
                grad[:n-lg] -= wk * h[lg:]
        else:
            p = -lg
            grad[:n-p] += wk * (1.0 - h[p:])
            grad[p:] -= wk * h[:n-p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n/2.0)
    return val, grad

def lbfgs_step(h0, T, lam, lag_w, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def f(x): return sm_obj_and_grad(x, T, lam, lag_w)[0]
    def g(x): return sm_obj_and_grad(x, T, lam, lag_w)[1]
    try:
        r = minimize(f, x0, jac=g, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)]*n,
                     options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(r.x)
    except:
        return x0

            # Compute lin and quad for coordinate i
            lin = np.zeros(m)
            quad = np.zeros(m)
            for k in range(m):

                if lg == 0:
                    lin[k] = 1.0
                    quad[k] = -1.0
                elif lg > 0:

                else:

            # Base correlation (h[i] set to 0)
```

```python
def _min_f1d(base, lin, quad):
    """Minimize max_k (base[k] + lin[k]*x + quad[k]*x^2) over x in [0,1]
    Here quad is 0 for all k except k=0 where it's -1."""
    # Evaluate at boundaries and at critical points
    candidates = [0.0, 1.0]
    # The quadratic (k=0) has derivative 1 + 2*quad[0]*x = 1 - 2x, critical at x=0.5
    candidates.append(0.5)
    # For each line k, it intersects the quadratic at:
    # base[0] + x - x^2 = base[k] + lin[k]*x
    # x^2 + (lin[k]-1)*x + (base[k]-base[0]) = 0
    for k in range(len(base)):
        if k == 0: continue
        a = 1.0
        b = lin[k] - 1.0
        c = base[k] - base[0]
        disc = b*b - 4*a*c
        if disc >= 0:
            r1 = (-b + np.sqrt(disc)) / (2*a)
            r2 = (-b - np.sqrt(disc)) / (2*a)
            if 0 <= r1 <= 1: candidates.append(r1)
            if 0 <= r2 <= 1: candidates.append(r2)
    # Also crossings between lines
    # f_k(x) = f_j(x) => (lin[k]-lin[j])*x + (base[k]-base[j]) = 0
    # This gives x = (base[j]-base[k])/(lin[k]-lin[j])
    # But there are O(m²) pairs... for m=319, that's ~50K pairs. Too many.
    # Instead, just sample and use a few iterations of bisection
    
    # Simpler: ternary search or golden section on the piecewise function
    # Since f is continuous and piecewise quadratic (actually: max of one concave quad and lines),
    # f is continuous. We can use a simple approach:
    
    # Evaluate f at candidates and also do a fine grid
    # Actually, let's just do: sample 200 points, find best, then zoom in
    
    xs = np.linspace(0, 1, 200)
    vals = np.array([np.max(base + x*lin + x*x*quad) for x in xs])
    idx = np.argmin(vals)
    # Zoom in
    if idx == 0: lo, hi = 0.0, xs[1]
    elif idx == len(xs)-1: lo, hi = xs[-2], 1.0
    else: lo, hi = xs[idx-1], xs[idx+1]
    # Bisection/golden section on [lo, hi]
    for _ in range(50):
        m1 = lo + (hi-lo)/3
        m2 = hi - (hi-lo)/3
        f1 = np.max(base + m1*lin + m1*m1*quad)
        f2 = np.max(base + m2*lin + m2*m2*quad)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2
```

This is O(200 * m + 50 * m) = O(100 * m) per coordinate. For m=319, that's ~32K per coordinate, ~5M per sweep. In Python that might be slow...

The list comprehension `[np.max(base + x*lin + x*x*quad) for x in xs]` is 200 iterations each doing O(m) work = 200*319 ≈ 64K. Plus the 50 iterations of golden section = 50*319 ≈ 16K. Total ~80K per coordinate. Per sweep: 160*80K = 12.8M. In Python (not vectorized), this might take ~1-2 seconds per sweep. In 400 seconds, that's 200-400 sweeps. Should be enough.

```python
vals = np.max(base[None, :] + xs[:, None]*lin[None, :] + (xs[:, None])**2 * quad[None, :], axis=1)
```

This creates a 200×319 array. Much faster.

Similarly for the golden section, each evaluation is O(m) which is fine.

Let me also think about whether I can vectorize the lin/quad computation across all coordinates. For coordinate i:
- lin[k] for lag lg>0: (1-best[i-lg]) if i>=lg, plus -best[i+lg] if i+lg<n
- This is essentially: lin[k] = (1-best[i-la[k]]) * (i>=la[k]) + (-best[i+la[k]]) * (i+la[k]<n) for lg>0

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.full(n, 0.5)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _sm_val_grad(h, T, lam, lag_w):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cw = corr * lag_w
    mc = np.max(cw)
    Tc = T * (cw - mc)
    Tc -= np.max(Tc)
    ex = np.exp(Tc)
    ws = ex / ex.sum()
    val = dx * (mc + np.log(ex.sum()) / T) + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    la = np.arange(m) - n + 1
    for k in range(m):
        lg = la[k]
        wk = ws[k] * lag_w[k]
        if wk < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wk * (1.0 - h[:n - lg])
                grad[:n - lg] -= wk * h[lg:]
        else:
            p = -lg
            grad[:n - p] += wk * (1.0 - h[p:])
            grad[p:] -= wk * h[:n - p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_step(h0, T, lam, lag_w, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def f(x):
        return _sm_val_grad(x, T, lam, lag_w)[0]
    def g(x):
        return _sm_val_grad(x, T, lam, lag_w)[1]
    try:
        r = minimize(f, x0, jac=g, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(r.x)
    except Exception:
        return x0

def _f1d_vec(base, lin, quad, x):
    return np.max(base + x * lin + x * x * quad)

def _min_1d(base, lin, quad):
    xs = np.linspace(0.0, 1.0, 100)
    vals = np.array([_f1d_vec(base, lin, quad, x) for x in xs])
    idx = int(np.argmin(vals))
    if idx == 0:
        lo, hi = 0.0, xs[1]
    elif idx == len(xs) - 1:
        lo, hi = xs[-2], 1.0
    else:
        lo, hi = xs[idx - 1], xs[idx + 1]
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = _f1d_vec(base, lin, quad, m1)
        f2 = _f1d_vec(base, lin, quad, m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def cd_sweeps(h, t_end, max_sweeps):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    m = 2 * n - 1
    la = np.arange(m) - n + 1
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full').copy()
        for i in range(n):
            if time.time() > t_end:
                break
            vi = best[i]
            ci = np.zeros(m)
            lin = np.zeros(m)
            quad = np.zeros(m)
            for k in range(m):
                lg = la[k]
                if lg == 0:
                    ci[k] = vi * (1.0 - vi)
                    lin[k] = 1.0
                    quad[k] = -1.0
                elif lg > 0:
                    c = 0.0
                    if i >= lg:
                        c += vi * (1.0 - best[i - lg])
                        lin[k] += (1.0 - best[i - lg])
                    if i + lg < n:
                        c += best[i + lg] * (1.0 - vi)
                        lin[k] += -best[i + lg]
                    ci[k] = c
                else:
                    p = -lg
                    c = 0.0
                    if i + p < n:
                        c += vi * (1.0 - best[i + p])
                        lin[k] += (1.0 - best[i + p])
                    if i >= p:
                        c += best[i - p] * (1.0 - vi)
                        lin[k] += -best[i - p]
                    ci[k] = c
            base = corr - ci
            xnew = _min_1d(base, lin, quad)
            if abs(xnew - vi) < 1e-12:
                continue
            best[i] = xnew
            ci_new = np.zeros(m)
            for k in range(m):
                lg = la[k]
                if lg == 0:
                    ci_new[k] = xnew * (1.0 - xnew)
                elif lg > 0:
                    c = 0.0
                    if i >= lg:
                        c += xnew * (1.0 - best[i - lg])
                    if i + lg < n:
                        c += best[i + lg] * (1.0 - xnew)
                    ci_new[k] = c
                else:
                    p = -lg
                    c = 0.0
                    if i + p < n:
                        c += xnew * (1.0 - best[i + p])
                    if i >= p:
                        c += best[i - p] * (1.0 - xnew)
                    ci_new[k] = c
            corr = base + ci_new
            c_new = np.max(corr) * dx
            if c_new < best_c5 - 1e-15:
                best_c5 = c_new
                improved = True
        if not improved and sweep > 2:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s * 0.97
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    m = 2 * n - 1
    # Phase 1: L-BFGS warmup
    p1 = t0 + budget_s * 0.07
    for T in [300, 2000, 8000]:
        if time.time() > p1:
            break
        hc = lbfgs_step(best_h, T, 250.0, np.ones(m), 2000, 1e-16)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5, best_h = c, hc
    # Phase 2: Extended CD
    p2 = t0 + budget_s * 0.35
    best_h, best_c5 = cd_sweeps(best_h, p2, 99999)
    # Phase 3: Active-set L-BFGS + CD interleaved
    p3 = t0 + budget_s * 0.55
    for it in range(40):
        if time.time() > p3:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        thr = np.percentile(corr, 85)
        lw = np.where(corr >= thr, 3.0, 1.0)
        T = 3000 + 2000 * it
        hc = lbfgs_step(best_h, T, 250.0, lw, 1500, 1e-16)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5, best_h = c, hc
        elif hc is not None:
            c2 = compute_c5(hc)
            if c2 < best_c5 + 0.0001:
                best_h = hc
        best_h, best_c5 = cd_sweeps(best_h, min(time.time() + 8, p3), 80)
    # Phase 4: Alternating CD + full L-BFGS
    p4 = t0 + budget_s * 0.78
    for cyc in range(8):
        if time.time() > p4:
            break
        t_cd_end = min(time.time() + 25, p4 - 15)
        best_h, best_c5 = cd_sweeps(best_h, t_cd_end, 3000)
        if time.time() > p4:
            break
        hc = lbfgs_step(best_h, 30000, 400.0, np.ones(m), 3000, 1e-17)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5, best_h = c, hc
    # Phase 5: Basin hopping
    p5 = t0 + budget_s * 0.93
    for it in range(50):
        if time.time() > p5:
            break
        cool = 1.0 - it / 50.0
        sc = 0.003 + 0.025 * cool
        hp = project_h(best_h + sc * np.random.randn(n))
        T_bh = 5000 * cool + 2000
        hc = lbfgs_step(hp, T_bh, 300.0, np.ones(m), 1000, 1e-15)
        c = compute_c5(hc)
        temp = 0.0002 * cool + 0.00003
        if c < best_c5:
            best_c5, best_h = c, hc
        elif np.random.rand() < np.exp(-(c - best_c5) / temp):
            best_h = hc
            best_h, c2 = cd_sweeps(best_h, min(time.time() + 4, p5), 60)
            if c2 < best_c5:
                best_c5 = c2
    # Phase 6: Final polish
    if time.time() < t_end - 15:
        best_h, best_c5 = cd_sweeps(best_h, t_end - 15, 5000)
    if time.time() < t_end - 5:
        hc = lbfgs_step(best_h, 150000, 500.0, np.ones(m), 3000, 1e-17)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5, best_h = c, hc
        best_h, best_c5 = cd_sweeps(best_h, t_end, 2000)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
