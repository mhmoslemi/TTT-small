def proj_h(h, n):

    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:

def _sobj_grad(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    obj = (m + np.log(e.sum()) / T) * 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return obj, g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=400):
    def f_obj(x):
        return _sobj_grad(x, T)[0]
    def f_grad(x):
        return _sobj_grad(x, T)[1]
    try:
        r = minimize(h, f_obj, f_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 40})
        return proj_h(r.x, n)
    except:
        return h

def bin_swap_delta(hb, i, j, n):
    """Compute delta for corr when swapping hb[i]=1->0, hb[j]=0->1. Returns new_corr."""
    corr = np.correlate(hb, 1.0 - hb, mode='full')
    for k in range(2 * n - 1):
        ii = i + k - n + 1
        jj = j + k - n + 1
        pi = i - k + n - 1
        qj = j - k + n - 1
        d = 0.0
        if 0 <= i <= n - 1 and 0 <= ii < n:
            d -= (1.0 - hb[ii])
        if 0 <= pi < n:
            d += hb[pi]
        if 0 <= j <= n - 1 and 0 <= jj < n:
            d += (1.0 - hb[jj])
        if 0 <= qj < n:
            d -= hb[qj]
        corr[k] += d
    return corr

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-start L-BFGS-B with soft-max
    for T in [100, 500, 2000, 8000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.18, sweeps=40)

    # Phase 3: Binary SA with multiple restarts (key improvement)
    t_sa_end = t0 + budget_s * 0.50
    for restart in range(3):
        if time.time() > t_sa_end:
            break
        h_bin = (best_h > 0.5).astype(float)
        if restart > 0:
            # Perturb the binary solution
            swap_count = 3 + restart * 2
            ones_i = np.where(h_bin == 1)[0]
            zeros_i = np.where(h_bin == 0)[0]
            for _ in range(swap_count):
                i = np.random.choice(ones_i)
                j = np.random.choice(zeros_i)
                h_bin[i], h_bin[j] = 0, 1
                ones_i = np.where(h_bin == 1)[0]
                zeros_i = np.where(h_bin == 0)[0]
        h_b, c5_b = bin_sa_fast(h_bin, n, t_sa_end - (budget_s * 0.15) * (3 - restart),
                                max_iters=150000)
        if c5_b < best_c5:
            h_cont = lbfgs_opt(h_b, 3000.0, n, iters=300)
            c5_cont = compute_c5(h_cont)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont
            else:
                best_c5 = c5_b
                best_h = h_b

    # Phase 4: Basin hopping on continuous
    t_bh = t0 + budget_s * 0.50
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.72, n_iters=100)

    # Phase 5: Another round of binary SA from best continuous
    h_bin2 = (best_h > 0.5).astype(float)
    h_b2, c5_b2 = bin_sa_fast(h_bin2, n, t0 + budget_s * 0.82, max_iters=120000)
    if c5_b2 < best_c5:
        best_h = h_b2
        best_c5 = c5_b2

    # Phase 6: Final continuous polish
    if time.time() - t0 < budget_s * 0.92:
        h_new = lbfgs_opt(best_h, 20000.0, n, iters=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.985, sweeps=25)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
