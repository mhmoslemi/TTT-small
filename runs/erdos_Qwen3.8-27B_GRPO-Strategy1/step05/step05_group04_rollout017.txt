def struct_perturb(h, n, rng, scale, mode):
    h_p = h.copy()
    if mode == 0:
        blk = min(n // 8, max(4, int(scale * n)))
        start = rng.randint(0, n - blk)
        seg = h_p[start:start+blk]
        shift = rng.randint(1, blk)
        h_p[start:start+blk] = np.concatenate([seg[shift:], seg[:shift]])
    elif mode == 1:

        amp = scale * 0.08
        phase = rng.uniform(0, 2 * np.pi)
        h_p += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)

        idx = rng.choice(n, size=k, replace=False)
        h_p[idx] += scale * 0.15 * rng.randn(k)

def soft_grad_h(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    exponents = np.clip(T * (corr - corr.max()), -50.0, 50.0)
    p = np.exp(exponents)
    p /= p.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = p[m]
        if w < 1e-12:
            continue
        s = m - (n - 1)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
    return dx * grad

    best_c5 = compute_c5(best_h)

        frac = it / max_iters
        T = 30.0 + 800.0 * frac

        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h, cur_c5 = h_opt, c5_opt

                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end_total = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial CD
    t1 = t0 + budget_s * 0.18
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=8000, seed=seed)

    # Phase 2: Gradient phase
    if time.time() < t0 + budget_s * 0.30:
        h_g, c5_g = grad_phase(best_h, n, t0 + budget_s * 0.30, max_iters=3000, seed=seed+5)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.34, max_sweeps=5000, seed=seed+10)

    # Phase 3: Basin hopping (main search)
    if time.time() < t0 + budget_s * 0.68:
        best_h, best_c5 = basin_hopping(best_h, n, t0 + budget_s * 0.68, n_hops=600, seed=seed+50)

    # Phase 4: CD polish
    if time.time() < t0 + budget_s * 0.82:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82, max_sweeps=8000, seed=seed+200)

    # Phase 5: Second basin hop with different perturbs
    if time.time() < t0 + budget_s * 0.90:
        best_h, best_c5 = basin_hopping(best_h, n, t0 + budget_s * 0.90, n_hops=300, seed=seed+500)

    # Phase 6: Gradient + CD final
    if time.time() < t0 + budget_s * 0.95:
        h_g2, c5_g2 = grad_phase(best_h, n, t0 + budget_s * 0.94, max_iters=1000, seed=seed+999)
        if c5_g2 < best_c5:
            best_h, best_c5 = h_g2, c5_g2
        best_h, best_c5 = coord_descent(best_h, n, min(t_end_total - 2, t0 + budget_s * 0.97), max_sweeps=5000, seed=seed+300)

    # Final polish
    if time.time() < t_end_total - 1:
        best_h, best_c5 = coord_descent(best_h, n, t_end_total - 0.5, max_sweeps=2000, seed=seed+9999)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
