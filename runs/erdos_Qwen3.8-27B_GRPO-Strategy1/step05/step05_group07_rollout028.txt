        return proj_h(r.x, n)
    except Exception:

def coord_desc(h, n, t_end, sweeps=60):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr_cache = np.correlate(best, 1.0 - best, mode='full')
        mf = 2 * n - 1
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            # Compute coefficients for the 1D problem
            coeff = np.zeros(mf)
            shift_range = range(mf)
            for k in shift_range:
                shift = k - (n - 1)
                j2 = i + shift
                if 0 <= j2 < n:
                    coeff[k] += 1.0 - best[j2]
                j1 = i - shift
                if 0 <= j1 < n:
                    coeff[k] -= best[j1]
            base = corr_cache - old * coeff
            # minimize max(base[i] + new_val * coeff[i]) over new_val in [0,1]
            # This is a 1D problem
            def _obj(v):
                return float(np.max(base + v * coeff))
            r = minimize_scalar(_obj, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-10})
            nv = float(r.x)
            if abs(nv - old) > 1e-10:
                trial = best.copy()
                trial[i] = nv
                trial = proj_h(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_c5 - 1e-14:
                    best = trial
                    best_c5 = c5t
                    corr_cache = np.correlate(best, 1.0 - best, mode='full')
                    imp = True
        if not imp:
            break
    return best, best_c5

def penalty_refine(h, n, t_end, n_rounds=5):
    """Focus on active constraints using penalty method"""
    best = h.copy()
    best_c5 = compute_c5(best)
    for rnd in range(n_rounds):
        if time.time() > t_end:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        c5 = corr.max() * 2.0 / n
        # Find active lags (within 0.5% of max)
        threshold = c5 - 0.005 * 2.0 / n
        active = np.where(corr * 2.0 / n > threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[-5:]
        # Define penalty objective: sum of (c_k - target)^2 for active k
        target = max(best_c5 * n / 2.0 - 0.002, 0.0)  # slightly below current
        def pen_obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            excess = c[active] - target
            excess = np.maximum(excess, 0)
            return np.sum(excess ** 2) * 2.0 / n
        def pen_grad(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            excess = np.maximum(c[active] - target, 0)
            g = np.zeros(n)
            for idx in active:
                if excess[np.searchsorted(active, idx)] < 1e-15:
                    continue
                shift = idx - (n - 1)
                j1a, j1b = max(0, -shift), min(n, n - shift)
                if j1a < j1b:
                    g[j1a:j1b] += 2 * excess[np.searchsorted(active, idx)] * (1.0 - x[j1a + shift:j1b + shift])
                j2a, j2b = max(0, shift), min(n, n + shift)
                if j2a < j2b:
                    g[j2a:j2b] -= 2 * excess[np.searchsorted(active, idx)] * x[j2a - shift:j2b - shift]
            return g * 2.0 / n
        try:
            r = minimize(best, pen_obj, pen_grad, method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-10})
            hp = proj_h(r.x, n)
            c5p = compute_c5(hp)
            if c5p < best_c5:
                best = hp
                best_c5 = c5p
        except Exception:
            pass
    return best, best_c5

def swap_sa(h, n, t_end, max_iters=100000):

    ones = np.where(h_b == 1)[0]
    n_ones = n // 2
    if len(ones) > n_ones:
        h_b[np.random.choice(ones, len(ones) - n_ones, replace=False)] = 0
    elif len(ones) < n_ones:

    dx = 2.0 / n
    best_h = h_b.copy()

    T = 0.02
    for _ in range(max_iters):

        j = np.random.randint(n)

        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        T = max(0.0003, T * 0.99995)

def basin_hop(h, n, t_end, n_iters=80):

        nc5 = float(np.max(np.correlate(h_b, 1.0 - h_b, mode='full')) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        T = max(0.0002, T * 0.99993)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=70):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        scale = 0.03 + 0.07 * cool
        pert = scale * np.random.randn(n)
        f = np.random.randint(1, 10)
        ph = np.random.uniform(0, 2 * np.pi)
        pert += 0.015 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        hp = proj_h(work + pert, n)
        T_val = 400.0 + 3000.0 * cool
        ht = lbfgs_smooth(hp, T_val, n, iters=180)
        c5t = compute_c5(ht)
        temp = 0.0004 * cool + 0.00002
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def nesterov_refine(h, n, t_end, n_steps=8000):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    h_prev = h.copy()
    base_step = 0.015
    for t in range(n_steps):
        if time.time() > t_end:
            break
        momentum = min(0.9, 0.3 + 0.0005 * t)
        y = h_curr + momentum * (h_curr - h_prev)
        corr = np.correlate(y, 1.0 - y, mode='full')
        topk = np.argsort(corr)[-5:][::-1]
        m = corr.max()
        w = np.exp(200.0 * (corr[topk] - m))
        w = w / w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, topk):
            if wi < 1e-6:
                continue
            shift = int(ki) - (n - 1)
            j1a, j1b = max(0, -shift), min(n, n - shift)
            if j1a < j1b:
                g[j1a:j1b] += wi * (1.0 - y[j1a + shift:j1b + shift])
            j2a, j2b = max(0, shift), min(n, n + shift)
            if j2a < j2b:
                g[j2a:j2b] -= wi * y[j2a - shift:j2b - shift]
        step = base_step / np.sqrt(1.0 + 0.01 * t)
        h_new = proj_h(y - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_prev = h_curr
        h_curr = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS smooth with aggressive T
    t_p1 = t0 + budget_s * 0.10
    for T in [80, 400, 2000, 8000, 40000, 200000]:
        if time.time() > t_p1:
            break
        h_new = lbfgs_smooth(best_h, T, n, iters=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Penalty refinement
    best_h, best_c5 = penalty_refine(best_h, n, t0 + budget_s * 0.20, n_rounds=3)

    # Phase 3: Coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.34, sweeps=70)

    # Phase 4: Nesterov subgradient
    best_h, best_c5 = nesterov_refine(best_h, n, t0 + budget_s * 0.44, n_steps=6000)
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.48, sweeps=30)

    # Phase 5: Binary SA
    h_bin, c5_bin = swap_sa(best_h, n, t0 + budget_s * 0.58)
    if c5_bin < best_c5:
        best_h = lbfgs_smooth(h_bin, 3000.0, n, iters=300)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.62, sweeps=35)

    # Phase 6: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.76, n_iters=55)

    # Phase 7: Penalty + coord
    best_h, best_c5 = penalty_refine(best_h, n, t0 + budget_s * 0.82, n_rounds=3)
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.88, sweeps=50)

    # Phase 8: Try nearby n
    if time.time() - t0 < budget_s * 0.91:
        for n_try in [n + 8, n - 8]:
            if time.time() - t0 > budget_s * 0.94:
                break
            if n_try < 40:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_try = np.interp(x_new, x_old, best_h)
            h_try = proj_h(h_try, n_try)
            h_try = lbfgs_smooth(h_try, 5000.0, n_try, iters=250)
            c5_try = compute_c5(h_try)
            h_try, c5_try = coord_desc(h_try, n_try, t0 + budget_s * 0.95, sweeps=25)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                n = n_try

    # Phase 9: Final polish
    if time.time() - t0 < budget_s * 0.96:
        best_h = lbfgs_smooth(best_h, 500000.0, len(best_h), iters=600)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.99:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t0 + budget_s * 0.995, sweeps=35)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
