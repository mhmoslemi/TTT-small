def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(np.clip(T * (corr - m), -500, 500)))) / T
    n = len(h)
    dx = 2.0 / n
    return val * dx

    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()

    if diff > 0:

                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt(h, n, i, niter=50):

    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)

        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-10:

            h_try[i] = hi_new

                best_h = h_try
                improved = True
        if not improved:

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(np.clip(T * (corr - m), -400, 400)))) / T
    n = len(h)
    return val * (2.0 / n)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_1d(h, n, i, niter=45):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_opt_2d(h, n, i, j, niter=30):
    h_base = h.copy()
    h_base[i] = 0.0
    h_base[j] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    def make_coefs(idx):
        vm1 = idx - m_vals
        vm2 = idx + m_vals
        v1 = (vm1 >= 0) & (vm1 < n)
        v2 = (vm2 >= 0) & (vm2 < n)
        c = np.zeros(2 * n - 1)
        c[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
        c[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
        return c
    ci = make_coefs(i)
    cj = make_coefs(j)
    lo1, hi1, lo2, hi2 = 0.0, 1.0, 0.0, 1.0
    for _ in range(niter):
        m1a = lo1 + (hi1 - lo1) / 3.0
        m1b = hi1 - (hi1 - lo1) / 3.0
        m2a = lo2 + (hi2 - lo2) / 3.0
        m2b = hi2 - (hi2 - lo2) / 3.0
        f1 = np.max(corr_base + m1a * ci + m2a * cj)
        f2 = np.max(corr_base + m1b * ci + m2a * cj)
        f3 = np.max(corr_base + m1a * ci + m2b * cj)
        f4 = np.max(corr_base + m1b * ci + m2b * cj)
        best = min(f1, f2, f3, f4)
        if best == f1:
            hi1 = m1b
            hi2 = m2b
        elif best == f2:
            lo1 = m1a
            hi2 = m2b
        elif best == f3:
            hi1 = m1b
            lo2 = m2a
        else:
            lo1 = m1a
            lo2 = m2a
    return (lo1 + hi1) / 2.0, (lo2 + hi2) / 2.0

def coord_sweep(h, n, t_end, use_pairs=False, n_pair=20):
    improved = False
    order = np.random.permutation(n)
    for i in order:
        if time.time() > t_end:
            return h, False
        hi_new = coord_opt_1d(h, n, i)
        if abs(hi_new - h[i]) > 1e-10:
            h[i] = hi_new
            improved = True
    if use_pairs:
        for _ in range(n_pair):
            if time.time() > t_end:
                return h, improved
            i, j = np.random.choice(n, 2, replace=False)
            hi, hj = coord_opt_2d(h, n, i, j)
            if abs(hi - h[i]) > 1e-10 or abs(hj - h[j]) > 1e-10:
                h[i] = hi
                h[j] = hj
                improved = True
    h = project_h(h)
    return h, improved

def coord_descent(h, n, t_end, max_sweeps=100, use_pairs=False):
    best_c5 = compute_c5(h)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        h, imp = coord_sweep(h, n, t_end, use_pairs=use_pairs)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
        if not imp:
            break
    return h, best_c5

def fourier_perturb(h, n, scale, rng):
    hf = np.fft.rfft(h)
    nf = len(hf)
    noise = (rng.randn(nf) + 1j * rng.randn(nf))
    noise[0] = 0
    for k in range(1, min(6, nf)):
        noise[k] *= 0.4
    hp = np.fft.irfft(hf + scale * noise, n=n)
    return project_h(hp.real)

def gaussian_perturb(h, n, scale, rng):
    return project_h(h + scale * rng.randn(n))

def block_perturb(h, n, rng):
    blk = max(2, n // 10)
    a = rng.randint(0, n - blk)
    b = rng.randint(0, n - blk)
    pert = np.zeros(n)
    s = (h[a:a+blk].sum() - h[b:b+blk].sum())
    if abs(s) < 1e-8:
        return h
    pert[a:a+blk] = -0.3 * s / blk
    pert[b:b+blk] = 0.3 * s / blk
    return project_h(h + pert)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP warm-up (10%)
    for T in [80, 800, 8000]:
        if elapsed() > budget_s * 0.10:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=1500, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent (15%)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.25,
                                     max_sweeps=50, use_pairs=False)

    # Phase 3: Basin hopping (45%) - diverse perturbations + coord descent
    work_h = best_h.copy()
    work_c5 = best_c5
    t_bh_end = t0 + budget_s * 0.70
    n_iters = 0
    while time.time() < t_bh_end:
        n_iters += 1
        cool = max(0.0, 1.0 - n_iters / 120.0)
        mod = n_iters % 10
        scale = 0.03 + 0.12 * cool
        if mod == 0:
            h_p = gaussian_perturb(work_h, n, scale, rng)
        elif mod == 1:
            h_p = fourier_perturb(work_h, n, scale * 1.5, rng)
        elif mod == 2:
            h_p = block_perturb(work_h, n, rng)
        elif mod == 3:
            h_p = gaussian_perturb(work_h, n, scale * 0.5, rng)
        elif mod == 4:
            # Frequency-specific perturbation
            freq = rng.randint(2, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = scale * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
            h_p = project_h(work_h + pert)
        elif mod == 5:
            h_p = fourier_perturb(work_h, n, scale, rng)
        elif mod == 6:
            # Swap-based
            idx = rng.permutation(n)
            a, b = idx[0], idx[1]
            h_p = work_h.copy()
            h_p[a], h_p[b] = h_p[b], h_p[a]
            h_p = project_h(h_p)
        elif mod == 7:
            # Mirror perturbation
            h_p = work_h.copy()
            seg = n // 6
            a = rng.randint(0, n - 2 * seg)
            h_p[a:a+seg] = work_h[a+seg:a+2*seg]
            h_p[a+seg:a+2*seg] = work_h[a:a+seg]
            h_p = project_h(h_p)
        elif mod == 8:
            h_p = gaussian_perturb(work_h, n, scale * 0.7, rng)
        else:
            h_p = fourier_perturb(work_h, n, scale * 1.3, rng)

        # Local optimization: SLSQP + coordinate descent
        T_loc = 300 + 10000 * cool
        hc = slsqp_opt(h_p, T_loc, n, maxiter=400, ftol=1e-14)
        hc, c5v = coord_descent(hc, n, min(t_bh_end, time.time() + 3),
                                 max_sweeps=8, use_pairs=(n_iters % 3 == 0))

        temp = 0.0006 * cool + 0.00004
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 4: SLSQP high-T polish (10%)
    for T in [80000, 400000]:
        if time.time() > t0 + budget_s * 0.82:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=2500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Extended coordinate descent with pairs (12%)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94,
                                     max_sweeps=80, use_pairs=True)

    # Phase 6: Final fine-tuning (5%)
    if elapsed() < budget_s * 0.97:
        hc = slsqp_opt(best_h, 1500000, n, maxiter=2000, ftol=1e-16)

        if c5v < best_c5:
            best_c5 = c5v

    if elapsed() < budget_s * 0.99:

                                         max_sweeps=30, use_pairs=False)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
