import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_1d(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n - 1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    h[i] = old
    def f(t):
        return np.max(corr_base + t * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13, 'maxiter': 300})
    return res.x

def pair_opt(h, n, i, j):
    oi, oj = h[i], h[j]
    h[i] = 0.0; h[j] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    ci = np.zeros(2 * n - 1); cj = np.zeros(2 * n - 1)
    lags = np.arange(-(n - 1), n)
    for idx, c in [(i, ci), (j, cj)]:
        j1 = idx + lags; j2 = idx - lags
        v1 = (j1 >= 0) & (j1 < n); v2 = (j2 >= 0) & (j2 < n)
        c[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
        c[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    h[i] = oi; h[j] = oj
    def f(x):
        return np.max(corr_base + x[0] * ci + x[1] * cj)
    res = minimize(f, [oi, oj], method='Nelder-Mead',
                   options={'xatol': 1e-10, 'fatol': 1e-12, 'maxiter': 300, 'adaptive': True})
    return np.clip(res.x, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip((corr - m) * T, -400, 400)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_sweep(h, n, t_end, tol=1e-14):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    sweeps = 0
    while time.time() < t_end:

        order = np.random.permutation(n)

                best_h = h_try
                improved = True
        sweeps += 1
        if not improved:
            break
    return best_h, best_c5
