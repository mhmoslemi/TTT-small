```python
def exact_1d_min(base, coef, lo=0.0, hi=1.0):
    """Find min of max_k (base[k] + coef[k]*x) for x in [lo, hi]"""
    # Build upper envelope of lines
    # Sort by slope
    order = np.argsort(coef)
    slopes = coef[order]
    intercepts = base[order]
    
    # Convex hull trick for upper envelope
    # A line (m_i, c_i) is on the upper envelope if it's not dominated
    # Process lines in order of slope
    envelope = []  # list of (slope, intercept, x_start)
    for i in range(len(slopes)):
        m, c = slopes[i], intercepts[i]
        if not envelope:
            envelope.append((m, c, -np.inf))
            continue
        # Check if new line is above the last line in envelope at the intersection
        while envelope:
            m_prev, c_prev, x_start_prev = envelope[-1]
            if m == m_prev:
                # Same slope, keep the one with higher intercept
                if c <= c_prev:
                    break
                else:
                    envelope.pop()
                    continue
            # Intersection of (m, c) and (m_prev, c_prev):
            # m*x + c = m_prev*x + c_prev => x = (c_prev - c) / (m - m_prev)
            x_int = (c_prev - c) / (m - m_prev)
            if x_int <= x_start_prev:
                envelope.pop()
            else:
                envelope.append((m, c, x_int))
                break
        else:
            if not envelope:
                envelope.append((m, c, -np.inf))
            else:
                m_prev, c_prev, x_start_prev = envelope[-1]
                if m == m_prev:
                    if c > c_prev:
                        envelope.append((m, c, x_start_prev))
                else:
                    x_int = (c_prev - c) / (m - m_prev)
                    if x_int > x_start_prev:
                        envelope.append((m, c, x_int))
    
    # Now evaluate the envelope on [lo, hi]
    # The envelope is a piecewise linear function
    # Find minimum on [lo, hi]
    best_val = np.inf
    best_x = lo
    
    for i, (m, c, x_start) in enumerate(enumarate(envelope)):
        if i + 1 < len(envelope):
            x_end = envelope[i+1][2]
        else:
            x_end = np.inf
        
        # Evaluate at max(lo, x_start) and min(hi, x_end)
        eval_points = set()
        if x_start <= hi and lo <= x_end:
            if lo > x_start:
                eval_points.add(lo)
            if hi < x_end:
                eval_points.add(hi)
            # The minimum of a line on an interval is at an endpoint
            # But we only care about the interval [max(lo,x_start), min(hi,x_end)]
            x_left = max(lo, x_start)
            x_right = min(hi, x_end)
            if x_left <= x_right:
                val_left = m * x_left + c
                val_right = m * x_right + c
                if val_left < best_val:
                    best_val = val_left
                    best_x = x_left
                if val_right < best_val:
                    best_val = val_right
                    best_x = x_right
    
    # Also check the breakpoints within [lo, hi]
    for i in range(1, len(envelope)):
        x_bp = envelope[i][2]
        if lo <= x_bp <= hi:
            val = envelope[i][0] * x_bp + envelope[i][1]
            if val < best_val:
                best_val = val
                best_x = x_bp
    
    return best_x
```

```python
    b = len(block_indices)
    # Compute base correlation (with block coords set to 0)
    h_base = h.copy()
    h_base[block_indices] = 0

    m = len(corr_base)  # 2n-1
    coefs = []
    for idx in block_indices:
        m_vals = np.arange(-(n-1), n)
        # For the term h[idx]*(1-h[idx+k]): contributes (1-h[idx+k]) to lag k
        # For the term h[idx-k]*(1-h[idx]): contributes -h[idx-k] to lag... 
        # Actually let me just compute it directly:
        # Change in corr[k] when h[idx] goes from 0 to v:
        # corr[n-1+k] = sum_i h[i]*(1-h[i+k])
        # Terms involving h[idx]:
        #   h[idx]*(1-h[idx+k]) for valid idx+k: coefficient is (1-h[idx+k])
        #   h[idx-k]*(1-h[idx]) for valid idx-k: coefficient is -h[idx-k]
        coef = np.zeros(m)
        # h[idx] as first factor: appears in corr[n-1+k] where i=idx, j=idx+k
        for k in range(0, n):
            j = idx + k
            if 0 <= j < n:
                coef[n-1+k] += (1 - h[j])
        # h[idx] as second factor (in (1-h[j]) where j=idx): appears in corr[n-1+k] where j=idx, i=idx-k
        for k in range(0, n):
            i = idx - k
            if 0 <= i < n:
                coef[n-1+k] += (-h[i])
        coefs.append(coef)
    
    # LP: min t s.t. corr_base + v1*c1 + v2*c2 + ... + vb*cb <= t, 0<=vi<=1
    # Variables: t, v1, v2, ..., vb
    # Inequality: for each k: -corr_base[k] - c1[k]*v1 - ... + t >= 0
    # i.e., (1, -c1[k], -c2[k], ..., -cb[k]) · (t, v1, ..., vb) >= corr_base[k]
    # linprog: minimize c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    
    c_obj = np.zeros(b+1)

    A_ub = np.zeros((m + 2*b, b+1))
    b_ub = np.zeros(m + 2*b)

    # i.e., (-1, c1[k], c2[k], ..., cb[k]) · (t, v1, ..., vb) <= -corr_base[k]
    A_ub[:m, 0] = -1
    for i in range(b):
        A_ub[:m, i+1] = coefs[i]
    b_ub[:m] = -corr_base
    
    # 0 <= vi <= 1: handled by bounds
    bounds = [(None, None)] + [(0, 1)] * b
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    else:
        new_vals = h[block_indices]
    
    h_new = h.copy()
    h_new[block_indices] = new_vals
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_lp(h, n, indices, niter=10):
    """Optimize a subset of coordinates using LP (min-max)"""
    m = 2 * n - 1
    h_base = h.copy()
    h_base[indices] = 0.0
    corr_base = get_corr(h_base)
    
    b = len(indices)
    coefs = np.zeros((b, m))
    for ii, idx in enumerate(indices):
        # coefficient of h[idx] in each lag of the correlation
        # corr[n-1+k] involves h[i]*(1-h[i+k])
        # h[idx] appears as:
        # 1) first factor: h[idx]*(1-h[idx+k]) for k=0..n-1-idx
        # 2) second factor: h[idx-k]*(1-h[idx]) for k=0..idx
        #    i.e., contributes -h[idx-k] to lag (n-1+k)
        for k in range(n):
            j = idx + k
            if j < n:
                coefs[ii, n-1+k] += (1.0 - h[j])
            i = idx - k
            if i >= 0:
                coefs[ii, n-1+k] += (-h[i])
    
    # LP: min t s.t. corr_base + V@coefs^T <= t, 0<=v<=1
    # Variables: [v_1, ..., v_b, t]
    # Constraints: coefs[i,k]*v_i + ... - t <= -corr_base[k]
    n_var = b + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((m, n_var))
    A_ub[:, :b] = coefs.T  # coef_i[k] for variable v_i
    A_ub[:, b] = -1.0     # for t
    b_ub_vec = -corr_base
    
    bounds = [(0.0, 1.0)] * b + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_vec, bounds=bounds, method='highs')
        if res.success:
            return res.x[:b]
    except:
        pass
    return h[indices].copy()

def block_coord_descent(h, n, t_end, block_size=2, max_sweeps=50):
    """Coordinate descent with LP-based block optimization"""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        # Process blocks
        order = np.random.permutation(n)
        for start in range(0, n, block_size):
            if time.time() > t_end:
                return best_h, best_c5
            # Get a random block
            block = order[start:start+block_size]
            if len(block) < 2:
                # Single coordinate: use ternary
                v = coord_opt_ternary(best_h, n, block[0], niter=50)
                if abs(v - best_h[block[0]]) > 1e-12:
                    h_try = best_h.copy()
                    h_try[block[0]] = v
                    h_try = project_h(h_try)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
            else:
                vals = coord_opt_lp(best_h, n, block)
                if np.abs(vals - best_h[block]).max() > 1e-12:
                    h_try = best_h.copy()
                    h_try[block] = vals
                    h_try = project_h(h_try)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def coord_opt_ternary(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n-1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2*n-1)
    coefs[valid1] += (1.0 - h[vm1[valid1]])
    coefs[valid2] += (-h[vm2[valid2]])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi-lo)/3.0
        m2 = hi - (hi-lo)/3.0
        f1 = np.max(corr_base + m1*coefs)
        f2 = np.max(corr_base + m2*coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2.0
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_1d(h, n, i, niter=50):
    h0 = h.copy()
    h0[i] = 0.0
    cb = get_corr(h0)
    mv = np.arange(-(n-1), n)
    v1 = i - mv; v2 = i + mv
    w1 = (v1 >= 0) & (v1 < n); w2 = (v2 >= 0) & (v2 < n)
    cf = np.zeros(2*n-1)
    cf[w1] += (1.0 - h[v1[w1]])
    cf[w2] += (-h[v2[w2]])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi-lo)/3.0; m2 = hi - (hi-lo)/3.0
        if np.max(cb + m1*cf) < np.max(cb + m2*cf):
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2.0

def coord_2d(h, n, i, j, ni=35, nj=35):
    h0 = h.copy()
    h0[i] = 0.0; h0[j] = 0.0
    cb = get_corr(h0)
    mv = np.arange(-(n-1), n)
    cfi = np.zeros(2*n-1); cfj = np.zeros(2*n-1)
    for idx, cf in [(i, cfi), (j, cfj)]:
        v1 = idx - mv; v2 = idx + mv
        w1 = (v1 >= 0) & (v1 < n); w2 = (v2 >= 0) & (v2 < n)
        cf[w1] += (1.0 - h[v1[w1]])
        cf[w2] += (-h[v2[w2]])
    lo_i, hi_i, lo_j, hi_j = 0.0, 1.0, 0.0, 1.0
    for _ in range(ni):
        mi1 = lo_i + (hi_i-lo_i)/3; mi2 = hi_i - (hi_i-lo_i)/3
        def eval_j(mix):
            l, h = 0.0, 1.0
            bestv = 0.0

                    h = mj2
                else:
                    l = mj1
            return (l+h)/2
        vj1 = eval_j(mi1); vj2 = eval_j(mi2)

        f2 = np.max(cb + mi2*cfi + vj2*cfj)
        if f1 < f2: hi_i = mi2
        else: lo_i = mi1
    vi = (lo_i+hi_i)/2

def coord_descent(h, n, t_end, max_sweeps=80, use_pairs=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

def perturb(h, n, it, cool, rng):

def slsqp_step(h, n, T, maxiter=800):
    def obj(x):
        corr = get_corr(np.clip(x,0,1))
        m = corr.max()
        return (m + np.log(np.sum(np.exp(T*(corr-m))))/T) * (2.0/n)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0}],
                       options={'maxiter':maxiter,'ftol':1e-14})
        return project_h(np.clip(res.x,0,1))
    except:
        return h
```
