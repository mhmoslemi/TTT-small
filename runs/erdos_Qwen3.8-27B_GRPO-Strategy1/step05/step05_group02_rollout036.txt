import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def smooth_c5(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(sh))) / T) * 2.0 / n

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    d = t - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        if r.sum() < 1e-14:
            return h
        h = h + r * d / r.sum()
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def constr_eq(x, n):
    return x.sum() - n / 2.0

def slsqp(h, T, n, mi=1500, ft=1e-16):
    try:
        r = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': constr_eq, 'args': (n,)}],
                     options={'maxiter': mi, 'ftol': ft})
        return project_h(np.clip(r.x, 0, 1))
    except:
        return h

def c5_1d(hi, cb, cf, dx):
    return float(np.max(cb + hi * cf) * dx)

def coord_one(h, n, i):
    hb = h.copy(); hb[i] = 0.0
    cb = np.correlate(hb, 1.0 - hb, mode='full')
    kr = np.arange(-(n - 1), n)
    j1, j2 = i + kr, i - kr
    m1, m2 = (j1 >= 0) & (j1 < n), (j2 >= 0) & (j2 < n)
    cf = np.zeros(2 * n - 1)
    cf[m1] += (1.0 - h[np.clip(j1[m1], 0, n - 1)])
    cf[m2] += (-h[np.clip(j2[m2], 0, n - 1)])
    try:
        r = minimize_scalar(c5_1d, args=(cb, cf, 2.0/n), bounds=(0, 1), method='bounded', options={'xatol': 1e-14})
        return float(r.x)
    except:
        return h[i]

def coord_desc(h, n, te, ms=300):
    bh = h.copy(); bc = compute_c5(h)
    for sw in range(ms):
        if time.time() > te:
            break
        imp = 0
        for i in range(n):
            hn = coord_one(bh, n, i)
            if abs(hn - bh[i]) < 1e-13:
                continue
            ht = bh.copy(); ht[i] = hn; ht = project_h(ht)
            ct = compute_c5(ht)
            if ct < bc - 1e-14:
                bc, bh, imp = ct, ht, imp + 1
        if imp == 0:
            break
    return bh, bc

def compute_subgrad(h, n, active):
    dx = 2.0 / n
    sg = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                sg[:n - lag] += (1.0 - h[lag:])
            if lag > 0:
                sg[lag:] += -h[:n - lag]
        else:
            ll = -lag
            if ll < n:
                sg[ll:] += (1.0 - h[:n - ll])
            if ll > 0:
                sg[:n - ll] += -h[ll:]
    sg *= dx / max(len(active), 1)
    return sg

def subgrad_phase(h, n, te, n_iter=20):
    bh = h.copy(); bc = compute_c5(h)
    for it in range(n_iter):
        if time.time() > te:
            break
        corr = np.correlate(bh, 1.0 - bh, mode='full')
        mx = corr.max()
        act = np.where(corr >= mx - 0.5)[0]
        alpha = 0.3
        for _ in range(30):
            sg = compute_subgrad(bh, n, act)
            hnew = project_h(bh - alpha * sg)
            cnew = compute_c5(hnew)
            if cnew < bc:
                bc, bh = cnew, hnew
            else:
                alpha *= 0.6
                if alpha < 1e-5:
                    break
            if time.time() > te:
                break
    return bh, bc

def nm_subset(h, n, idx, te, mi=3000):
    idx = np.array(idx)
    x0 = h[idx].copy()
    def nm_obj(xb):
        ht = h.copy(); ht[idx] = np.clip(xb, 0, 1)
        ht = project_h(ht)
        return compute_c5(ht)
    try:
        r = minimize(nm_obj, x0, method='Nelder-Mead',
                     options={'maxiter': mi, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True})
        ht = h.copy(); ht[idx] = np.clip(r.x, 0, 1)
        ht = project_h(ht)
        return ht, compute_c5(ht)
    except:
        return h.copy(), compute_c5(h)

def make_pert(n, cool, rng):
    m = rng.randint(0, 5)
    if m == 0:
        bs = rng.randint(4, max(5, n // 8))
        st = rng.randint(0, n - bs)
        p = np.zeros(n); p[st:st + bs] = (0.04 + 0.12 * cool) * rng.standard_normal(bs)
    elif m == 1:
        f = rng.randint(2, max(3, n // 15))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif m == 2:
        np_ = rng.randint(4, max(5, n // 8))
        ix = rng.choice(n, np_, replace=False)
        p = np.zeros(n); p[ix] = (0.08 + 0.25 * cool) * rng.standard_normal(np_)
    elif m == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    else:
        hf = n // 2
        p = np.zeros(n)
        p[:hf] = 0.1 * cool * rng.standard_normal(hf)
        p[hf:] = -0.1 * cool * rng.standard_normal(n - hf)
    return p

def bin_push(h, n, te):
    bh = h.copy(); bc = compute_c5(h)
    for _ in range(500):
        if time.time() > te:
            break
        d = np.minimum(bh, 1.0 - bh)
        i = int(np.argmin(d))
        if d[i] > 0.3:
            break
        ft = 0.0 if bh[i] > 0.5 else 1.0
        ht = bh.copy(); ht[i] = ft; ht = project_h(ht)
        ct = compute_c5(ht)
        if ct < bc - 1e-15:
            bc, bh = ct, ht
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        best_h = np.zeros(n0); best_h[:n0 // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP smooth refinement with increasing T
    for T in [20000, 100000, 500000]:
        if time.time() - t0 > budget_s * 0.10:

        hc = slsqp(best_h, T, n, mi=2000, ft=1e-16)

        if cv < best_c5:

    best_h, best_c5 = subgrad_phase(best_h, n, t0 + budget_s * 0.20, n_iter=20)

    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.38, ms=400)
