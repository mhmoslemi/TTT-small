import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                if i <= n - 1 - s:
                    lin[mp] -= best_h[i + s]
                if i <= n - 1 - s:
                    lin[mn] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[mn] -= best_h[i - s]
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-14:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def softmax_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exponents = np.clip(T * (corr - m), -50.0, 50.0)
    p = np.exp(exponents)
    p /= p.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = p[m_idx]
        if w < 1e-14:
            continue
        s = m_idx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

def adam_grad(h, n, t_end, max_iters=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = min(1.0, it / max_iters)
        T = 30.0 * 60.0 ** frac
        g = softmax_grad(h_c, T, n)
        t_i = it + 1
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * g * g
        m_hat = m_v / (1 - beta1 ** t_i)
        v_hat = v_v / (1 - beta2 ** t_i)
        step = lr / (1.0 + frac * 3.0)
        update = m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_h(h_c - step * update, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c5

def fourier_perturb(h, rng, amp):
    n = len(h)
    spec = np.fft.rfft(h)
    freqs = np.arange(len(spec))
    mask = freqs >= 2
    mask[0] = False
    mask[1] = False
    pert_spec = np.zeros_like(spec)
    pert_spec[mask] = amp * (rng.randn() + 1j * rng.randn())
    h_pert = h + np.fft.irfft(pert_spec, n=n)
    return h_pert

def block_perturb(h, rng, amp):
    n = len(h)
    nb = rng.randint(3, 8)
    sizes = np.zeros(nb, dtype=int)
    idx = 0
    for b in range(nb - 1):
        sizes[b] = max(1, int((n - idx) / (nb - b)))
        idx += sizes[b]
    sizes[nb - 1] = n - idx
    h_new = h.copy()
    offsets = []
    start = 0
    for s in sizes:
        offsets.append(rng.randint(-max(1, s // 2), max(1, s // 2) + 1))
        start += s
    for b in range(nb):
        s = sizes[b]
        start_b = b * 0
        for bb in range(b + 1):
            start_b += sizes[bb]
        end_b = start_b + s
        shift = offsets[b] % s
        h_new[start_b:end_b] = np.roll(h[start_b:end_b], shift)
    h_new = h + amp * 0.3 * (h_new - h)
    return h_new

def basin_hop_v2(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    stuck = 0
    step_base = 0.04
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - hop / max(n_hops, 1))
        step = step_base * (0.3 + 0.7 * cool)
        if stuck > 10:
            step *= 1.5
            stuck = 0
        mode = hop % 6
        if mode == 0:
            h_pert = fourier_perturb(cur_h, rng, step)
        elif mode == 1:
            h_pert = block_perturb(cur_h, rng, step)
        elif mode == 2:
            f = rng.randint(1, 8)
            ph = rng.uniform(0, 2 * np.pi)
            h_pert = cur_h + step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 3:
            mask = rng.random(n) < 0.2
            h_pert = cur_h.copy()
            h_pert[mask] += step * 2 * rng.randn(mask.sum())
        elif mode == 4:
            h_pert = cur_h + step * 0.5 * rng.randn(n)
        else:
            idx = rng.choice(n, size=max(8, n // 8), replace=False)
            h_pert = cur_h.copy()
            h_pert[idx] += step * rng.randn(len(idx))
        h_pert = project_h(h_pert, n)
        hop_time = max(1.5, 6.0 * cool + 1.0)
        hop_end = min(t_end, time.time() + hop_time)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=60, seed=seed + hop)
        temp = 0.003 * cool + 0.0002
        accept = c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp))
        if accept:
            if c5_opt >= cur_c5:
                stuck += 1
            else:
                stuck = 0
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        else:
            stuck += 1
    return best_h, best_c5

def upsample_h(h, n_from, n_to):
    dx_old = 2.0 / n_from
    x_old = (np.arange(n_from) + 0.5) * dx_old
    x_new = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    h_up = np.interp(x_new, x_old, h, period=2.0)
    return project_h(h_up, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    t1 = t0 + budget_s * 0.18
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=2500, seed=seed)

    if time.time() < t0 + budget_s * 0.35:
        h_g, c5_g = adam_grad(best_h, n, t0 + budget_s * 0.33, max_iters=3000)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.40, max_sweeps=1500, seed=seed + 1)

    if time.time() < t0 + budget_s * 0.68:
        best_h, best_c5 = basin_hop_v2(best_h, n, t0 + budget_s * 0.68, n_hops=500, seed=seed + 10)

    if time.time() < t0 + budget_s * 0.78:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.78, max_sweeps=3000, seed=seed + 100)

    if time.time() < t0 + budget_s * 0.88:
        h_g2, c5_g2 = adam_grad(best_h, n, t0 + budget_s * 0.86, max_iters=1500)
        if c5_g2 < best_c5:

    assert abs(best_h.sum() - n / 2.0) < 1e-6
