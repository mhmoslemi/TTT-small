from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def exact_coord_sweep(h, n, n_top=0, max_iters=50):
    """Exact coordinate descent: solve 1D subproblem exactly for each coordinate."""
    best_h = h.copy()
    for it in range(max_iters):
        corr = compute_corr(best_h)
        improved = False
        for i in range(n):
            h_old = best_h[i]
            # Compute coefficients for each shift
            k_range = np.arange(-(n-1), n)
            valid_plus = (i + k_range >= 0) & (i + k_range < n)
            valid_minus = (i - k_range >= 0) & (i - k_range < n)
            coeff = np.zeros(2*n-1)
            coeff[valid_plus] = 1.0 - best_h[i + k_range[valid_plus]]
            coeff[valid_minus] -= best_h[i - k_range[valid_minus]]
            
            base = corr - h_old * coeff
            
            # Minimize max(base + t*coeff) over t in [0,1]
            def obj(t):
                return np.max(base + t * coeff)
            
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                   options={'xatol': 1e-10})
            t_new = res.x
            if abs(t_new - h_old) > 1e-10:
                best_h[i] = t_new
                improved = True
        if not improved:
            break

    corr_np1 = np.arange(-(n-1), n)  # shift values
    prev_c5 = compute_c5(best_h)
    no_improve = 0
    for it in range(max_iters):
        corr = compute_corr(best_h)
        sweep_improved = 0
        for i in range(n):
            h_old = best_h[i]
            # coeff[s] for shift k_s = s - (n-1)

            valid_p = (i + ks >= 0) & (i + ks < n)

            coeff[valid_m] -= best_h[i - ks[valid_m]]
            base = corr - h_old * coeff
            # minimize max(base + t*coeff) for t in [0,1]
            res = minimize_scalar(lambda t: np.max(base + t * coeff),
                                   bounds=(0,1), method='bounded', options={'xatol':1e-10})

                sweep_improved += 1

            break
        c5_new = compute_c5(best_h)
        if c5_new < prev_c5 - 1e-12:
            prev_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break
    return best_h

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def exact_coord_descent(h, n, max_iters=200, t_end=None):
    h = project_h(h.copy())
    ks = np.arange(-(n-1), n)
    for it in range(max_iters):
        if t_end is not None and time.time() > t_end:
            break
        corr = compute_corr(h)
        improved = 0
        for i in range(n):
            if t_end is not None and (i % 50 == 0) and time.time() > t_end:
                return h
            h_old = h[i]
            valid_p = (i + ks >= 0) & (i + ks < n)
            valid_m = (i - ks >= 0) & (i - ks < n)
            coeff = np.zeros(2*n-1)
            coeff[valid_p] = 1.0 - h[i + ks[valid_p]]
            coeff[valid_m] -= h[i - ks[valid_m]]
            base = corr - h_old * coeff
            res = minimize_scalar(lambda t: np.max(base + t * coeff),
                                   bounds=(0,1), method='bounded', options={'xatol':1e-10})
            t_new = res.x
            delta = t_new - h_old
            if abs(delta) > 1e-10:
                h[i] = t_new
                corr += delta * coeff
                improved += 1
        if improved == 0:
            break
    return h

def shift_targeted_adjust(h, n, n_top=5, t_end=None):
    """Target the top n_top shifts specifically."""
    h = h.copy()
    for _ in range(50):
        if t_end is not None and time.time() > t_end:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-n_top:]
        shifts = top_idx - (n-1)
        # For each top shift, compute gradient and take a small step
        grad = np.zeros(n)
        for s in shifts:
            for i in range(n):
                v_p = (i + s >= 0) and (i + s < n)
                v_m = (i - s >= 0) and (i - s < n)
                g = 0.0
                if v_p:
                    g += (1.0 - h[i+s])
                if v_m:
                    g -= h[i-s]
                grad[i] += g
        grad /= n_top
        # Step
        alpha = 0.01
        h -= alpha * grad
        h = project_h(h)
    return exact_coord_descent(h, n, 50, t_end)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    t_end = t0 + budget_s

    # Small random perturbations
